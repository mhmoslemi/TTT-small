def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 1:

            break

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm = float(c_.max())
            shifted = (c_ - cm) / beta
            shifted = shifted - shifted.max()
            e_ = np.exp(shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * float(np.log(s_)))

        def grad(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            g = _compute_logsumexp_grad(a_, c_, beta, len(c_))
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z.copy(), jac=grad,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_level, 'ftol': 1e-22,
                         'gtol': 1e-14, 'maxls': 50, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                    print(f"    L-BFGS beta_frac={beta_frac:.6f}: score={best_score:.10f}")
        except Exception:
            pass

    return best_a, best_score


def _subgradient_momentum(a_init, deadline, lr_init=0.01, decay=0.9995, mu=0.9, K_top=5):
    """Sub-gradient with Nesterov momentum using top-K peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    velocity = np.zeros(n)
    lr = lr_init

    iter_count = 0
    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        K = min(K_top, len(c))
        top_idx = np.argsort(c)[-K:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / K
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1] - W
        grad = 4.0 * n * a * h

        velocity = mu * velocity - lr * grad
        a_new = a - velocity
        a_new = _project_simplex(a_new)

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            a = a.copy()

        lr *= decay
        velocity *= 0.999
        iter_count += 1

        if iter_count % 500 == 0:
            print(f"    SubGrad iter={iter_count}, lr={lr:.8f}, score={best_score:.10f}")

    return best_a, best_score


def _peak_coordinate_descent(a_init, deadline, n_passes_max=500):
    """Coordinate descent targeting the peak convolution lag."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for p in range(n_passes_max):
        if time.monotonic() > deadline - 2.0:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        c_peak = float(c[k_star])

        # Compute contribution of each element to the peak
        # c[k_star] = sum_i a[i]*a[k_star-i]
        contrib = np.zeros(n)
        if k_star < n:
            contrib[:k_star + 1] = a[:k_star + 1] * a[k_star::-1]
        else:
            start = k_star - n + 1
            contrib[start:] = a[start:] * a[n - 1:start - 1:-1]

        # Find the element with largest contribution
        i_peak = int(np.argmax(contrib))

        # Find "safe" elements (low contribution to top lags)
        top_lags = np.argsort(c)[-min(5, len(c)):]
        danger = np.zeros(n)
        for kl in top_lags:
            kl = int(kl)
            if kl < n:
                danger[:kl + 1] += a[:kl + 1] * a[kl::-1]
            else:
                s = kl - n + 1
                danger[s:] += a[s:] * a[n - 1:s - 1:-1]

        # Pick a safe element (low danger, not the peak element)
        safe_mask = np.ones(n, dtype=bool)
        safe_mask[i_peak] = False
        danger_sorted = np.argsort(danger)
        add_idx = -1
        for idx in danger_sorted:
            if safe_mask[idx] and a[idx] > 1e-10:
                add_idx = int(idx)
                break
        if add_idx < 0:
            add_idx = int(rng.integers(n))
            if add_idx == i_peak:
                add_idx = (add_idx + 1) % n

        # Try several fractions to move
        best_frac = 0
        best_sc_trial = best_score
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            delta = a[i_peak] * frac
            if delta < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i_peak] -= delta
            a_trial[add_idx] += delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            sc = _score_norm(a_trial)
            if sc < best_sc_trial:
                best_sc_trial = sc
                best_frac = frac

        if best_frac > 0:
            delta = a[i_peak] * best_frac
            a[i_peak] -= delta
            a[add_idx] += delta
            a = np.maximum(a, 0.0)
            a /= a.sum()
            if best_sc_trial < best_score:
                best_score = best_sc_trial
                best_a = a.copy()

    return best_a, best_score


def _simulated_annealing(a_init, deadline, n_trials=50000, T_init=0.005, T_final=0.0001):
    """Simulated annealing with small multiplicative perturbations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(67890)

    n_total = max(1, n_trials)
    for it in range(n_total):
        if time.monotonic() > deadline - 2.0:
            break

        T = T_init * (T_final / T_init) ** (it / n_total)

        # Perturbation: multiply random block by random factor
        block_size = int(rng.integers(3, min(32, n // 10)))
        start = int(rng.integers(0, n - block_size))
        a_trial = a.copy()
        mult = np.ones(block_size)
        mult[:block_size] = np.exp(rng.normal(0, T, size=block_size))
        a_trial[start:start + block_size] *= mult
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        sc_trial = _score_norm(a_trial)
        sc_curr = _score_norm(a) if it % 10 == 0 else best_score

        if sc_trial < best_score or (rng.random() < np.exp(-(sc_trial - best_score) / max(T * 0.1, 1e-10))):
            a = a_trial.copy()
            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a_trial.copy()

    return best_a, best_score


def _topk_lbfgs(a_init, deadline, K=3, maxiter_per_level=3000):
    """L-BFGS using top-K gradient (cheaper than logsumexp)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    betas = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004]

    for beta_frac in betas:
        beta = beta_frac * max_c0
        if time.monotonic() > deadline - 5.0:
            break

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm = float(c_.max())
            shifted = (c_ - cm) / beta
            shifted = shifted - shifted.max()
            e_ = np.exp(shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * float(np.log(s_)))

        def grad(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            g = _compute_logsumexp_grad(a_, c_, beta, len(c_))
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z.copy(), jac=grad,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_level, 'ftol': 1e-22,
                         'gtol': 1e-15, 'maxls': 40, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS continuation, sub-gradient momentum,
    coordinate descent, simulated annealing, and final polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_work = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_work = 4096
        i = np.arange(n_work, dtype=np.float64)
        parent = np.maximum((i + 1) * (n_work - i), 1e-10)
        parent /= parent.sum()
        sc_parent = _score_norm(parent)
        print(f"No parent, triangular n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    print(f"Starting optimization with n={n_work}, current best={best_score:.10f}")

    # Phase 1: L-BFGS continuation - fine (0-400s)
    phase1_end = min(deadline, t_start + 400.0)
    print(f"Phase 1: L-BFGS fine continuation (target {phase1_end - t_start:.0f}s)")
    a1, sc1 = _lbfgs_continuation(
        parent, phase1_end,
        beta_starts=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
        maxiter_per_level=3000
    )
    update_best(a1, sc1, "lbfgs_fine")

    # Phase 2: Sub-gradient with momentum (400-800s)
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end:
        print(f"Phase 2: Sub-gradient momentum (target {phase2_end - time.monotonic():.0f}s)")
        a2, sc2 = _subgradient_momentum(
            best_seq, phase2_end,
            lr_init=0.005, decay=0.9998, mu=0.85, K_top=3
        )
        update_best(a2, sc2, "subgrad_K3")

        a2b, sc2b = _subgradient_momentum(
            best_seq, min(deadline, time.monotonic() + 150.0),
            lr_init=0.003, decay=0.9999, mu=0.9, K_top=7
        )
        update_best(a2b, sc2b, "subgrad_K7")

    # Phase 3: Peak coordinate descent (800-1200s)
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Peak coordinate descent (target {phase3_end - time.monotonic():.0f}s)")
        a3, sc3 = _peak_coordinate_descent(best_seq, phase3_end, n_passes_max=300)
        update_best(a3, sc3, "coord_desc")

    # Phase 4: Simulated annealing (1200-1500s)
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: Simulated annealing (target {phase4_end - time.monotonic():.0f}s)")
        a4, sc4 = _simulated_annealing(
            best_seq, phase4_end,
            n_trials=40000, T_init=0.003, T_final=0.00005
        )
        update_best(a4, sc4, "sim_anneal")

    # Phase 5: Top-K L-BFGS (1500-1700s)
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Top-K L-BFGS (target {phase5_end - time.monotonic():.0f}s)")
        a5, sc5 = _topk_lbfgs(best_seq, phase5_end, K=3, maxiter_per_level=5000)
        update_best(a5, sc5, "topk_lbfgs")

    # Phase 6: Final intensive polish (1700s to deadline)
    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"Phase 6: Final polish (remaining {remaining:.0f}s)")

        if remaining > 100:
            a6, sc6 = _lbfgs_continuation(
                best_seq, min(deadline - 20, time.monotonic() + remaining * 0.5),
                beta_starts=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                maxiter_per_level=5000
            )
            update_best(a6, sc6, "final_lbfgs")

        if time.monotonic() < deadline - 30:
            a7, sc7 = _peak_coordinate_descent(
                best_seq, min(deadline - 25, time.monotonic() + 60.0),
                n_passes_max=100
            )
            update_best(a7, sc7, "final_coord")

        if time.monotonic() < deadline - 20:
            a8, sc8 = _simulated_annealing(
                best_seq, min(deadline - 15, time.monotonic() + 40.0),
                n_trials=15000, T_init=0.001, T_final=0.00002
            )
            update_best(a8, sc8, "final_anneal")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            i = np.arange(4096, dtype=np.float64)
            final_seq = np.maximum((i + 1) * (4096 - i), 1e-10)
            final_seq /= final_seq.sum()
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_norm(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        # Validate with evaluator if feasible
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    i = np.arange(4096, dtype=np.float64)
    fallback = np.maximum((i + 1) * (4096 - i), 1e-10)
    fallback /= fallback.sum()
    print(f"FALLBACK: n=4096, score={_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
