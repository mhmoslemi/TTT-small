"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Uses diverse initializations, FISTA with
adaptive beta, peak-flattening refinement, and L-BFGS polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    """Project v onto the probability simplex (sum = 1, all >= 0)"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    """Compute gradient of sum_k w_k * c_k where c = a*a (full conv).
    w is a vector of length 2n-1 (weights for each conv position).
    Returns gradient of length n.
    """
    n = len(a)
    a_rev = a[::-1]
    # gradient_j = 2 * sum_k w_k * a[k-j] (approximately)
    # This is 2 * (w * a_rev)[n-1 : 2n-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_weights(c, beta):
    """Compute softmax weights from correlation values"""
    cm = float(c.max())
    z = (c - cm) / beta
    # Clip to avoid overflow
    z = np.clip(z, -50, 50)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _topk_weights(c, k, temperature=0.01):
    """Compute weights concentrated on the top-k correlation values"""
    cm = float(c.max())
    # Use a "top-k" style weight: exp((c - cm) / temperature) for top k
    # and very small weight elsewhere
    sorted_idx = np.argsort(c)[::-1]
    w = np.zeros_like(c)
    top_vals = c[sorted_idx[:k]]
    top_min = top_vals[-1]
    # For top k: use softmax
    z_top = (top_vals - top_vals[0]) / temperature
    z_top = np.clip(z_top, -50, 50)
    e_top = np.exp(z_top)
    w_top = e_top / e_top.sum()
    for i in range(k):
        w[sorted_idx[i]] = w_top[i]
    return w


def _fista_optimize(a_init, beta, deadline, max_iter):
    """FISTA optimization with simplex constraint and given beta smoothing"""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    L = 4.0 * n / beta  # Lipschitz constant estimate
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _softmax_weights(c_y, beta)
        grad = _compute_grad(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _peak_flatten(a, top_k=10, iterations=5000, deadline=None):
    """
    Additional phase: specifically target the top-k peaks in the autocorrelation.
    Uses a "peak variance" penalty to encourage flattening.
    """
    n = len(a)
    a_work = a.copy()
    best_a = a_work.copy()
    best_score = _score_norm(a_work)

    # Use log-space parameterization
    z = np.log(np.maximum(a_work, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.005

    for it in range(iterations):
        if deadline is not None and time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Compute top-k weights (sharper than full softmax)
        beta_top = 0.001 * (cm + 1e-10)
        w = _softmax_weights(c, beta_top)

        # Also compute a "peak variance" gradient
        # Top k values
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        top_mean = top_vals.mean()

        # Gradient of max is standard
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.2:
            update = update * (0.2 / update_norm)

        z = z - update
        z = np.clip(z, -30.0, 30.0)
        z = z - z.mean()

        if t % 100 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
            # Decay learning rate
            lr *= 0.999

    return best_a, best_score


def _make_gaussian(n, width_frac=0.25):
    """Create a Gaussian profile"""
    center = (n - 1) / 2.0
    sigma = n * width_frac
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    """Create a raised cosine (Hann) window profile"""
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_interior_conc(n, frac=0.6):
    """Sequence that's 1 in the interior and 0 at edges (with smooth transition)"""
    t = np.linspace(0, 1, n)
    edge = (1 - frac) / 2.0
    a = np.ones(n)
    # Smooth edges
    left = t < edge
    right = t > 1 - edge
    a[left] = np.maximum(0.5 * (1 + np.cos(np.pi * t[left] / edge)), 1e-10)
    a[right] = np.maximum(0.5 * (1 + np.cos(np.pi * (1 - t[right]) / edge)), 1e-10)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _scipy_lbfgs_polish(a_init, deadline, max_fun=5000):
    """Use scipy L-BFGS-B on log-parameterized variables for final polish"""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def objective(z):
        a = _softmax_compute(z)
        return _score_norm(a)

    def gradient(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Use very small beta for near-exact max
        beta = max(1e-6, cm * 1e-4)
        w = _softmax_weights(c, beta)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 4.0 * n * a * (h - W)
        # Chain rule: d/dz = d/da * da/dz
        # da/dz = a * (1 - a) (for softmax)
        grad_z = grad_a * a * (1.0 - a)
        return grad_z

    # Track best
    best_z = z0.copy()
    best_val = objective(z0)

    t_start = time.monotonic()
    try:
        result = scipy_minimize(
            objective, z0, method='L-BFGS-B',
            jac=gradient,
            options={'maxiter': 200, 'maxfun': max_fun, 'gtol': 1e-8,
                     'ftol': 1e-12}
        )
        if result.fun < best_val:
            best_val = result.fun
            best_z = result.x
    except Exception:
        pass

    a_result = _softmax_compute(best_z)
    return a_result, best_val

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Combines diverse FISTA initialization,
adaptive-beta Adam in log-space, peak-flattening refinement,
multi-restart diversification, and large-n exploration.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _softmax_weights(c, beta):
    cm = float(c.max())
    z = (c - cm) / beta
    z = np.clip(z, -60, 60)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _compute_grad_h(a, w):
    """Compute h[j] = sum_k w[k] * a[k-j] (the gradient building block)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]


def _fista_optimize(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _softmax_weights(c_y, beta)
        h = _compute_grad_h(y, w)
        grad = 2.0 * h

        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_list, lr_init=0.01, max_iter_total=400000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    v_max_buf = np.zeros(n)  # AMSGrad
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    best_score_history = {}

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        no_improve_count = 0
        phase_best = best_score

        while it_in_phase < max_iter_total and t < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = np.clip((c - cm) / beta, -60, 60)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            h = _compute_grad_h(a, w)
            W = float(np.dot(w, c))
            grad_z = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            v_max_buf = np.maximum(v_max_buf, v_buf)  # AMSGrad
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_max_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_update = 0.4
            if update_norm > max_update:
                update = update * (max_update / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 40 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                elif sc >= phase_best - 1e-12:
                    no_improve_count += 1
                else:
                    no_improve_count = 0

            # Learning rate: cosine decay within phase
            progress = it_in_phase / max(1, max_iter_total / len(beta_list))
            lr_decay = 0.5 * (1 + np.cos(np.pi * min(1.0, progress)))
            current_lr = lr_init * lr_decay
            if no_improve_count > 50:
                current_lr *= 0.9

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _make_gaussian(n, width_frac=0.25):
    center = (n - 1) / 2.0
    sigma = n * width_frac
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_interior(n, frac=0.7):
    t = np.linspace(0, 1, n)
    edge = (1 - frac) / 2.0
    a = np.ones(n)
    left_mask = t < edge
    right_mask = t > 1 - edge
    a[left_mask] = 0.5 * (1 + np.cos(np.pi * t[left_mask] / edge))
    a[right_mask] = 0.5 * (1 + np.cos(np.pi * (1 - t[right_mask]) / edge))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n):
    """Two Gaussian bumps"""
    n_half = n // 2
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - n_half * 0.4) / (n * 0.1)) ** 2)
    a += np.exp(-0.5 * ((t - n_half * 1.6) / (n * 0.1)) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        betas_fine.append(beta)
        beta *= 0.45
    betas_fine = betas_fine[:14]

    per_time = phase2_end - time.monotonic() - 15
    if per_time > 30:
        a_opt, _ = _adam_logspace(
            a_work, time.monotonic() + per_time,
            betas_fine, lr_init=0.012, max_iter_total=400000
        )
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  Adam fine: {sc_opt:.10f} t={time.monotonic()-t_start:.0f}s")

    a_work = best_seq.copy() / best_seq.sum()
    print(f"Phase 2 best: {best_score:.10f}")

    # ===== PHASE 3: Peak flattening (800 - 1100s) =====
    print("=== Phase 3: Peak flatten ===")
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end - 30:
        per_t = phase3_end - time.monotonic() - 10
        a_pf, _ = _peak_flatten(a_work, time.monotonic() + per_t, top_k=20,
                                lr_init=0.003, max_iter=100000)
        sc_pf = _score(a_pf)
        if sc_pf < best_score:
            best_score = sc_pf
            best_seq = a_pf.copy()
            print(f"  Peak flatten: {sc_pf:.10f} t={time.monotonic()-t_start:.0f}s")

    print(f"Phase 3 best: {best_score:.10f}")

    # ===== PHASE 4: Multi-restart diversification (1100 - 1450s) =====
    print("=== Phase 4: Multi-restart ===")
    phase4_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase4_end - 50:
        a_base = best_seq.copy() / best_seq.sum()
        n_restarts = 6
        per_restart = max(30, (phase4_end - time.monotonic() - 20) / n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break

            if i == 0:
                p_start = _perturb(a_base, 0.015, rng)
            elif i == 1:
                # Spectral perturbation
                p_start = a_base.copy()
                freq = rng.integers(3, max(4, n_main // 50))
                t_arr = np.arange(n_main, dtype=np.float64)
                p_start = p_start + 0.008 * np.cos(2 * np.pi * freq * t_arr / n_main + rng.uniform(0, 2 * np.pi))
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif i == 2:
                p_start = _perturb(a_base, 0.06, rng)
            elif i == 3:
                # Smooth with small kernel
                kernel = np.array([0.15, 0.35, 0.35, 0.15])
                p_start = np.convolve(a_base, kernel, mode='same')
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif i == 4:
                # Frequency-domain perturbation
                F = np.fft.rfft(a_base)
                F_mag = np.abs(F)
                F_phase = np.angle(F)
                F_mag *= (1 + 0.02 * rng.standard_normal(len(F_mag)))
                F_mag = np.maximum(F_mag, 0)
                F_new = F_mag * np.exp(1j * F_phase)
                p_start = np.fft.irfft(F_new, n=n_main)
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                # Anti-perturbation: inverse of perturbation
                z = np.log(np.maximum(a_base, 1e-15))
                z = z - 0.03 * rng.standard_normal(len(z))
                z = z - z.mean()
                p_start = _softmax_compute(z)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.001 * max_cs, 0.0004 * max_cs, 0.00015 * max_cs, 0.00006 * max_cs]

            p_best, _ = _adam_logspace(
                p_start, time.monotonic() + per_restart,
                bs, lr_init=0.008, max_iter_total=60000
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: {sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    print(f"Phase 4 best: {best_score:.10f}")

    # ===== PHASE 5: Larger n exploration (1450 - 1680s) =====
    print("=== Phase 5: Larger n ===")
    phase5_end = min(deadline, t_start + 1680.0)
    if time.monotonic() < phase5_end - 50 and best_seq is not None:
        for n_try in [8192, 16384]:
            if time.monotonic() > phase5_end - 40:
                break
            a_try = _interp_profile(best_seq, n_try)
            sc_try = _score(a_try)
            print(f"  n={n_try} interpolated: {sc_try:.10f}")

            if sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()

            # Quick optimize
            if time.monotonic() < phase5_end - 40:
                c_try = _full_convolve(a_try, a_try)
                max_ct = float(c_try.max())
                beta_t = 0.003 * max_ct
                a_try2, _ = _fista_optimize(a_try, beta_t, time.monotonic() + 25, 15000)
                a_try2 = a_try2 / a_try2.sum()

                c_try2 = _full_convolve(a_try2, a_try2)
                max_ct2 = float(c_try2.max())
                betas_t = [0.001 * max_ct2, 0.0004 * max_ct2, 0.00015 * max_ct2]
                per_t5 = min(30, phase5_end - time.monotonic() - 10)
                if per_t5 > 15:
                    a_try2, _ = _adam_logspace(
                        a_try2, time.monotonic() + per_t5,
                        betas_t, lr_init=0.005, max_iter_total=40000
                    )
                sc_try2 = _score(a_try2)
                if sc_try2 < best_score:
                    best_score = sc_try2
                    best_seq = a_try2.copy()
                    print(f"  n={n_try} optimized: {sc_try2:.10f}")

    print(f"Phase 5 best: {best_score:.10f}")

    # ===== PHASE 6: Final ultra-fine polish (1680 - deadline) =====
    print("=== Phase 6: Ultra-fine polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())

        # Very fine betas
        betas_ult = []
        beta = 0.00005 * max_cp
        while beta > 0.000001 * max_cp:
            betas_ult.append(beta)
            beta *= 0.4
        betas_ult = betas_ult[:8]

        per_t6 = deadline - time.monotonic() - 8
        if per_t6 > 10:
            a_pol2, _ = _adam_logspace(
                a_pol, time.monotonic() + per_t6,
                betas_ult, lr_init=0.0015, max_iter_total=150000
            )
            sc_pol2 = _score(a_pol2)
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_seq = a_pol2.copy()
                print(f"  Ultra-fine: {sc_pol2:.10f}")

            # One more peak flatten pass
            if time.monotonic() < deadline - 15:
                a_pf2, _ = _peak_flatten(
                    best_seq, time.monotonic() + min(15, deadline - time.monotonic() - 5),
                    top_k=10, lr_init=0.001, max_iter=20000
                )
                sc_pf2 = _score(a_pf2)
                if sc_pf2 < best_score:
                    best_score = sc_pf2
                    best_seq = a_pf2.copy()
                    print(f"  Final peak flatten: {sc_pf2:.10f}")

    # ===== FINAL =====
    print(f"\n=== FINAL RESULT ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"Score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Eval: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(512) / 512.0
    print(f"FALLBACK: {_score(fallback):.10f}")
    return [float(x) for x in fallback]
