import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _convolve_self(a):
    return _fft_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _norm_score(a):
    n = len(a)
    c = _convolve_self(a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _cross_correlation(w, a):
    """Compute h[i] = sum_k w[k]*a[k-i] for i=0..n-1 using FFT."""
    n = len(a)
    N = _next_pow2(2 * n)
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a[::-1], N)
    result = np.fft.irfft(W * A_rev, N)
    return result[n - 1:2 * n - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _norm_score(a):
    n = len(a)
    c = _convolve_self(a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _cross_correlation(w, a):
    """Compute h[i] = sum_k w[k]*a[k-i] for i=0..n-1 using FFT."""
    n = len(a)
    N = _next_pow2(2 * n)
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a[::-1], N)
    result = np.fft.irfft(W * A_rev, N)
    return result[n - 1:2 * n - 1]


def _smoothed_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a on simplex."""
    n = len(a)
    c = _convolve_self(a)
    cm = float(c.max())
    shifted = beta * c - beta * cm
    e = np.exp(shifted)
    w = e / e.sum()
    h = _cross_correlation(w, a)
    grad = 2.0 * h
    return grad, c, w


def _mirror_descent(a_init, deadline, rng, lr_init=1.0, lr_decay=0.9997,
                    beta_start=30.0, beta_end=300.0, n_phases=8,
                    momentum_decay=0.9):
    """Mirror descent with adaptive smoothing and momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _norm_score(a_init)

    momentum = np.zeros(n)
    lr = lr_init

    betas = np.geomspace(beta_start, beta_end, n_phases)

    for phase_i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        phase_time = (deadline - time.monotonic()) / (n_phases - phase_i)
        max_iters = int(phase_time * 80)  # ~80 iterations per second target
        max_iters = max(max_iters, 100)

        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            a = _softmax_fn(z)
            grad, c, w = _smoothed_gradient(a, beta)

            # Center gradient
            grad_centered = grad - grad.mean()

            # Momentum update
            momentum = momentum_decay * momentum + (1.0 - momentum_decay) * grad_centered

            # Update
            z = z - lr * momentum
            z = np.clip(z, -60, 60)
            z = z - z.mean()

            # Evaluate
            if it % 50 == 0:
                sc = _norm_score(_softmax_fn(z))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = _softmax_fn(z).copy()

            # Decay lr
            lr *= lr_decay

        # Evaluate at end of phase
        sc = _norm_score(_softmax_fn(z))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = _softmax_fn(z).copy()
        print(f"    Phase {phase_i+1}/{n_phases} (beta={beta:.1f}): score={best_score:.10f}")

    return best_a, best_score


def _peak_shaving(a, deadline, rng, max_rounds=3000):
    """Surgically reduce the top convolution peaks."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()
    best_score = _norm_score(a)
    no_improve_count = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _convolve_self(a_cur)
        cm = float(c.max())
        avg_c = float(c.mean())

        if cm < avg_c * 1.0005:
            break

        peak_idx = int(np.argmax(c))
        valid_i = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
        if len(valid_i) == 0:
            continue

        prod = a_cur[valid_i] * a_cur[peak_idx - valid_i]
        order = np.argsort(prod)[::-1][:8]

        improved = False
        for pos in order:
            idx = int(valid_i[pos])
            # Try reducing this element
            for delta in [0.003, 0.007, 0.015, 0.03]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _norm_score(a_trial)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing small elements not contributing to peak
            small_idx = np.argsort(a_cur)[:10]
            for sidx in small_idx:
                if sidx not in valid_i:
                    for delta in [0.005, 0.01, 0.02]:
                        z_trial = z.copy()
                        z_trial[sidx] += delta
                        a_trial = _softmax_fn(z_trial)
                        sc = _norm_score(a_trial)
                        if np.isfinite(sc) and sc < best_score - 1e-12:
                            best_score = sc
                            best_a = a_trial.copy()
                            z = z_trial.copy()
                            improved = True
                            break
                if improved:
                    break

        if not improved:
            no_improve_count += 1
            if no_improve_count > 200:
                # Perturb and reset counter
                z += 0.03 * rng.normal(size=n)
                z = z - z.mean()
                no_improve_count = 0
        else:
            no_improve_count = 0

    return best_a, best_score


def _lbfgs_optimize(a_init, deadline, rng, max_betas=6):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _norm_score(a_init)

    def make_obj_grad(beta):
        def obj(zv):
            av = _softmax_fn(zv)
            cv = _convolve_self(av)
            cm = float(cv.max())
            shifted = beta * cv - beta * cm
            e = np.exp(shifted)
            return (beta * cm + float(np.log(np.sum(e))))
        def grad(zv):
            av = _softmax_fn(zv)
            cv = _convolve_self(av)
            cm = float(cv.max())
            shifted = beta * cv - beta * cm
            e = np.exp(shifted)
            w = e / e.sum()
            h = _cross_correlation(w, av)
            grad_a = 2.0 * h
            offset = float(np.dot(av, grad_a))
            return av * (grad_a - offset)
        return obj, grad

    betas = [50.0, 100.0, 200.0, 400.0, 800.0, 1600.0][:max_betas]
    for beta in betas:
        if time.monotonic() > deadline:
            break
        obj, grad = make_obj_grad(beta)
        remaining_iters = int((deadline - time.monotonic()) * 5)
        remaining_iters = min(5000, max(100, remaining_iters))
        try:
            result = minimize(obj, z, method='L-BFGS-B', jac=grad,
                              options={'maxiter': remaining_iters, 'ftol': 1e-15,
                                       'gtol': 1e-9, 'maxls': 40,
                                       'maxfun': remaining_iters * 3})
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -60, 60)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                sc = _norm_score(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _spectral_equality(a, deadline, rng, max_iters=3000):
    """Minimize variance of the autocorrelation using gradient descent."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()
    best_score = _norm_score(a)
    lr = 0.05

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        a_cur = _softmax_fn(z)
        c = _convolve_self(a_cur)
        avg_c = float(c.mean())

        # Gradient of variance: d(sum (c[k]-avg)^2)/da[i]
        # = 2 * sum_k (c[k]-avg) * dc[k]/da[i] = 2 * sum_k (c[k]-avg) * 2*a[k-i]
        weights = 2.0 * (c - avg_c)
        h = _cross_correlation(weights, a_cur)
        grad = 2.0 * h
        grad = grad - grad.mean()

        z = z - lr * grad
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        lr *= 0.9999

        if it % 100 == 0:
            sc = _norm_score(_softmax_fn(z))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = _softmax_fn(z).copy()

            # Periodic perturbation
            if it % 1000 == 999:
                z += 0.02 * rng.normal(size=n)
                z = z - z.mean()

    return best_a, best_score


def _make_ramp(n, slope=1.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_p = _norm_score(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(4096)
        update_best(a0, _norm_score(a0), "default_ramp")
        parent = a0

    n_target = len(best_seq)
    print(f"Target n = {n_target}")

    # --- Phase 1: Mirror descent (35% time) ---
    p1_end = t_start + (budget_s - 30.0) * 0.35
    print(f"\nPhase 1: Mirror descent [deadline t={p1_end - t_start:.0f}s]")

    a_work = best_seq.copy()
    configs = [
        {'lr_init': 1.5, 'lr_decay': 0.9998, 'beta_start': 20.0, 'beta_end': 200.0, 'n_phases': 8},
        {'lr_init': 0.5, 'lr_decay': 0.9995, 'beta_start': 50.0, 'beta_end': 500.0, 'n_phases': 6},
        {'lr_init': 3.0, 'lr_decay': 0.9999, 'beta_start': 10.0, 'beta_end': 100.0, 'n_phases': 10},
    ]
    for ci, cfg in enumerate(configs):
        if time.monotonic() > p1_end:
            break
        sub_dl = min(p1_end, time.monotonic() + (p1_end - time.monotonic()) / (len(configs) - ci))
        a_md, sc_md = _mirror_descent(a_work, sub_dl, rng, **cfg)
        update_best(a_md, sc_md, f"mirror_{ci}")
        if sc_md < best_score:
            a_work = a_md.copy()

    print(f"  After Phase 1: best={best_score:.10f}")

    # --- Phase 2: L-BFGS (20% time) ---
    p2_end = t_start + (budget_s - 30.0) * 0.55
    print(f"\nPhase 2: L-BFGS [deadline t={p2_end - t_start:.0f}s]")

    if time.monotonic() < p2_end - 10:
        a_lb, sc_lb = _lbfgs_optimize(best_seq, p2_end - 5, rng, max_betas=5)
        update_best(a_lb, sc_lb, "lbfgs")

        # Perturbed restart
        if time.monotonic() < p2_end - 20:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z += 0.08 * rng.normal(size=n_target)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a_pert = _softmax_fn(z)
            a_lp, sc_lp = _lbfgs_optimize(a_pert, p2_end - 3, rng, max_betas=3)
            update_best(a_lp, sc_lp, "lbfgs_pert")

    print(f"  After Phase 2: best={best_score:.10f}")

    # --- Phase 3: Peak shaving (25% time) ---
    p3_end = t_start + (budget_s - 30.0) * 0.80
    print(f"\nPhase 3: Peak shaving [deadline t={p3_end - t_start:.0f}s]")

    if time.monotonic() < p3_end - 10 and best_seq is not None:
        a_ps, sc_ps = _peak_shaving(best_seq, p3_end, rng, max_rounds=5000)
        update_best(a_ps, sc_ps, "peak_shave")

    print(f"  After Phase 3: best={best_score:.10f}")

    # --- Phase 4: Spectral + final polish (20% time) ---
    p4_end = deadline
    print(f"\nPhase 4: Spectral + final polish [deadline t={p4_end - t_start:.0f}s]")

    if time.monotonic() < p4_end - 15 and best_seq is not None:
        a_sp, sc_sp = _spectral_equality(best_seq, p4_end - 30, rng, max_iters=3000)
        update_best(a_sp, sc_sp, "spectral")

        # Final mirror descent with small lr
        if time.monotonic() < p4_end - 10:
            a_f, sc_f = _mirror_descent(best_seq, p4_end - 5, rng,
                                        lr_init=0.2, lr_decay=0.99999,
                                        beta_start=200.0, beta_end=500.0,
                                        n_phases=4)
            update_best(a_f, sc_f, "final_mirror")

        # Last peak shaving
        if time.monotonic() < p4_end - 3:
            a_ps2, sc_ps2 = _peak_shaving(best_seq, p4_end - 1, rng, max_rounds=1000)
            update_best(a_ps2, sc_ps2, "peak_shave_final")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(4096)
        sc_final = _score(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, norm_score={best_score:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    fallback = _make_ramp(4096)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
