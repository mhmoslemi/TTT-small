"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS on smooth log-sum-exp surrogate, periodic LP corrections,
multi-scale optimization, and adaptive multi-restart.

import numpy as np

from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_safe(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(f, n):
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(f, M, time_limit=20):
    n = len(f)
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _convolve_gradient(a, w):
    """Compute gradient of sum_k w_k * conv(a,a)_k w.r.t. a.
    grad_i = 2 * sum_{k: 0<=k-i<n} w_k * a_{k-i}
    """
    n = len(a)
    # w has length 2n-1
    # For each i, grad_i = 2 * sum_{j=0}^{n-1} w_{i+j} * a_j  (where i+j < 2n-1)
    # This is a convolution: grad = 2 * conv(w, a_reversed)[n-1:2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h


def _smooth_objective_and_grad(z, beta):
    """Compute smooth surrogave and gradient in log-space.
    z is log(a) (unnormalized), a = softmax(z).
    Returns (objective, grad_z) where objective = beta * log(sum(exp(c_k/beta)))
    and grad_z is the gradient w.r.t. z.
    """
    a = _softmax_safe(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        obj = cm
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = cm + beta * np.log(s)
        w = e / s
    
    # Gradient of F(a) = sum_k w_k * c_k w.r.t. a
    grad_a = _convolve_gradient(a, w)
    
    # Gradient w.r.t. z: grad_z = grad_a * a - a * sum(grad_a * a)
    # (Jacobian of softmax)
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z

def _make_uniform(n):

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_safe(z)


def _structured_perturb(a, rng):
    """Apply structured perturbations: circular shift, flip, scale segments."""
    n = len(a)
    result = a.copy()
    op = rng.integers(0, 4)
    if op == 0:
        # Circular shift
        shift = rng.integers(1, n // 4)
        result = np.roll(result, shift)
    elif op == 1:
        # Flip
        result = result[::-1]
    elif op == 2:
        # Scale a random segment
        start = rng.integers(0, n // 2)
        length = rng.integers(n // 8, n // 2)
        scale = rng.uniform(0.5, 2.0)
        result[start:start+length] *= scale
    else:
        # Multiply by a smooth random function
        freq = rng.uniform(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi * freq, n)
        mod = 1.0 + 0.3 * np.sin(x + phase)
        result = result * mod
    result = np.maximum(result, 1e-12)
    return result / result.sum()

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS on smooth log-sum-exp surrogate, LP-based guaranteed descent,
adaptive multi-scale optimization, and structured multi-restart.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_safe(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(f, n, support=None):
    if support is not None and len(support) < n:
        idx = support
    else:
        idx = np.arange(n)
    k = np.arange(2 * n - 1)
    j = idx
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, len(idx)))


def _lp_solve(f, M, support, time_limit=25):
    n = len(f)
    if support is None:
        support = np.arange(n)
    A = _build_lp_matrix(f, n, support)
    c_obj = -np.ones(len(support))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * len(support)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g_full = np.zeros(n)
            g_full[support] = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g_full.sum() > 1e-15:
                return g_full
    except Exception:
        pass
    return None


def _convolve_gradient_weights(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _smooth_obj_grad(z, beta):
    a = _softmax_safe(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        obj = cm
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = cm + beta * np.log(s)
        w = e / s
    grad_a = _convolve_gradient_weights(a, w)
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    return obj, grad_z


def _adam_smooth(a_init, deadline, lr=0.01, beta_init=0.015):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = best_score

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    beta_phases = [1.0, 0.6, 0.35, 0.2, 0.1, 0.05, 0.02, 0.008]
    lr_phases = [1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.12, 0.06]
    total_time = deadline - time.monotonic()
    if total_time <= 0:
        return best_a, best_score

    phase_fracs = np.array([0.18, 0.16, 0.15, 0.14, 0.12, 0.11, 0.09, 0.05])
    phase_fracs = phase_fracs / phase_fracs.sum()

    cumulative_frac = 0.0
    for phase_idx in range(len(beta_phases)):
        if time.monotonic() >= deadline:
            break
        phase_frac = phase_fracs[phase_idx]
        phase_end = t_start_global + (cumulative_frac + phase_frac) * total_time
        phase_end = min(phase_end, deadline)
        cumulative_frac += phase_frac

        beta = beta_phases[phase_idx] * max_c_init
        current_lr = lr * lr_phases[phase_idx]

        local_old_score = best_score
        old_score_time = time.monotonic()

        while time.monotonic() < phase_end:
            t_step += 1
            a = _softmax_safe(z)
            obj, grad_z = _smooth_obj_grad(z, beta)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.8:
                update = update * (0.8 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t_step % 200 == 0:
                a_check = _softmax_safe(z)
                c_check = _full_convolve(a_check, a_check)
                sc = float(c_check.max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
                # Adaptive: if no improvement for a while, reduce beta
                if best_score == local_old_score and time.monotonic() - old_score_time > max(5.0, 0.15 * (phase_end - old_score_time)):
                    beta *= 0.75
                    current_lr *= 0.8
                    old_score_time = time.monotonic()

    a_final = _softmax_safe(z)
    c_final = _full_convolve(a_final, a_final)
    sc_final = float(c_final.max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _lbfgs_optimize(a_init, deadline, beta=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = best_score
    beta_val = beta * max_c_init

    call_count = [0]

    def obj_grad(z_local):
        call_count[0] += 1
        obj, grad = _smooth_obj_grad(z_local, beta_val)
        if call_count[0] % 20 == 0:
            if time.monotonic() > deadline:
                raise KeyboardInterrupt
        return obj, grad

    # Run L-BFGS in chunks
    chunk_size = 300
    for _ in range(100):
        if time.monotonic() >= deadline:
            break
        try:
            result = minimize(obj_grad, z, method='L-BFGS-B',
                             jac=True, options={'maxiter': chunk_size,
                                               'gtol': 1e-10, 'ftol': 1e-15})
            z = result.x
            z = z - z.mean()
            z = np.clip(z, -50, 50)
        except (KeyboardInterrupt, Exception):
            break

        a_check = _softmax_safe(z)
        c_check = _full_convolve(a_check, a_check)
        sc = float(c_check.max())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_check.copy()

    return best_a, best_score


def _make_uniform(n):
    return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_safe(z)


def _structured_perturb(a, rng):
    n = len(a)
    result = a.copy()
    op = rng.integers(0, 5)
    if op == 0:
        shift = rng.integers(1, max(2, n // 4))
        result = np.roll(result, shift)
    elif op == 1:
        result = result[::-1]
    elif op == 2:
        start = rng.integers(0, n // 3)
        length = rng.integers(n // 6, n // 2)
        scale = rng.uniform(0.4, 2.5)
        result[start:start + length] *= scale
    elif op == 3:
        freq = rng.uniform(1.0, 8.0)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi * freq, n)
        mod = 1.0 + 0.4 * np.sin(x + phase)
        result = result * mod
    else:
        result = result * (1.0 + 0.2 * rng.standard_normal(n))
    result = np.maximum(result, 1e-12)
    return result / result.sum()


def _lp_phase(f, n, deadline, update_best):
    c = _full_convolve(f, f)
    M = float(c.max())
    current_t = 0.4
    lp_steps = 0
    stuck_count = 0
    support = np.arange(n)
    A_cache = None
    f_for_matrix = None

    while time.monotonic() < deadline - 20:
        if time.monotonic() >= deadline - 20:
            break
        c = _full_convolve(f, f)
        M = float(c.max())

        # Build LP matrix
        if A_cache is None or f_for_matrix is None or np.max(np.abs(f - f_for_matrix)) > 1e-9:
            A_cache = _build_lp_matrix(f, n, support)
            f_for_matrix = f.copy()

        g_local = None
        try:
            c_obj = -np.ones(len(support))
            b_ub = np.full(2 * n - 1, M)
            bounds_lp = [(0, None)] * len(support)
            res = linprog(c_obj, A_ub=A_cache, b_ub=b_ub, bounds=bounds_lp,
                          method='highs', options={'time_limit': 25, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g_full = np.zeros(n)
                g_full[support] = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
                if g_full.sum() > 1e-15:
                    g_local = g_full
        except Exception:
            g_local = None

        if g_local is None:
            stuck_count += 1
            if stuck_count > 8:
                break
            g_local = _perturb_log(f, 0.02, np.random.default_rng(lp_steps))
            g_local = g_local / g_local.sum()

        g_sum = float(g_local.sum())
        if g_sum < 1e-15:
            break
        g_norm = g_local / g_sum

        # Backtracking line search
        t_try = current_t
        improved = False
        for _ in range(8):
            f_new = (1 - t_try) * f + t_try * g_norm
            f_new = np.maximum(f_new, 0)
            f_new_sum = float(f_new.sum())
            if f_new_sum < 1e-15:
                t_try *= 0.5
                continue
            f_new = f_new / f_new_sum

            c_new = _full_convolve(f_new, f_new)
            M_new = float(c_new.max())

            if M_new < M * (1 - 1e-10):
                f = f_new
                current_t = min(1.0, t_try * 1.4)
                improved = True
                lp_steps += 1
                update_best(f)
                print(f"  LP step {lp_steps}: M={M:.12f} -> {M_new:.12f} "
                      f"score={_score(f):.10f} t={time.monotonic()-t_start_global:.1f}s")
                break
            else:
                t_try *= 0.5

        if not improved:
            current_t *= 0.5
            stuck_count += 1
            if stuck_count > 12:
                break

        # Update support periodically
        if lp_steps > 0 and lp_steps % 5 == 0:
            threshold = 0.005 * np.max(g_local) if g_local.any() else 0
            new_support = np.where(g_local > threshold)[0]
            if len(new_support) < n // 2:
                new_support = np.where(g_local > 0.001 * np.max(g_local))[0]
            if len(new_support) >= 100:
                support = new_support
                A_cache = None
                f_for_matrix = None
                print(f"  Support reduced to {len(support)}")

    return f, lp_steps
