    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_solve(f):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if g.value is not None and t.value is not None:
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 1e-15:
                g_val = g_val / gs
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate_bt(f, n_iters, deadline, step_init=0.4):
    """LP iteration with backtracking line search and periodic perturbation."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    a_cur = f.copy()
    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_cur, a_cur)
        M = float(c.max())

        g, t_val = _lp_solve(a_cur)
        if g is None:
            break

        # Backtracking line search
        alpha = step_init * (1.0 / (1.0 + 0.02 * it))
        best_trial_score = _score_norm_fn(a_cur)
        best_trial_a = a_cur.copy()

        for trial in range(12):
            a_trial = (1.0 - alpha) * a_cur + alpha * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            sc = _score_norm_fn(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial_a = a_trial.copy()
            alpha *= 0.6
            if alpha < 0.003:
                break

        a_cur = best_trial_a
        if best_trial_score < best_score:
            best_score = best_trial_score
            best_a = best_trial_a.copy()

        # Periodic perturbation to escape local minima
        if it > 0 and it % 40 == 0:
            noise_scale = 0.02 * (1.0 - float(it) / max(1, n_iters))
            z = np.log(np.maximum(a_cur, 1e-15))
            z = z + noise_scale * rng.normal(size=n)
            z = z - z.mean()
            a_pert = _softmax_fn(z)
            sc_pert = _score_norm_fn(a_pert)
            if sc_pert < best_trial_score:
                a_cur = a_pert
                best_trial_score = sc_pert
                if sc_pert < best_score:
                    best_score = sc_pert
                    best_a = a_pert.copy()

        if it % 20 == 0:
            print(f"    LP iter {it}: score={_score_norm_fn(a_cur):.10f}, "
                  f"best={best_score:.10f}, t*={t_val:.8f}, gap={M - t_val:.6f}")

    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _lbfgs_continuation(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                 0.0002, 0.00015, 0.0001, 0.00008, 0.00005,
                 0.00003, 0.00002, 0.000015, 0.00001]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        maxiter = max(200, int(time_per * 40))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                         'maxls': 60, 'maxfun': maxiter * 6}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
                        if idx % 5 == 0:
                            print(f"      L-BFGS beta_idx={idx}: score={best_score:.10f}")
        except Exception:
            pass

    return best_a, best_score


def _single_beta_lbfgs(a_init, beta, deadline, maxiter=10000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    try:
        result = minimize(
            _logsumexp_obj, z, args=(beta, n),
            method='L-BFGS-B', jac=_logsumexp_grad,
            options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                     'maxls': 60, 'maxfun': maxiter * 6}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            s_new = _score_norm_fn(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_score


def _nesterov_subgrad(a_init, deadline, max_iter=30000):
    """Nesterov-accelerated subgradient on max(conv(a,a)) with averaging."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    t_k = 1.0
    running_sum = np.zeros(n)
    n_accepted = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        theta = (1.0 - t_k) / t_new

        a_y = a + theta * (a - a_prev)
        a_y = np.maximum(a_y, 0)
        sy = a_y.sum()
        if sy < 1e-15:
            break
        a_y = a_y / sy

        # Compute subgradient of max
        c = _full_convolve(a_y, a_y)
        M = float(c.max())
        active = c >= (M - 0.005 * M)
        n_active = int(active.sum())
        if n_active == 0:
            idx_max = int(np.argmax(c))
            active[idx_max] = True
            n_active = 1

        active_indices = np.where(active)[0]
        g = np.zeros(n)
        for k in active_indices:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                g[j_lo:j_hi + 1] += 2.0 * a_y[k - j_hi:k - j_lo + 1]
        g /= n_active

        # Step size
        step = 2.0 / (t_k + 1.0) / np.sqrt(n)
        a_new = a_y - step * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        # Track running average
        running_sum += a_new
        n_accepted += 1
        a_avg = running_sum / n_accepted
        sc_avg = _score_norm_fn(a_avg)

        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

        a_prev = a
        a = a_new
        t_k = t_new

        if it % 10000 == 0 and it > 0:
            print(f"      Nesterov iter {it}: score={sc:.10f}, avg={sc_avg:.10f}, best={best_score:.10f}")

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=50):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        M = c.max()
        # Focus on coordinates where gradient is largest
        active = c >= (M - 0.01 * M)
        active_indices = np.where(active)[0]
        # Compute which a[j] are most involved in active lags
        involvement = np.zeros(n)
        for k in active_indices[:50]:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            involvement[j_lo:j_hi + 1] += 1.0
        # Prioritize coordinates with high involvement
        order = np.argsort(-involvement)
        if involvement.max() > 0:
            order = np.concatenate([order, np.setdiff1d(np.arange(n), order)])

        improved = False
        for j in order[:min(n, 2000)]:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local_score = _score_norm_fn(a)
            for delta in [-0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 2048
        parent = np.ones(n_init) / n_init
        update_best(parent, _score_norm_fn(parent), "default")

    n_parent = len(parent)

    # Phase 1: Heavy LP iteration (600s)
    phase1_end = min(deadline, t_start + 600.0)
    print(f"\nPhase 1: Heavy LP iteration (n={n_parent})")

    n_lp = n_parent
    a_lp = parent.copy()

    for lp_round in range(8):
        if time.monotonic() > phase1_end:
            break
        step = max(0.1, 0.5 / (lp_round + 1))
        a_lp, sc_lp = _lp_iterate_bt(a_lp, n_iters=50, deadline=phase1_end, step_init=step)
        update_best(a_lp, sc_lp, f"lp_round{lp_round}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: L-BFGS continuation (450s)
    phase2_end = min(deadline, t_start + 1050.0)
    print(f"\nPhase 2: L-BFGS continuation (n={len(best_seq)})")

    n_restarts = 4
    per_restart_time = (phase2_end - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase2_end - 30:
            break

        if r == 0:
            a_cur = best_seq.copy()
        else:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.03 * (r + 1)
            z = z + noise * rng.normal(size=len(z))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax_fn(z)

        per_dl = min(phase2_end - 10, time.monotonic() + per_restart_time)
        p_best, sc_best = _lbfgs_continuation(a_cur, per_dl)
        update_best(p_best, sc_best, f"lbfgs_r{r}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Nesterov subgradient (300s)
    phase3_end = min(deadline, t_start + 1350.0)
    print(f"\nPhase 3: Nesterov subgradient")

    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()))
        p_best, sc_best = _nesterov_subgrad(best_seq, per_dl, max_iter=50000)
        update_best(p_best, sc_best, "nesterov")

        # Follow up with L-BFGS on the nesterov result
        if time.monotonic() < phase3_end - 20 and best_seq is not None:
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            per_dl2 = min(phase3_end - 5, time.monotonic() + 60.0)
            p_best2, sc_best2 = _single_beta_lbfgs(best_seq, 0.0002 * max_c_final, per_dl2, maxiter=2000)
            update_best(p_best2, sc_best2, "post_nesterov_lbfgs")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Multi-n exploration + polish (remaining time)
    phase4_end = deadline
    print(f"\nPhase 4: Multi-n exploration + polish")

    if time.monotonic() < phase4_end and best_seq is not None:
        # Try different n values
        n_alternatives = [3072, 5120]
        n_alternatives = [x for x in n_alternatives if x != len(best_seq)]

        for n_alt in n_alternatives:
            if time.monotonic() > phase4_end - 60:
                break
            p_interp = _interp_profile(best_seq, n_alt)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{n_alt}")

            # Quick LP at this n
            if time.monotonic() < phase4_end - 40:
                per_dl = min(phase4_end - 30, time.monotonic() + 80.0)
                p_best, sc_best = _lp_iterate_bt(p_interp, n_iters=30, deadline=per_dl, step_init=0.3)
                update_best(p_best, sc_best, f"lp_n{n_alt}")

                if time.monotonic() < phase4_end - 30:
                    per_dl2 = min(phase4_end - 20, time.monotonic() + 50.0)
                    p_best2, sc_best2 = _lbfgs_continuation(p_best, per_dl2)
                    update_best(p_best2, sc_best2, f"lbfgs_n{n_alt}")

        # Coordinate descent polish on best
        if time.monotonic() < phase4_end - 20 and best_seq is not None:
            per_dl = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic()) * 0.6)
            p_best, sc_best = _coord_descent(best_seq, per_dl, max_rounds=30)
            update_best(p_best, sc_best, "coord_descent")

        # Final ultra-fine L-BFGS
        if time.monotonic() < phase4_end - 10 and best_seq is not None:
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            per_dl2 = min(phase4_end - 5, time.monotonic() + 40.0)
            p_best2, sc_best2 = _single_beta_lbfgs(best_seq, 5e-5 * max_c_final, per_dl2, maxiter=5000)
            update_best(p_best2, sc_best2, "ultra_fine")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 6000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
