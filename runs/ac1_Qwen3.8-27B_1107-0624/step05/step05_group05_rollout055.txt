"""
Optimized Adam with adaptive temperature annealing for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses FFT-based gradient, active-set
refinement, and multi-phase optimization strategy.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_soft_gradient(a, c, T):
    """
    Compute gradient of 2*n*T*log(sum exp(c/T)) w.r.t. a.
    Returns gradient of shape (n,).
    p_k = exp(c_k/T) / sum(exp(c/T))
    g[i] = 2*n * sum_k p_k * 2*a[k-i] (valid indices)
         = 4*n * sum_j p[i+j] * a[j]
    """
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    
    # g[i] = 4*n * sum_j p[i+j] * a[j]
    # = 4*n * conv_full[n-1:2n-1][::-1] where conv_full = conv(a, p[::-1])
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n-1:2*n-1][::-1]
    return g

def _compute_active_gradient(a, c, top_k=10):
    """
    Compute subgradient using top-k active lags.
    g[i] = (2/n_active) * sum_{k in active} 2*n * 2*a[k-i]
         = (4*n/n_active) * sum_{k in active} a[k-i]
    """
    n = len(a)
    # Find top-k lags
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n-1:2*n-1][::-1]
    return g

def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)

def _lp_step(a, rows, cols, time_limit=30):
    """LP step: maximize sum(g) subject to conv(a,g) <= M, g >= 0."""
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()
        # Line search
        best_t = 0.0
        best_sc = _score_norm(a)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t) * a + t * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        if best_t == 0.0:
            for t in [0.02, 0.05, 0.08]:
                a_cand = (1.0 - t) * a + t * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a, M
    except Exception:
        return a, M

def _adam_z_space(a, deadline, T_start, T_end, lr_start, lr_end, 
                  max_iter=500000, use_active_set=False, active_k=5):
    """Adam optimization in z-space with annealed temperature."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Current temperature
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        if use_active_set and T < T_start * 0.01:
            g = _compute_active_gradient(a, c, active_k)
        else:
            g = _compute_soft_gradient(a, c, T)
        
        # Project to tangent space of simplex (in z-space)
        # z_grad[i] = a[i] * (g[i] - sum_j a[j]*g[j])
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        # Adam update
        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1**t)
        vh = v_buf / (1 - b2**t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip update
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        # Periodic score evaluation
        if t % 500 == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _fista(a_init, deadline, T_start, T_end, max_iter=100000):
    """FISTA with simplex projection."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    
    # Lipschitz constant estimate (conservative)
    L = 4.0 * n * 2.0  # rough upper bound
    step = 1.0 / L
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        T = T_start * (T_end / T_start) ** (t / max_iter)
        c = _fft_convolve(y, y)
        g = _compute_soft_gradient(y, c, T)
        
        # Gradient step
        a_new = y - step * g
        
        # Project onto simplex
        a_new = _project_simplex(a_new)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k**2))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        
        a = a_new
        
        if t % 500 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient computation, LP for structure, Adam with adaptive
temperature annealing in z-space, active-set gradients for final polishing,
and multi-scale restarts (n=2048 and n=4096) for diversity.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=5):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols, time_limit=20):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t == 0.0:
            for t_val in [0.02, 0.05, 0.08]:
                a_cand = (1.0 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a, M
    except Exception:
        return a, M


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _adam_z_space(a, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, use_active_set=False, active_k=5,
                  eval_interval=300, clip_norm=0.8):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    best_patience_score = best_score
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if use_active_set and T < T_start * 0.005:
            g = _compute_active_gradient(a, c, active_k)
        else:
            g = _compute_soft_gradient(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        if update_norm > clip_norm:

        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:

                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1

                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)

def _fista_simplex(a_init, deadline, T_start, T_end, max_iter=100000):
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    L_init = 8.0 * n
    step = 1.0 / L_init
    eval_interval = 500

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        c = _fft_convolve(y, y)
        g = _compute_soft_gradient(y, c, T)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-10:
            step = step * min(2.0, max(0.5, 1.0 / (step * g_norm + 1e-10)))
        step = min(step, 2.0 / L_init)
        step = max(step, 1e-10)
        a_new = y - step * g
        a_new = _project_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            break
        prev_sc = sc
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: FISTA with annealing (120s - 400s)
    phase2_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase2_end:
        c_cur = _fft_convolve(a, a)
        max_c = float(c_cur.max())
        a_fista, sc_fista = _fista_simplex(a, phase2_end,
                                            T_start=0.01 * max_c, T_end=0.0001 * max_c,
                                            max_iter=150000)
        full_sc = _score(a_fista)
        print(f"Phase 2 FISTA: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_fista.copy()

    # Phase 3: Adam z-space, soft gradient, medium annealing (400s - 800s)
    phase3_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase3_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_adam, sc_adam = _adam_z_space(best_a, phase3_end - 5,
                                         T_start=0.005 * max_c, T_end=0.00005 * max_c,
                                         lr_start=0.04, lr_end=0.005,
                                         max_iter=400000, eval_interval=200)
        full_sc = _score(a_adam)
        print(f"Phase 3 Adam-soft: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_adam.copy()

    # Phase 4: Adam z-space, active-set gradient, low temperature (800s - 1200s)
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_active, sc_active = _adam_z_space(best_a, phase4_end - 5,
                                             T_start=0.001 * max_c, T_end=0.00001 * max_c,
                                             lr_start=0.02, lr_end=0.002,
                                             max_iter=400000,
                                             use_active_set=True, active_k=5,
                                             eval_interval=200)
        full_sc = _score(a_active)
        print(f"Phase 4 Adam-active: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_active.copy()

    # Phase 5: Multi-scale exploration - try n=4096 (1200s - 1450s)
    phase5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase5_end:
        n_large = 4096
        a_large = _interp_to_length(best_a, n_large)
        c_large = _fft_convolve(a_large, a_large)
        max_cl = float(c_large.max())
        a_large_opt, sc_large = _adam_z_space(a_large, phase5_end - 5,
                                               T_start=0.002 * max_cl, T_end=0.00002 * max_cl,
                                               lr_start=0.03, lr_end=0.003,
                                               max_iter=200000,
                                               use_active_set=True, active_k=5,
                                               eval_interval=200)
        full_sc = _score(a_large_opt)
        print(f"Phase 5 n=4096 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_large_opt.copy()
        # Also try interpolating back to n=2048 from the large solution
        if full_sc < 2.0:
            a_interp = _interp_to_length(a_large_opt, n)
            c_int = _fft_convolve(a_interp, a_interp)
            max_ci = float(c_int.max())
            a_interp_opt, sc_int = _adam_z_space(a_interp, phase5_end - 5,
                                                   T_start=0.001 * max_ci, T_end=0.00001 * max_ci,
                                                   lr_start=0.02, lr_end=0.002,
                                                   max_iter=100000,
                                                   use_active_set=True, active_k=3,
                                                   eval_interval=200)
            full_sc_int = _score(a_interp_opt)
            print(f"Phase 5 interp back n={n}: score={full_sc_int:.10f}")
            if np.isfinite(full_sc_int) and full_sc_int < best_score:
                best_score = full_sc_int
                best_a = a_interp_opt.copy()

    # Phase 6: Perturbation restarts on best (1450s - 1650s)
    phase6_end = min(deadline, t_start + 1650.0)
    n_best = len(best_a)
    if time.monotonic() < phase6_end:
        for attempt in range(5):
            if time.monotonic() > phase6_end - 40:
                break
            sigma = 0.015 + 0.008 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            a_pert_opt, sc_pert = _adam_z_space(a_pert, min(phase6_end - 5, t_start + 1650.0),
                                                 T_start=0.001 * max_cp, T_end=0.00003 * max_cp,
                                                 lr_start=0.025, lr_end=0.002,
                                                 max_iter=80000,
                                                 use_active_set=True, active_k=3,
                                                 eval_interval=150)
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # Phase 7: Final ultra-fine polish (1650s - deadline)
    if time.monotonic() < deadline - 8:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_final = _adam_z_space(best_a, deadline - 5,
                                           T_start=0.0001 * max_cf, T_end=0.000002 * max_cf,
                                           lr_start=0.004, lr_end=0.0003,
                                           max_iter=300000,
                                           use_active_set=True, active_k=3,
                                           eval_interval=100)
        full_sc = _score(a_final)
        print(f"Phase 7 final polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
