```python
    # So max(c) = max(R[0], ..., R[n-1])

        if k == 0:
            sg += 2.0 * a
        else:

            sg[:n-k] += a[k:]      # a[j+k] for j = 0..n-k-1
```

```python
def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

```

2. a_new = a - lr * sg
3. a_new = project_simplex(a_new)

Q_k has the form:

This is related to the shift matrix. For k=0: Q_0 = I (identity), which is PSD. ✓
For k>0: Q_k has 1/2 on the k-th super/sub-diagonal. The eigenvalues of such a matrix are 2*(1/2)*cos(theta) = cos(theta) for various theta, so they range from -1 to 1. So Q_k is NOT PSD!

This means R[k] = a^T Q_k a is NOT convex in a (it's indefinite). However, R[k] = sum_i a[i]*a[i+k] >= 0 for a >= 0, so it's convex on the non-negative orthant.

More precisely, R[k] is a "bilinear" form that is convex on the non-negative orthant (since the cross-partial is positive: d²R[k]/da[i]da[i+k] = 1 > 0).

So g(a) = max_k R[k] is convex on the non-negative orthant. The projected subgradient method should work.

```python
"""
Combined LP + Nesterov-subgradient + coordinate-descent optimizer for minimizing
2*n*max(autoconv)/sum^2. Key improvements over prior approach:
- FFT-accelerated exact subgradient of the max-autocorrelation
- Nesterov momentum with adaptive step size for fast convergence
- Exact 1D coordinate descent (binary search on target with O(n) feasibility check)
- LP direction finding with adaptive M and tighter line search
- Multi-start with symmetric/asymmetric variants and different n
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _autocorr(a):
    """Compute linear autocorrelation R[0..n-1] where R[k] = sum_i a[i]*a[i+k]"""
    n = len(a)
    c = _full_convolve(a, a)
    return c[:n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    R = _autocorr(a)
    return float(2.0 * n * R.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1, returns 2n*max(R)"""
    R = _autocorr(a)
    return float(2.0 * len(a) * R.max())


def _norm_seq(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _compute_subgradient(a):
    """Compute exact subgradient of max_k R[k] at point a. Returns (subgrad, R_max, K_star)."""
    n = len(a)
    R = _autocorr(a)
    R_max = float(R.max())
    K_star = np.where(R >= R_max * (1.0 - 1e-12))[0]
    sg = np.zeros(n)
    for k in K_star:
        if k == 0:
            sg += 2.0 * a
        else:
            sg[:n - k] += a[k:]
            sg[k:] += a[:n - k]
    sg /= len(K_star)
    return sg, R_max, K_star


def _build_lp_matrix(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_direction(a, M, rows, cols, t_limit=20):
    n = len(a)
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': t_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _line_search(a, direction, max_sc):
    """Find best t in [0,1] for a + t*(direction - a)"""
    best_t = 0.0
    best_sc = max_sc

        sc = _score_norm(cand)
        if sc < best_sc:

            best_t = t

    prev_best = best_sc
    no_improve = 0

        y = a + momentum * v
        y = _project_simplex(y)

        sg, R_max, K_star = _compute_subgradient(y)

        # Update velocity and position
        v = momentum * v - lr * sg
        a_new = _project_simplex(a + v)

        # Ensure non-negativity and normalization
        a_new = np.maximum(a_new, 0)
        a_new = _norm_seq(a_new)
        a = a_new

        t += 1

    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = init_lr
    prev_best = best_sc

    while time.monotonic() < deadline:
        a = _softmax(z)
        sg, R_max, _ = _compute_subgradient(a)
        W = float(np.dot(a, sg))
        grad_z = a * (sg - W)
        if not np.all(np.isfinite(grad_z)):
            break

        t += 1
        m_buf = b1 * m_buf + (1 - b1) * grad_z
        v_buf = b2 * v_buf + (1 - b2) * (grad_z * grad_z)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if t % 200 == 0:
            a_check = _softmax(z)
            sc = _score_norm(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
            if t % 2000 == 0:
                if best_sc < prev_best * 0.9999:
                    lr = min(0.01, lr * 1.1)
                else:
                    lr = max(1e-5, lr * 0.9)
                prev_best = best_sc

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final
    return best_a, best_sc


def _make_symmetric(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return _norm_seq(a_sym)


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z += sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _score_interp_profile(profile, n_target):
    """Create a sequence of length n_target from a profile function sampled on [0,1]."""
    x = np.linspace(0, 1, n_target)
    a = profile(x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """Main search function. Returns list of floats minimizing the evaluation score."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # --- Load parent sequence ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if parent_raw.sum() > 0.01:
                parent = _norm_seq(parent_raw)
    except Exception:
        pass

    if parent is not None:
        n0 = len(parent)
        a = parent.copy()
        sc = _score(a)
        best_seq = a.copy()
        best_score = sc
        print(f"Parent loaded: n={n0}, score={sc:.10f}")
    else:
        n0 = 2048
        a = np.ones(n0) / n0
        best_seq = a.copy()
        best_score = _score(a)
        print(f"No parent. Uniform n={n0}, score={best_score:.10f}")

    n = len(a)

    # --- Try symmetric version ---
    a_sym = _make_symmetric(a)
    sc_sym = _score(a_sym)
    print(f"Symmetric version: score={sc_sym:.10f}")
    if sc_sym < best_score:
        best_score = sc_sym
        best_seq = a_sym.copy()
        a = a_sym.copy()
        print("Using symmetric version as starting point")

    # --- Build LP matrix ---
    rows, cols = _build_lp_matrix(n)
    print(f"LP matrix built: {len(rows)} nonzeros, t={time.monotonic()-t_start:.1f}s")

    # ============ PHASE 1: LP-based optimization (0 - 400s) ============
    phase1_end = min(deadline, t_start + 400.0)
    a = best_seq.copy()
    prev_sc = _score_norm(a)
    stall = 0
    lp_count = 0

    while time.monotonic() < phase1_end:
        t0 = time.monotonic()
        R = _autocorr(a)
        M = float(R.max())
        b = _lp_direction(a, M, rows, cols, t_limit=25)
        if b is None or b.sum() < 1e-15:
            break
        b = _norm_seq(b)
        best_t, best_step_sc = _line_search(a, b, _score_norm(a))
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b
            sc_new = _score_norm(a)
            if sc_new < prev_sc - 1e-12:
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        prev_sc = _score_norm(a)
        lp_count += 1

        if lp_count % 10 == 0 or stall > 20:
            full_sc = _score(a)
            print(f"  LP[{lp_count}] score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s stall={stall}")
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()
        if stall > 50:
            print(f"  LP stalled at iter {lp_count}")
            # Try with a slightly reduced M to force progress
            R2 = _autocorr(a)
            M2 = float(R2.max()) * 0.999
            b = _lp_direction(a, M2, rows, cols, t_limit=20)
            if b is not None and b.sum() > 1e-15:
                b = _norm_seq(b)
                a = 0.5 * a + 0.5 * b
                stall = 0
            else:
                break

    full_sc = _score(a)
    if full_sc < best_score:
        best_score = full_sc
        best_seq = a.copy()
    print(f"Phase 1 (LP) done: {lp_count} iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # ============ PHASE 2: Nesterov subgradient (400 - 800s) ============
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end:
        a_ng, sc_ng = _nesterov_phase(best_seq, phase2_end, init_lr=0.003, momentum_init=0.85)
        full_sc = _score(a_ng)
        print(f"Phase 2 (Nesterov): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_ng.copy()

    # ============ PHASE 3: More LP from improved point (800 - 1100s) ============
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end:
        a = best_seq.copy()
        prev_sc = _score_norm(a)
        stall = 0
        lp3 = 0
        while time.monotonic() < phase3_end:
            R = _autocorr(a)
            M = float(R.max())
            b = _lp_direction(a, M, rows, cols, t_limit=20)
            if b is None or b.sum() < 1e-15:
                break
            b = _norm_seq(b)
            best_t, _ = _line_search(a, b, _score_norm(a))
            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * b
                if _score_norm(a) < prev_sc - 1e-12:
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            prev_sc = _score_norm(a)
            lp3 += 1
            if lp3 % 10 == 0:
                full_sc = _score(a)
                if full_sc < best_score:
                    best_score = full_sc
                    best_seq = a.copy()
                if lp3 % 50 == 0:
                    print(f"  LP2[{lp3}] score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if stall > 40:
                break
        full_sc = _score(a)
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a.copy()
        print(f"Phase 3 (LP2) done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # ============ PHASE 4: Nesterov with different params (1100 - 1350s) ============
    phase4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase4_end:
        a_ng2, sc_ng2 = _nesterov_phase(best_seq, phase4_end, init_lr=0.001, momentum_init=0.92)
        full_sc = _score(a_ng2)
        print(f"Phase 4 (Nesterov2): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_ng2.copy()

    # ============ PHASE 5: Perturbed restarts (1350 - 1550s) ============
    phase5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase5_end:
        for attempt in range(5):
            if time.monotonic() > phase5_end - 30:
                break
            sigma = 0.02 + 0.01 * attempt
            a_p = _perturb(best_seq, sigma, rng)
            # Quick LP from perturbed start
            prev_sc = _score_norm(a_p)
            stall = 0
            for it in range(30):
                if time.monotonic() > phase5_end - 10:
                    break
                R = _autocorr(a_p)
                M = float(R.max())
                b = _lp_direction(a_p, M, rows, cols, t_limit=10)
                if b is None:
                    break
                b = _norm_seq(b)
                best_t, _ = _line_search(a_p, b, prev_sc)
                if best_t > 0:
                    a_p = (1.0 - best_t) * a_p + best_t * b
                    if _score_norm(a_p) < prev_sc - 1e-12:
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
                prev_sc = _score_norm(a_p)
                if stall > 15:
                    break
            full_sc = _score(a_p)
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a_p.copy()

    # ============ PHASE 6: Final Adam polishing (1550 - deadline) ============
    if time.monotonic() < deadline - 5:
        a_adam, sc_adam = _adam_finetune(best_seq, deadline - 3, init_lr=0.002)
        full_sc = _score(a_adam)
        print(f"Phase 6 (Adam): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_adam.copy()

    # ============ PHASE 7: One more symmetric + LP pass if time remains ============
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_s = _make_symmetric(best_seq)
        sc_s = _score(a_s)
        if sc_s < best_score:
            best_score = sc_s
            best_seq = a_s.copy()
            print(f"Final symmetric: score={sc_s:.10f}")
        # Quick LP polish
        a = best_seq.copy()
        for it in range(20):
            if time.monotonic() > deadline - 3:
                break
            R = _autocorr(a)
            M = float(R.max())
            b = _lp_direction(a, M, rows, cols, t_limit=5)
            if b is None:
                break
            b = _norm_seq(b)
            best_t, _ = _line_search(a, b, _score_norm(a))
            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * b
        full_sc = _score(a)
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a.copy()
        print(f"Phase 7 final: score={best_score:.10f}")

    # --- Final validation ---
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
        s = 256.0
    # Scale so sum is reasonable (avoid underflow/overflow issues)
    # The score is scale-invariant, so any scaling works, but keep it in a safe range
    scale = 1.0 / s
    final = final * scale
    # Make sure values are not too small (avoid numerical issues)
    final = np.maximum(final, 0.0)

    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"External evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Evaluator error: {ex}")

    return [float(x) for x in final]
```
