import numpy as np
import cvxpy as cp
from scipy.sparse import csr_matrix

def solve_alternating_lp(a, n):
    """Solve LP: min t s.t. conv(a, b) <= t, b >= 0, sum(b) = sum(a)"""
    S = np.sum(a)
    b = cp.Variable(n)
    t = cp.Variable()
    
    # Build constraint matrix C where C[k, j] = a[k-j] if valid
    # We need C @ b <= t * ones
    # C is (2n-1) x n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i  # b index
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    
    C = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [
        C @ b <= t * np.ones(2*n-1),
        b >= 0,
        cp.sum(b) == S
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.GLPK)
    
    if b.value is not None:
        return b.value
    return a  # fallback

    return csr_matrix((vals, (k_idx, j_idx)), shape=(2 * n - 1, n))


def _lp_step(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0, sum(b) = sum(a)."""
    n = len(a)
    S = float(a.sum())
    C = _build_C_sparse(a)
    res = linprog(
        c=-np.ones(n),
        A_ub=C,
        b_ub=np.full(2 * n - 1, M),
        A_eq=np.ones((1, n)),
        b_eq=np.array([S]),
        bounds=[(0, None)] * n,
        method='highs',
        options={'time_limit': 60, 'output_flag': False}
    )
    if res.success:
        b = np.maximum(res.x, 0)
        s_b = b.sum()
        if s_b > 1e-15:
            b = b * (S / s_b)
        return b
    return a.copy()


def _alternating_lp(a_init, deadline, max_iter=2000):
    """Frank-Wolfe style alternating LP optimization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-15:
        return a_init.copy()
    a = a / S

    best_a = a.copy()
    best_sc = _score(a)
    alpha = 0.5
    patience = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        if M < 1e-15:
            break

        b = _lp_step(a, M)
        if b is None or not np.all(np.isfinite(b)):
            break

        # Try full replacement first, blend if worse
        for al in [1.0, 0.5, 0.3, 0.1]:
            a_new = (1 - al) * a + al * b
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new <= best_sc + 1e-12:
                a = a_new
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    patience = 0
                else:
                    patience += 1
                break
        else:
            # None worked, take smallest step
            a_new = (1 - 0.05) * a + 0.05 * b
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            a = a_new
            patience += 1

        if it % 20 == 0:
            print(f"    LP iter {it}: score={_score(a):.10f}, M={M:.8f}, patience={patience}")

        if patience > 200:
            # Reduce alpha and continue
            alpha = max(0.01, alpha * 0.5)
            patience = 0

    return best_a, best_sc


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_optimize(a_init, beta, n, max_iter=2000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxcor': 30, 'maxls': 50, 'disp': False}
        )
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _subgrad_polish(a_init, deadline, max_steps=3000):
    """Subgradient descent with projection to simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-15:
        return a_init, _score(a_init)
    a = a / S

    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.02
    best_lr = lr

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = wc[n - 1:2 * n - 1]

        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            best_lr = lr
        else:
            # Don't accept, reduce lr
            lr *= 0.9
            if lr < 1e-6:
                lr = best_lr * 0.7

        if i % 200 == 0:
            print(f"    SG iter {i}: score={_score(a):.10f}, lr={lr:.6f}")

    return best_a, best_sc


def _perturb_asymmetric(a, rng, strength=0.1):
    """Create an asymmetric perturbation to break symmetry."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    # Add a linear ramp (breaks reflection symmetry)
    t = np.linspace(0, 1, n)
    z = z + strength * (t - 0.5) * (rng.uniform(1, 3))
    # Add random high-freq noise
    z = z + strength * 0.5 * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _lbfgs_schedule(a_start, deadline, min_remaining=30.0):
    """Multi-beta L-BFGS continuation schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(3000, max(100, int(remaining / 0.05)))
        a_new = _lbfgs_optimize(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid LP-based alternating optimization with gradient polishing.
    Uses Frank-Wolfe LP oracle for global progress, then L-BFGS and
    subgradient methods for fine polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_main = len(parent)
    except Exception:
        parent = None
        n_main = 2048

    if parent is not None:
        print(f"Parent: n={n_main}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
        a0 = parent.copy()
    else:
        a0 = np.ones(n_main) / n_main
        print(f"No parent, uniform n={n_main}, score={_score(a0):.10f}")
        update_best(a0, _score(a0), "init")

    # --- Phase 1: LP-based alternating optimization ---
    print(f"\nPhase 1: Alternating LP (t={time.monotonic() - t_start:.1f}s)")
    lp_deadline = deadline - 600.0
    if time.monotonic() < lp_deadline:
        a_lp, sc_lp = _alternating_lp(a0, lp_deadline, max_iter=1000)
        update_best(a_lp, sc_lp, "alternating_lp")

    # --- Phase 2: L-BFGS continuation from current best ---
    print(f"\nPhase 2: L-BFGS continuation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_l = _lbfgs_schedule(best_seq, deadline, min_remaining=200.0)
        update_best(a_l, _score(a_l), "lbfgs_sched")

    # --- Phase 3: Subgradient polish ---
    print(f"\nPhase 3: Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 100, max_steps=2000)
        update_best(a_sg, sc_sg, "subgrad")

    # --- Phase 4: Fine L-BFGS ---
    print(f"\nPhase 4: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([0.0005, 0.0002, 0.0001, 0.00005])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 40:
                break
            a_f = _lbfgs_optimize(best_seq, beta, n, max_iter=1000)
            sc_f = _score(a_f)
            if sc_f < best_score:
                update_best(a_f, sc_f, f"fine_{i}")
            else:
                break

    # --- Phase 5: Asymmetric restarts with LP ---
    print(f"\nPhase 5: Asymmetric restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        n_restarts = min(3, max(1, int(remaining / 200)))
        for i in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break
            print(f"  Restart {i} (t={time.monotonic() - t_start:.1f}s)")
            a_r = _perturb_asymmetric(best_seq, rng, strength=0.05 * (i + 1))
            sc_r = _score(a_r)
            if sc_r < best_score:
                update_best(a_r, sc_r, f"restart_{i}_init")

            # Short LP phase
            lp_dl = min(deadline - 25, time.monotonic() + 120)
            if time.monotonic() < lp_dl:
                a_r2, sc_r2 = _alternating_lp(a_r, lp_dl, max_iter=300)
                update_best(a_r2, sc_r2, f"restart_{i}_lp")

            # Quick L-BFGS
            if time.monotonic() < deadline - 20:
                a_r3 = _lbfgs_schedule(a_r2 if sc_r2 < sc_r else a_r, deadline - 20, min_remaining=20)
                update_best(a_r3, _score(a_r3), f"restart_{i}_lbfgs")

    # --- Phase 6: Final subgradient + L-BFGS cycle ---
    print(f"\nPhase 6: Final cycle (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _subgrad_polish(best_seq, deadline - 3, max_steps=500)
        update_best(a_fin, sc_fin, "final_sg")

        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta_final = mb * 0.00003
            a_fin2 = _lbfgs_optimize(best_seq, beta_final, n, max_iter=500)
            update_best(a_fin2, _score(a_fin2), "final_lbfgs")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
