Adaptive gradient-based optimization with multi-profile initialization,
adaptive temperature scheduling, and multi-length exploration for
minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _ fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _gradient_smoothed(a, beta):
    """Compute gradient of 2n * LSE_beta(conv(a,a)) / S^2."""
    n = len(a)
    c = _fft_convolve(a, a)
    s = float(a.sum())
    
    # Weight computation with log-sum-exp trick
    c_max = c.max()
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    w_sum = float(e.sum())
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
    else:
        w = e / w_sum
    
    lse = (c_max * w_sum + beta * float(np.sum(c * w))) / w_sum  # Hmm, not quite
    
    # Actually: LSE_beta(c) = (1/beta) * log(sum(exp(beta*c)))
    # = (1/beta) * log(exp(beta*c_max) * sum(exp(beta*(c-c_max))))
    # = c_max + (1/beta) * log(sum(exp(beta*(c-c_max))))
    # = c_max + (1/beta) * log(sum(e))
    # where e[k] = exp(beta*(c[k]-c_max))
    
    lse_val = c_max + np.log(w_sum) / beta
    
    # Gradient of sum(w_k * c_k) w.r.t. a:
    # = 2 * cross_corr(w, a)
    # cross_corr(w, a)_j = sum_k w_k * a_{k-j}
    # = (w * a_rev)_{n-1+j}
    
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    # wc has length len(w) + len(a_rev) - 1 = (2n-1) + n - 1 = 3n-2
    # We want elements at indices n-1 to 2n-2
    grad_sum_wc = 2.0 * wc[n-1:2*n-1]
    
    # Full gradient of 2n * lse / S^2
    # = (2n/S^2) * grad_sum_wc - (4n*lse/S^3) * 1
    grad = (2.0 * n / (s * s)) * grad_sum_wc - (4.0 * n * lse_val / (s ** 3)) * np.ones(n)
    
    return grad, lse_val

def _make_profile(n, kind='uniform', param=0.0):
    """Generate various initial profiles."""
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1)  # [0, 1]
    
    if kind == 'uniform':
        p = np.ones(n)
    elif kind == 'raised_cosine':
        p = np.cos(np.pi * (x - 0.5)) ** 2
        p = np.maximum(p, 1e-10)
    elif kind == 'super_gauss':
        p = np.exp(-((x - 0.5) ** 2) / (2 * param ** 2)) if param > 0 else np.ones(n)
    elif kind == 'polynomial':
        p = (x * (1 - x)) ** param
        p = np.maximum(p, 1e-10)
    elif kind == 'bimodal':
        p = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    elif kind == 'triangular':
        p = 1.0 - np.abs(2.0 * x - 1.0)
        p = np.maximum(p, 1e-10)
    else:
        p = np.ones(n)
    
    return p / p.sum()


def _lp_direction(f, M, time_limit=20):
    """LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    n = len(f)
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except:
        pass
    return None


def _lbfgs_polish(a_init, deadline, max_iter=5000):
    """L-BFGS-B polish using the smoothed objective."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    beta = 0.01  # Fixed moderate smoothing
    
    def objective(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _fft_convolve(a, a)
        c_max = c.max()
        c_shifted = (c - c_max) / beta
        e = np.exp(c_shifted)
        lse = c_max + np.log(e.sum()) / beta
        return 2.0 * n * lse  # Scale-invariant part (sum(a) is fixed by softmax)
    
    def gradient(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _fft_convolve(a, a)
        c_max = c.max()
        c_shifted = (c - c_max) / beta
        e = np.exp(c_shifted)
        w = e / e.sum()
        lse = c_max + np.log(e.sum()) / beta
        
        a_rev = a[::-1]
        wc = _fft_convolve(w, a_rev)
        grad_lse = 2.0 * wc[n-1:2*n-1]

        # a = softmax(z), so da_j/dz_k = a_j (delta_jk - a_k)
        # grad_z = a * (grad_a - sum(grad_a * a))
        grad_a = (2.0 * n) * grad_lse / (a.sum() ** 2)
        grad_z = a * (grad_a - np.dot(grad_a, a))
        
        return grad_z
    
    try:
        result = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'maxfun': 10000,
                                  'ftol': 1e-15, 'gtol': 1e-10})
        z_opt = result.x
        a_opt = np.exp(z_opt - z_opt.max())
        a_opt = a_opt / a_opt.sum()
        return a_opt
    except:
        return a_init


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    a_new = np.exp(z - z.max())
    return a_new / a_new.sum()

Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive-temperature gradient descent with Nesterov momentum,
LP-based projection for guaranteed descent, multi-profile initialization,
multi-length exploration, and L-BFGS polishing.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(z, beta_rel):
    """Compute gradient of F wrt z, where a=softmax(z), F=2n*LSE_beta(c)/S^2.
    beta_rel is relative to max(c)."""
    a = _softmax(z)
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    s = float(a.sum())
    if c_max < 1e-300 or s < 1e-15:
        return np.zeros_like(z), c_max
    beta = beta_rel / c_max
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    w_sum = float(e.sum())
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
        lse_val = c_max

    grad_a = (2.0 * n / (s * s)) * 2.0 * wc[n - 1:2 * n - 1] - (4.0 * n * lse_val / (s ** 3)) * np.ones(n)
    grad_z = a * (grad_a - np.dot(grad_a, a))

    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _optimize_main(a_init, deadline, lr=0.01, beta_start=0.5, beta_end=0.05):
    """Nesterov-accelerated gradient descent in log-space with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z_prev = z.copy()
    momentum = np.zeros(n)
    best_a = a_init.copy()
    best_score = _score(a_init)
    t_step = 0
    mu = 0.9
    start_time = time.monotonic()
    total_time = max(1.0, deadline - start_time)
    no_improve = 0
    patience = 5000

    while time.monotonic() < deadline:
        frac = min(1.0, (time.monotonic() - start_time) / total_time)
        beta_rel = beta_start + (beta_end - beta_start) * frac
        cur_lr = lr * (1.0 + 2.0 * (1.0 - frac))

        a = _softmax(z)
        grad_z, c_max = _compute_grad_z(z, beta_rel)
        if not np.all(np.isfinite(grad_z)):
            break

        t_step += 1
        mu_new = (1.0 + (t_step + 1.0) / (t_step + 2.0)) * (t_step / (t_step + 1.0)) * (1.0 - mu) + mu
        momentum = (1.0 - mu_new) * grad_z + mu_new * momentum
        z = z - cur_lr * momentum
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t_step % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > patience and t_step % 2000 == 0:
                cur_lr *= 0.8
                no_improve = 0
        if t_step % 10000 == 0:
            cur_lr = max(0.001, cur_lr * 0.95)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _optimize_lp_based(a_init, deadline):
    """LP-based iterative improvement."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    n = len(a_init)
    f = a_init.copy()
    f = f / f.sum()
    c = _fft_convolve(f, f)
    best_score = _score(f)
    best_f = f.copy()
    stuck = 0
    t = 0.5
    step_count = 0

    while time.monotonic() < deadline - 10:
        c = _fft_convolve(f, f)
        M = float(c.max())
        k_idx = np.arange(2 * n - 1)
        j_idx = np.arange(n)
        diff = k_idx[:, np.newaxis] - j_idx[np.newaxis, :]
        mask = (diff >= 0) & (diff < n)
        rows, cols = np.where(mask)
        vals = f[diff[rows, cols]]
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        g = None
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 10, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
                if g.sum() < 1e-15:
                    g = None
        except Exception:
            g = None
        if g is None:
            stuck += 1
            if stuck > 8:
                break
            continue
        stuck = 0
        g = g / g.sum()
        improved = False
        t_try = t
        for _ in range(5):
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0)
            s_new = f_new.sum()
            if s_new < 1e-15:
                t_try *= 0.5
                continue
            f_new = f_new / s_new
            c_new = _fft_convolve(f_new, f_new)
            M_new = float(c_new.max())
            if M_new < M * (1.0 - 1e-12):
                f = f_new
                t = min(1.0, t_try * 1.4)
                improved = True
                step_count += 1
                sc = _score(f)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                break
            else:
                t_try *= 0.5
        if not improved:
            t *= 0.5
    return best_f, best_score


def _lbfgs_polish(a_init, deadline):
    """L-BFGS-B polish with fixed moderate beta."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    beta_fixed = 0.1

    def obj(z):
        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = c.max()
        beta = beta_fixed / max(c_max, 1e-15)
        c_sh = (c - c_max) / beta
        e = np.exp(c_sh)
        lse = c_max + np.log(max(e.sum(), 1e-300)) / beta
        return 2.0 * n * lse

    def obj_grad(z):
        grad_z, _ = _compute_grad_z(z, beta_fixed)
        return (2.0 * n) * grad_z * 0.5

    try:
        res = minimize(obj, z0, jac=obj_grad, method='L-BFGS-B',
                       options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': 5000})
        a_opt = _softmax(res.x)
        return a_opt
    except Exception:
        return a_init


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: adaptive gradient descent, LP refinement,
    L-BFGS polish, multi-restart, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_score = np.inf
    best_seq = None

    def try_update(p):
        nonlocal best_score, best_seq
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = p.copy()
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s", flush=True)

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if parent_raw.sum() > 0.01:
                parent = parent_raw / parent_raw.sum()
    except Exception:
        parent = None

    # === Phase 0: Initialization (0-40s) ===
    print("=== Phase 0: Initialization ===", flush=True)
    phase0_end = min(deadline, t_start + 40.0)

    if parent is not None:
        try_update(parent)
        print(f"  Parent: n={len(parent)}, score={best_score:.10f}")

    init_candidates = []
    if parent is not None:
        init_candidates.append(parent.copy())

    n_vals = [2048, 4096, 8192]
    if parent is not None:
        n_vals = [len(parent), max(2048, len(parent) * 2), max(4096, len(parent) * 2)]
        n_vals = list(set(n_vals))

    n_main = len(parent) if parent is not None else 8192

    profiles = [
        ('uniform', 0.0),
        ('raised_cosine', 0.0),
        ('super_gauss', 0.15),
        ('super_gauss', 0.2),
        ('super_gauss', 0.25),
        ('super_gauss', 0.3),
        ('polynomial', 0.5),
        ('polynomial', 1.0),
        ('polynomial', 2.0),
        ('airy_like', 0.0),
        ('bimodal', 0.0),
    ]

    for n_try in n_vals[:3]:
        for kind, param in profiles[:5]:
            if time.monotonic() > phase0_end:
                break
            p = _make_profile(n_try, kind, param)
            init_candidates.append(p)
            sc = _score(p)
            if sc < best_score:
                try_update(p)

    print(f"  Phase 0 best: {best_score:.10f}, {len(init_candidates)} candidates", flush=True)

    # === Phase 1: Main gradient optimization (40s-600s) ===
    print("\n=== Phase 1: Main gradient optimization ===", flush=True)
    phase1_end = min(deadline, t_start + 600.0)

    # Sort candidates by score
    scored_cands = []
    for p in init_candidates:
        sc = _score(p)
        scored_cands.append((sc, p))
    scored_cands.sort(key=lambda x: x[0])

    top_k = min(4, len(scored_cands))
    per_cand = (phase1_end - time.monotonic() - 20.0) / max(1, top_k)

    for i in range(top_k):
        if time.monotonic() > phase1_end - 15:
            break
        sc_init, p_init = scored_cands[i]
        print(f"  Opt [{i}]: init_score={sc_init:.10f}, n={len(p_init)}, budget={per_cand:.0f}s", flush=True)
        dl = min(phase1_end - 10, time.monotonic() + per_cand)
        a_opt, sc_opt = _optimize_main(p_init, dl, lr=0.01, beta_start=0.5, beta_end=0.03)
        try_update(a_opt)
        print(f"    -> score={sc_opt:.10f}", flush=True)

    print(f"  Phase 1 best: {best_score:.10f}", flush=True)

    # === Phase 2: LP refinement (600s-850s) ===
    print("\n=== Phase 2: LP refinement ===", flush=True)
    phase2_end = min(deadline, t_start + 850.0)

    if best_seq is not None and time.monotonic() < phase2_end - 20:
        a_init = best_seq / best_seq.sum()
        a_lp, sc_lp = _optimize_lp_based(a_init, min(phase2_end - 10, time.monotonic() + 200.0))
        try_update(a_lp)
        print(f"  Phase 2 best: {best_score:.10f}", flush=True)

    # === Phase 3: L-BFGS polish (850s-1100s) ===
    print("\n=== Phase 3: L-BFGS polish ===", flush=True)
    phase3_end = min(deadline, t_start + 1100.0)

    if best_seq is not None and time.monotonic() < phase3_end - 20:
        a_init = best_seq / best_seq.sum()
        remaining = phase3_end - time.monotonic() - 15
        if remaining > 30:
            a_bfgs = _lbfgs_polish(a_init, time.monotonic() + min(remaining, 150.0))
            try_update(a_bfgs)
            # Second BFGS pass
            if time.monotonic() < phase3_end - 30:
                a_init2 = best_seq / best_seq.sum()
                a_bfgs2 = _lbfgs_polish(a_init2, time.monotonic() + (phase3_end - time.monotonic() - 10))
                try_update(a_bfgs2)
        print(f"  Phase 3 best: {best_score:.10f}", flush=True)

    # === Phase 4: Multi-restart (1100s-1400s) ===
    print("\n=== Phase 4: Multi-restart ===", flush=True)
    phase4_end = min(deadline, t_start + 1400.0)

    if best_seq is not None and time.monotonic() < phase4_end - 20:
        a_best = best_seq / best_seq.sum()
        n_restarts = 6
        per_r = (phase4_end - time.monotonic() - 20.0) / max(1, n_restarts)
        sigmas = [0.02, 0.04, 0.08, 0.12, 0.2, 0.35]
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 15:
                break
            p_start = _perturb(a_best, sigmas[i], rng)
            dl = min(phase4_end - 10, time.monotonic() + per_r)
            a_opt, _ = _optimize_main(p_start, dl, lr=0.008, beta_start=0.3, beta_end=0.02)
            try_update(a_opt)
        print(f"  Phase 4 best: {best_score:.10f}", flush=True)

    # === Phase 5: Multi-length exploration (1400s-1650s) ===
    print("\n=== Phase 5: Multi-length exploration ===", flush=True)
    phase5_end = min(deadline, t_start + 1650.0)

    if best_seq is not None and time.monotonic() < phase5_end - 20:
        cur_n = len(best_seq)
        targets = []
        for tn in [1024, 2048, 4096, 8192, 16384, 32768]:
            if tn != cur_n and 512 <= tn <= 65536:
                targets.append(tn)
        targets.sort(key=lambda x: abs(x - cur_n))
        targets = targets[:3]

        for tn in targets:
            if time.monotonic() > phase5_end - 30:
                break
            p_int = _interp_profile(best_seq, tn)
            try_update(p_int)
            remaining = phase5_end - time.monotonic() - 25
            if remaining > 60:
                a_opt, _ = _optimize_main(p_int, time.monotonic() + min(remaining, 130.0),
                                          lr=0.006, beta_start=0.3, beta_end=0.02)
                try_update(a_opt)
        print(f"  Phase 5 best: {best_score:.10f}", flush=True)

    # === Phase 6: Final fine-tuning (1650s-deadline) ===
    print("\n=== Phase 6: Final fine-tuning ===", flush=True)

    if best_seq is not None and time.monotonic() < deadline - 20:
        a_best = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            a_fin, _ = _optimize_main(a_best, time.monotonic() + min(remaining, 120.0),
                                      lr=0.002, beta_start=0.1, beta_end=0.01)
            try_update(a_fin)
        if time.monotonic() < deadline - 10:
            a_best2 = best_seq / best_seq.sum()
            remaining2 = deadline - time.monotonic() - 5
            if remaining2 > 20:
                a_fin2, _ = _optimize_main(a_best2, time.monotonic() + min(remaining2, 60.0),
                                           lr=0.001, beta_start=0.05, beta_end=0.005)
                try_update(a_fin2)

    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}, "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"  Verified score: {sc_final:.10f}", flush=True)
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"  Evaluator: {sc_eval:.10f}", flush=True)
        except Exception as ex:
            print(f"  Eval error: {ex}", flush=True)
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"  FALLBACK: {_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
