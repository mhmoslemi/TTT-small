    # = (1/k*beta) * [conv(w, a)[m] + conv(w, a)[2n-2-m]]  ... hmm not quite
    # 
    # Actually: sum_j w_j * a_{j-m} = (w * a_reversed)[m + n - 1] ... 
    # Let me just compute it directly using the convolution structure.
    
    # The Jacobian of c = conv(a,a) is:
    # J[m, i] = dc_m / da_i = a_{m-i} + a_{i-m} (with a_j = 0 for j not in [0,n-1])
    # 
    # So grad_a = J^T * (w / (k * beta)) where w is the softmax over top-k
    # grad_a[m] = sum_j (w_j / (k*beta)) * (a_{j-m} + a_{m-j})
    
    # We can compute this as:
    # sum_j w_j * a_{j-m} = cross-correlation of w and a at lag m
    # = conv(w, a_reversed)[m + n - 1] ... 
    # 
    # Actually, let's use the fact that:
    # sum_j w_j * a_{j-m} for fixed m = sum_j w_j * a_{j-m}
    # If we define b = a reversed, then a_{j-m} = b[n-1-j+m] = b[n-1-(j-m)]
    # So sum_j w_j * b[n-1-(j-m)] = (w * b)[n-1-m + ... ] 
    # 
    # This is getting complicated. Let me just use the direct formula:
    # grad_a = (2/(k*beta)) * conv(w, a) ... no that's not right either.
    
    # Let me think again. c_k = sum_j a_j * a_{k-j} where k-j must be in [0, n-1].
    # dc_k / da_i = a_{k-i} + a_{i-k} (when 0 <= k-i <= n-1 and 0 <= i-k <= n-1)
    # Note that a_{i-k} is nonzero only when 0 <= i-k <= n-1, i.e., k <= i <= k+n-1.
    # And a_{k-i} is nonzero only when 0 <= k-i <= n-1, i.e., i-k <= n-1, i.e., i >= k-(n-1).
    
    # So dc_k / da_i = a_{k-i} * [0 <= k-i <= n-1] + a_{i-k} * [0 <= i-k <= n-1]
    
    # grad_a_i = sum_k (W_k) * dc_k / da_i where W_k = w_k / (k*beta) for k in topk
    # = sum_k W_k * a_{k-i} * [0 <= k-i <= n-1] + sum_k W_k * a_{i-k} * [0 <= i-k <= n-1]
    
    # First term: sum_k W_k * a_{k-i} = sum_{j=0}^{n-1} W_{i+j} * a_j (where we define W_k = 0 for k not in topk or out of range)
    # = conv(W, a)[i] ... wait, conv(W, a)[i] = sum_j W_j * a_{i-j}
    # That's sum_j W_j * a_{i-j}, not sum_j W_{i+j} * a_j.
    
    # sum_j W_{i+j} * a_j = cross-correlation... 
    # = sum_j a_j * W_{i+j}
    # If I reverse W: W_rev[m] = W_{2n-2-m}, then sum_j a_j * W_{i+j} = sum_j a_j * W_rev[2n-2-i-j]
    # = conv(a, W_rev)[2n-2-i]
    
    # OK this is getting too complicated analytically. Let me just compute the gradient
    # using the formula: grad_a_i = sum_k W_k * (a_{max(0, k-i)} + a_{max(0, i-k)})
    # where we only sum over k in the top-k set and where the indices are valid.
    
    # For efficiency, I'll use the convolution-based approach:
    # sum_k W_k * a_{k-i} (for k-i in [0,n-1]) = sum_j W_{i+j} * a_j for j in [max(0, -i), min(n-1, n-1-i)]
    # This is the "anti-convolution" or cross-correlation.
    
    # Let me just compute both terms using FFT:
    # Term 1: h1[i] = sum_j a_j * W[i+j] for valid i,j
    #         = sum_j a_j * W[i+j]
    # Let W_ext be W padded to length 2n-1. Then h1[i] = sum_{j=0}^{min(n-1, 2n-2-i)} a_j * W_ext[i+j]
    # This is a "sliding dot product" which can be computed as a reverse cross-correlation.
    
    # Actually, the simplest efficient way:
    # h1 = cross-correlation of a and W, where (a ⋆ W)[i] = sum_j a[j] W[j+i]
    # This equals conv(a, W_reversed)[n-1 + i - (n-1)] ... 
    
    # You know what, let me just compute it the naive way for correctness, and optimize later if needed.
    # For n up to 6144 and k up to 50, the computation should be fast enough.
    
    a_rev = a[::-1].copy()  # a_rev[j] = a[n-1-j]
    
    # dc_k/da_i = a_{k-i} + a_{i-k}
    # grad_a_i = sum_{k in topk} W_k * (a_{k-i} + a_{i-k})
    
    # sum_k W_k * a_{k-i}: 
    # Let's define U = W (length 2n-1, mostly zero, nonzero only at topk indices)
    # Then sum_k U_k * a_{k-i} = (U * a_flipped)[...] 
    # 
    # a_{k-i} = a_flipped[n-1-(k-i)] = a_flipped[n-1-k+i]
    # So sum_k U_k * a_flipped[n-1-k+i] = conv(U, a_flipped)[n-1+i]
    # where conv is the full convolution.
    
    # Similarly, sum_k U_k * a_{i-k} = sum_k U_k * a_flipped[n-1-i+k]
    # = conv(a_flipped, U)[n-1-i + ... ] hmm
    
    # Let me just use: sum_k U_k * a_{i-k} = sum_j (i-j) ... 
    # a_{i-k} is nonzero when 0 <= i-k <= n-1, i.e., i-(n-1) <= k <= i
    # sum_k U_k * a_{i-k} = sum_{k=max(0,i-n+1)}^{min(2n-2,i)} U_k * a_{i-k}
    # Let j = i-k, so k = i-j:
    # = sum_j U_{i-j} * a_j for j where i-j is in [0, 2n-2] and j in [0, n-1]
    # = conv(U, a)[i]  (standard convolution!)
    
    # And: sum_k U_k * a_{k-i}
    # = sum_j U_{i+j} * a_j (letting j = k-i, k = i+j)
    # This is NOT a standard convolution. It's a "reversed cross-correlation".
    # = sum_j a_j * U_{i+j}
    # = cross-correlation of a and U, evaluated at i.
    # = conv(a, U_reversed)[len(U)-1-i + ... ]
    
    # U has length 2n-1. U_reversed[m] = U[2n-2-m].
    # conv(a, U_reversed)[p] = sum_j a_j * U_reversed[p-j] = sum_j a_j * U[2n-2-p+j]
    # We want sum_j a_j * U[i+j]. Set 2n-2-p+j = i+j => p = 2n-2-i.
    # So sum_j a_j * U[i+j] = conv(a, U_reversed)[2n-2-i]
    
    # Therefore:
    # grad_a_i = (1/k) * sum_k W_k * (a_{k-i} + a_{i-k})
    #          = (1/k) * [conv(a, U_reversed)[2n-2-i] + conv(U, a)[i]]
    # where U_k = W_k (the softmax weights at topk positions, 0 elsewhere)
    
    # Wait, I need to be more careful. W_k here is the weight for the k-th convolution coefficient.
    # Let me redefine:
    # W[k] = w_k / (k*beta) for k in topk, 0 otherwise. (k here is the number of top elements, confusing notation)
    
    # Let me use different notation:
    # Let S = set of top-k indices in the convolution.
    # Let w_s be the softmax weights for s in S, so sum_{s in S} w_s = 1.
    # The objective is (1/|S|) * log(sum_{s in S} exp(c_s / beta))
    # Its gradient w.r.t. c is: grad_c[s] = w_s / (|S| * beta) for s in S, 0 otherwise.
    # 
    # So U[s] = w_s / (|S| * beta) for s in S, 0 otherwise.
    # 
    # grad_a_i = sum_s U[s] * (a_{s-i} + a_{i-s})
    # = conv(a, U_rev)[2n-2-i] + conv(U, a)[i]  ... need to verify indices
    
    # For now, let me just compute it directly. For n <= 6144 and |S| <= 50, 
    # the direct computation takes |S| * n operations, which is at most 50 * 6144 = 307200.
    # That's fast enough.
    
    # Direct computation:
    grad = np.zeros(n)
    topk_idx = np.argsort(c)[::-1][:k]
    # w is already computed as softmax weights
    for idx in range(k):
        s = topk_idx[idx]
        weight = w[idx] / (k * beta)
        # dc_s / da_i = a_{s-i} + a_{i-s}
        # a_{s-i} is nonzero when 0 <= s-i <= n-1, i.e., s-(n-1) <= i <= s
        # a_{i-s} is nonzero when 0 <= i-s <= n-1, i.e., s <= i <= s+(n-1)
        
        lo1 = max(0, s - (n - 1))
        hi1 = min(n - 1, s)
        if lo1 <= hi1:
            # grad[i] += weight * a[s-i] for i in [lo1, hi1]
            # a[s-i] for i in [lo1, hi1] means a[s-lo1] down to a[s-hi1]
            # = a[hi1...lo1...s-lo1] reversed... 
            # a[s-i] for i = lo1, lo1+1, ..., hi1 gives a[s-lo1], a[s-lo1-1], ..., a[s-hi1]
            # s-lo1 >= 0 and s-hi1 >= 0
            # s-lo1 = s - max(0, s-n+1) = min(s, n-1)
            # s-hi1 = s - min(n-1, s) = max(0, s-n+1)
            j_lo = s - hi1  # = max(0, s-n+1)
            j_hi = s - lo1  # = min(s, n-1)
            # grad[lo1:hi1+1] += weight * a[j_hi:j_lo-1:-1]
            grad[lo1:hi1 + 1] += weight * a[j_lo:j_hi + 1][::-1]
        
        lo2 = max(0, s)
        hi2 = min(n - 1, s + n - 1)
        if lo2 <= hi2:
            # grad[i] += weight * a[i-s] for i in [lo2, hi2]
            # a[i-s] for i = lo2, ..., hi2 gives a[lo2-s], ..., a[hi2-s]
            j_lo = lo2 - s  # = 0
            j_hi = hi2 - s  # = min(n-1, n-1) = n-1
            grad[lo2:hi2 + 1] += weight * a[j_lo:j_hi + 1]
    
    # The full gradient of the score (2n * topk_logsumexp) w.r.t. a is:
    # 2n * grad (from above)
    # But we're using softmax parameterization, so we need to convert.
    return grad  # This is the gradient of topk_logsumexp w.r.t. a


def topk_lse_obj(z, k, beta, n):
    """Objective: 2n * topk_logsumexp(conv(softmax(z), softmax(z)), k, beta)"""
    a = softmax_fn(z)
    c = full_convolve(a, a)
    return 2.0 * n * topk_logsumexp(c, k, beta)


def topk_lse_grad(z, k, beta, n):
    """Gradient of the top-k log-sum-exp objective w.r.t. z."""
    a = softmax_fn(z)
    c = full_convolve(a, a)
    
    # Gradient w.r.t. a
    grad_a = topk_logsumexp_grad(c, k, beta, n)
    
    # Convert to gradient w.r.t. z using softmax Jacobian
    # d(softmax(z))_i / dz_j = a_i * (delta_ij - a_j)
    # So grad_z_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = a_j * grad_a_j - a_j * sum_i (a_i * grad_a_i)
    # = a_j * (grad_a_j - dot(a, grad_a))
    
    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot)
    
    # Multiply by 2n (from the score)
    return 2.0 * n * grad_z


def lp_active_set(a, k_active=20, max_time=120):
    """
    LP-based active set method:
    Given current a, find direction g that minimizes max(conv(a, g)) 
    over the active set, then update a.
    """
    n = len(a)
    t_start = time.monotonic()
    
    for iteration in range(50):
        if time.monotonic() - t_start > max_time:
            break
        
        c = full_convolve(a, a)
        # Get active set: top-k convolution indices
        active_idx = np.argsort(c)[::-1][:k_active]
        M = float(c[active_idx[0]])  # max value
        
        # Build LP: find g >= 0, sum(g) = 1, minimizing t
        # subject to conv(a, g)[k] <= t for k in active_idx
        # 
        # conv(a, g)[k] = sum_j a_j * g[k-j]
        # This is linear in g!
        # 
        # Let A be a matrix where A[m, j] = coefficient of g[j] in conv(a,g)[m]
        # A[m, j] = a[m-j] if 0 <= m-j <= n-1, else 0
        # 
        # Constraints: A[m, :] @ g <= t for m in active_idx
        # sum(g) = 1
        # g >= 0
        
        # Build constraint matrix
        n_constraints = len(active_idx)
        # Variables: g[0..n-1], t (total n+1 variables)
        # Constraints:
        # For each m in active_idx: sum_j A[m,j] * g[j] - t <= 0
        # sum_j g[j] = 1 (handled separately)
        # g[j] >= 0 (handled by bounds)
        # t >= 0 (handled by bounds)
        
        # A[m, j] = a[m-j] if 0 <= m-j <= n-1 else 0
        A_cons = np.zeros((n_constraints, n))
        for idx, m in enumerate(active_idx):
            # A_cons[idx, j] = a[m-j] if 0 <= m-j <= n-1
            j_min = max(0, m - (n - 1))
            j_max = min(n - 1, m)
            if j_min <= j_max:
                A_cons[idx, j_min:j_max + 1] = a[m - j_max:m - j_min + 1][::-1]
        
        # LP: minimize t
        # subject to: A_cons @ g - t <= 0  (n_constraints inequalities)
        #             sum(g) = 1
        #             g >= 0, t >= 0
        #
        # Using scipy linprog:
        # c_obj = [0, 0, ..., 0, 1] (minimize t, which is the last variable)
        # A_ub = [A_cons, -I_n] (for A_cons @ g - t <= 0)
        #        Actually: A_cons @ g - t <= 0 means [A_cons, -1] @ [g; t] <= 0
        # b_ub = zeros
        # A_eq = [ones(1, n), 0] (sum(g) = 1)
        # b_eq = [1]
        # bounds = [(0, None)] * n + [(0, None)]
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t
        
        # Inequality constraints: A_cons @ g - t <= 0
        A_ub = np.hstack([A_cons, -np.ones((n_constraints, 1))])
        b_ub = np.zeros(n_constraints)
        
        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.ones(1)
        
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                g = res.x[:n]
                g = np.maximum(g, 0)
                g = g / g.sum()
                
                # Update: a_new = (1 - step) * a + step * g
                # Use adaptive step size
                for step in [0.5, 0.3, 0.2, 0.1, 0.05]:
                    a_new = (1 - step) * a + step * g
                    a_new = np.maximum(a_new, 0)
                    s = a_new.sum()
                    if s < 1e-15:
                        continue
                    a_new = a_new / s
                    sc = score_normalized(a_new)
                    if sc < score_normalized(a):
                        a = a_new
                        break
        except Exception:
            pass
    
    return a


def lbfgs_topk(a_init, k, beta_list, deadline):
    """Run L-BFGS with top-k log-sum-exp objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = score_normalized(a_init)
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        remaining_betas = beta_list[beta_list.index(beta):]
        time_left = deadline - time.monotonic()
        time_per = time_left / max(1, len(remaining_betas))
        time_per = min(time_per, 120.0)
        level_deadline = time.monotonic() + time_per
        maxiter = max(500, int(time_per * 20))
        
        try:
            result = minimize(
                topk_lse_obj, z, args=(k, beta, n),
                method='L-BFGS-B', jac=topk_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': maxiter * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = softmax_fn(result.x)
                sc = score_normalized(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def subgradient_refine(a_init, deadline, k=20, lr_init=0.05, max_iter=100000):
    """Subgradient method with top-k averaging and Adam."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = score_normalized(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = softmax_fn(z)
        c = full_convolve(a, a)
        
        # Compute top-k gradient
        grad_a = topk_logsumexp_grad(c, k, 0.01 * float(c.max()), n)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        # Evaluate
        a = softmax_fn(z)
        sc = score_normalized(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Decay learning rate
        if t % 5000 == 0:
            lr = max(0.001, lr * 0.8)
    
    return best_a, best_score


def coord_descent_active(a_init, deadline, k=30, max_rounds=100):
    """Coordinate descent focusing on variables that affect the top-k convolution coefficients."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = score_normalized(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a = softmax_fn(z)
        c = full_convolve(a, a)
        topk_idx = np.argsort(c)[::-1][:k]
        
        # Identify which a_j values affect the top-k coefficients
        active_vars = set()
        for s in topk_idx:
            j_min = max(0, s - (n - 1))
            j_max = min(n - 1, s)
            for j in range(j_min, j_max + 1):
                active_vars.add(j)
        
        active_vars = sorted(active_vars)
        improved = False
        
        for j in active_vars:
            if time.monotonic() > deadline:
                    break
            best_delta = 0.0
            current_score = score_normalized(softmax_fn(z))
            
            for delta in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = softmax_fn(z_trial)
                sc = score_normalized(a_trial)
                if sc < current_score - 1e-10:
                    current_score = sc
                    best_delta = delta
            
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                improved = True
        
        if not improved:
            break
    
    a = softmax_fn(z)
    sc = score_normalized(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    return profiles


def interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
