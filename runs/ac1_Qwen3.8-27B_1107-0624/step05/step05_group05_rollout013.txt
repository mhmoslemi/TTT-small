"""
Hybrid optimizer: LP + FISTA (smoothed log-sum-exp) + coordinate descent
for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _conv_grad(a, w):
    """Compute gradient of sum_k w_k * (a*a)_k with respect to a.
    w has length 2n-1, a has length n. Returns gradient of length n."""
    n = len(a)
    w_rev = w[::-1]
    conv_wa = _full_convolve(w, a)  # length 2n-1
    conv_wrev_a = _full_convolve(w_rev, a)  # length 2n-1
    return conv_wa[:n] + conv_wrev_a[n-1:2*n-1]

def _smoothed_objective(a, beta):

    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / max(beta, 1e-300)
    # Handle potential overflow
    max_shifted = shifted.max()
    if max_shifted > 500:

        e = np.exp(shifted)
        s = e.sum()
        return c_max + beta * (max_shifted + np.log(s))
    else:
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            return c_max
        return c_max + beta * np.log(s)

def _smoothed_grad(a, beta):

    c = _full_convolve(a, a)

def _coord_descent(a, deadline, max_rounds=1000):
    """Coordinate descent: for each coordinate, find the best value to reduce the max convolution."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = _full_convolve(best_a, best_a)
        c_max = c.max()
        
        # Try modifying each coordinate
        for j in range(n):
            if time.monotonic() > deadline:
                break
            a_old = best_a[j]
            # Try reducing a[j] and redistributing
            # The max convolution position is argmax(c)
            # Element a[j] contributes to c[k] for k = j to j + n - 1 (as a_{k-j})
            # and for k = 0 to j (as a_{j-k})
            
            # Try a[j] = 0
            best_a[j] = 0
            # Renormalize
            s = best_a.sum()
            if s < 1e-15:
                best_a[j] = a_old
                continue
            best_a /= s
            c_new = _full_convolve(best_a, best_a)
            sc = c_new.max()
            c_new_max = sc
            if c_new_max < c_max - 1e-12:
                c_max = c_new_max
                improved = True
            else:
                best_a[j] = a_old
                best_a /= s  # undo normalization
        
        sc = _score_norm(best_a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
        
        if not improved and rnd > 10:
            break
    
    return best_a, best_score

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-300))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)

def _solve_lp(f, M, rows, cols):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None

def _lp_step(f, rows, cols):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.03, 0.05]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M

    """Find optimal t ≥ 0 to minimize max_k conv(f+t*d, f+t*d)_k."""
    q = _full_convolve(f, d)  # length 2n-1
    r = _full_convolve(d, d)  # length 2n-1
