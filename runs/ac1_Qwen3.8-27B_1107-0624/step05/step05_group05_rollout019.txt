"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based large steps, coordinate-wise exact optimization, 
smoothed Adam with log-sum-exp, and peak-shaving heuristics.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'mip_rel_gap': 0.01})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(f, rows, cols, time_limit=30):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f.copy(), M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.02, 0.05, 0.08]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M


def _coord_sweep(a, c, num_sweeps=1, deadline=None):
    """Coordinate-wise optimization: for each coordinate, try several multipliers."""
    n = len(a)
    a = a.copy()
    for sweep in range(num_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Shuffle order for diversity
        order = np.random.permutation(n)
        for i in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            v = a[i]
            if v < 1e-15:
                continue
            # Try several multipliers
            best_x = v
            best_max = c.max()
            for mult in [0.5, 0.6, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.5]:
                x = v * mult
                if x < 0:
                    continue
                # Compute new convolution max efficiently
                # Change: a[i] goes from v to x, delta = x - v
                delta = x - v
                if abs(delta) < 1e-15:
                    new_max = best_max
                else:
                    # c_new[k] = c[k] + effect of changing a[i]
                    # For k = 2*i: c changed by (x^2 - v^2)
                    # For k = i+j (j valid, j != i): c changed by delta * a[j]
                    # For k = j+i (j valid, j != i): c changed by a[j] * delta
                    # Net effect vector:
                    c_new = c.copy()
                    c_new[2 * i] += (x * x - v * v)
                    # For all j != i where both i and j are in [0, n-1]:
                    # position i+j gets delta * a[j]
                    # position j+i gets a[j] * delta (same position! since addition is commutative)
                    # So position (i+j) gets 2*delta*a[j] for j != i
                    # Wait no: conv(a,a)[k] = sum_j a[j]*a[k-j]
                    # Changing a[i] to x:
                    #   term where j=i: x * a[k-i] (was v * a[k-i]), change = delta * a[k-i]
                    #   term where k-j=i, i.e., j=k-i: a[k-i] * x (was a[k-i] * v), change = a[k-i] * delta
                    #   So total change at position k (k != 2i) = 2 * delta * a[k-i] (if k-i is valid)
                    #   At position k = 2i: the term is a[i]*a[i] = x^2 (was v^2), change = x^2 - v^2
                    # But we also need to handle the case where k-i is out of range
                    j_valid = np.arange(0, n)
                    positions = i + j_valid  # positions i+0, i+1, ..., i+n-1
                    valid_pos = positions[(positions >= 0) & (positions < 2 * n - 1)]
                    valid_j = valid_pos - i
                    # Exclude the diagonal (j=i, position 2i)
                    mask = valid_j != i
                    if np.any(mask):
                        c_new[valid_pos[mask]] += 2.0 * delta * a[valid_j[mask]]
                    new_max = c_new.max()
                if new_max < best_max:
                    best_max = new_max
                    best_x = x
            a[i] = best_x
            # Update c in place for next iteration
            if best_x != v:
                delta = best_x - v
                c[2 * i] += (best_x * best_x - v * v)
                j_valid = np.arange(0, n)
                positions = i + j_valid
                valid_pos = positions[(positions >= 0) & (positions < 2 * n - 1)]
                valid_j = valid_pos - i
                mask = valid_j != i
                if np.any(mask):
                    c[valid_pos[mask]] += 2.0 * delta * a[valid_j[mask]]
    return a, c


def _coord_sweep_exact(a, c, deadline=None, max_sweeps=10):
    """More precise coordinate-wise optimization using bisection on the 1D subproblem."""
    n = len(a)
    a = a.copy()
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        improved = False
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            v = a[i]
            if v < 1e-15:
                continue
            # Binary search for optimal x in [0, 3*v]
            # Objective: max of (c[2i] - v^2 + x^2) and (c[k] + 2*(x-v)*a[k-i]) for valid k
            lo, hi = 0.0, 3.0 * v
            # Quick check: is current v already optimal?
            best_sc = c.max()
            # Try a few points
            candidates = [0.0, 0.3 * v, 0.5 * v, 0.7 * v, 1.3 * v, 2.0 * v]
            best_x = v
            for xc in candidates:
                delta = xc - v
                new_c_2i = c[2 * i] - v * v + xc * xc
                # Max of linear functions
                j_valid = np.arange(0, n)
                positions = i + j_valid
                valid_mask = (positions >= 0) & (positions < 2 * n - 1)
                valid_pos = positions[valid_mask]
                valid_j = valid_pos - i
                mask = valid_j != i
                if np.any(mask):
                    linear_vals = c[valid_pos[mask]] + 2.0 * delta * a[valid_j[mask]]
                    new_max = max(new_c_2i, linear_vals.max())
                else:
                    new_max = new_c_2i
                if new_max < best_sc:
                    best_sc = new_max
                    best_x = xc
            if best_x != v:
                a[i] = best_x
                improved = True
        if not improved:
            break
    return a


def _adam_smooth(a_init, deadline, beta_start, beta_end, max_iter=500000, lr=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    patience = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1

        c = _full_convolve(a, a)
        c_max = float(c.max())
        ratio = max(0.0, (t - max_iter // 2)) / (max_iter // 2 + 1)
        beta = beta_start * (max(beta_end, 1e-15) / beta_start) ** ratio
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):

        else:

        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        if t % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

                    current_lr = min(0.05, current_lr * 1.15)
                else:
                    current_lr = max(0.001, current_lr * 0.9)
                prev_best = best_score
                patience = 0

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _peak_shave(a, c, num_peaks=5, deadline=None):
    """Identify top convolution peaks and try to reduce them."""
    n = len(a)
    a = a.copy()
    # Find top peaks
    sorted_idx = np.argsort(c)[::-1]
    for peak_k in sorted_idx[:num_peaks]:
        if deadline is not None and time.monotonic() > deadline:
            break
        peak_val = c[peak_k]
        # Which coordinates contribute to this peak?
        # c[peak_k] = sum_j a[j] * a[peak_k - j]
        j_start = max(0, peak_k - n + 1)
        j_end = min(peak_k, n - 1)
        contributing = np.arange(j_start, j_end + 1)
        if len(contributing) == 0:
            continue
        # For each contributing coordinate, try reducing it
        best_reduction = 0
        best_coord = -1
        best_new_val = 0
        for j in contributing:
            v = a[j]
            if v < 1e-15:
                continue
            partner = peak_k - j
            contrib = v * a[partner] if 0 <= partner < n else 0
            # Try reducing a[j] by a fraction
            for frac in [0.7, 0.5, 0.3]:
                new_v = v * frac
                reduction = contrib * (1 - frac)
                if reduction > best_reduction:
                    best_reduction = reduction
                    best_coord = j
                    best_new_val = new_v
        if best_coord >= 0 and best_reduction > 0:
            a[best_coord] = best_new_val
            # Update c
            c = _full_convolve(a, a)
    return a

        full_sc = _score(a_polished)
        print(f"Phase 6 final Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_polished.copy()

    # Phase 7: Perturbation + quick refinement (last 30s)
    if time.monotonic() < deadline - 5:
        for attempt in range(3):
            if time.monotonic() > deadline - 5:
                break
            p_start = _perturb(best_seq, 0.02 + 0.01 * attempt, rng)
            c_p = _full_convolve(p_start, p_start)
            p_start, c_p = _coord_sweep(p_start, c_p, num_sweeps=1, deadline=deadline - 5)
            full_sc = _score(p_start)
            print(f"  Perturb[{attempt}]: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = p_start.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
