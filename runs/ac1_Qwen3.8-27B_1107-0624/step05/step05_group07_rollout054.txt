```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)

    return np.maximum(v - theta, 0)
```

```python
"""
Step function sequence optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: multi-restart with diverse initializations, L-BFGS on log-sum-exp smoothed
objective with adaptive temperature, tight-set focused optimization, peak deflation,
coordinate polish, and multi-n exploration with interpolation.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -100, 100)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -100, 100)
    e = np.exp(vals)
    w = e / e.sum()
    grad_a = 2.0 * n * _full_convolve(w, a[::-1])[n - 1:2 * n - 1]
    dot_ga_a = float(np.dot(grad_a, a))
    return a * (grad_a - dot_ga_a)


def _tightset_obj(z, S, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return float(2.0 * n * c[S].max())


def _tightset_grad(z, S, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    max_val = float(c[S].max())
    tol = max_val * 1e-10 + 1e-15
    active = c[S] >= max_val - tol
    S_active = S[active]
    if len(S_active) == 0:
        S_active = S[:1]
    w = np.zeros(2 * n - 1)
    w[S_active] = 1.0 / len(S_active)
    grad_a = 2.0 * n * _full_convolve(w, a[::-1])[n - 1:2 * n - 1]
    dot_ga_a = float(np.dot(grad_a, a))
    return a * (grad_a - dot_ga_a)


def _run_lbfgs_lse(a_init, deadline, beta_list=None, max_total_iter=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                 0.0002, 0.00015, 0.0001, 0.00008, 0.00005, 0.00003, 0.00002]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 90.0)
        time_per = max(time_per, 5.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(100, int(time_per * 20))

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_lbfgs_tightset(a_init, deadline, top_k_list=None, max_iter_per_k=500):
    n = len(a_init)
    if top_k_list is None:
        top_k_list = [5, 10, 20, 50, 100, 200, 500]
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for k in top_k_list:
        if time.monotonic() > deadline:
            break
        c = _full_convolve(_softmax(z), _softmax(z))
        S = np.argsort(c)[-k:]
        S = np.sort(S)

        def obj_tight(z_inner):
            return _tightset_obj(z_inner, S, n)

        def grad_tight(z_inner):
            return _tightset_grad(z_inner, S, n)

        try:
            result = minimize(
                obj_tight, z,
                method='L-BFGS-B', jac=grad_tight,
                options={'maxiter': max_iter_per_k, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 50, 'maxfun': max_iter_per_k * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _peak_deflation(a, n_peaks=5, alpha=0.01, n_iter=100):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iter):
        c = _full_convolve(a, a)
        peak_indices = np.argsort(c)[-n_peaks:]
        max_c = float(c.max())

        # Compute deflation direction: reduce a at positions contributing to peaks
        deflate_dir = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            deflate_dir[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]

        deflate_dir = np.maximum(deflate_dir, 0)
        d_max = deflate_dir.max()
        if d_max < 1e-15:
            break
        deflate_dir /= d_max

        a_new = a - alpha * deflate_dir
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            alpha *= 0.7
            if alpha < 1e-6:
                break

    return best_a, best_score


def _subgradient_ascent(a_init, deadline, n_iter=3000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    beta = 0.05 * float(_full_convolve(a, a).max())

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-10, 0.03 * cm)
        vals = np.clip((c - cm) / beta, -100, 100)
        e = np.exp(vals)
        w = e / e.sum()
        grad_a = 2.0 * n * _full_convolve(w, a[::-1])[n - 1:2 * n - 1]
        dot_ga_a = float(np.dot(grad_a, a))
        grad = a * (grad_a - dot_ga_a)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 500 == 0:
            lr = max(0.005, lr * 0.9)

    return best_a, best_score


def _coord_polish(a_init, deadline, max_rounds=20):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        peak_idx = np.argsort(c)[-20:]
        improved = False
        for j in peak_idx:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            cur_score = _score_norm(a)
            for delta in [-0.2, -0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1, 0.2]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax(z_trial)
                sc = _score_norm(a_trial)
                if sc < cur_score:
                    cur_score = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax(z)
                sc = _score_norm(z)
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _make_diverse_inits(n, parent, rng):
    inits = []
    t = np.linspace(0, 1, n)
    u = 2 * t - 1

    if parent is not None:
        p_interp = _interp_profile(parent, n)
        inits.append(p_interp)
        z = np.log(np.maximum(p_interp, 1e-15))
        z = z - 0.02 * rng.normal(size=n)
        z = z - z.mean()
        inits.append(_softmax(z))
        z = np.log(np.maximum(p_interp, 1e-15))
        z = z - 0.05 * rng.normal(size=n)
        z = z - z.mean()
        inits.append(_softmax(z))

    # Various profiles
    for slope in [0.5, 1.0, 2.0, -1.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())

    a = np.ones(n)
    inits.append(a / n)

    a = 1.0 / (0.5 + u ** 2)
    inits.append(a / a.sum())

    a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** 2
    inits.append(a / a.sum())

    a = np.abs(u) + 0.2
    inits.append(a / a.sum())

    # Symmetric random
    r = np.abs(rng.normal(size=n // 2 + 1)) + 0.1
    a_sym = np.concatenate([r, r[-2::-1]])
    inits.append(a_sym / a_sym.sum())

    return inits


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    # Peak deflation + re-optimize
    if time.monotonic() < phase5_end - 20:
        a_new, sc_new = _peak_deflation(best_seq, n_peaks=3, alpha=0.003, n_iter=80)
        update_best(a_new, sc_new, "polish_deflate")
        if sc_new < best_score:
            a_cur = a_new
            if time.monotonic() < phase5_end - 10:
                per_dl3 = min(phase5_end, time.monotonic() + 30.0)
                a_new2, sc_new2 = _run_lbfgs_tightset(a_cur, per_dl3, top_k_list=[5, 10], max_iter_per_k=300)
                update_best(a_new2, sc_new2, "polish_after_deflate")

    # Coordinate polish
    if time.monotonic() < phase5_end - 10:
        per_dl4 = min(phase5_end, time.monotonic() + (phase5_end - time.monotonic()) * 0.6)
        a_new, sc_new = _coord_polish(best_seq, per_dl4, max_rounds=15)
        update_best(a_new, sc_new, "polish_coord")

    # Final ultra-fine L-BFGS
    if time.monotonic() < phase5_end - 5:
        c_ult = _full_convolve(best_seq, best_seq)
        max_c_ult = float(c_ult.max())
        final_betas = [b * max_c_ult for b in [0.00005, 0.00002, 0.00001]]
        per_dl5 = min(phase5_end, time.monotonic() + 25.0)
        a_new, sc_new = _run_lbfgs_lse(best_seq, per_dl5, beta_list=final_betas)
        update_best(a_new, sc_new, "final_ultra")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
