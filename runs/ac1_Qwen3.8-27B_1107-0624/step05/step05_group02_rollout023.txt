"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with adaptive smooth max, Adam fine-tuning, multi-restart,
and larger-n exploration to improve upon the LP-based approach.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_max(c, beta):
    """Compute smooth maximum and its gradient w.r.t. c."""
    c_max = np.max(c)
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c_max + beta * 0.0, w
    w = e / s
    smooth_val = c_max + beta * np.log(s)
    return smooth_val, w


def _grad_a(a, w):
    """Compute gradient of smooth_max(conv(a,a), beta) w.r.t. a.
    
    w: weights from smooth max over the convolution.
    Returns: gradient vector of same length as a.
    """
    n = len(a)
    a_rev = a[::-1]
    hc = _full_convolve(w, a_rev)
    return 2.0 * hc[n - 1:2 * n - 1]


def _lbfgs_optimize(a_init, deadline, beta_schedule=None, max_iter=200):
    """L-BFGS-B optimization in log-space with smooth max objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    
    if beta_schedule is None:
        max_c = best_score
        beta_schedule = [max_c * s for s in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]]
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        def objective(z_vec):
            a = _softmax(z_vec)
            c = _full_convolve(a, a)
            val, w = _smooth_max(c, beta)
            g_a = _grad_a(a, w)
            g_z = a * (g_a - np.dot(a, g_a))
            return val, g_z
        
        val, grad = objective(z)
        if not np.all(np.isfinite(grad)):
            continue
        
        try:
            res = minimize(objective, z, method='L-BFGS-B',
                          jac=True, options={'maxiter': max_iter, 'maxfun': 500,
                                             'gtol': 1e-8, 'iprint': 0})
            if res.x is not None:
                a_new = _softmax(res.x)
                sc = float(_full_convolve(a_new, a_new).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.01, beta_start=None, beta_end=None):
    """Adam optimizer with adaptive log-sum-exp smoothed max (multi-phase beta)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = best_score

    if beta_start is None:
        beta_start = 0.05 * max_c_init
    if beta_end is None:
        beta_end = 0.0001 * max_c_init

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    total_iterations = 50000
    start_time = time.monotonic()
    
    it = 0
    while it < total_iterations:
        if time.monotonic() > deadline:
            break
        it += 1
        t_step += 1
        
        # Adaptive beta: cosine schedule from beta_start to beta_end
        progress = it / total_iterations
        beta = beta_start + (beta_end - beta_start) * (1 + np.cos(np.pi * min(progress, 1.0))) / 2.0
        beta = max(beta, 1e-10 * max_c_init)

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        current_lr = lr * (1.0 + np.cos(np.pi * min(progress, 1.0))) / 2.0
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if it % 100 == 0:
            sc = float(c.max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _block_perturb(a, rng, n_blocks=10, block_size=None):
    """Perturb by swapping and scaling blocks."""
    n = len(a)
    if block_size is None:
        block_size = max(1, n // (2 * n_blocks))
    a_new = a.copy()
    for _ in range(n_blocks):
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if i != j and abs(i - j) >= block_size:
            a_new[i:i+block_size], a_new[j:j+block_size] = a_new[j:j+block_size].copy(), a_new[i:i+block_size].copy()
    s = a_new.sum()
    return a_new / s if s > 0 else a


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _make_gaussian(n, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) / width) ** 2)
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _make_raised_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-15)
    return a / a.sum()

    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rhos = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rhos > 0
    rho = np.max(rhos[cond])
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0)
