"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam in log-space with adaptive temperature, multiple initializations,
peak-targeted corrections, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize as sp_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _chirp_init(n, rng):
    """Create a chirp-like initialization with approximately flat autocorrelation."""
    t = np.arange(n) / n
    # Linear chirp
    for trial in range(20):
        lam = rng.uniform(2, 20)
        phi = rng.uniform(0, 2 * np.pi)
        raw = np.cos(2 * np.pi * lam * t * (t + 1) / 2 + phi)
        # Shift to make non-negative
        raw_shifted = raw - raw.min() + 0.1
        a = raw_shifted / raw_shifted.sum()
        if a.min() > 1e-10:
            return a
    return np.ones(n) / n


def _random_smooth_init(n, rng, sigma=0.3):
    """Random smooth initialization via Gaussian process-like construction."""
    freqs = rng.uniform(0, 3, size=min(n, 200))
    amps = rng.exponential(1.0, size=len(freqs))
    phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
    t = np.arange(n) / n
    raw = np.ones(n) / n
    for f, a, p in zip(freqs, amps, phases):
        raw += a * 0.05 * np.cos(2 * np.pi * f * t + p)
    raw = np.maximum(raw, 1e-8)
    return raw / raw.sum()


def _adam_optimize(a_init, deadline, lr=0.01, beta_start=0.5, beta_end=0.001,
                   max_iters=500000, momentum_reset_interval=20000, verbose=True):
    """
    Adam optimizer in log-space with adaptive temperature (beta).
    Minimizes 2*n*LSE(conv(a,a)) where a = softmax(z), sum(a)=1.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = max(best_score, 1e-10)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    # Adaptive beta: geometric decay from beta_start to beta_end
    total_planned = max_iters
    beta_log_start = np.log(max(beta_start * max_c_init, 1e-10))
    beta_log_end = np.log(max(beta_end * max_c_init, 1e-12))

    # Learning rate schedule: warmup then cosine decay
    warmup_steps = min(500, total_planned // 10)

    iters_done = 0
    last_score_check = 0
    score_check_interval = max(20, min(200, n // 20))

    while iters_done < max_iters:
        if time.monotonic() > deadline:
            break
        iters_done += 1
        t_step += 1

        # Current beta (geometric decay)
        progress = min(1.0, iters_done / total_planned)
        log_beta = beta_log_start + (beta_log_end - beta_log_start) * progress
        beta = float(np.exp(log_beta))

        # Current learning rate
        if iters_done < warmup_steps:
            current_lr = lr * (iters_done / warmup_steps)
        else:
            decay_progress = (iters_done - warmup_steps) / (total_planned - warmup_steps)
            current_lr = lr * 0.5 * (1 + np.cos(np.pi * min(1.0, decay_progress)))

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Log-sum-exp weights
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s_e = float(e.sum())
        if s_e < 1e-300 or not np.isfinite(s_e):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e

        # Gradient computation
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        # Momentum reset to escape local minima
        if iters_done % momentum_reset_interval == 0:
            m_buf *= 0.5
            v_buf *= 0.5

        # Score check
        if iters_done - last_score_check >= score_check_interval:
            last_score_check = iters_done
            sc = float(c.max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                if verbose and (iters_done % (score_check_interval * 10) == 0):
                    obj = 2.0 * n * best_score
                    print(f"    Adam iter {iters_done}: max_c={best_score:.2e}, "
                          f"obj={obj:.6f}, beta={beta:.4e}")

    # Final check
    a_final = _softmax(z)
    c_final = _full_convolve(a_final, a_final)
    sc_final = float(c_final.max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

    c_init = _full_convolve(a, a)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)

    b1, b2, eps = 0.85, 0.99, 1e-8

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        update = lr * mh / (np.sqrt(vh) + eps)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if step % 20 == 0:
            sc = float(c.max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_smooth(a, sigma, rng, bandwidth=10):
    """Smooth perturbation using a Gaussian kernel."""
    n = len(a)
    noise = rng.standard_normal(n)
    # Smooth the noise
    kernel = np.zeros(n)
    k = np.arange(-bandwidth, bandwidth + 1)
    kernel_vals = np.exp(-k**2 / (2 * (bandwidth // 3)**2))
    for j in range(n):
        kernel[j] = kernel_vals[j + bandwidth] if j + bandwidth < len(kernel_vals) else 0
    # Use FFT for convolution
    N = _next_pow2(2 * n)
    smooth = np.fft.irfft(np.fft.rfft(noise, N) * np.fft.rfft(kernel, N), N)[:n]
    smooth = smooth / (np.std(smooth) + 1e-15)

    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * smooth
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    Uses Adam in log-space with adaptive temperature, multiple initializations,
    peak-targeted corrections, and multi-scale exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) t={elapsed:.1f}s")

    # === Load parent ===
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_main = 8192
    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent loaded: n={n_main}, score={sc0:.10f}")
        update_best(a0)
    else:
        a0 = np.ones(n_main) / n_main
        print(f"No parent, using uniform n={n_main}")
        update_best(a0)

    # === Phase 1: Multi-start quick optimization (0 - 400s) ===
    phase1_end = min(deadline, t_start + 400.0)
    print(f"\n=== Phase 1: Multi-start optimization (target n={n_main}) ===")

    # Generate diverse starting points
    starts = []
    start_names = []

    # Parent (or uniform)
    starts.append(a0)
    start_names.append("parent")

    # Chirp
    starts.append(_chirp_init(n_main, rng))
    start_names.append("chirp")

    # Random smooth
    starts.append(_random_smooth_init(n_main, rng))
    start_names.append("random_smooth")

    # Perturbed parent
    if parent is not None:
        starts.append(_perturb(parent, 0.1, rng))
        start_names.append("perturb_0.1")
        starts.append(_perturb_smooth(parent, 0.05, rng))
        start_names.append("perturb_smooth_0.05")

    # Interpolate parent to different n
    if parent is not None:
        for target_n in [4096, 16384]:
            if target_n != n_main:
                p_interp = _interp_profile(parent, target_n)
                starts.append(p_interp)
                start_names.append(f"interp_n{target_n}")

    # Quick Adam on each start
    per_start_time = (phase1_end - t_start - 30) / len(starts)
    per_start_time = max(30.0, min(60.0, per_start_time))

    for i, (a_start, name) in enumerate(zip(starts, start_names)):
        if time.monotonic() > phase1_end - 30:
            break
        n_try = len(a_start)
        deadline_i = min(time.monotonic() + per_start_time, phase1_end - 20)
        print(f"  Start '{name}' (n={n_try}, {time.monotonic()-t_start:.0f}s elapsed):")
        a_opt, sc_opt = _adam_optimize(
            a_start, deadline_i, lr=0.02, beta_start=0.3, beta_end=0.01,
            max_iters=200000, momentum_reset_interval=10000, verbose=False
        )
        obj = 2.0 * n_try * sc_opt
        print(f"    Done: obj={obj:.6f}, max_c={sc_opt:.2e}")
        update_best(a_opt)

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Main long Adam optimization (400s - 1100s) ===
    phase2_end = min(deadline, t_start + 1100.0)
    print(f"\n=== Phase 2: Main Adam optimization ===")

    if best_seq is not None:
        a_main = best_seq / best_seq.sum()
        n_main_opt = len(a_main)
        print(f"  Starting from best (n={n_main_opt}, score={best_score:.10f})")

        # Run with different parameter sets
        param_sets = [
            {'lr': 0.015, 'beta_start': 0.2, 'beta_end': 0.003, 'mr': 15000},
            {'lr': 0.01, 'beta_start': 0.1, 'beta_end': 0.001, 'mr': 20000},
            {'lr': 0.008, 'beta_start': 0.05, 'beta_end': 0.0005, 'mr': 25000},
        ]

        time_per_set = (phase2_end - time.monotonic() - 30) / len(param_sets)
        time_per_set = max(60.0, time_per_set)

        for pi, ps in enumerate(param_sets):
            if time.monotonic() > phase2_end - 30:
                break
            deadline_i = min(time.monotonic() + time_per_set, phase2_end - 20)
            print(f"  Param set {pi+1}/{len(param_sets)}: lr={ps['lr']}, "
                  f"beta [{ps['beta_start']}->{ps['beta_end']}], "
                  f"time={time.monotonic()-t_start:.0f}s")
            a_opt, sc_opt = _adam_optimize(
                a_main, deadline_i, lr=ps['lr'], beta_start=ps['beta_start'],
                beta_end=ps['beta_end'], max_iters=500000,
                momentum_reset_interval=ps['mr'], verbose=True
            )
            update_best(a_opt)
            a_main = a_opt / a_opt.sum()  # Chain: use result as next start

        print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Peak-targeted correction (1100s - 1350s) ===
    phase3_end = min(deadline, t_start + 1350.0)
    print(f"\n=== Phase 3: Peak-targeted correction ===")

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        a_peak = best_seq / best_seq.sum()
        deadline_i = time.monotonic() + min(
            (phase3_end - time.monotonic() - 15),
            phase3_end - t_start - time.monotonic()
        )
        # Multiple peak-targeted runs with different n_peaks
        for n_pk in [3, 5, 8]:
            if time.monotonic() > phase3_end - 20:
                break
            a_opt, sc_opt = _peak_targeted_correction(
                a_peak, n_peaks=n_pk, n_steps=500, lr=0.003,
                deadline=phase3_end - 15
            )
            update_best(a_opt)
            a_peak = a_opt / a_opt.sum()

        # Follow up with regular Adam
        if time.monotonic() < phase3_end - 30:
            a_opt, sc_opt = _adam_optimize(
                a_peak, phase3_end - 20, lr=0.005, beta_start=0.02,
                beta_end=0.002, max_iters=100000, momentum_reset_interval=5000,
                verbose=False
            )
            update_best(a_opt)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-restart with diverse perturbations (1350s - 1550s) ===
    phase4_end = min(deadline, t_start + 1550.0)
    print(f"\n=== Phase 4: Multi-restart ===")

    if best_seq is not None and time.monotonic() < phase4_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 5
        per_restart = (phase4_end - time.monotonic() - 20) / n_restarts
        per_restart = max(20.0, min(40.0, per_restart))

        sigmas = [0.02, 0.04, 0.08, 0.12, 0.2]
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 20:
                break
            if i % 2 == 0:
                p_start = _perturb(a_best_norm, sigmas[i], rng)
            else:
                p_start = _perturb_smooth(a_best_norm, sigmas[i] * 0.5, rng)
            deadline_i = time.monotonic() + per_restart
            a_r, sc_r = _adam_optimize(
                p_start, deadline_i, lr=0.015, beta_start=0.1,
                beta_end=0.005, max_iters=100000,
                momentum_reset_interval=10000, verbose=False
            )
            update_best(a_r)

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Larger/n smaller n exploration (1550s - 1700s) ===
    phase5_end = min(deadline, t_start + 1700.0)
    print(f"\n=== Phase 5: Multi-n exploration ===")

    if best_seq is not None and time.monotonic() < phase5_end - 30:
        tried_ns = set([len(best_seq)])
        cand_ns = [4096, 8192, 12288, 16384, 24576]
        # Sort by closeness to current best n
        cand_ns.sort(key=lambda x: abs(x - len(best_seq)))

        for target_n in cand_ns:
            if time.monotonic() > phase5_end - 40:
                break
            if target_n in tried_ns:
                continue
            tried_ns.add(target_n)

            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp)

            remaining = phase5_end - time.monotonic() - 30
            if remaining > 30:
                a_r, _ = _adam_optimize(
                    p_interp, time.monotonic() + remaining,
                    lr=0.01, beta_start=0.1, beta_end=0.003,
                    max_iters=200000, momentum_reset_interval=10000,
                    verbose=False
                )
                update_best(a_r)

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine polish (1700s - deadline) ===
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_final_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 5
        if remaining > 20:
            # Very fine Adam
            a_pol, _ = _adam_optimize(
                a_final_norm, time.monotonic() + remaining * 0.6,
                lr=0.003, beta_start=0.01, beta_end=0.001,
                max_iters=100000, momentum_reset_interval=20000,
                verbose=False
            )
            update_best(a_pol)

            # Peak-targeted micro-correction
            if time.monotonic() < deadline - 10:
                a_pk, _ = _peak_targeted_correction(
                    a_pol / a_pol.sum(), n_peaks=3, n_steps=200,
                    lr=0.002, deadline=deadline - 5
                )
                update_best(a_pk)

    print(f"Phase 6 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
