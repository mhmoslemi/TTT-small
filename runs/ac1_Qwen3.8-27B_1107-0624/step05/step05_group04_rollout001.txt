import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    """Score with sum normalization (scale-invariant)."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_grad_z(a, w, n):
    """Compute gradient of sum_k w[k]*conv_k w.r.t. z (log-softmax params)."""
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1: 2 * n - 1]
    wc = float(np.dot(w, _full_convolve(a, a)))
    grad_z = 2.0 * a * (h - wc)
    return grad_z


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    return _compute_grad_z(a, w, n)


def _iterative_reweight_obj(z, alpha, n):
    """Objective using iteratively reweighted peaks: sum w_k * c_k where w ∝ c^alpha."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    # Weight peaks by their value raised to alpha (concentrates on max)
    c_pos = np.maximum(c, 1e-30)
    w = c_pos ** alpha
    sw = w.sum()
    if sw < 1e-300:
        return 0.0
    w = w / sw
    return float(np.dot(w, c))


def _iterative_reweight_grad(z, alpha, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    c_pos = np.maximum(c, 1e-30)
    w = c_pos ** alpha
    sw = w.sum()
    if sw < 1e-300:
        return np.zeros(n)
    w = w / sw
    return _compute_grad_z(a, w, n)


def _peak_pressure_adam(a_init, deadline, rng, n_max_iter=500000):
    """Adam optimization with iteratively reweighted peak focus."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    # Alpha schedule: start moderate, increase to focus on max
    alpha_start = 2.0
    alpha_end = 50.0
    alpha_ramp = 5000  # iterations to ramp up alpha

        if time.monotonic() > deadline:

        t += 1
        a = _softmax_fn(z)

        # Adaptive alpha
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / alpha_ramp)

        # Compute weights
        c_pos = np.maximum(c, 1e-30)
        w = c_pos ** alpha
        sw = w.sum()
        if sw < 1e-300:
            break
        w = w / sw

        grad_z = _compute_grad_z(a, w, n)

        if not np.all(np.isfinite(grad_z)):
            break

        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = np.clip(z, -70, 70)
        z = z - z.mean()
        a = _softmax_fn(z)

def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _make_ramp(n, slope=1.5, curvature=0.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t + curvature * (t - 0.5) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exponential(n, rate=2.0):
    t = np.linspace(0, 1, n)
    a = np.exp(rate * t)
    return a / a.sum()


def _make_bilinear(n, slope1=1.0, slope2=1.0, break_point=0.6):
    t = np.linspace(0, 1, n)
    a = np.where(t < break_point,
                 1.0 + slope1 * t / break_point,
                 1.0 + slope1 + slope2 * (t - break_point) / (1 - break_point))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    print(f"  After Phase 3: best_score={best_score:.10f}")

    # --- Phase 4: Coordinate descent + final polish (20% of time) ---
    phase4_end = deadline
    print(f"\nPhase 4: Coordinate descent + polish (deadline t={phase4_end - t_start:.0f}s)")

    # Coordinate descent
    if time.monotonic() < phase4_end - 60:
        cd_budget = min(phase4_end - 60, time.monotonic() + total_budget * 0.08)
        a_cd, sc_cd = _coordinate_descent(
            best_seq, cd_budget, rng,
            max_rounds=max(100, int((cd_budget - time.monotonic()) * 10))
        )
        update_best(a_cd, sc_cd, "coord_descent")

    # Final L-BFGS with very small beta
    if time.monotonic() < phase4_end - 20 and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        beta_tiny = max(1e-8 * max_c_final, 1e-12)

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta_tiny, n_final),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 20000, 'ftol': 1e-18, 'gtol': 1e-13,
                         'maxls': 40, 'maxfun': 50000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(np.clip(result.x, -70, 70))
                sc_new = _score_norm_fn(a_new)
                update_best(a_new, sc_new, "final_lbfgs")
        except Exception as ex:
            print(f"  Final L-BFGS failed: {ex}")

    # Another coordinate descent round
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        cd2_budget = min(phase4_end - 5, time.monotonic() + 40.0)
        a_cd2, sc_cd2 = _coordinate_descent(
            best_seq, cd2_budget, rng,
            max_rounds=max(50, int((cd2_budget - time.monotonic()) * 5))
        )
        update_best(a_cd2, sc_cd2, "coord_descent_2")

    # Final Adam with very small lr
    if time.monotonic() < phase4_end - 5 and best_seq is not None:
        adam_final_budget = min(phase4_end - 3, time.monotonic() + 30.0)
        a_af, sc_af = _peak_pressure_adam(
            best_seq, adam_final_budget, rng,
            n_max_iter=max(100, int((adam_final_budget - time.monotonic()) * 50))
        )
        update_best(a_af, sc_af, "final_adam")

    # --- Final validation and return ---
    print(f"\nFinal Phase 4 complete: best_score={best_score:.10f}")

    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
