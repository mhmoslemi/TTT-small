    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _peak_heights(c, k=5):
    """Return the top-k peak heights and their indices."""
    order = np.argsort(c)[::-1]
    idx = order[:k]
    return c[idx], idx


def _subgradient_at_peak(a, c, k_star):
    """Compute subgradient of max(a*a) at peak k_star in log-space.
    grad_z[j] = 2 * a[j] * (a[k_star - j] - c[k_star])
    where a[k_star - j] = 0 if out of range.
    """
    n = len(a)
    h = np.zeros(n)
    # h[j] = a[k_star - j] if 0 <= k_star - j < n
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    # For j in [lo, hi): k_star - j is in [0, n-1]
    # h[j] = a[k_star - j]
    j_range = np.arange(lo, hi)
    h[j_range] = a[k_star - j_range]
    cm = float(c[k_star])
    grad_z = 2.0 * a * (h - cm)
    return grad_z


def _topk_logsumexp_obj(z, beta, k_top):
    """Objective: beta * logsumexp(top-k of (a*a) / beta)."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    order = np.argsort(c)[::-1]
    top_k = order[:min(k_top, len(c))]
    ct = c[top_k]
    cm = float(ct.max())
    if cm < 1e-30:
        return 0.0
    shifted = (ct - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _topk_logsumexp_grad(z, beta, k_top):
    """Gradient of top-k log-sum-exp objective."""
    n = len(z)
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    order = np.argsort(c)[::-1]
    top_k = order[:min(k_top, len(c))]
    ct = c[top_k]
    cm = float(ct.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (ct - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # Weighted combination of subgradients at top peaks
    grad_a = np.zeros(n)
    for wi, ki in zip(w, top_k):
        h = np.zeros(n)
        lo = max(0, int(ki) - n + 1)
        hi_idx = min(n, int(ki) + 1)
        j_range = np.arange(lo, hi_idx)
        h[j_range] = a[int(ki) - j_range]
        grad_a += wi * 2.0 * h
    # Transform to z-space
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta):
    n = len(z)
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # Weighted subgradient
    h = np.zeros(n)
    # w is over all convolution indices, compute weighted h
    # h[j] = sum_k w[k] * a[k-j] for valid k-j
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _adam_peak_gd(a_init, deadline, rng, lr_init=0.1, max_iter=200000, k_top=3):
    """Adam gradient descent using exact subgradient at top peaks."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    no_improve = 0
    patience = 5000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        score = float(2.0 * n * c.max())

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient using top-k peaks
        k_used = min(k_top, len(c))
        order = np.argsort(c)[::-1][:k_used]
        cm = float(c[order[0]])

        if cm < 1e-30:
            break

        # Use top-k weighted subgradient
        # Simple: use the argmax subgradient
        k_star = int(order[0])
        grad_z = _subgradient_at_peak(a, c, k_star)

        # If multiple peaks are close, blend
        if k_top > 1 and k_used > 1:
            gap = c[order[0]] - c[order[-1]]
            if gap < 0.1 * cm:
                # Blend top-k gradients
                for ki in order[1:]:
                    grad_z += 0.5 * _subgradient_at_peak(a, c, int(ki))

        if not np.all(np.isfinite(grad_z)):
            break

        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        # Adaptive learning rate
        if t % 5000 == 0:
            lr = max(0.005, lr * 0.85)

        # Restart if stuck
        if no_improve > patience:
            noise_scale = min(0.05, 0.01 * no_improve / patience)
            z = z + noise_scale * rng.normal(size=n)
            z = z - z.mean()
            z = np.clip(z, -60, 60)
            no_improve = 0
            if t % 10000 == 0:
                lr = lr * 1.2  # bump lr on restart

    return best_a, best_score


def _lbfgs_continuation(a_init, deadline, rng, n_betas=15, use_topk=False, k_top=3):
    """L-BFGS with continuation on log-sum-exp or top-k log-sum-exp."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_start = 0.5 * max_c0
    beta_end = 1e-7 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + time_budget

        maxiter = 300 if beta > 0.1 * max_c0 else (1000 if beta > 0.01 * max_c0 else 3000)
        maxfun = maxiter * 8

        if use_topk:
            try:
                result = minimize(
                    _topk_logsumexp_obj, z, args=(beta, k_top),
                    method='L-BFGS-B', jac=_topk_logsumexp_grad,
                    options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                             'maxls': 40, 'maxfun': maxfun}
                )
            except Exception:
                continue
        else:
            try:
                result = minimize(
                    _logsumexp_obj, z, args=(beta,),
                    method='L-BFGS-B', jac=_logsumexp_grad,
                    options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                             'maxls': 40, 'maxfun': maxfun}
                )
            except Exception:
                continue

        if result.x is not None and np.all(np.isfinite(result.x)):
            z_new = result.x
            z_new = np.clip(z_new, -80, 80)
            z_new = z_new - z_new.mean()
            a_new = _softmax_fn(z_new)
            if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()

    return best_a, best_score


def _coordinate_refine(a, deadline, rng, max_rounds=500):
    """Coordinate descent in log-space: try adjusting one coordinate at a time."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    c = _full_convolve(a, a)
    cm = float(c.max())

    # Find the top contributors to the peak
    k_star = int(np.argmax(c))
    valid_i = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
    if len(valid_i) == 0:
        return best_a, best_score

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c_cur = _full_convolve(a_cur, a_cur)
        sc_cur = float(2.0 * n * c_cur.max())

        if sc_cur < best_score:
            best_score = sc_cur
            best_a = a_cur.copy()

        k_star = int(np.argmax(c_cur))
        valid_i = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        if len(valid_i) == 0:
            break

        prod = a_cur[valid_i] * a_cur[k_star - valid_i]
        # Try top contributors
        order = np.argsort(prod)[::-1][:12]

        improved = False
        for idx_pos in order:
            idx = valid_i[idx_pos]
            # Try small adjustments in both directions
            for delta in [0.02, 0.01, 0.005, 0.002]:
                for direction in [-1, 1]:
                    z_trial = z.copy()
                    z_trial[idx] += direction * delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random coordinate adjustments
            for _ in range(5):
                idx = rng.integers(0, n)
                for delta in [0.01, 0.005]:
                    for direction in [-1, 1]:
                        z_trial = z.copy()
                        z_trial[idx] += direction * delta
                        a_trial = _softmax_fn(z_trial)
                        sc = _score_norm_fn(a_trial)
                        if sc < best_score - 1e-10:
                            best_score = sc
                            best_a = a_trial.copy()
                            z = z_trial.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved and rnd > 50:
            break

    return best_a, best_score


def _peak_surgery(a, deadline, rng, max_rounds=300):
    """Surgically reduce peaks by identifying and adjusting top contributors."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        sc_cur = float(2.0 * n * cm)

        if sc_cur < best_score:
            best_score = sc_cur
            best_a = a_cur.copy()

        # Find top 3 peaks
        order = np.argsort(c)[::-1][:3]
        improved = False

        for pk in order:
            pk = int(pk)
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            order_idx = np.argsort(prod)[::-1][:6]

            for idx_pos in order_idx[:4]:
                idx = valid_i[idx_pos]
                for delta in [0.03, 0.015, 0.008]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            if rnd > 30:
                break
            # Add small noise to escape
            z = z + 0.01 * rng.normal(size=n)
            z = z - z.mean()
            z = np.clip(z, -60, 60)

    return best_a, best_score


def _adaptive_beta(c, top_k=10):
    """Adaptive beta for log-sum-exp based on peak structure."""
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(top_k, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.02 * cm:
        return max(1e-10, 0.02 * cm)
    elif gap > 0.3 * cm:
        return max(1e-10, 0.005 * cm)
    else:
        return max(1e-10, 0.5 * gap)


def _simulated_annealing_polish(a, deadline, rng, n_iter=5000):
    """SIMMAGIC-style: random perturbations accepted if they reduce the peak."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    current_score = best_score
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))

    temp = 0.05
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        # Pick a coordinate to perturb (bias toward peak contributors)
        a_cur = _softmax_fn(z)
        valid_i = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        if len(valid_i) == 0:
            break

        # 70% chance pick from peak contributors, 30% random
        if rng.random() < 0.7:
            idx = valid_i[rng.integers(0, len(valid_i))]
        else:
            idx = rng.integers(0, n)

        delta = rng.normal(0, temp)
        z_trial = z.copy()
        z_trial[idx] += delta
        z_trial = z_trial - z_trial.mean()
        z_trial = np.clip(z_trial, -60, 60)
        a_trial = _softmax_fn(z_trial)
        sc_trial = _score_norm_fn(a_trial)

        # Metropolis acceptance
        if sc_trial < current_score or (np.random.random() < np.exp(-(sc_trial - current_score) / (temp * current_score))):
            z = z_trial
            current_score = sc_trial
            if current_score < best_score:
                best_score = current_score
                best_a = a_trial.copy()
                c = _full_convolve(a_trial, a_trial)
                k_star = int(np.argmax(c))

        # Cool down
        if it % 500 == 0:
            temp *= 0.9

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        _update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(4096, 1.5)
        _update_best(a0, _score_norm_fn(a0), "default_ramp")

    n_best = len(best_seq) if best_seq is not None else 4096

    # === PHASE 1: L-BFGS continuation from parent (n=4096) ===
    # Time: ~40%
    phase1_end = t_start + (budget_s - 20.0) * 0.35
    print(f"\nPhase 1: L-BFGS continuation n={n_best} (deadline +{phase1_end-t_start:.0f}s)")

    if time.monotonic() < phase1_end - 10 and best_seq is not None:
        a_p1, sc_p1 = _lbfgs_continuation(best_seq, phase1_end, rng, n_betas=20)
        _update_best(a_p1, sc_p1, "phase1_lbfgs")

        # Try top-k variant
        if time.monotonic() < phase1_end - 30:
            a_p1b, sc_p1b = _lbfgs_continuation(best_seq, phase1_end, rng, n_betas=10, use_topk=True, k_top=5)
            _update_best(a_p1b, sc_p1b, "phase1_topk")

    print(f"  After Phase 1: best={best_score:.10f}")

    # === PHASE 2: Adam peak GD (n=4096) ===
    phase2_end = t_start + (budget_s - 20.0) * 0.55
    print(f"\nPhase 2: Adam peak GD n={n_best} (deadline +{phase2_end-t_start:.0f}s)")

    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        # Multiple Adam runs with different hyperparameters
        alloc = (phase2_end - time.monotonic()) / 3.0
        for trial in range(3):
            if time.monotonic() > phase2_end - 5:
                break
            dl = time.monotonic() + min(alloc, 200.0)
            lr = [0.05, 0.1, 0.02][trial]
            kt = [3, 5, 2][trial]
            a_p2, sc_p2 = _adam_peak_gd(best_seq, dl, rng, lr_init=lr, k_top=kt)
            _update_best(a_p2, sc_p2, f"phase2_adam_{trial}")

    print(f"  After Phase 2: best={best_score:.10f}")

    # === PHASE 3: Coordinate refinement + peak surgery (n=4096) ===
    phase3_end = t_start + (budget_s - 20.0) * 0.65
    print(f"\nPhase 3: Coordinate refinement + peak surgery (deadline +{phase3_end-t_start:.0f}s)")

    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        dl = min(phase3_end - 5, time.monotonic() + 60.0)
        a_p3, sc_p3 = _coordinate_refine(best_seq, dl, rng)
        _update_best(a_p3, sc_p3, "phase3_coord")

        if time.monotonic() < phase3_end - 10:
            dl2 = min(phase3_end - 3, time.monotonic() + 40.0)
            a_p3b, sc_p3b = _peak_surgery(best_seq, dl2, rng)
            _update_best(a_p3b, sc_p3b, "phase3_surgery")

    print(f"  After Phase 3: best={best_score:.10f}")

    # === PHASE 4: Upsample to n=8192 and optimize ===
    phase4_end = t_start + (budget_s - 20.0) * 0.85
    print(f"\nPhase 4: Upsample to n=8192 (deadline +{phase4_end-t_start:.0f}s)")

    n8192 = 8192
    if time.monotonic() < phase4_end - 30 and best_seq is not None:
        a_8192 = _upsample(best_seq, n8192)
        sc_8192 = _score_norm_fn(a_8192)
        print(f"  Upsampled to n={n8192}: score={sc_8192:.10f}")
        _update_best(a_8192, sc_8192, "upsampled_8192")

        # L-BFGS on 8192
        if time.monotonic() < phase4_end - 60:
            alloc = min(300.0, phase4_end - time.monotonic() - 30)
            dl = time.monotonic() + alloc
            a_p4, sc_p4 = _lbfgs_continuation(a_8192, dl, rng, n_betas=15)
            _update_best(a_p4, sc_p4, "phase4_lbfgs_8192")

        # Adam on 8192
        if time.monotonic() < phase4_end - 30:
            alloc = min(200.0, phase4_end - time.monotonic() - 5)
            dl = time.monotonic() + alloc
            a_p4b, sc_p4b = _adam_peak_gd(best_seq if len(best_seq) < 8192 else best_seq, dl, rng, lr_init=0.03, k_top=3)
            if len(a_p4b) == n8192:
                _update_best(a_p4b, sc_p4b, "phase4_adam_8192")
            else:
                # It optimized at smaller n, try to use it
                a_up = _upsample(a_p4b, n8192)
                _update_best(a_up, _score_norm_fn(a_up), "phase4_adam_8192_up")

    print(f"  After Phase 4: best={best_score:.10f}")

    # === PHASE 5: Final polish ===
    print(f"\nPhase 5: Final polish (deadline +{deadline-t_start:.0f}s)")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_final = len(best_seq)

        # Coordinate refinement
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 20, time.monotonic() + 40.0)
            a_pf, sc_pf = _coordinate_refine(best_seq, dl, rng, max_rounds=200)
            _update_best(a_pf, sc_pf, "final_coord")

        # Peak surgery
        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 10, time.monotonic() + 20.0)
            a_pf2, sc_pf2 = _peak_surgery(best_seq, dl2, rng, max_rounds=100)
            _update_best(a_pf2, sc_pf2, "final_surgery")

        # Short Adam
        if time.monotonic() < deadline - 5:
            dl3 = min(deadline - 5, time.monotonic() + 15.0)
            a_pf3, sc_pf3 = _adam_peak_gd(best_seq, dl3, rng, lr_init=0.01, k_top=2, max_iter=50000)
            _update_best(a_pf3, sc_pf3, "final_adam")

    # === Phase 6: Try different starting points (if time) ===
    if time.monotonic() < deadline - 10 and best_seq is not None:
        print(f"\nPhase 6: Diversified restarts")
        n_cur = len(best_seq)

        # Exponential profile
        t = np.linspace(0, 1, n_cur)
        for rate in [2.0, 3.0, 4.0]:
            if time.monotonic() > deadline - 5:
                break
            a_exp = np.exp(rate * t)
            a_exp = a_exp / a_exp.sum()
            sc_exp = _score_norm_fn(a_exp)
            if sc_exp < best_score * 1.05:  # Only if close
                dl = min(deadline - 5, time.monotonic() + 30.0)
                a_r, sc_r = _adam_peak_gd(a_exp, dl, rng, lr_init=0.05, k_top=3)
                _update_best(a_r, sc_r, f"restart_exp_{rate}")

        # Perturbed parent
        if time.monotonic() < deadline - 15:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            for noise_scale in [0.05, 0.1, 0.2]:
                if time.monotonic() > deadline - 5:
                    break
                z_pert = z + noise_scale * rng.normal(size=n_cur)
                z_pert = np.clip(z_pert, -60, 60)
                z_pert = z_pert - z_pert.mean()
                a_pert = _softmax_fn(z_pert)
                dl = min(deadline - 5, time.monotonic() + 20.0)
                a_r, sc_r = _adam_peak_gd(a_pert, dl, rng, lr_init=0.03, k_top=3)
                _update_best(a_r, sc_r, f"restart_noise_{noise_scale}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.5)

        sc_final = _score_fn_full(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn_full(fallback):.10f}")
    return [float(x) for x in fallback]
