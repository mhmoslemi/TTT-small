```python
    n = len(a)
    c = full_convolve(a, a)
    M = c.max()
    K = np.where(c >= M - 1e-12)[0]
    
    # Compute g[j] = (1/|K|) * sum_{k in K} a[k-j] for valid indices
    g = np.zeros(n)
    for k in K:
        # a[k-j] is valid when 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        vals = a[k - j_end : k - j_start + 1][::-1]
        g[j_start:j_end+1] += vals
    g /= len(K)
    
    # Project to sum(a)=1 hyperplane
    g_proj = g - g.mean()
    
    # Update
    a_new = a - eta * g_proj
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    
    return a_new
```
