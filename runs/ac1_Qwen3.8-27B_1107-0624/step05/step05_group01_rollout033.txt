"""
Optimized multi-phase search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based peak balancing, coordinate descent with incremental convolution
updates, line search blending, and random restarts.
"""
import numpy as np
from scipy.optimize import linprog, minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_from_conv(n, c, s):
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))

def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a

def _softmax_fn(z):

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return e / s

def _lp_blend(a, M, max_constraints=2000, time_limit=60):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k (or top K), b >= 0.
    Return the optimal b.
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Build constraint matrix A where A[k, j] = a[k-j] if valid
    # (a*b)[k] = sum_j a[k-j] * b[j] for valid j
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # But for j in range(n), k-j is valid when k-n+1 <= j <= min(k, n-1)
    
    # Use all constraints if feasible, else top K by current c[k]
    c = _full_convolve(a)
    
    if max_constraints < conv_len:
        # Use top K constraints
        active = np.argsort(c)[::-1][:max_constraints]
    else:
        active = np.arange(conv_len)
    
    K = len(active)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(active):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    
    b_vec = np.full(K, M)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def _line_search_blend(a, b, S_target, max_iters=50):

    fc = f(c_pt)

            d_pt = c_pt

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

            off = k_idx - i

            p = 2.0 * x * a[off]
            if 2 * i >= k_lo and 2 * i <= k_hi:
                p[2 * i - k_lo] = x * x
            d = c[k_idx] - p

            best_val = float(c.max()) / (S * S)
            best_x = x
            best_c_aff = None

            for frac in [0.0, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                x_try = x * frac
                S_try = S - x + x_try
                if S_try < 1e-12:
                    continue
                p_try = 2.0 * x_try * a[off]
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    p_try[2 * i - k_lo] = x_try * x_try
                c_aff = d + p_try

                # Compute new max efficiently
                # New c: unchanged outside [k_lo, k_hi], c_aff inside
                # max = max(max(c outside range), max(c_aff))
                # For efficiency, just compute max of the full new array
                # But that requires a copy. Instead, use the fact that
                # the new max is max(c.max() if old max not in range, c_aff.max())
                # Simplest: just compute max(c_aff) and compare with a stored bg_max
                # For correctness, do the full thing but avoid full copy:
                new_max = max(c_aff.max(), c[k_idx].max())
                # This isn't quite right either since c[k_idx].max() might have been
                # the global max and is now reduced. Let's be safe:
                # The new global max = max(new values in range, old max outside range)
                # Old max outside range: we don't track this easily.
                # Just use the conservative upper bound:
                # new_max <= max(c.max(), c_aff.max())
                # And new_max >= c_aff.max()
                # For the objective, use new_max = max(c.max(), c_aff.max()) as upper bound
                # If it doesn't improve, we skip (safe)
                ub = max(c.max(), c_aff.max())
                val = ub / (S_try * S_try)

                if val < best_val - 1e-15:
                    # Verify with exact computation
                    c_new = c.copy()
                    c_new[k_idx] = c_aff
                    exact_max = c_new.max()
                    exact_val = exact_max / (S_try * S_try)
                    if exact_val < best_val - 1e-15:
                        best_val = exact_val
                        best_x = x_try
                        best_c_aff = c_aff.copy()

            if best_x != x and best_c_aff is not None:
                c[k_idx] = best_c_aff
                S = S - x + best_x
                a[i] = best_x
                any_improved = True

        if not any_improved:
            break

    c_final = _full_convolve(a)
    S_final = float(a.sum())
    return a, float(2.0 * n * c_final.max() / (S_final * S_final))


def _random_perturb(a, sigma):
    n = len(a)
    noise = np.random.randn(n) * sigma
    a_new = np.maximum(a + noise, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    S = a.sum()
    return a_new * (S / s)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lbfgs_softmin(a_init, deadline, beta_list=None, maxiter=3000):
    """L-BFGS on log-sum-exp surrogate of the max convolution."""
    from scipy.optimize import minimize
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init)
        m0 = float(c0.max())
        beta_list = [m0 * b for b in
                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break

        def obj_fn(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return 0.0
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
            return float(2.0 * n * lse)

        def grad_fn(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return np.zeros_like(z_in)
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            e = np.exp(shifted - max_s)
            w = e / e.sum()
            a_rev = a_in[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            W = float(np.dot(w, c_in))
            grad = 4.0 * n * a_in * (h - W)
            if not np.all(np.isfinite(grad)):
                return np.zeros_like(z_in)
            return grad

        try:
            result = minimize(obj_fn, z, method='L-BFGS-B', jac=grad_fn,
                              options={'maxiter': maxiter, 'ftol': 1e-18,
                                       'gtol': 1e-12, 'maxls': 80,
                                       'maxfun': 10000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. LP peak balancing with line-search blending (primary)
    2. L-BFGS log-sum-exp continuation (secondary)
    3. Coordinate descent polishing
    4. Random restarts + LP
    5. Final LP + CD
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 60.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_score = np.inf

    def update_best(a, score, label=""):
        nonlocal best_a, best_score
        if a is not None and np.isfinite(score) and score < best_score:
            best_a = a.copy()
            best_score = float(score)
            print(f"  NEW BEST: {score:.10f} (n={len(a)}) [{label}] "
                  f"t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score(parent), "greedy")

    print(f"Starting: n={n}, best={best_score:.10f}")

    # === Phase 1: LP with K=500 (500s) ===
    print("Phase 1: LP K=500")
    p1_end = min(deadline, t_start + 500.0)
    a1, sc1 = _lp_iterate(best_a, p1_end, max_rounds=200, K=500)
    update_best(a1, sc1, "lp_K500")

    # === Phase 2: LP with K=1000 (300s) ===
    print("Phase 2: LP K=1000")
    p2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < p2_end:
        a2, sc2 = _lp_iterate(best_a, p2_end, max_rounds=100, K=1000)
        update_best(a2, sc2, "lp_K1000")

    # === Phase 3: L-BFGS log-sum-exp (250s) ===
    print("Phase 3: L-BFGS continuation")
    p3_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < p3_end and best_a is not None:
        c_b = _full_convolve(best_a)
        m_b = float(c_b.max())
        betas = [m_b * b for b in [0.002, 0.001, 0.0005, 0.0002, 0.0001]]
        a3, sc3 = _lbfgs_softmin(best_a, p3_end, beta_list=betas, maxiter=5000)
        update_best(a3, sc3, "lbfgs")

    # === Phase 4: LP with K=2000 (200s) ===
    print("Phase 4: LP K=2000")
    p4_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < p4_end:
        a4, sc4 = _lp_iterate(best_a, p4_end, max_rounds=50, K=2000)
        update_best(a4, sc4, "lp_K2000")

    # === Phase 5: Coordinate descent (200s) ===
    print("Phase 5: Coordinate descent")
    p5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < p5_end and best_a is not None:
        a5, sc5 = _coord_descent(best_a, time_budget=180)
        update_best(a5, sc5, "coord_desc")

    # === Phase 6: Random restarts + quick LP (150s) ===
    print("Phase 6: Random restarts")
    p6_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < p6_end and best_a is not None:
        for attempt in range(8):
            if time.monotonic() > p6_end - 30:
                break
            sigma = 0.003 * (1.0 + 0.3 * attempt)
            a_r = _random_perturb(best_a, sigma)
            sc_r = _score(a_r)
            if sc_r < best_score:
                update_best(a_r, sc_r, f"perturb_{attempt}")
            # Quick LP
            a_lr, sc_lr = _lp_iterate(a_r, min(p6_end - 15, time.monotonic() + 40),
                                      max_rounds=8, K=300)
            update_best(a_lr, sc_lr, f"perturb_lp_{attempt}")

    # === Phase 7: Final LP + CD (remaining) ===
    if time.monotonic() < deadline - 20 and best_a is not None:
        print("Phase 7: Final polish")
        a_f, sc_f = _lp_iterate(best_a, deadline - 50, max_rounds=20, K=500)
        update_best(a_f, sc_f, "final_lp")
        if time.monotonic() < deadline - 20:
            a_fc, sc_fc = _coord_descent(a_f, time_budget=20)
            update_best(a_fc, sc_fc, "final_cd")

    # Final validation
    if best_a is not None:
        final = np.clip(best_a, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / (s * np.sqrt(s) / (s ** 1.5))  # keep same sum
            final = final
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}")

        if len(final) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
                else:
                    print("Evaluator returned inf, using greedy fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score(fb):.10f}")
    return [float(x) for x in fb]
