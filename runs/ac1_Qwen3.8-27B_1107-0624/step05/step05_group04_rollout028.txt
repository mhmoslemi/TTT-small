"""

Key improvements over previous:

5. Multi-restart strategy with spectral-aware perturbations

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _convolve(a):
    return _full_convolve(a, a)


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _topk_logsumexp_obj(z, beta, K):
    """Log-sum-exp over only the top K peaks for more focused optimization."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if K >= len(c):
        cm = float(c.max())
        if cm < 1e-30:
            return 0.0
        shifted = (c - cm) / beta
        return beta * (cm / beta + float(np.log(np.sum(np.exp(shifted)))))
    # Get top K values
    thresholds = np.partition(c, len(c) - K)[len(c) - K]
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    # Soft top-K: usesigmoid gating
    gate = 1.0 / (1.0 + np.exp(-(c - thresholds) / (beta * 0.1)))
    # Weighted log-sum-exp
    effective = c * gate
    em = float(effective.max())
    if em < 1e-30:
        return 0.0
    shifted = (effective - em) / beta
    return beta * (em / beta + float(np.log(np.sum(np.exp(shifted)))))


def _adaptive_beta(c, strength=1.0):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(10, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.01 * cm:
        return max(1e-12, 0.01 * cm * strength)
    elif gap > 0.3 * cm:
        return max(1e-12, 0.005 * cm * strength)
    else:
        return max(1e-12, 0.2 * gap * strength)


def _run_lbfgs_continuation(a_init, deadline, rng, n_betas=20):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # More aggressive schedule: start at 0.2*max_c, end at 1e-7*max_c
    beta_start = 0.2 * max_c0
    beta_end = 1e-7 * max_c0
    if beta_end < 1e-15:
        beta_end = 1e-15

    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 300.0)
        level_deadline = time.monotonic() + max(1.0, time_budget)

        # Scale iterations based on n and beta
        if n <= 512:
            maxiter_base = 2000
        elif n <= 2048:
            maxiter_base = 5000
        else:
            maxiter_base = 10000

        # More iterations for smaller beta (finer optimization)
        beta_ratio = beta / max_c0
        if beta_ratio > 0.1:
            maxiter = maxiter_base // 3
        elif beta_ratio > 0.01:
            maxiter = maxiter_base // 2
        else:
            maxiter = maxiter_base

        maxfun = maxiter * 8

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 80, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                z_new = np.clip(z_new, -100, 100)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_peak_targeted_descent(a_init, deadline, rng, max_rounds=500):
    """Directly target and reduce the top peaks of the convolution."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    round_count = 0
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Find top peaks
        top_k = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        improved = False
        for pk in peak_indices[:3]:
            # Valid indices for this peak: a[i] * a[pk-i] for valid i
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            # Sort by contribution (descending)
            order = np.argsort(prod)[::-1]

            # Try reducing top contributors
            for idx_pos in order[:6]:
                idx = valid_i[idx_pos]
                # Try various reduction amounts
                for delta in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

            # Also try increasing small elements that don't contribute to peaks
            if not improved:
                small_idx = np.argsort(a_cur)[:3]
                for idx in small_idx:
                    for delta in [0.005, 0.01, 0.02]:
                        z_trial = z.copy()
                        z_trial[idx] += delta
                        a_trial = _softmax_fn(z_trial)
                        sc = _score_norm_fn(a_trial)
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a_trial.copy()
                            z = z_trial.copy()
                            improved = True
                            break
                    if improved:
                        break

            if improved:
                break

        round_count += 1
        if not improved:
            # Try a spectral perturbation to escape local minimum
            if rnd % 50 == 49 and rnd < max_rounds - 10:
                z_pert = z + 0.05 * rng.normal(size=n)
                z_pert = z_pert - z_pert.mean()
                a_pert = _softmax_fn(z_pert)
                sc_pert = _score_norm_fn(a_pert)
                if sc_pert < best_score:
                    best_score = sc_pert
                    best_a = a_pert.copy()
                    z = z_pert.copy()
                else:
                    # Small random kick
                    z = z + 0.02 * rng.normal(size=n)
                    z = z - z.mean()

            if rnd > 100 and not improved:
                break

    return best_a, best_score


def _run_adam_refinement(a_init, deadline, rng, lr_init=0.05, max_iter=50000):
    """Adam gradient descent with adaptive beta for fine-tuning."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    c = _full_convolve(a_init, a_init)
    beta = _adaptive_beta(c)
    best_beta = beta

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)

        if it % 200 == 0:
            beta = _adaptive_beta(c)
            if beta < best_beta * 0.3:
                best_beta = beta

        cm = float(c.max())
        if cm < 1e-30:
            break

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()

        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 5000 == 0:
            lr = max(0.005, lr * 0.85)

        if t % 15000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _make_hann(n):
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.sin(np.pi * (t + 0.5 / n)) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_super_gaussian(n, p=4.0, center=0.5, width=0.3):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - center) / width) ** p)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_asymmetric(n, alpha=0.3):
    """Asymmetric profile: peaks near position alpha."""
    t = np.linspace(0, 1, n)
    # Skewed Gaussian-like
    sigma_l = 0.25
    sigma_r = 0.35
    a = np.where(t <= alpha,
                 np.exp(-((t - alpha) ** 2) / (2 * sigma_l ** 2)),
                 np.exp(-((t - alpha) ** 2) / (2 * sigma_r ** 2)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gmm(n, K=3):
    """Gaussian mixture model initialization."""
    rng = np.random.default_rng(42)
    weights = rng.dirichlet(np.ones(K))
    means = np.sort(rng.uniform(0.1, 0.9, K))
    sigmas = rng.uniform(0.1, 0.3, K)
    t = np.linspace(0, 1, n)
    a = np.zeros(n)
    for k in range(K):
        a += weights[k] * np.exp(-0.5 * ((t - means[k]) / sigmas[k]) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    names = []
    # Hann window
    profiles.append(_make_hann(n))
    names.append("hann")
    # Super-Gaussian with various parameters
    for p in [3.0, 4.0, 6.0]:
        for center in [0.35, 0.5, 0.65]:
            profiles.append(_make_super_gaussian(n, p=p, center=center, width=0.35))
            names.append(f"sg_p{p}_c{center}")
    # Asymmetric
    for alpha in [0.3, 0.4, 0.6, 0.7]:
        profiles.append(_make_asymmetric(n, alpha=alpha))
        names.append(f"asym_{alpha}")
    # GMM
    for K in [2, 3, 4]:
        profiles.append(_make_gmm(n, K=K))
        names.append(f"gmm_{K}")
    # Ramps
    for s in [1.0, 2.0, 3.0]:
        profiles.append(_make_ramp(n, s))
        names.append(f"ramp_{s}")
    # Raised cosine variants
    t = np.linspace(0, 1, n, endpoint=False)
    for phase in [0.0, 0.2, 0.5]:
        a = 0.5 * (1 + np.cos(2 * np.pi * (t + phase)))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        names.append(f"raised_cos_{phase}")

    return profiles, names


def _try_parent_sequence():
    """Try to load and use the parent sequence."""
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                return parent_raw / s
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization function."""
    t_start = time.monotonic()
    effective_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_start + effective_budget
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf
    best_n = 0

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score, best_n
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Initialization ---
    parent = _try_parent_sequence()
    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")

    if best_seq is None:
        a0 = _make_hann(256)
        update_best(a0, _score_norm_fn(a0), "default_hann")

    print(f"Starting best: {best_score:.10f}, n={best_n}")

    # --- Phase 1: Multi-scale optimization (55% of time) ---
    phase1_end = t_start + effective_budget * 0.55
    print(f"\nPhase 1: Multi-scale optimization (deadline t={phase1_end - t_start:.0f}s)")

    # Try multiple initializations at n=128 and pick the best
    n_start = 128
    profiles_128, names_128 = _make_profiles(n_start)

    best_init_a = None
    best_init_sc = np.inf
    for idx, prof in enumerate(profiles_128):
        sc = _score_norm_fn(prof)
        if sc < best_init_sc:
            best_init_sc = sc
            best_init_a = prof.copy()
            best_init_name = names_128[idx]

    print(f"  Best initial profile: {best_init_name}, score_norm={best_init_sc:.10f}")

    # Quick optimization at n=128
    quick_deadline = min(phase1_end, t_start + 25.0)
    a_cur, sc_cur = _run_lbfgs_continuation(best_init_a, quick_deadline, rng, n_betas=10)
    print(f"  After n=128 optimization: score_norm={sc_cur:.10f}")

    # Multi-scale up
    scales = [256, 512, 1024, 2048, 4096]
    time_alloc = {256: 30, 512: 50, 1024: 80, 2048: 130, 4096: 180}
    n_betas_per_scale = {256: 12, 512: 15, 1024: 18, 2048: 20, 4096: 22}

    for target_n in scales:
        if time.monotonic() > phase1_end - 15:
            break

        a_target = _upsample(a_cur, target_n)
        sc_target = _score_norm_fn(a_target)

        alloc = time_alloc.get(target_n, 60)
        opt_deadline = min(phase1_end - 5, time.monotonic() + alloc)
        nb = n_betas_per_scale.get(target_n, 15)
        a_opt, sc_opt = _run_lbfgs_continuation(a_target, opt_deadline, rng, n_betas=nb)
        print(f"  n={target_n}: upsample={sc_target:.10f}, optimized={sc_opt:.10f}")

        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt

    if sc_cur < best_score:
        update_best(a_cur, sc_cur, "multi_scale")
    else:
        a_cur = best_seq.copy()
        sc_cur = best_score

    # Try n=8192 if time allows and improvement is significant
    if time.monotonic() < phase1_end - 60 and sc_cur < 1.53:
        print(f"  Trying n=8192...")
        a_8192 = _upsample(a_cur, 8192)
        sc_8192 = _score_norm_fn(a_8192)
        opt_dl_8192 = min(phase1_end - 5, time.monotonic() + 120.0)
        a_8192_opt, sc_8192_opt = _run_lbfgs_continuation(a_8192, opt_dl_8192, rng, n_betas=15)
        print(f"  n=8192: upsample={sc_8192:.10f}, optimized={sc_8192_opt:.10f}")
        if sc_8192_opt < best_score:
            update_best(a_8192_opt, sc_8192_opt, "n8192")
            a_cur = a_8192_opt
            sc_cur = sc_8192_opt

    print(f"\n  Phase 1 complete: best_score={best_score:.10f}, n={best_n}")

    # --- Phase 2: Aggressive fine L-BFGS (25% of time) ---
    phase2_end = t_start + effective_budget * 0.80
    print(f"\nPhase 2: Fine L-BFGS (deadline t={phase2_end - t_start:.0f}s)")

    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        # Run with many small-beta steps
        opt_deadline = min(phase2_end - 20, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
        a_p2, sc_p2 = _run_lbfgs_continuation(best_seq, opt_deadline, rng, n_betas=25)
        update_best(a_p2, sc_p2, "phase2_lbfgs")

        # Restart with perturbation
        if time.monotonic() < phase2_end - 30:
            n_best = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()

            # Try multiple perturbations
            for pert_scale in [0.05, 0.1, 0.2]:
                if time.monotonic() > phase2_end - 20:
                    break
                noise = pert_scale * rng.normal(size=n_best)
                z_pert = z + noise
                z_pert = np.clip(z_pert, -80, 80)
                z_pert = z_pert - z_pert.mean()
                a_pert = _softmax_fn(z_pert)
                opt_dl2 = min(phase2_end - 10, time.monotonic() + 80.0)
                a_p2b, sc_p2b = _run_lbfgs_continuation(a_pert, opt_dl2, rng, n_betas=12)
                update_best(a_p2b, sc_p2b, f"phase2_restd_{pert_scale}")

    print(f"\n  Phase 2 complete: best_score={best_score:.10f}, n={best_n}")

    # --- Phase 3: Peak targeting + Adam (15% of time) ---
    phase3_end = t_start + effective_budget * 0.95
    print(f"\nPhase 3: Peak targeting + Adam (deadline t={phase3_end - t_start:.0f}s)")

    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        # Peak-targeted descent
        ps_deadline = min(phase3_end - 40, time.monotonic() + (phase3_end - time.monotonic()) * 0.4)
        a_ps, sc_ps = _run_peak_targeted_descent(best_seq, ps_deadline, rng)
        update_best(a_ps, sc_ps, "peak_target")

        # Adam refinement
        if time.monotonic() < phase3_end - 20:
            gd_deadline = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()) * 0.5)
            a_gd, sc_gd = _run_adam_refinement(best_seq, gd_deadline, rng, lr_init=0.03)
            update_best(a_gd, sc_gd, "adam_refine")

        # Another peak targeting round from best
        if time.monotonic() < phase3_end - 10:
            ps_dl2 = min(phase3_end - 5, time.monotonic() + 40.0)
            a_ps2, sc_ps2 = _run_peak_targeted_descent(best_seq, ps_dl2, rng, max_rounds=200)
            update_best(a_ps2, sc_ps2, "peak_target2")

    print(f"\n  Phase 3 complete: best_score={best_score:.10f}, n={best_n}")

    # --- Phase 4: Final tiny-beta L-BFGS polish (5% of time) ---
    print(f"\nPhase 4: Final polish")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        beta_tiny = max(1e-8, 1e-6 * max_c_final)

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()

        n_final = len(best_seq)
        maxiter = max(2000, min(20000, int(8000 * (4096 / max(n_final, 1)))))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta_tiny,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 100, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    print(f"  Tiny-beta polish: {best_score:.10f}")
        except Exception as ex:
            print(f"  Tiny-beta failed: {ex}")

        # One more Adam pass
        if time.monotonic() < deadline - 3:
            gd_dl = min(deadline - 1, time.monotonic() + 20.0)
            a_gd3, sc_gd3 = _run_adam_refinement(best_seq, gd_dl, rng, lr_init=0.005, max_iter=5000)
            update_best(a_gd3, sc_gd3, "final_adam")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_hann(256)

        sc_final = _score_fn(final_seq)
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}, elapsed={elapsed:.0f}s")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_hann(512)
    print(f"FALLBACK: n=512, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
