        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter=5000, ftol=1e-16):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-12,
                     'maxcor': 50, 'maxls': 300, 'disp': False}
        )
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _simplex_project(v, target=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.where(cond)[0][0]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _mirror_descent(a_init, deadline, n_steps=10000, eta_init=0.05, decay=0.9995):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    eta = eta_init

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        n_act = active.sum()
        if n_act == 0:
            active[int(np.argmax(c))] = True
            n_act = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]

        # Subgradient of max(c) is 2*h (factor of 2 from quadratic derivative)
        g = 2.0 * h

        # Mirror descent update
        z = np.log(np.maximum(a, 1e-30))
        z = z - eta * g
        z = z - z.max()
        exp_z = np.exp(z)
        a_new = exp_z / exp_z.sum()

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new
        eta *= decay

    return best_a, best_sc


def _proj_subgrad(a_init, deadline, n_steps=5000, eta_init=0.02):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    eta = eta_init

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        n_act = active.sum()
        if n_act == 0:
            active[int(np.argmax(c))] = True
            n_act = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        # Subgradient of f(a) = 2n*max(c) is 4n*(h - W) ... actually just 4n*h
        # Since on simplex, the projected gradient is: g = 4n*h - (sum of 4n*h)*a
        # But simpler: just use g = 4n*h and project
        g = 4.0 * n * h
        # Center the gradient (projection handles the constraint)
        a_temp = a - eta * (g - g.mean())
        a_new = _simplex_project(a_temp, 1.0)

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new
        eta *= 0.999

    return best_a, best_sc


def _perturb_small(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3, amp_scale=0.01):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 2))
        amp = amp_scale * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t) + 0.1 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lbfgs_schedule(a_start, deadline, min_time_remaining=50.0, verbose=True):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    # More aggressive and longer beta schedule
    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.012, 0.006,
                              0.003, 0.0015, 0.0007, 0.0003, 0.00015, 0.00007])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * (n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 10))))

        a_new = _lbfgs_run(current_a, beta, n, max_iter=max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-14:
            if verbose:
                print(f"    beta={beta:.2e}: score={sc:.10f} (iter={i})")
            current_a = a_new
            current_sc = sc
        elif verbose:
            print(f"    beta={beta:.2e}: no improvement, skipping")

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}")
        update_best(a0, _score(a0), "init")

    # === Phase 1: Main L-BFGS optimization ===
    print(f"\nPhase 1: Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _lbfgs_schedule(a0, deadline, min_time_remaining=400.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "main_lbfgs")

    # === Phase 2: Mirror descent fine-tuning ===
    print(f"\nPhase 2: Mirror descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        md_deadline = min(deadline - 150.0, time.monotonic() + 120.0)
        a_md, sc_md = _mirror_descent(best_seq, md_deadline, n_steps=8000, eta_init=0.03)
        update_best(a_md, sc_md, "mirror_descent")

    # === Phase 3: Second L-BFGS pass with very small betas ===
    print(f"\nPhase 3: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([0.00005, 0.00002, 0.00001, 0.000005, 0.000002])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - 60 - time.monotonic()
            if remaining < 3:
                break
            max_iter = min(3000, max(200, int(remaining / 0.03)))
            a_new = _lbfgs_run(best_seq, beta, n, max_iter=max_iter)
            sc = _score(a_new)
            update_best(a_new, sc, f"fine_{i}")
            if sc < best_score:
                best_seq = a_new

    # === Phase 4: Projected subgradient ===
    print(f"\nPhase 4: Projected subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        ps_deadline = min(deadline - 50.0, time.monotonic() + 60.0)
        a_ps, sc_ps = _proj_subgrad(best_seq, ps_deadline, n_steps=3000, eta_init=0.01)
        update_best(a_ps, sc_ps, "proj_subgrad")

    # === Phase 5: Iterated local search ===
    print(f"\nPhase 5: Iterated local search (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        n_trials = min(8, max(2, int(remaining / 60)))

        for i in range(n_trials):
            if time.monotonic() > deadline - 15:
                break

            # Different perturbation strategies
            if i % 3 == 0:
                p_start = _perturb_small(best_seq, 0.005, rng)
            elif i % 3 == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=2, amp_scale=0.005)
            else:
                p_start = _perturb_small(best_seq, 0.002, rng)

            sc_start = _score(p_start)

            # Quick L-BFGS from perturbed point
            a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=20.0, verbose=False)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"ils_{i}")

            # Quick mirror descent on the result
            if sc_r < best_score + 0.01 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _mirror_descent(a_r, deadline - 10, n_steps=2000, eta_init=0.02)
                update_best(a_r2, sc_r2, f"ils_{i}_md")

    # === Phase 6: Final ultra-fine polish ===
    print(f"\nPhase 6: Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta_final = mb * 0.000001
        a_fin = _lbfgs_run(best_seq, beta_final, n, max_iter=500)
        sc_fin = _score(a_fin)
        update_best(a_fin, sc_fin, "final_lbfgs")

        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _mirror_descent(best_seq, deadline - 2, n_steps=1000, eta_init=0.005)
            update_best(a_fin2, sc_fin2, "final_md")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
