```python
def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'feasibility_tolerance': 1e-10})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None

    return g, M
```

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive LP with binary-search line search, sharp-to-smooth
log-sum-exp gradient descent (Adam), targeted peak shaving, and
multi-restart local search to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _conv(a):
    return _full_convolve(a, a)


def _conv_max(a):
    c = _conv(a)
    return float(c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        if len(j_range) > 0:
            rows_list.append(np.full(len(j_range), k, dtype=np.int32))
            cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=20):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-10})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_direction(f, rows, cols):
    c = _conv(f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols)
    if g_0 is None or g_0.sum() < 1e-15:
        return None, M
    g = g_0 / g_0.sum()
    return g, M


def _binary_search_step(f, g, M_target, max_iters=20):
    lo = 0.0
    hi = 1.0
    c_max_g = _conv_max(g)
    if c_max_g <= M_target:
        return 1.0
    for _ in range(max_iters):
        mid = (lo + hi) / 2.0
        f_mid = (1.0 - mid) * f + mid * g
        if _conv_max(f_mid) <= M_target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-5:
            break
    return lo


def _lp_phase(a, rows, cols, deadline, label="LP"):
    prev_m = _conv_max(a)
    stall = 0
    count = 0
    reductions = [0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    while time.monotonic() < deadline:
        t0 = time.monotonic()
        g, M = _lp_direction(a, rows, cols)
        if g is None:
            stall += 5
            if stall > 40:
                break
            continue
        found = False
        for red in reductions:
            M_target = M * (1.0 - red)
            t_opt = _binary_search_step(a, g, M_target, max_iters=18)
            if t_opt > 0.003:
                a_new = (1.0 - t_opt) * a + t_opt * g
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s > 1e-15:
                    a_new = a_new / s
                new_m = _conv_max(a_new)
                if new_m < M - 1e-12:
                    a = a_new
                    found = True
                    break
        if not found:
            t_small = 0.03
            a_new = (1.0 - t_small) * a + t_small * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            new_m = _conv_max(a_new)
            if new_m < M - 1e-12:
                a = a_new
        count += 1
        dt = time.monotonic() - t0
        cur_m = _conv_max(a)
        if cur_m < prev_m - 1e-13:
            stall = 0
        else:
            stall += 1
        if count % 10 == 0 or stall > 15:
            sc = _score(a)
            print(f"  {label}[{count}] score={sc:.10f} M={cur_m:.10f} stall={stall} dt={dt:.2f}s t={time.monotonic()-t0:.0f}s")
        if stall > 50:
            print(f"  {label} stalled at iter {count}")
            break
        prev_m = cur_m
    return a, count


def _adam_phase(a_init, deadline, beta_start, beta_end, lr_init, max_iter, label="Adam"):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _conv_max(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    cur_lr = lr_init
    prev_best = best_sc
    lr_history = []

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _conv(a)
        c_max = float(c.max())
        progress = min(1.0, t / max_iter)
        beta = beta_start * (max(beta_end, beta_start * 1e-6) / beta_start) ** progress
        c_shifted = (c - c_max) / max(beta, 1e-12)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = cur_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.4:
            update = update * (0.4 / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        if t % 200 == 0:
            sc = _conv_max(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
        if t % 8000 == 0:
            if best_sc < prev_best * 0.9998:
                cur_lr = min(0.04, cur_lr * 1.08)
            elif best_sc > prev_best:
                cur_lr = max(0.002, cur_lr * 0.92)
            prev_best = best_sc
            lr_history.append(cur_lr)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _conv_max(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _peak_shave_step(a, n_peaks=3, alpha=0.3):
    n = len(a)
    c = _conv(a)
    c_max = float(c.max())
    top_idx = np.argsort(c)[::-1][:n_peaks]
    pressure = np.zeros(n)
    for k in top_idx:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        partners = k - np.arange(j_start, j_end + 1)
        valid = (partners >= 0) & (partners < n)
        if valid.any():
            pressure[j_start:j_end + 1][valid] += c[k] * a[partners[valid]]
    max_p = pressure.max()
    if max_p < 1e-30:
        return a, c_max
    w = np.exp(-alpha * pressure / max_p)
    w = w / w.sum()
    step = min(0.15, alpha * 0.1)
    a_new = (1.0 - step) * a + step * w
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    new_max = _conv_max(a_new)
    return a_new, new_max


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return np.exp(z - z.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

            full_sc = _score(a4)
            print(f"  FineAdam: score={full_sc:.10f}")
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a4.copy()

    # === Phase 5: Multi-restart (1500 - 1785s) ===
    p5_end = min(deadline, t_start + 1785.0)
    if time.monotonic() < p5_end:
        print(f"\n--- Phase 5: Multi-restart ---")
        attempts = 5
        attempts_per = max(1, int((p5_end - time.monotonic()) / attempts / 30.0))
        for att in range(attempts):
            if time.monotonic() > p5_end - 20:
                break
            sigma = 0.02 + 0.015 * att
            a_r = _perturb(best_seq, sigma, rng)
            a_r = _normalize(a_r)

            # Quick LP
            sub_deadline_lp = min(p5_end - 10, time.monotonic() + 40.0)
            a_r, _ = _lp_phase(a_r, rows, cols, sub_deadline_lp, f"R{att}_LP")

            # Quick Adam
            sub_deadline_ad = min(p5_end - 5, time.monotonic() + 60.0)
            if time.monotonic() < sub_deadline_ad:
                c_r = _conv(a_r)
                max_cr = float(c_r.max())
                a_r, _ = _adam_phase(a_r, sub_deadline_ad,
                                     beta_start=0.0002 * max_cr,
                                     beta_end=0.002 * max_cr,
                                     lr_init=0.006, max_iter=100000,
                                     label=f"R{att}_Adam")

            full_sc = _score(a_r)
            print(f"  Restart[{att}] (sigma={sigma:.3f}): score={full_sc:.10f}")
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a_r.copy()
                print(f"    *** NEW BEST: {best_score:.10f} ***")

    # === Final: Ultra-fine Adam on best ===
    final_deadline = deadline - 3.0
    if time.monotonic() < final_deadline:
        print(f"\n--- Final polishing ---")
        c_f = _conv(best_seq)
        max_cf = float(c_f.max())
        best_seq, scf = _adam_phase(best_seq, final_deadline,
                                    beta_start=0.00005 * max_cf,
                                    beta_end=0.0005 * max_cf,
                                    lr_init=0.003, max_iter=200000,
                                    label="Final")
        full_sc = _score(best_seq)
        print(f"Final Adam: score={full_sc:.10f}")
        if full_sc < best_score:
            best_score = full_sc

    # Final validation
    if best_seq is not None:
        final_out = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_out.sum())
        if s < 0.01:
            final_out = np.ones(256)
        sc_final = _score(final_out)
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_out)}, time={elapsed:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_out]))
            print(f"Evaluator confirms: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final_out]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
