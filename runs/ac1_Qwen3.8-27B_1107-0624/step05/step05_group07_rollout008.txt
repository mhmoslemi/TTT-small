import numpy as np
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    """Score with sum(a) assumed to be 1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _simplex_proj(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

        sc = _score_norm_fn(a_new)

    n = len(a)
    g = np.zeros(n)
    for k in lags:
        # Term 1: a[k-j] for valid j
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += a[k - idx]
        # Term 2: a[j-k] for valid j
        # j ranges from max(k, 0) to n-1, and j-k ranges from 0 to n-1
        j_start2 = max(k, 0)
        if j_start2 < n:
            idx2 = np.arange(j_start2, n)
            g[idx2] += a[idx2 - k]
    g /= len(lags)
    return g

    n = len(a)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _simplex_proj(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _compute_subgrad_one(a, k):

def _upsample(a, n_new):
    """Upsample a sequence to length n_new using interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()

    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002, 0.0001]
        beta_list = [b * max_c0 for b in fracs]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        
        def obj(z):
            return _logsumexp_obj_and_grad(z, beta, n)[0]
        
        def grad(z):
            return _logsumexp_obj_and_grad(z, beta, n)[1]
        
        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',

            if result.x is not None and np.all(np.isfinite(result.x)):

                    best_a = a_new.copy()

def _make_asymmetric_profiles(n):

    a = t * (1-t)**2 + 0.01

    return profiles

The max of (2-x)*e^{-ax} on [1,2]: derivative is -e^{-ax} - a(2-x)e^{-ax} = -e^{-ax}(1+a(2-x)). This is always negative for a > 0, so the max is at x=1, value = e^{-a}.

So for a >= 1: max(f*f) = C^2 * (1/a) * e^{-1}.
C is determined by C * integral_0^1 e^{-at} dt = 1 => C * (1-e^{-a})/a = 1 => C = a/(1-e^{-a}).
So max(f*f) = [a/(1-e^{-a})]^2 * (1/a) * e^{-1} = a * e^{-1} / (1-e^{-a})^2.

For a = 1: max = 1 * e^{-1} / (1-e^{-1})^2 = 0.368 / (0.632)^2 = 0.368 / 0.4 = 0.92.
Score ≈ 2 * 0.92 = 1.84.

For a = 2: max = 2 * e^{-1} / (1-e^{-2})^2 = 0.736 / (0.865)^2 = 0.73
