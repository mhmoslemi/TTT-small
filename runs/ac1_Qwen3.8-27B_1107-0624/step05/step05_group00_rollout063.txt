def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv(a,a))"""
    n = len(a)

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _subgradient(a, kstar):
    """Subgradient of max(c) at a, where kstar = argmax c.
    g[m] = 2*a[kstar-m] if 0<=kstar-m<n else 0."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, kstar - (n - 1))
    hi = min(n - 1, kstar)
    for m in range(lo, hi + 1):
        g[m] = 2.0 * a[kstar - m]
    return g


def _fista_smoothed(a_init, betas, deadline, max_iter_per_beta=20000):
    """FISTA on the simplex with softmax-smoothed max, geometric beta schedule."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        # Recompute t_val for each beta phase (restart momentum)
        t_val = 1.0
        a_curr = best_a.copy()
        a_prev = a_curr.copy()
        
        L = 4.0 * n / max(beta, 1e-10)
        step = 1.0 / L
        # Cap step to avoid instability
        step = min(step, 0.5 / max(1.0, n))
        
        for it in range(max_iter_per_beta):
            if time.monotonic() > deadline:
                break
            
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            c_y = _full_convolve(y, y)
            grad = _grad_smooth_max(y, c_y, beta)
            
            a_next = _proj_simplex(y - step * grad)
            
            if it % 500 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
            
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
        
        sc = _score_norm(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score

            grad = _grad_smooth_max(a, c, max(beta, 1e-10))
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_from_z(z)
                sc = _score_norm(a_chk)

    sc_final = _score_norm(a_final)

def _subgrad_averaging(a_init, deadline, eta0=0.1, decay=500):

    a_avg = a_curr.copy()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    T = 0
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        T += 1
        eta = eta0 / (1.0 + T / decay)
        
        c, cm, kstar = _conv_and_max(a_curr)
        g = _subgradient(a_curr, kstar)
        
        # Gradient step with simplex projection
        a_next = _proj_simplex(a_curr - eta * g)
        
        # Update average
        a_avg = (T - 1) / T * a_avg + (1.0 / T) * a_next
        
        a_curr = a_next
        
        if T % 200 == 0:
            sc_avg = _score_norm(a_avg)
            sc_curr = _score_norm(a_curr)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()
            if sc_curr < best_score:
                best_score = sc_curr
                best_a = a_curr.copy()

    return best_a, best_score


def _mirror_descent(a_init, deadline, eta0=0.05, decay=200):
    """Mirror descent with entropy kernel (multiplicative weight update).
    
    a_new = a * exp(-eta * grad) / sum
    This naturally maintains the simplex constraint.
    """
    n = len(a_init)
    a_curr = a_init.copy()
    a_avg = a_curr.copy()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    T = 0
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        T += 1
        eta = eta0 / (1.0 + T / decay)
        
        c, cm, kstar = _conv_and_max(a_curr)
        g = _subgradient(a_curr, kstar)
        
        # Multiplicative weight update
        log_a = np.log(np.maximum(a_curr, 1e-30))
        log_a_new = log_a - eta * g
        log_a_new = log_a_new - np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s = a_new.sum()
        if s < 1e-300:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        a_avg = (T - 1) / T * a_avg + (1.0 / T) * a_new
        a_curr = a_new
        
        if T % 100 == 0:
            sc_avg = _score_norm(a_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()
            sc_curr = _score_norm(a_curr)
            if sc_curr < best_score:
                best_score = sc_curr
                best_a = a_curr.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "geometric":
        # Geometric with ratio chosen to balance first few conv coefficients
        r = 0.7
        a = r ** np.arange(n, dtype=np.float64)
        return a / a.sum()
    elif kind == "linear_decr":
        a = np.linspace(1.0, 0.1, n)
        return a / a.sum()
    elif kind == "bimodal":
        a = np.zeros(n)
        a[:n//3] = 1.0
        a[2*n//3:] = 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_from_z(z)


def _perturb_block(a, rng):
    """Randomly scale a contiguous block."""
    n = len(a)
    block_len = rng.integers(max(2, n // 16), max(3, n // 4))
    start = rng.integers(0, n - block_len)
    factor = rng.uniform(0.3, 3.0)
    a_new = a.copy()
    a_new[start:start+block_len] *= factor
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def _perturb_shift(a, rng):
    """Circular shift by a small amount."""
    n = len(a)
    shift = int(rng.integers(1, max(2, n // 64)))
    a_new = np.roll(a, shift)
    return a_new / a_new.sum()


def _truncation_trick(a, n_target, top_fraction=0.8):
    """Zero out the smallest (1-top_fraction) elements and renormalize."""
    n = len(a)
    n_keep = max(1, int(n * top_fraction))
    idx = np.argsort(a)
    a_new = a.copy()
    a_new[idx[:n - n_keep]] = 0.0
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

    print("Phase 2: Adam log-space")
    phase2_end = min(deadline, t_start + 700.0)
