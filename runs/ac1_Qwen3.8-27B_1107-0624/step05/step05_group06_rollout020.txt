def active_set_direction(a, active_indices):
    n = len(a)
    # LP: maximize sum(d) subject to (d*a)[j] <= 0 for j in active, d[i] >= -a[i]
    # linprog minimizes c^T x, so we minimize -sum(d)
    c = -np.ones(n)

    # (d*a)[j] = sum_i d[i] * a[j-i] (with a[k]=0 if k<0 or k>=n)
    A_ub = np.zeros((len(active_indices), n))
    for row, j in enumerate(active_indices):
        for i in range(n):
            k = j - i
            if 0 <= k < n:
                A_ub[row, i] = a[k]
    
    b_ub = np.zeros(len(active_indices))
    
    # Bounds: d[i] >= -a[i], d[i] <= some upper bound (say 1)
    bounds = [(-a[i], 1.0) for i in range(n)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x if result.success else None

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = sfft.rfft(a, N)
    fb = sfft.rfft(b, N)
    r = sfft.irfft(fa * fb, N)[:out_n]

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)
    c_max = c.max()

    """
    Solve LP to find direction d that maximizes sum(d)
    subject to (d*a)[j] <= 0 for j in active_indices,
    and -a[i] <= d[i] <= 1.
    Returns the direction d or None if infeasible/unbounded.
    """
    n = len(a)
    K = len(active_indices)
    if K == 0:
        return np.ones(n) / n  # just increase everything
    
    # Objective: minimize -sum(d)
    c_obj = -np.ones(n)
    
    # Constraint matrix: (d*a)[j] <= 0 for each j in active
    # (d*a)[j] = sum_i d[i] * a[j-i] where a[k]=0 if k<0 or k>=n
    A_ub = np.zeros((K, n))
    for row, j in enumerate(active_indices):
        # d*a at position j: sum over i of d[i] * a[j-i]
        # a[j-i] is non-zero when 0 <= j-i < n, i.e., j-n+1 <= i <= j
        i_start = max(0, j - n + 1)
        i_end = min(n - 1, j)
        for i in range(i_start, i_end + 1):
            A_ub[row, i] = a[j - i]
    
    b_ub = np.zeros(K)
    
    # Bounds
    lower = -a.copy()
    upper = np.ones(n)
    bounds = list(zip(lower, upper))
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    elif result.status == 3:  # unbounded
        # This shouldn't happen with bounds, but just in case
        return np.ones(n) / n
    else:
        return None

    LP-based active set polishing.
    At each step, identify active set, solve LP for direction, take a step.
    """

    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        S = a.sum()
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Identify active set: positions within tol of max
        active_thresh = c_max * (1 - tol)
        active_mask = c >= active_thresh
        active_indices = np.where(active_mask)[0]
        
        if len(active_indices) == 0:
            break
        
        # Solve LP
        d = _lp_direction(a, active_indices)
        if d is None:
            # Try with smaller active set
            top_k = min(20, len(active_indices))
            sorted_idx = np.argsort(c)[::-1][:top_k]
            d = _lp_direction(a, sorted_idx)
            if d is None:
                break
        
        # Check if direction is useful
        d_sum = d.sum()
        if d_sum < 1e-10:
            # No useful direction found, try with fewer active constraints
            top_k = min(5, len(active_indices))
            sorted_idx = np.argsort(c)[::-1][:top_k]
            d = _lp_direction(a, sorted_idx)
            if d is None or d.sum() < 1e-10:
                break
        
        # Line search: find best step size
        # We want a + t*d >= 0 and the max conv doesn't increase too much
        t_max = min(1.0, min(a[i] / max(-d[i], 1e-15) for i in range(n) if d[i] < 0)) if np.any(d < 0) else 1.0
        t_max = max(0.01, min(1.0, t_max))
        
        # Try a few step sizes
        best_t = 0.0
        best_t_sc = sc
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            t = min(t, t_max)
            a_try = a + t * d
            if np.any(a_try < 0):
                continue
            sc_try = _score(a_try)
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t = t
        
        if best_t == 0:
            # Small step
            best_t = 0.01 * t_max
            a_try = a + best_t * d
            if np.any(a_try < 0):
                break
        
        a = a + best_t * d
        a = np.maximum(a, 0)
    
    # Final check
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc

        if remaining < 3:
            break

        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()

        est_iter_time = 0.004 * (n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_iter_time * 6))))

        try:
            result = minimize(
                _smoobj, z0, args=(beta, n),
                jac=lambda z: _smograd(z, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 40, 'maxls': 150, 'disp': False}
            )
            a_new = _softmax(result.x)
            sc = _score(a_new)
            if sc < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc
        except Exception:
            pass

    return current_a

def _lp_minimax_direction(a, active_indices):
    """
    Solve LP: minimize t subject to (d*a)[j] - t <= 0 for active j,
    d[i] >= -a[i], sum(d) = 1.
    Returns (d, t_opt) or (None, inf).
    """
    n = len(a)
    K = len(active_indices)
    if K == 0:
        return np.ones(n) / n, 0.0

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.zeros((K + n, n + 1))
    b_ub = np.zeros(K + n)

    for row, j in enumerate(active_indices):
        i_start = max(0, j - n + 1)
        i_end = min(n - 1, j)
        for i in range(i_start, i_end + 1):
            A_ub[row, i] = a[j - i]
        A_ub[row, n] = -1.0

    for i in range(n):
        A_ub[K + i, i] = -1.0
        b_ub[K + i] = a[i]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     method='highs')

    if result.success:
        return result.x[:n], float(result.x[n])
    return None, np.inf

def _lp_polish(a_init, deadline, max_iter=500, active_tol=1e-6):
    """
    LP-based minimax active set polishing.
    Identifies active set, solves LP for minimax direction, takes a step.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    print(f"    LP polish start: score={best_sc:.10f}, n={n}")

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 50 == 0:
                print(f"    LP iter {it}: score={sc:.10f}")

        active_mask = c >= c_max * (1.0 - active_tol)
        active_indices = np.where(active_mask)[0]

        if len(active_indices) > 200:
            top_k = np.argsort(c)[::-1][:200]
            active_indices = top_k

        d, t_opt = _lp_minimax_direction(a, active_indices)

        if d is None:
            # Try smaller active set
            if len(active_indices) > 10:
                top_10 = np.argsort(c)[::-1][:10]
                d, t_opt = _lp_minimax_direction(a, top_10)
            if d is None:
                break

        if t_opt > 1e-12:
            # Can't decrease max while increasing sum - near optimal
            # Try with just top 5
            top_5 = np.argsort(c)[::-1][:5]
            d5, t5 = _lp_minimax_direction(a, top_5)
            if d5 is not None and t5 < t_opt:
                d, t_opt = d5, t5
            else:
                break

        # Line search
        t_max = 1.0
        neg_mask = d < 0
        if np.any(neg_mask):
            ratios = a[neg_mask] / (-d[neg_mask])
            t_max = min(t_max, float(ratios.min()))
            t_max = max(0.001, t_max)

        best_t = 0.0
        best_t_sc = sc
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            t = min(frac, t_max)
            a_try = a + t * d
            if np.any(a_try < -1e-12):
                continue
            a_try = np.maximum(a_try, 0.0)
            sc_try = _score(a_try)
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t = t

        if best_t == 0.0:
            best_t = 0.01 * t_max
            a_try = np.maximum(a + best_t * d, 0.0)
            if _score(a_try) >= sc:
                break

        a = np.maximum(a + best_t * d, 0.0)

        # Adapt active tolerance
        if it % 100 == 0 and active_tol > 1e-8:
            active_tol *= 0.5

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    print(f"    LP polish done: best_score={best_sc:.10f}")
    return best_a, best_sc

def _subgrad_polish(a_init, deadline, max_steps=3000, lr_init=0.03):
    """Subgradient polishing in z-space with exact active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = max(1e-6, 0.001 * c_max)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if i % 100 == 0:
            sc_check = _score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()
            else:
                lr *= 0.95

        z = z - lr * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        lr *= 0.9999

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "oscillatory":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n

def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid L-BFGS + LP minimax active set optimizer.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        print(f"Parent: n={n_main}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}")
        update_best(a0, _score(a0), "init")

    # Phase 1: L-BFGS
    print(f"\nPhase 1: L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _lbfgs_phase(a0, deadline, min_time_remaining=600.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Subgradient polish
    print(f"\nPhase 2: Subgradient (t={time.monotonic() - t_start:.1f}s)")
    sg_deadline = min(deadline - 400.0, time.monotonic() + 120.0)
    a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, max_steps=2000, lr_init=0.02)
    update_best(a_sg, sc_sg, "subgrad")

    # Phase 3: LP minimax polish
    print(f"\nPhase 3: LP minimax polish (t={time.monotonic() - t_start:.1f}s)")
    lp_deadline = min(deadline - 200.0, time.monotonic() + 300.0)
    a_lp, sc_lp = _lp_polish(best_seq, lp_deadline, max_iter=400, active_tol=5e-7)
    update_best(a_lp, sc_lp, "lp_polish")

    # Phase 4: Fine L-BFGS + LP cycle
    print(f"\nPhase 4: Fine cycle (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        # Very fine L-BFGS
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        n = len(best_seq)
        fine_betas = mb * np.array([0.0001, 0.00005, 0.00002, 0.00001])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 80:
                break
            z0 = np.log(np.maximum(best_seq, 1e-15))
            z0 = z0 - z0.mean()
            try:
                result = minimize(
                    _smoobj, z0, args=(beta, n),
                    jac=lambda z: _smograd(z, beta, n),
                    method='L-BFGS-B',
                    options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-13,
                             'maxcor': 40, 'maxls': 100, 'disp': False}
                )
                a_f = _softmax(result.x)
                sc_f = _score(a_f)
                update_best(a_f, sc_f, f"fine_{i}")
            except Exception:
                pass

        # More LP polish with tighter tolerance
        if time.monotonic() < deadline - 50:
            a_lp2, sc_lp2 = _lp_polish(best_seq, deadline - 30,
                                       max_iter=200, active_tol=1e-8)
            update_best(a_lp2, sc_lp2, "lp_tight")

    # Phase 5: Try n=4096 if time permits
    print(f"\nPhase 5: Try n=4096 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        try:
            n4096 = 4096
            # Upsample parent to 4096
            if parent is not None:
                from scipy.interpolate import interp1d
                f_interp = interp1d(np.arange(len(parent)), parent,
                                    kind='cubic', fill_value='extrapolate')
                a_4096 = f_interp(np.linspace(0, len(parent) - 1, n4096))
                a_4096 = np.maximum(a_4096, 0)
                a_4096 = a_4096 / a_4096.sum()
            else:
                a_4096 = _make_profile(n4096, "uniform")

            print(f"  n=4096 initial score: {_score(a_4096):.10f}")
            a_4k = _lbfgs_phase(a_4096, deadline, min_time_remaining=150.0)
            sc_4k = _score(a_4k)
            update_best(a_4k, sc_4k, "n4096_lbfgs")

            if sc_4k < best_score + 0.01:
                a_4k_lp, sc_4k_lp = _lp_polish(a_4k, deadline - 30,
                                                max_iter=200, active_tol=5e-7)
                update_best(a_4k_lp, sc_4k_lp, "n4096_lp")
        except Exception as e:
            print(f"  n=4096 failed: {e}")

    # Phase 6: Final polish
    print(f"\nPhase 6: Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin, sc_fin = _lp_polish(best_seq, deadline - 5,
                                   max_iter=100, active_tol=1e-9)
        update_best(a_fin, sc_fin, "final_lp")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
