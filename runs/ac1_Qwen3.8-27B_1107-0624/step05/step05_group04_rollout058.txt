```python
"""
Peak-targeted optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a combination of:
1. L-BFGS with log-sum-exp continuation
2. Projected subgradient with Nesterov acceleration
3. Coordinate descent in log-space
4. Adaptive peak equalization
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    """Returns 2*n*max_c (assuming sum(a)=1)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum

def _logsumexp_grad(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(len(a))
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    n = len(a)
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z

def _peak_grad(z):
    """Gradient focusing on the top few peaks (not smoothed)."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    # Use top 5 peaks with weights proportional to their values
    top_k = min(5, len(c))
    idx = np.argsort(c)[::-1][:top_k]
    w = c[idx].copy()
    w = w / w.sum()
    
    h = np.zeros(n)
    for ki in range(top_k):
        k = idx[ki]
        # h_i += w_ki * a_{k-i}
        valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
        h[valid_i] += w[ki] * a[k - valid_i]
    
    h *= 2.0
    offset = float(np.dot(a, h))
    grad_z = a * (h - offset)
    return grad_z

def _run_lbfsgs(z_init, deadline, n_betas=12, beta_start_scale=0.3, beta_end_scale=1e-5):
    """L-BFGS with continuation schedule."""
    n = len(z_init)
    a_init = _softmax_fn(z_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    z = z_init.copy()
    best_z = z.copy()
    best_score = _score_norm_fn(a_init)
    
    beta_start = beta_start_scale * max_c0
    beta_end = beta_end_scale * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + time_budget
        
        maxiter = 3000
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_z = z_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_z, best_score

def _nesterov_subgrad(z_init, deadline, lr_init=0.05, top_k=5):
    """Nesterov accelerated subgradient method on z-space."""
    n = len(z_init)
    z = z_init.copy()
    y = z.copy()  # Nesterov momentum point
    best_z = z.copy()
    best_score = _score_norm_fn(_softmax_fn(z))
    
    t = 0
    gamma = 2.0  # Nesterov parameter
    
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        t = it + 1
        t_prev = it
        
        # Nesterov: y = z + (t_prev/(t+gamma)) * (z - z_prev)
        # Simplified: momentum
        if t > 1:
            momentum = (t_prev) / (t + gamma)
            z = z + momentum * (z - y)
        y = z.copy()
        
        # Compute gradient at y
        g = _peak_grad(y)
        
        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.01 * t)

def _coordinate_descent(z_init, deadline, max_rounds=1000):
    """Coordinate descent in z-space: optimize one z_i at a time."""
    n = len(z_init)
    z = z_init.copy()
    best_z = z.copy()
    best_score = _score_norm_fn(_softmax_fn(z))
    
    # Precompute: for efficiency, we'll evaluate the full score after each coordinate update
    # but use a fast check to decide direction
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try small perturbations of z_i
            for delta in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                z_trial = z.copy()
                z_trial[i] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_z = z_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
        
        if not improved:
            # Try larger steps
            for i in range(0, n, max(1, n//200)):
                if time.monotonic() > deadline:
                    break
                for delta in [-0.05, -0.1, 0.05, 0.1, -0.2, 0.2]:
                    z_trial = z.copy()
                    z_trial[i] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_z = z_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
        
        if not improved and rnd > 50:
            # Add noise
            z = z + 0.01 * np.random.default_rng(rnd).normal(size=n)
            z = z - z.mean()
    
    return best_z, best_score

def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    for s in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        profiles.append(_make_ramp(n, s))
    for rate in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    # Symmetric bump
    a = np.exp(-3.0 * (t - 0.5)**2)
    profiles.append(a / a.sum())
    return profiles

def _peak_equalize_v2(z, deadline, rng, max_rounds=500):
    """Improved peak equalization: target the top contributors to each peak."""
    best_z = z.copy()
    best_score = _score_norm_fn(_softmax_fn(z))
    z_cur = z.copy()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z_cur)
        n = len(a)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Find top 3 peaks
        top_k = min(3, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        improved = False
        for pk in peak_indices:
            if time.monotonic() > deadline:
                break
            # Find top contributors to this peak
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a[valid_i] * a[pk - valid_i]
            # The reduction in c_pk from reducing a_i by delta is approximately 2*delta*a[pk-i]
            # (for the dominant term)
            order = np.argsort(prod)[::-1][:10]
            
            for idx_pos in order[:6]:
                idx = valid_i[idx_pos]
                partner = pk - idx
                # Try reducing idx and/or partner
                for delta in [0.01, 0.02, 0.05, 0.1]:
                    z_trial = z_cur.copy()
                    z_trial[idx] -= delta
                    if 0 <= partner < n and partner != idx:
                        z_trial[partner] -= delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_z = z_trial.copy()
                        z_cur = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try redistributing mass from high to low
            sorted_idx = np.argsort(a)
            # Reduce top, increase bottom
            for ratio in [0.005, 0.01, 0.02]:
                z_trial = z_cur.copy()
                z_trial[sorted_idx[0]] += ratio  # increase smallest
                z_trial[sorted_idx[-1]] -= ratio  # decrease largest
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_z = z_trial.copy()
                    z_cur = z_trial.copy()
                    improved = True
                    break
            if improved:
                continue

        if rnd % 50 == 0 and rnd > 0:

            z_cur = z_cur + noise_level * rng.normal(size=n)
            z_cur = z_cur - z_cur.mean()
    
    return best_z, best_score

def _simulated_annealing(z_init, deadline, rng, T_init=0.1, T_min=0.001, cooling=0.9995):
    """Simulated annealing in z-space."""
    n = len(z_init)
    z = z_init.copy()
    best_z = z.copy()
    best_score = _score_norm_fn(_softmax_fn(z))
    T = T_init
    
    a = _softmax_fn(z)
    sc = best_score
    
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        if T < T_min:
            break
        
        # Random perturbation
        idx = rng.integers(0, n)
        delta = rng.normal(0, 0.05)
        z_trial = z.copy()
        z_trial[idx] += delta
        z_trial = z_trial - z_trial.mean()
        
        a_trial = _softmax_fn(z_trial)
        sc_trial = _score_norm_fn(a_trial)
        
        if sc_trial < sc or rng.random() < np.exp(-(sc_trial - sc) / max(T, 1e-10)):
            z = z_trial
            sc = sc_trial
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
        
        T *= cooling
    
    return best_z, best_score
```

```python
def _project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _subgradient_step(a, step_size, top_k=5):
    """One subgradient step with projection onto simplex."""
    n = len(a)
    c = _full_convolve(a, a)
    # Get top k peaks
    idx = np.argsort(c)[::-1][:top_k]
    w = c[idx].copy()
    w = w / w.sum()
    
    # Subgradient: sum_k w_k * 2 * C_k * a
    g = np.zeros(n)
    for ki in range(top_k):
        k = idx[ki]
        # (C_k * a)_i = a_{k-i} if 0 <= k-i < n
        valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
        g[valid_i] += 2.0 * w[ki] * a[k - valid_i]
    
    # Projected gradient step
    a_new = _project_simplex(a - step_size * g)
    return a_new
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1)
    rho = np.nonzero(u * rng > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _subgrad_at(a, top_k=5):
    """Compute subgradient of max(conv(a,a)) w.r.t. a, focusing on top_k peaks."""
    n = len(a)
    c = _full_convolve(a, a)
    idx = np.argsort(c)[::-1][:top_k]
    w = c[idx].copy()
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    g = np.zeros(n)
    for ki in range(top_k):
        k = idx[ki]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        valid_i = np.arange(lo, hi)
        g[valid_i] += 2.0 * w[ki] * a[k - valid_i]
    return g, c, idx, w


def _nesterov_simplex(a_init, deadline, lr_init=0.5, top_k=5, momentum_init=0.5):
    """Nesterov accelerated projected subgradient on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    t = 0
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        t = it + 1
        
        # Nesterov momentum
        mu = (t - 1) / (t + 1.0) if t > 1 else 0.0
        y_new = (1.0 - mu) * a + mu * (a - (a - y))
        y = y_new
        
        # Subgradient at y
        g, c, idx, w = _subgrad_at(y, top_k)
        
        # Backtracking line search
        lr = lr_init / (1.0 + 0.005 * t)
        a_trial = _project_simplex(y - lr * g)
        sc_trial = _score_norm_fn(a_trial)
        
        retry = 0
        while sc_trial > _score_norm_fn(y) and retry < 5 and lr > 1e-6:
            lr *= 0.5
            a_trial = _project_simplex(y - lr * g)
            sc_trial = _score_norm_fn(a_trial)
            retry += 1
        
        a = a_trial
        
        if np.isfinite(sc_trial) and sc_trial < best_score:
            best_score = sc_trial
            best_a = a.copy()
    
    return best_a, best_score


def _peak_pairing(a, deadline, rng, max_rounds=5000):
    """
    Peak pairing: identify pairs (i, j) where i+j is a peak index.
    Reduce such pairs to bring down multiple peaks simultaneously.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    a_cur = a.copy()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        
        # Top 4 peaks
        top_k = min(4, len(c))
        peak_idx = np.argsort(c)[::-1][:top_k]
        
        improved = False
        for pk in peak_idx:
            if time.monotonic() > deadline:
                break
            lo = max(0, pk - n + 1)
            hi = min(n, pk + 1)
            valid_i = np.arange(lo, hi)
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            order = np.argsort(prod)[::-1][:8]
            
            for op in order[:5]:
                i = valid_i[op]
                j = pk - i
                # Try reducing i and j together
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    delta_a = frac * min(a_cur[i], a_cur[j])
                    if delta_a < 1e-15:
                        continue
                    a_trial = a_cur.copy()
                    a_trial[i] -= delta_a
                    if 0 <= j < n and j != i:
                        a_trial[j] -= delta_a
                    # Redistribute to the smallest elements
                    total_removed = delta_a * (1 + (1 if 0 <= j < n and j != i else 0))
                    if total_removed > 0:
                        sorted_idx = np.argsort(a_trial)
                        # Distribute to bottom 20%
                        n_bot = max(1, n // 5)
                        bottom = sorted_idx[:n_bot]
                        bot_sum = a_trial[bottom].sum()
                        if bot_sum > 1e-15:
                            for bi in bottom:
                                a_trial[bi] += total_removed * a_trial[bi] / bot_sum
                        else:
                            a_trial[bottom[0]] += total_removed
                    a_trial = a_trial / a_trial.sum()
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        a_cur = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try frequency-domain flattening
            if rnd % 20 == 0:
                A = np.fft.rfft(a_cur)
                mag = np.abs(A)
                # Try to flatten the magnitude spectrum (reduces time-domain peaks)
                target_mag = np.median(mag[1:]) * np.ones_like(mag)
                if target_mag[0] > 0:
                    target_mag[0] = mag[0]  # Keep DC
                phase = np.angle(A)
                A_new = target_mag * np.exp(1j * phase)
                a_spec = np.fft.irfft(A_new, n * 2 - 1)[:n]
                a_spec = np.maximum(a_spec, 0)
                if a_spec.sum() > 1e-10:
                    a_spec = a_spec / a_spec.sum()
                    sc_spec = _score_norm_fn(a_spec)
                    if sc_spec < best_score - 1e-14:
                        best_score = sc_spec
                        best_a = a_spec.copy()
                        a_cur = a_spec.copy()
                        improved = True
            
            if not improved and rnd > 100:
                # Add small noise
                noise = 0.005 * rng.normal(size=n)
                a_trial = a_cur + noise
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                a_cur = a_trial
    
    return best_a, best_score


def _run_lbfgs_continuation(z_init, deadline, n_betas=12, beta_start_scale=0.3, beta_end_scale=1e-5):

        return best_z, best_score
    
    beta_start = beta_start_scale * max_c0
    beta_end = max(beta_end_scale * max_c0, 1e-10)
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        time_budget = max(time_budget, 3.0)
        level_deadline = time.monotonic() + time_budget
        
        try:

                _logsumexp_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_logsumexp_grad,

def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    for s in [0.3, 0.7, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0]:
        profiles.append(_make_ramp(n, s))

    profiles.append(a / a.sum())
    # Bump
    a = np.exp(-2.0 * (t - 0.3)**2 / 0.1)

def _frequency_flatten(a, deadline, rng, max_iters=5000):
    """
    Iterate: go to frequency domain, flatten |A|, come back, project to positive.
    This is similar to the "Gerchberg-Saxton" algorithm but for autocorrelation flattening.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    a_cur = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Go to frequency domain
        A = np.fft.rfft(a_cur, 2 * n - 1)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # We want the IFFT of |A|^2 to be flat (i.e., the autocorrelation is flat)
        # |A|^2 should have a flat IFFT, meaning |A|^2 should be concentrated at DC
        # Actually: conv(a,a) = IFFT(|A|^2). We want IFFT(|A|^2) to be flat.
        # IFFT(|A|^2) flat means |A|^2 has all its mass at DC → |A| is a delta at DC → a is constant.
        # That's not quite right. Let me reconsider.
        
        # The constraint is: a >= 0, sum(a) = 1
        # In frequency: A[0] = sum(a) = 1, a = IFFT(A) >= 0
        # We want: IFFT(|A|^2) to be as flat as possible
        
        # Strategy: modify |A| to make IFFT(|A|^2) flatter
        # This is the "Sato" approach: minimize variance of IFFT(|A|^2)
        
        # Alternative: use the constraint that a >= 0 more directly
        # Project back to time domain and enforce non-negativity
        a_new = np.fft.irfft(A, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:

        a_new = a_new / a_new.sum()
        
        # Blend with current
        alpha = min(1.0, 0.5 / (1.0 + it * 0.01))
        a_blend = (1 - alpha) * a_cur + alpha * a_new
        a_blend = np.maximum(a_blend, 0)
        a_blend = a_blend / a_blend.sum()

            print(f"  n={target_n}: no improvement ({sc_opt:.10f} vs {sc_cur:.10f})")
    
    if sc_cur < best_score:
        update_best(a_cur, sc_cur, "multiscale")
    
    # Also try a second multi-scale path with different initialization
    if time.monotonic() < phase1_end - 30:
        # Use a different profile (e.g., exponential)
        z128b = np.log(np.maximum(profiles[8] if len(profiles) > 8 else profiles[0], 1e-15))
        z128b = z128b - z128b.mean()
        # Quick optimize
        z128b, sc128b = _run_lbfgs_continuation(z128b, min(phase1_end - 20, time.monotonic() + 10.0), n_betas=4)
        a128b = _softmax_fn(z128b)
        
        if sc128b < _score_norm_fn(a_cur):
            print(f"  Alt path n=128: {sc128b:.10f} (better than main)")
            # Continue this path
            for target_n in [256, 512, 1024, 2048, 4096][:3]:
                if time.monotonic() > phase1_end - 5:
                    break
                a_t = _upsample(a128b, target_n)
                z_t = np.log(np.maximum(a_t, 1e-15))
                z_t = z_t - z_t.mean()
                z_t, sc_t = _run_lbfgs_continuation(z_t, min(phase1_end - 3, time.monotonic() + 15.0), n_betas=5)
                a128b = _softmax_fn(z_t)
            if _score_norm_fn(a128b) < best_score:
                update_best(a128b, _score_norm_fn(a128b), "alt_multiscale")
    
    # === Phase 2: Fine L-BFGS from best (20% of time) ===
    phase2_end = t_start + (budget_s - 30.0) * 0.50
    print(f"\n=== Phase 2: Fine L-BFGS (until {phase2_end - t_start:.0f}s) ===")
    
    if best_seq is not None and time.monotonic() < phase2_end - 10:
        n_best = len(best_seq)
        z_best = np.log(np.maximum(best_seq, 1e-15))
        z_best = z_best - z_best.mean()
        
        # Run multiple L-BFGS with different parameters
        attempts = [
            {'n_betas': 15, 'bs': 0.1, 'be': 1e-6, 'perturb': 0.0},
            {'n_betas': 12, 'bs': 0.05, 'be': 1e-7, 'perturb': 0.03},
            {'n_betas': 18, 'bs': 0.02, 'be': 1e-8, 'perturb': 0.01},
        ]
        
        for ai, att in enumerate(attempts):
            if time.monotonic() > phase2_end - 10:
                break
            z_try = z_best.copy()
            if att['perturb'] > 0:
                z_try = z_try + att['perturb'] * rng.normal(size=n_best)
                z_try = z_try - z_try.mean()
            
            alloc = min(90.0, max(10.0, (phase2_end - time.monotonic()) / (len(attempts) - ai)))
            opt_dl = min(phase2_end - 5, time.monotonic() + alloc)
            z_r, sc_r = _run_lbfgs_continuation(z_try, opt_dl, n_betas=att['n_betas'],
                                                 beta_start_scale=att['bs'], beta_end_scale=att['be'])
            a_r = _softmax_fn(z_r)
            update_best(a_r, sc_r, f"phase2_{ai}")
            if sc_r < _score_norm_fn(_softmax_fn(z_best)):
                z_best = z_r
    
    # === Phase 3: Nesterov subgradient + Peak pairing (30% of time) ===
    phase3_end = t_start + (budget_s - 30.0) * 0.80
    print(f"\n=== Phase 3: Subgradient + Peak pairing (until {phase3_end - t_start:.0f}s) ===")
    
    if best_seq is not None and time.monotonic() < phase3_end - 10:
        a_cur = best_seq.copy()
        
        # Nesterov subgradient
        ns_dl = min(phase3_end - 120, time.monotonic() + 180.0)
        a_ns, sc_ns = _nesterov_simplex(a_cur, ns_dl, lr_init=1.0, top_k=5)
        update_best(a_ns, sc_ns, "nesterov_subgrad")
        if sc_ns < _score_norm_fn(a_cur):
            a_cur = a_ns
        
        # Peak pairing
        pp_dl = min(phase3_end - 60, time.monotonic() + 180.0)
        a_pp, sc_pp = _peak_pairing(a_cur, pp_dl, rng)
        update_best(a_pp, sc_pp, "peak_pairing")
        if sc_pp < _score_norm_fn(a_cur):
            a_cur = a_pp
        
        # Another Nesterov round
        if time.monotonic() < phase3_end - 20:
            ns_dl2 = min(phase3_end - 5, time.monotonic() + 90.0)
            a_ns2, sc_ns2 = _nesterov_simplex(a_cur, ns_dl2, lr_init=0.5, top_k=3)
            update_best(a_ns2, sc_ns2, "nesterov2")
    
    # === Phase 4: Final polish (20% of time) ===
    phase4_end = deadline
    print(f"\n=== Phase 4: Final polish (until {phase4_end - t_start:.0f}s) ===")
    
    if best_seq is not None and time.monotonic() < phase4_end - 10:
        n_final = len(best_seq)
        
        # Fine L-BFGS with very small beta
        z_fin = np.log(np.maximum(best_seq, 1e-15))
        z_fin = z_fin - z_fin.mean()
        if time.monotonic() < phase4_end - 30:
            z_f, sc_f = _run_lbfgs_continuation(z_fin, min(phase4_end - 15, time.monotonic() + 120.0),
                                                  n_betas=20, beta_start_scale=0.01, beta_end_scale=1e-8)
            a_f = _softmax_fn(z_f)
            update_best(a_f, sc_f, "final_lbfgs")
        
        # Peak pairing fine
        if time.monotonic() < phase4_end - 15:
            a_cur = best_seq.copy()
            pp_dl2 = min(phase4_end - 5, time.monotonic() + 60.0)
            a_pp2, sc_pp2 = _peak_pairing(a_cur, pp_dl2, rng, max_rounds=3000)
            update_best(a_pp2, sc_pp2, "peak_pairing_final")
        
        # Quick Nesterov
        if time.monotonic() < phase4_end - 5:
            a_cur = best_seq.copy()
            ns_dl3 = min(phase4_end - 2, time.monotonic() + 30.0)
            a_ns3, sc_ns3 = _nesterov_simplex(a_cur, ns_dl3, lr_init=0.3, top_k=3)
            update_best(a_ns3, sc_ns3, "nesterov_final")
    
    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.5)
        
        # Verify with the actual evaluator
        sc_final = evaluate_sequence([float(x) for x in final_seq])
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        # If for some reason the score is bad, try a fallback
        if not np.isfinite(sc_final) or sc_final > 2.0:
            print("WARNING: Final score is bad, trying fallback optimizations")
            # Try the parent again
            if parent is not None:
                sc_parent = evaluate_sequence([float(x) for x in parent / parent.sum()])
                if np.isfinite(sc_parent) and sc_parent < sc_final:
                    final_seq = parent / parent.sum()
                    sc_final = sc_parent
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_ramp(256, 1.5)
    sc_fb = evaluate_sequence([float(x) for x in fallback])
    print(f"FALLBACK: n=256, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
```
