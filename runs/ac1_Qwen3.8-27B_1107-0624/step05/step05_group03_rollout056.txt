def _next_fft_len(n):
    """Next fast FFT length >= n."""
    if n <= 1:
        return max(1, n)
    # Use powers of 2 for simplicity
    k = 1
    while k < n:
        k <<= 1
    return k


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_fft_len(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    """Stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    """Score assuming sum(a)=1: returns 2n * max_conv."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """General score: 2n * max_conv / sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _build_subset_constraint_matrix(n, f, k_list):
    """Build sparse matrix for subset of convolution constraints.
    For each k in k_list: sum_{i valid} f[k-i] * g[i] <= M
    Returns (A, shape) where A @ g gives the constraint LHS values.
    """
    rows = []
    cols = []
    vals = []
    for k in k_list:
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(k_list), n))
    # Remap row indices to 0..len(k_list)-1
    # Actually we need to fix the row indexing
    rows2 = []
    cols2 = []
    vals2 = []
    for idx, k in enumerate(k_list):
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows2.append(idx)
            cols2.append(j)
            vals2.append(f[k - j])
    A = csr_matrix((vals2, (rows2, cols2)), shape=(len(k_list), n))
    return A


def _lp_direction(f, M, k_list, timeout=30.0):
    """Solve LP: max sum(g) s.t. conv constraints on k_list, g >= 0.
    Returns g or None."""
    n = len(f)
    try:
        A = _build_subset_constraint_matrix(n, f, k_list)
        g = cp.Variable(n, non_neg=True)
        constraints = [A @ g <= M]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)
        
        t0 = time.monotonic()
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
            try:
                prob.solve(solver=solver, threads=1)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    if g.value is not None:
                        return np.maximum(g.value, 0.0)
            except Exception:
                continue
            if time.monotonic() - t0 > timeout:
                break
    except Exception:
        pass
    return None


def _fw_step_with_line_search(f, M, g_dir, c_max, n):

        f_new = np.maximum(f_new, 0.0)

        if s < 1e-15:

        f_new = f_new / s
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_t = t
    
    f_new = (1.0 - best_t) * f + best_t * g_dir
    f_new = np.maximum(f_new, 0.0)
    f_new = f_new / f_new.sum()
    return f_new, best_score, best_t


def _fw_lp_method(f_init, deadline, rng, max_iters=50):
    """LP-based conditional gradient with line search and constraint subset."""
    n = len(f_init)
    f = f_init.copy() / f_init.sum()
    
    c = _fft_convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = float(_score_norm_fn(f))
    
    print(f"  FW-LP start: score={best_score:.10f}")
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        # Get top K lags for constraints
        K = min(500, 2 * n - 1)
        k_list = np.argsort(c)[-K:].tolist()
        
        # Solve LP for direction
        g_dir = _lp_direction(f, M, k_list, timeout=45.0)
        if g_dir is None:
            # Fallback: use uniform direction
            g_dir = np.ones(n) / n
            print(f"  FW iter {it}: LP failed, using uniform direction")
        else:
            s_g = g_dir.sum()
            if s_g < 1e-15:
                g_dir = np.ones(n) / n
            else:
                g_dir = g_dir / s_g
        
        # Line search
        f_new, sc_new, t = _fw_step_with_line_search(f, M, g_dir, c.max(), n)
        
        if sc_new < best_score:
            best_score = sc_new
            best_f = f_new.copy()
            f = f_new.copy()
            c = _fft_convolve(f, f)
            M = float(c.max())
            print(f"  FW iter {it}: score={sc_new:.10f}, t={t:.3f}")
        else:
            # Stagnation: try smaller step
            f_new2 = (1.0 - 0.05) * f + 0.05 * g_dir
            f_new2 = np.maximum(f_new2, 0.0) / np.maximum(f_new2.sum(), 1e-15)
            sc2 = _score_norm_fn(f_new2)
            if sc2 < best_score:
                best_score = sc2
                best_f = f_new2.copy()
                f = f_new2.copy()
                c = _fft_convolve(f, f)
                M = float(c.max())
                print(f"  FW iter {it}: small step, score={sc2:.10f}")
            else:
                # Try random perturbation
                f_pert = f + 0.01 * rng.standard_normal(n)
                f_pert = np.maximum(f_pert, 0.0) / np.maximum(f_pert.sum(), 1e-15)
                sc_pert = _score_norm_fn(f_pert)
                if sc_pert < best_score:
                    best_score = sc_pert
                    best_f = f_pert.copy()
                    f = f_pert.copy()
                    c = _fft_convolve(f, f)
                    M = float(c.max())
                    print(f"  FW iter {it}: perturbed, score={sc_pert:.10f}")
                else:
                    print(f"  FW iter {it}: stagnation, best={best_score:.10f}")
    
    return best_f, best_score


def _lbfgs_continuation(a_init, deadline, rng, beta_scale=1.0, min_beta_frac=0.0001, max_wall=None):
    """L-BFGS with logsumexp continuation and adaptive beta."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    c0 = _fft_convolve(a, a)
    max_c0 = float(c0.max())
    
    # Adaptive beta schedule
    beta_starts = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
                   0.0002, 0.0001, min_beta_frac]
    betas = [b * max_c0 * beta_scale for b in beta_starts]
    
    if max_wall is not None:
        per_level = max(2.0, max_wall / len(betas))
    else:
        per_level = 20.0
    
    for level, beta in enumerate(betas):
        if time.monotonic() > deadline - 5.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        
        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _fft_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)
        
        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _fft_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            # Gradient of smooth objective w.r.t. a
            a_rev = a_[::-1]
            w_conv = _fft_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g_a = 4.0 * n * beta * a_ * (h - W)
            # Convert to z-gradient via softmax
            g_z = a_ * (g_a - float(np.dot(a_, g_a)))
            if not np.all(np.isfinite(g_z)):
                return np.zeros_like(z_)
            return g_z
        
        try:
            result = minimize(
                obj, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxls': 40, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
                    elapsed = time.monotonic()
                    print(f"  L-BFGS beta={beta:.6f}: score={s_new:.10f}")
        except Exception as e:
            print(f"  L-BFGS beta={beta:.6f} failed: {e}")
    
    return best_a, best_score


def _spectral_flatten(a_init, deadline, rng, n_iters=200):
    """Frequency-domain equalization to flatten the PSD."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    N = _next_fft_len(2 * n)
    
    for it in range(n_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress dynamic range of magnitude
        # Try different compression exponents
        alpha = 0.3 + 0.3 * (it % 5) / 4.0  # Cycle through 0.3, 0.375, 0.45, 0.525, 0.6
        
        mag_compressed = mag ** alpha
        # Add a small floor to avoid zeroing out
        floor = 0.01 * mag_compressed.mean()
        mag_compressed = np.maximum(mag_compressed, floor)
        
        A_new = mag_compressed * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_score


def _peak_equalize(a_init, deadline, rng, n_trials=50000):
    """Local search that targets equalizing the top peaks."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 3.0:
            break
        
        c = _fft_convolve(a, a)
        
        # Find top K lags
        K = min(8, 2 * n - 1)
        top_lags = np.argsort(c)[-K:]
        max_lag = int(top_lags[-1])
        max_val = c[max_lag]
        
        strategy = it % 4
        
        if strategy == 0:
            # Reduce the dominant product at the max lag
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min > i_max:
                continue
            pairs = np.arange(i_min, i_max + 1)
            partners = max_lag - pairs
            products = a[pairs] * a[partners]
            idx = int(pairs[np.argmax(products)])
            
            a_trial = a.copy()
            frac = rng.uniform(0.1, 0.4)
            delta = a_trial[idx] * frac
            a_trial[idx] -= delta
            a_trial[idx] = max(0.0, a_trial[idx])
            
            # Find safe position (low contribution to top lags)
            contrib = np.zeros(n)
            for kl in top_lags:
                kl = int(kl)
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                for j in range(lo, hi + 1):
                    partner = kl - j
                    if 0 <= partner < n:
                        contrib[j] += a[partner]
            
            safe_pool = np.argsort(contrib)[:max(10, n // 4)]
            add_idx = int(safe_pool[rng.integers(len(safe_pool))])
            a_trial[add_idx] += delta
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
        
        elif strategy == 1:
            # Block multiplicative perturbation
            block_size = int(rng.integers(3, min(32, n // 4)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.7, 1.3, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
        
        elif strategy == 2:
            # Pair mass exchange
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.1, 0.5)
            amt1 = a_trial[i1] * frac
            amt2 = a_trial[i2] * frac
            a_trial[i1] = max(0.0, a_trial[i1] - amt1 + amt2)
            a_trial[i2] = max(0.0, a_trial[i2] - amt2 + amt1)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
        
        else:
            # Gaussian bump perturbation
            center = int(rng.integers(n))
            width = float(rng.uniform(0.01, 0.1))
            delta = float(rng.uniform(0.01, 0.08))
            a_trial = a.copy()
            positions = np.arange(n, dtype=np.float64)
            bump = delta * np.exp(-(positions - center) ** 2 / (2 * width * n ** 2))
            a_trial += bump
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
    
    return best_a, best_sc


def _adam_topk(a_init, deadline, rng, lr=0.015, k_top=3, max_iters=100000):
    """Adam optimizer with top-K peaked gradient."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break
        
        t += 1
        a_curr = _softmax_fn(z)
        c = _fft_convolve(a_curr, a_curr)
        
        # Top-K gradient
        K = min(k_top, len(c))
        top_idx = np.argsort(c)[-K:]
        grad_a = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                grad_a[lo:hi + 1] += 2.0 * a_curr[k_val - hi:k_val - lo + 1][::-1]
        grad_a /= K
        grad_a *= 4.0 * n
        
        # Softmax gradient
        grad_z = a_curr * (grad_a - float(np.dot(a_curr, grad_a)))
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        lr_t = lr * (1.0 - 0.5 * t / max_iters)
        z = z - max(0.001, lr_t) * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        if t % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    
    return best_a, best_score


def _try_different_n(a_best, deadline, rng):
    """Try interpolating to different n values and optimizing."""
    n_curr = len(a_best)
    best_a = a_best.copy()
    best_score = _score_norm_fn(a_best)
    
    target_ns = sorted(set([n_curr // 2, n_curr, n_curr * 2, 4096, 8192, 16384]))
    target_ns = [x for x in target_ns if 1024 <= x <= 32768]
    
    for tn in target_ns:
        if time.monotonic() > deadline - 60.0:
            break
        if tn == n_curr:
            continue
        
        p_interp = _interp_profile(a_best, tn)
        sc_interp = _score_norm_fn(p_interp)
        if sc_interp < best_score:
            best_score = sc_interp
            best_a = p_interp.copy()
            print(f"  Multi-n: n={tn}, score={sc_interp:.10f}")
        
        # Brief L-BFGS polish at this n
        if time.monotonic() < deadline - 30.0:
            dl = min(deadline - 20.0, time.monotonic() + 40.0)
            a_pol, sc_pol = _lbfgs_continuation(p_interp, dl, rng,
                                                  beta_scale=0.5, max_wall=35.0)
            if sc_pol < best_score:
                best_score = sc_pol
                best_a = a_pol.copy()
                print(f"  Multi-n: n={tn}, polished score={sc_pol:.10f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 0: Diverse initialization
    Phase 1: LP-based conditional gradient with line search
    Phase 2: L-BFGS logsumexp continuation (adaptive)
    Phase 3: Spectral flattening + peak equalization
    Phase 4: Multi-n exploration
    Phase 5: Final intensive polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST [{label}]: {best_score:.10f} (n={len(p)}, t={elapsed:.0f}s)")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        _update(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_triangular_weight(n_parent)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using triangular n={n_parent}, score={sc_parent:.10f}")
        _update(parent, sc_parent, "init")
    
    n_work = n_parent
    
    # --- Phase 0: Diverse initialization (20s) ---
    phase0_end = min(deadline, t_start + 20.0)
    n_try = n_work
    
    profiles = [
        _make_linear_profile(n_try, 0.5),
        _make_linear_profile(n_try, 1.0),
        _make_linear_profile(n_try, 1.5),
        _make_linear_profile(n_try, 2.0),
        _make_quadratic_profile(n_try, 0.5, 0.3),
        _make_quadratic_profile(n_try, 1.0, -0.2),
        _make_quadratic_profile(n_try, 0.8, 0.4),
        _make_quadratic_profile(n_try, -0.5, 0.5),
        _make_bimodal_profile(n_try, 0.3, 0.12),
        _make_bimodal_profile(n_try, 0.4, 0.15),
        _make_bimodal_profile(n_try, 0.5, 0.2),
        _make_bimodal_profile(n_try, 0.6, 0.1),
        _make_bimodal_profile(n_try, 0.7, 0.08),
        _make_w_profile(n_try, 2, 0.3),
        _make_w_profile(n_try, 2, 0.5),
        _make_w_profile(n_try, 3, 0.3),
        _make_w_profile(n_try, 4, 0.2),
        _make_w_profile(n_try, 5, 0.15),
        _make_power_profile(n_try, 1.0),
        _make_power_profile(n_try, 1.5),
        _make_power_profile(n_try, 2.0),
        _make_power_profile(n_try, 3.0),
        _make_power_profile(n_try, 5.0),
        _make_power_profile(n_try, 8.0),
        _make_triangular_weight(n_try),
        _make_bspline_profile(n_try, 3),
        _make_bspline_profile(n_try, 4),
        _make_bspline_profile(n_try, 5),
        _make_bspline_profile(n_try, 6),
        _make_raised_cosine(n_try, 1.0),
        _make_raised_cosine(n_try, 2.0),
        _make_hann_window(n_try),
        _make_blackman_window(n_try),
        _make_flattop_window(n_try),
    ]
    
    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy()))
        _update(p, sc, f"prof{idx}")
    
    candidates.sort(key=lambda x: x[0])
    if candidates:
        print(f"Phase 0 best: {candidates[0][0]:.10f}, overall best: {best_score:.10f}")
    
    # Determine best starting point for main optimization
    start_point = candidates[0][1].copy() if candidates else parent.copy()
    if _score_norm_fn(start_point) > _score_norm_fn(parent):
        start_point = parent.copy()
    
    # --- Phase 1: LP-based conditional gradient (350s) ---
    phase1_end = min(deadline, t_start + 370.0)
    if time.monotonic() < phase1_end - 30:
        print(f"Phase 1: FW-LP, start score={_score_norm_fn(start_point):.10f}")
        a_fw, sc_fw = _fw_lp_method(start_point, phase1_end, rng, max_iters=40)
        _update(a_fw, sc_fw, "fw_lp")
    
    # --- Phase 2: L-BFGS continuation (400s) ---
    phase2_end = min(deadline, t_start + 770.0)
    if best_seq is not None and time.monotonic() < phase2_end - 30:
        print(f"Phase 2: L-BFGS continuation, current best={best_score:.10f}")
        
        dl = phase2_end - 10.0
        max_wall = max(20.0, dl - time.monotonic() - 10.0)
        a_best, sc_best = _lbfgs_continuation(best_seq, dl, rng,
                                                beta_scale=1.0, min_beta_frac=0.00005,
                                                max_wall=max_wall)
        _update(a_best, sc_best, "lbfgs_main")
        
        # Second pass with more aggressive schedule
        if time.monotonic() < phase2_end - 30:
            dl2 = phase2_end - 10.0
            max_wall2 = max(10.0, dl2 - time.monotonic() - 5.0)
            a_best2, sc_best2 = _lbfgs_continuation(best_seq, dl2, rng,
                                                      beta_scale=0.4, min_beta_frac=0.00002,
                                                      max_wall=max_wall2)
            _update(a_best2, sc_best2, "lbfgs_aggr")
    
    # --- Phase 3: Spectral flattening + Peak equalization (300s) ---
    phase3_end = min(deadline, t_start + 1070.0)
    
    if best_seq is not None and time.monotonic() < phase3_end - 20:
        print(f"Phase 3: Spectral + peak equalization, current best={best_score:.10f}")
        
        # 3a: Spectral flattening
        if time.monotonic() < phase3_end - 100:
            dl_spec = min(phase3_end - 100, time.monotonic() + 80.0)
            a_spec, sc_spec = _spectral_flatten(best_seq, dl_spec, rng, n_iters=300)
            _update(a_spec, sc_spec, "spectral")
        
        # 3b: Peak equalization
        if time.monotonic() < phase3_end - 20:
            dl_peak = phase3_end - 10.0
            a_peak, sc_peak = _peak_equalize(best_seq, dl_peak, rng, n_trials=50000)
            _update(a_peak, sc_peak, "peak_eq")
    
    # --- Phase 4: Multi-n exploration (200s) ---
    phase4_end = min(deadline, t_start + 1270.0)
    if best_seq is not None and time.monotonic() < phase4_end - 30:
        print(f"Phase 4: Multi-n exploration")
        a_mn, sc_mn = _try_different_n(best_seq, phase4_end, rng)
        _update(a_mn, sc_mn, "multi_n")
    
    # --- Phase 5: Final intensive polishing (300s) ---
    phase5_end = min(deadline, t_start + 1570.0)
    if best_seq is not None and time.monotonic() < phase5_end - 30:
        remaining = max(30.0, phase5_end - time.monotonic())
        print(f"Phase 5: Final polishing, remaining={remaining:.0f}s")
        
        # 5a: Fine L-BFGS with very small beta
        if remaining > 80:
            dl = min(phase5_end - 30, time.monotonic() + remaining * 0.35)
            a_fine, sc_fine = _lbfgs_continuation(best_seq, dl, rng,
                                                    beta_scale=0.2, min_beta_frac=0.00001,
                                                    max_wall=remaining * 0.35)
            _update(a_fine, sc_fine, "final_lbfgs")
        
        # 5b: Adam with tiny learning rate
        if time.monotonic() < phase5_end - 40:
            dl_adam = min(phase5_end - 30, time.monotonic() + remaining * 0.25)
            a_adam, sc_adam = _adam_topk(best_seq, dl_adam, rng, lr=0.005, k_top=2,
                                         max_iters=60000)
            _update(a_adam, sc_adam, "final_adam")
        
        # 5c: More peak equalization
        if time.monotonic() < phase5_end - 20:
            dl_pe = min(phase5_end - 15, time.monotonic() + remaining * 0.2)
            a_pe, sc_pe = _peak_equalize(best_seq, dl_pe, rng, n_trials=30000)
            _update(a_pe, sc_pe, "final_peak")
    
    # --- Phase 6: Multi-restart with L-BFGS (if time remains) ---
    phase6_end = min(deadline, t_start + 1700.0)
    if best_seq is not None and time.monotonic() < phase6_end - 30:
        n_restarts = 3
        per_restart = max(15.0, (phase6_end - time.monotonic() - 20.0) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 20:
                break
            noise = 0.01 + 0.01 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(phase6_end - 15, time.monotonic() + per_restart)
            a_r, sc_r = _lbfgs_continuation(p_start, dl, rng,
                                             beta_scale=0.5, max_wall=per_restart - 5.0)
            _update(a_r, sc_r, f"restart{i}")
    
    # --- Final validation ---
    print(f"\n=== FINAL RESULTS ===")
    print(f"Best score: {best_score:.10f}")
    if best_seq is not None:
        print(f"Best n: {len(best_seq)}")
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_fn(final_seq)
        print(f"Final validated score: {sc_final:.10f}")
        
        # Validate with evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"Evaluator score: {sc_eval:.10f}")
            if np.isfinite(sc_eval) and sc_eval < best_score:
                best_score = sc_eval
        except Exception as ex:
            print(f"Evaluator call failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_triangular_weight(8192)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
