import numpy as np
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bspline_profile(n, order):
    t = np.linspace(0.0, 1.0, n)
    u = t * 0.85 + 0.075
    u = np.clip(u, 0.001, 0.999)
    a = u ** (order - 1) * (1.0 - u) ** (order - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, separation, width):

    c1 = 0.5 - separation / 2.0
    c2 = 0.5 + separation / 2.0
    a = np.exp(-(t - c1)**2 / (2 * width**2)) + np.exp(-(t - c2)**2 / (2 * width**2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_profile(n, freq, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u**2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_symmetric(a):
    """Enforce symmetry: a[i] = a[n-1-i]"""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 1e-15)
    return a_sym / a_sym.sum()


def _compute_topk_gradient(a, k_top):
    """Compute gradient using top-K lags. Returns (gradient, top_lags, max_val)."""
    n = len(a)
    c = _full_convolve(a, a)
    k_actual = min(k_top, len(c))
    top_vals = np.partition(c, -k_actual)[-k_actual:]
    threshold = top_vals[0]
    top_idx = np.where(c >= threshold - 1e-12)[0]
    
    max_val = float(c.max())
    grad = np.zeros(n)
    for k_val in top_idx:
        k_val = int(k_val)
        lo = max(0, k_val - (n - 1))
        hi = min(n - 1, k_val)
        if lo <= hi:
            # grad[j] += 2*a[k-j] for j in [lo, hi]
            # a[k-j] for j in [lo, hi] = a[k-hi : k-lo+1] reversed
            segment = a[k_val - hi:k_val - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * segment
    grad /= len(top_idx)
    return grad, top_idx, max_val


def _fast_mw_method(a_init, deadline, rng, k_top=5, lr_init=None, n_iters=None):
    """
    Fast multiplicative weights method with adaptive step size.
    Each iteration: FFT conv -> top-K gradient -> MW update.
    """
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    if lr_init is None:
        lr_init = 0.5
    if n_iters is None:
        n_iters = 500000
    
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    lr = lr_init
    patience = 5000
    no_improve = 0
    
    for it in range(n_iters):
        if time.monotonic() > deadline - 2.0:
            break
        
        z_curr = np.log(np.maximum(a, 1e-15))
        z_curr = z_curr - z_curr.mean()
        
        # Compute gradient
        c = _full_convolve(a, a)
        k_actual = min(k_top, len(c))
        top_vals = np.partition(c, -k_actual)[-k_actual:]
        threshold = top_vals[0]
        top_idx = np.where(c >= threshold)[0]
        
        max_val = float(c.max())
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                segment = a[k_val - hi:k_val - lo + 1][::-1]
                grad[lo:hi + 1] += 2.0 * segment
        grad /= len(top_idx)
        
        # Normalize gradient
        g_norm = np.max(np.abs(grad))
        if g_norm < 1e-15:
            break
        
        # MW update: a[j] *= exp(-lr * grad[j] / g_norm)
        scale = lr / g_norm
        z_curr = z_curr - scale * grad
        z_curr = np.clip(z_curr, -50, 50)
        z_curr = z_curr - z_curr.mean()
        a = _softmax_fn(z_curr)
        
        # Check improvement periodically
        if (it + 1) % 100 == 0:
            sc = _score_norm_fn(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 100
            
            if no_improve >= patience:
                # Reduce learning rate
                lr *= 0.5
                no_improve = 0
                if lr < 1e-4:
                    break
        
    return best_a, best_score


def _hybrid_mw_lbfgs(a_init, deadline, rng, k_top=5, mw_chunk=50000, lbfgs_chunk=200.0):
    """
    Alternating fast MW iterations with periodic L-BFGS refinement.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    lr = 0.8
    no_improve = 0
    
    while time.monotonic() < deadline - 10.0:
        # MW phase
        mw_end = min(deadline - 30.0, time.monotonic() + 30.0)
        for it in range(mw_chunk):
            if time.monotonic() > mw_end:
                break
            
            a = _softmax_fn(z)
            c = _full_convolve(a, a)
            k_actual = min(k_top, len(c))
            top_vals = np.partition(c, -k_actual)[-k_actual:]
            threshold = top_vals[0]
            top_idx = np.where(c >= threshold)[0]
            
            grad = np.zeros(n)
            for k_val in top_idx:
                k_val = int(k_val)
                lo = max(0, k_val - (n - 1))
                hi = min(n - 1, k_val)
                if lo <= hi:
                    segment = a[k_val - hi:k_val - lo + 1][::-1]
                    grad[lo:hi + 1] += 2.0 * segment
            grad /= len(top_idx)
            
            g_norm = np.max(np.abs(grad))
            if g_norm < 1e-15:
                break
            
            scale = lr / g_norm
            z = z - scale * grad
            z = np.clip(z, -50, 50)
            z = z - z.mean()
        
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.7
            if lr < 0.01:
                lr = 0.5
        
        # L-BFGS refinement
        if time.monotonic() < deadline - 30.0:
            lbfgs_end = min(deadline - 10.0, time.monotonic() + 80.0)
            a_best, sc_best = _lbfgs_short(a_best, lbfgs_end, rng)
            if sc_best < best_score:
                best_score = sc_best
                best_a = a_best.copy()
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
    
    return best_a, best_score


def _lbfgs_obj_grad(z, beta, n):
    """Compute logsumexp-smoothed objective and gradient."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * cm, np.zeros_like(z)
    log_sum = float(np.log(s))
    obj = 2.0 * n * (cm + beta * log_sum)
    
    w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    g = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(z)
    return obj, g


def _lbfgs_short(a_init, deadline, rng, max_levels=5):
    """Short L-BFGS with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    betas = [0.1, 0.04, 0.015, 0.005, 0.002, 0.0008, 0.0003]
    betas = [b * max_c0 for b in betas[:max_levels]]
    
    per_level = max(10.0, (deadline - time.monotonic()) / len(betas))
    
    for beta in betas:
        if time.monotonic() > deadline - 3.0:
            break
        level_end = min(deadline - 2.0, time.monotonic() + per_level)
        
        def obj(z_):
            val, _ = _lbfgs_obj_grad(z_, beta, n)
            return val
        
        def grad_fn(z_):
            _, g = _lbfgs_obj_grad(z_, beta, n)
            return g
        
        try:
            iterations = 0
            while time.monotonic() < level_end:
                result = minimize(
                    obj, z, jac=grad_fn,
                    method='L-BFGS-B',
                    options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                             'maxls': 40, 'maxfun': 3000}
                )
                iterations += 1
                if result.x is not None and np.all(np.isfinite(result.x)):
                    z = result.x
                    a_new = _softmax_fn(z)
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                if result.status == 'convergence' or not np.all(np.isfinite(result.x)):
                    break
        except Exception:
            pass
    
    return best_a, best_score


def _active_set_polish(a_init, deadline, rng):
    """
    Active-set method: identify active lags, solve KKT conditions.
    At optimality with single active lag k*, for all j in support:
      2*a[k*-j] = lambda (plus boundary corrections)
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    max_rounds = 50
    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = _full_convolve(a, a)
        max_val = float(c.max())
        max_lag = int(np.argmax(c))
        
        # Find all lags within 1e-6 of max
        tol = max_val * 1e-8
        active_lags = np.where(c >= max_val - tol)[0]
        
        # Get support
        threshold = 1e-10
        support = np.where(a > threshold)[0]
        sup_set = set(support.tolist())
        
        # Build linear system for single active lag case
        # For j in support: sum_k alpha_k * 2*a[k-j] = lambda
        # Simplified: if only one active lag k*, then a[k*-j] = lambda/2 for j in support
        # This means a is determined by a[k*-j] for j in support
        # i.e., a[j] = a[k*-(k*-j)] ... this is circular
        
        # Alternative: use the condition that for the maximizing lag,
        # the gradient should be constant on the support.
        # grad[j] = 2*a[k*-j] for j in [k*-n+1, k*] (clipped to [0, n-1])
        # At optimum: grad[j] = const for all j in support
        
        lo = max(0, max_lag - (n - 1))
        hi = min(n - 1, max_lag)
        
        if lo < hi:
            # Gradient on the support should be constant
            grad_vals = 2.0 * a[max_lag - hi:max_lag - lo + 1][::-1]
            # For j in support ∩ [lo, hi], grad_vals[j-lo] should equal lambda
            sup_in_range = [j - lo for j in support if lo <= j <= hi]
            
            if len(sup_in_range) >= 2:
                target = np.mean(grad_vals[sup_in_range])
                # Adjust: make a[max_lag - j] = target/2 for j in support
                # This means a[i] = target/2 for i in [max_lag-hi, max_lag-lo] where max_lag-i in support
                # i.e., i in [max_lag-hi, max_lag-lo] and (max_lag-i) in support
                
                # Simpler: set a[i] proportional to satisfy the condition
                # For i in [lo, hi]: the constraint is a[max_lag-i] = target/2
                # Let u = max_lag - i, so u ranges over [max_lag-hi, max_lag-lo]
                # We want a[u] = target/2 for u in [max_lag-hi, max_lag-lo] ∩ support_shifted
                
                # This is getting complex. Let's use a simpler approach:
                # Iteratively adjust a to make the top lag's gradient flat on support
                alpha = 0.1
                for it in range(100):
                    c_curr = _full_convolve(a, a)
                    if c_curr.max() < max_val * (1 - 1e-10):
                        break
                    ml = int(np.argmax(c_curr))
                    lo2 = max(0, ml - (n - 1))
                    hi2 = min(n - 1, ml)
                    if lo2 < hi2:
                        gv = 2.0 * a[ml - hi2:ml - lo2 + 1][::-1]
                        mean_gv = np.mean(gv)
                        # nudge a to flatten the gradient
                        # a[ml-j] should all equal mean_gv/2
                        target_a = mean_gv / 2.0
                        idx_range = slice(ml - hi2, ml - lo2 + 1)
                        a[idx_range] = (1 - alpha) * a[idx_range] + alpha * target_a
                        a = np.maximum(a, 1e-15)
                        a = a / a.sum()
        
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            # converged or stuck
            if abs(sc - best_score) < 1e-12:
                break
    
    return best_a, best_score


def _symmetric_optimize(a_init, deadline, rng):
    """Optimize restricted to symmetric sequences."""
    n = len(a_init)
    a_sym = _make_symmetric(a_init)
    
    # Work with half the sequence
    half = (n + 1) // 2
    a_half = a_sym[:half].copy()
    best_a = a_sym.copy()
    best_score = _score_norm_fn(a_sym)
    
    # Fast MW on symmetric constraint
    z = np.log(np.maximum(a_half, 1e-15))
    z = z - z.mean()
    lr = 0.5
    
    def sym_expand(h):
        if n % 2 == 0:
            return np.concatenate([h[::-1], h])
        else:
            return np.concatenate([h[:0:-1], h])
    
    full_len = 2 * half - 1 if n % 2 == 1 else 2 * half
    # Actually let's just work with full symmetric vectors
    a = a_sym.copy()
    z_full = np.log(np.maximum(a, 1e-15))
    z_full = z_full - z_full.mean()
    
    for it in range(200000):
        if time.monotonic() > deadline - 3.0:
            break
        
        c = _full_convolve(a, a)
        k_top = 5
        k_actual = min(k_top, len(c))
        top_vals = np.partition(c, -k_actual)[-k_actual:]
        threshold = top_vals[0]
        top_idx = np.where(c >= threshold)[0]
        
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                segment = a[k_val - hi:k_val - lo + 1][::-1]
                grad[lo:hi + 1] += 2.0 * segment
        grad /= len(top_idx)
        
        # Symmetrize gradient
        grad = 0.5 * (grad + grad[::-1])
        
        g_norm = np.max(np.abs(grad))
        if g_norm < 1e-15:
            break
        
        scale = lr / g_norm
        z_full = z_full - scale * grad
        z_full = np.clip(z_full, -50, 50)
        z_full = z_full - z_full.mean()
        a_new = _softmax_fn(z_full)
        a = _make_symmetric(a_new)
        z_full = np.log(np.maximum(a, 1e-15))
        z_full = z_full - z_full.mean()
        
        if (it + 1) % 500 == 0:
            sc = _score_norm_fn(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            else:
                lr *= 0.9
                if lr < 0.01:
                    lr = 0.5
    
    # Short L-BFGS polish
    if time.monotonic() < deadline - 5.0:
        dl = min(deadline - 3.0, time.monotonic() + 60.0)
        a_best, sc_best = _lbfgs_short(best_a, dl, rng, max_levels=3)
        a_best = _make_symmetric(a_best)
        sc_best = _score_norm_fn(a_best)
        if sc_best < best_score:
            best_score = sc_best
            best_a = a_best.copy()
    
    return best_a, best_score


def _evolutionary_search(a_init, deadline, rng, pop_size=8):
    """Simple evolutionary strategy with tournament selection."""
    n = len(a_init)
    pop = []
    scores = []
    
    for i in range(pop_size):
        noise_scale = 0.05 * (1 + i * 0.5)
        z = np.log(np.maximum(a_init, 1e-15))
        z = z + noise_scale * rng.normal(size=n)
        z = np.clip(z, -40, 40)
        p = _softmax_fn(z)
        pop.append(p)
        scores.append(_score_norm_fn(p))
    
    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]
    
    # Also include the parent
    sc_parent = _score_norm_fn(a_init)
    if sc_parent < best_score:
        best_a = a_init.copy()
        best_score = sc_parent
        pop.append(a_init.copy())
        scores.append(sc_parent)
    
    generations = 0
    while time.monotonic() < deadline - 30.0:
        generations += 1
        
        # Selection: keep top half
        if len(pop) > pop_size:
            order = np.argsort(scores)
            keep = order[:pop_size]
            pop = [pop[i] for i in keep]
            scores = [scores[i] for i in keep]
        
        # Crossover and mutation
        new_pop = list(pop[max(0, len(pop)-2):])  # Keep best 2
        for _ in range(pop_size - len(new_pop)):
            # Pick parents by tournament
            i1 = int(rng.integers(len(pop)))
            i2 = int(rng.integers(len(pop)))
            if scores[i1] > scores[i2]:
                i1, i2 = i2, i1
            
            # Blend
            t = rng.uniform(0.3, 0.7)
            child = t * pop[i1] + (1 - t) * pop[i2]
            child = np.maximum(child, 1e-12)
            
            # Mutation
            noise = rng.uniform(-0.02, 0.02, size=n)
            child = child + noise * child
            child = np.maximum(child, 1e-12)
            child = child / child.sum()
            
            new_pop.append(child)
        
        pop = new_pop
        scores = [_score_norm_fn(p) for p in pop]
        
        # Short polish on best
        best_idx = int(np.argmin(scores))
        best_a = pop[best_idx].copy()
        best_score = scores[best_idx]
        
        if time.monotonic() < deadline - 60.0:
            dl = min(deadline - 30.0, time.monotonic() + 30.0)
            a_pol, sc_pol = _lbfgs_short(best_a, dl, rng, max_levels=2)
            if sc_pol < best_score:
                best_score = sc_pol
                best_a = a_pol.copy()
                pop[best_idx] = best_a
                scores[best_idx] = best_score
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: fast MW -> L-BFGS continuation -> active-set polish
    -> symmetric optimization -> evolutionary search -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            p = np.asarray(p, dtype=np.float64).copy()
            p = np.maximum(p, 0.0)
            s = p.sum()
            if s > 1e-10:
                p = p / s
            best_seq = p
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()

            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_work = len(parent)
        sc_parent = _score_norm_fn(parent)

    print(f"Working with n={n_work}")
    
    # --- Phase 0: Diverse initialization (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    profiles = [
        _make_triangular_weight(n_work),
        _make_power_profile(n_work, 1.5),

        _make_power_profile(n_work, 3.0),

            break
        sub_deadline = min(phase1_end - 30.0, time.monotonic() + 120.0)
        a_mk, sc_mk = _fast_mw_method(start_a, sub_deadline, rng, k_top=k_top, lr_init=0.6)
        update_best(a_mk, sc_mk, f"mw_k{k_top}")
        if a_mk is not None and sc_mk < _score_norm_fn(start_a):
            start_a = a_mk.copy()
    
    print(f"Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: Hybrid MW + L-BFGS (350s) ---
    phase2_end = min(deadline, t_start + 780.0)
    if time.monotonic() < phase2_end - 30.0:
        print(f"Phase 2: Hybrid MW+L-BFGS")
        a_hyb, sc_hyb = _hybrid_mw_lbfgs(best_seq.copy(), phase2_end, rng, k_top=5)
        update_best(a_hyb, sc_hyb, "hybrid")
    
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: L-BFGS continuation with aggressive schedule (300s) ---
    phase3_end = min(deadline, t_start + 1080.0)
    if time.monotonic() < phase3_end - 30.0:
        print(f"Phase 3: L-BFGS continuation")
        a_lb, sc_lb = _lbfgs_short(best_seq.copy(), phase3_end, rng, max_levels=7)
        update_best(a_lb, sc_lb, "lbfgs_main")
    
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Active-set polish (150s) ---
    phase4_end = min(deadline, t_start + 1230.0)
    if time.monotonic() < phase4_end - 10.0:
        print(f"Phase 4: Active-set polish")
        a_act, sc_act = _active_set_polish(best_seq.copy(), phase4_end, rng)
        update_best(a_act, sc_act, "active_set")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Symmetric optimization (150s) ---
    phase5_end = min(deadline, t_start + 1380.0)
    if time.monotonic() < phase5_end - 10.0:
        print(f"Phase 5: Symmetric optimization")
        a_sym, sc_sym = _symmetric_optimize(best_seq.copy(), phase5_end, rng)
        update_best(a_sym, sc_sym, "symmetric")
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Evolutionary restarts (180s) ---
    phase6_end = min(deadline, t_start + 1560.0)
    if time.monotonic() < phase6_end - 30.0:
        print(f"Phase 6: Evolutionary search")
        evo_deadline = min(phase6_end, time.monotonic() + 150.0)
        a_evo, sc_evo = _evolutionary_search(best_seq.copy(), evo_deadline, rng, pop_size=6)
        update_best(a_evo, sc_evo, "evolutionary")
    
    print(f"Phase 6 done: best={best_score:.10f}")
    
    # --- Phase 7: Try different n (60s) ---
    phase7_end = min(deadline, t_start + 1620.0)
    if best_seq is not None and time.monotonic() < phase7_end - 10.0:
        n_curr = len(best_seq)
        for target_n in [n_curr // 2, n_curr * 2]:
            if target_n < 512 or target_n > 16384:
                continue
            if time.monotonic() > phase7_end - 20:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Interp n={target_n}: score={sc_interp:.10f}")
            update_best(p_interp, sc_interp, f"interp_n{target_n}")
            
            if time.monotonic() < phase7_end - 15 and sc_interp < best_score * 1.01:
                dl = min(phase7_end - 10, time.monotonic() + 40.0)
                a_r, sc_r = _lbfgs_short(p_interp, dl, rng, max_levels=3)
                update_best(a_r, sc_r, f"lbfgs_n{target_n}")
    
    print(f"Phase 7 done: best={best_score:.10f}")
    
    # --- Phase 8: Final intensive polishing ---
    remaining = max(10.0, deadline - time.monotonic())
    if best_seq is not None and remaining > 20.0:
        print(f"Phase 8: Final polish, remaining={remaining:.0f}s")
        
        # 8a: Very fine L-BFGS
        if remaining > 50:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.4)
            a_f, sc_f = _lbfgs_short(best_seq.copy(), dl, rng, max_levels=5)
            update_best(a_f, sc_f, "final_lbfgs")
        
        # 8b: Fast MW with tiny lr
        if time.monotonic() < deadline - 15:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.3)
            a_mw, sc_mw = _fast_mw_method(best_seq.copy(), dl, rng, k_top=3, lr_init=0.1)
            update_best(a_mw, sc_mw, "final_mw")
        
        # 8c: Active set again
        if time.monotonic() < deadline - 10:
            dl = min(deadline - 5, time.monotonic() + remaining * 0.2)
            a_as, sc_as = _active_set_polish(best_seq.copy(), dl, rng)
            update_best(a_as, sc_as, "final_active")
    
    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_norm_fn(final_seq)
        print(f"FINAL: norm_score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        # Validate with evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite: {sc_eval}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_triangular_weight(8192)
    print(f"FALLBACK: n=8192, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
