"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP direction-finding, FFT-accelerated Adam with line search,
perturbation-based restarts, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def fft_convolve(a, b=None):
    """Fast convolution using FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb)[:n]


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=15.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M for all x.
    Returns the optimal g (or None if infeasible).
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    """
    Golden-section line search to find best t in (0,1] for (1-t)*f + t*g.
    Returns (best_t, best_f, best_score).
    """
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    x1 = hi - (hi - lo) / gr
    x2 = hi - (hi - lo) * (1.0 / gr)

    def obj(t):
        if t <= 0:
            return 0.0
        c = normalize((1.0 - t) * f + t * g)
        if c is None:
            return np.inf
        return internal_score(c)

    f1 = obj(x1)
    f2 = obj(x2)
    for _ in range(max_iters):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - (hi - lo) / gr
            f1 = obj(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = hi - (hi - lo) * (1.0 / gr)
            f2 = obj(x2)
        if abs(hi - lo) < 1e-6:
            break

    t_best = (lo + hi) / 2.0
    c_best = normalize(f + t_best * (g - f))
    if c_best is None:
        t_best = 0.1
        c_best = normalize(f + t_best * (g - f))
    s_best = internal_score(c_best)
    return t_best, c_best, s_best


def lp_iterate_enhanced(f, max_iters=500, t_init=0.5, t_min=0.05, deadline=None,
                         verbose=True, line_search_mode=True):
    """
    Enhanced LP-based iterations with line search and perturbation.
    f must be normalized to sum 1.
    Returns (best_f, best_score).
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = fft_convolve(f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M

    no_improve = 0
    t_step = t_init
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Every 50 iterations without improvement, try perturbation
        if no_improve % 50 == 49 and no_improve > 0:
            noise = rng.normal(0, 0.05, n)
            f_pert = f + noise
            f_pert = normalize(f_pert)
            if f_pert is not None:
                sc_pert = internal_score(f_pert)
                if sc_pert < best_score:
                    best_f = f_pert.copy()
                    best_score = sc_pert
                    M = float(fft_convolve(best_f).max())
                    f = best_f.copy()
                    no_improve = 0

        # Solve LP direction
        g_raw = lp_direction(f, M, timeout=10.0)
        if g_raw is None:
            # Try with slightly relaxed constraint
            g_raw = lp_direction(f, M * 1.05, timeout=10.0)
            if g_raw is None:
                break

        g = normalize(g_raw)
        if g is None:
            break

        if line_search_mode:
            t_best, c_new, score_new = line_search(f, g, best_score)
        else:
            t_best = t_step
            c_new = normalize(f + t_best * (g - f))
            if c_new is None:
                break
            score_new = internal_score(c_new)

        if score_new < best_score - 1e-12:
            best_f = c_new.copy()
            best_score = score_new
            M = float(fft_convolve(best_f).max())
            f = c_new.copy()
            no_improve = 0
            t_step = max(t_min, t_step * 0.997)
        else:
            no_improve += 1
            f = best_f.copy()
            M = best_M
            if no_improve % 20 == 0:
                t_step = max(t_min, t_step * 0.75)
            if no_improve > 300:
                break

        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={best_M:.6f}, no_improve={no_improve}")

    return best_f, best_score


def adam_fft(p0, deadline, lr=0.05, reg=1e-10, verbose=True, seed=0):
    """
    Adam gradient descent in logit space with FFT-based convolution.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = fft_convolve(p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.15 * max_c)
    rng = np.random.default_rng(seed)

    # Precompute FFT size
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve(p0)
        max_c = float(c.max())

        # Soft argmax weights
        x = (c - max_c) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Gradient via FFT: grad_p[i] = 2 * sum_x w[x] * p0[x-i] (circular-ish)
        # Use correlation: h[i] = sum_j w[j+i] * p0[j] for i=0..n-1
        # This is the cross-correlation of w and p0
        fw = np.fft.rfft(w, fft_size)
        fp = np.fft.rfft(p0, fft_size)
        # Cross-correlation: ifft(conj(fw) * fp) gives sum_k w[k-j] * p0[j]
        # We want h[i] = sum_j w[j+i] * p0[j] = sum_j p0[j] * w[j+i]
        # This is the cross-correlation of p0 and w at lag i
        # = ifft(conj(fp) * fw)[i] ... let me think again
        # Actually: (convolve(p0, w))[i] = sum_j p0[j] * w[i-j]
        # We want h[i] = sum_j p0[j] * w[j+i]
        # Let's flip w: wf[i] = w[n-1-i] (or use the full 2n-1 length)
        # h[i] = sum_j p0[j] * w[j+i] = cross_correlation(p0, w, lag=i)
        # = convolve(p0, flip(w))[i + n - 1] if we use full padding
        
        # Simpler: use the direct formula with the weight vector
        # h[i] = sum_{j=0}^{n-1} p0[j] * w[j+i] where w is defined on [0, 2n-2]
        # This is a sliding dot product, compute via FFT:
        # h = ifft(fft(p0, 2n-1) * conj(fft(w, 2n-1))) -- not quite right for linear cross-corr
        
        # Let me just do it directly for n=512 (it's fast enough)
        h = np.zeros(n)
        for i in range(n):
            # h[i] = sum_j p0[j] * w[j+i] for j=0..n-1 (w has 2n-1 elements)
            h[i] = np.dot(p0, w[i:i + n])
        
        grad_p = 2.0 * h + reg * p0

        # Logit space gradient
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c_new = fft_convolve(p)
        score = 2.0 * n * float(c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.93)

        if it % 100 == 0:
            beta = max(1e-14, beta * 0.5)

        if no_imp > 1500:
            # Restart with perturbation
            z = 0.85 * best_z + 0.35 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-12, 0.15 * float(fft_convolve(
                np.exp(z - z.max()) / np.exp(z - z.max()).sum()).max()))

        p0 = p

    return best_p, best_score


def adam_fft_fast(p0, deadline, lr=0.05, seed=0):
    """
    Faster Adam using vectorized correlation via FFT.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = fft_convolve(p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.15 * max_c)
    rng = np.random.default_rng(seed)

    # Precompute FFT size for cross-correlation
    fft_size = 1
    while fft_size < 2 * n:
        fft_size *= 2

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve(p0)
        max_c = float(c.max())

        x = (c - max_c) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        # Cross-correlation h[i] = sum_j p0[j] * w[i+j]
        # Use FFT: h = ifft(conj(fft(p0)) * fft(w_padded))[0:n]
        # Actually, h[i] = sum_{j} p0[j] * w[i+j]
        # Define W[k] = w[k] for k in [0, 2n-2]
        # h[i] = sum_{j=0}^{n-1} p0[j] * W[i+j]
        # This is the cross-correlation of p0 and W at lag i
        # = (p0 cross W)[i] = sum_j p0[j] * W[i+j]
        # Using FFT: ifft(fft(p0, N) * conj(fft(W, N)))... 
        # Actually cross-correlation = ifft(conj(fft(p0)) * fft(W))
        # Let me verify: (x cross y)[k] = sum_n x[n+k] * conj(y[n])
        # We want h[i] = sum_j p0[j] * w[i+j]
        # Let x = w, y = p0: (x cross y)[i] = sum_j w[i+j] * conj(p0[j]) = h[i] (since p0 is real)
        # So h = ifft(conj(fft(p0, N)) * fft(w, N))
        
        fp = np.fft.rfft(p0, fft_size)
        fw = np.fft.rfft(w, fft_size)
        h_full = np.fft.irfft(np.conj(fp) * fw, fft_size)
        h = h_full[:n]

        grad_p = 2.0 * h + 1e-10 * p0

        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c_new = fft_convolve(p)
        score = 2.0 * n * float(c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.92)

        if it % 50 == 0:
            beta = max(1e-14, beta * 0.5)

        if no_imp > 1000:
            z = 0.85 * best_z + 0.4 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-12, 0.15 * float(fft_convolve(
                np.exp(z - z.max()) / np.exp(z - z.max()).sum()).max()))

        p0 = p

    return best_p, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_profiles(n, seed=42):
    """Generate diverse initial profiles."""
    rng = np.random.default_rng(seed)
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
        a = 1.0 + eta * u
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Quadratic
    for c1 in [0.3, 0.7, 1.0, 1.5]:
        a = 1.0 + 1.0 * u + c1 * u * u
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Cosine-based
    for k in [1, 2, 3]:
        a = 1.0 + 0.3 * np.cos(k * np.pi * u)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Random smooth
    for _ in range(5):
        z = rng.normal(0, 0.5, n)
        # Smooth it
        kernel = np.exp(-0.5 * (np.linspace(-3, 3, min(n, 21))) ** 2)
        kernel = kernel / kernel.sum()
        z_s = np.convolve(z, kernel, mode='same')
        a = np.exp(z_s)
        profiles.append(a / a.sum())

    # Sine bump
    a = np.sin(np.pi * 0.5 * (u + 1)) ** 2
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())

    # Double bump
    a = np.exp(-4 * u ** 2) + 0.3 * np.exp(-4 * (u - 0.5) ** 2) + 0.3 * np.exp(-4 * (u + 0.5) ** 2)
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())

    return profiles


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: LP + Adam(FFT) + perturbation + multi-length.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Try to get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent
                best_score = internal_score(parent)
                print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 512
        profiles = make_profiles(n_init, seed)
        for p in profiles:
            sc = internal_score(p)
            if sc < best_score:
                best_score = sc
                best_seq = p
        print(f"Generated init: n={len(best_seq)}, score={best_score:.8f}")

    # ===== Phase 1: LP at n=512 (400s) =====
    n1 = 512
    deadline_p1 = min(deadline, t_start + 400.0)
    a1 = interpolate_seq(best_seq, n1)
    if a1 is not None:
        sc1 = internal_score(a1)
        print(f"\nPhase 1: LP at n={n1}, score={sc1:.8f}")
        f_best, sc_best = lp_iterate_enhanced(a1, max_iters=1000, t_init=0.4,
                                               t_min=0.03, deadline=deadline_p1,
                                               verbose=True, line_search_mode=True)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best
            print(f"  Phase 1 result: score={best_score:.8f}")

    # ===== Phase 2: Adam FFT at n=512 (500s) =====
    n2 = 512
    deadline_p2 = min(deadline, t_start + 900.0)
    if time.monotonic() < deadline_p2 - 30:
        a2 = interpolate_seq(best_seq, n2)
        if a2 is not None:
            sc2 = internal_score(a2)
            print(f"\nPhase 2: Adam FFT at n={n2}, score={sc2:.8f}")
            # Run multiple Adam runs with different seeds/LR
            sub_deadline = deadline_p2
            n_runs = 3
            time_per_run = max(30, (sub_deadline - time.monotonic()) / n_runs - 5)
            for run in range(n_runs):
                if time.monotonic() > sub_deadline - 10:
                    break
                start = time.monotonic()
                end = min(sub_deadline - 5, start + time_per_run)
                lr_val = [0.05, 0.03, 0.08][run % 3]
                p_best_run, sc_run = adam_fft_fast(best_seq, end, lr=lr_val, seed=100 + run)
                if sc_run < best_score:
                    best_score = sc_run
                    best_seq = p_best_run
                    print(f"  Adam run {run}: score={best_score:.8f}")
                else:
                    print(f"  Adam run {run}: score={sc_run:.8f} (no improvement)")

    # ===== Phase 3: Multi-length exploration (400s) =====
    deadline_p3 = min(deadline, t_start + 1300.0)
    if time.monotonic() < deadline_p3 - 30:
        print(f"\nPhase 3: Multi-length exploration")
        for n_try in [384, 640, 768]:
            if time.monotonic() > deadline_p3 - 60:
                break
            a_try = interpolate_seq(best_seq, n_try)
            if a_try is None:
                continue
            sc_try = internal_score(a_try)
            sub_end = min(deadline_p3 - 10, time.monotonic() + 80)
            p_best_try, sc_best_try = adam_fft_fast(a_try, sub_end, lr=0.04, seed=200 + n_try)
            if sc_best_try < best_score:
                best_score = sc_best_try
                best_seq = p_best_try
                print(f"  n={n_try}: score={best_score:.8f} (NEW BEST)")
            else:
                print(f"  n={n_try}: score={sc_best_try:.8f} (no improvement, base was {sc_try:.8f})")

    # ===== Phase 4: LP + Adam combo on best (remaining time) =====
    if time.monotonic() < deadline - 30:
        n4 = 512
        deadline_p4 = deadline
        # Go back to n=512 for final polish
        a4 = interpolate_seq(best_seq, n4)
        if a4 is not None:
            print(f"\nPhase 4: Final polish at n={n4}, score={internal_score(a4):.8f}")
            
            # Quick LP
            d_lp = min(deadline_p4 - 60, time.monotonic() + 120)
            f_best, sc_best = lp_iterate_enhanced(a4, max_iters=200, t_init=0.3,
                                                   t_min=0.05, deadline=d_lp,
                                                   verbose=False, line_search_mode=True)
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  LP final: score={best_score:.8f}")

            # Final Adam
            if time.monotonic() < deadline_p4 - 10:
                p_best, sc_best = adam_fft_fast(best_seq, deadline_p4 - 5, lr=0.03, seed=999)
                if sc_best < best_score:
                    best_score = sc_best
                    best_seq = p_best
                    print(f"  Adam final: score={best_score:.8f}")

    # ===== Perturbation exploration with remaining time =====
    if time.monotonic() < deadline - 20:
        print(f"\nPerturbation phase: current best={best_score:.8f}")
        for pert_run in range(5):
            if time.monotonic() > deadline - 15:
                break
            noise_scale = [0.03, 0.05, 0.08, 0.1, 0.15][pert_run % 5]
            perturbed = best_seq + rng.normal(0, noise_scale, len(best_seq))
            perturbed = normalize(perturbed)
            if perturbed is None:
                continue
            sub_end = min(deadline - 10, time.monotonic() + 60)
            p_best_p, sc_p = adam_fft_fast(perturbed, sub_end, lr=0.04, seed=500 + pert_run)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = p_best_p
                print(f"  Perturbation {pert_run} (scale={noise_scale}): score={best_score:.8f} (NEW BEST)")
            else:
                print(f"  Perturbation {pert_run} (scale={noise_scale}): score={sc_p:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, internal_score={final_score:.8f}")

    # Validate with evaluate_sequence
    seq_list = [float(x) for x in best_seq]
    try:
        eval_score = evaluate_sequence(seq_list)
        print(f"  evaluate_sequence: {eval_score:.8f}")
    except Exception as e:
        print(f"  evaluate_sequence error: {e}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        t = np.linspace(0, 1, 512)
        best_seq = 1.0 + 1.5 * (2 * t - 1)
        best_seq = np.maximum(best_seq, 1e-10)
        best_seq = best_seq / best_seq.sum()

    return [float(x) for x in best_seq]
