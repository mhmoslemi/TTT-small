2*n*max(conv(a,a))/sum(a)^2. Uses FFT-based gradient computation,

def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))

def compute_M(f):
    """Compute max of linear convolution of f with itself."""
    c = np.convolve(f, f)
    return float(c.max())

def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s

def cross_corr_for_grad(w, a):
    """
    Compute grad[i] = sum_k w[k] * a[k-i] for i=0..n-1.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    N = 1
    while N < (2*n-1) + n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    result = np.fft.irfft(fw * fa, N)
    return result[n-1:2*n-1]

def softmax_weights(c, beta):
    """Compute softmax weights for convolution values."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s

def gradient_descent_step(a, beta, eta):
    """
    One step of mirror descent in log-space.
    Returns new a (normalized, non-negative).
    """
    n = len(a)
    c = np.convolve(a, a)
    w = softmax_weights(c, beta)
    # grad[i] = 2 * sum_k w[k] * a[k-i]
    h = cross_corr_for_grad(w, a)
    grad = 2.0 * h
    # Project: subtract mean to satisfy sum constraint
    grad = grad - float(np.mean(grad))
    
    # Log-space step
    z = np.log(np.maximum(a, 1e-30))
    z = z - eta * grad
    # Clip to prevent overflow
    z = np.clip(z, -50, 50)
    z = z - z.max()  # numerical stability
    a_new = np.exp(z)
    a_new = a_new / a_new.sum()
    return a_new, c, w

def smooth_value(c, beta):
    """Compute the smoothed objective value."""
    mval = float(c.max())
    x = (c - mval) / beta
    return mval + beta * np.log(float(np.exp(x).sum()))

def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A

def solve_lp_for_g(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()

    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, f)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 0:
        result = result / s
    return result

    grad = 2.0 * h
    return grad, c, w


def gradient_optimize(a_init, deadline, rng, init_beta=0.01, max_iters=50000, verbose=False):
    """Adam-based optimization in log-space with adaptive beta."""
    a = normalize_nonneg(a_init.copy())
    if a is None:
        return a_init, compute_M(a_init)

    n = len(a)
    beta = init_beta
    best_a = a.copy()
    best_M = compute_M(a)
    no_imp = 0

    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    lr = 0.05

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, c, w = compute_grad(a, beta)
        grad = grad - float(np.mean(grad))
        grad += 1e-8 * a

        t_adam += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)

        z = np.log(np.maximum(a, 1e-30))
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a_new = np.exp(z)
        a_new = a_new / a_new.sum()

        if not np.all(np.isfinite(a_new)):
            a = best_a.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        a = a_new
        M_new = compute_M(a)

        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.01, lr * 0.95)

        if it % 100 == 0 and it > 0:
            beta = max(1e-5, beta * 0.97)

        if no_imp > 800:
            z = np.log(np.maximum(best_a, 1e-30))
            z = z + 0.08 * rng.normal(size=n)
            z = z - z.mean()
            z = np.clip(z, -50, 50)
            a = np.exp(z)
            a = a / a.sum()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = 0.05
            no_imp = 0
            beta = max(0.01, 0.05 * compute_M(a))

        if verbose and it % 2000 == 0:
            print(f"      GD iter {it}: M={M_new:.10f}, best={best_M:.10f}, beta={beta:.6f}")

    return best_a, best_M


def make_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp_for_g(f, M):
    n = len(f)
    c_vec = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 4.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, max_iters=1000):

            f = normalize_nonneg(f)

                f = best_f.copy()
            M = compute_M(f)
            continue

        if g_sum < 1e-15:

            f_try = (1.0 - t) * f + t * g
            f_try = normalize_nonneg(f_try)
            if f_try is None:
                continue
            M_try = compute_M(f_try)
            if
