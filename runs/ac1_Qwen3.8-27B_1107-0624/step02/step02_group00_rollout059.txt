"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses power-norm gradient descent with FFT-based cross-correlation, smooth-max polish,
multiple lengths, and diverse initializations to break local minima.
Key innovation: clean gradient formula grad[k] = 2*(w cross-correlate a)[k]
enables efficient optimization in power-norm space with adaptive p.

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _cross_correlate_fft(w, a):
    """Compute h[k] = sum_j w[j+k]*a[j] for k=0..n-1 using FFT."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    h = np.fft.irfft(np.conj(fw) * fa, N)[:n]
    return h


def _grad_power_norm(a, p):
    """
    Gradient of (sum(max(c,0)^p))^(1/p) where c = conv(a,a).
    Returns (gradient, objective_value).
    """
    n = len(a)
    c = _fft_convolve(a)
    c_pos = np.maximum(c, 0.0)
    if c_pos.max() < 1e-30:
        return np.zeros(n), 0.0
    
    # w_j = c_pos[j]^(p-1)
    w = c_pos ** (p - 1)
    S = float(np.sum(c_pos * w))  # = sum(c_pos^p)
    if S < 1e-30:
        return np.zeros(n), 0.0
    
    # grad[k] = 2 * S^(1/p - 1) * h[k] where h = cross_correlate(w, a)
    h = _cross_correlate_fft(w, a)
    factor = 2.0 * S ** (1.0 / p - 1.0)
    grad = factor * h
    val = S ** (1.0 / p)
    return grad, val


def _grad_smooth_max(a, beta):
    """
    Gradient of smooth_max(conv(a,a)) using log-sum-exp.
    Returns (gradient_unscaled, smooth_max_value).
    Gradient is w cross-correlate a (times 2 for the full gradient).
    """
    n = len(a)
    c = _fft_convolve(a)
    cv = float(c.max())
    x = (c - cv) / beta
    # Clip for numerical stability
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    h = _cross_correlate_fft(w, a)
    grad = 2.0 * h
    val = cv + beta * float(np.log(s))
    return grad, val


def _normalize(a):
    """Normalize to sum=1, ensure non-negative."""
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_raised_cosine(n):
    t = np.linspace(0.0, np.pi, n)
    a = 1.0 + 0.5 * np.cos(t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_sym(n, eta=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - np.abs(t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_asym(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_sym(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * (1.0 - u * u)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_window(n, sigma=0.4):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-0.5 * (u / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _random_smooth(n, rng, n_fourier=10):
    """Generate a random smooth non-negative profile using Fourier series."""
    k = np.arange(1, n_fourier + 1)
    coeffs = rng.normal(size=n_fourier)
    phases = rng.uniform(0, 2 * np.pi, size=n_fourier)
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i in range(n_fourier):
        a += (coeffs[i] / (i + 1)) * np.cos(2 * np.pi * k[i] * t + phases[i])
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _peak_shave(a, n_steps=100):
    """Simple peak-shaving: reduce coordinates that contribute most to the peak."""
    a = a.copy().astype(np.float64)
    n = len(a)
    a = _normalize(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    for step in range(n_steps):
        c = _fft_convolve(a)
        i_peak = int(np.argmax(c))
        
        # Find which coordinates contribute most to c[i_peak]
        # c[i_peak] = sum_j a[j]*a[i_peak-j]
        j_low = max(0, i_peak - n + 1)
        j_high = min(n - 1, i_peak)
        contrib = np.zeros(n)
        for j in range(j_low, j_high + 1):
            contrib[j] += a[j] * a[i_peak - j]
            if j != i_peak - j:
                contrib[i_peak - j] += a[j] * a[i_peak - j]
        
        # Reduce the top contributors slightly
        top_k = max(2, n // 32)
        top_idx = np.argsort(contrib)[-top_k:]
        reduce_amt = 0.01 * a[top_idx].min()
        a[top_idx] -= reduce_amt
        a = np.maximum(a, 0.0)
        a = _normalize(a)
        
        sc = _score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
    
    return best_a

            inits.append(_interpolate_profile(parent, n_t))
            # Perturbed parent
            for _ in range(2):
                pert = _interpolate_profile(parent, n_t)
                pert = pert * (1.0 + 0.05 * rng.normal(size=n_t))
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                inits.append(pert)
        
        inits.append(_make_raised_cosine(n_t))
        inits.append(_make_linear_sym(n_t, 2.0))
        inits.append(_make_linear_asym(n_t, 1.5))
        inits.append(_make_quadratic_sym(n_t, 1.0))
        inits.append(_make_gaussian_window(n_t, 0.4))
        inits.append(_random_smooth(n_t, rng, 8))
        
        print(f"\n=== Length n={n_t}, {len(inits)} inits, budget={t_alloc:.0f}s ===")
        
        # For each initialization, run optimization
        per_init_time = max(5.0, t_alloc / max(1, len(inits)))
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > t_phase_end - 5:
                break
            
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            
            t_init_deadline = min(t_phase_end - 5, time.monotonic() + per_init_time)
            
            if time.monotonic() > t_init_deadline:
                continue
            
            # Phase 1: Power-norm Adam (main workhorse)
            t_p1 = t_init_deadline - (t_init_deadline - time.monotonic()) * 0.6
            a1 = _adam_optimize_power_norm(a0, t_p1, max_iter=30000,
                                            p_start=20, p_end=150, lr=0.03)
            sc1 = _score(a1)
            if np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_seq = a1.copy()
            
            # Phase 2: Smooth-max Adam polish
            t_p2 = t_init_deadline - (t_init_deadline - time.monotonic()) * 0.3
            if time.monotonic() < t_p2:
                a2 = _adam_optimize_smooth_max(a1, t_p2, max_iter=20000, lr=0.015)
                sc2 = _score(a2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a2.copy()
            
            # Phase 3: Nesterov power-norm (different dynamics)
            if time.monotonic() < t_init_deadline - 2:
                a3 = _nesterov_power_norm(a2 if time.monotonic() < t_p2 else a1,
                                           t_init_deadline - 2, max_iter=20000, p=80, lr=0.01)
                sc3 = _score(a3)
                if np.isfinite(sc3) and sc3 < best_score:
                    best_score = sc3
                    best_seq = a3.copy()
            
            print(f"  Init {ii}: {sc0:.8f} -> P1:{sc1:.10f} -> P2:{sc2:.10f} -> P3:{sc3:.10f}")
    
    # Final polish on the best sequence found
    if best_seq is not None:
        print(f"\n=== Final polish phase, best_score={best_score:.10f} ===")
        remaining = deadline - time.monotonic()
        
        if remaining > 30:
            # Smooth-max with very small beta for final squeeze
            t_final = time.monotonic() + min(remaining * 0.4, 200.0)
            a_fp = _adam_optimize_smooth_max(best_seq, t_final, max_iter=50000,
                                              beta_start=1e-4, lr=0.005)
            sc_fp = _score(a_fp)
            if np.isfinite(sc_fp) and sc_fp < best_score:
                best_score = sc_fp
                best_seq = a_fp.copy()
            print(f"  Final smooth-max: {sc_fp:.10f}")
        
        if remaining > 30 and time.monotonic() < deadline - 20:
            # Nesterov with high p
            t_fin2 = min(deadline - 10, time.monotonic() + min(remaining * 0.3, 150.0))
            a_fn = _nesterov_power_norm(best_seq, t_fin2, max_iter=30000, p=200, lr=0.005)
            sc_fn = _score(a_fn)
            if np.isfinite(sc_fn) and sc_fn < best_score:
                best_score = sc_fn
                best_seq = a_fn.copy()
            print(f"  Final Nesterov: {sc_fn:.10f}")
        
        # Peak shaving (fast, just a few iterations)
        if time.monotonic() < deadline - 10:
            a_ps = _peak_shave(best_seq, n_steps=50)
            sc_ps = _score(a_ps)
            if np.isfinite(sc_ps) and sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
            print(f"  Peak shave: {sc_ps:.10f}")
        
        # One more round of smooth-max on the very best
        if time.monotonic() < deadline - 10:
            t_last = deadline - 10
            a_last = _adam_optimize_smooth_max(best_seq, t_last, max_iter=30000,
                                               beta_start=5e-5, lr=0.003)
            sc_last = _score(a_last)
            if np.isfinite(sc_last) and sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Very final: {sc_last:.10f}")
    
    # Fallback
    if best_seq is None:
        best_seq = _make_raised_cosine(256)
        best_score = _score(best_seq)
    
    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s <= 0.01:
        final_seq = _make_raised_cosine(256)
    
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
