"""
Multi-resolution LP-based optimization with active-set reduction,
adaptive step sizing, and Adam polish for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses progressive dimension increase and multiple initialization strategies.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction_active(f, M, active_frac=0.3, timeout=10.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M for active constraints.
    Uses only the most active (largest) fraction of constraints for speed.
    """
    n = len(f)
    A_full = build_conv_matrix(f)
    
    # Identify active constraints: rows where A_full @ f is close to M
    Af = A_full @ f
    # Active if Af > 0.7 * M (or just use top fraction)
    n_constraints = 2 * n - 1
    n_active = max(n, int(n_constraints * active_frac))
    
    # Get indices of most active constraints
    active_idx = np.argsort(Af)[-n_active:]
    active_idx = np.sort(active_idx)
    
    A = A_full[active_idx, :]
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            # Verify against full constraints
            full_conv = A_full @ g
            if np.max(full_conv) <= M * 1.01:
                return g
            # If violated, scale down
            scale = M / max(np.max(full_conv), 1e-30)
            g = g * min(1.0, scale)
            return g
    return None


def lp_direction_full(f, M, timeout=10.0):
    """Solve LP with all constraints (slower but more accurate)."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterate(f, max_iters=500, t_init=0.4, t_min=0.01, deadline=None,
               verbose=False, use_active=True, active_frac=0.3):
    """
    LP-based iterations to reduce max(conv(f,f)).
    f must be normalized to sum 1.
    Returns (best_f, best_score).
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M

    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        if use_active:
            g_raw = lp_direction_active(f, M, active_frac=active_frac, timeout=5.0)
        else:
            g_raw = lp_direction_full(f, M, timeout=5.0)
        
        if g_raw is None:
            # Try full LP as fallback
            g_raw = lp_direction_full(f, M, timeout=10.0)
            if g_raw is None:
                break

        g = normalize(g_raw)
        if g is None:
            break

        # Adaptive step: try several step sizes
        best_local_f = f.copy()
        best_local_score = best_score
        
        for t_candidate in [t_step, t_step * 2.0, t_step * 0.5]:
            if t_candidate < 0.01 or t_candidate > 0.8:
                continue
            f_new = (1.0 - t_candidate) * f + t_candidate * g
            f_new = normalize(f_new)
            if f_new is None:
                continue
            c_new = np.convolve(f_new, f_new)
            M_new = float(c_new.max())
            score_new = 2.0 * n * M_new
            
            if score_new < best_local_score:
                best_local_score = score_new
                best_local_f = f_new.copy()
                best_local_M = M_new
                best_local_t = t_candidate

        if best_local_score < best_score - 1e-14:
            best_f = best_local_f.copy()
            best_score = best_local_score
            best_M = best_local_M
            f = best_local_f.copy()
            M = best_local_M
            t_step = max(t_min, best_local_t)
            no_improve = 0
        else:
            no_improve += 1
            f = best_f.copy()
            M = best_M
            if no_improve % 10 == 0:
                t_step = max(t_min, t_step * 0.8)
            if no_improve > 150:
                break

        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={M:.6f}, t={t_step:.4f}")

    return best_f, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return normalize(a.copy())
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def upsample_duplicate(a, factor=2):
    """Upsample by duplicating each element."""
    n = len(a)
    a_new = np.repeat(a, factor)
    return normalize(a_new)


def upsample_insert_zero(a, factor=2):
    """Upsample by inserting zeros between elements."""
    n = len(a)
    if factor == 2:
        a_new = np.zeros(2 * n)
        a_new[0::2] = a
    else:
        a_new = np.zeros(n * factor)
        a_new[::factor] = a
    return normalize(a_new)


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_multi_profile(n, coeffs):
    """Create profile from polynomial coefficients."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.ones(n)
    for k, c in enumerate(coeffs, start=1):
        a += c * (u ** k)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_random_profile(n, rng, sparsity=0.5):
    """Create a random non-negative profile."""
    a = rng.random(n)
    if sparsity < 1.0:
        mask = rng.random(n) < sparsity
        a[~mask] = 0.0
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def make_bimodal_profile(n, width=0.3):
    """Create a bimodal profile with peaks at the ends."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.0) / width) ** 2) + np.exp(-((t - 1.0) / width) ** 2) + 0.1
    return a / a.sum()

    # So it's just 2*a[i] (derivative of a[i]^2 is 2*a[i]).
    
    # Let me just compute it directly using the formula:
    # grad[i] = sum_k w[k] * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] if 0 <= k-i <= n-1 (counting both j=i and j=k-i, but they're the same when k=2i)
    # Actually the cleanest: grad[i] = 2 * sum_k w[k] * a[k-i] * [0<=k-i<=n-1] - sum_k w[k] * a[k-i] * [k=2i, 0<=k-i<=n-1]
    # The subtraction handles the double-counting at k=2i.
    # But actually for k=2i: dc[2i]/da[i] = 2*a[i], and 2*a[k-i]*[valid] = 2*a[i]. So no correction needed!
    # Because when k=2i, j=i gives a[i]*a[i], and the derivative is 2*a[i].
    # And 2*a[k-i] = 2*a[i]. So the formula grad[i] = 2*sum_k w[k]*a[k-i]*[valid] is correct!
    
    # Let me verify with a simple case: a=[a0,a1], c = [a0^2, 2*a0*a1, a1^2]
    # dc/da0 = [2*a0, 2*a1, 0]
    # Formula: grad[0] = 2*(w[0]*a[0-0] + w[1]*a[1-0] + w[2]*a[2-0]) = 2*(w[0]*a0 + w[1]*a1 + w[2]*0)
    # = 2*w[0]*a0 + 2*w[1]*a1. But dc/da0[0] = 2*a0, dc/da0[1] = 2*a1, dc/da0[2] = 0.
    # So grad[0] = w[0]*2*a0 + w[1]*2*a1 + w[2]*0 = 2*w[0]*a0 + 2*w[1]*a1. ✓
    
    # grad[1] = 2*(w[0]*a[0-1](invalid) + w[1]*a[1-1] + w[2]*a[2-1]) = 2*(0 + w[1]*a0 + w[2]*a1)
    # = 2*w[1]*a0 + 2*w[2]*a1. And dc/da1 = [0, 2*a0, 2*a1].
    # grad[1] = w[0]*0 + w[1]*2*a0 + w[2]*2*a1 = 2*w[1]*a0 + 2*w[2]*a1. ✓
    
    # Great, so the formula is: grad[i] = 2 * sum_{k: 0<=k-i<=n-1} w[k] * a[k-i]
    # = 2 * (w * flip_a)[i] where flip_a is a reversed and padded...
    # Actually it's a correlation: grad[i] = 2 * sum_j w[i+j] * a[j] for valid indices.
    # Equivalently: grad = 2 * correlate(w, a)[:n] (with appropriate alignment)
    
    # Use FFT for efficiency
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1, 0<=i+j<=2n-2
    # This is like a cross-correlation of w and a.
    # corr[i] = sum_j a[j] * w[i+j]
    # = sum_j a[j] * w[(i+j)]
    # Let's pad w to length 2n-1 (it already is) and a to length 2n-1 (zero-padded)
    # Then corr = ifft(fft(w, 2n) * conj(fft(a_padded, 2n)))... 
    
    # Actually, let's just use the direct approach for n <= 2048 (it's O(n^2) but with small constant)
    # Or use the np.correlate approach
    
    # Better: use the fact that grad[i] = 2 * (a * w_reflected)[i] where * is cross-correlation
    # Let's just do it with a loop over i, using slicing of w
    
    grad = np.zeros(n)
    for i in range(n):
        # Need w[i : i+n] (indices i to i+n-1), and a[0:n]
        k_lo = i
        k_hi = i + n  # exclusive
        if k_lo < 2 * n - 1 and k_hi > 0:
            actual_lo = max(0, k_lo)
            actual_hi = min(2 * n - 1, k_hi)
            if actual_lo < actual_hi:
                # w[actual_lo:actual_hi] corresponds to a[actual_lo-i : actual_hi-i]
                j_lo = actual_lo - i
                j_hi = actual_hi - i
                grad[i] = 2.0 * np.dot(w[actual_lo:actual_hi], a[j_lo:j_hi])
    
    return grad, max_c


def gradient_max_conv_fft(a, beta=0.01):
    """FFT-based gradient of soft-max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    
    x = (c - max_c) / beta
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    
    # grad[i] = 2 * sum_j a[j] * w[i+j] for valid j
    # This is a cross-correlation: (a corr w)[i] = sum_j a[j] w[i+j]
    # Use FFT: pad both to size 2n
    size = 2 * n
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    # Cross-correlation via FFT
    corr = np.fft.irfft(A * np.conj(W), size)
    # Actually, let me think more carefully.
    # sum_j a[j] * w[i+j] for j=0..n-1
    # = sum_j a[j] * w[i+j]
    # Let's define w2 = w (length 2n-1, padded to 2n)
    # This is like (a * w2)[i] where * is correlation (not convolution)
    # In numpy: np.correlate(w2_padded, a, mode='full') gives...
    # Actually let me just use: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is the i-th element of the "anti-diagonal" sum.
    # 
    # Alternative: define b = w (length 2n-1). Then grad[i] = 2*sum_j a[j]*b[i+j]
    # = 2 * (a flipped, convolved with b) at appropriate index...
    # 
    # Let's just use a simple approach: build a matrix or use sliding window.
    # For n up to 2048, O(n^2) with numpy dot should be fast enough.
    
    grad = np.zeros(n)
    for i in range(n):
        # sum_j a[j] * w[i+j] for j = 0..n-1, valid when 0 <= i+j <= 2n-2
        j_lo = max(0, -i)  # always 0 since i >= 0
        j_hi = min(n, 2 * n - 1 - i + 1)  # i+j <= 2n-2 -> j <= 2n-2-i
        j_hi = min(n, 2 * n - 1 - i)
        if j_lo < j_hi:
            grad[i] = 2.0 * np.dot(a[j_lo:j_hi], w[i + j_lo:i + j_hi])
    
    return grad, max_c


def adam_polish(p0, deadline, lr=0.05, reg=1e-10, verbose=True):
    """
    Adam gradient descent in logit space to polish a non-negative sequence.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.1 * max_c)

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Soft argmax weights
        x = (c - max_c) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w_vec = np.zeros_like(c)
            w_vec[np.argmax(c)] = 1.0
        else:
            w_vec = ex / s

        # Gradient computation (vectorized)
        grad_p = np.zeros(n)
        for i in range(n):
            j_hi = min(n, 2 * n - 1 - i)
            if i < j_hi:
                grad_p[i] = 2.0 * np.dot(p0[:j_hi], w_vec[i:i + j_hi])
        
        grad_p += reg * p0

        # Logit space gradient
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 500 == 0:
            beta = max(1e-14, beta * 0.5)

        if no_imp > 3000:
            # Restart with perturbation
            rng_r = np.random.default_rng(it)
            z = 0.7 * best_z + 0.5 * rng_r.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0

        p0 = p

    return best_p, best_score


def perturb_and_retry(f, n_perturbations=5, t_perturb=0.1, max_lp_iters=20, deadline=None):
    """Try random perturbations and re-optimize with a few LP steps."""
    n = len(f)
    best_f = f.copy()
    best_score = 2.0 * n * float(np.convolve(f, f).max())
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
    
    for _ in range(n_perturbations):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Add noise
        noise = rng.normal(0, t_perturb, size=n)
        f_pert = f + noise
        f_pert = normalize(f_pert)
        if f_pert is None:
            continue
        
        # Quick LP refinement
        f_best, sc_best = lp_iterate(f_pert, max_iters=max_lp_iters, t_init=0.3,
                                      t_min=0.05, deadline=deadline, use_active=True,
                                      active_frac=0.2)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_f = f_best.copy()
    
    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-resolution LP + Adam search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
      1. Multiple initializations at small n
      2. Multi-resolution LP refinement (128 -> 256 -> 512 -> 1024)
      3. Adam polish at the best resolution
      4. Perturbation-based local search for final improvement
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 10.0)
    rng = np.random.default_rng(seed)

    # Get initial sequence
    best_seq = None
    best_score = np.inf

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent.copy()
                best_score = internal_score(parent)
                print(f"Parent sequence: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass

    # Phase 0: Try multiple initializations at n=128
    n0 = 128
    deadline_p0 = min(deadline, t_start + 200.0)
    
    candidates_0 = []
    
    # Start from parent if available
    if best_seq is not None:
        cand = interpolate_seq(best_seq, n0)
        if cand is not None:
            candidates_0.append(cand)
    
    # Analytic profiles
    profiles = [
        make_linear_profile(n0, 1.0),
        make_linear_profile(n0, 1.5),
        make_linear_profile(n0, 2.0),
        make_linear_profile(n0, 2.5),
        make_linear_profile(n0, 3.0),
        make_multi_profile(n0, [1.5, -0.5]),
        make_multi_profile(n0, [2.0, -1.0, 0.3]),
        make_multi_profile(n0, [1.0, 0.5, -0.3, 0.1]),
        make_bimodal_profile(n0, 0.2),
        make_bimodal_profile(n0, 0.4),
        make_bimodal_profile(n0, 0.6),
    ]
    candidates_0.extend(profiles)
    
    # Random profiles
    for _ in range(5):
        candidates_0.append(make_random_profile(n0, rng, sparsity=0.7))
    
    # Evaluate all candidates
    for cand in candidates_0:
        sc = internal_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    
    print(f"\nPhase 0: Best initial n={n0}, score={best_score:.8f}")
    
    # Phase 1: LP at n=128
    deadline_p1 = min(deadline, t_start + 400.0)
    print(f"\nPhase 1: LP at n={n0}")
    f_best, sc_best = lp_iterate(best_seq, max_iters=500, t_init=0.4, t_min=0.02,
                                  deadline=deadline_p1, verbose=True, use_active=True,
                                  active_frac=0.4)
    if f_best is not None and sc_best < best_score:
        best_score = sc_best
        best_seq = f_best.copy()
    print(f"  Phase 1 result: score={best_score:.8f}")

    # Phase 2: LP at n=256
    n2 = 256
    deadline_p2 = min(deadline, t_start + 800.0)
    if time.monotonic() < deadline_p2 - 60:
        print(f"\nPhase 2: LP at n={n2}")
        # Try different upsampling strategies
        a2_interp = interpolate_seq(best_seq, n2)
        a2_dup = upsample_duplicate(best_seq, 2)
        
        # Optimize the better one
        sc_interp = internal_score(a2_interp)
        sc_dup = internal_score(a2_dup)
        
        if sc_interp <= sc_dup:
            a2 = a2_interp
        else:
            a2 = a2_dup
        
        print(f"  Initial n={n2} score: interp={sc_interp:.8f}, dup={sc_dup:.8f}")
        
        f_best, sc_best = lp_iterate(a2, max_iters=400, t_init=0.35, t_min=0.02,
                                      deadline=deadline_p2, verbose=True, use_active=True,
                                      active_frac=0.3)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best.copy()
        
        # Also try the other upsampling if it was close
        other_a2 = a2_dup if a2 is not a2_dup else a2_interp
        if time.monotonic() < deadline_p2 - 30:
            f_best2, sc_best2 = lp_iterate(other_a2, max_iters=200, t_init=0.35,
                                            t_min=0.02, deadline=deadline_p2, verbose=False,
                                            use_active=True, active_frac=0.3)
            if f_best2 is not None and sc_best2 < best_score:
                best_score = sc_best2
                best_seq = f_best2.copy()
        
        print(f"  Phase 2 result: score={best_score:.8f}")

    # Phase 3: LP at n=512
    n3 = 512
    deadline_p3 = min(deadline, t_start + 1400.0)
    if time.monotonic() < deadline_p3 - 60:
        print(f"\nPhase 3: LP at n={n3}")
        a3_interp = interpolate_seq(best_seq, n3)
        a3_dup = upsample_duplicate(best_seq, 2) if len(best_seq) == 256 else interpolate_seq(best_seq, n3)
        
        sc_interp = internal_score(a3_interp)
        sc_dup = internal_score(a3_dup)
        
        if sc_interp <= sc_dup:
            a3 = a3_interp
        else:
            a3 = a3_dup
        
        print(f"  Initial n={n3} score: interp={sc_interp:.8f}, dup={sc_dup:.8f}")
        
        f_best, sc_best = lp_iterate(a3, max_iters=300, t_init=0.3, t_min=0.02,
                                      deadline=deadline_p3, verbose=True, use_active=True,
                                      active_frac=0.3)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best.copy()
        
        print(f"  Phase 3 result: score={best_score:.8f}")

    # Phase 3b: Try n=1024 if time permits
    n4 = 1024
    deadline_p3b = min(deadline, t_start + 1650.0)
    if time.monotonic() < deadline_p3b - 120 and len(best_seq) <= 512:
        print(f"\nPhase 3b: LP at n={n4}")
        a4 = interpolate_seq(best_seq, n4)
        sc_4 = internal_score(a4)
        print(f"  Initial n={n4} score: {sc_4:.8f}")
        
        f_best, sc_best = lp_iterate(a4, max_iters=150, t_init=0.25, t_min=0.02,
                                      deadline=deadline_p3b, verbose=True, use_active=True,
                                      active_frac=0.25)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best.copy()
            print(f"  Phase 3b result (n={n4}): score={best_score:.8f}")
        else:
            # Revert to n=512 if 1024 didn't help
            if len(best_seq) != 512:
                best_seq = interpolate_seq(best_seq, 512)
    
    # Phase 4: Adam polish
    if time.monotonic() < deadline - 30:
        n_final = len(best_seq)
        if n_final not in [256, 512, 1024]:
            best_seq = interpolate_seq(best_seq, 512)
            n_final = 512
        
        print(f"\nPhase 4: Adam polish at n={n_final}, initial score={internal_score(best_seq):.8f}")
        deadline_adam = deadline - 20
        p_best, sc_best = adam_polish(best_seq, deadline_adam, lr=0.04, reg=1e-10, verbose=True)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best.copy()
            print(f"  Adam result: score={best_score:.8f}")

    # Phase 5: Perturbation-based local search
    if time.monotonic() < deadline - 10:
        print(f"\nPhase 5: Perturbation search")
        n_p = len(best_seq)
        f_best, sc_best = perturb_and_retry(best_seq, n_perturbations=3,
                                             t_perturb=0.05, max_lp_iters=30,
                                             deadline=deadline - 5)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = f_best.copy()
            print(f"  Perturbation result: score={best_score:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    if final_score != best_score:
        print(f"Warning: score mismatch! Computed={final_score:.8f}, tracked={best_score:.8f}")
    best_score = final_score
    
    print(f"\nFinal: n={len(best_seq)}, score={best_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    # Final evaluation
    final_eval = evaluate_sequence([float(x) for x in best_seq])
    print(f"Final evaluate_sequence: {final_eval:.8f}")

    return [float(x) for x in best_seq]
