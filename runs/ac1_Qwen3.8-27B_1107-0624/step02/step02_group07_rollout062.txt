```python
def softmax_gradient(s, t):
    n = len(s)
    conv = np.convolve(s, s)
    tc = t * conv
    tc -= tc.max()
    exp_tc = np.exp(tc)
    w = exp_tc / exp_tc.sum()
    
    # Compute g[m] = 2*t * sum_k w[k]*s[k-m] for m=0..n-1
    # This is the cross-correlation of w and s
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    W = np.fft.rfft(w, N_fft)
    S = np.fft.rfft(s, N_fft)
    # Cross-correlation: ifft(conj(W) * S)
    c = np.fft.irfft(np.conj(W) * S, N_fft)
    grad = 2.0 * t * c[0:n]
    return grad
```

```python
    s = s / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)
    
    eta = 0.5
    t = 5.0
    no_improve = 0
    patience = 200
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        
        # Choose gradient type based on iteration
        if it < 500:
            # Phase 1: L2 optimization
            grad = grad_l2(s)
        else:
            # Phase 2: Soft-max optimization
            grad = grad_softmax(s, t)
        
        grad = np.maximum(grad, 0)  # Gradient should be non-negative for s >= 0
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        
        # Multiplicative update
        s_new = s * np.exp(-eta * grad / grad_max)
        s_new = np.maximum(s_new, 1e-15)
        s_new = s_new / s_new.sum()
        
        score = fast_score(s_new)
        
        if score < best_score - 1e-12:
            best_score = score
            best_s = s_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        s = s_new
        
        # Increase temperature in phase 2
        if it >= 500:
            t = min(t * 1.005, 100.0)
        
        # Adaptive step size
        if no_improve == patience:
            eta = max(eta * 0.7, 0.01)
            no_improve = 0
        elif no_improve == patience * 3:
            # Perturbation restart
            s = s * (1 + 0.02 * rng.standard_normal(n))
            s = np.maximum(s, 1e-10)
            s = s / s.sum()
            eta = 0.5
            t = max(t * 0.5, 1.0)
            no_improve = 0
    
    return best_s, best_score

# === LP polishing ===

def lp_polish(s, deadline, rng):
    """LP-based polishing step."""
    n = len(s)
    if n > 512:
        return s, fast_score(s)  # LP too slow for large n
    
    s = s.copy()
    best_score = fast_score(s)
    t = 0.4
    no_lp_improve = 0
    
    while time.monotonic() < deadline and no_lp_improve < 10:
        n_local = len(s)
        conv = compute_conv(s)
        M = conv.max()
        
        # LP: maximize sum(b) s.t. conv(s, b)[k] <= M, b >= 0
        # conv(s, b)[k] = sum_j s[j] * b[k-j] (with valid indices)
        # This is a linear constraint in b for fixed s.
        A = np.zeros((2 * n_local - 1, n_local))
        for j in range(n_local):
            A[j:j + n_local, j] = s
        c = -np.ones(n_local)
        b_ub = np.full(2 * n_local - 1, M)
        bounds = [(0.0, None)] * n_local
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': 2.0, 'presolve': True})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0.0)
                if b.sum() > 1e-10:
                    b = b / b.sum()
                    # Line search
                    best_t = 0.0
                    best_sc = best_score
                    for tt in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                        cand = (1 - tt) * s + tt * b
                        cand = np.maximum(cand, 0)
                        cand = cand / cand.sum()
                        sc = fast_score(cand)
                        if sc < best_sc:
                            best_sc = sc
                            best_t = tt
                    if best_t > 0:
                        s = (1 - best_t) * s + best_t * b
                        s = np.maximum(s, 0) / s.sum()
                        if best_sc < best_score:
                            best_score = best_sc
                            no_lp_improve = 0
                        else:
                            no_lp_improve += 1
                    else:
                        no_lp_improve += 1
                else:
                    no_lp_improve += 1
            else:
                no_lp_improve += 1
        except Exception:
            no_lp_improve += 1
    
    return s, best_score

def init_flat_spectrum(n, rng):
    """Initialize with flat (random phase) spectrum."""
    N = 1
    while N < 2 * n - 1:
        N *= 2
    # Desired magnitudes
    mags = np.full(N // 2 + 1, 1.0 / np.sqrt(n - 1))
    mags[0] = 1.0  # DC
    # Random phases
    phases = rng.uniform(0, 2 * np.pi, size=mags.shape)
    phases[0] = 0
    # For real signal, need conjugate symmetry
    S = mags * np.exp(1j * phases)
    # Make it conjugate symmetric
    for k in range(1, (N // 2) + 1):
        if k <= N // 2:
            S[N - k] = np.conj(S[k])  # This doesn't work for rfft
    # Simpler: use rfft approach
    S_r = mags * np.exp(1j * phases)
    s = np.fft.irfft(S_r, N)[:n]
    s = np.maximum(s, 0.01)
    s = s / s.sum()
    return s

def init_difference_set(n, rng):
    """Initialize based on a difference set pattern (if available for this n)."""
    s = np.ones(n) / n
    # For n=7, use the (7,3,1) difference set
    # For other n, use a pseudo-random pattern
    if n == 7:
        s = np.array([1, 1, 0, 1, 0, 0, 0], dtype=float)
        s = s + 0.1  # Make all positive
        s = s / s.sum()
    else:
        # Use a "spread out" pattern
        k = max(3, n // 3)
        indices = np.sort(rng.choice(n, size=k, replace=False))
        s = np.zeros(n)
        s[indices] = 0.5
        s[:] += 0.05  # Small baseline
        s = s / s.sum()
    return s
```
