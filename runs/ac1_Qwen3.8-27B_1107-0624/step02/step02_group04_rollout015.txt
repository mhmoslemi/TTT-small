```python
"""
Optimized subgradient method for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-k subgradient averaging, adaptive step sizes, and multi-phase strategy.
"""
import numpy as np
import time

def _convolve_fft(a):
    n = a.size
    if n <= 64:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    c = np.maximum(c, 0)
    return c

def _score(a):
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _convolve_fft(a)
    return 2.0 * n * c.max() / (s*s)

def _topk_avg_subgrad(c, a, k):
    """Compute the average subgradient of max(conv) using top-k maximizers."""
    n = a.size
    # Find top-k positions in convolution
    if k >= c.size:
        k = c.size
    topk_idx = np.argpartition(c, -k)[-k:]
    # For each topk position m, the subgradient w.r.t. a_i is a[m-i] (if valid)
    g = np.zeros(n)
    for m in topk_idx:
        # g_i += a[m-i] for i where 0 <= m-i < n
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        g[lo:hi+1] += a[m-hi:m-lo+1][::-1]
    g /= k
    return g
```

```python
        sc = _score(p)
        if np.isfinite(sc):
            candidates.append((sc, p, label))
            update_best(p, sc, f"init {label}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: best={best_score:.10f}, {len(candidates)} candidates")

    # Phase 1: Projected gradient with BB (300s)
    phase1_end = min(deadline, t0 + 300.0)
    print(f"Phase 1: Projected BB, deadline={phase1_end - t0:.0f}s")
    top_cands = candidates[:4]
    per_cand = (phase1_end - time.monotonic()) / max(1, len(top_cands))
    for sc, p, label in top_cands:
        if time.monotonic() > phase1_end:
            break
        per_end = min(phase1_end, time.monotonic() + per_cand)
        p_best, sc_best = _opt_projected(p, per_end, rng, eta0=0.05, k_init=64,
                                         use_nesterov=False, use_bb=True, name=f"proj_{label}")
        update_best(p_best, sc_best, f"projBB {label}")

    print(f"Phase 1: best={best_score:.10f}")

    # Phase 2: Mirror descent (300s)
    phase2_end = min(deadline, t0 + 600.0)
    print(f"Phase 2: Mirror descent, deadline={phase2_end - t0:.0f}s")
    if best_seq is not None:
        a_start = best_seq.copy()
        # Multiple mirror descent runs from best with different params
        for i, (eta0, k_init) in enumerate([(0.3, 32), (0.5, 16), (0.8, 8)]):
            if time.monotonic() > phase2_end:
                break
            per_end = min(phase2_end, time.monotonic() + 80.0)
            if i > 0:
                a_start = best_seq + 0.02 * rng.normal(size=n_target)
                a_start = _project_simplex(a_start)
            p_best, sc_best = _opt_mirror(a_start, per_end, rng, eta0=eta0, k_init=k_init,
                                          name=f"mirror{i}")
            update_best(p_best, sc_best, f"mirror{i}")

    print(f"Phase 2: best={best_score:.10f}")

    # Phase 3: Nesterov accelerated (300s)
    phase3_end = min(deadline, t0 + 900.0)
    print(f"Phase 3: Nesterov, deadline={phase3_end - t0:.0f}s")
    if best_seq is not None:
        a_start = best_seq.copy()
        for i, eta0 in enumerate([0.02, 0.01, 0.03]):
            if time.monotonic() > phase3_end:
                break
            per_end = min(phase3_end, time.monotonic() + 90.0)
            if i > 0:
                a_start = best_seq + 0.01 * rng.normal(size=n_target)
                a_start = _project_simplex(a_start)
            p_best, sc_best = _opt_nesterov(a_start, per_end, rng, eta0=eta0, k_init=16,
                                            name=f"nest{i}")
            update_best(p_best, sc_best, f"nesterov{i}")

    print(f"Phase 3: best={best_score:.10f}")

    # Phase 4: Multi-restart from best (300s)
    phase4_end = min(deadline, t0 + 1200.0)
    print(f"Phase 4: Multi-restart, deadline={phase4_end - t0:.0f}s")
    if best_seq is not None:
        n_restarts = 6
        per_restart = (phase4_end - time.monotonic()) / max(1, n_restarts)
        for i in range(n_restarts):
            if time.monotonic() > phase4_end:
                break
            per_end = min(phase4_end, time.monotonic() + per_restart)
            noise_scale = 0.03 + 0.03 * i
            a_start = best_seq + noise_scale * rng.normal(size=n_target)
            a_start = _project_simplex(a_start)
            # Alternate between methods
            if i % 3 == 0:
                p_best, sc_best = _opt_projected(a_start, per_end, rng, eta0=0.03, k_init=32,
                                                 use_nesterov=False, use_bb=True, name=f"r{i}")
            elif i % 3 == 1:
                p_best, sc_best = _opt_mirror(a_start, per_end, rng, eta0=0.4, k_init=16,
                                              name=f"r{i}")
            else:
                p_best, sc_best = _opt_nesterov(a_start, per_end, rng, eta0=0.015, k_init=8,
                                                name=f"r{i}")
            update_best(p_best, sc_best, f"restart{i}")

    print(f"Phase 4: best={best_score:.10f}")

    # Phase 5: Final fine polish (remaining time)
    phase5_end = deadline
    print(f"Phase 5: Final polish, deadline={phase5_end - t0:.0f}s")
    if best_seq is not None:
        # Very small step projected gradient
        per_end_1 = min(phase5_end, time.monotonic() + 120.0)
        p_best, sc_best = _opt_projected(best_seq, per_end_1, rng, eta0=0.01, k_init=8,
                                         k_min=1, use_nesterov=False, use_bb=True,
                                         name="final1")
        update_best(p_best, sc_best, "final_proj")

        if time.monotonic() < phase5_end - 30:
            per_end_2 = min(phase5_end, time.monotonic() + 120.0)
            p_best, sc_best = _opt_nesterov(best_seq, per_end_2, rng, eta0=0.005, k_init=4,
                                            k_min=1, name="final2")
            update_best(p_best, sc_best, "final_nest")

        # Try mirror descent with very small eta
        if time.monotonic() < phase5_end - 20:
            per_end_3 = min(phase5_end, time.monotonic() + 60.0)
            p_best, sc_best = _opt_mirror(best_seq, per_end_3, rng, eta0=0.1, k_init=4,
                                          k_min=1, name="final3")
            update_best(p_best, sc_best, "final_mirror")

    print(f"Phase 5: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final = best_seq.copy()
        final = np.clip(final, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_ramp(n_target, 2.0)

        # Exact score check
        sc_exact = _score(final)
        print(f"Final score (internal): {sc_exact:.10f}")

        # Try evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(sc_eval):
                print(f"Final score (evaluator): {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Evaluator error: {e}")

        print(f"FINAL: score={best_score:.10f}, n={final.size}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_ramp(n_target, 2.0)
    print(f"FALLBACK: n={n_target}")
    return [float(x) for x in fallback]
```
