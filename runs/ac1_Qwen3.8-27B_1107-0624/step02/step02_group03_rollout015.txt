    # For each peak, compute the gradient contribution

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def adam_polish(p0, deadline, lr_init=0.05, lr_min=0.005, verbose=True):
    """
    Adam gradient descent in logit space with cosine decay and restarts.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    beta = max(1e-12, 0.15 * max_c)
    total_iters_planned = 100000

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Softmax weights for differentiating the max
        x = (c - max_c) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Compute h[i] = sum_x w[x] * p[x-i] (correlation)
        # Efficient computation using FFT for large n
        if n > 256:
            # Use FFT for efficiency
            w_padded = np.zeros(2 * n - 1)
            w_padded[:len(w)] = w
            p_padded = np.zeros(2 * n - 1)
            p_padded[:n] = p0
            # corr[i] = sum_k w[k] * p[k-i] = (w * p)[i] in correlation sense
            # This is the cross-correlation
            h = np.zeros(n)
            # h[i] = sum_k w[k] * p[k-i] for valid k-i
            # = sum_{j} w[j+i] * p[j]
            for i in range(n):
                h[i] = np.dot(w[i:i + n], p0)
        else:
            h = np.zeros(n)
            for i in range(n):
                h[i] = np.dot(w[i:i + n], p0)

        grad_p = 2.0 * h + 1e-10 * p0
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            no_imp = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        # Cosine decay learning rate
        progress = min(1.0, it / total_iters_planned)
        lr = lr_min + 0.5 * (lr_init - lr_min) * (1.0 + np.cos(np.pi * progress))

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 500 == 0:
            beta = max(1e-14, beta * 0.7)

        if no_imp > 1500:
            # Restart with perturbation
            rng = np.random.default_rng(1000 + it)
            z = 0.7 * best_z + 0.5 * rng.normal(size=n) * max_c
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            no_imp = 0
            beta = max(1e-12, 0.15 * float(np.convolve(
                np.exp(z - z.max()), np.exp(z - z.max())).max()))

        p0 = p

        if verbose and it % 5000 == 0:
            print(f"    Adam iter {it}: score={best_score:.8f}, lr={lr:.6f}, beta={beta:.2e}")

    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    alpha = 0.05  # temperature for peak selection

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Focused weights: only top peaks with sharp softmax
        x = (c - max_c) / (alpha * max_c + 1e-15)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Gradient
        h = np.zeros(n)
        for i in range(n):
            h[i] = np.dot(w[i:i + n], p0)

        grad_p = 2.0 * h + 1e-10 * p0
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        lr = lr_init * (1.0 / (1.0 + 0.001 * it))
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp % 1000 == 0:
            alpha = max(0.001, alpha * 0.8)

        if no_imp > 2000:
            rng = np.random.default_rng(2000 + it)
            z = 0.5 * best_z + 0.7 * rng.normal(size=n) * max_c
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            no_imp = 0
            alpha = 0.05

        p0 = p

        if verbose and it % 5000 == 0:
            print(f"    Focused iter {it}: score={best_score:.8f}, alpha={alpha:.4f}")

    return best_p, best_score


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Improved LP at n=512 (line search, multiple M)
    Phase 2: Interpolate to n=1024, brief LP
    Phase 3: Adam polish (standard + focused)
    Phase 4: Multi-restart and final refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 10.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent.copy()
                best_score = internal_score(parent)
                print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 512
        profiles = [
            make_linear_profile(n_init, 1.5),
            make_linear_profile(n_init, 2.0),
            make_linear_profile(n_init, 1.0),
        ]
        for p in profiles:
            sc = internal_score(p)
            if sc < best_score:
                best_score = sc
                best_seq = p
        print(f"Generated initial: n={len(best_seq)}, score={best_score:.8f}")

    # === Phase 1: Improved LP at n=512 ===
    n1 = 512
    deadline_p1 = min(deadline, t_start + 350.0)
    if time.monotonic() < deadline_p1 - 10:
        a1 = interpolate_seq(best_seq, n1)
        if a1 is not None:
            print(f"\n=== Phase 1: LP at n={n1}, score={internal_score(a1):.8f} ===")
            # Time one LP solve
            t0 = time.monotonic()
            M_init = float(np.convolve(a1, a1).max())
            _ = lp_direction(a1, M_init, timeout=5)
            t_lp = time.monotonic() - t0
            print(f"  LP time: {t_lp*1000:.1f}ms")

            f_best, sc_best = lp_iterate_improved(
                a1, max_iters=2000, deadline=deadline_p1, verbose=True)
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  Phase 1 result: score={best_score:.8f}")
            else:
                print(f"  Phase 1: no improvement, keeping score={best_score:.8f}")

    # === Phase 2: Interpolate to n=1024 ===
    n2 = 1024
    deadline_p2 = min(deadline, t_start + 550.0)
    if time.monotonic() < deadline_p2 - 10:
        a2 = interpolate_seq(best_seq, n2)
        if a2 is not None:
            sc2 = internal_score(a2)
            print(f"\n=== Phase 2: n={n2}, interpolated score={sc2:.8f} ===")
            if sc2 < best_score:
                best_score = sc2
                best_seq = a2
                print(f"  Interpolation improved to {best_score:.8f}")

            # Brief LP at n=1024 (limited)
            if time.monotonic() < deadline_p2 - 30:
                t0 = time.monotonic()
                M_init = float(np.convolve(a2, a2).max())
                _ = lp_direction(a2, M_init, timeout=15)
                t_lp2 = time.monotonic() - t0
                print(f"  LP time at n=1024: {t_lp2*1000:.1f}ms")
                if t_lp2 < 20:  # Only if fast enough
                    f_best, sc_best = lp_iterate_improved(
                        a2, max_iters=50, deadline=deadline_p2, verbose=True)
                    if f_best is not None and sc_best < best_score:
                        best_score = sc_best
                        best_seq = f_best
                        print(f"  Phase 2 LP result: score={best_score:.8f}")

    # === Phase 3: Adam polish ===
    # Decide which n to use for Adam
    # Try both n=512 and n=1024, keep the best
    n_adam_1 = 512
    n_adam_2 = 1024

    # First Adam pass at n=512
    deadline_p3a = min(deadline, t_start + 1100.0)
    if time.monotonic() < deadline_p3a - 10:
        p1 = interpolate_seq(best_seq, n_adam_1)
        if p1 is not None:
            print(f"\n=== Phase 3a: Adam at n={n_adam_1}, score={internal_score(p1):.8f} ===")
            p_best, sc_best = adam_polish(p1, deadline_p3a, lr_init=0.05, lr_min=0.003, verbose=True)
            if sc_best < best_score:
                best_score = sc_best
                best_seq = p_best
                print(f"  Phase 3a result: score={best_score:.8f}")

    # Second Adam pass at n=1024
    deadline_p3b = min(deadline, t_start + 1500.0)
    if time.monotonic() < deadline_p3b - 10:
        p2 = interpolate_seq(best_seq, n_adam_2)
        if p2 is not None:
            print(f"\n=== Phase 3b: Adam at n={n_adam_2}, score={internal_score(p2):.8f} ===")
            p_best, sc_best = adam_polish(p2, deadline_p3b, lr_init=0.04, lr_min=0.002, verbose=True)
            if sc_best < best_score:
                best_score = sc_best
                best_seq = p_best
                print(f"  Phase 3b result: score={best_score:.8f}")

    # === Phase 4: Focused Adam + multi-restart ===
    deadline_p4 = min(deadline, t_start + 1750.0)
    if time.monotonic() < deadline_p4 - 10:
        n4 = len(best_seq)
        print(f"\n=== Phase 4: Focused Adam at n={n4}, score={internal_score(best_seq):.8f} ===")

        # Try focused Adam
        sub_deadline = min(deadline_p4, time.monotonic() + 150)
        if time.monotonic() < sub_deadline - 5:
            p_best, sc_best = adam_polish_focused(
                best_seq, sub_deadline, lr_init=0.015, verbose=True)
            if sc_best < best_score:
                best_score = sc_best
                best_seq = p_best
                print(f"  Focused result: score={best_score:.8f}")

        # Multi-restart: try a few perturbations
        if time.monotonic() < deadline_p4 - 30:
            print(f"\n  Multi-restart phase...")
            for restart in range(3):
                sub_dl = min(deadline_p4, time.monotonic() + 80)
                if time.monotonic() >= sub_dl - 5:
                    break
                rng_r = np.random.default_rng(seed + 10000 * (restart + 1))
                perturbation = rng_r.normal(size=len(best_seq)) * 0.02 * best_seq.sum() / len(best_seq)
                p_pert = normalize(best_seq + perturbation)
                if p_pert is None:
                    continue
                p_best, sc_best = adam_polish(
                    p_pert, sub_dl, lr_init=0.03, lr_min=0.002, verbose=False)
                if sc_best < best_score:
                    best_score = sc_best
                    best_seq = p_best
                    print(f"  Restart {restart+1} improved to: {best_score:.8f}")

    # === Phase 5: Final validation ===
    print(f"\n=== Final ===")
    final_score = internal_score(best_seq)
    print(f"  Internal score: {final_score:.8f}, n={len(best_seq)}")

    # Validate with evaluate_sequence
    try:
        test_list = [float(x) for x in best_seq]
        eval_score = evaluate_sequence(test_list)
        print(f"  Evaluator score: {eval_score:.8f}")
        if eval_score < np.inf and eval_score < final_score + 0.1:
            final_score = eval_score
    except Exception as e:
        print(f"  Evaluation error: {e}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(512, 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    print(f"  Final: n={len(best_seq)}, score={internal_score(best_seq):.8f}")
    return [float(x) for x in best_seq]
