    # = grad_lse_a[i] * a[i] - a[i] * sum_j grad_lse_a[j] * a[j]
    
    grad_lse_a = grad_a / beta
    dot = np.dot(grad_lse_a, a)
    grad_z = grad_lse_a * a - a * dot
    
    return obj, grad_z

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP direction finding with line search, smooth L-BFGS in log-space,
and coordinate descent polish, with multi-resolution strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

# --- Helper functions ---

def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] approximates convolve(f,g)[x]."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=15.0):
    """Solve LP: max sum(g) s.t. conv(f,g)[x] <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def score_at_t(f, g, t):
    """Score of (1-t)*f + t*g where both have sum 1."""
    a = (1.0 - t) * f + t * g
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    a = a / s
    c = np.convolve(a, a)
    return 2.0 * len(a) * c.max()


def line_search_direction(f, g, max_iter=80):
    """Golden section search for optimal t in [0,1] along direction g from f."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = score_at_t(f, g, c)
    fd = score_at_t(f, g, d)
    for _ in range(max_iter):
        if b - a < 1e-7:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = score_at_t(f, g, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = score_at_t(f, g, d)
    return (a + b) / 2.0


def lp_with_linesearch(f_init, deadline, max_iters=300, verbose=True):
    """LP direction finding with golden section line search."""
    n = len(f_init)
    f = normalize(f_init.astype(np.float64).copy())
    if f is None:
        return f_init, internal_score(f_init)

    M = float(np.convolve(f, f).max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0
    prev_dir = None

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=20.0)
        if g_raw is None:
            break
        g = normalize(g_raw)
        if g is None:
            break

        # Try blending with previous direction for diversity
        directions = [g]
        if prev_dir is not None:
            g_blend = 0.4 * g + 0.6 * prev_dir
            g_blend_n = normalize(g_blend)
            if g_blend_n is not None:
                directions.append(g_blend_n)

        best_t = 0.0
        best_sc = 2.0 * n * M
        for d in directions:
            t_opt = line_search_direction(f, d)
            sc = score_at_t(f, d, t_opt)
            if sc < best_sc:
                best_sc = sc
                best_t = t_opt
                best_dir = d

        if best_sc < best_score - 1e-14:
            f_new = normalize((1.0 - best_t) * f + best_t * best_dir)
            if f_new is not None:
                f = f_new
                M = float(np.convolve(f, f).max())
                best_f = f.copy()
                best_score = 2.0 * n * M
                no_improve = 0
            prev_dir = best_dir
        else:
            no_improve += 1
            if no_improve > 30:
                break

        if verbose and it % 30 == 0:
            print(f"    LP-ls iter {it}: score={best_score:.8f}, M={best_M:.8f}, t={best_t:.4f}")
            best_M = M

    return best_f, best_score


def smooth_obj_grad(z, beta):
    """Smooth objective and gradient in log-space (softmax parameterization)."""
    n = len(z)
    a = softmax(z)
    c = np.convolve(a, a)
    max_c = float(c.max())
    x = (c - max_c) / beta
    x_shifted = x - x.max()
    log_sum = x.max() + np.log(np.exp(x_shifted).sum())
    lse = max_c + beta * log_sum
    obj = 2.0 * n * lse

    # Softmax weights over convolution
    w = np.exp(x_shifted)
    w = w / w.sum()

    # Gradient of lse w.r.t. a:
    # grad_a[i] = (1/beta) * sum_k w[k] * d c[k]/d a[i]
    # d c[k]/d a[i] = a[k-i] (from j=i) + a[k-i] (from j=k-i) = 2a[k-i] if k!=2i
    #                = a[k-i] if k=2i
    # So grad_a[i] = (2/beta) * h[i] - (1/beta) * w[2i] * a[i] (if 2i < 2n-1)
    # where h[i] = sum_k w[k] * a[k-i] (valid k only)

    h = np.zeros(n)
    # h[i] = sum_{k: 0<=k-i<n, 0<=k<2n-1} w[k] * a[k-i]
    for i in range(n):
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        if k_lo <= k_hi:
            h[i] = np.dot(w[k_lo:k_hi + 1], a[k_lo - i:k_hi - i + 1])

    grad_a = (2.0 / beta) * h
    # Subtract diagonal: where k = 2i
    for i in range(n):
        k = 2 * i
        if k < 2 * n - 1:
            grad_a[i] -= (1.0 / beta) * w[k] * a[i]

    # Chain rule through softmax:
    # d obj/d z[i] = sum_j (d obj/d a[j]) * a[j] * (delta_{ji} - a[i])
    # = (d obj/d a[i]) * a[i] - a[i] * sum_j (d obj/d a[j]) * a[j]

    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)

    return obj, grad_z


def smooth_lbgfs(f_init, deadline, verbose=True):
    """L-BFGS optimization in log-space with decreasing temperature."""
    n = len(f_init)
    f = normalize(f_init.astype(np.float64).copy())
    if f is None:
        return f_init, internal_score(f_init)

    best_f = f.copy()
    best_score = 2.0 * n * float(np.convolve(f, f).max())

    z = np.log(np.maximum(f, 1e-15))
    z -= z.mean()

    # Anneal temperature
    beta_start = 0.05 * float(np.convolve(f, f).max())
    beta_min = 1e-6
    betas = [beta_start, beta_start * 0.5, beta_start * 0.25, beta_start * 0.1,
             beta_start * 0.05, beta_start * 0.02, beta_min]
    betas = [max(b, beta_min) for b in betas]

    for beta in betas:
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline - 30:
            break

        phase_deadline = min(deadline - 10, time.monotonic() + 180)

        def fun_wrapper(z_vec):
            obj, _ = smooth_obj_grad(z_vec, beta)
            return obj, _

        res = minimize(fun_wrapper, z, jac=True, method='L-BFGS-B',
                       options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10,
                                'maxfun': 1000})
        if res.x is not None:
            a_new = softmax(res.x)
            sc = 2.0 * n * float(np.convolve(a_new, a_new).max())
            if sc < best_score - 1e-14:
                best_score = sc
                best_f = a_new.copy()
                z = res.x.copy()
            elif res.x is not None:
                z = res.x.copy()

        if verbose:
            print(f"    L-BFGS beta={beta:.6f}: score={best_score:.8f}")

    return best_f, best_score


def coordinate_descent(a_init, deadline, max_sweeps=500, verbose=True):
    """Coordinate descent: optimize one element at a time."""
    n = len(a_init)
    a = normalize(a_init.astype(np.float64).copy())
    if a is None:
        return a_init

    best_a = a.copy()
    best_score = 2.0 * n * float(np.convolve(a, a).max())
    no_improve_sweeps = 0

    factors = np.array([0.2, 0.4, 0.6, 0.75, 0.9, 1.1, 1.3, 1.6, 2.0, 3.0, 5.0])

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        sweep_improved = False
        for k in range(n):
            if time.monotonic() > deadline:
                break
            a_k = a[k]
            if a_k < 1e-15:
                # Try introducing a small value
                for trial in [1e-6, 1e-5, 1e-4]:
                    a[k] = trial
                    a_n = a / a.sum()
                    sc = 2.0 * n * float(np.convolve(a_n, a_n).max())
                    if sc < best_score:
                        a[k] = trial
                        a = a / a.sum()
                        sweep_improved = True
                continue

            cur_sc = 2.0 * n * float(np.convolve(a, a).max())
            best_local = cur_sc
            best_val = a_k

            for fac in factors:
                a[k] = a_k * fac
                s = a.sum()
                if s < 1e-10:
                    a[k] = a_k
                    continue
                a_n = a / s
                sc = 2.0 * n * float(np.convolve(a_n, a_n).max())
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = a_k * fac

            if best_val != a_k:
                a[k] = best_val
                a = a / a.sum()
                sweep_improved = True

        if sweep_improved:
            cur_sc = 2.0 * n * float(np.convolve(a, a).max())
            if cur_sc < best_score - 1e-14:
                best_score = cur_sc
                best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > 5:
                break

        if verbose and sweep % 50 == 0:
            print(f"    CD sweep {sweep}: score={best_score:.8f}")

    return best_a


def interpolate_seq(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_initial_profile(n, eta=1.5):
    """Create a linear profile for initialization."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def perturbed_restart(a, sigma, rng):
    """Add multiplicative log-normal perturbation."""
    z = np.log(np.maximum(a, 1e-15))
    z += sigma * rng.normal(size=len(z))
    a_new = softmax(z)
    return a_new
