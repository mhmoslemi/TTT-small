"""
Soft-max gradient descent with Adam for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses softmax parameterization, FFT-based gradient computation, and multi-start
strategy with beta annealing.
"""

import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def softmax_from_z(z):
    """Compute softmax from logits z."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / z.size
    return e / s


def compute_gradient(z, a, c, w):
    """
    Compute gradient of soft-max(conv(a,a)) with respect to z.
    a = softmax(z), c = conv(a,a), w = softmax(c/beta) weights.
    
    grad_z[l] = a[l] * (h[l] - 2 * L)
    where h[l] = sum_k w[k] * (a[k-l] + a[l-k])
    and L = sum_k w[k] * c[k]
    """
    n = a.size
    # Compute h[l] = sum_k w[k] * (a[k-l] + a[l-k])
    # f2[l] = (w * a)[l] = sum_k w[k] * a[l-k]
    # f1[l] = (w_rev * a)[2n-2-l] where w_rev[k] = w[2n-2-k]
    
    N_corr = next_pow2((2 * n - 1) + n - 1)  # >= 3n-2
    
    fw = np.fft.rfft(w, N_corr)
    fa = np.fft.rfft(a, N_corr)
    
    # f2 = IFFT(fw * fa) gives (w * a)[l] for l = 0..3n-3
    f2_full = np.fft.irfft(fw * fa, N_corr)
    f2 = f2_full[:n]
    
    # f1[l] = (w_rev * a)[2n-2-l]
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, N_corr)
    f1_full = np.fft.irfft(fw_rev * fa, N_corr)
    # f1[l] = f1_full[2n-2-l] for l = 0..n-1
    idx = 2 * n - 2 - np.arange(n)
    f1 = f1_full[idx]
    
    h = f1 + f2
    
    L = float(np.dot(w, c))
    
    grad_z = a * (h - 2.0 * L)
    
    return grad_z


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=8):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.6) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_deterministic_profiles(n):
    profs = []
    for eta in (0.5, 0.8, 1.0, 1.2, 1.5, 1.8, 2.0, 2.5):
        profs.append(make_linear_profile(n, eta))
    for a2 in (-0.3, -0.5, -0.8, -1.0, 0.3, 0.5, 0.8, 1.0):
        profs.append(make_quadratic_profile(n, a2))
    # Parabolic: high at edges, low in middle
    t = np.linspace(0, 1, n)
    a = 1.0 + 2.0 * (2*t - 1)**2
    profs.append(a / a.sum())
    # Anti-parabolic: low at edges, high in middle
    a = 1.0 - 0.8 * (2*t - 1)**2
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    return profs


def softmax_gradient_optimize(a0, deadline, rng, lr=0.05, beta_init=None, 
                               beta_min=1e-10, beta_decay=0.95, decay_every=200,
                               max_iter=100000, b1=0.9, b2=0.999, eps=1e-8,
                               restart_after=500):
    """
    Optimize using soft-max gradient in logit space with Adam.
    a0: initial sequence (will be normalized).
    """
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = a.sum()
    if s < 1e-30:
        return a0, np.inf
    a = a / s
    n = a.size
    
    z = np.log(a)
    z -= z.mean()
    
    if beta_init is None:
        c_init = fft_convolve(a)
        beta = 0.1 / max(c_init.max(), 1e-30)
    else:
        beta = beta_init
    
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    
    c = fft_convolve(a)
    M_val = float(c.max())
    
    best_a = a.copy()
    best_M = M_val
    no_imp = 0
    
    N_corr = next_pow2(3 * n - 1)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute soft-max weights
        x = (c - M_val) / beta
        x = np.clip(x, -200.0, 10.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex
        
        # Compute gradient
        grad_z = compute_gradient(z, a, c, w)
        
        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_a)
            z -= z.mean()
            a = best_a.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            continue
        
        # Adam update
        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * grad_z
        v_adam = b2 * v_adam + (1.0 - b2) * (grad_z * grad_z)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)
        
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -100.0, 100.0)
        z_new -= z_new.mean()
        
        a_new = softmax_from_z(z_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        c_new = fft_convolve(a_new)
        M_new = float(c_new.max())
        
        z = z_new
        a = a_new
        c = c_new
        M_val = M_new
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Beta decay
        if t_adam % decay_every == 0:
            beta = max(beta_min, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > restart_after:
            # Mix with random perturbation
            z_pert = 0.7 * np.log(best_a) + 0.3 * z + 0.5 * rng.normal(size=n)
            z_pert = np.clip(z_pert, -100.0, 100.0)
            z_pert -= z_pert.mean()
            a = softmax_from_z(z_pert)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            z = np.log(a)
            z -= z.mean()
            c = fft_convolve(a)
            M_val = float(c.max())
            beta = max(beta_min, 0.1 / max(M_val, 1e-30))
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            no_imp = 0
    
    return best_a, best_M


def lp_optimize_fast(f0, deadline, rng, max_iter=10000):
    """
    LP-based direction finding (faster version with time limit on LP).
    """
    from scipy.optimize import linprog
    
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size
    
    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    # Precompute the A matrix structure
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f  # This will need updating
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Update A matrix (since f changes)
        A[:] = 0
        for j in range(n):
            A[j:j + n, j] = f
        
        c_obj = -np.ones(n)
        b_vec = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        
        if not res.success:
            no_imp += 1
            if no_imp > 30:
                break
            continue
        
        g = res.x
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 100:
                break
            continue
        
        g = g / g_sum
        
        improved = False
        for t in (0.3, 0.1, 0.5, 0.2, 0.15, 0.4, 0.05, 0.6):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(fft_convolve(f_new).max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break
        
        if not improved:
            no_imp += 1
            if no_imp > 50:
                f = best_f + 0.01 * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-30:
                    f = best_f.copy()
                else:
                    f = f / fs
                M = float(fft_convolve(f).max())
                no_imp = 0
    
    return best_f, best_M


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None
