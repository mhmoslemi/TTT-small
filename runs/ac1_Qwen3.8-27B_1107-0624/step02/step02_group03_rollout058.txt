"""
FFT-accelerated Adam optimization with animated smoothness parameter for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multi-resolution strategy (n=512, 1024, 2048) with multi-start and random perturbation.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def softmax(z):
    """Stable softmax."""
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(z)) / len(z)
    return a / s


def fft_conv(a, b=None):
    """FFT-based linear convolution."""
    if b is None:
        b = a
    n_out = len(a) + len(b) - 1
    N = 1
    while N < n_out:
        N *= 2
    fa = fft(a, N)
    fb = fft(b, N)
    c = ifft(fa * fb).real
    return c[:n_out]


def smooth_logsumexp(c, beta):
    """Smooth max approximation. Returns (value, softmax_weights)."""
    c_shifted = (c - c.max()) / beta
    ex = np.exp(c_shifted)
    s = ex.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return float(c.max()), w
    w = ex / s
    val = float(c.max() + beta * np.log(s))
    return val, w


def compute_grad_fft(w, a):
    """
    Compute grad[j] = 2 * sum_m w[m+j] * a[m] for j=0,...,n-1 using FFT.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N *= 2
    W = fft(w, N)
    A = fft(a, N)
    corr = ifft(W * np.conj(A)).real[:n]
    return 2.0 * corr


def gradient_logit(z, beta):
    """
    Compute gradient of 2*n*smooth_max(conv(a,a)) w.r.t. logit z,
    where a = softmax(z). Returns (value, grad_z).
    """
    n = len(z)
    a = softmax(z)
    c = fft_conv(a, a)
    val, w = smooth_logsumexp(c, beta)
    grad_a = compute_grad_fft(w, a)
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    return 2.0 * n * val, grad_z


def adam_optimize(p0, deadline, lr=0.02, beta_init=0.01, beta_decay=0.998,
                  beta_min=0.0002, verbose=True):
    """
    Adam optimization in logit space with FFT-accelerated gradient
    and animated smoothness parameter.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    z = np.log(p0)
    z -= z.mean()

    best_z = z.copy()
    best_score = 2.0 * n * float(np.convolve(p0, p0).max())
    best_p = p0.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0

    beta = beta_init
    no_improve = 0
    base_lr = lr

    while time.monotonic() < deadline:
        t += 1

        val, grad_z = gradient_logit(z, beta)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t = 0
            continue

        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * grad_z * grad_z
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        p = softmax(z)
        score = 2.0 * n * float(np.convolve(p, p).max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay beta (smoothness)
        beta = max(beta_min, beta * beta_decay)

        # Adaptive learning rate
        if no_improve > 2000:
            lr = max(0.005, lr * 0.8)

        # Restart if stuck for too long
        if no_improve > 8000:

            z = 0.85 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t = 0
            no_improve = 0
            lr = base_lr
            beta = max(beta_min, beta_init * 0.3)

        if verbose and t % 2000 == 0:
            print(f"    Adam it={t}: best={best_score:.8f}, beta={beta:.6f}, lr={lr:.5f}")

    return best_p, best_score


def upsample(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_gaussian_profile(n, sigma_frac=0.3):
    """Gaussian-like profile."""
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_linear_profile(n, eta):
    """Linear profile: 1 + eta*(2t-1)."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_polynomial_profile(n, coeffs):
    """Polynomial profile: sum_k coeffs[k] * (2t-1)^k."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.ones(n)
    for k, c in enumerate(coeffs, start=1):
        a += c * (u ** k)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def conjugate_gradient_optimize(p0, deadline, lr_init=0.1, max_cg_iters=50):
    """
    Fletcher-Reeves conjugate gradient in logit space with FFT gradient.
    Uses exact line search (golden section).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    z = np.log(p0)
    z -= z.mean()

    best_z = z.copy()
    best_score = 2.0 * n * float(np.convolve(p0, p0).max())
    best_p = p0.copy()

    beta = 0.005
    no_improve = 0
    it = 0
    d = None

    while time.monotonic() < deadline:
        it += 1

        val, grad = gradient_logit(z, beta)

        if d is None or np.dot(grad, d) > 0:
            d = -grad
            cg_count = 0
        else:
            beta_cg = np.dot(grad, grad) / max(np.dot(d_prev, d_prev), 1e-30)
            d = -grad + beta_cg * d
            cg_count += 1

        d_prev = d.copy()

        # Line search (backtracking)
        alpha = lr_init
        f_cur = val
        for _ in range(30):
            z_try = z + alpha * d
            z_try = np.clip(z_try, -50, 50)
            z_try -= z_try.mean()
            f_try, _ = gradient_logit(z_try, beta)
            if f_try < f_cur - 1e-12 * alpha * np.dot(grad, -d):
                break
            alpha *= 0.5

        z = z + alpha * d
        z = np.clip(z, -50, 50)
        z -= z.mean()

        p = softmax(z)
        score = 2.0 * n * float(np.convolve(p, p).max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if cg_count > max_cg_iters:
            d = None

        if no_improve > 100:
            beta = max(0.0002, beta * 0.7)
            if beta <= 0.0003:
                break

        if it % 50 == 0:
            print(f"    CG it={it}: best={best_score:.8f}, beta={beta:.6f}")

    return best_p, best_score
