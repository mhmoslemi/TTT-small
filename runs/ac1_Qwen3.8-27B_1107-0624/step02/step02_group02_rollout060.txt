    return np.maximum(c, 0)

def soft_max_and_weights(c, beta):
    m = c.max()
    x = (c - m) / beta
    ex = np.exp(x)
    s = ex.sum()
    w = ex / s
    return m + beta * np.log(s), w

def grad_a_space(a, w):
    """Compute dL/da[i] = 2 * sum_m a[m] * w[i+m]."""
    n = len(a)
    a_rev = a[::-1]
    N = next_pow2(n + len(w) - 1)
    fa = rfft(a_rev, N)
    fw = rfft(w, N)
    conv = irfft(fa * fw, N)
    # (a_rev * w)[n-1+i] = sum_m a[m]*w[m+i]
    result = conv[n-1 : n-1+n]
    return 2.0 * result

def logit_to_a(z):
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    if a_sum < 1e-300:
        return np.ones(len(z)) / len(z)
    return a / a_sum

def a_to_logit(a):
    a = np.maximum(a, 1e-15)
    z = np.log(a)
    z -= z.mean()
    return z

def obj_grad(z, beta, n):
    """Objective and gradient in logit space."""
    a = logit_to_a(z)
    c = compute_conv_fft(a)
    val, w = soft_max_and_weights(c, beta)
    obj = 2.0 * n * val
    
    if not np.isfinite(obj):
        return obj, np.zeros(n)
    
    ga = grad_a_space(a, w)
    mean_ga = np.dot(a, ga)
    gz = a * (ga - mean_ga)
    
    if not np.all(np.isfinite(gz)):
        return obj, np.zeros(n)
    
    return obj, gz

def lbfgs_stage(z, beta, n, max_iter=500, time_limit=None, t_start=None):
    """Single L-BFGS stage at fixed beta."""
    if t_start is None:
        t_start = time.monotonic()
    if time_limit is not None and time.monotonic() >= t_start + time_limit:
        return z
    
    n_params = n
    
    best_z = z.copy()
    best_obj = obj_grad(z, beta, n_params)[0]
    
    def fun(z_in):
        return obj_grad(z_in, beta, n_params)
    
    try:
        result = minimize(fun, z, method='L-BFGS-B', jac=True,
                         options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                  'maxls': 50, 'factr': 1e7, 'gtol': 1e-8})
        if result.success or result.x is not None:
            new_obj = obj_grad(result.x, beta, n_params)[0]
            if new_obj < best_obj:
                best_z = result.x
                best_obj = new_obj
    except Exception:
        pass
    
    return best_z

def full_optimize(z_init, deadline, verbose=True):
    """Multi-stage L-BFGS with beta annealing."""
    n = len(z_init)
    z = z_init.copy()
    t_start = time.monotonic()
    
    # Get initial max to set beta
    a = logit_to_a(z)
    c = compute_conv_fft(a)
    max_c = c.max()
    
    # Beta schedule: from smooth to sharp
    beta = max_c * 0.1
    best_obj = 2.0 * n * max_c
    best_z = z.copy()
    
    stage = 0
    while time.monotonic() < deadline:
        time_left = deadline - time.monotonic()
        if time_left < 2.0:
            break
        
        # Allocate time to this stage
        alloc = min(time_left, max(3.0, time_left / 5.0))
        
        new_z = lbfgs_stage(z, beta, n, max_iter=300, time_limit=alloc, t_start=t_start)
        new_obj = obj_grad(new_z, beta, n)[0]
        
        if new_obj < best_obj:
            best_obj = new_obj
            best_z = new_z
            z = new_z
        else:
            z = best_z
        
        stage += 1
        if verbose and stage % 3 == 0:
            exact_a = logit_to_a(z)
            exact_c = compute_conv_fft(exact_a)
            exact_score = 2.0 * n * exact_c.max()
            print(f"  Stage {stage}: beta={beta:.2e}, soft_obj={new_obj:.8f}, exact={exact_score:.8f}, best={best_obj:.8f}")
        
        # Anneal beta
        beta *= 0.5
        if beta < 1e-8:
            beta = 1e-8
    
    return best_z, best_obj

def adam_polish(a_init, deadline, lr=0.05, beta_init=None, verbose=True):
    """Adam polish in logit space."""
    n = len(a_init)
    z = a_to_logit(a_init)
    
    a = logit_to_a(z)
    c = compute_conv_fft(a)
    if beta_init is None:
        beta = 0.05 * c.max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_z = z.copy()
    best_score = np.inf
    no_improve = 0
    
    while time.monotonic() < deadline:
        obj, gz = obj_grad(z, beta, n)
        
        if np.isfinite(obj) and obj < best_score:
            best_score = obj
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        t += 1
        m = b1 * m + (1-b1) * gz
        v = b2 * v + (1-b2) * gz**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50, 50)
        z_new -= z_new.mean()
        
        if not np.all(np.isfinite(z_new)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t = 0
            continue
        
        z = z_new
        
        if t % 100 == 0:
            beta = max(beta * 0.85, 1e-10)
        
        if no_improve > 2000:

            z = best_z + 0.2 * np.random.default_rng(t).normal(size=n)

            z -= z.mean()
            beta = max(0.03 * compute_conv_fft(logit_to_a(z)).max(), 1e-8)
            m.fill(0); v.fill(0); t = 0
            no_improve = 0
    
    a = logit_to_a(best_z)
    c = compute_conv_fft(a)
    score = 2.0 * n * c.max()
    return a, score

Multi-phase L-BFGS and Adam optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

import numpy as np

def next_pow2(n):
    """Smallest power of 2 >= n, at least 2."""
    if n <= 2:
        return 2
    return 1 << (n - 1).bit_length()


def compute_conv_fft(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    if n <= 64:
        c = np.convolve(a, a)
    else:
        N = next_pow2(2 * n - 1)
        fa = rfft(a, N)
        c = irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def soft_max_and_weights(c, beta):
    """Compute soft max value and gradient weights."""
    m = c.max()
    x = (c - m) / beta
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = ex / s
    return m + beta * np.log(s), w


def grad_a_space(a, w):
    """
    Compute dL/da[i] = 2 * sum_m a[m] * w[i+m] for i=0..n-1.
    Uses FFT cross-correlation via convolution with reversed array.
    """
    n = len(a)
    a_rev = a[::-1]
    N = next_pow2(n + len(w) - 1)
    fa = rfft(a_rev, N)
    fw = rfft(w, N)
    conv = irfft(fa * fw, N)
    result = conv[n - 1:n - 1 + n]
    return 2.0 * result

    """Generate diverse smooth starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    # Linear profiles
    for eta in [1.0, 1.5, 2.0, 3.0, 4.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Cosine profiles
    for k in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Gaussian-like
    for sigma in [0.15, 0.2, 0.25, 0.35]:
        a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Quadratic
    for a_c in [-0.5, 0.0, 0.5, 1.0]:
        a = 1.0 + a_c * (t - 0.5) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Random smooth (Fourier basis)
    for _ in range(6):
        coeffs = rng.normal(0, 0.3, 8)
        a = np.ones(n)
        for i, c in enumerate(coeffs[1:], 1):
            a = a + c * np.cos(2.0 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles
