def lp_iterate(f_init, deadline, rng, max_iters=500, use_min_peak=False):
    """LP-based iterative optimization."""
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_max_conv(f_init)
    
    n = len(f)
    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        mode = 'min_peak' if use_min_peak else 'max_sum'
        g = lp_direction(f, M, mode=mode)
        
        if g is None:
            f = f + 0.05 * rng.normal(size=n)
            f = normalize_nonneg(f)
            if f is None:
                f = best_f.copy()
            M = compute_max_conv(f)
            continue
        
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Line search
        best_t = 0.0
        best_M_try = M
        for t in [0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            f_try = (1.0-t)*f + t*g
            f_try = normalize_nonneg(f_try)
            if f_try is None:
                continue
            M_try = compute_max_conv(f_try)
            if M_try < best_M_try - 1e-13:
                best_M_try = M_try
                best_t = t
            if t <= best_t:
                break
        
        if best_t > 0:
            f_try = (1.0-best_t)*f + best_t*g
            f_try = normalize_nonneg(f_try)
            if f_try is not None:
                f = f_try
                M = best_M_try
                if M < best_M - 1e-13:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                else:
                    no_imp += 1
        else:
            no_imp += 1
        
        if no_imp > 20:
            # Restart
            pert = rng.normal(size=n) * 0.2
            f = 0.8 * best_f + 0.2 * normalize_nonneg(best_f + pert)
            if f is None:
                f = best_f.copy()
            M = compute_max_conv(f)
            no_imp = 0
    
    return best_f, best_M

def compute_score(f):
    n = len(f)

    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))

    return float(np.convolve(f, f).max())

def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def make_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_direction_min_peak(f, time_limit=3.0):
    """Minimize t s.t. conv(f,g)[k] <= t, sum(g)=1, g>=0."""
    n = len(f)
    A = make_conv_matrix(f)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def soft_max_w(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def corr_weights_fft(w, p):
    n = len(p)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def adam_optimize(f_init, deadline, rng, lr=0.04, max_iters=10000,
                  beta_init_frac=0.3, beta_decay=0.85, beta_period=80,
                  restart_thresh=250):
    """
    Adam optimization in logit space with adaptive beta.
    Returns (best_f, best_M).
    """
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = np.convolve(f, f)
    M = float(c.max())
    beta = max(1e-10, beta_init_frac * M)

    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        w = soft_max_w(c, beta)
        gp = 2.0 * corr_weights_fft(w, f)
        vp = gp + 1e-8 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1**adam_t)
        vh = v / (1 - b2**adam_t)
        z = z - lr_cur * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -50, 50)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-13:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr_cur = max(0.005, lr_cur * 0.95)

        if adam_t % beta_period == 0:
            beta = max(1e-12, beta * beta_decay)

        if no_imp > restart_thresh:
            # Annealing restart: partially randomize
            z = 0.6 * best_z + 0.4 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, beta_init_frac * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr_cur = base_lr
            no_imp = 0

    return best_f, best_M


def lp_adam_interleave(f_init, deadline, rng, lr=0.03):
    """
    Interleave LP steps and Adam steps for combined exploration/exploitation.
    """
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_max_conv(f_init)

    n = len(f)
    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    z = np.log(np.maximum(f, 1e-15))
    z -= z.mean()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    beta = max(1e-10, 0.3 * M)
    lr_cur = lr

    for it in range(2000):
        if time.monotonic() > deadline:
            break

        # Every 10 iterations, do an LP step
        if it % 10 == 0 and M > 0:
            g = lp_direction_max_sum(f, M, time_limit=2.0)
            if g is not None:
                g = g / g.sum()
                # Small LP step
                t_lp = 0.3
                f_try = (1.0 - t_lp) * f + t_lp * g
                f_try = normalize_nonneg(f_try)
                if f_try is not None:
                    M_try = compute_max_conv(f_try)
                    if M_try < M - 1e-13:
                        f = f_try
                        M = M_try
                        z = np.log(np.maximum(f, 1e-15))
                        z -= z.mean()
                        if M < best_M - 1e-13:
                            best_M = M
                            best_f = f.copy()
                            no_imp = 0
                        else:
                            no_imp += 1
                        continue

        # Adam step
        c = np.convolve(f, f)
        w = soft_max_w(c, beta)
        gp = 2.0 * corr_weights_fft(w, f)
        vp = gp + 1e-8 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1**adam_t)
        vh = v / (1 - b2**adam_t)
        z = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-13:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        if adam_t % 80 == 0:
            beta = max(1e-12, beta * 0.85)

        if no_imp > 100:
            z = 0.7 * z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.3 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr_cur = lr
            no_imp = 0

    return best_f, best_M


def lp_iterate(f_init, deadline, rng, max_iters=300, use_min_peak=False):
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_max_conv(f_init)

    n = len(f)
    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        if use_min_peak:
            g = lp_direction_min_peak(f, time_limit=2.0)
        else:
            g = lp_direction_max_sum(f, M, time_limit=2.0)

        if g is None:
            f = f + 0.05 * rng.normal(size=n)
            f = normalize_nonneg(f)
            if f is None:
                f = best_f.copy()
            M = compute_max_conv(f)
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum

        best_t = 0.0
        best_M_try = M
        best_f_try = f
        for t in [0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02]:
            f_try = (1.0 - t) * f + t * g
            f_try = normalize_nonneg(f_try)
            if f_try is None:
                continue
            M_try = compute_max_conv(f_try)
            if M_try < best_M_try - 1e-13:
                best_M_try = M_try
                best_f_try = f_try
                best_t = t

        if best_t > 0:
            f = best_f_try
            M = best_M_try
            if M < best_M - 1e-13:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 15:
            pert = rng.normal(size=n) * 0.15
            f_new = 0.85 * best_f + 0.15 * normalize_nonneg(best_f + pert)
            if f_new is not None:
                f = f_new
            else:
                f = best_f.copy()
            M = compute_max_conv(f)
            no_imp = 0

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)

    # Try parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
                sc = compute_score(parent)
                try_update(parent, sc)
                print(f"Parent score: {sc:.10f}")
    except Exception:
        pass

    # Phase 1: Diverse Adam starts at n=256
    n1 = 256
    print(f"Phase 1: Adam optimization at n={n1}")
    phase1_end = t_start + 500.0

    starts = get_starting_points(n1, rng, count=12)
    if parent is not None and parent.size != n1:
        starts.insert(0, interpolate_profile(parent, n1))
    elif parent is not None:
        starts.insert(0, parent.copy())

    # Score and sort starts
    scored = []
    for s in starts:
        sc = compute_score(s)
        scored.append((sc, s))
        try_update(s, sc)
    scored.sort(key=lambda x: x[0])

    # Select top diverse set
    selected = [x[1] for x in scored[:12]]
    # Ensure diverse: add some from middle and end
    if len(scored) > 12:
        selected.append(scored[len(scored) // 2][1])
        selected.append(scored[-1][1])

    results = []
    for i, s in enumerate(selected):
        remaining = phase1_end - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining / max(1, len(selected) - i), 55.0)
        if alloc < 8:
            break
        dl = time.monotonic() + alloc

        f_best, M_best = adam_optimize(s, dl, rng, lr=0.04, max_iters=8000,
                                       beta_init_frac=0.3, beta_decay=0.85,
                                       beta_period=60, restart_thresh=200)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results.append((sc, f_best))
        print(f"  Adam {i + 1}/{len(selected)}: score={sc:.10f}, best={best_score:.10f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.10f}")

    # Phase 2: LP+Adam interleave on top results
    print("Phase 2: LP+Adam interleave")
    phase2_end = t_start + 850.0
    top_k = min(6, len(results))

    for i in range(top_k):
        remaining = phase2_end - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining / max(1, top_k - i), 50.0)
        if alloc < 8:
            break
        dl = time.monotonic() + alloc

        f_best, M_best = lp_adam_interleave(results[i][1], dl, rng, lr=0.03)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results.append((sc, f_best))
        print(f"  Interleave {i + 1}/{top_k}: score={sc:.10f}, best={best_score:.10f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best: {best_score:.10f}")

    # Phase 3: Pure LP on top results
    print("Phase 3: LP refinement")
    phase3_end = t_start + 1100.0
    top_k2 = min(4, len(results))

    for i in range(top_k2):
        remaining = phase3_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_k2 - i), 45.0)
        if alloc < 5:
            break
        dl = time.monotonic() + alloc

        # Try both LP modes
        f_best, M_best = lp_iterate(results[i][1], dl, rng, max_iters=400, use_min_peak=False)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results.append((sc, f_best))

        if time.monotonic() < dl:
            f_best2, M_best2 = lp_iterate(results[i][1], min(time.monotonic() + 20, dl),
                                           rng, max_iters=200, use_min_peak=True)
            sc2 = 2.0 * n1 * M_best2
            try_update(f_best2, sc2)
            results.append((sc2, f_best2))

        print(f"  LP {i + 1}/{top_k2}: best={best_score:.10f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 3 done. Best: {best_score:.10f}")

    # Phase 4: Final Adam polish with sharper beta
    print("Phase 4: Final Adam polish (sharp beta)")
    phase4_end = t_start + 1400.0
    top_k3 = min(5, len(results))

    for i in range(top_k3):
        remaining = phase4_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_k3 - i), 45.0)
        if alloc < 5:
            break
        dl = time.monotonic() + alloc

        f_best, M_best = adam_optimize(results[i][1], dl, rng, lr=0.015, max_iters=15000,
                                       beta_init_frac=0.1, beta_decay=0.9,
                                       beta_period=50, restart_thresh=300)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results.append((sc, f_best))
        print(f"  Polish {i + 1}/{top_k3}: score={sc:.10f}, best={best_score:.10f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 4 done. Best: {best_score:.10f}")

    # Phase 5: Multi-scale
    print("Phase 5: Multi-scale")
    phase5_end = min(deadline, t_start + 1650.0)

    # n=128
    n_small = 128
    for i in range(min(3, len(results))):
        remaining = phase5_end - time.monotonic()
        if remaining < 15:
            break
        f_interp = interpolate_profile(results[i][1], n_small)
        dl = time.monotonic() + min(30, remaining / max(1, 3 - i))
        f_best, M_best = adam_optimize(f_interp, dl, rng, lr=0.04, max_iters=5000,
                                       beta_init_frac=0.3, beta_decay=0.85,
                                       beta_period=60, restart_thresh=150)
        sc = 2.0 * n_small * M_best
        try_update(f_best, sc)
        print(f"  n={n_small} start {i + 1}: score={sc:.10f}, best={best_score:.10f}")

    # n=512
    n_large = 512
    for i in range(min(3, len(results))):
        remaining = phase5_end - time.monotonic()
        if remaining < 20:
            break
        f_interp = interpolate_profile(results[i][1], n_large)
        dl = time.monotonic() + min(40, remaining / max(1, 3 - i))
        f_best, M_best = adam_optimize(f_interp, dl, rng, lr=0.03, max_iters=5000,
                                       beta_init_frac=0.3, beta_decay=0.85,
                                       beta_period=60, restart_thresh=200)
        sc = 2.0 * n_large * M_best
        try_update(f_best, sc)
        print(f"  n={n_large} start {i + 1}: score={sc:.10f}, best={best_score:.10f}")

    print(f"Phase 5 done. Best: {best_score:.10f}")

    # Phase 6: Final refinement on the best
    print("Phase 6: Final refinement")
    if best_seq is not None and time.monotonic() < deadline - 30:
        f_best = interpolate_profile(best_seq, 256)
        dl = deadline - 5
        f_refined, M_refined = adam_optimize(f_best, dl, rng, lr=0.01, max_iters=20000,
                                             beta_init_frac=0.05, beta_decay=0.92,
                                             beta_period=100, restart_thresh=400)
        sc = 2.0 * 256 * M_refined
        try_update(f_refined, sc)
        print(f"  Final Adam: score={sc:.10f}, best={best_score:.10f}")

        if time.monotonic() < deadline - 20:
            f_lp, M_lp = lp_iterate(f_refined, deadline - 5, rng, max_iters=200, use_min_peak=False)
            sc_lp = 2.0 * 256 * M_lp
            try_update(f_lp, sc_lp)
            print(f"  Final LP: score={sc_lp:.10f}, best={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = make_linear_profile(256, 1.5)
        best_score = compute_score(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
    else:
        best_seq = best_seq / s

    final_score = compute_score(best_seq)
    print(f"Final: n={len(best_seq)}, score={final_score:.10f}")

    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence score: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
