    """Project v onto {x: x >= 0, sum(x) = target_sum}."""
    # Use water-filling
    # Find lambda such that sum(max(v - lambda, 0)) = target_sum
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find rho such that v_sorted[rho] > (cumsum[rho] - target_sum) / (rho + 1) >= v_sorted[rho+1]
    # lambda = (cumsum[rho] - target_sum) / (rho + 1)
    
    n = len(v)
    # Bisection on lambda
    lo = v.min()
    hi = v.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lambda_val = (lo + hi) / 2

def score_fn(s):
    """Compute 2*n*max(conv(s,s))/sum(s)^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def conv_max_and_index(s):
    """Compute convolution, return (max_value, argmax_index)."""
    conv = np.convolve(s, s)
    return np.max(conv), np.argmax(conv), conv

def subgradient_at_max(s, k_star, top_k=1):
    """Compute sub-gradient at the maximum position(s).
    Returns gradient of max(conv[s,s]) using top_k peak positions.
    """
    n = len(s)
    conv = np.convolve(s, s)
    max_val = conv.max()
    
    # Find positions within 0.1% of max
    threshold = max_val * 0.999
    active = np.where(conv >= threshold)[0]
    
    # Use at most top_k active positions
    if len(active) > top_k:
        # Use the top_k with highest values
        idx = np.argsort(conv[active])[::-1][:top_k]
        active = active[idx]
    
    g = np.zeros(n)
    for k in active:

        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            g[j_lo:j_hi+1] += 2.0 * s[k - j_hi : k - j_lo - 1 : -1]
    
    # Average over active set
    g /= len(active)
    return g, conv

def project_simplex(v, target_sum):
    """Project v onto {x: x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Water-filling: find lambda such that sum(max(v - lambda, 0)) = target_sum
    # Bisection
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_val = np.sum(np.maximum(v - mid, 0.0))
        if s_val > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.maximum(v - lam, 0.0)
    # Fix any numerical issues
    result_sum = result.sum()
    if result_sum > 0:
        result = result * (target_sum / result_sum)
    return result

def lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M, b >= 0."""
    n = len(s)
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = s[j] if 0 <= k-j < n, else 0 (for conv(s,b)[k] = sum_j s[k-j]*b[j])
    # Actually conv(s,b)[k] = sum_j s[j] * b[k-j] (valid indices)
    # = sum_j b[j] * s[k-j] (valid indices)
    # So A[k, j] = s[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # b[j] appears in conv[k] for k in [j, j+n-1] with coefficient s[k-j]
        A[j:j+n, j] = s
    
    c = -np.ones(n)  # Maximize sum(b) = minimize -sum(b)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None
