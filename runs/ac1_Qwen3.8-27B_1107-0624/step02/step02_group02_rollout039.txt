    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = np.maximum(c, 0.0)
    return c

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_direction(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception:
        pass
    return None

def line_search_blend(f, g, best_M, max_iter=20):
    """Find best blend parameter t in [0, 1] that minimizes max(conv)."""
    def obj(t):
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            return 1e10
        f_new = f_new / s
        c = np.convolve(f_new, f_new)
        return c.max()
    
    # Try t values
    best_t = 0.0
    best_val = obj(0.0)
    
    # Coarse search
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        v = obj(t)
        if v < best_val:
            best_val = v
            best_t = t
    
    # Refine around best_t
    if best_t > 0:
        t_lo = max(0, best_t - 0.1)
        t_hi = min(1, best_t + 0.1)
        result = minimize_scalar(obj, bounds=(t_lo, t_hi), method='bounded', options={'maxiter': max_iter})
        if result.fun < best_val:
            best_t = result.x
            best_val = result.fun
    
    return best_t, best_val

def lp_optimize_v2(f_init, deadline, rng, verbose=True):
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)
    
    best_M = float(np.convolve(f, f).max())
    best_f = f.copy()
    stagnant = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        M_current = float(np.convolve(f, f).max())
        
        # Try multiple M values
        best_g = None
        best_g_M = np.inf
        for M_mult in [0.995, 1.0, 1.005]:
            M_try = M_current * M_mult
            g = lp_direction(f, M_try)
            if g is not None:
                # Check if this direction actually helps
                t_test = 0.1
                f_test = (1 - t_test) * f + t_test * g
                f_test = np.maximum(f_test, 0.0)
                s_test = f_test.sum()
                if s_test > 1e-10:
                    f_test = f_test / s_test
                    M_test = float(np.convolve(f_test, f_test).max())
                    if M_test < best_g_M:
                        best_g_M = M_test
                        best_g = g
        
        if best_g is None:
            # Try restart
            f = best_f + 0.05 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            stagnant += 1
            if stagnant > 20:
                break
            continue
        
        # Line search
        t_opt, M_new = line_search_blend(f, best_g, best_M)
        
        if t_opt > 0 and M_new <= best_M + 1e-12:
            f_new = (1 - t_opt) * f + t_opt * best_g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new > 1e-10:
                f_new = f_new / s_new
            f = f_new
            f = f / f.sum()
            
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
            
            if verbose and iteration % 20 == 0:
                print(f"    LP iter {iteration}: M={best_M:.10f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            # Restart with different perturbation
            scale = 0.02 + 0.03 * min(stagnant / 30, 2.0)
            f = best_f + scale * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            stagnant = 0
    
    return best_f

def subgradient_polish(f_init, deadline, rng, verbose=True):
    """Subgradient descent on the max correlation."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    f = f / f.sum()
    n = len(f)
    
    best_M = float(np.convolve(f, f).max())
    best_f = f.copy()
    no_imp = 0
    
    # Polyak-like step size
    M_target = best_M * 0.99  # optimistic target
    k = 0
    
    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        M_current = float(c.max())
        k_max = int(np.argmax(c))
        
        # Subgradient of max(c) w.r.t. f
        # d c[k_max] / d f[i] = f[k_max - i] + f[i] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] += f[j]
            # The other term: f[i] appears when k_max - j = i, i.e., j = k_max - i
            # Actually, d(f*f)[k]/df[i] = f[k-i] + f[i] if both in range
            # Wait, let me redo this.
            # (f*f)[k] = sum_j f[j] * f[k-j]
            # d/d f[i] = f[k-i] (from the first f) + f[i] (from the second f, when k-j=i, j=k-i)
            # So d/d f[i] (f*f)[k] = f[k-i] + f[i] (with zero padding)
        
        # Recompute properly
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            j1 = k_max - i
            if 0 <= j1 < n:
                val += f[j1]
            val += f[i]  # from the other term (always in range since i is in [0,n))
            grad[i] = val
        
        # Project: we want to decrease max(c), so move in -grad direction
        # But we also need to maintain sum(f) = 1 and f >= 0
        # Use a simple approach: f = f - alpha * grad, then project
        alpha = 1.0 / (k + 10) * 0.1
        f_new = f - alpha * grad
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / s
        
        M_new = float(np.convolve(f_new, f_new).max())
        
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_f = f_new.copy()
            no_imp = 0
        else:
            f_new = f  # reject
            no_imp += 1
            # Try smaller step
            if no_imp % 10 == 0:
                alpha *= 0.5
                f_new = f - alpha * grad
                f_new = np.maximum(f_new, 0.0)
                s = f_new.sum()
                if s > 1e-10:
                    f_new = f_new / s
                    M_new = float(np.convolve(f_new, f_new).max())
                    if M_new < best_M - 1e-13:
                        best_M = M_new
                        best_f = f_new.copy()
                        no_imp = 0
        
        f = best_f if no_imp > 50 else f_new
        k += 1
        
        if no_imp > 200:
            # Restart
            f = best_f + 0.01 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            no_imp = 0
            k = 0
        
        if verbose and k % 500 == 0:
            print(f"    Subgrad iter {k}: M={best_M:.10f}, no_imp={no_imp}")
    
    return best_f

def make_profile(n, type='linear', **kwargs):
    t = np.linspace(0.0, 1.0, n)
    if type == 'linear':
        eta = kwargs.get('eta', 2.0)
        a = 1.0 + eta * t
    elif type == 'cosine':
        k = kwargs.get('k', 1)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * k * t)
    elif type == 'exp':
        rate = kwargs.get('rate', 3.0)
        a = np.exp(rate * t)
    elif type == 'quadratic':
        a = 1.0 + t ** 2
    elif type == 'random_smooth':
        rng = kwargs.get('rng', np.random.default_rng(0))
        coeffs = rng.uniform(-0.4, 0.4, 4)
        a = np.ones(n)
        for i, c in enumerate(coeffs, 1):
            a = a + c * np.cos(2 * np.pi * i * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -40.0, 40.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / e.size
    return e / s

def logit_adam_polish(p0, deadline, rng, lr=0.05):
    """Logit-space Adam polishing."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    c = full_convolve_fft(p0)
    maxc = float(c.max())
    beta = max(1e-12, 0.3 * maxc)
    
    best_p = p0.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        # Soft max weight
        mval = float(c.max())
        if beta <= 1e-14:
            w = np.zeros(len(c))
            mask = c >= mval - 1e-12
            w[mask] = 1.0 / np.count_nonzero(mask)
        else:
            x = (c - mval) / beta
            ex = np.exp(x)
            s = float(ex.sum())
            w = ex / s
        
        # Gradient: d max(c) / d p[i] ≈ 2 * sum_j w[k] * (p[k-i] + p[i])
        # For the soft version: d L / d p[i] = 2 * sum_k w[k] * dc[k]/dp[i]
        # dc[k]/dp[i] = p[k-i] + p[i] (with bounds)
        # So dL/dp[i] = 2 * sum_k w[k] * (p[k-i] + p[i])
        # = 2 * (sum_k w[k] * p[k-i]) + 2 * p[i] * sum_k w[k]
        # = 2 * (w * p_reversed)[i] + 2 * p[i]  (approximately)
        
        # More precisely, let's compute it via the convolution approach
        # h[i] = sum_k w[k] * p[k-i]  -- this is like a cross-correlation
        # We can compute this efficiently:
        # h = correlate(w, p) -- need to be careful about indexing
        
        # Let's use FFT:
        N = next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp, N)
        # h[i] = sum_k w[k] * p[k-i+padded]... 
        # Actually, let me think about this more carefully.
        # We want: grad[i] = sum_k w[k] * (p[k-i] + p[i])
        # = sum_k w[k] * p[k-i] + p[i] * sum_k w[k]
        # = (w * p)[i] (cross-corr) + p[i]
        
        # (w * p)[i] = sum_k w[k] * p[k-i]
        # This is the cross-correlation of w and p at lag i.
        # Using FFT: if we compute conv(w, p_reversed), we get the cross-correlation.
        
        # Let me just compute it directly for clarity (n is at most 1024):
        grad_p = np.zeros(n)
        for i in range(n):
            # sum_k w[k] * p[k-i] where k-i must be in [0, n-1]
            val = 0.0
            for k in range(n):
                j = k - i  # index into p
                if 0 <= j < n:
                    val += w[k] * p[j]
            grad_p[i] = val + p[i]  # + p[i] * sum(w) = + p[i] since sum(w)=1
        
        # Hmm, this O(n^2) loop might be slow for n=1024. Let me use the FFT approach.
        # Actually for n=512 it should be fine (512*1023 ≈ 500K operations per iteration)
        
        # Gradient in z-space: dz/dp * dL/dp
        # p = softmax(z), so dp/dz = p * (I - 1 1^T) (outer product)
        # dL/dz[i] = p[i] * (dL/dp[i] - sum_j p[j] * dL/dp[j])
        
        dot = float(np.dot(p, grad_p))
        grad_z = p * (2.0 * grad_p + 1e-8 * p - dot)  # factor of 2 from the objective
        grad_z -= np.mean(grad_z)
        
        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_p + 1e-12)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * grad_z ** 2
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        z = np.clip(z, -40, 40)
        z -= np.mean(z)
        p = softmax_from_z(z)
        
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        
        if score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if adam_t % 100 == 0:
            beta = max(1e-14, beta * 0.7)
        
        if no_imp > max(500, 3 * n):
            z = 0.7 * np.log(best_p + 1e-12) + 0.3 * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= np.mean(z)
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            beta = max(1e-12, 0.3 * c.max())
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP with line search + subgradient polish + Adam polish.
    Explores multiple sequence lengths and starting profiles.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
    
    # Try to get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass
    
    print(f"Starting. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)
    
    # Phase 1: LP optimization at n=512
    n_main = 512
    phase1_end = min(deadline, t_start + 600.0)
    
    # Starting profile
    if init_seq is not None and init_seq.size == n_main:
        p_main = init_seq.copy()
    elif init_seq is not None:
        p_main = interpolate_profile(init_seq, n_main)
    else:
        p_main = make_profile(n_main, 'linear', eta=2.0)
    
    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1: n={n_main}, start score={sc:.10f}")
    
    # Main LP run
    lp_deadline = min(phase1_end, time.monotonic() + 400.0)
    result = lp_optimize_v2(p_main, lp_deadline, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"After LP: score={sc:.10f}")
    
    # Try other profiles
    profiles = [
        ('linear_e3', make_profile(n_main, 'linear', eta=3.0)),
        ('linear_e1', make_profile(n_main, 'linear', eta=1.0)),
        ('exp_r2', make_profile(n_main, 'exp', rate=2.0)),
        ('exp_r4', make_profile(n_main, 'exp', rate=4.0)),
        ('quad', make_profile(n_main, 'quadratic')),
        ('cos_1', make_profile(n_main, 'cosine', k=1)),
        ('cos_2', make_profile(n_main, 'cosine', k=2)),
        ('rand1', make_profile(n_main, 'random_smooth', rng=rng)),
        ('rand2', make_profile(n_main, 'random_smooth', rng=rng)),
    ]
    
    for name, p in profiles:
        if time.monotonic() > phase1_end - 5:
            break
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.1:
            continue
        per_dl = min(phase1_end, time.monotonic() + 30.0)
        result = lp_optimize_v2(p, per_dl, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        if sc < best_score + 0.05:
            print(f"  Profile {name}: {sc:.10f}")
    
    print(f"After Phase 1: best={best_score:.10f}")
    
    # Phase 2: Subgradient polish
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        n_polish = best_seq.size
        print(f"Phase 2: subgradient polish, n={n_polish}")
        sub_dl = min(phase2_end, time.monotonic() + 150.0)
        result = subgradient_polish(best_seq, sub_dl, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After subgradient: score={sc:.10f}")
    
    # Phase 3: LP at larger n
    n_large = 1024
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 3: LP at n={n_large}, score={sc:.10f}")
        
        lp3_dl = min(phase3_end, time.monotonic() + 300.0)
        result = lp_optimize_v2(p_large, lp3_dl, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After large LP: score={sc:.10f}")
    
    # Phase 4: Adam polish on best
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        n_apol = best_seq.size
        print(f"Phase 4: Adam polish, n={n_apol}, current best={best_score:.10f}")
        
        for lr in [0.05, 0.03, 0.08]:
            if time.monotonic() > phase4_end - 5:
                break
            per_dl = min(phase4_end, time.monotonic() + 100.0)
            p0 = best_seq + 0.002 * rng.normal(size=n_apol)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = logit_adam_polish(p0, per_dl, rng, lr=lr)
            # Convert score (which is 2n*maxc with sum=1) to actual score
            sc_actual = compute_score(result)
            update(result, sc_actual)
            print(f"  Adam lr={lr}: score={sc_actual:.10f}")
    
    # Phase 5: Try n=2048 if time and score warrant
    n_xl = 2048
    phase5_end = min(deadline, t_start + 1720.0)
    if (time.monotonic() < phase5_end - 30 and best_seq is not None 
            and best_score > 1.503):
        p_xl = interpolate_profile(best_seq, n_xl)
        sc = compute_score(p_xl)
        update(p_xl, sc)
        print(f"Phase 5: n={n_xl}, score={sc:.10f}")
        lp5_dl = min(phase5_end, time.monotonic() + 80.0)
        result = lp_optimize_v2(p_xl, lp5_dl, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After xl LP: score={sc:.10f}")
    
    # Phase 6: Final short polish
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = min(deadline - 5, time.monotonic() + 20.0)
        n_final = best_seq.size
        p0 = best_seq + 0.001 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, _ = logit_adam_polish(p0, per, rng, lr=0.03)
        sc = compute_score(result)
        update(result, sc)
        print(f"Final polish: score={sc:.10f}")
    
    # Final validation
    if best_seq is not None:
        final_seq = np.maximum(best_seq, 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
    else:
        final_seq = make_profile(512, 'linear', eta=2.0)
    
    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")
    
    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):

**Key mathematical insight**: The problem `min max_k (a*a)[k]` subject to `sum(a) = 1, a >= 0` can be reformulated.

"""

import numpy as np

from scipy.optimize import linprog, minimize_scalar


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Return normalized g."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def adaptive_blend(f, g, max_trials=6):

def fft_cross_corr(w, p):
    """Compute h[i] = sum_k w[k] * p[k-i] for i=0..n-1 using FFT."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp_rev = np.fft.rfft(p[::-1], N)
    corr = np.fft.irfft(fw * fp_rev, N)
    # corr[m] = sum_k w[k] * p_rev[m-k] = sum_k w[k] * p[n-1-(m-k)]
    # We want h[i] = sum_k w[k] * p[k-i]
    # p[k-i] = p_rev[n-1-(k-i)] = p_rev[n-1-k+i]
    # So h[i] = sum_k w[k] * p_rev[n-1-k+i]
    # conv(w, p_rev)[m] = sum_k w[k] * p_rev[m-k]
    # Set m-k = n-1-k+i => m = n-1+i
    # So h[i] = conv(w, p_rev)[n-1+i] = corr[n-1+i]
    return corr[n - 1:2 * n - 1]


def gradient_polish(f_init, deadline, rng, verbose=True):
    """Subgradient/gradient method on the soft-max correlation."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    f = f / f.sum()
    n = len(f)

    c = full_convolve_fft(f)
    best_M = float(c.max())
    best_f = f.copy()
    no_imp = 0
    k_iter = 0
    beta = 0.3 * best_M

    while time.monotonic() < deadline:
        c = full_convolve_fft(f)
        mval = float(c.max())

        # Soft max weights
        if beta <= 1e-15:
            w = np.zeros(len(c))
            mask = c >= mval - 1e-12
            if np.any(mask):
                w[mask] = 1.0 / np.count_nonzero(mask)
            else:
                w[int(np.argmax(c))] = 1.0
        else:
            x = (c - mval) / beta
            ex = np.exp(x)
            s = float(ex.sum())
            w = ex / s

        # Gradient: d/dp[i] L = 2 * sum_k w[k] * dc[k]/dp[i]
        # dc[k]/dp[i] = p[k-i] + p[i] (with zero padding for out-of-range)
        # = h[i] + p[i] where h[i] = sum_k w[k] * p[k-i]
        h = fft_cross_corr(w, f)
        grad_p = 2.0 * (h + f)  # sum(w) = 1

        # Projected gradient step (maintain sum=1, f>=0)
        alpha = 0.5 / (np.sqrt(k_iter + 10) + 1.0)
        f_new = f - alpha * grad_p
        # Remove mean to maintain sum
        f_new = f_new - (f_new.sum() - 1.0) / n
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / s

        M_new = float(full_convolve_fft(f_new).max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f_new.copy()
            no_imp = 0
            f = f_new
        else:
            no_imp += 1
            # Backtrack
            if no_imp % 5 == 0:
                alpha *= 0.5
                f_new = f - alpha * grad_p
                f_new = f_new - (f_new.sum() - 1.0) / n
                f_new = np.maximum(f_new, 0.0)
                s = f_new.sum()
                if s > 1e-10:
                    f_new = f_new / s
                    M_new = float(full_convolve_fft(f_new).max())
                    if M_new < best_M - 1e-14:
                        best_M = M_new
                        best_f = f_new.copy()
                        no_imp = 0
                        f = f_new

        k_iter += 1

        # Anneal beta
        if k_iter % 50 == 0:
            beta = max(1e-15, beta * 0.85)

        if no_imp > 200:
            f = best_f + 0.01 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            no_imp = 0
            k_iter = 0
            beta = 0.3 * best_M

        if verbose and k_iter % 200 == 0:
            print(f"    Grad iter {k_iter}: M={best_M:.12f}, no_imp={no_imp}")

    return best_f


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -40.0, 40.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / e.size
    return e / s


def adam_polish(p0, deadline, rng, lr=0.06):
    """Logit-space Adam with FFT-based gradient."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = full_convolve_fft(p0)
    maxc = float(c.max())
    beta = max(1e-12, 0.25 * maxc)

    best_p = p0.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        c = full_convolve_fft(p0)
        mval = float(c.max())

        # Soft max weights
        if beta <= 1e-15:
            w = np.zeros(len(c))
            mask = c >= mval - 1e-12
            if np.any(mask):
                w[mask] = 1.0 / np.count_nonzero(mask)
            else:
                w[int(np.argmax(c))] = 1.0
        else:
            x = (c - mval) / beta
            ex = np.exp(x)
            s = float(ex.sum())
            if not np.isfinite(s) or s <= 0:
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
                s = 1.0
            w = ex / s

        # Gradient in p-space
        h = fft_cross_corr(w, p0)
        grad_p = 2.0 * (h + p0)

        # Gradient in z-space: dL/dz[i] = p[i] * (dL/dp[i] - sum_j p[j] dL/dp[j])
        dot = float(np.dot(p0, grad_p))
        grad_z = p0 * (grad_p - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_p + 1e-12)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            no_imp += 1
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * grad_z * grad_z
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        z = np.clip(z, -40, 40)
        z -= np.mean(z)
        p0 = softmax_from_z(z)

        c = full_convolve_fft(p0)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if score < best_score - 1e-14:
            best_score = score
            best_p = p0.copy()
            no_imp = 0
        else:
            no_imp += 1

        if adam_t % 80 == 0:
            beta = max(1e-15, beta * 0.8)

        if no_imp > max(400, 2 * n):
            z = 0.7 * np.log(best_p + 1e-12) + 0.3 * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= np.mean(z)
            p0 = softmax_from_z(z)
            c = full_convolve_fft(p0)
            beta = max(1e-12, 0.3 * float(c.max()))
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def make_profile(n, ptype, rng=None):
    t = np.linspace(0.0, 1.0, n)
    if ptype.startswith('lin'):
        try:
            eta = float(ptype.split('_')[1])
        except (ValueError, IndexError):
            eta = 2.0
        a = 1.0 + eta * t
    elif ptype.startswith('exp'):
        try:
            rate = float(ptype.split('_')[1])
        except (ValueError, IndexError):
            rate = 3.0
        a = np.exp(rate * t)
    elif ptype.startswith('cos'):
        try:
            k = int(ptype.split('_')[1])
        except (ValueError, IndexError):
            k = 1
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t)
    elif ptype.startswith('quad'):
        a = 1.0 + 0.5 * t + 0.5 * t * t
    elif ptype.startswith('rand'):
        if rng is None:
            rng = np.random.default_rng(0)
        coeffs = rng.uniform(-0.3, 0.3, 4)
        a = np.ones(n)
        for i, c in enumerate(coeffs, 1):
            a = a + c * np.cos(2 * np.pi * i * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP with adaptive line search, gradient polish,
    Adam polish. Explores n in {384, 512, 768, 1024}.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc_init = compute_score(init_seq)
                print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
                update(init_seq, sc_init)
    except Exception:
        pass

    # ============ Phase 1: Quick exploration of n values ============
    n_candidates = [384, 512, 768, 1024]
    phase1_end = min(deadline, t_start + 350.0)

    # Determine which n to focus on
    for n_try in n_candidates:
        if time.monotonic() > phase1_end - 20:
            break
        # Starting profile
        if init_seq is not None:
            p0 = interpolate_profile(init_seq, n_try)
        else:
            p0 = make_profile(n_try, 'lin_2.0')

        sc = compute_score(p0)
        update(p0, sc)

        # Quick LP run
        per_dl = min(phase1_end, time.monotonic() + 50.0)
        result = lp_optimize(p0, per_dl, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        if sc < best_score + 0.05:
            print(f"  n={n_try}: score={sc:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # ============ Phase 2: Extended LP on best n ============
    n_best = best_seq.size if best_seq is not None else 512
    phase2_end = min(deadline, t_start + 750.0)

    if time.monotonic() < phase2_end - 10:
        print(f"Phase 2: LP at n={n_best}")
        p0 = best_seq.copy()
        lp_dl = min(phase2_end, time.monotonic() + 350.0)
        result = lp_optimize(p0, lp_dl, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After extended LP: score={sc:.10f}")

        # Try a few alternative profiles at same n
        profiles = ['lin_1.5', 'lin_3.0', 'exp_2.0', 'exp_4.0', 'quad',
                    'cos_1', 'cos_2', 'rand_0', 'rand_1']
        for pname in profiles:
            if time.monotonic() > phase2_end - 5:
                break
            p_alt = make_profile(n_best, pname, rng)
            sc_p = compute_score(p_alt)
            if sc_p >= best_score + 0.15:
                continue
            per_dl = min(phase2_end, time.monotonic() + 30.0)
            result = lp_optimize(p_alt, per_dl, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
            if sc < best_score + 0.02:
                print(f"  Alt profile {pname}: {sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: Gradient polish ============
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        print(f"Phase 3: gradient polish, n={best_seq.size}")
        grad_dl = min(phase3_end, time.monotonic() + 200.0)
        result = gradient_polish(best_seq, grad_dl, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After gradient polish: score={sc:.10f}")

        # Another round with noise
        if time.monotonic() < phase3_end - 10:
            p_noisy = best_seq + 0.005 * rng.normal(size=best_seq.size)
            p_noisy = np.maximum(p_noisy, 1e-12)
            p_noisy = p_noisy / p_noisy.sum()
            grad_dl2 = min(phase3_end, time.monotonic() + 120.0)
            result = gradient_polish(p_noisy, grad_dl2, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Adam polish ============
    phase4_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        n_ap = best_seq.size
        print(f"Phase 4: Adam polish, n={n_ap}, current={best_score:.10f}")

        lrs = [0.06, 0.04, 0.08, 0.03]
        for lr in lrs:
            if time.monotonic() > phase4_end - 5:
                break
            per_dl = min(phase4_end, time.monotonic() + 100.0)
            p0 = best_seq.copy()
            # Tiny perturbation
            p0 = p0 + 0.001 * rng.normal(size=n_ap)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, _ = adam_polish(p0, per_dl, rng, lr=lr)
            sc = compute_score(result)
            update(result, sc)
            print(f"  Adam lr={lr}: score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ============ Phase 5: Try larger/smaller n if beneficial ============
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end - 30 and best_seq is not None:
        # Try nearby n values
        for n_adj in [n_best + 128, n_best - 128, n_best + 256]:
            if n_adj < 128 or n_adj > 2048:
                continue
            if time.monotonic() > phase5_end - 10:
                break
            p_adj = interpolate_profile(best_seq, n_adj)
            sc = compute_score(p_adj)
            update(p_adj, sc)
            # Quick LP on this
            lp_dl = min(phase5_end, time.monotonic() + 40.0)
            result = lp_optimize(p_adj, lp_dl, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
            if sc < best_score + 0.01:
                print(f"  n={n_adj}: {sc:.10f}")

    print(f"After Phase 5: best={best_score:.10f}")

    # ============ Phase 6: Final short polish ============
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = min(deadline - 5, time.monotonic() + 25.0)
        n_fin = best_seq.size
        p0 = best_seq + 0.0005 * rng.normal(size=n_fin)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, _ = adam_polish(p0, per, rng, lr=0.03)
        sc = compute_score(result)
        update(result, sc)
        print(f"Final polish: score={sc:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = np.maximum(best_seq, 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
    else:
        final_seq = make_profile(512, 'lin_2.0')

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
