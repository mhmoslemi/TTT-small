The gradient of the objective w.r.t. z is:

= d(obj)/df[i] * f[i] - f[i] * sum_j d(obj)/df[j] * f[j]
= f[i] * (grad_f[i] - dot(f, grad_f))

where grad_f = d(obj)/df is the gradient with respect to f.

This is the standard softmax parameterization gradient.

import numpy as np
import time

def compute_conv(f):
    return np.convolve(f, f)

def compute_score_norm(f):
    """Score for normalized f (sum=1)."""
    n = len(f)
    c = compute_conv(f)
    return 2.0 * n * float(c.max())

def softmax_normalize(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def compute_soft_max(c, beta):
    mval = c.max()
    x = (c - mval) / beta
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w

def compute_grad_f(f, w, N):
    """Gradient of soft_max(conv(f,f)) w.r.t. f."""
    n = len(f)
    gw = np.fft.fft(w, N)
    ff = np.fft.fft(f, N)
    corr = np.fft.ifft(gw * np.conj(ff)).real[:n]
    return 2.0 * corr

def compute_grad_z(f, grad_f):
    """Gradient w.r.t. log-parameters z (softmax parameterization)."""
    dot = float(np.dot(f, grad_f))
    return f * (grad_f - dot)

def adam_update(m, v, t, b1, b2, eps, grad, lr):
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * (grad * grad)
    mh = m / (1 - b1**t)
    vh = v / (1 - b2**t)
    return z - lr * mh / (np.sqrt(vh) + eps)

import numpy as np
import time

def compute_conv(f):
    return np.convolve(f, f)

def compute_score_norm(f):
    n = len(f)
    c = compute_conv(f)
    return 2.0 * n * float(c.max())

def softmax_normalize(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(z), 1.0/len(z))
    return e / s

def soft_max_val_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w

def grad_soft_max_conv(f, w, N):
    """Gradient of soft_max(conv(f,f)) w.r.t. f. Returns array of length n."""
    n = len(f)
    gw = np.fft.fft(w, N)
    ff = np.fft.fft(f, N)
    corr = np.fft.ifft(gw * np.conj(ff)).real[:n]
    return 2.0 * corr

def grad_z_from_grad_f(f, gf):
    """Convert grad w.r.t. f to grad w.r.t. z (log-space)."""
    dot = float(np.dot(f, gf))
    gz = f * (gf - dot)
    return gz - gz.mean()

def make_power_law(n, alpha):
    t = np.arange(1, n+1, dtype=np.float64)
    f = t ** alpha
    return f / f.sum()

def make_smooth_random(n, rng, n_harmonics=10):

    f = np.ones(n)
    for k in range(1, n_harmonics+1):
        amp = rng.uniform(-0.5, 0.5) / k
        f += amp * np.cos(2*np.pi*k*t + rng.uniform(0, 2*np.pi))
    f = np.maximum(f, 0.01)
    return f / f.sum()

def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)

def coordinate_refine(f, deadline, max_rounds=100):
    """Coordinate-wise descent: try adjusting each element slightly."""
    n = len(f)
    f = f.copy()
    c = compute_conv(f)
    M = float(c.max())
    best_M = M
    best_f = f.copy()
    
    # For each coordinate, try a range of values
    # To make this efficient, only adjust coordinates that are "important"
    # (i.e., contribute to the peak convolution values)
    
    steps = [0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 0.3, 0.1]
    
    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            f_old = f[i]
            for s in steps:
                f_try = f.copy()
                f_try[i] = f_old * s
                # Renormalize
                f_try = np.maximum(f_try, 0)
                s_sum = f_try.sum()
                if s_sum < 1e-10:
                    continue
                f_try = f_try / s_sum
                c_try = compute_conv(f_try)
                M_try = float(c_try.max())
                if M_try < M - 1e-12:
                    f = f_try
                    M = M_try
                    improved = True
                    break
            if M < best_M - 1e-12:
                best_M = M
                best_f = f.copy()
        
        if not improved:
            break
    
    return best_f, best_M

                  restart_patience=200, restart_freq=500, max_iters=100000):

    Adam optimization in log-space for minimizing max(conv(f,f)) with sum(f)=1.
    """
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    
    z = np.log(f)
    z = z - z.mean()

    best_z = z.copy()
    no_imp = 0

        if time.monotonic() > deadline:
            break
        
        _, w = soft_max_val_weights(c, beta)
        gf = grad_soft_max_conv(f, w, N)
        gz = grad_z_from_grad_f(f, gf)

            z = best_z.copy()
            m[:] = 0
            v[:] = 0
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1**t_adam)
        vh = v / (1 - b2**t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        f = softmax_normalize(z)
        c = compute_conv(f)
        M = float(c.max())

            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0

import numpy as np
import time

"""
Multi-phase gradient optimization in log-space for minimizing 2*n*max(conv(f,f))/sum(f)^2.
Uses Adam with adaptive soft-max parameter, multiple sequence lengths (256, 512, 1024),
diverse asymmetric starting points, coordinate-wise refinement, and smart restarts.
Key insight: asymmetric sequences reduce the central convolution peak.
"""


def compute_conv(f):
    return np.convolve(f, f)


def compute_score_norm(f):
    n = len(f)
    c = compute_conv(f)
    return 2.0 * n * float(c.max())


def softmax_normalize(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(z), 1.0 / len(z))
    return e / s


def soft_max_val_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def grad_soft_max_conv(f, w, N):
    """Gradient of soft_max(conv(f,f)) w.r.t. f using FFT cross-correlation."""
    n = len(f)
    gw = np.fft.fft(w, N)
    ff = np.fft.fft(f, N)
    corr = np.fft.ifft(gw * np.conj(ff)).real[:n]
    return 2.0 * corr


def grad_z_from_grad_f(f, gf):
    """Convert gradient w.r.t. f to gradient w.r.t. log-parameters z."""
    dot = float(np.dot(f, gf))
    gz = f * (gf - dot)
    return gz - gz.mean()


def make_power_law(n, alpha):
    t = np.arange(1, n + 1, dtype=np.float64)
    f = t ** alpha
    return f / f.sum()


def make_beta_shape(n, a, b):
    """Beta-distribution-like shape: f[j] ~ (j+1)^a * (n-j)^b."""
    j = np.arange(n, dtype=np.float64)
    left = (j + 1) ** a
    right = (n - j) ** b
    f = left * right
    f = np.maximum(f, 1e-15)
    return f / f.sum()


def make_smooth_random(n, rng, n_harmonics=12):
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(-0.4, 0.4) / (k ** 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        f += amp * np.cos(2 * np.pi * k * t + phase)
    f = np.maximum(f, 0.01)
    return f / f.sum()


def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def get_starting_points(n, rng, parent=None):
    """Generate diverse starting points for optimization."""
    starts = []
    if parent is not None:
        starts.append(interpolate_profile(parent, n))
        starts.append(interpolate_profile(parent[::-1], n))
    # Asymmetric power laws (key: asymmetry reduces central peak)
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        starts.append(make_power_law(n, alpha))
    # Beta-like shapes with varying asymmetry
    for a, b in [(1.0, 0.0), (0.0, 1.0), (2.0, 0.5), (0.5, 2.0),
                 (1.5, 0.0), (0.0, 1.5), (3.0, 1.0), (1.0, 3.0),
                 (2.0, -0.3), (-0.3, 2.0)]:
        starts.append(make_beta_shape(n, a, b))
    # Smooth random
    for _ in range(5):
        starts.append(make_smooth_random(n, rng))
    # Uniform
    starts.append(np.ones(n) / n)
    # Skewed Gaussian
    t = np.linspace(0, 1, n)
    for mu in [0.3, 0.4, 0.6, 0.7]:
        f = np.exp(-0.5 * ((t - mu) / 0.25) ** 2)
        f = np.maximum(f, 1e-12)
        starts.append(f / f.sum())
    return starts


def adam_optimize(f_init, deadline, rng, lr_init=0.08, beta_init=None,
                  beta_min=0.005, beta_decay=0.94, beta_interval=40,
                  restart_patience=200, max_iters=200000):
    """
    Adam optimization in log-space for minimizing max(conv(f,f)) with sum(f)=1.
    Returns (best_f, best_M).
    """
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()

    z = np.log(f)
    z = z - z.mean()

    N = 1
    while N < 2 * n:
        N <<= 1

    c = compute_conv(f)
    M = float(c.max())

    if beta_init is None:
        beta_init = 0.5 * M
    beta = float(beta_init)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_M = M
    best_f = f.copy()
    best_z = z.copy()
    no_imp = 0
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        _, w = soft_max_val_weights(c, beta)
        gf = grad_soft_max_conv(f, w, N)
        gz = grad_z_from_grad_f(f, gf)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m[:] = 0
            v[:] = 0
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        step = lr * mh / (np.sqrt(vh) + eps)
        z = z - step

        z = np.clip(z, -50, 50)
        z = z - z.mean()
        f = softmax_normalize(z)
        c = compute_conv(f)
        M = float(c.max())

        if M < best_M - 1e-13:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta decay
        if it % beta_interval == 0 and it > 0:
            beta = max(beta_min, beta * beta_decay)

        # Learning rate decay on stagnation
        if no_imp > 0 and no_imp % 300 == 0:
            lr = max(0.002, lr * 0.85)

        # Restart on stagnation
        if no_imp > restart_patience:
            scale = rng.choice([0.1, 0.15, 0.2, 0.3, 0.5])
            z = best_z + scale * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            f = softmax_normalize(z)
            c = compute_conv(f)
            M = float(c.max())
            beta = max(0.2 * M, beta_min)
            m[:] = 0
            v[:] = 0
            t_adam = 0
            lr = lr_init
            no_imp = 0

    return best_f, best_M


def coordinate_refine(f_init, deadline, max_rounds=80):
    """Coordinate-wise descent: adjust one element at a time."""
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    c = compute_conv(f)
    M = float(c.max())
    best_M = M
    best_f = f.copy()

    steps = [0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 3.0, 0.1, 0.01]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Shuffle order for diversity
        order = list(range(n))
        np.random.shuffle(order)
        for i in order:
            if time.monotonic() > deadline:
                break
            f_old = f[i]
            if f_old < 1e-15:
                continue
            for s in steps:
                f_try = f.copy()
                f_try[i] = f_old * s
                f_try = np.maximum(f_try, 0)
                ss = f_try.sum()
                if ss < 1e-10:
                    continue
                f_try = f_try / ss
                c_try = compute_conv(f_try)
                M_try = float(c_try.max())
                if M_try < M - 1e-14:
                    f = f_try
                    M = M_try
                    improved = True
                    break
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
        if not improved:
            break

    return best_f, best_M
