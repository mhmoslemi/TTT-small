s = softmax(x) = exp(x) / sum(exp(x))
ds/dx[i,j] = s[i] * (delta(i,j) - s[j])

The gradient of f(s) w.r.t. x is:
df/dx[j] = sum_i (df/ds[i]) * (ds[i]/dx[j])
= sum_i (df/ds[i]) * s[i] * (delta(i,j) - s[j])
= s[j] * df/ds[j] - s[j] * sum_i (df/ds[i]) * s[i]
= s[j] * (df/ds[j] - sum_i df/ds[i] * s[i])

So if g = df/ds (the gradient w.r.t. s), then:
df/dx[j] = s[j] * (g[j] - sum_i g[i]*s[i])
= s[j] * (g[j] - g·s)

This is clean! The gradient w.r.t. x is s * (g - (g·s) * 1).

"""
Minimize 2*n*max(conv(s,s))/(sum(s))^2 using L-BFGS-B optimization
of a p-norm surrogate in softmax parameterization, with LP refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def softmax(x):
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    return ex / ex.sum()

def pnorm_obj_and_grad(x, p):
    """
    Compute p-norm of conv(s,s) where s = softmax(x), and its gradient w.r.t. x.
    Returns (value, gradient).
    """
    s = softmax(x)
    n = len(s)
    conv = np.convolve(s, s, mode='full')
    conv = np.maximum(conv, 1e-30)
    
    # p-norm value
    log_conv = np.log(conv)
    plogc = p * log_conv
    plogc_max = plogc.max()
    log_val = (1.0/p) * (plogc_max + np.log(np.sum(np.exp(plogc - plogc_max))))
    val = np.exp(log_val)
    
    # Gradient w.r.t. s (direction only, ignoring positive scalar factor)
    # grad_s[i] ∝ sum_k conv_k^(p-1) * s[k-i]
    log_w = (p-1) * log_conv
    lw_max = log_w.max()
    w = np.exp(log_w - lw_max)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(w)) / len(w)
    else:
        w = w / w_sum
    
    # grad_s[i] = 2 * sum_k w[k] * s[k-i]
    # Compute via convolution:
    # sum_k w[k] * s[k-i] for i = 0..n-1
    # This is the cross-correlation of w and s
    conv_ws = np.convolve(s, w[::-1], mode='full')
    grad_s = 2.0 * conv_ws[len(w)-1:len(w)-1+n]
    
    # Gradient w.r.t. x: dx[j] = s[j] * (grad_s[j] - dot(grad_s, s))
    gs_s = np.dot(grad_s, s)
    grad_x = s * (grad_s - gs_s)
    
    return val, grad_x

def lp_refine(s, M, time_limit=5.0):
    """
    Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M, b >= 0.
    Returns b or None.
    """
    n = len(s)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j+n, j] = s
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-12:
                return b
    except Exception:
        pass
    return None

def optimize_lbfgsb(s_init, p_values, max_iter_per_p, deadline, rng, verbose=True):
    """
    Run L-BFGS-B optimization with increasing p.
    """
    n = len(s_init)
    s = s_init.copy()
    s = np.maximum(s, 1e-10)
    s = s / s.sum()
    
    best_s = s.copy()
    best_score = fast_score(s)
    
    # Initialize x from s: x = log(s) (approximately)
    # Actually, we want softmax(x) = s, so x = log(s) works (up to a constant)
    x = np.log(np.maximum(s, 1e-15))
    
    for p in p_values:
        if time.monotonic() > deadline:
            break
        if verbose:
            print(f"    L-BFGS-B with p={p}, current score={best_score:.6f}")
        
        def obj(x):
            val, grad = pnorm_obj_and_grad(x, p)
            return val, grad
        
        # Run L-BFGS-B
        result = minimize(obj, x, jac=True, method='L-BFGS-B',
                         options={'maxiter': max_iter_per_p, 'gtol': 1e-8,
                                  'maxls': 50})
        
        x = result.x
        s = softmax(x)
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        score = fast_score(s)
        
        if verbose:
            print(f"    After L-BFGS-B (p={p}): score={score:.6f}, "
                  f"func_calls={result.nfev}, success={result.success}")
        
        if score < best_score:
            best_score = score
            best_s = s.copy()
            print(f"    ** New best: {best_score:.6f}")
    
    return best_s, best_score

def gradient_polish(s, deadline, rng, max_iter=5000):
    """
    Gradient polishing with the exact max (subgradient method).
    """
    n = len(s)
    s = np.maximum(s, 0.0)
    s = s / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)
    
    eta = 0.05
    no_improve = 0
    patience = 200
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        max_conv = conv.max()
        # Find active lags (within 1% of max)
        active = conv >= 0.99 * max_conv
        
        # Subgradient: average of gradients at active lags
        # grad_s[i] = (1/|active|) * sum_{k in active} 2*s[k-i]
        grad_s = np.zeros(n)
        for k in np.where(active)[0]:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad_s[i] += 2.0 * s[ki]
        grad_s /= active.sum()
        
        # Project to tangent space
        grad_s = grad_s - grad_s.mean()
        
        # Backtracking line search
        alpha = eta
        cur_score = best_score  # use best score as reference
        for _ in range(10):
            s_new = np.maximum(s - alpha * grad_s, 0.0)
            s_new = s_new / s_new.sum()
            sc = fast_score(s_new)
            if sc < cur_score:
                break
            alpha *= 0.5
        
        s = s_new
        score = fast_score(s)
        
        if score < best_score - 1e-12:
            best_score = score
            best_s = s.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve == patience:
            eta *= 0.7
            no_improve = 0
            if eta < 0.003:
                s = np.maximum(s + 0.005 * rng.standard_normal(n), 0.0)
                s = s / s.sum()
                eta = 0.05
                no_improve = 0
    
    return best_s, best_score

def make_initializations(n, rng, hseq=None):
    """Generate diverse initializations."""
    inits = []
    
    # From hseq
    if hseq is not None and len(hseq) > 0:
        x_old = np.linspace(0.0, 1.0, len(hseq))
        x_new = np.linspace(0.0, 1.0, n)
        h_r = np.interp(x_new, x_old, np.maximum(hseq, 0))
        h_r = np.maximum(h_r, 0.01)
        h_r = h_r / h_r.sum()
        inits.append(h_r)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangular
    x = np.abs(np.arange(n) - (n-1)/2.0) / ((n-1)/2.0)
    s = 1.0 - x
    s = np.maximum(s, 0.01)
    s = s / s.sum()
    inits.append(s)
    
    # Exponential
    for alpha in [1.0, 2.0, -1.0, -2.0]:
        s = np.exp(alpha * np.arange(n) / (n-1))
        s = s / s.sum()
        inits.append(s)
    
    # Cosine
    x = np.arange(n) / (n-1)
    s = np.cos(np.pi * x / 2.0)
    s = np.maximum(s, 0.01)
    s = s / s.sum()
    inits.append(s)
    
    # Random (a few)
    for _ in range(3):
        s = rng.random(n) + 0.1
        s = s / s.sum()
        inits.append(s)
    
    return inits
