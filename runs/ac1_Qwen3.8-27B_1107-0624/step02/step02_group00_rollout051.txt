**Better approach for coordinate descent:**

Instead of ternary search (which requires many evaluations), I can find the optimal t analytically or with a simpler method.

The key observation: the score as a function of t = a[i] (with all other a[j] fixed) is:
score(t) = 2n * max_k c_new[k](t)

where c_new[k](t) is piecewise:
- c_new[k] = c[k] + 2*(t - old) * a[k-i] for k in [i, i+n-1], k ≠ 2i, k-i in [0,n-1]
- c_new[2i] = c[2i] + (t^2 - old^2) for 2i in [0, 2n-2]
- c_new[k] = c[k] for all other k

Let M_fixed = max of c_new[k] for k not in the affected range.
Let the affected terms be:
- Linear: L_k(t) = c[k] + 2*a[k-i]*(t - old) for k in affected, k ≠ 2i
- Quadratic: Q(t) = c[2i] + t^2 - old^2 (if 2i is in range)

The max of affected = max(max_k L_k(t), Q(t), (nothing else))
The overall max = max(M_fixed, max_affected(t))

Since max_affected(t) is convex (max of convex functions), the overall max is convex in t.

To find the minimum:
1. If max_affected(0) <= M_fixed: the score is M_fixed for all t where max_affected(t) <= M_fixed. We want to find the range of t where this holds, and pick any t in that range (e.g., t=0 to simplify).

2. If max_affected(t) > M_fixed for all t: we need to minimize max_affected(t). The minimum of a convex function can be found by finding where its subgradient changes sign.

"""
Coordinate descent + smooth-max gradient optimization for minimizing
2n * max(convolve(a,a)) / sum(a)^2.

Key innovations over previous approach:
1. Fast coordinate descent with incremental convolution updates and golden-section 1D subproblem solving.
2. Vectorized inner loops for speed.
3. Alternate between coordinate descent and gradient polish to escape local minima.
4. Multiple initializations with diverse profiles (sine, raised-cosine, parabolic, asymmetric).
5. Multi-length exploration (256-640) with interpolation from parent.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _smooth_max_val_weight(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_val_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _cd_score_at(t, i, old, a, c, k_lo, k_hi, n):
    """Compute max(c) after hypothetically setting a[i]=t (without modifying a or c)."""
    delta = t - old
    if abs(delta) < 1e-16:
        return float(np.maximum(c, 0).max())

    # Unaffected max
    if k_lo > 0:
        m_unaff_l = float(c[:k_lo].max())
    else:
        m_unaff_l = 0.0
    if k_hi < 2 * n - 2:
        m_unaff_r = float(c[k_hi + 1:].max())
    else:
        m_unaff_r = 0.0
    m_unaff = max(m_unaff_l, m_unaff_r)

    # Affected: vectorized
    ks = np.arange(k_lo, k_hi + 1)
    k_is = ks - i
    new_vals = c[ks].copy()
    new_vals += 2.0 * delta * a[k_is]
    if k_lo <= 2 * i <= k_hi:
        idx = 2 * i - k_lo
        new_vals[idx] = c[2 * i] + delta * (2.0 * old + delta)
    m_aff = float(new_vals.max())
    return max(m_unaff, m_aff)


def _coord_descent_optimize(a, deadline, max_sweeps=200000, seed=0):
    """
    Coordinate descent on normalized sequence (sum=1).
    Minimizes max(convolve(a,a)) via per-coordinate golden section search.
    """
    a = a.copy()
    n = len(a)
    a = _normalize(a)
    c = _fft_convolve(a)
    c = np.maximum(c, 0.0)

    best_a = a.copy()
    best_score = float(c.max())

    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate ~0.618

    no_imp_sweeps = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            old = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            if old < 1e-15:
                # Try a few small values
                best_t = 0.0
                best_f = _cd_score_at(0.0, i, 0.0, a, c, k_lo, k_hi, n)
                for t_try in [1e-4, 1e-3, 0.01 / n, 0.05 / n]:
                    f_try = _cd_score_at(t_try, i, 0.0, a, c, k_lo, k_hi, n)
                    if f_try < best_f:
                        best_f = f_try
                        best_t = t_try
                t_new = best_t
            else:
                # Golden section search on [0, min(1, 3*old)]
                lo = 0.0
                hi = min(1.0, 3.0 * old + 0.01)
                if hi <= lo:
                    hi = lo + 1e-6

                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f_c1 = _cd_score_at(c1, i, old, a, c, k_lo, k_hi, n)
                f_c2 = _cd_score_at(c2, i, old, a, c, k_lo, k_hi, n)

                for _ in range(25):
                    if f_c1 < f_c2:
                        hi = c2
                        c2 = c1
                        f_c2 = f_c1
                        c1 = hi - gr * (hi - lo)
                        f_c1 = _cd_score_at(c1, i, old, a, c, k_lo, k_hi, n)
                    else:
                        lo = c1
                        c1 = c2
                        f_c1 = f_c2
                        c2 = lo + gr * (hi - lo)
                        f_c2 = _cd_score_at(c2, i, old, a, c, k_lo, k_hi, n)

                t_new = (lo + hi) / 2.0
                f_new = _cd_score_at(t_new, i, old, a, c, k_lo, k_hi, n)

                # Also check boundaries
                f_0 = _cd_score_at(0.0, i, old, a, c, k_lo, k_hi, n)
                if f_0 < f_new:
                    t_new = 0.0
                    f_new = f_0

                # Only update if there's improvement
                f_old = _cd_score_at(old, i, old, a, c, k_lo, k_hi, n)
                if f_new >= f_old - 1e-15:
                    t_new = old

            # Apply update
            if abs(t_new - old) > 1e-16:
                delta = t_new - old
                a[i] = t_new
                # Update c incrementally
                for k in range(k_lo, k_hi + 1):
                    ki = k - i
                    if k == 2 * i:
                        c[k] += delta * (2.0 * old + delta)
                    else:
                        c[k] += 2.0 * delta * a[ki]  # a[k-i] unchanged
                # Ensure non-negativity of c (numerical)
                # Don't clip here to avoid bias; check at end

        # Periodic cleanup and check
        if sweep % 50 == 0:
            c = np.maximum(c, 0.0)
            s = a.sum()
            if s < 1e-10:
                a = np.ones(n) / n
                c = _fft_convolve(a)
                continue
            a = a / s
            c = _fft_convolve(a)
            c = np.maximum(c, 0.0)
            cur_score = float(c.max())
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_a = a.copy()
                sweep_improved = True
                no_imp_sweeps = 0
            else:
                no_imp_sweeps += 1
                if no_imp_sweeps > 100:
                    a = best_a.copy()
                    c = _fft_convolve(a)
                    c = np.maximum(c, 0.0)
                    no_imp_sweeps = 0
                    # Perturb to escape
                    a = a * (1.0 + 0.05 * rng.normal(size=n))
                    a = _normalize(a)
                    c = _fft_convolve(a)
                    c = np.maximum(c, 0.0)

    return best_a, best_score


def _gradient_polish(a, deadline, max_iter=50000, lr=0.05):
    """Smooth-max gradient descent in logit space (simplex param)."""
    a = a.copy()
    n = len(a)
    a = _normalize(a)

    z = np.log(np.maximum(a, 1e-30))
    z -= float(np.mean(z))

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    best_z = z.copy()

    beta = max(1e-10, 0.05 * float(_fft_convolve(a).max()))
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = _normalize(np.exp(z + momentum * v))
        grad, maxc = _gradient_smooth_max(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            z = best_z.copy()
            v = np.zeros(n)
            beta = max(1e-10, beta * 2.0)
            continue

        # Gradient in logit space: a_i * (g_i - sum(a_j * g_j))
        dot = float(np.dot(a_nest, grad))
        gz = a_nest * (grad - dot)
        gz -= float(np.mean(gz))

        v = momentum * v - lr * gz
        z_new = z + v
        z_new = np.clip(z_new, -50.0, 50.0)
        a_new = _normalize(np.exp(z_new))

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.005, lr * 0.95)
            if no_imp > max(2000, 4 * n):
                z = best_z.copy() + 0.3 * np.random.default_rng(it).normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                a = _normalize(np.exp(z))
                v = np.zeros(n)
                beta = max(1e-10, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        z = z_new
        a = a_new

        if it % 300 == 0:
            beta = max(1e-12, beta * 0.7)

    return best_a, best_score


def _make_sine_profile(n):
    t = (np.arange(n) + 0.5) / n
    a = np.sin(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raises_cos_profile(n):
    t = np.arange(n) / n
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_parabolic_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_asymmetric_profile(n, skew=2.0, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    center = 0.5 + 0.2 * skew / (1.0 + abs(skew))
    a = np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    if n0 == 0:
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _cd_then_gradient(a, deadline_cd, deadline_grad, seed=0):
    """Run coordinate descent then gradient polish."""
    a_cd, sc_cd = _coord_descent_optimize(a, deadline_cd, max_sweeps=200000, seed=seed)
    if time.monotonic() < deadline_grad - 5:
        a_gp, sc_gp = _gradient_polish(a_cd, deadline_grad, max_iter=30000, lr=0.04)
        if sc_gp < sc_cd:
            return a_gp, sc_gp
    return a_cd, sc_cd

    across multiple initializations and sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(min(parent.size, 640))
    for ln in [256, 320, 384, 448, 512, 576, 640]:
        if ln not in lengths:
            lengths.append(ln)

    # Sort by expected difficulty (shorter first for quick results)
    lengths.sort()

    # Generate initializations
    def get_inits(n, parent_arr, rng_local):
        inits = []
        if parent_arr is not None:
            inits.append(_interpolate_profile(parent_arr, n))
            for _ in range(3):
                pert = _interpolate_profile(parent_arr, n) * (1.0 + 0.1 * rng_local.normal(size=n))
                inits.append(_normalize(pert))
        inits.append(_make_sine_profile(n))
        inits.append(_make_raises_cos_profile(n))
        inits.append(_make_parabolic_profile(n))
        inits.append(_make_linear_profile(n, 1.0))
        inits.append(_make_linear_profile(n, 2.0))
        inits.append(_make_asymmetric_profile(n, 2.0, 0.3))
        inits.append(_make_asymmetric_profile(n, -2.0, 0.3))
        inits.append(_make_asymmetric_profile(n, 3.0, 0.25))
        # Random smooth
        for _ in range(2):
            z = rng_local.normal(size=n) * 0.5
            # Smooth it
            for _s in range(3):
                z = 0.5 * z + 0.25 * np.roll(z, 1) + 0.25 * np.roll(z, -1)
            a_r = _normalize(np.exp(z))
            inits.append(a_r)
        return inits

    # Phase 1: Coordinate descent on multiple inits
    time_per_len = (deadline - t_start - 60.0) / len(lengths)
    print(f"Time budget per length: {time_per_len:.1f}s, n_lengths={len(lengths)}")

    for li, n_t in enumerate(lengths):
        len_start = time.monotonic()
        len_deadline = min(deadline - 60.0, len_start + time_per_len)

        if li >= len(lengths) - 2 and time.monotonic() > deadline - 120:
            break

        inits = get_inits(n_t, parent, rng)

        for init_idx, a0 in enumerate(inits):
            if time.monotonic() > len_deadline:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Coordinate descent
            cd_deadline = len_deadline - 10.0
            a_cd, sc_cd = _coord_descent_optimize(a0, cd_deadline, max_sweeps=100000, seed=seed + li * 100 + init_idx)
            sc_cd_full = sc_cd * 2 * n_t  # convert back to full score
            if np.isfinite(sc_cd_full) and sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
            print(f"  n={n_t}, init={init_idx}: CD score={sc_cd_full:.10f} (time={time.monotonic()-t_start:.0f}s)")

    # Phase 2: Gradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"Phase 2: Gradient polish on best (score={best_score:.10f})")
        gp_deadline = min(deadline - 20.0, time.monotonic() + 180.0)
        a_gp, sc_gp_norm = _gradient_polish(best_seq, gp_deadline, max_iter=50000, lr=0.04)
        sc_gp = float(2.0 * len(a_gp) * sc_gp_norm)
        if np.isfinite(sc_gp) and sc_gp < best_score:
            best_score = sc_gp
            best_seq = a_gp.copy()
        print(f"  Gradient polish result: {sc_gp:.10f}")

    # Phase 3: Another round of CD on the best (with perturbation)
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"Phase 3: CD refinement round 2")
        cd_deadline2 = min(deadline - 15.0, time.monotonic() + 120.0)
        # Slightly perturb
        a_pert = best_seq * (1.0 + 0.03 * rng.normal(size=len(best_seq)))
        a_pert = _normalize(a_pert)
        a_cd2, sc_cd2_norm = _coord_descent_optimize(a_pert, cd_deadline2, max_sweeps=100000, seed=seed + 9999)
        sc_cd2 = float(2.0 * len(a_cd2) * sc_cd2_norm)
        if np.isfinite(sc_cd2) and sc_cd2 < best_score:
            best_score = sc_cd2
            best_seq = a_cd2.copy()
        print(f"  CD round 2: {sc_cd2:.10f}")

    # Phase 4: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        gp_deadline2 = min(deadline - 5.0, time.monotonic() + 60.0)
        a_final, sc_final_norm = _gradient_polish(best_seq, gp_deadline2, max_iter=20000, lr=0.03)
        sc_final = float(2.0 * len(a_final) * sc_final_norm)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        print(f"  Final gradient polish: {sc_final:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_sine_profile(256)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_sine_profile(256)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
