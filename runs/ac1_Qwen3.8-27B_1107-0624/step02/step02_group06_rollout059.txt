"""
Improved optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-K peak focused gradient in logit space with adaptive stepping,
multi-scale search, and block coordinate polish.
"""
import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_gradient(weights, a):
    """Compute gradient of sum_k weights[k]*conv(a,a)[k] w.r.t. a.
    Returns a vector of same shape as a."""
    n = len(a)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(weights, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    return 2.0 * h[n - 1:2 * n - 1]


def topk_weights(c, k=3):
    """Create weights concentrated on top-k peaks."""
    c = np.asarray(c, dtype=np.float64)
    m = len(c)
    k = min(k, m)
    idx = np.partition(c, m-k)[m-k:]
    idx = np.sort(idx)[::-1]  # top k indices
    w = np.zeros_like(c)
    # Weight by how much each peak exceeds the second-highest
    # Simple: equal weight on top k
    w[idx] = 1.0 / k
    return w


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-30 or not np.isfinite(s):
        return None
    return a / s


def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def true_score(a):
    """Exact score using numpy convolve."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def make_profiles(n, rng):
    """Generate diverse initial profiles."""
    profiles = []
    t = np.linspace(0, 1, n)
    
    # Linear
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    # Exponential
    for eta in [1.0, 2.0, 4.0, 8.0]:
        a = np.exp(eta * t)
        profiles.append(a / a.sum())
    
    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = t**p + 1e-10
        profiles.append(a / a.sum())
    
    # Semicircle (Wigner semicircle)
    a = np.sqrt(np.maximum(1.0 - (2*t - 1)**2, 0.0))
    if a.sum() > 0:
        profiles.append(a / a.sum())
    
    # Gaussian-like
    for sigma in [0.15, 0.2, 0.3, 0.4]:
        a = np.exp(-((t - 0.5)**2) / (2 * sigma**2))
        profiles.append(a / a.sum())
    
    # Two-hump
    for center in [0.3, 0.35, 0.4]:
        a = np.exp(-((t - center)**2) / 0.02) + np.exp(-((t - (1-center))**2) / 0.02)
        profiles.append(a / a.sum())
    
    # Cosine window
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())
    
    # Hamming-like
    a = 0.54 + 0.46 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())
    
    # Random smooth
    for _ in range(10):
        coeffs = rng.normal(0, 1, 8)
        a = np.sum(coeffs[j] * np.cos(2 * np.pi * j * t) for j in range(len(coeffs)))
        a = np.abs(a) + 0.5
        profiles.append(a / a.sum())
    
    return profiles


def optimize_topk(a0, deadline, rng, lr=0.05, topk=3, restart_interval=500):
    """Optimize using top-K peak focused gradient in logit space with Adam."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    n = a0.size
    
    z = np.log(a0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    a = a0.copy()
    c = full_convolve_fft(a)
    best_z = z.copy()
    best_a = a0.copy()
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        # Compute top-k weights
        w = topk_weights(c, topk)
        
        # Gradient
        gp = compute_gradient(w, a)
        # Project to simplex tangent space
        vp = gp + 1e-8 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t = 0
            continue
        
        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t = 0
            continue
        
        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        a = softmax_from_z(z)
        c = full_convolve_fft(a)
        score = float(2.0 * n * c.max())
        
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.01, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0); v.fill(0); t = 0
            continue
        
        # Restart if stuck
        if no_imp > restart_interval:
            # Perturb around best
            z = best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            a = softmax_from_z(z)
            c = full_convolve_fft(a)
            m.fill(0); v.fill(0); t = 0
            lr = base_lr
            no_imp = 0
    
    return best_a, best_score


def optimize_softmax(a0, deadline, rng, lr=0.08, beta_frac=0.15):

    a0 = a0 / a0.sum()

    c = full_convolve_fft(a)

def block_coordinate_polish(a, deadline, rng, block_size=8, n_sweeps=None):
    """Polish using block coordinate descent."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    
    if n_sweeps is None:
        n_sweeps = max(5, int((deadline - time.monotonic()) * 3))
    
    best_a = a.copy()
    best_score = internal_score(a)
    no_imp = 0
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random block order
        blocks = list(range(0, n, block_size))
        rng.shuffle(blocks)
        
        for start in blocks:
            if time.monotonic() > deadline:
                break
            end = min(start + block_size, n)
            block = a[start:end].copy()
            block_sum = float(block.sum())
            if block_sum < 1e-20:
                continue
            
            # Try various redistributions within the block
            best_block = block.copy()
            best_sc = best_score
            
            # Generate candidate redistributions
            t_block = np.linspace(0, 1, end - start)
            candidates = [block.copy()]
            
            # Uniform
            candidates.append(np.full(end - start, block_sum / (end - start)))
            
            # Linear
            for slope in [-2, -1, 0.5, 1, 2]:
                c = 1 + slope * t_block
                c = np.maximum(c, 0.01)
                candidates.append(block_sum * c / c.sum())
            
            # Convex/concave
            for p in [0.5, 2.0, 3.0]:
                c = t_block**p + 0.1
                candidates.append(block_sum * c / c.sum())
            
            # Gaussian bump
            for center in [0.3, 0.5, 0.7]:
                c = np.exp(-((t_block - center)**2) / 0.1) + 0.1
                candidates.append(block_sum * c / c.sum())
            
            # Keep original
            # Try each candidate
            for cand in candidates:
                cand = np.maximum(cand, 1e-15)
                # Scale to maintain block sum
                cand_sum = float(cand.sum())
                if cand_sum < 1e-20:
                    continue
                cand = cand * (block_sum / cand_sum)
                
                new_a = a.copy()
                new_a[start:end] = cand
                new_a = new_a / new_a.sum()
                sc = internal_score(new_a)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_block = cand
            
            a[start:end] = best_block
            a = a / a.sum()
            
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        
        if not improved and sweep > 3:
            if no_imp > n // block_size:
                break
    
    return best_a, best_score


def interpolate_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        return normalize_nonneg(p)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return normalize_nonneg(np.interp(x1, x0, p))

    n = len(a)
    N = next_pow2(max(2*n-1, len(w)))
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # Cross-correlation: sum_m a[m]*w[m+j]
    # = IFFT(conj(fa) * fw)[j] for j = 0, ..., n-1
    corr = np.fft.irfft(np.conj(fa) * fw, N)
    R = corr[:n]  # First n elements
    # Correction term
    correction = np.zeros(n)
    for j in range(n):
        if 2*j < len(w):
            correction[j] = w[2*j] * a[j]
    return 2 * R - correction
