def fft_convolve(a, b):
    """Fast convolution using FFT."""
    n, m = len(a), len(b)
    N = 1
    while N < n + m - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:n+m-1]

def self_convolve(a):
    """Compute a*a (autocorrelation)."""
    n = len(a)
    if n <= 64:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*n-1]

def build_lp_matrix(p0, c0):
    """Build the LP matrix A such that (A @ p - t) <= c0."""
    n = len(p0)
    # A has shape (2n-1, n), where A[k, i] = 2*p0[k-i] if valid
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = k - np.arange(lo, hi + 1)
        A[k, lo:hi+1] = 2.0 * p0[idx]
    return A

def solve_lp(p0):
    """Solve the linearized QCP LP. Returns p_new."""
    n = len(p0)
    c0 = self_convolve(p0)
    A = build_lp_matrix(p0, c0)
    
    # Variables: x = [p0, ..., p_{n-1}, t]
    # Minimize: t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A @ p - t <= c0
    # In matrix form: [A, -1] @ x <= c0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = c0.copy()
    
    # Equality: sum(p) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: p[i] >= 0, t >= 0 (t should be >= c0.max() > 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

        c_try = self_convolve(p_try)
        if c_try.max() <= M_max + 1e-10:
            lo = mid
        else:

    s = float(f.sum())
    if s < 0.01:
        return np.inf
    c = self_convolve_fft(f)
    return float(2.0 * n * c.max() / (s * s))

def build_lp(A_base, c0, n):
    """Build LP for linearized QCP.
    A_base: (2n-1, n) matrix where A_base[k,i] = p0[k-i] (not 2*p0[k-i])
    We want: 2 * A_base @ p <= t + c0
    In standard form: 2*A_base @ p - t <= c0
    Variables: [p (n), t (1)]
    """
    # A_ub shape: (2n-1, n+1)
    A_ub = np.hstack([2.0 * A_base, -np.ones((2*n-1, 1))])
    b_ub = c0.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0, None)] * n + [(0, None)]
    return A_ub, b_ub, A_eq, b_eq, c_obj, bounds

def solve_linearized_lp(p):
    """Solve the linearized QCP LP at point p."""
    n = len(p)
    c0 = self_convolve_fft(p)
    
    # Build A_base: A_base[k, i] = p[k-i] if 0 <= k-i <= n-1
    A_base = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_base[k, lo:hi+1] = p[k - np.arange(lo, hi+1)]
    
    A_ub, b_ub, A_eq, b_eq, c_obj, bounds = build_lp(A_base, c0, n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': 2.0, 'presolve': True})
    if res.success and res.x is not None:
        p_new = np.maximum(res.x[:n], 0)
        s = p_new.sum()
        if s > 1e-15:
            return p_new / s
    return None

def line_search_alpha(p, p_new, M):
    """Find largest alpha in [0,1] s.t. max(self_conv(p + alpha*(p_new-p))) <= M + 1e-9."""
    d = p_new - p
    # Check alpha=1
    p1 = (1.0 - 1.0) * p + 1.0 * p_new  # = p_new
    c1 = self_convolve_fft(p1)
    if c1.max() <= M + 1e-9:
        return 1.0
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        pm = p + mid * d
        pm = np.maximum(pm, 0)
        sm = pm.sum()
        if sm < 1e-15:
            hi = mid
            continue
        pm = pm / sm
        cm = self_convolve_fft(pm)
        if cm.max() <= M + 1e-9:
            lo = mid
        else:
            hi = mid
    return lo

def optimize_lp(p_init, deadline, rng, max_iters=50000):
    """Main LP-based optimization."""
    p = np.maximum(p_init, 0).astype(np.float64)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)
    
    c = self_convolve_fft(p)
    M = float(c.max())
    best_p = p.copy()
    best_M = M
    stall = 0
    lp_fail_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Solve LP
        p_new = solve_linearized_lp(p)
        if p_new is None:
            lp_fail_count += 1
            if lp_fail_count > 10:
                break
            # Perturb and retry
            p = p + 0.01 * rng.standard_normal(n)
            p = np.maximum(p, 0)
            p = p / p.sum()
            continue
        lp_fail_count = 0
        
        # Check if meaningful direction
        diff = np.max(np.abs(p_new - p))
        if diff < 1e-13:
            stall += 1
            if stall > 50:
                break
            # Perturb
            p = p + 0.005 * rng.standard_normal(n)
            p = np.maximum(p, 0)
            p = p / p.sum()
            continue
        
        # Line search
        c = self_convolve_fft(p)
        M = float(c.max())
        alpha = line_search_alpha(p, p_new, M)
        
        if alpha < 1e-12:
            stall += 1
            if stall > 50:
                break
            p = p + 0.003 * rng.standard_normal(n)
            p = np.maximum(p, 0)
            p = p / p.sum()
            continue
        
        # Update
        p = (1.0 - alpha) * p + alpha * p_new
        c = self_convolve_fft(p)
        M_new = float(c.max())
        
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_p = p.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Bigger perturbation
                perturbation = rng.standard_normal(n) * 0.1
                p = best_p + perturbation
                p = np.maximum(p, 0)
                p = p / p.sum()
                stall = 0
    
    return best_p, best_M

"""
Optimization of step function to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses successive linear programming (linearized QCP) with line search,
Adam polish, and multi-start multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def self_conv(a):
    """Auto-convolution a*a via FFT for n>64, direct otherwise."""
    n = len(a)
    if n <= 64:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def score(f):
    """Compute 2n*max(conv)/sum^2."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-15:
        return np.inf
    c = self_conv(f)
    return float(2.0 * n * c.max() / (s * s))


def make_conv_grad_matrix(p):
    """Build matrix G where G[k,i] = 2*p[k-i] if 0<=k-i<=n-1 else 0.
    Then G @ p = 2*c (where c = p*p, the autocorrelation)."""
    n = len(p)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        G[k, lo:hi + 1] = 2.0 * p[k - np.arange(lo, hi + 1)]
    return G


def solve_lp_linearized(p):
    """Solve: min t s.t. G @ p_new <= t + c0, sum(p_new)=1, p_new >= 0.
    Returns p_new or None."""
    n = len(p)
    c0 = self_conv(p)
    G = make_conv_grad_matrix(p)
    
    # Variables: [p_new (n), t (1)]
    # Constraint: G @ p_new - t <= c0  =>  [G, -1] @ [p_new, t] <= c0
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = c0.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            p_new = np.maximum(res.x[:n], 0.0)
            s = p_new.sum()
            if s > 1e-14:
                return p_new / s
    except Exception:
        pass
    return None


def line_search(p, p_new, M_tol):
    """Find max alpha in [0,1] s.t. max(self_conv(p + alpha*(p_new-p))) <= M_tol."""
    d = p_new - p
    # Test alpha = 1
    c1 = self_conv(p_new)
    if c1.max() <= M_tol:
        return 1.0
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(35):
        mid = 0.5 * (lo + hi)
        pm = p + mid * d
        pm = np.maximum(pm, 0.0)
        sm = pm.sum()
        if sm < 1e-14:
            hi = mid
            continue
        pm = pm / sm
        cm = self_conv(pm)
        if cm.max() <= M_tol:
            lo = mid
        else:
            hi = mid
    return lo


def lp_optimize(p_init, deadline, rng, max_iters=200000):
    """Successive LP optimization."""
    p = np.maximum(p_init, 0.0).astype(np.float64)
    s = p.sum()
    if s < 1e-14:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)
    
    c = self_conv(p)
    M = float(c.max())
    best_p = p.copy()
    best_M = M
    stall = 0
    n_iters = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        p_new = solve_lp_linearized(p)
        if p_new is None:
            stall += 1
            if stall > 20:
                break
            p = p + 0.01 * rng.standard_normal(n)
            p = np.maximum(p, 0.0)
            p = p / p.sum()
            continue
        
        diff = np.max(np.abs(p_new - p))
        if diff < 1e-14:
            stall += 1
            if stall > 100:
                break
            if stall % 10 == 0:
                scale = min(0.05, 0.01 * stall / 10)
                p = p + scale * rng.standard_normal(n)
                p = np.maximum(p, 0.0)
                p = p / p.sum()
            continue
        
        M_cur = float(self_conv(p).max())
        alpha = line_search(p, p_new, M_cur + 1e-12)
        
        if alpha < 1e-14:
            stall += 1
            if stall > 50:
                break
            p = p + 0.005 * rng.standard_normal(n)
            p = np.maximum(p, 0.0)
            p = p / p.sum()
            continue
        
        p = (1.0 - alpha) * p + alpha * p_new
        c = self_conv(p)
        M_new = float(c.max())
        n_iters += 1
        
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_p = p.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                # Aggressive restart
                p = 0.85 * best_p + 0.15 * rng.dirichlet(np.ones(n) * 0.5)
                stall = 0
    
    return best_p, best_M, n_iters


def softmax_grad(c, beta):
    """Compute soft-max and its gradient weights."""
    m = c.max()
    x = (c - m) / beta
    e = np.exp(x)
    w = e / e.sum()
    return m + beta * np.log(e.sum()), w


def adam_polish(p_init, deadline, rng, lr=0.1, max_iters=50000):
    """Adam polish using soft-max gradient."""
    p = np.maximum(p_init, 1e-12).astype(np.float64)
    p = p / p.sum()
    n = len(p)
    z = np.log(p)
    z -= z.mean()
    
    m_ = np.zeros(n)
    v_ = np.zeros(n)
    t_ = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c = self_conv(p)
    M = float(c.max())
    beta = 0.1 * M + 1e-10
    
    best_p = p.copy()
    best_M = M
    best_z = z.copy()
    stall = 0
    base_lr = lr
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        L, w = softmax_grad(c, beta)
        # Gradient of max(c) w.r.t. p[j] = 2 * sum_i w[i+j] * p[i] (approximately)
        # More precisely: d/dp[j] sum_k w[k] * c[k] where c = p*p
        # = sum_k w[k] * 2*p[k-j] (for valid k-j)
        # = 2 * (w_reversed_correlated with p)
        
        # Compute h[j] = sum_k w[k] * p[k-j] for j = 0, ..., n-1
        # This is a cross-correlation of w and p
        # h[j] = sum_k w[k] * p[k-j] where k-j must be in [0, n-1]
        # Using FFT:
        N = 1
        while N < 2 * n - 1 + len(w) - 1:
            N <<= 1
        # Actually, let me compute h[j] = sum_{k} w[k] * p[k-j] for j=0..n-1
        # k ranges such that 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
        # h[j] = sum_{k=j}^{j+n-1} w[k] * p[k-j]
        # This is a valid cross-correlation
        
        # Simpler: h = correlate(w, p) shifted
        # h[j] = sum_k w[k] * p[k-j]
        # Let's use: if we define w_ext of length n with w_ext[j] = w[j] for valid j
        # Then h[j] = sum_k w_ext[k] * p[k-j]
        
        # Actually the cleanest way:
        # h[j] = sum_k w[k] * p[k-j] (k from 0 to 2n-2, p index from 0 to n-1)
        # = cross-correlation
        # h = np.correlate(w, p, mode='valid') won't work directly because lengths differ
        
        # Let me just do it with the conv matrix approach:
        # h[j] = sum_k w[k] * p[k-j]
        # For fixed j: k ranges from j to j+n-1 (so that k-j is in [0,n-1])
        # h[j] = sum_{k=j}^{j+n-1} w[k] * p[k-j]
        
        # Using FFT: h = ifft(fft(w, 2n-1) * conj(fft(p, 2n-1))) ... no that's not right either.
        
        # Let me just use the matrix approach for clarity (n is at most 512):
        # h[j] = sum_k w[k] * p[k-j] for k where 0<=k-j<=n-1, i.e., j<=k<=j+n-1
        # = sum_i w[j+i] * p[i] for i=0..n-1 (where j+i <= 2n-2)
        
        # So h[j] = sum_{i=0}^{min(n-1, 2n-2-j)} w[j+i] * p[i]
        
        # This is: h = W @ p where W[j,i] = w[j+i] if j+i <= 2n-2, else 0
        # W is a (n x n) upper-triangular-like matrix
        
        # For efficiency, use FFT:
        # h[j] = sum_i w[j+i] * p[i]
        # Let w_flip[i] = w[n-1-i] (reverse w, but w has length 2n-1 so this doesn't quite work)
        
        # Actually, let me just compute it directly for small n, or use the following trick:
        # h[j] = sum_i p[i] * w[j+i]
        # Define w_pad of length 2n-1 (already is)
        # h = cross-correlation of p and w, specifically h[j] = sum_i p[i] * w[j+i]
        # This is the "valid" cross-correlation with a specific shift
        
        # Using np.correlate: np.correlate(w, p, mode='full') gives
        # full[j] = sum_i w[i+j] * p[i] (with appropriate indexing)
        # Actually np.correlate(w, p, 'full')[j] = sum_i w[i] * p[i-j+something]
        
        # Let me just compute it the direct way for now (n <= 512, so n^2 = 262144, fast enough)
        h = np.zeros(n)
        for j in range(n):
            # h[j] = sum_i p[i] * w[j+i] where j+i <= 2n-2
            max_i = min(n-1, 2*n-2-j)
            if max_i >= 0:
                h[j] = np.dot(p[:max_i+1], w[j:j+max_i+1])
        
        # Gradient of L w.r.t. p[j] = 2 * h[j] (from the chain rule)
        # But we need to be careful: we're optimizing in log space z, and p = softmax(z) (approximately)
        # dp/dz[j] = p[j] * (delta[j,l] - p[l]) for each l
        # So grad_z[l] = sum_j (dL/dp[j]) * (dp[j]/dz[l]) = sum_j 2*h[j] * p[j] * (delta[j,l] - p[l])
        # = 2*p[l]*h[l] - 2*p[l]*sum_j h[j]*p[j]
        # = 2*p[l]*(h[l] - sum_j h[j]*p[j])
        
        grad_p = 2.0 * h
        dot = float(np.dot(p, grad_p))
        grad_z = p * (grad_p - dot)
        grad_z -= grad_z.mean()  # Remove mean for invariance
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m_ = np.zeros(n)
            v_ = np.zeros(n)
            t_ = 0
            continue
        
        t_ += 1
        m_ = b1 * m_ + (1 - b1) * grad_z
        v_ = b2 * v_ + (1 - b2) * (grad_z * grad_z)
        mhat = m_ / (1 - b1 ** t_)
        vhat = v_ / (1 - b2 ** t_)
        z = z - lr * mhat / (np.sqrt(vhat) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        p = np.exp(z)
        p = p / p.sum()
        
        c = self_conv(p)
        M = float(c.max())
        
        if M < best_M - 1e-13:
            best_M = M
            best_p = p.copy()
            best_z = z.copy()
            stall = 0
        else:
            stall += 1
            if stall % 200 == 0:
                lr = max(0.01, lr * 0.9)
        
        if t_ % 100 == 0:
            beta = max(1e-12, beta * 0.85)
        
        if stall > max(200, 2 * n):
            z = 0.8 * best_z + 0.3 * rng.standard_normal(n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p = np.exp(z)
            p = p / p.sum()
            c = self_conv(p)
            M = float(c.max())
            beta = 0.2 * M
            m_ = np.zeros(n)
            v_ = np.zeros(n)
            t_ = 0
            lr = base_lr
            stall = 0
    
    return best_p, best_M


def make_initializations(n, rng, count=10):
    """Generate diverse initializations."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Linear
    t = np.arange(n, dtype=np.float64)
    for slope in [0.5, 1.0, 2.0, 3.0]:
        f = 1.0 + slope * t / (n - 1)
        inits.append(f / f.sum())
    # Reverse linear
    for slope in [1.0, 2.0]:
        f = 1.0 + slope * (1.0 - t / (n - 1))
        inits.append(f / f.sum())
    # Quadratic (convex)
    u
