import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max_and_weights(c, beta):
    """Compute soft max value and weights."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_grad_fft(a, w):
    """Compute gradient of soft_max(conv(a,a)) w.r.t. a using FFT.
    Returns grad[i] = dL/da[i] (factor of 2 included)."""
    n = a.size
    N = next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w[::-1], N)
    full_conv = np.fft.irfft(A * W, N)[:3 * n - 2]
    h = full_conv[n - 1:2 * n - 1][::-1]
    return 2.0 * h


def softmax_stable(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z) / z.size
    return e / s


def generate_profiles(n, rng):
    """Generate diverse initial profiles at length n."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    # Power-law decreasing: (1 - i/n + eps)^p
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 7.0, 10.0]:
        a = (1.0 - t + 1e-6) ** p
        a = np.maximum(a, 1e-12)
        profiles.append(("pow_d_%s" % p, a / a.sum()))
    # Power-law increasing
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = (t + 1e-6) ** p
        a = np.maximum(a, 1e-12)
        profiles.append(("pow_i_%s" % p, a / a.sum()))
    # Exponential decay
    for gamma in [0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0, 20.0]:
        a = np.exp(-gamma * t)
        profiles.append(("exp_%s" % gamma, a / a.sum()))
    # Rational decay
    for gamma in [1.0, 3.0, 5.0, 10.0, 20.0]:
        a = 1.0 / (1.0 + gamma * t)
        profiles.append(("rat_%s" % gamma, a / a.sum()))
    # Gaussian centered left
    for sigma in [0.2, 0.3, 0.4, 0.5]:
        a = np.exp(-((t - 0.2) ** 2) / (2 * sigma ** 2))
        a = np.maximum(a, 1e-12)
        profiles.append(("gauss_%s" % sigma, a / a.sum()))
    # Cosine taper
    for k in [1, 2, 3]:
        a = 0.5 + 0.5 * np.cos(np.pi * k * t)
        a = np.maximum(a, 1e-12)
        profiles.append(("cos_%d" % k, a / a.sum()))
    # Linear decreasing
    for slope in [1.0, 2.0, 3.0, 5.0]:
        a = 1.0 + slope * (1.0 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(("lin_%s" % slope, a / a.sum()))
    # Mixed: decreasing with bump
    for peak_pos in [0.1, 0.2, 0.3]:
        for width in [0.1, 0.2, 0.3]:
            a = np.exp(-((t - peak_pos) ** 2) / (2 * width ** 2)) + 0.1 * np.exp(-t * 2)
            a = np.maximum(a, 1e-12)
            profiles.append(("mix_%s_%s" % (peak_pos, width), a / a.sum()))
    # Random smooth (low-frequency Fourier)
    for seed_idx in range(8):
        coeffs = rng.uniform(-0.6, 0.6, 6)
        a = np.ones(n)
        for i in range(1, 6):
            a = a + coeffs[i] * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(("rand_%d" % seed_idx, a / a.sum()))
    return profiles


def logit_adam_optimize(p0, deadline, rng, lr=0.05, beta_init=None, beta_decay=0.995,
                        beta_min=1e-10, restart_every=2000, verbose=False):
    """Log-space Adam optimization with adaptive soft-max temperature."""
    n = p0.size
    p = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 1e-10)
    p = p / p.sum()

    # Initialize log-parameters
    z = np.log(p)
    z -= float(np.mean(z))

    if beta_init is None:
        c0 = full_convolve_fft(p)
        beta = max(1e-6, 0.3 * float(c0.max()))
    else:
        beta = beta_init

    b1, b2, eps = 0.90, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = full_convolve_fft(p)
    maxc = float(c.max())
    best_p = p.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    step_count = 0

    while time.monotonic() < deadline:
        L, w = soft_max_and_weights(c, beta)
        gp = compute_grad_fft(p, w)
        # Project to tangent space of simplex
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_stable(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                # Adaptive: if stuck, reduce LR
                if no_imp % 500 == 0:
                    lr = max(0.01, lr * 0.97)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        step_count += 1

        # Temperature schedule: decay gradually, but accelerate if making progress
        if no_imp == 0:
            beta *= beta_decay
        else:
            beta *= (beta_decay + 0.001 * min(no_imp / 1000.0, 1.0))
        beta = max(beta_min, min(beta, 0.5 * maxc + 1e-8))

        # Restart if very stuck
        if no_imp > restart_every:
            z = 0.85 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_stable(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-6, 0.3 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = max(lr, 0.04)
            no_imp = 0
            if verbose:
                print("    restart at step %d, score=%.10f" % (step_count, best_score))

    return best_p, best_score


def interpolate_profile(p0, n_new):
    """Linear interpolation of profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def coordinate_descent_refine(p, deadline, rng, n_steps=5000, verbose=False):
    """Local refinement using random coordinate perturbations."""
    n = p.size
    p = np.maximum(p, 1e-12).copy()
    p = p / p.sum()
    c = np.convolve(p, p)
    best_score = float(2.0 * n * c.max())
    best_p = p.copy()
    accepted = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Pick 1-3 random coordinates to perturb
        n_pert = rng.integers(1, 4)
        idxs = rng.choice(n, size=n_pert, replace=False)
        p_new = p.copy()
        for idx in idxs:
            delta = rng.uniform(-0.05, 0.05) * max(p[idx], 1e-4)
            p_new[idx] = max(0.0, p[idx] + delta)
        s = p_new.sum()
        if s < 1e-6:
            continue
        p_new = p_new / s
        c_new = np.convolve(p_new, p_new)
        sc = float(2.0 * n * c_new.max())
        if sc < best_score - 1e-14:
            best_score = sc
            best_p = p_new.copy()
            p = p_new
            accepted += 1
        elif verbose and i % 500 == 0:
            pass

    if verbose:
        print("    coord descent: %d accepted, score=%.10f" % (accepted, best_score))
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/sum(a)^2.
    Phase 1: Diverse profile generation and scoring at n=512.
    Phase 2: Log-space Adam optimization from top profiles.
    Phase 3: Extended refinement with multiple hyperparameter settings.
    Phase 4: Upsample to n=1024 and optimize.
    Phase 5: Final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    best_n = 0

    def update(p, sc):
        nonlocal best_seq, best_score, best_n
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            best_n = best_seq.size

    # Try to get initial sequence from global
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print("Starting. budget=%ds" % int(budget_s))
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print("Initial from height_sequence_1: n=%d, score=%.10f" % (init_seq.size, sc_init))
        update(init_seq, sc_init)

    # ============ Phase 1: Profile generation and scoring ============
    n_main = 512
    phase1_end = min(deadline, t_start + 120.0)

    # Generate profiles
    profiles = generate_profiles(n_main, rng)
    # Add the initial sequence if available
    if init_seq is not None:
        if init_seq.size != n_main:
            p_interp = interpolate_profile(init_seq, n_main)
        else:
            p_interp = init_seq.copy()
        profiles.append(("init", p_interp))

    # Score all profiles
    scored = []
    for name, p in profiles:
        if time.monotonic() > phase1_end:
            break
        sc = compute_score(p)
        scored.append((sc, name, p))
        update(p, sc)

    scored.sort(key=lambda x: x[0])
    print("Phase 1: %d profiles scored. Best: %s (%.10f)" % (len(scored), scored[0][1], scored[0][0]))
    print("Top 5:", " | ".join(["%s: %.6f" % (s[1], s[0]) for s in scored[:5]]))

    # ============ Phase 2: Adam optimization from top profiles ============
    phase2_end = min(deadline, t_start + 650.0)
    top_k = min(5, len(scored))
    per_profile_time = max(5.0, (phase2_end - time.monotonic()) / top_k)

    for rank in range(top_k):
        if time.monotonic() > phase2_end - 10:
            break
        sc_init, name, p_init = scored[rank]
        per_deadline = min(phase2_end - 5, time.monotonic() + max(20.0, per_profile_time))
        print("Phase 2: optimizing profile '%s' (init score=%.8f), time budget=%.0fs" %
              (name, sc_init, per_deadline - time.monotonic()))

        result, sc = logit_adam_optimize(p_init, per_deadline, rng, lr=0.05, verbose=(rank == 0))
        update(result, sc)
        print("  Result: score=%.10f (improvement: %.8f)" % (sc, sc_init - sc))

    print("After Phase 2: best=%.10f at n=%d" % (best_score, best_n))

    # ============ Phase 3: Extended refinement with varied hyperparams ============
    phase3_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        p3 = best_seq.copy()
        if p3.size != n_main:
            p3 = interpolate_profile(p3, n_main)

        configs = [
            {"lr": 0.03, "beta_decay": 0.998, "restart_every": 3000},
            {"lr": 0.08, "beta_decay": 0.993, "restart_every": 1500},
            {"lr": 0.05, "beta_decay": 0.996, "restart_every": 2000},
            {"lr": 0.10, "beta_decay": 0.990, "restart_every": 1000},
            {"lr": 0.02, "beta_decay": 0.999, "restart_every": 4000},
        ]

        remaining = phase3_end - time.monotonic()
        per_cfg = max(10.0, remaining / len(configs))
        for cfg_i, cfg in enumerate(configs):
            if time.monotonic() > phase3_end - 10:
                break
            per_deadline = min(phase3_end - 5, time.monotonic() + per_cfg)
            # Add tiny noise for diversity
            p0 = p3 + 0.002 * rng.normal(size=p3.size)
            p0 = np.maximum(p0, 1e-10)
            p0 = p0 / p0.sum()

            result, sc = logit_adam_optimize(p0, per_deadline, rng,
                                             lr=cfg["lr"], beta_decay=cfg["beta_decay"],
                                             restart_every=cfg["restart_every"],
                                             verbose=(cfg_i == 0))
            update(result, sc)
            print("Phase 3 cfg%d (lr=%.2f): score=%.10f" % (cfg_i, cfg["lr"], sc))

            # Quick coordinate descent refinement
            if best_seq is not None and best_seq.size == n_main:
                cd_deadline = min(phase3_end - 5, time.monotonic() + 15.0)
                cd_result, cd_sc = coordinate_descent_refine(best_seq, cd_deadline, rng,
                                                             n_steps=2000, verbose=False)
                update(cd_result, cd_sc)

    print("After Phase 3: best=%.10f at n=%d" % (best_score, best_n))

    # ============ Phase 4: Try larger n ============
    n_large = 1024
    phase4_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase4_end - 60 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc_large = compute_score(p_large)
        update(p_large, sc_large)
        print("Phase 4: n=%d, interpolated score=%.10f" % (n_large, sc_large))

        if sc_large < best_score + 0.02:  # Worth trying
            optimal_configs = [
                {"lr": 0.04, "beta_decay": 0.997, "restart_every": 2000},
                {"lr": 0.06, "beta_decay": 0.995, "restart_every": 1500},
            ]
            remaining4 = phase4_end - time.monotonic()
            per_cfg4 = max(10.0, remaining4 / len(optimal_configs))
            for cfg4 in optimal_configs:
                if time.monotonic() > phase4_end - 20:
                    break
                per_dl = min(phase4_end - 10, time.monotonic() + per_cfg4)
                p0 = p_large + 0.001 * rng.normal(size=n_large)
                p0 = np.maximum(p0, 1e-10)
                p0 = p0 / p0.sum()
                result, sc = logit_adam_optimize(p0, per_dl, rng,
                                                 lr=cfg4["lr"], beta_decay=cfg4["beta_decay"],
                                                 restart_every=cfg4["restart_every"],
                                                 verbose=False)
                update(result, sc)
                print("  Phase 4 (n=1024, lr=%.2f): score=%.10f" % (cfg4["lr"], sc))

    print("After Phase 4: best=%.10f at n=%d" % (best_score, best_n))

    # ============ Phase 5: Final polish ============
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = best_seq.size
        per = min(deadline - 5, time.monotonic() + 40.0)
        p0 = best_seq + 0.0005 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-10)
        p0 = p0 / p0.sum()
        result, sc = logit_adam_optimize(p0, per, rng, lr=0.02,
                                         beta_decay=0.999, restart_every=5000,
                                         verbose=True)
        update(result, sc)

        # Coordinate descent on the result
        if time.monotonic() < deadline - 3:
            cd_dl = min(deadline - 2, time.monotonic() + 15.0)
            cd_result, cd_sc = coordinate_descent_refine(best_seq, cd_dl, rng,
                                                         n_steps=3000, verbose=True)
            update(cd_result, cd_sc)

    print("After Phase 5: best=%.10f at n=%d" % (best_score, best_n))

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = interpolate_profile(generate_profiles(512, rng)[0][1], 512)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        t = np.linspace(0, 1, 512)
        final_seq = (1.0 + 2.0 * (1.0 - t)) / 512.0

    final_score = compute_score(final_seq)
    print("FINAL: n=%d, score=%.10f" % (final_seq.size, final_score))

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print("evaluate_sequence confirms: %.10f" % eval_score)
        else:
            print("evaluate_sequence returned non-finite, retrying with simpler sequence")
            final_seq = generate_profiles(512, rng)[0][1]
            final_score = compute_score(final_seq)
            print("Fallback: n=%d, score=%.10f" % (final_seq.size, final_score))
    except Exception as e:
        print("Could not call evaluate_sequence: %s" % str(e))

    return [float(x) for x in final_seq]
