    """Exact subgradient of 2n*conv[k_star]/S^2. O(n)."""
    n = len(s)
    M = conv[k_star]
    # grad[j] = 4n*(S*s[k_star-j] - M) / S^3
    # s[k_star-j] is valid when 0 <= k_star-j < n
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    grad = np.zeros(n)
    # s[k_star - j] for j in [lo, hi] means s values at indices [k_star-hi, k_star-lo] = [lo', hi']
    # k_star - j ranges from k_star-hi to k_star-lo as j goes from lo to hi
    idx = np.arange(k_star - hi, k_star - lo + 1)
    grad[lo:hi + 1] = 4.0 * n * (S * s[idx] - M) / (S * S * S)
    return grad


def _gradient_soft(s, conv, S, beta, top_m=5):
    """Soft gradient using top-m peaks with exponential weighting."""
    n = len(s)
    M = conv.max()
    if M < 1e-15:
        return np.zeros(n)
    # Weights: exp(beta * conv / M) over top-m peaks
    vals, idx = np.sort(conv)[::-1]
    m = min(top_m, 2 * n - 1)
    # Use exponential weighting over top m
    w = np.exp(beta * vals[:m] / M - 1.0)  # -1 for numerical stability
    w = w / w.sum()
    W = float(np.dot(w, vals[:m]))
    # g[j] = sum_{k in top_m} w[k] * s[k-j]
    # Compute via direct summation (m is small)
    g = np.zeros(n)
    for t in range(m):
        k = idx[t]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_s = np.arange(k - hi, k - lo + 1)
        g[lo:hi + 1] += w[t] * s[idx_s]
    grad = 4.0 * n * (S * g - W) / (S * S * S)
    return grad


def _equalization_step(s, conv, S, num_peaks=5, shrink=0.3):
    """Gauss-Newton step to equalize the top peaks. Returns new s or None."""
    n = len(s)
    M = conv.max()
    if M < 1e-15:
        return None
    # Find top peaks
    sorted_idx = np.argsort(conv)[::-1]
    # Select peaks: the max and any others within a threshold
    m = 1
    for i in range(1, num_peaks):
        if conv[sorted_idx[i]] > 0.9 * M:
            m = i + 1
        else:
            break
    m = min(m, num_peaks)
    k_peaks = sorted_idx[:m]
    # Target: shrink the max by 'shrink' fraction
    T = M * (1.0 - shrink)
    # Build the (m+1) x (m+1) system
    # A[j,j'] = sum_i s[k_j - i] * s[k_{j'} - i] = aperiodic autocorrelation at lag k_{j'}-k_j
    # A[j, last] = P[k_j] = sum_i s[k_j - i] (partial sum)
    # A[last, last] = n
    # b[j] = (T - conv[k_j]) / 2
    # b[last] = 0
    # Then delta = M_mat^T * A^{-1} * b where M_mat is (m+1) x n

    A = np.zeros((m + 1, m + 1))
    b = np.zeros(m + 1)

    for j in range(m):
        for jp in range(m):
            lag = k_peaks[jp] - k_peaks[j]
            # autocorr at 'lag': sum_i s[i] s[i+lag] (aperiodic)
            if lag >= 0:
                lo_i = 0
                hi_i = n - 1 - lag
            else:
                lo_i = -lag
                hi_i = n - 1
            A[j, jp] = np.dot(s[lo_i:hi_i + 1], s[lo_i + lag:hi_i + 1 + lag])
        # P[k_j] = sum_i s[k_j - i] for i=0..n-1 (valid range)
        lo = max(0, k_peaks[j] - (n - 1))
        hi = min(n - 1, k_peaks[j])
        A[j, m] = np.sum(s[lo:hi + 1]) if hi >= lo else 0.0
        b[j] = (T - conv[k_peaks[j]]) / 2.0
    A[m, m] = float(n)
    np.fill_diagonal(A, A.diagonal() + 1e-10)  # Regularization for stability

    try:
        lam = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        return None

    # delta[j] = sum_t lam[t] * s[k_t - j] + lam[m]
    delta = lam[m] * np.ones(n)
    for t in range(m):
        k = k_peaks[t]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_s = np.arange(k - hi, k - lo + 1)
        delta[lo:hi + 1] += lam[t] * s[idx_s]

    # Find max step size: s + alpha*delta >= 0
    neg_mask = delta < -1e-15
    if not np.any(neg_mask):
        alpha = 1.0
    else:
        alpha = np.min(-s[neg_mask] / delta[neg_mask])
        alpha = max(alpha, 1e-10)

    # Damp the step for safety
    alpha = min(alpha, 1.0) * 0.8
    s_new = s + alpha * delta
    s_new = np.maximum(s_new, 0.0)
    s_sum = s_new.sum()
    if s_sum < 1e-12:
        return None
    s_new /= s_sum
    return s_new


def mirror_descent_step(s, grad, eta):
    """Exponentiated gradient update projected to simplex."""
    # s_new[j] ∝ s[j] * exp(-eta * grad[j])
    # Normalize grad for numerical stability
    grad_shifted = grad - grad.max()
    log_update = np.log(s + 1e-30) + (-eta * grad_shifted)
    # Shift for numerical stability
    log_update -= log_update.max()
    s_new = np.exp(log_update)
    s_sum = s_new.sum()
    if s_sum < 1e-30:
        return s.copy()
    return s_new / s_sum


def optimize_sequence(s0, deadline, rng, verbose=False):
    """Main optimization loop. Returns (best_s, best_score)."""
    n = len(s0)
    s = s0.astype(np.float64).copy()
    s = np.maximum(s, 0.0)
    s_sum = s.sum()
    if s_sum < 1e-12:
        s = np.ones(n) / n
    else:
        s /= s_sum

    conv = _fft_conv(s)
    best_score = _score(s)
    best_s = s.copy()

    eta = 1.0
    beta = 3.0
    no_improve = 0
    patience = 80
    it = 0

    while time.monotonic() < deadline:
        s_sum = s.sum()
        conv = _fft_conv(s)
        cur_score = 2.0 * n * conv.max() / (s_sum * s_sum)

        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_s = s.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Determine gradient type
        M = conv.max()
        if M < 1e-15:
            break

        # Check if we should do equalization step
        should_equalize = (it % 50 == 49) or (no_improve > patience * 0.7 and it % 20 == 19)

        if should_equalize:
            s_eq = _equalization_step(s, conv, s_sum, num_peaks=5, shrink=0.2)
            if s_eq is not None:
                eq_score = _score(s_eq)
                if eq_score < cur_score:
                    s = s_eq
                    if verbose and eq_score < best_score:
                        best_score = eq_score
                        best_s = s.copy()
                    it += 1
                    continue
            else:
                pass

        # Choose gradient
        # Use hard max if dominant, soft otherwise
        scaled = conv / M
        top2 = np.sort(scaled)[::-1]
        if len(top2) >= 2 and top2[1] > 0.95:
            # Multiple comparable peaks: use soft
            grad = _gradient_soft(s, conv, s_sum, beta=beta, top_m=5)
        else:
            k_star = int(np.argmax(conv))
            grad = _gradient_hard(s, conv, k_star, s_sum)

        # Mirror descent step
        eta_eff = max(eta, 0.01)
        s_new = mirror_descent_step(s, grad, eta_eff)

        # Check improvement
        new_score = _score(s_new)
        if new_score < cur_score * (1 + 1e-6):
            s = s_new
        else:
            # Try smaller step
            s_new2 = mirror_descent_step(s, grad, eta_eff * 0.3)
            s = s_new2

        it += 1

        # Adaptive parameters
        if no_improve == patience:
            eta *= 0.6
            no_improve = 0
            if eta < 0.05:
                # Perturb and restart
                s = s + 0.01 * rng.standard_normal(n)
                s = np.maximum(s, 1e-6)
                s /= s.sum()
                eta = 1.0
                beta = float(rng.choice([2.0, 4.0, 6.0, 10.0, 15.0]))
                no_improve = 0
                patience = int(patience * 0.8)
                if patience < 30:
                    patience = 30
        elif no_improve == patience // 2:
            beta = min(beta * 1.5, 30.0)

        # Slowly increase beta
        if it % 200 == 199:
            beta = min(beta + 0.5, 25.0)

    return best_s, best_score


def generate_candidates(n, rng, hseq=None):
    """Generate diverse starting candidates of length n."""
    candidates = []

    # Interpolated height_sequence_1
    if hseq is not None and len(hseq) > 5:
        x_old = np.linspace(0.0, 1.0, len(hseq))
        x_new = np.linspace(0.0, 1.0, n)
        h_r = np.interp(x_new, x_old, np.maximum(hseq, 0))
        h_r = np.maximum(h_r, 1e-8)
        h_r /= h_r.sum()
        candidates.append(h_r.copy())

    # Uniform
    candidates.append(np.ones(n) / n)

    # Cosine window
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        x = np.arange(n) / max(n - 1, 1)
        s = np.cos(alpha * np.pi * (x - 0.5)) ** 2
        s = np.maximum(s, 1e-6)
        s /= s.sum()
        candidates.append(s.copy())

    # Asymmetric: heavy on one side
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for side in ['left', 'right']:
            s = np.ones(n)
            k = int(frac * n)
            if side == 'left':
                s[:k] *= 2.0
                s[k:] *= 0.5
            else:
                s[:n-k] *= 0.5
                s[n-k:] *= 2.0
            s = np.maximum(s, 1e-6)
            s /= s.sum()
            candidates.append(s.copy())

    # Gaussian bumps (asymmetric)
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            x = np.arange(n) / max(n - 1, 1)
            s = np.exp(-(x - center) ** 2 / (2 * width ** 2))
            s = np.maximum(s, 1e-6)
            s /= s.sum()
            candidates.append(s.copy())

    # Random
    for _ in range(8):
        s = rng.random(n) + 0.05
        s /= s.sum()
        candidates.append(s.copy())

    # Random with structure (low-pass filtered)
    for _ in range(4):
        s = rng.random(n)
        # Simple low-pass: moving average
        kernel = np.ones(10) / 10
        s = np.convolve(s, kernel, mode='same')
        s = np.maximum(s, 1e-4)
        s /= s.sum()
        candidates.append(s.copy())

    # Sort by score
    scored = []
    for s in candidates:
        sc = _score(s)
        scored.append((sc, s))
    scored.sort(key=lambda x: x[0])
    return scored


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(s,s))/sum(s)^2.
    Uses mirror descent with exact subgradients, Gauss-Newton equalization,
    diverse initialization, and multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 60.0

    global_best_score = float('inf')
    global_best_seq = np.ones(256) / 256.0

    # Load height_sequence_1
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            print(f"Loaded height_sequence_1, length={h.size}, score={_score(h):.6f}")
    except Exception:
        pass

    # Try various lengths
    lengths = [192, 256, 320, 384, 448, 512]

    for n in lengths:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        print(f"\n=== n={n}, remaining={remaining:.0f}s ===")

        candidates = generate_candidates(n, rng, hseq)
        print(f"  Generated {len(candidates)} candidates, best initial score={candidates[0][0]:.6f}")

        # Run optimization on top candidates
        num_to_try = min(5, len(candidates))
        time_per = remaining / (num_to_try + 2)  # +2 for random restarts and refinement

        for i in range(num_to_try):
            if time.monotonic() > deadline - 60:
                break
            sc0, s0 = candidates[i]
            if sc0 >= global_best_score and i > 0:
                continue  # Skip candidates worse than current best (except first)

            cand_deadline = time.monotonic() + time_per * 1.2
            best_s, best_sc = optimize_sequence(s0, cand_deadline, rng, verbose=False)

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Candidate {i}: new best {best_sc:.6f} (n={n})")
            else:
                print(f"  Candidate {i}: score={best_sc:.6f}")

        # Random restarts
        for r in range(3):
            if time.monotonic() > deadline - 90:
                break
            s_r = rng.random(n) + 0.05
            # Apply a light smoothing
            k = min(n // 8, 20)
            if k > 1:
                s_r = np.convolve(s_r, np.ones(k) / k, mode='same')
                s_r = np.maximum(s_r, 1e-4)
            s_r /= s_r.sum()
            best_s, best_sc = optimize_sequence(s_r, time.monotonic() + 30, rng)
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Restart {r}: new best {best_sc:.6f} (n={n})")

    # Final refinement on the best found
    if time.monotonic() < deadline - 20:
        print(f"\n=== Final refinement, score={global_best_score:.6f}, n={len(global_best_seq)} ===")
        n = len(global_best_seq)
        best_s, best_sc = optimize_sequence(global_best_seq, deadline - 10, rng)
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s.copy()
            print(f"  ** Refined: {best_sc:.6f}")

    # Additional polish: try equalization-only on the best
    if time.monotonic() < deadline - 5:
        s = global_best_seq.copy()
        s /= s.sum()
        for _ in range(200):
            if time.monotonic() > deadline - 3:
                break
            conv = _fft_conv(s)
            S = s.sum()
            s_eq = _equalization_step(s, conv, S, num_peaks=8, shrink=0.1)
            if s_eq is not None:
                sc = _score(s_eq)
                if sc < _score(s):
                    s = s_eq
                    if sc < global_best_score:
                        global_best_score = sc
                        global_best_seq = s.copy()

    # Verify
    final_score = float(evaluate_sequence(global_best_seq.tolist()))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(global_best_seq)}")

    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result
