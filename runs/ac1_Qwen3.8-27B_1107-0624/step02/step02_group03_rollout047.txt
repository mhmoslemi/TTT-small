```python
"""
Multi-scale FFT-accelerated optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines fast FFT-based Adam in logit space with LP direction finding,
operating at multiple sequence lengths (512, 1024, 2048) to exploit
increased degrees of freedom at larger n.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Fast convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def compute_gradient(w, a):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # part2[i] = sum_k w[k] * a[i-k] = (w * a)[i] = convolution at index i
    conv_wa = fft_convolve(w, a)
    part2 = conv_wa[:n]
    # part1[i] = sum_j a[j] * w[i+j]
    # = conv(a_rev, w)[i + n - 1] where a_rev = a[::-1]
    b = a[::-1]
    conv_bw = fft_convolve(b, w)
    part1 = conv_bw[n - 1:2 * n - 1]
    return part1 + part2


def adam_polish_fftp(p0, deadline, lr=0.05, reg=1e-10, verbose=True, tag=""):
    """
    Adam gradient descent in logit space using FFT for convolution.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = fft_convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.15 * max_c)

    if verbose:
        print(f"  [{tag}] Adam start: n={n}, score={best_score:.8f}, beta={beta:.2e}")

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve(p0, p0)
        max_c = float(c.max())

        # Softmax weights
        x = (c - max_c) / beta
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_ex

        # Gradient
        h = compute_gradient(w, p0)
        grad_p = 2.0 * h + reg * p0

        # Logit space gradient
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = fft_convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0 and lr > 0.005:
                lr = lr * 0.92

        if it % 100 == 0:
            beta = max(1e-14, beta * 0.5)

        if no_imp > 1500:
            rng = np.random.default_rng(it + 12345)
            z = 0.85 * best_z + 0.25 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-12, 0.15 * float(fft_convolve(
                np.exp(z - z.max()) / np.exp(z - z.max()).sum(),
                np.exp(z - z.max()) / np.exp(z - z.max()).sum()
            ).max()))

        p0 = p

    if verbose:
        print(f"  [{tag}] Adam done: iters={it}, best_score={best_score:.8f}")
    return best_p, best_score


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=15.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterate(f, max_iters=500, t_init=0.4, t_min=0.02, deadline=None, verbose=True, tag=""):
    """LP-based iterations to reduce max(conv(f,f)). f must be normalized to sum 1."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = fft_convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M

    no_improve = 0
    t_step = t_init

    if verbose:
        print(f"  [{tag}] LP start: n={n}, score={best_score:.8f}")

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Try full step
        g_score = 2.0 * n * float(fft_convolve(g, g).max())

        if g_score < best_score * 0.998:
            f = g.copy()
            M = float(fft_convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
        else:
            f_new = (1.0 - t_step) * f + t_step * g
            f_new = normalize(f_new)
            if f_new is None:
                break
            M_new = float(fft_convolve(f_new, f_new).max())
            score_new = 2.0 * n * M_new

            if score_new < best_score - 1e-13:
                best_f = f_new.copy()
                best_score = score_new
                best_M = M_new
                f = f_new.copy()
                M = M_new
                no_improve = 0
                t_step = max(t_min, t_step * 0.997)
            else:
                no_improve += 1
                if no_improve % 10 == 0:
                    t_step = max(t_min, t_step * 0.85)
                if no_improve > 300:
                    t_step = t_init * 0.5
                    no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    [{tag}] LP iter {it}: score={best_score:.8f}, t={t_step:.4f}")

    if verbose:
        print(f"  [{tag}] LP done: score={best_score:.8f}")
    return best_f, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length using cubic spline."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    # Use cubic interpolation via numpy
    from scipy.interpolate import CubicSpline
    cs = CubicSpline(x_old, a, bc_type='clamped')
    a_new = cs(x_new)
    a_new = np.maximum(a_new, 0.0)
    return normalize(a_new)


def make_profile(n, type_, **kw):
    """Generate various initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0  # in [-1, 1]

    if type_ == 'uniform':
        a = np.ones(n)
    elif type_ == 'linear':
        eta = kw.get('eta', 1.5)
        a = 1.0 + eta * u
    elif type_ == 'quadratic':
        c1 = kw.get('c1', 1.0)
        c2 = kw.get('c2', -0.5)
        a = 1.0 + c1 * u + c2 * u * u
    elif type_ == 'cubic':
        c1 = kw.get('c1', 1.2)
        c2 = kw.get('c2', -0.3)
        c3 = kw.get('c3', 0.5)
        a = 1.0 + c1 * u + c2 * u * u + c3 * u * u * u
    elif type_ == 'bessel':
        # Approximate raised cosine / Bessel-like
        a = np.cos(np.pi * u / 2) ** 2
    elif type_ == 'gaussian_bump':
        sigma = kw.get('sigma', 0.3)
        a = np.exp(-u**2 / (2 * sigma**2))
    elif type_ == 'nollinz':
        # Nollinz-like window
        a = 0.42 + 0.5 * np.cos(np.pi * u) + 0.08 * np.cos(2 * np.pi * u)
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-resolution FFT+LP search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
      1. Initialize from height_sequence_1 or analytic profiles.
      2. Adam FFT polish at n=1024.
      3. LP refinement at n=1024.
      4. Upsample to n=2048, Adam + LP.
      5. Final polish at best n.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 20.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Phase 0: Get initial sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent
                best_score = internal_score(parent)
                print(f"Parent sequence: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 512
        profiles = [
            make_profile(n_init, 'linear', eta=1.5),
            make_profile(n_init, 'quadratic', c1=1.0, c2=-0.5),
            make_profile(n_init, 'cubic', c1=1.2, c2=-0.3, c3=0.4),
            make_profile(n_init, 'bessel'),
            make_profile(n_init, 'nollinz'),
        ]
        for p in profiles:
            sc = internal_score(p)
            if sc < best_score:
                best_score = sc
                best_seq = p
        print(f"Generated initial: n={len(best_seq)}, score={best_score:.8f}")

    # ============ Phase 1: Adam at n=1024 (300s) ============
    n1 = 1024
    deadline_p1 = min(deadline, t_start + 250.0)
    print(f"\n=== Phase 1: Adam at n={n1} ===")
    a1 = interpolate_seq(best_seq, n1)
    if a1 is not None:
        p_best, sc_best = adam_polish_fftp(a1, deadline_p1, lr=0.05, tag="P1")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 1 result: score={best_score:.8f}")

    # ============ Phase 2: LP at n=1024 (500s) ============
    deadline_p2 = min(deadline, t_start + 750.0)
    if time.monotonic() < deadline_p2 - 30:
        print(f"\n=== Phase 2: LP at n={n1} ===")
        f_init = best_seq.copy()
        if len(f_init) != n1:
            f_init = interpolate_seq(f_init, n1)
        if f_init is not None:
            # Time one LP solve to estimate iteration budget
            t0 = time.monotonic()
            M_test = float(fft_convolve(f_init, f_init).max())
            _ = lp_direction(f_init, M_test, timeout=10)
            t_lp = time.monotonic() - t0
            print(f"  LP time estimate: {t_lp*1000:.0f}ms")

            max_iters = max(20, int((deadline_p2 - time.monotonic()) * 0.6 / max(t_lp, 0.05)))
            max_iters = min(max_iters, 2000)
            f_best, sc_best = lp_iterate(f_init, max_iters=max_iters, t_init=0.4,
                                         t_min=0.02, deadline=deadline_p2,
                                         verbose=True, tag="P2")
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  Phase 2 result: score={best_score:.8f}")

    # ============ Phase 3: Adam polish at n=1024 (150s) ============
    deadline_p3 = min(deadline, t_start + 900.0)
    if time.monotonic() < deadline_p3 - 10:
        print(f"\n=== Phase 3: Adam polish at n={n1} ===")
        p_best, sc_best = adam_polish_fftp(best_seq, deadline_p3, lr=0.03, tag="P3")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 3 result: score={best_score:.8f}")

    # ============ Phase 4: Try n=2048 ============
    n2 = 2048
    deadline_p4 = min(deadline, t_start + 1500.0)
    if time.monotonic() < deadline_p4 - 60:
        print(f"\n=== Phase 4: Attempt n={n2} ===")
        a2 = interpolate_seq(best_seq, n2)
        if a2 is not None:
            # Quick Adam at n=2048
            deadline_p4a = min(deadline_p4, time.monotonic() + 250.0)
            p_best2, sc_best2 = adam_polish_fftp(a2, deadline_p4a, lr=0.04, tag="P4a")
            if sc_best2 < best_score:
                best_score = sc_best2
                best_seq = p_best2
                print(f"  Phase 4a result: score={best_score:.8f}")

            # LP at n=2048 if time remains
            deadline_p4b = deadline_p4
            if time.monotonic() < deadline_p4b - 30:
                t0 = time.monotonic()
                M_test = float(fft_convolve(best_seq, best_seq).max())
                _ = lp_direction(best_seq, M_test, timeout=20)
                t_lp = time.monotonic() - t0
                print(f"  LP time at n=2048: {t_lp*1000:.0f}ms")

                max_iters = max(10, int((deadline_p4b - time.monotonic()) * 0.5 / max(t_lp, 0.1)))
                max_iters = min(max_iters, 500)
                f_best, sc_best = lp_iterate(best_seq, max_iters=max_iters, t_init=0.35,
                                             t_min=0.02, deadline=deadline_p4b,
                                             verbose=True, tag="P4b")
                if f_best is not None and sc_best < best_score:
                    best_score = sc_best
                    best_seq = f_best
                    print(f"  Phase 4b result: score={best_score:.8f}")

    # ============ Phase 5: Final Adam polish ============
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 5: Final polish ===")
        n_final = len(best_seq)
        p_best, sc_best = adam_polish_fftp(best_seq, deadline, lr=0.02, tag="P5")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 5 result: score={best_score:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_profile(512, 'linear', eta=1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
```
