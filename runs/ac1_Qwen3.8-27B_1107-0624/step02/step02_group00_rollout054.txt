"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP spreading, smooth-max gradient descent, spectral balancing,
and adaptive peak deflation.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _smooth_max_grad(a, beta):
    """Compute the value and gradient weight of the smooth max of conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip for numerical stability
    x = np.clip(x, -100, 100)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        val = mval
    else:
        w = ex / s
        val = mval + beta * float(np.log(s))
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (with zero padding)
    # This is 2 * (cross-correlation of a and w) at lag j
    # Using FFT: cross_corr[j] = sum_k w[k] * a[k-j]
    # = (a * flip(w))[n-1-j] ... let me think
    # Actually: sum_k w[k] * a[k-j] for valid k,j
    # Let's use: conv(flip(w), a)[n-1+j] ... 
    
    # Simpler: use the convolution matrix structure
    # grad[j] = 2 * sum_{k: valid} w[k] * a[k-j]
    # This is the cross-correlation (a ⋆ w)[j] = sum_k a[k] w[k+j]
    # Wait, I need to be careful with indices.
    
    # From our derivation: ∂(a*a)[k]/∂a_j = 2*a[k-j] (if 0<=k-j<n)
    # So: grad[j] = 2 * sum_k w[k] * a[k-j] (with a[m]=0 if m out of range)
    # = 2 * sum_k w[k] * a[k-j]
    # Let m = k-j: = 2 * sum_m w[m+j] * a[m]
    # = 2 * (a * reverse-pad(w))[n-1-j] ... 
    
    # Easiest: use np.correlate or FFT
    # (a ⋆ w)[j] = sum_m a[m] * w[m+j] (non-circular cross-correlation)
    # In FFT: ifft(fft(a, N) * conj(fft(w, N)))[:n] gives (a ⋆ w)[0..n-1]
    
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w[:N], N)  # w has length 2n-1, pad to N
    # cross-correlation: ifft(fa * conj(fw))
    cc = np.fft.irfft(fa * np.conj(fw), N)[:n]
    grad = 2.0 * cc
    
    return val, grad, w

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _lp_refine(a, deadline, max_iter=10000, inflate=1.001, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)
        
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.1)
            if no_imp > 500:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        
        a = a_new
    
    return best_a

def _gradient_descent_smooth_max(a, deadline, max_iter=20000, beta_init=None, lr_init=0.02):
    """Smooth-max gradient descent with adaptive beta and Nesterov momentum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    c = _fft_convolve(a)
    cmax = float(c.max())
    if beta_init is None:
        beta = 0.05 * cmax
    else:
        beta = beta_init
    
    lr = lr_init
    base_lr = lr
    momentum = 0.9
    v = np.zeros(n)
    
    best_a = a.copy()
    best_score = float(2.0 * n * cmax)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0)
        s_nest = a_nest.sum()
        if s_nest < 1e-10:
            a_nest = a
        else:
            a_nest = a_nest / s_nest
        
        val, grad, w = _smooth_max_grad(a_nest, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue
        
        # Normalize gradient by the sum to make it scale-invariant
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        v = momentum * v - lr * grad / grad_norm * cmax
        a_new = _project_simplex(a + v)
        a = a_new
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.95)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = 0.05 * float(_fft_convolve(a).max())
                lr = base_lr
                no_imp = 0
        
        # Adaptive beta: decrease slowly
        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)
    
    return best_a

def _spectral_balance(a, deadline, max_iter=5000):
    """Spectral balancing: flatten the spectrum of a using Gerchberg-Saxton style iterations."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    N = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    target_mult = 1.0  # Target level = RMS * target_mult
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        rms = np.sqrt(np.mean(mag**2))
        target = rms * target_mult
        
        # Soft cap: compress values above target
        ratio = mag / (target + 1e-30)
        # Use a soft minimum: min(mag, target) approximated by sigmoid
        # Or just hard clip
        mag_new = np.minimum(mag, target)
        
        # But we need to preserve the DC component (sum of a)
        # mag_new[0] = sum(a) = 1, so don't clip DC
        mag_new[0] = mag[0]
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project onto non-negative simple (non-negative, sum=1)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            no_imp += 1
            if no_imp > 100:
                break
            continue
        
        a_new = a_new / s
        score = _score(a_new)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            # Try to reduce target_mult for next iteration
            if it % 50 == 0:
                target_mult = max(0.8, target_mult * 0.99)
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                target_mult = min(2.0, target_mult * 1.1)
            if no_imp > max(1000, 3 * n):
                a = best_a.copy()
                target_mult = 1.0
                no_imp = 0

def _logit_adam(a, deadline, max_iter=20000, lr=0.03):

    z -= float(np.mean(z))

    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    
    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = 0.05 * float(c.max())
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        p = a
        c = _fft_convolve(p)
        mval = float(c.max())

            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        fw = np.fft.rfft(w[:N], N)
        fp = np.fft.rfft(p, N)
        cc = np.fft.irfft(fp * np.conj(fw), N)[:n]
        vp = 2.0 * cc + 1e-6 * p
        
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z -= float(np.mean(z))

        score = float(2.0 * n * c.max())

            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)
        
        if it % 200 == 0:
            beta = max(1e-10, beta * 0.7)
        
        if no_imp > max(3000, 5 * n):
            rng = np.random.default_rng(it)
            z = 0.7 * best_z + 0.3 * rng.normal(size=n) * 0.1
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = 0.05 * float(c.max())
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0

def _next_pow2(n):

    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):
    """Compute smooth max value and its gradient for conv(a,a).
    Returns (value, gradient) where gradient[j] = 2*sum_k w[k]*a[k-j]."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = np.clip((c - mval) / beta, -100.0, 100.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        val = mval
    else:
        w = ex / s
        val = mval + beta * float(np.log(s))

    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * (cross-correlation h)[j] where h[j] = sum_m a[m] * w[m+j]
    # In FFT: h = IFFT(FFT(a) * conj(FFT(w))) with N >= 3n-2
    N = _next_pow2(3 * n - 2)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(fa * np.conj(fw), N)[:n]
    grad = 2.0 * h
    return val, grad


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=10000, inflate=1.001, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.05)
            if no_imp > 500:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _gradient_descent_smooth_max(a, deadline, max_iter=20000, beta_init=None, lr_init=0.015):
    """Smooth-max gradient descent with Nesterov momentum and adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    c = _fft_convolve(a)
    cmax = float(c.max())
    if beta_init is None:
        beta = 0.04 * cmax
    else:
        beta = beta_init

    lr = lr_init
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)

    best_a = a.copy()
    best_score = float(2.0 * n * cmax)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0)
        s_nest = a_nest.sum()
        if s_nest > 1e-10:
            a_nest = a_nest / s_nest
        else:
            a_nest = a

        val, grad = _smooth_max_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        # Scale gradient for stability
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm * cmax

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.001, lr * 0.95)
            if no_imp > max(3000, 6 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = 0.04 * float(_fft_convolve(a).max())
                lr = base_lr
                no_imp = 0

        if it % 80 == 0:
            beta = max(1e-10, beta * 0.82)

    return best_a


def _spectral_balance(a, deadline, max_iter=5000):
    """Gerchberg-Saxton style spectral flattening with non-negative projection."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    N = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    target_mult = 1.2

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        rms = np.sqrt(np.mean(mag ** 2))
        target = rms * target_mult

        # Soft compression: for values above target, pull them down
        mag_new = mag.copy()
        above = mag_new > target
        if np.any(above):
            excess = mag_new[above] - target
            mag_new[above] = target + excess * 0.3  # Soft cap

        # Preserve DC
        mag_new[0] = mag[0]

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            no_imp += 1
            if no_imp > 200:
                break
            continue
        a_new = a_new / s
        score = _score(a_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            if it % 100 == 0:
                target_mult = max(0.7, target_mult * 0.99)
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                target_mult = min(3.0, target_mult * 1.05)
            if no_imp > max(2000, 4 * n):
                a = best_a.copy()
                target_mult = 1.2
                no_imp = 0

        a = a_new

    return best_a


def _logit_adam(a, deadline, max_iter=20000, lr=0.03):
    """Adam optimization in logit space with correct gradient."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = 0.04 * float(c.max())
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        mval = float(c.max())
        x = np.clip((c - mval) / beta, -100.0, 100.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        # Gradient of smooth max w.r.t. p: g_p[j] = 2*sum_k w[k]*p[k-j]
        N_cr = _next_pow2(3 * n - 2)
        fp = np.fft.rfft(p, N_cr)
        fw = np.fft.rfft(w, N_cr)
        gp = 2.0 * np.fft.irfft(fp * np.conj(fw), N_cr)[:n]
        gp += 1e-7 * p  # Small regularization

        # Convert to z-gradient: gz_j = p_j * (gp_j - sum(p*gp))
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 150 == 0:
            beta = max(1e-10, beta * 0.72)

        if no_imp > max(3000, 6 * n):
            rng = np.random.default_rng(it * 7 + 1)
            z = 0.6 * best_z + 0.4 * rng.normal(size=n) * 0.1
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = 0.04 * float(c.max())
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _make_hann_window(n):
    t = np.linspace(0, 2 * np.pi, n, endpoint=False)
    w = 0.5 * (1 + np.cos(t))
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def _make_gaussian(n, sigma=0.3):
    t = np.linspace(-1, 1, n)
    g = np.exp(-t ** 2 / (2 * sigma ** 2))
    return g / g.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _peak_deflation(a, deadline, max_iter=500):
    """Targeted peak reduction: identify top contributing pairs and reduce them."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        cmax_val = float(c.max())
        # Find top few peaks
        k_top = min(5, len(c))
        idx = np.argsort(c)[::-1][:k_top]
        # Weight by how much above median
        cmed = float(np.median(c))
        weights = np.maximum(c[idx] - cmed, 0.0)
        if weights.sum() < 1e-15:
            break
        weights = weights / weights.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        for k in idx:
            ki = int(k)
            for j in range(n):
                diff = ki - j
                if 0 <= diff < n:
                    grad[j] += weights[list(idx).index(k)] * 2.0 * a[diff]

        # Step in negative gradient direction
        lr = 0.01
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)

        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:
                break

    return best_a
