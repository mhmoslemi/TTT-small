        c = _fft_convolve(p)
        mval = float(c.max())
        x = np.clip((c - mval) / beta, -50, 50)
        ex = np.exp(x)
        ss = float(ex.sum())
        if not np.isfinite(ss) or ss <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / ss
        N = _next_pow2(3 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p, N)
        r = np.fft.irfft(fw * np.conj(fp), N)
        vp = 2.0 * r[:n] + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t_adam = 0
            continue
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v_arr / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t_adam = 0
            continue
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        if it % 150 == 0:
            beta = max(1e-12, beta * 0.5)
        if no_imp > 5000:
            z = 0.7 * best_z + 0.3 * np.random.default_rng(it + 999).normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t_adam = 0
            no_imp = 0
    return best_a


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max()) * inflate
        c_obj = -np.ones(n)
        A = _make_conv_matrix(a)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 200:
                break
            continue
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 300:
                break
        a = a_new
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale smooth-max gradient optimization with aggressive beta annealing,
    Nesterov momentum, diverse initializations, and LP-based final refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Phase 1: Optimize at n=512 from multiple initializations
    n512 = 512
    inits = []
    if parent is not None:
        inits.append(_interpolate_profile(parent, n512))
        for _ in range(2):
            p_pert = _interpolate_profile(parent, n512)
            p_pert = p_pert * (1.0 + 0.05 * rng.normal(size=n512))
            p_pert = np.maximum(p_pert, 0)
            p_pert = p_pert / p_pert.sum()
            inits.append(p_pert)
    for eta in [1.0, 1.5, 2.0, 2.5, 3.0]:
        inits.append(_make_linear_profile(n512, eta))
    for a1, a3 in [(1.5, -1.0), (2.0, -1.5), (2.5, -2.0)]:
        inits.append(_make_cubic_profile(n512, a1, a3))

    phase1_deadline = t_start + 600.0
    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        sc0 = _score(a0)
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()
        # Budget per init
        per_init_budget = min(60.0, phase1_deadline - time.monotonic())
        if per_init_budget < 5:
            break
        a_opt = _gradient_optimize(a0, t_start + (time.monotonic() - t_start) + per_init_budget,
                                   max_iter=20000, beta_factor=0.05, momentum=0.85)
        sc_opt = _score(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
        print(f"  Init {i}: start={sc0:.8f} -> opt={sc_opt:.10f}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # Phase 2: LP refinement on best
    if best_seq is not None and time.monotonic() < deadline - 200:
        lp_deadline = min(t_start + 1000.0, deadline - 200)
        a_lp = _lp_refine(best_seq, lp_deadline, max_iter=3000, inflate=1.001, blend=0.25)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        print(f"Phase 2 LP: score={sc_lp:.10f}")

    # Phase 3: Try n=1024 (interpolate best 512 to 1024)
    if best_seq is not None and time.monotonic() < deadline - 300:
        n1024 = 1024
        a_1024 = _interpolate_profile(best_seq, n1024)
        sc_1024 = _score(a_1024)
        if np.isfinite(sc_1024) and sc_1024 < best_score:
            best_score = sc_1024
            best_seq = a_1024.copy()
        print(f"  n=1024 interpolated: score={sc_1024:.10f}")

        # Optimize at 1024
        grad_deadline = min(t_start + 1400.0, deadline - 200)
        a_1024_opt = _gradient_optimize(a_1024, grad_deadline,
                                        max_iter=30000, beta_factor=0.04, momentum=0.85)
        sc_1024_opt = _score(a_1024_opt)
        if np.isfinite(sc_1024_opt) and sc_1024_opt < best_score:
            best_score = sc_1024_opt
            best_seq = a_1024_opt.copy()
        print(f"  n=1024 optimized: score={sc_1024_opt:.10f}")

    # Phase 4: Logit Adam polish
    if best_seq is not None and time.monotonic() < deadline - 60:
        adam_deadline = min(t_start + budget_s - 60.0, deadline - 30)
        a_adam = _logit_adam_polish(best_seq, adam_deadline, max_iter=30000, lr=0.03)
        sc_adam = _score(a_adam)
        if np.isfinite(sc_adam) and sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()
        print(f"Phase 4 Adam: score={sc_adam:.10f}")

    # Phase 5: Final tiny LP pass
    if best_seq is not None and time.monotonic() < deadline - 10:
        lp_deadline2 = deadline - 5
        a_last = _lp_refine(best_seq, lp_deadline2, max_iter=1000, inflate=1.0005, blend=0.15)
        sc_last = _score(a_last)
        if np.isfinite(sc_last) and sc_last < best_score:
            best_score = sc_last
            best_seq = a_last.copy()
        print(f"Phase 5 final LP: score={sc_last:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
