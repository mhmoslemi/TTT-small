import numpy as np

from scipy.optimize import linprog


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a, b):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size + b.size - 1
    N = next_pow2(n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n]
    return np.maximum(c, 0.0)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    idx = np.arange(n)
    for j in range(n):
        A[j:j+n, j] = f
    return A


def lp_descent_step(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def soft_max_grad(c, beta):
    """Compute soft max value and weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w

def lp_optimize(f_init, deadline, rng, verbose=True):
    """LP-based optimization with adaptive step size."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_M = float(np.convolve(f, f).max())
    t = 0.2
    stagnant = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())
        
        g = lp_descent_step(f, M)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()
        
        # Adaptive step size: try to decrease M
        improved = False
        for _ in range(5):  # Try up to 5 step sizes
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-10:
                t *= 0.5
                continue
            f_new = f_new / sn
            M_new = float(np.convolve(f_new, f_new).max())
            
            if M_new < M - 1e-14:
                f = f_new
                if M_new < best_M - 1e-12:
                    best_M = M_new
                    best_f = f.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                improved = True
                iteration += 1
                break
            else:
                t *= 0.5
                t = max(t, 0.001)
        
        if not improved:
            stagnant += 1
        
        t = max(0.005, min(0.3, t * 1.1))  # Slowly increase t
        
        if stagnant > 50:
            # Random restart
            noise_scale = 0.15 * np.std(best_f) + 0.01 / n
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.2
            stagnant = 0
        
        if verbose and iteration % 50 == 0:
            print(f"  LP iter {iteration}: M={best_M:.10f}, t={t:.4f}")
    
    return best_f
