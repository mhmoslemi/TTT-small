    Given f (length n, sum = sqrt(2n)), find the LP direction.

    return g, M

"""
LP-direction method with line search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Core idea: at each step, solve an LP to find a direction g that keeps f*g bounded,
then take an optimal line search step towards g. Combined with multi-length search,
diverse initializations, and gradient polish for final refinement.
"""
import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    mx = float(c.max())
    return float(2.0 * n * mx / (s * s))


def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(f):
    n = len(f)
    ff = np.convolve(f, f)
    M = float(ff.max())
    if M <= 0:
        return None
    C = build_conv_matrix(f)
    c_obj = -np.ones(n)
    A_ub = np.vstack([C, -np.eye(n)])
    b_ub = np.concatenate([M * np.ones(2 * n - 1), np.zeros(n)])
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if not res.success or res.x is None:
        return None
    g = res.x.copy()
    S = float(g.sum())
    if S < 1e-10:
        return None
    target = np.sqrt(2.0 * n)
    g = g * target / S
    g = np.maximum(g, 0.0)
    return g


def line_search(f, g):
    ff = np.convolve(f, f)
    fg = np.convolve(f, g)
    gg = np.convolve(g, g)
    best_t = 0.0
    best_val = float(ff.max())
    ts = np.linspace(0.0, 1.0, 41)
    for t in ts:
        hh = (1 - t) ** 2 * ff + 2 * t * (1 - t) * fg + t ** 2 * gg
        val = float(hh.max())
        if val < best_val:
            best_val = val
            best_t = t
    if 0 < best_t < 1:
        ts2 = np.linspace(max(0.0, best_t - 0.03), min(1.0, best_t + 0.03), 21)
        for t in ts2:
            hh = (1 - t) ** 2 * ff + 2 * t * (1 - t) * fg + t ** 2 * gg
            val = float(hh.max())
            if val < best_val:
                best_val = val
                best_t = t
    return best_t


def lp_optimize(f0, deadline, max_iter=1000):
    f = f0.copy()
    n = len(f)
    target = np.sqrt(2.0 * n)
    s = float(f.sum())
    if s < 1e-10:
        return f
    f = f * target / s
    f = np.maximum(f, 0.0)
    best_f = f.copy()
    best_peak = float(np.convolve(f, f).max())
    no_improve = 0
    prev_f = f.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        g = lp_direction(f)
        if g is None:
            break
        t = line_search(f, g)
        if t < 1e-7:
            no_improve += 1
            if no_improve > 20:
                break
            f = f + 0.01 * (g - f)
            f = np.maximum(f, 0.0)
            f = f * target / f.sum()
            continue
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new * target / f_new.sum()
        new_peak = float(np.convolve(f_new, f_new).max())
        if new_peak < best_peak:
            best_peak = new_peak
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
        prev_f = f.copy()
        f = f_new
    return best_f


def make_profile(n, ptype, param):
    j = np.arange(n, dtype=np.float64)
    t = j / max(1, n - 1)
    u = 2.0 * t - 1.0
    if ptype == 'uniform':
        a = np.ones(n)
    elif ptype == 'quadratic':
        a = 1.0 + param * (1.0 - u ** 2)
    elif ptype == 'gaussian':
        a = np.exp(-param * u ** 2)
    elif ptype == 'cosine':
        a = 0.5 + 0.5 * np.cos(np.pi * u) ** max(0.1, param)
    elif ptype == 'power':
        a = (1.0 - np.abs(u)) ** max(0.1, param)
    elif ptype == 'sigmoid':
        a = 1.0 / (1.0 + np.exp(-max(0.1, param) * u))
    elif ptype == 'cubic':
        a = 1.0 + param * u - 0.5 * param * u ** 3
    elif ptype == 'two_pole':
        a = 1.0 + param * np.cos(2 * np.pi * u)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    return a


def soft_max_adam_polish(p0, deadline, rng, lr=0.08):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n = p0.size
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.2 * maxc)
    best_p = p0.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        mval = float(c.max())
        if beta <= 1e-14:
            w = np.zeros(n)
            mask = c >= mval - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                w[np.argmax(c)] = 1.0
            else:
                w[mask] = 1.0 / cnt
        else:
            x = (c - mval) / beta
            ex = np.exp(x)
            s = float(ex.sum())
            if not np.isfinite(s) or s <= 0:
                ex = np.zeros(n)
                ex[np.argmax(c)] = 1.0
                s = 1.0
            w = ex / s
        N = next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p[::-1], N)
        gp = 2.0 * np.fft.irfft(fw * fp, N)[n - 1:2 * n - 1]
        vp = gp + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            z = np.log(np.maximum(best_p, 1e-15))
            z -= float(np.mean(z))
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_p, 1e-15))
            z -= float(np.mean(z))
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        p = e / e.sum()
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.03, lr * 0.95)
        if adam_t % 80 == 0:
            beta = max(1e-14, beta * 0.7)
        if no_imp > max(300, 4 * n):
            z = 0.8 * np.log(np.maximum(best_p, 1e-15)) + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            p = e / e.sum()
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.2 * maxc)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    return best_p


def coordinate_descent_polish(p, deadline, rng, max_sweeps=50):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    best_p = p.copy()
    best_score = compute_score(p)
    no_imp = 0
    step = 0.1

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score
            for factor in [1 + step, 1 - step, 1 + step * 0.5, 1 - step * 0.5]:
                nv = orig * factor
                if nv < 1e-15 or nv > 1000:
                    continue
                p[i] = nv
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                sc = compute_score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = nv
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        if improved:
            step = max(0.03, step * 0.85)
        if no_imp > n:
            break
    return best_p, best_score


def interpolate_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    out = np.interp(x1, x0, p)
    out = np.maximum(out, 1e-12)
    return out / out.sum()
