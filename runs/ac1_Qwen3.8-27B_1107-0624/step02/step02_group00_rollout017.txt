def _smooth_max_val_grad(c, beta):
    """Compute smooth max value and gradient (weights) of c with parameter beta."""
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip for numerical stability
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w


def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _interpolate_profile(p0, n_new):

        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_conv_matrix(a):

    A = np.zeros((2 * n - 1, n))

    return A

                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _lbfgs_smooth_max(a, deadline, max_iter=5000, beta_init=None):

        return smax, gz

        return val

        _, g = objective_and_grad(x)
        return g

    x0 = np.log(np.maximum(a, 1e-15))

def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w

    gz = p * (vp - dot)
    gz -= gz.mean()
    return gz

"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP spreading, L-BFGS on smooth max (new), gradient polish with Nesterov,
and simulated annealing. Multi-length exploration with adaptive beta schedules.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_val_grad(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=10000, inflate=1.001, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)
        if g is None:
            break
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.1)
            if no_imp > 500:
                break
            continue
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        a = a_new
    return best_a


def _lbfgs_obj(x, n, beta):
    x_clipped = np.clip(x, -50, 50)
    x_shifted = x_clipped - x_clipped.max()
    e = np.exp(x_shifted)
    p = e / e.sum()
    c = _fft_convolve(p)
    val, _ = _smooth_max_val_grad(c, beta)
    return val


def _lbfgs_jac(x, n, beta):
    x_clipped = np.clip(x, -50, 50)
    x_shifted = x_clipped - x_clipped.max()
    e = np.exp(x_shifted)
    p = e / e.sum()
    c = _fft_convolve(p)
    _, w = _smooth_max_val_grad(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp_rev = np.fft.rfft(p[::-1], N)
    h_full = np.fft.irfft(fw * fp_rev, N)
    h = h_full[n - 1:2 * n - 1]
    vp = 2.0 * h
    dot = float(np.dot(p, vp))
    gz = p * (vp - dot)
    gz -= gz.mean()
    return gz


def _lbfgs_optimize(a, deadline, max_iter=5000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    base_beta = max(1e-6, 0.05 * float(c.max()))
    x0 = np.log(np.maximum(a, 1e-15))
    x0 = x0 - x0.max()
    no_imp = 0
    for stage in range(10):
        if time.monotonic() > deadline:
            break
        beta = base_beta * (0.5 ** stage)
        if beta < 1e-10:
            beta = 1e-10
        iter_per_stage = max(100, max_iter // (10 - stage))
        try:
            res = minimize(
                _lbfgs_obj, x0,
                jac=_lbfgs_jac,
                args=(n, beta),
                method='L-BFGS-B',
                options={
                    'maxiter': iter_per_stage,
                    'ftol': 1e-16,
                    'gtol': 1e-10,
                    'maxls': 30,
                    'maxfun': iter_per_stage * 3
                }
            )
            x0 = np.clip(res.x, -50, 50)
            x0 = x0 - x0.max()
            e = np.exp(x0)
            p = e / e.sum()
            c = _fft_convolve(p)
            score = float(2.0 * n * c.max())
            if score < best_score - 1e-14:
                best_score = score
                best_a = p.copy()
                no_imp = 0
            else:
                no_imp += 1
        except Exception:
            no_imp += 1
        if no_imp >= 3:
            break
    return best_a


def _gradient_polish(a, deadline, max_iter=20000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.1 * float(c.max()))
    lr = 0.02
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a_nest = np.maximum(a + momentum * v, 0)
        s_nest = a_nest.sum()
        if s_nest < 1e-10:
            a_nest = a
        a_nest = a_nest / s_nest
        c = _fft_convolve(a_nest)
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex <= 0 or not np.isfinite(s_ex):
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_nest[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h + 1e-7 * a_nest
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(2000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.1 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0
        if it % 100 == 0:
            beta = max(1e-12, beta * 0.7)
    return best_a


def _simulated_annealing(a, deadline, max_iter=15000, T_init=0.003):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    current_score = best_score
    rng = np.random.default_rng(777)
    T = T_init
    cooling = (1e-7 / T_init) ** (1.0 / max(1, max_iter))
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a_pert = a.copy()
        k = max(2, n // 20)
        idx = rng.choice(n, size=k, replace=False)
        for i in idx:
            a_pert[i] = max(0.0, a_pert[i] * (1.0 + rng.normal(0, 0.08)))
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s
        c_pert = _fft_convolve(a_pert)
        score_pert = float(2.0 * n * c_pert.max())
        delta = score_pert - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_pert
            current_score = score_pert
            if score_pert < best_score - 1e-14:
                best_score = score_pert
                best_a = a.copy()
        T *= cooling
    return best_a

        if gp_budget > 5:
            gp_end = time.monotonic() + gp_budget
            a_gp2 = _gradient_polish(best_seq, gp_end, max_iter=10000)
            sc_gp2 = _score(a_gp2)
            if np.isfinite(sc_gp2) and sc_gp2 < best_score:
                best_score = sc_gp2
                best_seq = a_gp2.copy()
            print(f"  Final GP: score={sc_gp2:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    # Authoritative evaluation
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
