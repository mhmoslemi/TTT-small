"""
Multi-resolution LP + FISTA optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous approach:
1. Extended to n=1024 for better asymptotic performance.
2. Line search on LP blending step.
3. FISTA polish using structured gradient.
4. Multiple diverse starting points.

import numpy as np
import time
from scipy.optimize import linprog
import scipy.signal

def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    # Use the structure: A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    m_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = m_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = f[diff[valid]]
    return A

def lp_direction(f, M, timeout=10.0):
    """Solve LP: maximize sum(g) s.t. g>=0, conv(f,g) <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def interpolate_seq(a, n_new):
    """Interpolate a normalized sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

def lp_iterate(f, max_iters, t_init=0.3, t_min=0.01, deadline=None, verbose=True, tag=""):
    """
    LP-based iterations to reduce max(conv(f,f)).
    f must be normalized to sum 1.
    Uses line search on blending parameter.

    n = len(f)

    f = normalize(f)

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M

    no_improve = 0
    t_step = t_init
    it = 0

        if g is None:

        best_new_score = np.inf
        best_new_f = None

    best_score = 2.0 * n * float(np.convolve(x, x).max())
    no_imp = 0
    it = 0
