    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)

    elif type == 'parabola':

    theta = (cssv[rho] - 1) / (rho + 1.0)

"""
Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses projected gradient descent in log-space with adaptive soft-argmax,
combined with diverse initialization and multi-resolution strategy.
Key insight: asymmetric profiles (e.g., ramps) dramatically outperform
symmetric ones by shifting the convolution peak away from the center.
"""

import numpy as np
import time

def score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def interp_seq(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

def make_power_profile(n, alpha):
    """p[i] ∝ ((i+0.5)/n)^alpha."""
    t = (np.arange(n) + 0.5) / n
    p = t ** alpha
    return normalize(p)

def make_exp_profile(n, alpha):
    """p[i] ∝ exp(alpha * i/n)."""
    t = np.arange(n) / n
    p = np.exp(alpha * t)
    return normalize(p)

def make_poly_profile(n, coeffs):
    """p[i] ∝ sum_k coeffs[k] * (i/n)^k."""
    t = np.arange(n) / n
    p = np.zeros(n)
    for k, c in enumerate(coeffs):
        p += c * t**k
    return normalize(p)

def compute_soft_gradient(p, beta):
    """
    Compute the soft gradient of max(conv(p,p)) w.r.t. p.
    Uses soft argmax with temperature beta.
    p must be normalized (sum=1, p>=0).
    Returns (gradient, max_c) where gradient has same shape as p.
    """
    n = len(p)
    c = np.convolve(p, p)
    max_c = float(c.max())
    
    # Soft weights
    x = (c - max_c) / beta
    x = np.clip(x, -100.0, 0.0)
    w = np.exp(x)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # Gradient: g[k] = sum_i w[i] * (p[i-k] + p[k])
    # where p[m] = 0 for m outside [0, n-1]
    # = sum_i w[i] * p[i-k] + p[k] * sum_i w[i]
    # = A[k] + p[k]  (since sum w = 1)
    # A[k] = sum_{j=0}^{n-1} w[j+k] * p[j]
    
    # Compute A using the structure: A[k] = dot(w[k:k+n], p)
    A = np.zeros(n)
    for k in range(n):
        A[k] = np.dot(w[k:k+n], p)
    
    g = A + p
    return g, max_c

    A = np.zeros(n)
