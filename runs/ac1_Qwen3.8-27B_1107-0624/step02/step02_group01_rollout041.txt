"""
Power-law ramp optimization with gradient polishing to minimize 2n*max(conv(a,a))/sum(a)^2.
Key insight: asymmetric power-law ramps f[j] = ((j+1)/n)^p achieve scores well below 2,
with the optimal p balancing continuous-approximation gains against discrete peak effects.

import numpy as np
import time
import math


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def make_power_ramp(n, p, amp=1000.0):
    """Create f[j] = amp * ((j+1)/n)^p"""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = amp * (j / n) ** p
    return a


def make_power_ramp_normalized(n, p):
    """Create normalized power ramp with sum=1"""
    a = make_power_ramp(n, p, 1.0)
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s


def make_bilateral_ramp(n, p, alpha):
    """f[j] = alpha*((j+1)/n)^p + (1-alpha)*((n-j)/n)^p"""
    j = np.arange(1, n + 1, dtype=np.float64)
    x = j / n
    a = alpha * x ** p + (1.0 - alpha) * (1.0 - x + 1.0/n) ** p
    a = np.maximum(a, 1e-30)
    return a / a.sum()


def softmax_gradient_polish(f0, deadline, rng, lr=0.05, max_iter=100000, beta_init=None):
    """Polish using soft-max gradient in logit space."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())
    if beta_init is None:
        beta = max(1e-12, 0.1 * M_val)
    else:
        beta = beta_init

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_corr = next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam % 500 == 0:
            beta = max(beta * 0.6, 1e-10)

        if no_imp > max(500, 2 * n):
            z = 0.8 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

    return best_f, best_M


def coordinate_descent_polish(f0, deadline, rng, max_rounds=5000):
    """Polish by adjusting pairs of elements to reduce max convolution."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    best_f = f.copy()
    best_score = compute_score(f)
    c = fft_convolve(f)
    M = float(c.max())

    step = 0.01
    no_imp = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Find where the max convolution is
        k_max = int(np.argmax(c))

        # Try to reduce f at positions that contribute to the peak
        # (f*f)[k_max] = sum_j f[j] * f[k_max - j]
        # Positions contributing: j where 0<=j<n and 0<=k_max-j<n
        j_lo = max(0, k_max - n + 1)
        j_hi = min(k_max, n - 1)

        # Try random pairs of adjustments
        improved = False
        n_tries = min(50, n // 4)
        for _ in range(n_tries):
            j1 = rng.integers(j_lo, j_hi + 1)
            j2 = rng.integers(0, n)
            if j1 == j2:
                continue
            # Transfer mass from j1 to j2
            delta = step * f[j1]
            if j2 not in range(j_lo, j_hi + 1):
                f_new = f.copy()
                f_new[j1] -= delta
                f_new[j2] += delta
                f_new = np.maximum(f_new, 0)
                s = f_new.sum()
                if s < 1e-30:
                    continue
                f_new = f_new / s
                c_new = fft_convolve(f_new)
                M_new = float(c_new.max())
                if M_new < M - 1e-15:
                    f = f_new
                    c = c_new
                    M = M_new
                    k_max = int(np.argmax(c))
                    j_lo = max(0, k_max - n + 1)
                    j_hi = min(k_max, n - 1)
                    sc = compute_score(f)
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                    no_imp = 0
                    improved = True
                    break
            else:
                # Both in the peak region - try to reduce the larger one
                if f[j1] > f[j2]:
                    delta = step * f[j1]
                    f_new = f.copy()
                    f_new[j1] -= delta
                    f_new[j2] += delta
                    f_new = np.maximum(f_new, 0)
                    s = f_new.sum()
                    if s < 1e-30:
                        continue
                    f_new = f_new / s
                    c_new = fft_convolve(f_new)
                    M_new = float(c_new.max())
                    if M_new < M - 1e-15:
                        f = f_new
                        c = c_new
                        M = M_new
                        j_lo = max(0, k_max - n + 1)
                        j_hi = min(k_max, n - 1)
                        sc = compute_score(f)
                        if sc < best_score:
                            best_score = sc
                            best_f = f.copy()
                        no_imp = 0
                        improved = True
                        break

        if not improved:
            no_imp += 1
            if no_imp > 100:
                # Perturb and restart
                f = best_f * (1.0 + 0.05 * rng.normal(size=n))
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                c = fft_convolve(f)
                M = float(c.max())
                k_max = int(np.argmax(c))
                no_imp = 0

    return best_f, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-15)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search:
    1. Evaluate power-law ramps with various exponents p.
    2. Find optimal p by direct score computation.
    3. Polish the best shape with gradient methods.
    4. Try different sequence lengths.
    5. Final refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  New best: score={sc:.10f}, n={p.size}")

    # Try to load parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
                sc = compute_score(parent)
                try_update(parent, sc)
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        parent = None

    # Phase 1: Evaluate power-law ramps at n=512
    n1 = 512
    print(f"\nPhase 1: Evaluating power-law ramps at n={n1}")

    p_values = [1, 1.5, 2, 2.5, 3, 4, 5, 6, 7, 8, 9, 10, 12, 14, 16, 20, 25, 30, 40, 50]
    ramp_results = []

    for p in p_values:
        a = make_power_ramp(n1, p, 1.0)
        sc = compute_score(a)
        ramp_results.append((sc, p))
        print(f"    p={p}: score={sc:.10f}")

    ramp_results.sort(key=lambda x: x[0])
    best_p = ramp_results[0][1]
    print(f"  Best p={best_p}, score={ramp_results[0][0]:.10f}")

    # Now refine p around the best with finer steps
    p_fine = best_p
    for delta in [0.1, 0.05, 0.02]:
        improved = True
        while improved:
            improved = False
            for ds in [-delta, delta]:
                p_try = p_fine + ds
                if p_try < 0.1:
                    continue
                a = make_power_ramp(n1, p_try, 1.0)
                sc = compute_score(a)
                if sc < compute_score(make_power_ramp(n1, p_fine, 1.0)):
                    p_fine = p_try
                    improved = True
            delta /= 2.5
            if delta < 0.001:
                break

    print(f"  Refined best p={p_fine:.4f}")
    a_best = make_power_ramp(n1, p_fine, 1.0)
    sc_best = compute_score(a_best)
    try_update(a_best, sc_best)

    # Phase 2: Gradient polish the best ramp
    print(f"\nPhase 2: Gradient polishing (n={n1})")
    phase2_end = min(deadline, t_start + 400.0)

    best_f = a_best / a_best.sum()
    per = min(phase2_end, t_start + 300.0)
    p_best, M_best = softmax_gradient_polish(best_f, per, rng, lr=0.04)
    sc = 2.0 * n1 * M_best
    try_update(p_best, sc)

    # Try different learning rates
    for lr_try in [0.02, 0.08, 0.01]:
        if time.monotonic() > phase2_end:
            break
        per = min(phase2_end, time.monotonic() + 60.0)
        p_best, M_best = softmax_gradient_polish(best_f, per, rng, lr=lr_try)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)

    # Coordinate descent polish
    if time.monotonic() < phase2_end:
        per = min(phase2_end, time.monotonic() + 80.0)
        p_best, sc_best = coordinate_descent_polish(best_f, per, rng, max_rounds=3000)
        try_update(p_best, sc_best)

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Try different sequence lengths
    print(f"\nPhase 3: Exploring different lengths")
    phase3_end = min(deadline, t_start + 900.0)

    for n_try in [256, 384, 640, 768, 1024]:
        if time.monotonic() > phase3_end:
            break

        # Interpolate best shape
        p_int = interpolate_profile(best_seq, n_try)
        if p_int is None:
            continue

        # Also try fresh power ramps at this length
        a_ramp = make_power_ramp(n_try, p_fine, 1.0)

        # Pick better starting point
        sc_int = compute_score(p_int)
        sc_ramp = compute_score(a_ramp)
        start = p_int if sc_int < sc_ramp else a_ramp
        sc_start = min(sc_int, sc_ramp)
        try_update(start, sc_start)

        # Polish
        per = min(phase3_end, time.monotonic() + 120.0)
        p_best, M_best = softmax_gradient_polish(start, per, rng, lr=0.04)
        sc = 2.0 * n_try * M_best
        try_update(p_best, sc)

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Try bilateral ramps (mix of increasing and decreasing)
    print(f"\nPhase 4: Bilateral ramps and other shapes")
    phase4_end = min(deadline, t_start + 1200.0)

    n4 = 512
    for alpha in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8]:
        if time.monotonic() > phase4_end:
            break
        for p in [2, 3, 5, 8, 12]:
            if time.monotonic() > phase4_end:
                break
            a = make_bilateral_ramp(n4, p, alpha)
            sc = compute_score(a)
            try_update(a, sc)

    # Try "flat top" ramps: f[j] = min((j+1)/n, c)^p
    for c_val in [0.5, 0.7, 0.85]:
        if time.monotonic() > phase4_end:
            break
        j = np.arange(1, n4 + 1, dtype=np.float64)
        for p in [2, 4, 6, 10]:
            if time.monotonic() > phase4_end:
                break
            a = np.minimum(j / n4, c_val) ** p
            a = np.maximum(a, 1e-15)
            sc = compute_score(a)
            try_update(a, sc)

    print(f"After Phase 4: best={best_score:.10f}")

    # Phase 5: Final intensive polishing
    print(f"\nPhase 5: Final polishing")
    phase5_end = deadline

    if best_seq is not None:
        n_best = best_seq.size
        best_norm = best_seq / best_seq.sum()

        # Multiple rounds of gradient polish with different parameters
        for lr in [0.03, 0.05, 0.02]:
            if time.monotonic() > phase5_end - 60:
                break
            per = min(phase5_end - 30, time.monotonic() + 150.0)
            p_best, M_best = softmax_gradient_polish(best_norm, per, rng, lr=lr)
            sc = 2.0 * n_best * M_best
            try_update(p_best, sc)

        # Coordinate descent
        if time.monotonic() < phase5_end - 60:
            per = min(phase5_end - 30, time.monotonic() + 200.0)
            p_best, sc_best = coordinate_descent_polish(best_norm, per, rng, max_rounds=10000)
            try_update(p_best, sc_best)

        # Try to improve by local perturbations
        if time.monotonic() < phase5_end - 30:
            for _ in range(5):
                if time.monotonic() > phase5_end - 30:
                    break
                # Random smooth perturbation
                noise = np.cumsum(rng.normal(0, 0.1, size=n_best))
                noise -= noise.mean()
                f_pert = best_norm * (1.0 + 0.1 * noise / max(1.0, np.abs(noise).max()))
                f_pert = np.maximum(f_pert, 1e-15)
                f_pert = f_pert / f_pert.sum()
                sc = compute_score(f_pert)
                if sc < best_score:
                    try_update(f_pert, sc)
                    # Polish the improved version
                    per = min(phase5_end - 30, time.monotonic() + 60.0)
                    p_best, M_best = softmax_gradient_polish(f_pert, per, rng, lr=0.03)
                    sc = 2.0 * n_best * M_best
                    try_update(p_best, sc)

    # Also try a different approach: optimize in log-space directly
    if best_seq is not None and time.monotonic() < phase5_end - 60:
        n_f = best_seq.size
        base = best_seq / best_seq.sum()
        z = np.log(np.maximum(base, 1e-15))
        z -= z.mean()

        # Simple gradient descent in log-space
        lr = 0.01
        for it in range(20000):
            if time.monotonic() > phase5_end - 30:
                break
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M = float(c.max())
            k_max = int(np.argmax(c))

            # Subgradient: derivative of (f*f)[k_max] w.r.t. f
            # d(f*f)[k_max]/df[j] = f[k_max - j] (if valid)
            grad = np.zeros(n_f)
            for j in range(n_f):
                k2 = k_max - j
                if 0 <= k2 < n_f:
                    grad[j] = f[k2]

            # Normalize gradient to log-space
            grad_z = grad - f * grad.sum() / f.sum()

            z = z - lr * grad_z
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()

        f_final = np.exp(z - z.max())
        f_final = f_final / f_final.sum()
        sc = compute_score(f_final)
        try_update(f_final, sc)

    print(f"\nFinal best: score={best_score:.10f}")

    # Final: ensure valid output
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            # Scale to be well within [0, 1000]
            max_val = final.max()
            if max_val > 999.0:
                final = final * 999.0 / max_val
            # Verify
            sc_final = compute_score(final)
            print(f"Final verified: score={sc_final:.10f}, n={final.size}, max={final.max():.2f}")
            return [float(x) for x in final]

    # Fallback: use the best power ramp
    a = make_power_ramp(512, 3.0, 1000.0)
    return [float(x) for x in a]
