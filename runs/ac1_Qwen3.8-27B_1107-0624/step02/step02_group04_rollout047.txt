import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0 and mn > -1e-10:
        c = c - mn
    return c


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = _full_conv(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _topk_softmax_grad(c, beta, k_top):
    """Return (smooth_max_value, gradient_weights) using top-k soft-max."""
    n = len(c)
    if n == 0:
        return 0.0, np.array([])
    idx = np.argpartition(c, -min(k_top, n))[-min(k_top, n):]
    top = c[idx]
    m = float(top.max())
    if beta < 1e-15:
        w = np.zeros(n)
        mx = float(c.max())
        mask = c >= mx - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return m, w
    # Use full soft-max but reweight toward top-k
    c_shift = c - m
    ex = np.exp(c_shift / beta)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(n)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = ex / s
    val = m + beta * float(np.log(s))
    return val, w


def _corr_weights(w, p):
    """Compute correlation of w with p: result[i] = sum_j w[j]*p[i+j] (valid part)."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _deflation_direction(a, c, peak_lag):
    """Compute deflation direction targeting the peak lag."""
    n = len(a)
    d = np.zeros(n)
    # d[i] should minimize sum_i d[i]*a[peak_lag - i] subject to sum(d)=0
    # Optimal: d[i] = -(a[peak_lag - i] - mean) for valid i
    for i in range(n):
        j = peak_lag - i
        if 0 <= j < n:
            d[i] = a[j]
    d -= d.mean()
    # Ensure a + eps*d >= 0 for reasonable eps
    # Clip: where a is small, reduce d[i] if positive
    min_a = float(a.min())
    max_pos = float(d.max()) if d.max() > 0 else 0
    if max_pos > 0:
        safe_eps = min_a / max_pos * 0.5
        d = d * min(1.0, max_pos * 1.0)  # normalize
    return d

def _coordinate_descent(p0, deadline, rng, n_iter_max=200000):
    """Coordinate descent polish: perturb small groups of coordinates."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    p = p0.copy()
    c = _full_conv(p)
    maxc = float(c.max())
    best_p = p0.copy()
    best_score = float(2.0 * n * maxc)

    T = 0.02
    T_min = 0.0005
    cool = 0.9997

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        # Pick a small patch
        ps = int(np.clip(rng.integers(3, min(15, n // 4)), 3, n))
        start = int(rng.integers(0, n - ps + 1))
        end = start + ps

        scale = T * np.sqrt(ps) * 0.3
        p_new = p.copy()
        p_new[start:end] += rng.normal(0, scale, ps)
        p_new = np.maximum(p_new, 1e-12)
        p_new = p_new / p_new.sum()

        c_new = _full_conv(p_new)
        maxc_new = float(c_new.max())
        score_new = float(2.0 * n * maxc_new)

        if score_new < best_score:
            p = p_new
            c = c_new
            best_p = p_new.copy()
            best_score = score_new
            T = max(T_min, T * 0.995)
        else:
            delta = score_new - best_score
            if T > T_min and rng.random() < np.exp(-delta / (T * 0.5)):
                p = p_new
                c = c_new
            T = max(T_min, T * cool)

        if it % 20000 == 0 and it > 0:
            print(f"    CD: iter={it}, T={T:.6f}, best={best_score:.10f}")

    return best_p, best_score


def _make_taper_profile(n, taper_frac):
    """Create a tapered profile: uniform with triangular taper at ends."""
    a = np.ones(n)
    t = int(taper_frac * n)
    if t > 0:
        ramp = np.linspace(0.1, 1.0, t)
        a[:t] = ramp
        a[n-t:] = ramp[::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_exp_profile(n, alpha):
    """Exponential profile: more mass at one end."""
    t = np.arange(n)
    a = np.exp(-alpha * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 1e-12)
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-12)
    return p1 / p1.sum()

                                      name=f"restart{r} n={target_n}")
            update_best(pr, scr, f"restart{r} n={target_n}")

        current = best_seq.copy() if best_seq is not None else p2

    print(f"\nPhase 2: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # --- Phase 3: Coordinate descent polish ---
    phase3_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        print(f"\n  Coordinate descent on n={best_seq.size}")
        p_cd, sc_cd = _coordinate_descent(best_seq, phase3_end, rng, n_iter_max=100000)
        update_best(p_cd, sc_cd, "coord_descent")

    # --- Phase 4: Final fine optimization ---
    phase4_end = deadline
    if time.monotonic() < phase4_end and best_seq is not None:
        remaining = phase4_end - time.monotonic()
        if remaining > 20:
            print(f"\n  Final polish ({remaining:.0f}s remaining)")
            p_fin, sc_fin = _optimize_logit(best_seq, phase4_end, rng,
                                           lr=0.02, top_k=3,
                                           beta_init_frac=0.05, beta_decay=0.80,
                                           no_imp_limit=100, deflate_every=100,
                                           name="final")
            update_best(p_fin, sc_fin, "final_polish")

    # --- Final validation ---
    if best_seq is not None:
        final = np.asarray(best_seq, dtype=np.float64).copy()
        final = np.clip(final, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s

        # Exact score
        sc_exact = _score_exact(final)
        print(f"\nExact score: {sc_exact:.10f} (n={final.size})")

        # Try evaluator
        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception as e:
                print(f"Evaluator error: {e}")

        return [float(x) for x in final]

    # Fallback
    print("FALLBACK to uniform n=256")
    fb = np.ones(256)
    return [float(x) for x in fb]
