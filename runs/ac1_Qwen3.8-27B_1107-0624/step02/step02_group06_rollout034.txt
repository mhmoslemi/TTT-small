    best_p = None
    best_sc = np.inf
    t_end = time.monotonic() + budget_s

    be = np.array([0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0, -1.0, -2.0, -3.0])
    gs = np.array([0.0, 0.5, 1.0, 2.0, 3.0, 4.0, -1.0, -2.0, -3.0])
    ds = np.array([0.0, 0.5, -0.5, 1.0, -1.0, 1.5, -1.5, 2.0, -2.0])

    step = 0
    total = len(be) * len(gs) * len(ds)
    for b in be:
        for g in gs:
            for d in ds:
                if time.monotonic() > t_end:
                    return best_p, best_sc
                p = make_bump_profile(n, b, g, d)
                sc = exact_score(p)
                if sc < best_sc:
                    best_sc = sc
                    best_p = p.copy()
                step += 1

    # Beta profiles
    alphas = [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 1.2, 0.7]
    betas = [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 1.2, 0.7]
    for al in alphas:
        for bt in betas:
            if time.monotonic() > t_end:
                return best_p, best_sc
            p = make_beta_profile(n, al, bt)
            sc = exact_score(p)
            if sc < best_sc:
                best_sc = sc
                best_p = p.copy()

    # Cosine profiles
    for a0 in [1.0, 1.5, 2.0]:
        for a2 in [-0.1, -0.3, -0.5, -0.7, -1.0, 0.1, 0.3]:
            for a4 in [-0.05, -0.1, -0.2, 0.0, 0.05, 0.1]:
                if time.monotonic() > t_end:
                    return best_p, best_sc
                p = make_cosine_profile(n, a0, a2, a4)
                sc = exact_score(p)
                if sc < best_sc:
                    best_sc = sc
                    best_p = p.copy()

    return best_p, best_sc


def adam_optimize(p0, deadline, rng, base_lr=0.1, beta_init=0.15,
                  use_exact_score=True, exact_every=50):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-14)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    lr = base_lr
    steps = 0

    while time.monotonic() < deadline:
        L, w = soft_max_grad(c, beta)
        gp = 2.0 * corr_weights(w, p)
        reg = 1e-7
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)

        if use_exact_score and steps % exact_every == 0:
            maxc = float(c.max())
            score = float(2.0 * n * maxc)
        else:
            maxc = float(c.max())
            score = float(2.0 * n * maxc)

        steps += 1

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 400 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 100 == 0:
            beta = max(1e-14, beta * 0.75)

        max_no_imp = max(500, 8 * n)
        if no_imp > max_no_imp:
            scale = 0.5 + 0.5 * rng.random()
            z = scale * best_z + 0.4 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
