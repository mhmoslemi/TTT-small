    k = 1
    while k < n:
        k <<= 1

    n = a.size

    return c

    ex = np.exp(x)
    s = np.sum(ex)
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w

    fq = np.fft.rfft(q[::-1], N)

    return h[n-1:2*n-1]

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n = a.size

    if mn < -1e-10:

        c = np.maximum(c, 0)
    return c

    n = p.size

"""
Continuation-based soft-max optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses geometric beta schedule with Adam in logit space, diverse initializations,
and multi-restart polish. Focuses on n=1024 with optional n=2048 escalation.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < -1e-10:
        c = c - mn
    elif mn < 0:
        c = np.maximum(c, 0.0)
    return c


def _corr_fft(w, p):
    n = p.size
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    corr = np.fft.irfft(fw * np.conj(fp), N)
    return corr[:n]


def _softmax_z(z, floor=1e-15):
    z = np.clip(z, -30.0, 30.0)
    z = z - np.max(z)
    e = np.exp(z)
    p = e + floor
    s = p.sum()
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / e.size
    return p / s


def _score(p):
    n = p.size
    c = _conv_fft(p)
    return 2.0 * n * float(c.max())


def _normalize(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, None)
    s = p.sum()
    if s < 1e-10:
        return np.ones_like(p) / p.size
    return p / s


def _make_linear(n, slope=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_quad(n, a1=1.5, a2=0.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_cubic(n, a1=1.8, a3=-1.2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_sine(n, freq=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * freq * t) ** 2 + 0.01
    return _normalize(a)


def _make_exp(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(alpha * (t - 0.5))
    return _normalize(a)


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def _optimize_adam(p0, deadline, rng, beta, lr=0.05, max_iters=20000,
                   no_imp_limit=3000, name=""):
    n = p0.size
    z = np.log(np.maximum(p0, 1e-15))
    z -= np.max(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0

    p = p0.copy()
    best_p = p0.copy()
    best_score = _score(p)
    no_imp = 0
    it = 0

    while it < max_iters and time.monotonic() < deadline:
        c = _conv_fft(p)
        maxc = float(c.max())

        x = (c - maxc) / max(beta, 1e-30)
        ex = np.exp(x)
        w = ex / ex.sum()

        grad_p = 2.0 * _corr_fft(w, p)
        dot = float(np.dot(p, grad_p))
        grad_z = p * (grad_p - dot)

        if not np.all(np.isfinite(grad_z)):
            break

        t_count += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_count)
        vh = v / (1.0 - b2 ** t_count)

        step = lr * mh / (np.sqrt(vh) + eps)
        z = z - step
        z = np.clip(z, -30.0, 30.0)
        z -= np.max(z)
        p = _softmax_z(z)

        sc = _score(p)
        it += 1

        if np.isfinite(sc) and sc < best_score - 1e-13:
            best_score = sc
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > no_imp_limit:
                break

    return best_p, best_score, z


def _continuation_optimize(p0, deadline, rng, n_levels=30, lr_init=0.06,
                           beta_start_frac=0.4, beta_end_frac=1e-5,
                           iters_per_level=8000, name=""):
    n = p0.size
    z = np.log(np.maximum(p0, 1e-15))
    z -= np.max(z)
    p = _softmax_z(z)

    c = _conv_fft(p)
    maxc = float(c.max())

    best_p = p.copy()
    best_score = _score(p)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0

    for level in range(n_levels):
        if time.monotonic() >= deadline:
            break
        frac = level / max(1, n_levels - 1)
        beta = maxc * max(beta_start_frac * (beta_end_frac / beta_start_frac) ** frac, 1e-12)

        lr = lr_init * (1.0 - 0.5 * frac)
        no_imp = 0
        no_imp_limit = max(500, 3 * n)
        it = 0

        while it < iters_per_level and time.monotonic() < deadline and no_imp < no_imp_limit:
            c = _conv_fft(p)
            cur_maxc = float(c.max())
            beta_cur = cur_maxc * max(beta / maxc, 1e-12)

            x = (c - cur_maxc) / max(beta_cur, 1e-30)
            ex = np.exp(x)
            w = ex / ex.sum()

            grad_p = 2.0 * _corr_fft(w, p)
            dot = float(np.dot(p, grad_p))
            grad_z = p * (grad_p - dot)

            if not np.all(np.isfinite(grad_z)):
                no_imp = no_imp_limit + 1
                break

            t_count += 1
            m = b1 * m + (1.0 - b1) * grad_z
            v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
            mh = m / (1.0 - b1 ** t_count)
            vh = v / (1.0 - b2 ** t_count)

            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -30.0, 30.0)
            z -= np.max(z)
            p = _softmax_z(z)

            sc = _score(p)
            it += 1

            if np.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_p = p.copy()
                no_imp = 0
            else:
                no_imp += 1

        if level % 5 == 0 or level == n_levels - 1:
            print(f"    [{name}] L{level+1:2d}/{n_levels} beta={beta:.2e} sc={best_score:.10f} it={it}")

    return best_p, best_score


def _active_set_polish(p0, deadline, rng, n_iter=5000, lr=0.01, name=""):
    n = p0.size
    p = p0.copy()
    z = np.log(np.maximum(p, 1e-15))
    z -= np.max(z)

    best_p = p0.copy()
    best_score = _score(p)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0

    it = 0
    while it < n_iter and time.monotonic() < deadline:
        c = _conv_fft(p)
        maxc = float(c.max())

        # Use exact active set: all lags within 1% of max
        threshold = maxc * 0.99
        active = c >= threshold
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        grad_p = 2.0 * _corr_fft(w, p)
        dot = float(np.dot(p, grad_p))
        grad_z = p * (grad_p - dot)

        if not np.all(np.isfinite(grad_z)):
            break

        t_count += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_count)
        vh = v / (1.0 - b2 ** t_count)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -30.0, 30.0)
        z -= np.max(z)
        p = _softmax_z(z)

        sc = _score(p)
        it += 1
        if np.isfinite(sc) and sc < best_score - 1e-13:
            best_score = sc
            best_p = p.copy()

    return best_p, best_score


def _multi_restart(p0, deadline, rng, n_restarts=6, noise_scales=None, name=""):
    n = p0.size
    if noise_scales is None:
        noise_scales = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]

    best_p = p0.copy()
    best_score = _score(p0)

    z_base = np.log(np.maximum(p0, 1e-15))
    z_base -= np.max(z_base)

    per_time = (deadline - time.monotonic()) / max(1, n_restarts)

    for i in range(n_restarts):
        if time.monotonic() >= deadline:
            break
        per_end = min(deadline, time.monotonic() + max(30.0, per_time))
        ns = noise_scales[i] if i < len(noise_scales) else 0.3

        if i == 0:
            p_start = p0.copy()
        else:
            z_try = z_base + ns * rng.normal(size=n)
            z_try = np.clip(z_try, -30, 30)
            z_try -= np.max(z_try)
            p_start = _softmax_z(z_try)

        p_try, sc_try = _continuation_optimize(
            p_start, per_end, rng, n_levels=15, lr_init=0.04,
            beta_start_frac=0.3, beta_end_frac=1e-4,
            iters_per_level=3000, name=f"{name}_r{i}")

        if sc_try < best_score:
            best_score = sc_try
            best_p = p_try.copy()
            print(f"    [{name}] restart {i}: {sc_try:.10f} (noise={ns:.2f})")

    return best_p, best_score
