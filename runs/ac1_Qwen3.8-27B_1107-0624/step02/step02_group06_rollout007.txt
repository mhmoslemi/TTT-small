```python
def conv_gradient(dc, p):
    """Backprop through c = conv(p, p). dc has length 2n-1, p has length n. Returns grad_p of length n."""
    n = len(p)
    # grad_p[j] = sum_k dc[k] * (p[k-j] + p[j-k]) with bounds
    # Use the fact that this is a cross-correlation
    npad = n
    # First term: sum_k dc[k] * p[k-j]
    # For j=0: sum_k dc[k] * p[k-padded]... 
    # Let me just use np.correlate
    # np.correlate(a, v)[k] = sum_n a[n+k] * v[n] (for full mode, k ranges appropriately)
    
    # Term 1: T1[j] = sum_k dc[k] * p[k-j] where 0<=k-j<n
    # = sum_m dc[j+m] * p[m] where m = k-j, 0<=m<n, 0<=j+m<2n-1
    # This is the cross-correlation of dc with p, at lag j
    # np.correlate(dc, p, mode='full')[offset + j] for appropriate offset
    
    full1 = np.correlate(dc, p, mode='full')
    # full1[k] = sum_n dc[n+k] * p[n] for k from -(n-1) to (2n-2)
    # We want k = 0, 1, ..., n-1
    # In the full output, k=0 is at index n-1
    term1 = full1[n-1:2*n-1]  # indices n-1 to 2n-2, giving k=0 to n-1
    
    # Term 2: T2[j] = sum_k dc[k] * p[j-k] where 0<=j-k<n
    # = sum_m dc[j-m] * p[m] where m = j-k, 0<=m<n, 0<=j-m<2n-1
    # This is the cross-correlation of p with dc, at lag j
    full2 = np.correlate(p, dc, mode='full')
    # full2[k] = sum_n p[n+k] * dc[n] for k from ... to ...
    # We want the result at "lag j" for j = 0, ..., n-1
    # In the full output, k=0 is at index (2n-2) (len(dc)-1)
    term2 = full2[(2*n-2):3*n-3]  # indices (2n-2) to (3n-3), giving k=0 to n-1
    
    return term1 + term2
```
