"""
FFT-based softmax gradient descent for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive temperature annealing, multi-scale search (n=512,1024,2048),
diverse starting points, and scipy L-BFGS-B polish.
"""

import numpy as np

from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    m = c.max()
    return 2.0 * n * m / (s * s)


def compute_grad_z(z, beta):
    """
    Compute gradient of soft-max(max conv(a,a)) w.r.t. logit parameter z,
    where a = softmax(z) normalized to sum 1.
    Returns (soft_max_value, gradient).
    """
    n = z.size
    a = np.exp(z - z.max())
    s = a.sum()
    a = a / s

    c = fft_convolve(a)
    c_max = c.max()

    # Soft-max weights (numerically stable)
    x = (c - c_max) / beta
    ex = np.exp(x)
    ex_sum = ex.sum()
    if ex_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / ex_sum

    sm_val = c_max + beta * np.log(max(ex_sum, 1e-300))

    # Gradient w.r.t. a: grad_a[j] = 2 * sum_k w[k] * a[k-j]
    # Computed as cross-correlation via FFT
    N = next_pow2(3 * n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(fw * np.conj(fa), N)[:n]
    grad_a = 2.0 * corr

    # Project to tangent space of sum(a)=1: grad_z = a * (grad_a - <a, grad_a>)
    a_dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - a_dot)

    return sm_val, grad_z, a

        nonlocal best_seq, best_score
        if a is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(a, dtype=np.float64).copy()
            best_score = float(sc)

    # Load parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
                sc = compute_score(parent)
                try_update(parent, sc)
                print(f"Parent: n={parent.size}, score={sc:.10f}")
    except Exception as e:
        print(f"No parent available: {e}")

    candidates = []  # (score, n, sequence)

    # Phase 1: Exploration at multiple scales
    print(f"\n=== Phase 1: Exploration ===")
    n_values = [512, 1024, 2048]
    phase1_end = min(deadline, t_start + 500.0)

    for n in n_values:
        if time.monotonic() > phase1_end - 30:
            break
        starts = make_starts(n, rng, parent)
        # Limit starts to keep time manageable
        max_starts = min(len(starts), 12 if n <= 512 else 8)
        per_start = max(15.0, (phase1_end - time.monotonic()) / max(1, max_starts) * 0.8)

        print(f"  n={n}, {len(starts)} starts available, using {max_starts}")
        for i in range(max_starts):
            if time.monotonic() > phase1_end - 10:
                break
            a0 = starts[i]
            a0 = np.maximum(a0, 1e-12)
            a0 = a0 / a0.sum()
            end_t = min(phase1_end - 5, time.monotonic() + per_start)
            a_best, sc = grad_descend(a0, end_t, rng,
                                       beta_start=0.1, beta_end=0.01,
                                       lr=0.12, max_iter=50000,
                                       beta_decay_start=30,
                                       beta_decay_rate=0.997)
            try_update(a_best, sc)
            candidates.append((sc, n, a_best))
            if i % 4 == 0:
                print(f"    start {i}/{max_starts}: score={sc:.10f}, global_best={best_score:.10f}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.10f} (n={candidates[0][1] if candidates else 0})")

    # Phase 2: Deep refinement of top candidates
    print(f"\n=== Phase 2: Deep Refinement ===")
    phase2_end = min(deadline, t_start + 1200.0)
    top_k = min(4, len(candidates))

    for i in range(top_k):
        if time.monotonic() > phase2_end - 30:
            break
        sc_i, n_i, a_i = candidates[i]
        # Only refine if this candidate is competitive
        if sc_i > best_score * 1.05:
            continue
        end_t = min(phase2_end - 10, time.monotonic() + 200.0)
        a_best, sc = grad_descend(a_i, end_t, rng,
                                   beta_start=0.04, beta_end=0.0005,
                                   lr=0.06, max_iter=300000,
                                   beta_decay_start=100,
                                   beta_decay_rate=0.999,
                                   restart_threshold=600)
        try_update(a_best, sc)
        print(f"  Deep refine cand {i} (n={n_i}, was {sc_i:.6f}): score={sc:.10f}")

    print(f"Phase 2 done. Best: {best_score:.10f}")

    # Phase 3: scipy L-BFGS-B polish
    print(f"\n=== Phase 3: L-BFGS-B Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = best_seq.size
        a0 = best_seq.copy()

        def lbbf_obj(z):
            a = np.exp(z - z.max())
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            return compute_score(a)

        z0 = np.log(np.maximum(a0, 1e-15))
        z0 -= z0.mean()

        try:
            res = minimize(lbbf_obj, z0, method='L-BFGS-B',
                           options={'maxiter': 3000, 'maxfun': 30000, 'gtol': 1e-8})
            if np.isfinite(res.fun):
                z_res = res.x
                a_res = np.exp(z_res - z_res.max())
                a_res = np.maximum(a_res, 1e-15)
                a_res = a_res / a_res.sum()
                sc = compute_score(a_res)
                try_update(a_res, sc)
                print(f"  L-BFGS-B (n={n_best}): score={sc:.10f}, status={res.status}")
        except Exception as e:
            print(f"  L-BFGS-B error: {e}")

    # Phase 4: Cross-scale - try interpolating best to other n values
    print(f"\n=== Phase 4: Cross-Scale ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for n_target in [512, 1024, 2048]:
            if time.monotonic() > deadline - 40:
                break
            if n_target == best_seq.size:
                continue
            a_ext = interp_to_length(best_seq, n_target)
            end_t = min(deadline - 30, time.monotonic() + 80.0)
            a_best, sc = grad_descend(a_ext, end_t, rng,
                                       beta_start=0.05, beta_end=0.005,
                                       lr=0.08, max_iter=100000,
                                       beta_decay_start=50,
                                       beta_decay_rate=0.998)
            try_update(a_best, sc)
            print(f"  n={n_target}: score={sc:.10f}")

    # Phase 5: Final short polish on absolute best
    print(f"\n=== Phase 5: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        end_t = min(deadline, time.monotonic() + 60.0)
        a_best, sc = grad_descend(best_seq, end_t, rng,
                                   beta_start=0.02, beta_end=0.001,
                                   lr=0.04, max_iter=100000,
                                   beta_decay_start=20,
                                   beta_decay_rate=0.999)
        try_update(a_best, sc)
        print(f"  Final polish: score={sc:.10f}")

    # Final result
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        # Verify with evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(sc_eval):
                print(f"Final evaluated score: {sc_eval:.10f}")
                if sc_eval < sc_final:
                    print("NOTE: evaluator gave lower score (clamping effect)")
        except Exception as e:
            print(f"Evaluation check error: {e}")

        return [float(x) for x in final]

    # Fallback
    print("Fallback: uniform sequence")
    fb = np.ones(512)
    return [float(1.0) for _ in range(512)]
