def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_direction(f, M, timeout=10.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M for all x.
    Returns the optimal g (or None if infeasible).
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def spectral_init(n, rng, n_iter=200):
    """
    Initialize a sequence using spectral shaping (Gerchberg-Saxton-like).
    Aims for flat magnitude spectrum with non-negative time-domain samples.
    """
    a = rng.random(n)
    a = a / a.sum()
    
    for _ in range(n_iter):
        # Go to frequency domain
        A = np.fft.fft(a)
        # Flatten magnitude: set all to geometric mean
        mags = np.abs(A)
        mags[mags < 1e-10] = 1e-10
        geo_mean = np.exp(np.mean(np.log(mags)))
        # Keep phase, set magnitude to geometric mean
        phases = np.angle(A)
        A_new = geo_mean * np.exp(1j * phases)
        # But keep DC component fixed (sum constraint)
        A_new[0] = np.abs(A[0])
        # Go back to time domain
        a = np.real(np.fft.ifft(A_new))
        # Project onto non-negative
        a = np.maximum(a, 0)
        # Normalize
        s = a.sum()
        if s < 1e-10:
            a = rng.random(n)
            a = a / a.sum()
        else:
            a = a / s
    
    return a

def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..n-1."""
    n = len(p)
    h = np.zeros(n)
    for i in range(n):
        h[i] = np.dot(w[i:i + n], p)
    return h

            m.fill(0)

            beta = max(1e-10, 0.1 * float(np.convolve(

def interpolate_seq(a, n_new):

    x_new = np.linspace(0.0, 1.0, n_new)

    return normalize(a_new)
