Hybrid coordinate descent + soft-max mirror descent for minimizing

def compute_score(p):

    n = len(p)
    c = np.convolve(p, p)

    p = np.maximum(p, 0.0)

    """Linearly interpolate a profile to new length, then renormalize."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)

    w = soft_max_weights(c, beta)
    # grad[j] = 4n * sum_m p[m] * w[m+j] = 4n * correlate(w, p, 'full')[j]
    corr = np.correlate(w, p, mode='full')
    grad = 4.0 * n * corr[:n]
    return grad, float(c.max())


def coord_step_exact(p, j, conv, S):
    """
    Find optimal value of p[j] given all other elements fixed.
    Uses analytical critical points + sampling.
    Returns new value for coordinate j.
    """
    n = len(p)
    x = p[j]
    S_prime = S - x

    if S_prime < 1e-15:
        return x

    # p[j] appears in conv[k] for k in [j, min(j+n-1, 2n-2)]
    k_lo = j
    k_hi = min(j + n - 1, 2 * n - 2)
    ks = np.arange(k_lo, k_hi + 1)

    # conv[k] = base + x * D where D[k] = p[k-j]
    D = p[ks - j]  # shape (len(ks),)
    base = conv[ks] - x * D  # shape (len(ks),)

    # Unchanged conv max
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[ks] = False
    if mask.any():
        unchanged_max = conv[mask].max()
    else:
        unchanged_max = 0.0

    # Generate candidate x values
    candidates = [0.0, x, S_prime * 0.1, S_prime * 0.5]

    for m in range(len(D)):
        if D[m] > 1e-15:
            # Critical point: minimize (base[m] + x*D[m]) / (S_prime + x)^2
            # d/dx = [D*(S'+x) - 2*(base+x*D)] / (S'+x)^3 = 0
            # D*S' + Dx - 2*base - 2xD = 0
            # D*S' - 2*base - xD = 0
            # x = (D*S' - 2*base) / D = S' - 2*base/D
            xc = S_prime - 2.0 * base[m] / D[m]
            if xc > 0 and xc < S_prime * 100:
                candidates.append(xc)
            # Breakpoint with unchanged_max
            xb = (unchanged_max - base[m]) / D[m]
            if 0 < xb < S_prime * 100:
                candidates.append(xb)

    # Top-constraint breakpoints
    vals = base + x * D
    if len(D) > 2:
        top_k = min(8, len(D))
        top_idx = np.argsort(vals)[-top_k:]
        for i in range(len(top_idx)):
            for j2 in range(i + 1, len(top_idx)):
                m1, m2 = top_idx[i], top_idx[j2]
                dd = D[m1] - D[m2]
                if abs(dd) > 1e-15:
                    xb = (base[m2] - base[m1]) / dd
                    if 0 < xb < S_prime * 100:
                        candidates.append(xb)

    # Evaluate candidates
    best_x = x
    best_score = compute_score(p)  # current score as baseline

    for xc in candidates:
        if xc < 0 or not np.isfinite(xc):
            continue
        new_change = base + xc * D
        max_changed = new_change.max()
        max_total = max(unchanged_max, max_changed)
        denom = (S_prime + xc) ** 2
        if denom < 1e-30:
            continue
        sc = 2.0 * n * max_total / denom
        if sc < best_score:
            best_score = sc
            best_x = xc

    return best_x


def coord_descent_polish(p, deadline, max_sweeps=5000):
    """
    Run coordinate descent sweeps to polish p.
    Reconvolves after each coordinate update (fast for n<=512).
    """
    n = len(p)
    p = p.copy().astype(np.float64)
    S = p.sum()
    conv = np.convolve(p, p)
    best_p = p.copy()
    best_sc = compute_score(p)
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        changed = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            new_val = coord_step_exact(p, j, conv, S)
            if abs(new_val - p[j]) > 1e-14:
                delta = new_val - p[j]
                p[j] = new_val
                S += delta
                # Update conv incrementally
                ks = np.arange(j, min(j + n, 2 * n - 1))
                # For k != 2j: conv[k] += delta * p[k-j]
                # For k == 2j: conv[k] += delta * (new + old) = delta * (2*new - delta)
                for k in ks:
                    if k == 2 * j and 2 * j < 2 * n - 1:
                        conv[k] += delta * (2.0 * new_val - delta)
                    else:
                        conv[k] += delta * p[k - j]
                changed = True

        if changed:
            sc = compute_score(p)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_p = p.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 5:
            break

    return best_p, best_sc


def mirror_descent_polish(p, deadline, rng, lr_init=0.1, beta_init=None):
    """
    Soft-max mirror descent in log domain for global optimization.
    """
    n = len(p)
    p = p.copy().astype(np.float64)
    p = np.maximum(p, 1e-15)
    p = p / p.sum()

    z = np.log(p)
    z -= z.mean()

    beta = beta_init if beta_init is not None else 0.05 * np.convolve(p, p).max()
    lr = lr_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_p = p.copy()
    best_z = z.copy()
    best_sc = compute_score(p)
    no_imp = 0

    while time.monotonic() < deadline:
        grad, cmax = compute_gradient(p, beta)

        # Center the gradient (remove mean to preserve sum)
        # In log domain: dz = -lr * p * (grad - sum(p*grad))
        pg = p * grad
        dz = -lr * (grad - pg.sum())
        dz -= dz.mean()

        t_adam += 1
        m = b1 * m + (1 - b1) * dz
        v = b2 * v + (1 - b2) * (dz * dz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * 2 * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            continue

        z = np.clip(z, -20, 20)
        z -= z.mean()
        p = np.exp(z)
        p = p / p.sum()

        sc = compute_score(p)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta decay
        if t_adam % 200 == 0:
            beta *= 0.85

        # Learning rate decay
        if no_imp > 300:
            lr = max(0.01, lr * 0.9)

        # Restart
        if no_imp > max(500, 2 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n) * np.sqrt(no_imp / 1000.0)
            z = np.clip(z, -20, 20)
            z -= z.mean()
            p = np.exp(z)
            p = p / p.sum()
            beta = 0.05 * np.convolve(p, p).max()
            lr = lr_init
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            no_imp = 0

    return best_p, best_sc


def make_starting_points(n, rng, count=20):
    """Generate diverse starting points."""
    starts = []
    t = np.linspace(0, 1, n)

    # Linear profiles
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        f = 1.0 + slope * t
        starts.append(f / f.sum())

    # Quadratic
    for a in [-1.0, -0.5, 0.5, 1.0]:
        f = 1.0 + a * (2 * t - 1) ** 2
        f = np.maximum(f, 1e-6)
        starts.append(f / f.sum())

    # Cubic (antisymmetric)
    for a1 in [1.0, 1.5, 2.0, 3.0]:
        f = 1.0 + 0.5 * a1 * (2 * t - 1) - 0.3 * a1 * (2 * t - 1) ** 3
        f = np.maximum(f, 1e-6)
        starts.append(f / f.sum())

    # Bimodal
    for sigma in [0.1, 0.15, 0.2]:
        f = np.exp(-((t - 0.2) / sigma) ** 2) + np.exp(-((t - 0.8) / sigma) ** 2) + 0.05
        starts.append(f / f.sum())

    # Trigonometric
    for k in [2, 3, 4]:
        for amp in [0.3, 0.5]:
            f = 1.0 + amp * np.cos(2 * np.pi * k * t)
            f = np.maximum(f, 1e-6)
            starts.append(f / f.sum())

    # Random Fourier
    for _ in range(count):
        coeffs = rng.normal(0, 1, 6)
        f = 1.0
        for k in range(3):
            f += 0.3 * coeffs[2 * k] * np.cos(2 * np.pi * (k + 1) * t)
            f += 0.3 * coeffs[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * t)
        f = np.maximum(f, 1e-6)
        starts.append(f / f.sum())

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Coordinate descent from diverse starts at n=256
    Phase 2: Mirror descent polish on best results at n=256
    Phase 3: Interpolate to n=512, coordinate descent + mirror descent
    Phase 4: Try n=1024 if time permits
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: score={sc:.10f}, n={len(p)}, t={time.monotonic()-t_start:.1f}s")

    # Get parent sequence if available
    parent = None
    try:
        p_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if p_arr.ndim == 1 and p_arr.size > 0:
            pn = np.maximum(p_arr, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
                print(f"Using parent sequence: n={len(parent)}, score={compute_score(parent):.10f}")
                try_update(parent, compute_score(parent))
    except Exception:
        pass

    # ===== Phase 1: Coordinate descent at n=256 =====
    n1 = 256
    phase1_end = min(deadline, t_start + 300.0)
    print(f"\n=== Phase 1: Coordinate descent at n={n1} ===")

    starts = make_starting_points(n1, rng, count=15)
    if parent is not None:
        if len(parent) == n1:
            starts.insert(0, parent.copy())
        else:
            starts.insert(0, interpolate_profile(parent, n1))

    # Add parent perturbed
    if parent is not None:
        for _ in range(5):
            pert = parent + 0.1 * rng.normal(size=n1)
            pn = normalize(pert)
            if pn is not None:
                starts.append(pn)

    results_1 = []
    for i, s in enumerate(starts):
        remaining = phase1_end - time.monotonic()
        if remaining < 2:
            break
        alloc = min(remaining / max(1, len(starts) - i), 15.0)
        if alloc < 1:
            break
        t_dl = time.monotonic() + alloc
        p_best, sc_best = coord_descent_polish(s, t_dl)
        results_1.append((sc_best, p_best))
        try_update(p_best, sc_best)
        if (i + 1) % 5 == 0:
            print(f"  Phase 1: {i+1}/{len(starts)} done, best={best_score:.10f}")

    results_1.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.10f}")

    # ===== Phase 2: Mirror descent polish at n=256 =====
    phase2_end = min(deadline, t_start + 600.0)
    print(f"\n=== Phase 2: Mirror descent polish at n={n1} ===")

    top_k = min(5, len(results_1))
    results_2 = []
    for i in range(top_k):
        remaining = phase2_end - time.monotonic()
        if remaining < 5:
            break
        alloc = min(remaining / max(1, top_k - i), 50.0)
        if alloc < 3:
            break
        t_dl = time.monotonic() + alloc
        p_best, sc_best = mirror_descent_polish(results_1[i][1], t_dl, rng, lr_init=0.05)
        results_2.append((sc_best, p_best))
        try_update(p_best, sc_best)
        print(f"  Mirror {i+1}/{top_k}: score={sc_best:.10f}")

        # Follow up with more coordinate descent
        remaining2 = phase2_end - time.monotonic()
        if remaining2 > 5:
            t_dl2 = time.monotonic() + min(remaining2, 20.0)
            p_cd, sc_cd = coord_descent_polish(p_best, t_dl2)
            results_2.append((sc_cd, p_cd))
            try_update(p_cd, sc_cd)

    all_results = results_1 + results_2
    all_results.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best: {best_score:.10f}")

    # ===== Phase 3: Interpolate to n=512 and optimize =====
    n3 = 512
    phase3_end = min(deadline, t_start + 1300.0)
    print(f"\n=== Phase 3: Optimization at n={n3} ===")

    results_3 = []
    top_c = min(4, len(all_results))
    for i in range(top_c):
        remaining = phase3_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_c - i), 100.0)
        if alloc < 5:
            break

        p_ext = interpolate_profile(all_results[i][1], n3)
        t_dl = time.monotonic() + alloc * 0.5
        p_cd, sc_cd = coord_descent_polish(p_ext, t_dl)
        results_3.append((sc_cd, p_cd))
        try_update(p_cd, sc_cd)
        print(f"  n=512 CD {i+1}/{top_c}: score={sc_cd:.10f}")

        # Mirror descent polish
        remaining2 = phase3_end - time.monotonic()
        if remaining2 > 10:
            t_dl2 = time.monotonic() + min(remaining2 * 0.5, 60.0)
            p_md, sc_md = mirror_descent_polish(p_cd, t_dl2, rng, lr_init=0.03)
            results_3.append((sc_md, p_md))
            try_update(p_md, sc_md)
            print(f"  n=512 MD {i+1}: score={sc_md:.10f}")

            # Final CD polish
            remaining3 = phase3_end - time.monotonic()
            if remaining3 > 5:
                t_dl3 = time.monotonic() + min(remaining3, 30.0)
                p_final, sc_final = coord_descent_polish(p_md, t_dl3)
                results_3.append((sc_final, p_final))
                try_update(p_final, sc_final)

    all_results_3 = all_results + results_3
    all_results_3.sort(key=lambda x: x[0])
    print(f"Phase 3 done. Best: {best_score:.10f}")

    # ===== Phase 4: Try n=1024 if time permits =====
    n4 = 1024
    phase4_end = deadline
    remaining = phase4_end - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Optimization at n={n4} (time={remaining:.0f}s) ===")
        results_4 = []
        top_d = min(3, len(all_results_3))
        for i in range(top_d):
            remaining = phase4_end - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining / max(1, top_d - i), 60.0)
            if alloc < 5:
                break

            p_ext = interpolate_profile(all_results_3[i][1], n4)
            t_dl = time.monotonic() + alloc * 0.4
            p_cd, sc_cd = coord_descent_polish(p_ext, t_dl)
            results_4.append((sc_cd, p_cd))
            try_update(p_cd, sc_cd)
            print(f"  n=1024 CD {i+1}: score={sc_cd:.10f}")

            # Light mirror descent
            remaining2 = phase4_end - time.monotonic()
            if remaining2 > 15:
                t_dl2 = time.monotonic() + min(remaining2 * 0.4, 30.0)
                p_md, sc_md = mirror_descent_polish(p_cd, t_dl2, rng, lr_init=0.02)
                results_4.append((sc_md, p_md))
                try_update(p_md, sc_md)
                print(f"  n=1024 MD {i+1}: score={sc_md:.10f}")

        all_results_3 = all_results_3 + results_4
        all_results_3.sort(key=lambda x: x[0])
        print(f"Phase 4 done. Best: {best_score:.10f}")

    # ===== Phase 5: Final intensive polish on best solution =====
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_seq is not None:
        print(f"\n=== Phase 5: Final polish (time={remaining:.0f}s) ===")
        n_best = len(best_seq)
        # Alternate between CD and MD
        t_dl = deadline
        for round_idx in range(5):
            if time.monotonic() > deadline:
                break
            # CD
            p_cd, sc_cd = coord_descent_polish(best_seq, min(deadline, time.monotonic() + remaining * 0.3))
            try_update(p_cd, sc_cd)
            # MD
            if time.monotonic() < deadline - 5:
                p_md, sc_md = mirror_descent_polish(p_cd, min(deadline, time.monotonic() + remaining * 0.3), rng, lr_init=0.02)
                try_update(p_md, sc_md)
                best_seq = p_md
        # Final CD
        p_final, sc_final = coord_descent_polish(best_seq, deadline)
        try_update(p_final, sc_final)

    # ===== Final validation =====
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = compute_score(best_seq)

    # Ensure validity
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = np.ones(256) / 256.0
        best_score = compute_score(best_seq)
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final score
    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}")

    # Cross-check with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
        if np.isfinite(ev_score) and ev_score < final_score:
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    # Pick the best from all results (in case tracking missed something)
    for sc, p in all_results_3:
        if np.isfinite(sc) and sc < final_score:
            p = np.maximum(p, 0.0)
            ps = p.sum()
            if ps > 0.01:
                p = p / ps
                p = np.clip(p, 0.0, 1000.0)
                final_score = compute_score(p)
                best_seq = p

    print(f"FINAL ANSWER: n={len(best_seq)}, score={compute_score(best_seq):.10f}")
    return [float(x) for x in best_seq]
