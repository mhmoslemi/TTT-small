Optimized step function search using exponential profiles, gradient methods,
and LP refinement. Targets score < 1.5027.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w / w.sum()

def _gradient_optimize(a, deadline, max_iter=50000):
    """Nesterov accelerated gradient with smooth max and simplex projection."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _score(a)  # = 2n * max_c (since sum=1)
    
    c = _fft_convolve(a)
    beta = 0.05 * float(c.max())
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    it = 0
    no_imp = 0
    
    while it < max_iter and time.monotonic() < deadline:
        # Nesterov lookahead
        y = a + momentum * v
        
        val, grad, c = _smooth_max_grad(y, beta)
        
        # Scale gradient: we're optimizing 2n * smooth_max, gradient is 2n * grad_smooth
        # But since we project to simplex (sum=1), we need to account for that
        # The effective gradient on the simplex is grad - mean(grad) * a (tangent to simplex)
        # Actually, for projection, we just use the raw gradient
        grad_effective = grad - np.mean(grad)  # tangent component
        
        v = momentum * v - lr * grad_effective
        a_new = _project_simplex(a + v)
        a = a_new
        it += 1
        
        # Evaluate
        c = _fft_convolve(a)
        score = 2.0 * n * float(c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive parameters
        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)
        
        if no_imp > 2000:
            lr = max(0.001, lr * 0.9)
        
        if no_imp > 10000:
            # Restart from best with perturbation
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.05 * float(_fft_convolve(a).max())
            lr = 0.01
            no_imp = 0
    
    return best_a

def _lp_spread(a, M, max_constraints=None):
    """LP to find best direction that keeps convolution <= M."""
    n = len(a)
    c_obj = -np.ones(n)
    
    # Build constraint matrix: A[j, k] = a[k-j] if valid
    # To save memory, only include constraints where a is significantly non-zero
    # Or just build the full matrix for n <= 1024
    
    if n > 1024:
        # Use a subset of constraints
        # Only include constraints that are near-active
        c = _fft_convolve(a)
        threshold = M * 0.9
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([n-1])  # at least the center
        # Build sparse constraint
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for k in active:
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    rows.append(k)
                    cols.append(j)
                    vals.append(a[ki])
        A_ub = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
        b_ub = np.full(len(active), M)
    else:
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = a
        if max_constraints is not None:
            # Only keep the most active constraints
            c = _fft_convolve(a)
            idx = np.argsort(c)[::-1][:max_constraints]
            A = A[idx]
        else:
            A = A
        b_ub = np.full(A.shape[0], M)
        A_ub = A
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _lp_optimize(a, deadline, max_iter=500, inflate=1.001, blend=0.3):
    """LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)

    best_a = a.copy()

        if g is None:

            if no_imp > 200:
                break
            continue

def _make_exp_profile(n, alpha):
    """Exponential profile: a[j] = exp(alpha * j / (n-1))"""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * j / (n - 1))
    return a / a.sum()

def _make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_power_profile(n, p):
    """Power law: a[j] = (j+1)^p"""
    j = np.arange(n, dtype=np.float64) + 1
    a = j ** p
    return a / a.sum()

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 0 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 0 else np.ones(n_new) / n_new

"""
Optimized step function search: combines exponential/linear/power profile initializations,
Nesterov accelerated gradient with adaptive smooth-max, and LP refinement.
Targets score < 1.5027 by trying multiple lengths and intensive per-candidate optimization.
"""

import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1, so score = 2n * max_c."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_and_grad(a, beta):
    """Compute smooth max value and its gradient."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))

    # grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices only)
    # = 2 * cross_correlation(w, a)[j]
    N2 = _next_pow2(3 * n)
    fw = np.fft.rfft(w, N2)
    fa = np.fft.rfft(a, N2)
    cross = np.fft.irfft(np.conj(fw) * fa, N2)
    grad = 2.0 * cross[:n]
    return val, grad, c


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_optimize(a, deadline, max_iter=100000, verbose=True):
    """Nesterov accelerated gradient with smooth max, simplex projection."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    max_c = float(c.max())
    best_score = float(2.0 * n * max_c)

    beta = 0.1 * max_c
    lr = 0.005
    momentum = 0.9
    v = np.zeros(n)
    it = 0
    no_imp = 0
    base_lr = lr

    while it < max_iter and time.monotonic() < deadline:
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0.0)

        val, grad, c = _smooth_max_and_grad(y, beta)

        # Project gradient onto simplex tangent space
        grad_tan = grad - np.mean(grad)

        # Adaptive: check magnitude

        if gmag < 1e-15:
            break

        # Scale lr by gradient magnitude for stability
        lr_eff = lr * min(1.0, 0.01 / (gmag + 1e-12))

        v = momentum * v - lr_eff * grad_tan
        a_new = _project_simplex(a + v)
        a = a_new
        it += 1

        # Evaluate (less frequently to save time)
        if it % 5 == 0:
            c = _fft_convolve(a)
            score = float(2.0 * n * c.max())
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1

        # Adaptive beta (decrease geometrically)

            beta = max(1e-12, beta * 0.85)

        # Adaptive lr

            lr = max(0.0005, lr * 0.95)
        if no_imp > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.1 * float(_fft_convolve(a).max())
            lr = base_lr
            no_imp = 0

    # Final evaluation
    c = _fft_convolve(a)
    score = float(2.0 * n * c.max())
    if score < best_score:

        best_a = a.copy()

def _lp_optimize(a, deadline, max_iter=300, inflate=1.001, blend=0.3):
    """LP-based refinement using only active constraints for speed."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    no_imp = 0
    cur_blend = blend
    cur_inflate = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * cur_inflate

        # Build constraint matrix (only active constraints for speed)
        # Active: top min(200, 2n-1) values of c
        n_constraints = min(300, 2 * n - 1)
        idx = np.argsort(c)[::-1][:n_constraints]

        A = np.zeros((n_constraints, n))
        for i, k in enumerate(idx):
            jmin = max(0, k - n + 1)
            jmax = min(k, n - 1)
            for j in range(jmin, jmax + 1):
                A[i, j] = a[k - j]

        b_ub = np.full(n_constraints, M)
        c_obj = -np.ones(n)

        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')

        if not res.success:
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 30 == 0:
                cur_inflate = min(1.02, cur_inflate * 1.05)
            if no_imp > 150:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - cur_blend) * a + cur_blend * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                cur_blend = max(0.05, cur_blend * 0.9)
            if no_imp > 200:
                a = best_a.copy()
                cur_blend = blend
                cur_inflate = inflate
                no_imp = 0

        a = a_new

    print(f"    LP: {it} iters, score={best_score:.10f}")
    return best_a


def _make_exp_profile(n, alpha):
    j = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * j / max(1, n - 1))
    return a / a.sum()


def _make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    j = np.arange(n, dtype=np.float64) + 1.0
    a = j ** p
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):

    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 0 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 0 else np.ones(n_new) / n_new

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: n={parent.size}, score={sc:.10f}")
    except Exception:
        pass

    # Strategy: Focus most time on the parent (n=512), then try n=1024

    # === MAIN: Optimize parent ===
    if parent is not None and parent.size <= 1024:
        main_deadline = t_start + (deadline - t_start) * 0.65
        print(f"\n=== Main optimization on parent (n={parent.size}), "
              f"budget={main_deadline - t_start:.0f}s ===")
        a_opt, sc_opt = _optimize_candidate(parent, main_deadline, phase_label="parent")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

    # === SECONDARY: Try n=1024 ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        n_try = 1024
        sec_deadline = t_start + (time.monotonic() - t_start) + remaining * 0.5
        print(f"\n=== Secondary: n={n_try}, budget={sec_deadline - t_start:.0f}s ===")

        # Best initialization for n
        if best_seq is not None:
            a_init = _interpolate_profile(best_seq, n_try)
        else:
            a_init = _make_exp_profile(n_try, 2.0)

        # Also prepare a few alternatives
        alts = [
            _make_exp_profile(n_try, 1.5),
            _make_exp_profile(n_try, 2.5),
            _make_exp_profile(n_try, 3.0),
            _make_linear_profile(n_try, 2.5),
            _make_linear_profile(n_try, 3.5),
            _make_power_profile(n_try, 1.5),
            _make_quadratic_profile(n_try, 2.0, -1.0),
        ]

        # Score all and pick best 2
        candidates = [(a_init, "interp")] + [(a, f"alt{i}") for i, a in enumerate(alts)]
        scored = []
        for a_c, label in candidates:
            sc = _score(a_c)
            if np.isfinite(sc):
                scored.append((sc, a_c, label))
        scored.sort(key=lambda x: x[0])

        # Optimize the best one
        if scored:
            sc_best, a_best, label = scored[0]
            print(f"  Best init for n={n_try}: {label} with score {sc_best:.6f}")
            a_opt, sc_opt = _optimize_candidate(a_best, sec_deadline, phase_label=label)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()

    # === TERTIARY: If time remains, try optimizing the best seq further ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        print(f"\n=== Tertiary: Final polish, budget={remaining:.0f}s ===")
        a_opt, sc_opt = _optimize_candidate(best_seq, deadline, phase_label="final")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

    # Fallback
    if best_seq is None:
        best_seq = _make_exp_profile(512, 2.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_exp_profile(512, 2.0)

    final_sc = _score(final_seq)
    print(f"\nFINAL RESULT: score={final_sc:.10f}, n={final_seq.size}")

    # Verify with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        print(f"evaluate_sequence confirms: {sc_eval:.10f}")
        if np.isfinite(sc_eval) and sc_eval < final_sc:
            final_sc = sc_eval
    except Exception as e:
        print(f"  Evaluation check: {e}")

    return [float(x) for x in final_seq]
