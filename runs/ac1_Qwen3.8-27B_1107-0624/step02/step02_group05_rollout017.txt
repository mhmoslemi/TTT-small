def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]

def fft_convolve_self(f):
    """Fast f*f using one FFT (since f is real)."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(f, N)
    return np.fft.irfft(fa * fa, N)[:2*n-1]

def compute_score_fast(f):
    """Compute 2*n*max(conv)/sum^2 using FFT."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve_self(f)
    return float(2.0 * n * c.max() / (s * s))

def compute_grad_soft(f, beta):
    """
    Compute gradient of soft-max approximation of max(conv(f,f)).
    Returns (L, grad_f) where grad_f = dL/df (length n).
    """
    n = len(f)
    c = fft_convolve_self(f)  # length 2n-1
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros(2*n-1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    L = mval + beta * np.log(s)
    
    # grad[m] = 2 * sum_j w[j+m] * f[j] for j=0..n-1
    # = 2 * (convolve(reverse(f), w))[n-1 : 2n-1]
    fr = f[::-1]
    # Need convolve(fr, w) of length n + (2n-1) - 1 = 3n-2
    # We only need indices n-1 to 2n-2
    N = 1
    while N < 3*n - 2:
        N <<= 1
    frf = np.fft.rfft(fr, N)
    wf = np.fft.rfft(w, N)
    conv = np.fft.irfft(frf * wf, N)
    grad = 2.0 * conv[n-1 : 2*n-1]
    
    return L, grad

def adam_anneal_optimize(f_init, deadline, rng, base_lr=0.03, max_iters=10000):
    """
    Adam optimization with simulated annealing in log-space.
    Uses FFT-based gradient computation.
    """
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    
    z = np.log(f)
    z -= z.mean()

    v = np.zeros(n)
    adam_t = 0

    M = float(c.max())

    best_z = z.copy()

    T = 0.5 * M  # initial temperature
    T_min = 1e-6 * M

        if time.monotonic() > deadline:
            break
        
        L, grad_f = compute_grad_soft(f, beta)
        
        # Project to tangent space of simplex in log-space
        dot = float(np.dot(f, grad_f))
        gz = f * (grad_f - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * gz

        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)

            f = f_new

                v.fill(0)

                no_imp += 1

def make_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A

            return np.maximum(res.x, 0.0)

        pass
    return None

import numpy as np
from scipy.optimize import linprog
import time
import math

"""
Multi-strategy optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-accelerated Adam with simulated annealing, power-mean objectives,
LP refinement, iterated local search, and multi-n exploration.
"""


def fft_convolve_self(f):
    """Fast f*f linear convolution using one FFT."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(f, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def compute_score_fast(f):
    """Compute 2*n*max(conv)/sum^2 using FFT."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve_self(f)
    return float(2.0 * n * c.max() / (s * s))


def compute_grad_softmax(f, beta):
    """Gradient of soft-max of conv(f,f). Returns (L, grad_f)."""
    n = len(f)
    c = fft_convolve_self(f)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    L = mval + beta * np.log(s)
    fr = f[::-1]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    frf = np.fft.rfft(fr, N)
    wf = np.fft.rfft(w, N)
    conv = np.fft.irfft(frf * wf, N)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return L, grad


def compute_grad_powermean(f, p):
    """Gradient of power mean (sum c^p)^(1/p) of conv(f,f). Returns (L, grad_f)."""
    n = len(f)
    c = fft_convolve_self(f)
    cp = np.maximum(c, 0) ** p
    sum_cp = float(cp.sum())
    if sum_cp < 1e-30:
        L = 0.0
        grad = np.zeros(n)
        return L, grad
    L = sum_cp ** (1.0 / p)
    w = cp / sum_cp  # normalized weights
    fr = f[::-1]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    frf = np.fft.rfft(fr, N)
    wf = np.fft.rfft(w, N)
    conv = np.fft.irfft(frf * wf, N)
    grad = 2.0 * L ** (1.0 - p) * conv[n - 1:2 * n - 1]
    return L, grad


def normalize_nonneg(f):
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def make_linear_profile(n, slope=2.0):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_gaussian_bump(n, mu=0.5, sigma=0.2):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_dual_bump(n):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * (t / 0.15) ** 2) + np.exp(-0.5 * ((t - 1) / 0.15) ** 2) + 0.1
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_random_profile(n, rng):
    ncomp = int(rng.integers(3, 8))
    theta = rng.normal(size=2 * ncomp)
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    for k in range(ncomp):
        f = f + 0.5 * theta[2 * k] * np.cos(2 * np.pi * (k + 1) * t)
        f = f + 0.5 * theta[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * t)
    f = np.maximum(f, 1e-6)
    return f / f.sum()


def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def get_starting_points(n, rng, parent=None, count=15):
    starts = []
    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy())
        else:
            starts.append(interpolate_profile(parent, n))
    starts.append(make_linear_profile(n, 1.0))
    starts.append(make_linear_profile(n, 1.5))
    starts.append(make_linear_profile(n, 2.0))
    starts.append(make_linear_profile(n, 3.0))
    starts.append(make_cubic_profile(n, 1.5, -1.0))
    starts.append(make_cubic_profile(n, 2.0, -1.5))
    starts.append(make_cubic_profile(n, 2.5, -2.0))
    starts.append(make_dual_bump(n))
    starts.append(make_gaussian_bump(n, 0.3, 0.25))
    starts.append(make_gaussian_bump(n, 0.7, 0.25))
    for _ in range(count):
        starts.append(make_random_profile(n, rng))
    return starts


def adam_optimize(f_init, deadline, rng, base_lr=0.03, use_power_mean=False, p_power=50,
                  T_init_factor=0.3, T_decay=0.998, beta_init_factor=0.15):
    """
    Adam optimization with simulated annealing in log-space.
    Uses FFT-based gradient computation.
    """
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = fft_convolve_self(f)
    M = float(c.max())
    beta = beta_init_factor * M

    best_f = f.copy()
    best_z = z.copy()
    best_M = M

    T = T_init_factor * M
    T_min = 1e-7 * max(M, 1e-10)
    T_decay_val = T_decay

    no_imp = 0
    lr = base_lr

    for it in range(50000):
        if time.monotonic() > deadline:
            break

        if use_power_mean:
            L, grad_f = compute_grad_powermean(f, p_power)
        else:
            L, grad_f = compute_grad_soft(f, beta)

        dot = float(np.dot(f, grad_f))
        gz = f * (grad_f - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps_adam)

        if not np.all(np.isfinite(z_new)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z_new = np.clip(z_new, -50, 50)
        z_new = z_new - z_new.mean()
        e = np.exp(z_new)
        f_new = e / e.sum()

        c_new = fft_convolve_self(f_new)
        M_new = float(c_new.max())

        delta = M_new - M
        if delta < -1e-14:
            z = z_new
            f = f_new
            if M_new < best_M - 1e-14:
                best_M = M_new
                best_f = f_new.copy()
                best_z = z_new.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            if delta < 0:
                accept = True
            else:
                accept = bool(rng.random() < math.exp(-delta / max(T, 1e-30)))
            if accept:
                z = z_new
                f = f_new
                m.fill(0)
                v.fill(0)
                adam_t = 0
                no_imp += 1
            else:
                no_imp += 1

        c = fft_convolve_self(f)
        M = float(c.max())

        T = max(T_min, T * T_decay_val)

        if no_imp > 300:
            lr = max(0.005, lr * 0.92)
        elif no_imp == 0:
            lr = min(0.08, lr * 1.01)

        if not use_power_mean and adam_t % 120 == 0:
            beta = max(1e-10, beta * 0.82)

        if no_imp > max(400, 2 * n):
            z = best_z + 0.4 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = fft_convolve_self(f)
            M = float(c.max())
            beta = max(1e-10, beta_init_factor * M)
            T = T_init_factor * M
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_f, best_M


def compute_grad_soft(f, beta):
    """Alias for compute_grad_softmax."""
    return compute_grad_softmax(f, beta)


def make_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp_for_g(f, M):
    n = len(f)
    c_vec = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, max_iters=300):
    """LP-based iterative optimization with FFT."""
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_score_fast(f_init)

    n = len(f)
    M = float(fft_convolve_self(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = solve_lp_for_g(f, M)
        if g is None:
            f = normalize_nonneg(f + 0.1 * rng.normal(size=n))
            if f is None:
                f = best_f.copy()
            M = float(fft_convolve_self(f).max())
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum

        best_t = 0.0
        best_f_try = f
        best_M_try = M
        for t in [0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
            f_try = (1.0 - t) * f + t * g
            f_try = normalize_nonneg(f_try)
            if f_try is None:
                continue
            M_try = float(fft_convolve_self(f_try).max())
            if M_try < best_M_try - 1e-12:
                best_M_try = M_try
                best_f_try = f_try
                best_t = t

        if best_t > 0:
            f = best_f_try
            M = best_M_try
            if M < best_M - 1e-12:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 15:
            pert = normalize_nonneg(best_f + 0.5 * rng.normal(size=n))
            if pert is not None:
                f = normalize_nonneg(0.75 * best_f + 0.25 * pert)
            if f is None:
                f = best_f.copy()
            M = float(fft_convolve_self(f).max())
            no_imp = 0

    return best_f, best_M


def iterated_local_search(f_init, deadline, rng, num_restarts=20):
    """
    ILS: repeatedly perturb the best solution and re-optimize with short Adam runs.
    """
    best_f = f_init.copy()
    best_M = float(fft_convolve_self(best_f).max())
    n = len(best_f)

    for r in range(num_restarts):
        if time.monotonic() > deadline:
            break

        # Perturb
        scale = 0.3 * (1.0 + 0.5 * rng.random())
        f_pert = normalize_nonneg(best_f + scale * rng.normal(size=n))
        if f_pert is None:
            f_pert = best_f.copy()

        # Short Adam optimization
        sub_deadline = min(deadline, time.monotonic() + max(10, 600.0 / num_restarts))
        f_opt, M_opt = adam_optimize(f_pert, sub_deadline, rng, base_lr=0.04,
                                     T_init_factor=0.2, T_decay=0.995)
        if M_opt < best_M - 1e-12:
            best_M = M_opt
            best_f = f_opt.copy()

    return best_f, best_M
