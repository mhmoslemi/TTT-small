def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of two >= n, at least 2."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def logsumexp_smooth_max(c, beta):
    """Compute smooth max and weights."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = float(np.count_nonzero(mask))
        w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -50, 50))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * np.log(s)
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n-1:2*n-1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(z) / len(z)
    return e / s

    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = full_convolve_fft(p0)
    maxc = float(c.max())
    if beta_init is None:
        beta = 0.15 * maxc
    else:
        beta = beta_init
    beta = max(beta_min, beta)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    patience = max(500, 3 * n)

    while time.monotonic() < deadline:
        L, w = logsumexp_smooth_max(c, beta)
        gp = 2.0 * corr_weights(w, p0)
        dot = float(np.dot(p0, gp))
        gz = p0 * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1**t_adam)
        vh = v / (1 - b2**t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))

        p0 = softmax_from_z(z)
        c = full_convolve_fft(p0)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p0.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Anneal beta
        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)

        if t_adam % 2000 == 0:
            lr = lr * 0.95

            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            no_imp = 0
            lr = 0.05

        if verbose and t_adam % 500 == 0:
            print(f"  Adam t={t_adam}, score={best_score:.10f}, beta={beta:.2e}")

    return best_p, best_score

        profiles.append(("cos_%d" % k, a / a.sum()))

    a = 1.0 - 2.0 * np.abs(t - 0.5)

    # Exponential
    for base in [2.0, 4.0, 8.0]:
        a = base ** t
        profiles.append(("exp_%.0f" % base, a / a.sum()))

    # Gaussian bump
    for sigma in [0.2, 0.3, 0.4]:
        a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 0.01
        profiles.append(("gauss_%.1f" % sigma, a / a.sum()))

    # Power law
    for p in [0.5, 1.0, 2.0]:
        a = (t + 0.1) ** p

        a = np.maximum(a, 1e-10)
        profiles.append(("rand_%d" % i, a / a.sum()))

def interpolate_profile(p0, n_new):
    """Cubic interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.arange(n0)
    x1 = np.linspace(0, n0 - 1, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-15)
    return p1 / p1.sum()

"""
Multi-phase gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Adam in logit space with log-sum-exp smoothing, multi-scale search (256->512->1024->2048),
LP-assisted refinement, simulated annealing for escape, and adaptive hyperparameter scheduling.

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of two >= n, at least 2."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def logsumexp_smooth_max(c, beta):
    """Compute smooth max and weights."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = float(np.count_nonzero(mask))
        w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -50.0, 50.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * np.log(s)
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(z) / len(z)
    return e / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def adam_logit_opt(p0, deadline, rng, lr=0.05, beta_init=None, beta_decay=0.95,
                   beta_min=1e-14, patience_mult=3.0, verbose=False, use_lp=False, lp_every=500):
    """Adam optimization in logit space with log-sum-exp smoothing, optional LP refinement."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = full_convolve_fft(p0)
    maxc = float(c.max())
    if beta_init is None:
        beta = 0.12 * maxc
    else:
        beta = beta_init
    beta = max(beta_min, beta)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    patience = max(400, int(patience_mult * n))
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = logsumexp_smooth_max(c, beta)
        gp = 2.0 * corr_weights(w, p0)
        dot = float(np.dot(p0, gp))
        gz = p0 * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))

        p0 = softmax_from_z(z)
        c = full_convolve_fft(p0)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p0.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Anneal beta
        if t_adam % 40 == 0:
            beta = max(beta_min, beta * beta_decay)

        # Decay learning rate
        if t_adam % 1500 == 0 and t_adam > 0:
            lr = max(0.005, lr * 0.93)

        # LP refinement step
        if use_lp and t_adam > lp_every and no_imp > patience // 2 and time.monotonic() < deadline - 10:
            M_curr = float(np.convolve(p0, p0).max())
            g = lp_step(p0, M_curr)
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                for t_ratio in [0.3, 0.2, 0.1, 0.05, 0.02]:
                    p_new = (1 - t_ratio) * p0 + t_ratio * g
                    p_new = np.maximum(p_new, 1e-15)
                    s_new = p_new.sum()
                    if s_new < 1e-10:
                        continue
                    p_new = p_new / s_new
                    c_new = full_convolve_fft(p_new)
                    M_new = float(c_new.max())
                    M_orig = float(np.convolve(p0, p0).max())
                    if M_new <= M_orig + 1e-12:
                        p0 = p_new
                        z = np.log(np.maximum(p0, 1e-15))
                        z -= float(np.mean(z))
                        c = c_new
                        maxc = M_new
                        score = float(2.0 * n * maxc)
                        if score < best_score - 1e-14:
                            best_score = score
                            best_p = p0.copy()
                            best_z = z.copy()
                            no_imp = 0
                        break
            lr = max(0.01, lr * 0.9)

        # Restart if stuck
        if no_imp > patience:
            mix = np.random.choice([0.5, 0.7, 0.8, 0.9], p=[0.2, 0.3, 0.3, 0.2])
            noise_scale = rng.uniform(0.2, 0.6)
            z = mix * best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p0 = softmax_from_z(z)
            c = full_convolve_fft(p0)
            beta = max(beta_min, 0.12 * float(c.max()))
            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            no_imp = 0
            lr = base_lr
            base_lr = max(0.02, lr * 0.95)

        if verbose and t_adam % 1000 == 0:
            print(f"    Adam t={t_adam}, best={best_score:.10f}, beta={beta:.2e}")

    return best_p, best_score


def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append(("ramp_%.1f" % eta, a / a.sum()))

    # Strongly asymmetric ramps
    for base in [2.0, 4.0, 8.0, 16.0]:
        a = base ** (t - 0.5)
        profiles.append(("asym_%.0f" % base, a / a.sum()))

    # Cosine windows
    for k in [1, 2, 3, 4]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * k * t)
        a = np.maximum(a, 1e-10)
        profiles.append(("cos_%d" % k, a / a.sum()))

    # Triangle
    a = 1.0 - 2.0 * np.abs(t - 0.5)
    a = np.maximum(a, 1e-10)
    profiles.append(("tri", a / a.sum()))

    # Skewed triangle
    for skew in [0.3, 0.4, 0.6]:
        peak = 1.0 - skew
        for i in range(n):
            if i / (n - 1) <= peak:
                a[i] = (i / (n - 1)) / peak
            else:
                a[i] = (1 - i / (n - 1)) / (1 - peak)
        a = np.maximum(a, 1e-10)
        profiles.append(("skewtri_%.1f" % skew, a / a.sum()))

    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = (t + 0.05) ** p
        profiles.append(("pow_%.1f" % p, a / a.sum()))

    # Gaussian bumps
    for sigma in [0.15, 0.25, 0.35]:
        a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 0.005
        profiles.append(("gauss_%.2f" % sigma, a / a.sum()))

    # Random smooth (Fourier)
    for i in range(8):
        coeffs = rng.uniform(-0.4, 0.4, 7)
        phases = rng.uniform(0, 2 * np.pi, 7)
        a = np.ones(n)
        for j in range(1, 7):
            a = a + coeffs[j] * np.cos(2 * np.pi * j * t + phases[j])
        a = np.maximum(a, 1e-10)
        profiles.append(("rand_%d" % i, a / a.sum()))

    return profiles


def interpolate_profile(p0, n_new):
    """Interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.arange(n0, dtype=np.float64)
    x1 = np.linspace(0, n0 - 1, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-15)
    return p1 / p1.sum()


def simulated_annealing(p0, deadline, rng, n_iter=50000):
    """Brief SA phase to escape local minima."""
    p = np.asarray(p0, dtype=np.float64).ravel()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    score = float(2.0 * n * float(full_convolve_fft(p).max()))
    best_p = p.copy()
    best_score = score

    T = 0.0002
    T_min = 0.000005
    cool_rate = (T_min / T) ** (1.0 / max(1, n_iter))

    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        # Perturb: randomly swap or nudge two elements
        idx = rng.integers(0, n)
        if rng.random() < 0.3 and n > 10:
            # Swap-like perturbation
            j = rng.integers(0, n)
            if j != idx:
                delta = rng.normal(0, 0.01 * p.mean())
                p[idx] += delta
                p[j] -= delta
        else:
            # Nudge
            p[idx] += rng.normal(0, 0.005 * p.mean())

        p = np.maximum(p, 1e-15)
        p = p / p.sum()

        c = full_convolve_fft(p)
        new_score = float(2.0 * n * float(c.max()))

        delta = new_score - score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            score = new_score
            if score < best_score:
                best_score = score
                best_p = p.copy()

        T *= cool_rate

    return best_p, best_score

                                        patience_mult=3.0, use_lp=True, lp_every=300, verbose=True)
        sc_final = compute_score(result)
        update(result, sc_final)
        print(f"  Main: {sc_final:.10f}")

        # Try variants
        for i in range(3):
            if time.monotonic() > phase2_end - 10:
                break
            p2v = interpolate_profile(best_seq, n2)
            p2v = p2v + 0.003 * rng.normal(size=n2)
            p2v = np.maximum(p2v, 1e-10)
            p2v = p2v / p2v.sum()
            per_dead = min(phase2_end, time.monotonic() + 60.0)
            result, sc_res = adam_logit_opt(p2v, per_dead, rng, lr=0.04, beta_decay=0.95, use_lp=True, lp_every=200)
            sc_final = compute_score(result)
            update(result, sc_final)
            print(f"  Variant {i}: {sc_final:.10f}")

    print(f"Phase 2 done: best={best_score:.10f}")

    # ============ Phase 3: Extend to n=1024 ============
    n3 = 1024
    phase3_end = t_start + 1250.0

    if time.monotonic() < phase3_end - 20 and best_seq is not None:
        print(f"\n=== Phase 3: Extend to n={n3} (until t={phase3_end - t_start:.0f}s) ===")
        p3 = interpolate_profile(best_seq, n3)
        sc3 = compute_score(p3)
        update(p3, sc3)
        print(f"  Interpolated: score={sc3:.10f}")

        per_dead = min(phase3_end, time.monotonic() + 500.0)
        result, sc_res = adam_logit_opt(p3, per_dead, rng, lr=0.04, beta_decay=0.955,
                                        patience_mult=4.0, use_lp=True, lp_every=500, verbose=True)
        sc_final = compute_score(result)
        update(result, sc_final)
        print(f"  n={n3} main: {sc_final:.10f}")

        # Variants at n=1024
        for i in range(3):
            if time.monotonic() > phase3_end - 15:
                break
            p3v = interpolate_profile(best_seq, n3)
            p3v = p3v + 0.003 * rng.normal(size=n3)
            p3v = np.maximum(p3v, 1e-10)
            p3v = p3v / p3v.sum()
            per_dead = min(phase3_end, time.monotonic() + 120.0)
            result, sc_res = adam_logit_opt(p3v, per_dead, rng, lr=0.035, beta_decay=0.955, use_lp=True, lp_every=300)
            sc_final = compute_score(result)
            update(result, sc_final)
            print(f"  n={n3} variant {i}: {sc_final:.10f}")

    print(f"Phase 3 done: best={best_score:.10f}")

    # ============ Phase 4: Extend to n=2048 if beneficial ============
    n4 = 2048
    phase4_end = t_start + 1650.0

    if time.monotonic() < phase4_end - 30 and best_seq is not None and best_score > 1.503:
        print(f"\n=== Phase 4: Extend to n={n4} ===")
        p4 = interpolate_profile(best_seq, n4)
        sc4 = compute_score(p4)
        update(p4, sc4)
        print(f"  Interpolated: score={sc4:.10f}")

        per_dead = min(phase4_end, time.monotonic() + 350.0)
        result, sc_res = adam_logit_opt(p4, per_dead, rng, lr=0.03, beta_decay=0.96,
                                        patience_mult=5.0, use_lp=False, verbose=True)
        sc_final = compute_score(result)
        update(result, sc_final)
        print(f"  n={n4}: {sc_final:.10f}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # ============ Phase 5: SA escape + final polish ============
    if time.monotonic() < deadline - 20 and best_seq is not None:
        n5 = best_seq.size
        print(f"\n=== Phase 5: SA + polish at n={n5} ===")

        # Brief SA
        sa_dead = min(deadline - 60, time.monotonic() + 80.0)
        p_sa, sc_sa = simulated_annealing(best_seq, sa_dead, rng, n_iter=30000)
        sc_sa_final = compute_score(p_sa)
        update(p_sa, sc_sa_final)
        print(f"  SA: {sc_sa_final:.10f}")

        # Final fine polish
        if time.monotonic() < deadline - 15:
            maxc = float(full_convolve_fft(best_seq).max())
            per_dead = min(deadline, time.monotonic() + 50.0)
            result, sc_res = adam_logit_opt(best_seq, per_dead, rng, lr=0.015,
                                            beta_decay=0.97, beta_init=0.03 * maxc,
                                            patience_mult=2.0)
            sc_final = compute_score(result)
            update(result, sc_final)
            print(f"  Final polish: {sc_final:.10f}")

    # ============ Final validation ============
    if best_seq is None:
        best_seq = np.ones(512) / 512.0

    final_seq = np.maximum(best_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
