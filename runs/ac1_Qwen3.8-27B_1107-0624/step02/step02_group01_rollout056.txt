    c0 = fft_convolve(a)
    max_c0 = float(c0.max())
    beta = beta_init_frac * max_c0 if beta_init_frac else 0.25 * max_c0

    z = np.log(a)
    z -= z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a.copy()
    best_score = compute_score(a)
    no_imp = 0
    patience = 8000
    stall_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, c, W, w = compute_gradient(a, n, beta)

        if not np.all(np.isfinite(grad)):
            z = np.log(np.maximum(a, 1e-15))
            z -= z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        # Anneal beta periodically
        if t_adam % 200 == 0:
            beta *= beta_decay
            min_beta = 1e-8 * float(c.max())
            beta = max(beta, min_beta)

        sc = compute_score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
            stall_count = 0
        else:
            no_imp += 1

        # Restart strategies
        if no_imp > patience:
            stall_count += 1
            strategy = rng.integers(0, 6)
            if strategy == 0:
                # Gaussian perturbation in log-space
                z_pert = np.log(np.maximum(best_a, 1e-15))
                z_pert += 0.1 * rng.normal(size=n)
            elif strategy == 1:
                # Blend with smooth random
                a_r = make_initial(n, rng, 'random_smooth_large')
                a_pert = 0.6 * best_a + 0.4 * a_r
                z_pert = np.log(np.maximum(a_pert, 1e-15))
            elif strategy == 2:
                # Blend with uniform
                a_pert = 0.85 * best_a + 0.15 / n
                z_pert = np.log(np.maximum(a_pert, 1e-15))
            elif strategy == 3:
                # Large log-space perturbation
                z_pert = np.log(np.maximum(best_a, 1e-15))
                z_pert += 0.3 * rng.normal(size=n)
            elif strategy == 4:
                # Partial randomization
                n_rand = n // 4
                idx = rng.choice(n, n_rand, replace=False)
                a_pert = best_a.copy()
                a_r = make_initial(n, rng, 'random_smooth')
                a_pert[idx] = a_r[idx]
                z_pert = np.log(np.maximum(a_pert, 1e-15))
            else:
                # Fresh smooth random
                a_pert = make_initial(n, rng, 'random_smooth_large')
                z_pert = np.log(a_pert)

            z_pert = np.clip(z_pert, -60.0, 60.0)
            z_pert -= z_pert.mean()
            a_new = np.exp(z_pert - z_pert.max())
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()

            sc_new = compute_score(a_new)
            if sc_new < best_score:
                best_a = a_new.copy()
                best_score = sc_new

            a = a_new
            z = z_pert
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            beta = 0.25 * float(fft_convolve(a).max())
            no_imp = 0
            patience = max(3000, patience // 2)

        if verbose and it % 20000 == 0:
            print(f"    GD iter {it}: best={best_score:.10f}, beta={beta:.8f}, NO={no_imp}")

    return best_a, best_score


def fine_polish(a_init, deadline, rng, lr=0.05, verbose=False):
    """Final polishing with very small learning rate and small beta."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size

    c0 = fft_convolve(a)
    beta = 0.05 * float(c0.max())

    z = np.log(a)
    z -= z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a.copy()
    best_score = compute_score(a)
    no_imp = 0

    for it in range(50000):
        if time.monotonic() > deadline:
            break

        grad, c, W, w = compute_gradient(a, n, beta)
        if not np.all(np.isfinite(grad)):
            break

        t_adam += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        if t_adam % 500 == 0:
            beta *= 0.95
            beta = max(beta, 1e-10 * float(c.max()))

        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 10000:
            break

        if verbose and it % 10000 == 0:
            print(f"    Polish iter {it}: best={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage search:
    1. Try parent sequence if available.
    2. Multi-start gradient descent at n=512 from diverse profiles.
    3. Best candidate gets extended to n=1024.
    4. Final polish on the absolute best.
    """
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_score = np.inf

    def try_update(a, sc):
        nonlocal best_a, best_score
        if a is not None and np.isfinite(sc) and sc < best_score:
            best_a = a.copy()
            best_score = float(sc)

    # Try parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent: n={parent.size}, score={sc:.10f}")

    # Phase 1: Multi-start GD at n=512
    n1 = 512
    phase1_end = min(deadline, t_start + 700.0)

    starts = []
    if parent is not None:
        p_int = interpolate_seq(parent, n1)
        if p_int is not None:
            starts.append(p_int)
            starts.append(p_int[::-1].copy())  # reversed

    # Diverse deterministic initializations
    modes = ['u_shape', 'u_strong', 'u_quad', 'center_peak', 'bimodal',
             'cos_taper', 'sin_taper', 'polynomial', 'polynomial2',
             'linear', 'random_smooth', 'random_smooth_large']
    for mode in modes:
        starts.append(make_initial(n1, rng, mode))

    # Additional random starts
    for _ in range(6):
        starts.append(make_initial(n1, rng, 'random_smooth_large'))

    # Shuffle to spread risk
    perm = rng.permutation(len(starts))
    starts = [starts[i] for i in perm]

    candidates = []
    per_start_time = max(5.0, (phase1_end - t_start) / len(starts))

    for i, a0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        a0 = np.maximum(a0, 1e-12)
        a0 = a0 / a0.sum()
        sc0 = compute_score(a0)
        if not np.isfinite(sc0):
            continue

        per = min(phase1_end, time.monotonic() + per_start_time)
        a_best, sc_best = gradient_descent(a0, per, rng, lr=0.3,
                                           beta_init_frac=0.25,
                                           beta_decay=0.9997,
                                           max_iter=100000)
        try_update(a_best, sc_best)
        candidates.append((sc_best, a_best))
        if i % 4 == 0:
            print(f"  Phase1 start {i+1}/{len(starts)}: best={best_score:.10f}")

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1 (n={n1}): best={best_score:.10f}")

    # Phase 2: Extended GD on top candidates
    phase2_end = min(deadline, t_start + 1100.0)
    top_k = min(4, len(candidates))
    if top_k > 0 and time.monotonic() < phase2_end:
        per_cand = max(20.0, (phase2_end - t_start) / top_k)
        for i in range(top_k):
            if time.monotonic() > phase2_end:
                break
            sc_i, a_i = candidates[i]
            if sc_i >= best_score:
                # Use the current best instead
                a_i = best_a.copy()
            per = min(phase2_end, time.monotonic() + per_cand)
            a_best, sc_best = gradient_descent(a_i, per, rng, lr=0.2,
                                               beta_init_frac=0.15,
                                               beta_decay=0.9998,
                                               max_iter=200000)
            try_update(a_best, sc_best)
            print(f"  Phase2 cand {i}: score={sc_best:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Fine polish on best
    phase3_end = min(deadline, t_start + 1350.0)
    if best_a is not None and time.monotonic() < phase3_end:
        per = min(phase3_end, time.monotonic() + 150.0)
        a_best, sc_best = fine_polish(best_a, per, rng, lr=0.05)
        try_update(a_best, sc_best)
        print(f"After Phase 3 polish (n={best_a.size}): best={best_score:.10f}")

    # Phase 4: Try n=1024
    n2 = 1024
    phase4_end = min(deadline, t_start + 1600.0)
    if best_a is not None and time.monotonic() < phase4_end - 60:
        a_ext = interpolate_seq(best_a, n2)
        if a_ext is not None:
            per = min(phase4_end, time.monotonic() + 120.0)
            a_best, sc_best = gradient_descent(a_ext, per, rng, lr=0.25,
                                               beta_init_frac=0.2,
                                               beta_decay=0.9997,
                                               max_iter=100000)
            try_update(a_best, sc_best)
            print(f"Phase 4 (n={n2}): score={sc_best:.10f}")

            if time.monotonic() < phase4_end - 20:
                per2 = min(phase4_end, time.monotonic() + 60.0)
                a_best2, sc_best2 = fine_polish(a_best, per2, rng, lr=0.03)
                try_update(a_best2, sc_best2)
                print(f"Phase 4 polish (n={n2}): score={sc_best2:.10f}")

    # Phase 5: Try n=2048 if time permits
    n3 = 2048
    if best_a is not None and time.monotonic() < deadline - 80:
        a_ext = interpolate_seq(best_a, n3)
        if a_ext is not None:
            per = min(deadline - 30, time.monotonic() + 80.0)
            a_best, sc_best = gradient_descent(a_ext, per, rng, lr=0.2,
                                               beta_init_frac=0.2,
                                               beta_decay=0.9998,
                                               max_iter=50000)
            try_update(a_best, sc_best)
            print(f"Phase 5 (n={n3}): score={sc_best:.10f}")

    # Phase 6: Final refinement on absolute best
    if best_a is not None and time.monotonic() < deadline - 10:
        n_best = best_a.size
        per = min(deadline - 5, time.monotonic() + 60.0)
        a_best, sc_best = gradient_descent(best_a, per, rng, lr=0.1,
                                           beta_init_frac=0.1,
                                           beta_decay=0.9999,
                                           max_iter=50000)
        try_update(a_best, sc_best)
        print(f"After Phase 6: best={best_score:.10f}, n={best_a.size}")

    # Final validation
    if best_a is not None:
        final = np.maximum(best_a, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final: score={sc_final:.10f}, n={final.size}")

        # Verify with evaluator
        if final.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                print(f"Evaluated score: {sc_eval:.10f}")
                if np.isfinite(sc_eval) and sc_eval < best_score:
                    best_score = sc_eval
            except Exception as e:
                print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_initial(512, rng, 'u_shape')
    print(f"Fallback: score={compute_score(fb):.10f}")
    return [float(x) for x in fb]
