```python
from scipy.optimize import linprog
import numpy as np

def lp_find_direction(f, M):
    """Find g that maximizes sum(g) subject to f*g <= M and g >= 0."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: for each k from 0 to 2n-2
    # sum_{i=max(0,k-(n-1))}^{min(n-1,k)} f[i] * g[k-i] <= M
    A_rows = []
    b_vals = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        for i in range(max(0, k-n+1), min(n, k+1)):
            row[k-i] = f[i]
        A_rows.append(row)
        b_vals.append(M)
    
    A = np.array(A_rows)
    b = np.array(b_vals)
    
    # Solve: minimize c^T g subject to A g <= b, g >= 0
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```

```python
from scipy.optimize import linprog
import time

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s

def score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2 * n * c.max() / (s * s)

def lp_direction(f, M, n):
    """Solve LP to find g maximizing sum(g) s.t. f*g <= M, g >= 0."""
    c_obj = -np.ones(n)
    A_rows = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            row[k - i] = f[i]
        A_rows.append(row)
    A = np.array(A_rows)
    b = np.full(2*n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def lp_optimize(f0, deadline, max_iter=1000):
    """LP-based optimization."""
    f = normalize(f0.copy())
    n = len(f)
    best_f = f.copy()
    best_score = score(f)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        M = np.convolve(f, f).max()
        g = lp_direction(f, M, n)
        if g is None:
            break
        g = normalize(g)
        if g is None:
            break
        
        # Line search
        best_t = None
        best_sc = best_score
        for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            f_new = (1 - t) * f + t * g
            f_new = normalize(f_new)
            sc = score(f_new)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_t = t
                best_f = f_new
        
        if best_t is None:
            # Try smaller steps
            for t in [0.005, 0.001]:
                f_new = (1 - t) * f + t * g
                f_new = normalize(f_new)
                sc = score(f_new)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_t = t
                    best_f = f_new
        
        if best_t is None:
            break  # Stuck
        
        f = best_f
        if best_score > best_sc:
            best_score = best_sc
    
    return best_f, best_score
```

```python
def build_A_matrix(n, f):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    return A


def lp_direction(f, n, M):

def power_sum_optimize(f0, deadline, verbose=True, beta_start=50.0, beta_end=200.0, lr=0.05):
    f = normalize_nonneg(np.asarray(f0, dtype=np.float64).copy())
    if f is None:
        return None, np.inf
    n = len(f)
    best_f = f.copy()
    best_score = compute_score_normalized(best_f)
    it = 0
    total_iters = 20000
    stuck = 0
    rng = np.random.default_rng(77)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        frac = it / total_iters
        beta = beta_start + (beta_end - beta_start) * frac

        c = np.convolve(f, f)
        c_max = float(c.max())
        x = (c - c_max) / beta
        ex = np.exp(np.clip(x, -50, 50))
        w = ex / ex.sum()

        conv_w = np.convolve(w, f)[n - 1:2 * n - 1]
        grad = 2.0 * conv_w / beta
        dot = float(np.dot(f, grad))
        g = grad - dot

        if not np.all(np.isfinite(g)):
            continue

        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        mh = m / (1 - b1 ** (it + 1))
        vh = v / (1 - b2 ** (it + 1))
        step = lr * mh / (np.sqrt(vh) + eps)

        f_new = normalize_nonneg(f - step)
        if f_new is None:
            continue

        sc = compute_score_normalized(f_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = f_new.copy()
            stuck = 0
        else:
            stuck += 1

        f = f_new

        if stuck > 5000:
            f = normalize_nonneg(best_f + 0.02 * np.maximum(rng.dirichlet(np.ones(n) * 60.0), 0.0))
            m.fill(0.0)
            v.fill(0.0)
            stuck = 0

        if it % 5000 == 0 and verbose:
            print(f"    PowerSum iter {it}, beta={beta:.1f}, score={best_score:.10f}")

    if verbose:
        print(f"  PowerSum done: {it} iters, best={best_score:.10f}")
    return best_f, best_score


def interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def coord_descent_polish(p, deadline, rng, max_sweeps=200):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = normalize_nonneg(p)
    if p is None:
        return None, np.inf
    n = p.size
    best_p = p.copy()
    best_score = compute_score_normalized(p)
    step_frac = 0.1
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            for factor in [1.1, 0.9, 1.2, 0.8]:
                new_val = orig * factor
                if new_val < 1e-15:
                    continue
                p[i] = new_val
                sc = compute_score_normalized(p)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_p = p.copy()
                    improved = True
                    no_imp = 0
                    break
            p[i] = p[i] if improved and p[i] != orig else orig
            if p[i] != orig:
                p = normalize_nonneg(p)
            if not improved:
                no_imp += 1

        if improved:
            step_frac = max(0.03, step_frac * 0.95)
        if no_imp > n:
            break

    return best_p, best_score

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        sc = compute_score_general(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            if label:
                print(f"  NEW BEST: {best_score:.10f} n={p.size} [{label}]")

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(parent >= 0):
            parent = normalize_nonneg(parent)
    except Exception:
        pass

    print("Phase 0: Initialization")
    if parent is not None:
        try_update(parent, "parent")
        for tn in [256, 512]:
            if parent.size != tn:
                try_update(interp_profile(parent, tn), f"parent_interp_{tn}")

    for n_init in [256, 512]:
        for p0 in make_deterministic_profiles(n_init)[:6]:
            try_update(p0, f"det_n{n_init}")

    print(f"  After init: best={best_score:.10f}")

    # Phase 1: LP-based optimization at n=512
    print("Phase 1: LP optimization at n=512")
    n_lp = 512
    phase1_deadline = min(deadline, t_start + 700.0)

    if parent is not None and parent.size == n_lp:
        p_start = parent.copy()
    elif parent is not None:
        p_start = interp_profile(parent, n_lp)
    else:
        p_start = make_deterministic_profiles(n_lp)[3].copy()

    p_best, sc = lp_optimize(p_start, phase1_deadline, verbose=True)
    try_update(p_best, f"LP_n{n_lp}")

    # Phase 2: LP at n=256
    print("Phase 2: LP optimization at n=256")
    n_lp2 = 256
    phase2_deadline = min(deadline, t_start + 1100.0)

    p_start256 = interp_profile(p_best if p_best is not None else p_start, n_lp2)
    p_best256, sc256 = lp_optimize(p_start256, phase2_deadline, verbose=True)
    try_update(p_best256, f"LP_n{n_lp2}")

    # Phase 3: Power-sum gradient at n=512
    print("Phase 3: Power-sum gradient at n=512")
    phase3_deadline = min(deadline, t_start + 1400.0)
    if p_best is not None:
        p_start3 = p_best.copy()
    else:
        p_start3 = interp_profile(p_start256, n_lp) if p_start256 is not None else make_deterministic_profiles(n_lp)[2].copy()
    p_best3, sc3 = power_sum_optimize(p_start3, phase3_deadline, verbose=True)
    try_update(p_best3, f"PS_n{n_lp}")

    # Phase 4: Subgradient at n=256
    print("Phase 4: Subgradient at n=256")
    phase4_deadline = min(deadline, t_start + 1600.0)
    if p_best256 is not None:
        p_start4 = p_best256.copy()
    else:
        p_start4 = interp_profile(p_start3, n_lp2)
    p_best4, sc4 = subgradient_optimize(p_start4, phase4_deadline, verbose=True)
    try_update(p_best4, f"SG_n{n_lp2}")

    # Phase 5: Try a few more LP runs from different starting points
    print("Phase 5: Additional LP runs")
    phase5_deadline = min(deadline, t_start + 1720.0)
    for i, prof in enumerate(make_deterministic_profiles(n_lp)[:4]):
        if time.monotonic() > phase5_deadline:
            break
        per = min(phase5_deadline, time.monotonic() + 120.0)
        p_b, s_b = lp_optimize(prof, per, verbose=False)
        try_update(p_b, f"LP_det_{i}")

    # Phase 6: Coordinate descent polish
    print("Phase 6: Coordinate descent polish")
    phase6_deadline = min(deadline, t_start + 1785.0)
    if best_seq is not None and best_seq.size <= 512:
        per = min(phase6_deadline, time.monotonic() + 50.0)
        p_pol, sc_pol = coord_descent_polish(best_seq, per, rng, max_sweeps=50)
        try_update(p_pol, "polish")

    # Final: verify with evaluate_sequence
    print("Phase 7: Final verification")
    final_seq = best_seq
    if final_seq is not None:
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"  Verified score: {sc_eval:.10f}")
            else:
                print(f"  WARNING: evaluate returned non-finite")
        except Exception as e:
            print(f"  Eval error: {e}")

        try_score = compute_score_general(final_seq)
        print(f"  Internal score: {try_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = make_deterministic_profiles(256)[2]
    print(f"  FALLBACK: score={compute_score_general(fallback):.10f}")
    return [float(x) for x in fallback]
```
