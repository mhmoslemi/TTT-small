- In the time domain: `a_new = proj_{nonneg, sum=1}(a - mu * grad)`
- In the frequency domain: modify the phase to reduce the peak

Let me try a **frequency domain approach**:

1. Compute `A = FFT(a)`
2. The power spectrum is `|A|^2`
3. The autocorrelation is `IFFT(|A|^2)`
4. To reduce the peak of the autocorrelation, we can:
   a. Find the peak position m
   b. Compute the "error" at that position
   c. Adjust the phase of A to reduce this error
   d. Take IFFT to get new a
   e. Project onto {a >= 0, sum(a) = 1}

This is similar to the "projection onto convex sets" (POCS) approach.

Actually, I think the most promising approach is a **hybrid of time-domain gradient descent and frequency-domain spectral shaping**.

1. A clean, efficient implementation
2. Multiple diverse initializations
3. Soft-max gradient with Adam
4. Aggressive annealing
5. Multi-length search
6. Coordinate descent polish

"""
Improved optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Spectral initialization (flat power spectrum target)
- Two-phase annealing (coarse then fine)
- Adaptive beta based on autocorrelation spread
- Nesterov momentum in logit space
- Frequency-domain spectral shaping
- Multi-length search including n=384, 640
- More diverse restarts
"""
import numpy as np
import time

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2*n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    if c.min() < 0:
        c = np.maximum(c, 0.0)
    return c

def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv(a,a))"""
    n = a.size
    c = full_convolve_fft(a)
    return 2.0 * n * float(c.max())

def true_score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * max(c) / (s*s))

def soft_max_grad(c, beta):
    """Compute soft-max value and gradient weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 50))
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w

def corr_with_weights(w, p):
    """Compute correlation of weights w with reversed p, giving gradient in p-space."""
    n = len(p)
    N = next_pow2(2*n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n-1:2*n-1]

def softmax_z(z):
    z = z - float(z.max())
    e = np.exp(np.clip(z, -50, 50))
    p = e + 1e-15
    s = p.sum()
    return p / s if s > 0 else np.ones_like(e) / e.size

def make_linear(n, slope):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def make_exp(n, rate):
    t = np.linspace(0, 1, n)
    a = np.exp(rate * t)
    return a / a.sum()

def make_poly(n, coeffs):
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    for k, c in enumerate(coeffs, 1):
        a = a + c * t**k
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def make_rational(n, p, q):
    t = np.linspace(0, 1, n)
    a = (t + 0.01)**p / ((t + 0.5)**q + 0.1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def make_spectral_init(n, rng, target_flatness=0.5):
    """
    Initialize by designing the power spectrum.
    Target: partially flat power spectrum (interpolate between flat and concentrated).
    """
    N = next_pow2(2*n - 1)
    # Design power spectrum
    # Flat spectrum gives minimum possible peak autocorrelation
    # But IFFT of flat spectrum is a delta, which is not non-negative
    # So we interpolate between flat and a "feasible" spectrum
    
    # Start with a random phase
    phases = rng.uniform(0, 2*np.pi, N//2 + 1)
    # Power: interpolate between flat and concentrated
    freqs = np.arange(N//2 + 1)
    freq_norm = freqs / (N//2)
    # Concentrated: more power at low frequencies
    concentrated_power = np.exp(-3 * freq_norm**2)
    flat_power = np.ones_like(freq_norm)
    power = (1 - target_flatness) * concentrated_power + target_flatness * flat_power
    
    # Construct complex spectrum
    A = np.zeros(N//2 + 1, dtype=complex)
    A[0] = np.sqrt(power[0])  # DC component (real)
    for k in range(1, N//2 + 1):
        A[k] = np.sqrt(power[k]) * np.exp(1j * phases[k])
    
    # Hermitian symmetry
    A_full = np.zeros(N, dtype=complex)
    A_full[:N//2+1] = A
    A_full[N//2+1:] = np.conj(A[1:N//2][::-1]) if N > 2 else 0
    
    a_full = np.fft.irfft(A_full)  # Wait, this is wrong for full spectrum
    # Actually, let me use rfft convention
    a_full = np.fft.ifft(A_full).real
    a = a_full[:n]
    
    # Project to non-negative
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return make_linear(n, 1.0)
    return a / s

def optimize_soft_max_adam(p0, deadline, rng, phase="coarse"):
    """
    Two-phase soft-max Adam optimization.
    Phase 'coarse': high beta, large lr, fast annealing
    Phase 'fine': low beta, small lr, slow annealing
    """
    n = p0.size
    p = np.maximum(p0, 1e-15).copy()
    p = p / p.sum()
    
    z = np.log(p)
    z -= z.mean()
    
    if phase == "coarse":
        lr = 0.15
        beta_frac = 0.5
        beta_decay_interval = 40
        beta_decay_rate = 0.6
        restart_threshold = 150
        base_lr = lr
    else:
        lr = 0.05
        beta_frac = 0.15
        beta_decay_interval = 60
        beta_decay_rate = 0.75
        restart_threshold = 400
        base_lr = lr
    
    # Nesterov momentum
    m = np.zeros(n)
    v_z = z.copy()  # velocity (previous z)
    
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-10, beta_frac * maxc)
    
    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        L, w = soft_max_grad(c, beta)
        gp = 2.0 * corr_with_weights(w, p)
        
        # Center the gradient (remove mean to preserve sum)
        gz = p * (gp - float(np.dot(p, gp)))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v_z = z.copy()
            continue
        
        # Nesterov: look-ahead step
        z_new = z + m
        gz_new = p * (gp - float(np.dot(p, gp)))  # Approximate (using old p)
        
        m = 0.9 * m - lr * gz_new
        
        z = z + m
        z = np.clip(z, -40, 40)
        z -= z.mean()
        
        p = softmax_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0)
            continue
        
        if it % beta_decay_interval == 0:
            beta = max(1e-12, beta * beta_decay_rate)
        
        if no_imp > restart_threshold:
            # Perturb and restart
            z = best_z + 0.5 * rng.standard_normal(n)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            p = softmax_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m.fill(0)
            no_imp = 0
    
    return best_p, best_score

def optimize_adam_v2(p0, deadline, rng, lr_init=0.10, beta_init=0.3, 
                     beta_decay=0.7, beta_interval=50, restart_at=300):
    """
    Adam optimizer in logit space with soft-max gradient.
    More aggressive hyperparameters than previous.
    """
    n = p0.size
    p = np.maximum(p0, 1e-15).copy()
    p = p / p.sum()
    
    z = np.log(p)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-10, beta_init * maxc)
    
    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    lr = lr_init
    
    while time.monotonic() < deadline:
        L, w = soft_max_grad(c, beta)
        gp = 2.0 * corr_with_weights(w, p)
        gz = p * (gp - float(np.dot(p, gp)))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1-b1) * gz
        v = b2 * v + (1-b2) * gz**2
        mh = m / (1 - b1**t_adam)
        vh = v / (1 - b2**t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        z = np.clip(z, -40, 40)
        z -= z.mean()
        p = softmax_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > 0 and no_imp % 200 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        if t_adam % beta_interval == 0:
            beta = max(1e-12, beta * beta_decay)
        
        if no_imp > restart_at:
            z = best_z + 0.7 * rng.standard_normal(n)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            p = softmax_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init * maxc)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = lr_init
            no_imp = 0
    
    return best_p, best_score

def coordinate_descent(p, deadline, rng, step_frac=0.1, max_sweeps=1000):
    """Coordinate descent polish."""
    n = p.size
    p = np.maximum(p, 1e-15).copy()
    p = p / p.sum()
    
    best_p = p.copy()
    best_score = score_normalized(p)
    no_imp = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score
            for factor in [1.3, 1.15, 0.85, 0.7]:
                new_val = orig * factor
                if new_val < 1e-14:
                    continue
                p[i] = new_val
                s = p.sum()
                p_norm = p / s
                sc = score_normalized(p_norm)
                if sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        if not improved:
            if no_imp > n:
                break
        if improved:
            step_frac = max(0.03, step_frac * 0.95)
    
    return best_p, best_score

def interpolate_to_length(p, n_new):
    n_old = p.size
    if n_old == n_new:
        return p.copy() / p.sum()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    p_new = np.interp(x_new, x_old, p)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()

**Alternative idea: Use the "flat spectrum" as a target**

The ideal autocorrelation is a delta at lag 0 (for a Cauchy-Schwarz-like bound). The next best thing is a flat autocorrelation (all lags equal). A flat autocorrelation corresponds to a flat power spectrum (by Parseval).

So: design a sequence whose power spectrum is as flat as possible, while maintaining non-negativity.

This is a constraint: `FFT(a)` should have approximately constant magnitude.

We can enforce this by:
1. Compute `A = FFT(a)`
2. Replace `|A|` with a modified target (e.g., flatten the peaks)
3. Keep the phase: `A_new = A / |A| * |A|_target`
4. IFFT to get new `a`
5. Project to non-negative

This is the "spectral shaping" approach. Let me implement it.

Actually, for the autocorrelation (not just correlation), the power spectrum is `|FFT(a)|^2`. So we want `|FFT(a)|^2` to be flat, i.e., `|FFT(a)|` to be constant.

The algorithm:
1. `A = rfft(a, N)`
2. `|A|_current = |A|`
3. `|A|_target = target_flatness * mean(|A|) + (1-target_flatness) * |A|_current`
4. `A_new = A / |A|_current * |A|_target` (preserve phase)
5. `a_new = irfft(A_new, N)[:n]`
6. `a_new = max(a_new, 0)`
7. Normalize

This is similar to the "projection" methods in MPSL literature.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    n = a.size
    c = _convolve_fft(a)
    if c.size == 0:
        return np.inf
    return 2.0 * n * float(c.max())

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        idx = int(np.argmax(c))
        w[idx] = 1.0
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 50.0))
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w

    z = np.log(p)
    z -= z.mean()

    b1, b2, eps = 0.90, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_adam = 0

    c = _convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-10, beta_frac * maxc)

    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    lr = lr_init
    it = 0

    while time.monotonic() < deadline:
        it += 1
        _, w = _soft_max_weights(c, beta)
        gp = 2.0 * _corr_weights(w, p)
        gz = p * (gp - float(np.dot(p, gp)))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        z = np.clip(z, -max_z, max_z)
        z -= z.mean()
        p = _softmax_z(z)
        c = _convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > 0 and no_imp % 150 == 0:
                    lr = max(0.02, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        if t_adam % beta_interval == 0:
            beta = max(1e-12, beta * beta_decay)

        if no_imp > restart_at:
            noise_scale = 0.5 + 0.3 * float(no_imp) / restart_at
            z = best_z + noise_scale * rng.standard_normal(n)
            z = np.clip(z, -max_z, max_z)
            z -= z.mean()
            p = _softmax_z(z)
            c = _convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = lr_init
            no_imp = 0

    return best_p, best_score


def _optimize_heavy_ball(p0, deadline, rng, lr_init=0.06, beta_frac=0.3,
                         momentum=0.85, beta_decay=0.7, beta_interval=50,
                         restart_at=250):
    n = p0.size
    p = np.maximum(p0, 1e-15).astype(np.float64).copy()
    p = p / p.sum()

    z = np.log(p)
    z -= z.mean()
    vel = np.zeros(n, dtype=np.float64)

    c = _convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-10, beta_frac * maxc)

    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    lr = lr_init
    it = 0

    while time.monotonic() < deadline:
        it += 1
        _, w = _soft_max_weights(c, beta)
        gp = 2.0 * _corr_weights(w, p)
        gz = p * (gp - float(np.dot(p, gp)))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            vel.fill(0)
            continue

        vel = momentum * vel - lr * gz
        z = z + vel
        z = np.clip(z, -45.0, 45.0)
        z -= z.mean()
        p = _softmax_z(z)
        c = _convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.01, lr * 0.92)
        else:
            z = best_z.copy()
            vel.fill(0)
            continue

        if it % beta_interval == 0:
            beta = max(1e-12, beta * beta_decay)

        if no_imp > restart_at:
            z = best_z + 0.6 * rng.standard_normal(n)
            z = np.clip(z, -45.0, 45.0)
            z -= z.mean()
            p = _softmax_z(z)
            c = _convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            vel.fill(0)
            lr = lr_init
            no_imp = 0

    return best_p, best_score


def _coord_descent(p, deadline, rng, step_sizes=None):
    n = p.size
    p = np.maximum(p, 1e-15).astype(np.float64).copy()
    p = p / p.sum()

    if step_sizes is None:
        step_sizes = [1.4, 1.2, 1.1, 0.9, 0.8, 0.6]

    best_p = p.copy()
    best_score = _score_norm(p)
    no_imp_sweeps = 0

    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score
            for mult in step_sizes:
                new_val = orig * mult
                if new_val < 1e-14 or new_val > 1e6:
                    continue
                p[i] = new_val
                p = p / p.sum()
                sc = _score_norm(p)
                if sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                improved = True
        if improved:
            no_imp_sweeps = 0
        else:
            no_imp_sweeps += 1
            if no_imp_sweeps >= 3:
                break

    return best_p, best_score


def _track(candidates, p, sc):
    candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
    candidates.sort(key=operator.itemgetter(0))
    if len(candidates) > 15:
        del candidates[15:]


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent.size > 0:
            parent = np.maximum(parent, 0.0)
            s = parent.sum()
            if s > 0:
                parent = parent / s
    except Exception:
        parent = None

    # === Phase 1: Coarse-to-fine multi-scale (0 - 400s) ===
    print("Phase 1: Multi-scale coarse-to-fine")
    p1_end = min(deadline, t_start + 400.0)

    scale_lengths = [32, 64, 128, 256, 512]
    scale_times = [3.0, 8.0, 25.0, 60.0, 120.0]

    current_best = None

    for li, (n_scale, t_alloc) in enumerate(zip(scale_lengths, scale_times)):
        if time.monotonic() > p1_end:
            break

        # Generate initial candidates at this scale
        profs = _init_profiles(n_scale, rng)
        if parent is not None and parent.size != n_scale:
            profs.append(_interp_length(parent, n_scale))
        elif parent is not None:
            profs.append(parent.copy())

        if current_best is not None and current_best.size != n_scale:
            profs.append(_interp_length(current_best, n_scale))

        # Score all initial profiles
        scored = []
        for p0 in profs:
            p0 = _norm_nonneg(p0)
            if p0 is None:
                continue
            sc = _score_norm(p0)
            if np.isfinite(sc):
                scored.append((sc, p0))

        scored.sort(key=operator.itemgetter(0))

        # Track best
        if scored:
            for sc, p in scored[:5]:
                if sc < best_score:
                    best_score = sc
                    best_seq = p.copy()
                _track(candidates, p, sc)

        # Optimize top candidates
        n_optimize = min(5, len(scored))
        time_per = t_alloc / max(1, n_optimize)

        for oi in range(n_optimize):
            if time.monotonic() > p1_end:
                break
            sc_o, p_o = scored[oi]
            per = min(p1_end, time.monotonic() + time_per)
            p_best, sc_best = _optimize_adam(p_o, per, rng,
                                              lr_init=0.15, beta_frac=0.4,
                                              beta_decay=0.6, beta_interval=30,
                                              restart_at=100)
            sc_final = _score_norm(p_best)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = p_best.copy()
                print(f"  scale n={n_scale} opt {oi}: {sc_final:.10f}")
            _track(candidates, p_best, sc_final)

        # Update current_best for next scale
        if scored:
            current_best = scored[0][1].copy()
            if candidates:
                current_best = candidates[0][1].copy()

        print(f"  After n={n_scale}: best={best_score:.10f}")

    # === Phase 2: Fine optimization at multiple lengths (400 - 1000s) ===
    print("Phase 2: Fine optimization")
    p2_end = min(deadline, t_start + 1000.0)

    fine_lengths = [256, 384, 512, 640]

    for n_fine in fine_lengths:
        if time.monotonic() > p2_end:
            break

        # Get starting points for this length
        starts = []
        for sc_c, p_c in candidates[:8]:
            if p_c.size == n_fine:
                starts.append(p_c)
            else:
                starts.append(_interp_length(p_c, n_fine))
        if parent is not None and parent.size != n_fine:
            starts.append(_interp_length(parent, n_fine))

        # Add some fresh random starts
        for _ in range(3):
            starts.append(_make_spectral(n_fine, rng, flatness=rng.uniform(0.4, 0.7)))

        # Optimize each start
        n_starts = len(starts)
        time_per = (p2_end - time.monotonic()) / max(1, n_starts)
        time_per = max(time_per, 5.0)
        time_per = min(time_per, 50.0)

        for si in range(n_starts):
            if time.monotonic() > p2_end:
                break
            p_s = _norm_nonneg(starts[si])
            if p_s is None:
                continue
            per = min(p2_end, time.monotonic() + time_per)

            # Try Adam
            p_best1, sc1 = _optimize_adam(p_s, per, rng,
                                          lr_init=0.08, beta_frac=0.2,
                                          beta_decay=0.75, beta_interval=50,
                                          restart_at=300)
            sc1 = _score_norm(p_best1)
            if sc1 < best_score:
                best_score = sc1
                best_seq = p_best1.copy()
                print(f"  fine n={n_fine} adam {si}: {sc1:.10f}")
            _track(candidates, p_best1, sc1)

            # Try heavy ball if time allows
            if time.monotonic() < p2_end:
                per2 = min(p2_end, time.monotonic() + time_per * 0.7)
                p_best2, sc2 = _optimize_heavy_ball(p_s, per2, rng,
                                                    lr_init=0.04, beta_frac=0.25,
                                                    momentum=0.88,
                                                    beta_decay=0.75,
                                                    beta_interval=50,
                                                    restart_at=300)
                sc2 = _score_norm(p_best2)
                if sc2 < best_score:
                    best_score = sc2
                    best_seq = p_best2.copy()
                    print(f"  fine n={n_fine} hb {si}: {sc2:.10f}")
                _track(candidates, p_best2, sc2)

        print(f"  After fine n={n_fine}: best={best_score:.10f}")

    # === Phase 3: Extended polish of top candidates (1000 - 1500s) ===
    print("Phase 3: Extended polish")
    p3_end = min(deadline, t_start + 1500.0)

    for i in range(min(6, len(candidates))):
        if time.monotonic() > p3_end:
            break
        sc_c, p_c = candidates[i]
        per = min(p3_end, time.monotonic() + 60.0)
        p_best, sc_best = _optimize_adam(p_c, per, rng,
                                         lr_init=0.04, beta_frac=0.12,
                                         beta_decay=0.85, beta_interval=80,
                                         restart_at=500, max_z=40.0)
        sc_best = _score_norm(p_best)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best.copy()
            print(f"  polish {i}: {sc_best:.10f}")
        _track(candidates, p_best, sc_best)

    print(f"  After phase 3: best={best_score:.10f}")

    # === Phase 4: Different hyperparameter sweep (1500 - 1650s) ===
    print("Phase 4: Hyperparameter sweep")
    p4_end = min(deadline, t_start + 1650.0)

    top_cands = candidates[:4]
    hp_configs = [
        (0.20, 0.5, 0.55, 25, 80),
        (0.10, 0.25, 0.65, 35, 150),
        (0.06, 0.15, 0.80, 60, 400),
        (0.15, 0.35, 0.60, 30, 120),
    ]

    for ci, (sc_c, p_c) in enumerate(top_cands):
        if time.monotonic() > p4_end:
            break
        for hi, (lr_h, bf_h, bd_h, bi_h, ra_h) in enumerate(hp_configs):
            if time.monotonic() > p4_end:
                break
            per = min(p4_end, time.monotonic() + 20.0)
            p_best, sc_best = _optimize_adam(p_c, per, rng,
                                             lr_init=lr_h, beta_frac=bf_h,
                                             beta_decay=bd_h,
                                             beta_interval=bi_h,
                                             restart_at=ra_h)
            sc_best = _score_norm(p_best)
            if sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  hp sweep {ci},{hi}: {sc_best:.10f}")
            _track(candidates, p_best, sc_best)

    # === Phase 5: Coordinate descent (1650 - 1770s) ===
    print("Phase 5: Coordinate descent")
    p5_end = min(deadline, t_start + 1770.0)

    for i in range(min(6, len(candidates))):
        if time.monotonic() > p5_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size > 700:
            continue
        per = min(p5_end, time.monotonic() + 15.0)
        p_best, sc_best = _coord_descent(p_c, per, rng)
        sc_best = _score_norm(p_best)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best.copy()
            print(f"  CD {i}: {sc_best:.10f}")
        _track(candidates, p_best, sc_best)

    # === Final verification ===
    print("Final verification")
    if best_seq is not None:
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(best_seq.sum())
        if s > 0:
            best_seq = best_seq / s

        # Check all candidates with true score
        for sc_c, p_c in candidates:
            if p_c.size <= 2048:
                p_clip = np.clip(p_c, 0.0, 1000.0)
                p_norm = p_clip / p_clip.sum()
                sc_true = _true_score(p_norm)
                if np.isfinite(sc_true) and sc_true < best_score:
                    best_score = sc_true
                    best_seq = p_norm.copy()

        # Final exact check
        sc_final = _true_score(best_seq)
        print(f"  FINAL: score={sc_final:.10f} n={best_seq.size}")

        try:
            sc_eval = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"  Evaluator confirms: {sc_eval:.10f}")
        except Exception:
            pass

        return [float(x) for x in best_seq]

    # Fallback
    fallback = _make_linear(512, 2.0)
    print(f"  FALLBACK: {_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
