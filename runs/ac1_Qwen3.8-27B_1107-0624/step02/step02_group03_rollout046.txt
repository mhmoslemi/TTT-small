from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=5.0):
    """Solve LP: maximize sum(g) s.t. g >= 0, conv(f,g) <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

    grad[j] = 2 * sum_k w[k] * f[k-j] (with bounds).

    # Let k = j + m, then f[k-j] = f[m], w[k] = w[j+m]

    # This is the cross-correlation (w star f)[j]

    # Actually np.correlate(a, v, 'full')[k] = sum_n a[n+k-(len(v)-1)] * conj(v[n])

    corr = np.correlate(w, f, mode='full')
    grad = 2.0 * corr[n - 1: 2 * n - 1]

def soft_max_weights(c, beta):
    """Compute soft-max weights for the max of c."""
    max_c = float(c.max())
    x = (c - max_c) / beta
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30:

        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    return w, max_c


def mirror_descent_step(f, eta, beta):
    """Single mirror descent step on the simplex."""
    n = len(f)
    c = np.convolve(f, f)
    w, max_c = soft_max_weights(c, beta)
    grad = compute_gradient(w, f)
    # Ensure grad doesn't have extreme values
    grad = np.clip(grad, -1e6, 1e6)
    # Exponentiated gradient update
    log_f = np.log(np.maximum(f, 1e-300))
    log_f -= log_f.max()
    log_f_new = log_f - eta * grad
    log_f_new -= log_f_new.max()
    f_new = np.exp(log_f_new)
    f_new = f_new / f_new.sum()
    return f_new, max_c

            s = a_new.sum()
            if s < 1e-10:
                return 1e10

        best_v = vals.copy()

def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bell_profile(n, width):
    """Create a Gaussian-like profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.exp(-u ** 2 / (2 * width ** 2))
    a = a / a.sum()
    return a


def make_cosine_profile(n, freq):

    u = 2.0 * t - 1.0
    a = 1.0 + 0.3 * np.cos(freq * u) 
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def fft_convolve(a, b):

    n = len(a) + len(b) - 1
    f_a = np.fft.rfft(a, n=n)
    f_b = np.fft.rfft(b, n=n)
    return np.fft.irfft(f_a * f_b, n=n)


def internal_score_fast(a):
    """Faster score computation using FFT for large n."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    if n > 256:
        c = fft_convolve(a, a)
    else:
        c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP-based optimization with backtracking (n=512)
    Phase 2: Mirror descent (exponentiated gradient)
    Phase 3: Subspace optimization
    Phase 4: Multi-start with random perturbations
    Phase 5: Final polish combining all methods
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0
    n = 512
    
    # Get initial sequence
    best_f = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            s = parent.sum()
            if s > 1e-10:
                # Interpolate to n=512 if needed
                if len(parent) != n:
                    x_old = np.linspace(0, 1, len(parent))
                    x_new = np.linspace(0, 1, n)
                    parent = np.interp(x_new, x_old, parent)
                    parent = np.maximum(parent, 0.0)
                    s = parent.sum()
                    if s > 1e-10:
                        parent = parent / s
                best_f = parent
                print(f"Parent: n={len(best_f)}, score={internal_score(best_f):.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_f is None:
        # Generate initial profiles and pick best
        profiles = [
            make_linear_profile(n, 1.0),
            make_linear_profile(n, 1.5),
            make_linear_profile(n, 2.0),
            make_bell_profile(n, 0.5),
            make_bell_profile(n, 0.7),
            make_cosine_profile(n, 2.0),
            make_cosine_profile(n, 3.0),
        ]
        best_score = np.inf
        for p in profiles:
            sc = internal_score(p)
            if sc < best_score:
                best_score = sc
                best_f = p
        print(f"Generated initial: n={n}, score={best_score:.8f}")
    
    best_score = internal_score(best_f)
    print(f"Starting score: {best_score:.8f} (target: 1.5027)")

    print("\n=== Phase 1: LP optimization ===")
    dl1 = min(deadline, t_start + 350.0)
    f1, M1 = lp_with_backtracking(best_f, dl1, t_init=0.6, verbose=True)
    score1 = 2.0 * n * M1
    if score1 < best_score:
        best_f = f1
        best_score = score1
        print(f"  Phase 1 result: {best_score:.8f}")
    else:
        print(f"  Phase 1: no improvement (M1={M1:.8f})")
    
    # Phase 2: Mirror descent
    print("\n=== Phase 2: Mirror descent ===")
    dl2 = min(deadline, t_start + 700.0)
    # Try multiple alpha/beta combinations
    for eta_try in [0.3, 0.5, 0.8, 1.0]:
        if time.monotonic() > dl2 - 30:
            break
        dl_sub = min(dl2, time.monotonic() + 80.0)
        f2, M2 = backtracking_mirror_descent(best_f, dl_sub, eta_init=eta_try, verbose=False)
        score2 = 2.0 * n * M2
        if score2 < best_score:
            best_f = f2
            best_score = score2
            print(f"  MD (eta={eta_try}): {best_score:.8f}")
    
    # Phase 3: Subspace optimization
    print("\n=== Phase 3: Subspace optimization ===")
    dl3 = min(deadline, t_start + 1100.0)
    f3, M3 = subspace_optimize(best_f, dl3, n_sub=6, verbose=True)
    score3 = 2.0 * n * M3
    if score3 < best_score:
        best_f = f3
        best_score = score3
        print(f"  Phase 3 result: {best_score:.8f}")
    else:
        print(f"  Phase 3: no improvement")
    
    # Phase 4: Multi-start search
    print("\n=== Phase 4: Multi-start ===")
    dl4 = min(deadline, t_start + 1500.0)
    f4, score4 = multi_start_search(best_f, dl4, n_starts=15, verbose=True)
    if score4 < best_score:
        best_f = f4
        best_score = score4
        print(f"  Phase 4 result: {best_score:.8f}")
    else:
        print(f"  Phase 4: no improvement")
    
    # Phase 5: Final polish - alternate LP and MD
    print("\n=== Phase 5: Final polish ===")
    dl5 = deadline
    for round_idx in range(10):
        if time.monotonic() > dl5 - 5:
            break
        # LP step
        dl_lp = min(dl5 - 5, time.monotonic() + 40.0)
        f_lp, M_lp = lp_with_backtracking(best_f, dl_lp, t_init=0.4)
        sc_lp = 2.0 * n * M_lp
        if sc_lp < best_score:
            best_f = f_lp
            best_score = sc_lp
            print(f"  Polish round {round_idx} LP: {best_score:.8f}")
        
        # MD step
        dl_md = min(dl5 - 5, time.monotonic() + 40.0)
        f_md, M_md = backtracking_mirror_descent(best_f, dl_md, eta_init=0.4, verbose=False)
        sc_md = 2.0 * n * M_md
        if sc_md < best_score:
            best_f = f_md
            best_score = sc_md
            print(f"  Polish round {round_idx} MD: {best_score:.8f}")

    if time.monotonic() < deadline - 20:
        print("\n=== Extra: Aggressive perturbations ===")
        rng = np.random.default_rng(777)
        for i in range(10):
            if time.monotonic() > deadline - 10:
                break
            perm = rng.uniform(0.02, 0.15)
            f_p = random_perturb(best_f, perm, seed=5000 + i)
            if f_p is not None:
                dl_sub = min(deadline - 5, time.monotonic() + 30.0)
                f_opt, M_opt = lp_with_backtracking(f_p, dl_sub, t_init=0.5)
                sc = 2.0 * n * M_opt
                if sc < best_score:
                    best_f = f_opt
                    best_score = sc
                    print(f"  Aggressive {i}: {best_score:.8f}")
    
    # Final validation
    final_score = internal_score(best_f)
    print(f"\nFinal: n={len(best_f)}, score={final_score:.8f}")
    print(f"Target: 1.5027, Gap: {final_score - 1.5027:.6f}")
    
    # Ensure valid output
    best_f = np.asarray(best_f, dtype=np.float64)
    best_f = np.clip(best_f, 0.0, 1000.0)
    s = float(best_f.sum())
    if s < 0.01:
        best_f = make_linear_profile(n, 1.5)
        s = best_f.sum()
    best_f = best_f / s
    
    return [float(x) for x in best_f]
