import numpy as np

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_gradient(a, beta):
    """Compute gradient of softmax-smoothed max of conv(a,a)."""
    n = a.size
    c = fft_convolve(a)

    x = c / beta
    x = x - x.max()
    w = np.exp(x)

    if w_sum < 1e-30:

        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    W = np.fft.rfft(w, N)
    grad = np.fft.irfft(np.conj(A) * W, N)[:n]

    t = np.linspace(0.0, 1.0, n)

def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u**3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=6):
    t = np.linspace(0.0, 1.0, n)

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

"""
Gradient-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses softmax-smoothed max objective with Adam optimizer in log-space,
monotonicity enforcement, adaptive temperature, and multi-length search.
Combines fast FFT-based gradients with careful temperature scheduling
and plateau escape mechanisms.
"""

import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u**3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=6):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-15)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def gradient_optimize(f0, deadline, rng, enforce_monotone=True,
                      init_beta_frac=0.3, min_beta_frac=0.02,
                      init_lr=0.1, min_lr=0.001,
                      beta_decay=0.995, lr_decay=0.999,
                      max_iter=100000, plateau_patience=400,
                      verbose=False):
    """Adam-based gradient optimization in log-space with adaptive temperature."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    z = np.log(f)
    z -= z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = fft_convolve(f)
    M_cur = float(c.max())
    beta = init_beta_frac * M_cur
    lr = init_lr

    best_f = f.copy()
    best_M = M_cur
    best_z = z.copy()
    no_imp = 0

    N_corr = next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute convolution and softmax weights
        c = fft_convolve(f)
        M_cur = float(c.max())

        x = c / beta
        x = x - x.max()
        w = np.exp(x)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum

        # Cross-correlation for gradient: grad_i = sum_j f_j * w_{j+i}
        A = np.fft.rfft(f, N_corr)
        W = np.fft.rfft(w, N_corr)
        grad = np.fft.irfft(np.conj(A) * W, N_corr)[:n]

        # Project to simplex tangent space
        grad_z = f * (grad - np.dot(f, grad))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1**t_adam)
        vh = v / (1.0 - b2**t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if enforce_monotone:
            z_sorted = np.sort(z)
            z = z_sorted

        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        f_new = np.exp(z - z.max())
        fs = f_new.sum()
        if fs < 1e-30 or not np.isfinite(fs):
            z = best_z.copy()
            f = best_f.copy()
            continue
        f = f_new / fs

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-13:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive schedule
        if it % 100 == 0:
            beta = max(min_beta_frac * M_new, beta * beta_decay)
            lr = max(min_lr, lr * lr_decay)

        # Plateau escape
        if no_imp > plateau_patience:
            strength = 0.2 + 0.2 * rng.random()
            z = (1.0 - strength) * best_z + strength * rng.normal(size=n) * 0.5
            if enforce_monotone:
                z = np.sort(z)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_cur = float(c.max())
            beta = max(0.1 * M_cur, min_beta_frac * M_cur)
            lr = max(init_lr * 0.3, lr)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

        if verbose and it % 500 == 0:
            print(f"      iter {it}: M={M_new:.12f}, best={best_M:.12f}, beta={beta:.6f}")

    return best_f, best_M


def local_peak_reduction(f0, deadline, rng, max_iter=5000):
    """Try to directly reduce the convolution peak by local adjustments."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    best_f = f.copy()
    best_M = float(fft_convolve(f).max())
    step = 0.05

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(f)
        M_cur = float(c.max())
        peak_pos = int(np.argmax(c))

        # Find positions contributing to the peak
        # c[peak_pos] = sum_i f[i] * f[peak_pos - i]
        contribs = []
        for i in range(n):
            j = peak_pos - i
            if 0 <= j < n:
                contribs.append((i, j, f[i] * f[j]))

        # Sort by contribution (largest first)
        contribs.sort(key=lambda x: -x[2])

        # Try reducing the top contributors and adding to "quiet" positions
        top_k = min(5, len(contribs))
        proposed = f.copy()

        # Reduce top contributors
        for idx in range(top_k):
            i, j, val = contribs[idx]
            reduction = val * step * 0.5
            proposed[i] -= reduction * (f[i] / f[i] if f[i] > 0 else 0)
            if i != j:
                proposed[j] -= reduction * (f[j] / f[j] if f[j] > 0 else 0)

        # Find quiet positions (low contribution to any c_k near peak)
        # Add mass to positions far from the peak
        min_dist = n // 3
        candidates = [i for i in range(n) if abs(i - peak_pos) >= min_dist or i < min_dist]
        if not candidates:
            candidates = list(range(n))

        # Add mass proportionally to current values
        total_reduction = f.sum() - proposed.sum()
        if total_reduction > 0:
            cand_sum = sum(f[i] for i in candidates)
            if cand_sum > 1e-30:
                for i in candidates:
                    proposed[i] += total_reduction * (f[i] / cand_sum)

        proposed = np.maximum(proposed, 1e-15)
        ps = proposed.sum()
        if ps < 1e-30:
            step *= 0.5
            continue
        proposed = proposed / ps

        c_new = fft_convolve(proposed)
        M_new = float(c_new.max())

        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = proposed.copy()
            f = proposed.copy()
        else:
            step *= 0.95
            if step < 1e-6:
                break

    return best_f, best_M

        try_update(p_best, sc)
        candidates.append((sc, p_best))
        print(f"  P1 [{name}]: score={sc:.10f}, best={best_score:.10f}")

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")

    # === Phase 2: Extended optimization on top candidates ===
    phase2_end = min(deadline, t_start + 1150.0)
    top_k = min(6, len(candidates))
    per_cand = max(20.0, (phase2_end - t_start) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates[i]
        per = min(phase2_end, time.monotonic() + per_cand)
        p_best, M_best = gradient_optimize(p0, per, rng, enforce_monotone=True,
                                           init_beta_frac=0.15, min_beta_frac=0.008,
                                           init_lr=0.06, min_lr=0.0005,
                                           beta_decay=0.998, lr_decay=0.9995,
                                           plateau_patience=500)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  P2 cand {i}: score={sc:.10f}, best={best_score:.10f}")

    # Local peak reduction on best
    if best_seq is not None and best_seq.size == n1 and time.monotonic() < phase2_end:
        per = min(phase2_end, time.monotonic() + 30.0)
        p_best, M_best = local_peak_reduction(best_seq, per, rng)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  P2 local: score={sc:.10f}, best={best_score:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # === Phase 3: Non-monotone exploration ===
    phase3_end = min(deadline, t_start + 1300.0)
    if best_seq is not None and best_seq.size == n1 and time.monotonic() < phase3_end:
        per = min(phase3_end, time.monotonic() + 60.0)
        p_best, M_best = gradient_optimize(best_seq, per, rng, enforce_monotone=False,
                                           init_beta_frac=0.1, min_beta_frac=0.01,
                                           init_lr=0.04, min_lr=0.001,
                                           beta_decay=0.998, lr_decay=0.999,
                                           plateau_patience=300)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  P3 non-monotone: score={sc:.10f}, best={best_score:.10f}")

    # === Phase 4: Try n=1024 ===
    n2 = 1024
    phase4_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and time.monotonic() < phase4_end - 30:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase4_end, time.monotonic() + 100.0)
            p_best, M_best = gradient_optimize(p_ext, per, rng, enforce_monotone=True,
                                               init_beta_frac=0.2, min_beta_frac=0.01,
                                               init_lr=0.05, min_lr=0.001,
                                               beta_decay=0.997, lr_decay=0.999,
                                               plateau_patience=400)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"  P4 (n={n2}): score={sc:.10f}, best={best_score:.10f}")

    # === Phase 5: Try n=2048 briefly ===
    n3 = 2048
    if best_seq is not None and time.monotonic() < phase4_end - 30:
        p_ext = interpolate_profile(best_seq, n3)
        if p_ext is not None:
            per = min(phase4_end, time.monotonic() + 50.0)
            p_best, M_best = gradient_optimize(p_ext, per, rng, enforce_monotone=True,
                                               init_beta_frac=0.2, min_beta_frac=0.01,
                                               init_lr=0.04, min_lr=0.001,
                                               beta_decay=0.998, lr_decay=0.999,
                                               plateau_patience=300)
            sc = 2.0 * n3 * M_best
            try_update(p_best, sc)
            print(f"  P5 (n={n3}): score={sc:.10f}, best={best_score:.10f}")

    print(f"After Phase 5: best={best_score:.10f}")

    # === Phase 6: Final intensive polish ===
    phase6_end = deadline
    if best_seq is not None and time.monotonic() < phase6_end - 10:
        n_best = best_seq.size
        per = min(phase6_end, time.monotonic() + 100.0)
        p_best, M_best = gradient_optimize(best_seq, per, rng, enforce_monotone=True,
                                           init_beta_frac=0.05, min_beta_frac=0.003,
                                           init_lr=0.02, min_lr=0.0003,
                                           beta_decay=0.9995, lr_decay=0.9998,
                                           plateau_patience=300)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        print(f"  P6 polish: score={sc:.10f}, n={n_best}")

        # One more local reduction
        if time.monotonic() < phase6_end - 5:
            p_best2, M_best2 = local_peak_reduction(p_best, phase6_end, rng, max_iter=1000)
            sc2 = 2.0 * n_best * M_best2
            try_update(p_best2, sc2)
            print(f"  P6 local: score={sc2:.10f}")

    # Final evaluation and return
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(sc_eval):
                print(f"Final evaluated score: {sc_eval:.10f}")
        except Exception as e:
            print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
