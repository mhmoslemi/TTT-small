from scipy.optimize import linprog

def solve_lp(a, M):
    n = len(a)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (a*g)[k] <= M for k = 0, ..., 2n-2
    # (a*g)[k] = sum_i a[i] * g[k-i] (where 0 <= i < n and 0 <= k-i < n)
    
    # Build constraint matrix
    rows = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = a[i]
        rows.append(row)
    
    A_ub = np.array(rows)  # shape (2n-1, n)
    b_ub = np.full(2*n - 1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x
