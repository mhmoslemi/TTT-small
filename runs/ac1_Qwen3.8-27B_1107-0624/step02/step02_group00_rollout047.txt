Hybrid bottleneck-LP + spectral shaping + gradient optimization for minimizing
2n*max(conv(a,a))/sum(a)^2. Key improvements over previous: (1) bottleneck LP
that directly minimizes max cross-correlation, (2) spectral domain shaping for
diversification, (3) aggressive M deflation schedule, (4) multi-length search,
(5) trust-region QP via cvxpy for better local moves.

import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _bottleneck_lp(a, target_t=None):
    """Solve: min t s.t. C(a)@b <= t, sum(b)=1, b>=0.
    Returns (b_opt, t_opt) or (None, None) if infeasible."""
    n = len(a)
    C = _make_conv_matrix(a)
    b_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        C @ b_var <= t_var,
        cp.sum(b_var) == 1,
        b_var >= 0
    ]
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000, abstol=1e-8, reltol=1e-6)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=5000)
        except Exception:
            return None, None
    if b_var.value is None or t_var.value is None:
        return None, None
    b_opt = np.asarray(b_var.value, dtype=np.float64)
    b_opt = np.maximum(b_opt, 0.0)
    s = b_opt.sum()
    if s < 1e-10:
        return None, None
    b_opt = b_opt / s
    t_opt = float(t_var.value)
    return b_opt, t_opt


def _bottleneck_lp_scipy(a):
    """Scipy-based bottleneck LP using the trick: min t s.t. C@b - t*1 <= 0, sum(b)=1, b>=0."""
    n = len(a)
    C = _make_conv_matrix(a)
    # Variables: [b_0, ..., b_{n-1}, t]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    # Inequality: C @ b - t <= 0  =>  [C, -1] @ [b; t] <= 0
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        b_opt = np.maximum(res.x[:n], 0.0)
        s = b_opt.sum()
        if s > 1e-10:
            b_opt = b_opt / s
            t_opt = float(res.x[-1])
            return b_opt, t_opt
    return None, None


def _lp_spread(a, M):
    """Original LP: max sum(b) s.t. C(a)@b <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine_v2(a, deadline, max_iter=20000, mode='bottleneck'):
    """Improved LP refinement with bottleneck LP and aggressive M deflation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    blend = 0.4
    no_imp = 0
    total_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M_cur = float(c.max())

        if mode == 'bottleneck':
            b_opt, t_opt = _bottleneck_lp_scipy(a)
            if b_opt is None:
                # Fall back to spread LP
                M_lp = M_cur * 1.005
                b_opt = _lp_spread(a, M_lp)
                if b_opt is None:
                    break
                t_opt = M_lp

            t_opt = max(t_opt, 1e-12)
            # Only move if the bottleneck is significantly less than current max
            if t_opt >= M_cur * 0.999:
                # Try spread LP with slightly inflated M
                M_lp = M_cur * 1.002
                b_spread = _lp_spread(a, M_lp)
                if b_spread is not None and float(b_spread.sum()) > 1.0 + 1e-6:
                    b_opt = b_spread / b_spread.sum()
                else:
                    no_imp += 1
                    if no_imp > 200:
                        blend = max(0.1, blend * 0.95)
                    if no_imp > 1000:
                        break
                    continue

            sum_b = float(b_opt.sum())
            if sum_b <= 1e-10:
                break
            b_opt = b_opt / sum_b

        else:
            M_lp = M_cur * 1.002
            b_opt = _lp_spread(a, M_lp)
            if b_opt is None:
                break
            sum_b = float(b_opt.sum())
            if sum_b <= 1.0 + 1e-6:
                no_imp += 1
                if no_imp > 500:
                    break
                continue
            b_opt = b_opt / sum_b

        a_new = (1.0 - blend) * a + blend * b_opt
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            total_imp += 1
            # Slowly decrease blend for stability
            blend = max(0.15, blend * 0.999)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend = max(0.1, blend * 0.9)
            if no_imp > 500:
                a = best_a.copy()
                blend = 0.4
                no_imp = 0
                if total_imp == 0:
                    break

        a = a_new

    return best_a


def _smooth_max_and_grad(a, beta):
    """Compute smooth max of autocorrelation and its gradient."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    # Gradient: d/dc (smooth_max) = w, and dc/da[i] = 2*(a_rev * w)[i]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    val = mval + beta * float(np.log(max(s, 1e-300)))
    return val, w, grad


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a, deadline, max_iter=30000):
    """Adam + Nesterov gradient descent on simplex with smooth max."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    c_max = float(c.max())
    beta = max(1e-6, 0.05 * c_max)
    lr = 0.03
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        _, _, grad = _smooth_max_and_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            v = np.zeros(n)
            beta = max(1e-6, beta * 2.0)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.93)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 100 == 0:
            c_max = float(_fft_convolve(a).max())
            beta = max(1e-9, 0.05 * c_max)

    return best_a


def _logit_adam_polish(a, deadline, max_iter=30000, lr=0.04):
    """Adam in logit parameterization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-6, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        _, _, grad_p = _smooth_max_and_grad(p, beta)
        # Convert gradient from p-space to z-space: gz = p * (gp - mean(p*gp))
        dot = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.95)

        if it % 150 == 0:
            c_max = float(_fft_convolve(a).max())
            beta = max(1e-9, 0.05 * c_max)

        if no_imp > max(3000, 4 * n):
            rng = np.random.default_rng(it * 7 + 3)
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-6, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _spectral_shaping(a, n_iter=200, target_flatness=0.7):
    """Fuchs-like spectral shaping: flatten the power spectrum to reduce autocorrelation peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = float(_score(a))

    N = _next_pow2(2 * n - 1)

    for it in range(n_iter):
        c = _fft_convolve(a)
        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Compute power spectrum
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2

        # Target: flatten the spectrum
        # Where autocorrelation has peaks, we want to reduce corresponding spectral components
        # Simple approach: geometric mean of current and flat spectrum
        total_power = power.sum()
        flat_power = total_power / len(power)

        # Blend towards flat
        alpha = target_flatness
        new_power = (1 - alpha) * power + alpha * flat_power
        new_power = np.maximum(new_power, 1e-12)

        # Preserve phase
        new_A = np.sqrt(new_power) * np.exp(1j * np.angle(A))
        # Make sure A[0] is real and positive (DC component)
        new_A[0] = abs(new_A[0])

        a_new = np.fft.irfft(new_A, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        a = a_new

    return best_a


def _make_profile(n, profile='linear', param=1.5):
    t = np.linspace(0.0, 1.0, n)
    if profile == 'linear':
        a = 1.0 + param * (t - 0.5)
    elif profile == 'cosine':
        a = 1.0 + param * np.cos(np.pi * t)
    elif profile == 'raised_cosine':
        a = 1.0 + param * np.cos(2.0 * np.pi * t) ** 2
    elif profile == 'super_gauss':
        a = np.exp(-param * ((t - 0.5) * 2) ** 4)
    elif profile == 'cubic':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u - 0.5 * param * u ** 3
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _cvxpy_trust_region(a, M_target, radius=0.1):
    """Solve: min max_i (b*b)_i approximately via:
    min t s.t. C(b)@b <= t (linearized around a), sum(b)=1, b>=0, ||b-a||<=r
    Linearization: C(a)@b <= t (use cross-correlation as proxy)
    """
    n = len(a)
    C = _make_conv_matrix(a)
    b_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        C @ b_var <= t_var,
        cp.sum(b_var) == 1,
        b_var >= 0,
        cp.norm(b_var - a, 2) <= radius
    ]
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000, abstol=1e-9, reltol=1e-7)
    except Exception:
        return None
    if b_var.value is None:
        return None
    b_opt = np.asarray(b_var.value, dtype=np.float64)
    b_opt = np.maximum(b_opt, 0.0)
    s = b_opt.sum()
    if s < 1e-10:
        return None
    return b_opt / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Bottleneck LP refinement (direct max cross-corr minimization)
    2. Spectral shaping (Fuchs-like)
    3. Gradient polish (smooth max + Adam in logit space)
    4. Trust-region QP for local moves
    5. Multi-length search with diverse initializations
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    lengths = [512]
    if parent is not None:
        lengths.insert(0, parent.size)
    for ln in [384, 768, 1024]:
        if ln not in lengths:
            lengths.append(ln)

    # Phase 1: Bottleneck LP refinement on each length
    for n_t in lengths:
        if time.monotonic() > deadline - 200:
            break

        if parent is not None:
            a0 = _interpolate_profile(parent, n_t)
        else:
            a0 = _make_profile(n_t, 'linear', 1.5)

        sc0 = _score(a0)
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # LP refinement
        lp_budget = min(400.0, deadline - time.monotonic() - 200.0)
        if lp_budget < 20:
            continue
        lp_end = time.monotonic() + lp_budget
        print(f"Phase 1 LP (bottleneck): n={n_t}, start={sc0:.10f}, budget={lp_budget:.0f}s")

        a_lp = _lp_refine_v2(a0, lp_end, max_iter=15000, mode='bottleneck')
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        print(f"  LP result: score={sc_lp:.10f}")

        # Quick gradient polish
        gp_budget = min(80.0, deadline - time.monotonic() - 200.0)

            a_gp = _gradient_polish(a_lp, gp_end, max_iter=8000)

                best_seq = a_gp.copy()
            print(f"  Grad polish: score={sc_gp:.10f}")

    # Phase 2: Spectral shaping diversification
    if best_seq is not None and time.monotonic() < deadline - 200:

        if ss_budget > 20:
            ss_end = time.monotonic() + ss_budget

            sc_ss = _score(a_ss)
            if np.isfinite(sc_ss) and sc_ss < best_score:
                best_score = sc_ss
                best_seq = a_ss.copy()
            print(f"  Spectral result: score={sc_ss:.10f}")

            # Polish the spectral result
            if sc_ss < best_score + 0.01 and time.monotonic() < deadline - 100:
                gp_end = time.monotonic() + min(60.0, deadline - time.monotonic() - 100.0)
                a_gp = _gradient_polish(a_ss, gp_end, max_iter=5000)
                sc = _score(a_gp)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_gp.copy()
                print(f"  Polish spectral: score={sc:.10f}")

    if time.monotonic() < deadline - 150:

        for prof, par in [('linear', 1.0), ('linear', 1.5), ('linear', 2.0),

                best_seq = a_ad.copy()
            print(f"  Adam result: score={sc_ad:.10f}")

    if best_seq is not None and time.monotonic() < deadline - 40:
        lp_budget = min(80.0, deadline - time.monotonic() - 40.0)
        if lp_budget > 15:
            lp_end = time.monotonic() + lp_budget
            print(f"Phase 5: Final LP, budget={lp_budget:.0f}s")
            a_final_lp = _lp_refine_v2(best_seq, lp_end, max_iter=5000, mode='bottleneck')
            sc_final = _score(a_final_lp)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final_lp.copy()
            print(f"  Final LP: score={sc_final:.10f}")

    # Phase 6: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        gp_budget = min(60.0, deadline - time.monotonic() - 20.0)
        if gp_budget > 10:
            gp_end = time.monotonic() + gp_budget
            a_final_gp = _gradient_polish(best_seq, gp_end, max_iter=10000)
            sc_final = _score(a_final_gp)

                best_score = sc_final
                best_seq = a_final_gp.copy()
            print(f"  Final grad: score={sc_final:.10f}")

    # Final cleanup
    if best_seq is None:
        best_seq = _make_profile(256, 'linear', 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_profile(256, 'linear', 1.5)

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
