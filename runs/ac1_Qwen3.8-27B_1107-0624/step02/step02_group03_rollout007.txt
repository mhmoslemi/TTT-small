"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines parameterized optimization, Adam gradient descent with temperature
annealing, and LP-based polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def compute_grad_p_fft(p, w):
    """Compute gradient of weighted sum of convolution values."""
    n = len(p)
    p_flip = p[::-1]
    conv_result = np.convolve(p_flip, w)
    grad_p = 2.0 * conv_result[n-1:2*n-1]
    return grad_p

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def lp_direction(f, M, timeout=30.0):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_iterate(f, max_iters=200, t_init=0.3, t_min=0.01, deadline=None, verbose=True):
    n = len(f)
    f = normalize(f.astype(np.float64).copy())
    if f is None:
        return None, np.inf
    
    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0
    t_step = t_init
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        g_raw = lp_direction(f, M, timeout=min(30.0, max(5.0, (deadline - time.monotonic()) / 3 if deadline else 30.0)))
        if g_raw is None:
            break
        
        g = normalize(g_raw)
        if g is None:
            break
        
        # Try blending
        for t_try in [t_step, 0.5*t_step, 0.25*t_step]:
            f_new = normalize((1-t_try)*f + t_try*g)
            if f_new is None:
                continue
            c_new = np.convolve(f_new, f_new)
            M_new = float(c_new.max())
            score_new = 2.0 * n * M_new
            
            if score_new < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = score_new
                f = f_new.copy()
                M = M_new
                no_improve = 0
                t_step = min(0.4, t_step * 1.01)
                break
        else:
            no_improve += 1
            f = best_f.copy()
            M = float(np.convolve(f, f).max())
            if no_improve % 10 == 0:
                t_step = max(t_min, t_step * 0.8)
            if no_improve > 100:
                break
        
        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={M:.6f}")
    
    return best_f, best_score

def parametric_profile(t, coeffs):
    """a[i] = exp(sum_k coeffs[k] * t^k) normalized."""
    a = np.zeros_like(t)
    for k, c in enumerate(coeffs):
        a += c * (t ** k)
    # Ensure numerical stability
    a = a - a.max()
    a = np.exp(a)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def parametric_score(coeffs, n=512):
    t = np.linspace(0, 1, n)
    a = parametric_profile(t, coeffs)
    return internal_score(a), a

def multi_phase_optimize(p0, deadline, verbose=True):
    """Multi-phase optimization: gradient descent + temperature annealing + LP polish."""
    n = len(p0)
    p = p0.copy()
    p = normalize(p)
    
    best_p = p.copy()
    best_score = internal_score(p)
    
    if verbose:
        print(f"  Start: score={best_score:.8f}")
    
    # Phase 1: Adam with moderate temperature
    T = 0.05 * float(np.convolve(p, p).max())
    phase1_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
    p, best_score = adam_phase(p, phase1_deadline, T_init=T, lr=0.05, verbose=verbose)
    best_p = p.copy()
    
    if verbose:
        print(f"  After Phase 1: score={best_score:.8f}")
    
    # Phase 2: Lower temperature
    if time.monotonic() < deadline:
        T = 0.01 * float(np.convolve(p, p).max())
        phase2_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
        p, sc = adam_phase(p, phase2_deadline, T_init=T, lr=0.03, verbose=verbose)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
    
    if verbose:
        print(f"  After Phase 2: score={best_score:.8f}")
    
    # Phase 3: LP polishing
    if time.monotonic() < deadline - 60:
        phase3_deadline = deadline - 30
        f_best, sc_best = lp_iterate(best_p, max_iters=100, deadline=phase3_deadline, verbose=verbose)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_p = f_best
    
    if verbose:
        print(f"  After Phase 3: score={best_score:.8f}")
    
    return best_p, best_score

    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    
    T = T_init
    best_p = p.copy()
    best_score = internal_score(p)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() < deadline:
        t += 1
        iter_count += 1
        
        c = np.convolve(p, p)
        M = float(c.max())
        
        # Softmax weights
        x = (c - M) / T
        ex = np.exp(x)
        sw = ex / ex.sum()
        
        # Gradient
        grad_p = compute_grad_p_fft(p, sw)
        
        # Logit gradient
        dot = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot)
        
        # Adam
        m = beta1 * m + (1 - beta1) * gz
        v = beta2 * v + (1 - beta2) * gz * gz
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        lr_t = lr / np.sqrt(1 - beta2**t)
        
        z = np.log(p)
        z -= z.mean()
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()
        
        p_new = np.exp(z - z.max())
        p_new = p_new / p_new.sum()
        
        c_new = np.convolve(p_new, p_new)
        score_new = 2.0 * n * float(c_new.max())
        
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_p = p_new.copy()
            p = p_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 1000 == 0:
                p = best_p.copy()
                m.fill(0)
                v.fill(0)
                t = 0
        
        # Temperature decay
        if t % 200 == 0:
            T = max(1e-12, T * 0.5)
        
        if verbose and iter_count % 1000 == 0:
            print(f"    Adam iter {iter_count}: score={best_score:.8f}, T={T:.2e}")
        
        if no_improve > 10000:
            break
    
    return best_p, best_score

exact subgradient methods, and LP-based polishing to minimize the peak autocorrelation
ratio 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.optimize import linprog

def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def compute_grad_p_fft(p, w):
    n = len(p)
    p_flip = p[::-1]
    conv_result = np.convolve(p_flip, w)
    grad_p = 2.0 * conv_result[n-1:2*n-1]
    return grad_p

def softmax_weights(c, T):
    c_max = c.max()
    x = (c - c_max) / T
    ex = np.exp(x - x.max())
    return ex / ex.sum()

def project_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    n = len(v)
    if v.sum() == 0:
        return np.ones(n) / n
    # Find lambda such that sum(max(v - lambda, 0)) = target_sum
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0) / np.maximum(v - theta, 0).sum()

def adam_phase(p, deadline, T_init, lr=0.05, T_decay_interval=200, T_decay_factor=0.5, 
               max_no_improve=15000, verbose=True):
    """Adam optimization in logit space with temperature annealing."""
    n = len(p)
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    
    T = T_init
    best_p = p.copy()
    best_score = internal_score(p)
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() < deadline:
        t += 1
        iter_count += 1
        
        c = np.convolve(p, p)
        M = float(c.max())
        
        sw = softmax_weights(c, T)
        grad_p = compute_grad_p_fft(p, sw)
        
        dot = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot)
        
        if not np.all(np.isfinite(gz)):
            p = best_p.copy()
            m.fill(0)
            v.fill(0)
            t = 0
            continue
        
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        m_hat = m / (1.0 - b1**t)
        v_hat = v / (1.0 - b2**t)
        
        lr_t = lr * np.sqrt(1 - b2**t) / (1 - b1**t)
        
        z = np.log(p)
        z -= z.mean()
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= z.mean()
        
        p_new = np.exp(z - z.max())
        p_new = p_new / p_new.sum()
        
        c_new = np.convolve(p_new, p_new)
        score_new = 2.0 * n * float(c_new.max())
        
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_p = p_new.copy()
            p = p_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 2000 == 0:
                p = best_p.copy()
                m.fill(0)
                v.fill(0)
                t = 0
        
        if t % T_decay_interval == 0:
            T = max(1e-14, T * T_decay_factor)
        
        if verbose and iter_count % 2000 == 0:
            print(f"      Adam iter={iter_count}, score={best_score:.10f}, T={T:.2e}, no_imp={no_improve}")
        
        if no_improve > max_no_improve:
            break
    
    return best_p, best_score

def subgradient_phase(p, deadline, n_lags=None, lr=0.001, verbose=True):
    """Exact subgradient method with line search."""
    n = len(p)
    p = normalize(p.copy())
    best_p = p.copy()
    best_score = internal_score(p)
    best_M = float(np.convolve(p, p).max())
    
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        c = np.convolve(p, p)
        M = float(c.max())
        
        # Find active lags
        if n_lags is not None:
            # Use top n_lags
            top_idx = np.argsort(c)[-n_lags:]
            active = np.zeros_like(c)
            active[top_idx] = 1.0 / n_lags
        else:
            # All lags at max
            thresh = M - 1e-10
            active = (c >= thresh).astype(float)
            if active.sum() == 0:
                active[np.argmax(c)] = 1.0
            else:
                active /= active.sum()
        
        grad_p = compute_grad_p_fft(p, active)
        
        # Try multiple step sizes
        best_step = None
        best_step_score = best_score
        
        c = np.convolve(p, p)
        M_cur = float(c.max())
        
        for lr_try in [lr, lr*0.5, lr*0.1, lr*2, lr*5]:
            p_try = p - lr_try * grad_p
            p_try = project_simplex(p_try)
            if p_try is None:
                continue
            c_try = np.convolve(p_try, p_try)
            score_try = 2.0 * n * float(c_try.max())
            if score_try < best_step_score:
                best_step_score = score_try
                best_step = p_try
        
        if best_step is not None and best_step_score < best_score - 1e-13:
            best_score = best_step_score
            best_p = best_step.copy()
            p = best_step
            no_improve = 0
            # Increase lr slightly
            lr = min(0.01, lr * 1.01)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.9
            if no_improve > 500:
                break
        
        if verbose and iter_count % 500 == 0:
            print(f"      Subgrad iter={iter_count}, score={best_score:.10f}")
    
    return best_p, best_score

def lp_direction(f, M, timeout=30.0):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_phase(f, deadline, max_iters=200, verbose=True):
    """LP-based iterative optimization."""
    n = len(f)
    f = normalize(f.astype(np.float64).copy())
    if f is None:
        return None, np.inf
    
    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0
    t_step = 0.3
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        remaining = (deadline - time.monotonic()) if deadline else 300.0
        lp_timeout = min(60.0, max(5.0, remaining / 5.0))
        
        g_raw = lp_direction(f, M, timeout=lp_timeout)
        if g_raw is None:
            break
        
        g = normalize(g_raw)
        if g is None:
            break
        
        improved = False
        for t_try in [t_step, 0.5, 0.7, 0.3, 0.15, 0.1, 0.05]:
            if t_try > 1:
                continue
            f_new = normalize((1-t_try)*f + t_try*g)
            if f_new is None:
                continue
            c_new = np.convolve(f_new, f_new)
            M_new = float(c_new.max())
            score_new = 2.0 * n * M_new
            
            if score_new < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = score_new
                f = f_new.copy()
                M = M_new
                no_improve = 0
                improved = True
                t_step = min(0.4, t_try * 1.1)
                break
        
        if not improved:
            no_improve += 1
            f = best_f.copy()
            M = float(np.convolve(f, f).max())
            if no_improve % 5 == 0:
                t_step = max(0.05, t_step * 0.85)
            if no_improve > 50:
                break
        
        if verbose and it % 20 == 0:
            print(f"      LP iter={it}, score={best_score:.10f}, t={t_step:.3f}")
    
    return best_f, best_score
