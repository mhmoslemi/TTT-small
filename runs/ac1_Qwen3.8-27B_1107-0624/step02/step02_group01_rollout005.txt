        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_max_conv(a):
    c = fft_convolve(a)
    return float(c.max())


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quintic_profile(n, a1, a3, a5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3 + a5 * u ** 5
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=10):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.03, 1.0) / (k ** 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_deterministic_profiles(n):
    profs = []
    for eta in (0.8, 1.0, 1.2, 1.5, 1.8, 2.0, 2.5, 3.0):
        profs.append(make_linear_profile(n, eta))
    for a1, a3 in ((1.0, -0.7), (1.2, -0.9), (1.5, -1.2), (1.5, -2.0),
                   (1.8, -1.5), (2.0, -2.0), (2.0, -3.0), (2.5, -3.5),
                   (0.8, 0.5), (1.5, 0.8), (2.0, -4.0)):
        profs.append(make_cubic_profile(n, a1, a3))
    profs.append(make_quintic_profile(n, 1.5, -1.0, 0.5))
    profs.append(make_quintic_profile(n, 2.0, -1.5, 0.8))
    profs.append(make_quintic_profile(n, 1.8, -2.5, -0.3))
    t = np.linspace(0, 1, n)
    a = 1.0 + 1.5 * t - 0.8 * t * t
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    a = 1.0 + 1.0 * np.sin(np.pi * t)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    a = 1.0 + 0.5 * np.cos(2 * np.pi * t) + 0.3 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    return profs


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def nesterov_optimize(f0, deadline, rng, lr=0.1, beta_nest=0.9,
                      beta_start=None, beta_end=1e-6, max_iter=200000,
                      restart_interval=500, verbose=False):
    """Nesterov-accelerated soft-max gradient descent in logit space."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()
    z_prev = z.copy()

    c = fft_convolve(f)
    M_val = float(c.max())
    if beta_start is None:
        beta_start = 0.2 * M_val
    beta = beta_start

    best_f = f.copy()
    best_M = M_val
    no_imp = 0
    N_corr = next_pow2(3 * n - 2)

    # Precompute FFT of a for reuse (will update when f changes significantly)
    fa_cached = np.fft.rfft(f, N_corr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute soft-max weights
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -200.0, 10.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        # Compute h = cross-correlation of w and a
        fw = np.fft.rfft(w, N_corr)
        h = np.fft.irfft(fw * np.conj(fa_cached), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = z_prev.copy()
            continue

        # Nesterov momentum
        z = z + beta_nest * (z - z_prev) - lr * grad_z
        z_prev = z.copy()

        # Project: keep z within bounds and re-normalize
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        # Update FFT cache
        fa_cached = np.fft.rfft(f, N_corr)

        # Compute new max convolution
        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta: decrease over time
        frac = min(1.0, it / max(1, max_iter * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac

        # Restart if stuck
        if no_imp > restart_interval:
            z = 0.8 * np.log(best_f) + 0.5 * rng.normal(size=n)
            z -= z.mean()
            z_prev = z.copy()
            f = np.exp(z - z.max())
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            fa_cached = np.fft.rfft(f, N_corr)
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(beta_start * 0.5, 0.1 * M_val)
            no_imp = 0

        # Occasionally reduce learning rate
        if it > 0 and it % 5000 == 0:
            lr *= 0.9

        if verbose and it % 5000 == 0:
            sc = 2.0 * n * best_M
            print(f"      Nest it {it}: M={best_M:.10f}, score={sc:.10f}, beta={beta:.2e}")

    return best_f, best_M


def adam_optimize(f0, deadline, rng, lr=0.05, max_iter=100000,
                  beta_start=None, beta_end=1e-6, restart_interval=300,
                  verbose=False):
    """Adam-based soft-max gradient descent in logit space."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    c = fft_convolve(f)
    M_val = float(c.max())
    if beta_start is None:
        beta_start = 0.15 * M_val
    beta = beta_start

    best_f = f.copy()
    best_M = M_val
    no_imp = 0
    N_corr = next_pow2(3 * n - 2)

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -200.0, 10.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        fa = np.fft.rfft(f, N_corr)
        fw = np.fft.rfft(w, N_corr)
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_f.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        frac = min(1.0, it / max(1, max_iter * 0.85))
        beta = beta_start * (beta_end / beta_start) ** frac

        if no_imp > restart_interval:
            z = 0.85 * np.log(best_f) + 0.4 * rng.normal(size=n)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(0.1 * M_val, 1e-8)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

        if verbose and it % 2000 == 0:
            sc = 2.0 * n * best_M
            print(f"      Adam it {it}: M={best_M:.10f}, score={sc:.10f}")

    return best_f, best_M


def solve_lp_direction(f, M, n, time_limit=60):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True,
                                               'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_refine(f0, deadline, rng, max_iter=200, verbose=False):
    """LP-based iterative refinement."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline - 10:
            break

        g = solve_lp_direction(f, M, n, time_limit=30)
        if g is None:
            no_imp += 1
            if no_imp > 20:
                break
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 50:
                break
            continue

        g = g / g_sum
        improved = False
        for t in (0.3, 0.15, 0.5, 0.1, 0.4, 0.2, 0.05, 0.6):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = compute_max_conv(f_new)
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1

        if verbose and it % 10 == 0:
            sc = 2.0 * n * best_M
            print(f"      LP it {it}: M={best_M:.10f}, score={sc:.10f}")

    return best_f, best_M


def get_parent_sequence():
    """Try to access the parent sequence from global scope."""
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                return pv / s
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Nesterov gradient descent from parent at n=512
    Phase 2: Multi-start exploration at n=512
    Phase 3: Nesterov refinement of best candidate
    Phase 4: Adam polishing with fine-tuned parameters
    Phase 5: LP refinement
    Phase 6: Try n=1024 and interpolate
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent
    parent = get_parent_sequence()

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    n1 = 512

    # Phase 1: Nesterov from parent (if available) ~200s
    phase1_end = min(deadline, t_start + 200.0)
    print(f"Phase 1: Nesterov from parent, deadline={phase1_end - t_start:.0f}s")

    if parent is not None:
        p0 = interpolate_profile(parent, n1)
        if p0 is not None:
            p_best, M_best = nesterov_optimize(
                p0, phase1_end, rng, lr=0.08, beta_nest=0.9,
                beta_start=0.15, max_iter=100000, restart_interval=200,
                verbose=True)
            sc = 2.0 * n1 * M_best
            try_update(p_best, sc)
            print(f"  Phase 1 (parent Nesterov): score={sc:.10f}")

    # Phase 2: Multi-start with diverse profiles ~250s
    phase2_end = min(deadline, t_start + 500.0)
    print(f"\nPhase 2: Multi-start, budget={phase2_end - t_start:.0f}s")

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(("parent", p_int))

    # Add diverse deterministic profiles
    dets = make_deterministic_profiles(n1)
    for i, p in enumerate(dets[:12]):
        starts.append((f"det_{i}", p))

    # Add random profiles
    for i in range(10):
        p = make_smooth_random(n1, rng, n_harm=12)
        starts.append((f"rand_{i}", p))

    per_start = max(5.0, (phase2_end - t_start) / max(1, len(starts)))
    candidates_512 = []

    for i, (name, p0) in enumerate(starts):
        if time.monotonic() > phase2_end - 5:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s

        per = min(phase2_end - 5, time.monotonic() + per_start)
        p_best, M_best = nesterov_optimize(
            p0, per, rng, lr=0.07, beta_nest=0.9,
            beta_start=0.15, max_iter=50000, restart_interval=150)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates_512.append((sc, p_best))
        if i % 5 == 0:
            print(f"  Phase 2 {name} ({i}/{len(starts)}): score={sc:.10f}, best={best_score:.10f}")

    # Also try Adam from a few starts
    for i in range(min(4, len(starts))):
        if time.monotonic() > phase2_end - 5:
            break
        name, p0 = starts[i]
        p0 = np.maximum(p0, 0.0)
        p0 = p0 / p0.sum()
        per = min(phase2_end - 5, time.monotonic() + 20.0)
        p_best, M_best = adam_optimize(
            p0, per, rng, lr=0.05, max_iter=30000,
            beta_start=0.12, restart_interval=200)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 2 Adam {name}: score={sc:.10f}")

    candidates_512.sort(key=lambda x: x[0])
    print(f"\nAfter Phase 2: best={best_score:.10f}, n candidates={len(candidates_512)}")

    # Phase 3: Extended Nesterov on best candidates ~300s
    phase3_end = min(deadline, t_start + 850.0)
    print(f"\nPhase 3: Extended refinement, budget={phase3_end - t_start:.0f}s")

    top_k = min(4, len(candidates_512))
    per_cand = max(30.0, (phase3_end - t_start) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase3_end - 10:
            break
        sc0, p0 = candidates_512[i]
        per = min(phase3_end - 10, time.monotonic() + per_cand)

        # Try Nesterov with different hyperparameters
        p_best, M_best = nesterov_optimize(
            p0, per, rng, lr=0.04, beta_nest=0.95,
            beta_start=0.08, max_iter=150000, restart_interval=400,
            verbose=(i == 0))
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)

        # Also try Adam with low lr
        if time.monotonic() < phase3_end - 20:
            per2 = min(phase3_end - 10, time.monotonic() + 40.0)
            p_best2, M_best2 = adam_optimize(
                p_best, per2, rng, lr=0.02, max_iter=50000,
                beta_start=0.05, restart_interval=500)
            sc2 = 2.0 * n1 * M_best2
            try_update(p_best2, sc2)

        print(f"  Phase 3 cand {i} (initial {sc0:.6f}): best={best_score:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: LP refinement ~200s
    phase4_end = min(deadline, t_start + 1100.0)
    print(f"\nPhase 4: LP refinement, budget={phase4_end - t_start:.0f}s")

    if best_seq is not None and best_seq.size == n1:
        per = min(phase4_end - 10, time.monotonic() + 120.0)
        p_best, M_best = lp_refine(best_seq, per, rng, max_iter=100, verbose=True)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 4 LP: score={sc:.10f}")

    # Also try LP on second best
    if len(candidates_512) > 1 and time.monotonic() < phase4_end - 20:
        _, p0 = candidates_512[1]
        per = min(phase4_end - 10, time.monotonic() + 50.0)
        p_best, M_best = lp_refine(p0, per, rng, max_iter=50)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 4 LP (2nd): score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # Phase 5: Final Nesterov polish with very small steps ~150s
    phase5_end = min(deadline, t_start + 1300.0)
    print(f"\nPhase 5: Final polish, budget={phase5_end - t_start:.0f}s")

    if best_seq is not None and best_seq.size == n1:
        per = min(phase5_end - 5, time.monotonic() + 80.0)
        p_best, M_best = nesterov_optimize(
            best_seq, per, rng, lr=0.01, beta_nest=0.99,
            beta_start=0.03, max_iter=200000, restart_interval=1000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 5 Nesterov (fine): score={sc:.10f}")

        per2 = min(phase5_end - 5, time.monotonic() + 40.0)
        p_best2, M_best2 = adam_optimize(
            best_seq, per2, rng, lr=0.005, max_iter=50000,
            beta_start=0.02, restart_interval=2000)
        sc2 = 2.0 * n1 * M_best2
        try_update(p_best2, sc2)
        print(f"  Phase 5 Adam (fine): score={sc2:.10f}")

    print(f"After Phase 5: best={best_score:.10f}")

    # Phase 6: Try n=1024
    n2 = 1024
    phase6_end = min(deadline, t_start + 1500.0)
    print(f"\nPhase 6: Try n={n2}, budget={phase6_end - t_start:.0f}s")

    if best_seq is not None and time.monotonic() < phase6_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase6_end - 30, time.monotonic() + 80.0)
            p_best, M_best = nesterov_optimize(
                p_ext, per, rng, lr=0.05, beta_nest=0.9,
                beta_start=0.1, max_iter=80000, restart_interval=300,
                verbose=True)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"  Phase 6 Nesterov (n={n2}): score={sc:.10f}")

            # Polish
            if time.monotonic() < phase6_end - 20:
                per2 = min(phase6_end - 10, time.monotonic() + 40.0)
                p_best2, M_best2 = adam_optimize(
                    p_best, per2, rng, lr=0.02, max_iter=30000,
                    beta_start=0.05, restart_interval=500)
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"  Phase 6 Adam (n={n2}): score={sc2:.10f}")

    # Also try n=256 (sometimes shorter is better)
    n0 = 256
    if time.monotonic() < deadline - 60:
        print(f"\nPhase 6b: Try n={n0}")
        per = min(deadline - 30, time.monotonic() + 40.0)
        if best_seq is not None:
            p_short = interpolate_profile(best_seq, n0)
            if p_short is not None:
                p_best, M_best = nesterov_optimize(
                    p_short, per, rng, lr=0.08, beta_nest=0.9,
                    beta_start=0.2, max_iter=50000, restart_interval=200)
                sc = 2.0 * n0 * M_best
                try_update(p_best, sc)
                print(f"  Phase 6b (n={n0}): score={sc:.10f}")

    # Final LP if time permits
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        if n_best <= 1024:
            per = min(deadline - 10, time.monotonic() + 60.0)
            p_best, M_best = lp_refine(best_seq, per, rng, max_iter=50, verbose=True)
            sc = compute_score(p_best)
            try_update(p_best, sc)
            print(f"\nFinal LP (n={n_best}): score={sc:.10f}")

    # Final verification
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"\nFinal internal score: {sc_final:.10f}, n={final.size}")

        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    print(f"Fallback: n={fb.size}")
    return [float(x) for x in fb]
