```python
    """
    Solve LP: maximize sum(b_half) s.t. conv(a, b)_k <= M for k=0,...,n-1, b_half >= 0.
    where a and b are symmetric sequences of length n determined by their first n/2 elements.

    # a_full is determined by a_half: a_full[i] = a_half[i] for i < m, a_full[i] = a_half[n-1-i] for i >= m

    # For each k, conv(a_full, b_full)_k = sum_j a_full[j] * b_full[k-j]
    # b_full[j] = b_half[j] if j < m, else b_half[n-1-j]

            if idx < m:
                col = idx
            else:

    c_obj = -np.ones(m)

    return None
```

```python
"""
Symmetric LP-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key innovation: Exploit the symmetry of autocorrelation to halve the optimization
variables. The LP-based spreading method operates in the reduced space, enabling
faster convergence and exploration of longer sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_fn(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _symmetrize(a):
    """Make a sequence symmetric: a[i] = a[n-1-i]."""
    n = a.size
    return 0.5 * (a + a[::-1])


def _expand_symmetric(a_half, n):
    """Expand a half-sequence to a full symmetric sequence of length n."""
    m = n // 2
    a_full = np.empty(n)
    a_full[:m] = a_half
    a_full[m:] = a_half[::-1]
    return a_full


def _build_lp_matrix(a_half, n):
    """Build the LP constraint matrix for the symmetric problem.
    
    Returns A (n x m) such that A @ b_half = conv(a_full, b_full)[:n].
    """
    m = n // 2
    a_full = _expand_symmetric(a_half, n)
    A = np.zeros((n, m))
    for k in range(n):
        for col in range(m):
            j1 = k - col
            j2 = k - n + 1 + col
            if 0 <= j1 < n:
                A[k, col] += a_full[j1]
            if 0 <= j2 < n:
                A[k, col] += a_full[j2]
    return A


def _symmetric_lp(a_half, n, M):
    """Solve LP: maximize sum(b_half) s.t. A @ b_half <= M, b_half >= 0."""
    m = n // 2

def _smooth_max_and_weights(c, beta):
    """Compute smooth max value and softmax weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w
```

```python
def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    # Scale to target_sum = 1, project, then scale back
    v_scaled = v / target_sum
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v_scaled - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    return (w / s) * target_sum
```

```python
"""
Symmetric LP-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovation: exploit autocorrelation symmetry to halve optimization variables,
enabling much faster LP iterations and exploration of longer sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_fn(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _symmetrize(a):
    n = a.size
    return 0.5 * (a + a[::-1])


def _expand_symmetric(a_half, n):
    m = n // 2
    a_full = np.empty(n)
    a_full[:m] = a_half
    a_full[m:] = a_half[::-1]
    return a_full


def _build_lp_matrix(a_half, n):
    m = n // 2
    a_full = _expand_symmetric(a_half, n)
    k_idx = np.arange(n)[:, None]
    col_idx = np.arange(m)[None, :]
    j1 = k_idx - col_idx
    j2 = k_idx - n + 1 + col_idx
    A = np.zeros((n, m))
    mask1 = (j1 >= 0) & (j1 < n)
    mask2 = (j2 >= 0) & (j2 < n)
    A[mask1] += a_full[j1[mask1]]
    A[mask2] += a_full[j2[mask2]]
    return A


def _symmetric_lp(a_half, n, M):
    m = n // 2
    A = _build_lp_matrix(a_half, n)
    c_obj = -np.ones(m)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(n, M),
                  bounds=[(0, None)] * m, method='highs')
    if res.success:
        return res.x
    return None


def _project_simplex(v, target_sum=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    if target_sum <= 0:
        return np.zeros(n)
    v_scaled = v / target_sum
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v_scaled - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    return (w / s) * target_sum


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _grad_sym(a_full, n, beta):
    m = n // 2
    c = _fft_convolve(a_full)
    c_first = c[:n]
    w = _smooth_max_weights(c_first, beta)
    w_rev = w[::-1]
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a_full, N)
    fw = np.fft.rfft(w_rev, N)
    conv_res = np.fft.irfft(fa * fw, N)[:2 * n - 1]
    grad_full = 2.0 * conv_res[n - 1::-1]
    grad_half = np.zeros(m)
    for i in range(m):
        j = n - 1 - i
        if i == j:
            grad_half[i] = grad_full[i]
        else:
            grad_half[i] = grad_full[i] + grad_full[j]
    return grad_half, float(c_first.max())

    best_score = float(2.0 * n * best_M)

    inflate = 1.003
    no_imp = 0
    base_blend = blend
    base_inflate = inflate

    it = 0
    while time.monotonic() < deadline and it < max_iter:

        if it % 5000 == 0:

        b_half = _symmetric_lp(a_half, n, M_lp)

        if b_half is None:
            break

        b_full = _expand_symmetric(b_half, n)
        s_b = float(b_full.sum())
        if s_b < 1e-10 or s_b <= 1.0 + 1e-8:

def _interpolate_sym(parent, n):
    p = np.asarray(parent, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n:
        a = p.copy()
    else:
        x0 = np.linspace(0.0, 1.0, n0)
        x1 = np.linspace(0.0, 1.0, n)
        a = np.interp(x1, x0, p)
    a = _symmetrize(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = _default_init(n)
    return a


def _default_init(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 1.5 * (t - 0.5)
    a = np.maximum(a, 1e-12)
    a = _symmetrize(a)
    a = a / a.sum()
    return a
```
