        max_val = conv.max()
        max_idx = np.where(conv == max_val)[0]

        grad = np.zeros(n)
        for k in max_idx:

            for i in range(n):

                if 0 <= j < n:
                    grad[i] += s[j]

        alpha = 1.0 / (10 + 0.1 * it)
        
        # Gradient step
        s_new = s - alpha * grad
        
        # Project onto simplex
        s_new = proj_simplex(s_new)
        
        s = s_new
    
    return s
