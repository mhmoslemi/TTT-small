def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.

    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

def next_pow2(n):

def full_convolve_fft(a):

def param_adam_search(theta0, basis, deadline, rng, lr=0.3):

    theta = np.asarray(theta0, dtype=np.float64).ravel().copy()
    dim = theta.size
    n = basis.shape[0]

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(dim, dtype=np.float64)
    v = np.zeros(dim, dtype=np.float64)
    adam_t = 0

    p = profile_from_basis(theta, basis)
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.25 * maxc)

    best_theta = theta.copy()
    best_p = p.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gtheta = basis.T @ gz

        if not np.all(np.isfinite(gtheta)):
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gtheta
        v = b2 * v + (1.0 - b2) * (gtheta * gtheta)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        theta = theta - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(theta)):
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

def coordinate_descent_polish(p, deadline, rng, n_sweeps=None, step_frac=0.1):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    if n_sweeps is None:
        n_sweeps = max(3, int((deadline - time.monotonic()) * 2))

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved_this_sweep = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            candidates_hi = [1.0 + step_frac, 1.0 + 0.5 * step_frac, 1.0 + 2.0 * step_frac]
            candidates_lo = [1.0 - step_frac, 1.0 - 0.5 * step_frac, 1.0 - 0.3 * step_frac]
            best_val = orig
            best_sc = best_score
            for factor in candidates_hi + candidates_lo:
                new_val = orig * factor
                if new_val < 1e-15 or new_val > 1000.0:
                    continue
                p[i] = new_val
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                p[i] /= (s - orig + new_val)
                sc = internal_score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                improved_this_sweep = True
                no_imp = 0
            else:
                no_imp += 1
        if improved_this_sweep:
            step_frac = max(0.02, step_frac * 0.9)
        if no_imp > 2 * n:
            break

    return best_p, best_score
