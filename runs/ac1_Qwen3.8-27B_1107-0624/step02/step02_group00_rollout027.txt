def _conv_gradient(a, w):
    """
    Compute gradient of sum_j w_j * c_j where c = conv(a, a).
    grad_a[i] = 2 * (conv(a, w_rev))[2n-2-i] for i = 0,...,n-1

    """

    w_rev = w[::-1]

    N = _next_pow2(3 * n - 2)

    grad_a = _conv_gradient(a, w)
    
    # Gradient w.r.t. z (chain rule through softmax)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

                tag += f" ** {sc_opt:.10f} **"
            else:
                tag += f"  {sc_opt:.10f}"
            print(tag)

    # Final Adam polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final Adam polish (n={best_seq.size}) ===")
        z_best = _make_z_from_a(best_seq)
        a_final, z_final, sc_final = _adam_polish(
            z_best, budget_end=deadline - 10, lr=0.03, n_T=8, max_iter=30000)
        print(f"  Adam: {sc_final:.10f}")
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(256) / 256.0

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    sc_final_eval = _score(final_seq)
    print(f"\nFINAL: score={sc_final_eval:.10f}, n={final_seq.size}")

    try:
        sc_auth = evaluate_sequence(final_seq.tolist())
        print(f"  evaluate_sequence: {sc_auth:.10f}")
        if np.isfinite(sc_auth) and sc_auth < best_score:
            best_score = sc_auth
    except Exception:
        pass

    return [float(x) for x in final_seq]
