import time
import cvxpy as cp
from scipy.optimize import linprog

def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def cvxpy_solve(n, timeout=60):
    """Solve the convex problem directly using CVXPY."""
    a = cp.Variable(n)
    t = cp.Variable()
    
    # Build the objective: minimize t
    objective = cp.Minimize(t)
    
    constraints = [cp.sum(a) == 1, a >= 0]
    
    # Add quadratic constraints: conv(a,a)[m] <= t for m = 0,...,n-1
    for m in range(n):
        # conv(a,a)[m] = sum_j a[j]*a[m-j] where 0<=j<n and 0<=m-j<n
        # This is a quadratic form
        Q = np.zeros((n, n))
        for j in range(max(0, m-(n-1)), min(n-1, m)+1):
            k = m - j
            if 0 <= k < n:
                Q[j, k] = 1.0
        # Make symmetric
        Q = (Q + Q.T) / 2
        # The quadratic form a^T Q a = sum_{j,k} Q[j,k]*a[j]*a[k]
        # = sum_j a[j]^2 * Q[j,j] + sum_{j<k, Q[j,k]>0} 2*Q[j,k]*a[j]*a[k]
        # But we want sum_j a[j]*a[m-j]
        # Let's verify: a^T Q a = sum_{j,k} Q[j,k] a[j] a[k]
        # For m=0: Q[j,j] = 1 for all j, so a^T Q a = sum_j a[j]^2 = conv[0] ✓
        # For m=1: Q[j,j+1] = Q[j+1,j] = 1/2, so a^T Q a = sum_j Q[j,j+1]*2*a[j]*a[j+1] = sum_j a[j]*a[j+1] = conv[1] ✓
        constraints.append(cp.quad_form(a, Q) <= t)
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        # or: prob.solve(solver=cp.ECOS)
        if a.value is not None:
            solution = np.maximum(a.value, 0)
            solution = solution / solution.sum()
            return solution
    except Exception as e:
        print(f"CVXPY MOSEK failed: {e}")
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if a.value is not None:
                solution = np.maximum(a.value, 0)
                solution = solution / solution.sum()
                return solution
        except Exception as e2:
            print(f"CVXPY SCS failed: {e2}")
    
    return None

def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def build_conv_quadratic_matrix(n, m):
    """Build n×n PSD matrix Q such that a^T Q a = conv(a,a)[m]."""
    Q = np.zeros((n, n))
    for j in range(max(0, m - (n - 1)), min(n - 1, m) + 1):
        k = m - j
        if 0 <= k < n:
            if j == k:
                Q[j, j] += 1.0
            else:
                Q[j, k] += 0.5
                Q[k, j] += 0.5
    return Q


def cvxpy_solve(n, time_budget=120):
    """Solve min t s.t. conv(a,a)[m] <= t, sum(a)=1, a>=0 using CVXPY."""
    t0 = time.monotonic()

    t_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1, a_var >= 0]

    # Only need m = 0, ..., n-1 by symmetry
    for m in range(n):
        Q = build_conv_quadratic_matrix(n, m)
        constraints.append(cp.quad_form(a_var, Q) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.SCS, cp.ECOS, cp.CLARABEL]

def lp_direction(f, M, timeout=15.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M for all x.
    """
    n = len(f)
    # Build conv matrix: (2n-1) x n where A[x, j] = f[x-j] if valid
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def interpolate_seq(a, n_new):
    """Interpolate sequence to new length, maintaining non-negativity and normalization."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a_new
    return a_new / s


def make_profile(n, eta=0.0, gamma=0.0):

    u = 2.0 * t - 1.0
    a = 1.0 + eta * u + gamma * u * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    print(f"After Phase 4: best_score={best_score:.8f}")

    # --- Phase 5: LP polish ---
    deadline_p5 = deadline
    if time.monotonic() < deadline_p5 - 30:
        print(f"\nPhase 5: LP polish at n={len(best_seq)}")
        lp_deadline = deadline_p5 - 20
        best_seq, best_score = lp_refine(best_seq, lp_deadline, max_iters=100, verbose=True)
        print(f"After Phase 5: best_score={best_score:.8f}")

    # --- Final validation ---
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")
    if final_score < best_score:
        best_score = final_score

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_profile(512, 1.0)
        s = best_seq.sum()
    best_seq = best_seq / s

    # Final check with evaluate_sequence
    try:
        eval_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"evaluate_sequence score: {eval_score:.8f}")
        if eval_score != final_score:
            print(f"  Warning: mismatch! internal={final_score:.8f}, eval={eval_score:.8f}")
    except Exception as e:
        print(f"  evaluate_sequence check failed: {e}")

    return [float(x) for x in best_seq]
