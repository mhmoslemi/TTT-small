def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):

    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _make_cosine_profile(n, freq=1, amplitude=1.0, phase=0.0):
    """Create a cosine-modulated profile: max(0, 1 + amplitude * cos(2*pi*freq*i/n + phase))"""
    i = np.arange(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()

def _smooth_max_and_grad(a, beta):
    """Compute smooth max of convolution and its gradient."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip for numerical stability
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        val = mval
    else:
        w = ex / s
        val = mval + beta * float(np.log(s))
    
    # Gradient: d/d a_i [sum_k w_k * (a*a)_k] = 2 * sum_k w_k * a_{k-i}
    # = 2 * (w * a_reversed) circular... actually it's:
    # grad_i = 2 * sum_{k} w_k * a_{k-i} (where a_{k-i} is 0 if k-i out of range)
    # This is 2 * correlation of w with a, which is 2 * (w_star * a) where star is reversed
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return val, grad, c, mval

    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    else:
        beta = beta_init

    for it in range(max_iter):

        c = _fft_convolve(a)
        mval = float(c.max())

        x = np.clip(x, -50.0, 50.0)

        if not np.isfinite(s_ex) or s_ex <= 0:

        # Move mass from high peak_load to low peak_load (among trough positions)
        # Score each position: want to decrease where peak_load high, increase where peak_load low

The problem is: minimize $\max_k (a*a)_k$ subject to $\sum a_i = 1, a_i \geq 0$.

By the max-min theorem / LP duality, this is equivalent to:
$\max_k (a*a)_k = \max_{p \in \Delta_{2n-1}} p^T (a*a) = \max_{p \in \Delta_{2n-1}} (\sum_k p_k \sum_i a_i a_{k-i})$

$= \max_{p \in \Delta_{2n-1}} \sum_i \sum_j a_i a_j p_{i+j}$

$= \max_{p \in \Delta_{2n-1}} a^T Q(p) a$

where $Q(p)_{ij} = p_{i+j}$ (a symmetric matrix).

So the problem is: $\min_a \max_p a^T Q(p) a$ subject to $a \in \Delta_n$.

This is a zero-sum game! The "row player" chooses $a$ and the "column player" chooses $p$.

By the minimax theorem (since the feasible sets are convex and compact, and the payoff is bilinear... wait, it's quadratic in $a$, not bilinear).

Actually, it's $\min_a \max_p a^T Q(p) a$ which is not a standard zero-sum game because the payoff is quadratic in $a$.

But we can use this perspective: at the optimum, there exists a $p^*$ (a distribution over the convolution indices) such that $a$ minimizes $a^T Q(p^*) a$ over the simplex. This gives us a way to check optimality and also a way to iterate:
1. Given $a$, find $p^* = \arg\max_p a^T Q(p) a$ (which is concentrated on the argmax of $(a*a)_k$)
2. Given $p^*$, find $a^* = \arg\min_a a^T Q(p^*) a$ over the simplex

Step 2 is a quadratic programming problem! We want to minimize $a^T Q a$ subject to $\sum a_i = 1, a_i \geq 0$.

This is a QCQP (quadratic constraint/quadratic program). If $Q$ is positive semidefinite, it's convex. But $Q(p)$ with $p$ on the simplex is NOT necessarily PSD.

However, we can still use this as an iteration:
1. Compute $p^*$ (concentrated on peaks of $a*a$)
2. Solve the QP: $\min a^T Q(p^*) a$ s.t. $a \geq 0, \sum a_i = 1$
3. Update $a \leftarrow (1-t)a + t a^*$

This is essentially a **Frank-Wolfe / Mirror Descent** type iteration!

Actually, this is very similar to what the LP approach does, but using a QP instead. The QP approach might be more effective because it captures the quadratic nature of the problem.

But solving a QP at each iteration might be expensive. Let me think about whether we can solve it efficiently.

The QP is: $\min a^T Q a$ s.t. $\sum a_i = 1, a_i \geq 0$ where $Q_{ij} = p_{i+j}$.

Note that $Q$ is a Hankel matrix (constant along anti-diagonals). This special structure might allow efficient solving.

Actually, for the iteration to work, I don't need to solve the QP exactly. I can use a few steps of projected gradient on the QP, or use the KKT conditions.

The KKT conditions for $\min a^T Q a$ s.t. $a \geq 0, \mathbf{1}^T a = 1$ are:
- $2Qa + \mathbf{1}\lambda - \mu = 0$ where $\mu \geq 0$ is the dual variable for $a \geq 0$
- $\mu_i a_i = 0$ (complementary slackness)
- $\mathbf{1}^T a = 1$

At the solution, for active variables ($a_i > 0$): $2(Qa)_i + \lambda = 0$, so $(Qa)_i = -\lambda/2$ (constant).
For inactive variables ($a_i = 0$): $2(Qa)_i + \lambda \geq 0$.

So the solution has the property that $(Qa)_i$ is constant on the support of $a$.

This gives us a "water-filling" type algorithm:
1. Start with some support set $S$
2. On $S$, solve $(Q_S a_S) = -\lambda/2 \cdot \mathbf{1}$ with $\sum_{i \in S} a_i = 1$
3. Check if for $i \notin S$: $(Qa)_i \geq -\lambda/2$
4. If not, add the violating $i$ to $S$ and go to 2

This is similar to the active set method for QPs.

def _qp_step(a, p, lambda_init=None):
    """
    Solve min a^T Q a s.t. sum(a)=1, a>=0 where Q_ij = p_{i+j}
    using active set / water-filling.
    """
    n = len(a)
    # Q is Hankel: Q_ij = p_{i+j}
    
    # Start with full support
    S = set(range(n))
    
    for _ in range(50):  # max iterations
        S_list = sorted(S)
        k = len(S_list)
        
        # Build Q_S (k x k matrix)
        QS = np.zeros((k, k))
        for i, si in enumerate(S_list):
            for j, sj in enumerate(S_list):
                QS[i, j] = p[si + sj]
        
        # Solve QS * a_S = -lambda/2 * ones, sum(a_S) = 1
        # This is a (k+1) x (k+1) system:
        # [QS   1] [a_S]   [0]
        # [1^T  0] [lam] = [1]
        
        A = np.zeros((k+1, k+1))
        A[:k, :k] = QS
        A[:k, k] = 1.0
        A[k, :k] = 1.0
        b = np.zeros(k+1)
        b[k] = 1.0
        
        try:
            sol = np.linalg.solve(A, b)
        except np.linalg.LinAlgError:
            # Use least squares
            sol, _, _, _ = np.linalg.lstsq(A, b, rcond=None)
        
        a_S = sol[:k]
        lam = sol[k]
        
        # Check non-negativity
        if np.all(a_S >= -1e-12):
            a_S = np.maximum(a_S, 0)
            a_S = a_S / a_S.sum()
            
            # Build full a
            a_full = np.zeros(n)
            for i, si in enumerate(S_list):
                a_full[si] = a_S[i]
            
            # Check KKT: for i not in S, (Qa)_i + lambda/2 >= 0
            Qa = np.zeros(n)
            for i in range(n):
                Qa[i] = 0.0
                for j in S_list:
                    Qa[i] += p[i + j] * a_full[j]
            
            inactive_OK = True
            for i in range(n):
                if i not in S:
                    if Qa[i] + lam / 2 < -1e-10:
                        inactive_OK = False
                        S.add(i)
                        break
            
            if inactive_OK:
                return a_full
        
        # If some a_S[i] < 0, remove that index from S
        min_idx = int(np.argmin(a_S))
        if a_S[min_idx] < -1e-12:
            S.discard(S_list[min_idx])
        else:
            # Numerical issue, just return
            a_full = np.zeros(n)
            a_S = np.maximum(a_S, 0)
            a_S = a_S / a_S.sum()
            for i, si in enumerate(S_list):
                a_full[si] = a_S[i]
            return a_full
    
    # Fallback: return uniform on S
    a_full = np.zeros(n)
    for i in S:
        a_full[i] = 1.0 / len(S)
    return a_full

"""
Optimized smooth-max gradient descent with diverse initializations
for minimizing 2n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
import math

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _make_profile_type1(n, freq, amp):
    """Cosine modulated: 1 + amp * cos(2*pi*freq*i/n)"""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_profile_type2(n, freq1, amp1, freq2, amp2):
    """Two-frequency cosine modulation."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp1 * np.cos(2.0 * np.pi * freq1 * i / n) + amp2 * np.cos(2.0 * np.pi * freq2 * i / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_profile_type3(n, p=2):
    """Bounded power: (i+1)^(-alpha) type, or periodic sawtooth."""
    i = np.arange(n, dtype=np.float64)
    # Periodic triangular wave
    t = 2.0 * ((i / n) - np.floor(i / n + 0.5))  # triangle in [-1, 1]
    a = 1.0 + 0.8 * np.abs(t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _adam_polish(a, deadline, lr=0.03, beta_decay=0.85, beta_floor=1e-10):
    """
    Adam smooth-max polish in logit space.
    Returns (best_a, best_score).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    mval = float(c.max())
    beta = 0.03 * mval  # initial smoothing
    best_score = float(2.0 * n * mval)
    best_a = a.copy()

    z = np.log(a)
    z -= float(np.mean(z))
    best_z = z.copy()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            w = np.zeros(2*n-1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        # grad in a-space: 2 * (w correlated with a)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_r = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_r, N)[n-1:2*n-1]
        ga = 2.0 * h + 1e-10 * a

        # grad in z-space
        dot = float(np.dot(a, ga))
        gz = a * (ga - dot)
        gz -= np.mean(gz)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m[:] = 0; v[:] = 0; t = 0
            continue

        t += 1
        m = b1 * m + (1-b1) * gz
        v = b2 * v + (1-b2) * gz**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -35.0, 35.0)
        z -= np.mean(z)
        
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(1500, 2*n):
                # Restart with perturbation
                z = best_z + 0.2 * np.random.default_rng(no_imp).standard_normal(n)
                z = np.clip(z, -35.0, 35.0)
                z -= np.mean(z)

                a = e / e.sum()
                c = _fft_convolve(a)
                beta = 0.03 * float(c.max())
                m[:] = 0; v[:] = 0; t = 0
                lr = base_lr
                no_imp = 0

        # Decay beta
        if t % 200 == 0:
            beta = max(beta_floor, beta * beta_decay)

    return best_a, best_score

"""
Multi-frequency diverse initialization + aggressive Adam smooth-max optimization
with adaptive restarts and size exploration for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key improvements over previous approach:
1. Much more diverse initializations (cosine-modulated, multi-frequency, random)

3. Smarter restart strategy using best-so-far + perturbation
4. Size exploration (256-1024)

6. Peak-directed gradient weighting

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_cosine(n, freq, amp, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multi_cosine(n, freqs, amps):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for f, amp in zip(freqs, amps):
        a += amp * np.cos(2.0 * np.pi * f * i / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, width_frac=0.3):
    """Mass concentrated at two locations."""
    i = np.arange(n, dtype=np.float64)
    t = i / max(n - 1, 1)
    center1 = 0.15
    center2 = 0.85
    sigma = width_frac * 0.3
    a = np.exp(-((t - center1) / sigma) ** 2) + np.exp(-((t - center2) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quadruplet(n, sigma=0.15):
    """Mass at four locations."""
    i = np.arange(n, dtype=np.float64)
    t = i / max(n - 1, 1)
    centers = [0.1, 0.35, 0.65, 0.9]
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-((t - c) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_power_law(n, alpha=0.5):
    """Decaying power law from edges."""
    i = np.arange(n, dtype=np.float64)
    t = np.minimum(i, n - 1 - i).astype(np.float64) + 1
    a = t ** (-alpha)
    return a / a.sum()


def _adam_smooth_max(a, deadline, lr=0.03, beta_start_ratio=0.03,
                     beta_decay=0.82, beta_decay_every=150,
                     beta_floor=1e-10, restart_perturb=0.25,
                     max_restart_noimp=3000, seed=0):
    """
    Adam smooth-max optimization in logit space.
    Two phases: coarse (large beta) then fine (small beta).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = _fft_convolve(a)
    mval = float(c.max())
    best_score = float(2.0 * n * mval)
    best_a = a.copy()

    z = np.log(a)
    z -= float(np.mean(z))
    best_z = z.copy()

    beta = beta_start_ratio * mval

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_imp = 0
    base_lr = lr
    rng = np.random.default_rng(seed + n)

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        # Gradient in a-space: 2 * correlation(w, reversed_a)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_r = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_r, N)[n - 1:2 * n - 1]
        ga = 2.0 * h + 1e-10 * a

        # Convert to z-space
        dot = float(np.dot(a, ga))
        gz = a * (ga - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m[:] = 0
            v[:] = 0
            t = 0
            beta = max(beta_floor, 0.03 * float(_fft_convolve(best_a).max()))
            no_imp = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * gz * gz
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m[:] = 0
            v[:] = 0
            t = 0
            continue

        z = np.clip(z, -35.0, 35.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max_restart_noimp:
                # Restart: perturb around best
                perturb = rng.standard_normal(n) * restart_perturb
                z = best_z + perturb
                z = np.clip(z, -35.0, 35.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                c2 = _fft_convolve(a)
                beta = 0.05 * float(c2.max())
                m[:] = 0
                v[:] = 0
                t = 0
                lr = base_lr
                no_imp = 0

        # Decay beta
        if t % beta_decay_every == 0 and t > 0:
            beta = max(beta_floor, beta * beta_decay)

    return best_a, best_score


def _refine_peaks(a, deadline, max_iter=20000):
    """
    Direct peak-refinement: identify which elements contribute most to the peak
    and redistribute their mass to reduce the peak.
    Uses incremental convolution updates.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    no_imp = 0
    step = 0.002 / math.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        peak_idx = int(np.argmax(c))
        peak_val = float(c[peak_idx])

        # Influence of position i on peak: a[peak_idx - i] if valid
        influence = np.zeros(n)
        valid = (peak_idx < n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                influence[i] = a[j]

        # Find high-influence source and low-influence target
        src_candidates = np.where((a > step * 2) & (influence > 0.3 * influence.max()))[0]
        tgt_candidates = np.where(influence < 0.3 * influence.min() + 1e-15)[0]
        if len(tgt_candidates) == 0:
            tgt_candidates = np.where(influence < influence.mean())[0]

        if len(src_candidates) > 0 and len(tgt_candidates) > 0:
            # Pick the source with highest a[i]*influence[i]
            src_scores = a[src_candidates] * influence[src_candidates]
            src = src_candidates[int(np.argmax(src_scores))]

            # Pick target with lowest influence
            tgt = tgt_candidates[int(np.argmin(influence[tgt_candidates]))]

            # Try the move
            a_new = a.copy()
            a_new[src] -= step
            a_new[tgt] += step
            if a_new[src] < 0:
                step *= 0.9
                continue

            # Update convolution incrementally
            # Delta c_k = 2*step*(a[tgt-k] - a[src-k]) for valid k
            delta = np.zeros(2 * n - 1)
            # Contribution from moving -step from src: affects c[src+j] by -step*a[j]
            for j in range(n):
                delta[src + j] -= step * a[j]
            # Contribution from moving +step to tgt: affects c[tgt+j] by +step*a[j]
            for j in range(n):
                delta[tgt + j] += step * a[j]

            c_candidate = c + delta
            new_peak = float(c_candidate.max())
            new_score = 2.0 * n * new_peak

            if new_score < best_score - 1e-15:
                a = a_new
                c = c_candidate
                best_score = new_score
                best_a = a.copy()
                no_imp = 0
                step = min(step * 1.02, 0.01 / math.sqrt(n))
            else:
                no_imp += 1
                step = max(step * 0.95, 0.0001 / math.sqrt(n))
        else:
            no_imp += 1

        if no_imp > 5000:
            break

    return best_a, best_score

        final_seq = final_seq / s
    else:
        final_seq = _make_cosine(512, 2, 1.0)

    # Verify with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        print(f"\nevaluate_sequence score: {sc_eval:.10f}")
        if np.isfinite(sc_eval):
            best_score = min(best_score, sc_eval)
    except Exception as e:
        print(f"  Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
