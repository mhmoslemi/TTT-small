import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N <<= 1
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]


def compute_score(f):
    """Compute 2*n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    s = float(np.sum(f))
    if s < 1e-30:
        return np.inf
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    conv = np.fft.irfft(F * F, N)[:2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def compute_conv_max(f):
    """Compute max of linear auto-convolution."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    conv = np.fft.irfft(F * F, N)[:2 * n - 1]
    return float(conv.max())


def soft_max_and_weights(conv, beta):
    """Compute soft-max value and probability weights."""
    m = float(conv.max())
    x = (conv - m) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    w = ex / s
    return m + beta * np.log(s), w


def gradient_from_weights(w, f):
    """Compute 2*sum_k w[k]*f[k-i] for all i via FFT cross-correlation."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.fft(w, N)
    ff = np.fft.fft(f, N)
    h = np.fft.ifft(fw * np.conj(ff)).real
    return 2.0 * h[:n]


def normalize(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def make_linear_profile(n, slope=2.0):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cosine_profile(n, power=1.0):
    t = np.linspace(0.0, 1.0, n)
    f = np.cos(np.pi * t) ** (2 * power) + 0.1
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_dual_bump(n):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * (t / 0.15) ** 2) + np.exp(-0.5 * ((t - 1) / 0.15) ** 2) + 0.1
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def spectral_init(n, rng, num_freq=None):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    if num_freq is None:
        num_freq = n // 2
    # Random phases in frequency domain
    phases = rng.uniform(0, 2 * np.pi, num_freq)
    # Power spectrum: flat for non-DC, with DC component set by normalization
    power = np.ones(num_freq)
    power[0] = n  # DC component corresponds to sum
    # Complex spectrum
    spectrum = np.zeros(n, dtype=np.complex128)
    spectrum[:num_freq] = np.sqrt(power[:num_freq]) * np.exp(1j * phases)
    # Ensure Hermitian symmetry
    for k in range(1, n - 1):
        if k < num_freq:
            spectrum[n - k] = np.conj(spectrum[k])
    if n % 2 == 0:
        spectrum[n // 2] = np.abs(spectrum[n // 2])
    # Inverse FFT
    f = np.fft.ifft(spectrum).real
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return None
    return f / s


def smooth_random_init(n, rng, num_modes=8, amp=0.4):
    """Random smooth profile using Fourier modes."""
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    for k in range(1, num_modes + 1):
        coef_c = rng.uniform(-amp, amp)
        coef_s = rng.uniform(-amp, amp)
        f = f + coef_c * np.cos(2 * np.pi * k * t) + coef_s * np.sin(2 * np.pi * k * t)
    f = np.maximum(f, 1e-8)
    return f / f.sum()


def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def adam_optimize(f_init, deadline, rng, max_iters=20000, base_lr=0.05,
                  beta_start=0.5, beta_end=0.01, lookahead_period=50,
                  lookahead_alpha=0.5, restart_every=500):
    """
    Adam optimization in logit space with lookahead and adaptive soft-max.
    Minimizes max(conv(f,f)) subject to f >= 0, sum(f) = 1.
    """
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    s = float(f.sum())
    f = f / s

    # Logit parameterization
    z = np.log(f + 1e-30)
    z -= z.mean()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    # Lookahead state
    z_slow = z.copy()
    la_count = 0

    # Current parameters
    beta = beta_start
    lr = base_lr
    total_iters = max(1, max_iters)

    # Compute initial conv and score
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    conv = np.fft.irfft(F * F, N)[:2 * n - 1]
    M = float(conv.max())

    best_M = M
    best_f = f.copy()
    best_z = z.copy()
    no_improve = 0
    restart_count = 0
    max_no_improve = 300

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive beta: geometric decay
        progress = it / total_iters
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.5)
        beta = max(beta, 1e-6)

        # Learning rate: cosine annealing
        phase = (it % restart_every) / restart_every
        lr = base_lr * 0.5 * (1.0 + np.cos(np.pi * phase))

        # Compute soft-max and gradient
        sm_val, w = soft_max_and_weights(conv, beta)
        g_f = gradient_from_weights(w, f)

        # Convert to logit gradient: dL/dz[i] = f[i]*(g_f[i] - sum(f*g_f))
        dot = float(np.dot(f, g_f))
        g_z = f * (g_f - dot)
        g_z -= g_z.mean()

        # Check for numerical issues
        if not np.all(np.isfinite(g_z)):
            z = best_z.copy()
            z_slow = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        m = b1 * m + (1 - b1) * g_z
        v = b2 * v + (1 - b2) * (g_z * g_z)
        m_hat = m / (1 - b1 ** t_adam)
        v_hat = v / (1 - b2 ** t_adam)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Clip for numerical stability
        z = np.clip(z, -50, 50)
        z -= z.mean()

        # Lookahead
        la_count += 1
        if la_count >= lookahead_period:
            z_slow = z_slow + lookahead_alpha * (z - z_slow)
            la_count = 0

        # Convert back to f
        e = np.exp(z)
        f = e / e.sum()

        # Update convolution
        F = np.fft.rfft(f, N)
        conv = np.fft.irfft(F * F, N)[:2 * n - 1]
        M = float(conv.max())

        # Track best
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > max_no_improve:
            restart_count += 1
            # Perturb around best
            noise_scale = 0.2 * (1.0 + 0.1 * restart_count)
            z = best_z + noise_scale * rng.standard_normal(n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            z_slow = z.copy()
            e = np.exp(z)
            f = e / e.sum()
            F = np.fft.rfft(f, N)
            conv = np.fft.irfft(F * F, N)[:2 * n - 1]
            M = float(conv.max())
            m.fill(0)
            v.fill(0)
            t_adam = 0
            beta = max(beta_start * 0.5, 0.1)
            no_improve = 0
            max_no_improve = max(200, max_no_improve + 100)

    return best_f, best_M


def valley_fill_step(f, M, rng):
    """
    Fast heuristic: identify where convolution is below average and add mass there.
    This approximates the LP direction without solving the LP.
    """
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    conv = np.fft.irfft(F * F, N)[:2 * n - 1]

    # Compute gradient direction: where to add mass
    # We want to reduce max conv. The "valley" approach:
    # Add mass where it contributes least to the peak convolution values.
    # grad[i] = 2*sum_k w_k * f[k-i] where w_k concentrated on peak
    peak_idx = np.argmax(conv)
    # Use a window around the peak
    half_w = max(1, n // 8)
    lo = max(0, peak_idx - half_w)
    hi = min(2 * n - 2, peak_idx + half_w)

    # Weight: higher where conv is higher
    w_local = conv[lo:hi + 1]
    w_local = w_local ** 2  # Emphasize peaks

    # Gradient contribution from peak region
    g = np.zeros(n)
    for k in range(lo, hi + 1):
        wk = w_local[k - lo]
        # d(conv[k])/d(f[i]) = 2*f[k-i] if 0 <= k-i <= n-1
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            g[i] += wk * 2.0 * f[k - i]

    # Direction: reduce g (add mass where g is small)
    g_norm = g / (np.sum(g * f) + 1e-30)
    direction = 1.0 / (g_norm + 0.1)
    direction = direction / direction.sum()

    # Blend
    t = 0.3
    f_new = (1 - t) * f + t * direction
    return normalize(f_new)


def lp_optimize_fast(f_init, deadline, rng, max_iters=100, blend_factor=0.3):
    """
    Fast LP-inspired optimization without actual LP solves.
    Uses valley-filling heuristic with line search.
    """
    f = f_init.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return f_init, compute_conv_max(f_init)

    n = len(f)
    M = compute_conv_max(f)
    best_M = M
    best_f = f.copy()
    no_imp = 0

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Get direction
        g = valley_fill_step(f, M, rng)
        if g is None:
            break

        # Normalize direction
        g = normalize(g)
        if g is None:
            break

        # Line search
        best_t = 0.0
        best_M_try = M
        for t in [blend_factor, 0.2, 0.1, 0.05, 0.02]:
            f_try = (1 - t) * f + t * g
            f_try = normalize(f_try)
            if f_try is None:
                continue
            F = np.fft.rfft(f_try, N)
            conv = np.fft.irfft(F * F, N)[:2 * n - 1]
            M_try = float(conv.max())
            if M_try < best_M_try - 1e-15:
                best_M_try = M_try
                best_t = t

        if best_t > 0:
            f_try = (1 - best_t) * f + best_t * g
            f = normalize(f_try)
            M = best_M_try
            if M < best_M - 1e-14:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 20:
            # Random perturbation
            f = f + 0.1 * rng.standard_normal(n)
            f = normalize(f)
            M = compute_conv_max(f)
            no_imp = 0

    return best_f, best_M


def get_diverse_starts(n, rng, parent=None, count=12):
    """Generate diverse starting points."""
    starts = []
    starts.append(make_linear_profile(n, 1.0))
    starts.append(make_linear_profile(n, 2.0))
    starts.append(make_linear_profile(n, 3.0))
    starts.append(make_cubic_profile(n, 1.5, -1.0))
    starts.append(make_cubic_profile(n, 2.5, -2.0))
    starts.append(make_cosine_profile(n, 1.0))
    starts.append(make_cosine_profile(n, 0.5))
    starts.append(make_dual_bump(n))

    # Spectral inits
    for _ in range(3):
        s = spectral_init(n, rng, num_freq=n // 2)
        if s is not None:
            starts.append(s)

    # Smooth random
    for _ in range(2):
        starts.append(smooth_random_init(n, rng, num_modes=6, amp=0.3))

    # Parent if available
    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy())
        else:
            starts.append(interpolate_profile(parent, n))

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
    Uses improved Adam with lookahead, spectral initialization, and multi-scale search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)
            print(f"  New best: score={sc:.10f}, n={len(f)}")

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
                print(f"Parent sequence loaded: n={len(parent)}, score={compute_score(parent):.10f}")
                try_update(parent, compute_score(parent))
    except Exception:
        pass

    results = []  # (score, f) pairs

    # ============================================================
    # Phase 1: Adam optimization at n=256 with multiple starts
    # ============================================================
    n1 = 256
    phase1_end = min(deadline, t_start + 800.0)
    print(f"\n=== Phase 1: Adam optimization at n={n1} ===")

    starts1 = get_diverse_starts(n1, rng, parent, count=12)

    # Evaluate all starts
    for i, s in enumerate(starts1):
        sc = compute_score(s)
        try_update(s, sc)

    print(f"  Initial best: {best_score:.10f}")

    # Run Adam from top starts (sorted by initial score)
    start_scores = [(compute_score(s), s) for s in starts1]
    start_scores.sort(key=lambda x: x[0])

    num_astarts = min(8, len(starts1))
    # Allocate time: more time to better starts
    total_alloc = phase1_end - time.monotonic()
    if total_alloc > 10:
        for i in range(num_astarts):
            remaining = phase1_end - time.monotonic()
            if remaining < 10:
                break
            # Better starts get more time
            weight = 1.0 / (i + 1)
            alloc = min(remaining * weight / sum(1.0 / (j + 1) for j in range(i, num_astarts)), 200.0)
            alloc = max(alloc, 20.0)
            run_deadline = time.monotonic() + alloc

            sc, f = adam_optimize(starts1[i], run_deadline, rng,
                                  max_iters=15000, base_lr=0.04,
                                  beta_start=0.3, beta_end=0.02,
                                  lookahead_period=40, lookahead_alpha=0.5,
                                  restart_every=400)
            score_val = 2.0 * n1 * sc
            try_update(f, score_val)
            results.append((score_val, f))
            print(f"  Adam start {i+1}/{num_astarts}: M={sc:.10f}, score={score_val:.10f}, time_left={phase1_end-time.monotonic():.0f}s")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.10f}")

    # ============================================================
    # Phase 2: Valley-fill refinement on top results
    # ============================================================
    phase2_end = min(deadline, t_start + 1100.0)
    print(f"\n=== Phase 2: Valley-fill refinement ===")

    top_k = min(4, len(results))
    for i in range(top_k):
        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_k - i), 100.0)
        alloc = max(alloc, 10.0)
        run_deadline = time.monotonic() + alloc

        sc, f = lp_optimize_fast(results[i][1], run_deadline, rng, max_iters=200, blend_factor=0.25)
        score_val = 2.0 * n1 * sc
        try_update(f, score_val)
        results.append((score_val, f))
        print(f"  Valley-fill {i+1}/{top_k}: score={score_val:.10f}")

    results.sort(key=lambda x: x[0])

    # ============================================================
    # Phase 3: Extended Adam on best result (more iterations)
    # ============================================================
    phase3_end = min(deadline, t_start + 1400.0)
    print(f"\n=== Phase 3: Extended Adam refinement ===")

    if len(results) > 0:
        remaining = phase3_end - time.monotonic()
        if remaining > 30:
            sc, f = adam_optimize(results[0][1], phase3_end, rng,
                                  max_iters=30000, base_lr=0.02,
                                  beta_start=0.1, beta_end=0.005,
                                  lookahead_period=60, lookahead_alpha=0.6,
                                  restart_every=600)
            score_val = 2.0 * n1 * sc
            try_update(f, score_val)
            results.append((score_val, f))
            print(f"  Extended Adam: score={score_val:.10f}")

    results.sort(key=lambda x: x[0])

    # ============================================================
    # Phase 4: Try n=512
    # ============================================================
    n4 = 512
    phase4_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 4: Try n={n4} ===")

    if len(results) > 0:
        top_4 = min(3, len(results))
        for i in range(top_4):
            remaining = phase4_end - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining / max(1, top_4 - i), 150.0)
            alloc = max(alloc, 20.0)
            run_deadline = time.monotonic() + alloc

            f_ext = interpolate_profile(results[i][1], n4)
            sc, f = adam_optimize(f_ext, run_deadline, rng,
                                  max_iters=10000, base_lr=0.03,
                                  beta_start=0.2, beta_end=0.01,
                                  lookahead_period=50, lookahead_alpha=0.5,
                                  restart_every=500)
            score_val = 2.0 * n4 * sc
            try_update(f, score_val)
            results.append((score_val, f))
            print(f"  n={n4} Adam {i+1}/{top_4}: score={score_val:.10f}, time_left={phase4_end-time.monotonic():.0f}s")

    results.sort(key=lambda x: x[0])

    # ============================================================
    # Phase 5: Final polish on overall best
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: Final polish ===")

    if best_seq is not None and time.monotonic() < phase5_end - 30:
        n_best = len(best_seq)
        remaining = phase5_end - time.monotonic()
        if remaining > 20:
            sc, f = adam_optimize(best_seq, phase5_end, rng,
                                  max_iters=20000, base_lr=0.01,
                                  beta_start=0.05, beta_end=0.002,
                                  lookahead_period=80, lookahead_alpha=0.7,
                                  restart_every=800)
            score_val = 2.0 * n_best * sc
            try_update(f, score_val)
            print(f"  Final polish (n={n_best}): score={score_val:.10f}")

    # ============================================================
    # Final: validate and return
    # ============================================================
    if best_seq is None:
        best_seq = make_linear_profile(256, 1.5)
        best_score = compute_score(best_seq)

    # Ensure validity
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, computed score={final_score:.10f}")

    # Validate with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence score: {ev_score:.10f}")
        if np.isfinite(ev_score):
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    print(f"Time used: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    return [float(x) for x in best_seq]
