"""
Phase-annealed multiplicative gradient descent for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses power-law starts, aggressive soft-max gradient descent with adaptive temperature,
multi-resolution refinement, and targeted sub-gradient polishing.
"""
import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a, mode='full')
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = np.maximum(c, 0.0)
    return c


def softmax_weights(c, beta):
    """Compute softmax weights for soft-max gradient."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.array([])
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def conv_grad(c, a, w):
    """Compute gradient: grad[i] = sum_k w[k] * 2*a[k-i] (with bounds)."""
    n = len(a)
    N = next_pow2(2 * n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    h = np.fft.irfft(fw * fa, N)
    # h[k] = sum_i w[i] * a[k-i]
    # We need for each i: sum_k w[k] * a[k-i]
    # This is h with indices shifted: result[i] = 2 * h[i + n - 1] ... 
    # Actually: h[m] = sum_k w[k]*a[m-k]. We want g[i] = sum_k w[k]*a[k-i].
    # Let j = k-i, k = j+i: g[i] = sum_j w[j+i]*a[j] = (a * w_reversed)[i] ... 
    # Simpler: g[i] = sum_k w[k]*a[k-i]. Let m = k-i: g[i] = sum_m w[m+i]*a[m].
    # This is the cross-correlation of w and a at lag i.
    # conv(a_reversed, w)[n-1+i] = sum_k a[n-1-k]*w[k+n-1+i] ... hmm getting confused.
    # Let me just use: g[i] = sum_k w[k]*a[k-i] for k-i in [0,n-1], i in [0,n-1]
    # h = conv(w, a) via FFT gives h[m] = sum_k w[k]*a[m-k] for m=0..2n-2
    # g[i] = sum_k w[k]*a[k-i] = sum_k w[k-i+i]*a[k-i] ... let j=k-i: = sum_j w[j+i]*a[j]
    # This equals conv(a, w_reversed)[i + n - 1] ... 
    # Easiest: g[i] = h_from_conv[i] where h_from_conv = conv(a_rev, w)[n-1:]
    # Because conv(a_rev, w)[n-1+i] = sum_k a_rev[k]*w[n-1+i-k] = sum_k a[n-1-k]*w[n-1+i-k]
    # Let j = n-1-k: = sum_j a[j]*w[i+j] = sum_j a[j]*w[j+i] = g[i] (renaming). YES!
    fa_rev = np.fft.rfft(a[::-1], N)
    g = np.fft.irfft(fa_rev * fw, N)[n - 1:2 * n - 1]
    return 2.0 * g


def interp_profile(p, n_new):
    """Linearly interpolate profile to new length."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        return p / s if s > 1e-10 else p
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p)


def make_ramp(n):
    """Pure linear ramp [1, 2, ..., n]."""
    a = np.arange(1.0, n + 1.0)
    return a / a.sum()


def make_ramp_zero(n):
    """Ramp starting from 0: [0, 1, 2, ..., n-1]."""
    a = np.arange(0.0, float(n))
    # Replace first zero with tiny value to avoid issues
    a[0] = 1e-10
    return a / a.sum()


def make_power_law(n, p):
    """Power law: [1^p, 2^p, ..., n^p]."""
    a = np.arange(1.0, n + 1.0) ** p
    return a / a.sum()


def make_flipped_ramp(n):
    """Decreasing ramp: [n, n-1, ..., 1]."""
    a = np.arange(float(n), 0.0, -1.0)
    return a / a.sum()


def make_double_ramp(n, peak_frac=0.5):
    """Ramp up then down."""
    peak = int(n * peak_frac)
    a = np.zeros(n)
    a[:peak] = np.arange(1.0, peak + 1.0)
    a[peak:] = np.arange(peak, 0.0, -1.0)
    if a[peak] == 0:
        a[peak] = 1.0
    return a / a.sum()


def make_log_ramp(n):
    """Log ramp: [log(2), log(3), ..., log(n+1)]."""
    a = np.log(np.arange(2.0, n + 2.0))
    return a / a.sum()


def make_exponential(n, rate=1.0):
    """Exponential: [e^{rate*i} for i=0..n-1], capped."""
    a = np.exp(rate * np.arange(n, dtype=np.float64) / max(1, n - 1))
    return a / a.sum()


def gradient_descend(a0, deadline, rng, lr_init=0.1, beta_ratio=0.15, verbose=True, label=""):
    """Adaptive soft-max gradient descent with momentum."""
    a = np.maximum(np.asarray(a0, dtype=np.float64).ravel(), 1e-12)
    a = a / a.sum()
    n = a.size

    c = full_convolve_fft(a)
    maxc = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * maxc)
    beta = max(1e-10, beta_ratio * maxc)
    lr = lr_init
    no_imp = 0
    stagnant = 0
    iteration = 0

    # Momentum buffers
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    adam_t = 0

    while time.monotonic() < deadline:
        w = softmax_weights(c, beta)
        g = conv_grad(c, a, w)

        # Center the gradient (invariance to scaling)
        g -= float(np.dot(a, g)) * a / max(1e-15, float(np.dot(a, a)))

        if not np.all(np.isfinite(g)):
            a = best_a.copy()
            m_buf.fill(0)
            v_buf.fill(0)
            adam_t = 0
            continue

        # Adam update
        adam_t += 1
        m_buf = b1 * m_buf + (1 - b1) * g
        v_buf = b2 * v_buf + (1 - b2) * g * g
        mh = m_buf / (1 - b1 ** adam_t)
        vh = v_buf / (1 - b2 ** adam_t)
        
        # Multiplicative update with Adam
        step = lr * mh / (np.sqrt(vh) + eps)
        step = np.clip(step, -2.0, 2.0)
        a = np.maximum(a * np.exp(step), 1e-15)
        s = a.sum()
        if s < 1e-10 or not np.isfinite(s):
            a = best_a.copy()
            continue
        a = a / s

        c = full_convolve_fft(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
            stagnant = 0
        else:
            no_imp += 1
            stagnant += 1

        # Anneal beta
        if iteration % 30 == 0:
            beta = max(1e-10, beta * 0.85)

        # Adaptive lr
        if no_imp > 200:
            lr = max(0.005, lr * 0.95)

        # Restart if very stagnant
        if stagnant > max(500, 2 * n):
            a = best_a + 0.05 * rng.normal(size=n)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            c = full_convolve_fft(a)
            maxc = float(c.max())
            beta = max(1e-10, 0.2 * maxc)
            lr = lr_init * 0.5
            m_buf.fill(0)
            v_buf.fill(0)
            adam_t = 0
            stagnant = 0
            no_imp = 0
            if verbose:
                print(f"    [{label}] Restart at iter {iteration}, best={best_score:.8f}")

        iteration += 1
        if verbose and iteration % 200 == 0:
            print(f"    [{label}] iter={iteration}, score={best_score:.8f}, beta={beta:.2e}, lr={lr:.4f}")

    return best_a, best_score


def topk_subgrad(a0, deadline, rng, topk=20, verbose=True, label=""):
    """Targeted subgradient on top-k lag positions."""
    a = np.maximum(np.asarray(a0, dtype=np.float64).ravel(), 1e-12)
    a = a / a.sum()
    n = a.size

    c = full_convolve_fft(a)
    maxc = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * maxc)
    iteration = 0
    no_imp = 0

    # Learnable weights for top-k positions (in log space)
    log_w = np.zeros(min(topk, 2 * n - 1))

    while time.monotonic() < deadline:
        # Find top-k lags
        kn = min(topk, 2 * n - 1)
        top_idx = np.argsort(c)[-kn:]
        
        # Update log_w based on how peaked each top-k position is
        top_vals = c[top_idx]
        # Normalize: positions with higher conv values get more weight
        w_k = top_vals / max(1e-12, top_vals.sum())
        
        # Update log_w with exponential moving average
        log_w = 0.8 * log_w + 0.2 * np.log(w_k + 1e-10)
        log_w = log_w - float(log_w.mean())
        w_k = np.exp(log_w - log_w.max())
        w_k = w_k / w_k.sum()

        # Compute combined gradient: sum of topk w_k[j] * grad_j
        N = next_pow2(2 * n)
        grad = np.zeros(n)
        fa = np.fft.rfft(a, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        
        for j in range(kn):
            k = top_idx[j]
            # grad_j[i] = 2*a[k-i] for valid i
            # Contribution to total grad: w_k[j] * 2*a[k-i]
            # Sum over j: 2 * sum_j w_k[j] * a[k_j - i]
            # = 2 * conv(a_rev, w_combined)[n-1+i] where w_combined[k_j] = w_k[j]
            pass
        
        # More efficient: build combined weight vector and do one FFT
        w_combined = np.zeros(2 * n - 1)
        for j in range(kn):
            w_combined[top_idx[j]] = w_k[j]
        
        fw = np.fft.rfft(w_combined, N)
        h = np.fft.irfft(fa_rev * fw, N)[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        # Center gradient
        grad -= float(np.dot(a, grad)) * a / max(1e-15, float(np.dot(a, a)))

        if not np.all(np.isfinite(grad)):
            a = best_a.copy()
            continue

        # Multiplicative step
        lr = 0.05
        step = np.clip(lr * grad, -1.0, 1.0)
        a_new = np.maximum(a * np.exp(step), 1e-15)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        
        c = full_convolve_fft(a_new)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            lr *= 0.95

        iteration += 1
        if verbose and iteration % 100 == 0:
            print(f"    [{label}] topk iter={iteration}, score={best_score:.8f}")

        if no_imp > 500:
            # Perturb
            a = best_a + 0.02 * rng.normal(size=n)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            c = full_convolve_fft(a)
            no_imp = 0

    return best_a, best_score

    # Final cleanup
    if best_seq is None:
        best_seq = make_ramp(512)

    final_seq = np.maximum(best_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_ramp(256)

    final_score = compute_score(final_seq)
    print(f"\n=== FINAL: n={final_seq.size}, score={final_score:.8f} ===")

    # Validate
    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.8f}")
        else:
            print(f"WARNING: evaluate_sequence returned {eval_score}")
            # Fallback to a safe sequence
            final_seq = make_ramp(512)
            final_score = compute_score(final_seq)
            print(f"Fallback ramp score: {final_score:.8f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    elapsed = time.monotonic() - t_start
    print(f"Total time: {elapsed:.1f}s")

    return [float(x) for x in final_seq]
