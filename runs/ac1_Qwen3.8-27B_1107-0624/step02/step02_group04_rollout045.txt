def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):

def _soft_max_gradient_step(p, c, beta):
    """Compute soft-max weights and gradient."""
    # Soft-max of c
    m = float(c.max())
    x = (c - m) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # Gradient: d/dp_i max(c) ≈ (c * w) * (p reversed)
    # Actually, the gradient of c_k w.r.t. p_i is p_{k-i}
    # So the gradient of sum_k w_k * c_k w.r.t. p_i is sum_k w_k * p_{k-i}
    # = (w * p_reversed)[i + n - 1] (cross-correlation)
    
    n = len(p)
    # Compute w * p_reversed (cross-correlation)
    # result[i] = sum_k w[k] * p[k-i] for valid k

    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    gp = h[n-1:2*n-1]
    
    return gp, m + beta * np.log(s)

"""
Adaptive optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses logit-space optimization with adaptive soft-max temperature,
LP-guided direction updates, and multi-scale search.

import numpy as np

# Helper functions (top-level as required)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c[:]+= -mn
    return c

def _score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2n * max(c)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _full_convolve_fft(a)
    return float(2.0 * n * c.max())

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = _full_convolve_fft(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

def _softmax_exp(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / float(e.size)
    return p / s

def _make_profile(n, kind='lin', param=1.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == 'lin':
        a = 1.0 + param * u
    elif kind == 'quad':
        a = 1.0 + param * u + 0.3 * param * u * u
    elif kind == 'cubic':
        a = 1.0 + param * u - 0.3 * param * u**3
    elif kind == 'cos':
        a = 1.0 + param * np.cos(np.pi * t)
    elif kind == 'bump':
        a = 1.0 + param * np.exp(-4.0 * u * u)
    elif kind == 'edge':
        a = 1.0 + param * (1.0 - u * u)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-8)
    return a / a.sum()

def _interp_to_n(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-10)
    s = p.sum()
    return p / s if s > 1e-30 else None

def _corr(w, p):
    """Cross-correlation: result[i] = sum_k w[k] * p[k-i] (valid range)."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n-1:2*n-1]

def _grad_softmax(c, p, beta):
    """Gradient of soft-max beta(c) w.r.t. p."""
    n = len(p)
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        L = m
    else:
        w = ex / s
        L = m + beta * np.log(s)
    gp = _corr(w, p)
    return gp, L, w

def _lp_step(f, M, timeout=30.0):
    """
    Solve LP: max sum(b) s.t. (f*b)_k <= M for all k, b >= 0.
    Returns normalized b (sum=1) or None.
    """
    import cvxpy as cp
    from scipy.sparse import csr_matrix
    
    n = f.size
    t0 = time.monotonic()
    if t0 > timeout:
        return None
    
    # Build sparse constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, 2*n - 1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    b_var = cp.Variable(n)
    constraints = [A @ b_var <= M, b_var >= 0]
    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=3000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, verbose=False)
        except Exception:
            return None
    
    if b_var.value is None:
        return None
    
    b_val = np.asarray(b_var.value).ravel()
    b
