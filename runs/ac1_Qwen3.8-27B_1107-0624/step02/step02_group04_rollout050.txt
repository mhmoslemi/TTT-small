def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:

    c = _full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def _make_hann(n):

def _compute_grad_smooth_max(a, c, beta):
    """Compute gradient of soft_max(c, beta) w.r.t. a where c = conv(a,a)."""
    n = a.size
    c = np.asarray(c, dtype=np.float64)
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / beta)
    w = exp_c / exp_c.sum()
    # grad of b[k] w.r.t. a[i] is a[k-i] (for valid indices)
    # So grad_a soft_max = sum_k w[k] * (a shifted by k)
    # Using correlation: grad[i] = sum_k w[k] * a[k-i] + sum_k w[k] * a[i-k]
    # Actually for conv(a,a)[k] = sum_j a[j]*a[k-j], d/db_k/d a[i] = a[k-i] (if valid)
    # So sum_k w[k] * a[k-i] over valid k-i
    # This is the correlation of w with a (reversed)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    grad = h[n - 1:2 * n - 1]
    return grad


def _objective_and_grad(z, beta, n):
    """Objective and gradient in logit space for L-BFGS-B."""
    a = _softmax_from_z(z)
    c = _full_convolve_fft(a)
    if c.size == 0:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    c_shifted = c - c_max
    exp_c = np.exp(c_shifted / beta)
    s = float(exp_c.sum())
    if not np.isfinite(s) or s <= 0:
        return 1e10, np.zeros(n)
    obj = 2.0 * n * (c_max + beta * np.log(s))
    w = exp_c / s
    grad_a = _compute_grad_smooth_max(a, c, beta)
    # Chain rule for softmax: grad_z = a * (grad_a - sum(a * grad_a))
    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros(n)
    return float(obj), grad_z


def _lbfgs_optimize(p0, deadline, beta_init=0.05, beta_final=0.002,
                    maxiter=500, tol=1e-12):
    """Run L-BFGS-B on the soft-max objective in logit space."""
    n = p0.size
    z0 = np.log(np.maximum(p0, 1e-12))
    z0 -= z0.mean()

    state = {'beta': beta_init, 'iter': 0, 'deadline': deadline}
    total_iters = max(50, int(maxiter))
    n_param = n

    def obj_grad(z):
        if time.monotonic() > state['deadline']:
            return 1e10, np.zeros(n_param)
        state['iter'] += 1
        # Decay beta over iterations
        frac = min(1.0, state['iter'] / total_iters)
        state['beta'] = beta_init * (beta_final / beta_init) ** frac
        return _objective_and_grad(z, state['beta'], n_param)

    try:
        res = minimize(obj_grad, z0, jac=True, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': tol,
                                'gtol': 1e-8, 'maxfun': maxiter * 3})
    except Exception:
        res = None

    if res is not None and hasattr(res, 'x'):
        z_best = res.x
        p_best = _softmax_from_z(z_best)
        sc = _internal_score(p_best)
        return p_best, sc
    else:
        sc = _internal_score(p0)
        return p0.copy(), sc

        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z -= z.mean()
        p = _softmax_from_z(z)
        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.005, lr * 0.95)

        if t % 30 == 0:
            beta = max(1e-12, beta * beta_decay)

        if no_imp > no_imp_limit:
            break

    return best_p, best_score


def _soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _local_peak_reduction(p, n_reductions=50, deadline=None):
    """Try to reduce the peak of the autocorrelation by adjusting elements near the peak."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    c = _full_convolve_fft(p)
    best_score = float(2.0 * n * c.max())

    for _ in range(n_reductions):
        if deadline is not None and time.monotonic() > deadline:
            break
        peak_idx = int(np.argmax(c))
        # The peak at lag k means: sum_i a[i]*a[k-i] is large
        # Try reducing elements that contribute most to this peak
        # Find which a[i] values contribute most
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        if lo >= hi:
            break
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] += p[j]
                contrib[j] += p[i]
        # Reduce the largest contributor slightly
        max_contrib_idx = int(np.argmax(contrib))
        if contrib[max_contrib_idx] <= 0:
            break
        # Try slightly reducing this element
        old_val = p[max_contrib_idx]
        for frac in [0.95, 0.9, 0.85, 0.8, 0.75]:
            p_try = p.copy()
            p_try[max_contrib_idx] = old_val * frac
            p_try = np.maximum(p_try, 0.0)
            s = p_try.sum()
            if s <= 0:
                continue
            p_try = p_try / s
            c_try = _full_convolve_fft(p_try)
            sc_try = float(2.0 * n * c_try.max())
            if sc_try < best_score:
                best_score = sc_try
                p = p_try
                c = c_try
                break
    return p, best_score

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses L-BFGS-B with soft-max smoothing, diverse initialization,
    multi-length escalation, and Adam fine-tuning.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={p.size}) [{label}]")

    # --- Get parent sequence ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent available: n={parent.size}, score={_internal_score(parent):.10f}")

    # --- Phase 1: Quick scan of profiles at multiple lengths ---
    print("Phase 1: Quick profile scan")
    t_phase1_end = min(deadline, t_start + 60.0)
    lengths = [128, 256, 512, 1024]
    if parent is not None:
        pn = parent.size
        if pn not in lengths and pn <= 4096:
            lengths.append(pn)
            lengths.sort()

    candidates = []
    for n in lengths:
        if time.monotonic() > t_phase1_end:
            break
        profs = _generate_diverse_profiles(n)
        if parent is not None and parent.size != n:
            p_parent = _interp_profile(parent, n)
            if p_parent is not None:
                profs.append(('parent', p_parent))
        if parent is not None and parent.size == n:
            profs.append(('parent', parent.copy()))

        for label, p in profs:
            sc = _internal_score(p)
            if np.isfinite(sc):
                candidates.append((sc, p.copy(), n, label))
                update_best(p, sc, f"scan {label} n={n}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}, {len(candidates)} candidates")

    # --- Phase 2: L-BFGS-B optimization on top candidates ---
    print("Phase 2: L-BFGS-B optimization")
    t_phase2_end = min(deadline, t_start + 600.0)
    top_candidates = []
    seen = set()
    for sc, p, n, label in candidates:
        key = n
        if key not in seen:
            seen.add(key)
            top_candidates.append((sc, p, n, label))
        if len(top_candidates) >= 6:
            break

    for sc, p, n, label in top_candidates:
        if time.monotonic() > t_phase2_end:
            break
        per_time = min(80.0, t_phase2_end - time.monotonic())
        if per_time < 10:
            break
        print(f"  L-BFGS-B on {label} n={n} (score={sc:.10f})")
        p_opt, sc_opt = _lbfgs_optimize(p, t_start + time.monotonic() + per_time,
                                         beta_init=0.08, beta_final=0.003,
                                         maxiter=200)
        update_best(p_opt, sc_opt, f"lbfgs {label} n={n}")

    # --- Phase 3: Adam refinement on best ---
    print("Phase 3: Adam refinement")
    t_phase3_end = min(deadline, t_start + 900.0)
    if best_seq is not None and time.monotonic() < t_phase3_end:
        per_time = min(120.0, t_phase3_end - time.monotonic())
        p_opt, sc_opt = _adam_refine(best_seq, t_start + time.monotonic() + per_time,
                                      lr=0.06, beta_init_frac=0.08, beta_decay=0.70)
        update_best(p_opt, sc_opt, "adam_refine")

        # Second pass with smaller lr
        if time.monotonic() < t_phase3_end:
            per_time2 = min(60.0, t_phase3_end - time.monotonic())
            p_opt2, sc_opt2 = _adam_refine(best_seq, t_start + time.monotonic() + per_time2,
                                            lr=0.03, beta_init_frac=0.05, beta_decay=0.75,
                                            no_imp_limit=300)
            update_best(p_opt2, sc_opt2, "adam_refine2")

    # --- Phase 4: Multi-restart with diverse perturbations ---
    print("Phase 4: Multi-restart")
    t_phase4_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < t_phase4_end:
        n_best = best_seq.size
        n_restarts = 6
        per_restart = min(60.0, (t_phase4_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > t_phase4_end:
                break
            # Create perturbation
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            noise_scale = 0.1 + 0.15 * i
            z = z + noise_scale * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_start = _softmax_from_z(z)

            # Alternate between L-BFGS-B and Adam
            if i % 2 == 0:
                p_try, sc_try = _lbfgs_optimize(p_start, t_start + time.monotonic() + per_restart,
                                                 beta_init=0.06, beta_final=0.003, maxiter=150)
            else:
                p_try, sc_try = _adam_refine(p_start, t_start + time.monotonic() + per_restart,
                                              lr=0.05, beta_init_frac=0.07, beta_decay=0.70)
            update_best(p_try, sc_try, f"restart{i} n={n_best}")

    # --- Phase 5: Length escalation ---
    print("Phase 5: Length escalation")
    t_phase5_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < t_phase5_end:
        n_cur = best_seq.size
        target_lengths = [l for l in [256, 512, 1024, 2048] if l != n_cur and l <= 2048]
        # Try to escalate to a longer length
        for target_n in sorted(target_lengths):
            if time.monotonic() > t_phase5_end:
                break
            p_interp = _interp_profile(best_seq, target_n)
            if p_interp is None:
                continue
            sc_interp = _internal_score(p_interp)
            update_best(p_interp, sc_interp, f"interp n={target_n}")

            per_time = min(80.0, t_phase5_end - time.monotonic())
            if per_time < 15:
                break
            p_opt, sc_opt = _lbfgs_optimize(p_interp, t_start + time.monotonic() + per_time,
                                             beta_init=0.08, beta_final=0.005, maxiter=150)
            update_best(p_opt, sc_opt, f"lbfgs_escalate n={target_n}")

        # Also try shorter lengths if current is large
        if n_cur > 512:
            for target_n in [256, 512]:
                if time.monotonic() > t_phase5_end:
                    break
                p_interp = _interp_profile(best_seq, target_n)
                if p_interp is None:
                    continue
                sc_interp = _internal_score(p_interp)
                update_best(p_interp, sc_interp, f"interp_down n={target_n}")
                per_time = min(40.0, t_phase5_end - time.monotonic())
                if per_time < 10:
                    break
                p_opt, sc_opt = _lbfgs_optimize(p_interp, t_start + time.monotonic() + per_time,
                                                 beta_init=0.06, beta_final=0.003, maxiter=100)
                update_best(p_opt, sc_opt, f"lbfgs_down n={target_n}")

    # --- Phase 6: Final polish with local peak reduction and tiny Adam ---
    print("Phase 6: Final polish")
    if best_seq is not None:
        # Local peak reduction
        p_lp, sc_lp = _local_peak_reduction(best_seq, n_reductions=30,
                                             deadline=deadline)
        update_best(p_lp, sc_lp, "peak_reduce")

        # Tiny Adam
        per_final = min(60.0, deadline - time.monotonic())
        if per_final > 5:
            p_fin, sc_fin = _adam_refine(best_seq, t_start + time.monotonic() + per_final,
                                          lr=0.02, beta_init_frac=0.03, beta_decay=0.80,
                                          no_imp_limit=500)
            update_best(p_fin, sc_fin, "final_adam")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_hann(256)

        # Exact score for verification
        if final_seq.size <= 2048:
            sc_exact = _true_score(final_seq)
            print(f"Exact score: {sc_exact:.10f}")

        # Try evaluate_sequence
        if final_seq.size <= 2048:
            try:
                eval_list = [float(x) for x in final_seq]
                sc_eval = float(evaluate_sequence(eval_list))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception as e:
                print(f"evaluate_sequence error: {e}")

        print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_hann(256)
    print(f"FALLBACK: n=256, score={_internal_score(fallback):.10f}")
    return [float(x) for x in fallback]
