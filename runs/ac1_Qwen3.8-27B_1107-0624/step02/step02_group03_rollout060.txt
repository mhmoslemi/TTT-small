import numpy as np

def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:

    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[k] = convolve(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=5.0):
    """Solve LP: maximize sum(g) s.t. g >= 0, conv(f,g) <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

        # Adaptive beta: proportional to the gap
        sorted_c = np.sort(c)[::-1]
        gap = max_c - sorted_c[min(1, len(sorted_c) - 1)]
        beta = max(1e-6, 0.5 * gap)
        beta = max(beta, 0.3 * beta0)  # don't go below 30% of initial

        # Soft argmax weights
        x_c = (c - max_c) / beta
        ex = np.exp(x_c)
        w = ex / ex.sum()

        # Gradient computation
        h = compute_h(w, p_y)
        grad_p = 2.0 * h
        dot_val = float(np.dot(p_y, grad_p))
        grad_z = p_y * (grad_p - dot_val)

        # Check for finite gradient
        if not np.all(np.isfinite(grad_z)):
            y = best_z.copy()
            x = y.copy()
            x_prev = y.copy()
            continue

        # Nesterov accelerated update
        x_new = y - lr * grad_z
        t += 1
        beta_t = (t - 1) / (t + 1)
        y = x_new + beta_t * (x_new - x)
        x = x_new

        # Clip and center
        y = np.clip(y, -60.0, 60.0)
        y -= np.mean(y)

        # Divergence check
        if not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 55:
            y = best_z.copy()
            x = y.copy()
            x_prev = y.copy()
            t = 1
            lr = max(0.003, lr * 0.5)

        # Periodic decay of lr
        if it % 20000 == 0:
            lr = max(0.002, lr * 0.85)

        # Restart mechanism
        if it % 10000 == 0:
            # Check KKT residual: how uniform is p_y * h?
            ph = p_y * h
            ph_cv = (ph.max() - ph.min()) / (ph.mean() + 1e-30)
            if ph_cv > 0.5:
                rng = np.random.default_rng(it)
                z_new = 0.85 * best_z + 0.4 * rng.normal(size=n)
                z_new = np.clip(z_new, -50, 50)
                z_new -= np.mean(z_new)
                y = z_new
                x = z_new.copy()
                x_prev = z_new.copy()
                t = 1
                lr = max(0.005, lr * 1.1)

        if verbose and it % 5000 == 0:
            print(f"    Nesterov iter {it}: score={best_score:.8f}, beta={beta:.6f}, lr={lr:.6f}")

    return best_p, best_score


def mul_weight_polish(p0, deadline, lr=0.5, sharpness=50.0, verbose=False):
    """
    Multiplicative weight update for polishing.
    p_j <- p_j * (mean(h)/h_j)^lr where h_j = sum_k w_k * p_{k-j}.
    """
    n = len(p0)
    p = np.maximum(p0, 1e-12)
    p = p / p.sum()
    best_p = p.copy()
    best_score = internal_score(p)
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p, p)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()

        # Adaptive sharpness based on gap
        sorted_c = np.sort(c)[::-1]
        gap = max_c - sorted_c[1] if len(sorted_c) > 1 else max_c
        sharp = max(10.0, min(200.0, 5.0 / (gap + 1e-10)))

        x_c = (c - max_c) * sharp
        ex = np.exp(x_c)
        w = ex / ex.sum()

        h = compute_h(w, p)
        mean_h = h.mean()
        ratio = (mean_h / (h + 1e-30)) ** lr
        p = p * ratio
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

        if verbose and it % 10000 == 0:
            print(f"    MW iter {it}: score={best_score:.8f}")

    return best_p, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def propose_candidate():
    """
    Multi-strategy search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
      1. Load parent sequence (n=512).
      2. LP-based direction finding for coarse improvement.
      3. Nesterov-accelerated gradient polish with adaptive smoothing.
      4. Multiplicative weight polish.
      5. Try different resolutions (n=1024) for potential improvement.
      6. Final fine polish.
    """
    t_start = time.monotonic()
    budget = 1800.0
    deadline = t_start + budget - 20.0
    rng = np.random.default_rng(123)

    # Load parent sequence
    best_seq = None
    best_score = np.inf

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent
                best_score = internal_score(parent)
                print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        n_init = 512
        t_arr = np.linspace(0, 1, n_init)
        a = 1.0 + 1.5 * (2 * t_arr - 1)
        a = np.maximum(a, 1e-10)
        best_seq = a / a.sum()
        best_score = internal_score(best_seq)
        print(f"Initial: n={len(best_seq)}, score={best_score:.8f}")

    # Ensure n=512 for main optimization
    if len(best_seq) != 512:
        best_seq = interpolate_seq(best_seq, 512)
        best_score = internal_score(best_seq)
        print(f"Interpolated to n=512: score={best_score:.8f}")

    # Phase 1: LP iterations (coarse)
    deadline_p1 = min(deadline, t_start + 250.0)
    print(f"\nPhase 1: LP at n=512, score={best_score:.8f}")
    f_best, sc_best = lp_iterate(best_seq.copy(), max_iters=500, deadline=deadline_p1, verbose=True)
    if f_best is not None and sc_best < best_score:
        best_score = sc_best
        best_seq = f_best
        print(f"  Phase 1 result: score={best_score:.8f}")

    # Phase 2: Nesterov polish at n=512 (main workhorse)
    deadline_p2 = min(deadline, t_start + 900.0)
    print(f"\nPhase 2: Nesterov at n=512, score={best_score:.8f}")
    p_best, sc_best = nesterov_polish(best_seq, deadline_p2, lr=0.012, beta0=0.025, verbose=True)
    if sc_best < best_score:
        best_score = sc_best
        best_seq = p_best
        print(f"  Phase 2 result: score={best_score:.8f}")

    # Phase 3: Multiplicative weight polish
    deadline_p3 = min(deadline, t_start + 1200.0)
    if time.monotonic() < deadline_p3 - 30:
        print(f"\nPhase 3: MW polish at n=512, score={best_score:.8f}")
        p_best, sc_best = mul_weight_polish(best_seq, deadline_p3, lr=0.4, verbose=True)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 3 result: score={best_score:.8f}")

    # Phase 4: Try n=1024
    deadline_p4 = min(deadline, t_start + 1550.0)
    if time.monotonic() < deadline_p4 - 60:
        a_1024 = interpolate_seq(best_seq, 1024)
        sc_1024 = internal_score(a_1024)
        print(f"\nPhase 4: Nesterov at n=1024, initial score={sc_1024:.8f}")
        p_1024, sc_1024 = nesterov_polish(a_1024, deadline_p4, lr=0.008, beta0=0.015, verbose=True)
        if sc_1024 < best_score:
            best_score = sc_1024
            best_seq = p_1024
            print(f"  Phase 4 result: score={best_score:.8f}")
        else:
            print(f"  Phase 4: no improvement ({sc_1024:.8f} >= {best_score:.8f})")

    # Phase 5: Final Nesterov polish on best sequence
    if time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        print(f"\nPhase 5: Final polish at n={n_final}, score={best_score:.8f}")
        p_final, sc_final = nesterov_polish(best_seq, deadline, lr=0.005, beta0=0.01, verbose=True)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = p_final
            print(f"  Phase 5 result: score={best_score:.8f}")

    # Also try a quick n=256 check
    if time.monotonic() < deadline - 5:
        a_256 = interpolate_seq(best_seq, 256)
        sc_256 = internal_score(a_256)
        if sc_256 < best_score:
            print(f"  n=256 is better: {sc_256:.8f}")
            best_seq = a_256
            best_score = sc_256

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = np.ones(512) / 512.0
        s = 1.0
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
