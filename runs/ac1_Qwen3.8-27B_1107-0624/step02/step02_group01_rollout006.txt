def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_conv_and_max(a):
    c = fft_convolve(a)
    return c, float(c.max())


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_arcsine_profile(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_semicircle_profile(n):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.sqrt(np.maximum(center ** 2 - (i - center) ** 2, 1e-12))
    return a / a.sum()


def make_power_profile(n, alpha):
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = t ** alpha + (1 - t) ** alpha + 1e-12
    return a / a.sum()


def make_cosine_profile(n, power):
    t = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * t) ** (2 * power) + 1e-12
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    center = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - center) / center)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def solve_sparse_lp(f, M, n, top_k_indices, max_time=5.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for k in top_k_indices, g >= 0."""
    K = len(top_k_indices)
    A = np.zeros((K, n), dtype=np.float64)
    for idx, k in enumerate(top_k_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]
    c_obj = -np.ones(n)
    b = np.full(K, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max_time})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def softmax_gradient_polish(f0, deadline, rng, lr=0.05, max_iter=100000, beta_init=None):
    """Adaptive soft-max gradient polishing in logit space."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c, M_val = compute_conv_and_max(f)
    if beta_init is None:
        beta = max(1e-12, 0.1 * M_val)
    else:
        beta = beta_init

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_corr = next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Soft-max weights over convolution positions
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 10.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        # Gradient in z-space
        # h[j] = sum_p w[p] * a[p-j] (cross-correlation)
        fw = np.fft.rfft(w, N_corr)
        fa_n = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(fa_n), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c, M_new = compute_conv_and_max(f)

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta
        if t_adam % 200 == 0:
            beta = max(1e-12, beta * 0.85)

        # Adaptive learning rate
        if t_adam % 1000 == 0 and no_imp > 500:
            lr *= 0.5

        # Restart if stuck
        if no_imp > max(500, 5 * n):
            z = 0.8 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c, M_val = compute_conv_and_max(f)
            beta = max(1e-12, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
            lr = 0.05

    return best_f, best_M


def peak_shaving(f0, deadline, rng, max_iter=50000):
    """Directly shave the top peaks by adjusting contributing elements."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    c, M = compute_conv_and_max(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Find top peak position
        k_star = int(np.argmax(c))
        # Find elements contributing to this peak
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib_indices = np.arange(lo, hi + 1)
        contrib_vals = f[contrib_indices] * f[k_star - contrib_indices]

        # Reduce the largest contributors
        if len(contrib_indices) > 0:
            top_contrib = np.argsort(contrib_vals)[-3:]
            for idx in top_contrib:
                j = contrib_indices[idx]
                # Try reducing a_j slightly
                reduction = rng.uniform(0.01, 0.05) * f[j]
                f_new = f.copy()
                f_new[j] = max(0.0, f[j] - reduction)
                # Redistribute to minimize new peak
                # Add to the element that least increases the max
                if f_new[j] > 0 or reduction > 0:
                    # Spread reduction across nearby elements
                    neighbors = np.arange(max(0, j - 2), min(n, j + 3))
                    for ni in neighbors:
                        f_new[ni] += reduction / len(neighbors)
                    f_new = np.maximum(f_new, 0.0)
                    fs = f_new.sum()
                    if fs > 1e-30:
                        f_new = f_new / fs
                        c_new, M_new = compute_conv_and_max(f_new)
                        if M_new < M - 1e-15:
                            f = f_new
                            c = c_new
                            M = M_new
                            no_imp = 0
                            if M < best_M:
                                best_M = M
                                best_f = f.copy()
                            break
                if M <= best_M + 1e-15:
                    no_imp += 1

        # Also try increasing non-contributing elements
        if no_imp > 50:
            non_contrib = np.setdiff1d(np.arange(n), contrib_indices)
            if len(non_contrib) > 0:
                j = rng.choice(non_contrib)
                f_new = f.copy()
                increase = rng.uniform(0.01, 0.03) * f.mean()
                f_new[j] += increase
                # Reduce the peak contributor
                pk = int(np.argmax(c))
                f_new[max(0, pk - 1)] = max(0.0, f_new[max(0, pk - 1)] - 0.5 * increase)
                f_new = np.maximum(f_new, 0.0)
                fs = f_new.sum()
                if fs > 1e-30:
                    f_new = f_new / fs
                    c_new, M_new = compute_conv_and_max(f_new)
                    if M_new < M - 1e-15:
                        f = f_new
                        c = c_new
                        M = M_new
                        no_imp = 0
                        if M < best_M:
                            best_M = M
                            best_f = f.copy()

        if no_imp > 200:
            z = np.log(np.maximum(f, 1e-15))
            z -= z.mean()
            z = 0.85 * z + 0.15 * rng.normal(size=n)
            f = np.exp(z - z.max())
            f = f / f.sum()
            c, M = compute_conv_and_max(f)
            no_imp = 0

    return best_f, best_M


def coordinate_descent(f0, deadline, rng, max_iter=200000):
    """Cyclic coordinate descent optimizing one element at a time."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    c, M = compute_conv_and_max(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0
    idx = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Pick a coordinate to update (cyclic with randomization)
        if it % 7 == 0:
            idx = rng.integers(0, n)
        else:
            idx = (idx + 1) % n

        j = idx
        # Try different values for a_j
        best_val = f[j]
        best_M_loc = M
        
        # The optimal a_j minimizes max over k of C_k(a_j)
        # C_k depends on a_j for k in [j, 2j+n-1-j] = [j, n+j-1]
        # C_k = a_j * a_{k-j} + (contribution from other terms)
        # So C_k = a_j * a_{k-j} + R_k where R_k doesn't depend on a_j
        
        # Find the range of k affected
        k_lo = j
        k_hi = n + j - 1
        if k_lo >= 2 * n - 1:
            continue
        k_hi = min(k_hi, 2 * n - 2)
        
        # Compute R_k for affected k
        # R_k = C_k - a_j * a_{k-j} (for k-j in range)
        # But we need to be careful: C_k = sum_i a_i * a_{k-i}
        # The terms involving a_j are: a_j * a_{k-j} (i=j) and a_{k-j} * a_j (i=k-j if k-j != j)
        # So the total contribution of a_j to C_k is:
        #   if k-j in [0,n-1] and k-j != j: a_j * a_{k-j} + a_{k-j} * a_j = 2*a_j*a_{k-j}
        #   if k-j == j (k=2j): a_j^2, derivative 2*a_j
        # Actually from earlier: dC_k/da_j = 2*a_{k-j} if 0<=k-j<=n-1
        # So C_k = a_j * (2*a_{k-j}) + R_k for k-j in [0,n-1]
        # And C_k = R_k for k-j not in [0,n-1]
        
        # For the optimal a_j >= 0, we want to minimize max(C_k)
        # C_k = 2*a_{k-j}*a_j + R_k for k in [j, n+j-1] (where k-j in [0,n-1])
        # This is a linear function of a_j for each k, so max is convex in a_j
        # Optimal: equalize the top few
        
        # Simple approach: try a grid of values
        a_j_current = f[j]
        candidates = [0.0, a_j_current * 0.5, a_j_current * 0.75, a_j_current,
                      a_j_current * 1.25, a_j_current * 1.5, a_j_current * 2.0]
        # Add some targeted values
        max_slope = 0
        for k in range(k_lo, k_hi + 1):
            kj = k - j
            if 0 <= kj < n:
                slope = 2.0 * f[kj]
                if slope > max_slope:
                    max_slope = slope
        
        if max_slope > 0:
            # Current peak C_k = a_j_current * slope + R_k
            # R_k at current: C_k - a_j_current * slope_k
            # Try to reduce: a_j_new = a_j_current * (1 - delta)
            for delta in (0.05, 0.1, 0.15, 0.2, 0.3):
                candidates.append(max(0.0, a_j_current * (1 - delta)))
        
        random_val = rng.uniform(0, 2 * a_j_current)
        candidates.append(random_val)
        
        best_local_M = M
        best_local_val = a_j_current
        
        for val in candidates:
            val = max(0.0, val)
            f[j] = val
            # Quick recompute of only affected convolution values
            # Full recompute for correctness (n is small enough)
            c_tmp, M_tmp = compute_conv_and_max(f)
            if M_tmp < best_local_M:
                best_local_M = M_tmp
                best_local_val = val
        
        f[j] = best_local_val
        s = f.sum()
        f = f / s
        c, M = compute_conv_and_max(f)
        
        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > 500:
            no_imp = 0

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage hybrid search:
    1. Evaluate parent and analytical profiles.
    2. Sparse LP optimization at n=512.
    3. Soft-max gradient polishing.
    4. Peak shaving and coordinate descent.
    5. Extension to n=1024.
    6. Final refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent if available
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            sp = pv.sum()
            if sp > 1e-30:
                parent = pv / sp
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # Phase 0: Try analytical profiles at n=512
    n1 = 512
    phase0_end = min(deadline, t_start + 30.0)
    
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            sc = compute_score(p_int)
            try_update(p_int, sc)
            print(f"Interpolated parent to n={n1}: score={sc:.10f}")

    profiles = make_deterministic_profiles(n1)
    for p0 in profiles:
        if time.monotonic() > phase0_end:
            break
        sc = compute_score(p0)
        try_update(p0, sc)
    
    # Add random starts
    for _ in range(10):
        if time.monotonic() > phase0_end:
            break
        p0 = make_smooth_random(n1, rng)
        sc = compute_score(p0)
        try_update(p0, sc)
    
    print(f"After Phase 0: best={best_score:.10f}")

    # Phase 1: Sparse LP optimization at n=512
    phase1_end = min(deadline, t_start + 600.0)
    
    # Sort candidates by score and optimize the top ones
    candidates = []
    if best_seq is not None:
        candidates.append(best_seq.copy())
    
    # Add top analytical profiles
    prof_scores = []
    for p0 in profiles:
        sc = compute_score(p0)
        prof_scores.append((sc, p0))
    prof_scores.sort(key=lambda x: x[0])
    for sc, p0 in prof_scores[:5]:
        candidates.append(p0.copy())
    
    # Add a few random ones
    for _ in range(3):
        candidates.append(make_smooth_random(n1, rng))
    
    per_cand = max(10.0, (phase1_end - t_start) / max(1, len(candidates)))
    
    for i, p0 in enumerate(candidates):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s
        
        per = min(phase1_end, time.monotonic() + per_cand)
        p_best, M_best = lp_sparse_optimize(p0, per, rng, max_iter=8000, top_k=15)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        if i % 3 == 0:
            print(f"  Phase 1 cand {i}/{len(candidates)}: score={sc:.10f}, overall best={best_score:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: Soft-max gradient polishing
    phase2_end = min(deadline, t_start + 1080.0)
    
    if best_seq is not None and best_seq.size == n1:
        per = min(phase2_end, time.monotonic() + 200.0)
        p_best, M_best = softmax_gradient_polish(best_seq, per, rng, lr=0.04)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"Phase 2 polish 1: score={sc:.10f}")
    
    # Polish with different hyperparameters
    if best_seq is not None and best_seq.size == n1:
        per = min(phase2_end, time.monotonic() + 150.0)
        p_best, M_best = softmax_gradient_polish(best_seq, per, rng, lr=0.02, beta_init=0.05 * compute_conv_and_max(best_seq)[1])
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"Phase 2 polish 2: score={sc:.10f}")
    
    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Peak shaving
    phase3_end = min(deadline, t_start + 1320.0)
    
    if best_seq is not None and best_seq.size == n1:
        per = min(phase3_end, time.monotonic() + 120.0)
        p_best, M_best = peak_shaving(best_seq, per, rng, max_iter=50000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"Phase 3 peak shave: score={sc:.10f}")
    
    if best_seq is not None and best_seq.size == n1:
        per = min(phase3_end, time.monotonic() + 120.0)
        p_best, M_best = coordinate_descent(best_seq, per, rng, max_iter=100000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"Phase 3 coord descent: score={sc:.10f}")
    
    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Try n=1024
    n2 = 1024
    phase4_end = min(deadline, t_start + 1620.0)
    
    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            sc = compute_score(p_ext)
            try_update(p_ext, sc)
            print(f"Interpolated to n={n2}: score={sc:.10f}")
            
            per = min(phase4_end, time.monotonic() + 100.0)
            p_best, M_best = lp_sparse_optimize(p_ext, per, rng, max_iter=5000, top_k=20)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"Phase 4 LP (n={n2}): score={sc:.10f}")
            
            if time.monotonic() < phase4_end - 30:
                per2 = min(phase4_end, time.monotonic() + 60.0)
                p_best2, M_best2 = softmax_gradient_polish(p_best, per2, rng, lr=0.03)
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"Phase 4 polish (n={n2}): score={sc2:.10f}")

    # Also try n=640
    n3 = 640
    if best_seq is not None and time.monotonic() < phase4_end - 30:
        p_ext640 = interpolate_profile(best_seq, n3)
        if p_ext640 is not None:
            per = min(phase4_end, time.monotonic() + 50.0)
            p_best, M_best = lp_sparse_optimize(p_ext640, per, rng, max_iter=5000, top_k=15)
            sc = 2.0 * n3 * M_best
            try_update(p_best, sc)
            print(f"Phase 4 LP (n={n3}): score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # Phase 5: Final refinement on absolute best
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 80.0)
        p_best, M_best = lp_sparse_optimize(best_seq, per, rng, max_iter=10000, top_k=10)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        print(f"Phase 5 LP refine: score={sc:.10f}, n={best_seq.size}")
        
        if time.monotonic() < deadline - 10:
            per2 = min(deadline - 5, time.monotonic() + 40.0)
            p_best2, M_best2 = softmax_gradient_polish(best_seq, per2, rng, lr=0.02)
            sc2 = compute_score(p_best2)
            try_update(p_best2, sc2)
            print(f"Phase 5 final polish: score={sc2:.10f}")

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")
        
        # Verify with evaluator
        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned inf, using internal score")
            except Exception as e:
                print(f"Evaluation error: {e}")
        
        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    print(f"Fallback used, score={compute_score(fb):.10f}")
    return [float(x) for x in fb]
