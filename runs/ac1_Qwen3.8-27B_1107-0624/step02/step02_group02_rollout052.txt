def next_pow2(n):

    while k < n:
        k <<= 1
    return max(2, k)

def convolve_fft(a):

        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

def lp_solve(f, M, method='highs'):

    A = build_conv_matrix(f)
    c_obj = -np.ones(n)

    if res.success and res.x is not None:

    return None

**Yet another approach: Use the "method of conditional gradients" (Frank-Wolfe).**

The problem: `min f(a) = max_k (a*a)[k]` s.t. `a in simplex`.

Frank-Wolfe: at each step, solve `min_{v in simplex} ∇f(a) · v`, then update `a = (1-γ)a + γv`.

The gradient of `f(a)` at the peak position `k*` is `2 * a_reversed_shifted_by_k*` (as computed earlier). The linearization `min_{v in simplex} grad · v` picks the coordinate with the minimum gradient entry (since `v` is a vertex of the simplex, it's a standard basis vector).

So Frank-Wolfe would: at each step, find the index `j` that minimizes the gradient at that position, and move mass from `a` toward `e_j` (the `j`-th basis vector).

This is actually a very simple algorithm:
1. Compute gradient at current peak
2. Find the index `j` with the smallest gradient
3. Move a fraction of mass from `a` to `e_j`

This is essentially a "mass redistribution" algorithm. It might be slow but it's simple and guaranteed to make progress.

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based descent with line search at n=512, then gradient-based 
optimization at n=1024, combined with Adam polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_solve(f, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize_linesearch(f_init, deadline, rng, verbose=True):
    """LP-based optimization with line search on step size."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    c_f = convolve_fft(f)
    best_M = float(c_f.max())
    best_f = f.copy()
    t_current = 0.3
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        M = float(convolve_fft(f).max())
        
        # Solve LP
        g = lp_solve(f, M)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()
        
        # Line search: try several step sizes
        best_t = 0.0
        best_val = best_M
        
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7]
        # Also try around the current best
        t_candidates = sorted(set([max(0.02, min(0.95, x)) for x in 
                                     [t_current * 0.5, t_current * 0.8, t_current, 
                                      t_current * 1.2, t_current * 1.5]] + t_candidates))
        
        for t in t_candidates:
            f_try = (1 - t) * f + t * g
            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try < 1e-10:
                continue
            f_try = f_try / s_try
            val = float(convolve_fft(f_try).max())
            if val < best_val - 1e-15:
                best_val = val
                best_t = t
        
        if best_t > 0:
            f_new = (1 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            f_new = f_new / s_new
            M_new = float(convolve_fft(f_new).max())
            
            if M_new < best_M - 1e-14:
                f = f_new
                best_M = M_new
                best_f = f.copy()
                t_current = best_t
                stagnant = 0
            else:
                stagnant += 1
                t_current = max(0.05, t_current * 0.8)
        else:
            stagnant += 1
            t_current = max(0.05, t_current * 0.7)
        
        iteration += 1
        if verbose and iteration % 50 == 0:
            print(f"  LP iter {iteration}: M={best_M:.10f}, t={t_current:.4f}")
        
        # Random restart
        if stagnant > 50:
            noise_scale = 0.05 * np.std(best_f)
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t_current = 0.3
            stagnant = 0

    return best_f


def soft_mx(c, beta):
    """Smooth max with log-sum-exp."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p, N):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 using FFT."""
    n = len(p)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def adam_polish(p0, deadline, rng, lr=0.05, reg=1e-8):
    """Adam optimization in logit space for peak minimization."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    N = next_pow2(2 * n - 1)

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.15 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    start_time = time.monotonic()

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p, N)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Decay beta periodically
        if adam_t % 50 == 0:
            beta = max(1e-14, beta * 0.85)
        
        # Decay learning rate slowly
        if no_imp % 500 == 0 and no_imp > 0:
            lr = max(0.01, lr * 0.95)

        # Restart if stuck
        max_no_imp = max(500, 3 * n)
        if no_imp > max_no_imp:
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_concentrated_profile(n, width_ratio=0.5):
    """Profile concentrated in the middle."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / (width_ratio * 0.5)) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_linear_asym_profile(n, slope=1.0):
    """Asymmetric linear profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_bimodal_profile(n, center1=0.3, center2=0.7, width=0.1):
    """Two-peaked profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center1) / width) ** 2) + np.exp(-((t - center2) / width) ** 2) + 0.1
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    return result / result.sum()
