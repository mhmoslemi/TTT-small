        M = conv0.max()

        # Variables: s[0], ..., s[n-1], M (total n+1 variables)
        # Constraint for each k: 2*sum_j s0[k-j]*s[j] - conv0[k] <= M
        # i.e., 2*sum_j s0[k-j]*s[j] - M <= conv0[k]

        A = np.zeros((2*n-1, n+1))

import numpy as np
import time
from scipy.optimize import linprog

"""
Combined LP successive-linearization and softmax FFT gradient descent
for minimizing 2n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
The LP approach takes large principled steps via linearized constraints,
while gradient descent provides fine-tuning. Multiple lengths and
initializations are explored with adaptive parameter scheduling.
"""


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def softmax_gradient_step(s, beta, eta, N):
    """One step of softmax gradient descent. Returns new s."""
    n = len(s)
    conv = np.convolve(s, s)
    # Numerically stable softmax
    shifted = conv - conv.max()
    exp_vals = np.exp(beta * shifted)
    p_sum = exp_vals.sum()
    if p_sum < 1e-300:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = exp_vals / p_sum
    # Gradient via FFT cross-correlation: grad[m] = 2*sum_j p[j+m]*s[j]
    P = np.fft.rfft(p, N)
    S = np.fft.rfft(s, N)
    cross = np.fft.irfft(P * np.conj(S), N)
    grad = 2.0 * cross[:n]
    # Gradient step with projection
    s_new = s - eta * grad
    s_new = np.maximum(s_new, 0.0)
    s_sum = s_new.sum()
    if s_sum < 1e-15:
        return s.copy()
    return s_new / s_sum


def lp_linearization_step(s0, damping=0.5, time_limit=10.0):
    """
    One LP successive-linearization step.
    Linearizes (s*s)[k] around s0, solves LP to minimize M,
    returns damped update.
    """
    n = len(s0)
    s0 = np.maximum(s0, 0.0)
    s0 = s0 / s0.sum()
    conv0 = np.convolve(s0, s0)
    
    # Build LP: minimize M
    # vars: [s_0, ..., s_{n-1}, M]
    # constraints: 2*sum_j s0[k-j]*s[j] - M <= conv0[k] for k=0,...,2n-2
    #              sum(s) = 1
    #              s[j] >= 0, M >= 0
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            A[k, j] = 2.0 * s0[k - j]
        A[k, n] = -1.0
        b_ub[k] = conv0[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return s0.copy()
    
    if not res.success or res.x is None:
        return s0.copy()
    
    s_lp = np.maximum(res.x[:n], 0.0)
    s_lp_sum = s_lp.sum()
    if s_lp_sum < 1e-15:
        return s0.copy()
    s_lp = s_lp / s_lp_sum
    
    # Damped update
    s_new = (1.0 - damping) * s0 + damping * s_lp
    s_new = np.maximum(s_new, 0.0)
    s_new = s_new / s_new.sum()
    return s_new


def optimize_lp(s_init, deadline, rng, max_iters=500):
    """LP successive-linearization optimization."""
    n = len(s_init)
    s = np.maximum(s_init, 0.0).copy()
    s = s / s.sum()
    
    best_s = s.copy()
    best_score = fast_score(s)
    damping = 0.5
    consecutive_worse = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        s_new = lp_linearization_step(s, damping=damping, time_limit=5.0)
        score_new = fast_score(s_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_s = s_new.copy()
            s = s_new
            consecutive_worse = 0
            # Slightly increase damping when making progress
            damping = min(damping * 1.02, 0.9)
        else:
            s = s_new
            consecutive_worse += 1
            if consecutive_worse >= 10:
                # Reduce damping for smaller steps
                damping = max(damping * 0.7, 0.05)
                consecutive_worse = 0
            if consecutive_worse >= 50:
                # Restart from best
                s = best_s.copy()
                damping = 0.3
                consecutive_worse = 0
    
    return best_s, best_score

        s = np.maximum(s, 0)
        if s.sum() > 0.01:
            inits.append(s / s.sum())
    
    inits.append(np.ones(n) / n)
    
    # Cosine window
    s = np.cos(np.pi * x / 2) ** 2
    inits.append(np.maximum(s, 0.01) / np.maximum(s, 0.01).sum())
    
    # Hann
    s = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(np.maximum(s, 0.01) / np.maximum(s, 0.01).sum())
    
    # Raised cosine variants
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        s = np.cos(np.pi * (x - 0.5) * alpha) ** 2
        inits.append(np.maximum(s, 0.01) / np.maximum(s, 0.01).sum())
    
    # Random
    for _ in range(4):
        s = rng.random(n) + 0.1
        inits.append(s / s.sum())
    
    # Exponential
    for a in [1.0, 2.0, 3.0, 5.0]:
        s = np.exp(-a * np.arange(n) / n)
        inits.append(s / s.sum())
    
    # Two-sided exponential
    for a in [1.0, 2.0, 3.0]:
        s = np.exp(-a * np.abs(np.arange(n) - n / 2) / n)
        inits.append(s / s.sum())
    
    # Modulated
    for freq in [2, 3, 5, 8, 13]:
        s = 1 + 0.5 * np.cos(2 * np.pi * freq * x)
        inits.append(np.maximum(s, 0.01) / np.maximum(s, 0.01).sum())
    
    # Sort by score
    scored = [(fast_score(s), i, s) for i, s in enumerate(inits)]
    scored.sort(key=lambda t: t[0])
    return [s for _, _, s in scored]
