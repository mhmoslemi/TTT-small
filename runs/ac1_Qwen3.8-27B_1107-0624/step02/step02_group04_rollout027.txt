```python
from scipy.optimize import linprog

def solve_lp(f, M):
    n = len(f)
    # Maximize sum(g)
    # Subject to: g >= 0, (f*g)[m] <= M for m = 0, ..., 2n-2
    # c = -1 (we minimize -sum(g))
    c = -np.ones(n)
    # (f*g)[m] = sum_i f[i] * g[m-i] (with bounds)
    # For each m, the constraint is: sum_i A[m,i] * g[i] <= M
    # where A[m,i] = f[i] if 0 <= m-i < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        for i in range(n):
            j = m - i
            if 0 <= j < n:
                A_ub[m, i] = f[i]
    b_ub = np.full(2*n-1, M)
    # g >= 0
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x
```

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _soft_max_weights(c, beta):
    """Compute soft-max weights and the soft-max value."""
    c = np.asarray(c, dtype=np.float64)

def _cross_corr_weights(w, p):
    """Compute cross-correlation of w with p, aligned to same length as p."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _solve_lp_escape(f, M, max_iter=50000):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[m] <= M for all m.
    Returns the optimal g (normalized to sum 1) or None if infeasible.
    """
    n = len(f)
    # Build constraint matrix A (2n-1 rows, n cols)
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    # This is a Toeplitz-like matrix
    
    # Use sparse for efficiency
    rows = []
    cols = []
    data = []
    for m in range(2 * n - 1):
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        for j in range(j_start, j_end + 1):
            rows.append(m)
            cols.append(j)
            data.append(f[m - j])
    
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=(0, None), 
                  method='highs', options={'maxiter': max_iter})
    
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None
