import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

def _full_convolve(a):
    """Full convolution using FFT for speed."""
    n = len(a)
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0)  # numerical errors can give tiny negatives

def _score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2n * max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a)
    return 2.0 * n * c.max()

def _lp_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns optimal b (unnormalized) or None if infeasible."""
    n = len(a)
    # Build constraint matrix: conv(a,b)[k] = sum_i a[k-i] * b[i]
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # This is a banded (Toeplitz) matrix

    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2*n-1, M)

    if res.status == 0:

LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses iterative LP direction finding combined with gradient polishing.

# Helper functions (top-level, no closures)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve_fft(a):
    n = len(a)
    if n <= 128:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s <= 0:
        return np.inf
    c = _full_convolve_fft(a)
    return 2.0 * n * c.max() / (s * s)

def _build_conv_matrix(a):
    """Build sparse matrix A where A[k,i] = a[k-i] if 0<=k-i<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for d in range(n):
        i_start = 0
        i_end = min(n-1, 2*n-2-d)
        if i_start > i_end:
            continue
        i_vals = np.arange(i_start, i_end+1)
        k_vals = i_vals + d
        rows.append(k_vals)
        cols.append(i_vals)

    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)

    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                  method='highs', options={'disp': False})
    if res.status == 0:
        return res.x
    return None

    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    
    c = _full_convolve_fft(a)
    maxc = c.max()
    beta = beta_init * maxc
    best_a = a.copy()
    best_score = maxc
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Soft-max weights
        c_max = c.max()
        x = (c - c_max) / beta
        ex = np.exp(x)
        s = ex.sum()
        w = ex / s
        
        # Gradient: 2 * cross_corr(w, a)
        # cross_corr(w, a)[i] = sum_j w[j] * a[i-j] (with appropriate bounds)
        # Using FFT: this is the convolution of w with reversed a
        N = _next_pow2(2*n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        gp = 2.0 * np.fft.irfft(fw * fa_rev, N)[n-1:2*n-1]
        
        # Grad on z (logit space)
        dot = np.dot(a, gp)
        gz = a * (gp - dot)
        gz -= gz.mean()
        
        # Adam update
        t += 1
        m = b1*m + (1-b1)*gz
        v = b2*v + (1-b2)*gz*gz
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z -= lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        c = _full_convolve_fft(a)
        maxc = c.max()
        
        if maxc < best_score - 1e-12:
            best_score = maxc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decay beta
        if it % 20 == 0:
            beta = max(1e-10, beta * 0.85)
        
        if no_imp > 200:
            break
    
    return best_a, best_score

"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding with soft-max gradient descent.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Fix tiny negative values from FFT numerical errors
    if c.size > 0 and c.min() < 0:
        c = c - c.min()
    return c

def _score_from_norm1(a):
    """Score for sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = _full_conv(a)
    if c.size == 0:
        return np.inf
    return 2.0 * n * float(c.max())

def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for d in range(n):
        i_end = min(n - 1, 2 * n - 2 - d)
        if i_end < 0:
            continue
        i_vals = np.arange(0, i_end + 1)
        k_vals = i_vals + d
        rows.append(k_vals)
        cols.append(i_vals)
        vals.append(np.full(len(i_vals), a[d]))
    if not rows:
        return csr_matrix((2 * n - 1, n))
    return csr_matrix((np.concatenate(vals), (np.concatenate(rows), np.concatenate(cols))),
                      shape=(2 * n - 1, n))

def _lp_solve(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'disp': False, 'presolve': True})
    if res.status == 0 and res.x is not None:
        return res.x
    return None

def _make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_linear(n, eta):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_cubic(n, a1, a3):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _lp_optimize(a, deadline, max_iters=10000, verbose=True, print_interval=100):
    """Iteratively improve using LP direction finding."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-30:

    a = a / s

    best_M = M

"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses iterative LP direction finding combined with soft-max gradient descent.
Multi-scale approach: 512 -> 1024.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.size > 0 and c.min() < 0:
        c = c - c.min()
    return c


def _score_norm1(a):
    """Score for sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = _full_conv(a)
    if c.size == 0:
        return np.inf
    return 2.0 * n * float(c.max())


def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for d in range(n):
        i_end = min(n - 1, 2 * n - 2 - d)
        if i_end < 0:
            continue
        i_vals = np.arange(0, i_end + 1)
        k_vals = i_vals + d
        rows.append(k_vals)
        cols.append(i_vals)
        vals.append(np.full(len(i_vals), a[d]))
    if not rows:
        return csr_matrix((2 * n - 1, n))
    return csr_matrix(
        (np.concatenate(vals), (np.concatenate(rows), np.concatenate(cols))),
        shape=(2 * n - 1, n)
    )


def _lp_solve(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'disp': False, 'presolve': True})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def _make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear(n, eta):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic(n, a1, a3):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic(n, a1, a2):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _lp_optimize(a, deadline, max_iters=10000, verbose=True, print_interval=200):
    """Iteratively improve using LP direction finding with line search."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-30:
        return a, np.inf
    a = a / s

    c = _full_conv(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    no_imp = 0
    total_iters = 0
    t_values = [0.5, 0.3, 0.7, 0.4, 0.6, 0.2, 0.8, 0.45, 0.55, 0.9, 1.0, 0.1]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = _lp_solve(a, M)
        if b is None:
            no_imp += 1
            if no_imp > 50:
                break
            continue

        sum_b = float(b.sum())
        if sum_b <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 50:
                break
            continue

        b = b / sum_b

        improved = False
        for t in t_values:
            a_try = (1.0 - t) * a + t * b
            M_try = float(_full_conv(a_try).max())
            if M_try < M - 1e-14:
                a = a_try
                M = M_try
                improved = True
                if M < best_M - 1e-14:
                    best_M = M
                    best_a = a.copy()
                no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp > 80:
                break

        total_iters += 1
        if verbose and total_iters % print_interval == 0:
            print(f"    LP iter {total_iters}: M={M:.12f}, best={best_M:.12f}, S={sum_b:.5f}")

    return best_a, best_M


def _grad_optimize(a, deadline, beta_frac=0.08, lr=0.05, max_iters=20000,
                   verbose=True, print_interval=1000, rng=None):
    """Soft-max gradient descent in logit space with Adam."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    c = _full_conv(a)
    maxc = float(c.max())
    beta = max(1e-12, beta_frac * maxc)
    best_a = a.copy()
    best_M = maxc
    no_imp = 0

    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    N = _next_pow2(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c_max = float(c.max())
        x = (c - c_max) / beta
        x = np.clip(x, -50, 0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex <= 0 or not np.isfinite(s_ex):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        fw = np.fft.rfft(w[::-1], N)
        fa = np.fft.rfft(a, N)
        gp = 2.0 * np.fft.irfft(fa * fw, N)[:2 * n - 1]
        gp = gp[n - 1:2 * n - 1]

        dot = float(np.dot(a, gp))
        gz = a * (gp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = np.log(best_a)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            a = best_a.copy()
            c = _full_conv(a)
            maxc = float(c.max())
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()

        c = _full_conv(a)
        maxc = float(c.max())

        if maxc < best_M - 1e-14:
            best_M = maxc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 30 == 0:
            beta = max(1e-12, beta * 0.82)

        if verbose and it % print_interval == 0:
            print(f"    Grad iter {it}: M={maxc:.12f}, best={best_M:.12f}, beta={beta:.2e}")

        if no_imp > 400:
            z = np.log(best_a)
            z -= z.mean()
            z += 0.15 * rng.standard_normal(n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_conv(a)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            no_imp = 0

    return best_a, best_M

                    M_try = float(_full_conv(a_try).max())
                    if M_try < M - 1e-14:
                        a = a_try
                        M = M_try
                        improved = True
                        break

                if improved:
                    if M < best_M - 1e-14:
                        best_M = M
                        best_a = a.copy()
                    no_imp = 0
                    grad_streak = 0
                    lp_streak += 1
                    continue
                else:
                    lp_streak += 1
            else:
                lp_streak = 0

        # Gradient step (single Adam step)
        c_max = float(c.max())
        beta = max(1e-12, 0.05 * c_max)
        x = (c - c_max) / beta
        x = np.clip(x, -50, 0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex > 0 and np.isfinite(s_ex):
            w = ex / s_ex
        else:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w[::-1], N)
        fa = np.fft.rfft(a, N)
        gp = 2.0 * np.fft.irfft(fa * fw, N)[:2 * n - 1]
        gp = gp[n - 1:2 * n - 1]

        # Simple gradient step with momentum
        a_new = a - 0.01 * gp
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()
        M_new = float(_full_conv(a_new).max())

        if M_new < M - 1e-14:
            a = a_new
            M = M_new
            lp_streak = 0
            grad_streak += 1
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
            no_imp = 0
        else:
            grad_streak += 1
            no_imp += 1

        if verbose and it % 500 == 0:
            print(f"    Hybrid iter {it}: M={M:.12f}, best={best_M:.12f}, mode={'LP' if use_lp else 'Grad'}")

        if no_imp > 500:
            break

    return best_a, best_M
