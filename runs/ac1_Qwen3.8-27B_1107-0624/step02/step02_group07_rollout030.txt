def fast_score(s):
    """Compute 2n * max(conv(s,s)) / (sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)


def compute_subgradient(s, active_list):
    """Compute mean subgradient of max_k conv(s,s)[k] at s, averaged over active_list."""
    n = len(s)
    grad = np.zeros(n)
    for k in active_list:
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if hi1 >= lo1:
            idx = np.arange(lo1, hi1 + 1)
            grad[lo1:hi1 + 1] += s[k - idx]
        lo2 = max(0, k)
        hi2 = min(k + n - 1, n - 1)
        if hi2 >= lo2:
            idx = np.arange(lo2, hi2 + 1)
            grad[lo2:hi2 + 1] += s[idx - k]
    grad /= len(active_list)
    return grad


def variance_gradient(s, conv):
    """Compute gradient of sum(conv[k]^2) w.r.t. s. O(n^2) direct."""
    n = len(s)
    M = 2 * n - 1
    grad = np.zeros(n)
    for j in range(n):
        ks1_start = j
        ks1_end = min(j + n, M)
        if ks1_end > ks1_start:
            ks1 = np.arange(ks1_start, ks1_end)
            grad[j] += np.dot(conv[ks1], s[ks1 - j])
        ks2_start = max(j - n + 1, 0)
        ks2_end = min(j + 1, M)
        if ks2_end > ks2_start:
            ks2 = np.arange(ks2_start, ks2_end)
            grad[j] += np.dot(conv[ks2], s[j - ks2])
    return 2.0 * grad


def soft_max_gradient(s, conv, beta=20.0):
    """Soft approximation of max-gradient using exponential weighting."""
    n = len(s)
    M = 2 * n - 1
    max_conv = conv.max()
    if max_conv <= 0:
        return np.zeros(n)
    w = np.exp(beta * (conv - max_conv) / max_conv)
    w_sum = w.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    w = w / w_sum
    grad = np.zeros(n)
    for j in range(n):
        ks1_start = j
        ks1_end = min(j + n, M)
        if ks1_end > ks1_start:
            ks1 = np.arange(ks1_start, ks1_end)
            grad[j] += np.dot(w[ks1], s[ks1 - j])
        ks2_start = max(j - n + 1, 0)
        ks2_end = min(j + 1, M)
        if ks2_end > ks2_start:
            ks2 = np.arange(ks2_start, ks2_end)
            grad[j] += np.dot(w[ks2], s[j - ks2])
    return grad
