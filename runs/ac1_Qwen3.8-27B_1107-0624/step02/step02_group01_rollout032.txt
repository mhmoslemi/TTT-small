def compute_max_conv(a):
    c = fft_convolve(a)
    return float(c.max())

def frequency_perturb(f, rng, num_modes=5):
    """Perturb in frequency domain to explore different structures."""
    n = f.size
    N = 1
    while N < n:
        N <<= 1
    F = np.fft.rfft(f, N)
    # Add random phase perturbation to some modes
    for _ in range(num_modes):
        k = rng.integers(1, N // 2)
        amp = rng.uniform(0.01, 0.1)
        phase = rng.uniform(0, 2 * np.pi)
        F[k] += amp * np.exp(1j * phase)
        if k < N - k:  # Hermitian symmetry
            F[N - k] = np.conj(F[k])
    f_new = np.fft.irfft(F, N)[:n]
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-30:
        return f
    return f_new / s

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    if n > 1024:
        # Too large for dense LP, use a simplified version
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            return res.x
    except:
        pass
    return None

def make_linear_profile(n, eta):

    a = 1.0 + eta * (t - 0.5)

def make_cubic_profile(n, a1, a3):

    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_smooth_random(n, rng, n_harm=10):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.6) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_max_conv(a):
    c = fft_convolve(a)
    return float(c.max())


def softmax_max(c, beta):
    c_max = float(c.max())
    if not np.isfinite(c_max):
        return c_max, np.zeros_like(c)
    x = (c - c_max) / beta
    ex = np.exp(np.clip(x, -60.0, 0.0))
    s = float(ex.sum())
    if s < 1e-30:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    L = c_max + beta * np.log(s)
    w = ex / s
    return L, w


def compute_gradient(f, beta, N_fft):
    """Compute softmax-approximated gradient of max(conv(f,f)) wrt f."""
    c = fft_convolve(f)
    L, w = softmax_max(c, beta)
    F = np.fft.rfft(f, N_fft)
    W = np.fft.rfft(w, N_fft)
    cross = np.fft.irfft(np.conj(F) * W, N_fft)[:f.size]
    grad = 2.0 * cross
    return L, grad, c

def solve_lp_direction(f, M, n):
    if n > 1200:
        return None
    try:
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f
        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t_adam = 0
                no_imp = 0

        f = f_new

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Evaluate parent
    2. Gradient optimization at n=512 from diverse starts
    3. Symmetric variant optimization
    4. Extended optimization with LP escapes on top candidates
    5. Multi-length exploration (256, 768, 1024, 2048)
    6. Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent: n={parent.size}, score={sc:.10f}")

    # Phase 1: Gradient optimization at n=512
    n1 = 512
    phase1_end = min(deadline, t_start + 500.0)

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(('parent', p_int))
        # Symmetrized parent
        p_sym = symmetric_project(p_int)
        starts.append(('parent_sym', p_sym))
        # Slightly perturbed parent
        p_pert = parent + 0.01 * rng.normal(size=n1)
        p_pert = np.maximum(p_pert, 0.0)
        p_pert = p_pert / p_pert.sum()
        starts.append(('parent_pert', p_pert))

    # Deterministic profiles
    starts.append(('lin_1.5', make_linear_profile(n1, 1.5)))
    starts.append(('lin_2.0', make_linear_profile(n1, 2.0)))
    starts.append(('lin_2.5', make_linear_profile(n1, 2.5)))
    starts.append(('lin_3.0', make_linear_profile(n1, 3.0)))
    starts.append(('cub_1', make_cubic_profile(n1, 1.5, -1.0)))
    starts.append(('cub_2', make_cubic_profile(n1, 2.0, -1.5)))
    starts.append(('cub_3', make_cubic_profile(n1, 1.8, -1.2)))
    starts.append(('quar_1', make_quartic_profile(n1, 1.5, 0.3, -0.8)))
    starts.append(('quar_2', make_quartic_profile(n1, 2.0, -0.2, -1.2)))

    # Random smooth
    for i in range(4):
        starts.append((f'srand_{i}', make_smooth_random(n1, rng, n_harm=6 + 2 * i)))

    candidates = []
    per_start = max(10.0, (phase1_end - t_start) / max(1, len(starts)))

    for i, (name, p0) in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s

        per = min(phase1_end, time.monotonic() + per_start)
        p_best, M_best = gradient_descent_optimize(
            p0, per, rng,
            beta_init=0.15, lr_init=0.005,
            use_adam=True, max_iter=30000
        )
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates.append((sc, p_best))
        print(f"  P1 [{name}]: score={sc:.10f} (best={best_score:.10f})")

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: Symmetric variant on best
    phase2_end = min(deadline, t_start + 700.0)
    if candidates:
        sc0, p0 = candidates[0]
        per = min(phase2_end, time.monotonic() + 100.0)
        p_best, M_best = gradient_descent_symmetric(
            p0, per, rng, beta_init=0.15, lr_init=0.004
        )
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  P2 symmetric: score={sc:.10f} (best={best_score:.10f})")

    # Phase 3: Extended optimization with LP escapes on top-3
    phase3_end = min(deadline, t_start + 1100.0)
    top_k = min(3, len(candidates))
    per_cand = max(40.0, (phase3_end - time.monotonic()) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase3_end:
            break
        sc, p0 = candidates[i]
        per = min(phase3_end, time.monotonic() + per_cand)

        p_best, M_best = gradient_descent_optimize(
            p0, per, rng,
            beta_init=0.10, lr_init=0.003,
            use_adam=True, max_iter=80000,
            restart_every=500,
            lp_callback=solve_lp_direction
        )
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  P3 ext cand {i}: score={sc:.10f} (best={best_score:.10f})")

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Multi-length exploration
    phase4_end = min(deadline, t_start + 1550.0)

    for n_try in [256, 768, 1024, 1536, 2048, 3072, 4096]:
        if time.monotonic() > phase4_end - 20:
            break
        if best_seq is None:
            continue

        p_ext = interpolate_profile(best_seq, n_try)
        if p_ext is None:
            continue

        remaining = phase4_end - time.monotonic()
        # Allocate more time to promising lengths
        if n_try in (1024, 2048):
            per = min(remaining - 10, max(40.0, remaining * 0.25))
        elif n_try in (768, 1536, 3072, 4096):
