```python
import numpy as np
import cvxpy as cp
import time

def solve_lp(f, M, solver=None):
    """Solve the LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix: (f*g)_k = sum_i f_i * g_{k-i}
    # This is a matrix A such that A @ g gives the convolution f*g
    # A has shape (2n-1, n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver)
    except:
        prob.solve()
    
    if g.value is None:
        return None
    return g.value
```

```python
    return A


def convolve_fast(a, b):
    """Fast linear convolution using FFT for large arrays."""
    n = len(a)
    m = len(b)
    if n * m <= 4096:
        return np.convolve(a, b)
    N = 1
    while N < n + m - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    # Fix tiny negative values from FFT numerical error
    if c.min() < 0 and c.min() > -1e-10:
        c = np.maximum(c, 0)
    return c


def score_fn(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = convolve_fast(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return None
    return a / s


def golden_section_min(val_fn, a_lo, a_hi, tol=1e-12, max_iter=100):
    """Golden section minimization of val_fn on [a_lo, a_hi]."""
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c_pt = a_hi - gr * (a_hi - a_lo)
    d_pt = a_lo + gr * (a_hi - a_lo)
    fc = val_fn(c_pt)
    fd = val_fn(d_pt)
    for _ in range(max_iter):
        if a_hi - a_lo < tol:
            break
        if fc < fd:
            a_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = a_hi - gr * (a_hi - a_lo)
            fc = val_fn(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_lo + gr * (a_hi - a_lo)
            fd = val_fn(d_pt)
    t_opt = 0.5 * (a_lo + a_hi)
    return t_opt, val_fn(t_opt)


def mix_line_search(ff, fg, gg):
    """
    Find optimal t for h = (1-t)*f + t*g given precomputed convolutions.
    ff = f*f, fg = f*g, gg = g*g. All assumed normalized (sum=1).
    Returns (t_opt, min_max_val).
    """
    def val_at_t(t):
        vals = (1.0 - t) ** 2 * ff + 2.0 * t * (1.0 - t) * fg + t * t * gg
        return float(vals.max())
    t_opt, min_val = golden_section_min(val_at_t, 0.0, 1.0, tol=1e-13, max_iter=80)
    return t_opt, min_val


def solve_lp_for_direction(f, M, solver=None):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)_k <= M for all k.
    Returns optimal g or None if infeasible.
    """
    n = len(f)
    A = build_conv_matrix(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    if solver is not None:
        try:
            prob.solve(solver=solver)
        except Exception:
            prob.solve()
    else:
        prob.solve()
    if g.value is None:
        return None
    g_val = np.asarray(g.value, dtype=np.float64)
    g_val = np.maximum(g_val, 0.0)
    s = float(g_val.sum())
    if s < 1e-15:
        return None
    return g_val / s


def solve_lp_for_direction_dual(f, M, solver=None):
    """
    Alternative LP formulation using dual structure.
    Primal: max 1^T g s.t. A g <= M*1, g >= 0
    Dual:   min M * 1^T y s.t. A^T y >= 1, y >= 0
    We solve the primal directly (smaller in most cases).
    """
    return solve_lp_for_direction(f, M, solver=solver)


def subgradient_step(a, lr):
    """
    Subgradient step for minimizing max(a*a) subject to sum(a)=1, a>=0.
    Returns updated a.
    """
    n = len(a)
    aa = convolve_fast(a, a)
    max_val = aa.max()
    # Find active constraints (within tolerance)
    tol = max(1e-10, 0.01 * max_val)
    active = np.where(aa >= max_val - tol)[0]
    # Compute subgradient: average gradient over active constraints
    # grad_i of (a*a)_k is 2*a_{k-i} (with a_j=0 out of range)
    sg = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                sg[i] += 2.0 * a[j]
    sg /= len(active)
    # Update
    a_new = a - lr * sg
    # Project onto {a >= 0, sum(a) = 1}
    a_new = project_simplex(a_new)
    return a_new, max_val


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w /= w.sum()
    return w


def lp_alternating_optimize(f0, deadline, solver=None, max_iter=500):
    """
    LP-based alternating optimization.
    f0: initial sequence (will be normalized internally).
    Returns (best_f, best_score).
    """
    f = normalize(f0)
    if f is None:
        return f0, np.inf
    n = len(f)

    ff = convolve_fast(f, f)
    M = float(ff.max())
    best_f = f.copy()
    best_score = float(2.0 * n * M)
    no_improve = 0
    base_steps = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
    step_idx = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Solve LP for direction
        g = solve_lp_for_direction(f, M, solver=solver)
        if g is None:
            # Try with slightly larger M
            g = solve_lp_for_direction(f, M * 1.01, solver=solver)
            if g is None:
                break

        # Precompute convolutions for line search
        fg = convolve_fast(f, g)
        gg = convolve_fast(g, g)

        # Find optimal mixing parameter
        t_opt, min_max_val = mix_line_search(ff, fg, gg)

        # Update
        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = normalize(f_new)
        if f_new is None:
            break

        ff_new = convolve_fast(f_new, f_new)
        M_new = float(ff_new.max())
        score_new = float(2.0 * n * M_new)

        if not np.isfinite(score_new):
            break

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new.copy()
            ff = ff_new
            M = M_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            if no_improve > 3:
                # Try multiple step sizes
                improved = False
                for frac in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                    t_try = t_opt * frac
                    f_try = (1.0 - t_try) * f + t_try * g
                    f_try = normalize(f_try)
                    if f_try is None:
                        continue
                    ff_try = convolve_fast(f_try, f_try)
                    M_try = float(ff_try.max())
                    sc_try = float(2.0 * n * M_try)
                    if sc_try < best_score - 1e-14:
                        best_score = sc_try
                        best_f = f_try.copy()
                        f = f_try.copy()
                        ff = ff_try
                        M = M_try
                        no_improve = 0
                        improved = True
                        break
                if not improved:
                    # Reset with perturbation
                    f = best_f + 0.01 * np.random.default_rng(it).normal(size=n)
                    f = normalize(f)
                    if f is None:
                        f = best_f.copy()
                    ff = convolve_fast(f, f)
                    M = float(ff.max())
                    no_improve = 0

        if no_improve > 50:
            # Perturb and restart
            rng = np.random.default_rng(it * 7 + 3)
            f = best_f.copy()
            f = f + 0.05 * rng.normal(size=n)
            f = normalize(f)
            if f is None:
                f = best_f.copy()
            ff = convolve_fast(f, f)
            M = float(ff.max())
            no_improve = 0

    return best_f, best_score


def subgradient_optimize(f0, deadline, lr_init=0.01, decay=0.999):
    """
    Subgradient method for minimizing max(a*a) subject to sum(a)=1, a>=0.
    """
    f = normalize(f0)
    if f is None:
        return f0, np.inf
    n = len(f)

    ff = convolve_fast(f, f)
    best_score = float(2.0 * n * ff.max())
    best_f = f.copy()
    lr = lr_init

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        aa = convolve_fast(f, f)
        max_val = float(aa.max())
        score = float(2.0 * n * max_val)

        if score < best_score - 1e-12:
            best_score = score
            best_f = f.copy()

        # Subgradient step
        tol = max(1e-12, 0.005 * max_val)
        active = np.where(aa >= max_val - tol)[0]

        # Compute subgradient efficiently
        sg = np.zeros(n)
        for k in active:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                sg[i] += 2.0 * a_idx(f, k - i)
        sg /= float(len(active))

        f_new = f - lr * sg
        f_new = project_simplex(f_new)
        f = f_new
        lr *= decay

        if lr < 1e-8:
            break

    return best_f, best_score


def a_idx(a, i):
    """Safe index with zero padding."""
    n = len(a)
    if 0 <= i < n:
        return a[i]
    return 0.0


def coordinate_descent_polish(f0, deadline, n_sweeps=50):
    """
    Coordinate descent: try adjusting each element individually.
    """
    f = normalize(f0)
    if f is None:
        return f0, np.inf
    n = len(f)
    best_f = f.copy()
    best_score = score_fn(f)
    no_improve = 0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        rng = np.random.default_rng(sweep * 13 + 7)
        for i in rng.permutation(n):
            if time.monotonic() > deadline:
                break
            orig = f[i]
            best_local = orig
            best_local_sc = best_score
            # Try multiplicative adjustments
            for factor in [1.1, 1.05, 1.02, 0.98, 0.95, 0.9, 1.2, 0.8]:
                new_val = orig * factor
                if new_val < 0 or new_val > 10.0:
                    continue
                f[i] = new_val
                f = normalize(f)
                if f is None:
                    break
                sc = score_fn(f)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = new_val
            f[i] = best_local
            f = normalize(f)
            if best_local_sc < best_score:
                best_score = best_local_sc
                best_f = f.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1
        if not improved:
            if no_improve > 2 * n:
                break
            no_improve = 0

    return best_f, best_score


def get_initializations(n, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0.0, 1.0, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

    # Exponential profiles
    for eta in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(eta * (t - 0.5))
        inits.append(a / a.sum())

    # Power profiles
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(t ** p + (1-t)**p, 1e-10)
        inits.append(a / a.sum())

    # Hann window
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())

    # Raised cosine variants
    for bw in [0.3, 0.5, 0.7, 1.0]:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) / bw))
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

    # Sigmoid profiles
    for eta in [3.0, 5.0, 8.0, 12.0, 20.0]:
        a = 1.0 / (1.0 + np.exp(-eta * (t - 0.5)))
        inits.append(a / a.sum())

    # Two-block profiles
    for frac in [0.3, 0.4, 0.5, 0.6]:
        for ratio in [0.5, 0.7, 1.5, 2.0]:
            a = np.ones(n)
            k = int(n * frac)
            a[k:] = ratio
            inits.append(a / a.sum())

    # Cubic
    u = 2.0 * t - 1.0
    for a1, a3 in [(1.0, -0.5), (1.5, -1.0), (2.0, -1.5), (0.5, -0.2)]:
        a = 1.0 + a1 * u + a3 * u ** 3
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

    # Parabolic (concave)
    for amp in [0.3, 0.5, 0.7, 1.0]:
        a = 1.0 + amp * (1.0 - 4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

    # Random smooth
    rng = np.random.default_rng(42)
    for _ in range(5):
        # Smooth random via convolution with Gaussian
        raw = rng.uniform(0.5, 1.5, n)
        # Smooth
        kernel = np.exp(-0.5 * (np.arange(-10, 11) / 3.0) ** 2)
        kernel /= kernel.sum()
        a = np.convolve(raw, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

    # Parent solution
    if parent is not None:
        p = normalize(parent)
        if p is not None and len(p) == n:
            inits.append(p)
        elif p is not None:
            # Interpolate
            x0 = np.linspace(0, 1, len(p))
            x1 = np.linspace(0, 1, n)
            p_interp = np.interp(x1, x0, p)
            p_interp = normalize(p_interp)
            if p_interp is not None:
                inits.append(p_interp)

    return inits


def detect_best_solver():
    """Detect the best available LP solver."""
    available = cp.installed_solvers()
    # Prefer order for LP
    preference = ['GUROBI', 'MOSEK', 'CLARABEL', 'ECOS', 'GLPK', 'CBC', 'XPRESS', 'PDLP', 'SCIP', 'GLOP']
    for solver in preference:
        if solver in available:
            return solver
    return None


def socp_direct_solve(n, deadline, solver=None):
    """
    Direct SOCP formulation:
    Minimize t subject to a^T Q_k a <= t for all k, sum(a)=1, a>=0.
    Only practical for small n.
    """
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable(nonneg=True)

    constraints = [cp.sum(a) == 1]

    # Build Q_k matrices and add SOC constraints
    for k in range(2 * n - 1):
        # (a*a)_k = sum_i a_i * a_{k-i}
        # = a^T Q_k a where Q_k is symmetric
        # Q_k[i,j] = 0.5 if i+j=k and i!=j, 1 if 2i=k
        # For SOC: a^T Q_k a <= t means ||Q_k^{1/2} a|| <= sqrt(t)
        # But Q_k may not be full rank, so use a different approach
        # Actually, (a*a)_k = sum_{i: valid} a_i * a_{k-i}
        # This is a convex quadratic (Q_k is PSD)
        # We can use cp.quad_form(a, Q_k) <= t
        Qk = np.zeros((n, n))
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if i == j:
                    Qk[i, j] = 1.0
                else:
                    Qk[i, j] = 0.5
                    Qk[j, i] = 0.5
        # Check PSD
        # Add constraint using sumsquares for rank-1 part
        # (a*a)_k = sum_i a_i * a_{k-i}
        # Let's just use the explicit quadratic form
        constraints.append(cp.quad_form(a, cp.psd_wrap(Qk)) <= t)

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)

    try:
        if solver:
            prob.solve(solver=solver)
        else:
            prob.solve()
    except Exception as e:
        print(f"  SOCP solve failed: {e}")
        return None, np.inf

    if a.value is None:
        return None, np.inf

    a_val = np.maximum(a.value, 0)
    a_val = normalize(a_val)
    if a_val is None:
        return None, np.inf
    sc = score_fn(a_val)
    return a_val, sc

        a = a / a.sum()
        per = min(phase4_end, time.monotonic() + 60.0)
        result_f, result_sc = lp_alternating_optimize(a, per, solver=solver, max_iter=80)
        update_best(result_f, result_sc, f"LP_n1024_lin{eta}")

    print(f"  After Phase 4: best={best_score:.10f}")

    # Phase 5: Coordinate descent polish
    print("Phase 5: Coordinate descent polish")
    phase5_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and best_seq.size <= 1024:
        per = min(phase5_end, time.monotonic() + 100.0)
        result_f, result_sc = coordinate_descent_polish(best_seq, per, n_sweeps=30)
        update_best(result_f, result_sc, "CD_polish")

    # Second round of LP on the CD-polished result
    if best_seq is not None and best_seq.size <= 512 and time.monotonic() < phase5_end:
        per = min(phase5_end, time.monotonic() + 60.0)
        result_f, result_sc = lp_alternating_optimize(best_seq, per, solver=solver, max_iter=100)
        update_best(result_f, result_sc, "LP_after_CD")

    print(f"  After Phase 5: best={best_score:.10f}")

    # Phase 6: Subgradient refinement on best
    print("Phase 6: Subgradient refinement")
    phase6_end = min(deadline, t_start + 1750.0)
    if best_seq is not None and best_seq.size <= 512:
        per = min(phase6_end, time.monotonic() + 50.0)
        result_f, result_sc = subgradient_optimize(best_seq, per, lr_init=0.005, decay=0.9995)
        update_best(result_f, result_sc, "SG_refine")

    print(f"  After Phase 6: best={best_score:.10f}")

    # Phase 7: Final polish
    print("Phase 7: Final polish")
    phase7_end = deadline
    if best_seq is not None and best_seq.size <= 512:
        # One more CD round
        per = min(phase7_end, time.monotonic() + 30.0)
        result_f, result_sc = coordinate_descent_polish(best_seq, per, n_sweeps=20)
        update_best(result_f, result_sc, "final_CD")

    # Final verification
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s
            final_score = score_fn(final_seq)
            print(f"  FINAL (internal): score={final_score:.10f} n={final_seq.size}")

            # Verify with exact evaluation
            try:
                ev_score = float(evaluate_sequence(final_seq.tolist()))
                if np.isfinite(ev_score):
                    print(f"  FINAL (evaluated): score={ev_score:.10f}")
                    if ev_score < final_score:
                        final_score = ev_score
            except Exception as e:
                print(f"  Evaluation error: {e}")

            return [float(x) for x in final_seq]

    # Fallback
    print("  Using fallback...")
    fallback = np.ones(256) / 256.0
    fb_score = score_fn(fallback)
    print(f"  FALLBACK: score={fb_score:.10f}")
    return [float(x) for x in fallback]
```
