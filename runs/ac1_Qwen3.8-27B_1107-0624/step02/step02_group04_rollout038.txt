"""
Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses L_p-based smooth gradient in logit space with Adam, diverse restarts,
and convolution flattening penalty.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = _full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _corr_weights(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

"""
Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive soft-max gradient in logit space with Adam, L2 convolution
flatness penalty, coarse-to-fine escalation, and diverse multi-restart.

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = _full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _corr(w, p):
    """Compute sliding inner product: result[i] = sum_j w[j+i] * p[j] for i=0..n-1"""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _make_profile(n, kind, param=1.5):
    t = np.linspace(0.0, 1.0, n)
    if kind == 'linear':
        a = 1.0 + param * (t - 0.5)
    elif kind == 'quad':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u * u
    elif kind == 'cos':
        a = 1.0 + param * np.cos(2.0 * np.pi * t)
    elif kind == 'sine':
        a = 1.0 + param * np.sin(2.0 * np.pi * t)
    elif kind == 'tanh':
        a = 1.0 + param * np.tanh(4.0 * (t - 0.5))
    elif kind == 'sech':
        a = 1.0 / np.cosh(3.0 * (t - 0.5))
    elif kind == 'bump':
        a = 1.0 + param * np.exp(-8.0 * (t - 0.5) ** 2)
    elif kind == 'anti_bump':
        a = 1.0 + param * np.exp(-8.0 * (t) ** 2) + param * np.exp(-8.0 * (t - 1.0) ** 2)
    elif kind == 'smooth_noise':
        rng = np.random.default_rng(12345)
        freqs = rng.uniform(0.5, 3.0, 8)
        phases = rng.uniform(0, 2 * np.pi, 8)
        a = 1.0
        for f, ph in zip(freqs, phases):
            a += 0.3 * param * np.sin(2 * np.pi * f * t + ph)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    s = float(a.sum())
    return a / s


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_interp = np.interp(x1, x0, p0)
    return _normalize_nonneg(p_interp)


def _optimize_logit(p0, deadline, rng, lr=0.05, l2 penalty_weight=0.001,
                    beta_init_frac=0.15, beta_min=1e-8,
                    no_imp_threshold=200, restart_noise=0.2):
    """
    Adam optimization in logit space with adaptive soft-max gradient
    and optional L2 convolution flatness penalty.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    # Adam
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(beta_min, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    step = 0

    while time.monotonic() < deadline:
        # Soft-max weights
        c_clipped = np.maximum(c, 1e-30)
        log_w = c_clipped / beta
        log_w -= float(np.max(log_w))
        w = np.exp(log_w)
        w_sum = float(w.sum())
        if not np.isfinite(w_sum) or w_sum <= 0:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # Gradient of soft-max objective: 2 * corr(w, p)
        gp = 2.0 * _corr(w, p)

        # L2 penalty gradient: 4 * corr(c, p)
        if l2_penalty_weight > 0.0:
            gp = gp + l2_penalty_weight * 4.0 * _corr(c, p)

        # Map to logit space
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        # Adam update
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1

        # Adaptive beta
        if step % 20 == 0:
            # Estimate effective number of active constraints
            w_entropy = float(-np.sum(w * np.log(np.maximum(w, 1e-30))))
            n_active = math.exp(w_entropy)
            # Target: n_active should be between n//32 and n//4
            target_active = max(4, n // 16)
            if n_active > target_active * 2:
                beta = max(beta_min, beta * 0.8)
            elif n_active < target_active * 0.5:
                beta = min(maxc * 0.5, beta * 1.3)

        # Decay beta gradually
        if step % 100 == 0:
            beta = max(beta_min, beta * 0.995)

        # Restart if stuck
        if no_imp > no_imp_threshold:
            noise = restart_noise
            if best_score < 1.52:
                noise = 0.1
            elif best_score < 1.55:
                noise = 0.15
            z = best_z + noise * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(beta_min, beta_init_frac * maxc)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            no_imp = 0
            lr = max(0.02, lr * 1.2)

        step += 1

        if step % 5000 == 0:
            print(f"    opt: step={step}, score={score:.10f}, best={best_score:.10f}, beta={beta:.2e}")

    return best_p, best_score


def _local_search(p0, deadline, rng, n_patches=2000, patch_size_range=(3, 20)):
    """
    Local patch-based search: randomly perturb small patches.
    """
    p = p0.copy()
    n = p.size
    c = _full_convolve_fft(p)
    best_p = p0.copy()
    best_score = float(2.0 * n * c.max())
    T = 0.02

    step = 0
    while time.monotonic() < deadline:
        step += 1
        ps = int(rng.integers(patch_size_range[0], min(patch_size_range[1], n // 4) + 1))
        start = int(rng.integers(0, n - ps + 1))

        p_new = p.copy()
        scale = T * 0.3
        p_new[start:start + ps] += rng.normal(0, scale, ps)
        p_new = np.maximum(p_new, 1e-12)
        p_new = p_new / p_new.sum()

        c_new = _full_convolve_fft(p_new)
        score_new = float(2.0 * n * c_new.max())
        delta = score_new - best_score

        if delta < 0:
            p = p_new
            c = c_new
            best_p = p_new.copy()
            best_score = score_new
            T = max(0.001, T * 0.995)
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-10)):
                p = p_new
                c = c_new
            T = max(0.001, T * 0.999)

        if step % 10000 == 0:
            print(f"    LS: step={step}, best={best_score:.10f}")

    return best_p, best_score
