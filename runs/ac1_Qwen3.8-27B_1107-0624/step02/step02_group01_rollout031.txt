Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max gradient descent with FFT-accelerated computation, 
LP-based direction finding, coordinate refinement, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

    c = fft_convolve(a)
    return float(c.max())

    a = t ** alpha

    return a / a.sum()

def make_asymmetric_tent(n, peak_pos):
    # Peak at position peak_pos (0 to 1)
    t = np.linspace(0.0, 1.0, n)
    if peak_pos <= 0.5:
        rise = t / peak_pos if peak_pos > 0 else np.ones(n)

    a = np.minimum(rise, fall)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

    # FFT(w, N) * FFT(f, N) -> IFFT
    # result[j] = sum_k w[k] * f[(j-k) mod N]
    # For j=0: sum_k w[k]*f[-k mod N] = w[0]*f[0] + w[1]*f[N-1] (f[N-1]=0 if N large)
    #         = w[0]*f[0] ✓
    # For j=1: sum_k w[k]*f[1-k] = w[0]*f[1] + w[1]*f[0] ✓
    
    # So conv[:n] gives the first n values of the convolution.
    
    grad_f = h1 + h2
    
    # Now, in logit space (z = log(f)):
    # f = softmax(z) (approximately, since we normalize)
    # df/dz_j = f[j] * (delta_jk - f[k]) for the softmax
    # But since we're using f directly and re-normalizing, the gradient in z-space is:
    # grad_z[j] = f[j] * (grad_f[j] - sum_k f[k]*grad_f[k])
    # Simplified: grad_z[j] = f[j] * grad_f[j] (the subtraction is a constant that doesn't affect the direction much)
    
    # Actually, for the constraint sum(f) = 1, the gradient in the feasible space is:
    # grad_z[j] = f[j] * (grad_f[j] - mean(f * grad_f))
    # But for Adam, the centering doesn't matter much. Let me just use:
    # grad_z[j] = f[j] * grad_f[j]
    
    grad_z = f * grad_f
    
    # Adam update
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1.0 - b1) * grad_z
    v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
    t_adam += 1
    mh = m / (1.0 - b1 ** t_adam)
    vh = v / (1.0 - b2 ** t_adam)
    z = z - lr * mh / (np.sqrt(vh) + eps)
    z = np.clip(z, -50.0, 50.0)
    z -= z.mean()
    
    f_new = np.exp(z - z.max())
    f_new = f_new / f_new.sum()
    f_new = np.maximum(f_new, 1e-15)
    f_new = f_new / f_new.sum()
    
    return f_new, z, m, v, t_adam, M_val

def softmax_optimize(f0, deadline, rng, lr=0.08, max_iter=100000, beta_init_frac=0.1):
    """Optimize using soft-max gradient in logit space."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size
    
    z = np.log(f)
    z -= z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-10, beta_init_frac * M_val)
    
    best_f = f.copy()
    best_M = M_val
    no_imp = 0
    imp_count = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        f, z, m, v, t_adam, M_val = softmax_gradient_step(f, beta, z, m, v, t_adam, lr)
        c = fft_convolve(f)
        M_new = float(c.max())
        M_val = M_new
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
            imp_count += 1
        else:
            no_imp += 1
        
        # Anneal beta
        if t_adam % 200 == 0:
            beta = max(1e-12, beta * 0.65)
        
        # Decrease learning rate over time
        lr = lr * (1.0 - 0.0001 * it)
        lr = max(lr, 0.001)
        
        # Restart if stuck
        if no_imp > max(500, 5 * n):
            z = 0.7 * np.log(best_f) + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            no_imp = 0
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.1 * M_val)
            lr = 0.08
    
    return best_f, best_M

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def lp_optimize(f0, deadline, rng, max_iter=20000):
    """LP-based iterative optimization."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size
    
    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        g = lp_direction(f, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 30:
                break
            f = best_f + 0.01 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                M = best_M
                continue
            f = f / fs
            M = compute_max_conv(f)
            if M < best_M:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            continue
        
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 100:
                f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = compute_max_conv(f)
                no_imp = 0
            continue
        
        g = g / g_sum
        improved = False
        for t in (0.3, 0.15, 0.5, 0.1, 0.7, 0.2, 0.4):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = compute_max_conv(f_new)
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break
        
        if not improved:
            no_imp += 1
            if no_imp % 50 == 0:
                f = 0.85 * best_f + 0.15 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = compute_max_conv(f)
                no_imp = 0
    
    return best_f, best_M

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None

def coordinate_polish(f, deadline, rng, n_steps=500):
    """Simple coordinate descent polish."""
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f
    f = f / s
    n = f.size
    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    
    idx = rng.permutation(n)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        j = idx[step % n]
        # Try to adjust f[j]
        # Compute gradient of max conv w.r.t. f[j]
        c = fft_convolve(f)
        k_star = int(np.argmax(c))
        # dc[k_star]/df[j] = f[k_star - j] + f[j - k_star] (with bounds)
        d1 = f[k_star - j] if 0 <= k_star - j < n else 0.0
        d2 = f[j - k_star] if 0 <= j - k_star < n else 0.0
        grad_j = d1 + d2
        
        # Try decreasing f[j] if grad > 0
        if grad_j > 0:
            for frac in (0.01, 0.005, 0.02):
                delta = -frac * f[j]
                f_new = f.copy()
                f_new[j] += delta
                f_new[j] = max(0, f_new[j])
                fs = f_new.sum()
                if fs < 1e-30:
                    continue
                f_new = f_new / fs
                M_new = compute_max_conv(f_new)
                if M_new < M - 1e-15:
                    f = f_new
                    M = M_new
                    if M < best_M:
                        best_M = M
                        best_f = f.copy()
                    break
        else:
            # Try increasing
            for frac in (0.01, 0.005):
                delta = frac * f[j]
                f_new = f.copy()
                f_new[j] += delta
                fs = f_new.sum()
                if fs < 1e-30:
                    continue
                f_new = f_new / fs
                M_new = compute_max_conv(f_new)
                if M_new < M - 1e-15:
                    f = f_new
                    M = M_new
                    if M < best_M:
                        best_M = M
                        best_f = f.copy()
                    break
    
    return best_f, best_M

Wait, I need to double-check the gradient computation. Let me trace through it more carefully.

We have f of length n. The convolution c = conv(f, f) has length 2n-1.
c[k] = sum_{j} f[j] * f[k-j] where the sum is over j such that both 0 <= j < n and 0 <= k-j < n.

So c[k] = sum_j f[j] * f[k-j] (with implicit bounds).

The gradient of c[k] with respect to f[j]:
dc[k]/df[j] = f[k-j] + f[j-k]

(where we define f[x] = 0 if x < 0 or x >= n)

The soft-max objective: L = beta * log(sum_k exp(c[k]/beta))

dL/dz_j (where z = log(f)):
First, dL/df_j = sum_k (exp(c[k]/beta) / sum_l exp(c[l]/beta)) * dc[k]/df[j]
= sum_k w[k] * (f[k-j] + f[j-k])

Then, since z_j = log(f_j) and f = softmax(z) (approximately, since we normalize):
dL/dz_j = f_j * (dL/df_j - sum_k f_k * dL/df_k)

But for the Adam update, the constant subtraction doesn't matter much. So I'll use:
grad_z[j] = f[j] * (dL/df_j)

Now, dL/df_j = sum_k w[k] * (f[k-j] + f[j-k])

The first term: sum_k w[k] * f[k-j]
This is the cross-correlation of w and f at lag j.
Let's call this xc[j] = sum_k w[k] * f[k-j]

The second term: sum_k w[k] * f[j-k]
This is the convolution of w and f at position j.
Let's call this cv[j] = sum_k w[k] * f[j-k]

So dL/df_j = xc[j] + cv[j]

For the FFT computation:
- xc[j] = IFFT(FFT(w, N) * conj(FFT(f, N)))[j] for j in [0, n-1]
  (with N >= len(w) + len(f) - 1 = 2n-1 + n - 1 = 3n-2)
  
- cv[j] = IFFT(FFT(w, N) * FFT(f, N))[j] for j in [0, n-1]
  (with N >= 3n-2)

Wait, I need to be careful about the cross-correlation convention.

The standard cross-correlation is: (a ⊛ b)[n] = sum_k a[k] * conj(b[k-n])

In numpy/scipy, `np.correlate(a, b, mode='full')` computes:
result[n] = sum_k a[k] * b[k - (M-B-1)+n] where M=len(a), B=len(b)

Actually, let me just use the FFT definition directly.

If we define:
- A = FFT(a, N) (zero-padded to length N)
- B = FFT(b, N)

Then IFFT(A * conj(B)) gives the circular cross-correlation:
result[n] = sum_k a[k] * b[(k-n) mod N]

For N >= len(a) + len(b) - 1, this equals the linear cross-correlation for n in [0, N-1].

Specifically:
result[n] = sum_{k=0}^{N-1} a[k] * b[(k-n) mod N]

For n in [0, len(a)-1] (and assuming b is zero-padded):
= sum_{k=n}^{n+len(b)-1} a[k] * b[k-n]
= sum_{j=0}^{len(b)-1} a[j+n] * b[j]

Hmm, this is the cross-correlation of a and b at lag n, where positive lag means a is shifted right relative to b.

In our case:
a = w (length 2n-1), b = f (length n)

xc[j] = sum_k w[k] * f[k-j] = sum_{k=j}^{j+n-1} w[k] * f[k-j] (where f is 0 outside [0,n-1])

Using the FFT: IFFT(FFT(w) * conj(FFT(f)))[j] = sum_k w[k] * f[(k-j) mod N]
For k in [j, j+n-1] and assuming N is large enough:
= sum_{k=j}^{j+n-1} w[k] * f[k-j] ✓

Great, so xc = IFFT(FFT(w, N) * conj(FFT(f, N)))[:n] gives the first n values.

For the convolution:
IFFT(FFT(w, N) * FFT(f, N))[j] = sum_k w[k] * f[(j-k) mod N]
For j in [0, n-1]:
= sum_{k=0}^{j} w[k] * f[j-k] + sum_{k=j+1}^{j+N-n} w[k] * f[j-k+N]
The second sum is 0 if N is large enough (since f is 0-padded).
So = sum_{k=0}^{j} w[k] * f[j-k]

But we want cv[j] = sum_k w[k] * f[j-k] for k in [max(0, j-n+1), min(j, 2n-2)]
= sum_{k=j-n+1}^{j} w[k] * f[j-k]

The FFT gives sum_{k=0}^{j} w[k] * f[j-k] (assuming N large enough that the wrap-around doesn't contribute).

But w has length 2n-1, and for k in [0, j], w[k] is valid. And f[j-k] is valid for 0 <= j-k < n, i.e., j-n+1 <= k <= j.

So the FFT convolution at position j gives:
sum_{k=max(0, j-n+1)}^{min(j, 2n-2)} w[k] * f[j-k]

Which is exactly cv[j]! (Since w[k] = 0 for k >= 2n-1, and f[j-k] = 0 for j-k >= n or j-k < 0.)

Wait, but the FFT convolution at position j gives sum_{k=0}^{N-1} w[k] * f[(j-k) mod N]. For N >= 3n-2:
- When (j-k) mod N is in [0, n-1] and k is in [0, 2n-2], we get w[k] * f[j-k]
- All other terms are 0

So yes, cv = IFFT(FFT(w, N) * FFT(f, N))[:n] gives the correct values.

But wait, there's a subtlety. The standard circular convolution at position j is:
sum_{k=0}^{N-1} w[k] * f[(j-k) mod N]

For j in [0, n-1] and N >= 3n-2:
- (j-k) mod N = j-k when j-k >= 0, i.e., k <= j
- (j-k) mod N = j-k+N when j-k < 0, i.e., k > j

"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines soft-max gradient descent with FFT, LP-based direction finding,
diverse initial profiles, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_max_conv(a):
    c = fft_convolve(a)
    return float(c.max())


def make_power_profile(n, alpha):
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t ** alpha
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_exp_profile(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(alpha * t)
    return a / a.sum()


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_asymmetric_tent(n, peak_pos):
    t = np.linspace(0.0, 1.0, n)
    if peak_pos <= 0.01:
        a = 1.0 - t
    elif peak_pos >= 0.99:
        a = t
    else:
        rise = t / peak_pos
        fall = 1.0 - (t - peak_pos) / (1.0 - peak_pos)
        a = np.minimum(rise, fall)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_gaussian_profile(n, center, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_bezier_profile(n, p0, p1, p2, p3):
    t = np.linspace(0.0, 1.0, n)
    u = 1.0 - t
    a = u ** 3 * p0 + 3 * u ** 2 * t * p1 + 3 * u * t ** 2 * p2 + t ** 3 * p3
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=10):

    a = np.ones(n)

        a += amp * np.cos(2 * np.pi * k * t + phase)

    return a / a.sum()

def softmax_gradient_step(f, beta, z, m, v, t_adam, lr):
    n = f.size
    c = fft_convolve(f)
    M_val = float(c.max())

    x = (c - M_val) / max(beta, 1e-15)
    ex = np.exp(np.clip(x, -100.0, 5.0))
    s_ex = float(ex.sum())
    if s_ex < 1e-30:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s_ex = 1.0
    w = ex / s_ex

    N = next_pow2(3 * n)
    fw = np.fft.rfft(w, N)
    ff = np.fft.rfft(f, N)

    cross = np.fft.irfft(fw * np.conj(ff), N)
    conv = np.fft.irfft(fw * ff, N)

    h1 = cross[:n]
    h2 = conv[:n]

    grad_f = h1 + h2

    if not np.all(np.isfinite(grad_f)):
        return f, z, m, v, t_adam, M_val

    grad_z = f * grad_f
    grad_z = grad_z - np.mean(grad_z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1.0 - b1) * grad_z
    v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
    t_adam += 1
    mh = m / (1.0 - b1 ** t_adam)
    vh = v / (1.0 - b2 ** t_adam)
    z_new = z - lr * mh / (np.sqrt(vh) + eps)
    z_new = np.clip(z_new, -50.0, 50.0)
    z_new -= z_new.mean()

    f_new = np.exp(z_new - z_new.max())
    f_new = f_new / f_new.sum()
    f_new = np.maximum(f_new, 1e-15)
    f_new = f_new / f_new.sum()

    return f_new, z_new, m, v, t_adam, M_val

def lp_direction(f, M, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def coordinate_polish(f, deadline, rng, n_rounds=3):
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f, np.inf
    f = f / s
    n = f.size
    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = fft_convolve(f)
        k_star = int(np.argmax(c))
        for j in range(n):
            if time.monotonic() > deadline:
                break
            d1 = f[k_star - j] if 0 <= k_star - j < n else 0.0
            d2 = f[j - k_star] if 0 <= j - k_star < n else 0.0
            grad_j = d1 + d2
            if grad_j <= 0:
                continue
            for frac in (0.005, 0.002, 0.01):
                f_new = f.copy()
                f_new[j] = max(0.0, f_new[j] - frac * f[j])
                fs = f_new.sum()
                if fs < 1e-30:
                    continue
                f_new = f_new / fs
                M_new = compute_max_conv(f_new)
                if M_new < M - 1e-16:
                    f = f_new
                    M = M_new
                    if M < best_M:
                        best_M = M
                        best_f = f.copy()
                    break

    return best_f, best_M

        try_update(p_best, sc)

        p_best2, M_best2 = coordinate_polish(p_best, min(phase3_end, time.monotonic() + 20.0), rng)
        sc2 = 2.0 * n1 * M_best2
        try_update(p_best2, sc2)

        if time.monotonic() < phase3_end - 5:
            per_lp = min(phase3_end, time.monotonic() + 30.0)
            p_best3, M_best3 = lp_optimize(p_best, per_lp, rng, max_iter=6000)
            sc3 = 2.0 * n1 * M_best3
            try_update(p_best3, sc3)

        print(f"Phase 3 cand {i}: best={best_score:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Try n=1024
    n2 = 1024
    phase4_end = min(deadline, t_start + 1700.0)

    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase4_end, time.monotonic() + 120.0)
            p_best, M_best = softmax_optimize(p_ext, per, rng, lr=0.05, beta_init_frac=0.08)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"Phase 4 (n=1024) soft-max: score={sc:.10f}")

            if time.monotonic() < phase4_end - 30:
                per2 = min(phase4_end, time.monotonic() + 50.0)
                p_best2, M_best2 = lp_optimize(p_best, per2, rng, max_iter=4000)
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"Phase 4 (n=1024) LP: score={sc2:.10f}")

    # Phase 5: Final refinement on absolute best
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = best_seq.size
        per = min(deadline - 10, time.monotonic() + 60.0)
        p_best, M_best = softmax_optimize(best_seq, per, rng, lr=0.015, beta_init_frac=0.02)
        sc = compute_score(p_best)
        try_update(p_best, sc)

        p_best2, M_best2 = coordinate_polish(p_best, min(deadline - 5, time.monotonic() + 15.0), rng)
        sc2 = compute_score(p_best2)
        try_update(p_best2, sc2)

        print(f"After Phase 5: best={best_score:.10f}, n={best_seq.size}")

    # Final
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"\nFinal internal score: {sc_final:.10f}, n={final.size}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(sc_eval):
                print(f"Final evaluated score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
