Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based spreading, active-set gradient descent, and multiplicative updates.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _make_conv_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ b = conv(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    if constrain_sum:
        # Add equality constraint sum(b) = 1
        A_eq = np.ones((1, n))
        b_eq = np.ones(1)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    else:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None

def _lp_refine_v2(a, deadline, max_iter=5000, mode='spread'):
    """
    Improved LP refinement.
    mode='spread': max sum(b) s.t. conv(a,b) <= M*inflate
    mode='unit':   max sum(b) s.t. conv(a,b) <= M*inflate, sum(b)=1
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    
    inflate = 1.0005
    blend = 0.5
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        
        if mode == 'unit':
            g = _lp_spread(a, M_lp, constrain_sum=True)
        else:
            g = _lp_spread(a, M_lp, constrain_sum=False)
        
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1e-10:
            break
        
        g = g / sum_g
        
        # Line search for best blend factor
        best_blend_score = best_score
        best_blend_a = a.copy()
        for t in [blend, blend * 0.5, blend * 1.5, 0.1, 0.3, 0.7, 0.9]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc = _score(a_try)
            if sc < best_blend_score:
                best_blend_score = sc
                best_blend_a = a_try.copy()
                best_t = t
        
        if best_blend_score < best_score - 1e-14:
            best_score = best_blend_score
            best_a = best_blend_a.copy()
            no_imp = 0
            # Adapt inflate and blend
            if no_imp == 0:
                inflate = max(1.0001, inflate * 0.95)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                inflate = min(1.01, inflate * 1.05)
                blend = max(0.1, blend * 0.9)
            if no_imp > 200:
                a = best_a.copy()
                no_imp = 0
                inflate = 1.001
                blend = 0.5
        
        a = best_blend_a
    
    return best_a

def _active_set_gradient(a, k=10):
    """
    Compute subgradient using top-k active constraints.
    Returns gradient vector and the max value.
    """
    n = len(a)
    c = _fft_convolve(a)
    M = float(c.max())
    
    # Get top-k indices
    if k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argsort(c)[::-1][:k]
    
    # Gradient: grad_j = 2 * sum_{i in idx} c_weight_i * a[i-j] (for valid i-j)
    # More precisely: grad_j = 2 * sum_{i: i-j in [0,n)} w_i * a[i-j]
    # where w_i = 1/k for i in idx (uniform weights on active set)
    # This equals 2/k * (w * a_reversed)[j] where w is the indicator of idx
    
    w = np.zeros(2 * n - 1)
    w[idx] = 1.0 / k
    
    # conv(w, a) gives us what we need
    # conv(w, a)[j] = sum_i w[i] * a[j-i] (for valid indices)
    # But we want sum_i w[i] * a[i-j] which is conv(w_flipped, a)[j]
    # Actually: sum_i w[i] * a[i-j] = (w * a_reversed)[j] in convolution sense
    # Let's use: h = convolve(w, a[::-1]) and take the right part
    
    N = _next_pow2(2 * n - 1 + n)  # need size for w (len 2n-1) convolved with a (len n)
    # Actually, let's just use np.convolve for simplicity
    w_full = np.zeros(2 * n - 1)
    w_full[idx] = 1.0 / k
    
    # We want g[j] = sum_i w_full[i] * a[i-j] for 0<=j<n, 0<=i<2n-1
    # This is the cross-correlation of w_full and a
    # = conv(w_full, a_reversed) at certain offset
    # Let's compute: r = convolve(w_full, a[::-1])
    # r[m] = sum_i w_full[i] * a[::-1][m-i] = sum_i w_full[i] * a[n-1-(m-i)]
    # = sum_i w_full[i] * a[n-1-m+i]
    # We want sum_i w_full[i] * a[i-j], so set n-1-m+i = i-j, giving m = n-1+j
    # So g[j] = r[n-1+j]
    
    r = np.convolve(w_full, a[::-1])
    g = 2.0 * r[n - 1:2 * n - 1]
    
    return g, M, c

def _gradient_refine_v2(a, deadline, max_iter=50000, k=20):
    """
    Active-set gradient descent with backtracking line search.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    lr = 0.05
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        g, M, c = _active_set_gradient(a, k)
        
        # Project gradient onto tangent space of simplex (remove mean)
        g_tan = g - np.mean(g)
        
        # Try different step sizes (backtracking)
        best_score_iter = best_score
        best_a_iter = a.copy()
        best_lr = 0
        
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            a_try = a - lr_try * g_tan
            a_try = _project_simplex(a_try)
            sc = _score(a_try)
            if sc < best_score_iter:
                best_score_iter = sc
                best_a_iter = a_try
                best_lr = lr_try
        
        if best_score_iter < best_score - 1e-14:
            best_score = best_score_iter
            best_a = best_a_iter.copy()
            no_imp = 0
            if best_lr > 0:
                lr = best_lr
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.8)
            if no_imp > 1000:
                a = best_a.copy()
                lr = base_lr
                no_imp = 0
                k = max(5, k // 2)
        
        a = best_a_iter
    
    return best_a

def _multiplicative_update(a, deadline, max_iter=50000, lr=0.1):
    """
    Multiplicative (Exponentiated Gradient) updates.
    a_k <- a_k * exp(-lr * grad_k), then normalize.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        
        # Subgradient: 2 * a[argmax - k] for the argmax position
        # Or use top-k average
        idx = np.argmax(c)
        # grad_k = 2 * a[idx - k] (for valid idx-k)
        k_idx = np.arange(n)
        valid = (idx >= k_idx) & (idx < 2 * n - 1)
        # Actually, grad_k = 2 * a[idx-k] if 0 <= idx-k < n
        # This is a shifted version of a
        shift = idx  # we want a[idx - k]
        if 0 <= shift < n:
            grad = np.zeros(n)
            # grad[k] = 2*a[shift-k] for k in [0, shift]
            # grad[k] = 0 for k > shift
            grad[:shift+1] = 2.0 * a[shift::-1]
        elif shift >= n:
            # grad[k] = 2*a[shift-k] for k in [shift-n+1, n-1]
            start = shift - n + 1
            grad[start:] = 2.0 * a[n-1:start-1:-1]  # a[n-1], a[n-2], ..., a[n-1-(start)]
            # Let me redo this
            # We want grad[k] = 2*a[shift-k] for k where 0 <= shift-k < n
            # i.e., shift-n < k <= shift, intersected with [0, n-1]
            # So k in [max(0, shift-n+1), min(n-1, shift)]
            k_lo = max(0, shift - n + 1)
            k_hi = min(n - 1, shift)
            for k in range(k_lo, k_hi + 1):
                grad[k] = 2.0 * a[shift - k]
        else:
            grad = np.zeros(n)
        
        # Remove mean to project onto simplex tangent
        grad_tan = grad - np.mean(grad)
        
        # Multiplicative update
        a = a * np.exp(-lr * grad_tan)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        sc = _score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.01, lr * 0.9)
            if no_imp > 2000:
                a = best_a.copy()
                lr = base_lr
                no_imp = 0
    
    return best_a

def _smooth_max_gradient(a, beta):
    """Compute gradient of smooth max of conv(a,a) using FFT."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x - x.max())  # numerical stability
    s = float(ex.sum())
    w = ex / s
    
    # Gradient: 2 * sum_i w_i * a[i-k]
    N = _next_pow2(2 * n - 1 + n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    # h_full[m] = sum_i w[i] * a_rev[m-i] = sum_i w[i] * a[n-1-(m-i)]
    # We want g[k] = sum_i w[i] * a[i-k]
    # a[i-k] = a[n-1-(n-1-i+k)] ... let me think differently
    # g[k] = sum_i w[i] * a[i-k]
    # Let's define b = a reversed: b[j] = a[n-1-j]
    # g[k] = sum_i w[i] * a[i-k] = sum_i w[i] * b[n-1-(i-k)] = sum_i w[i] * b[n-1-i+k]
    # This is conv(w, b)[n-1+k] = conv(w, a_rev)[n-1+k]
    # conv(w, b)[m] = sum_i w[i] * b[m-i]
    # So conv(w, b)[n-1+k] = sum_i w[i] * b[n-1+k-i] = sum_i w[i] * a[n-1-(n-1+k-i)] = sum_i w[i] * a[i-k]
    # Yes! So g[k] = conv(w, a_rev)[n-1+k]
    
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, mval

def _smooth_max_refine(a, deadline, max_iter=50000):
    """Smooth max gradient descent with Nesterov acceleration."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    c = _fft_convolve(a)
    beta = 0.05 * float(c.max())
    lr = 0.03
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0)
        s_nest = a_nest.sum()
        if s_nest > 1e-10:
            a_nest = a_nest / s_nest
        
        grad, _ = _smooth_max_gradient(a_nest, beta)
        
        # Project gradient
        grad_tan = grad - np.mean(grad)
        
        if not np.all(np.isfinite(grad_tan)):
            beta = beta * 2
            v = np.zeros(n)
            continue
        
        v = momentum * v - lr * grad_tan
        a_new = _project_simplex(a + v)
        a = a_new
        
        sc = _score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > 3000:
                a = best_a.copy()
                v = np.zeros(n)
                c = _fft_convolve(a)
                beta = 0.05 * float(c.max())
                lr = base_lr
                no_imp = 0
        
        if it % 100 == 0:
            beta = max(1e-8, beta * 0.8)
    
    return best_a

def _adam_logit_refine(a, deadline, max_iter=50000):
    """Adam optimization in logit space with smooth max."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t_adam = 0
    lr = 0.03
    
    c = _fft_convolve(a)
    beta = 0.05 * float(c.max())
    
    best_a = a.copy()
    best_z = z.copy()
    best_score = _score(a)
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad, _ = _smooth_max_gradient(a, beta)
        # Project gradient in probability space, then convert to logit space
        grad_tan = grad - np.mean(grad)
        # In logit space: dL/dz_j = p_j * (dL/dp_j - sum(p_k * dL/dp_k))
        # where p = a (normalized)
        gz = a * grad_tan
        gz -= np.mean(gz)
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v_arr = b2 * v_arr + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v_arr / (1 - b2 ** t_adam)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= np.mean(z)
        
        e = np.exp(z)
        a = e / e.sum()
        
        sc = _score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.95)
            if no_imp > 5000:
                z = best_z + 0.3 * np.random.default_rng(it).normal(size=n)
                z = np.clip(z, -50, 50)
                z -= np.mean(z)
                e = np.exp(z)
                a = e / e.sum()
                c = _fft_convolve(a)
                beta = 0.05 * float(c.max())
                m = np.zeros(n)
                v_arr = np.zeros(n)
                t_adam = 0
                lr = base_lr
                no_imp = 0
        
        if it % 200 == 0:
            beta = max(1e-8, beta * 0.75)
    
    return best_a

def _make_initial(n, type='linear', params=None):
    """Create initial sequence of given length and type."""
    if type == 'uniform':
        return np.ones(n) / n
    elif type == 'linear':
        eta = params if params is not None else 1.5
        t = np.linspace(0, 1, n)
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif type == 'quadratic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + 0.5 * u * u
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif type == 'sine':
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif type == 'random':
        rng = np.random.default_rng(0)
        a = rng.random(n)
        return a / a.sum()
    else:
        return np.ones(n) / n

def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimization: LP spreading + active-set gradient + smooth max + Adam.
    Tries multiple initializations and lengths, keeping the best result.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed % (2**31 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Define optimization pipeline for a given initial sequence
    def _optimize_pipeline(a0, time_budget):
        """Run a full optimization pipeline on a0, returning best result."""
        a = a0.copy()
        sc = _score(a)
        print(f"  Initial: score={sc:.10f}, n={a.size}")
        
        remaining = time_budget
        phases = [
            ('lp', 0.3),
            ('active_grad', 0.2),
            ('smooth_max', 0.2),
            ('adam', 0.3),
        ]
        
        for phase_name, frac in phases:
            phase_time = frac * time_budget
            if time.monotonic() > deadline - 10:
                break
            phase_end = min(deadline - 5, time.monotonic() + phase_time)
            
            if phase_name == 'lp':
                a = _lp_refine_v2(a, phase_end, max_iter=3000)
            elif phase_name == 'active_grad':
                a = _active_set_refine(a, phase_end)
            elif phase_name == 'smooth_max':
                a = _smooth_max_refine(a, phase_end)
            elif phase_name == 'adam':
                a = _adam_logit_refine(a, phase_end)
            
            sc = _score(a)
            print(f"  After {phase_name}: score={sc:.10f}")
            # Global best
            nonlocal best_score, best_seq
            # Actually can't use nonlocal in this context easily, let me restructure
        
        return a
    
    # Since we can't easily use nonlocal, let me restructure
    # Main optimization loop
    lengths_to_try = [512]
    if parent is not None:
        if parent.size not in lengths_to_try:
            lengths_to_try.insert(0, parent.size)
    
    # For each length, try multiple initializations
    time_per_length = (deadline - time.monotonic()) / len(lengths_to_try)
    
    for n_t in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        
        # Create initializations
        inits = []
        if parent is not None:
            p_interp = _interpolate_to(parent, n_t)
            inits.append(('parent', p_interp))
            for _ in range(2):
                p_pert = p_interp * (1 + 0.2 * rng.normal(size=n_t))
                p_pert = np.maximum(p_pert, 0)
                p_pert = p_pert / p_pert.sum()
                inits.append(('perturbed_parent', p_pert))
        
        inits.append(('linear_1.5', _make_initial(n_t, 'linear', 1.5)))
        inits.append(('linear_2.0', _make_initial(n_t, 'linear', 2.0)))
        inits.append(('sine', _make_initial(n_t, 'sine')))
        inits.append(('random', _make_initial(n_t, 'random')))
        
        # Time budget per initialization
        time_per_init = min(200.0, time_per_length / len(inits))
        
        for init_name, a0 in inits:
            if time.monotonic() > deadline - 30:
                break
            
            # Check if we have enough time
            if time.monotonic() + 30 > deadline:
                break
            
            sc0 = _score(a0)
            print(f"n={n_t}, init={init_name}: score={sc0:.10f}")
            
            if sc0 < best_score:

    # Final polish on the best

def _next_pow2(n):
    k = 1
    while k < n:

def _fft_convolve(a):

        return np.array([])

        return np.convolve(a, a)

    if mn < 0.0:
        c = c - mn
    return c


def _score(a):

def _make_conv_matrix(a):

    A = np.zeros((2 * n - 1, n))
    for j in range(n):

    return A


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

    return None

        v_arr = b2 * v_arr + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v_arr / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= np.mean(z)

        e = np.exp(z)
        a = e / e.sum()

        sc = _score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.008, lr * 0.95)
            if no_imp > 5000:
                z = 0.7 * best_z + 0.3 * np.random.default_rng(it).normal(size=n)
                z = np.clip(z, -50, 50)
                z -= np.mean(z)
                e = np.exp(z)
                a = e / e.sum()
                c = _fft_convolve(a)
                beta = 0.04 * float(c.max())
                m = np.zeros(n)
                v_arr = np.zeros(n)
                t_adam = 0
                lr = base_lr
                no_imp = 0

        if it % 150 == 0:
            beta = max(1e-9, beta * 0.72)

    return best_a


def _interpolate_to(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        if s > 1e-30:
            return p / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 0)
    s = p_new.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p_new / s


def _make_initial(n, profile='linear', eta=1.5):
    if profile == 'uniform':
        return np.ones(n) / n
    elif profile == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif profile == 'cosine':
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.6 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif profile == 'quadratic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + 0.8 * (1 - u * u)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
