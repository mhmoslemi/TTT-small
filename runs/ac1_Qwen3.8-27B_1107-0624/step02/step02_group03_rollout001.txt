"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp gradient descent with adaptive temperature, combined with
LP-based direction finding and multi-resolution strategy.
Key improvements: proper LSE gradient, better temperature schedule,
multiple restarts, and hybrid LP+gradient optimization.

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_quadratic_profile(n, a2, a1):
    """Create quadratic profile 1 + a1*u + a2*u^2, u=2t-1."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_fourier_profile(n, harmonics):
    """Create profile from Fourier harmonics: 1 + sum amp_k*cos(k*2*pi*t + phi_k)."""
    t = np.arange(n) / n
    a = np.ones(n)
    for k, amp, phi in harmonics:
        a += amp * np.cos(k * 2.0 * np.pi * t + phi)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def compute_lse_gradient(p, w):
    """
    Compute gradient of LSE(conv(p,p), beta) w.r.t. p.
    grad_p[i] = 2 * sum_k w[k] * p[k-i] (valid indices)
    w is the softmax of conv(p,p)/beta, length 2n-1.
    """
    n = len(p)
    grad_p = np.zeros(n)
    for i in range(n):
        k_lo = i
        k_hi = min(2 * n - 1, i + n)
        if k_hi > k_lo:
            grad_p[i] = 2.0 * np.dot(w[k_lo:k_hi], p[0:k_hi - k_lo])
    return grad_p


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=15.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M.
    Returns optimal g or None.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_optimize(f, max_iters=300, t_init=0.5, t_min=0.05, deadline=None, verbose=True):
    """LP-based iterative optimization. f normalized to sum 1."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=10.0)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Try full step
        c_g = np.convolve(g, g)
        score_g = 2.0 * n * float(c_g.max())

        if score_g < best_score * 0.998:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            no_improve = 0
        else:
            # Blended step with multiple t values
            improved = False
            for t in [t_step, t_step * 0.5, t_step * 0.25, 0.1, 0.05]:
                f_new = (1.0 - t) * f + t * g
                f_new = normalize(f_new)
                if f_new is None:
                    continue
                c_new = np.convolve(f_new, f_new)
                score_new = 2.0 * n * float(c_new.max())
                if score_new < best_score - 1e-13:
                    best_f = f_new.copy()
                    best_score = score_new
                    best_M = float(c_new.max())
                    f = f_new.copy()
                    M = best_M
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
                f = best_f.copy()
                M = 2.0 * best_score / (2.0 * n)
                if no_improve % 30 == 0:
                    t_step = max(t_min, t_step * 0.7)
                if no_improve > 150:
                    break

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, t={t_step:.4f}")

    return best_f, best_score


def lse_adam_optimize(p0, deadline, beta_init=None, lr_init=0.02, verbose=True,
                      beta_decay=0.9995, restart_every=5000, max_restarts=20):
    """
    Adam gradient descent in logit space minimizing LSE(conv(p,p), beta).
    Beta decreases over time (annealing). Periodic restarts for exploration.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    c0 = np.convolve(p0, p0)
    max_c0 = float(c0.max())
    if beta_init is None:
        beta_init = 0.05 * max_c0

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c0
    no_imp = 0
    beta = beta_init
    lr = lr_init
    base_lr = lr_init
    n_restarts = 0

    while time.monotonic() < deadline and n_restarts < max_restarts:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # LSE weights
        z_c = c / beta
        z_c -= z_c.max()
        w = np.exp(z_c)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        # Gradient w.r.t. p
        grad_p = compute_lse_gradient(p0, w)

        # Gradient w.r.t. z (logit space)
        mean_grad = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - mean_grad)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            p0 = np.exp(z - z.max())
            p0 /= p0.sum()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        # Adam update
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Anneal beta
        beta *= beta_decay
        beta = max(beta, 1e-10 * float(np.convolve(best_p, best_p).max()))

        # Adaptive LR
        if no_imp % 1000 == 0:
            lr = max(0.003, lr * 0.92)

        # Restart if stuck
        if no_imp > restart_every:
            n_restarts += 1
            rng = np.random.default_rng(it + n_restarts * 1000)
            # Perturb from best
            z = best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= np.mean(z)
            p0 = np.exp(z - z.max())
            p0 /= p0.sum()
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = 0.05 * float(np.convolve(p0, p0).max())
            if verbose:
                print(f"    LSE restart #{n_restarts}, beta reset, best={best_score:.8f}")

        p0 = p

        if verbose and it % 2000 == 0:
            print(f"    LSE iter {it}: score={best_score:.8f}, beta={beta:.6f}, lr={lr:.5f}")

    return best_p, best_score

def multiplicative_update(p, gamma=0.1, iterations=100):
    """
    Multiplicative "repulsion" update: push mass away from high-convolution regions.
    p_new[i] proportional to p[i] / (c[i] + c[2n-2-i])^gamma
    """
    n = len(p)
    p = p.copy()
    for _ in range(iterations):
        c = np.convolve(p, p)
        # For each position i, the "local congestion" is related to c values
        # involving p[i]. Use c[i] and c[2n-2-i] as proxies.
        congestion = c[np.minimum(np.arange(n), 2*n-2-np.arange(n))]
        # Actually, let's use a simpler proxy: the contribution of p[i] to max(c)
        # p[i] contributes p[i]*p[j] to c[i+j] for all j
        # Simple approach: use the row of the convolution matrix
        weights = 1.0 / (1.0 + gamma * congestion / max(c.max(), 1e-30))
        p_new = p * weights
        p_new = normalize(p_new)
        if p_new is None:
            break
        p = p_new
    return p

        p4_best, p4_score = lse_optimize_fixed_beta(best_seq, deadline_p4,
                                                     beta=0.01 * max_c_cur,
                                                     lr=0.008, verbose=True)
        if p4_score < best_score:
            best_score = p4_score
            best_seq = p4_best.copy()
            print(f"  Phase 4a result: score={best_score:.8f}")

    # --- Phase 5: Final multiplicative polish ---
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nPhase 5: Final polish at n={len(best_seq)}")
        
        # Try different gamma values for multiplicative update
        for gamma in [0.05, 0.1, 0.2, 0.3]:
            p_try = multiplicative_update(best_seq, gamma=gamma, iterations=50)
            sc_try = internal_score(p_try)
            if sc_try < best_score:
                best_score = sc_try
                best_seq = p_try
                print(f"  Multiplicative (gamma={gamma}): score={best_score:.8f}")
        
        # Short LSE refinement
        if time.monotonic() < deadline - 10:
            c_final = np.convolve(best_seq, best_seq)
            p5_best, p5_score = lse_optimize_fixed_beta(
                best_seq, deadline - 5,
                beta=0.005 * float(c_final.max()),
                lr=0.005, verbose=False)
            if p5_score < best_score:
                best_score = p5_score
                best_seq = p5_best.copy()
                print(f"  Phase 5 LSE result: score={best_score:.8f}")

    # --- Try n=1024 if time permits ---
    remaining = deadline - time.monotonic()
    if remaining > 120:
        n5 = 1024
        print(f"\nPhase 6: Try n={n5}")
        a5 = interpolate_seq(best_seq, n5)
        if a5 is not None:
            sc5 = internal_score(a5)
            print(f"  Interpolated to n={n5}: score={sc5:.8f}")
            if sc5 < best_score:
                best_score = sc5
                best_seq = a5.copy()
            # Quick LSE at n=1024
            deadline_p6 = deadline - 10
            p6_best, p6_score = lse_adam_optimize(a5, deadline_p6,
                                                   beta_init=0.02 * float(np.convolve(a5, a5).max()),
                                                   lr_init=0.01, verbose=True,
                                                   beta_decay=0.9999, restart_every=2000, max_restarts=5)
            if p6_score < best_score:
                best_score = p6_score
                best_seq = p6_best.copy()
                print(f"  Phase 6 result: score={best_score:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
