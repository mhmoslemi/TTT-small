def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_gaussian_window(n):
    j = np.arange(n)
    sigma = n / 4.0
    a = np.exp(-0.5 * ((j - n / 2.0 + 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bell(n, width_param):
    j = np.arange(n)
    t = (j - n / 2.0 + 0.5) / (n / 2.0)
    a = 1.0 / (1.0 + (t / width_param) ** 4)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def generate_initializations(n, rng):
    inits = []
    inits.append(make_uniform(n))
    inits.append(make_raised_cosine(n))
    inits.append(make_hann(n))
    inits.append(make_parabolic(n))
    inits.append(make_linear(n, 0.8))
    inits.append(make_linear(n, 1.2))
    inits.append(make_linear(n, 1.6))
    inits.append(make_linear(n, 2.0))
    inits.append(make_gaussian_window(n))
    inits.append(make_bell(n, 0.7))
    inits.append(make_bell(n, 1.0))
    inits.append(make_bell(n, 1.5))
    for _ in range(10):
        inits.append(make_smooth_random(n, rng))
    # Asymmetric profiles
    t = np.linspace(0, 1, n)
    a = 1.0 + 1.5 * t - 0.8 * t * t
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    a = 1.0 + 0.5 * t + 1.0 * t ** 3
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    a = 0.5 + 1.0 * np.sin(np.pi * t)
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    return inits


def softmax_gradient_step(z, beta, lr, m, v, t_adam, b1, b2, eps, n, N_corr):
    """One step of Adam on log-space with softmax-smoothed max-conv objective."""
    f = np.exp(z - z.max())
    f = f / f.sum()
    
    c = fft_convolve(f)
    M_val = float(c.max())
    
    # Softmax weights
    x = (c - M_val) / beta
    ex = np.exp(np.clip(x, -100.0, 5.0))
    s_ex = float(ex.sum())
    if s_ex < 1e-30:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s_ex = 1.0
    w = ex / s_ex
    
    # Gradient in a-space: grad_a[j] = 2 * sum_k w[k] * a[k-j]
    # This is a cross-correlation of w and f
    fw = np.fft.rfft(w, N_corr)
    ff = np.fft.rfft(f, N_corr)
    h = np.fft.irfft(fw * np.conj(ff), N_corr)[:n]
    
    # Gradient in z-space: grad_z[j] = f[j] * (grad_a[j] - sum(f * grad_a))
    grad_a = 2.0 * h
    grad_z = f * (grad_a - float(np.dot(f, grad_a)))
    
    if not np.all(np.isfinite(grad_z)):
        return z, f, M_val, m, v, t_adam, False
    
    t_adam += 1
    m = b1 * m + (1.0 - b1) * grad_z
    v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
    mh = m / (1.0 - b1 ** t_adam)
    vh = v / (1.0 - b2 ** t_adam)
    
    z_new = z - lr * mh / (np.sqrt(vh) + eps)
    z_new = np.clip(z_new, -50.0, 50.0)
    z_new -= z_new.mean()
    
    return z_new, f, M_val, m, v, t_adam, True


def optimize_gradient(f0, deadline, rng, beta_start=0.08, beta_end=0.005,
                      lr_start=0.08, lr_end=0.001, max_iter=200000,
                      restart_every=500, restart_strength=0.3):
    """Adam-based optimization in log-space with beta annealing."""
    f = np.asarray(f0, dtype=np.float64).copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size
    
    z = np.log(f)
    z -= z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_f = f.copy()
    best_M = float(fft_convolve(f).max())
    no_imp = 0
    
    N_corr = next_pow2(3 * n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta and learning rate
        progress = min(1.0, it / max(1, max_iter * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = max(beta, beta_end)
        lr = lr_start * (lr_end / lr_start) ** (it / max(1, max_iter))
        lr = max(lr, 1e-5)
        
        z, f_curr, M_val, m, v, t_adam, success = softmax_gradient_step(
            z, beta, lr, m, v, t_adam, b1, b2, eps, n, N_corr)
        
        if not success:
            z = np.log(np.maximum(best_f, 1e-15))
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        if M_val < best_M - 1e-15:
            best_M = M_val
            best_f = f_curr.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Restart if stuck
        if no_imp > restart_every:
            z_best = np.log(np.maximum(best_f, 1e-15))
            z_best -= z_best.mean()
            noise = rng.normal(size=n) * restart_strength
            z = z_best + noise
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
            # Reset beta to intermediate
            beta = beta_start * 0.5
    
    return best_f, best_M


def coordinate_descent_polish(a, deadline, max_rounds=500):
    """Polish using coordinate descent: for each coordinate, find optimal value."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return a, np.inf
    a = a / s
    n = a.size
    
    c = fft_convolve(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    
    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Random order for diversity
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() > deadline + 5:
                break
            
            current_val = a[j]
            if current_val < 1e-15:
                continue
            
            # Try small perturbations
            worst_k = int(np.argmax(c))
            
            # Binary search for best delta
            # (a*a)[k] changes by 2*delta*a[k-j] + delta^2 (if k=2j)
            # We want to find delta that minimizes max_k (c[k] + delta * 2*a[k_j])
            
            # For this coordinate, the gradient direction is:
            # d/d(delta) c[k] = 2*a[k-j] for k != 2j, and 2*a[k-j] + delta for k = 2j
            # To reduce the max, we want to decrease c at the max position(s)
            
            # Try decreasing a[j]
            if worst_k is not None:
                idx_c = worst_k - j
                if 0 <= idx_c < n:
                    coeff = 2.0 * a[idx_c]
                    if coeff > 1e-15:
                        # How much can we decrease?
                        # c[worst_k] + delta * 2*a[worst_k-j] <= M (trivially satisfied for delta<0)
                        # But other positions might increase... actually for delta<0, c decreases at positions where a[k-j]>0
                        # So decreasing a[j] always helps (or is neutral) for the max
                        # The question is: how much can we decrease before other positions become new max?
                        
                        # Try several values
                        for frac in (0.01, 0.005, 0.002, 0.001, 0.0005):
                            delta = -frac * current_val
                            a_new = a.copy()
                            a_new[j] = current_val + delta
                            if a_new[j] < 0:
                                continue
                            s_new = s + delta
                            if s_new < 1e-15:
                                continue
                            # Rescale to maintain sum
                            a_new = a_new / s_new * s  # keep sum same
                            c_new = fft_convolve(a_new)
                            M_new = float(c_new.max())
                            if M_new < M - 1e-14:
                                a = a_new
                                c = c_new
                                M = M_new
                                improved = True
                                if M < best_M:
                                    best_M = M
                                    best_a = a.copy()
                                break
            # Try increasing a[j] slightly (might help if it reduces peaks elsewhere)
            for frac in (0.005, 0.002, 0.001):
                delta = frac * s / n
                a_new = a.copy()
                a_new[j] = a[j] + delta
                s_new = a_new.sum()
                a_new = a_new / s_new * s
                c_new = fft_convolve(a_new)
                M_new = float(c_new.max())
                if M_new < M - 1e-14:
                    a = a_new
                    c = c_new
                    M = M_new
                    improved = True
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()
                    break
        
        if not improved and round_idx > 50:
            # Try a random perturbation restart
            if rng is not None:
                break
    
    return best_a, best_M


def interpolate_to_length(a, n_new):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        a = np.maximum(a, 0.0)
        s = a.sum()
        return a / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    return a_new / s if s > 1e-30 else None


def l_bfgs_polish(f0, deadline, max_iter=200, beta=0.1):
    """Use scipy L-BFGS-B on the softmax-smoothed objective in log-space."""
    from scipy.optimize import minimize
    
    f0 = np.asarray(f0, dtype=np.float64).copy()
    f0 = np.maximum(f0, 1e-15)
    f0 = f0 / f0.sum()
    n = f0.size
    
    z0 = np.log(f0)
    z0 -= z0.mean()
    
    N_corr = next_pow2(3 * n)
    
    def objective_and_grad(z):
        z = np.clip(z, -50, 50)
        f = np.exp(z - z.max())
        f = f / f.sum()
        
        c = fft_convolve(f)
        M_val = float(c.max())
        
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100, 5))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        
        L = M_val + beta * float(np.log(s_ex))
        
        fw = np.fft.rfft(w, N_corr)
        ff = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(ff), N_corr)[:n]
        
        grad_a = 2.0 * h
        grad_z = f * (grad_a - float(np.dot(f, grad_a)))
        
        if not np.all(np.isfinite(grad_z)):
            return 1e10, np.zeros(n)
        
        return L, grad_z
    
    # Custom wrapper for L-BFGS-B (needs separate fun and jac)
    def fun(z):
        val, _ = objective_and_grad(z)
        return val
    
    def jac(z):
        _, g = objective_and_grad(z)
        return g
    
    start_time_local = time.monotonic()
    result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                               'maxfun': 500})
    
    z_opt = np.clip(result.x, -50, 50)
    z_opt -= z_opt.mean()
    f_opt = np.exp(z_opt - z_opt.max())
    f_opt = f_opt / f_opt.sum()
    
    return f_opt, float(fft_convolve(f_opt).max())


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimization:
    1. Gradient-based optimization at n=512 from diverse starts.
    2. Interpolation to n=1024 with continued optimization.
    3. L-BFGS polishing and coordinate descent.
    4. Final exact evaluation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
    
    # Check parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None
    
    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")
    
    # Phase 1: Optimize at n=512
    n1 = 512
    phase1_deadline = min(deadline, t_start + 600.0)
    
    starts = []
    if parent is not None:
        p_int = interpolate_to_length(parent, n1)
        if p_int is not None:
            starts.append(p_int)
    
    inits_512 = generate_initializations(n1, rng)
    starts.extend(inits_512)
    
    print(f"Phase 1: {len(starts)} starts at n={n1}")
    
    # Try each start with gradient optimization
    candidates_512 = []
    per_start_time = max(5.0, (phase1_deadline - t_start) / max(1, len(starts)))
    
    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_deadline - 2:
            break
        
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s
        
        per = min(phase1_deadline, time.monotonic() + per_start_time)
        
        # Use different beta schedules for different starts
        if i == 0 and parent is not None:
            p_best, M_best = optimize_gradient(p0, per, rng,
                                                beta_start=0.06, beta_end=0.003,
                                                lr_start=0.05, lr_end=0.001,
                                                max_iter=100000, restart_every=400)
        elif i < 4:
            p_best, M_best = optimize_gradient(p0, per, rng,
                                                beta_start=0.10, beta_end=0.005,
                                                lr_start=0.10, lr_end=0.002,
                                                max_iter=80000, restart_every=300)
        else:
            p_best, M_best = optimize_gradient(p0, per, rng,
                                                beta_start=0.08, beta_end=0.004,
                                                lr_start=0.08, lr_end=0.001,
                                                max_iter=60000, restart_every=350)
        
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates_512.append((sc, p_best))
        
        if i % 3 == 0:
            print(f"  Start {i}/{len(starts)}: best={best_score:.10f}")
    
    candidates_512.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")
    
    # Phase 2: Extended optimization on top candidates at n=512
    phase2_deadline = min(deadline, t_start + 1000.0)
    top_k = min(4, len(candidates_512))
    
    for i in range(top_k):
        if time.monotonic() > phase2_deadline - 5:
            break
        _, p0 = candidates_512[i]
        per = min(phase2_deadline, time.monotonic() + 80.0)
        
        # Use L-BFGS for more precise optimization
        if time.monotonic() < phase2_deadline - 40:
            p_best, M_best = l_bfgs_polish(p0, per, max_iter=100, beta=0.05)
            sc = 2.0 * n1 * M_best
            try_update(p_best, sc)
            print(f"  Phase 2 L-BFGS cand {i}: score={sc:.10f}")
        
        # Then continue with gradient
        if time.monotonic() < phase2_deadline - 10:
            per2 = min(phase2_deadline, time.monotonic() + 50.0)
            p_best2, M_best2 = optimize_gradient(p_best if np.isfinite(sc) else p0,
                                                  per2, rng,
                                                  beta_start=0.03, beta_end=0.002,
                                                  lr_start=0.03, lr_end=0.0005,
                                                  max_iter=100000, restart_every=200,
                                                  restart_strength=0.15)
            sc2 = 2.0 * n1 * M_best2
            try_update(p_best2, sc2)
    
    print(f"After Phase 2: best={best_score:.10f}")
    
    # Phase 3: Try n=1024
    n2 = 1024
    phase3_deadline = min(deadline, t_start + 1450.0)
    
    if best_seq is not None and time.monotonic() < phase3_deadline - 60:
        p_ext = interpolate_to_length(best_seq, n2)
        if p_ext is not None:
            per = min(phase3_deadline, time.monotonic() + 120.0)
            p_best, M_best = optimize_gradient(p_ext, per, rng,
                                                beta_start=0.05, beta_end=0.003,
                                                lr_start=0.06, lr_end=0.001,
                                                max_iter=150000, restart_every=400)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"Phase 3 gradient (n={n2}): score={sc:.10f}")
            
            # L-BFGS polish at n=1024
            if time.monotonic() < phase3_deadline - 30:
                per2 = min(phase3_deadline, time.monotonic() + 60.0)
                p_best2, M_best2 = l_bfgs_polish(p_best, per2, max_iter=80, beta=0.04)
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"Phase 3 L-BFGS (n={n2}): score={sc2:.10f}")
    
    print(f"After Phase 3: best={best_score:.10f}")
    
    # Phase 4: Coordinate descent polish on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        per = min(deadline - 5, time.monotonic() + 120.0)
        p_cd, M_cd = coordinate_descent_polish(best_seq, per, max_rounds=200)
        sc = compute_score(p_cd)
        try_update(p_cd, sc)
        print(f"After coord descent: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")
    
    # Phase 5: One more gradient pass on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 60.0)
        p_best, M_best = optimize_gradient(best_seq, per, rng,
                                            beta_start=0.03, beta_end=0.002,
                                            lr_start=0.03, lr_end=0.0005,
                                            max_iter=200000, restart_every=300,
                                            restart_strength=0.1)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        print(f"After Phase 5: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")
    
    # Final: try also at a different length for the best shape
    if best_seq is not None and time.monotonic() < deadline - 20:
        # Try n=768
        n3 = 768
        p_ext3 = interpolate_to_length(best_seq, n3)
        if p_ext3 is not None:
            per = min(deadline - 5, time.monotonic() + 40.0)
            p_b3, M_b3 = optimize_gradient(p_ext3, per, rng,
                                            beta_start=0.04, beta_end=0.003,
                                            lr_start=0.05, lr_end=0.001,
                                            max_iter=100000, restart_every=300)
            sc3 = 2.0 * n3 * M_b3
            try_update(p_b3, sc3)
            print(f"Try n={n3}: score={sc3:.10f}")
    
    # Final output
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(sc_eval):
                print(f"Final evaluated score: {sc_eval:.10f}")
            else:
                print(f"Warning: evaluate_sequence returned {sc_eval}")
        except Exception as e:
            print(f"Evaluation error: {e}")
        
        # Ensure all values are valid for the evaluator
        final = np.clip(final, 0.0, 1000.0)
        if final.sum() < 0.01:
            final = final * 1.0 / final.sum() * 1.0
        
        return [float(x) for x in final]
    
    # Fallback
    fb = make_linear(512, 1.2)
    return [float(x) for x in fb]
