    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..n-1 using np.correlate."""
    n = len(p)
    return np.correlate(w, p, mode='valid')


def make_profile(n, func):
    """Create a profile from a function of t in [0,1]."""
    t = np.linspace(0.0, 1.0, n)
    a = func(t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def adam_optimize(p0, deadline, lr=0.05, beta_init=None, beta_decay=0.7,
                  beta_decay_interval=150, beta_min=1e-12, verbose=True,
                  restart_threshold=2000, restart_perturb=0.5):
    """
    Adam optimization in logit space with adaptive soft-max smoothing.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1, p>=0.
    """
    n = len(p0)
    p = np.maximum(p0, 1e-12).astype(np.float64)
    p = p / p.sum()

    z = np.log(p)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p, p)
    max_c = float(c.max())
    if beta_init is None:
        beta = 0.1 * max_c
    else:
        beta = beta_init

    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    last_print = 0
    best_beta = beta

    while time.monotonic() < deadline:
        it += 1

        c = np.convolve(p, p)
        max_c = float(c.max())

        # Soft argmax weights
        x = (c - max_c) / beta
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_ex

        # Gradient via correlation
        h = corr_weights(w, p)
        grad_p = 2.0 * h

        # Logit space gradient
        dot = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            beta = max(0.05 * float(np.convolve(p, p).max()), beta_min)
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p_new = np.exp(z - z.max())
        p_sum = p_new.sum()
        if p_sum < 1e-30 or not np.all(np.isfinite(p_new)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue
        p_new = p_new / p_sum

        c_new = np.convolve(p_new, p_new)
        score = 2.0 * n * float(c_new.max())

        if score < best_score - 1e-14:
            best_score = score
            best_p = p_new.copy()
            best_z = z.copy()
            best_beta = beta
            no_imp = 0
        else:
            no_imp += 1

        p = p_new
        z = z.copy()

        # Decay beta
        if it % beta_decay_interval == 0:
            beta = max(beta_min, beta * beta_decay)

        # Adaptive learning rate
        if no_imp > 400:
            lr = max(0.005, lr * 0.95)

        # Restart if stuck
        if no_imp > restart_threshold:
            rng = np.random.default_rng(it + 7777)
            z = best_z + rng.normal(0, restart_perturb, size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(beta_min, 0.05 * float(np.convolve(
                np.exp(z - z.max()) / np.exp(z - z.max()).sum(),
                np.exp(z - z.max()) / np.exp(z - z.max()).sum()
            ).max()))

        if verbose and it - last_print >= 500:
            last_print = it
            elapsed = time.monotonic() - (deadline - (time.monotonic() - time.monotonic()))
            print(f"    iter {it}: score={best_score:.8f}, beta={beta:.8f}, "
                  f"lr={lr:.4f}, no_imp={no_imp}")

    return best_p, best_score


def multi_restart_adam(p0, deadline, n_restarts=3, verbose=True):
    """
    Run Adam optimization multiple times with different perturbations and hyperparameters.
    Return the best result.
    """
    n = len(p0)
    best_p = p0.copy()
    best_score = internal_score(p0)

    configs = [
        {'lr': 0.06, 'beta_decay': 0.7, 'beta_decay_interval': 100},
        {'lr': 0.04, 'beta_decay': 0.75, 'beta_decay_interval': 150},
        {'lr': 0.08, 'beta_decay': 0.65, 'beta_decay_interval': 80},
        {'lr': 0.05, 'beta_decay': 0.8, 'beta_decay_interval': 200},
        {'lr': 0.03, 'beta_decay': 0.7, 'beta_decay_interval': 120},
    ]

    time_per_restart = (deadline - time.monotonic()) / max(n_restarts, 1)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        sub_deadline = min(deadline, time.monotonic() + time_per_restart)
        cfg = configs[r % len(configs)]

        if r == 0:
            start = best_p.copy()
        else:
            rng = np.random.default_rng(1000 + r)
            z = np.log(np.maximum(best_p, 1e-12))
            z -= np.mean(z)
            z += rng.normal(0, 0.3, size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            start = np.exp(z - z.max())
            start = start / start.sum()

        p_best, sc_best = adam_optimize(
            start, sub_deadline,
            lr=cfg['lr'],
            beta_decay=cfg['beta_decay'],
            beta_decay_interval=cfg['beta_decay_interval'],
            beta_min=1e-12,
            restart_threshold=1500,
            restart_perturb=0.4,
            verbose=verbose
        )

        if sc_best < best_score:
            best_score = sc_best
            best_p = p_best
            if verbose:
                print(f"  Restart {r}: improved to {best_score:.8f}")

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-resolution Adam search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
      1. Initialize from parent or analytic profiles.
      2. Optimize at n=128 (fast, many iterations).
      3. Upsample to n=256, optimize.
      4. Upsample to n=512, multi-restart optimization.
      5. Final polish at n=512.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 20.0)

    best_seq = None
    best_score = np.inf

    # Get initial sequence from parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent_norm = parent / parent.sum()
                best_seq = parent_norm
                best_score = internal_score(parent_norm)
                print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    # Generate analytic profile candidates
    analytic_profiles = {}
    for n in [128, 256, 512]:
        t = np.linspace(0, 1, n)
        profiles = [
            make_profile(n, lambda t: 1.0 + 0.5 * np.cos(2 * np.pi * t)),
            make_profile(n, lambda t: 1.0 + 0.3 * np.cos(4 * np.pi * t)),
            make_profile(n, lambda t: 1.0 + 0.4 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t)),
            make_profile(n, lambda t: 1.0 + 0.6 * (2 * t - 1)),
            make_profile(n, lambda t: 1.0 + 0.3 * np.sin(2 * np.pi * t)),
            make_profile(n, lambda t: np.exp(-4.0 * (t - 0.5) ** 2 / 0.05)),
            make_profile(n, lambda t: 1.0 + 0.5 * np.cos(6 * np.pi * t) - 0.2 * np.cos(2 * np.pi * t)),
        ]
        analytic_profiles[n] = profiles

    # Find best initial candidate
    for n in [128, 256, 512]:
        for prof in analytic_profiles[n]:
            sc = internal_score(prof)
            if sc < best_score:
                best_score = sc
                best_seq = prof

    if best_seq is not None:
        print(f"Best initial: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: Optimize at n=128
    n1 = 128
    deadline_p1 = min(deadline, t_start + 300.0)
    print(f"\n{'='*50}")
    print(f"Phase 1: n={n1} (budget: {deadline_p1 - t_start:.0f}s)")
    print(f"{'='*50}")

    # Initialize from best available at n=128
    a1 = interpolate_seq(best_seq, n1)
    sc_init_1 = internal_score(a1)
    print(f"  Initial score: {sc_init_1:.8f}")

    p_best, sc_best = adam_optimize(
        a1, deadline_p1,
        lr=0.1, beta_decay=0.65, beta_decay_interval=80,
        beta_min=1e-13, verbose=True,
        restart_threshold=1000, restart_perturb=0.6
    )
    if sc_best < best_score:
        best_score = sc_best
        best_seq = p_best
        print(f"  Phase 1 result: score={best_score:.8f}")

    # Phase 2: Optimize at n=256
    n2 = 256
    deadline_p2 = min(deadline, t_start + 800.0)
    if time.monotonic() < deadline_p2 - 100:
        print(f"\n{'='*50}")
        print(f"Phase 2: n={n2} (budget: {deadline_p2 - time.monotonic():.0f}s)")
        print(f"{'='*50}")

        a2 = interpolate_seq(best_seq, n2)
        sc_init_2 = internal_score(a2)
        print(f"  Initial score: {sc_init_2:.8f}")

        # Try multiple configurations
        time_left_p2 = deadline_p2 - time.monotonic()
        if time_left_p2 > 200:
            p_best, sc_best = multi_restart_adam(a2, deadline_p2, n_restarts=2, verbose=True)
        else:
            p_best, sc_best = adam_optimize(
                a2, deadline_p2,
                lr=0.06, beta_decay=0.7, beta_decay_interval=120,
                beta_min=1e-13, verbose=True,
                restart_threshold=1500, restart_perturb=0.5
            )

        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 2 result: score={best_score:.8f}")

    # Phase 3: Optimize at n=512 with multiple restarts
    n3 = 512
    deadline_p3 = min(deadline, t_start + 1600.0)
    if time.monotonic() < deadline_p3 - 100:
        print(f"\n{'='*50}")
        print(f"Phase 3: n={n3} multi-restart (budget: {deadline_p3 - time.monotonic():.0f}s)")
        print(f"{'='*50}")

        a3 = interpolate_seq(best_seq, n3)
        sc_init_3 = internal_score(a3)
        print(f"  Initial score: {sc_init_3:.8f}")

        time_left_p3 = deadline_p3 - time.monotonic()
        n_restarts_3 = max(2, min(4, int(time_left_p3 / 250)))

        p_best, sc_best = multi_restart_adam(a3, deadline_p3, n_restarts=n_restarts_3, verbose=True)

        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 3 result: score={best_score:.8f}")

    # Phase 4: Final polish with very small beta
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*50}")
        print(f"Phase 4: Final polish n={len(best_seq)}")
        print(f"{'='*50}")

        a4 = best_seq.copy()
        if len(a4) != 512:
            a4 = interpolate_seq(a4, 512)

        c4 = np.convolve(a4, a4)
        beta_small = 0.005 * float(c4.max())

        p_best, sc_best = adam_optimize(
            a4, deadline,
            lr=0.015, beta_init=beta_small,
            beta_decay=0.95, beta_decay_interval=500,
            beta_min=1e-15, verbose=True,
            restart_threshold=3000, restart_perturb=0.2
        )

        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 4 result: score={best_score:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={final_score:.8f}")
    print(f"{'='*50}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01 or not np.isfinite(s):
        best_seq = np.ones(256) / 256.0
        s = 1.0
    best_seq = best_seq / s

    # Verify with evaluate_sequence if available
    try:
        eval_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"evaluate_sequence confirms: {eval_score:.8f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
