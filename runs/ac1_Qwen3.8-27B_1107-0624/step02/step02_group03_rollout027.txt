            no_improve = 0
        else:
            f_new = (1.0 - t_step) * f + t_step * g
            f_new = normalize(f_new)
            if f_new is None:
                break
            c_new = np.convolve(f_new, f_new)
            M_new = float(c_new.max())
            score_new = 2.0 * n * M_new

            if score_new < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = score_new
                best_M = M_new
                f = f_new.copy()
                M = M_new
                no_improve = 0
                t_step = max(t_min, t_step * 0.995)
            else:
                no_improve += 1
                f = best_f.copy()
                M = best_M
                if no_improve % 20 == 0:
                    t_step = max(t_min, t_step * 0.7)
                if no_improve > 200:
                    break

        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={best_M:.6f}")

    return best_f, best_score


def soft_max_gradient_descent(p0, deadline, lr=0.05, beta_init=0.05,
                              beta_min=0.0005, verbose=True, use_fft=True):
    """
    Adaptive soft-max gradient descent in logit space.
    Uses correlation-based gradient with FFT for speed.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    beta = beta_init
    base_lr = lr

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Adaptive soft-max weights
        x = (c - max_c) / beta
        x_shifted = x - x.max()
        ex = np.exp(x_shifted)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Gradient via correlation
        if use_fft and n > 128:
            h = corr_weights_fft(w, p0)
        else:
            h = corr_weights(w, p0)
        grad_p = 2.0 * h

        # Logit space gradient
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.01, lr * 0.95)
                beta = max(beta_min, beta * 0.85)

        # Adaptive beta: decrease when gap between soft-max and true max is small
        if it % 200 == 0:
            soft_max_val = beta * float(np.log(s)) if s > 0 else 0
            beta = max(beta_min, beta * 0.75)

        if no_imp > 3000:
            # Restart with larger perturbation
            z = 0.7 * best_z + 0.4 * np.random.default_rng(it * 7).normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = 0.05

        p0 = p

    return best_p, best_score


def polynomial_optimize(n, degree, deadline, verbose=True):
    """
    Optimize a polynomial parameterization: a[j] = max(0, p(j/(n-1)))
    where p is evaluated on u = 2*j/(n-1) - 1 in [-1, 1].
    """
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    def params_to_a(params):
        a = np.ones(n) * params[0]
        for k in range(1, degree + 1):
            a += params[k] * (u ** k)
        a = np.maximum(a, 1e-10)
        return a / a.sum()

    def obj(params):
        a = params_to_a(params)
        return internal_score(a)

    best_params = None
    best_obj = np.inf
    rng = np.random.default_rng(123)

    # Multiple starting points
    inits = [
        [1.0, 1.5, 0.0, 0.0] + [0.0] * (degree - 3),
        [1.0, 0.0, 1.0, 0.0] + [0.0] * (degree - 3),
        [1.0, 2.0, -1.0, 0.5] + [0.0] * max(0, degree - 3),
        [1.0, 0.5, -0.5, 0.3, -0.1] + [0.0] * max(0, degree - 4),
    ]
    for _ in range(5):
        inits.append([1.0] + list(rng.normal(0, 0.8, degree)))

    for init in inits:
        if time.monotonic() > deadline:
            break
        init = np.array(init[:degree + 1], dtype=np.float64)
        if degree + 1 > len(init):
            init = np.concatenate([init, np.zeros(degree + 1 - len(init))])
        try:
            res = minimize(obj, init, method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-9,
                                    'fatol': 1e-11, 'adaptive': True})
            if res.fun < best_obj:
                best_obj = res.fun
                best_params = res.x.copy()
        except Exception:
            pass

    if best_params is not None:
        best_a = params_to_a(best_params)
        if verbose:
            print(f"  Poly(deg={degree}, n={n}): score={best_obj:.8f}")
        return best_a
    return None
