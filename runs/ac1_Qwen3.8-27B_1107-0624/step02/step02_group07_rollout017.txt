def subgradient_step(s, eta, soft_max_beta=0):
    """One subgradient step using the max (or soft-max) of the convolution."""
    n = len(s)
    conv = np.convolve(s, s)
    
    if soft_max_beta > 0:
        # Soft maximum weighting
        shifted = conv - conv.max()
        weights = np.exp(soft_max_beta * shifted / conv.max())
        weights /= weights.sum()
    else:

import numpy as np
import time
from scipy.optimize import linprog

def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def compute_subgradient_fft(s, weights):
    """g[i] = 2 * sum_k weights[k] * s[k-i] (valid range only)"""
    n = len(s)
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    w_rev = weights[::-1]
    conv_full = np.fft.irfft(np.fft.rfft(s, N_fft) * np.fft.rfft(w_rev, N_fft), N_fft)
    return 2.0 * conv_full[2*n-2 : n-2 : -1]

def soft_subgradient(s, beta):
    """Compute soft-subgradient of max(conv(s,s)) using log-sum-exp."""
    n = len(s)
    conv = np.convolve(s, s)
    # Soft-max weighting
    max_conv = conv.max()
    if max_conv < 1e-30:
        return np.zeros(n), 0.0
    shifted = conv / max_conv
    weights = np.exp(beta * (shifted - 1.0))
    weights /= weights.sum()
    g = compute_subgradient_fft(s, weights)
    return g, max_conv

def gradient_descent(s, deadline, rng, beta_start=1.0, beta_end=50.0, 
                     eta_start=0.5, use_lp=False, n_lp_calls=5):
    """Main optimization loop."""
    n = len(s)
    best_s = s.copy()
    best_score = fast_score(s)
    s = s.copy()
    
    eta = eta_start
    beta = beta_start
    no_improve = 0
    patience = 100
    lp_interval = max(100, n_lp_calls * 100) if use_lp else 10**9
    lp_calls = 0
    
    max_iterations = 50000
    for it in range(max_iterations):
        if time.monotonic() > deadline:
            break
        
        cur_score = fast_score(s)
        conv = np.convolve(s, s)
        max_conv = conv.max()
        
        # Compute subgradient
        shifted = conv / max_conv if max_conv > 0 else conv
        weights = np.exp(beta * (shifted - 1.0))
        weights /= weights.sum()
        g = compute_subgradient_fft(s, weights)
        
        # Multiplicative gradient step
        g_max = g.max()
        if g_max < 1e-15:
            break
        s_new = s * np.exp(-eta * g / g_max)
        s_new = np.maximum(s_new, 1e-12)
        s_new /= s_new.sum()
        
        score_new = fast_score(s_new)
        
        if score_new < best_score:
            best_score = score_new
            best_s = s_new.copy()
            no_improve = 0
        elif score_new < cur_score:
            no_improve += 1
        else:
            no_improve += 1
        
        s = s_new
        
        # Adaptive parameters
        if no_improve >= patience:
            if use_lp and lp_calls < n_lp_calls and n <= 512:
                # Try LP direction
                b = solve_lp_direction(s, max_conv, time_limit=3.0)
                if b is not None and b.sum() > 1e-8:
                    b /= b.sum()
                    # Line search
                    best_t = 0.0
                    best_ts = score_new
                    for t in np.linspace(0.1, 0.8, 8):
                        cand = (1-t)*s + t*b
                        cand = np.maximum(cand, 0) / np.maximum(cand.sum(), 1e-15)
                        ts = fast_score(cand)
                        if ts < best_ts:
                            best_ts = ts
                            best_t = t
                    if best_t > 0:
                        s = (1-best_t)*s + best_t*b
                        s = np.maximum(s, 0) / s.sum()
                        lp_calls += 1
                        no_improve = 0
                        if fast_score(s) < best_score:
                            best_score = fast_score(s)
                            best_s = s.copy()
                        continue
                
                # If LP failed or exhausted, try restart
                s = s + 0.05 * rng.standard_normal(n)
                s = np.maximum(s, 0.01)
                s /= s.sum()
                eta = eta_start
                beta = rng.choice([3.0, 5.0, 10.0, 20.0])
                no_improve = 0
            else:
                # Simple restart with perturbation
                s = s + 0.03 * rng.standard_normal(n)
                s = np.maximum(s, 0.01)
                s /= s.sum()
                eta = eta_start * rng.uniform(0.5, 1.5)
                beta = rng.choice([3.0, 5.0, 10.0, 20.0, 50.0])
                no_improve = 0
        
        # Gradually increase beta
        if it % 200 == 0:
            beta = min(beta * 1.1, beta_end)
        
        # Decay eta slowly
        if it % 100 == 0 and no_improve == 0:
            eta = max(eta * 0.99, 0.01)
    
    return best_s, best_score

def solve_lp_direction(s, M, time_limit=3.0):
    """Solve LP: max sum(b) s.t. conv(s,b)[k] <= M, b >= 0."""
    n = len(s)
    # conv(s, b)[k] = sum_j s[j] * b[k-j] <= M
    # Variables: b[0], ..., b[n-1]
    # Constraints: for k = 0, ..., 2n-2: sum_j s[j] * b[k-j] <= M
    #              (where b[k-j] = 0 if k-j not in [0, n-1])
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # b[j] appears in conv[k] when k-j is a valid index of s, i.e., 0 <= k-j < n
        # So k ranges from j to j+n-1
        A[j:j+n, j] = s
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

import time
from scipy.optimize import linprog

def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def compute_subgradient_fft(s, weights):
    n = len(s)
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    w_rev = weights[::-1]
    conv_full = np.fft.irfft(np.fft.rfft(s, N_fft) * np.fft.rfft(w_rev, N_fft), N_fft)
    return 2.0 * conv_full[2*n-2 : n-2 : -1]

def get_weights(conv, beta):
    max_conv = conv.max()
    if max_conv < 1e-30:
        return np.ones(len(conv)) / len(conv), 0.0
    shifted = conv / max_conv
    w = np.exp(beta * (shifted - 1.0))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    return w, max_conv

def subgradient_direction(s, beta):
    conv = np.convolve(s, s)
    weights, max_conv = get_weights(conv, beta)
    g = compute_subgradient_fft(s, weights)
    return g, weights, max_conv

        sc = fast_score(s_new)

    n = len(s)

                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            x = np.maximum(res.x, 0.0)
            if x.sum() > 1e-10:

        pass
    return None
