import numpy as np
import time

def full_conv(n, a):
    """Compute linear convolution of a with itself using FFT."""
    N = 1
    while N < 2*n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    c = np.maximum(c, 0)  # remove numerical negatives
    return c

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = full_conv(n, a)
    return 2.0 * n * c.max() / (s * s)

def gradient_of_max_conv(a, w):
    """
    Compute gradient of sum_k w[k] * (a*a)[k] with respect to a.
    w should be a probability vector of length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    N = 1
    while N < 3*n:
        N *= 2
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    corr = np.fft.irfft(fw * fa_rev, N)
    # g[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * (w * a_rev)[i + n - 1]
    g = 2.0 * corr[n-1 : 2*n-1]
    return g

def projected_grad_desc(a0, deadline, lr_init=0.05, beta_init=5.0):

    Projected gradient descent to minimize max_k (a*a)[k] subject to a>=0, sum(a)=1.

    n = len(a)
    a = np.maximum(0, a)
    a /= a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    lr = lr_init
    beta = beta_init
    no_improve = 0
    
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        c = full_conv(n, a)
        # Soft max weights
        x = beta * (c - c.max())
        w = np.exp(x)
        w /= w.sum()
        
        # Gradient
        g = gradient_of_max_conv(a, w)
        
        # Step
        a_new = a - lr * g
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s > 1e-30:
            a_new /= s
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a = a_new.copy()
        
        # Adaptive LR
        if no_improve > 50:
            lr *= 0.9
            if no_improve > 500:

                a = np.maximum(0, a)

                lr = lr_init * 0.5
                no_improve = 0

        if it % 1000 == 0:
            print(f"  PGD iter {it}: score={best_score:.10f}, lr={lr:.6f}, beta={beta:.2f}")
    
    return best_a, best_score
