    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_min_peak(a):
    """Find g >= 0, sum(g)=1 minimizing max conv(a,g). Returns (g, M')."""
    n = len(a)
    A = _make_conv_matrix(a)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        g = res.x[:n]
        M_prime = float(res.x[-1])
        return g, M_prime
    return None, None


def _lp_max_sum(a, M):
    """Find g >= 0 maximizing sum(g) subject to conv(a, g) <= M."""
    n = len(a)
    A = _make_conv_matrix(a)
    c_obj = -np.ones(n + 1)
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * (n + 1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        g = res.x[:n]
        return g
    return None


def _lp_refine(a, deadline, max_iter=10000):
    """Iterative LP-based refinement using min-peak direction finding."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    rng = np.random.default_rng(777)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g, M_lp = _lp_min_peak(a)
        if g is None:
            break

        current_M = float(_fft_convolve(a).max())

        if M_lp >= current_M * (1.0 - 1e-10):
            no_imp += 1
            if no_imp > 200:
                a = best_a.copy()
                if no_imp > 800:
                    # Add noise to escape
                    noise = rng.normal(size=n) * 0.05
                    a = a + noise
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-10:
                        break
                    a = a / s
                no_imp = 0
            continue

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            no_imp += 1
            continue
        g = g / g_sum

        # Line search over blend factors
        best_candidate = None
        best_candidate_M = best_M

        for t in [0.4, 0.6, 0.8, 1.0]:
            a_cand = (1.0 - t) * a + t * g
            a_cand = np.maximum(a_cand, 0)
            s = a_cand.sum()
            if s < 1e-10:
                continue
            a_cand = a_cand / s
            c_cand = _fft_convolve(a_cand)
            M_cand = float(c_cand.max())
            if M_cand < best_candidate_M:
                best_candidate_M = M_cand
                best_candidate = a_cand

        if best_candidate is not None:
            a = best_candidate
            if best_candidate_M < best_M - 1e-12:
                best_M = best_candidate_M
                best_a = a.copy()
                best_score = float(2.0 * n * best_M)
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

    return best_a


def _coordinate_descent(a, deadline, max_sweeps=100000):
    """Fast vectorized coordinate descent."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_M = float(c.max())
    best_a = a.copy()
    no_imp = 0
    rng = np.random.default_rng(4242)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        sweep_improved = False
        order = rng.permutation(n)

        for i in order:
            ai = a[i]

            if ai < 1e-14:
                cands = [0.01 / n, 0.005 / n]
            else:
                cands = [ai * 0.4, ai * 0.7, ai * 1.4, ai * 2.0, 1e-8]
                # Try to equalize c[2i] with peak
                ki = 2 * i
                if ki < 2 * n - 1:
                    target = best_M - (c[ki] - ai * ai)
                    if target > 0:
                        cands.append(np.sqrt(target))

            best_delta = 0.0
            best_new_M = float(c.max())

            for nv in cands:
                delta = nv - ai
                if abs(delta) < 1e-14:
                    continue

                k_start = max(0, i)
                k_end = min(2 * n - 1, i + n - 1) + 1
                idx = np.arange(k_start, k_end) - i
                updates = 2.0 * delta * a[idx]
                if 2 * i >= k_start and 2 * i < k_end:
                    pos = 2 * i - k_start
                    updates[pos] = 2.0 * ai * delta + delta * delta

                local_new_max = float((c[k_start:k_end] + updates).max())
                global_new_max = max(best_M, local_new_max)
                # Also need to check non-updated part
                if k_start > 0:
                    global_new_max = max(global_new_max, float(c[:k_start].max()))
                if k_end < 2 * n - 1:
                    global_new_max = max(global_new_max, float(c[k_end:].max()))

                if global_new_max < best_new_M - 1e-12:
                    best_new_M = global_new_max
                    best_delta = delta

            if best_delta != 0.0:
                k_start = max(0, i)
                k_end = min(2 * n - 1, i + n - 1) + 1
                idx = np.arange(k_start, k_end) - i
                updates = 2.0 * best_delta * a[idx]
                if 2 * i >= k_start and 2 * i < k_end:
                    pos = 2 * i - k_start
                    updates[pos] = 2.0 * ai * best_delta + best_delta * best_delta
                c[k_start:k_end] += updates
                a[i] += best_delta
                sweep_improved = True

        if sweep_improved:
            current_M = float(c.max())
            if current_M < best_M - 1e-12:
                best_M = current_M
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 80:
            a = best_a.copy()
            c = _fft_convolve(a)
            no_imp = 0
            if no_imp > 500:
                break

    return best_a


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a, deadline, max_iter=50000):
    """Smooth-max gradient with Nesterov momentum and adaptive beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_M = float(c.max())

    beta = max(1e-8, 0.05 * best_M)
    lr = 0.03
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    rng = np.random.default_rng(555)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        grad, maxc = _gradient_smooth_max(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.95)
            if no_imp > max(1000, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-10, 0.05 * best_M)
                lr = 0.03
                no_imp = 0

        if it % 100 == 0:
            beta = max(1e-12, beta * 0.8)

    return best_a


def _adam_polish(a, deadline, max_iter=30000):
    """Adam in logit space for final refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    lr = 0.04

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    beta = max(1e-8, 0.05 * best_M)
    no_imp = 0
    rng = np.random.default_rng(888)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c_tmp = _fft_convolve(p)
        w = _smooth_max_weight(c_tmp, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_M = M
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                noise = rng.normal(size=n) * 0.3
                z = best_z + noise
                z = np.clip(z, -40.0, 40.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-8, 0.05 * best_M)
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr = 0.04
                no_imp = 0

        if it % 150 == 0:
            beta = max(1e-12, beta * 0.75)

    return best_a


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_triangle_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_two_bump_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-50.0 * (t - 0.15) ** 2) + np.exp(-50.0 * (t - 0.85) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP direction finding + coordinate descent + gradient polish.
    Tries multiple lengths and initializations to avoid local minima.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define candidate lengths and their time budgets
    # Prioritize n=512 (where parent lives) and n=384
    candidate_lengths = [512, 384, 256, 640]

    # Allocate time proportionally, with more time for 512
    time_allocation = {512: 0.35, 384: 0.25, 256: 0.20, 640: 0.15}
    # Reserve 5% for final polish
    total_alloc = sum(time_allocation.values())
    final_polish_time = budget_s * 0.05

    for n_t in candidate_lengths:
        remain_time = deadline - time.monotonic()
        if remain_time < 30:
            break

        alloc = (budget_s - final_polish_time) * time_allocation.get(n_t, 0.1)
        phase_end = time.monotonic() + alloc
        if phase_end > deadline - final_polish_time:
            phase_end = deadline - final_polish_time

        print(f"\n=== Processing n={n_t}, budget={alloc:.0f}s ===")

        # Generate initial candidates
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
            # Perturbed parent
            for _ in range(2):
                pp = _interpolate_profile(parent, n_t)
                pp = pp * (1.0 + 0.15 * rng.normal(size=n_t))
                pp = np.maximum(pp, 0)
                pp = pp / pp.sum()
                inits.append(pp)
        inits.append(_make_linear_profile(n_t, 1.5))
        inits.append(_make_linear_profile(n_t, 2.0))
        inits.append(_make_cosine_profile(n_t))
        inits.append(_make_triangle_profile(n_t))
        inits.append(_make_two_bump_profile(n_t))

        # Pick the best initial candidate
        best_init = None
        best_init_score = np.inf
        for(init) in inits:
            if init is None:
                continue
            sc = _score(init)
            if np.isfinite(sc) and sc < best_init_score:
                best_init_score = sc
                best_init = init
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = init.copy()

        if best_init is None:
            best_init = _make_linear_profile(n_t, 1.5)
            best_init_score = _score(best_init)

        print(f"  Best init: score={best_init_score:.10f}")

        # Phase A: LP refinement (main workhorse)
        lp_end = min(phase_end - alloc * 0.4, time.monotonic() + alloc * 0.55)
        if time.monotonic() < lp_end:
            a_lp = _lp_refine(best_init, lp_end, max_iter=20000)
            sc_lp = _score(a_lp)
            print(f"  After LP: score={sc_lp:.10f}")
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()

            # Also try LP from second-best init
            if len(inits) > 1:
                second_best = None
                second_score = np.inf
                for init in inits:
                    if init is not None and init is not best_init:
                        sc = _score(init)
                        if np.isfinite(sc) and sc < second_score and sc < best_init_score + 0.1:
                            second_score = sc
                            second_best = init
                if second_best is not None and time.monotonic() < lp_end:
                    a_lp2 = _lp_refine(second_best, lp_end, max_iter=5000)
                    sc_lp2 = _score(a_lp2)
                    if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                        best_score = sc_lp2
                        best_seq = a_lp2.copy()
                    print(f"  LP from 2nd init: score={sc_lp2:.10f}")
        else:
            a_lp = best_init

        # Phase B: Coordinate descent
        cd_time = alloc * 0.25
        cd_end = time.monotonic() + cd_time
        if cd_end > phase_end:
            cd_end = phase_end
        if time.monotonic() < cd_end:
            a_cd = _coordinate_descent(a_lp, cd_end, max_sweeps=200000)
            sc_cd = _score(a_cd)
            print(f"  After CD: score={sc_cd:.10f}")
            if np.isfinite(sc_cd) and sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()

        # Phase C: Gradient polish
        gp_time = alloc * 0.15
        gp_end = time.monotonic() + gp_time
        if gp_end > phase_end:
            gp_end = phase_end
        if time.monotonic() < gp_end:
            a_gp = _gradient_polish(a_cd, gp_end, max_iter=20000)
            sc_gp = _score(a_gp)
            print(f"  After Grad: score={sc_gp:.10f}")
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()

    # === FINAL POLISH PHASE ===
    print(f"\n=== Final Polish Phase ===")
    if best_seq is not None and time.monotonic() < deadline:
        final_end = deadline

        # Adam polish
        ad_time = final_polish_time * 0.4
        ad_end = time.monotonic() + ad_time
        if ad_end > final_end:
            ad_end = final_end
        if time.monotonic() < ad_end:
            a_ad = _adam_polish(best_seq, ad_end, max_iter=30000)
            sc_ad = _score(a_ad)
            print(f"  Adam: score={sc_ad:.10f}")
            if np.isfinite(sc_ad) and sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()

        # Coordinate descent
        cd_end2 = time.monotonic() + final_polish_time * 0.3
        if cd_end2 > final_end:
            cd_end2 = final_end
        if time.monotonic() < cd_end2:
            a_cd2 = _coordinate_descent(best_seq, cd_end2, max_sweeps=200000)
            sc_cd2 = _score(a_cd2)
            print(f"  Final CD: score={sc_cd2:.10f}")
            if np.isfinite(sc_cd2) and sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = a_cd2.copy()

        # One more LP pass
        lp_end2 = time.monotonic() + final_polish_time * 0.2
        if lp_end2 > final
