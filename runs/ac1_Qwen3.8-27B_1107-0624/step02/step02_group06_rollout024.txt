from scipy.optimize import linprog

def solve_lp_for_g(f, M):
    """
    Solve: maximize sum(g) subject to g >= 0, conv(f, g)[k] <= M for all k.
    f is fixed non-negative sequence of length n.
    M is the bound.
    Returns g (normalized to sum 1).
    """
    n = len(f)
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: conv(f, g)[k] <= M for k = 0, ..., 2n-2
    # conv(f,g)[k] = sum_i f[i] * g[k-i] for valid i, k-i
    # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * g[k-i]
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A_ub[k, k - i] = f[i]
    
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None

"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal update directions, combined with
gradient-based polishing and multi-length search.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c

def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))

def build_conv_matrix(f, n):
    """Build sparse matrix A where (A @ g)[k] = conv(f, g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_for_g(f, M):
    """
    Solve: maximize sum(g) subject to g >= 0, conv(f, g) <= M (componentwise).
    Returns g normalized to sum 1, or None if infeasible.
    """
    n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def lp_iteration(f, t=0.1):
    """
    One LP-based iteration: find g that maximizes sum(g) subject to conv(f,g) <= max(conv(f,f)).
    Update f toward g.
    Returns new_f, score.
    """
    f = np.asarray(f, dtype=np.float64).ravel()
    n = len(f)
    c = full_convolve_fft(f)
    M = float(c.max())
    
    g = solve_lp_for_g(f, M)
    if g is None:
        return f, internal_score(f)
    
    new_f = (1 - t) * f + t * g
    new_f = np.maximum(new_f, 0)
    s = new_f.sum()
    if s <= 1e-30:
        return f, internal_score(f)
    new_f = new_f / s
    
    return new_f, internal_score(new_f)

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def gradient_stp(p0, deadline, rng, lr=0.10, beta_init=0.05, max_iters=5000):
    """
    Gradient-based optimization in logit space with soft-max.
    Uses exact max when beta is small.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = beta_init * maxc

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline and adam_t < max_iters:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Decrease beta periodically
        if adam_t % 100 == 0:
            beta = max(1e-15, beta * 0.6)

        # Restart if stuck
        if no_imp > max(500, 5 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = beta_init * maxc
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score

def coordinate_descent_polish(p, deadline, rng, step_frac=0.05):
    """Coordinate descent with exact score evaluation."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0
    sweep = 0

    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            factors = [1.0 + step_frac, 1.0 + 0.5*step_frac, 1.0 + 2.0*step_frac,
                       1.0 - step_frac, 1.0 - 0.5*step_frac, 1.0 - 0.3*step_frac]
            best_val = orig
            best_sc = best_score
            for factor in factors:
                new_val = orig * factor
                if new_val < 1e-15 or new_val > 1000.0:
                    continue
                p[i] = new_val
                sc = internal_score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        
        sweep += 1
        if improved:
            step_frac = max(0.01, step_frac * 0.95)
        if no_imp > 2 * n:
            break
        if sweep > 50:
            break

    return best_p, best_score

def true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def make_deterministic_profiles(n, count=10):
    """Generate diverse deterministic initial profiles."""
    profs = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear profiles
    for eta in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    
    # Exponential profiles
    for eta in (1.0, 2.0, 3.0, 5.0, 8.0):
        a = np.exp(eta * t)
        profs.append(a / a.sum())
        a = np.exp(-eta * t)
        profs.append(a / a.sum())
    
    # Power laws
    for p in (0.3, 0.5, 1.0, 2.0, 3.0):
        a = t**p + 1e-8
        profs.append(a / a.sum())
        a = (1-t)**p + 1e-8
        profs.append(a / a.sum())
    
    # Sigmoid
    for eta in (3.0, 5.0, 8.0, 15.0):
        a = 1.0 / (1.0 + np.exp(-eta * (t - 0.5)))
        profs.append(a / a.sum())
    
    # Gaussian bump
    for sigma in (0.1, 0.2, 0.3, 0.5):
        a = np.exp(-((t - 0.5)**2) / (2*sigma**2))
        profs.append(a / a.sum())
    
    # Two-peak
    a = np.exp(-((t-0.25)**2)/0.02) + np.exp(-((t-0.75)**2)/0.02)
    profs.append(a / a.sum())
    
    # Quartic
    u = 2*t - 1
    a = 1.0 + 0.5*u**2 - 0.3*u**4
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    
    # Cosine
    a = 0.5 + 0.4 * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    
    # Triangular
    a = 1.0 - np.abs(2*t - 1)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    
    return profs[:count]
