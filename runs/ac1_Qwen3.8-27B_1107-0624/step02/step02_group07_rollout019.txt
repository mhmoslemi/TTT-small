```python
import numpy as np
import time
from scipy.optimize import linprog

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    N = 1
    while N < 2 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    conv = np.fft.irfft(S * S, N)[:2*n-1]
    return float(2.0 * n * conv.max() / (total * total))

def compute_autocorr(s):
    """Compute full autocorrelation."""
    n = len(s)
    N = 1
    while N < 2 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    return np.fft.irfft(S * S, N)[:2*n-1]

def autocorr_gradient(s, w):
    """
    Compute gradient of sum_m w[m] * (s*s)[m] with respect to s.
    s: array of length n, w: array of length 2n-1.
    """
    n = len(s)
    s_rev = s[::-1]
    total_len = n + (2*n - 1) - 1  # = 3n - 2
    N = 1
    while N < total_len:
        N *= 2
    S_rev = np.fft.rfft(s_rev, N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(S_rev * W, N)
    return 2.0 * conv[n-1 : 2*n-1]

def lp_flat_direction(s, M, time_limit=5.0):
    """
    Solve LP: maximize t subject to conv(s,b)[k] >= t, conv(s,b)[k] <= M, sum(b)=1, b >= 0.
    Returns (b, t) or (None, None).
    """
    n = len(s)
    num_lags = 2 * n - 1
    
    # Decision variables: b[0..n-1], t -> total n+1 variables
    # Objective: maximize t -> minimize -t
    c = np.zeros(n + 1)
    c[n] = -1.0  # minimize -t
    
    # Constraints:
    # conv(s,b)[k] - t <= 0 for all k (upper bound)
    # -conv(s,b)[k] + t <= 0 for all k (lower bound, i.e., conv >= t)
    # sum(b) = 1 (handled by A_eq)
    
    # conv(s,b)[k] = sum_j s[j] b[k-j] (with b out of range = 0)
    # For variable b[j]: coefficient in conv[k] is s[k-j] if 0 <= k-j < n, else 0
    
    # Build A_ub matrix: (2*(2n-1), n+1)
    A_ub = np.zeros((2 * num_lags, n + 1))
    b_ub = np.zeros(2 * num_lags)
    
    for k in range(num_lags):
        # Row k: conv(s,b)[k] - t <= M
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = s[idx]
        A_ub[k, n] = -1.0  # -t
        b_ub[k] = M
        
        # Row k + num_lags: -(conv(s,b)[k]) + t <= 0
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k + num_lags, j] = -s[idx]
        A_ub[k + num_lags, n] = 1.0  # +t
        b_ub[k + num_lags] = 0.0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # b >= 0, t free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x[:n], 0.0)
            b = b / b.sum()  # renormalize
            t = res.x[n]
            return b, t
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None, None

def lp_peak_direction(s, M, time_limit=5.0):
    """
    Solve LP: maximize sum(b) subject to conv(s,b)[k] <= M, b >= 0.
    Returns b or None.
    """
    n = len(s)
    num_lags = 2 * n - 1
    
    c = -np.ones(n)  # minimize -sum(b)
    
    A = np.zeros((num_lags, n))
    for k in range(num_lags):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = s[idx]
    
    b_ub = np.full(num_lags, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-10:
                b = b / b.sum()
                return b
    except Exception:
        pass
    return None
```

```python
"""
Minimize 2*n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
Uses FFT-based subgradient descent with adaptive peak-focusing,
LP-guided direction finding (flatness optimization), diverse initialization,
and adaptive restart strategies.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2 using FFT."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    N = 1
    while N < 2 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    conv = np.fft.irfft(S * S, N)[:2 * n - 1]
    return float(2.0 * n * np.max(conv) / (total * total))


def compute_autocorr(s):
    """Compute full autocorrelation (s*s) of length 2n-1 using FFT."""
    n = len(s)
    N = 1
    while N < 2 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    return np.fft.irfft(S * S, N)[:2 * n - 1]


def autocorr_gradient_fft(s, w):
    """
    Compute gradient of sum_m w[m]*(s*s)[m] w.r.t. s using FFT.
    s: length n, w: length 2n-1. Returns gradient of length n.
    Uses: grad[i] = 2 * (s_rev * w)[i+n-1] where s_rev = s[::-1]
    """
    n = len(s)
    s_rev = s[::-1]
    total_len = n + (2 * n - 1) - 1
    N = 1
    while N < total_len:
        N *= 2
    S_rev = np.fft.rfft(s_rev, N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(S_rev * W, N)
    return 2.0 * conv[n - 1:2 * n - 1]


def lp_flat_direction(s, M, time_limit=4.0):
    """
    LP: maximize t s.t. conv(s,b)[k] >= t, conv(s,b)[k] <= M, sum(b)=1, b>=0.
    Finds direction b that makes conv(s,b) as flat as possible under peak cap M.
    """
    n = len(s)
    num_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = -1.0

    A_ub = np.zeros((2 * num_lags, n + 1))
    b_ub = np.zeros(2 * num_lags)

    for k in range(num_lags):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A_ub[k, j] = s[k - j]
        A_ub[k, n] = -1.0
        b_ub[k] = M

        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A_ub[k + num_lags, j] = -s[k - j]
        A_ub[k + num_lags, n] = 1.0
        b_ub[k + num_lags] = 0.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x[:n], 0.0)
            if b.sum() > 1e-10:
                return b / b.sum()
    except Exception:
        pass
    return None


def lp_peak_direction(s, M, time_limit=4.0):
    """
    LP: maximize sum(b) s.t. conv(s,b)[k] <= M for all k, b >= 0.
    Finds a 'large' direction that doesn't increase the peak.
    """
    n = len(s)
    num_lags = 2 * n - 1
    c = -np.ones(n)
    A = np.zeros((num_lags, n))
    for k in range(num_lags):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, j] = s[k - j]
    b_ub = np.full(num_lags, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-10:
                return b / b.sum()
    except Exception:
        pass
    return None


def subgradient_step(s, beta, eta):
    """One multiplicative subgradient step with exponential peak focusing."""
    n = len(s)
    conv = compute_autocorr(s)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return s.copy()

    w = np.exp(beta * (conv - max_conv) / max_conv)
    w = w / w.sum()

    grad = autocorr_gradient_fft(s, w)
    grad = np.maximum(grad, 0.0)
    grad_max = grad.max()
    if grad_max < 1e-15:
        return s.copy()

    s_new = s * np.exp(-eta * grad / grad_max)
    s_new = np.maximum(s_new, 1e-15)
    s_new = s_new / s_new.sum()
    return s_new

    beta = 4.0
    no_improve = 0
    patience = 400
    lp_interval = 25 if n <= 384 else 50
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # Stalemate recovery
        if no_improve > patience:
            perturbation_scale = min(0.1, 0.02 + 0.01 * (no_improve // patience))
            s = best_s + perturbation_scale * rng.standard_normal(n)
            s = np.maximum(s, 0.005)
            s = s / s.sum()
            eta = 0.15
            beta = float(rng.choice([3.0, 5.0, 8.0, 10.0, 15.0]))
            no_improve = 0
            continue

        # LP-guided step
        if use_lp and it % lp_interval == lp_interval - 1:
            conv = compute_autocorr(s)
            M = conv.max()
            cur_score = fast_score(s)

            # Try flat direction LP
            b = lp_flat_direction(s, M, time_limit=3.0)
            improved = False
            if b is not None:
                best_t = 0.0
                best_sc = cur_score
                for alpha in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6]:
                    cand = (1 - alpha) * s + alpha * b
                    cand = np.maximum(cand, 0.0)
                    csum = cand.sum()
                    if csum < 1e-10:
                        continue
                    cand = cand / csum
                    sc = fast_score(cand)
                    if sc < best_sc:
                        best_sc = sc
                        best_t = alpha
                if best_t > 0:
                    s = (1 - best_t) * s + best_t * b
                    s = np.maximum(s, 0.0) / s.sum()
                    if best_sc < best_score:
                        best_score = best_sc
                        best_s = s.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    improved = True

            if not improved:
                # Try peak direction LP
                b = lp_peak_direction(s, M, time_limit=3.0)
                if b is not None:
                    best_t = 0.0
                    best_sc = cur_score
                    for alpha in [0.1, 0.2, 0.3, 0.5]:
                        cand = (1 - alpha) * s + alpha * b
                        cand = np.maximum(cand, 0.0)
                        cand = cand / cand.sum()
                        sc = fast_score(cand)
                        if sc < best_sc:
                            best_sc = sc
                            best_t = alpha
                    if best_t > 0:
                        s = (1 - best_t) * s + best_t * b
                        s = np.maximum(s, 0.0) / s.sum()
                        if best_sc < best_score:
                            best_score = best_sc
                            best_s = s.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
            continue

        # Subgradient step
        s_new = subgradient_step(s, beta, eta)
        sc_new = fast_score(s_new)
        sc_cur = fast_score(s)

        if sc_new < best_score:
            best_score = sc_new
            best_s = s_new.copy()
            s = s_new
            no_improve = 0
            # Reward improvement: slightly increase step
            eta = min(eta * 1.02, 0.4)
        elif sc_new < sc_cur:
            s = s_new
            no_improve += 1
        else:
            no_improve += 1
            # Penalize non-improvement
            if no_improve % 20 == 0:
                eta *= 0.92

        # Gradually increase beta (focus more on peaks)
        if it % 200 == 0:
            beta = min(beta + 0.5, 40.0)

        # Periodic re-seeding: boost very small elements
        if it % 500 == 0:
            min_val = s.min()
            if min_val < 1e-6:
                s = s + 1e-5
                s = s / s.sum()

    return best_s, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(s,s))/(sum(s))^2.
    Combines FFT-based adaptive subgradient descent with LP-guided corrections,
    diverse initialization, multiple sequence lengths, and final polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 40.0

    global_best_score = float('inf')
    global_best_seq = np.ones(256) / 256.0

    # Load previous best
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0, 1000)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            print(f"Loaded height_sequence_1, length={h.size}, score={fast_score(h)}")
    except Exception:
        pass

    # Phase 1: Broad search across lengths
    lengths = [256, 384, 512]

    for n in lengths:
        if time.monotonic() > deadline - 90:
            break
        remaining = deadline - time.monotonic()
        print(f"\n=== Phase 1: n={n}, remaining={remaining:.0f}s ===")

        inits = generate_initializations(n, rng, hseq)

        # Score and sort
        scored = []
        for s in inits:
            s = np.maximum(s, 0).astype(np.float64)
            if s.sum() < 1e-10:
                continue
            s = s / s.sum()
            sc = fast_score(s)
            scored.append((sc, s))
        scored.sort(key=lambda x: x[0])

        print(f"  {len(scored)} initializations, best={scored[0][0]:.6f}")

        # Optimize top candidates
        num_top = min(3, len(scored))
        time_per = max(30.0, (remaining - 60) / num_top)

        for i in range(num_top):
            if time.monotonic() > deadline - 30:
                break
            sc0, s0 = scored[i]
            print(f"  Opt init {i}: score={sc0:.6f}")

            use_lp = (n <= 384)
            best_s, best_sc = main_optimize(s0, time.monotonic() + time_per, rng, use_lp=use_lp)

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** New best: {best_sc:.6f} (n={n})")

    # Phase 2: Intensive refinement on the best found
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 2: Refinement, current best={global_best_score:.6f} ===")
        n = len(global_best_seq)
        remaining = deadline - time.monotonic()

        # Run multiple refinement rounds with different params
        for round_num in range(3):
            if time.monotonic() > deadline - 10:
                break
            sub_deadline = min(time.monotonic() + remaining / (3 - round_num), deadline - 10)
            best_s, best_sc = main_optimize(global_best_seq, sub_deadline, rng, use_lp=(n <= 384))
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Refinement {round_num}: {best_sc:.6f}")
            else:
                print(f"  Refinement {round_num}: no improvement ({best_sc:.6f})")

    # Phase 3: Try also n matching the best length with new random starts
    if time.monotonic() < deadline - 20:
        n_best = len(global_best_seq)
        print(f"\n=== Phase 3: Random restarts at n={n_best} ===")
        for restart in range(5):
            if time.monotonic() > deadline - 10:
                break
            s_r = rng.random(n_best) ** 0.5
            s_r = np.maximum(s_r, 0.01)
            s_r = s_r / s_r.sum()
            # Blend with current best
            s_r = 0.7 * s_r + 0.3 * global_best_seq
            s_r = np.maximum(s_r, 0) / s_r.sum()

            best_s, best_sc = main_optimize(s_r, time.monotonic() + 25, rng, use_lp=(n_best <= 384))
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Restart {restart}: new best {best_sc:.6f}")

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_score = float(evaluate_sequence(result))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(global_best_seq)}")

    return result
```
