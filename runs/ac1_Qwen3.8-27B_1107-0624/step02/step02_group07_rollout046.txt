    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def softmax_gradient(s, beta):
    """Compute gradient of soft-max objective using FFT."""
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = np.max(conv)
    if max_conv <= 0:
        return np.zeros(n), conv
    w = np.exp(beta * (conv - max_conv) / max_conv)
    # Compute cross-correlation w ⊛ s
    N_fft = 1
    while N_fft < 3 * n:
        N_fft *= 2
    W = np.fft.rfft(w, N_fft)
    S = np.fft.rfft(s, N_fft)
    cross = np.fft.irfft(np.conj(S) * W, N_fft)
    grad = 2.0 * cross[0:n]
    return grad, conv

def multiplicative_update(s, grad, eta):
    """Multiplicative gradient update."""
    grad_max = np.max(grad)
    if grad_max < 1e-15:
        return s.copy()
    s_new = s * np.exp(-eta * grad / grad_max)
    s_new = np.maximum(s_new, 1e-15)
    return s_new / s_new.sum()

    n = len(s)

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


"""
Minimizes 2*n*max(conv(s,s))/(sum(s))^2 using:
- Multi-scale soft-max gradient descent via FFT
- LP-based descent directions
- Spectral equalization (gentle flattening of |FFT(s)|^2)
- Backtracking line search
- Diverse asymmetric initializations including Fourier perturbations
- Multiple sequence lengths with adaptive time allocation
"""


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def compute_conv(s):
    """Compute full convolution of s with itself."""
    return np.convolve(s, s)


def softmax_gradient_fft(s, beta):
    """
    Compute gradient of soft-max weighted convolution sum using FFT.
    Returns (gradient, conv, max_conv).
    """
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = np.max(conv)
    if max_conv <= 0:
        return np.zeros(n), conv, max_conv
    w = np.exp(beta * (conv - max_conv) / max_conv)
    # Cross-correlation: grad[j] = 2 * sum_k w[j+k] * s[k]
    N_fft = 1
    while N_fft < 3 * n:
        N_fft *= 2
    W = rfft(w, N_fft)
    S = rfft(s, N_fft)
    cross = irfft(np.conj(S) * W, N_fft)
    grad = 2.0 * cross[0:n]
    return grad, conv, max_conv


def spectral_equalize(s, strength=0.02):
    """
    Gently flatten the power spectrum of s to reduce convolution sidelobes.
    Returns a new non-negative sequence.
    """
    n = len(s)
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    S = rfft(s, N_fft)
    magnitude = np.abs(S)
    # Target: flat spectrum (geometric mean of magnitudes as target)
    nonzero = magnitude > 1e-12
    if np.sum(nonzero) < 2:
        return s.copy()
    target = np.median(magnitude[nonzero])
    # Blend: new_magnitude = (1-strength)*old + strength*target
    new_mag = (1.0 - strength) * magnitude + strength * target
    # Keep phase
    phase = S / np.where(magnitude > 1e-12, magnitude, 1.0)
    S_new = new_mag * phase
    # Ensure DC component is preserved (real sum)
    s_new = irfft(S_new, N_fft)[:n]
    s_new = np.maximum(s_new, 0.0)
    if s_new.sum() < 1e-12:
        return s.copy()
    return s_new / s_new.sum()


def backtracking_line_search(s, direction, step_size, max_iters=20, back_factor=0.5, c1=0.01):
    """
    Backtracking line search for multiplicative updates.
    direction: the "force" (negative gradient direction in log-space)
    Returns updated s and the step taken.
    """
    cur_score = fast_score(s)
    step = step_size
    best_s = s.copy()
    best_sc = cur_score
    for _ in range(max_iters):
        s_try = s * np.exp(step * direction)
        s_try = np.maximum(s_try, 1e-15)
        s_try = s_try / s_try.sum()
        sc_try = fast_score(s_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_s = s_try.copy()
        if sc_try <= cur_score - c1 * step * np.sum(direction * np.log(np.maximum(s, 1e-15))):
            return best_s, step
        step *= back_factor
        if step < 1e-6:
            break
    return best_s, step


def lp_direction(s, M, time_limit=2.0):
    """
    Solve LP: maximize sum(b) s.t. conv(s,b)[m] <= M for all m, b >= 0.
    Returns b (normalized) or None.
    """
    n = len(s)
    if n > 512:
        return None
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        m_start = j
        m_end = min(j + n - 1, 2 * n - 2)
        A[m_start:m_end + 1, j] = s[0:m_end - m_start + 1]
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-12:
                return b / b.sum()
    except Exception:
        pass
    return None

            h_r = np.maximum(h_r, 0.0)

                h_r = h_r / h_r.sum()
                candidates.insert(0, (fast_score(h_r), h_r.copy()))

        if hseq is not None and hseq.size == n:
            h_norm = hseq / hseq.sum()
            candidates.insert(0, (fast_score(h_norm), h_norm.copy()))

        candidates.sort(key=lambda x: x[0])

        # Run local search on top candidates
        num_cands = min(5, len(candidates))
        time_per_cand = alloc / num_cands

        for i in range(num_cands):
            if time.monotonic() > deadline - 20:
                break
            sc0, s0 = candidates[i]
            # Allocate more time to better candidates
            t_alloc = time_per_cand * (1.0 + 0.5 * (1.0 / (1.0 + i)))
            t_alloc = min(t_alloc, deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break

            cand_deadline = time.monotonic() + t_alloc
            use_lp = (n <= 400)

            best_s, best_sc = local_search(s0, cand_deadline, rng, use_lp=use_lp)

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** New global best: {best_sc:.6f} (n={n}, cand={i}, init_sc={sc0:.6f})")

    # Final refinement on the best sequence
    if time.monotonic() < deadline - 10:
        n_best = len(global_best_seq)
        print(f"\n=== Final refinement (n={n_best}), current best={global_best_score:.6f} ===")
        # Refine with LP
        best_s, best_sc = local_search(global_best_seq, deadline - 5, rng, use_lp=(n_best <= 400))
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s.copy()

        # One more spectral equalization pass
        for strength in [0.01, 0.02, 0.05, 0.1]:
            s_eq = spectral_equalize(global_best_seq, strength=strength)
            sc_eq = fast_score(s_eq)
            if sc_eq < global_best_score:
                global_best_score = sc_eq
                global_best_seq = s_eq.copy()
                print(f"  Spectral eq (strength={strength}): {sc_eq:.6f}")

        # Final gradient polish
        if time.monotonic() < deadline - 5:
            best_s, best_sc = local_search(global_best_seq, deadline - 2, rng, use_lp=False)
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()

    # Verify with actual evaluator
    final_seq = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_score = float(evaluate_sequence(final_seq))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(final_seq)}")

    return final_seq
