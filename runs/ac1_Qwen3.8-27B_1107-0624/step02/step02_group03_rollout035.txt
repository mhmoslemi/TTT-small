            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    h[i] += a[j]
        
        h /= len(S_idx)
        
        # Support: where h is significant
        support = h > h.max() * 1e-6
        
        if not support.any():
            break
        
        a_new = np.zeros(n)
        a_new[support] = h[support]
        a_new = norm1(a_new)
        
        if a_new is None:
            break
        
        # Check convergence
        if np.max(np.abs(a_new - a)) < tol:
            a = a_new
            break
        
        # Blend to avoid divergence
        a = 0.7 * a + 0.3 * a_new
        a = norm1(a)
    
    return a


def adam_polish(p0, deadline, lr_init=0.05, reg=1e-10, verbose=True):
    """
    Adam gradient descent in logit space.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    
    z = np.log(p0)
    z -= np.mean(z)
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    
    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    lr = lr_init
    beta = max(1e-12, 0.1 * max_c)
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())
        
        # Soft argmax
        x = (c - max_c) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s
        
        # Gradient
        h = np.zeros(n)
        for i in range(n):
            lo = max(0, i - (n - 1))
            hi = min(2 * n - 1, i + n - 1)
            # h[i] = sum_{k} w[k] * a[k-i] where 0 <= k-i < n
            # = sum_{k=i}^{i+n-1} w[k] * a[k-i] = sum_{j=0}^{n-1} w[i+j] * a[j]
            k_lo = max(0, i)
            k_hi = min(2*n-1, i + n - 1)
            j_lo = k_lo - i
            j_hi = k_hi - i
            h[i] = np.dot(w[k_lo:k_hi+1], p0[j_lo:j_hi+1])
        
        grad_p = 2.0 * h + reg * p0
        
        # Logit gradient
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0); v.fill(0); it = 0
            continue
        
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        
        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum
        
        c = np.convolve(p, p)
        sc = 2.0 * n * float(c.max())
        
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.005, lr * 0.9)
        
        if it % 100 == 0:
            beta = max(1e-14, beta * 0.7)
        
        if no_imp > 3000:
            # Restart
            rng = np.random.default_rng(it)
            z = 0.7 * best_z + 0.3 * rng.normal(size=n) * 0.5
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0); v.fill(0); it = 0
            lr = lr_init
            no_imp = 0
            beta = max(1e-12, 0.1 * float(np.convolve(
                np.exp(z - z.max()) / np.exp(z - z.max()).sum(),
                np.exp(z - z.max()) / np.exp(z - z.max()).sum()
            ).max()))
        
        p0 = p
    
    return best_p, best_score

"""
Multi-phase optimization: LP direction-finding with line search, KKT fixed-point,
and Adam gradient polish for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, timeout=5.0):
    n = len(f)
    A = build_conv_matrix(f)
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds,
                  method='highs', options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    """KKT fixed point: at optimum, correlation with active set is constant on support."""
    n = len(a)
    a = a.copy().astype(np.float64)
    
    for it in range(max_iters):
        c = np.convolve(a, a)
        M = c.max()
        
        # Active set (positions where c is close to max)
        active = c >= M * 0.9995
        n_active = active.sum()
        if n_active == 0:
            break

        h = np.zeros(n)

            break
