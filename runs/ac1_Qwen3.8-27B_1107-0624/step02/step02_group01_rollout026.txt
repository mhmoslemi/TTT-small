Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines DCT-basis L-BFGS optimization, soft-max gradient with Adam,
and LP-based direction finding. Explores multiple sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_max_conv(a):
    c = fft_convolve(a)
    return float(c.max())


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=8):

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def next_pow2(n):

def fft_convolve(a):
    n = a.size

    c = irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 20})
    if res.success:
        return res.x
    return None

    # Additional LP from a perturbed version

        p_pert = best_seq + 0.01 * rng.normal(size=n1)
        p_pert = np.maximum(p_pert, 0.0)
        p_pert = p_pert / p_pert.sum()
        per = min(p3_end, time.monotonic() + 100.0)
        a_lp2, M_lp2 = lp_optimize(p_pert, per, rng, max_iter=10000)
        try_update(a_lp2, 2.0 * n1 * M_lp2)

    print(f"After Phase 3: best={best_score:.10f}")

    # === Phase 4: Try n=1024 ===
    n2 = 1024
    p4_end = min(deadline, t_start + 1600.0)
    if best_seq is not None and time.monotonic() < p4_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            # Quick gradient at n=1024
            per = min(p4_end, time.monotonic() + 120.0)
            a_g1024, M_g1024 = softmax_adam(p_ext, per, rng, lr=0.03, t_init=15.0, t_min=500.0,
                                             max_iter=200000)
            try_update(a_g1024, 2.0 * n2 * M_g1024)
            print(f"Phase 4 gradient (n=1024): score={2.0 * n2 * M_g1024:.10f}")

            # LP at n=1024
            if time.monotonic() < p4_end - 40:
                per2 = min(p4_end, time.monotonic() + 150.0)
                a_lp1024, M_lp1024 = lp_optimize(a_g1024, per2, rng, max_iter=15000)
                try_update(a_lp1024, 2.0 * n2 * M_lp1024)
                print(f"Phase 4 LP (n=1024): score={2.0 * n2 * M_lp1024:.10f}")

    # Also try direct LP at n=1024 from deterministic profiles
    if time.monotonic() < p4_end - 50:
        for p0 in make_deterministic_profiles(n2)[:3]:
            if time.monotonic() > p4_end - 30:
                break
            per = min(p4_end - 30, time.monotonic() + 40.0)
            a_b, M_b = softmax_adam(p0, per, rng, lr=0.04, t_init=20.0, t_min=300.0,
                                     max_iter=100000)
            try_update(a_b, 2.0 * n2 * M_b)

    print(f"After Phase 4: best={best_score:.10f}")

    # === Phase 5: Final LP polishing on the absolute best ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        per = min(deadline - 10, time.monotonic() + 120.0)
        a_final, M_final = lp_optimize(best_seq, per, rng, max_iter=20000)
        try_update(a_final, 2.0 * n_best * M_final)
        print(f"After Phase 5: best={best_score:.10f}, n={best_seq.size}")

    # === Final evaluation ===
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        # Verify with evaluator for small sequences
        if final.size <= 1024:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
