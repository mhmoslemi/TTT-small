"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP descent, accelerated smooth gradient descent, and coordinate descent.

import numpy as np

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = np.maximum(c, 0.0)
    return c

def build_conv_matrix(f):
    n = len(f)
    N = 2 * n - 1
    A = np.zeros((N, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_step(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def softmax_from_z(z):

    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:

    return c


def _build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_step(f, M):
    n = len(f)
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_optimize(f_init, deadline, rng, verbose=False):
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    conv_f = _full_convolve_fft(f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.3
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        if stagnant > 2:
            M = best_M
        else:
            M = float(_full_convolve_fft(f).max())

        g = _lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        # Line search: try decreasing t values
        best_new_f = None
        best_new_M = M + 1e-15
        for t_try in [t, t * 0.6, t * 0.3, t * 0.15]:
            f_new = (1 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new = f_new / s_new
            M_new = float(_full_convolve_fft(f_new).max())
            if M_new < best_new_M:
                best_new_M = M_new
                best_new_f = f_new
                if M_new <= M * 0.999:
                    break

        if best_new_f is not None:
            f = best_new_f
            if best_new_M < best_M - 1e-12:
                best_M = best_new_M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
            t = max(t * 0.995, 0.05)
        else:
            t *= 0.5
            stagnant += 1

        if verbose and iteration % 50 == 0:
            print(f"  LP iter {iteration}: M={best_M:.10f}, t={t:.3f}")

        if stagnant > 80:
            noise = 0.03 * rng.normal(size=n)
            f = best_f + noise
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.3
            stagnant = 0

    return best_f


def _softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-14
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _compute_grad_smoothed(p, c, beta):
    """Compute gradient of beta*logsumexp(c/beta) wrt p."""
    c_max = float(c.max())
    x = (c - c_max) / beta
    ex = np.exp(np.clip(x, -50, 0))
    s_ex = float(np.sum(ex))
    if s_ex <= 0 or not np.isfinite(s_ex):
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s_ex = 1.0
    w = ex / s_ex

    n = len(p)
    N_fft = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_fft)
    fp = np.fft.rfft(p[::-1], N_fft)
    h = np.fft.irfft(fw * fp, N_fft)
    gp = 2.0 * h[n - 1:2 * n - 1]
    return gp, w


def _gradient_optimize(p0, deadline, rng, lr=0.05, beta_scale=0.15, verbose=False):
    """Adam-based optimization in logit space with active-set guided restarts."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = beta_scale * maxc

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    iter_count = 0

    while time.monotonic() < deadline:
        gp, w = _compute_grad_smoothed(p, c, beta)
        vp = gp + 1e-9 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        iter_count += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        # Cosine annealing
        progress = min(iter_count / 3000.0, 1.0)
        lr_t = lr * 0.5 * (1.0 + np.cos(np.pi * progress))

        z = z - lr_t * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Beta schedule
        if iter_count % 25 == 0:
            beta = max(1e-16, beta * 0.72)

        # Active-set guided restart
        restart_thresh = max(300, int(1.5 * n))
        if no_imp > restart_thresh:
            # Find active set: where conv is close to max
            c_best = _full_convolve_fft(best_p)
            c_max_best = float(c_best.max())
            threshold = c_max_best * 0.95
            active_mask = c_best >= threshold

            # Perturb z more strongly at "hot" coordinates
            # Hot coords are those p[j] that contribute to active conv terms
            n_fft = _next_pow2(2 * n - 1)
            fw = np.fft.rfft(active_mask.astype(np.float64), n_fft)
            fp = np.fft.rfft(best_p[::-1], n_fft)
            hot = np.fft.irfft(fw * fp, n_fft)[n - 1:2 * n - 1]
            hot_norm = hot / (hot.max() + 1e-15)

            strength = min(0.5, 0.1 + 0.1 * (no_imp / restart_thresh))
            noise = rng.normal(size=n) * (0.3 + 0.7 * hot_norm)
            z = (1.0 - strength) * best_z + strength * noise
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            no_imp = 0
            iter_count = 0

        if verbose and iter_count % 200 == 0:
            print(f"    grad iter {iter_count}: score={best_score:.10f}, beta={beta:.2e}")

    return best_p, best_score


def _make_profile_linear(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, k=1, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_triangle(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exp(n, rate=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def _coord_descent_polish(p0, deadline, rng, max_sweeps=30):
    """Simple coordinate descent: scale individual coordinates and renormalize."""
    p = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 1e-14).copy()
    p = p / p.sum()
    n = p.size
    score = _compute_score(p)
    best_p = p.copy()
    best_score = score

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            old_val = p[j]
            if old_val < 1e-12:
                continue
            for factor in [0.6, 0.8, 1.25, 1.5]:
                new_val = old_val * factor
                p[j] = new_val
                p_sum = p.sum()
                if p_sum < 0.01:
                    p[j] = old_val
                    continue
                p = p / p_sum
                new_score = _compute_score(p)
                if new_score < score - 1e-13:
                    score = new_score
                    improved = True
                    if score < best_score:
                        best_score = score
                        best_p = p.copy()
                    break
                else:
                    p = best_p.copy() if score > best_score else p
                    p[j] = old_val
                    p_sum = p.sum()
                    p = p / p_sum if p_sum > 0.01 else p
        if not improved:
            break
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800):
    global _G_best_seq, _G_best_score
    _G_best_seq = None
    _G_best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def _upd(p, sc):
        global _G_best_seq, _G_best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < _G_best_score:
            _G_best_seq = np.asarray(p, dtype=np.float64).copy()
            _G_best_score = float(sc)

    print(f"=== Multi-phase search started, budget={budget_s}s ===")

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    if init_seq is not None:
        sc = _compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc:.10f}")
        _upd(init_seq, sc)
    else:
        init_seq = _make_profile_linear(512, 1.5)
        sc = _compute_score(init_seq)
        print(f"Default init: score={sc:.10f}")
        _upd(init_seq, sc)

    # === Phase 1: LP at n=512 ===
    n1 = 512
    phase1_deadline = min(deadline, t_start + 380.0)

    if init_seq.size != n1:
        p1 = _interpolate_profile(init_seq, n1)
    else:
        p1 = init_seq.copy()

    sc = _compute_score(p1)
    _upd(p1, sc)
    print(f"Phase 1: LP n={n1}, start={sc:.10f}")

    result = _lp_optimize(p1, phase1_deadline, rng, verbose=True)
    sc = _compute_score(result)
    _upd(result, sc)
    print(f"After Phase 1 LP: score={sc:.10f}")

    # Quick gradient after LP at n=512
    if time.monotonic() < phase1_deadline + 50 and _G_best_seq is not None and _G_best_seq.size == n1:
        gd_dead = min(deadline, time.monotonic() + 50.0)
        rp, rs = _gradient_optimize(_G_best_seq, gd_dead, rng, lr=0.06, beta_scale=0.12)
        _upd(rp, rs)
        print(f"  Quick gradient n=512: score={rs:.10f}")

    # === Phase 2: Extended gradient at n=512 ===
    phase2_deadline = min(deadline, t_start + 800.0)

    if _G_best_seq is not None and _G_best_seq.size == n1:
        print(f"Phase 2: Gradient n={n1}, current={_G_best_score:.10f}")
        for lr in [0.05, 0.08, 0.03]:
            if time.monotonic() > phase2_deadline - 20:
                break
            run_dl = min(phase2_deadline, time.monotonic() + 100.0)
            p0 = _G_best_seq + 0.001 * rng.normal(size=n1)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            rp, rs = _gradient_optimize(p0, run_dl, rng, lr=lr, beta_scale=0.12)
            _upd(rp, rs)
            print(f"  Grad lr={lr}: score={rs:.10f}")

        # Coordinate descent polish
        if time.monotonic() < phase2_deadline - 20:
            cd_dead = min(phase2_deadline, time.monotonic() + 60.0)
            rp, rs = _coord_descent_polish(_G_best_seq, cd_dead, rng)
            _upd(rp, rs)
            print(f"  Coord descent: score={rs:.10f}")

    print(f"After Phase 2: best={_G_best_score:.10f}")

    # === Phase 3: Try n=1024 ===
    n3 = 1024
    phase3_deadline = min(deadline, t_start + 1300.0)

    if _G_best_seq is not None and time.monotonic() < phase3_deadline - 50:
        p3 = _interpolate_profile(_G_best_seq, n3)
        sc3 = _compute_score(p3)
        print(f"Phase 3: n={n3}, interp={sc3:.10f}")
        _upd(p3, sc3)

        if sc3 < _G_best_score + 0.03:
            lp7_dead = min(phase3_deadline, time.monotonic() + 120.0)
            result3 = _lp_optimize(p3, lp7_dead, rng, verbose=False)
            sc3 = _compute_score(result3)
            _upd(result3, sc3)
            print(f"  After LP n=1024: score={sc3:.10f}")

            if time.monotonic() < phase3_deadline - 30:
                g7_dead = min(phase3_deadline, time.monotonic() + 100.0)
                rp, rs = _gradient_optimize(result3, g7_dead, rng, lr=0.04, beta_scale=0.1)
                _upd(rp, rs)
                print(f"  After grad n=1024: score={rs:.10f}")

    print(f"After Phase 3: best={_G_best_score:.10f}")

    # === Phase 4: Try other lengths if best is at 512 ===
    if _G_best_seq is not None and _G_best_seq.size == 512 and time.monotonic() < deadline - 100:
        for n_try in [768, 640]:
            if time.monotonic() > deadline - 80:
                break
            p_try = _interpolate_profile(_G_best_seq, n_try)
            sc_try = _compute_score(p_try)
            if sc_try >= _G_best_score + 0.01:
                continue
            print(f"  Trying n={n_try}: interp={sc_try:.10f}")
            _upd(p_try, sc_try)
            # Short gradient
            gt_dead = min(deadline - 60, time.monotonic() + 80.0)
            rp, rs = _gradient_optimize(p_try, gt_dead, rng, lr=0.04, beta_scale=0.12)
            _upd(rp, rs)
            print(f"  After grad n={n_try}: score={rs:.10f}")

    # === Phase 5: Final intensive polish at best length ===
    phase5_deadline = min(deadline, t_start + budget_s - 40.0)
    if _G_best_seq is not None and time.monotonic() < phase5_deadline - 20:
        n5 = _G_best_seq.size
        print(f"Phase 5: Final polish n={n5}, score={_G_best_score:.10f}")

        # Aggressive gradient
        for lr in [0.02, 0.03, 0.015]:
            if time.monotonic() > phase5_deadline - 15:
                break
            run_dl = min(phase5_deadline, time.monotonic() + 70.0)
            p0 = _G_best_seq + 0.0005 * rng.normal(size=n5)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            rp, rs = _gradient_optimize(p0, run_dl, rng, lr=lr, beta_scale=0.08)
            _upd(rp, rs)
            print(f"  Final grad lr={lr}: score={rs:.10f}")

    # === Phase 6: Very final LP if time and n allows ===
    if _G_best_seq is not None and _G_best_seq.size <= 768 and time.monotonic() < deadline - 15:
        lp_final_dead = min(deadline, time.monotonic() + 40.0)
        result_f = _lp_optimize(_G_best_seq, lp_final_dead, rng, verbose=False)
        sc_f = _compute_score(result_f)
        _upd(result_f, sc_f)
        print(f"Final LP: score={sc_f:.10f}")

    # === Output ===
    if _G_best_seq is not None:
        final_seq = _G_best_seq
    else:
        final_seq = _make_profile_linear(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_profile_linear(256, 1.5)

    final_score = _compute_score(final_seq)
    print(f"=== FINAL: n={final_seq.size}, score={final_score:.10f} ===")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
