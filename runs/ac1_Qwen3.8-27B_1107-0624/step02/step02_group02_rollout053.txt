import numpy as np

from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M, max_iter=200):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'max_iter': max_iter, 'primal_feasibility_tolerance': 1e-9,
                               'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def soft_mx(c, beta):
    """Smooth max with log-sum-exp."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-14
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def gradient_polish(p0, deadline, rng, lr=0.05, reg=1e-8, beta_init=None,

    """Logit-space Adam polish with aggressive soft-max annealing."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    if beta_init is None:
        beta = max(1e-12, 0.1 * maxc)
    else:
        beta = beta_init

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    step = 0

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        step += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.005, lr * 0.97)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if step % beta_update_freq == 0:
            beta = max(1e-14, beta * beta_decay)

        if no_imp > max(500, 2 * n):
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
            if verbose:
                print(f"    Restart at step {step}, score={best_score:.10f}")

    return best_p, best_score


def make_ramp_profile(n, sharpness=1.0):
    """Ramp profile: (1+t)^sharpness normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 + t) ** sharpness
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_asymmetric_profile(n, alpha=0.5, k=1):
    """Asymmetric profile: ramp * (1 + alpha*cos(2*pi*k*t))."""
    t = np.linspace(0.0, 1.0, n)
    base = 1.0 + t
    mod = 1.0 + alpha * np.cos(2.0 * np.pi * k * t)
    a = base * mod
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_two_scale_profile(n, env_sharp=1.5, mod_amp=0.3, mod_freq=3):
    """Two-scale: smooth envelope * oscillatory modulation."""
    t = np.linspace(0.0, 1.0, n)
    env = (1.0 + t) ** env_sharp
    mod = 1.0 + mod_amp * np.sin(2.0 * np.pi * mod_freq * t)
    a = env * mod
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 1e-12)
    return result / result.sum()

    best_score = np.inf
    best_n = 0

    def update(p, sc):
        nonlocal best_seq, best_score, best_n
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            best_n = best_seq.size

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ============ Phase 1: LP at n=512 (0-350s) ============
    n1 = 512
    phase1_end = min(deadline, t_start + 350.0)
    print(f"\n=== Phase 1: LP at n={n1} ===")

    if init_seq is not None and init_seq.size != n1:
        p1 = interpolate_profile(init_seq, n1)
    elif init_seq is not None:
        p1 = init_seq.copy()
    else:
        p1 = make_ramp_profile(n1, 1.5)

    sc = compute_score(p1)
    update(p1, sc)
    print(f"Start: score={sc:.10f}")

    # Main LP run
    lp_end = min(phase1_end, time.monotonic() + 250.0)
    result = lp_optimize(p1, lp_end, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"After main LP: score={sc:.10f}")

    # Try alternate profiles with LP
    alt_profiles = [
        ("ramp_2", make_ramp_profile(n1, 2.0)),
        ("ramp_0.5", make_ramp_profile(n1, 0.5)),
        ("asym_1", make_asymmetric_profile(n1, 0.3, 1)),
        ("two_scale", make_two_scale_profile(n1, 1.5, 0.2, 2)),
        ("ramp_3", make_ramp_profile(n1, 3.0)),
    ]
    for name, p in alt_profiles:
        if time.monotonic() > phase1_end - 10:
            break
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.1:
            continue
        per_dead = min(phase1_end, time.monotonic() + 30.0)
        res = lp_optimize(p, per_dead, rng, verbose=False)
        sc = compute_score(res)
        update(res, sc)
        if sc < best_score + 0.01:
            print(f"  {name}: {sc:.10f}")

    # Quick gradient polish after LP
    if time.monotonic() < phase1_end - 10 and best_seq is not None:
        pol_end = min(phase1_end, time.monotonic() + 80.0)
        p_pol = best_seq.copy() + 0.001 * rng.normal(size=best_seq.size)
        p_pol = np.maximum(p_pol, 1e-12)
        p_pol = p_pol / p_pol.sum()
        res_pol, sc_pol = gradient_polish(p_pol, pol_end, rng, lr=0.04,
                                           beta_decay=0.93, verbose=True)
        sc = compute_score(res_pol)
        update(res_pol, sc)
        print(f"After Phase 1 polish: score={sc:.10f}")

    print(f"Phase 1 done: best={best_score:.10f} (n={best_n})")

    # ============ Phase 2: LP at n=1024 (350-700s) ============
    n2 = 1024
    phase2_end = min(deadline, t_start + 700.0)
    print(f"\n=== Phase 2: LP at n={n2} ===")

    if time.monotonic() < phase2_end - 20 and best_seq is not None:
        p2 = interpolate_profile(best_seq, n2)
        sc = compute_score(p2)
        update(p2, sc)
        print(f"Interpolated to n={n2}: score={sc:.10f}")

        lp_end2 = min(phase2_end, time.monotonic() + 250.0)
        result = lp_optimize(p2, lp_end2, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After n={n2} LP: score={sc:.10f}")

        # Gradient polish at n=1024
        if time.monotonic() < phase2_end - 10 and best_n == n2:
            pol_end2 = min(phase2_end, time.monotonic() + 60.0)
            p_pol = best_seq.copy() + 0.0005 * rng.normal(size=n2)
            p_pol = np.maximum(p_pol, 1e-12)
            p_pol = p_pol / p_pol.sum()
            res_pol, sc_pol = gradient_polish(p_pol, pol_end2, rng, lr=0.03,
                                               beta_decay=0.94, verbose=True)
            sc = compute_score(res_pol)
            update(res_pol, sc)
            print(f"After Phase 2 polish: score={sc:.10f}")

    print(f"Phase 2 done: best={best_score:.10f} (n={best_n})")

    # ============ Phase 3: Multi-size exploration (700-1200s) ============
    phase3_end = min(deadline, t_start + 1200.0)
    print(f"\n=== Phase 3: Multi-size exploration ===")

    test_sizes = [640, 720, 768, 800, 896, 960]
    for n_try in test_sizes:
        if time.monotonic() > phase3_end - 30 or best_seq is None:
            break
        p_try = interpolate_profile(best_seq, n_try)
        sc_try = compute_score(p_try)
        if sc_try >= best_score + 0.02:
            continue
        sc_try = compute_score(p_try)
        update(p_try, sc_try)
        print(f"  n={n_try}: interpolated score={sc_try:.10f}")

        # Quick LP + gradient
        per_lp = min(phase3_end, time.monotonic() + 40.0)
        res_lp = lp_optimize(p_try, per_lp, rng, verbose=False)
        sc_lp = compute_score(res_lp)
        update(res_lp, sc_lp)

        if time.monotonic() < phase3_end - 20:
            per_pol = min(phase3_end, time.monotonic() + 30.0)
            p_pol = res_lp.copy() + 0.001 * rng.normal(size=n_try)
            p_pol = np.maximum(p_pol, 1e-12)
            p_pol = p_pol / p_pol.sum()
            res_pol, sc_pol = gradient_polish(p_pol, per_pol, rng, lr=0.03,
                                               beta_decay=0.93)
            sc_pol = compute_score(res_pol)
            update(res_pol, sc_pol)
            print(f"  n={n_try}: after LP+polish={sc_pol:.10f}")

    print(f"Phase 3 done: best={best_score:.10f} (n={best_n})")

    # ============ Phase 4: Intensive gradient polish (1200-1650s) ============
    phase4_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 4: Intensive polishing ===")

    if best_seq is not None:
        # Multiple polishing rounds with different parameters
        polish_configs = [
            {"lr": 0.06, "beta_decay": 0.92, "time": 120},
            {"lr": 0.03, "beta_decay": 0.95, "time": 120},
            {"lr": 0.04, "beta_decay": 0.93, "time": 100},
            {"lr": 0.02, "beta_decay": 0.96, "time": 100},
        ]
        for i, cfg in enumerate(polish_configs):
            if time.monotonic() > phase4_end - 10:
                break
            per = min(phase4_end, time.monotonic() + cfg["time"])
            noise = 0.002 / (i + 1)
            p_pol = best_seq.copy() + noise * rng.normal(size=best_n)
            p_pol = np.maximum(p_pol, 1e-12)
            p_pol = p_pol / p_pol.sum()
            res_pol, sc_pol = gradient_polish(p_pol, per, rng, lr=cfg["lr"],
                                               beta_decay=cfg["beta_decay"], verbose=True)
            sc_pol = compute_score(res_pol)
            update(res_pol, sc_pol)
            print(f"  Polish round {i+1} (lr={cfg['lr']}): {sc_pol:.10f}")

    print(f"Phase 4 done: best={best_score:.10f} (n={best_n})")

    # ============ Phase 5: Final LP + micro-polish (1650-1775s) ============
    phase5_end = min(deadline, t_start + 1775.0)
    print(f"\n=== Phase 5: Final refinement ===")

    if best_seq is not None and time.monotonic() < phase5_end - 20:
        # One more LP pass
        lp_end5 = min(phase5_end, time.monotonic() + 60.0)
        res_lp = lp_optimize(best_seq, lp_end5, rng, verbose=False)
        sc_lp = compute_score(res_lp)
        update(res_lp, sc_lp)
        print(f"  Final LP: {sc_lp:.10f}")

        # Micro gradient polish
        if time.monotonic() < phase5_end - 5:
            per = min(phase5_end, time.monotonic() + 30.0)
            p_pol = best_seq.copy() + 0.0003 * rng.normal(size=best_n)
            p_pol = np.maximum(p_pol, 1e-12)
            p_pol = p_pol / p_pol.sum()
            res_pol, sc_pol = gradient_polish(p_pol, per, rng, lr=0.01,
                                               beta_decay=0.97)
            sc_pol = compute_score(res_pol)
            update(res_pol, sc_pol)
            print(f"  Micro polish: {sc_pol:.10f}")

    print(f"Phase 5 done: best={best_score:.10f} (n={best_n})")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_ramp_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_ramp_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
