"""
Optimized search for minimizing 2n*max(conv(a,a))/sum(a)^2 using:
1. Soft-max gradient descent in logit space (primary method)
2. LP-based direction finding (auxiliary)
3. Multi-scale exploration and adaptive temperature
4. Multiple diverse initializations
"""

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def softmax_from_logit(z):
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(z) / z.size
    return a / s


def compute_softmax_gradient(a, beta, N_fft):
    """Compute gradient of soft-max approx to max(conv(a,a)) w.r.t. logit z.
    Returns grad_z of shape (n,)."""
    n = a.size
    c = fft_convolve(a)
    M = c.max()
    # Softmax weights
    x = (c - M) / beta
    x = np.clip(x, -100.0, 50.0)
    ex = np.exp(x)
    S = ex.sum()
    if S < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / S
    
    # h_j = sum_i w_i * a_{i-j} (cross-correlation of w and a)
    # Using FFT: h = IFFT(FFT(w) * conj(FFT(a)))[:n]
    fw = np.fft.rfft(w, N_fft)
    fa = np.fft.rfft(a, N_fft)
    h = np.fft.irfft(fw * np.conj(fa), N_fft)[:n]
    
    # grad_z_k = 2 * a_k * (h_k - sum_j h_j * a_j)
    h_dot_a = float(np.dot(h, a))
    grad_z = 2.0 * a * (h - h_dot_a)
    
    return grad_z, M + beta * np.log(S + 1e-300)


def softmax_gradient_descent(a0, deadline, rng, beta_init=0.01, beta_min=0.001,
                              beta_decay=0.95, lr=0.5, max_iter=100000,
                              patience=2000, restart_mix=0.3):
    """
    Optimize a = softmax(z) to minimize soft-max approximation of max(conv(a,a)).
    Uses Adam optimizer.
    """
    n = a0.size
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    z = np.log(a + 1e-30)
    z = z - z.mean()
    
    beta = beta_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    N_fft = next_pow2(3 * n)
    
    c = fft_convolve(a)
    M_val = float(c.max())
    
    best_a = a.copy()
    best_M = M_val
    no_imp = 0
    best_z = z.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad_z, L = compute_softmax_gradient(a, beta, N_fft)
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            a = softmax_from_logit(z)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp += 1
            if no_imp > patience:
                break
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        a = softmax_from_logit(z)
        
        c = fft_convolve(a)
        M_new = float(c.max())
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive beta
        if t_adam % 500 == 0:
            beta = max(beta_min, beta * beta_decay)
        
        # Restart
        if no_imp > patience:
            z = (1.0 - restart_mix) * best_z + restart_mix * rng.normal(size=n) * 0.1
            z = np.clip(z, -100.0, 100.0)
            z = z - z.mean()
            a = softmax_from_logit(z)
            beta = max(beta_min, beta * 2)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
            c = fft_convolve(a)
            M_val = float(c.max())
    
    return best_a, best_M


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 20})
    if res.success and res.x is not None:
        return res.x
    return None


def lp_optimize(f0, deadline, rng, max_iter=10000):
    """LP-based iterative optimization."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size
    
    c = fft_convolve(f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        g = solve_lp_direction(f, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 30:
                f = best_f + 0.01 * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-30:
                    f = best_f.copy()
                    continue
                f = f / fs
                c = fft_convolve(f)
                M = float(c.max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
            continue
        
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 100:
                f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n)) + 1e-10
                f = f / f.sum()
                c = fft_convolve(f)
                M = float(c.max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
            continue
        
        g = g / g_sum
        
        improved = False
        for t in (0.3, 0.5, 0.2, 0.7, 0.1, 0.4, 0.6, 0.8):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            c_new = fft_convolve(f_new)
            M_new = float(c_new.max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break
        
        if not improved:
            no_imp += 1
            if no_imp % 50 == 0:
                f = 0.85 * best_f + 0.15 * np.abs(rng.normal(size=n)) + 1e-10
                f = f / f.sum()
                M = float(fft_convolve(f).max())
                no_imp = 0
    
    return best_f, best_M


def make_profiles(n, rng):
    """Generate diverse initial profiles."""
    profiles = []
    
    # Uniform
    profiles.append(np.ones(n) / n)
    
    # Linear ramps
    t = np.linspace(0, 1, n)
    for slope in (1.0, 1.5, 2.0, 3.0, 5.0):
        a = 1.0 + slope * t
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
        a = 1.0 + slope * (1.0 - t)
        profiles.append(a / a.sum())
    
    # Quadratic
    for a0, a1, a2 in ((1, 0, -1), (1, 0, -2), (1, 1, -2), (1, -1, 2),
                        (0.5, 0, -0.5), (1, 0, -0.5)):
        a = a0 + a1 * t + a2 * t * t
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    # Sine-based
    for k in (1, 2, 3):
        for amp in (0.3, 0.5, 0.7):
            a = 1.0 + amp * np.sin(2 * np.pi * k * t)
            a = np.maximum(a, 1e-10)
            profiles.append(a / a.sum())
            a = 1.0 + amp * np.cos(2 * np.pi * k * t)
            a = np.maximum(a, 1e-10)
            profiles.append(a / a.sum())
    
    # Smooth random
    for _ in range(5):
        a = np.ones(n)
        for k in range(1, 6):
            amp = rng.uniform(0.1, 0.5) / k
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + phase)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    # U-shaped (high at edges)
    a = 1.0 + 2.0 * (2 * t - 1) ** 2
    profiles.append(a / a.sum())
    a = 1.0 + 4.0 * (2 * t - 1) ** 2
    profiles.append(a / a.sum())
    
    # Triangular (peak in middle)
    a = 1.0 - 2.0 * np.abs(t - 0.5)
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())
    
    return profiles


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None
