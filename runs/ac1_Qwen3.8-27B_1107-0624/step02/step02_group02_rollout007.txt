import numpy as np

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def build_conv_matrix_rows(f, rows):
    """Build subset of rows of the conv matrix."""
    n = len(f)
    A = np.zeros((len(rows), n), dtype=np.float64)
    for idx, row in enumerate(rows):
        # Row r of conv matrix: A[r, j] = f[j] if 0 <= r-j < n
        # i.e., j ranges from max(0, r-n+1) to min(n-1, r)
        j_start = max(0, row - n + 1)
        j_end = min(n - 1, row)
        if j_end >= j_start:
            A[idx, j_start:j_end+1] = f[j_start:j_end+1]
    return A

"""
Optimized LP + local search for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses constraint generation LP with line search, peak shaving, Adam polish,
and multi-length exploration. Key improvements over baseline: adaptive step
size via line search, faster LP via cutting-plane, targeted peak shaving,
and diverse initialization.

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.asarray(np.convolve(a, a), dtype=np.float64)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def build_conv_matrix_rows(f, rows):
    """Build subset of rows of the conv matrix for constraint generation."""
    n = len(f)
    A = np.zeros((len(rows), n), dtype=np.float64)
    for idx, row in enumerate(rows):
        j_start = max(0, row - n + 1)
        j_end = min(n - 1, row)
        if j_end >= j_start:
            A[idx, j_start:j_end + 1] = f[j_start:j_end + 1]
    return A


def lp_solve(f, M, time_limit=5.0):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0 using cutting plane."""
    n = len(f)
    t_start = time.monotonic()

    # Initial binding constraints: top-8 peaks of conv(f,f)
    conv_ff = np.convolve(f, f)
    top_idx = np.argsort(conv_ff)[-min(8, 2 * n - 1):][::-1].tolist()
    binding = list(set(top_idx))

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    best_b = None

    for iteration in range(80):
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit - 1.0:
            break

        remaining = max(1.0, time_limit - elapsed)
        A = build_conv_matrix_rows(f, binding)
        b_ub = np.full(len(binding), M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': remaining})
            if not res.success or res.x is None:
                # Fallback to full LP
                A_full = build_conv_matrix(f)
                res = linprog(c_obj, A_ub=A_full, b_ub=np.full(2 * n - 1, M),
                             bounds=bounds, method='highs',
                             options={'time_limit': max(1.0, time_limit - elapsed)})
                if res.success and res.x is not None:
                    return np.maximum(res.x, 0.0)
                break
        except Exception:
            break

        b = np.maximum(res.x, 0.0)
        if b.sum() < 1e-12:
            break

        # Check full feasibility
        conv_fb = np.convolve(f, b)
        max_v = float(conv_fb.max())

        if max_v <= M * (1 + 1e-6) + 1e-10:
            best_b = b
            break
        else:
            new_row = int(np.argmax(conv_fb))
            if new_row not in binding:
                binding.append(new_row)
            else:
                # Constraint already present but violated (numerical issue)
                best_b = b
                break

    if best_b is None:
        # As last resort, try the full LP
        try:
            A_full = build_conv_matrix(f)
            res = linprog(c_obj, A_ub=A_full, b_ub=np.full(2 * n - 1, M),
                         bounds=bounds, method='highs', options={'time_limit': 3.0})
            if res.success and res.x is not None:
                best_b = np.maximum(res.x, 0.0)
        except Exception:
            pass

    return best_b


def lp_optimize(f_init, deadline, rng):
    """Main LP-based optimization with line search."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    conv_f = np.convolve(f, f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        M = float(np.convolve(f, f).max())

        g = lp_solve(f, M, time_limit=4.0)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        # Line search: try multiple step sizes
        best_t = 0.1
        best_score_new = np.inf
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new = f_new / s_new
            M_new = float(np.convolve(f_new, f_new).max())
            score_new = 2.0 * n * M_new  # normalized by sum=1
            if score_new < best_score_new:
                best_score_new = score_new
                best_t = t

        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new
        M_new = float(np.convolve(f_new, f_new).max())

        if M_new <= best_M + 1e-12:
            f = f_new
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
        else:
            stagnant += 1

        # Restart when stuck
        if stagnant > 80:
            noise_scale = 0.15 * max(1.0 / n, np.std(best_f))
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            stagnant = 0
            if time.monotonic() > deadline:
                break

    return best_f


def soft_mx(c, beta):
    """Smooth max with log-sum-exp."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights_fft(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 using FFT."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def adam_polish(p0, deadline, rng, lr=0.06):
    """Logit-space Adam polish with cosine annealing."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.15 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights_fft(w, p)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.02, lr * 0.97)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 60 == 0:
            beta = max(1e-14, beta * 0.75)

        if no_imp > max(600, 3 * n):
            z = 0.7 * best_z + 0.35 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_profile_cosine(n, k, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_profile_gaussian(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_profile_chirp(n, beta=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + beta * (t - 0.5) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_profile_power(n, gamma=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (t + 0.1) ** gamma
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_profile_mixed(n, rng):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0
    for k in range(1, 6):

    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 1e-12)
    return result / result.sum()


def peak_shave_local(f, max_iter=200, rng=None):
    """Local peak shaving via coordinate-like descent."""
    if rng is None:
        rng = np.random.default_rng(12345)
    f = f.copy().astype(np.float64)
    n = len(f)
    s = f.sum()
    f = f / s
    best_f = f.copy()
    best_M = float(np.convolve(f, f).max())
    step = 0.005

    for _ in range(max_iter):
        conv_f = np.convolve(f, f)
        k_star = int(np.argmax(conv_f))

        # Compute sensitivity: how much each f[i] contributes to conv_f[k_star]
        sens = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            contrib = 0.0
            if 0 <= j1 < n:
                contrib += f[j1]
            if 0 <= j2 < n:
                contrib += f[j2]
            sens[i] = contrib

        # Reduce top contributors, increase low contributors
        top_k = min(5, n // 4)
        bottom_k = min(5, n // 4)
        top_idx = np.argsort(sens)[-top_k:]
        bottom_idx = np.argsort(sens)[:bottom_k]

        # Pick one to reduce and one to increase
        i_red = top_idx[rng.integers(top_k)]
        i_add = bottom_idx[rng.integers(bottom_k)]
        if i_red == i_add:
            continue

        delta = step * max(f[i_red], 1e-10)
        new_f = f.copy()
        new_f[i_red] -= delta
        new_f[i_add] += delta
        new_f = np.maximum(new_f, 0.0)
        ns = new_f.sum()
        if ns < 1e-10:
            continue
        new_f = new_f / ns
        new_M = float(np.convolve(new_f, new_f).max())

        if new_M < best_M - 1e-14:
            best_M = new_M
            best_f = new_f.copy()
            f = new_f
            step = min(step * 1.05, 0.02)
        else:
            step = max(step * 0.9, 0.001)

    return best_f

        # Try a couple profiles at large n
        for p_small in profiles[:3]:
            if time.monotonic() > phase2_deadline - 10:
                break
            p_lp = interpolate_profile(p_small, n_large)
            sc_p = compute_score(p_lp)
            if sc_p >= best_score + 0.02:
                continue
            per_dead = min(phase2_deadline - 5, time.monotonic() + 60.0)
            result = lp_optimize(p_lp, per_dead, rng)
            sc = compute_score(result)
            update(result, sc)

    print(f"After Phase2: best={best_score:.10f}")

    # ============ Phase 3: Try n=768 if promising ============
    n_mid = 768
    phase3_deadline = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase3_deadline - 30 and best_seq is not None and best_score > 1.504:
        p_mid = interpolate_profile(best_seq, n_mid)
        sc = compute_score(p_mid)
        update(p_mid, sc)
        print(f"Phase3: n={n_mid}, initial score={sc:.10f}")

        lp_dead3 = min(phase3_deadline - 5, time.monotonic() + 150.0)
        result = lp_optimize(p_mid, lp_dead3, rng)
        sc = compute_score(result)
        update(result, sc)
        print(f"After mid LP: score={sc:.10f}")

    print(f"After Phase3: best={best_score:.10f}")

    # ============ Phase 4: Adam polish ============
    phase4_deadline = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase4_deadline - 10 and best_seq is not None:
        n_polish = best_seq.size
        print(f"Phase4: polishing n={n_polish}, best={best_score:.10f}")

        for lr in [0.06, 0.03, 0.04]:
            if time.monotonic() > phase4_deadline - 10:
                break
            per_dead = min(phase4_deadline - 5, time.monotonic() + 90.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.001 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = adam_polish(p0, per_dead, rng, lr=lr)
            update(result, sc)
            print(f"  Adam lr={lr}: score={sc:.10f}")

    print(f"After Phase4: best={best_score:.10f}")

    # ============ Phase 5: Final polish + peak shave ============
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = best_seq.size
        # Peak shave
        ps = peak_shave_local(best_seq, max_iter=300, rng=rng)
        sc_ps = compute_score(ps)
        update(ps, sc_ps)
        print(f"Final peak shave: score={sc_ps:.10f}")

        # Short Adam
        per = min(deadline - 10, time.monotonic() + 25.0)
        p0 = best_seq + 0.0005 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, sc = adam_polish(p0, per, rng, lr=0.02)
        update(result, sc)
        print(f"Final Adam: score={sc:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_profile_linear(512, 2.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_profile_linear(256, 2.0)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
