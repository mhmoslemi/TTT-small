"""
Optimized subgradient search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses momentum-accelerated soft-max subgradient descent in simplex space,
adaptive beta scheduling, multi-length escalation, and diverse initializations
including parent interpolation, raised-cosine, and spectral approaches.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c


def _softmax_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _corr_weights(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _score_normalized(p):
    n = p.size
    if n == 0:
        return np.inf
    s = float(p.sum())
    if s <= 1e-30:
        return np.inf
    p = p / s
    c = _full_convolve_fft(p)
    mval = float(c.max())
    return float(2.0 * n * mval)


def _make_parent_interp(parent, n_new):
    p0 = np.asarray(parent, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        a = np.maximum(p0, 0.0)
        s = a.sum()
        if s < 1e-30:
            return None
        return a / s
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    a = np.interp(x1, x0, p0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinc_squared(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * t) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3

def _make_poly(n, coeffs):

    a = np.zeros(n)
    for c in coeffs:
        a += c * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _proj_simplex(v):
    n = v.size
    v = v - np.mean(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _momentum_optimize(p0, deadline, rng, mu=0.9, lr_init=1.0,
                       beta_init=0.05, beta_decay=0.95, beta_min=1e-7,
                       restart_limit=300, noise_scale=0.2):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p = np.maximum(p0, 1e-12)
    p = p / p.sum()
    n = p.size

    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(beta_min, beta_init * maxc)

    best_p = p.copy()
    best_score = float(2.0 * n * maxc)

    v = np.zeros(n, dtype=np.float64)
    no_imp = 0
    it = 0
    lr = lr_init

    while time.monotonic() < deadline:
        it += 1
        # Soft-max approximation of max
        L, w = _softmax_weights(c, beta)
        # Gradient of max(conv(p,p)) w.r.t. p
        gp = 2.0 * _corr_weights(w, p)
        # Project to tangent space of simplex (remove mean component)
        gp = gp - np.mean(gp)

        # Step in p-space (not z-space) for better conditioning
        p_trial = p - lr * gp
        # Project to simplex
        p_trial = _proj_simplex(p_trial)
        # Ensure non-negative and normalize
        p_trial = np.maximum(p_trial, 0.0)
        s = p_trial.sum()
        if s < 1e-30:
            p_trial = p.copy()
            s = 1.0
        p_trial = p_trial / s

        # Nesterov momentum
        v = mu * v + (p_trial - p)
        p_new = p + v
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-30:
            p_new = p_trial
            s = 1.0
        p_new = p_new / s

        c_new = _full_convolve_fft(p_new)
        maxc_new = float(c_new.max())
        score = float(2.0 * n * maxc_new)

        if score < best_score - 1e-12:
            best_score = score
            best_p = p_new.copy()
            no_imp = 0
            # Slightly increase momentum on improvement
            mu = min(0.99, mu * 1.002)
        else:
            no_imp += 1
            # Reduce momentum slightly when stuck
            mu = max(0.7, mu * 0.999)

        p = p_new
        c = c_new
        maxc = maxc_new

        # Adaptive beta
        if it % 20 == 0:
            beta *= beta_decay
            if c.size >= 2:
                sorted_idx = np.argsort(c)[::-1]
                gap = c[sorted_idx[0]] - c[sorted_idx[1]]
                if gap < beta:
                    beta = max(beta_min, gap * 0.5)

        # Decay learning rate slowly
        if it % 100 == 0:
            lr *= 0.995

        # Restart
        if no_imp > restart_limit:
            noise = noise_scale * rng.normal(size=n)
            p = best_p + noise
            p = np.maximum(p, 1e-12)
            p = p / p.sum()
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(beta_min, 0.05 * maxc)
            v = np.zeros(n, dtype=np.float64)
            lr = lr_init * 0.5
            mu = 0.9
            no_imp = 0
            noise_scale = max(0.02, noise_scale * 0.7)

    return best_p, best_score


def _adam_logit_optimize(p0, deadline, rng, lr=0.1, beta_init_frac=0.1,
                         beta_decay=0.93, restart_limit=200):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0 + 1e-15)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    vhat = np.zeros(n)
    t_adam = 0

    c = _full_convolve_fft(p0)
    maxc = float(c.max())
    beta = max(1e-14, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        L, w = _softmax_weights(c, beta)
        gp = 2.0 * _corr_weights(w, p0)
        dot = float(np.dot(p0, gp))
        gz = p0 * (gp - dot)
        gz -= np.mean(gz)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m[:] = 0
            vhat[:] = 0
            t_adam = 0
            p0 = best_p.copy()
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        vhat = b2 * vhat + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = vhat / (1 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        e = np.exp(z - z.max())
        p_new = e + 1e-12
        p_new = p_new / p_new.sum()

        c = _full_convolve_fft(p_new)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if score < best_score - 1e-14:
            best_score = score
            best_p = p_new.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        p0 = p_new

        if t_adam % 30 == 0:
            beta = max(1e-14, beta * beta_decay)
            if c.size >= 2:
                si = np.argsort(c)[::-1]
                gap = c[si[0]] - c[si[1]]
                if gap < beta * 0.1:
                    beta = max(1e-14, beta * 0.3)

        if no_imp > restart_limit:
            ns = 0.15 if best_score < 1.53 else 0.3
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()
            c = _full_convolve_fft(p0)
            maxc = float(c.max())
            beta = max(1e-14, beta_init_frac * maxc)
            m[:] = 0
            vhat[:] = 0
            t_adam = 0
            no_imp = 0

    return best_p, best_score


def _coordinate_descent_polish(p0, deadline, rng, n_sweeps_max=2000):
    """
    Coordinate descent in simplex space. For each coordinate, try small perturbations
    and keep the best one. This is a simple but effective local search.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p = np.maximum(p0, 1e-12)
    p = p / p.sum()
    n = p.size

    c = _full_convolve_fft(p)
    best_score = float(2.0 * n * c.max())
    best_p = p.copy()

    step_sizes = [0.01 / n, 0.005 / n, 0.001 / n, 0.0005 / n]

    sweep = 0
    while time.monotonic() < deadline and sweep < n_sweeps_max:
        sweep += 1
        improved = False

        # Random coordinate order
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break
            for step in step_sizes:
                # Try increasing and decreasing p[i]
                for sign in [1, -1]:
                    dp = -sign * step * p[i]  # change to p[i]
                    if dp >= 0 and p[i] + dp < 0:
                        continue
                    if dp < 0 and p[i] + dp < -1e-15:
                        continue
                    # Keep sum constant: redistribute dp to a random other coordinate
                    j = int(rng.integers(0, n))
                    if j == i:
                        j = (i + 1) % n
                    p_trial = p.copy()
                    p_trial[i] += dp
                    p_trial[j] -= dp
                    if p_trial[i] < 0 or p_trial[j] < 0:
                        continue
                    p_trial = p_trial / p_trial.sum()

                    c_trial = _full_convolve_fft(p_trial)
                    sc = float(2.0 * n * c_trial.max())
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_p = p_trial.copy()
                        p = p_trial.copy()
                        c = c_trial
                        improved = True
                        break
                if improved:
                    break

        if not improved and sweep % 50 == 0:
            # Add small noise to escape
            p = best_p + 0.01 / n * rng.normal(size=n)
            p = np.maximum(p, 1e-12)
            p = p / p.sum()
            c = _full_convolve_fft(p)

        if not improved and sweep % 200 == 0:
            break

    return best_p, best_score


def _true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={p.size}) [{label}]")

    # Get parent
    parent = None
    parent_n = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                parent_n = parent.size
    except Exception:
        parent = None

    print(f"Parent: {'n=' + str(parent_n) if parent is not None else 'not available'}")

    # --- Build candidate pool ---
    candidates = []  # (p, n, label)

    # Deterministic profiles at multiple lengths
    for n in [256, 512, 1024]:
        candidates.append((_make_raised_cosine(n), n, f"rc_{n}"))
        candidates.append((_make_sinc_squared(n), n, f"sinc2_{n}"))
        candidates.append((_make_linear(n, 1.0), n, f"lin1_{n}"))
        candidates.append((_make_linear(n, 2.0), n, f"lin2_{n}"))
        candidates.append((_make_cubic(n, 1.5, -1.0), n, f"cubic_{n}"))
        candidates.append((_make_poly(n, [1.0, 0.3, -0.2, 0.1]), n, f"poly_{n}"))

    # Parent interpolations
    if parent is not None:
        for n in [512, 1024, 2048]:
            pi = _make_parent_interp(parent, n)
            if pi is not None:
                candidates.append((pi, n, f"parent_{n}"))
        if parent_n in [512, 1024, 2048]:
            candidates.append((parent.copy(), parent_n, f"parent_{parent_n}"))

    # Score all candidates
    scored = []
    for p, n, label in candidates:
        sc = _score_normalized(p)
        if np.isfinite(sc):
            scored.append((sc, p, n, label))
            update_best(p, sc, f"init_{label}")

    scored.sort(key=lambda x: x[0])
    print(f"Phase 0: {len(scored)} candidates, best={best_score:.10f}")

    # --- Phase 1: Optimize top candidates with momentum method ---
    phase1_end = min(deadline, t_start + 400.0)
    top_k = min(6, len(scored))
    for i in range(top_k):
        if time.monotonic() >= phase1_end:
            break
        sc, p, n, label = scored[i]
        per_end = min(phase1_end, time.monotonic() + 60.0)
        pb, scb = _momentum_optimize(p, per_end, rng,
                                     mu=0.9, lr_init=0.05,
                                     beta_init=0.05, beta_decay=0.97,
                                     restart_limit=200, noise_scale=0.15)
        update_best(pb, scb, f"momentum_{label}")

    print(f"Phase 1: best={best_score:.10f}")

    # --- Phase 2: Optimize with Adam-logit method ---
    phase2_end = min(deadline, t_start + 800.0)
    # Take best from phase 1 and top initial candidates
    phase2_candidates = []
    if best_seq is not None:
        phase2_candidates.append((best_seq.copy(), best_seq.size, "best_p1"))
    for i in range(min(3, len(scored))):
        sc, p, n, label = scored[i]
        phase2_candidates.append((p.copy(), n, f"adam_{label}"))

    for p, n, label in phase2_candidates:
        if time.monotonic() >= phase2_end:
            break
        per_end = min(phase2_end, time.monotonic() + 80.0)
        pb, scb = _adam_logit_optimize(p, per_end, rng, lr=0.08,
                                       beta_init_frac=0.1, beta_decay=0.94,
                                       restart_limit=150)
        update_best(pb, scb, label)

    print(f"Phase 2: best={best_score:.10f}")

    # --- Phase 3: Escalate to longer length ---
    phase3_end = min(deadline, t_start + 1300.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        for target_n in [2048, 4096]:
            if time.monotonic() >= phase3_end - 60:
                break
            p_interp = _make_parent_interp(best_seq, target_n)
            if p_interp is None:
                continue
            sc_interp = _score_normalized(p_interp)
            update_best(p_interp, sc_interp, f"interp_{target_n}")

            # Optimize at new length
            per_end = min(phase3_end, time.monotonic() + 120.0)
            pb, scb = _momentum_optimize(p_interp, per_end, rng,
                                         mu=0.9, lr_init=0.03,
                                         beta_init=0.04, beta_decay=0.97,
                                         restart_limit=250, noise_scale=0.1)
            update_best(pb, scb, f"momentum_{target_n}")

            # Adam-logit at new length
            if time.monotonic() < phase3_end - 30:
                per_end2 = min(phase3_end - 30, time.monotonic() + 60.0)
                pb2, scb2 = _adam_logit_optimize(pb, per_end2, rng, lr=0.05,
                                                 beta_init_frac=0.08, beta_decay=0.95,
                                                 restart_limit=100)
                update_best(pb2, scb2, f"adam_{target_n}")

    print(f"Phase 3: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # --- Phase 4: Coordinate descent polish ---
    phase4_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"  CD polish at n={best_seq.size}")
        pb, scb = _coordinate_descent_polish(best_seq, phase4_end, rng)
        update_best(pb, scb, "CD_polish")

    # --- Phase 5: Final intensive optimization ---
    phase5_end = deadline
    if best_seq is not None and time.monotonic() < phase5_end:
        remaining = phase5_end - time.monotonic()
        if remaining > 30:
            per_end = min(phase5_end, time.monotonic() + remaining * 0.6)
            pb, scb = _momentum_optimize(best_seq, per_end, rng,
                                         mu=0.95, lr_init=0.01,
                                         beta_init=0.02, beta_decay=0.98,
                                         beta_min=1e-8,
                                     restart_limit=400, noise_scale=0.05)
            update_best(pb, scb, "final_momentum")

            if time.monotonic() < phase5_end - 10:
                per_end2 = min(phase5_end, time.monotonic() + 20.0)
                pb2, scb2 = _adam_logit_optimize(best_seq, per_end2, rng, lr=0.02,
                                                 beta_init_frac=0.03, beta_decay=0.97,
                                                 restart_limit=100)
                update_best(pb2, scb2, "final_adam")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_raised_cosine(1024)

        # Exact score
        final_n = final_seq.size
        if final_n <= 4096:
            sc_exact = _true_score(final_seq)
            print(f"Exact score: {sc_exact:.10f} (n={final_n})")
            if sc_exact < best_score:
                best_score = sc_exact

        # Try evaluate_sequence
        if final_n <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception as e:
                print(f"Eval error: {e}")

        print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fb = _make_raised_cosine(1024)
    print(f"FALLBACK: n=1024, score={_score_normalized(fb):.10f}")
    return [float(x) for x in fb]
