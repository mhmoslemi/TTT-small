            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.005, lr * 0.95)
        
        # Update beta
        if t_adam % 30 == 0:
            beta = max(1e-15, beta * beta_decay)
        
        # Check for convergence
        if no_imp > 500:
            break
        
        # Time the iteration for debugging
        if t_adam % 1000 == 0:
            print(f"    GD: iter={t_adam}, score={score_new:.10f}, beta={beta:.2e}")
    
    return _normalize(best_p), 2.0 * n * best_score  # wait, best_score already includes the 2*n factor
    # Actually best_score = 2.0 * n * maxc, so it's the full score (with sum=1)
    return _normalize(best_p), best_score

def _lp_iterate(a, deadline, rng, n_max_lp_iters=200, t_init=0.1):
    """
    LP-based iterative refinement.
    At each step:
    1. Compute M = max(conv(a, a))
    2. Solve LP: max sum(g) s.t. conv(a, g) <= M, g >= 0
    3. Rescale g to have sum = sum(a)
    4. Update: a_new = (1-t)*a + t*g
    5. Check improvement, adjust t
    """
    n = len(a)
    a = a / a.sum()  # normalize to sum 1
    best_a = a.copy()
    best_score = _score(a)
    
    t = t_init
    no_imp = 0
    
    for it in range(n_max_lp_iters):
        if time.monotonic() > deadline:
            break
        
        M = np.max(_full_convolve(a))
        
        # Solve LP
        g = _lp_solve(a, M, n)
        if g is None or g is None:
            # LP failed, reduce threshold or break
            M *= 0.99
            continue
        
        sum_g = g.sum()
        if sum_g < 1e-10:
            break
        
        # Rescale g to match sum of a
        g = g * (1.0 / sum_g)  # sum(g) = 1 = sum(a)
        
        # Try the update with backtracking
        improved = False
        for _ in range(10):
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            score_new = _score(a_new)
            if score_new < best_score - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_score = score_new
                improved = True
                t = min(1.0, t * 1.1)  # increase step size slightly
                no_imp = 0
                break
            else:
                t = max(0.01, t * 0.7)  # reduce step size
        
        if not improved:
            no_imp += 1
            if no_imp > 20:
                break
    
    return best_a, best_score
