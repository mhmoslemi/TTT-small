import time

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    a_norm = a / s
    c = fft_convolve(a_norm)
    mval = float(c.max())

def coordinate_descent(f0, deadline, rng, max_iter=50000):
    """Coordinate descent: optimize one element at a time."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size
    
    c = fft_convolve(f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Pick a coordinate to optimize
        # Bias towards coordinates that appear in max convolution positions
        max_positions = np.where(c >= M - 1e-10)[0]
        # Coordinates involved in max positions
        involved = set()
        for kp in max_positions:
            for j in range(max(0, kp - n + 1), min(n, kp + 1)):
                involved.add(j)
        
        if len(involved) == 0:
            idx = rng.integers(n)
        else:
            idx = list(involved)[rng.integers(len(involved))]
        
        # Optimize f[idx] to minimize max(conv)
        # Try a few values
        best_val = f[idx]
        best_max = M
        
        # The optimal value should reduce the peak
        # Try values around current
        candidates = [f[idx] * 0.5, f[idx] * 0.7, f[idx] * 1.3, f[idx] * 1.5,
                      f[idx] * 0.3, f[idx] * 2.0, f[idx] + 0.001, max(0, f[idx] - 0.001)]
        
        for new_val in candidates:
            if new_val < 0:
                new_val = 0.0
            f_old = f[idx]
            f[idx] = new_val
            f = f / f.sum()
            c_new = fft_convolve(f)
            M_new = float(c_new.max())
            if M_new < best_max:
                best_max = M_new
                best_val = new_val
            f[idx] = f_old
        
        f[idx] = best_val
        f = f / f.sum()
        c = fft_convolve(f)
        M = float(c.max())
        
        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > 500:
            break
    
    return best_f, best_M

def make_profiles(n, rng):
    """Generate diverse initial profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    
    # Monotone decreasing
    for alpha in (1, 2, 3, 4, 5):
        a = (1.0 - t) ** alpha + 0.01
        profiles.append(a / a.sum())
    
    # Monotone increasing
    for alpha in (1, 2, 3):
        a = t ** alpha + 0.01
        profiles.append(a / a.sum())
    
    # U-shaped (high at edges)
    for alpha in (2, 3, 4):
        u = 2.0 * t - 1.0
        a = u ** alpha + 0.01
        profiles.append(a / a.sum())
    
    # Linear
    for slope in (0.5, 1.0, 2.0, 3.0, 5.0):
        a = 1.0 + slope * t
        profiles.append(a / a.sum())
        a = 1.0 + slope * (1.0 - t)
        profiles.append(a / a.sum())
    
    # Exponential
    for rate in (2, 5, 10, 20):
        a = np.exp(-rate * t) + 0.001
        profiles.append(a / a.sum())
        a = np.exp(rate * t) + 0.001
        profiles.append(a / a.sum())
    
    # Smooth random
    for _ in range(10):
        a = np.ones(n)
        for k in range(1, 15):
            amp = rng.uniform(0.1, 1.0) / k
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + phase)
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())
    
    # "Spike at edge" profiles
    for frac in (0.1, 0.2, 0.3, 0.5):
        a = np.ones(n)
        k = int(frac * n)
        a[:k] += 5.0
        a = a / a.sum()
        profiles.append(a)
        a = np.ones(n)
        a[-k:] += 5.0
        a = a / a.sum()
        profiles.append(a)
    
    return profiles

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()

    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None

        if time.monotonic() > deadline:
            break

        idx = int(order[it % n])

        f_val = f[idx]
        if f_val < 1e-20:
            f[idx] = 1e-10
            f = f / f.sum()
            c = fft_convolve(f)
            M = float(c.max())
            continue

        base = c.copy()
        for k in range(2 * n - 1):
            j1 = idx
            j2 = k - idx
            contrib = 0.0
            if 0 <= j2 < n:
                contrib += f_val * f[j2]
            if 0 <= j1 < n and j2 != j1:
                contrib += f[j2] * f_val if 0 <= j2 < n else 0.0
            # Actually let me just recompute properly
            pass

        # Simpler approach: just try scaling
        best_val = f_val
        best_max = M

        for factor in (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 3.0, 0.1):
            new_val = f_val * factor
            f[idx] = new_val
            fs = f.sum()
            if fs < 1e-30:
                f[idx] = f_val
                continue
            f = f / fs
            c_new = fft_convolve(f)
            M_new = float(c_new.max())
            if M_new < best_max - 1e-16:
                best_max = M_new
                best_val = new_val
            f[idx] = f_val

        f[idx] = best_val
        f = f / f.sum()
        c = fft_convolve(f)
        M = float(c.max())

        if M < best_M - 1e-16:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= n * 3:
            break

    return best_f, best_M


def make_profiles(n, rng):
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    for alpha in (1, 2, 3, 5, 8):
        a = (1.0 - t) ** alpha + 0.005
        profiles.append(a / a.sum())

    for alpha in (1, 2, 3):
        a = t ** alpha + 0.005
        profiles.append(a / a.sum())

    u = 2.0 * t - 1.0
    for alpha in (2, 3, 4, 6):
        a = u ** alpha + 0.005
        profiles.append(a / a.sum())

    for slope in (1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        a = 1.0 + slope * t
        profiles.append(a / a.sum())
        a = 1.0 + slope * (1.0 - t)
        profiles.append(a / a.sum())

    for rate in (3, 7, 15, 30, 60):
        a = np.exp(-rate * t) + 0.001
        profiles.append(a / a.sum())
        a = np.exp(rate * t) + 0.001
        profiles.append(a / a.sum())

    for _ in range(12):
        a = np.ones(n)
        for k in range(1, 20):
            amp = rng.uniform(0.05, 0.8) / k
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + phase)
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())

    for frac in (0.1, 0.2, 0.3, 0.5, 0.7):
        k = max(1, int(frac * n))
        a = np.ones(n)
        a[:k] += 8.0
        profiles.append(a / a.sum())
        a = np.ones(n)
        a[-k:] += 8.0
        profiles.append(a / a.sum())

    # Two-hump asymmetric
    for center in (0.2, 0.3, 0.7, 0.8):
        a = 1.0 + 4.0 * np.exp(-((t - center) / 0.1) ** 2) + 2.0 * np.exp(-((t - (1 - center)) / 0.15) ** 2)
        profiles.append(a / a.sum())

    return profiles


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent if available
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = float(pv.sum())
            if s > 1e-30:
                parent = pv / s
                parent_score = compute_score(parent)
                best_seq = parent.copy()
                best_score = parent_score
                print(f"Parent score: {parent_score:.10f}, n={parent.size}")
    except (NameError, Exception):
        parent = None

    # Phase 1: Gradient optimization at n=512 from diverse starts
    n1 = 512
    phase1_end = min(deadline, t_start + 500.0)

    all_starts = []
    if parent is not None:
        if parent.size == n1:
            all_starts.append(parent.copy())
        else:
            p_int = interpolate_profile(parent, n1)
            if p_int is not None:
                all_starts.append(p_int)

    profiles_512 = make_profiles(n1, rng)
    all_starts.extend(profiles_512[:20])

    candidates_512 = []
    per_start = max(3.0, (phase1_end - t_start) / max(1, len(all_starts)))

    for i, p0 in enumerate(all_starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 0.0)
        s = float(p0.sum())
        if s < 1e-30:
            continue
        p0 = p0 / s

        per = min(phase1_end - 1.0, time.monotonic() + per_start)
        if per < 2.0:
            break
        p_best, M_best = gradient_optimize(p0, per, rng, lr=0.06)
        sc = 2.0 * n1 * M_best
        if np.isfinite(sc) and sc < best_score:
            best_seq = p_best.copy()
            best_score = sc
        candidates_512.append((sc, p_best))
        if i % 5 == 0:
            print(f"  Phase 1 start {i}/{len(all_starts)}: best={best_score:.10f}")

    candidates_512 = [c for c in candidates_512 if np.isfinite(c[0])]
    candidates_512.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: Extended gradient optimization on top candidates
    phase2_end = min(deadline, t_start + 950.0)
    top_k = min(4, len(candidates_512))
    per_cand = max(30.0, (phase2_end - t_start) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates_512[i]
        sc0 = 2.0 * n1 * (float(fft_convolve(p0 / p0.sum()).max()))
        per = min(phase2_end - 2.0, time.monotonic() + per_cand)
        if per < 5.0:
            break
        # Use smaller beta for finer optimization
        beta_init = 0.03 * (2.0 * n1 * float(fft_convolve(p0 / p0.sum()).max()) / n1)
        p_best, M_best = gradient_optimize(p0, per, rng, lr=0.03, beta_init=beta_init)
        sc = 2.0 * n1 * M_best
        if np.isfinite(sc) and sc < best_score:
            best_seq = p_best.copy()
            best_score = sc
        print(f"  Phase 2 cand {i}: score={sc:.10f} (was {sc0:.10f})")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Coordinate descent on best
    phase3_end = min(deadline, t_start + 1150.0)
    if best_seq is not None and best_seq.size == n1:
        per = min(phase3_end - 2.0, time.monotonic() + 150.0)
        if per > 10.0:
            p_best, M_best = coordinate_descent(best_seq, per, rng)
            sc = 2.0 * n1 * M_best
            if np.isfinite(sc) and sc < best_score:
                best_seq = p_best.copy()
                best_score = sc
            print(f"After Phase 3 (coord descent): best={best_score:.10f}")

    # Phase 4: Try other lengths
    phase4_end = min(deadline, t_start + 1550.0)
    for n2 in (1024, 2048, 256):
        if time.monotonic() > phase4_end - 60:
            break
        if best_seq is not None:
            p_ext = interpolate_profile(best_seq, n2)
            if p_ext is not None:
                per = min(phase4_end - 30.0, time.monotonic() + 100.0)
                if per > 10.0:
                    p_best, M_best = gradient_optimize(p_ext, per, rng, lr=0.05)
                    sc = 2.0 * n2 * M_best
                    if np.isfinite(sc) and sc < best_score:
                        best_seq = p_best.copy()
                        best_score = sc
                    print(f"  Phase 4 (n={n2}): score={sc:.10f}")

    # Phase 5: Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        c_best = fft_convolve(best_seq / best_seq.sum())
        M_best_cur = float(c_best.max())
        beta_final = 0.02 * M_best_cur

        per = min(deadline - 10.0, time.monotonic() + 100.0)
        if per > 10.0:
            p_best, M_best = gradient_optimize(best_seq, per, rng, lr=0.015, beta_init=beta_final)
            sc = compute_score(p_best)
            if np.isfinite(sc) and sc < best_score:
                best_seq = p_best.copy()
                best_score = sc

        if time.monotonic() < deadline - 15:
            per2 = min(deadline - 5.0, time.monotonic() + 60.0)
            if per2 > 10.0:
                p_best2, M_best2 = coordinate_descent(best_seq, per2, rng)
                sc2 = compute_score(p_best2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_seq = p_best2.copy()
                    best_score = sc2

        print(f"After Phase 5: best={best_score:.10f}, n={best_seq.size}")

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0).astype(np.float64)
        s = float(final.sum())
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        if final.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    t = np.linspace(0, 1, 512)
    fb = (1.0 - t) ** 2 + 0.01
    fb = fb / fb.sum()
    return [float(x) for x in fb]
