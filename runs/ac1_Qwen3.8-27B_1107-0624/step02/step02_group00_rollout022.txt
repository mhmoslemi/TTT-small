    #      = sum_m w[j+m] * a[m] (m in [0,n-1], j+m in [0, 2n-2])
    
    # This is the cross-correlation between w (length 2n-1) and a (length n).
    # In FFT: pad both to N, compute IFFT(FFT(w) * conj(FFT(a_pad)))
    
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:2*n-1] = w  # w has length 2n-1
    
    fw_full = np.fft.rfft(w_pad, N)
    fa_full = np.fft.rfft(a_pad, N)
    # Cross-correlation: IFFT(conj(FFT(w)) * FFT(a)) gives sum_k w[k] * a[k+m] ... 
    # Let me use a different approach.
    # h[j] = sum_m w[j+m] * a[m]
    # This is (w_rev * a)[j] where w_rev is w reversed and padded... 
    # Actually, it's simpler to just do:
    # h = np.correlate(w, a, mode='valid') ... but that gives sum_i w[i] * a[i + (n-1-j)] ...
    # Let me just use the FFT trick:
    # h[j] = sum_m w[j+m] * a[m] = (a * w)[j] in the convolution sense if we reverse w
    # Specifically: h = convolve(a, w)[n-1:] ... let me verify.
    # convolve(a, w)[k] = sum_i a[i] * w[k-i]
    # For k = j + n - 1: sum_i a[i] * w[j+n-1-i]
    # Let m = n-1-i, i.e., i = n-1-m:
    # = sum_m a[n-1-m] * w[j+m]
    # That's not quite what I want. I want sum_m a[m] * w[j+m].
    
    # OK let me just be direct. h[j] = sum_{m=0}^{n-1} a[m] * w[j+m] for j in [0, n-1]
    # (where w has length 2n-1, so j+m goes from j to j+n-1, max is 2n-2 which is valid)
    
    # This is a valid-mode cross-correlation. Let's compute via FFT:
    # Define b[k] = w[k] for k in [0, 2n-2], 0 otherwise (already the case)
    # h[j] = sum_m a[m] * b[j+m]
    # This is the cross-correlation R_ab at lag... hmm.
    # R_ab[lag] = sum_k a[k] * b[k+lag]
    # So h[j] = R_ab[j] for j in [0, n-1]
    
    # To compute cross-correlation via FFT:
    # R_ab[lag] = IFFT{ conj(FFT(a)) * FFT(b) }[lag] ... in circular sense with proper padding
    # With zero-padding to N >= 2n-1 + n - 1 = 3n-2, we get the linear cross-correlation.
    
    # But we only need lags 0 to n-1, and N = next_pow2(2n-1) >= 2n-1.
    # For the circular convolution of length N:
    # IFFT(FFT(a_pad) * conj(FFT(b_pad))) gives circular cross-correlation
    # If N >= len(a) + len(b) - 1 = n + (2n-1) - 1 = 3n-2, we get the linear one.
    # N = next_pow2(2n-1) might not be >= 3n-2!
    
    # For safety, let's use a larger N:
    N2 = _next_pow2(3 * n)
    a2 = np.zeros(N2)
    a2[:n] = a
    w2 = np.zeros(N2)
    w2[:2*n-1] = w
    fa2 = np.fft.rfft(a2, N2)
    fw2 = np.fft.rfft(w2, N2)
    # Cross-correlation: sum_k a[k] * conj(w[k+lag]) ... 
    # Actually, R_ab[l] = sum_k a[k] * conj(b[k-l]) for real signals = sum_k a[k] * b[k-l]
    # Hmm, I keep getting confused with conventions.
    
    # Let me just use the simplest approach: 
    # h[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    # = sum_{m=0}^{n-1} a[m] * w[m+j]
    # This is the cross-correlation of a and w at lag j (in the convention where 
    # we shift w forward by j and take the inner product with a).
    
    # In numpy: np.correlate(w, a, mode='full') gives corr[k] = sum_i w[i+k] * conj(a[i])
    # with appropriate index ranges. For real signals: corr[k] = sum_i w[i+k] * a[i]
    # The valid range for our h[j] = sum_m a[m] * w[m+j] corresponds to k=j in np.correlate(w, a)
    
    # But for large n, doing this directly is O(n^2). Let's use FFT.
    # 
    # np.correlate(w, a, 'full')[j] for j in [0, n-1]:
    # corr_full[k] = sum_i w[i+k] * a[i] for valid i where 0<=i<n and 0<=i+k<2n-1
    # For k=j (0<=j<n-1): sum_m a[m] * w[m+j] (where m+j < 2n-1 is always true for m<n, j<n)
    # So h[j] = np.correlate(w, a, 'full')[j]
    
    # FFT approach for cross-correlation:
    # corr = IFFT{ conj(FFT(w_padded)) * FFT(a_padded) } ... 
    # Actually for cross-correlation of real signals:
    # R[k] = sum_n x[n+k] * y[n] = IFFT{ X * conj(Y) }[k] (with proper zero-padding)
    # Wait: if X = FFT(x), Y = FFT(y), then IFFT(X * conj(Y))[k] = sum_n x[n] * y[n-k]
    # (circular cross-correlation with the "other" convention)
    
    # Ugh, let me just use the property that for real signals:
    # sum_n x[n+k] * y[n] = (x * y_flipped)[k + len(y) - 1] (linear convolution)
    # where y_flipped = y[::-1]
    
    # So h[j] = sum_m a[m] * w[m+j] = convolve(w, a[::-1])[j] ... let me verify.
    # convolve(w, a_rev)[k] = sum_i w[i] * a_rev[k-i] = sum_i w[i] * a[n-1-(k-i)]
    # For this to equal sum_m a[m] * w[m+j]:
    # Let m = n-1-(k-i), so i = k-n+1+m
    # sum_m a[m] * w[k-n+1+m]
    # For k-n+1 = j: k = j+n-1
    # So h[j] = convolve(w, a[::-1])[j+n-1]
    
    a_rev = a[::-1]
    # convolve(w, a_rev) has length (2n-1) + n - 1 = 3n-2
    # We want indices j+n-1 for j in [0, n-1], so indices [n-1, 2n-2]
    N3 = _next_pow2(3 * n - 1)
    fw3 = np.fft.rfft(w, N3)
    fa_rev3 = np.fft.rfft(a_rev, N3)
    conv_w_a_rev = np.fft.irfft(fw3 * fa_rev3, N3)
    h = conv_w_a_rev[n-1:2*n-1]  # h[j] = conv_w_a_rev[j+n-1] for j in [0,n-1]
    
    grad = 2.0 * h
    return grad, float(c.max()), c

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _smooth_max_adam(a, deadline, max_iter=50000, lr=0.03, beta_init_factor=0.05):
    """Smooth max Adam optimizer with backtracking line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()
    
    beta = beta_init_factor * float(c.max())
    beta = max(beta, 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute smooth max gradient in z-space
        p = a
        c_curr = _fft_convolve(p)
        _, w = _smooth_max_and_weights(c_curr, beta)
        
        # Gradient of smooth_max w.r.t. a
        n_local = len(p)
        N = _next_pow2(3 * n_local)
        w_arr = np.zeros(N)
        w_arr[:2*n_local-1] = w
        p_rev = p[::-1]
        p_pad = np.zeros(N)
        p_pad[:n_local] = p_rev
        fw = np.fft.rfft(w_arr, N)
        fp = np.fft.rfft(p_pad, N)
        conv_result = np.fft.irfft(fw * fp, N)
        h = conv_result[n_local-1:2*n_local-1]
        grad_a = 2.0 * h
        
        # Gradient in z-space: grad_z[j] = p[j] * (grad_a[j] - sum(p * grad_a))
        dot = float(np.dot(p, grad_a))
        grad_z = p * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))
        
        # Adam update
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        
        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        
        e = np.exp(z_new)
        a_new = e / e.sum()
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.002, lr_cur * 0.95)
            if no_imp > max(3000, 5 * n):
                # Restart with perturbation
                z = best_z + 0.3 * np.random.default_rng(it).normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_init_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue
        
        z = z_new
        a = a_new
        
        # Decay beta
        if it % 100 == 0:
            beta = max(1e-12, beta * 0.8)
        
        # Periodically print
        if it % 5000 == 0:
            print(f"    Adam iter {it}: score={best_score:.10f}, beta={beta:.2e}, lr={lr_cur:.4f}, no_imp={no_imp}")
    
    return best_a, best_score

def _subgradient_polish(a, deadline, max_iter=10000):
    """Subgradient method directly targeting the maximum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    # Use smooth max with very small beta for near-subgradient
    beta = 1e-6
    lr = 0.1 / (n ** 0.5)  # O(1/sqrt(t)) step size
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())
        max_idx = int(np.argmax(c_curr))
        
        # Subgradient: direction that increases c[max_idx]
        # We want to decrease it, so move opposite
        # dc[max_idx]/da[j] = 2 * a[max_idx - j] (if valid)
        # More precisely: dc[k]/da[j] = 2*a[k-j] if 0<=k-j<n
        # So the subgradient direction for a[j] is 2*a[max_idx-j] (if 0<=max_idx-j<n)
        
        # Compute subgradient properly
        sg = np.zeros(n)
        k = max_idx
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                sg[j] = 2.0 * a[idx]
        # Also need to account for the "mirror" contribution
        # c[k] = sum_i a[i]*a[k-i], dc[k]/da[j] = a[k-j] + a[j] ... no
        # dc[k]/da[j] = a[k-j]*1[0<=k-j<n] + a[j]*1[0<=k-j<n] ... 
        # Wait, I already worked this out: dc[k]/da[j] = 2*a[k-j]*1[0<=k-j<n]
        # But actually that's only correct if we're careful. Let me re-derive.
        # c[k] = sum_{i=0}^{n-1} a[i]*a[k-i] * 1[0<=k-i<n]
        # d(c[k])/d(a[j]) = a[k-j]*1[0<=k-j<n] + sum_{i: k-i=j} a[i] * 1[0<=i<n]
        #                   = a[k-j]*1[0<=k-j<n] + a[j]*1[0<=k-j<n] * 1[j != k-j]
        #                   ... this is getting confusing. Let me just use:
        # Since c = conv(a,a), and conv is bilinear:
        # dc[k]/da[j] = (a * delta_j)[k] + (delta_j * a)[k] where delta_j is unit vector
        # = a[k-j] + a[j-k] (with appropriate bounds)
        # For real sequences: = a[k-j]*1[0<=k-j<n] + a[j-k]*1[0<=j-k<n]
        # Note: a[j-k] with j-k in [0,n-1] means j>=k
        # And a[k-j] with k-j in [0,n-1] means j<=k
        
        # So dc[k]/da[j] = a[k-j]*1[j<=k, k-j<n] + a[j-k]*1[j>=k, j-k<n]
        
        sg = np.zeros(n)
        for j in range(n):
            val = 0.0
            if j <= k and k - j < n:
                val += a[k - j]
            if j >= k and j - k < n:
                val += a[j - k]
            sg[j] = val
        
        # Normalize and apply step
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break
        
        # Step in the direction that reduces the max
        # a_new = a - lr * sg / (sum(a) * sum(sg)) to maintain sum=1
        sum_sg = float(sg.sum())
        if abs(sum_sg) < 1e-20:
            a_new = a - lr * sg / (n * sg_norm)
        else:
            a_new = a - lr * sg / (n * sg_norm)
        
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
        
        # Decay step size
        lr = lr * 0.999
        
    return best_a, best_score

def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_cosine_profile(n, alpha=1.0):
    """Raised cosine type profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_bell_profile(n, width=1.0):
    """Gaussian-like bell profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _subgradient_inline(a, max_idx, n):
    """Compute subgradient of c[max_idx] w.r.t. a inline (for speed)."""
    sg = np.zeros(n)
    k = max_idx
    # dc[k]/da[j] = a[k-j]*1[0<=k-j<n] + a[j-k]*1[0<=j-k<n]
    # First term: j from max(0, k-n+1) to min(k, n-1), value a[k-j]
    # Second term: j from max(k, 0) to min(k+n-1, n-1), value a[j-k]
    # But since k in [0, 2n-2] and j in [0, n-1]:
    # First term: k-j in [0, n-1] means j in [k-n+1, k] intersect [0, n-1]
    # Second term: j-k in [0, n-1] means j in [k, k+n-1] intersect [0, n-1]
    
    # First term
    j_start1 = max(0, k - n + 1)
    j_end1 = min(k, n - 1) + 1
    if j_start1 < j_end1:
        idx = np.arange(j_start1, j_end1)
        sg[idx] += a[k - idx]
    
    # Second term
    j_start2 = max(k, 0)
    j_end2 = min(k + n - 1, n - 1) + 1
    if j_start2 < j_end2:
        idx = np.arange(j_start2, j_end2)
        sg[idx] += a[idx - k]
    
    return sg

def _fast_subgradient(a, deadline, max_iter=20000):
    """Fast subgradient with vectorized operations."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    lr = 0.5 / np.sqrt(n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())
        max_idx = int(np.argmax(c_curr))
        
        sg = _subgradient_inline(a, max_idx, n)
        
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break
        
        step = lr / (it + 1) ** 0.5
        a_new = a - step * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        
        if it % 5000 == 0:
            print(f"    Subgrad iter {it}: score={best_score:.10f}, lr*{step:.6f}")
    
    return best_a, best_score

def _lp_spread(a, M):
    """LP: maximize sum(b) s.t. a*b conv <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)
        
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 100:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
            if no_imp > 200:
                break
        
        a = a_new
    
    return best_a

def _multi_peak_gradient(a, beta, top_k=5):
    """Gradient that weights multiple peaks, not just the max."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _grad_smooth_max(a, w):
    """Gradient of sum_k w[k]*c[k] where c=conv(a,a). Vectorized via FFT."""
    n = len(a)
    N = _next_pow2(3 * n)

def _subgradient_at_max(a, k):
    """Compute dc[k]/da vectorized."""
    n = len(a)
    sg = np.zeros(n)
    # First term: a[k-j] for j in [max(0,k-n+1), min(k,n-1)]
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    # Second term: a[j-k] for j in [max(k,0), min(k+n-1, n-1)]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)

def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    best_a = a0.copy()
    best_sc = sc0

    # Phase 1: Adam smooth max (coarse to fine)
    if time.monotonic() < deadline - 60:
        d1 = min(deadline - 60, time.monotonic() + 250)
        a1, s1 = _adam_smooth_max(a0, d1, max_iter=25000, lr=0.03, beta_factor=0.05)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam1: {best_sc:.10f}")

    # Phase 2: Multi-subgradient
    if time.monotonic() < deadline - 40:
        d2 = min(deadline - 40, time.monotonic() + 100)
        a2, s2 = _multi_subgradient(best_a, d2, max_iter=8000, top_k=3)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} MultiSG: {best_sc:.10f}")

    # Phase 3: LP refinement
    if time.monotonic() < deadline - 20:
        d3 = min(deadline - 20, time.monotonic() + 150)
        a3 = _lp_refine(best_a, d3, max_iter=2500, inflate=1.001, blend=0.3)
        s3 = _score(a3)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} LP: {best_sc:.10f}")

    # Phase 4: Fine Adam
    if time.monotonic() < deadline - 10:
        d4 = deadline - 10
        a4, s4 = _adam_smooth_max(best_a, d4, max_iter=20000, lr=0.01, beta_factor=0.02)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} Adam2: {best_sc:.10f}")

    # Phase 5: Final subgradient
    if time.monotonic() < deadline - 5:
        d5 = deadline - 5
        a5, s5 = _subgradient_opt(best_a, d5, max_iter=5000)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  {label} FinalSG: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining Adam smooth-max, multi-peak subgradient,
    and LP refinement. Explores diverse initializations at multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is not None:
        p512 = _interpolate_profile(parent, 512)
        p1024 = _interpolate_profile(parent, 1024)
    else:
        p512 = _make_linear_profile(512, 1.5)
        p1024 = _make_linear_profile(1024, 1.5)

    # === Strategy 1: Full pipeline on parent@512 (highest priority) ===
    print("=== Strategy 1: Parent @ n=512 ===")
    d1 = min(deadline - 500, t_start + 500)
    a_best, s_best = _full_pipeline(p512, d1, "S1")
    if s_best < best_score:
        best_score = s_best
        best_seq = a_best.copy()

    # === Strategy 2: Full pipeline on parent@1024 ===
    if time.monotonic() < deadline - 400:
        print("=== Strategy 2: Parent @ n=1024 ===")
        d2 = min(deadline - 400, time.monotonic() + 500)
        a_best2, s_best2 = _full_pipeline(p1024, d2, "S2")
        if s_best2 < best_score:
            best_score = s_best2
            best_seq = a_best2.copy()

    # === Strategy 3: Diverse initializations at n=512 ===
    if time.monotonic() < deadline - 200:
        print("=== Strategy 3: Diverse inits @ n=512 ===")
        n = 512
        candidates = [
            _make_linear_profile(n, 1.0),
            _make_linear_profile(n, 2.0),
            _make_cubic_profile(n, 2.0, -2.0),
            _make_cubic_profile(n, 1.2, -0.5),
            _make_cosine_profile(n, 0.5),
            _make_cosine_profile(n, 1.5),
            _make_bell_profile(n, 0.35),
            _make_bell_profile(n, 0.6),
        ]
        # Perturbed parent
        p_pert = p512 * (1.0 + 0.15 * rng.normal(size=n))
        p_pert = np.maximum(p_pert, 0.0)
        candidates.append(p_pert / p_pert.sum())

        # Interpolated parent variants
        p_interp = _interpolate_profile(p512, 400)
        p_interp_back = _interpolate_profile(p_interp, n)
        candidates.append(p_interp_back)

        for i, a0 in enumerate(candidates):
            if time.monotonic() > deadline - 200:
                break
            per_budget = min(40.0, (deadline - 200 - time.monotonic()) / max(1, len(candidates) - i))
            if per_budget < 10:
                break
            d3i = min(deadline - 200 + i, time.monotonic() + per_budget)
            # Quick pipeline: just Adam + subgradient
            a_q, s_q = _adam_smooth_max(a0, d3i, max_iter=5000, lr=0.02, beta_factor=0.05)
            if s_q < best_score:
                best_score = s_q
                best_seq = a_q.copy()
            print(f"  S3-{i}: {s_q:.10f}")

    # === Strategy 4: Focused final optimization on best ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"=== Strategy 4: Final focus (current best={best_score:.10f}) ===")
        # Try both 512 and 1024 versions of the best
        best_n = best_seq.size
        if best_n != 512:
            best_512 = _interpolate_profile(best_seq, 512)
        else:
            best_512 = best_seq.copy()

        # Fine-tune at 512
        d4a = min(deadline - 10, time.monotonic() + 150)
        a_f1, s_f1 = _adam_smooth_max(best_512, d4a, max_iter=15000, lr=0.008, beta_factor=0.01)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  Final Adam 512: {best_score:.10f}")

        # Subgradient on result
        if time.monotonic() < deadline - 8:
            d4b = min(deadline - 8, time.monotonic() + 80)
            a_f2, s_f2 = _multi_subgradient(best_seq, d4b, max_iter=5000, top_k=3)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  Final MultiSG: {best_score:.10f}")

        # LP polish
        if time.monotonic() < deadline - 5:
            d4c = min(deadline - 5, time.monotonic() + 60)
            a_f3 = _lp_refine(best_seq, d4c, max_iter=1500, inflate=1.0005, blend=0.2)
            s_f3 = _score(a_f3)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"  Final LP: {best_score:.10f}")

        # One more tiny Adam
        if time.monotonic() < deadline - 3:
            a_f4, s_f4 = _adam_smooth_max(best_seq, deadline - 3, max_iter=5000, lr=0.003, beta_factor=0.005)
            if s_f4 < best_score:
                best_score = s_f4
                best_seq = a_f4.copy()
            print(f"  Ultra-final Adam: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
