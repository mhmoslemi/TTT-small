Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

from scipy.optimize import linprog


def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < -1e-10:
        c = c - mn
    return c


def _score_normalized(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    b = _fft_convolve(a)
    if b.size == 0:
        return np.inf
    return float(2.0 * n * b.max() / (s * s))


def _score_unit(a):
    """Score assuming sum(a) is already 1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    b = _fft_convolve(a)
    if b.size == 0:
        return np.inf
    return float(2.0 * n * b.max() / (s * s))


def _cross_corr_lags(w, a):
    """Compute cross-correlation cross[m] = sum_j w[j+m]*a[j] for m=0..n-1."""
    n = len(a)
    nw = len(w)
    N = 1
    while N < nw + n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    cross = np.fft.irfft(fw * np.conj(fa), N)
    return cross[:n]


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64)
    z = np.clip(z, -50.0, 50.0)
    z = z - z.max()
    e = np.exp(z)
    p = e + floor
    s = p.sum()
    if s < 1e-30 or not np.isfinite(s):
        return np.ones_like(e) / len(e)
    return p / s


def _adam_optimize(a_init, deadline, rng, lr=0.06, restart_noise=0.15):
    """Adam optimization in logit space with soft-max gradient."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_score = _score_unit(a)
    no_imp = 0

        b = _fft_convolve(a)

        if ws < 1e-30:

def _lp_optimize(a_init, deadline, rng, max_iters=500):
    """LP-based direction finding with line search."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_unit(a)
    b = _fft_convolve(a)
    M = float(b.max())
    
    it = 0
    stalled = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        a = a / a.sum()
        b = _fft_convolve(a)
        M = float(b.max())
        
        # Build LP constraint matrix: (a*g)[k] = sum_i a[k-i]*g[i] <= M
        A_mat = np.zeros((2 * n - 1, n))
        for i in range(n):
            start = i
            end = min(i + n, 2 * n - 1)
            A_mat[start:end, i] = a[0:end - start]
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                    'time_limit': 10.0})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        g = res.x
        sg = float(g.sum())
        if sg < 1e-10:
            g = a.copy()
            sg = 1.0
        g = g / sg
        
        # Line search: try several step sizes
        best_t = 0.0
        best_sc = _score_unit(a)
        
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            st = float(a_try.sum())
            if st < 1e-10:
                continue
            a_try = a_try / st
            sc_try = _score_unit(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t
        
        if best_t > 0:

            a = np.maximum(a, 0.0)
            a = a / a.sum()
            
            sc = _score_unit(a)

def _coord_descent(a_init, deadline, rng):
    """Coordinate descent: optimize one element at a time."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64).copy()

    best_a = a.copy()

    idx = 0

        idx += 1
        
        s_other = float(a.sum()) - a[i]
        if s_other < 1e-15:
            continue
        
        # b_k = a[i] * a[k-i] + a[k-i] * a[i] + (other terms not involving a[i])
        # = 2*a[i]*a[k-i] + rest_k  (for valid k-i)
        # We want to minimize max_k (2*a[i]*a[k-i] + rest_k) / (s_other + a[i])^2
        # Let's just try a few values of a[i]
        
        current_val = a[i]

            current_val * 0.7,
            current_val * 0.85,
            current_val * 1.15,
            current_val * 1.3,
            current_val * 2.0,
            current_val + 0.01,
            max(0.0, current_val - 0.01),
            0.0,
            s_other / n,  # uniform
        ]
        
        best_local = current_val
        best_local_sc = best_score
        
        for val in candidates:
            val = max(0.0, val)

            s_new = s_other + val

def _make_profile(n, kind='smooth'):
    """Create initial profile."""
    t = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'linear':
        a = 1.0 + 1.5 * (t - 0.5)
    elif kind == 'quadratic':
        u = 2.0 * t - 1.0
        a = 1.0 + 0.5 * u * u
    elif kind == 'cubic':
        u = 2.0 * t - 1.0
        a = 1.0 + 0.8 * u - 0.3 * u ** 3
    elif kind == 'smooth':
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_to_length(a, n_target):
    """Interpolate sequence to new length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n_src)
    x1 = np.linspace(0.0, 1.0, n_target)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new
