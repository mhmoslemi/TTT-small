For real non-negative sequences, the best we can do is related to the "prolate spheroidal wave functions" or similar orthogonal sequences.

One key insight: if we think of `a` as a probability distribution, then `conv(a,a)` is the distribution of the sum of two independent random variables with distribution `a`. We want this sum distribution to be as flat as possible. The flat distribution on `[0, 2n-1]` would have each value equal to `1/(2n-1)`, but that's not achievable with a convolution of two identical non-negative sequences (unless n=1).

The best achievable flatness depends on `n`. For larger `n`, we can get closer to flat.

Another insight: the problem has a connection to the "self-convolution flatness" problem in harmonic analysis. The optimal sequences are related to "Sidon sets" and "Bose-Chowla sequences" in number theory.

For practical purposes, I'll focus on numerical optimization.

"""
Multi-resolution optimization combining LP direction-finding and FFT-accelerated
gradient descent (Adam with Nesterov momentum) for minimizing the peak autocorrelation
ratio 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=10.0):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterate(f, max_iters=500, t_init=0.4, t_min=0.01, deadline=None, verbose=False):
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M
    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        g_self_score = 2.0 * n * float(np.convolve(g, g).max())

        if g_self_score < best_score * 0.998:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
        else:
            f_new = (1.0 - t_step) * f + t_step * g
            f_new = normalize(f_new)
            if f_new is None:
                break
            c_new = np.convolve(f_new, f_new)
            M_new = float(c_new.max())
            score_new = 2.0 * n * M_new

            if score_new < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = score_new
                best_M = M_new
                f = f_new.copy()
                M = M_new
                no_improve = 0
                t_step = max(t_min, t_step * 0.99)
            else:
                no_improve += 1
                f = best_f.copy()
                M = best_M
                if no_improve % 10 == 0:
                    t_step = max(t_min, t_step * 0.8)
                if no_improve > 100:
                    break

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={best_M:.8f}, t={t_step:.4f}")

    return best_f, best_score


def interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_poly_profile(n, coeffs):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.ones(n)
    for k, c in enumerate(coeffs, start=1):
        a += c * (u ** k)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def compute_gradient_fft(p, w, beta):
    """
    Compute gradient of soft-max objective in logit space.
    p: probability vector (sum=1, non-neg)
    w: soft-argmax weights over convolution values
    Returns: gradient in logit space (n,)
    """
    n = len(p)
    # conv(p,p) gradient: grad_p[j] = 2 * sum_k w[k] * p[k-j] (with p[x]=0 outside [0,n))
    # This is 2 * correlate(w, p)
    size = 2 * n - 1
    size_fft = 1
    while size_fft < size:
        size_fft *= 2
    
    W = np.fft.rfft(w, size_fft)
    P = np.fft.rfft(p, size_fft)
    # correlate(w, p)[j] = sum_k w[k] * p[k-j]
    # = IFFT(FFT(w) * conj(FFT(p)))
    corr = np.fft.irfft(W * np.conj(P), size_fft)[:n]
    
    grad_p = 2.0 * corr
    # Logit space gradient: gz[j] = p[j] * (grad_p[j] - sum(p * grad_p))
    dot = float(np.dot(p, grad_p))
    gz = p * (grad_p - dot)
    return gz


def adam_polish(p0, deadline, lr=0.05, reg=1e-8, verbose=True, beta_init=None, nesterov=True):
    """
    Adam (optionally with Nesterov) gradient descent in logit space.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    v_nest = np.zeros(n)  # For Nesterov
    it = 0
    t_start = time.monotonic()

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr

    if beta_init is None:
        beta_init = 0.2 * max_c
    beta = beta_init

    # FFT size
    size = 2 * n - 1
    size_fft = 1
    while size_fft < size:
        size_fft *= 2

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Soft argmax weights
        x = (c - max_c) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Compute gradient using FFT
        gz = compute_gradient_fft(p0, w, beta)
        gz += reg * p0

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()
            continue

        # Adam update
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        # Nesterov lookahead: compute gradient at z - lr * mh / (sqrt(vh) + eps)
        if nesterov:
            z_new = z - lr * mh / (np.sqrt(vh) + eps)
            z_new = np.clip(z_new, -50.0, 50.0)
            z_new -= np.mean(z_new)
            p_new = np.exp(z_new - z_new.max())
            p_sum = p_new.sum()
            if p_sum > 1e-30:
                p_new = p_new / p_sum
                # Recompute gradient at lookahead point
                c_new = np.convolve(p_new, p_new)
                max_c_new = float(c_new.max())
                x_new = (c_new - max_c_new) / beta
                x_new = np.clip(x_new, -50, 50)
                ex_new = np.exp(x_new)
                s_new = ex_new.sum()
                if s_new > 1e-30:
                    w_new = ex_new / s_new
                    gz_new = compute_gradient_fft(p_new, w_new, beta)
                    gz_new += reg * p_new
                    if np.all(np.isfinite(gz_new)):
                        m = b1 * m + (1 - b1) * gz_new
                        v = b2 * v + (1 - b2) * gz_new * gz_new
                        mh = m / (1.0 - b1 ** it)
                        vh = v / (1.0 - b2 ** it)
                        z = z - lr * mh / (np.sqrt(vh) + eps)
                        z = np.clip(z, -50.0, 50.0)
                        z -= np.mean(z)
                        p0 = np.exp(z - z.max())
                        p0 = p0 / p0.sum()
        else:
            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -50.0, 50.0)
            z -= np.mean(z)
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()

        c_new = np.convolve(p0, p0)
        score = 2.0 * n * float(c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p0.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.01, lr * 0.95)

        # Adapt beta: decrease over time for sharper max approximation
        if it % 100 == 0:
            beta = max(1e-8, beta * 0.7)

        if no_imp > 1000:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-6, 0.15 * float(np.convolve(
                np.exp(z - z.max()), np.exp(z - z.max())).max()))
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Adam iter {it}: score={best_score:.8f}, beta={beta:.2e}, "
                  f"lr={lr:.4f}, elapsed={elapsed:.1f}s")

    return best_p, best_score

"""
Multi-resolution optimization combining LP direction-finding and FFT-accelerated
Adam gradient descent in logit space for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Progresses through n=512 -> 1024 -> 2048.

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=15.0):
    """Solve LP: maximize sum(g) s.t. g>=0, conv(f,g)<=M. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterate(f, max_iters=500, t_init=0.4, t_min=0.01, deadline=None, verbose=False):
    """LP-based iterations to reduce max(conv(f,f)). f normalized to sum 1."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M
    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        g_self_score = 2.0 * n * float(np.convolve(g, g).max())

        if g_self_score < best_score * 0.998:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
        else:
            f_new = (1.0 - t_step) * f + t_step * g
            f_new = normalize(f_new)
            if f_new is None:
                break
            c_new = np.convolve(f_new, f_new)
            M_new = float(c_new.max())
            score_new = 2.0 * n * M_new

            if score_new < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = score_new
                best_M = M_new
                f = f_new.copy()
                M = M_new
                no_improve = 0
                t_step = max(t_min, t_step * 0.99)
            else:
                no_improve += 1
                f = best_f.copy()
                M = best_M
                if no_improve % 10 == 0:
                    t_step = max(t_min, t_step * 0.8)
                if no_improve > 150:
                    break

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={best_M:.8f}, t={t_step:.4f}")

    return best_f, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum=1."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_poly_profile(n, coeffs):
    """Create profile from polynomial coefficients: a(u) = 1 + sum coeffs[k]*u^(k+1)."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.ones(n)
    for k, c in enumerate(coeffs, start=1):
        a += c * (u ** k)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_quadratic_profile(n, alpha, beta):
    """a(u) = 1 + alpha*u + beta*u^2, u in [-1,1], normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * u + beta * u * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def compute_gradient_fft(p, w):
    """
    Compute gradient of soft-max(conv(p,p)) w.r.t. p using FFT.
    p: probability vector (length n, sum=1, non-neg)
    w: soft-argmax weights (length 2n-1)
    Returns: grad_p (length n) where grad_p[j] = 2 * sum_k w[k]*p[k-j]
    """
    n = len(p)
    size = 2 * n - 1
    size_fft = 1
    while size_fft < size:
        size_fft *= 2

    W = np.fft.rfft(w, size_fft)
    P = np.fft.rfft(p, size_fft)
    # correlate(w, p)[j] = sum_k w[k] * p[k-j] = IFFT(FFT(w)*conj(FFT(p)))[:n]
    corr = np.fft.irfft(W * np.conj(P), size_fft)[:n]
    return 2.0 * corr


def adam_polish(p0, deadline, lr=0.05, reg=1e-8, verbose=True, beta_init=None):
    """
    Adam gradient descent in logit space to polish a non-negative sequence.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12).copy()
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    t_start = time.monotonic()

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr

    if beta_init is None:
        beta_init = 0.15 * max_c
    beta = beta_init

    # FFT size for gradient computation
    size = 2 * n - 1
    size_fft = 1
    while size_fft < size:
        size_fft *= 2

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Soft argmax weights
        x = (c - max_c) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Compute gradient using FFT
        grad_p = compute_gradient_fft(p0, w)
        grad_p += reg * p0

        # Logit space gradient: gz[j] = p[j] * (grad_p[j] - sum(p * grad_p))
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()
            continue

        # Adam update
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p0 = np.exp(z - z.max())
        p_sum = p0.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()
            continue
        p0 = p0 / p_sum

        c_new = np.convolve(p0, p0)
        score = 2.0 * n * float(c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p0.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.01, lr * 0.95)

        # Adapt beta: decrease for sharper max approximation
        if it % 80 == 0:
            beta = max(1e-10, beta * 0.65)

        if no_imp > 1500:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            z = 0.6 * best_z + 0.8 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-8, 0.15 * float(np.convolve(
                np.exp(z - z.max()), np.exp(z - z.max())).max()))
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Adam iter {it}: score={best_score:.8f}, beta={beta:.2e}, "
                  f"lr={lr:.4f}, elapsed={elapsed:.1f}s")

    return best_p, best_score

            print(f"\nPhase 3: Adam n={n3}, init score={sc3:.8f}")
            p_best, sc_best = adam_polish(a3, deadline_p3, lr=0.04, reg=1e-8,
                                           verbose=True, beta_init=None)
            if sc_best < best_score:
                best_score = sc_best
                best_seq = p_best
                print(f"  Phase 3: score={best_score:.8f}")

    # === Phase 4: Upsample to n=2048 and Adam polish ===
    n4 = 2048
    if time.monotonic() < deadline - 60:
        a4 = interpolate_seq(best_seq, n4)
        if a4 is not None:
            sc4 = internal_score(a4)
            print(f"\nPhase 4: Adam n={n4}, init score={sc4:.8f}")
            p_best, sc_best = adam_polish(a4, deadline, lr=0.03, reg=1e-8,
                                           verbose=True, beta_init=None)
            if sc_best < best_score:
                best_score = sc_best
                best_seq = p_best
                print(f"  Phase 4: score={best_score:.8f}")

    # === Phase 5: Nesterov polish on best found ===
    if time.monotonic() < deadline and best_seq is not None:
        n5 = len(best_seq)
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\nPhase 5: Nesterov n={n5}, init score={internal_score(best_seq):.8f}")
            p_best, sc_best = adam_polish_nesterov(best_seq, deadline, lr=0.02, reg=1e-8,
                                                    verbose=True, beta_init=None)
            if sc_best < best_score:
                best_score = sc_best
                best_seq = p_best
                print(f"  Phase 5: score={best_score:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, internal_score={final_score:.8f}")

    try:
        eval_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"evaluate_sequence: {eval_score:.8f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(512, 2.0)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
