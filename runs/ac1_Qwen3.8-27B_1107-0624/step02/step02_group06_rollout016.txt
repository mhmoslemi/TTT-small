import numpy as np
import time
import scipy.optimize

def hankel_matrix(p):
    """Construct the (2n-1) x n matrix A where A[k,j] = p[k-j] if valid, else 0."""
    n = len(p)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = p[idx]
    return A

"""
LP-based active set optimization with gradient polish for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses a Frank-Wolfe-like approach where each step solves an LP to find
the best compatible direction, combined with softmax-Adam gradient descent
and coordinate descent for final polishing.
"""
import numpy as np
import time
import scipy.optimize as sco


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-9:
        c = np.maximum(c, 0.0)
    elif mn < 0:
        c = c - mn
    return c


def score_normalized(p):
    n = p.size
    c = full_convolve(p)
    return 2.0 * n * float(c.max())


def score_true(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def construct_A(p):
    n = len(p)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = p
    return A


def lp_solve(p, M):
    n = len(p)
    A = construct_A(p)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = sco.linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status not in (0, 2):
        return None
    x = np.asarray(res.x, dtype=np.float64)
    x = np.maximum(x, 0.0)
    return x


def lp_optimize(p0, deadline, verbose=True):
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-8)
    p = p / p.sum()
    n = p.size

    c = full_convolve(p)
    M = float(c.max())
    best_p = p.copy()
    best_score = 2.0 * n * M
    stall = 0
    t = 0.4

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        x = lp_solve(p, M)
        if x is None:
            M *= 1.01
            continue

        x_sum = x.sum()
        if x_sum < 1e-12:
            break
        x = x / x_sum

        t_fw = 2.0 / (it + 2.0)
        t_use = min(t, t_fw)

        p_new = (1.0 - t_use) * p + t_use * x
        p_new = np.maximum(p_new, 1e-10)
        p_new = p_new / p_new.sum()

        c_new = full_convolve(p_new)
        M_new = float(c_new.max())
        score_new = 2.0 * n * M_new

        if score_new < best_score:
            best_score = score_new
            best_p = p_new.copy()
            stall = 0
            t = min(0.5, t * 1.02)
        else:
            stall += 1
            t = max(0.05, t * 0.85)
            if stall > 200:
                p = best_p.copy()
                M = float(full_convolve(p).max())
                stall = 0
                t = 0.3

        p = p_new
        M = max(M_new, 1e-15)

        if verbose and it % 200 == 0:
            print(f"    LP iter {it}: best_score={best_score:.8f}, M={M:.8f}, t={t:.4f}")

    return best_p, best_score


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / e.size
    return e / s


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def gradient_adam(p0, deadline, lr=0.08, beta_frac=0.15):
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size

    z = np.log(p)
    z -= float(np.mean(z))

    b1, b2, eps = 0.90, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_frac * maxc)

    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 400 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        if adam_t % 100 == 0:
            beta = max(1e-14, beta * 0.65)

        if no_imp > max(500, 5 * n):
            z = 0.7 * best_z + 0.5 * np.random.default_rng(adam_t).normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def coordinate_polish(p0, deadline, step_frac=0.06):
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = score_normalized(p)
    no_imp = 0
    rng = np.random.default_rng(12345)

    factors = [1.0 + step_frac, 1.0 + 0.5 * step_frac, 1.0 + 2.0 * step_frac,
               1.0 - step_frac, 1.0 - 0.5 * step_frac, 1.0 - 0.3 * step_frac]

    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score
            for f in factors:
                nv = orig * f
                if nv < 1e-15 or nv > 1000.0:
                    continue
                p[i] = nv
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                p /= s
                sc = score_normalized(p)
                if sc < best_sc:
                    best_sc = sc
                    best_val = nv
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1

        if improved:
            step_frac = max(0.02, step_frac * 0.92)
        if no_imp > 3 * n:
            break

    return best_p, best_score


def make_barker_like(n, rng):
    """Create a sequence inspired by low-autocorrelation binary codes."""
    b = rng.choice([-1.0, 1.0], size=n)
    # Smooth it slightly
    kernel = np.array([0.1, 0.8, 0.1])
    b_smooth = np.convolve(b, kernel, mode='same')
    p = np.maximum(b_smooth + 0.5, 0.01)
    return p / p.sum()


def make_quadratic_phase(n, rng):
    """Quadratic phase mapped to non-negative."""
    t = np.linspace(0, 1, n)
    freq = rng.uniform(1.0, 4.0)
    phase = np.cos(2 * np.pi * freq * t * t)
    p = np.maximum(1.0 + 0.8 * phase, 0.01)
    return p / p.sum()


def make_smooth_random(n, rng):
    """Random smooth profile via Fourier features."""
    k = rng.integers(3, 12)
    coefs = rng.normal(0, 1, k + 1)
    t = np.linspace(0, 1, n)
    f = np.zeros(n)
    for m in range(1, k + 1):
        f += coefs[m] * np.cos(2 * np.pi * m * t)
    p = np.maximum(1.0 + 0.7 * f, 0.01)
    return p / p.sum()


def make_u_profile(n, alpha):
    t = np.linspace(0, 1, n)
    p = 1.0 + alpha * (2 * t - 1) ** 2
    return p / p.sum()


def make_v_profile(n, alpha):
    t = np.linspace(0, 1, n)
    p = 1.0 - alpha * (2 * t - 1) ** 2
    p = np.maximum(p, 0.01)
    return p / p.sum()


def make_two_peak(n, center, width, height):
    t = np.linspace(0, 1, n)
    p = 1.0 / (1.0 + ((t - center) / width) ** 2) + 1.0 / (1.0 + ((t - (1 - center)) / width) ** 2)
    p = p + 0.01
    return p / p.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, p0)
    return result / result.sum()

    print("Phase 4: Coordinate descent")
    phase4_end = min(deadline, t_start + 1720.0)

    top_seeds = sorted(candidates, key=lambda x: x[0])[:4]
    for i, (sc_c, p_c) in enumerate(top_seeds):
        if time.monotonic() > phase4_end:
            break
        p0 = interpolate_profile(p_c, n_main)
        per = min(phase4_end, time.monotonic() + 35.0)
        p_best, sc = coordinate_polish(p0, per)
        sc_actual = score_normalized(p_best)
        local_update(p_best, sc_actual, f"cd_{i}")

    print(f"  After phase 4: best={best_score:.8f}")

    # Phase 5: Try n=1024 if time permits
    print("Phase 5: Try n=1024")
    phase5_end = min(deadline, t_start + 1760.0)
    if time.monotonic() < phase5_end - 60:
        n_big = 1024
        top_c = sorted(candidates, key=lambda x: x[0])[:2]
        for i, (sc_c, p_c) in enumerate(top_c):
            if time.monotonic() > phase5_end:
                break
            p0 = interpolate_profile(p_c, n_big)
            per = min(phase5_end, time.monotonic() + 40.0)
            p_best, sc = lp_optimize(p0, per, verbose=False)
            sc_actual = score_normalized(p_best)
            local_update(p_best, sc_actual, f"lp1024_{i}")

    # Final selection with exact scoring
    print("Final selection")
    final_seq = best_seq
    final_score = best_score

    if final_seq is not None and final_seq.size <= 2048:
        sc_exact = score_true(final_seq)
        if np.isfinite(sc_exact):
            final_score = min(final_score, sc_exact)

    for sc_c, p_c in candidates[:10]:
        if p_c.size <= 2048:
            sc_e = score_true(p_c)
            if np.isfinite(sc_e) and sc_e < final_score:
                final_score = sc_e
                final_seq = p_c

    # Verify with evaluator
    if final_seq is not None and final_seq.size <= 2048:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                final_score = min(final_score, sc_eval)
        except Exception:
            pass

    if final_seq is not None:
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s
        print(f"  FINAL: score={final_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fb = make_u_profile(256, 0.8)
    print(f"  FALLBACK: {score_normalized(fb):.10f}")
    return [float(x) for x in fb]
