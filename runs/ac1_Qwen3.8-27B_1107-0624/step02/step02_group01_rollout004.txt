"""
Optimized search for minimizing 2n*max(conv(a,a))/sum(a)^2 using:
1. Direct peak-minimizing LP (better than previous sum-maximizing LP)
2. Away-step Frank-Wolfe for fast local improvements

import numpy as np
import time

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    k = 1
    while k < 2 * n - 1:
        k <<= 1
    fa = np.fft.rfft(a, k)
    c = np.fft.irfft(fa * fa, k)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=12):

        phase = rng.uniform(0, 2 * np.pi)

    return a / a.sum()


def make_quadratic_profile(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_exponential_profile(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t)
    return a / a.sum()


def make_deterministic_profiles(n):
    profs = []
    for eta in (1.0, 1.5, 2.0, 2.5, 3.0):
        profs.append(make_linear_profile(n, eta))
    for a1, a3 in ((1.2, -0.8), (1.5, -1.2), (2.0, -1.5), (2.5, -2.0),
                   (1.0, -0.5), (1.5, -2.5), (2.0, -3.5), (3.0, -2.0)):
        profs.append(make_cubic_profile(n, a1, a3))
    for alpha in (1.0, 2.0, 3.0, 4.0):
        profs.append(make_quadratic_profile(n, alpha))
    for rate in (1.0, 2.0, 3.0, 4.0):
        profs.append(make_exponential_profile(n, rate))
    return profs


def build_constraint_matrix(f, n):
    """Build A[k,j] = f[k-j] if 0<=k-j<=n-1 else 0, shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_peak_min(f, S, n):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=S, g>=0.
    Returns (g, t_val) or (None, None) on failure.
    """
    A = build_constraint_matrix(f, n)
    # x = [g_0,...,g_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.success and res.x is not None:
        g = res.x[:n]
        t_val = float(res.x[n])
        return g, t_val
    return None, None


def solve_lp_with_regularization(f, S, n, mu=0.01):
    """
    Solve: min (t + mu * ||g - f||^2 / (2*S)) s.t. (f*g)[k] <= t, sum(g)=S, g>=0.
    Approximate by linearizing the quadratic term at f:
    min (t + mu * (g-f)·(g-f)/(2S)) ≈ min (t + mu * (g·f - f·f - f·(g-f))/(2S))
    This is tricky. Instead, use a simpler approach:
    min t + mu * sum_j max(0, |g_j - f_j|)
    Approximate with: min t + mu * sum_j (g_j + |g_j - f_j|)
    Or just use the pure LP and add regularization in the objective via extra vars.
    
    For simplicity, just use the pure LP.
    """
    return solve_lp_peak_min(f, S, n)


def line_search(f, g, S, c, n, max_iter=50):
    """Golden section search for optimal t in [(1-t)f + tg]."""
    def obj(t):
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-30:
            return np.inf
        f_new = f_new / s
        return compute_score(f_new)
    
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = obj(c1), obj(c2)
    
    for _ in range(max_iter):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
        if hi - lo < 1e-8:
            break
    
    t_opt = (lo + hi) / 2.0
    return t_opt, min(f1, f2, obj(t_opt))


def lp_optimize(f0, deadline, rng, max_iter=50000, verbose=False):
    """LP-based iterative optimization with direct peak minimization."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    S = f.sum()
    if S < 1e-30:
        return f0, np.inf
    f = f / S
    S = 1.0
    n = f.size
    
    c = fft_convolve(f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    best_score = 2.0 * n * M
    no_imp = 0
    consec_fail = 0
    
    n_constraints = 2 * n - 1
    A_cache = None
    A_last_f = None
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Rebuild constraint matrix if f changed significantly
        if A_cache is None or A_last_f is None or np.abs(f - A_last_f).max() > 1e-10:
            A_cache = build_constraint_matrix(f, n)
            A_last_f = f.copy()
        
        # Solve LP: min t s.t. (f*g)[k] <= t, sum(g) = 1, g >= 0
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.hstack([A_cache, -np.ones((n_constraints, 1))])
        b_ub = np.zeros(n_constraints)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0.0, None)] * (n + 1)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        
        if not res.success or res.x is None:
            consec_fail += 1
            if consec_fail > 100:
                # Escape: add noise
                f = f + 0.05 * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                S = f.sum()
                if S < 1e-30:
                    f = best_f.copy()
                    continue
                f = f / S
                c = fft_convolve(f)
                M = float(c.max())
                A_cache = None
                consec_fail = 0
                no_imp = 0
            continue
        
        g = res.x[:n]
        t_lp = float(res.x[n])
        
        # Check if g ≈ f (KKT point)
        if np.abs(g - f).max() < 1e-8:
            no_imp += 1
            if no_imp > 200:
                f = f + 0.03 * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                S = f.sum()
                if S < 1e-30:
                    f = best_f.copy()
                    continue
                f = f / S
                c = fft_convolve(f)
                M = float(c.max())
                A_cache = None
                no_imp = 0
            continue
        
        consec_fail = 0
        
        # Line search
        t_opt, best_local = line_search(f, g, 1.0, c, n)
        
        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = np.maximum(f_new, 0.0)
        S_new = f_new.sum()
        if S_new < 1e-30:
            # Try smaller step
            for t_try in (0.5 * t_opt, 0.25 * t_opt, 0.1):
                f_new = (1.0 - t_try) * f + t_try * g
                f_new = np.maximum(f_new, 0.0)
                S_new = f_new.sum()
                if S_new > 1e-30:
                    break
            if S_new < 1e-30:
                t_opt = 0.1
                f_new = (1.0 - t_opt) * f + t_opt * g
                f_new = np.maximum(f_new, 0.0)
                S_new = f_new.sum()
        
        f_new = f_new / S_new
        c_new = fft_convolve(f_new)
        M_new = float(c_new.max())
        
        if M_new < M - 1e-12:
            f = f_new
            M = M_new
            A_cache = None  # Invalidate
            no_imp = 0
            if M < best_M:
                best_M = M
                best_f = f.copy()
                best_score = 2.0 * n * M
        else:
            # Try half step
            t_half = 0.5 * t_opt
            f_half = (1.0 - t_half) * f + t_half * g
            f_half = np.maximum(f_half, 0.0)
            Sh = f_half.sum()
            if Sh > 1e-30:
                f_half = f_half / Sh
                ch = fft_convolve(f_half)
                Mh = float(ch.max())
                if Mh < M - 1e-12:
                    f = f_half
                    M = Mh
                    A_cache = None
                    no_imp = 0
                    if M < best_M:
                        best_M = M
                        best_f = f.copy()
                        best_score = 2.0 * n * M
                else:
                    no_imp += 1
            else:
                no_imp += 1
        
        if no_imp > max(500, 2 * n):
            # Stronger perturbation
            f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n)) + 1e-6 * np.ones(n)
            f = f / f.sum()
            c = fft_convolve(f)
            M = float(c.max())
            A_cache = None
            no_imp = 0
        
        if verbose and it % 500 == 0:
            print(f"    LP iter {it}: M={M:.12f}, best={best_M:.12f}")
    
    return best_f, 2.0 * n * best_M


def fw_optimize(f0, deadline, rng, max_iter=200000):
    """Frank-Wolfe (away-step) optimization for fast local improvements."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    f = f / f.sum()
    n = f.size
    
    c = fft_convolve(f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Find active peaks (within 1% of max)
        threshold = M * 0.99
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient: grad[j] = mean of a[k-j] over active k
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Subgradient: grad[j] = (1/|active|) * sum_{k in active} a[k-j]
        grad = np.zeros(n)
        for k in active:
            # a[k-j] is valid when 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi + 1] += f[k - np.arange(j_hi - j_lo + 1)]
        grad /= len(active)
        
        # FW step: minimize grad · g over simplex => g = e_{argmin grad}
        j_star = int(np.argmin(grad))
        
        # Away-step: if f is close to a vertex, move away
        # Check if any coordinate dominates
        sorted_idx = np.argsort(f)[::-1]
        if f[sorted_idx[0]] > 0.1 * n / len(np.where(f > 1e-10)[0]) and f[sorted_idx[0]] > f[j_star] * 2:
            # Move away from the dominant coordinate
            j_away = int(sorted_idx[0])
            if j_away != j_star:
                # Away-step FW
                t_opt = 0.5
                f_new = f.copy()
                # Move from j_away to j_star
                amount = f_new[j_away] * 0.5
                f_new[j_away] -= amount
                f_new[j_star] += amount
                f_new = np.maximum(f_new, 0.0)
                S = f_new.sum()
                if S > 1e-30:
                    f_new = f_new / S
                    c_new = fft_convolve(f_new)
                    M_new = float(c_new.max())
                    if M_new < best_M - 1e-12:
                        best_M = M_new
                        best_f = f_new.copy()
                        f = f_new
                        M = M_new
                        no_imp = 0
                        continue
        
        # Standard FW step
        # g = e_{j_star}, update f = (1-t)f + t*e_{j_star}
        # Optimal t for FW: t* = <grad, f> / <grad, f - e_{j_star}>
        num = float(grad @ f)
        den = float(grad @ (f - np.eye(n, j_star)))
        if den > 1e-15:
            t_opt = num / den
        else:
            t_opt = 2.0 / (it + 2)
        t_opt = min(max(t_opt, 0.0), 1.0)
        
        f_new = (1.0 - t_opt) * f + t_opt * np.eye(n, j_star)
        f_new = np.maximum(f_new, 0.0)
        S = f_new.sum()
        if S < 1e-30:
            continue
        f_new = f_new / S
        
        c = fft_convolve(f_new)
        M_new = float(c.max())
        
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > 5000:
            f = f + 0.02 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            c = fft_convolve(f)
            M = float(c.max())
            no_imp = 0
    
    return best_f, 2.0 * n * best_M


def softmax_polish(f0, deadline, rng, lr=0.05, max_iter=100000):
    """Adam-based optimization in log-space with soft-max smoothing."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size
    
    z = np.log(f)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-12, 0.1 * M_val)
    
    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_corr = 1
    while N_corr < 3 * n:
        N_corr <<= 1
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))
        
        # Gradient in log-space
        fw_fft = np.fft.rfft(w, N_corr)
        fa_fft = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw_fft * np.conj(fa_fft), N_corr)[:n]
        
        grad_z = 2.0 * f * (h - L)
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()
        
        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        c = fft_convolve(f)
        M_new = float(c.max())
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if t_adam % 500 == 0:
            beta = max(1e-12, beta * 0.7)
            if t_adam % 2000 == 0:
                lr = max(0.001, lr * 0.95)
        
        if no_imp > max(500, 2 * n):
            z = 0.85 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
            lr = 0.05
    
    return best_f, 2.0 * n * best_M


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining LP, FW, and gradient methods.
    Phase 1: Quick LP at n=512 from diverse starts
    Phase 2: Extended LP on best candidates
    Phase 3: FW polish
    Phase 4: Softmax gradient polish
    Phase 5: Try n=1024
    Phase 6: Final refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
    
    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None
    
    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")
    
    # Phase 1: LP at n=512 from diverse starts
    n1 = 512
    phase1_end = min(deadline, t_start + 480.0)
    
    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)
    
    # Diverse deterministic profiles
    starts.extend(make_deterministic_profiles(n1)[:10])
    
    # Smooth random
    for _ in range(10):
        starts.append(make_smooth_random(n1, rng))
    
    candidates = []
    per_start = max(5.0, (phase1_end - t_start) / max(1, len(starts)))
    
    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s
        
        per = min(phase1_end, time.monotonic() + per_start)
        p_best, sc = lp_optimize(p0, per, rng, max_iter=10000)
        try_update(p_best, sc)
        candidates.append((sc, p_best))
        if i % 5 == 0:
            print(f"  Phase 1 [{i}/{len(starts)}]: best={best_score:.10f}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")
    
    # Phase 2: Extended LP on top candidates
    phase2_end = min(deadline, t_start + 1020.0)
    top_k = min(4, len(candidates))
    per_cand = max(20.0, (phase2_end - t_start) / max(1, top_k))
    
    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates[i]
        per = min(phase2_end, time.monotonic() + per_cand)
        p_best, sc = lp_optimize(p0, per, rng, max_iter=50000, verbose=(i == 0))
        try_update(p_best, sc)
        print(f"  Phase 2 [{i}]: score={sc:.10f}")
    
    print(f"After Phase 2: best={best_score:.10f}")
    
    # Phase 3: FW polish on best
    phase3_end = min(deadline, t_start + 1260.0)
    if best_seq is not None and best_seq.size == n1:
        per = min(phase3_end, time.monotonic() + 120.0)
        p_best, sc = fw_optimize(best_seq, per, rng, max_iter=500000)
        try_update(p_best, sc)
        print(f"After Phase 3 FW: best={best_score:.10f}")
    
    # Phase 4: Softmax polish
    phase4_end = min(deadline, t_start + 1500.0)
    if best_seq is not None and best_seq.size == n1:
        per = min(phase4_end, time.monotonic() + 150.0)
        p_best, sc = softmax_polish(best_seq, per, rng, lr=0.05)
        try_update(p_best, sc)
        print(f"After Phase 4 softmax: best={best_score:.10f}")
    
    # Also try FW on second best
    if len(candidates) > 1 and time.monotonic() < phase4_end - 30:
        _, p0 = candidates[1]
        per = min(phase4_end, time.monotonic() + 60.0)
        p_best, sc = fw_optimize(p0, per, rng, max_iter=200000)
        try_update(p_best, sc)
    
    print(f"After Phase 4: best={best_score:.10f}")
    
    # Phase 5: Try n=1024
    n2 = 1024
    phase5_end = min(deadline, t_start + 1680.0)
    if best_seq is not None and time.monotonic() < phase5_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase5_end, time.monotonic() + 120.0)
            p_best, sc = lp_optimize(p_ext, per, rng, max_iter=20000)
            try_update(p_best, sc)
            print(f"Phase 5 LP n={n2}: score={sc:.10f}")
            
            if time.monotonic() < phase5_end - 30:
                per2 = min(phase5_end, time.monotonic() + 60.0)
                p_best2, sc2 = fw_optimize(p_best, per2, rng, max_iter=100000)
                try_update(p_best2, sc2)
                print(f"Phase 5 FW n={n2}: score={sc2:.10f}")
    
    print(f"After Phase 5: best={best_score:.10f}")
    
    # Phase 6: Final refinement on absolute best
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 90.0)
        p_best, sc = lp_optimize(best_seq, per, rng, max_iter=30000)
        try_update(p_best, sc)
        
        if time.monotonic() < deadline - 10:
            per2 = min(deadline - 5, time.monotonic() + 30.0)
            p_best2, sc2 = fw_optimize(p_best, per2, rng, max_iter=100000)
            try_update(p_best2, sc2)
    
    print(f"After Phase 6: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")
    
    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(sc_eval):
                print(f"Final evaluated score: {sc_eval:.10f}")
        except Exception as e:
            print(f"Evaluation error: {e}")
        
        return [float(x) for x in final]
    
    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
