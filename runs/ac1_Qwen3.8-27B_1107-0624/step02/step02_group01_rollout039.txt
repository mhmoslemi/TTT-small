def solve_lp_direction(f, M, n, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(f0, deadline, rng, max_iter=10000, verbose=False):

    f = f0.copy().astype(np.float64)

"""
Multi-scale peak-flattening optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses top-k peak targeting with adaptive softmax parameterization in log-space,
combined with LP polish and multi-resolution search (n=512 -> 1024 -> 2048).
Key improvements over parent: focused peak targeting, better time allocation
to larger n, and hybrid soft+hard peak weighting.
"""

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=10):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.1, 1.2) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_barker_like(n, rng):
    """Create a sequence inspired by Barker codes - alternating high/low with smooth transitions."""
    a = np.ones(n)
    n_segments = rng.integers(4, 12)
    for i in range(n_segments):
        start = int(i * n / n_segments)
        end = int((i + 1) * n / n_segments)
        if end > n:
            end = n
        if end <= start:
            continue
        level = rng.uniform(0.3, 2.0)
        a[start:end] = level
    # Smooth it
    from scipy.ndimage import gaussian_filter1d
    a = gaussian_filter1d(a, sigma=max(1, n // 50))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def compute_gradient_weights(c, beta, top_k, boost):
    """Compute gradient weights combining softmax background with top-k boost."""
    c_shifted = c - c.max()
    w_sm = np.exp(c_shifted / beta)
    s = w_sm.sum()
    if s < 1e-30:
        w_sm = np.zeros_like(c)
        w_sm[int(np.argmax(c))] = 1.0
    else:
        w_sm = w_sm / s

    if top_k > 0 and top_k < len(c):
        top_idx = np.argsort(c)[-top_k:]
        w_total = w_sm.copy()
        w_total[top_idx] *= boost
        w_total = w_total / w_total.sum()
    else:
        w_total = w_sm

    return w_total


def peak_flatten_optimize(f0, deadline, rng, top_k=15, boost=5.0, lr=0.06,
                          beta_init=None, beta_decay=0.92, beta_floor=0.005,
                          max_iter=200000):
    """
    Peak-flattening optimization using combined softmax + top-k peak targeting.
    Works in log-space (softmax parameterization) with Adam optimizer.
    """
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())

    if beta_init is None:
        beta = 0.08 * M_val
    else:
        beta = beta_init

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_fft = next_pow2(4 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute gradient weights
        w = compute_gradient_weights(c, beta, top_k, boost)

        # Compute gradient via cross-correlation: grad[j] = 2 * sum_m w[j+m] * a[m]
        fw = np.fft.rfft(w, N_fft)
        fa = np.fft.rfft(f, N_fft)
        corr = np.fft.irfft(fw * np.conj(fa), N_fft)
        dL_da = 2.0 * corr[:n]

        # Log-space gradient: dL/dz[i] = a[i] * (dL/da[i] - mean(dL/da * a))
        mean_grad = float(np.dot(dL_da, f))
        dL_dz = f * (dL_da - mean_grad)

        if not np.all(np.isfinite(dL_dz)):
            z = best_z.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * dL_dz
        v_adam = b2 * v_adam + (1.0 - b2) * (dL_dz * dL_dz)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        f_new = np.exp(z - z.max())
        s_new = f_new.sum()
        if s_new < 1e-30 or not np.isfinite(s_new):
            z = best_z.copy()
            continue
        f_new = f_new / s_new
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        c_new = fft_convolve(f_new)
        M_new = float(c_new.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f_new.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        f = f_new
        c = c_new
        M_val = M_new

        # Decay beta periodically
        if t_adam % 400 == 0:
            beta = max(beta_floor, beta * beta_decay)

        # Restart if stuck
        if no_imp > max(400, 4 * n):
            pert_scale = min(0.5, 2.0 / np.sqrt(n))
            z = (1.0 - pert_scale) * best_z + pert_scale * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            if M_val < best_M:
                best_M = M_val
                best_f = f.copy()
                best_z = z.copy()
            beta = max(beta_floor, 0.08 * M_val)
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            no_imp = 0

    return best_f, best_M


def solve_lp_direction(f, M, n, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(f0, deadline, rng, max_iter=5000, verbose=False):
    """Run LP-based iterative optimization."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = solve_lp_direction(f, M, n, time_limit=12)
        if g is None:
            no_imp += 1
            if no_imp > 20:
                f = best_f + 0.01 * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-30:
                    f = best_f.copy()
                    M = best_M
                    continue
                f = f / fs
                M = float(fft_convolve(f).max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 80:
                f = best_f + 0.02 * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-30:
                    f = best_f.copy()
                    M = best_M
                    no_imp = 0
                    continue
                f = f / fs
                M = float(fft_convolve(f).max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
            continue

        g = g / g_sum

        improved = False
        for t in (0.3, 0.15, 0.5, 0.1, 0.4, 0.2, 0.05, 0.7, 0.6):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(fft_convolve(f_new).max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp % 40 == 0:
                f = 0.85 * best_f + 0.15 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = float(fft_convolve(f).max())
                no_imp = 0

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: M={M:.12f}, best={best_M:.12f}")

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-scale peak-flattening + LP optimization.
    Phase 1: Peak-flattening at n=512 from multiple diverse starts
    Phase 2: LP polish on best n=512 candidates
    Phase 3: Extension to n=1024 with optimization
    Phase 4: Extension to n=2048 (if time permits)
    Phase 5: Final polish on the best candidate
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = parent.copy()
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # Phase 1: Peak-flattening at n=512
    n1 = 512
    phase1_end = min(deadline, t_start + 360.0)

    starts = []
    if parent is not None:
        if parent.size == n1:
            starts.append(parent.copy())
        else:
            p_int = interpolate_profile(parent, n1)
            if p_int is not None:
                starts.append(p_int)

    # Diverse deterministic starts
    for eta in (0.6, 1.0, 1.5, 2.0, 2.5, 3.0):
        starts.append(make_linear_profile(n1, eta))
    for a1, a3 in ((1.2, -0.8), (1.8, -1.2), (2.2, -1.8), (1.0, -0.5), (2.5, -2.5)):
        starts.append(make_cubic_profile(n1, a1, a3))
    for a1, a2 in ((1.0, -0.5), (1.5, -1.0), (2.0, -1.5), (0.5, 0.5)):
        starts.append(make_quadratic_profile(n1, a1, a2))

    # Random smooth starts
    for _ in range(6):
        starts.append(make_smooth_random(n1, rng, n_harm=10))

    # Barker-like starts
    for _ in range(3):
        starts.append(make_barker_like(n1, rng))

    candidates = []
    per_start = max(8.0, (phase1_end - t_start) / max(1, len(starts)))

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s

        per = min(phase1_end, time.monotonic() + per_start)
        p_best, M_best = peak_flatten_optimize(p0, per, rng, top_k=15, boost=6.0, lr=0.07)
        sc = 2.0 * n1 * M_best
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = p_best.copy()
        candidates.append((sc, p_best))
        if i % 3 == 0:
            print(f"  Phase 1 start {i+1}/{len(starts)}: score={sc:.10f}, best={best_score:.10f}")

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: LP polish on top candidates at n=512
    phase2_end = min(deadline, t_start + 600.0)
    top_k2 = min(3, len(candidates))
    per_cand = max(15.0, (phase2_end - t_start) / max(1, top_k2))

    for i in range(top_k2):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates[i]
        per = min(phase2_end, time.monotonic() + per_cand)
        p_best, M_best = lp_optimize(p0, per, rng, max_iter=4000)
        sc = 2.0 * n1 * M_best
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = p_best.copy()
        print(f"  Phase 2 LP cand {i}: score={sc:.10f}, best={best_score:.10f}")

    # Also run peak-flatten on the LP result
    if time.monotonic() < phase2_end - 20:
        per = min(phase2_end, time.monotonic() + 60.0)
        p_best, M_best = peak_flatten_optimize(best_seq, per, rng, top_k=10, boost=8.0, lr=0.04)
        sc = 2.0 * n1 * M_best
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = p_best.copy()
        print(f"  Phase 2 final polish: best={best_score:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Extend to n=1024
    n2 = 1024
    phase3_end = min(deadline, t_start + 1100.0)

    if best_seq is not None and time.monotonic() < phase3_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            # Peak-flatten at n=1024
            per = min(phase3_end, time.monotonic() + 250.0)
            p_best, M_best = peak_flatten_optimize(p_ext, per, rng, top_k=20, boost=5.0, lr=0.05)
            sc = 2.0 * n2 * M_best
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_seq = p_best.copy()
            print(f"  Phase 3 peak-flatten n={n2}: score={sc:.10f}, best={best_score:.10f}")

            # LP polish at n=1024
            if time.monotonic() < phase3_end - 30:
                per2 = min(phase3_end, time.monotonic() + 120.0)
                p_best2, M_best2 = lp_optimize(p_best, per2, rng, max_iter=3000)
                sc2 = 2.0 * n2 * M_best2
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = float(sc2)
                    best_seq = p_best2.copy()
                print(f"  Phase 3 LP n={n2}: score={sc2:.10f}, best={best_score:.10f}")

            # Final peak-flatten polish
            if time.monotonic() < phase3_end - 10:
                per3 = min(phase3_end, time.monotonic() + 60.0)
                p_best3, M_best3 = peak_flatten_optimize(best_seq, per3, rng, top_k=15, boost=7.0, lr=0.03)
                sc3 = 2.0 * n2 * M_best3
                if np.isfinite(sc3) and sc3 < best_score:
                    best_score = float(sc3)
                    best_seq = p_best3.copy()
                print(f"  Phase 3 final polish n={n2}: best={best_score:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Try n=2048
    n3 = 2048
    phase4_end = min(deadline, t_start + 1500.0)

    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p_ext = interpolate_profile(best_seq, n3)
        if p_ext is not None:
            per = min(phase4_end, time.monotonic() + 300.0)
            p_best, M_best = peak_flatten_optimize(p_ext, per, rng, top_k=25, boost=5.0, lr=0.04)
            sc = 2.0 * n3 * M_best
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_seq = p_best.copy()
            print(f"  Phase 4 n={n3}: score={sc:.10f}, best={best_score:.10f}")

            if time.monotonic() < phase4_end - 20:
                per2 = min(phase4_end, time.monotonic() + 100.0)
                p_best2, M_best2 = peak_flatten_optimize(p_best, per2, rng, top_k=20, boost=6.0, lr=0.02)
                sc2 = 2.0 * n3 * M_best2
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = float(sc2)
                    best_seq = p_best2.copy()
                print(f"  Phase 4 polish n={n3}: best={best_score:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # Phase 5: Final polish on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        per = min(deadline - 10, time.monotonic() + 120.0)

        # Peak-flatten with very fine tuning
        p_best, M_best = peak_flatten_optimize(best_seq, per, rng,
                                               top_k=max(10, n_best // 32),
                                               boost=8.0, lr=0.02,
                                               beta_decay=0.98, beta_floor=0.002)
        sc = compute_score(p_best)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = p_best.copy()
        print(f"  Phase 5 final polish: score={sc:.10f}, n={p_best.size}")

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        # Verify with evaluator for small sequences
        if final.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
