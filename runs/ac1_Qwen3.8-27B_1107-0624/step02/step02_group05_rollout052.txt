import numpy as np
import time
import math

"""
Improved optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses adaptive-temperature soft-max gradient descent in log-domain with
simulated annealing restarts, diagonal gradient corrections, and
multi-scale exploration (n=64 -> 256 -> 512).
"""


def compute_score(f):
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_conv_max(f):
    c = np.convolve(f, f)
    return float(c.max())


def normalize_nonneg(f):
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def soft_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def compute_grad(f, beta):
    """Gradient of soft-max(conv(f,f)) w.r.t. f with diagonal correction."""
    n = len(f)
    c = np.convolve(f, f)
    w = soft_max_weights(c, beta)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(f[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    h = h[n - 1:2 * n - 1]
    grad = 2.0 * h
    # Diagonal correction: where k=2i, the derivative is f[i] not 2*f[i]
    k_vals = 2 * np.arange(n)
    valid = k_vals < (2 * n - 1)
    if np.any(valid):
        grad[valid] -= w[k_vals[valid]] * f[valid]
    return grad, w, c


def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def make_starts(n, rng, parent=None, count=10):
    """Generate diverse starting points."""
    starts = []
    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy())
        else:
            starts.append(interpolate_profile(parent, n))
    t = np.linspace(0, 1, n)
    f = 1.0 + 2.0 * t
    starts.append(f / f.sum())
    u = 2.0 * t - 1.0
    f = 1.0 + 1.5 * u - 1.0 * u ** 3
    f = np.maximum(f, 1e-6)
    starts.append(f / f.sum())
    f = 1.0 + 0.4 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t)
    starts.append(f / f.sum())
    f = 1.0 + 0.3 * np.sin(np.pi * t)
    starts.append(f / f.sum())
    for _ in range(count):
        coeffs = rng.normal(size=8)
        f = 1.0
        for k in range(4):
            f += 0.25 * coeffs[2 * k] * np.cos(2 * np.pi * (k + 1) * t)
            f += 0.25 * coeffs[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * t)
        f = np.maximum(f, 1e-6)
        starts.append(f / f.sum())
    return starts


def grad_optimize(f_init, deadline, rng, lr=0.04, beta_start=0.5, beta_end=0.003,
                  max_iters=200000, verbose=False, restart_every=1000,
                  noise_scale=0.25, momentum_decay=0.995):

    Adam optimization in log-domain with adaptive beta (cosine schedule).
    Minimizes soft-max(conv(f,f)) ~ max(conv(f,f)).
    Returns (best_f, best_M).
    """
    f = normalize_nonneg(f_init.copy())

        return f_init, compute_conv_max(f_init)

    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    no_imp = 0
    current_lr = lr
    # Two-phase beta: first 60% of time uses slow decay, last 40% uses fast
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        # Adaptive beta with two-phase cosine
        progress = min(1.0, it / max(1, max_iters))
        if progress < 0.6:

            f = f / f.sum()
            m.fill(0)
            v.fill(0)

            continue
        # Adam update

                z = 0.85 * np.log(np.maximum(best_f, 1e-30)) + ns * rng.normal(size=n)
                z = np.clip(z, -60, 60)
                z -= z.mean()
                exp_z = np.exp(z)
                f = exp_z / exp_z.sum()
                m.fill(0)
                v.fill(0)
                t_count = 0
                current_lr = lr
                no_imp = 0
                restart_every = max(500, int(restart_every * 0.9))
        if verbose and it % 3000 == 0:
            score = 2.0 * n * M
            print(f"      iter {it}: score={score:.10f}, best_score={2.0*n*best_M:.10f}, beta={beta:.5f}, lr={current_lr:.6f}")
    return best_f, best_M
