def adam_optimize_symmetric(a_init, deadline, rng, n_half, lr=0.05, beta0=0.05):

    z = np.log(a)

    t = 0

    base_lr = lr

            break
        if it % 1000 == 0 and it > 0:
            # Check time budget
            pass

        N = 1
        while N < 2*n - 1:
            N <<= 1
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2*n-1]
        max_c = float(c.max())
        score = 2.0 * n * max_c
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Compute gradient
        L, w = soft_max_and_grad(c, beta)
        
        # grad[m] = 2 * sum_k w[k] * a[k-m]
        Nc = 1
        while Nc < 2*n - 1:
            Nc <<= 1
        fw_c = np.fft.rfft(w, Nc)
        fa_rev_c = np.fft.rfft(a[::-1], Nc)
        h = np.fft.irfft(fw_c * fa_rev_c, Nc)
        corr = h[n-1:2*n-1]
        grad = 2.0 * corr
        
        # Project to tangent space (sum constraint)
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)  # gradient in log-space, projected
        gz -= gz.mean()  # remove mean for log-space
        
        if not np.all(np.isfinite(gz)):
            break
        
        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        # Enforce symmetry
        z[:n_half] = (z[:n_half] + z[n-1-n_half:n-1:-1]) / 2
        
        # Convert back
        z = np.clip(z, -50, 50)
        e = np.exp(z)
        a = e / e.sum()
        
        # Anneal temperature
        if it % 100 == 0:
            beta = beta * 0.95
            if beta < 1e-6:
                beta = 1e-6
        
        # Restart if stuck
        if no_imp > 500:
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + 0.1 * rng.normal(size=n)
            z[:n_half] = (z[:n_half] + z[n-1-n_half:n-1:-1]) / 2

            a = e / e.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0

def make_smooth_profile(n, coeffs):
    """Generate a smooth profile from Fourier coefficients."""
    t = np.linspace(0, 1, n)
    f = 1.0
    for k in range(len(coeffs)//2):
        f += coeffs[2*k] * np.cos(2*np.pi*(k+1)*t)
        f += coeffs[2*k+1] * np.sin(2*np.pi*(k+1)*t)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def make_ramp_profile(n, alpha=0.0, beta=1.0):
    """Ramp-like profile: a[i] proportional to (i/n)^alpha * ((n-1-i)/n)^beta."""
    t = np.linspace(0, 1, n)
    f = t**alpha * (1-t)**beta + 1e-10
    return f / f.sum()

def lp_direction(f, M, deadline):
    """Solve LP to find direction g that maximizes sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix
    # (f*g)[k] = sum_i A[k,i] * g[i] where A[k,i] = f[k-i] if valid
    # Number of constraints: 2n-1
    # This could be slow for large n. Let's only use it for n <= 512.
    if n > 512:
        return None
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[k - np.arange(lo, hi+1)]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 3.0})
        if res.success:
            return np.maximum(res.x, 0)
    except:
        pass
    return None
