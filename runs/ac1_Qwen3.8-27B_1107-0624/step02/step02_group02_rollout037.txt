Let me now think about specific improvements to the LP step:

The LP: max 1^T b s.t. A b <= M, b >= 0, where A is the (2n-1) x n convolution matrix with rows corresponding to lags and columns to positions, with A[k,j] = f[j] if k >= j and k-j < n, else 0.

The dual: min M * 1^T y s.t. A^T y >= 1, y >= 0.
The optimal dual solution y* tells us which constraints are "important" (y*[k] > 0 means constraint k is tight).

We can use y* to compute a "force" on each position: the gradient of the objective with respect to b is (A^T y* - 1) (for the LP). But more useful for us is: the dual tells us which lags are at the peak.

After computing g (the LP solution), we interpolate f_new = (1-t)*f + t*g. Then we compute the new peak. If the peak increased, we reduce t.

**Improvement: Use the LP dual to compute a gradient for f itself.**

The peak of conv(f,f) at lag k* is f^T T_k* f. The gradient w.r.t. f is 2 T_k* f (where T_k* is the shift matrix).

If we know which lags are "active" (from the LP dual or from the actual peak), we can compute:
grad_f = 2 * sum_{k in active} w_k * T_k f

where w_k are weights (from the LP dual or uniform over active peaks).

Then we do a projected gradient step: f_new = project_simplex(f - alpha * grad_f).

This is complementary to the LP approach and might escape from local minima that the LP gets stuck in.

**Another improvement: Use the "mirror descent" framework.**

Instead of Euclidean projection, use a multiplicative update:
f_new[i] = f[i] * exp(-alpha * grad_f[i]) / Z

where Z normalizes. This keeps f positive and is more natural for the non-negative constraint.

import numpy as np
import time
from scipy.optimize import linprog

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0: return np.inf
    s = float(a.sum())
    if s < 0.01: return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n: k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0: return np.array([])
    if n <= 256: return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0)

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_direction(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except:
        pass
    return None

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def conv_gradient(c, active_mask, f):
    """Compute gradient contribution: sum over active k of w_k * (T_k f)."""
    n = len(f)
    grad = np.zeros(n)
    active_vals = c[active_mask]
    total = active_vals.sum()
    if total < 1e-20:
        return grad
    for k in np.where(active_mask)[0]:
        w = c[k] / total
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            grad[i] += w * f[k - i]
    return grad


def conv_gradient_fast(c, active_mask, f):
    """Fast gradient computation using FFT for large n."""
    n = len(f)
    if n <= 128:
        return conv_gradient(c, active_mask, f)
    # Use correlation: grad[i] = sum_k w_k * f[k-i] (valid range)
    # This is equivalent to correlating f with the weight pattern
    w_pattern = np.zeros(2 * n - 1)
    indices = np.where(active_mask)[0]
    active_vals = c[indices]
    total = active_vals.sum()
    if total < 1e-20:
        return np.zeros(n)
    w_pattern[indices] = active_vals / total
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w_pattern, N)
    ff = np.fft.rfft(f[::-1], N)
    h = np.fft.irfft(fw * ff, N)
    return h[n - 1:2 * n - 1]


def build_conv_matrix_sparse(f):
    """Build convolution matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M):
    """Find LP direction: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    if n > 2048:
        return None
    A = build_conv_matrix_sparse(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-12:
        result = result / s
    return result

        update(result, sc)
        print(f"After large LP: score={sc:.10f}")

        # Also try ramp at large n
        if time.monotonic() < phase3_end - 30:
            p_ramp_large = make_ramp(n_large, 1.0)
            sc_r = compute_score(p_ramp_large)
            per_dl = min(phase3_end, time.monotonic() + 80.0)
            result = lp_optimize(p_ramp_large, per_dl, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
            print(f"  Large ramp: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Logit-space Adam polish ============
    phase4_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        n_polish = best_seq.size
        print(f"Phase 4: polishing at n={n_polish}, current best={best_score:.10f}")

        for lr in [0.05, 0.03, 0.08]:
            if time.monotonic() > phase4_end - 10:
                break
            per_dl = min(phase4_end, time.monotonic() + 100.0)
            p0 = best_seq + 0.001 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = logit_anneal(p0, per_dl, rng, lr=lr, verbose=False)
            update(result, sc)
            print(f"  Adam polish lr={lr}: score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ============ Phase 5: Try n=2048 if time permits ============
    n_xl = 2048
    phase5_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase5_end - 30 and best_seq is not None and best_score > 1.503:
        p_xl = interpolate_profile(best_seq, n_xl)
        sc = compute_score(p_xl)
        update(p_xl, sc)
        print(f"Phase 5: n={n_xl}, score={sc:.10f}")

        lp_dl4 = min(phase5_end, time.monotonic() + 120.0)
        result = lp_optimize(p_xl, lp_dl4, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After XL LP: score={sc:.10f}")

    print(f"After Phase 5: best={best_score:.10f}")

    # ============ Phase 6: Final polish ============
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = min(deadline - 5, time.monotonic() + 40.0)
        n_final = best_seq.size
        p0 = best_seq + 0.0005 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, sc = logit_anneal(p0, per, rng, lr=0.02, verbose=False)
        update(result, sc)
        print(f"Final polish: score={sc:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_ramp(512, 1.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_ramp(256, 1.0)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
