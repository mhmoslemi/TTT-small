def peak_equalization(s, eta, beta, n_peaks):
    """One step of peak equalization."""
    n = len(s)
    b = np.convolve(s, s)  # length 2n-1
    max_b = b.max()
    if max_b < 1e-15:
        return s, max_b
    
    # Compute weights for peaks
    # Use exponential weighting: w[k] = exp(beta * (b[k] - max_b) / max_b)
    # This focuses on the peaks when beta is large
    w = np.exp(beta * (b - max_b) / max_b)
    
    # Or use top-k selection
    # threshold = np.sort(b)[-n_peaks]
    # w = (b >= threshold).astype(float)
    
    # Compute subgradient using FFT
    N_fft = 1
    while N_fft < 4*n:
        N_fft *= 2
    W = np.fft.rfft(w, N_fft)
    S = np.fft.rfft(s, N_fft)
    subgrad = np.fft.irfft(np.conj(S) * W, N_fft)[0:n]
    
    # Multiplicative update
    s_new = s * np.exp(-eta * subgrad / max_b)
    s_new = np.maximum(s_new, 0)
    s_sum = s_new.sum()
    if s_sum < 1e-15:

    s_new = s_new / s_sum
    return s_new, max_b

   In the frequency domain: the circular autocorrelation's DFT is |S_padded[f]|^2. So minimizing the peak of the circular autocorrelation is equivalent to making |S_padded[f]|^2 as flat as possible (since the time-domain peak is bounded by the L1 norm of the frequency domain, and by Parseval, the energy is fixed).

   This gives a lower bound but doesn't directly help with the optimization.
   
   However, the insight is: to minimize the peak of the autocorrelation, we want the power spectrum |S[f]|^2 to be as flat as possible. This is the "flat spectrum" criterion.
   
   For a non-negative sequence s, the power spectrum |S[f]|^2 = |sum_i s[i] e^{-2πif/n'}|^2 where n' is the zero-padded length. Making this flat is a well-studied problem (related to "flat-top" sequences in signal processing).

"""
Peak-equalization optimization for minimizing 2n*max(conv(s,s))/(sum(s))^2.
Uses iterative multiplicative gradient steps with adaptive peak weighting,
diverse initialization, and line search. Combines time-domain peak reduction
with occasional spectral flattening insights.
"""

import numpy as np
import time


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    b = np.convolve(s, s)
    return 2.0 * n * b.max() / (total * total)


def compute_subgrad(s, w):
    """Compute subgradient: subgrad[j] = sum_k w[k] * s[k-j] for valid k-j."""
    n = len(s)
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    W = np.fft.rfft(w, N_fft)
    S = np.fft.rfft(s, N_fft)
    subgrad = np.fft.irfft(np.conj(S) * W, N_fft)[:n]
    return np.maximum(subgrad, 0.0)


def peak_eq_step(s, eta, beta, top_k_fraction=1.0):
    """One peak equalization step. Returns (new_s, max_b)."""
    n = len(s)
    b = np.convolve(s, s)
    max_b = b.max()
    if max_b < 1e-15:
        return s.copy(), max_b

    # Compute peak weights
    if top_k_fraction < 1.0:
        n_peaks = max(1, int(top_k_fraction * len(b)))
        threshold = np.partition(b, len(b) - n_peaks)[len(b) - n_peaks]
        w = np.exp(beta * (b - max_b) / max_b)
        w[b < threshold] *= 0.1
    else:
        w = np.exp(beta * (b - max_b) / max_b)

    # Compute subgradient
    subgrad = compute_subgrad(s, w)
    sg_max = subgrad.max()
    if sg_max < 1e-15:
        return s.copy(), max_b

    # Multiplicative update with line search over a few step sizes
    best_s = s.copy()
    best_score = fast_score(s)
    for trial_eta in [eta * 0.5, eta, eta * 2.0]:
        s_new = s * np.exp(-trial_eta * subgrad / sg_max)
        s_new = np.maximum(s_new, 0.0)
        s_sum = s_new.sum()
        if s_sum < 1e-15:
            continue
        s_new = s_new / s_sum
        sc = fast_score(s_new)
        if sc < best_score:
            best_score = sc
            best_s = s_new

    return best_s, max_b


def optimize_sequence(s, deadline, rng, verbose=False):
    """Optimize a sequence using adaptive peak equalization."""
    n = len(s)
    s = s.copy() / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)

    eta = 0.4
    beta = 3.0
    top_k_frac = 1.0
    no_improve = 0
    patience = 80
    it_count = 0

    phase = 0  # 0=explore, 1=exploit, 2=polish

    while time.monotonic() < deadline:
        it_count += 1

        # Adaptive phase transitions
        if no_improve > patience * 2:
            phase = 1
        if no_improve > patience * 4:
            phase = 2

        if phase == 0:
            # Exploration: moderate parameters
            eta = rng.uniform(0.2, 1.0)
            beta = rng.uniform(2.0, 8.0)
            top_k_frac = rng.choice([0.3, 0.5, 0.7, 1.0])
        elif phase == 1:

        else:

            eta = rng.uniform(0.05, 0.2)
            beta = rng.uniform(10.0, 50.0)

        score = fast_score(s_new)

        if score < best_score - 1e-12:
            best_score = score
            best_s = s_new.copy()
            no_improve = 0
        else:

    return best_s, best_score


def generate_candidates(n, rng, hseq=None):
    """Generate diverse initial candidates of length n."""
    candidates = []
    x = np.linspace(0.0, 1.0, n)

    # From previous best

        h_interp = np.maximum(h_interp, 0)
        if h_interp.sum() > 0.01:

            candidates.append((fast_score(h_interp), h_interp))

    # Uniform
    s = np.ones(n) / n
    candidates.append((fast_score(s), s.copy()))

    # Cosine window
    s = np.cos(np.pi * x) ** 2
    s = np.maximum(s, 0.01)
    s = s / s.sum()
    candidates.append((fast_score(s), s.copy()))

    # Triangle
    s = 1.0 - np.abs(2.0 * x - 1.0)
    s = s / s.sum()
    candidates.append((fast_score(s), s.copy()))

    # Gaussian bumps at various positions
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width in [0.08, 0.12, 0.15, 0.2]:
            s = np.exp(-((x - center) / width) ** 2) + 0.05
            s = s / s.sum()
            candidates.append((fast_score(s), s.copy()))

    for c1, c2 in [(0.25, 0.75), (0.3, 0.7), (0.2, 0.8)]:

            s = s / s.sum()
            candidates.append((fast_score(s), s.copy()))

    # Power law shapes
    for p in [0.3, 0.5, 0.7, 1.5, 2.0, 3.0]:
        s = (1.0 - x) ** p + 0.01

        candidates.append((fast_score(s), s.copy()))

        candidates.append((fast_score(s2), s2.copy()))

    # Periodic modulations

                s = np.maximum(s, 0.05)
                s = s / s.sum()
                candidates.append((fast_score(s), s.copy()))

    # Random smooth (low-pass filtered random)
    for _ in range(5):
        s = rng.random(n)
        # Simple smoothing
        for _ in range(3):
            s = 0.5 * s + 0.25 * np.roll(s, 1) + 0.25 * np.roll(s, -1)
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        candidates.append((fast_score(s), s.copy()))

    # Pure random
    for _ in range(3):
        s = rng.random(n) + 0.05
        s = s / s.sum()
        candidates.append((fast_score(s), s.copy()))

    candidates.sort(key=lambda t: t[0])
    return candidates


def symmetrize(s):
    """Try symmetrization: s_sym = (s + s[::-1]) / 2."""
    s_sym = (s + s[::-1]) / 2.0
    s_sym = np.maximum(s_sym, 0)
    if s_sym.sum() > 0.01:
        s_sym = s_sym / s_sym.sum()
    return s_sym


def refine_with_coord_descent(s, n_passes=10, deadline=None):
    """Transfer-mass coordinate descent: move mass from high-pressure to low-pressure positions."""
    n = len(s)
    s = s.copy() / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)

    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        for _ in range(n_passes):
            if deadline is not None and time.monotonic() > deadline:
                break
            b = np.convolve(s, s)
            max_b = b.max()

            # Compute "pressure" at each position (subgradient of max)
            w = np.exp(10.0 * (b - max_b) / max_b)
            subgrad = compute_subgrad(s, w)
            sg_max = subgrad.max()
            if sg_max < 1e-15:
                break

            # Transfer mass from high-pressure to low-pressure
            i_src = int(np.argmax(subgrad))
            i_dst = int(np.argmin(subgrad))
            if i_src == i_dst:
                break

            delta = 0.05 * s[i_src]
            s_new = s.copy()
            s_new[i_src] -= delta
            s_new[i_dst] += delta
            s_new = np.maximum(s_new, 0)
            s_new = s_new / s_new.sum()

            sc = fast_score(s_new)
            if sc < best_score:
                best_score = sc
                best_s = s_new.copy()
                s = s_new
            else:
                # Try smaller transfer
                delta = 0.01 * s[i_src]
                s_new = s.copy()
                s_new[i_src] -= delta
                s_new[i_dst] += delta
                s_new = np.maximum(s_new, 0)
                s_new = s_new / s_new.sum()
                sc = fast_score(s_new)
                if sc < best_score:
                    best_score = sc
                    best_s = s_new.copy()
                    s = s_new

    return best_s, best_score
