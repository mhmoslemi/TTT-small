    n = len(f)

    c_self = np.convolve(f, f)
    # Sort by value, take top max_constraints
    idx = np.argsort(-c_self)[:max_constraints]

"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA with softmax smoothing (fast O(n log n) gradient via FFT/correlate),
LP-based Frank-Wolfe direction finding, and diverse multi-n initializations.

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_gradient(c, w, y, n):
    """
    Compute gradient of softmax-smoothed max(conv(y,y)) w.r.t. y.
    grad[i] = 2*h[i] - w[2i]*y[i]
    where h[i] = sum_j y[j]*w[i+j]
    """
    # h[i] = sum_j y[j] * w[i+j] for valid j, i
    # This is the cross-correlation: np.correlate(w, y, 'full')[i]
    h = np.correlate(w, y, mode='full')
    h_sub = h[:n]

    # w[2i]*y[i] with bounds checking
    w_2i = np.zeros(n)
    valid = 2 * np.arange(n) < len(w)
    w_2i[valid] = w[2 * np.arange(n)[valid]] * y[np.arange(n)[valid]]

    grad = 2.0 * h_sub - w_2i
    return grad


def fista_smooth(a0, time_budget=10.0, beta=50.0, lr=None):

        ex = np.exp(x)

        else:
            w = ex / s

        step = lr

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=5.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def frank_wolfe_lp(a0, time_budget=20.0, t_init=0.5):
    """
    Frank-Wolfe with LP direction finding and line search.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    a = a / a.sum()

    t_start = time.monotonic()
    c = np.convolve(a, a)

    best_a = a.copy()

    no_improve = 0
    t_step = t_init

    # Time one LP to estimate budget
    t0 = time.monotonic()
    g_test = lp_direction(a, M, timeout=3.0)
    t_lp = max(time.monotonic() - t0, 0.001)

    max_iters = int((time_budget - (time.monotonic() - t_start)) / (t_lp + 0.005))
    max_iters = min(max_iters, 5000)

    for it in range(max_iters):
        if time.monotonic() - t_start > time_budget:
            break

        g = lp_direction(a, M, timeout=max(1.0, 0.3 * t_lp * 20))
        if g is None:
            break
        g = normalize(g)
        if g is None:
            break

        # Line search
        best_t = 0.0
        best_s = best_score
        candidates = [t_step, 0.5 * t_step, 0.25 * t_step, 0.75 * t_step, 0.1 * t_step, 0.9 * t_step]
        for t in candidates:
            a_new = (1.0 - t) * a + t * g
            a_new = normalize(a_new)
            if a_new is None:

            sc = internal_score(a_new)
            if sc < best_s:

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

def coordinate_polish(a, time_budget=5.0, step_sizes=None):
    """
    Greedy coordinate descent: try small perturbations of each coordinate.
    """
    n = len(a)
    a = a.copy()
    best_score = internal_score(a)
    t_start = time.monotonic()

    if step_sizes is None:
        step_sizes = [0.01, -0.01, 0.05, -0.05, 0.1, -0.1, 0.001, -0.001]

    improved = True
    while improved and time.monotonic() - t_start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - t_start > time_budget:
                break
            for ds in step_sizes:
                a_new = a.copy()
                a_new[i] = max(0.0, a_new[i] + ds)
                a_new = normalize(a_new)
                if a_new is None:
                    continue
                sc = internal_score(a_new)
                if sc < best_score - 1e-12:
                    a = a_new
                    best_score = sc
                    improved = True
                    break

    return a, best_score

        # Remove duplicates (similar inits)
        unique_inits = [inits[0]]
        for idx in range(1, len(inits)):
            is_dup = False
            for u in unique_inits:
                if np.linalg.norm(inits[idx] - u) < 0.05:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(inits[idx])
        inits = unique_inits

        n_init = len(inits)
        # Time budget per initialization
        time_left = deadline - time.monotonic()
        time_per = min(20.0, time_left / (n_init + 1))

        for ii, init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break

            t_rem = deadline - time.monotonic()
            # Allocate: FISTA gets 60%, LP gets 30%, polish gets 10%
            t_fista = min(time_per * 0.6, t_rem * 0.4)
            t_lp = min(time_per * 0.3, t_rem * 0.3)
            t_polish = min(time_per * 0.1, t_rem * 0.15)

            if t_fista < 1.0:
                break

            # Phase A: FISTA
            f, sc = fista_smooth(init, time_budget=t_fista, beta=50.0)
            if sc < best_score:
                best_score = sc
                best_seq = f.copy()
                print(f"  n={n}, init {ii}: FISTA {sc:.8f}")

            # Also try different beta
            if time.monotonic() < deadline - 30:
                f2, sc2 = fista_smooth(f, time_budget=min(3.0, t_rem * 0.15), beta=20.0)
                if sc2 < best_score:
                    best_score = sc2
                    best_seq = f2.copy()
                    print(f"  n={n}, init {ii}: FISTA-beta20 {sc2:.8f}")

            # Phase B: LP Frank-Wolfe (only for promising starts)
            if time.monotonic() < deadline - 30 and t_lp > 2.0:
                f3, sc3 = frank_wolfe_lp(f, time_budget=t_lp)
                if sc3 < best_score:
                    best_score = sc3
                    best_seq = f3.copy()
                    print(f"  n={n}, init {ii}: LP-FW {sc3:.8f}")

            # Phase C: Coordinate polish
            if time.monotonic() < deadline - 25 and t_polish > 1.0:
                f4, sc4 = coordinate_polish(f, time_budget=t_polish,
                                            step_sizes=[0.005, -0.005, 0.02, -0.02, 0.01, -0.01])
                if sc4 < best_score:
                    best_score = sc4
                    best_seq = f4.copy()
                    print(f"  n={n}, init {ii}: polish {sc4:.8f}")

    # Final refinement phase: take best and run longer optimization
    if time.monotonic() < deadline - 20 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Final refinement at n={n_best}, score={best_score:.8f} ===")

        t_rem = deadline - time.monotonic() - 10
        # FISTA with annealed beta
        for beta in [100, 200, 500]:
            if time.monotonic() > deadline - 15:
                break
            t_budget = min(10.0, (deadline - time.monotonic() - 15) / 3.0)
            if t_budget < 2:
                break
            f, sc = fista_smooth(best_seq, time_budget=t_budget, beta=float(beta))
            if sc < best_score:
                best_score = sc
                best_seq = f.copy()
                print(f"  Final FISTA beta={beta}: {sc:.8f}")

        # LP polish
        if time.monotonic() < deadline - 20:
            t_budget = min(30.0, deadline - time.monotonic() - 15)
            if t_budget > 5:
                f, sc = frank_wolfe_lp(best_seq, time_budget=t_budget, t_init=0.3)
                if sc < best_score:
                    best_score = sc
                    best_seq = f.copy()
                    print(f"  Final LP-FW: {sc:.8f}")

        # Coordinate polish
        if time.monotonic() < deadline - 15:
            t_budget = min(10.0, deadline - time.monotonic() - 10)
            if t_budget > 2:
                f, sc = coordinate_polish(best_seq, time_budget=t_budget,
                                         step_sizes=[0.002, -0.002, 0.005, -0.005, 0.01, -0.01,
                                                     0.001, -0.001])
                if sc < best_score:
                    best_score = sc
                    best_seq = f.copy()
                    print(f"  Final polish: {sc:.8f}")

    # Final validation
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        # Fallback
        n_fallback = 256
        t = np.linspace(0, 1, n_fallback)
        best_seq = np.maximum(1.0 + 1.5 * (2 * t - 1), 1e-10)
        best_seq = best_seq / best_seq.sum()
        s = best_seq.sum()

    best_seq = best_seq / s
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    return [float(x) for x in best_seq]
