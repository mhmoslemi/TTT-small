"""
Minimize 2n*max(conv(a,a))/sum(a)^2 using adaptive soft-max gradient descent.
Key improvements: top-k focus, Nesterov momentum, cosine LR, multi-scale search.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c

def _gradient_from_weights(w, a):
    """Compute sum_k w[k] * dc[k]/da using FFT cross-correlation."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    return h[n - 1:2 * n - 1]

def _softmax_weights_topk(c, beta, k=None):
    """Compute soft-max weights, optionally only on top-k values."""
    if k is None or k >= c.size:
        k = c.size
    # Find top-k indices
    if k < c.size:
        topk_idx = np.argpartition(c, -k)[-k:]
    else:
        topk_idx = np.arange(c.size)
    
    c_topk = c[topk_idx]
    mval = c_topk.max()
    x = (c_topk - mval) / beta
    ex = np.exp(x)
    s = ex.sum()
    if s <= 0 or not np.isfinite(s):
        ex = np.zeros_like(c_topk)
        ex[topk_idx[np.argmax(c_topk)]] = 1.0
        s = 1.0
    w_topk = ex / s
    
    # Place back
    w = np.zeros(c.size)
    w[topk_idx] = w_topk
    return w, mval + beta * np.log(s)

- c = conv(a, a)
- f = max(c) ≈ soft_max(c, beta)

Where ∂a_j/∂z_i = a_j * (δ_{ij} - a_i) (the Jacobian of softmax)

1. Compute c = conv(a, a) using FFT
2. Compute weights w[k] from soft-max of c
3. Compute g_a[j] = sum_k w[k] * ∂c_k/∂a_j (using FFT cross-correlation)
4. Compute g_z[i] = sum_j g_a[j] * a_j * (δ_{ij} - a_i) = a_i * (g_a[i] - sum_j g_a[j] * a_j)

This is O(n log n) per gradient evaluation, which is very fast.

"""
Homotopy L-BFGS for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive soft-max smoothing, multi-restart strategy, and diverse initializations.
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / z.size
    return e / s


def _soft_max_smooth(c, beta):
    """Compute soft-max value and weights for array c with temperature beta."""
    mval = c.max()
    if beta < 1e-30:
        # Hard max
        w = np.zeros_like(c)
        idx = np.argmax(c)
        w[idx] = 1.0
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = ex.sum()
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w


def _corr_grad(w, a):
    """Compute sum_k w[k] * dc[k]/da[i] using FFT."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    return h[n - 1:2 * n - 1]


def _smooth_objective(z, beta, n):
    """Objective and gradient in logit space. Returns (value, gradient)."""
    a = _softmax_from_z(z)
    c = _convolve_fft(a)
    val, w = _soft_max_smooth(c, beta)
    # Gradient of soft-max value w.r.t. a
    ga = _corr_grad(w, a)
    # Gradient w.r.t. z: gz[i] = a[i] * (ga[i] - sum_j ga[j]*a[j])
    dot = ga @ a
    gz = a * (ga - dot)
    # Scale by 2n (constant factor, doesn't affect optimization)
    return 2.0 * n * val, 2.0 * n * gz


def _internal_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2 for normalized a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = a.sum()
    if s <= 0:
        return np.inf
    c = _convolve_fft(a)
    mval = c.max()
    return 2.0 * n * mval / (s * s)


def _true_score(a):
    """Exact score using np.convolve."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 0:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return 2.0 * n * mval / (s * s)
