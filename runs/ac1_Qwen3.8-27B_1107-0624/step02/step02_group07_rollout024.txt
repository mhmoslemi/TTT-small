import numpy as np
import time
from scipy.optimize import linprog

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def lp_step(s_fixed, n):
    """Solve LP: min M s.t. conv(s, s_fixed)[k] <= M, s >= 0, sum(s) = 1.
    Returns optimal s and M."""
    # Build constraint matrix
    # A[k, i] = s_fixed[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                A[k, i] = s_fixed[idx]
        A[k, n] = -1  # coefficient of M
    
    c = np.zeros(n+1)
    c[n] = 1  # minimize M
    
    b_ub = np.ones(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        s_opt = res.x[:n]
        M_opt = res.x[n]
        return s_opt, M_opt
    return None, None

    best_score = fast_score(s)

def spectral_refine(s, n, time_limit=30.0):
    """Use FFT to flatten the spectrum (and hence the autocorrelation)."""
    s = s.copy()
    s = np.maximum(s, 0) / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)
    start_time = time.monotonic()
    
    beta = 0.5  # How much to flatten
    N_fft = 1
    while N_fft < 4*n:
        N_fft *= 2
    
    while time.monotonic() - start_time < time_limit:
        S = np.fft.rfft(s, N_fft)
        mag = np.abs(S)
        
        # Flatten: S_new = S * (1/mag)^beta = S / mag^beta
        # This reduces peaks in the magnitude
        mag_safe = np.maximum(mag, 1e-10)
        S_new = S / np.power(mag_safe, beta)
        
        s_new = np.fft.irfft(S_new, N_fft)[:n]

        s_new = s_new / s_sum

        if score < best_score:
            best_score = score
            best_s = s_new.copy()
        
        s = s_new
        
        # Decrease beta over time (less aggressive flattening)
        beta *= 0.99
    
    return best_s, best_score
