`x = (2*conv0[k] - delta_k*S) / (2*quad_k*S - delta_k)`

    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def coord_sweep(sweep, deadline=None, verbose=False, top_k=15, grid_size=25):
    """
    One full coordinate descent sweep (forward + backward).
    Uses incremental convolution updates.
    Returns (sweep, score).
    """
    n = len(sweep)
    conv = np.convolve(sweep, sweep)
    total_sum = sweep.sum()

    for j in range(n):
        if deadline is not None and time.monotonic() > deadline:
            break
        x_old = sweep[j]
        S = total_sum - x_old

        if S < 1e-30:
            continue

        # Compute delta and quad for this coordinate
        # conv_new[k] = conv[k] + delta[k]*(x-x_old) + quad[k]*(x^2-x_old^2)
        delta = np.zeros(2 * n - 1)
        quad = np.zeros(2 * n - 1)

        k_lo = j
        k_hi = j + n - 1
        for k in range(k_lo, k_hi + 1):
            idx = k - j
            if k == 2 * j:
                quad[k] = 1.0
            else:
                delta[k] = 2.0 * sweep[idx]

        # Build candidates
        candidates = [x_old]
        if x_old > 1e-8:
            candidates.append(x_old * 0.5)
            candidates.append(x_old * 2.0)
        candidates.append(0.0)

        # Grid
        x_max = max(2.0 * x_old, S * 0.3, 1e-8)
        if x_old > 1e-10:
            grid = np.geomspace(1e-12, x_max, grid_size)
        else:
            grid = np.linspace(0.0, x_max, grid_size)
        candidates.extend(grid.tolist())

        # Analytical candidates from top-k convolution positions
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
        for k in top_indices:
            A_k = conv[k] - delta[k] * x_old - quad[k] * x_old ** 2
            denom = 2.0 * quad[k] * S - delta[k]
            if abs(denom) > 1e-15:
                x_cand = (2.0 * A_k - delta[k] * S) / denom
                if 0.0 < x_cand < 1e6:
                    candidates.append(x_cand)

        candidates = np.array(candidates)
        candidates = candidates[(candidates >= 0) & (candidates < 1e6)]
        if len(candidates) == 0:
            continue

        # Evaluate f(x) = 2n * max_k conv_new[k] / (S+x)^2 at all candidates
        x = candidates[:, None]
        conv_new = conv[None, :] + delta[None, :] * (x - x_old) + quad[None, :] * (x**2 - x_old**2)
        max_vals = conv_new.max(axis=1)
        f_vals = 2.0 * n * max_vals / (S + candidates) ** 2

        idx = np.argmin(f_vals)
        x_new = candidates[idx]

        if abs(x_new - x_old) > 1e-14:
            # Update convolution incrementally
            conv[j:j + n] += delta[j:j + n] * (x_new - x_old) + quad[j:j + n] * (x_new**2 - x_old**2)
            sweep[j] = x_new
            total_sum += (x_new - x_old)

    # Backward pass
    for j in range(n - 1, -1, -1):
        if deadline is not None and time.monotonic() > deadline:
            break
        x_old = sweep[j]
        S = total_sum - x_old

        if S < 1e-30:
            continue

        delta = np.zeros(2 * n - 1)
        quad = np.zeros(2 * n - 1)
        for k in range(j, j + n):
            idx = k - j
            if k == 2 * j:
                quad[k] = 1.0
            else:
                delta[k] = 2.0 * sweep[idx]

        candidates = [x_old]
        if x_old > 1e-8:
            candidates.append(x_old * 0.5)
            candidates.append(x_old * 1.5)
        candidates.append(0.0)

        x_max = max(2.0 * x_old, S * 0.3, 1e-8)
        if x_old > 1e-10:
            grid = np.geomspace(1e-12, x_max, grid_size)
        else:
            grid = np.linspace(0.0, x_max, grid_size)
        candidates.extend(grid.tolist())

        top_indices = np.argpartition(conv, -top_k)[-top_k:]
        for k in top_indices:
            A_k = conv[k] - delta[k] * x_old - quad[k] * x_old ** 2
            denom = 2.0 * quad[k] * S - delta[k]
            if abs(denom) > 1e-15:
                x_cand = (2.0 * A_k - delta[k] * S) / denom
                if 0.0 < x_cand < 1e6:
                    candidates.append(x_cand)

        candidates = np.array(candidates)
        candidates = candidates[(candidates >= 0) & (candidates < 1e6)]
        if len(candidates) == 0:
            continue

        x = candidates[:, None]
        conv_new = conv[None, :] + delta[None, :] * (x - x_old) + quad[None, :] * (x**2 - x_old**2)
        max_vals = conv_new.max(axis=1)
        f_vals = 2.0 * n * max_vals / (S + candidates) ** 2

        idx = np.argmin(f_vals)
        x_new = candidates[idx]

        if abs(x_new - x_old) > 1e-14:
            conv[j:j + n] += delta[j:j + n] * (x_new - x_old) + quad[j:j + n] * (x_new**2 - x_old**2)
            sweep[j] = x_new
            total_sum += (x_new - x_old)

    score = internal_score(sweep)
    return sweep, score


def coordinate_descent(sweep, num_sweeps=100, deadline=None, verbose=False, top_k=15, grid_size=25):
    """
    Run coordinate descent for num_sweeps full sweeps.
    Returns (best_sweep, best_score).
    """
    n = len(sweep)
    sweep = sweep.astype(np.float64).copy()
    best_score = internal_score(sweep)
    best_sweep = sweep.copy()
    no_improve = 0

    for s in range(num_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        sweep, cur_score = coord_sweep(sweep, deadline=deadline, top_k=top_k, grid_size=grid_size)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_sweep = sweep.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and s % 50 == 0:
            print(f"    CD sweep {s}: best={best_score:.10f}, current={cur_score:.10f}")

        if no_improve > 50:
            # Slight perturbation to escape local minima
            rng = np.random.default_rng(s)
            pert = sweep * (1.0 + 0.02 * rng.normal(size=n))
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            # Do a few more sweeps from perturbed point
            for _ in range(5):
                if deadline is not None and time.monotonic() > deadline:
                    break
                pert, p_score = coord_sweep(pert, deadline=deadline, top_k=top_k, grid_size=grid_size)
            if p_score < best_score:
                best_score = p_score
                best_sweep = pert.copy()
                sweep = pert.copy()
            else:
                sweep = best_sweep.copy()
            no_improve = 0

    return best_sweep, best_score


def adam_polish(p0, deadline, lr=0.05, verbose=True):
    """
    Adam gradient descent in logit space to polish a non-negative sequence.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.1 * max_c)

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Soft argmax weights
        x = (c - max_c) / beta
        x_max = x.max()
        ex = np.exp(x - x_max)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_ex

        # Gradient: d(max_c)/d(p0[i]) = 2 * sum_k w[k] * p0[k-i] (with valid ranges)
        # conv(p0,p0)[k] = sum_i p0[i]*p0[k-i]
        # d(conv[k])/d(p0[j]) = p0[k-j] (if 0<=k-j<=n-1) + p0[k-j] (if k-j!=j, same term)
        # Actually: d(conv[k])/d(p0[j]) = 2*p0[k-j] if 0<=k-j<=n-1 and k!=2j, or 2*p0[j] if k=2j
        # So d(max_c)/d(p0[j]) = sum_k w[k] * (2*p0[k-j] if 0<=k-j<=n-1 and k!=2j, or 2*p0[j] if k=2j)
        h = np.zeros(n)
        for j in range(n):
            # k ranges from j to j+n-1
            k_vals = np.arange(j, j + n)
            # conv[k] involves p0[j]*p0[k-j], so d(conv[k])/d(p0[j]) = p0[k-j] + p0[k-j] = 2*p0[k-j]
            # (with the caveat that if k-j==j, it's p0[j]*p0[j], d/dp0[j] = 2*p0[j])
            # In both cases it's 2*p0[k-j]
            h[j] = 2.0 * np.dot(w[j:j + n], p0)

        grad_p = h
        # Normalize constraint: project gradient to tangent space of sum(p)=1
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 100 == 0:
            beta = max(1e-14, beta * 0.7)
            lr = base_lr * max(0.1, 1.0 - 0.001 * it)

        if no_imp > 1000:
            # Restart with perturbation
            rng = np.random.default_rng(it + 42)
            z = 0.9 * best_z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-12, 0.1 * float(np.convolve(
                np.exp(z - z.max()), np.exp(z - z.max())).max()))

        p0 = p

    return best_p, best_score


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized to sum 1."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_quadratic_profile(n, a1, a2):
    """Create a quadratic profile 1 + a1*u + a2*u^2, u=2t-1, normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    arr = 1.0 + a1 * u + a2 * u ** 2
    arr = np.maximum(arr, 1e-10)
    return arr / arr.sum()


def propose_candidate():
    """
    Multi-resolution coordinate descent + Adam polish for minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
      1. Initialize from known best or analytic profiles.
      2. Coordinate descent at n=128 (fast, many sweeps).
      3. Interpolate to n=256, coordinate descent.
      4. Interpolate to n=512, coordinate descent (many sweeps).
      5. Adam gradient polish at n=512.
      6. More coordinate descent + Adam interleave at n=512.
      7. Try n=1024 if time permits.
    """
    t_start = time.monotonic()
    budget = 1800.0
    deadline = t_start + budget - 10.0
    rng = np.random.default_rng(12345)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                best_seq = parent / parent.sum()
                best_score = internal_score(best_seq)
                print(f"Parent: n={len(best_seq)}, score={best_score:.10f}")
    except Exception:
        pass

    # Generate diverse initializations
    n_init = 128
    profiles = [
        make_linear_profile(n_init, 1.5),
        make_linear_profile(n_init, 2.0),
        make_linear_profile(n_init, 2.5),
        make_quadratic_profile(n_init, 1.5, -0.5),
        make_quadratic_profile(n_init, 2.0, -0.8),

    for p in profiles:

    print(f"Best init: n={len(best_seq)}, score={best_score:.10f}")

                                  verbose=True, top_k=15, grid_size=30)
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2
        print(f"  Phase 2 result: {best_score:.10f}")

    # Phase 3: Coordinate descent at n=512
    deadline_p3 = min(deadline, t_start + 700.0)

    a3 = interpolate_seq(best_seq, n3)

    # Phase 7: Final Adam polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\nPhase 7: Final Adam polish, remaining={remaining:.0f}s")
        p7, sc7 = adam_polish(best_seq, deadline - 5, lr=0.02, verbose=True)
        if sc7 < best_score:
            best_score = sc7
            best_seq = p7
            print(f"  Phase 7 result: {best_score:.10f}")

    # Phase 8: One more CD pass
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a8, sc8 = coordinate_descent(best_seq, num_sweeps=100, deadline=deadline - 2,
                                      verbose=False, top_k=20, grid_size=40)
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8
            print(f"  Phase 8 result: {best_score:.10f}")

    final_score = internal_score(best_seq)

    # Ensure valid output

        best_seq = make_linear_profile(256, 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
