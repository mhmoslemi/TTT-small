from scipy.optimize import linprog
import time

def conv_matrix(n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[k, lo:hi+1] = 1.0
    return A

def solve_lp(n, time_limit=60):
    """Solve the direct LP to find optimal M for given n."""
    A = conv_matrix(n)
    
    # Variables: a_0, ..., a_{n-1}, M
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Inequality constraints: A@a - M <= 0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality constraint: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    
    if res.success:
        M_opt = res.x[-1]
        a_opt = res.x[:n]
        score = 2.0 * n * M_opt
        return a_opt, M_opt, score
    return None, None, None

Direct LP solution for minimizing 2*n*max(conv(a,a))/sum(a)^2.
The key insight is that after normalizing sum(a)=1, the problem becomes:
    minimize M subject to sum(a)=1, a>=0, conv(a,a)[k]<=M for all k
This is a linear program! We solve it for multiple values of n and return
the sequence achieving the best score.
"""

import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(n):
    """Build (2n-1) x n binary matrix where A[k,j]=1 iff j<=k<=j+n-1."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[k, lo:hi + 1] = 1.0
    return A


def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def solve_lp_for_n(n, time_limit=30.0):
    """
    Solve: minimize M s.t. sum(a)=1, a>=0, conv(a,a)[k]<=M for all k.
    Returns (a_optimal, score) where score = 2*n*M_optimal.
    """
    A = build_conv_matrix(n)

    # Variables: [a_0, ..., a_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality: A@a - M <= 0  =>  [A | -1] @ [a; M] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * (n + 1)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True,
                           'primal_feasibility_tolerance': 1e-12,
                           'dual_feasibility_tolerance': 1e-12})

    if res.success and res.x is not None:
        a = np.maximum(res.x[:n], 0.0)
        M = float(res.x[-1])
        score_lp = 2.0 * n * M
        # Verify with actual computation
        actual = internal_score(a)
        score = min(score_lp, actual)
        return a, score
    elif hasattr(res, 'x') and res.x is not None:
        # Even if not flagged as success, check if we got a feasible solution
        a = np.maximum(res.x[:n], 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            score = internal_score(a)
            return a, score
    return None, np.inf


def adam_polish(p0, deadline, lr=0.02, verbose=False):
    """Adam gradient descent in logit space for final polish."""
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    beta = max(1e-12, 0.1 * max_c)

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        x = (c - max_c) / beta
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_ex

        # Gradient: grad_p[i] = 2 * sum_k w[k] * p[k-i] (with valid indices)
        h = np.zeros(n)
        for i in range(n):
            lo = max(0, i - n + 1)
            hi = min(i + 1, 2 * n - 1)
            h[i] = np.dot(w[lo:hi], p0[lo - (i - n + 1) if i - n + 1 > 0 else 0:
                          hi - (i - n + 1) if i - n + 1 > 0 else hi])
        # Simpler: use correlation
        h = np.correlate(w, p0, mode='full')
        # Take the n elements centered
        h = h[n - 1:n - 1 + n]

        grad_p = 2.0 * h

        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_p.copy()
            z = np.log(np.maximum(z, 1e-12))
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            continue
        p = p / p_sum

        score = 2.0 * n * float(np.convolve(p, p).max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 500 == 0:
            beta = max(1e-14, beta * 0.5)

        p0 = p

    return best_p, best_score
