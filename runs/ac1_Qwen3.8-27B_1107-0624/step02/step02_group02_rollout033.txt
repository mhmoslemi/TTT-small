"""
Multi-scale soft-max Adam optimization with chirp initialization for minimizing
2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses Nesterov momentum, adaptive beta annealing, and LP-based escape.
"""
import numpy as np
import time
from scipy.optimize import linprog

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c

def soft_max_compute(c, beta):
    """Compute soft-max value and weights."""
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def cross_corr_fft(w, a):
    """Compute h[i] = sum_j w[i+j] * a_j for i=0..len(a)-1."""
    n = len(a)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e / float(e.sum())
    return p

    Nesterov soft-max Adam optimization.
    Optimizes over log-space with Nesterov momentum and adaptive beta.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-12)
    a0 = a0 / a0.sum()
    n = a0.size
    
    # Initialize in log-space
    z = np.log(a0 + 1e-15)
    z -= float(np.mean(z))
    
    # Adam parameters
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    
    # Beta schedule
    c = full_convolve_fft(a0)
    maxc = float(c.max())
    if beta_init is None:
        beta = 0.3 * maxc
    else:
        beta = beta_init
    
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    beta_decay = 0.95  # Decay beta every 50 steps
    step_count = 0
    
    while time.monotonic() < deadline:
        step_count += 1
        
        # Nesterov look-ahead
        z_nest = z + nesterov_m * m / (1.0 - b1) if t > 0 else z
        a_nest = softmax_from_z(z_nest)
        
        # Compute soft-max on the Nesterov point
        c_nest = full_convolve_fft(a_nest)
        L, w = soft_max_compute(c_nest, beta)
        
        # Gradient of soft-max w.r.t. a
        grad_a = 2.0 * cross_corr_fft(w, a_nest)
        
        # Gradient w.r.t. z (log-space)
        dot = float(np.dot(a_nest, grad_a))
        grad_z = a_nest * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))  # Remove constant component
        
        # Adam update
        t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)

            m.fill(0.0)
            v.fill(0.0)

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_step(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    if n > 2048:
        return None  # Too large for LP
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def lp_optimize(f_init, deadline, rng):
    """LP-based optimization with restart."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 1e-12).copy()
    f = f / f.sum()
    n = len(f)
    
    conv_f = full_convolve_fft(f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.2
    stagnant = 0
    
    while time.monotonic() < deadline - 5:
        M = float(full_convolve_fft(f).max())
        
        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 1e-12)
        f_new = f_new / f_new.sum()
        
        M_new = float(full_convolve_fft(f_new).max())
        
        if M_new < best_M - 1e-12:
            f = f_new
            best_M = M_new
            best_f = f.copy()
            stagnant = 0
        elif M_new <= M + 1e-10:
            f = f_new
            stagnant += 1
        else:
            t *= 0.7
            stagnant += 1
        
        t = max(0.01, t * 0.999)
        
        if stagnant > 100:
            noise = 0.05 * rng.normal(size=n)
            f = best_f + noise
            f = np.maximum(f, 1e-12)
            f = f / f.sum()
            t = 0.2
            stagnant = 0
    
    return best_f, float(2.0 * n * full_convolve_fft(best_f).max())

def chirp_profile(n, delta=1.0):
    """Quadratic phase (chirp) profile, non-negative."""
    t = np.arange(n, dtype=np.float64)
    # Quadratic chirp: cos(pi * delta * (t/n)^2)
    a = 1.0 + 0.5 * np.cos(np.pi * delta * (t / max(n - 1, 1)) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def windowed_chirp_profile(n, delta=1.0, window='hann'):
    """Windowed chirp profile."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * np.cos(np.pi * delta * (t / max(n - 1, 1)) ** 2)
    if window == 'hann':
        w = 0.5 * (1 + np.cos(2 * np.pi * t / max(n - 1, 1)))
        a = a * (0.5 + 0.5 * w)
    elif window == 'hamming':
        w = 0.54 + 0.46 * np.cos(2 * np.pi * t / max(n - 1, 1))
        a = a * (0.5 + 0.5 * w)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def cosine_profile(n, k=1, phase=0):
    """Cosine profile."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.4 * np.cos(2 * np.pi * k * t / n + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def linear_profile(n, eta=1.0):
    """Linear profile."""
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p)
    result = np.maximum(result, 1e-12)
    return result / result.sum()

"""
Multi-scale optimization for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses chirp-based initialization, Nesterov soft-max Adam with adaptive beta, L-BFGS-B refinement,
LP-based escape, and multi-resolution upscaling.

import numpy as np
import time
from scipy.optimize import linprog, minimize as scipy_minimize


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of two >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max_compute(c, beta):
    """Compute soft-max value and weights efficiently."""
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def cross_corr_fft(w, a):
    """Compute h[i] = sum_j w[i+j] * a_j for i=0..len(a)-1."""
    n = len(a)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z) / z.size
    return e / s


def nesterov_adam_optimize(a0, deadline, rng, lr=0.08, beta_init=None, beta_decay=0.93,
                           nesterov_m=0.9, restart_interval=800, verbose=False):
    """
    Nesterov soft-max Adam optimization in log-space.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-12)
    a0 = a0 / a0.sum()
    n = a0.size

    z = np.log(a0 + 1e-15)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    c = full_convolve_fft(a0)
    maxc = float(c.max())
    if beta_init is None:
        beta = 0.4 * maxc
    else:
        beta = float(beta_init)

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    step_count = 0
    nv_buf = np.zeros(n, dtype=np.float64)

    while time.monotonic() < deadline - 0.5:
        step_count += 1

        # Nesterov look-ahead
        if t > 0:
            nv_buf = nesterov_m * m / (1.0 - b1)
            z_nest = z + nv_buf
        else:
            z_nest = z
        a_nest = softmax_from_z(z_nest)

        # Compute soft-max
        c_nest = full_convolve_fft(a_nest)
        L, w = soft_max_compute(c_nest, beta)

        # Gradient of soft-max w.r.t. a_nest
        grad_a = 2.0 * cross_corr_fft(w, a_nest)

        # Gradient w.r.t. z
        dot = float(np.dot(a_nest, grad_a))
        grad_z = a_nest * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        # Adam update
        t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue
        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))

        a = softmax_from_z(z)
        c = full_convolve_fft(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        # Beta annealing
        if step_count % 30 == 0:
            beta = max(1e-10, beta * beta_decay)

        # Restart if stuck
        if no_imp > restart_interval:
            scale = 0.05 + 0.1 * (step_count % 5) * 0.1
            z = best_z + scale * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            a = softmax_from_z(z)
            c = full_convolve_fft(a)
            beta = max(1e-10, 0.3 * float(c.max()))
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_imp = 0
            nesterov_m = min(0.99, nesterov_m + 0.01)
            if verbose:
                print(f"    Restart at step {step_count}, beta={beta:.2e}, score={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(f):
    """Build (2n-1) x n matrix where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    if n > 2048:
        return None
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 8.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, verbose=False):
    """LP-based optimization loop."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 1e-12).copy()
    f = f / f.sum()
    n = len(f)

    conv_f = full_convolve_fft(f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    alpha = 0.2
    stagnant = 0

    while time.monotonic() < deadline - 3:
        M = float(full_convolve_fft(f).max())

        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 1e-12)
        f_new = f_new / f_new.sum()

        M_new = float(full_convolve_fft(f_new).max())

        if M_new < best_M - 1e-12:
            f = f_new
            best_M = M_new
            best_f = f.copy()
            stagnant = 0
            if verbose and stagnant == 0:
                print(f"    LP improved: M={best_M:.10f}")
        elif M_new <= M + 1e-10:
            f = f_new
            stagnant += 1
        else:
            alpha *= 0.7
            stagnant += 1

        alpha = max(0.01, alpha * 0.999)

        if stagnant > 80:
            noise = 0.03 * rng.normal(size=n)
            f = best_f + noise
            f = np.maximum(f, 1e-12)
            f = f / f.sum()
            alpha = 0.2
            stagnant = 0

    return best_f, float(2.0 * n * full_convolve_fft(best_f).max())


def chirp_profile(n, delta=1.0):
    """Quadratic chirp (non-negative)."""
    t = np.arange(n, dtype=np.float64)
    x = t / max(n - 1, 1)
    a = 1.0 + 0.5 * np.cos(np.pi * delta * x * x)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def windowed_chirp_profile(n, delta=1.0, window='hann'):
    """Windowed chirp profile."""
    t = np.arange(n, dtype=np.float64)
    x = t / max(n - 1, 1)
    a = 1.0 + 0.5 * np.cos(np.pi * delta * x * x)
    if window == 'hann':
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a = a * (0.3 + 0.7 * w)
    elif window == 'hamming':
        w = 0.54 + 0.46 * np.cos(2.0 * np.pi * x)
        a = a * (0.3 + 0.7 * w)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def cosine_profile(n, k=1, phase=0.0, amp=0.4):
    """Cosine profile."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def linear_profile(n, eta=1.5):
    """Linear profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def quadratic_profile(n, c1=0.0, c2=0.5):
    """Quadratic profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + c1 * t + c2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def random_fourier_profile(n, rng, n_coeffs=4, max_amp=0.4):
    """Random smooth Fourier profile."""
    coeffs = rng.uniform(-max_amp, max_amp, n_coeffs)
    phases = rng.uniform(0, 2 * np.pi, n_coeffs)
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for i, (c, p) in enumerate(zip(coeffs, phases), 1):
        a += c * np.cos(2.0 * np.pi * i * t / n + p)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p, n_new):
    """Linear interpolation of a profile to a new length."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        return p / s if s > 1e-10 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p)
    result = np.maximum(result, 1e-12)
    return result / result.sum()

        a = np.ones(n, dtype=np.float64)
        for k in range(n_coeffs):
            a += c[k] * np.cos(2.0 * np.pi * (k + 1) * t / n + p[k])
        a = np.maximum(a, 1e-12)
        return a / a.sum()

    def objective(params):
        a = param_sequence(params)
        c = full_convolve_fft(a)

            options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-6}

    print(f"Phase 3 best: {best_score:.10f}")

    # ============ Phase 4: Try n=4096 if time allows ============
    n3 = 4096
    if best_seq is not None and time.monotonic() < deadline - 120 and best_score > 1.503:
        print(f"\n--- Phase 4: Upscale to n={n3} ---")
        p4096 = interpolate_profile(best_seq, n3)
        sc = compute_score(p4096)
        update(p4096, sc)
        print(f"  Interpolated to {n3}: {sc:.10f}")

        per = min(deadline - 30, time.monotonic() + 200.0)
        result, sc = nesterov_adam_optimize(p4096, per, rng, lr=0.05, beta_decay=0.95)
        update(result, sc)
        print(f"  Adam at {n3}: {sc:.10f}")

    # ============ Phase 5: Final polish ============
    if best_seq is not None and time.monotonic() < deadline - 3:
        n_final = best_seq.size
        per = min(deadline - 3, time.monotonic() + 40.0)
        p0 = best_seq + 0.0003 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, sc = nesterov_adam_optimize(p0, per, rng, lr=0.015, beta_decay=0.98)
        update(result, sc)
        print(f"\nFinal polish: {sc:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"\n=== FINAL: n={final_seq.size}, score={final_score:.10f} ===")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        else:
            print(f"evaluate_sequence returned non-finite: {eval_score}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
