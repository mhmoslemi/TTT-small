        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _gradient_smooth_max(a, beta, top_k=None):
    """Compute gradient of smooth max of convolution using FFT-based correlation."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    
    if top_k is not None and top_k < len(c):
        # Use top-k weighted approach for sharper signal
        sorted_idx = np.argsort(c)[::-1]
        top_indices = sorted_idx[:top_k]
        w = np.zeros_like(c)
        top_vals = c[top_indices]
        # Softmax over top-k only
        t = (top_vals - top_vals.max()) / beta
        ex = np.exp(t)
        s = float(ex.sum())
        if s <= 0 or not np.isfinite(s):
            w[top_indices[0]] = 1.0
        else:
            w[top_indices] = ex / s
    else:
        x = (c - mval) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
    
    # Compute gradient: grad[k] = 2 * sum_m w[m] * a[m-k] for valid indices
    # Using cross-correlation: h[k] = sum_m w[m] * a[m-k]
    # np.correlate(w, a, 'full')[n-1+k] = sum_m w[m] * a[m-k]
    h_full = np.correlate(w, a, mode='full')
    h = h_full[n - 1: 2 * n - 1]
    grad = 2.0 * h
    
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        return res.x
    return None


def _fista_optimize(a, deadline, max_iter=50000, initial_lr=None, top_k_init=None):
    """FISTA (accelerated projected gradient) with adaptive step size and top-k weighting."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    c = _fft_convolve(a)
    M_cur = float(c.max())
    beta = max(1e-10, 0.05 * M_cur)
    
    if initial_lr is None:
        lr = 0.1 / (2.0 * n * 1.0)  # Conservative initial
    else:
        lr = initial_lr
    
    if top_k_init is None:
        top_k = max(5, n // 16)
    else:
        top_k = top_k_init
    
    # FISTA variables
    y = a.copy()
    t = 1.0
    v_mom = np.zeros(n)  # Not used for standard FISTA, kept for hybrid
    
    no_imp = 0
    base_lr = lr
    
    # Adaptive Lipschitz estimate
    L = 2.0 * n  # Rough initial Lipschitz constant
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        grad_y, mval_y = _gradient_smooth_max(y, beta, top_k)
        
        # Adaptive step size (backtracking)
        step = 1.0 / L
        a_new = _project_simplex(y - step * grad_y)
        
        # Backtracking (max 5 iterations)
        for _ in range(5):
            c_new = _fft_convolve(a_new)
            s_new = float(a_new.sum())
            score_new = 2.0 * n * float(c_new.max()) / (s_new * s_new) if s_new > 1e-10 else np.inf
            c_cur = _fft_convolve(y)
            s_cur = float(y.sum())
            score_cur = 2.0 * n * float(c_cur.max()) / (s_cur * s_cur) if s_cur > 1e-10 else np.inf
            
            # Check sufficient decrease
            if score_new <= score_cur + 1e-10:
                break
            step *= 0.5
            a_new = _project_simplex(y - step * grad_y)
            L = max(L * 1.5, 1e-20)
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = float(y.sum())
        if s_y > 1e-30:
            y = y / s_y
        t = t_new
        a = a_new
        
        # Evaluate
        c_a = _fft_convolve(a)
        s_a = float(a.sum())
        score_a = 2.0 * n * float(c_a.max()) / (s_a * s_a) if s_a > 1e-10 else np.inf
        
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr *= 0.95
                L = max(L * 1.1, 1e-20)
        
        # Adaptive beta decay
        if it % 100 == 0 and it > 0:
            beta = max(1e-12, beta * 0.85)
        
        # Adaptive top_k (gradually decrease for sharper focus)
        if it % 500 == 0 and it > 0:
            top_k = max(3, top_k * 3 // 4)
        
        # Restart if stuck too long
        if no_imp > max(3000, 5 * n):
            a = best_a.copy()
            y = a.copy()
            t = 1.0
            no_imp = 0
            lr = base_lr
            L = 2.0 * n
            beta = max(1e-10, 0.05 * float(_fft_convolve(a).max()))
            top_k = max(5, n // 16)
            # Add small perturbation
            rng = np.random.default_rng(it)
            a = a * (1.0 + 0.05 * rng.normal(size=n))
            a = np.maximum(a, 0)
            a = _project_simplex(a)
            y = a.copy()
        
        # Periodically reduce L (the function gets "smoother" as beta decreases)
        if it % 500 == 0:
            L = max(2.0 * n * 0.8, L * 0.9)
    
    return best_a


def _lp_refine(a, deadline, max_iter=20000, inflate=1.001, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)
        
        if g is None:
            if no_imp > 100:
                break
            no_imp += 1
            inflate_cur = min(1.01, inflate_cur * 1.2)
            continue
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.1)
            if no_imp > 500:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        score_new = _score(a_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
            if no_imp > max(200, 2 * n):
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(500, 4 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        
        a = a_new
    
    return best_a


def _subgradient_optimize(a, deadline, max_iter=30000):
    """Subgradient method with Polyak-like step size tracking."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Upper bound on optimal value (from current best)
    M_upper = best_score
    # Lower bound (theoretical minimum is 2 for uniform, but we can do better)
    M_lower = 1.5
    
    c = _fft_convolve(a)
    M_cur = float(c.max())
    
    no_imp = 0
    lr = 0.01
    base_lr = lr
    
    # Track running average for Cesaro averaging
    a_avg = a.copy()
    count = 0
    
    active_tol = 0.1  # Fraction of M to consider "active"
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M_val = float(c.max())
        
        # Find active lags
        threshold = M_val * (1.0 - active_tol)
        active_mask = c >= threshold
        n_active = int(active_mask.sum())
        
        if n_active == 0:
            active_mask[int(np.argmax(c))] = True
            n_active = 1
        elif n_active > n // 8:
            # Too many active, use top n//8
            top_n = max(3, n // 8)
            sorted_idx = np.argsort(c)[::-1][:top_n]
            active_mask = np.zeros_like(c)
            active_mask[sorted_idx] = True
            n_active = top_n
        
        # Subgradient: average gradient over active lags
        # grad[k] = (2/n_active) * sum_{m active} a[m-k]
        w = active_mask.astype(np.float64) / n_active
        h_full = np.correlate(w, a, mode='full')
        h = h_full[n - 1: 2 * n - 1]
        grad = 2.0 * h
        
        # Polyak step size
        gap = M_val - M_lower
        if gap < 1e-6:
            lr_cur = 0.001
        else:
            lr_cur = base_lr * min(1.0, 2.0 * gap / (np.linalg.norm(grad) ** 2 + 1e-20))
        
        a_new = _project_simplex(a - lr_cur * grad)
        a = a_new
        
        # Cesaro average
        count += 1
        a_avg = (a_avg * (count - 1) + a) / count
        
        score_new = _score(a)
        score_avg = _score(a_avg)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a.copy()
            no_imp = 0
        elif score_avg < best_score - 1e-12:
            best_score = score_avg
            best_a = a_avg.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                base_lr *= 0.95
        
        # Adapt active_tol
        if it % 200 == 0 and it > 0:
            active_tol = max(0.01, active_tol * 0.9)
        
        if no_imp > max(3000, 4 * n):
            a = best_a.copy()
            a_avg = a.copy()
            count = 1
            no_imp = 0
            base_lr = lr
            active_tol = 0.1
            rng = np.random.default_rng(it)
            a = a * (1.0 + 0.08 * rng.normal(size=n))
            a = np.maximum(a, 0)
            a = _project_simplex(a)
    
    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, c1=2.0, c2=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + c1 * u + c2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u * u / (2.0 * sigma * sigma))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_spectrum_profile(n, n_iter=100):
    """Iteratively shape a to have flat spectrum (flat |DFT|^2)."""
    # Start with a random non-negative vector
    rng = np.random.default_rng(12345)
    a = np.abs(rng.normal(size=n)) + 0.1
    a = a / a.sum()
    
    N = _next_pow2(2 * n - 1)
    
    for _ in range(n_iter):
        # Compute |A|^2
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        
        # Target: flat spectrum
        P_target = np.full_like(P, P.mean())
        
        # Adjust amplitude: scale by sqrt(target/current)
        scale = np.sqrt(P_target / (P + 1e-20))
        A_new = A * scale
        A_new[0] = A[0]  # Keep DC component (sum is fixed)
        
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a = a_new / s
    
    return a


def _spectral_optimize(a, deadline, max_iter=10000):
    """Optimize in spectral domain: flatten |A|^2 while keeping a >= 0."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    N = _next_pow2(2 * n - 1)
    
    # Initial spectrum
    A = np.fft.rfft(a, N)
    
    lr = 0.3
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a = _project_simplex(a)
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        
        # Excess spectrum
        P_mean = P.mean()
        excess = P - P_mean  # Positive where spectrum peaks
        
        # Gradient in spectral domain: reduce excess
        # d/dA |A|^2 = 2*Re(A) (for real a)
        # We want to reduce |A|^2 where it's above mean
        # Update: A <- A * (1 - lr * excess_frac)
        excess_frac = excess / (P_mean + 1e-20)
        scale = 1.0 - lr * np.maximum(excess_frac, 0) + 0.5 * lr * np.maximum(-excess_frac, 0)
        
        A_new = A * scale
        A_new[0] = A[0]  # Fix DC
        
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-30:
            break
        a_new = a_new / s_new
        
        score_new = _score(a_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr *= 0.95
            if no_imp > 500:
                break
        
        a = a_new
    
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining FISTA, LP, subgradient, and spectral methods.
    
    Strategy:
    1. Initialize from parent + analytical profiles
    2. FISTA with adaptive parameters for fast convergence
    3. LP refinement for local improvement
    4. Subgradient with Cesaro averaging
    5. Spectral domain optimization
    6. Try multiple lengths (256, 512, 1024)
    7. Final intensive polish cycle
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={p.size}")
    except Exception:
        pass
    
    # ============ PHASE 1: FISTA on parent at its native length ============
    if parent is not None:
        n_p = parent.size
        print(f"\nPhase 1: FISTA on parent, n={n_p}")
        ph1_end = t_start + min(120.0, (deadline - t_start) * 0.1)
        a_f = _fista_optimize(parent, ph1_end, max_iter=30000, initial_lr=0.05)
        sc_f = _score(a_f)
        print(f"  FISTA result: score={sc_f:.10f}")
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
    
    # ============ PHASE 2: LP refinement on best ============
    if best_seq is not None and time.monotonic() < deadline - 200:
        n_b = best_seq.size
        print(f"\nPhase 2: LP refinement, n={n_b}")
        lp_end = t_start + min(250.0, deadline - 200.0)
        a_lp = _lp_refine(best_seq, lp_end, max_iter=10000, inflate=1.001, blend=0.25)
        sc_lp = _score(a_lp)
        print(f"  LP result: score={sc_lp:.10f}")
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
    
    # ============ PHASE 3: FISTA polish after LP ============
    if best_seq is not None and time.monotonic() < deadline - 150:
        n_b = best_seq.size
        print(f"\nPhase 3: FISTA polish, n={n_b}")
        f_end = t_start + min(100.0, deadline - 150.0)
        a_f2 = _fista_optimize(best_seq, f_end, max_iter=20000, initial_lr=0.03,
                               top_k_init=max(3, n_b // 32))
        sc_f2 = _score(a_f2)
        print(f"  FISTA polish: score={sc_f2:.10f}")
        if sc_f2 < best_score:
            best_score = sc_f2
            best_seq = a_f2.copy()
    
    # ============ PHASE 4: Try n=512 from scratch ============
    if time.monotonic() < deadline - 200:
        n_try = 512
        print(f"\nPhase 4: Fresh optimization at n={n_try}")
        
        # Multiple initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_try))
        inits.append(_make_linear_profile(n_try, 1.5))
        inits.append(_make_linear_profile(n_try, 2.0))
        inits.append(_make_quadratic_profile(n_try, 2.0, -1.0))
        inits.append(_make_flat_spectrum_profile(n_try, 200))
        
        # Perturbed parent
        if parent is not None:
            p512 = _interpolate_profile(parent, n_try)
            for _ in range(2):
                p_pert = p512 * (1.0 + 0.15 * rng.normal(size=n_try))
                p_pert = np.maximum(p_pert, 0)
                p_pert = p_pert / p_pert.sum()
                inits.append(p_pert)
        
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 180:
                break
            
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            print(f"  Init[{idx}]: score={sc0:.10f}")
            
            # FISTA
            f_budget = min(60.0, deadline - time.monotonic() - 180.0)
            if f_budget < 5:
                break
            f_end = time.monotonic() + f_budget
            a_f = _fista_optimize(a0, f_end, max_iter=10000, initial_lr=0.05)
            sc_f = _score(a_f)
            
            # LP
            lp_budget = min(60.0, deadline - time.monotonic() - 180.0)
            if lp_budget > 5:
                lp_end = time.monotonic() + lp_budget
                a_lp = _lp_refine(a_f, lp_end, max_iter=3000, inflate=1.001, blend=0.3)
                sc_f = _score(a_lp)
            
            print(f"    -> FISTA+LP: score={sc_f:.10f}")
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a0.copy() if a0 is a_f else a_lp.copy()
                best_seq = a_lp if 'a_lp' in dir() else a_f
    
    # ============ PHASE 5: Try n=1024 ============
    if time.monotonic() < deadline - 150:
        n_try = 1024
        print(f"\nPhase 5: Optimization at n={n_try}")
        
        if parent is not None:
            a0 = _interpolate_profile(parent, n_try)
        else:
            a0 = _make_linear_profile(n_try, 1.5)
        
        sc0 = _score(a0)
        print(f"  Initial: score={sc0:.10f}")
        
        # FISTA (longer budget for larger n)
        f_budget = min(100.0, deadline - time.monotonic() - 120.0)
        if f_budget > 10:
            f_end = time.monotonic() + f_budget
            a_f = _fista_optimize(a0, f_end, max_iter=15000, initial_lr=0.04)
            sc_f = _score(a_f)
            print(f"  FISTA: score={sc_f:.10f}")
            
            # LP
            lp_budget = min(80.0, deadline - time.monotonic() - 120.0)
            if lp_budget > 10:
                lp_end = time.monotonic() + lp_budget
                a_lp = _lp_refine(a_f, lp_end, max_iter=3000, inflate=1.001, blend=0.2)
                sc_lp = _score(a_lp)
                print(f"  LP: score={sc_lp:.10f}")
                sc_f = min(sc_f, sc_lp)
                if sc_lp < sc_f:
                    a_f = a_lp
            
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
    
    # ============ PHASE 6: Subgradient on best ============
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_b = best_seq.size
        print(f"\nPhase 6: Subgradient, n={n_b}")
        sg_budget = min(80.0, deadline - time.monotonic() - 80.0)
        if sg_budget > 10:
            sg_end = time.monotonic() + sg_budget
            a_sg = _subgradient_optimize(best_seq, sg_end, max_iter=10000)
            sc_sg = _score(a_sg)
            print(f"  Subgradient: score={sc_sg:.10f}")
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
    
    # ============ PHASE 7: Spectral optimization ============
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_b = best_seq.size
        print(f"\nPhase 7: Spectral optimization, n={n_b}")
        sp_budget = min(50.0, deadline - time.monotonic() - 60.0)
        if sp_budget > 5:
            sp_end = time.monotonic() + sp_budget
            a_sp = _spectral_optimize(best_seq, sp_end, max_iter=5000)
            sc_sp = _score(a_sp)
            print(f"  Spectral: score={sc_sp:.10f}")
            if sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
    
    # ============ PHASE 8: Final intensive LP + FISTA cycle ============
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_b = best_seq.size
        print(f"\nPhase 8: Final polish cycle, n={n_b}")
        
        # LP
        lp_end = time.monotonic() + min(60.0, deadline - time.monotonic() - 15.0)
        a_lp_final = _lp_refine(best_seq, lp_end, max_iter=5000, inflate=1.0005, blend=0.15)
        sc_lp_f = _score(a_lp_final)
        print(f"  Final LP: score={sc_lp_f:.10f}")
        if sc_lp_f < best_score:
            best_score = sc_lp_f
            best_seq = a_lp_final.copy()
        
        # FISTA
        if time.monotonic() < deadline - 10:
            f_end = time.monotonic() + min(40.0, deadline - time.monotonic() - 10.0)
            a_f_final = _fista_optimize(best_seq, f_end, max_iter=10000, initial_lr=0.02,
                                        top_k_init=max(3, n_b // 64))
            sc_f_f = _score(a_f_final)
            print(f"  Final FISTA: score={sc_f_f:.10f}")
            if sc_f_f < best_score:
                best_score = sc_f_f
                best_seq = a_f_final.copy()
    
    # ============ PHASE 9: One more LP with very tight constraints ============
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_b = best_seq.size
        print(f"\nPhase 9: Tight LP, n={n_b}")
        lp_end = time.monotonic() + min(30.0, deadline - time.monotonic() - 5.0)
        a_tight = _lp_refine(best_seq, lp_end, max_iter=3000, inflate=1.0001, blend=0.1)
        sc_tight = _score(a_tight)
        print(f"  Tight LP: score={sc_tight:.10f}")
        if sc_tight < best_score:
            best_score = sc_tight
            best_seq = a_tight.copy()
    
    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)
    
    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                # The evaluate function may give slightly different score due to
                # floating point, use it as final
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Evaluation error: {e}")
    
    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
