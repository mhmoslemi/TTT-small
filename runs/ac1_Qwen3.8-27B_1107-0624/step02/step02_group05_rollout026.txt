"""

import numpy as np

def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_conv(f):
    """Compute linear convolution of f with itself."""
    return np.convolve(f, f)


def soft_max_weights(c, beta):
    """Compute soft-max weights for approximating max(c)."""
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -100, 100)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval, w


def fft_cross_corr(w, f):
    """Compute g[i] = sum_j w[i+j] * f[j] for i=0..n-1 using FFT.
    
    w has length 2n-1, f has length n.
    Returns array of length n.
    """
    n = len(f)
    N = 1
    while N < 3 * n:
        N <<= 1
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f[::-1], N)
    G = np.fft.irfft(W * F, N)
    return G[n - 1:2 * n - 1]


def compute_gradient_z(f, beta):
    """Compute gradient of score w.r.t. log-parameters z.
    
    Returns grad_z of length n where f = softmax(z).
    """
    n = len(f)
    c = np.convolve(f, f)
    mval, w = soft_max_weights(c, beta)
    g = fft_cross_corr(w, f)
    # grad_z[i] = f[i] * (g[i] - dot(f, g))
    dot_fg = float(np.dot(f, g))
    grad_z = f * (g - dot_fg)
    return grad_z, mval


def normalize(f):

    f = np.maximum(f, 0.0)

def logit_to_f(z):

    z = z - z.max()
    e = np.exp(np.clip(z, -100, 100))
    s = float(e.sum())
    if s < 1e-30:
        return np.ones(len(z)) / len(z)
    return e / s


def f_to_logit(f):
    """Convert probability vector to log-parameters."""
    f = np.maximum(f, 1e-30)
    z = np.log(f)
    z -= z.mean()
    return z

def adam_optimize(f_init, deadline, rng, lr=0.1, max_iters=10000, beta_init=None, verbose=False):

    Returns (best_f, best_score).

    if f is None:
        return f_init, np.inf
    
    n = len(f)
    z = f_to_logit(f)

    v = np.zeros(n)

    M = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:

            v.fill(0)

def lp_direction(f, M):
    """Solve LP to find direction g maximizing sum(g) s.t. (f*g)[k] <= M, g >= 0.
    
    Returns g (normalized to sum 1) or None.
    """
    n = len(f)
    # Build constraint matrix: (f*g)[k] <= M for k=0..2n-2
    # (f*g)[k] = sum_i A[k,i] * g[i]
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            A[k, idx] = f[k - idx]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            g_sum = float(g.sum())
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def lp_blend_step(f, M, g):
    """Find best blend t in (1-t)*f + t*g that reduces M."""
    best_t = 0.0
    best_f = f
    best_M = M
    for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.15, 0.25, 0.4]:
        f_try = (1.0 - t) * f + t * g
        f_try = normalize(f_try)
        if f_try is None:
            continue
        M_try = float(np.convolve(f_try, f_try).max())
        if M_try < best_M - 1e-10:
            best_M = M_try
            best_f = f_try
            best_t = t
    return best_f, best_M, best_t

    print(f"Phase 4: n={n4}, time budget={phase4_end - time.monotonic():.0f}s")
    
    results_4 = []
    top_4 = min(2, len(results_3) if results_3 else len(results_1))
    source_4 = results_3 if results_3 else results_1
    
    for i in range(top_4):
        remaining = phase4_end - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining / max(1, top_4 - i), 150.0)
        if alloc < 20:
            break
        f_ext = interpolate_profile(source_4[i][1], n4)
        run_deadline = time.monotonic() + alloc
        f_best, sc_best = adam_optimize(f_ext, run_deadline, rng, lr=0.04, max_iters=3000)
        try_update(f_best, sc_best)
        results_4.append((sc_best, f_best))
        print(f"  n={n4} Adam {i+1}/{top_4}: score={sc_best:.8f}, global best={best_score:.8f}")
    
    # ============ Phase 5: Final refinement of best solution ============
    phase5_end = deadline
    if best_seq is not None and time.monotonic() < phase5_end - 60:
        n_best = len(best_seq)
        print(f"Phase 5: Final refinement at n={n_best}, remaining={phase5_end - time.monotonic():.0f}s")
        
        # Adam refinement
        f_best, sc_best = adam_optimize(best_seq, phase5_end - 30, rng, lr=0.03, max_iters=5000,
                                         beta_init=0.01 * float(np.convolve(best_seq, best_seq).max()))
        try_update(f_best, sc_best)
        print(f"  Final Adam: score={sc_best:.8f}")
        
        # LP refinement
        if time.monotonic() < phase5_end - 20:
            M = float(np.convolve(best_seq, best_seq).max())
            f_best, sc_best = lp_optimize(best_seq, phase5_end - 5, rng, max_iters=200)
            try_update(f_best, sc_best)
            print(f"  Final LP: score={sc_best:.8f}")
    
    # ============ Final validation ============
    if best_seq is None:
        best_seq = make_uniform(256)
        best_score = compute_score(best_seq)
    
    # Ensure validity
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_uniform(256)
        best_score = 2.0
    else:
        best_seq = best_seq / s
    
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    
    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}")
    
    # Try evaluate_sequence for confirmation
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence score: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")
    
    return [float(x) for x in best_seq]
