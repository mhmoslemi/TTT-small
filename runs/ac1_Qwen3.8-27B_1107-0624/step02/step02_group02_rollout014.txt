"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based descent with structured restarts, smooth gradient polishing,
and multi-scale (n) exploration. Key innovations over previous approach:
- Structured perturbations (low-frequency patterns, envelope shaping) for escaping local minima

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def make_envelope_profile(n, alpha=0.5, beta=0.5):
    """Create a bell-shaped envelope: (1 - alpha*(t-0.5)^2/beta^2)^gamma."""
    t = np.linspace(0.0, 1.0, n)
    x = (t - 0.5) / (0.5 * beta) if beta > 0 else np.zeros(n)
    a = np.maximum(1.0 - alpha * x * x, 0.01)
    return a / a.sum()


def make_cosine_envelope(n, gamma=1.0):
    """Cosine envelope raised to gamma power."""
    t = np.linspace(0.0, 1.0, n)
    a = np.maximum(0.5 + 0.5 * np.cos(np.pi * t), 1e-12) ** gamma
    return a / a.sum()


def make_tapered_uniform(n, taper=0.3):
    """Uniform in the middle, tapered at edges."""
    a = np.ones(n)
    m = int(n * taper / 2)
    if m > 0:
        ramp = np.linspace(0.3, 1.0, m)
        a[:m] = ramp
        a[-m:] = ramp[::-1]
    return a / a.sum()


def make_mixed_profile(n, k=2, amp=0.4, base=1.0):

    t = np.linspace(0.0, 1.0, n)
    a = base + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)

def make_random_smooth(n, rng, freq_max=5):

def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 0.0)
    s = result.sum()
    return result / s if s > 1e-10 else np.ones(n_new) / n_new


def structured_perturbations(f, rng):
    """Generate structured perturbations of f to escape local minima."""
    n = len(f)
    perturbations = []

    # Low-frequency cosine bump
    for freq in [1, 2, 3]:
        t = np.linspace(0.0, 1.0, n)
        bump = 0.15 * np.cos(2.0 * np.pi * freq * t - rng.uniform(0, 2 * np.pi))
        p = np.maximum(f + bump, 0.0)
        ps = p.sum()
        if ps > 1e-10:
            perturbations.append(p / ps)

    # Envelope reshaping
    t = np.linspace(0.0, 1.0, n)
    for gamma in [0.5, 1.5, 2.0]:
        envelope = (0.5 + 0.5 * np.cos(np.pi * t)) ** gamma
        envelope = envelope / envelope.mean()
        p = np.maximum(f * envelope, 0.0)
        ps = p.sum()
        if ps > 1e-10:
            perturbations.append(p / ps)

    # Partial zeroing: zero out random 20% of entries
    mask = rng.random(n) > 0.2
    p = f * mask
    ps = p.sum()
    if ps > 1e-10:
        perturbations.append(p / ps)

    # Mirror flip
    p = f[::-1].copy()
    perturbations.append(p / p.sum())

    # Blend with uniform
    for mix in [0.2, 0.4]:
        p = (1 - mix) * f + mix * np.ones(n) / n
        perturbations.append(p / p.sum())

def soft_max_and_weights(c, beta):
    """Smooth max and attention weights."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(cnt)
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights_fft(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 using FFT."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_stable(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e) / e.size
    return e / s


def gradient_polish(p0, deadline, rng, lr=0.05, reg=1e-6, beta_init=None, n_beta_annoals=5):
    """Gradient-based polish using smooth max in logit space with Nesterov momentum."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    m_nest = np.zeros(n, dtype=np.float64)  # Nesterov momentum
    v_nest = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    if beta_init is None:
        beta = max(1e-12, 0.15 * maxc)
    else:
        beta = beta_init

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    beta_annoal_count = 0

    while time.monotonic() < deadline:
        L, w = soft_max_and_weights(c, beta)
        gp = 2.0 * corr_weights_fft(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            m_nest.fill(0.0)
            v_nest.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)

        # Nesterov-style: use momentum to predict
        m_nest = mh + 0.3 * (mh - m_nest)
        upd = m_nest / (np.sqrt(vh) + eps)

        z_new = z - lr * upd
        if not np.all(np.isfinite(z_new)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            m_nest.fill(0.0)
            v_nest.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z_new, -60.0, 60.0)
        z -= float(np.mean(z))
        p = softmax_stable(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            m_nest.fill(0.0)
            v_nest.fill(0.0)
            adam_t = 0
            continue

        # Anneal beta
        if adam_t % 50 == 0:
            beta = max(1e-14, beta * 0.85)
            beta_annoal_count += 1

        # Adaptive LR
        if no_imp > 500:
            lr = max(0.005, lr * 0.95)

        # Restart when very stuck
        if no_imp > max(1000, 5 * n):
            # Try structured perturbations
            perturbs = structured_perturbations(best_p, rng)
            found_better = False
            for pp in perturbs:
                cc = full_convolve_fft(pp)
                ss = 2.0 * n * float(cc.max())
                if ss < best_score:
                    best_score = ss
                    best_p = pp.copy()
                    z = np.log(np.maximum(pp, 1e-12))
                    z -= float(np.mean(z))
                    p = pp.copy()
                    c = full_convolve_fft(p)
                    maxc = float(c.max())
                    beta = max(1e-12, 0.2 * maxc)
                    lr = base_lr
                    no_imp = 0
                    m.fill(0.0)
                    v.fill(0.0)
                    m_nest.fill(0.0)
                    v_nest.fill(0.0)
                    adam_t = 0
                    found_better = True
                    break
            if not found_better:
                z = 0.7 * best_z + 0.3 * rng.normal(size=n) * max(0.1, np.std(best_z))
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                p = softmax_stable(z)
                c = full_convolve_fft(p)
                maxc = float(c.max())
                beta = max(1e-12, 0.25 * maxc)
                m.fill(0.0)
                v.fill(0.0)
                m_nest.fill(0.0)
                v_nest.fill(0.0)
                adam_t = 0
                lr = base_lr
                no_imp = 0

    return best_p, best_score


def quick_lp_optimize(f_init, n_iters_max=200, verbose=False):
    """Quick LP optimization with fixed iterations (for trying many starts)."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    best_M = float(np.convolve(f, f).max())
    best_f = f.copy()
    t = 0.3

    for it in range(n_iters_max):
        M = float(np.convolve(f, f).max())
        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            # Small noise restart
            f = best_f + 0.03 * np.random.default_rng(it % 100).normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.3
            continue

        g = g / g.sum()
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new
        M_new = float(np.convolve(f_new, f_new).max())

        if M_new <= best_M + 1e-12:
            f = f_new
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
        else:
            t = max(0.01, t * 0.8)

        t = max(0.01, t * 0.998)

    return best_f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining LP descent, gradient polishing,
    structured perturbations, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting multi-strategy search. Budget: {budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial sequence: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ============ Phase 1: Quick multi-n exploration ============
    # Try several n values with quick optimization
    n_candidates = [256, 384, 512, 640, 768, 1024]
    phase1_end = min(deadline, t_start + 300.0)
    time_per_n = max(30.0, (phase1_end - t_start) / len(n_candidates))

    print(f"\n=== Phase 1: Quick multi-n exploration (0 to {phase1_end - t_start:.0f}s) ===")

    for n_try in n_candidates:
        if time.monotonic() > phase1_end - 5:
            break

        # Generate multiple starting profiles
        starts = []
        if init_seq is not None:
            starts.append(("interp", interpolate_profile(init_seq, n_try)))
        starts.append(("cos_env_1", make_cosine_envelope(n_try, 1.0)))
        starts.append(("cos_env_2", make_cosine_envelope(n_try, 2.0)))
        starts.append(("tapered", make_tapered_uniform(n_try, 0.4)))
        starts.append(("env_05", make_envelope_profile(n_try, 0.5, 0.7)))
        starts.append(("env_08", make_envelope_profile(n_try, 0.8, 0.8)))
        starts.append(("rand_smooth_1", make_random_smooth(n_try, rng)))
        starts.append(("rand_smooth_2", make_random_smooth(n_try, rng)))

        best_n_score = np.inf
        best_n_seq = None

        for name, p_start in starts:
            if time.monotonic() > phase1_end - 5:
                break
            sc_p = compute_score(p_start)
            # Quick LP optimization
            quick_deadline = min(phase1_end - 5, time.monotonic() + time_per_n / len(starts))
            if time.monotonic() > quick_deadline:
                update(p_start, sc_p)
                continue
            result = quick_lp_optimize(p_start, n_iters_max=80)
            sc_r = compute_score(result)
            update(result, sc_r)
            if sc_r < best_n_score:
                best_n_score = sc_r
                best_n_seq = result.copy()
            if sc_r < best_score + 0.02:
                print(f"  n={n_try}, {name}: score={sc_r:.10f}")

        if best_n_seq is not None:
            print(f"  Phase 1 best for n={n_try}: {best_n_score:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # ============ Phase 2: Deep LP optimization on best candidates ============
    phase2_end = min(deadline, t_start + 900.0)
    print(f"\n=== Phase 2: Deep LP optimization ({phase1_end - t_start:.0f}s to {phase2_end - t_start:.0f}s) ===")

    if best_seq is not None and time.monotonic() < phase2_end - 20:
        n_best = best_seq.size

        # Deep LP on the current best
        lp_deadline = min(phase2_end - 10, time.monotonic() + 300.0)
        print(f"  Deep LP on n={n_best}, current best={best_score:.10f}")
        result = lp_optimize(best_seq, lp_deadline, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"  After deep LP n={n_best}: score={sc:.10f}")

        # Try deep LP on a few other n values
        for n_alt in [n_best // 2, n_best * 2, 512]:
            if n_alt == n_best or n_alt < 64 or n_alt > 2048:
                continue
            if time.monotonic() > phase2_end - 30:
                break
            p_alt = interpolate_profile(best_seq, n_alt)
            sc_alt = compute_score(p_alt)
            update(p_alt, sc_alt)
            alt_deadline = min(phase2_end - 10, time.monotonic() + 150.0)
            result_alt = lp_optimize(p_alt, alt_deadline, rng, verbose=False)
            sc_alt_r = compute_score(result_alt)
            update(result_alt, sc_alt_r)
            if sc_alt_r < best_score + 0.01:
                print(f"  Deep LP n={n_alt}: score={sc_alt_r:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: Gradient polishing ============
    phase3_end = min(deadline, t_start + 1500.0)
    print(f"\n=== Phase 3: Gradient polishing ({phase2_end - t_start:.0f}s to {phase3_end - t_start:.0f}s) ===")

    if best_seq is not None and time.monotonic() < phase3_end - 20:
        n_polish = best_seq.size

        # Multiple gradient polish rounds with different parameters
        polish_configs = [
            {"lr": 0.05, "reg": 1e-6, "beta": None},
            {"lr": 0.03, "reg": 5e-7, "beta": None},
            {"lr": 0.08, "reg": 1e-5, "beta": None},
            {"lr": 0.02, "reg": 1e-7, "beta": None},
        ]

        for cfg in polish_configs:
            if time.monotonic() > phase3_end - 10:
                break
            per_deadline = min(phase3_end - 5, time.monotonic() + 120.0)
            if time.monotonic() > per_deadline:
                break

            p0 = best_seq.copy()
            # Add tiny noise for diversity
            p0 = p0 + 0.002 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()

            result, sc = gradient_polish(p0, per_deadline, rng,
                                          lr=cfg["lr"], reg=cfg["reg"],
                                          beta_init=cfg["beta"])
            update(result, sc)
            print(f"  Gradient polish lr={cfg['lr']}: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Targeted improvement attempts ============
    phase4_end = min(deadline, t_start + 1700.0)
    print(f"\n=== Phase 4: Targeted improvements ===")

    if best_seq is not None and time.monotonic() < phase4_end - 20:
        n_cur = best_seq.size

        # Try interpolation to slightly larger n
        for scale in [1.25, 1.5, 2.0]:
            n_new = int(n_cur * scale)
            if n_new > 2048 or n_new < 64:
                continue
            if time.monotonic() > phase4_end - 30:
                break
            p_new = interpolate_profile(best_seq, n_new)
            sc_new = compute_score(p_new)
            update(p_new, sc_new)

            # Quick LP + gradient
            lp_d = min(phase4_end - 10, time.monotonic() + 60.0)
            r1 = lp_optimize(p_new, lp_d, rng, verbose=False)
            sc1 = compute_score(r1)
            update(r1, sc1)

            if time.monotonic() < phase4_end - 20:
                gp_d = min(phase4_end - 5, time.monotonic() + 60.0)
                r2, sc2 = gradient_polish(r1, gp_d, rng, lr=0.04, reg=1e-6)
                update(r2, sc2)
            print(f"  n={n_new} (scale={scale}): best={best_score:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ============ Phase 5: Final polish ============
    if time.monotonic() < deadline - 5 and best_seq is not None:
        final_d = min(deadline - 3, time.monotonic() + 40.0)
        n_final = best_seq.size
        p0 = best_seq + 0.001 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, sc = gradient_polish(p0, final_d, rng, lr=0.02, reg=1e-7)
        update(result, sc)
        print(f"Final polish: score={sc:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_cosine_envelope(512, 1.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = make_cosine_envelope(512, 1.0)
        s = float(final_seq.sum())

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        else:
            print(f"WARNING: evaluate_sequence returned {eval_score}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
