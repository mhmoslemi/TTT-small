def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=10):

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)

def make_spectral_init(n, rng):
    """Initialize from spectral flat-convolution target."""
    N = next_pow2(2 * n - 1)
    target = np.zeros(N)
    target[:2 * n - 1] = 1.0 / (2 * n - 1)

def softmax_gradient_polish(f0, deadline, rng, lr=0.05, max_iter=100000,
                            beta_init=None, verbose=False):
    """
    Softmax gradient descent in logit space for minimizing max(conv(a,a)).
    Uses adaptive temperature annealing and periodic restarts.
    """

    n = f.size

    z -= z.mean()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-12, 0.1 * M_val)

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_corr = next_pow2(3 * n)
    start_time = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Soft-max weights over convolution values
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        # Gradient computation via FFT cross-correlation
        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * grad_z
        v_adam = b2 * v_adam + (1.0 - b2) * (grad_z * grad_z)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)

        # Adaptive learning rate with decay
        decay = 1.0 / (1.0 + 0.01 * it / max(1, n))
        z = z - lr * decay * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0

            # Update smoothing: track the new max
            if M_new < M_val - 0.5 * beta:
                M_val = M_new
                beta = max(1e-14, 0.3 * beta)
        else:
            no_imp += 1

        # Anneal temperature periodically
        if t_adam % 500 == 0:
            beta = max(1e-14, beta * 0.65)

        # Restart from best with perturbation
        if no_imp > max(200, 5 * n):
            # Mix best with random to escape
            mix = rng.uniform(0.7, 0.95)
            z = mix * best_z + (1.0 - mix) * rng.normal(size=n) * np.sqrt(beta / max(M_val, 1e-10))
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-14, 0.15 * M_val)
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            no_imp = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    [softmax] iter={it}, M={best_M:.10f}, beta={beta:.2e}, elapsed={elapsed:.1f}s")

    return best_f, best_M

def flatten_step(f, n_passes=5):
    """
    Heuristic flatten: iteratively reduce the max convolution position
    by redistributing mass from contributing elements.
    """
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f
    f = f / s
    n = f.size
    best_f = f.copy()
    best_M = float(fft_convolve(f).max())

    for _ in range(n_passes):
        c = fft_convolve(f)
        k_star = int(np.argmax(c))
        M_val = c[k_star]

        # Find contributing pairs
        i_max = max(0, k_star - n + 1)
        i_min = min(k_star, n - 1)
        contributions = np.zeros(n)
        for i in range(i_max, i_min + 1):
            j = k_star - i
            contributions[i] = f[i] * f[j]

        # Reduce the largest contributor, increase a non-contributing element
        if contributions.max() < 1e-15:
            break
        i_reduce = int(np.argmax(contributions))
        # Find element far from position k_star
        dist = np.abs(np.arange(n) - k_star)
        i_increase = int(np.argmax(dist * f))  # prefer large, distant elements

        if i_reduce == i_increase:
            i_increase = (i_reduce + n // 2) % n

        delta = f[i_reduce] * 0.05
        f[i_reduce] -= delta
        f[i_increase] += delta
        f = np.maximum(f, 0.0)
        s = f.sum()
        f = f / s

        M_new = float(fft_convolve(f).max())
        if M_new < best_M:
            best_M = M_new
            best_f = f.copy()

    return best_f

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None
