def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _corr_weights(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):

def _internal_score(a):

def _interp_profile(p0, n_new):

def _make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def _make_cosine_sum_profile(n, coeffs):
    """
    Create a profile as 1 + sum of cosines.
    coeffs: list of (amplitude, frequency, phase) tuples.
    """
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for amp, freq, phase in coeffs:
        a += amp * np.cos(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 1e-10)
    s = float(a.sum())
    return a / s


def _make_double_bump_profile(n, frac1, frac2, width):
    """Two Gaussian bumps at positions frac1 and frac2."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - frac1) ** 2) / (2 * width ** 2)) + \
        np.exp(-((t - frac2) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 1e-10)
    s = float(a.sum())
    return a / s


def _make_w_profile(n, depth):
    """W-shaped profile: high at edges and center, low in between."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(4.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    s = float(a.sum())
    return a / s

def _lbfgs_polish(p0, steps=200, stiffness=50.0):
    """
    L-BFGS-B polishing using log-sum-exp smoothing of the max convolution.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n = p0.size
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()

    z0 = np.log(p0)

    def objective(z):
        z_c = np.clip(z, -60.0, 60.0)
        z_c = z_c - z_c.max()
        e = np.exp(z_c)
        p = e + 1e-12
        p = p / p.sum()
        c = _full_convolve_fft(p)
        # Log-sum-exp for smooth max
        smax = float(c.max())
        if smax <= 0:
            smax = 1e-10
        lse = smax + (1.0 / stiffness) * np.log(np.sum(np.exp(stiffness * (c - smax))))
        return 2.0 * n * lse

    def gradient(z):
        z_c = np.clip(z, -60.0, 60.0)
        z_c = z_c - z_c.max()
        e = np.exp(z_c)
        p = e + 1e-12
        p = p / p.sum()
        c = _full_convolve_fft(p)
        smax = float(c.max())
        if smax <= 0:
            smax = 1e-10
        # Soft-max weights
        w_soft = np.exp(stiffness * (c - smax))
        w_soft = w_soft / w_soft.sum()
        # Gradient of LSE w.r.t. p
        gp = 2.0 * _corr_weights(w_soft, p)
        # Gradient w.r.t. z (simplex constraint)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        return 2.0 * n * gz

    result = scipy_minimize(objective, z0, method='L-BFGS-B',
                            jac=gradient, options={'maxiter': steps,
                                                   'maxfun': steps * 3,
                                                   'ftol': 1e-12,
                                                   'gtol': 1e-10})
    z_opt = np.clip(result.x, -60.0, 60.0)
    z_opt = z_opt - z_opt.max()
    p_opt = _softmax_from_z(z_opt)
    sc = _internal_score(p_opt)
    return p_opt, sc

def _spectral_refine(p0, deadline, rng, n_iters=200):
    """
    Spectral refinement: modify the DFT to flatten the power spectrum,
    then project back to non-negative.
    """
    p = np.asarray(p0, dtype=np.float64).ravel()
    n = p.size
    p = p / p.sum()
    best_p = p.copy()
    best_sc = _internal_score(p)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute DFT
        A = np.fft.rfft(p)
        # |A|^2 is proportional to the circular autocorrelation
        # We want to reduce the peak of the circular autocorrelation
        # which corresponds to reducing the peak of |A|^2... 
        # Actually, we want to reduce the peak of the autocorrelation,
        # which is IDFT(|A|^2). So we want |A|^2 to be as flat as possible.
        pow_spec = np.abs(A) ** 2
        target = pow_spec.sum() / pow_spec.size
        
        # Water-filling: reduce high powers, increase low powers
        alpha = 0.1  # mixing parameter
        new_pow = (1 - alpha) * pow_spec + alpha * target
        new_pow = np.maximum(new_pow, 1e-20)
        
        # Preserve phase, adjust magnitude
        phase = np.angle(A)
        new_mag = np.sqrt(new_pow)
        A_new = new_mag * np.exp(1j * phase)
        
        # Inverse FFT and project to non-negative
        p_new = np.fft.irfft(A_new, n)
        p_new = np.maximum(p_new, 0.0)
        s = float(p_new.sum())
        if s > 1e-10:
            p_new = p_new / s
            sc = _internal_score(p_new)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_p = p_new.copy()
                p = p_new.copy()
    
    return best_p, best_sc

                                       no_imp_mult=3)
        update_best(p_best, sc_best, f"adam_{label}")
        polished.append((sc_best, p_best, label))
    
    # Sort polished by score
    polished.sort(key=lambda x: x[0])
    print(f"Phase 1: best={best_score:.10f}")

    # --- Phase 2: Multi-length exploration ---
    phase2_end = min(deadline, t_start + 800.0)
    
    # Try a few other lengths
    other_lengths = [l for l in [900, 960, 1000, 1100, 1152, 1200, 1280] 
                     if l != primary_n and l <= 1800]
    
    for n_try in other_lengths[:4]:
        if time.monotonic() > phase2_end - 60:
            break
        # Interpolate best to this length
        base_p = best_seq if best_seq is not None else polished[0][1]
        p_interp = _interp_profile(base_p, n_try)
        if p_interp is None:
            continue
        sc_interp = _internal_score(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_try}")
        
        # Quick polish
        per_end = min(phase2_end - 30, time.monotonic() + 40.0)
        p_pol, sc_pol = _logit_adam(p_interp, per_end, rng, lr=0.08,
                                     beta_init_frac=0.10, beta_decay=0.70,
                                     no_imp_mult=3)
        update_best(p_pol, sc_pol, f"polish_n{n_try}")
    
    print(f"Phase 2: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # --- Phase 3: L-BFGS-B polishing ---
    phase3_end = min(deadline, t_start + 1100.0)
    
    if time.monotonic() < phase3_end and best_seq is not None:
        print(f"  L-BFGS-B polish (n={best_seq.size})")
        # Use multiple stiffness values
        for stiff in [30.0, 80.0, 200.0]:
            if time.monotonic() > phase3_end:
                break
            p_lb, sc_lb = _lbfgs_polish(best_seq, steps=150, stiffness=stiff)
            update_best(p_lb, sc_lb, f"lbfgs_st{stiff}")
    
    print(f"Phase 3: best={best_score:.10f}")

    # --- Phase 4: Spectral refinement ---
    phase4_end = min(deadline, t_start + 1350.0)
    
    if time.monotonic() < phase4_end and best_seq is not None:
        print(f"  Spectral refinement (n={best_seq.size})")
        p_spec, sc_spec = _spectral_refine(best_seq, phase4_end, rng, n_iters=300)
        update_best(p_spec, sc_spec, "spectral")
    
    print(f"Phase 4: best={best_score:.10f}")

    # --- Phase 5: More aggressive Adam restarts ---
    phase5_end = min(deadline, t_start + 1550.0)
    
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = best_seq.size
        n_starts = 4
        per_start = (phase5_end - time.monotonic()) / max(1, n_starts)
        
        for i in range(n_starts):
            if time.monotonic() > phase5_end:
                break
            per_end = min(phase5_end, time.monotonic() + per_start)
            
            # Different noise patterns
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            
            if i == 0:
                noise = 0.1 * rng.normal(size=n_best)
            elif i == 1:
                # Low-freq noise
                noise = np.fft.irfft(np.maximum(np.fft.rfft(rng.normal(size=n_best)), 0), n_best)
                noise = 0.15 * noise / (np.std(noise) + 1e-10)
            elif i == 2:
                # High-freq noise
                noise = 0.08 * rng.normal(size=n_best)
            else:
                # Structured perturbation
                noise = 0.12 * np.sin(2 * np.pi * rng.integers(2, 20) * np.linspace(0, 1, n_best))
            
            z = z + noise
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_start = _softmax_from_z(z)
            
            p_try, sc_try = _logit_adam(p_start, per_end, rng, lr=0.06,
                                         beta_init_frac=0.08, beta_decay=0.72,
                                         no_imp_mult=3)
            update_best(p_try, sc_try, f"restart{i}_n{n_best}")
    
    print(f"Phase 5: best={best_score:.10f}")

    # --- Phase 6: Local search ---
    phase6_end = min(deadline, t_start + 1680.0)
    
    if time.monotonic() < phase6_end and best_seq is not None:
        print(f"  Local search (n={best_seq.size})")
        p_ls, sc_ls = _local_search(best_seq, phase6_end, rng, n_trials_max=30000)
        update_best(p_ls, sc_ls, "local")
    
    print(f"Phase 6: best={best_score:.10f}")

    # --- Phase 7: Final fine-tuning with very small learning rate ---
    if time.monotonic() < deadline - 20 and best_seq is not None:
        per_final = min(40.0, deadline - time.monotonic() - 20)
        p_fin, sc_fin = _logit_adam(best_seq, time.monotonic() + per_final, rng,
                                     lr=0.015, beta_init_frac=0.04, beta_decay=0.82,
                                     no_imp_mult=5)
        update_best(p_fin, sc_fin, "final_fine")
    
    # --- Final validation ---
    if best_seq is not None:
        final_seq = best_seq.copy()
        final_n = final_seq.size
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(256, 1.5)
        
        # Exact score check
        if final_n <= 2048:
            sc_exact = _true_score(final_seq)
            print(f"Exact score: {sc_exact:.10f}")
            if np.isfinite(sc_exact):
                best_score = min(best_score, sc_exact)
        
        # Try evaluate_sequence
        if final_n <= 1800:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    best_score = min(best_score, sc_eval)
            except Exception as e:
                print(f"Evaluator error: {e}")
        
        print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(256, 1.5)
    print(f"FALLBACK: n=256, score={_internal_score(fallback):.6f}")
    return [float(x) for x in fallback]
