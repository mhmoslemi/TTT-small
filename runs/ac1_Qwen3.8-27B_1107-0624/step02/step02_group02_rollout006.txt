"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based peak-targeting descent, subgradient polishing, and Adam in log-space.
Explores n=512, 1024, and 2048 with diverse initializations and restarts.

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of two >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max(c, beta):
    """Smooth max via log-sum-exp, returns (value, weights)."""
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights_fft(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] using FFT."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_stable(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size) / a.size
    return a / s


def make_template(n, kind, rng):
    """Generate a template profile of length n."""
    t = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'linear':
        a = 1.0 + 1.5 * t
    elif kind == 'linear2':
        a = 1.0 + 2.5 * t
    elif kind == 'cos1':
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
    elif kind == 'cos2':
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
    elif kind == 'sinc':
        x = (t - 0.5) * 4.0
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0.01)
    elif kind == 'gauss':
        a = np.exp(-2.0 * (t - 0.5) ** 2)
    elif kind == 'quad':
        a = 1.0 + 0.8 * (2.0 * t - 1.0) ** 2
    elif kind == 'bump':
        a = 0.3 + 0.7 * np.sin(np.pi * t) ** 2
    elif kind == 'mixed':
        a = 1.0 + 0.25 * np.cos(2 * np.pi * t) + 0.15 * np.cos(4 * np.pi * t)
    elif kind == 'random_smooth':
        coeffs = rng.uniform(-0.4, 0.4, 6)
        a = np.ones(n)
        for i, c in enumerate(coeffs[1:], 1):
            a = a + c * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 0.05)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = max(0, j - (n - 1))
        hi = min(2 * n - 1, j + n)
        idx = np.arange(lo, hi)
        A[idx, j] = f[idx - j]
    return A


def lp_peak_minimize(a, M_target):
    """LP: min t s.t. (a*b)_k <= t for all k, sum(b)=1, b>=0.
    Returns optimal direction b."""
    n = len(a)
    A = build_conv_matrix(a)
    # Variables: [b_0,...,b_{n-1}, t]
    # Objective: min t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    # Inequality: (a*b)_k - t <= 0  =>  A_k @ b - t <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(-1e6, 1e6)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            b = np.maximum(res.x[:n], 0.0)
            bs = b.sum()
            if bs > 1e-10:
                return b / bs
    except Exception:
        pass
    return None


def lp_max_sum(a, M):
    """LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            bs = b.sum()
            if bs > 1e-10:
                return b / bs
    except Exception:
        pass
    return None


def subgradient_step(a, c, beta=0.05):
    """Compute subgradient of max_c using soft-max weights."""
    n = len(a)
    _, w = soft_max(c, beta)
    # Gradient of c_k w.r.t. a_j:
    # g[j] = sum_k w[k] * (a[k-j] if 0<=k-j<n else 0) + ...
    # Use FFT: g = corr(w, a) + corr(w_rev, a)
    # Actually: dc_k/da_j = a[k-j] (if valid) + a[j] (if valid and k!=2j)
    # Simplified: g = 2 * corr(w, a) where corr is cross-correlation
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # Cross-correlation: sum_k w[k] * a[k-j]
    fc = np.fft.rfft(np.concatenate([a, np.zeros(N - n)]), N)
    g = np.fft.irfft(fw * np.conj(fc), N)[:n]
    # Also add the symmetric part: sum_k w[k] * a[j] where k-j valid
    # This is: for each j, sum_{k: 0<=k-j<n} w[k] * a[j] = a[j] * sum_{k=j}^{j+n-1} w[k]
    # Use cumulative sum
    w_pad = np.concatenate([np.zeros(n - 1), w])  # shift
    # Actually let me just compute it differently
    # g[j] = sum_k w[k] * (a[k-j] if 0<=k-j<n) + sum_k w[k] * (a[j] if 0<=k-j<n and k!=2j)
    # First term: cross-correlation of w and a
    # Second term: a[j] * (sum of w[k] for k where 0<=k-j<n) - a[j]*w[2j] (if 2j valid)
    # For simplicity, use the FFT cross-correlation which gives the first part
    # and add a[j] * S[j] where S[j] = sum_{k=max(0,j)}^{min(2n-2, j+n-1)} w[k]
    S = np.zeros(n)
    w_ext = np.concatenate([w, np.zeros(n)])
    for j in range(n):
        lo = max(0, j)
        hi = min(2 * n - 1, j + n)
        S[j] = w_ext[lo:hi].sum()
    g = g + a * S
    # Subtract the constant component (since we only care about the direction)
    dot_a = float(np.dot(a, g))
    g = g - dot_a * a
    return g


def adam_logit_optimize(p0, deadline, rng, lr=0.05, reg=1e-8):
    """Adam optimization in logit space for peak minimization."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.15 * maxc)

    best_p = p0.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max(c, beta)
        # Gradient computation via FFT
        N = next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(a_pad(p, N), N)
        g = np.fft.irfft(fw * np.conj(fa), N)[:n]
        # Add symmetric component
        w_ext = np.concatenate([w, np.zeros(n)])
        S = np.array([w_ext[max(0,j):min(2*n-1, j+n)].sum() for j in range(n)])
        g = g + p * S
        # Subtract mean component for invariance
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        gz += reg * p

        if not np.all(np.isfinite(gz)):
            z = best_p_logit(best_p, p0)
            m.fill(0.0); v.fill(0.0); step = 0
            continue

        step += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** step)
        vh = v / (1.0 - b2 ** step)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_p_logit(best_p, p0)
            m.fill(0.0); v.fill(0.0); step = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_stable(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.008, lr * 0.97)

        if step % 60 == 0:
            beta = max(1e-14, beta * 0.75)

        if no_imp > max(500, 3 * n):
            z = 0.7 * best_p_logit(best_p, p0) + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_stable(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0); v.fill(0.0); step = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def a_pad(a, N):
    """Zero-pad array a to length N."""
    n = len(a)
    if n >= N:
        return a[:N]
    out = np.zeros(N)
    out[:n] = a
    return out


def best_p_logit(best_p, p0):
    """Recover logit from best_p."""
    return np.log(np.maximum(best_p, 1e-12))


def lp_descent_loop(a_init, deadline, rng, verbose=True):
    """Iterative LP-based descent."""
    a = normalize_nonneg(a_init)
    n = len(a)
    c = full_convolve_fft(a)
    best_M = float(c.max())
    best_a = a.copy()
    t_blend = 0.25
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        M = float(c.max())

        # Alternate between two LP formulations
        if iteration % 3 == 0:
            # Peak-minimizing direction
            b = lp_peak_minimize(a, M)
        else:
            # Max-sum direction
            b = lp_max_sum(a, M)

        if b is None:
            t_blend *= 0.5
            stagnant += 1
            if t_blend < 0.005:
                break
            continue

        a_new = (1.0 - t_blend) * a + t_blend * b
        a_new = normalize_nonneg(a_new)
        c_new = full_convolve_fft(a_new)
        M_new = float(c_new.max())

        if M_new <= best_M + 1e-10:
            a = a_new
            if M_new < best_M - 1e-11:
                best_M = M_new
                best_a = a.copy()
                stagnant = 0
                # Slightly increase blend when improving
                t_blend = min(0.4, t_blend * 1.02)
            else:
                stagnant += 1
            iteration += 1
            if verbose and iteration % 50 == 0:
                print(f"    LP iter {iteration}: M={best_M:.12f}, score={2*n*best_M:.10f}, t={t_blend:.4f}")
        else:
            t_blend *= 0.5
            stagnant += 1

        # Slowly decay blend
        t_blend = max(0.008, t_blend * 0.997)

        # Random restart when stuck
        if stagnant > 80:
            noise_scale = 0.05 * max(1.0 / n, np.std(best_a))
            a = best_a + rng.normal(0, noise_scale, n)
            a = normalize_nonneg(a)
            t_blend = 0.2
            stagnant = 0
            if verbose:
                print(f"    LP restart at iter {iteration}, M={best_M:.12f}")

    return best_a


def coordinate_descent_polish(a, deadline, rng):
    """Coordinate descent: try adjusting pairs of elements to reduce peak."""
    a = normalize_nonneg(a).copy()
    n = len(a)
    c = full_convolve_fft(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    epoch = 0

    while time.monotonic() < deadline:
        epoch += 1
        improved = False
        # Try random pairs
        for _ in range(200):
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            # Try to move mass from i to j (or vice versa)
            for direction in [1, -1]:
                for alpha in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    a_trial = a.copy()
                    a_trial[i] += direction * alpha * a[i]
                    a_trial[j] -= direction * alpha * a[i]
                    if a_trial[i] < 0 or a_trial[j] < 0:
                        continue
                    a_trial = normalize_nonneg(a_trial)
                    c_trial = full_convolve_fft(a_trial)
                    M_trial = float(c_trial.max())
                    score_trial = 2.0 * n * M_trial
                    if score_trial < best_score - 1e-12:
                        a = a_trial
                        best_a = a.copy()
                        best_score = score_trial
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and epoch % 20 == 0:
            # Add small noise to escape
            a = a + 0.01 * rng.normal(size=n)
            a = normalize_nonneg(a)

    return best_a


def interpolate_profile(p0, n_new):
    """Linear interpolation to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum() if p0.sum() > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)

    phase1_end = min(deadline, t_start + 700.0)

    if init_seq is not None and init_seq.size != n_main:
        p_main = interpolate_profile(init_seq, n_main)
    elif init_seq is not None:
        p_main = init_seq.copy()
    else:
        p_main = make_template(n_main, 'linear', rng)

    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1: n={n_main}, start score={sc:.10f}")

    # Main LP optimization
    lp_deadline = min(phase1_end, time.monotonic() + 450.0)
    result = lp_descent_loop(p_main, lp_deadline, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"After main LP: score={sc:.10f}")

    # Try diverse templates
    templates = ['uniform', 'linear', 'linear2', 'cos1', 'cos2', 'sinc', 'gauss', 'quad', 'bump', 'mixed']
    for tmpl in templates:
        if time.monotonic() > phase1_end - 10:
            break
        p_t = make_template(n_main, tmpl, rng)
        sc_t = compute_score(p_t)
        if sc_t >= best_score + 0.08:
            continue
        per_dl = min(phase1_end, time.monotonic() + 35.0)
        result = lp_descent_loop(p_t, per_dl, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        if sc < best_score + 0.01:
            print(f"  Template {tmpl}: score={sc:.10f}")

    # Random smooth templates
    for ri in range(5):
        if time.monotonic() > phase1_end - 10:
            break
        p_r = make_template(n_main, 'random_smooth', rng)
        sc_r = compute_score(p_r)
        if sc_r >= best_score + 0.05:
            continue
        per_dl = min(phase1_end, time.monotonic() + 25.0)
        result = lp_descent_loop(p_r, per_dl, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)

    print(f"After Phase 1: best={best_score:.10f} (n={n_main})")

    # ============ Phase 2: Adam polish at n=512 ============
    phase2_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase2_end - 20 and best_seq is not None and best_seq.size == n_main:
        print(f"Phase 2: Adam polish at n={n_main}")
        for lr in [0.05, 0.03, 0.02]:
            if time.monotonic() > phase2_end - 10:
                break
            p0 = best_seq + 0.002 * rng.normal(size=n_main)
            p0 = normalize_nonneg(p0)
            per_dl = min(phase2_end, time.monotonic() + 80.0)
            result, sc = adam_logit_optimize(p0, per_dl, rng, lr=lr)
            update(result, sc)
            print(f"  Adam lr={lr}: score={sc:.10f}")

    # ============ Phase 3: Try n=1024 ============
    n_large = 1024
    phase3_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 3: n={n_large}, score={sc:.10f}")

        lp_dl3 = min(phase3_end, time.monotonic() + 300.0)
        result = lp_descent_loop(p_large, lp_dl3, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After n={n_large} LP: score={sc:.10f}")

        # Short Adam polish at n=1024
        if time.monotonic() < phase3_end - 40 and best_seq is not None and best_seq.size == n_large:
            p0 = best_seq + 0.001 * rng.normal(size=n_large)
            p0 = normalize_nonneg(p0)
            per_dl = min(phase3_end, time.monotonic() + 100.0)
            result, sc = adam_logit_optimize(p0, per_dl, rng, lr=0.03)
            update(result, sc)
            print(f"  Adam at n={n_large}: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Try n=2048 if time remains ============
    n_xl = 2048
    phase4_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase4_end - 30 and best_seq is not None and best_score > 1.503:
        p_xl = interpolate_profile(best_seq, n_xl)
        sc = compute_score(p_xl)
        update(p_xl, sc)
        print(f"Phase 4: n={n_xl}, score={sc:.10f}")

        lp_dl4 = min(phase4_end, time.monotonic() + 150.0)
        result = lp_descent_loop(p_xl, lp_dl4, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        print(f"After n={n_xl}: score={sc:.10f}")

    # ============ Phase 5: Final polishing ============
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = best_seq.size
        print(f"Phase 5: final polish at n={n_final}, best={best_score:.10f}")

        # Adam with very small lr
        p0 = best_seq + 0.0005 * rng.normal(size=n_final)
        p0 = normalize_nonneg(p0)
        per = min(deadline - 5, time.monotonic() + 60.0)
        result, sc = adam_logit_optimize(p0, per, rng, lr=0.01)
        update(result, sc)

        # Coordinate descent
        if time.monotonic() < deadline - 5:
            per2 = min(deadline - 3, time.monotonic() + 30.0)
            result_cd = coordinate_descent_polish(best_seq, per2, rng)
            sc_cd = compute_score(result_cd)
            update(result_cd, sc_cd)
            print(f"  Coord descent: score={sc_cd:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_template(512, 'linear', rng)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_template(256, 'linear', rng)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
