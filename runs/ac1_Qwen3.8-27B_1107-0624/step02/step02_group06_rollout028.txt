    # (f*g)[k] = sum_i f[i] * g[k-i] for valid i

        constraints.append(row @ g <= M)

    return np.maximum(g.value, 0)

"""
LP-based iterative optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Core method: at each step, solve LP max{sum(g): g>=0, (f*g)<=M} to find the
best feasible direction, then take a convex combination step. Combined with

import time
import cvxpy as cp


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-9:
        c = np.maximum(c, 0.0)
    elif mn < 0.0:
        c = c - mn
    return c


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

    s = float(f.sum())

        return np.full_like(f, target_sum / len(f))
    return f * (target_sum / s)


def build_conv_matrix_for_f(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M, solver=None):
    n = len(f)
    g_var = cp.Variable(n, nonneg=True)
    A = build_conv_matrix_for_f(f)
    constraints = [A @ g_var <= M]
    objective = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(objective, constraints)
    if solver is None:
        solver = cp.CLARABEL
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None
    if g_var.value is None:
        return None
    return np.maximum(g_var.value, 0.0)

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()

def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):

    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 60 == 0:
            beta = max(1e-14, beta * 0.75)

        if no_imp > max(400, 4 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def coordinate_descent_polish(p, deadline, rng, step_frac=0.06):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0
    sweep = 0

    factors_hi = [1.0 + step_frac, 1.0 + 0.5 * step_frac, 1.0 + 2.0 * step_frac]
    factors_lo = [1.0 - step_frac, 1.0 - 0.5 * step_frac, 1.0 - 0.3 * step_frac]

    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score
            for factor in factors_hi + factors_lo:
                new_val = orig * factor
                if new_val < 1e-15 or new_val > 1000.0:
                    continue
                p[i] = new_val
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                sc = internal_score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        sweep += 1
        if improved:
            step_frac = max(0.02, step_frac * 0.95)
        if no_imp > n:
            break

    return best_p, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 0 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    if s <= 0:
        return np.ones(n_new) / n_new
    return p1 / s
