"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based descent with backtracking line search, smooth log-sum-exp
AdamW optimization with cosine annealing, Nesterov-accelerated subgradient,
and multi-scale search across sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of two >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def convolve_with(a, b):
    """Compute conv(a, b) using FFT for efficiency."""
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    m = b.size
    if n * m <= 100000:
        return np.convolve(a, b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b)[k] <= M for all k, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, verbose=True):
    """LP-based optimization with backtracking line search."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    conv_f = np.convolve(f, f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.25
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        M_current = float(np.convolve(f, f).max())
        # Use slightly tighter M for better direction
        M = M_current * 0.999 if stagnant > 0 else M_current

        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            g = lp_step(f, M_current * 1.001)
            if g is None or g.sum() < 1e-10:
                break
        g = g / g.sum()

        # Backtracking line search
        found = False
        for _ in range(10):
            f_new = np.maximum((1 - t) * f + t * g, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-10:
                t *= 0.5
                continue
            f_new = f_new / s_new
            M_new = float(np.convolve(f_new, f_new).max())
            if M_new <= best_M + 1e-10:
                f = f_new
                if M_new < best_M - 1e-12:
                    best_M = M_new
                    best_f = f.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                found = True
                iteration += 1
                break
            else:
                t *= 0.5
        if not found:
            t = max(0.01, t * 0.5)
            stagnant += 1

        if verbose and iteration % 50 == 0 and iteration > 0:
            print(f"    LP iter {iteration}: M={best_M:.10f}, t={t:.4f}, stag={stagnant}")

        # Decay and floor t
        t = max(0.05, t * 0.995)

        # Random restart with adaptive noise
        if stagnant > 80:
            noise_scale = max(0.05, 0.3 / np.sqrt(n)) * np.std(best_f)
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.25
            stagnant = 0
            if verbose:
                print(f"    LP restart at iter {iteration}")

    return best_f


def soft_max_grad(c, beta):
    """Compute soft max value and gradient weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = np.count_nonzero(mask)
        w[mask] = 1.0 / max(1, cnt)
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def corr_weights_fft(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 using FFT."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e / float(e.sum())
    return p


def smooth_optimize(a_init, deadline, rng, lr=0.05, reg=1e-6, beta_init=None):
    """Smooth log-sum-exp optimization with AdamW and cosine annealing."""
    p = np.maximum(np.asarray(a_init, dtype=np.float64).ravel(), 1e-12)
    p = p / p.sum()
    n = p.size

    z = np.log(p)
    z -= float(np.mean(z))

    c = full_convolve_fft(p)
    maxc = float(c.max())
    if beta_init is None:
        beta = max(1e-12, 0.15 * maxc)
    else:
        beta = beta_init

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    base_lr = lr
    total_steps = 0
    best_imp_step = 0

    best_p = p.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    # Cosine annealing schedule
    T = 20000  # total steps for cosine period

    while time.monotonic() < deadline:
        # Cosine annealing of beta
        step_frac = min(1.0, total_steps / T)
        beta_eff = max(1e-14, beta * (0.5 + 0.5 * np.cos(np.pi * step_frac)))

        L, w = soft_max_grad(c, beta_eff)
        gp = 2.0 * corr_weights_fft(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = np.log(best_p)
            z -= float(np.mean(z))
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)

        # Cosine annealing of learning rate
        lr_eff = base_lr * (0.5 + 0.5 * np.cos(np.pi * min(1.0, total_steps / T)))
        lr_eff = max(0.005, lr_eff)

        z = z - lr_eff * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        total_steps += 1

        if np.isfinite(score) and score < best_score:
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
            if no_imp > 500:
                z = np.log(best_p) + 0.1 * rng.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                p = softmax_from_z(z)
                c = full_convolve_fft(p)
                maxc = float(c.max())
                m.fill(0.0)
                v.fill(0.0)
                adam_t = 0
                no_imp = 0

        if total_steps % 5000 == 0:
            # Periodic beta reset
            if step_frac >= 1.0:
                beta = max(1e-14, 0.5 * maxc)
                T = 10000
                total_steps = 0

    return best_p, best_score


def nesterov_subgrad(a_init, deadline, rng):
    """Nesterov-accelerated subgradient method for max(c) minimization."""
    p = np.maximum(np.asarray(a_init, dtype=np.float64).ravel(), 1e-12)
    p = p / p.sum()
    n = p.size

    def project_simplex(v):
        """Project v onto the simplex {x >= 0, sum(x) = 1}."""
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return v / s

    x = p.copy()
    y = p.copy()
    best_p = p.copy()
    best_score = float(2.0 * n * np.convolve(p, p).max())

    # Polyak step size
    best_c = float(np.convolve(p, p).max())
    lower_bound = 1.0 / (2 * n - 1)  # theoretical lower bound on max(c) when sum=1
    c_lo = best_c * 0.9  # optimistic lower bound

    gamma = 0.1
    t_step = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(y)
        k_max = int(np.argmax(c))
        c_max = float(c[k_max])

        # Subgradient of max(c) at y: 2 * (conv matrix row k_max) @ y
        # d[dy_i] = 2 * sum_j y_j * A[k_max, i] where A is conv matrix
        # This equals 2 * conv(y, e_{k_max reversed})... let's compute directly
        # Actually, c[k] = sum_i y[i]*y[k-i], so d(c[k])/d(y[i]) = y[k-i] + y[i-(k-n+1)]... 
        # Simpler: grad = 2 * (the row of the conv matrix at k_max dotted with y)
        # conv(y,y)[k] = sum_i y[i] * y[k-i]
        # d/dy[j] = y[k-j] + (if valid) y[j-(2n-2-k)]... 
        # Actually for k fixed: d(c[k])/dy[j] = y[k-j] (if 0<=k-j<n) + y[j-k+n-1]... 
        # Let me just use the correlation formula
        # c[k] = sum_i y[i]*y[k-i], d/dy[j] = y[k-j] (if 0<=k-j<n)
        # But also the "other side": when we write it as sum over all valid pairs
        # d(c[k])/dy[j] = y[k-j] if 0 <= k-j < n (from the left)
        #                 + y[j-(k-n+1)] if 0 <= j-(k-n+1) < n (from the right, when k >= n)
        # Actually the simplest: conv(y,y)[k] = sum_{i} y[i]*y[k-i]
        # d/dy[j]: the j appears as y[i] when i=j (giving y[k-j]) and as y[k-i] when k-i=j i.e. i=k-j (giving y[k-j])
        # So d/dy[j] = 2*y[k-j] if 0<=k-j<n, or y[k-j] if only one term
        # More carefully: c[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} y[i]*y[k-i]
        # d/dy[j]: j appears in the sum when i=j (if valid) giving y[k-j],
        #          and when k-i=j i.e. i=k-j (if valid) giving y[k-j]
        # If j and k-j are both in [0,n-1] and j != k-j: contribution is 2*y[k-j]
        # If j == k-j (i.e., j = k/2): contribution is y[k-j] = y[j]

        # Simpler approach using FFT:
        # The gradient of c[k_max] w.r.t. y is: 2 * (y reversed and shifted)
        # Specifically, d(c[k])/dy = 2 * flip(y) shifted appropriately

        # Let's just compute it directly for correctness:
        k = k_max
        grad = np.zeros(n)
        # c[k] = sum_i y[i]*y[k-i]
        # d/dy[j] = y[k-j] (if 0<=k-j<n) + y[k-j] (if 0<=k-j<n, from the other term)
        # = 2*y[k-j] if 0<=k-j<n and j != k-j
        # = y[j] if j == k-j (=k/2)
        # Actually: c[k] = sum_i y[i]*y[k-i]. Term j appears as y[j]*y[k-j] (when i=j)
        # and as y[i]*y[j] (when k-i=j, i.e., i=k-j).
        # If j != k-j: two terms, each contributing y[k-j], total 2*y[k-j]
        # If j == k-j: one term y[j]*y[j], derivative 2*y[j]
        # So in all cases: d/dy[j] = 2*y[k-j] if 0<=k-j<n, else 0

        if 0 <= k < n:
            # k-j in [0,n-1] means j in [k-(n-1), k]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            grad[lo:hi + 1] = 2.0 * y[k - hi:k - lo:-1]
        else:
            # k >= n: k-j in [0,n-1] means j in [k-(n-1), k] intersect [0,n-1]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] = 2.0 * y[k - hi:k - lo:-1]

        # But wait, I think I'm double counting. Let me just verify with small example.
        # y = [a, b, c], conv(y,y) = [a^2, 2ab, a^2+2bc+b^2... wait no]
        # conv([a,b,c],[a,b,c]) = [a^2, 2ab, a^2+2bc... no]
        # = [a*a, a*b+b*a, a*c+b*b+c*a, b*c+c*b, c*c]
        # = [a^2, 2ab, ac+b^2+ca, 2bc, c^2]
        # = [a^2, 2ab, 2ac+b^2, 2bc, c^2]
        # d/dc[2] = d(2ac+b^2)/dc = 2a. And k=2, k-j=2-j. For j=0: k-j=2, y[2]=c. 2*diff? 
        # Hmm, let me just use the formula d(c[k])/dy[j] = y[k-j] if valid
        # For k=2, j=0: y[2]=c, but we expect 2a. So the formula gives y[k-j]=y[2]=c, not 2a.
        # 
        # I think the issue is: c[k] = sum_i y[i]*y[k-i]
        # For k=2: c[2] = y[0]*y[2] + y[1]*y[1] + y[2]*y[0] = 2*y[0]*y[2] + y[1]^2
        # d/dy[0] = 2*y[2] (from y[0]*y[2] and y[2]*y[0])
        # d/dy[1] = 2*y[1] (from y[1]*y[1])
        # d/dy[2] = 2*y[0] (from y[0]*y[2] and y[2]*y[0])
        #
        # So d(c[k])/dy[j] = 2*y[k-j] if 0<=k-j<n AND (j+k-j=k, which is always true)
        # Wait: y[i]*y[k-i], derivative w.r.t. y[j]:
        #   - i=j: contributes y[k-j] (if 0<=k-j<n)
        #   - k-i=j, i=k-j: contributes y[k-j] = y[j... no, y[i]=y[k-j]
        #   So total: y[k-j] + y[k-j] = 2*y[k-j] if both i=j and i=k-j are valid AND j != k-j
        #   If j = k-j: only one term, contributes 2*y[j]
        #
        # So in all cases: d(c[k])/dy[j] = 2*y[k-j] if 0 <= k-j < n
        # Let me verify: k=2, j=0: 2*y[2] = 2c. But we computed d/dy[0] = 2a. MISMATCH!
        #
        # I think I'm confusing myself. Let me be very explicit.
        # c[2] = y[0]*y[2] + y[1]*y[1] + y[2]*y[0]
        # d/dy[0]: y[0] appears in term y[0]*y[2] (as y[0]) and in term y[2]*y[0] (as y[0])
        # So d/dy[0] = y[2] + y[2] = 2*y[2]? No wait, the second term is y[2]*y[0], 
        # derivative w.r.t. y[0] is y[2]. And first term y[0]*y[2], derivative w.r.t. y[0] is y[2].
        # Total: 2*y[2] = 2c. But I said d/dy[0] = 2a earlier. Let me recheck.
        # c[2] = y[0]*y[2] + y[1]*y[1] + y[2]*y[0] = 2*y[0]*y[2] + y[1]^2
        # d/dy[0] = 2*y[2] = 2c. Yes! I made an error earlier.
        # So the formula d(c[k])/dy[j] = 2*y[k-j] (if 0<=k-j<n) is correct.

        # OK so my formula was right all along. Let me just use it.
        # d(c[k])/dy[j] = 2*y[k-j] if 0 <= k-j < n, else 0

        # This can be computed as: grad = 2 * y_rev shifted
        # where y_rev[j] = y[k-j] for valid j
        # Equivalently: grad[j] = 2*y[k-j] for j in [max(0,k-n+1), min(k,n-1)]

        # Let me just recompute properly:
        grad = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            # grad[j] = 2*y[k-j] for j = lo, ..., hi
            # k-j ranges from k-lo down to k-hi
            grad[lo:hi + 1] = 2.0 * y[k - hi:k - lo + 1][::-1]

        # Step size (Polyak)
        t_step = (c_max - c_lo) / (np.dot(grad, grad) + 1e-15)
        t_step = min(t_step, 0.5 / (np.sqrt(t_step + 1.0) + 1.0))

        # Nesterov step
        x_new = y - t_step * grad
        x_new = project_simplex(x_new)
        y_new = x_new + (t_step / (t_step + 1.0)) * (x_new - x)
        y_new = project_simplex(y_new)

        # Check improvement
        c_check = full_convolve_fft(y_new)
        score = float(2.0 * n * c_check.max())
        if score < best_score:
            best_score = score
            best_p = y_new.copy()
            c_lo = min(c_lo, float(c_check.max()))

        x, y = x_new, y_new

        if t_step % 2000 == 0:
            print(f"    Subgrad step {t_step}: score={best_score:.10f}, c_lo={c_lo:.6f}")

    return best_p, best_score


def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    if n0 == 0:
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 1e-12)
    return result / result.sum()


def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    # Raised cosine variants
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.cos(np.pi * t / 2.0) ** (2 * p)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Parabolic window
    for p in [0.5, 1.0, 2.0]:
        a = (1.0 - (2.0 * t - 1.0) ** 2) ** p
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Cosine bumps
    for k in [1, 2, 3]:
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Gaussian-like windows
    for sigma in [0.2, 0.3, 0.4, 0.5]:
        a = np.exp(-((t - 0.5) / sigma) ** 2)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Double bump
    for w in [0.25, 0.33, 0.4]:
        a = np.exp(-((t - w) / 0.15) ** 2) + np.exp(-((t - (1 - w)) / 0.15) ** 2)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Random smooth profiles
    for _ in range(3):
        coeffs = rng.uniform(-0.3, 0.3, 4)
        a = 1.0
        for i, c in enumerate(coeffs, 1):
            a = a + c * np.cos(2.0 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles


def try_truncations(best_seq, best_score):
    """Try truncating the best sequence to various sub-lengths."""
    results = []
    n = best_seq.size
    candidates = [n // 2, n // 3, n // 4, 256, 128, 192, 320, 384, 640]
    for nc in candidates:
        if nc < 2 or nc >= n:
            continue
        # Try different windows into the sequence
        for start in [0, (n - nc) // 2]:
            if start + nc > n:
                continue
            sub = best_seq[start:start + nc]
            s = sub.sum()
            if s < 0.01:
                continue
            sub_norm = sub / s
            sc = compute_score(sub_norm)
            if sc < best_score:
                results.append((sub_norm, sc))
    return results


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses LP-based descent, smooth AdamW optimization, Nesterov subgradient,
    and multi-scale search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting multi-strategy search. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: LP optimization at n=512 =====
    n_main = 512
    phase1_end = min(deadline, t_start + 400.0)

    if init_seq is not None and init_seq.size != n_main:
        p_main = interpolate_profile(init_seq, n_main)
    elif init_seq is not None:
        p_main = init_seq.copy()
    else:
        p_main = make_profiles(n_main, rng)[4]  # cosine window

    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1 start: n={n_main}, score={sc:.10f}")

    # LP optimization
    lp_deadline = min(phase1_end, time.monotonic() + 300.0)
    result = lp_optimize(p_main, lp_deadline, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"After main LP: score={sc:.10f}")

    # Try a few other profiles with short LP runs
    profiles = make_profiles(n_main, rng)
    for i, p in enumerate(profiles):
        if time.monotonic() > phase1_end - 5:
            break
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.1:
            continue
        per_dl = min(phase1_end, time.monotonic() + 25.0)
        result = lp_optimize(p, per_dl, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        if sc < best_score + 0.02:
            print(f"  Profile {i}: score={sc:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # ===== Phase 2: Smooth optimization at n=512 =====
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        n_cur = best_seq.size
        print(f"Phase 2: smooth opt at n={n_cur}, best={best_score:.10f}")

        for lr in [0.05, 0.03, 0.08]:
            if time.monotonic() > phase2_end - 5:
                break
            per_dl = min(phase2_end, time.monotonic() + 80.0)
            p0 = best_seq.copy() + 0.002 * rng.normal(size=n_cur)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = smooth_optimize(p0, per_dl, rng, lr=lr)
            update(result, sc)
            print(f"  Smooth lr={lr}: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # ===== Phase 3: Try larger n=1024 =====
    n_large = 1024
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end - 20 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 3: n={n_large}, score={sc:.10f}")

        # LP at large n
        lp_dl3 = min(phase3_end, time.monotonic() + 250.0)
        result = lp_optimize(p_large, lp_dl3, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After large LP: score={sc:.10f}")

        # Smooth at large n
        if time.monotonic() < phase3_end - 30:
            per_dl = min(phase3_end, time.monotonic() + 60.0)
            result, sc = smooth_optimize(best_seq, per_dl, rng, lr=0.04)
            update(result, sc)
            print(f"  Large smooth: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ===== Phase 4: Nesterov subgradient =====
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        n_cur = best_seq.size
        print(f"Phase 4: Nesterov subgrad at n={n_cur}")
        per_dl = min(phase4_end, time.monotonic() + 200.0)
        result, sc = nesterov_subgrad(best_seq, per_dl, rng)
        update(result, sc)
        print(f"After Nesterov: score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ===== Phase 5: Try other lengths =====
    phase5_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase5_end - 10 and best_seq is not None:
        for n_try in [256, 384, 640, 768]:
            if time.monotonic() > phase5_end - 10:
                break
            p_try = interpolate_profile(best_seq, n_try)
            sc = compute_score(p_try)
            update(p_try, sc)
            if sc < best_score + 0.01:
                per_dl = min(phase5_end, time.monotonic() + 40.0)
                result = lp_optimize(p_try, per_dl, rng, verbose=False)
                sc = compute_score(result)
                update(result, sc)
                print(f"  n={n_try}: score={sc:.10f}")
            else:
                print(f"  n={n_try}: score={sc:.10f} (skipped)")

    print(f"After Phase 5: best={best_score:.10f}")

    # ===== Phase 6: Try truncations =====
    if best_seq is not None:
        truns = try_truncations(best_seq, best_score)
        for tr, sc in truns:
            update(tr, sc)
            print(f"  Truncation: n={tr.size}, score={sc:.10f}")

    # ===== Phase 7: Final polishing =====
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = best_seq.size
        print(f"Phase 7: final polish at n={n_final}")
        per_dl = min(deadline - 5, time.monotonic() + 60.0)
        p0 = best_seq.copy() + 0.001 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, sc = smooth_optimize(p0, per_dl, rng, lr=0.02, beta_init=0.05)
        update(result, sc)
        print(f"  Final smooth: score={sc:.10f}")

    # Quick LP polish if time remains
    if time.monotonic() < deadline - 15 and best_seq is not None:
        per_dl = min(deadline - 5, time.monotonic() + 30.0)
        result = lp_optimize(best_seq, per_dl, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        print(f"  Final LP: score={sc:.10f}")

    print(f"FINAL: n={best_seq.size if best_seq is not None else 0}, score={best_score:.10f}")

    # Validate
    if best_seq is not None:
        final_seq = np.maximum(best_seq, 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(512) / 512.0

        final_score = compute_score(final_seq)
        print(f"Computed final score: {final_score:.10f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(eval_score):
                print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"Could not call evaluate_sequence: {e}")

        return [float(x) for x in final_seq]
    else:
        fallback = np.ones(512) / 512.0
        return [float(x) for x in fallback]
