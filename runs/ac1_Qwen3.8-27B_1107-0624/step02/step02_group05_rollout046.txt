import numpy as np
from scipy.optimize import linprog

import math

def compute_score(f):
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_M(f):
    c = np.convolve(f, f)
    return float(c.max())


def normalize_nonneg(f):
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def make_linear_profile(n, slope=2.0):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_gaussian_bump(n, mu=0.5, sigma=0.2):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_dual_bump(n):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * (t / 0.15) ** 2) + np.exp(-0.5 * ((t - 1) / 0.15) ** 2) + 0.1
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_quintic_profile(n, a1=2.0, a3=-1.5, a5=0.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3 + a5 * u ** 5
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_random_fourier(n, rng, n_terms=6, amp=0.4):
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    for k in range(n_terms):
        freq = k + 1
        phase_c = rng.uniform(-np.pi, np.pi)
        phase_s = rng.uniform(-np.pi, np.pi)
        f = f + amp * np.cos(2 * np.pi * freq * t + phase_c)
        f = f + amp * np.sin(2 * np.pi * freq * t + phase_s)
    f = np.maximum(f, 1e-6)
    return f / f.sum()

def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def make_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp_for_g(f, M):
    n = len(f)
    c_vec = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_gradient_fft(w, f, N):
    """Compute g[j] = 2 * sum_m w[m+j] * f[m] using FFT."""
    fw = np.fft.rfft(w, N)
    ff = np.fft.rfft(f, N)
    h = np.fft.irfft(fw * np.conj(ff), N)
    return 2.0 * h[:len(f)]


def adam_optimize(f_init, deadline, rng, lr=0.05, max_iters=20000, use_subgrad=False):
    """Adam optimization in log-space with soft-max gradient."""
    n = len(f_init)
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_score(f_init)

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    z = np.log(f + 1e-30)
    z -= z.max()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = np.convolve(f, f)
    M = float(c.max())
    beta = 0.05 * M

    best_f = f.copy()
    best_M = M
    no_imp = 0
    base_lr = lr

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute soft-max weights
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -60, 60)
        ex = np.exp(x)
        s = float(ex.sum())
        if s < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        # Check if we should use exact subgradient (one peak dominates)
        w_max = float(w.max())
        if use_subgrad and w_max > 0.95:
            w_exact = np.zeros_like(c)
            w_exact[int(np.argmax(c))] = 1.0
            w = w_exact

        # Compute gradient via FFT
        g = compute_gradient_fft(w, f, N)

        # Gradient in log-space
        dot = float(np.dot(f, g))
        gz = f * (g - dot)

        if not np.all(np.isfinite(gz)):
            f = best_f.copy()
            z = np.log(f + 1e-30)
            z -= z.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            c = np.convolve(f, f)
            M = float(c.max())
            beta = 0.05 * M
            continue

        # Adam update
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        z -= z.max()
        e = np.exp(z)
        f = e / e.sum()

        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive temperature: decrease when making progress, reset when not
        if no_imp == 0 and it % 200 == 0:
            beta = max(1e-10, beta * 0.9)

        if no_imp > 200:
            beta = max(1e-10, beta * 0.7)
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.9)

        # Restart if stuck
        if no_imp > max(400, 4 * n):
            z = np.log(best_f + 1e-30)
            pert_scale = 0.3 + 0.3 * (no_imp / (10 * n))
            z += pert_scale * rng.normal(size=n)
            z -= z.max()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.05 * M)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_f, 2.0 * n * best_M

def multiplicative_update(f_init, deadline, rng, eta_init=0.1, max_iters=50000):
    """
    Mirror descent (exponential gradient) on the simplex.
    f_j <- f_j * exp(-eta * g_j) / Z
    This is equivalent to Adam in log-space without momentum.
    Often works well for this type of problem.
    """
    n = len(f_init)
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_score(f_init)

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    c = np.convolve(f, f)
    M = float(c.max())
    beta = 0.03 * M
    eta = eta_init

    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Soft-max weights
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -60, 60)
        ex = np.exp(x)
        s = float(ex.sum())
        if s < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        # Gradient
        g = compute_gradient_fft(w, f, N)
        dot = float(np.dot(f, g))
        gz = f * (g - dot)

        # Multiplicative update in log-space
        z = np.log(f + 1e-30)
        z = z - eta * (g - dot) / (f + 1e-30) * f  # This is z - eta * gz / f = z - eta * (g - dot)
        # Actually: z_new = z - eta * gz / f = z - eta * (g - dot) (since gz = f * (g-dot))
        # Wait, the mirror descent update is: z_new = z - eta * (gradient of L w.r.t. z)
        # gradient of L w.r.t. z_j = f_j * (g_j - dot(f,g))
        # So z_new = z - eta * f_j * (g_j - dot)
        # But this is what we want for the exponential family.
        # Let me just do: z_new = z - eta * (g - dot) which is the "natural" gradient step
        # Actually for mirror descent with KL divergence, the update is:
        # f_new_j = f_j * exp(-eta * g_j_normalized) / Z
        # where g_j_normalized = g_j - sum_i f_i * g_i
        # So in log-space: z_new = z - eta * (g - dot)
        
        z = z - eta * (g - dot)
        z -= z.max()
        e = np.exp(z)
        f = e / e.sum()

        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive learning rate and temperature
        if no_imp > 50:
            eta *= 0.9
            beta = max(1e-10, beta * 0.8)

        if no_imp > max(300, 3 * n):
            # Restart
            f = best_f * (1.0 + 0.3 * rng.standard_t(3, size=n))
            f = normalize_nonneg(f)
            if f is None:
                f = best_f.copy()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.03 * M)
            eta = eta_init
            no_imp = 0

    return best_f, 2.0 * n * best_M

            f_best, sc = adam_optimize(s, run_deadline, rng, lr=0.04)
        else:
            f_best, sc = multiplicative_update(s, run_deadline, rng, eta_init=0.08)

        try_update(f_best, sc)
        results_1.append((sc, f_best))

        if (i + 1) % 4 == 0:
            print(f"  After {i+1}/{n_starts}: best={best_score:.10f}, time={time.monotonic()-t_start:.0f}s")

    results_1.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.10f}")

    # === Phase 2: LP+Adam hybrid on top candidates ===
    phase2_end = min(deadline, t_start + 1150.0)
    print(f"\nPhase 2: LP+Adam hybrid, budget={phase2_end - time.monotonic():.0f}s")

    top_k = min(5, len(results_1))
    results_2 = []
    for i in range(top_k):
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break
        alloc = remaining / max(1, top_k - i)
        alloc = min(alloc, 100.0)
        if alloc < 15:
            break
        run_deadline = time.monotonic() + alloc
        f_best, sc = lp_adam_hybrid(results_1[i][1], run_deadline, rng, max_iters=3000, lp_every=80)
        try_update(f_best, sc)
        results_2.append((sc, f_best))
        print(f"  LP+Adam {i+1}/{top_k}: score={sc:.10f}, best={best_score:.10f}")

    # Combine results
    all_results = results_1 + results_2
    all_results.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best: {best_score:.10f}")

    # === Phase 3: Extend to n=512 ===
    phase3_end = min(deadline, t_start + 1550.0)
    n3 = 512
    if phase3_end - time.monotonic() > 60:
        print(f"\nPhase 3: Extension to n={n3}, budget={phase3_end - time.monotonic():.0f}s")

        results_3 = []
        top_c = min(4, len(all_results))
        for i in range(top_c):
            remaining = phase3_end - time.monotonic()
            if remaining < 30:
                break
            alloc = remaining / max(1, top_c - i)
            alloc = min(alloc, 80.0)
            if alloc < 20:
                break
            f_ext = interpolate_profile(all_results[i][1], n3)
            run_deadline = time.monotonic() + alloc

            if i % 2 == 0:
                f_best, sc = adam_optimize(f_ext, run_deadline, rng, lr=0.03)
            else:
                f_best, sc = lp_adam_hybrid(f_ext, run_deadline, rng, max_iters=2000, lp_every=100)

            try_update(f_best, sc)
            results_3.append((sc, f_best))
            print(f"  n={n3} run {i+1}/{top_c}: score={sc:.10f}, best={best_score:.10f}")

        # Also try multiplicative update at n=512
        if phase3_end - time.monotonic() > 30:
            f_ext = interpolate_profile(all_results[0][1], n3)
            run_deadline = phase3_end
            f_best, sc = multiplicative_update(f_ext, run_deadline, rng, eta_init=0.05)
            try_update(f_best, sc)
            results_3.append((sc, f_best))
            print(f"  n={n3} multiplicative: score={sc:.10f}, best={best_score:.10f}")
    else:
        print("\nPhase 3: Skipped (insufficient time)")

    # === Phase 4: Final polish ===
    if time.monotonic() < deadline - 30 and best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 4: Final polish, budget={remaining:.0f}s")
        n_f = len(best_seq)
        # Use Adam with very small steps for final polish
        f_best, sc = adam_optimize(best_seq, deadline, rng, lr=0.01, max_iters=50000)
        try_update(f_best, sc)
        print(f"  Final polish: score={sc:.10f}, best={best_score:.10f}")

        # Also try LP polish if time allows
        if time.monotonic() < deadline - 20 and n_f <= 512:
            f_best, sc = lp_adam_hybrid(best_seq, deadline, rng, max_iters=2000, lp_every=20)
            try_update(f_best, sc)
            print(f"  LP polish: score={sc:.10f}, best={best_score:.10f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = make_linear_profile(256, 1.5)
        best_score = compute_score(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}")

    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
        if np.isfinite(ev_score) and ev_score < final_score:
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
