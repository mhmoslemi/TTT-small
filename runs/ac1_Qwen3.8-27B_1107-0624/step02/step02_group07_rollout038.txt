The problem min_s max_k (Cs)[k] s.t. sum(s)=1, s>=0 is related to the theory of "minimum peak power" sequences. In signal processing, this is the "minimum peak-to-average power ratio" (PAPR) problem for aperiodic autocorrelation.

One insight: for non-negative sequences, the "energy" at the center of the convolution (lag n-1) is sum_i s[i]*s[n-1-i]. If the sequence is symmetric, this equals sum_i s[i]^2 (the energy). If the sequence is "anti-symmetric" (large at the beginning, small at the end), this can be much smaller.

```python
import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar

def compute_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def compute_conv(s):
    """Compute convolution."""
    return np.convolve(s, s)

def gradient_descent_step(s, beta, max_iters=100):
    """
    One gradient descent step with backtracking line search.
    Uses multiplicative update to preserve non-negativity.
    """
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = conv.max()
    if max_conv <= 0:
        return s.copy()
    
    # Smooth max weighting
    shifted = (conv - max_conv) / max_conv  # in [-1, 0]
    w = np.exp(beta * shifted)  # weights emphasizing the peak
    
    # Gradient computation via FFT
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    W = np.fft.rfft(w, N_fft)
    term1 = np.fft.irfft(np.conj(S) * W, N_fft)[0:n]
    term2 = np.fft.irfft(S * W, N_fft)[0:n]
    grad = term1 + term2
    grad = np.maximum(grad, 0.0)
    
    if grad.max() < 1e-15:
        return s.copy()
    
    # Normalize gradient
    grad = grad / grad.max()
    
    # Backtracking line search with multiplicative update
    current_score = compute_score(s)
    eta = 0.5
    best_s = s.copy()
    best_score = current_score
    
    for it in range(max_iters):
        s_new = s * np.exp(-eta * grad)
        s_new = np.maximum(s_new, 1e-15)
        s_new = s_new / s_new.sum()
        
        score_new = compute_score(s_new)
        if score_new < best_score:
            best_score = score_new
            best_s = s_new.copy()
        
        if score_new < current_score * (1 - 1e-4):  # Sufficient decrease
            return s_new
        
        eta *= 0.7
    
    return best_s

def single_coordinate_update(s, j):
    """
    Update a single coordinate j of s by solving the 1D subproblem.
    Minimizes max(conv(s,s)) over s[j] (with sum normalization handled separately).
    """
    n = len(s)
    conv = np.convolve(s, s)
    sj = s[j]
    
    # Precompute base values
    # When s[j] changes to x:
    # conv[k] changes for k in [j, j+n-1]
    # Specifically: new_conv[k] = conv[k] - contribution_old + contribution_new
    
    # The contribution of s[j] to conv[k]:
    # For k where 0 <= k-j < n and k-j != j: contribution = 2 * s[j] * s[k-j]
    # For k = 2j: contribution = s[j]^2
    
    lower = 0.0
    upper = max(2.0 * sj, 0.1)  # upper bound for search
    
    # Define the 1D objective
    def obj_1d(x):
        # Compute max conv when s[j] = x
        # Only need to update the affected conv values
        # Affected k values: those where j or k-j is in [0, n)
        k_min = max(0, j - (n - 1))  # actually just j (since k >= j for s[j]*s[k-j])
        # Actually, s[j] appears in conv[k] when 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_vals = range(max(0, j), min(2*n-1, j+n))
        
        max_val = 0.0
        # First, check unaffected conv values
        for k in range(2*n-1):
            if k < j or k > j + n - 1:
                if conv[k] > max_val:
                    max_val = conv[k]
        
        # Check affected conv values
        for k in k_vals:
            old_contrib = 0.0
            new_contrib = 0.0
            if k - j == j:  # k = 2j
                # s[j]*s[j] = sj^2 -> x^2
                old_contrib = sj * sj
                new_contrib = x * x
            elif 0 <= k - j < n:
                # s[j]*s[k-j] + s[k-j]*s[j] = 2*sj*s[k-j] -> 2*x*s[k-j]
                old_contrib = 2.0 * sj * s[k-j]
                new_contrib = 2.0 * x * s[k-j]
            
            new_conv_k = conv[k] - old_contrib + new_contrib
            if new_conv_k > max_val:
                max_val = new_conv_k
        
        return max_val
    
    result = minimize_scalar(obj_1d, bounds=(lower, upper), method='bounded',
                             options={'maxiter': 50, 'xatol': 1e-10})
    
    if result.success:
        s[j] = max(0.0, result.x)
    
    return s

def coordinate_descent_pass(s, n_coords=None, rng=None):
    """
    One pass of coordinate descent over all (or a subset of) coordinates.
    """
    n = len(s)
    if n_coords is None:
        indices = list(range(n))
    elif rng is not None:
        indices = rng.choice(n, size=min(n_coords, n), replace=False)
    else:
        indices = list(range(n_coords))
    
    for j in indices:
        s = single_coordinate_update(s, j)
    
    # Renormalize
    s = np.maximum(s, 0.0)
    if s.sum() > 1e-15:
        s = s / s.sum()
    
    return s

def lp_direction(s, M, time_limit=5.0):
    """
    Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M for all k, b >= 0.
    Returns b or None.
    """
    n = len(s)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A: A[k, j] = coefficient of b[j] in conv(s,b)[k]
    # conv(s,b)[k] = sum_i s[i] * b[k-i]
    # For a fixed k, b[j] appears when k-i=j, i.e., i=k-j
    # So conv(s,b)[k] = sum_j s[k-j] * b[j] (where 0 <= k-j < n)
    # A[k, j] = s[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((num_constraints, n))
    for j in range(n):
        # b[j] appears in conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] includes s[k-j]*b[j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        k_start = j
        k_end = min(j + n, num_constraints)
        A[k_start:k_end, j] = s[0:k_end - k_start]
    
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def try_parameterized(n, rng):
    """Generate diverse parameterized starting points."""
    results = []

    s = np.ones(n)

    results.append(compute_score(s), s.copy())

        s = r ** x
        results.append(compute_score(s), s.copy())
        s2 = r ** (n - 1 - x)
        results.append(compute_score(s2), s2.copy())
    
    # Two-level

            s2 = np.zeros(n)

def optimize_sequence(s, deadline, rng, use_lp=True, n_max_lp=512):
    """
    Main optimization loop for a single sequence.
    Combines gradient descent, coordinate descent, and LP.
    """
    n = len(s)
    s = np.maximum(s, 0.0)
    s = s / s.sum()
    best_s = s.copy()
    best_score = compute_score(s)
    
    beta = 5.0
    phase = 0  # 0: gradient, 1: coord descent, 2: LP
    no_improve = 0
    patience = 100
    
    while time.monotonic() < deadline:
        cur_score = compute_score(s)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_s = s.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Switch phases
        if no_improve > patience * 3:
            phase = (phase + 1) % 3
            no_improve = 0
            if phase == 0:
                # Restart with perturbation
                s = s + 0.01 * rng.standard_normal(n)
                s = np.maximum(s, 0.01)
                s = s / s.sum()
                beta = rng.choice([3.0, 5.0, 8.0, 12.0, 20.0, 30.0])
        
        if phase == 0:
            # Gradient descent
            s_new = gradient_descent_step(s, beta)
            s_new_score = compute_score(s_new)
            if s_new_score < cur_score:
                s = s_new
            else:
                s = s_new = s  # keep old
                beta = min(beta * 1.2, 60.0)
                if beta > 50:
                    # Perturb
                    s = s + 0.02 * rng.standard_normal(n)
                    s = np.maximum(s, 0.01)
                    s = s / s.sum()
                    beta = 5.0
            
        elif phase == 1:
            # Coordinate descent
            # Do a few random coordinate updates
            num_updates = min(n, 50)
            indices = rng.choice(n, size=num_updates, replace=False)
            for j in indices:
                if time.monotonic() > deadline - 5:
                    break
                s = single_coordinate_update(s, j)
            s = np.maximum(s, 0.0)
            if s.sum() > 1e-15:
                s = s / s.sum()
                
        elif phase == 2:
            # LP-based direction
            if n <= n_max_lp:
                conv = np.convolve(s, s)
                M = conv.max()
                b = lp_direction(s, M, time_limit=3.0)
                if b is not None and b.sum() > 1e-12:
                    b = b / b.sum()
                    # Line search over blend
                    best_t = 0.0
                    best_ts = cur_score
                    for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
                        cand = (1 - t) * s + t * b
                        cand = np.maximum(cand, 0.0)
                        cand = cand / cand.sum()
                        ts = compute_score(cand)
                        if ts < best_ts:
                            best_ts = ts
                            best_t = t
                    if best_t > 0:
                        s = (1 - best_t) * s + best_t * b
                        s = np.maximum(s, 0.0) / s.sum()
                else:
                    # Fallback to gradient
                    s = gradient_descent_step(s, beta)
            else:
                s = gradient_descent_step(s, beta)
        
        # Periodically increase beta
        if no_improve % 50 == 49:
            beta = min(beta + 2.0, 60.0)
    
    return best_s, best_score
```

```python
def single_coordinate_update_fast(s, j, conv, max_unaffected):
    """
    Fast single coordinate update.
    Precomputes the max of unaffected conv values.
    """
    n = len(s)
    sj = s[j]
    
    # Precompute the max of unaffected conv values
    # Affected k: j <= k <= j+n-1
    k_affected_start = j
    k_affected_end = min(j + n, 2*n)  # exclusive
    
    # For each affected k, compute the "base" (conv[k] minus s[j]'s contribution)
    # and the coefficient of x (the new value of s[j])
    # conv[k] = base[k] + coeff[k] * x + (x^2 if k=2j)
    
    bases = np.zeros(n)  # for k in affected range
    coeffs = np.zeros(n)
    has_quad = np.zeros(n, dtype=bool)
    
    idx = 0
    for k in range(k_affected_start, k_affected_end):
        if k - j == j:  # k = 2j
            # conv[k] includes s[j]^2
            bases[idx] = conv[k] - sj*sj
            coeffs[idx] = 0.0
            has_quad[idx] = True
        else:
            # conv[k] includes 2*s[j]*s[k-j]
            bases[idx] = conv[k] - 2.0*sj*s[k-j]
            coeffs[idx] = 2.0 * s[k-j]
            has_quad[idx] = False
        idx += 1
    
    num_affected = k_affected_end - k_affected_start
    
    # Now minimize: max(max_unaffected, max_i(bases[i] + coeffs[i]*x + has_quad[i]*x^2))
    
    def obj(x):
        vals = bases + coeffs * x
        vals[has_quad] += x * x
        return max(max_unaffected, np.max(vals))
    
    result = minimize_scalar(obj, bounds=(0.0, max(2.0*sj, 0.05)), method='bounded',
                             options={'maxiter': 30, 'xatol': 1e-8})
    
    new_sj = max(0.0, result.x) if result.success else sj
    
    # Update s and conv
    s[j] = new_sj
    
    # Update affected conv values
    for i in range(num_affected):
        k = k_affected_start + i
        if k - j == j:
            conv[k] = bases[i] + new_sj * new_sj
        else:
            conv[k] = bases[i] + coeffs[i] * new_sj
    
    return s, conv

def coordinate_descent_pass_fast(s, num_coords=None, rng=None):
    """Fast coordinate descent pass with precomputed convolution."""
    n = len(s)
    conv = np.convolve(s, s)
    
    if num_coords is None:
        num_coords = n
    
    # Random order
    if rng is not None:
        indices = rng.choice(n, size=min(num_coords, n), replace=False)
    else:
        indices = np.arange(n)
    
    for j in indices:
        k_start = j
        k_end = min(j + n, 2*n)
        
        # Compute max of unaffected values
        if k_start > 0:
            max_left = np.max(conv[:k_start]) if k_start > 0 else 0.0
        else:
            max_left = 0.0
        if k_end < 2*n - 1:
            max_right = np.max(conv[k_end:]) if k_end < 2*n - 1 else 0.0
        else:
            max_right = 0.0
        max_unaffected = max(max_left, max_right)
        
        s, conv = single_coordinate_update_fast(s, j, conv, max_unaffected)
    
    s = np.maximum(s, 0.0)
    if s.sum() > 1e-15:
        s = s / s.sum()
    
    return s
```

```python
"""
Optimized search for non-negative sequences minimizing 2n*max(conv(s,s))/(sum(s))^2.
Uses FFT-based gradient descent with backtracking line search, coordinate descent
refinement, LP-guided corrections, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar


def compute_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def compute_conv(s):
    """Compute full convolution."""
    return np.convolve(s, s)


def fft_gradient(s, w):
    """Compute gradient of sum_k w[k]*conv(s,s)[k] using FFT."""
    n = len(s)
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    W = np.fft.rfft(w, N_fft)
    term1 = np.fft.irfft(np.conj(S) * W, N_fft)[0:n]
    term2 = np.fft.irfft(S * W, N_fft)[0:n]
    return term1 + term2


def gradient_step_bt(s, beta, max_backtrack=50):
    """
    Gradient descent step with backtracking line search.
    Uses multiplicative update for non-negativity preservation.
    """
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = conv.max()
    if max_conv <= 0:
        return s.copy()

    # Smooth max weighting - emphasize peaks
    shifted = (conv - max_conv) / max_conv
    w = np.exp(beta * shifted)

    grad = fft_gradient(s, w)
    grad = np.maximum(grad, 0.0)
    grad_max = grad.max()
    if grad_max < 1e-15:
        return s.copy()
    grad = grad / grad_max

    current_score = compute_score(s)

    # Backtracking line search
    eta = 0.8
    best_s = s.copy()
    best_score = current_score

    for _ in range(max_backtrack):
        s_new = s * np.exp(-eta * grad)
        s_new = np.maximum(s_new, 1e-15)
        s_new = s_new / s_new.sum()
        score_new = compute_score(s_new)

        if score_new < best_score:
            best_score = score_new
            best_s = s_new.copy()

        # Armijo condition
        if score_new < current_score - 1e-10 * eta:
            return s_new
        eta *= 0.6
        if eta < 0.001:
            break

    return best_s


def coord_update_single(s, j, conv):
    """
    Update single coordinate j using 1D minimization of max(conv).
    Returns updated s and conv arrays.
    """
    n = len(s)
    sj = s[j]
    if sj < 1e-12:
        # Try small positive values
        best_x = 0.0
        best_val = compute_conv_max_after_update(s, j, 0.0, conv)
        for trial in [1e-4, 1e-3, 1e-2, 0.01, 0.05, 0.1]:
            val = compute_conv_max_after_update(s, j, trial, conv)
            if val < best_val:
                best_val = val
                best_x = trial
        if best_x > 0:
            s[j] = best_x
            update_conv_inplace(s, j, sj, best_x, conv, n)
        return s, conv

    k_start = j
    k_end = min(j + n, 2 * n)

    # Max of unaffected conv values
    max_unaffected = 0.0
    if k_start > 0:
        max_unaffected = max(max_unaffected, np.max(conv[:k_start]))
    if k_end < 2 * n - 1:
        max_unaffected = max(max_unaffected, np.max(conv[k_end:]))

    def obj_1d(x):
        return compute_conv_max_after_update(s, j, x, conv, k_start, k_end, max_unaffected)

    upper = max(3.0 * sj, 0.5)
    result = minimize_scalar(obj_1d, bounds=(0.0, upper), method='bounded',
                             options={'maxiter': 40, 'xatol': 1e-10})

    new_sj = max(0.0, result.x) if result.success else sj

    if abs(new_sj - sj) > 1e-12:
        s[j] = new_sj
        update_conv_inplace(s, j, sj, new_sj, conv, n)

    return s, conv


def compute_conv_max_after_update(s, j, x, conv, k_start=None, k_end=None, max_unaffected=None):
    """Compute max of conv after changing s[j] to x (without actually updating)."""
    n = len(s)
    sj = s[j]

    if k_start is None:
        k_start = j
        k_end = min(j + n, 2 * n)
        max_unaffected = 0.0
        if k_start > 0:
            max_unaffected = max(max_unaffected, np.max(conv[:k_start]))
        if k_end < 2 * n - 1:
            max_unaffected = max(max_unaffected, np.max(conv[k_end:]))

    max_val = max_unaffected

    for k in range(k_start, k_end):
        old_contrib = 0.0
        new_contrib = 0.0
        diff = k - j
        if diff == j:  # k = 2j
            old_contrib = sj * sj
            new_contrib = x * x
        elif 0 <= diff < n:
            old_contrib = 2.0 * sj * s[diff]
            new_contrib = 2.0 * x * s[diff]
        new_val = conv[k] - old_contrib + new_contrib
        if new_val > max_val:
            max_val = new_val

    return max_val


def update_conv_inplace(s, j, old_val, new_val, conv, n):
    """Update convolution array in place after changing s[j]."""
    k_start = j
    k_end = min(j + n, 2 * n)
    for k in range(k_start, k_end):
        diff = k - k_start  # this is k - j
        if diff == j:  # k = 2j
            conv[k] = conv[k] - old_val * old_val + new_val * new_val
        elif 0 <= diff < n:
            conv[k] = conv[k] - 2.0 * old_val * s[diff] + 2.0 * new_val * s[diff]


def coordinate_descent_pass(s, num_passes=1, rng=None):
    """Run coordinate descent passes on the sequence."""
    n = len(s)
    conv = np.convolve(s, s)

    for _ in range(num_passes):
        if rng is not None:
            indices = rng.permutation(n)
        else:
            indices = np.arange(n)

        for j in indices:
            s, conv = coord_update_single(s, j, conv)

    s = np.maximum(s, 0.0)
    if s.sum() > 1e-15:
        s = s / s.sum()
    return s


def lp_direction(s, M, time_limit=5.0):
    """
    Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M for all k, b >= 0.
    """
    n = len(s)
    num_constraints = 2 * n - 1

    A = np.zeros((num_constraints, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, num_constraints)
        A[k_start:k_end, j] = s[0:k_end - k_start]

    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    best_s = s.copy()

            mode = (mode + 1) % 3

                if beta > 60:
                    # Perturb and reset
                    s = s + 0.02 * rng.standard_normal(n)
                    s = np.maximum(s, 0.005)
                    s = s / s.sum()
                    beta = 5.0
                    no_improve = max(0, no_improve - 20)

        elif mode == 1:
            # Coordinate descent (a few passes)
            remaining_time = deadline - time.monotonic()
            if remaining_time > 2.0:
                s = coordinate_descent_pass(s, num_passes=1, rng=rng)
                sc_cd = compute_score(s)
                if sc_cd > cur_score * 1.01:
                    # Bad move, revert
                    s = best_s.copy()
                no_improve = max(0, no_improve - 30)
            else:
                mode = 0

        elif mode == 2:
            # LP-based direction
            if n <= 512:
                conv = np.convolve(s, s)
                M = conv.max()
                b = lp_direction(s, M, time_limit=3.0)
                if b is not None and b.sum() > 1e-12:
                    b = b / b.sum()
                    best_t = 0.0
                    best_ts = cur_score
                    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                        cand = (1.0 - t) * s + t * b
                        cand = np.maximum(cand, 0.0)
                        if cand.sum() < 1e-12:
                            continue
                        cand = cand / cand.sum()
                        ts = compute_score(cand)
                        if ts < best_ts:
                            best_ts = ts
                            best_t = t
                    if best_t > 0:
                        s = (1.0 - best_t) * s + best_t * b
                        s = np.maximum(s, 0.0) / s.sum()
                        no_improve = max(0, no_improve - 40)
                    else:
                        mode = 0
                else:
                    mode = 0
            else:
                mode = 0

        # Periodic beta increase
        if no_improve % 30 == 29:
            beta = min(beta + 1.0, 80.0)

    return best_s, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for non-negative sequence minimizing 2n*max(conv(s,s))/(sum(s))^2.
    Combines FFT gradient descent with backtracking, coordinate descent,
    LP-guided corrections, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0

    global_best_score = float('inf')
    global_best_seq = np.ones(256)

    # Load height_sequence_1 if available
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            print(f"Loaded height_sequence_1, length={h.size}")
    except Exception:
        pass

    # Sequence lengths to explore
    lengths = [256, 320, 384, 448, 512]

    # Time allocation: give more time to promising lengths
    # Start with lengths close to the known good one (384)
    lengths_sorted = [384, 320, 448, 512, 256]

    for n in lengths_sorted:
        if time.monotonic() > deadline - 90:
            break
        remaining = deadline - time.monotonic()
        print(f"\n=== n={n}, remaining={remaining:.0f}s ===")

        # Generate candidates
        candidates = try_parameterized(n, rng)

        # Add resized height_sequence_1
        if hseq is not None:
            x_old = np.linspace(0.0, 1.0, hseq.size)
            x_new = np.linspace(0.0, 1.0, n)
            h_r = np.interp(x_new, x_old, hseq)
            h_r = np.maximum(h_r, 0.0)
            if h_r.sum() > 0.01:
                h_r = h_r / h_r.sum()
                sc_h = compute_score(h_r)
                candidates.insert(0, (sc_h, h_r))
                print(f"  Resized hseq: score={sc_h:.6f}")

        # Direct use if length matches
        if hseq is not None and hseq.size == n:
            h_norm = np.maximum(hseq, 0.0) / hseq.sum()
            sc_direct = compute_score(h_norm)
            candidates.insert(0, (sc_direct, h_norm.copy()))
            print(f"  Direct hseq: score={sc_direct:.6f}")

        candidates.sort(key=lambda t: t[0])

        # Optimize top candidates
        num_cands = min(3, len(candidates))
        time_per_cand = max(30.0, (remaining - 90) / (num_cands + 1))

        for i in range(num_cands):
            if time.monotonic() > deadline - 60:
                break
            sc0, s0 = candidates[i]
            print(f"  Candidate {i}: initial score={sc0:.6f}")

            cand_deadline = time.monotonic() + min(time_per_cand, deadline - 60)
            use_lp = (n <= 512)

            best_s, best_sc = optimize_single(s0, cand_deadline, rng, use_lp=use_lp)

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** New global best: {global_best_score:.6f} (n={n})")

        # Quick random restarts
        restarts = 0
        while time.monotonic() < deadline - 60 and restarts < 2:
            s_r = rng.random(n) + 0.05
            s_r = s_r / s_r.sum()
            rd = min(time.monotonic() + 30, deadline - 60)
            best_s, best_sc = optimize_single(s_r, rd, rng, use_lp=(n <= 384))
            restarts += 1
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Restart {restarts}: {global_best_score:.6f}")

    # Final refinement phase
    if time.monotonic() < deadline - 15:
        print(f"\n=== Final refinement: score={global_best_score:.6f}, n={len(global_best_seq)} ===")
        n = len(global_best_seq)

        # Run coordinate descent as final polish
        s_polish = global_best_seq.copy()
        for _ in range(3):
            if time.monotonic() > deadline - 10:
                break
            s_polish = coordinate_descent_pass(s_polish, num_passes=1, rng=rng)
            sc_polish = compute_score(s_polish)
            if sc_polish < global_best_score:
                global_best_score = sc_polish
                global_best_seq = s_polish.copy()
                print(f"  CD polish: {global_best_score:.6f}")

        # One more optimization round on the best
        if time.monotonic() < deadline - 10:
            best_s, best_sc = optimize_single(global_best_seq, deadline - 5, rng,
                                              use_lp=(n <= 512))
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Final: {global_best_score:.6f}")

    # Verify with actual evaluator
    final_score = float(evaluate_sequence(global_best_seq.tolist()))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(global_best_seq)}")

    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result
```
