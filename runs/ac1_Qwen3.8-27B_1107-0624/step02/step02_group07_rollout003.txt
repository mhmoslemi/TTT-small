def lp_move(s, M_target, time_limit=3.0):
    """Solve LP to find direction b that reduces max conv."""
    n = len(s)
    # LP: maximize sum(b) subject to conv(s, b)[k] <= M_target, b >= 0
    # conv(s, b)[k] = sum_m s[m] * b[k-m]
    # For fixed s, this is linear in b.
    # Variable: b (length n)
    # Constraint: for each k = 0, ..., 2n-2: sum_{m} s[m] * b[k-m] <= M_target
    #   where the sum is over m such that 0<=m<n and 0<=k-m<n.
    
    # Build constraint matrix
    # A_ub[k, j] = coefficient of b[j] in constraint k
    # b[j] appears in conv[k] when k-m = j, i.e., m = k-j.
    # So A_ub[k, j] = s[k-j] if 0 <= k-j < n, else 0.
    
    # This is a (2n-1) x n matrix.
    # A_ub[k, j] = s[k-j] if 0 <= k-j <= n-1, else 0
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                A_ub[k, j] = s[m]
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M_target)

            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None
