```python
from scipy.optimize import linprog

def lp_direction(f, M):
    """Find g >= 0 that maximizes sum(g) subject to conv(f,g) <= M."""
    n = len(f)
    # conv(f,g)[i] = sum_k f[k]*g[i-k] (valid range)
    # This is a linear function of g
    # We want to maximize sum(g) = 1^T g
    # Subject to: A @ g <= M (where A[i,j] = f contribution to lag i from position j)
    #             g >= 0
    
    # Build A matrix (2n-1 x n)
    # A[i,j] = f[i-j] if 0 <= i-j < n, else 0
    # Actually, conv(f,g)[i] = sum_j f[i-j] * g[j] where 0 <= i-j < n and 0 <= j < n
    # So A[i,j] = f[i-j] if max(0, i-n+1) <= i-j <= min(i, n-1) ... 
    # More simply: A[i,j] = f[k] where k = i-j, if 0 <= k < n and 0 <= j < n
    
    A = np.zeros((2*n-1, n))
    for i in range(2*n-1):
        for j in range(n):
            k = i - j
            if 0 <= k < n:
                A[i,j] = f[k]
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # A @ g <= M (vector of M's)
    b = np.full(2*n-1, M)
    # g >= 0 => bounds = (0, None)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return -res.x  # negate because we minimized
    return None
```

```python
"""
LP-polished optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses soft-max gradient descent for initial convergence, then LP-based
active set method for precise local optimization. The LP finds the best
non-negative direction g that minimizes max conv(f,g), enabling
guaranteed progress at each blend step.
"""
import numpy as np
from scipy.optimize import linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = _full_convolve(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def _true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return a
    return a / s


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if s < 1e-30 or not np.isfinite(s):
        p = np.ones_like(e) / e.size
    else:
        p = p / s
    return p


def _soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _corr_weights(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _grad_descent(p0, deadline, rng, lr=0.06, beta_frac=0.12, beta_decay=0.72,
                  reg=1e-7, no_imp_limit=2000):
    """Adam gradient descent in logit space with soft-max surrogate."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-14)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = _soft_max_weights(c, beta)
        gp = 2.0 * _corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0); v.fill(0); adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); adam_t = 0
            continue

        z = np.clip(z, -50, 50)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.92)

        if adam_t % 50 == 0:
            beta = max(1e-12, beta * beta_decay)

        if no_imp > no_imp_limit:
            z = best_z + 0.2 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m.fill(0); v.fill(0); adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _build_conv_matrix(f, active_lags=None):
    """Build the constraint matrix for conv(f, g) at specified lags.
    Returns A (len(active_lags) x n) where A[i,j] = f contribution.
    """
    n = len(f)
    if active_lags is None:
        active_lags = np.arange(2 * n - 1)
    k = len(active_lags)
    A = np.zeros((k, n))
    for idx, i in enumerate(active_lags):
        i = int(i)
        j_lo = max(0, i - n + 1)
        j_hi = min(i, n - 1)
        if j_lo <= j_hi:
            A[idx, j_lo:j_hi+1] = f[i - j_hi:i - j_lo + 1][::-1]
    return A


def _lp_polish(f, deadline, rng, top_k=30, max_iters=500, min_alpha=0.01):
    """LP-based active set polishing.
    At each step, find g>=0, sum(g)=1 minimizing max conv(f,g) using an LP
    with only the top-k most constrained lags. Blend and repeat.
    """
    n = len(f)
    p = np.maximum(f, 0.0).copy()
    p = p / p.sum()

    c = _full_convolve(p)
    M = float(c.max())
    best_p = p.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(p)
        M = float(c.max())

        # Get active lags: top k where conv value is close to max
        if top_k < 2 * n - 1:
            sorted_idx = np.argsort(c)[::-1]
            # Include lags with value > 0.85 * M, up to top_k
            threshold = 0.85 * M
            active = sorted_idx[np.maximum(c[sorted_idx], 0) > threshold]
            active = active[:top_k]
            # Always include the global max
            max_idx = int(np.argmax(c))
            if max_idx not in active:
                active = np.append(active, max_idx)
        else:
            active = np.arange(2 * n - 1)

        # Build LP: min t s.t. conv(f,g)[i] <= t for i in active, g>=0, sum(g)=1
        A_conv = _build_conv_matrix(p, active)
        n_active = len(active)

        # Variables: g (n) + t (1) => n+1 vars
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t

        # Constraints: A_conv @ g - t <= 0
        A_ub = np.hstack([A_conv, -np.ones((n_active, 1))])
        b_ub = np.zeros(n_active)

        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'time_limit': 5})
        except Exception:
            break

        if not res.success:
            break

        g = res.x[:n]
        t_opt = res.x[n]

        if t_opt >= M * 0.9999:
            no_imp += 1
        else:
            no_imp = 0

        # Try blending with various alphas
        best_alpha = 0
        best_M_trial = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            p_trial = (1 - alpha) * p + alpha * g
            p_trial = np.maximum(p_trial, 0)
            s_trial = float(p_trial.sum())
            if s_trial < 1e-10:
                continue
            p_trial = p_trial / s_trial
            c_trial = _full_convolve(p_trial)
            M_trial = float(c_trial.max())
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_alpha = alpha

        if best_alpha > 0 and best_M_trial < best_M - 1e-12:
            p = (1 - best_alpha) * p + best_alpha * g
            p = np.maximum(p, 0)
            p = p / p.sum()
            c = _full_convolve(p)
            M = float(c.max())
            if M < best_M - 1e-14:
                best_M = M
                best_p = p.copy()

        if no_imp > 30:
            # Perturb and restart
            z = np.log(np.maximum(p, 1e-14))
            z -= z.mean()
            z = z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            M = float(c.max())
            no_imp = 0

        if best_M < 2.0 / n * 1.001:
            break  # Essentially at theoretical bound

    return best_p, float(2.0 * n * best_M)

            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_alpha = alpha

        if best_alpha > 0 and best_M_trial < best_M - 1e-14:
            p = (1 - best_alpha) * p + best_alpha * g
            p = np.maximum(p, 0)
            p = p / p.sum()
            c = _full_convolve(p)
            M = float(c.max())
            if M < best_M:
                best_M = M
                best_p = p.copy()

        if no_imp > 50:
            z = np.log(np.maximum(p, 1e-14))
            z -= z.mean()
            noise_scale = 0.05 * (1 + no_imp / 50.0)
            z = z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            M = float(c.max())
            no_imp = 0
            adaptive_k = top_k

        if it % 100 == 0 and it > 0:
            print(f"    LP iter {it}: M={M:.10f}, best_M={best_M:.10f}, k={adaptive_k}")

    return best_p, float(2.0 * n * best_M)


def _make_profile(n, shape='linear', param=1.5):
    t = np.linspace(0.0, 1.0, n)
    if shape == 'linear':
        a = 1.0 + param * (t - 0.5)
    elif shape == 'quad':
        u = 2 * t - 1
        a = 1.0 + param * u + 0.3 * param * u * u
    elif shape == 'cubic':
        u = 2 * t - 1
        a = 1.0 + param * u - 0.5 * param * u ** 3
    elif shape == 'sigmoid':
        a = 1.0 + param * (np.tanh(3 * (t - 0.5)) / np.tanh(1.5))
    elif shape == 'uniform':
        a = np.ones(n)
    else:
        a = 1.0 + param * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Gradient descent from multiple initializations
    Phase 2: LP-based active set polishing (key innovation)
    Phase 3: Multi-restart combining gradient + LP
    Phase 4: Final fine-tuning
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={p.size}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={parent.size}, score={_score_normalized(parent):.10f}")
    else:
        print("No parent available, using profiles")

    # --- Phase 1: Initial candidates + gradient descent ---
    print(f"\n=== Phase 1: Gradient Descent ===")
    n_target = 1024

    # Build initial candidates
    candidates = []

    if parent is not None:
        p_par = _interp_profile(parent, n_target)
        if p_par is not None:
            candidates.append(('parent', p_par))
        # Also try parent at its native length if reasonable
        if 128 <= parent.size <= 2048:
            candidates.append(('parent_native', parent.copy()))

    # Various profiles at target length
    profiles = [
        ('lin1.0', _make_profile(n_target, 'linear', 1.0)),
        ('lin1.5', _make_profile(n_target, 'linear', 1.5)),
        ('lin2.0', _make_profile(n_target, 'linear', 2.0)),
        ('lin2.5', _make_profile(n_target, 'linear', 2.5)),
        ('quad1.0', _make_profile(n_target, 'quad', 1.0)),
        ('cubic1.5', _make_profile(n_target, 'cubic', 1.5)),
        ('sigmoid1.5', _make_profile(n_target, 'sigmoid', 1.5)),
    ]
    candidates.extend(profiles)

    # Score all
    scored = []
    for label, p in candidates:
        sc = _score_normalized(p)
        scored.append((sc, label, p))
        if best_seq is None or sc < best_score:
            update_best(p, sc, f"init {label}")

    scored.sort(key=lambda x: x[0])
    print(f"  Phase 1 init: best={best_score:.10f}")

    # Gradient descent on top candidates
    phase1_end = min(deadline, t_start + 300.0)
    top_n = min(4, len(scored))
    for i in range(top_n):
        if time.monotonic() > phase1_end:
            break
        sc, label, p = scored[i]
        per_end = min(phase1_end, time.monotonic() + 70.0)
        p_best, sc_best = _grad_descent(p, per_end, rng, lr=0.06, beta_frac=0.12)
        update_best(p_best, sc_best, f"gd {label}")
        scored[i] = (sc_best, label, p_best)

    scored.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: LP Polishing (key innovation) ---
    print(f"\n=== Phase 2: LP Polishing ===")
    phase2_end = min(deadline, t_start + 1000.0)

    if best_seq is not None and time.monotonic() < phase2_end:
        # Work at the best sequence's length
        p_work = best_seq.copy()
        n_work = p_work.size
        print(f"  LP polishing at n={n_work}, starting score={best_score:.10f}")

        # First a quick LP polish
        per_end = min(phase2_end, time.monotonic() + 200.0)
        p_lp, sc_lp = _lp_polish(p_work, per_end, rng, top_k=30, max_iters=300)
        update_best(p_lp, sc_lp, "LP quick")

        # Then a longer, more thorough LP polish with adaptive active set
        if time.monotonic() < phase2_end:
            per_end2 = phase2_end
            p_lp2, sc_lp2 = _lp_polish_full(p_work, per_end2, rng, top_k=50, max_iters=50000)
            update_best(p_lp2, sc_lp2, "LP full")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Multi-restart ---
    print(f"\n=== Phase 3: Multi-restart ===")
    phase3_end = min(deadline, t_start + 1500.0)

    if best_seq is not None and time.monotonic() < phase3_end:
        n_best = best_seq.size
        n_restarts = 4
        remaining = max(0, phase3_end - time.monotonic())
        per_restart = remaining / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase3_end:
                break
            # Perturb best sequence
            noise_scale = 0.1 + 0.1 * i
            z = np.log(np.maximum(best_seq, 1e-14))
            z -= z.mean()
            z = z + noise_scale * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_start = _softmax_from_z(z)

            # Quick gradient descent
            gd_end = min(phase3_end, time.monotonic() + per_restart * 0.4)
            p_gd, sc_gd = _grad_descent(p_start, gd_end, rng, lr=0.05)
            update_best(p_gd, sc_gd, f"restart{i}_gd")

            # Then LP polish
            lp_end = min(phase3_end, time.monotonic() + per_restart * 0.55)
            if time.monotonic() < lp_end:
                p_lp, sc_lp = _lp_polish(p_gd, lp_end, rng, top_k=40, max_iters=200)
                update_best(p_lp, sc_lp, f"restart{i}_lp")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Try different lengths ---
    print(f"\n=== Phase 4: Length exploration ===")
    phase4_end = min(deadline, t_start + 1700.0)

    if best_seq is not None and time.monotonic() < phase4_end:
        # Try upscaling to 2048
        for target_n in [2048]:
            if time.monotonic() > phase4_end:
                break
            p_up = _interp_profile(best_seq, target_n)
            if p_up is None:
                continue
            sc_up = _score_normalized(p_up)
            update_best(p_up, sc_up, f"upscale_{target_n}")

            # Quick gradient + LP at new length
            remaining = max(0, phase4_end - time.monotonic())
            if remaining > 60:
                gd_end = time.monotonic() + remaining * 0.4
                p_gd, sc_gd = _grad_descent(p_up, gd_end, rng, lr=0.04)
                update_best(p_gd, sc_gd, f"upscale_gd_{target_n}")

                if time.monotonic() < phase4_end:
                    lp_end = phase4_end
                    p_lp, sc_lp = _lp_polish(p_gd, lp_end, rng, top_k=50, max_iters=1000)
                    update_best(p_lp, sc_lp, f"upscale_lp_{target_n}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Final fine LP polish on absolute best ---
    print(f"\n=== Phase 5: Final Polish ===")
    phase5_end = deadline

    if best_seq is not None and time.monotonic() < phase5_end:
        n_fin = best_seq.size
        remaining = max(0, phase5_end - time.monotonic())
        if remaining > 30:
            print(f"  Final LP polish, n={n_fin}, remaining={remaining:.1f}s")
            p_fin, sc_fin = _lp_polish_full(best_seq, phase5_end, rng,
                                             top_k=80, max_iters=100000)
            update_best(p_fin, sc_fin, "final_lp")

    # --- Final validation ---
    print(f"\n=== Final Validation ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s

        # Exact score
        n_fin = final.size
        if n_fin <= 2048:
            sc_exact = _true_score(final)
            print(f"  Exact score: {sc_exact:.10f}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(sc_eval):
                print(f"  Evaluator score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

        print(f"  FINAL: score={best_score:.10f}, n={final.size}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(1024, 'linear', 1.5)
    print(f"  FALLBACK: n={1024}, score={_score_normalized(fallback):.10f}")
    return [float(x) for x in fallback]
```
