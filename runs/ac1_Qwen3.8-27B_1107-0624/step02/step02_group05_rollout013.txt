import numpy as np
import time
from scipy.optimize import minimize, linprog

"""
Multi-phase optimization for minimizing 2*n*max(convolve(f,f))/sum(f)^2.
Uses soft-max gradient descent (Adam) in logit-space, L-BFGS polishing,
and multi-scale exploration across different sequence lengths.
Exploits FFT-based gradient computation for efficiency at large n.
"""


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_score_fn(f):
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def soft_max_value_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def correlation_weights_fft(w, f):
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(f[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def objective_fn(z, beta):
    n = len(z)
    f = softmax_fn(z)
    c = np.convolve(f, f)
    val, _ = soft_max_value_weights(c, beta)
    return 2.0 * n * val


def gradient_fn(z, beta):
    n = len(z)
    f = softmax_fn(z)
    c = np.convolve(f, f)
    _, w = soft_max_value_weights(c, beta)
    grad_f = correlation_weights_fft(w, f)
    dot = float(np.dot(f, grad_f))
    grad_z = f * (grad_f - dot)
    grad_z -= grad_z.mean()
    return 2.0 * n * grad_z


def make_linear_profile(n, slope):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_quintic_profile(n, a1, a3, a5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3 + a5 * u ** 5
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_gaussian_bump(n, mu, sigma):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_fourier_profile(n, rng, num_harmonics=6):
    t = np.linspace(0.0, 1.0, n)
    f = np.ones(n)
    for k in range(1, num_harmonics + 1):
        amp = rng.uniform(0.1, 0.6) / k
        phase = rng.uniform(0, 2 * np.pi)
        f = f + amp * np.cos(2 * np.pi * k * t + phase)
    f = np.maximum(f, 1e-8)
    return f / f.sum()


def make_sparse_profile(n, rng, density=0.5):
    f = np.zeros(n)
    mask = rng.random(n) < density
    f[mask] = 1.0 / np.sqrt(mask.sum())
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def to_logit(f):
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    z = np.log(f)
    z -= z.mean()
    return z


def adam_optimize(z_init, deadline, beta_init, lr, max_iters, seed):
    rng = np.random.default_rng(seed)
    n = len(z_init)
    z = z_init.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    beta = beta_init
    base_lr = lr

    f = softmax_fn(z)
    c = np.convolve(f, f)
    M = float(c.max())
    best_score = 2.0 * n * M
    best_z = z.copy()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        val, w = soft_max_value_weights(c, beta)
        grad_f = correlation_weights_fft(w, f)
        dot = float(np.dot(f, grad_f))
        grad_z = f * (grad_f - dot)
        grad_z -= grad_z.mean()
        score = 2.0 * n * val

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            f = softmax_fn(z)
            c = np.convolve(f, f)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            z = best_z.copy()
            f = softmax_fn(z)
            c = np.convolve(f, f)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z_new = np.clip(z_new, -60, 60)
        z_new -= z_new.mean()
        f_new = softmax_fn(z_new)
        c_new = np.convolve(f_new, f_new)
        M_new = float(c_new.max())
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_z = z_new.copy()
            z = z_new
            f = f_new
            c = c_new
            no_imp = 0
        else:
            z = z_new
            f = f_new
            c = c_new
            no_imp += 1

        if adam_t % 80 == 0:
            beta = max(1e-12, beta * 0.8)

        if no_imp > 0 and no_imp % 200 == 0:
            lr = max(0.02, lr * 0.92)

        if no_imp > max(400, 3 * n):
            z = 0.6 * best_z + 0.4 * rng.standard_normal(n)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            f = softmax_fn(z)
            c = np.convolve(f, f)
            beta = max(0.1 * c.max(), 1e-6)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    f_best = softmax_fn(best_z)
    c_best = np.convolve(f_best, f_best)
    final_score = 2.0 * n * float(c_best.max())
    return best_z, f_best, final_score


def lbfgs_polish(z_init, beta, maxiter, gtol):
    n = len(z_init)
    z0 = z_init.copy()
    z0 -= z0.mean()
    z0 = np.clip(z0, -50, 50)

    try:
        result = minimize(
            objective_fn, z0, method='L-BFGS-B',
            jac=gradient_fn, args=(beta,),
            options={'maxiter': maxiter, 'gtol': gtol, 'maxls': 50,
                     'factr': 1e7}
        )
        z_out = result.x
        z_out = np.clip(z_out, -60, 60)
        z_out -= z_out.mean()
        f_out = softmax_fn(z_out)
        c_out = np.convolve(f_out, f_out)
        sc = 2.0 * n * float(c_out.max())
        return z_out, f_out, sc
    except Exception:
        f_out = softmax_fn(z0)
        c_out = np.convolve(f_out, f_out)
        sc = 2.0 * n * float(c_out.max())
        return z0, f_out, sc


def lbfgs_polish_multi_beta(z_init, deadline, max_beta, min_beta, n_betas=5):
    n = len(z_init)
    best_z = z_init.copy()
    best_f = softmax_fn(z_init)
    c_init = np.convolve(best_f, best_f)
    best_score = 2.0 * n * float(c_init.max())

    betas = np.geomspace(max_beta, min_beta, n_betas)
    for beta in betas:
        if time.monotonic() > deadline:
            break
        maxiter = min(100, int(max(10, (deadline - time.monotonic()) / 0.5)))
        z_new, f_new, sc = lbfgs_polish(best_z, beta, maxiter, 1e-7)
        if sc < best_score:
            best_score = sc
            best_z = z_new
            best_f = f_new

    return best_z, best_f, best_score


def generate_diverse_starts(n, rng, count=20):
    starts = []
    # Polynomials
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        starts.append(make_linear_profile(n, slope))
    starts.append(make_cubic_profile(n, 1.0, -0.5))
    starts.append(make_cubic_profile(n, 1.5, -1.0))
    starts.append(make_cubic_profile(n, 2.0, -1.5))
    starts.append(make_cubic_profile(n, 2.5, -2.0))
    starts.append(make_cubic_profile(n, 3.0, -2.5))
    starts.append(make_quintic_profile(n, 1.0, -0.5, 0.2))
    starts.append(make_quintic_profile(n, 2.0, -1.5, 0.5))
    starts.append(make_quintic_profile(n, 3.0, -2.5, 1.0))
    # Bumps
    starts.append(make_gaussian_bump(n, 0.5, 0.3))
    starts.append(make_gaussian_bump(n, 0.3, 0.15))
    starts.append(make_gaussian_bump(n, 0.7, 0.15))
    # Fourier
    for _ in range(count):
        starts.append(make_fourier_profile(n, rng, num_harmonics=rng.integers(3, 8)))
    # Sparse
    for dens in [0.3, 0.5, 0.7]:
        starts.append(make_sparse_profile(n, rng, density=dens))
    return starts
