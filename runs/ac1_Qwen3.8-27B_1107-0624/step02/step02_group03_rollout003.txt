4. If not, remove violating indices from `S` and repeat

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines Lagrangian dual method (multiplicative weight ascent), FFT-accelerated
gradient computation, soft-argmax Adam polish, and multi-resolution search
with diverse initializations.

import numpy as np
import time
from scipy.optimize import minimize as sp_minimize


def score_func(a):
    """Compute the objective score."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def hankel_matvec_fft(lam, a):
    """Compute B @ a where B[i,j] = lam[i+j], using FFT.
    lam has length 2*n-1, a has length n. Result has length n.
    """
    n = len(a)
    a_rev = a[::-1]
    N = 1
    while N < 3 * n:
        N *= 2
    fa = np.fft.rfft(a_rev, N)
    fl = np.fft.rfft(lam, N)
    conv = np.fft.irfft(fa * fl, N)
    return conv[n - 1:2 * n - 2]


def hankel_matvec_direct(lam, a):
    """Direct computation for verification."""
    n = len(a)
    result = np.zeros(n)
    for i in range(n):
        result[i] = np.dot(lam[i:i + n], a)
    return result


def lagrangian_solve_inner(a_init, lam, max_inner_iters=500, lr_init=0.01):
    """Solve inner QP: min a^T B a s.t. sum(a)=1, a>=0, where B[i,j]=lam[i+j].
    Uses FISTA (accelerated proximal gradient) with L2 regularization for stability.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = normalize(a)
    if a is None:
        return a_init.copy()
    a_old = a.copy()
    t = 1.0
    lr = lr_init
    best_a = a.copy()
    best_val = np.inf

    for it in range(max_inner_iters):
        # Compute gradient: 2 * B @ a
        ba = hankel_matvec_fft(lam, a)
        grad = 2.0 * ba

        # FISTA step
        a_new = a_old - lr * grad
        # Projection onto simplex {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        a_old = a_new + ((t - 1.0) / t_new) * (a_new - a)
        a = a_new
        t = t_new

        c = np.convolve(a, a)
        val = float(np.dot(lam, c))
        if val < best_val:
            best_val = val
            best_a = a.copy()

            # Reduce learning rate if oscillating
            lr *= 0.95

    return best_a


def multiplicative_weight_update(c, lam, eta):
    """Update dual variables: lam_k <- lam_k * exp(eta * c[k]) / norm."""
    # Use log-space for numerical stability
    log_lam = np.log(lam + 1e-30)
    log_lam += eta * c
    # Normalize in log-space
    log_lam -= log_lam.max()
    lam_new = np.exp(log_lam)
    lam_new = lam_new / lam_new.sum()
    return lam_new

def soft_argmax_gradient(c, beta):
    """Compute soft-argmax weights for the max of c."""
    c_shifted = c - c.max()
    x = c_shifted / beta
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    return w


def gradient_soft_max(c, a, beta):
    """Compute gradient of soft max(c) w.r.t. a."""
    n = len(a)
    w = soft_argmax_gradient(c, beta)
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (with bounds)
    # This is a correlation: (w * a_rev)[i + n - 1]
    a_rev = a[::-1]
    N = 1
    while N < 3 * n:
        N *= 2
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a_rev, N)
    corr = np.fft.irfft(fw * fa, N)
    return 2.0 * corr[n - 1:2 * n - 2]

    a0 = a0 / a0.sum()

    z = np.log(a0)
    z -= z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * float(c.max())
    beta = 0.1 * float(c.max())
    base_lr = lr
    no_imp = 0

    while time.monotonic() < deadline:
        it += 1
        p = np.exp(z - z.max())
        p = p / p.sum()

        max_c = float(c.max())
        beta = max(1e-15, beta * 0.999)

        grad_p = gradient_soft_max(c, p, beta)
        # Add small regularization
        grad_p += 1e-10 * p

        # Logit-space gradient
        dot = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)

        p_new = p_new / p_new.sum()
        c_new = np.convolve(p_new, p_new)

            best_z = z.copy()
            no_imp = 0

            m.fill(0)

def make_gaussian_profile(n, sigma_frac):
    """Gaussian profile centered in the middle."""
    center = (n - 1) / 2.0
    x = np.arange(n) - center
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    return normalize(a)


def make_parabolic_profile(n, power):
    """Parabolic window: (1 - (2t-1)^2)^power."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.maximum(1.0 - u ** 2, 0.0) ** power
    return normalize(a)


def make_cosine_profile(n, k, amplitude):
    """1 + amplitude * cos(2*pi*k*i/n) window."""
    i = np.arange(n)
    a = 1.0 + amplitude * np.cos(2.0 * np.pi * k * i / n)
    return normalize(a)


def make_multicomp_profile(n, freqs, amps):
    """Sum of cosines: 1 + sum amps[k]*cos(2*pi*freqs[k]*i/n)."""
    i = np.arange(n)
    a = np.ones(n)
    for f, amp in zip(freqs, amps):
        a += amp * np.cos(2.0 * np.pi * f * i / n)
    return normalize(a)


def random_smooth_sequence(n, rng, smoothness=3):
    """Generate a random smooth non-negative sequence."""
    # Start with random values and smooth them
    base = rng.random(n)
    # Apply Gaussian smoothing
    kernel_size = smoothness
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 2.0)) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(base, kernel, mode='same')
    return normalize(smoothed)


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        a = a.copy()
        return normalize(a)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def coordinate_descent(a_init, deadline, verbose=True):

            a[i] = max(a[i] * 1.5, 1e-15)
            a = normalize(a)
            sc2 = score_func(a)

            a[i] = max(a[i] + 0.01, 1e-15)
            a = normalize(a)
            sc3 = score_func(a)

            a[i] = max(a[i] - 0.01, 1e-15)
            a = normalize(a)
            sc4 = score_func(a)

            # Pick the best perturbation
