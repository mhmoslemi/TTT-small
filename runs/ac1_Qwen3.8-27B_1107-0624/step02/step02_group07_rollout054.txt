```python
def spectral_direction(s, target_ratio=0.999):
    """Find a direction δ that reduces the peak of conv(s,s)."""
    n = len(s)
    conv = np.convolve(s, s)
    peak = conv.max()
    M = peak * target_ratio
    
    # We want conv(s, δ)[k] <= (M - conv[k]) / 2
    # For the "critical" positions (where conv[k] > M), we need conv(s,δ)[k] < 0
    # Excess at position k:
    excess = np.maximum(0, conv - M)  # How much each position exceeds M
    
    # We want to reduce these. In the frequency domain:
    # conv(s, δ) ≈ ifft(FFT(s) * FFT(δ)) (circular)
    # We want ifft(FFT(s) * FFT(δ))[k] ≈ -excess[k] (for the excess positions)
    
    N_fft = 1
    while N_fft < 2 * (2*n - 1):
        N_fft *= 2
    
    S = np.fft.rfft(s, N_fft)
    E = np.fft.rfft(excess, N_fft)
    
    # D = -E / S (element-wise)
    with np.errstate(divide='ignore', invalid='ignore'):
        D = np.where(np.abs(S) > 1e-10, -E / S, 0)
    
    delta = np.fft.irfft(D, N_fft)[:n]
    
    return delta, peak
```

Then the update is: s_new = s + alpha * delta, with alpha chosen by line search.

This should give a good descent direction for reducing the peak.

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def compute_gradient_softmax(s, tau):
    """Compute gradient of soft-max of conv(s,s) w.r.t. s using FFT."""
    n = len(s)
    conv = np.convolve(s, s)
    # Soft-max weights
    conv_shifted = tau * (conv - conv.max())
    p = np.exp(conv_shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        return np.zeros(n)
    p = p / p_sum
    # Gradient: grad[i] = 2 * sum_k p[k] * s[k-i] (valid range only)
    # = 2 * (p cross-corr s)[i]
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    P = np.fft.rfft(p, N_fft)
    S = np.fft.rfft(s, N_fft)
    corr = np.fft.irfft(np.conj(P) * S, N_fft)[:n]
    return 2.0 * corr

def spectral_step(s, reduction=0.001):
    """Compute a spectral direction that reduces the peak of conv(s,s)."""
    n = len(s)
    conv = np.convolve(s, s)
    peak = conv.max()
    M = peak * (1.0 - reduction)
    excess = np.maximum(0.0, conv - M)
    if excess.max() < 1e-15:
        return None
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    E = np.fft.rfft(excess, N_fft)
    with np.errstate(divide='ignore', invalid='ignore'):
        D = np.where(np.abs(S) > 1e-12, -E / S, 0.0)
    delta = np.fft.irfft(D, N_fft)[:n]

def lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M, b >= 0."""
    n = len(s)
    if n > 600:
        return None
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            continue
        cand = cand / cand.sum()
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_alpha = alpha
    return best_alpha, best_score
