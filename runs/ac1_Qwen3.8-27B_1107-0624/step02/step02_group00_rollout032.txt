"""
Hybrid top-K gradient + LP + multi-scale optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key improvements:
1. Top-K subgradient with continuation (decreasing K from broad to sharp)
2. Coarse-to-fine multi-scale optimization
3. Correct FFT-based gradient via cross-correlation
4. Diverse initializations including Fourier-motivated profiles
5. Backtracking line search for guaranteed descent
6. LP refinement as occasional polish
7. Adaptive restarts to escape local minima

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        mn = float(c.min())
        if mn < 0:
            c = c - mn
        return c
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _conv_gradient(w, a):
    """Compute grad[m] = 2 * sum_k w[k] * a[k-m] for m=0..n-1 using FFT."""
    n = len(a)
    n_w = len(w)
    N = _next_pow2(max(n_w + n - 1, 1))
    Fw = np.fft.rfft(w, N)
    a_rev = np.zeros(N)
    a_rev[:n] = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    h_full = np.fft.irfft(Fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if mval <= 0:
        n_c = len(c)
        w = np.zeros(n_c)
        w[int(np.argmax(c))] = 1.0
        return w, mval
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    return w, mval


def _topk_weights(c, K):
    c = np.asarray(c, dtype=np.float64)
    n_c = len(c)
    K = min(K, n_c)
    if K <= 0:
        K = 1
    if K >= n_c:
        w = np.ones(n_c) / n_c
        return w
    threshold = np.partition(c, n_c - K)[n_c - K]
    top_mask = c >= threshold - 1e-15
    n_top = int(top_mask.sum())
    if n_top == 0:
        w = np.zeros(n_c)
        w[int(np.argmax(c))] = 1.0
        return w
    w = np.zeros(n_c)
    w[top_mask] = 1.0 / n_top
    return w

    """Adam in logit space with top-K continuation."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)
    if K_start is None:
        K_start = max(1, n // 8)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    base_lr = lr
    current_K = K_start
    K_decrease_rate = 0.995

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Adaptive K: decrease over iterations
        current_K = max(K_end, int(K_start * (K_decrease_rate ** it)))

        p = a
        c = _fft_convolve(p)
        w = _topk_weights(c, current_K)
        grad = _conv_gradient(w, p)
        dot = float(np.dot(p, grad))
        gz = p * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_a, 1e-12))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.9)

        if no_imp > max(3000, 4 * n):
            # Restart with perturbation
            rng = np.random.default_rng(hash((int(seed), it)) % (2**31))
            z = 0.7 * np.log(np.maximum(best_a, 1e-12)) + 0.3 * rng.normal(size=n)
            z -= float(np.mean(z))
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0
            current_K = max(K_start, int(current_K * 1.5))

    return best_a


def _backtracking_descent(a_init, deadline, max_iter=20000, lr_init=0.1, K=5):
    """Gradient descent with backtracking line search on simplex."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = _project_simplex(a)
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    lr = lr_init
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w = _topk_weights(c, K)
        grad = _conv_gradient(w, a)
        dot = float(np.dot(a, grad))
        g = a * (grad - dot)
        g -= float(np.mean(g))

        if not np.all(np.isfinite(g)):
            break

        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(20):
            a_new = _project_simplex(a - alpha * g)
            c_new = _fft_convolve(a_new)
            score_new = float(2.0 * n * c_new.max())
            if score_new < best_score - 1e-14:
                a = a_new
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_imp += 1
            lr *= 0.9
            if no_imp > 2000:
                a = best_a.copy()
                lr = lr_init
                no_imp = 0
            continue

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a.copy()
            no_imp = 0
            lr = min(lr * 1.05, 0.5)
        else:
            no_imp += 1

    return best_a


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.3):
    """LP-based refinement: find direction maximizing sum while respecting conv bound."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        # Build constraint matrix
        A = _make_conv_matrix(a)
        b_ub = np.full(2 * n - 1, M_lp)
        c_obj = -np.ones(n)

        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'maxiter': 5000})
        if not res.success:
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.05)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
            if no_imp > 200:
                a = best_a.copy()
                no_imp = 0
                blend_cur = blend
                inflate_cur = inflate

        a = a_new

    return best_a


def _make_uniform(n):
    return np.ones(n) / n


def _make_linear(n, slope=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sin(n, freq=1, amp=0.5):
    t = np.linspace(0.0, 2 * np.pi * freq, n)
    a = 1.0 + amp * np.cos(t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cos_bump(n, center=0.5, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + np.exp(-((t - center) ** 2) / (2 * width ** 2))
    return a / a.sum()


def _make_double_bump(n, sep=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + np.exp(-((t - 0.5 + sep) ** 2) / 0.02) + np.exp(-((t - 0.5 - sep) ** 2) / 0.02)
    return a / a.sum()


def _make_poly(n, coeffs):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * t ** i
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_random(n, rng):
    a = rng.random(n) + 0.1
    return a / a.sum()


def _interpolate_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        if s > 1e-30:
            return p / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p_new / s


def _peak_shave(a, n_peaks=5, strength=0.1):
    """Reduce mass at positions contributing most to top peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    c = _fft_convolve(a)
    top_positions = np.argsort(c)[::-1][:n_peaks]

    # For each top peak position j, find the position i that contributes most
    contributions = np.zeros(n)
    for j in top_positions:
        # c[j] = sum_i a[i] * a[j-i], contribution of a[i] is a[j-i]
        valid_i = np.arange(0, n)
        valid_jm = (j - valid_i)
        valid_mask = (valid_jm >= 0) & (valid_jm < n)
        contrib = np.zeros(n)
        contrib[valid_i[valid_mask]] = a[valid_jm[valid_mask]]
        contributions += contrib

    # Reduce mass at high-contribution positions, add to low-contribution
    max_contrib = contributions.max()
    if max_contrib > 0:
        # Identify top contributors
        top_contrib_idx = np.argsort(contributions)[::-1][:n // 4]
        bottom_contrib_idx = np.argsort(contributions)[:n // 4]

        mass_to_move = strength * a[top_contrib_idx].sum()
        a[top_contrib_idx] -= strength * a[top_contrib_idx]
        a[top_contrib_idx] = np.maximum(a[top_contrib_idx], 0)

        # Distribute to bottom contributors
        bottom_weights = 1.0 / (contributions[bottom_contrib_idx] + 1e-10)
        bottom_weights /= bottom_weights.sum()
        a[bottom_contrib_idx] += mass_to_move * bottom_weights

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s

    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-scale + top-K gradient + LP hybrid optimization.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define lengths to try
    lengths = [256, 384, 512]

    # Phase 1: Multi-scale coarse-to-fine
    # Start at n=64, optimize, expand to larger sizes
    print("Phase 1: Coarse-to-fine multi-scale")
    a_coarse = _make_linear(64, 1.5)
    if parent is not None:
        a_coarse = _interpolate_to_length(parent, 64)

    for n_scale in [64, 128, 256]:
        if time.monotonic() > deadline - 200:
            break
        scale_budget = min(60.0, deadline - time.monotonic() - 200)
        if scale_budget < 10:
            break
        scale_end = time.monotonic() + scale_budget

        a_coarse = _adam_logit_optimize(a_coarse, scale_end,
                                         K_start=max(1, n_scale // 8), K_end=1,
                                         max_iter=10000, lr=0.1)
        sc = _score(a_coarse)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_coarse.copy()
        print(f"  Scale n={n_scale}: score={sc:.10f}")

    # Phase 2: Optimize at target lengths with diverse inits
    print("Phase 2: Target length optimization with diverse inits")
    for n_t in lengths:
        if time.monotonic() > deadline - 100:
            break

        # Generate diverse initializations
        inits = []
        inits.append(_make_uniform(n_t))
        inits.append(_make_linear(n_t, 1.0))
        inits.append(_make_linear(n_t, 2.0))
        inits.append(_make_linear(n_t, 3.0))
        inits.append(_make_sin(n_t, 1, 0.3))
        inits.append(_make_sin(n_t, 1, 0.5))
        inits.append(_make_sin(n_t, 2, 0.3))
        inits.append(_make_cos_bump(n_t, 0.5, 0.2))
        inits.append(_make_cos_bump(n_t, 0.3, 0.15))
        inits.append(_make_double_bump(n_t, 0.3))
        inits.append(_make_poly(n_t, [1, 0, 3, 0, -2]))
        inits.append(_make_poly(n_t, [1, 2, -1]))
        inits.append(_make_random(n_t, rng))
        inits.append(_make_random(n_t, rng))

        # Add parent-based inits
        if parent is not None:
            p_t = _interpolate_to_length(parent, n_t)
            inits.append(p_t)
            for _ in range(2):
                p_pert = p_t * (1.0 + 0.15 * rng.normal(size=n_t))
                p_pert = np.maximum(p_pert, 0)
                p_pert = p_pert / p_pert.sum()
                inits.append(p_pert)

        # Add coarse-to-fine init
        if a_coarse is not None:
            inits.append(_interpolate_to_length(a_coarse, n_t))

        # Optimize each init
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 80:
                break

            a0 = _project_simplex(a0)
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Determine budget for this init
            remaining = deadline - time.monotonic() - 80
            n_remaining_inits = len(inits) - idx
            init_budget = max(5.0, min(120.0, remaining / max(1, n_remaining_inits)))
            if init_budget < 5:
                break

            opt_end = time.monotonic() + init_budget

            # Adam logit optimization
            a_opt = _adam_logit_optimize(a0, opt_end,
                                         K_start=max(1, n_t // 4), K_end=2,
                                         max_iter=15000, lr=0.06)
            sc_opt = _score(a_opt)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
            if (idx + 1) % 4 == 0:
                print(f"  n={n_t}, init {idx+1}/{len(inits)}: best so far={best_score:.10f}")

    # Phase 3: LP refinement on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        lp_budget = min(120.0, deadline - time.monotonic() - 30)
        if lp_budget > 15:
            print(f"Phase 3: LP refinement, budget={lp_budget:.0f}s")
            lp_end = time.monotonic() + lp_budget
            a_lp = _lp_refine(best_seq, lp_end, max_iter=1500, inflate=1.001, blend=0.3)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  LP result: score={sc_lp:.10f}")

    # Phase 4: Fine Adam polish with small K
    if best_seq is not None and time.monotonic() < deadline - 20:
        ad_budget = min(150.0, deadline - time.monotonic() - 20)
        if ad_budget > 10:
            print(f"Phase 4: Fine Adam polish, budget={ad_budget:.0f}s")
            ad_end = time.monotonic() + ad_budget
            a_fine = _adam_logit_optimize(best_seq, ad_end,
                                          K_start=3, K_end=1,
                                          max_iter=50000, lr=0.03)
            sc_fine = _score(a_fine)
            if np.isfinite(sc_fine) and sc_fine < best_score:
                best_score = sc_fine
                best_seq = a_fine.copy()
            print(f"  Fine Adam: score={sc_fine:.10f}")

    # Phase 5: Peak shaving + re-optimize
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_shaved = _peak_shave(best_seq, n_peaks=10, strength=0.05)
        sc_shaved = _score(a_shaved)
        if np.isfinite(sc_shaved) and sc_shaved < best_score:
            best_score = sc_shaved
            best_seq = a_shaved.copy()

        if time.monotonic() < deadline - 10:
            a_res = _adam_logit_optimize(best_seq, deadline - 5,
                                         K_start=2, K_end=1,
                                         max_iter=20000, lr=0.02)
            sc_res = _score(a_res)
            if np.isfinite(sc_res) and sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
            print(f"  Phase 5: score={sc_res:.10f}")

    # Phase 6: Final LP polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        lp_budget = max(5.0, deadline - time.monotonic() - 5)
        if lp_budget > 3:
            a_final = _lp_refine(best_seq, time.monotonic() + lp_budget,
                                  max_iter=500, inflate=1.0005, blend=0.15)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LP: score={sc_final:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_linear(256, 1.5)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear(256, 1.5)

    # Validate with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
