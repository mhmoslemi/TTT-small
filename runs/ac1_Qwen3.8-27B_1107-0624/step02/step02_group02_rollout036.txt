"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based descent, Nesterov-accelerated gradient descent, and L-BFGS polishing
in logit space with adaptive temperature schedules.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def full_convolve(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 256:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def soft_max(c, beta):
    """Smooth max via log-sum-exp. Returns (value, weights)."""
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    val = m + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_grad_smooth(w, a):
    """Gradient of sum_k w[k]*c[k] wrt a, where c = conv(a,a)."""
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    part1 = np.fft.irfft(fw * np.conj(fa), N)[:n]
    part2 = np.fft.irfft(fw * fa, N)[:n]
    g = part1 + part2
    return g


def softmax_fn(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def build_conv_matrix(f):
    """Build matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, max_iters=500):
    """LP-based optimization."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()

    s = float(a.sum())

def full_convolve(a):
    """Full linear convolution using FFT for efficiency."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 256:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def soft_max(c, beta):
    """Smooth max via log-sum-exp. Returns (value, weights)."""
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    val = m + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_grad_smooth(w, a):
    """Gradient of sum_k w[k]*c[k] wrt a, where c = conv(a,a)."""
    n = len(a)
    N = 1

    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    part1 = np.fft.irfft(fw * np.conj(fa), N)[:n]
    part2 = np.fft.irfft(fw * fa, N)[:n]
    return part1 + part2

    n = len(f)

    return A

    return None

def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""

    Minimizes 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"    NEW BEST: n={p.size}, score={sc:.10f}")

    # Get initial sequence
    init_a = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_a = raw / s
    except Exception:
        pass

    print(f"Starting search. budget={budget_s}s")
    if init_a is not None:
        sc_init = compute_score(init_a)
        print(f"Initial from height_sequence_1: n={init_a.size}, score={sc_init:.10f}")
        update(init_a, sc_init)
    else:
        init_a = make_linear_profile(512, 1.5)
        sc_init = compute_score(init_a)
        update(init_a, sc_init)

    n_main = 512

    # ============ Phase 1: LP optimization at n=512 ============
    if best_seq.size != n_main:
        p_main = interpolate_profile(best_seq, n_main)
    else:
        p_main = best_seq.copy()

    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1: LP at n={n_main}, score={sc:.10f}")

    phase1_end = min(deadline, t_start + 200.0)
    result = lp_optimize(p_main, phase1_end, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"  After Phase 1 LP: best={best_score:.10f}")

    # ============ Phase 2: Nesterov GD at n=512 ============
    print(f"Phase 2: Nesterov at n={n_main}, score={best_score:.10f}")
    phase2_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase2_end - 5:
        result, sc = nesterov_optimize(best_seq, phase2_end, rng,
                                       beta_init=0.005, lr=0.02, momentum=0.95)
        sc = compute_score(result)
        update(result, sc)
    print(f"  After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: L-BFGS at n=512 (multiple beta) ============
    print(f"Phase 3: L-BFGS at n={n_main}")
    phase3_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        for beta_val in [1e-3, 5e-4, 2e-4, 1e-4, 5e-5]:
            if time.monotonic() > phase3_end - 10:
                break
            per_dead = min(phase3_end, time.monotonic() + 35.0)
            a0 = best_seq + 0.001 * rng.normal(size=best_seq.size)
            a0 = np.maximum(a0, 1e-12)
            a0 = a0 / a0.sum()
            result, sc = lbfgs_polish(a0, per_dead, beta=beta_val, max_iter=2000)
            sc = compute_score(result)
            update(result, sc)
    print(f"  After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Nesterov with different params ============
    print(f"Phase 4: Extended Nesterov at n={n_main}")
    phase4_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        for (b_i, lr_v, mom) in [(0.01, 0.01, 0.93), (0.002, 0.03, 0.97), (0.001, 0.008, 0.96)]:
            if time.monotonic() > phase4_end - 10:
                break
            per_dead = min(phase4_end, time.monotonic() + 60.0)
            a0 = best_seq + 0.002 * rng.normal(size=best_seq.size)
            a0 = np.maximum(a0, 1e-12)
            a0 = a0 / a0.sum()
            result, sc = nesterov_optimize(a0, per_dead, rng,
                                           beta_init=b_i, lr=lr_v, momentum=mom)
            sc = compute_score(result)
            update(result, sc)
    print(f"  After Phase 4: best={best_score:.10f}")

    # ============ Phase 5: Interpolate to n=1024 and optimize ============
    n_large = 1024
    phase5_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase5_end - 30 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 5: n={n_large}, score={sc:.10f}")

        # Nesterov at large n
        per_dead = min(phase5_end, time.monotonic() + 150.0)
        result, sc = nesterov_optimize(p_large, per_dead, rng,
                                       beta_init=0.003, lr=0.01, momentum=0.94)
        sc = compute_score(result)
        update(result, sc)
        print(f"  After Phase 5 Nesterov: best={best_score:.10f}")

        # L-BFGS at large n
        if time.monotonic() < phase5_end - 20:
            per_dead = min(phase5_end, time.monotonic() + 60.0)
            for beta_val in [1e-3, 5e-4, 1e-4]:
                if time.monotonic() > phase5_end - 10:
                    break
                per_d = min(phase5_end, time.monotonic() + 25.0)
                a0 = best_seq + 0.001 * rng.normal(size=best_seq.size)
                a0 = np.maximum(a0, 1e-12)
                a0 = a0 / a0.sum()
                result, sc = lbfgs_polish(a0, per_d, beta=beta_val, max_iter=1500)
                sc = compute_score(result)
                update(result, sc)
            print(f"  After Phase 5 L-BFGS: best={best_score:.10f}")

    print(f"After Phase 5: best={best_score:.10f}")

    # ============ Phase 6: Final intensive polishing ============
    phase6_end = deadline
    if time.monotonic() < phase6_end - 10 and best_seq is not None:
        print(f"Phase 6: Final polishing at n={best_seq.size}")

        # Multiple L-BFGS rounds with decreasing beta
        for beta_val in [1e-4, 5e-5, 2e-5, 1e-5, 5e-6]:
            if time.monotonic() > phase6_end - 5:
                break
            per_dead = min(phase6_end, time.monotonic() + 25.0)
            a0 = best_seq.copy()
            result, sc = lbfgs_polish(a0, per_dead, beta=beta_val, max_iter=2000)
            sc = compute_score(result)
            update(result, sc)

        # Final Nesterov with very low beta
        if time.monotonic() < phase6_end - 10:
            per_dead = min(phase6_end, time.monotonic() + 30.0)
            result, sc = nesterov_optimize(best_seq, per_dead, rng,
                                           beta_init=1e-4, beta_min=1e-7,
                                           lr=0.005, momentum=0.96)
            sc = compute_score(result)
            update(result, sc)

    print(f"After Phase 6: best={best_score:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
            if eval_score < best_score:
                best_score = eval_score
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
