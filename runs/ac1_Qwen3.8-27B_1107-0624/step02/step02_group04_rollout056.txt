        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _adam_nesterov_logit(p0, deadline, rng, lr=0.06, momentum=0.9,
                         beta_init_frac=0.10, beta_decay=0.70,
                         var_weight=0.0, reg=1e-7,
                         no_imp_mult=4, exact_mode_after=0.6):
    """
    Nesterov-momentum logit-space optimization with adaptive temperature.
    Switches to exact subgradient mode after a fraction of time.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    v = np.zeros(n, dtype=np.float64)  # velocity (Nesterov momentum)
    p_prev = p0.copy()

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    t_start = time.monotonic()
    total_budget = deadline - t_start

    base_lr = lr
    step = 0

    while time.monotonic() < deadline:
        step += 1
        frac_done = (time.monotonic() - t_start) / max(1e-6, total_budget)

        # Determine mode: smooth vs exact
        if frac_done > exact_mode_after or beta < 1e-9:
            # Exact subgradient
            _, w = _soft_max_weights(c, 0.0)
            L = maxc
        else:
            L, w = _soft_max_weights(c, beta)

        gp = 2.0 * _corr_weights(w, p)

        if var_weight > 0.0:
            gp += var_weight * (p - p.mean())

        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            v.fill(0.0)
            p = best_p.copy()
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            continue

        # Nesterov momentum update
        v = momentum * v - lr * gz
        z_new = z + v
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        p_new = _softmax_from_z(z_new)
        c_new = _full_convolve_fft(p_new)
        maxc_new = float(c_new.max())
        score = float(2.0 * n * maxc_new)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p_new.copy()
                best_z = z_new.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.003, lr * 0.95)
                    momentum = min(0.98, momentum + 0.001)
            z = z_new
            p = p_new
            c = c_new
            maxc = maxc_new
        else:
            z = best_z.copy()
            v.fill(0.0)
            p = best_p.copy()
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            continue

        # Adaptive beta decay
        if step % 30 == 0:
            beta = max(1e-12, beta * beta_decay)
            if c.size >= 2:
                sorted_idx = np.argsort(c)[::-1]
                if sorted_idx[0] != sorted_idx[1]:
                    gap = c[sorted_idx[0]] - c[sorted_idx[1]]
                    if gap < beta * 0.1:
                        beta = max(1e-12, beta * 0.3)

        # Restart if stuck
        no_imp_limit = max(100, no_imp_mult * n // 10)
        if no_imp > no_imp_limit:
            if best_score < 1.51:
                noise_scale = 0.08
            elif best_score < 1.53:
                noise_scale = 0.15
            elif best_score < 1.56:
                noise_scale = 0.25
            else:
                noise_scale = 0.4
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            v.fill(0.0)
            lr = base_lr
            momentum = 0.9
            no_imp = 0
            score = float(2.0 * n * maxc)
            if score < best_score:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()

    return best_p, best_score


def _lp_best_response(p, deadline_check, max_time=5.0):
    """
    Solve LP: max sum(q) s.t. conv(p,q)[k] <= M for all k, q >= 0.
    Returns normalized q or None if timeout/failure.
    """
    n = p.size
    c = _full_convolve_fft(p)
    M = float(c.max())

    t0 = time.monotonic()
    if time.monotonic() > deadline_check or (t0 + max_time) > deadline_check:
        return None

    # Constraint matrix A[i,j] = p[i] if j >= i, p[j] if j < i
    # conv(p,q)[k] = sum_{i+j=k} p[i] q[j]
    # We need: for each k = 0,...,2n-2: sum_{j=max(0,k-(n-1))}^{min(n-1,k)} p[k-j] q[j] <= M
    #
    # Build A as (2n-1) x n matrix
    # A[k,j] = p[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # q[j] contributes to conv positions k = j, j+1, ..., j+n-1
        # conv[p, q][k] includes p[k-j] * q[j] when 0 <= k-j < n
        k_start = j
        k_end = j + n
        A[k_start:k_end, j] = p[:k_end - k_start]

    c_obj = -np.ones(n)  # minimize -sum(q) = maximize sum(q)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max_time - 1.0})
        if res.status == 0 and res.x is not None:
            q = res.x
            s = float(q.sum())
            if s > 1e-10:
                return q / s
    except Exception:
        pass
    return None


def _lp_refined_step(p, deadline, rng, n_lp_calls=20):
    """
    Use LP best-response iteratively to refine p.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = float(2.0 * n * _full_convolve_fft(p).max())
    t0 = time.monotonic()
    calls_made = 0

    while time.monotonic() < deadline and calls_made < n_lp_calls:
        q = _lp_best_response(p, deadline - 2.0, max_time=4.0)
        calls_made += 1
        if q is None:
            break

        # Blend p towards q
        t_step = 0.3
        p_new = (1.0 - t_step) * p + t_step * q
        p_new = np.maximum(p_new, 1e-12)
        p_new = p_new / p_new.sum()

        c_new = _full_convolve_fft(p_new)
        maxc_new = float(c_new.max())
        score = float(2.0 * n * maxc_new)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_p = p_new.copy()
            p = p_new

    return best_p, best_score, calls_made


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses Nesterov logit-space optimization, LP refinement,
    exact subgradient mode, and multi-length escalation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            if label:
                print(f"  NEW BEST: {best_score:.10f} (n={p.size}) [{label}]")

    # --- Get parent sequence ---
    parent = None
    parent_n = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                parent_n = parent.size
    except Exception:
        parent = None

    print(f"Parent: n={parent_n if parent is not None else 'N/A'}")
    if parent is not None:
        print(f"  Parent score: {_internal_score(parent):.10f}")

    # --- Phase 0: Initial candidates ---
    candidates = []
    n_candidates = [512, 1024]
    if parent is not None and parent_n in [1024, 2048]:
        n_candidates.append(parent_n)

    for n in n_candidates:
        profs = []
        profs.append(('cos0.5', _make_profile_cosine(n, 0.5)))
        profs.append(('cos1.0', _make_profile_cosine(n, 1.0)))
        profs.append(('lin1.5', _make_profile_linear(n, 1.5)))
        profs.append(('asy1', _make_profile_asymmetric(n, 1.5, -1.0)))
        profs.append(('asy2', _make_profile_asymmetric(n, 2.0, 1.5)))
        if parent is not None:
            p_interp = _interp_profile(parent, n)
            if p_interp is not None:
                profs.append(('parent', p_interp))
        for label, p in profs:
            sc = _internal_score(p)
            if np.isfinite(sc):
                candidates.append((sc, p, n, label))

    candidates.sort(key=lambda x: x[0])
    for sc, p, n, label in candidates[:5]:
        update_best(p, sc, f"init {label} n={n}")
    print(f"Phase 0: best={best_score:.10f}")

    # --- Phase 1: Polish best at n=1024 ---
    phase1_end = min(deadline, t_start + 400.0)
    # Use the parent or best candidate
    if parent is not None and parent_n == 1024:
        p_work = parent.copy()
    else:
        p_work = candidates[0][1].copy()
        if p_work.size != 1024:
            p_work = _interp_profile(p_work, 1024)

    sc_work = _internal_score(p_work)
    print(f"Phase 1: starting n=1024, score={sc_work:.10f}")

    # Multiple optimization passes with different hyperparameters
    p_cur = p_work.copy()
    for pass_i in range(3):
        per_end = min(phase1_end, time.monotonic() + 120.0)
        lr = [0.06, 0.04, 0.02][min(pass_i, 2)]
        mom = [0.90, 0.92, 0.94][min(pass_i, 2)]
        beta_frac = [0.10, 0.06, 0.03][min(pass_i, 2)]
        beta_dec = [0.70, 0.75, 0.80][min(pass_i, 2)]

        p_best, sc_best = _adam_nesterov_logit(
            p_cur, per_end, rng, lr=lr, momentum=mom,
            beta_init_frac=beta_frac, beta_decay=beta_dec,
            var_weight=0.0, reg=1e-7, no_imp_mult=4,
            exact_mode_after=0.5)
        update_best(p_best, sc_best, f"polish{pass_i} n=1024")
        p_cur = best_seq.copy() if best_seq is not None and best_seq.size == 1024 else p_cur

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: LP-based refinement ---
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end and best_seq is not None and best_seq.size == 1024:
        print(f"Phase 2: LP refinement starting, score={best_score:.10f}")
        p_lp, sc_lp, n_calls = _lp_refined_step(best_seq, phase2_end, rng, n_lp_calls=30)
        update_best(p_lp, sc_lp, "LP refine n=1024")
        print(f"  LP calls made: {n_calls}, best={best_score:.10f}")

        # Polish after LP
        if time.monotonic() < phase2_end - 30.0:
            per_end = phase2_end
            p_pol, sc_pol = _adam_nesterov_logit(
                best_seq, per_end, rng, lr=0.03, momentum=0.94,
                beta_init_frac=0.03, beta_decay=0.82,
                var_weight=0.001, reg=1e-7, no_imp_mult=3,
                exact_mode_after=0.4)
            update_best(p_pol, sc_pol, "post-LP polish")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Escalate to n=2048 ---
    phase3_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        print(f"Phase 3: escalating to n=2048")
        p_2048 = _interp_profile(best_seq, 2048)
        if p_2048 is not None:
            sc_2048 = _internal_score(p_2048)
            print(f"  Interpolated score: {sc_2048:.10f}")
            update_best(p_2048, sc_2048, "interp n=2048")

            # Optimize at n=2048
            per_end_1 = min(phase3_end - 60.0, time.monotonic() + 180.0)
            p_b1, sc_b1 = _adam_nesterov_logit(
                p_2048, per_end_1, rng, lr=0.05, momentum=0.90,
                beta_init_frac=0.08, beta_decay=0.72,
                var_weight=0.0, reg=1e-7, no_imp_mult=4,
                exact_mode_after=0.5)
            update_best(p_b1, sc_b1, "opt1 n=2048")

            if time.monotonic() < phase3_end - 30.0:
                per_end_2 = min(phase3_end - 10.0, time.monotonic() + 150.0)
                p_b2, sc_b2 = _adam_nesterov_logit(
                    best_seq if best_seq.size == 2048 else p_b1,
                    per_end_2, rng, lr=0.03, momentum=0.93,
                    beta_init_frac=0.04, beta_decay=0.78,
                    var_weight=0.005, reg=1e-7, no_imp_mult=3,
                    exact_mode_after=0.4)
                update_best(p_b2, sc_b2, "opt2 n=2048")

    print(f"Phase 3 done: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # --- Phase 4: Multi-restart at best length ---
    phase4_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_best = best_seq.size
        n_starts = 4
        print(f"Phase 4: multi-restart at n={n_best}")

        for i in range(n_starts):
            if time.monotonic() > phase4_end:
                break
            per_end = min(phase4_end, time.monotonic() + 55.0)
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            noise = 0.10 + 0.08 * i
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_start = _softmax_from_z(z)
            p_try, sc_try = _adam_nesterov_logit(
                p_start, per_end, rng, lr=0.04, momentum=0.92,
                beta_init_frac=0.05, beta_decay=0.75,
                var_weight=0.0, reg=1e-7, no_imp_mult=3,
                exact_mode_after=0.4)
            update_best(p_try, sc_try, f"restart{i} n={n_best}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Final fine-tuning ---
    phase5_end = deadline
    if time.monotonic() < phase5_end and best_seq is not None:
        print(f"Phase 5: final fine-tuning")
        per_end = time.monotonic() + min(40.0, phase5_end - time.monotonic())
        p_fin, sc_fin = _adam_nesterov_logit(
            best_seq, per_end, rng, lr=0.015, momentum=0.95,
            beta_init_frac=0.02, beta_decay=0.85,
            var_weight=0.0, reg=1e-8, no_imp_mult=2,
            exact_mode_after=0.3)
        update_best(p_fin, sc_fin, "final")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq.copy(), 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s

        # Exact score
        sc_exact = _true_score(final_seq)
        print(f"Exact score: {sc_exact:.10f}")

        # Try evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
                best_score = min(best_score, sc_eval)
        except Exception:
            pass

        print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile_cosine(1024, 0.8)
    print(f"FALLBACK: n=1024")
    return [float(x) for x in fallback]
