```python
import numpy as np
import time

def fast_conv(s):
    n = len(s)
    N = 1
    while N < 2*n - 1:
        N *= 2
    S = np.fft.rfft(s, N)
    return np.fft.irfft(S * S, N)[:2*n-1]

def fast_score(s):
    n = len(s)
    total = float(np.sum(s))
    if total < 1e-15:
        return float('inf')
    conv = fast_conv(s)
    return 2.0 * n * float(np.max(conv)) / (total * total)

def compute_grad(s, T):
    """Compute gradient of soft-max objective."""
    n = len(s)
    total = float(np.sum(s))
    if total < 1e-15:
        return np.zeros(n)
    
    conv = fast_conv(s)
    conv_max = float(np.max(conv))
    
    # Soft-max weights (numerically stable)
    shifted = conv - conv_max
    exp_vals = np.exp(T * shifted)
    Z = float(np.sum(exp_vals))
    w = exp_vals / Z
    
    # M_soft = (1/T) * log(Z) + conv_max
    M_soft = conv_max + np.log(Z) / T
    
    # Compute h_grad[j] = sum_k w[k] * (s_valid(k-j) + s_valid(j-k))
    s_pad = np.zeros(2*n-1)
    s_pad[:n] = s
    
    N = 1
    while N < 4*n:
        N *= 2
    
    Spad = np.fft.rfft(s_pad, N)
    W = np.fft.rfft(w, N)
    
    # A[j] = sum_m s_pad[m] * w[m+j] = IFFT(conj(Spad) * W)[j]
    A = np.fft.irfft(np.conj(Spad) * W, N)[:n]
    # B[j] = sum_m s_pad[m] * w[j-m] = IFFT(Spad * W)[j]
    B = np.fft.irfft(Spad * W, N)[:n]
    
    h_grad = A + B
    
    S2 = total * total
    S3 = S2 * total
    
    grad = 2.0 * n * (h_grad / S2 - 2.0 * M_soft / S3)
    
    return grad

def adam_optimize(s0, deadline, rng, T_start=10.0, T_end=200.0, lr_start=0.05, lr_end=0.001):
    """Adam optimization with adaptive temperature."""
    n = len(s0)
    s = s0.astype(np.float64).copy()
    s = np.maximum(s, 0)
    if s.sum() < 1e-15:
        s = np.ones(n) / n
    else:
        s = s / s.sum()
    
    best_s = s.copy()
    best_score = fast_score(s)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    no_improve = 0
    patience = 200
    
    while time.monotonic() < deadline:
        t += 1
        progress = min(t / 1000.0, 1.0)  # Normalize over first 1000 steps
        
        # Adaptive temperature
        T = T_start + (T_end - T_start) * progress
        # Adaptive learning rate
        lr = lr_start * (lr_end / lr_start) ** progress
        
        grad = compute_grad(s, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        s_new = s - lr * m_hat / (np.sqrt(v_hat) + eps)
        s_new = np.maximum(s_new, 0.0)
        s_sum = s_new.sum()
        if s_sum < 1e-15:
            s_new = s  # Reject
        else:
            s_new = s_new / s_sum
        
        score = fast_score(s_new)
        
        if score < best_score:
            best_score = score
            best_s = s_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        s = s_new
        
        # Patience-based restart
        if no_improve >= patience:
            if no_improve >= patience * 3:
                # Full restart with perturbation
                s = best_s + 0.01 * rng.standard_normal(n)
                s = np.maximum(s, 0.01)
                s = s / s.sum()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0
            else:
                # Small perturbation
                s = best_s + 0.005 * rng.standard_normal(n)
                s = np.maximum(s, 0.005)
                s = s / s.sum()
                no_improve = 0
        
        if t % 200 == 0:
            print(f"    t={t}, score={best_score:.6f}, T={T:.1f}")
    
    return best_s, best_score
```

```python
def coordinate_descent(s, num_passes=10, deadline=None):
    """Refine using coordinate descent."""
    n = len(s)
    s = s.astype(np.float64).copy()
    s = np.maximum(s, 0)
    s = s / s.sum()
    
    best_s = s.copy()
    best_score = fast_score(s)
    
    for p in range(num_passes):
        if deadline and time.monotonic() > deadline:
            break
        
        for j in range(n):
            if deadline and time.monotonic() > deadline:
                break
            
            # Optimize s[j] by trying several values
            old_val = s[j]
            # Try: 0, half, current, 2x current, 4x current, and some random values
            candidates = [0.0, old_val * 0.5, old_val, old_val * 1.5, old_val * 2.0,
                          old_val * 0.1, old_val * 5.0, 1.0/n, 2.0/n, 0.5/n]
            # Add a few random candidates around current
            candidates.extend(old_val * np.exp(rng.standard_normal(3) * 0.3))
            candidates = [max(0.0, c) for c in candidates]
            
            best_j_val = old_val
            best_j_score = best_score
            
            for cand in candidates:
                s[j] = cand
                s_norm = s / s.sum()
                sc = fast_score(s_norm)
                if sc < best_j_score:
                    best_j_score = sc
                    best_j_val = cand
            
            s[j] = best_j_val
            s = s / s.sum()
            
            if best_j_score < best_score:
                best_score = best_j_score
                best_s = s.copy()
    
    return best_s, best_score
```
