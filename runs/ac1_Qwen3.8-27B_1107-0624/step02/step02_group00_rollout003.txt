"""
Adaptive smooth-max gradient descent with peak shaving and coordinate descent
for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key improvements over previous:
- Faster FFT-based gradient computation
- Multi-scale smooth max with adaptive(beta) annealing

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of convolution via FFT."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, mval


def _subgradient_topm(a, top_m=10):
    """Subgradient using top-m peaks of convolution."""
    n = len(a)
    c = _fft_convolve(a)
    m = min(top_m, len(c))
    top_idx = np.argsort(c)[::-1][:m]
    top_vals = c[top_idx]
    max_val = top_vals[0]
    if max_val < 1e-30:
        return np.zeros(n), max_val
    # Softmax weights
    temp = max(1e-10, max_val * 0.15)
    weights = np.exp((top_vals - max_val) / temp)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w_k, k in zip(weights, top_idx):
        k = int(k)
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad[valid] += w_k * 2.0 * a[idx[valid]]
    return grad, float(max_val)


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gauss_profile(n, center=0.45, width=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_ramp_asym(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _coordinate_descent(a, deadline, max_iter=20000):
    """Coordinate descent: optimize one element at a time."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    rng = np.random.default_rng(123)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Pick a random coordinate
        i = rng.integers(0, n)

        # Optimize a[i] by 1D search
        # Score as function of a[i]: 2n*max(conv)/sum^2
        # conv depends quadratically on a[i]
        a_save = a[i]
        a_rest_sum = s - a_save

        # Try a grid of values
        best_val = a_save
        best_local = best_score

        # Scale around current value
        for frac in [0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 0.2]:
            trial = a_save * frac
            if trial < 0:
                continue
            a[i] = trial
            new_sum = a_rest_sum + trial
            if new_sum < 1e-10:
                a[i] = a_save
                continue
            c = _fft_convolve(a)
            sc = 2.0 * n * c.max() / (new_sum * new_sum)
            # Convert to normalized score (sum=1)
            sc_norm = 2.0 * n * c.max()
            if sc_norm < best_local:
                best_local = sc_norm
                best_val = trial

        a[i] = best_val
        s = a.sum()
        a = a / s

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def _lp_spread_step(a, M):
    """Single LP spread step."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.3):
    """LP-based refinement (limited iterations for speed)."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * inflate_cur
        g = _lp_spread_step(a, M)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.15)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:
                break

        a = a_new

    return best_a, best_score

        for a0 in inits:
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_local_score:
                best_local_score = sc0
                best_local = a0.copy()

        if best_local is None:
            continue

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()

        candidates.append((n_t, best_local, best_local_score))
        print(f"  n={n_t}: best_init_score={best_local_score:.10f}")

    # Phase 2: LP refinement on top candidates (time-limited)
    lp_time_per = 60.0  # seconds per LP refinement
    candidates.sort(key=lambda x: x[2])  # Sort by score
    for i, (n_t, a0, sc0) in enumerate(candidates[:4]):  # Top 4
        if time.monotonic() > deadline - 400:
            break
        lp_deadline = time.monotonic() + min(lp_time_per, deadline - time.monotonic() - 400)
        a_lp, sc_lp = _lp_refine(a0, lp_deadline, max_iter=500, inflate=1.002, blend=0.35)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        print(f"  LP n={n_t}: {sc0:.10f} -> {sc_lp:.10f}")
        if i < len(candidates):
            candidates[i] = (n_t, a_lp, sc_lp)

    # Phase 3: Adam smooth-max on top candidates
    candidates.sort(key=lambda x: x[2])
    adam_time_per = 120.0
    for i, (n_t, a0, sc0) in enumerate(candidates[:3]):
        if time.monotonic() > deadline - 200:
            break
        ad_deadline = time.monotonic() + min(adam_time_per, deadline - time.monotonic() - 200)
        a_ad, sc_ad = _adam_smooth_max(a0, ad_deadline, max_iter=50000, lr=0.04, momentum=0.9)
        if sc_ad < best_score:
            best_score = sc_ad
            best_seq = a_ad.copy()
        print(f"  Adam n={n_t}: {sc0:.10f} -> {sc_ad:.10f}")
        if i < len(candidates):
            candidates[i] = (n_t, a_ad, sc_ad)

    # Phase 4: Peak shaving on the best
    if time.monotonic() < deadline - 150:
        ps_deadline = time.monotonic() + min(80.0, deadline - time.monotonic() - 150)
        a_ps, sc_ps = _peak_shaving(best_seq, ps_deadline, max_iter=10000, start_lr=0.008)
        if sc_ps < best_score:
            best_score = sc_ps
            best_seq = a_ps.copy()
        print(f"  Peak shaving: {best_score:.10f} -> {sc_ps:.10f}")

    # Phase 5: Another Adam pass on the improved sequence
    if time.monotonic() < deadline - 100:
        ad_deadline = time.monotonic() + min(100.0, deadline - time.monotonic() - 100)
        a_ad2, sc_ad2 = _adam_smooth_max(best_seq, ad_deadline, max_iter=30000, lr=0.025, momentum=0.92)
        if sc_ad2 < best_score:
            best_score = sc_ad2
            best_seq = a_ad2.copy()
        print(f"  Adam pass 2: {best_score:.10f} -> {sc_ad2:.10f}")

    # Phase 6: Coordinate descent final polish
    if time.monotonic() < deadline - 40:
        cd_deadline = time.monotonic() + min(60.0, deadline - time.monotonic() - 40)
        a_cd, sc_cd = _coordinate_descent(best_seq, cd_deadline, max_iter=50000)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
        print(f"  Coord descent: {best_score:.10f} -> {sc_cd:.10f}")

    # Phase 7: Final short Adam polish
    if time.monotonic() < deadline - 15:
        fin_deadline = time.monotonic() + min(30.0, deadline - time.monotonic() - 10)
        a_fin, sc_fin = _adam_smooth_max(best_seq, fin_deadline, max_iter=10000, lr=0.015, momentum=0.93)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
        print(f"  Final polish: {sc_fin:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
