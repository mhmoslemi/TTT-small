minimize max_k (a_k * s_i + b_k) subject to s_i >= 0, s_i <= 1 - sum_{j≠i} s_j

This is a 1D problem that can be solved by finding the minimum of a piecewise-linear function. The minimum occurs at one of the breakpoints where two of the (a_k * s_i + b_k) values cross, or at a boundary.

This is exact but might be slow for large n (need to check all k). However, we only need to check the k values that are currently near the maximum (say, the top 10).

Let me implement both gradient descent and coordinate descent, and see which works better.

Actually, for time efficiency, let me focus on the gradient approach but with some improvements:

1. **Exact subgradient**: Instead of the soft gradient, use the gradient at the single (or top few) max points. This is more "aggressive" in reducing the actual peak.

2. **Adaptive beta**: Start with moderate beta and increase over time.

3. **Multiple restarts**: If stuck for too long, perturb and restart.

4. **Length exploration**: Try several lengths and keep the best.

import numpy as np
import time

"""
Minimizes 2n*max(conv(s,s))/(sum(s))^2 using projected subgradient descent
with adaptive smoothing, coordinate descent polishing, and multi-start strategy.
Key differences from previous approach:
- Uses exact subgradient (not just soft) for final polishing
- Adds coordinate descent as a local refinement
- More aggressive beta annealing
- Diversified perturbation strategies
"""

def compute_conv_fft(s):
    n = len(s)
    N_fft = 1
    while N_fft < 2 * n:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    return np.fft.irfft(S * S, N_fft)[:2*n-1]

def score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = compute_conv_fft(s)
    return 2.0 * n * np.max(conv) / (total * total)

def soft_score(s, beta):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = compute_conv_fft(s)
    max_c = np.max(conv)
    lse = max_c + np.log(np.sum(np.exp(beta * (conv - max_c))))
    return 2.0 * n * lse / (beta * total * total)

def gradient_soft(s, beta):
    n = len(s)
    total = np.sum(s)
    conv = compute_conv_fft(s)
    max_c = np.max(conv)
    p = np.exp(beta * (conv - max_c))
    p_sum = np.sum(p)
    p /= p_sum
    # h[i] = sum_j s[j] * p[j+i]
    h = np.correlate(p, s, mode='valid')
    grad = 4.0 * n * h / (total * total)
    return grad

def gradient_exact(s, top_k=1):
    n = len(s)
    total = np.sum(s)
    conv = compute_conv_fft(s)
    # Get top_k indices with largest conv values
    indices = np.argsort(conv)[::-1][:top_k]
    # Weight them equally (or by value)
    # Actually, for subgradient of max, we use the gradient at the argmax.
    # For a smooth approximation, we can average over the top few.
    p = np.zeros(2*n-1)
    for idx in indices:
        p[idx] = 1.0 / len(indices)
    h = np.correlate(p, s, mode='valid')
    grad = 4.0 * n * h / (total * total)
    return grad

def proj_step(s, grad, eta):
    s_new = s - eta * grad
    s_new = np.maximum(s_new, 0.0)
    s_sum = s_new.sum()
    if s_sum < 1e-15:
        return s.copy()
    s_new *= (np.sum(s) / s_sum)
    return s_new

def line_search(s, grad, eta_max, f_obj, max_iter=20):
    eta = eta_max
    f_cur = f_obj(s)
    for _ in range(max_iter):
        s_new = proj_step(s, grad, eta)
        f_new = f_obj(s_new)
        if f_new < f_cur:
            return s_new, eta
        eta *= 0.5
    return s, eta

    if s_sum < 1e-15:

    total_iters = 3000

                s = s / s.sum()
                no_improve = 0

        if cur_score < best_score:

    eta = 0.05
    for it in range(2000):

        top_k = min(3, n)
        grad = gradient_exact(s, top_k=top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        s_new, eta = line_search(s, grad, eta_max=eta * 2, f_obj=score)
        
        if eta < 1e-7:

            for tk in [1, 2, 5, 10]:

        s = s_new
        cur_score = score(s)
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_s = s.copy()
    
    # Phase 3: Coordinate descent polishing
    s = best_s.copy()
    cur_score = score(s)
    for it in range(5000):
        if time.monotonic() > deadline:
            break
        
        # Pick a coordinate to optimize
        i = it % n
        # Find optimal s[i] given others fixed
        # c_k = a_k * s[i] + b_k where a_k and b_k don't depend on s[i]
        # a_k = 2 * s[k-i] (if 0 <= k-i < n) -- wait, this isn't right.
        
        # c_k = sum_j s_j * s_{k-j}
        # The terms involving s[i]: j=i gives s[i]*s[k-i], and k-j=i gives s[k-i]*s[i]
        # If i != k-i: c_k = (sum_{j≠i, j≠k-i} s_j*s_{k-j}) + 2*s[i]*s[k-i]
        # If i == k-i (k=2i): c_k = (sum_{j≠i} s_j*s_{k-j}) + s[i]^2
        
        # Let's compute a_k = dc_k/ds_i = 2*s[k-i] (if 0<=k-i<n) and b_k = c_k - a_k*s[i] + (correction for k=2i)
        # Actually, c_k as a function of s_i:
        # If k != 2i: c_k = b_k + 2*s[k-i]*s_i (linear in s_i)
        # If k == 2i: c_k = b_k + s_i^2 (quadratic in s_i)
        
        # For the quadratic case (k=2i), the objective max_k c_k is the max of:
        # - linear functions (for k≠2i)
        # - one quadratic s_i^2 + b_{2i} (for k=2i)
        
        # This is a 1D minimax problem. The optimal s_i is where the maximum is minimized.
        # For the linear parts: max over k≠2i of (b_k + a_k * s_i) where a_k = 2*s[k-i]
        # And the quadratic: s_i^2 + b_{2i}
        
        # For simplicity, let's just do a 1D search over s_i in [0, 1-sum(others)].
        
        s_i_old = s[i]
        s_others_sum = s.sum() - s_i_old
        s_i_max = s_others_sum  # since sum(s) = 1, s_i <= 1 - sum_others... wait
        # Actually s_i can be at most 1 (since all others are non-negative and sum to 1-s_i)
        s_i_max = 1.0  # s_i <= 1 (since sum = 1 and s_i >= 0)
        
        # Do a simple 1D minimization
        best_s_i = s_i_old
        best_f = score(s)
        
        # Try a grid + golden section
        for s_i_try in np.concatenate([
            [s_i_old * 0.5, s_i_old * 1.5],
            [0.0, 0.01, 0.02, s_i_max * 0.1, s_i_max * 0.5],
            s_i_old + 0.01 * rng.standard_normal(10)
        ]):

            # Renormalize

            if s_sum < 1e-15:

                continue
            s = s / s_sum
            f = score(s)
            if f < best_f:
                best_f = f
                best_s_i = s[i]
        
        if best_f < cur_score - 1e-12:
            cur_score = best_f
            if cur_score < best_score:
                best_score = cur_score
                best_s = s.copy()
        else:
            s[i] = s_i_old  # revert
    
    return best_s, best_score

def generate_initializations(n, rng, hseq=None):
    inits = []
    
    # Uniform
    s = np.ones(n) / n
    inits.append(s)
    
    # Cosine (Hann)
    x = np.linspace(0, np.pi, n)
    s = 0.5 * (1 + np.cos(x))
    s = np.minimum(s, np.ones(n))  # clip to avoid zeros at edges
    s = np.maximum(s, 0.01)
    s = s / s.sum()
    inits.append(s)
    
    # Various Gaussian widths
    center = (n-1) / 2.0
    for width in [0.1, 0.2, 0.3, 0.5, 0.7]:
        x = np.arange(n)
        s = np.exp(-0.5 * ((x - center) / (width * n)) ** 2)
        s = np.maximum(s, 1e-6)
        s = s / s.sum()
        inits.append(s)
        
    # Linear ramp
    x = np.linspace(0.1, 1.0, n)
    inits.append(x / x.sum())
    inits.append(np.maximum(1.0 - np.linspace(0, 0.9, n), 0.01) / np.sum(np.maximum(1.0 - np.linspace(0, 0.9, n), 0.01)))
    
    # Two-level
    for frac in [0.3, 0.4, 0.5, 0.6]:
        k = int(frac * n)
        s = np.ones(n)
        s[:k] = 2.0
        s = s / s.sum()
        inits.append(s)
    
    # Anti-symmetric-ish (high at edges, low in middle)
    x = np.arange(n)
    s = np.cos(np.pi * x / (n-1)) ** 2 + 0.1
    s = s / s.sum()
    inits.append(s)
    
    # Random
    for _ in range(5):
        s = rng.random(n) + 0.05
        s = s / s.sum()
        inits.append(s)
    
    # Resized height_sequence_1
    if hseq is not None:
        x_old = np.linspace(0, 1, len(hseq))
        x_new = np.linspace(0, 1, n)
        s = np.interp(x_new, x_old, hseq)
        s = np.maximum(s, 0.0)
        if s.sum() > 0.01:
            s = s / s.sum()
            inits.insert(0, s)
    
    return inits
