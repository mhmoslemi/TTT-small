def proj_simplex(a):
    """Project onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(a)
    if a.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)
