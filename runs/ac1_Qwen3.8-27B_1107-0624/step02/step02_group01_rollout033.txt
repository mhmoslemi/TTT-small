For each coordinate i, optimize a[i] (or log(a[i])) while keeping the others fixed. The optimal a[i] for fixed other coordinates can be found by solving a one-dimensional optimization problem.

def solve_lp_direction(f, M, n, prev_x=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    options = {'presolve': True, 'time_limit': 20}
    if prev_x is not None:
        # Try to use warm start by providing interior point
        pass
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options=options)
    if res.success:
        return res.x
    return None

def softmax_gradient_polish(f0, deadline, rng, lr=0.05, max_iter=50000):
    """Polish using soft-max gradient in logit space with adaptive beta."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())
    # Adaptive initial beta
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)]
    beta = max(1e-12, gap * 0.3, 0.05 * M_val)

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_corr = next_pow2(2 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Soft-max weights
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 10.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        # Gradient computation via cross-correlation
        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0

            # Adaptively increase beta as we converge
            if no_imp == 0 and t_adam % 100 == 0:
                beta = max(1e-12, beta * 0.85)
        else:
            no_imp += 1

        if t_adam % 500 == 0:
            beta = max(1e-12, beta * 0.7)

        if no_imp > max(500, 5 * n):
            # Escape local optimum
            z = 0.8 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

    return best_f, best_M


def coordinate_descent_polish(f0, deadline, rng, max_sweeps=100):
    """Polish using coordinate descent on individual elements."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    c = fft_convolve(f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Random order for this sweep
        order = rng.permutation(n)

        for idx in order:
            if time.monotonic() > deadline:
                break

            old_val = f[idx]

            # Compute base convolution (without contribution of f[idx])
            # b[k] = sum_{i+j=k} f[i]*f[j]
            # Contribution of f[idx]: f[idx]*f[k-idx] for each k
            # New b[k] = c[k] - f[idx]*f[k-idx] + new_val*f[k-idx] (for valid indices)
            # = c[k] + (new_val - f[idx])*f[k-idx]

            # For k from idx to idx + n - 1: contribution is f[idx] * f[k-idx]
            # (both f[idx]*f[k-idx] and f[k-idx]*f[idx] when k-idx != idx)
            # Actually b[k] = sum_i f[i]*f[k-i], so f[idx] appears when i=idx or k-i=idx
            # i=idx: term is f[idx]*f[k-idx] (valid if 0<=k-idx<n)
            # k-i=idx: term is f[i]*f[idx] where i=k-idx (valid if 0<=k-idx<n)
            # If idx != k-idx (i.e., k != 2*idx), both terms appear: 2*f[idx]*f[k-idx]
            # If idx == k-idx (k=2*idx), only one term: f[idx]^2

            # Let's compute the effect:
            # delta[k] = b_new[k] - b_old[k]
            # For k in [idx, idx+n-1]: f[idx] pairs with f[k-idx]
            #   If k != 2*idx: delta = 2*(new_val - old_val)*f[k-idx]
            #   If k == 2*idx: delta = new_val^2 - old_val^2

            # We want to minimize max(c + delta)
            # Try a range of new values

            # Quick approach: try new_val = 0, old_val*0.5, old_val, old_val*1.5
            # Pick the one that minimizes the resulting max

            candidates = [0.0, old_val * 0.3, old_val * 0.7, old_val,
                          old_val * 1.3, old_val * 1.7, old_val * 2.5]

            best_new_val = old_val
            best_max = M

            base_c = c.copy()
            for nv in candidates:
                if nv == old_val:
                    continue
                # Compute new max
                delta = np.zeros(2 * n - 1)
                lo = idx
                hi = min(idx + n - 1, 2 * n - 2)
                # For k in [lo, hi], the pair (idx, k-idx) contributes
                # delta[k] = (nv - old_val) * f[k-idx] [for the i=idx term]
                #            + (nv - old_val) * f[k-idx] [for the k-i=idx term, if k-idx != idx]
                # Actually it's simpler:
                # b[k] = sum_i f[i]*f[k-i]
                # Terms involving f[idx]: i=idx gives f[idx]*f[k-idx], i=k-idx gives f[k-idx]*f[idx]
                # If idx != k-idx: both terms, total 2*f[idx]*f[k-idx]
                # If idx == k-idx: one term f[idx]^2

                # Change: replace f[idx] with nv
                # For k where k-idx is valid (0<=k-idx<n, k != 2*idx):
                #   delta = (nv - old_val) * f[k-idx] + f[idx] * (nv - old_val) ... no
                #   delta = nv*f[k-idx] - old_val*f[k-idx] + nv*f[k-idx] - old_val*f[k-idx]
                #   Wait, I'm overcomplicating. Let me just recompute.

                # b_old[k] includes: f[idx]*f[k-idx] (if 0<=k-idx<=n-1) + f[k-idx]*f[idx] (if same and k!=2idx)
                # = (1 + [k != 2*idx]) * f[idx] * f[k-idx] for 0<=k-idx<=n-1
                # b_new[k] = (1 + [k != 2*idx]) * nv * f[k-idx] for 0<=k-idx<=n-1
                # delta[k] = (1 + [k != 2*idx]) * (nv - old_val) * f[k-idx]

                # For the range of k where k-idx is in [0, n-1]:
                k_lo = max(0, idx)
                k_hi = min(2 * n - 2, idx + n - 1)

                # For k in [k_lo, k_hi], k_idx = k - idx is in [0, n-1]
                contrib = f[k_lo - idx:k_hi - idx + 1]  # f[0], f[1], ..., f[hi-idx]
                multiplicity = np.ones(k_hi - k_lo + 1)
                # For k = 2*idx: multiplicity is 1, else 2
                if 2 * idx >= k_lo and 2 * idx <= k_hi:
                    multiplicity[2 * idx - k_lo] = 1.0

                delta_vals = multiplicity * (nv - old_val) * contrib
                c_candidate = base_c.copy()
                c_candidate[k_lo:k_hi + 1] += delta_vals
                c_candidate = np.maximum(c_candidate, 0)
                new_max = float(c_candidate.max())
                if new_max < best_max - 1e-15:
                    best_max = new_max
                    best_new_val = nv

            if best_new_val != old_val:
                f[idx] = best_new_val
                # Update c
                k_lo = max(0, idx)
                k_hi = min(2 * n - 2, idx + n - 1)
                contrib = f[k_lo - idx:k_hi - idx + 1]
                multiplicity = np.ones(k_hi - k_lo + 1)
                if 2 * idx >= k_lo and 2 * idx <= k_hi:
                    multiplicity[2 * idx - k_lo] = 1.0
                delta_vals = multiplicity * (best_new_val - old_val) * contrib
                c[k_lo:k_hi + 1] = np.maximum(c[k_lo:k_hi + 1] + delta_vals, 0.0)
                M = float(c.max())
                no_imp = 0
            else:
                no_imp += 1

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()

    return best_f, best_M


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search:
    1. LP-based direction finding at n=512 from diverse starts.
    2. Extended LP optimization on best candidates.
    3. Soft-max gradient polishing.
    4. Coordinate descent refinement.
    5. Interpolation to larger n with further refinement.
    6. Final evaluation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent if available
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}", flush=True)

    # Phase 1: LP optimization at n=512 from multiple diverse starts
    n1 = 512
    phase1_end = min(deadline, t_start + 480.0)

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)
            # Also try a perturbed version
            p_pert = p_int + 0.03 * rng.normal(size=n1)
            p_pert = np.maximum(p_pert, 0)
            p_pert = p_pert / p_pert.sum()
            starts.append(p_pert)

    # Deterministic profiles
    prows = make_deterministic_profiles(n1)
    starts.extend(prows[:8])

    # Random smooth profiles
    for _ in range(10):
        starts.append(make_smooth_random(n1, rng, n_harm=12))

    candidates = []
    per_start = max(5.0, (phase1_end - t_start) / max(1, len(starts)))

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s

        per = min(phase1_end, time.monotonic() + per_start * 0.9)
        p_best, M_best = lp_optimize(p0, per, rng, max_iter=8000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates.append((sc, p_best))
        if i % 5 == 0:
            print(f"  Phase 1 start {i}/{len(starts)}: best={best_score:.10f}", flush=True)

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}", flush=True)

    # Phase 2: Extended LP on top candidates
    phase2_end = min(deadline, t_start + 960.0)
    top_k = min(4, len(candidates))
    per_cand = max(20.0, (phase2_end - t_start) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates[i]
        per = min(phase2_end, time.monotonic() + per_cand)
        p_best, M_best = lp_optimize(p0, per, rng, max_iter=50000, verbose=False)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 2 cand {i}: score={sc:.10f}", flush=True)

    print(f"After Phase 2: best={best_score:.10f}", flush=True)

    # Phase 3: Soft-max gradient polish on best candidates
    phase3_end = min(deadline, t_start + 1260.0)
    polish_list = []
    if best_seq is not None and best_seq.size == n1:
        polish_list.append(best_seq.copy())
    # Add top candidates for polishing
    for i in range(min(3, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        _, p0 = candidates[i]
        polish_list.append(p0)

    for pi, p0 in enumerate(polish_list):
        if time.monotonic() > phase3_end:
            break
        per = min(phase3_end, time.monotonic() + 90.0)
        p_best, M_best = softmax_gradient_polish(p0, per, rng, lr=0.04, max_iter=80000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 3 polish {pi}: score={sc:.10f}", flush=True)

    print(f"After Phase 3: best={best_score:.10f}", flush=True)

    # Phase 4: Coordinate descent on the best
    if best_seq is not None and best_seq.size == n1:
        phase4_end = min(deadline, t_start + 1440.0)
        per = min(phase4_end, time.monotonic() + 120.0)
        p_best, M_best = coordinate_descent_polish(best_seq, per, rng, max_sweeps=50)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"After Phase 4 (coord descent): best={best_score:.10f}", flush=True)

    # Phase 5: Try n=1024
    n2 = 1024
    phase5_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase5_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase5_end, time.monotonic() + 100.0)
            p_best, M_best = lp_optimize(p_ext, per, rng, max_iter=20000)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"Phase 5 LP (n={n2}): score={sc:.10f}", flush=True)

            if time.monotonic() < phase5_end - 30:
                per2 = min(phase5_end, time.monotonic() + 60.0)
                p_best2, M_best2 = softmax_gradient_polish(p_best, per2, rng, lr=0.03, max_iter=30000)
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"Phase 5 polish (n={n2}): score={sc2:.10f}", flush=True)

    # Phase 6: Final LP refinement on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 60.0)
        p_best, M_best = lp_optimize(best_seq, per, rng, max_iter=30000)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        print(f"After Phase 6: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}", flush=True)

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}", flush=True)

        # Evaluate with the actual function for verification
        if final.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}", flush=True)
            except Exception as e:
                print(f"Evaluation error: {e}", flush=True)

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
