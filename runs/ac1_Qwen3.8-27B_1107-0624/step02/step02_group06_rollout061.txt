```python
"""
Search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses analytical gradients in softmax logit space, multiple optimizers (Adam, L-BFGS-B),
diverse initialization, multi-length search, and coordinate descent polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -80.0, 80.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e / float(e.sum())
    return p


def soft_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def compute_grad_objective(a, w, c, n):
    """
    Compute 4*n*a*(h - S) where:
    h[m] = sum_k w[k] * a[k-m] (cross-correlation)
    S = sum_k w[k] * c[k]
    """
    S = float(np.dot(w, c))
    # Compute h using FFT
    a_flip = a[::-1]
    N = next_pow2(2 * (2 * n - 1) + n - 1)  # enough for conv(w, a_flip)
    W = np.fft.rfft(w, N)
    AF = np.fft.rfft(a_flip, N)
    H = np.fft.irfft(W * AF, N)
    # h[m] = conv(w, a_flip)[n-1+m] for m = 0,...,n-1
    h = H[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - S)
    return grad


def objective_and_grad_z(z, beta):
    """
    Compute objective f = 2*n*soft_max(conv(softmax(z), softmax(z)), beta)
    and its gradient with respect to z.
    """
    a = softmax_from_z(z)
    n = a.size
    c = full_convolve_fft(a)
    val, w = soft_max_and_weights(c, beta)
    f = 2.0 * n * val
    # Gradient
    S = float(np.dot(w, c))
    a_flip = a[::-1]
    N = next_pow2(2 * (2 * n - 1) + n - 1)
    W = np.fft.rfft(w, N)
    AF = np.fft.rfft(a_flip, N)
    H = np.fft.irfft(W * AF, N)
    h = H[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - S)
    return f, grad


def adam_optimize(z0, beta_start, deadline, rng, lr=0.05, beta_decay=0.95, beta_min=1e-4):
    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    z -= float(np.mean(z))
    n = z.size
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = beta_start

        # So we should center by subtracting the mean.
        grad_centered = grad - float(np.mean(grad))
        
        m = b1 * m + (1 - b1) * grad_centered
        v = b2 * v + (1 - b2) * grad_centered ** 2
        mh = m / (1 - b1 ** t)

        z_new = z - lr * mh / (np.sqrt(vh) + eps)

        if np.isfinite(f_new) and f_new < best_f:

            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr *= 0.98
        
        # Decay beta
        if t % 50 == 0:
            beta = max(beta_min, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > 500:

            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))

            lr = 0.05
            beta = max(beta_min, beta_start * 0.5)

def lbfgs_wrapper(z, beta, n):
    f, grad = objective_and_grad_z(z, beta)
    # Center to handle gauge
    grad = grad - float(np.mean(grad))
    if not np.all(np.isfinite(grad)):
        return f if np.isfinite(f) else 1e10, np.zeros(n)
    return f, grad


def lbfgs_optimize(z0, beta, deadline, maxiter=100):
    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    z -= float(np.mean(z))
    n = z.size
    
    best_z = z.copy()
    best_f = objective_and_grad_z(z, beta)[0]
    
    start = time.monotonic()
    max_time = max(1.0, deadline - start)
    
    try:
        # Use L-BFGS-B with time-bounded iterations
        func_calls = [0]
        def fun(z):
            func_calls[0] += 1
            f, g = lbfgs_wrapper(z, beta, n)
            return f
        def jac(z):
            _, g = lbfgs_wrapper(z, beta, n)
            return g
        
        res = minimize(fun, z, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': 5000, 'maxls': 20})
        if res.success or np.isfinite(res.fun) or (res.fun if hasattr(res, 'fun') else np.inf) < best_f:
            zf = res.x - float(np.mean(res.x))
            f_new = objective_and_grad_z(zf, beta)[0]
            if np.isfinite(f_new) and f_new < best_f:
                best_f = f_new
                best_z = zf.copy()
    except Exception:
        pass
    
    return best_z, best_f


def coordinate_descent(a, deadline, rng, n_steps_per_coord=5):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    
    best_a = a.copy()
    best_score = compute_score(a)
    no_imp = 0
    step_sizes = [0.5, 0.3, 0.1, 0.05, 0.02, -0.5, -0.3, -0.1, -0.05, -0.02]
    
    sweep = 0
    while time.monotonic() < deadline:
        sweep += 1
        improved = False
        idx = rng.permutation(n)
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = a[i]
            for ss in step_sizes:
                new_val = orig * (1 + ss)
                if new_val < 1e-15:
                    continue
                a[i] = new_val
                a = a / a.sum()
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_imp = 0
                else:
                    a[i] = orig
                    a = a / a.sum()
                    no_imp += 1
            a = best_a.copy()
        if not improved and no_imp > 2 * n:
            break
    
    return best_a, best_score


def interpolate_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()


def make_diverse_profiles(n, rng):
    """Generate diverse initial profiles."""
    profs = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5]:
        p = 1.0 + alpha * (2*t - 1)
        p = np.maximum(p, 0.01)
        profs.append(p / p.sum())
    
    # Cosine (with phase)
    for alpha in [0.3, 0.5, 0.8]:
        for phase in [0, 0.3, -0.3, 0.5]:
            p = 1.0 + alpha * np.cos(np.pi * t + phase)
            p = np.maximum(p, 0.01)
            profs.append(p / p.sum())
    
    # Power taper
    for p_exp in [0.5, 1.0, 2.0, 3.0]:
        for q_exp in [0.5, 1.0, 2.0, 3.0]:
            p = (t + 0.01) ** p_exp * (1 - t + 0.01) ** q_exp
            p = np.maximum(p, 1e-6)
            profs.append(p / p.sum())
    
    # Gaussian-like
    for sigma in [0.1, 0.15, 0.2, 0.3, 0.5]:
        for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
            p = np.exp(-((t - center) ** 2) / (2 * sigma ** 2)) + 0.05
            profs.append(p / p.sum())
    
    # Sigmoid
    for eta in [5, 10, 15, 20]:
        for center in [0.3, 0.4, 0.5]:
            p = 1.0 / (1.0 + np.exp(-eta * (t - center)))
            p = np.maximum(p, 0.01)
            profs.append(p / p.sum())
    
    # Random smooth (from Fourier modes)
    for _ in range(20):
        coeffs = rng.normal(0, 1, 15)
        p = np.ones(n)
        for k, c in enumerate(coeffs, 1):
            if k % 2 == 1:
                p += c * np.cos(2 * np.pi * k * t)
            else:
                p += c * np.sin(2 * np.pi * k * t)
        p = np.maximum(p, 0.01)
        profs.append(p / p.sum())
    
    return profs
