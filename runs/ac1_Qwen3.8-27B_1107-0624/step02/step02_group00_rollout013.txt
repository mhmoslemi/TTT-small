If `c_k` is a peak (close to the max), we want to reduce it. The value `c_k = sum_i a_i * a_{k-i}`. To reduce `c_k`, we can:
- Reduce `a_i` for `i` that contributes significantly to `c_k`
- But we need to compensate by increasing other `a_j` (to maintain the sum)
- The compensation should be at positions that don't increase other peaks

This is essentially what the LP approach does, but it's expensive. A cheaper approximation:
- For the top `p` peaks, compute the "pressure" on each coordinate
- The pressure on `a_i` is proportional to `sum_{k in top peaks} w_k * a_{k-i}` (where `w_k` is the weight of peak `k`)
- Reduce `a_i` proportional to its pressure, and distribute the freed mass to low-pressure coordinates

This is like a "water-filling" in the pressure space.

    n = len(a)
    c = _fft_convolve(a)
    # Get top n_peaks positions
    top_idx = np.argsort(c)[::-1][:n_peaks]
    top_vals = c[top_idx]
    # Weight by how much above the median
    median_c = np.median(c)
    weights = np.maximum(top_vals - median_c, 0)
    weights = weights / weights.sum()
    
    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k, w in zip(top_idx, weights):
        # c_k = sum_i a_i * a_{k-i}
        # gradient w.r.t. a_i is 2 * a_{k-i} (approximately)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            pressure[i] += w * a[k - i]
    
    # Reduce high-pressure coordinates, increase low-pressure ones
    # Use exponential tilting
    alpha = 0.1  # step size
    a_new = a * np.exp(-alpha * pressure)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _smooth_max_grad(a, beta):
    """Compute smooth max value and gradient w.r.t. a."""
    n = len(a)

    N = _next_pow2(2 * n - 1)

    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return val, grad, float(c.max())

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    best_z = z.copy()

    no_imp = 0
    base_lr = lr
    # BB step tracking
    prev_z = z.copy()
    prev_grad = None
    bb_scale = 1.0

    startup_until = time.monotonic() + 5.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_val(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            continue

        # BB step size adaptation
        if prev_grad is not None and np.all(np.isfinite(prev_grad)):
            sz = z - prev_z
            sg = gz - prev_grad
            dot_ss = float(np.dot(sz, sz))
            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20:
                bb_candidate = dot_ss / max(abs(dot_sg), 1e-20)
                bb_candidate = np.clip(bb_candidate, 0.1, 10.0)
                bb_scale = 0.7 * bb_scale + 0.3 * bb_candidate
        prev_z = z.copy()
        prev_grad = gz.copy()

        cur_lr = base_lr * bb_scale

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - cur_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                base_lr = max(0.003, base_lr * 0.92)

        # Adaptive beta: decrease when peaks are concentrated
        if it % 100 == 0 and it > 100:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                peak_gap = top5[0] - top5[-1]
                if peak_gap < beta * 0.5:
                    beta = max(1e-10, beta * 0.7)
                elif peak_gap > beta * 5.0:
                    beta = min(0.5 * c_max_new, beta * 1.2)

        # Annealing restart
        if no_imp > max(5000, 8 * n):
            noise_scale = 0.3 * best_score
            z_restart = best_z + noise_scale * np.random.default_rng(it * 7919 + 1).standard_normal(n)
            z_restart = np.clip(z_restart, -60.0, 60.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            bb_scale = 1.0
            prev_grad = None
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            no_imp = 0

        z = z_new

    return best_a


def _lp_refine(a, deadline, max_iter=20000, inflate=1.0008, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.005, inflate_cur * 1.05)
            if no_imp > 300:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.1, blend_cur * 0.95)
                inflate_cur = min(1.002, inflate_cur * 0.98)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _gradient_polish_fine(a, deadline, max_iter=30000):
    """Fine gradient polish with very small beta for final refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-8, 0.01 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a)
    z -= np.mean(z)
    lr = 0.01
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_val(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.95)

        if it % 300 == 0:
            beta = max(1e-12, beta * 0.8)

        if no_imp > max(3000, 6 * n):
            break

    return best_a


def _peak_shave(a, deadline, max_iter=5000, n_peaks=5, strength=0.05):
    """Targeted peak shaving: reduce coordinates that contribute most to top peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]
        # Weights proportional to excess over median
        med = np.median(c)
        weights = np.maximum(top_vals - med, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = weights / weights.sum()

        # Compute "pressure" on each coordinate
        pressure = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w * a[j]

        # Apply exponential tilting to reduce pressure
        a_new = a * np.exp(-str_cur * pressure / (pressure.max() + 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            # Slowly increase strength
            str_cur = min(0.2, str_cur * 1.01)
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                str_cur = max(0.01, str_cur * 0.9)
            if no_imp > 1000:
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a


def _multi_start_refine(a, deadline, n_starts=5, rng=None):
    """Try multiple perturbed versions and refine the best."""
    if rng is None:
        rng = np.random.default_rng(12345)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    for i in range(n_starts):
        if time.monotonic() > deadline:
            break
        noise_scale = 0.05 * (i + 1)
        a_pert = a * (1.0 + noise_scale * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        sc = _score(a_pert)
        if sc > best_score * 1.5:
            continue

        # Quick gradient polish
        sub_deadline = min(deadline, time.monotonic() + 30.0)
        a_ref = _logit_adam_optimize(a_pert, sub_deadline, max_iter=5000, lr=0.03)
        sc_ref = _score(a_ref)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Phases:
    1. Diversified initialization (parent, flat-spectrum, bell, linear)
    2. Logit-Adam optimization with adaptive beta and BB step
    3. LP-based spreading refinement
    4. Peak shaving
    5. Fine gradient polish
    6. Multi-start local refinement
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try
    lengths = [512]
    if parent is not None:
        lengths.insert(0, parent.size)
    for ln in [384, 768]:
        if ln not in lengths:
            lengths.append(ln)

    # Phase 1: Generate diverse initializations and pick best
    print(f"Phase 1: Initialization, deadline in {deadline - time.monotonic():.0f}s", flush=True)
    init_pool = []

    for n_t in lengths:
        # Parent interpolation
        if parent is not None:
            init_pool.append(('parent', n_t, _interpolate_profile(parent, n_t)))

        # Linear profile
        for eta in [1.0, 1.5, 2.0]:
            init_pool.append((f'linear_{eta}', n_t, _make_linear_profile(n_t, eta)))

        # Bell profile
        init_pool.append(('bell', n_t, _make_bell_profile(n_t, 0.3)))
        init_pool.append(('sym_bell', n_t, _make_symmetric_bell(n_t, 0.4, 0.3)))

        # Flat spectrum
        init_pool.append(('flat_spec', n_t, _make_flat_spectrum_init(n_t, rng)))

    # Quick evaluation of initializations
    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print(f"  Top 5 inits:", flush=True)
    for sc, name, n_t, _ in init_scores[:5]:
        print(f"    {name} (n={n_t}): {sc:.8f}", flush=True)

    # Phase 2: Main optimization on top candidates
    # Focus on the best 3-4 initializations
    top_inits = init_scores[:4]
    for sc0, name, n_t, a0 in top_inits:
        if time.monotonic() > deadline - 300:
            break
        if sc0 >= best_score and name != 'parent' and n_t != 512:
            continue  # Skip if clearly worse and not the main length

        # Logit Adam optimization
        opt_deadline = min(deadline - 200, time.monotonic() + 400.0)
        print(f"Phase 2: Logit-Adam on {name} (n={n_t}), start={sc0:.8f}, budget={opt_deadline - time.monotonic():.0f}s", flush=True)
        a_opt = _logit_adam_optimize(a0, opt_deadline, max_iter=50000, lr=0.05)
        sc_opt = _score(a_opt)
        print(f"  Logit-Adam result: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # LP refinement
        if time.monotonic() < deadline - 100:
            lp_deadline = min(deadline - 80, time.monotonic() + 200.0)
            print(f"  LP refine, budget={lp_deadline - time.monotonic():.0f}s", flush=True)
            a_lp = _lp_refine(a_opt, lp_deadline, max_iter=20000, inflate=1.0008, blend=0.25)
            sc_lp = _score(a_lp)
            print(f"  LP result: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            a_opt = a_lp

        # Peak shaving
        if time.monotonic() < deadline - 50:
            ps_deadline = min(deadline - 30, time.monotonic() + 60.0)
            a_ps = _peak_shave(a_opt, ps_deadline, max_iter=5000, n_peaks=5, strength=0.03)
            sc_ps = _score(a_ps)
            print(f"  Peak shave: {sc_ps:.10f}", flush=True)
            if sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
            a_opt = a_ps

    # Phase 3: Fine polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 20:
        gp_deadline = min(deadline - 10, time.monotonic() + 120.0)
        print(f"Phase 3: Fine polish, budget={gp_deadline - time.monotonic():.0f}s", flush=True)
        a_fine = _gradient_polish_fine(best_seq, gp_deadline, max_iter=30000)
        sc_fine = _score(a_fine)
        print(f"  Fine polish: {sc_fine:.10f}", flush=True)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()

    # Phase 4: Multi-start local refinement
    if best_seq is not None and time.monotonic() < deadline - 15:
        ms_deadline = deadline - 10
        print(f"Phase 4: Multi-start, budget={ms_deadline - time.monotonic():.0f}s", flush=True)
        a_ms = _multi_start_refine(best_seq, ms_deadline, n_starts=4, rng=rng)
        sc_ms = _score(a_ms)
        print(f"  Multi-start: {sc_ms:.10f}", flush=True)
        if sc_ms < best_score:
            best_score = sc_ms
            best_seq = a_ms.copy()

    # Phase 5: One final LP pass if time allows
    if best_seq is not None and time.monotonic() < deadline - 5:
        lp_deadline2 = deadline - 3
        if lp_deadline2 - time.monotonic() > 10:
            a_last = _lp_refine(best_seq, lp_deadline2, max_iter=5000, inflate=1.0005, blend=0.15)
            sc_last = _score(a_last)
            if sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LP: {sc_last:.10f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        # Keep as-is (score is scale invariant)
        pass
    else:
        final_seq = _make_linear_profile(512, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
