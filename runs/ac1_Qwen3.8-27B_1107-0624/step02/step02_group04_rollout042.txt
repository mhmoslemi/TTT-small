Optimized search for step function coefficients minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses soft-max gradient descent with top-k exact gradient blending, adaptive temperature,

import numpy as np
import time
import math

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_conv(a):
    """Full convolution via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c

def _soft_max(c, beta):
    """Compute softmax approximation of max and weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-10:
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
        return mval, w
    val = mval + beta * math.log(s)
    w = ex / s
    return val, w

def _corr_vec(w, p, n):
    """Compute correlation: (w * p_reversed) gives contribution of each p_i to weighted sum."""
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

    e = np.exp(z)
    p = e + floor
    s = float(p.sum())

    return p

def _make_profile(n, profile_type, param1=1.0, param2=0.0):

def _interp_to_length(p, n_new):
    """Interpolate profile to new length."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:

    x0 = np.linspace(0.0, 1.0, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    while time.monotonic() < deadline:
        t_elapsed = time.monotonic() - t_start
        
        # Compute soft-max gradient
        L, w = _soft_max(c, beta)
        g_soft = 2.0 * _corr_vec(w, p, n)
        
        # Top-k exact gradient blend
        if top_k_blend > 0.0:
            k = min(3, c.size)
            top_idx = np.argsort(c)[::-1][:k]
            g_exact = np.zeros(n)
            for ki in top_idx:
                ki = int(ki)
                # c_ki = sum_i p_i * p_{ki-i} (valid range)
                # d(c_ki)/d(p_i) = p_{ki-i} for valid i
                lo = max(0, ki - (n - 1))
                hi = min(n - 1, ki)
                for i in range(lo, hi + 1):
                    j = ki - i
                    if 0 <= j < n:
                        g_exact[i] += p[j] * (2.0 / len(top_idx))
            g_total = (1.0 - top_k_blend) * g_soft + top_k_blend * g_exact
        else:
            g_total = g_soft
        
        # Variance penalty
        if var_weight > 0.0:
            center = c[n - 1:2 * n - 1] if c.size >= n else c
            c_mean = float(center.mean())
            # Gradient of variance w.r.t. p_i
            # d(var(c_center))/dp_i = 2 * sum_k (c_k - c_mean) * d(c_k)/dp_i
            # Simplified: use (p_i - mean(p)) as proxy
            g_total += var_weight * 2.0 * (p - p.mean())
        
        # Project gradient to sum-p preserving direction
        vp = g_total + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            adam_t = 0
            continue
        
        adam_t += 1
        m_vec = b1 * m_vec + (1 - b1) * gz
        v_vec = b2 * v_vec + (1 - b2) * (gz * gz)
        mh = m_vec / (1 - b1 ** adam_t)
        vh = v_vec / (1 - b2 ** adam_t)
        
        # Adaptive learning rate: decay if stuck
        eff_lr = lr * max(0.3, 1.0 - 0.1 * min(no_imp / 100.0, 1.0))
        z = z - eff_lr * mh / (np.sqrt(vh) + eps)

            z = best_z.copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            adam_t = 0
            continue
        
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()
        p = _softmax_from_z(z)
        c = _full_conv(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        
        # Beta decay
        if adam_t % 30 == 0:
            beta = max(1e-12, beta * beta_decay)
            # Check gap for faster sharpening
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-12, beta * 0.5)
        
        # Restart
        if no_imp > restart_limit:
            noise_scale = 0.1 + 0.3 * min(no_imp / (2.0 * restart_limit), 1.0)
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_conv(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            adam_t = 0
            no_imp = 0
    
    return best_p, best_score

def _deflation_step(p, n_iter=50, step_size=0.05, deadline=None):
    """
    Deflation: reduce elements that contribute most to the peak convolution.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _score(p)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_conv(p)
        k_max = int(np.argmax(c))
        
        # Compute contribution of each p_i to c[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = p[j]
        
        # Normalize contribution
        csum = float(contrib.sum())
        if csum < 1e-30:
            break
        contrib = contrib / csum
        
        # Deflate: reduce high-contribution elements, increase low-contribution
        p_new = p.copy()
        p_new -= step_size * contrib
        p_new += step_size * contrib.mean()  # maintain sum (approximately)
        p_new = np.maximum(p_new, 1e-12)
        p_new = p_new / p_new.sum()
        
        sc_new = _score(p_new)
        if sc_new < best_score:
            best_p = p_new.copy()
            best_score = sc_new
            p = p_new
        else:
            # Try smaller step
            p_try = p.copy()
            p_try -= (step_size * 0.3) * contrib
            p_try += (step_size * 0.3) * contrib.mean()
            p_try = np.maximum(p_try, 1e-12)
            p_try = p_try / p_try.sum()
            sc_try = _score(p_try)
            if sc_try < best_score:
                best_p = p_try.copy()
                best_score = sc_try
                p = p_try
    
    return best_p, best_score

def _coordinate_descent(p, max_iterations=200, deadline=None):
    """
    1D coordinate descent: for each element, find optimal value via ternary search.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _score(p)
    
    idx_order = np.random.default_rng(0).permutation(n)
    
    for it in range(max_iterations):
        if deadline is not None and time.monotonic() > deadline:
            break
        for idx in idx_order:
            if deadline is not None and time.monotonic() > deadline:
                break
            idx = int(idx)
            # Find optimal value for p[idx]
            # p[idx] ranges from 0 to p[idx]_current * some_factor
            lo = 0.0
            hi = p[idx] * 5.0
            # Ternary search
            for _ in range(20):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                p1 = p.copy()
                p1[idx] = m1
                p1 = np.maximum(p1, 1e-12)
                p1 = p1 / p1.sum()
                s1 = _score(p1)
                
                p2 = p.copy()
                p2[idx] = m2
                p2 = np.maximum(p2, 1e-12)
                p2 = p2 / p2.sum()
                s2 = _score(p2)
                
                if s1 < s2:
                    hi = m2
                else:
                    lo = m1
            
            optimal = (lo + hi) / 2.0
            p_new = p.copy()
            p_new[idx] = optimal
            p_new = np.maximum(p_new, 1e-12)
            p_new = p_new / p_new.sum()
            sc_new = _score(p_new)
            
            if sc_new < best_score:
                best_p = p_new.copy()
                best_score = sc_new
                p = p_new
    
    return best_p, best_score

def _scipy_polish(p, deadline, beta=0.01):
    """
    Polish using scipy.optimize with soft-max objective.
    """
    from scipy.optimize import minimize
    
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _score(p)
    
    z0 = np.log(p)
    z0 -= z0.mean()
    
    def objective(z):
        z = np.clip(z, -60.0, 60.0)
        z = z - z.max()
        e = np.exp(z)
        a = e + 1e-15
        a = a / a.sum()
        c = _full_conv(a)
        mval = float(c.max())
        # Smoother approximation
        L, _ = _soft_max(c, beta)
        return float(2.0 * n * L)
    
    def jac(z):
        z = np.clip(z, -60.0, 60.0)
        z = z - z.max()
        e = np.exp(z)
        a = e + 1e-15
        s = a.sum()
        a = a / s
        c = _full_conv(a)
        L, w = _soft_max(c, beta)
        g = 2.0 * _corr_vec(w, a, n)
        dot = float(np.dot(a, g))
        gz = a * (g - dot) / s
        gz -= gz.mean()
        return gz
    
    t_start = time.monotonic()
    # Run with time limit
    max_iter = 200
    result = minimize(objective, z0, method='L-BFGS-B', jac=jac,
                      options={'maxiter': max_iter, 'maxfun': 5000,
                              'ftol': 1e-12, 'gtol': 1e-8})
    
    if np.all(np.isfinite(result.x)):
        z_res = np.clip(result.x, -60.0, 60.0)
        z_res -= z_res.mean()
        p_res = _softmax_from_z(z_res)
        sc_res = _score(p_res)
        if np.isfinite(sc_res) and sc_res < best_score:
            best_p = p_res
            best_score = sc_res
    
    return best_p, best_score

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses soft-max + top-k exact gradient, deflation, scipy polish, and coordinate descent.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
        return c
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c


def _soft_max(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
        return mval, w
    val = mval + beta * math.log(s)
    w = ex / s
    return val, w


def _exact_topk_grad(p, c, k_count=3):
    """Compute exact subgradient using top-k max positions."""
    n = p.size
    k_count = min(k_count, c.size)
    top_idx = np.argsort(c)[::-1][:k_count]
    g = np.zeros(n)
    weights = np.zeros(k_count)
    for ki in range(k_count):
        idx = int(top_idx[ki])
        # Weight: use soft-max weights among top-k
        weights[ki] = c[idx]
    wsum = float(weights.sum())
    if wsum < 1e-30:
        return g
    weights = weights / wsum
    for ki in range(k_count):
        idx = int(top_idx[ki])
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        for i in range(lo, hi + 1):
            j = idx - i
            if 0 <= j < n:
                g[i] += 2.0 * weights[ki] * p[j]
    return g


def _corr_vec(w, p, n):
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(z.max())
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    p = p / s
    return p


def _score(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.inf
    s = float(p.sum())
    if s < 1e-30:
        return np.inf
    c = _full_conv(p)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

def _make_profile(n, ptype, p1=1.0, p2=0.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if ptype == 'uniform':
        a = np.ones(n)
    elif ptype == 'linear':
        a = 1.0 + p1 * u
    elif ptype == 'quad_s':
        a = 1.0 + p1 * u + p2 * u * u
    elif ptype == 'cubic':
        a = 1.0 + p1 * u + p2 * u * u * u
    elif ptype == 'bump':
        a = 1.0 + p1 * u * u
    elif ptype == 'arch':
        a = 1.0 - p1 * u * u
    elif ptype == 'cosine':
        a = 1.0 + p1 * np.cos(np.pi * t)
    elif ptype == 'sine2':
        a = 1.0 + p1 * np.sin(np.pi * t) ** 2
    elif ptype == 'power':
        a = (t + 1.0 / n) ** p1
    elif ptype == 'exp_mid':
        a = np.exp(-p1 * (u * u))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    return a


def _interp_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        p = p.copy()
        s = p.sum()
        if s > 1e-30:
            p = p / s
        return p
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    p_new = p_new / p_new.sum()
    return p_new


def _soft_max_adam(p0, deadline, rng, lr=0.06, beta_frac=0.12, beta_decay=0.72,
                   var_weight=0.02, reg=1e-8, restart_limit=300,
                   top_k_blend=0.0, top_k_count=3, name=""):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(p0)
    z -= z.mean()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    adam_t = 0
    p = p0.copy()
    c = _full_conv(p)
    maxc = float(c.max())
    beta = max(1e-10, beta_frac * maxc)
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        L, w = _soft_max(c, beta)
        g_soft = 2.0 * _corr_vec(w, p, n)

        g_total = g_soft.copy()
        if top_k_blend > 0.0:
            g_exact = _exact_topk_grad(p, c, top_k_count)
            g_total = (1.0 - top_k_blend) * g_soft + top_k_blend * g_exact

        if var_weight > 0.0:
            center = c[n - 1:2 * n - 1] if c.size >= n else c
            c_mean = float(center.mean())
            c_var = float(np.mean((center - c_mean) ** 2))
            if c_var > 1e-20:
                g_total += var_weight * 2.0 * (p - p.mean())

        vp = g_total + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m_vec = b1 * m_vec + (1 - b1) * gz
        v_vec = b2 * v_vec + (1 - b2) * (gz * gz)
        mh = m_vec / (1 - b1 ** adam_t)
        vh = v_vec / (1 - b2 ** adam_t)

        eff_lr = lr * max(0.2, 1.0 - 0.15 * min(no_imp / 80.0, 1.0))
        z = z - eff_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()
        p = _softmax_from_z(z)
        c = _full_conv(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1

        if adam_t % 25 == 0:
            beta = max(1e-12, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.15:
                    beta = max(1e-12, beta * 0.4)

        if no_imp > restart_limit:
            noise_scale = 0.08 + 0.4 * min(float(no_imp - restart_limit) / float(restart_limit), 1.0)
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_conv(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            adam_t = 0
            no_imp = 0

    return best_p, best_score


def _deflation(p, n_iter=300, step_size=0.02, deadline=None):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _score(p)

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_conv(p)
        k_max = int(np.argmax(c))
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = p[j]
        csum = float(contrib.sum())
        if csum < 1e-30:
            break
        contrib = contrib / csum
        p_new = p.copy()
        p_new -= step_size * contrib
        p_new += step_size * contrib.mean()
        p_new = np.maximum(p_new, 1e-12)
        p_new = p_new / p_new.sum()
        sc_new = _score(p_new)
        if sc_new < best_score:
            best_p = p_new.copy()
            best_score = sc_new
            p = p_new

    return best_p, best_score


def _coord_descent_fast(p, n_passes=3, deadline=None):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _score(p)
    rng_cd = np.random.default_rng(12345)

    for _pass in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        order = rng_cd.permutation(n)
        for idx in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            idx = int(idx)
            cur_val = p[idx]
            if cur_val < 1e-10:
                continue
            candidates = [cur_val * 0.5, cur_val * 0.8, cur_val * 1.2, cur_val * 1.5, cur_val * 0.3]
            best_local = cur_val
            for cv in candidates:
                if cv < 0:
                    continue
                p_try = p.copy()
                p_try[idx] = cv
                s_try = float(p_try.sum())
                if s_try < 1e-30:
                    continue
                p_try = p_try / s_try
                sc_try = _score(p_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_local = cv
            if best_local != cur_val:
                p[idx] = best_local
                s = float(p.sum())
                p = p / s
                if _score(p) < best_score + 1e-12:
                    best_p = p.copy()

    return best_p, best_score


def _scipy_polish(p, deadline, beta=0.005):
    from scipy.optimize import minimize

    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _score(p)
    z0 = np.log(p)
    z0 -= z0.mean()

    def obj(z):
        z = np.clip(z, -60.0, 60.0)
        z = z - z.max()
        e = np.exp(z)
        a = e + 1e-15
        a = a / a.sum()
        c = _full_conv(a)
        L, _ = _soft_max(c, beta)
        return float(2.0 * n * L)

    def jac(z):
        z = np.clip(z, -60.0, 60.0)
        z = z - z.max()
        e = np.exp(z)
        a = e + 1e-15
        s = a.sum()
        a = a / s
        c = _full_conv(a)
        L, w = _soft_max(c, beta)
        g = 2.0 * _corr_vec(w, a, n)
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz = gz / s
        gz -= gz.mean()
        return gz

    result = minimize(obj, z0, method='L-BFGS-B', jac=jac,
                      options={'maxiter': 100, 'maxfun': 3000, 'ftol': 1e-14, 'gtol': 1e-10})
    if np.all(np.isfinite(result.x)):
        z_res = np.clip(result.x, -60.0, 60.0)
        z_res -= z_res.mean()
        p_res = _softmax_from_z(z_res)
        sc_res = _score(p_res)
        if np.isfinite(sc_res) and sc_res < best_score:
            best_p = p_res
            best_score = sc_res
    return best_p, best_score
