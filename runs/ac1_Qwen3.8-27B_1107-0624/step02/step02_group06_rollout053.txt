def solve_lp_blending(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0"""
    n = len(a)
    # conv(a, b)_k = sum_j a_j * b_{k-j} for k = 0, ..., 2n-2
    # This is linear in b.
    # Build constraint matrix A such that A @ b <= M * ones(2n-1)
    
    # For constraint k: sum_j a_j * b_{k-j} <= M
    # b_{k-j} is valid when 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k
    # So for constraint k, the coefficients of b_j (for j in [max(0,k-n+1), min(k,n-1)]) are a_{k-j}
    
    # A is (2n-1) x n matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k, n-1)+1):
            A[k, j] = a[k-j]
    
    # Maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    b_rhs = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_rhs, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

def lp_and_blend_step(a, M, t):
    """One LP + blend step."""
    n = len(a)
    
    # Build constraint matrix (sparse)
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k, n-1)+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Solve LP
    c_obj = -np.ones(n)
    b_rhs = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_rhs, 
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True})
    
    if not res.success:
        return None
    
    b = np.maximum(res.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum

# Precomputed LP structures (will be built on first use)
_lp_cache = {}

def get_lp_structure(n):
    """Precompute sparse matrix structure for LP constraints."""
    if n in _lp_cache:
        return _lp_cache[n]
    rows = []
    cols = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    _lp_cache[n] = (rows, cols)
    return rows, cols

def build_lp_matrix(a, n, rows, cols):
    """Build the LP constraint matrix values from sequence a."""
    # A[k,j] = a[k-j] for valid (k,j)
    vals = np.zeros(len(rows))
    for idx in range(len(rows)):
        k = rows[idx]
        j = cols[idx]
        vals[idx] = a[k - j]
    return vals

def solve_lp_for_b(a, M, n):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0."""
    rows, cols = get_lp_structure(n)
    vals = build_lp_matrix(a, n, rows, cols)
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)
    b_rhs = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_rhs,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.success:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-10:

            return b / s, res.ineqlin_dual if hasattr(res, 'ineqlin_dual') else None
    return None, None

            best_sc = best_score

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)
