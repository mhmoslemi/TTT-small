    a = np.array([1.0/np.sqrt(i+1) for i in range(n)])
    a /= a.sum()
    
    for it in range(max_iter):
        # Compute convolution
        c = np.convolve(a, a)
        max_c = c.max()
        
        # Determine active set: indices where c[k] is close to max
        # Use a tolerance that shrinks over iterations
        tol_k = max_c * 0.01 + it * 1e-6
        S = set(np.where(c >= max_c - tol_k)[0])
        
        # Build matrix B where B[i,j] = 1 if i+j in S
        B = np.zeros((n, n))
        for k in S:
            for i in range(max(0, k-n+1), min(k+1, n)):
                j = k - i
                if 0 <= j < n:
                    B[i, j] = 1.0

        # Using KKT: B @ a = lambda * ones, @ sum(a) = 1
        # This is a (n+1) x (n+1) system:
        # [B   1] [a]     [lambda]
        # [1^T 0] [lam] = [1     ]

        M[:n, n] = 1.0
        M[n, :n] = 1.0
        rhs = np.concatenate([np.ones(n), [1.0]])

            sol = np.linalg.solve(M, rhs)
        except np.linalg.LinAlgError:

        if np.min(a_new) >= -1e-10:
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
        else:
            # Some components are negative - need to fix them

            # (simplified: just clip and renormalize for now)
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()

            S = set(np.where(c >= max_c - 1e-4)[0])
            # ... re-solve
            if len(S) <= 1:

"""
Optimized minimax convolution search using 1/sqrt initialization,
soft-max Adam in logit space, active-set linear solving, and
coordinate descent polishing across multiple sequence lengths.
"""
import numpy as np
import time


def next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -50, 0))
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

                if n_free < 2:
                    break
                B_free = B[np.ix_(free, free)]
                Mf = np.zeros((n_free + 1, n_free + 1))
                Mf[:n_free, :n_free] = B_free
                Mf[:n_free, n_free] = 1.0
                Mf[n_free, :n_free] = 1.0
                rhsf = np.concatenate([np.ones(n_free), [1.0]])
                try:
                    solf = np.linalg.solve(Mf + 1e-12 * np.eye(n_free + 1), rhsf)
                except np.linalg.LinAlgError:
                    solf = np.linalg.lstsq(Mf, rhsf, rcond=None)[0]
                a_trial = np.zeros(n)
                a_trial[free] = solf[:n_free]
                a_new = a_trial
                if np.min(a_new) >= -1e-12:
                    break

        a_new = np.maximum(a_new, 1e-15)
        s = float(a_new.sum())
        if s <= 0 or not np.isfinite(s):
            break
        a_new = a_new / s

        c_new = full_convolve_fft(a_new)
        max_new = float(c_new.max())

        if max_new < best_score_unnorm - tol:
            best_score_unnorm = max_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            active_tol *= 0.5
            if active_tol < max_c * 1e-6:
                break

    return best_a, 2.0 * n * best_score_unnorm


def coordinate_descent_polish(p, deadline, rng, n_sweeps=20, step_frac=0.15):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = float(2.0 * n * full_convolve_fft(p).max())
    no_imp_total = 0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            factors = [1 + step_frac, 1 + 0.5 * step_frac, 1 + 2 * step_frac,
                       1 - step_frac, 1 - 0.5 * step_frac, 1 - 0.3 * step_frac]
            best_val = orig
            best_sc = best_score
            for f in factors:
                new_val = orig * f
                if new_val < 1e-15:
                    continue
                p[i] = new_val
                s = p.sum()
                p[i] /= s
                c = full_convolve_fft(p)
                sc = float(2.0 * n * c.max())
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            s = p.sum()
            p = p / s
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp_total = 0
            else:
                no_imp_total += 1

        if improved:
            step_frac = max(0.03, step_frac * 0.85)
        if no_imp_total > n:
            step_frac = max(0.02, step_frac * 0.7)
        if no_imp_total > 3 * n:
            break

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []

    def local_update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            if label:
                print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} {label}")
        if len(candidates) < 30:
            candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
            candidates.sort(key=lambda x: x[0])
        else:
            if float(sc) < candidates[-1][0]:
                candidates[-1] = (float(sc), np.asarray(p, dtype=np.float64).copy())
                candidates.sort(key=lambda x: x[0])

    # Load parent if available
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_nonneg(parent)
    except Exception:
        parent = None

    # Phase 0: Initialization
    print("Phase 0: Initialization")
    if parent is not None:
        sc = internal_score(parent)
        local_update(parent, sc, "parent")
        for tn in (512, 1024):
            if parent.size != tn:
                p_interp = interpolate_profile(parent, tn)
                if p_interp is not None:
                    sc_i = internal_score(p_interp)
                    local_update(p_interp, sc_i, f"parent_n{tn}")

    # Generate profiles at key lengths
    target_lengths = [512, 1024]
    for tn in target_lengths:
        profs = make_custom_profiles(tn)
        for pi, p0 in enumerate(profs[:12]):
            sc = internal_score(p0)
            local_update(p0, sc, f"init_n{tn}_{pi}")

    print(f"  After init: best={best_score:.10f}")

    # Phase 1: Soft-max Adam at n=512
    print("Phase 1: Adam optimization at n=512")
    n1 = 512
    phase1_end = min(deadline, t_start + 400.0)
    init_512 = []
    for sc_c, p_c in candidates:
        if p_c.size == n1:
            init_512.append(p_c)
        elif p_c.size < n1:
            init_512.append(interpolate_profile(p_c, n1))
    if not init_512:
        init_512 = make_custom_profiles(n1)[:6]

    for i in range(min(8, len(init_512))):
        if time.monotonic() > phase1_end:
            break
        per = min(phase1_end, time.monotonic() + 40.0)
        p_best, sc = soft_max_adam(init_512[i], per, rng, lr=0.12, beta_init_frac=0.3, max_restarts=3)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"adam512_{i}")

    print(f"  After phase 1: best={best_score:.10f}")

    # Phase 2: Active-set polish at n=512
    print("Phase 2: Active-set at n=512")
    phase2_end = min(deadline, t_start + 650.0)
    for i in range(min(6, len(candidates))):
        if time.monotonic() > phase2_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size != n1:
            p_c = interpolate_profile(p_c, n1)
        if p_c is None:
            continue
        per = min(phase2_end, time.monotonic() + 30.0)
        p_best, sc_unnorm = active_set_polish(p_c, per)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"as512_{i}")

    print(f"  After phase 2: best={best_score:.10f}")

    # Phase 3: Work at n=1024
    print("Phase 3: Optimization at n=1024")
    n2 = 1024
    phase3_end = min(deadline, t_start + 1200.0)

    # Initialize at n=1024 from best candidates and profiles
    init_1024 = []
    for sc_c, p_c in candidates[:10]:
        p_interp = interpolate_profile(p_c, n2)
        if p_interp is not None:
            init_1024.append(p_interp)
    profs_1024 = make_custom_profiles(n2)
    for p0 in profs_1024[:8]:
        init_1024.append(p0)

    for i in range(min(10, len(init_1024))):
        if time.monotonic() > phase3_end:
            break
        per = min(phase3_end, time.monotonic() + 45.0)
        p_best, sc = soft_max_adam(init_1024[i], per, rng, lr=0.10, beta_init_frac=0.25, max_restarts=3)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"adam1024_{i}")

    # Active-set at n=1024
    for i in range(min(4, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size != n2:
            p_c = interpolate_profile(p_c, n2)
        if p_c is None:
            continue
        per = min(phase3_end, time.monotonic() + 40.0)
        p_best, sc_unnorm = active_set_polish(p_c, per)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"as1024_{i}")

    print(f"  After phase 3: best={best_score:.10f}")

    # Phase 4: Extended Adam on best candidates
    print("Phase 4: Extended optimization")
    phase4_end = min(deadline, t_start + 1550.0)
    for i in range(min(8, len(candidates))):
        if time.monotonic() > phase4_end:
            break
        sc_c, p_c = candidates[i]
        per = min(phase4_end, time.monotonic() + (30.0 if p_c.size <= 512 else 50.0))
        p_best, sc = soft_max_adam(p_c, per, rng, lr=0.08, beta_init_frac=0.15, max_restarts=2)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"ext_{i}")

    print(f"  After phase 4: best={best_score:.10f}")

    # Phase 5: Coordinate descent polish
    print("Phase 5: Coordinate descent")
    phase5_end = min(deadline, t_start + 1750.0)
    for i in range(min(5, len(candidates))):
        if time.monotonic() > phase5_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size > 1024:
            continue
        per = min(phase5_end, time.monotonic() + (30.0 if p_c.size <= 512 else 45.0))
        p_best, sc = coordinate_descent_polish(p_c, per, rng, n_sweeps=15, step_frac=0.12)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"cd_{i}")

    # Active-set on the very best
    if best_seq is not None and best_seq.size <= 1024:
        per = min(phase5_end, time.monotonic() + 30.0)
        p_best, sc_unnorm = active_set_polish(best_seq, per)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, "as_final")

    print(f"  After phase 5: best={best_score:.10f}")

    # Phase 6: Final selection
    print("Phase 6: Final selection")
    final_seq = best_seq
    final_score = best_score

    if final_seq is not None:
        for sc_c, p_c in candidates:
            if p_c.size <= 2048:
                sc_cx = true_score(p_c)
                if np.isfinite(sc_cx) and sc_cx < final_score:
                    final_score = sc_cx
                    final_seq = p_c.copy()

        if final_seq is not None and final_seq.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval) and sc_eval < final_score:
                    final_score = sc_eval
            except Exception:
                pass

        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s
        print(f"  FINAL: score={final_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    fallback = make_inv_sqrt_profile(512)
    print(f"  FALLBACK: score={internal_score(fallback):.10f}")
    return [float(x) for x in fallback]
