Step function optimization: adaptive soft-max gradient descent in logit space,
multi-length exploration (1024, 2048), spectral initialization, and

import numpy as np
import time
from scipy.optimize import minimize_scalar

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = _full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

def _true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e) / float(e.size)
    return p / s

def _soft_max_val_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

            no_imp = 0
            restart_noise *= 0.8
    
    return best_p, best_score

def _coord_descent(p0, deadline, n_sweeps=3):
    """Coordinate descent post-processing: optimize each element in turn."""
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p.size
    best_p = p.copy()
    best_score = _internal_score(p)
    
    for sweep in range(n_sweeps):
        if time.monotonic() >= deadline:
            break
        for i in range(n):
            if time.monotonic() >= deadline:
                break
            old_val = p[i]
            
            # Compute current conv without element i's contribution
            # Actually, it's easier to just evaluate the score at different values of p[i]
            # Score as function of x = p[i]:
            # S = sum(p) - old_val + x
            # conv changes: conv[k] changes by x * (contribution of p[i] to conv[k])
            # For the peak, we need to find which k is the argmax and how it changes
            
            # Simple approach: try a few candidate values
            s_base = float(p.sum()) - old_val
            c_full = _full_convolve_fft(p)
            peak_k = int(np.argmax(c_full))
            
            # The derivative of conv[peak_k] w.r.t. p[i]:
            # d(conv[peak_k])/d(p[i]) = 2*p[peak_k - i] if 0 <= peak_k - i < n
            dk = 0.0
            if 0 <= peak_k - i < n:
                dk = 2.0 * p[peak_k - i]
            
            # Score(x) = 2*n*max_k conv_k(x) / (s_base + x)^2
            # Approximate: if peak_k remains the max, score ≈ 2*n*(c[peak_k] + dk*(x-old_val)) / (s_base+x)^2
            # Minimize: d/dx = 0 -> (c[peak_k] + dk*(x-old_val)) / (s_base+x)^2 is minimized
            # Let A = c[peak_k] - dk*old_val, B = dk
            # f(x) = (A + B*x) / (s_base + x)^2
            # f'(x) = [B*(s_base+x)^2 - (A+B*x)*2*(s_base+x)] / (s_base+x)^4
            #        = [B*(s_base+x) - 2*(A+B*x)] / (s_base+x)^3
            #        = [B*s_base + B*x - 2A - 2B*x] / (s_base+x)^3
            #        = [B*s_base - 2A - B*x] / (s_base+x)^3
            # Set to 0: B*s_base - 2A - B*x = 0 -> x = (B*s_base - 2A) / B = s_base - 2A/B
            
            A = float(c_full[peak_k]) - dk * old_val
            if dk > 1e-15:
                x_opt = s_base - 2.0 * A / dk
            else:
                x_opt = old_val
            
            x_opt = max(0.0, min(x_opt, 50.0 * old_val + 1.0))
            
            # Evaluate score at x_opt vs old_val
            p[i] = x_opt
            s_new = s_base + x_opt
            if s_new > 0.01:
                c_new = _full_convolve_fft(p)
                score_new = float(2.0 * n * c_new.max() / (s_new * s_new))
                if score_new < best_score:
                    best_score = score_new
                    best_p = p.copy()
                # Also try staying at old_val
            else:
                p[i] = old_val
            
            # If x_opt didn't help, try old_val
            p[i] = old_val
            c_old = _full_convolve_fft(p)
            score_old = float(2.0 * n * c_old.max() / (s_base + old_val)**2)
            if score_old <= (score_new if s_new > 0.01 else np.inf):
                p[i] = old_val
        
        # After sweep, accept if improved
        p_test = p.copy()
        s_test = float(p_test.sum())
        if s_test > 0.01:
            c_test = _full_convolve_fft(p_test)
            sc_test = float(2.0 * n * c_test.max() / (s_test * s_test))
            if sc_test < best_score:
                best_score = sc_test
                best_p = p_test.copy()
    
    return best_p, best_score

def _spectral_init(n, rng):
    """Initialize with a flat-spectrum approach."""
    # Start with random phase, flat magnitude
    N = _next_pow2(2 * n - 1)
    A = np.zeros(N // 2 + 1, dtype=np.complex128)
    A[0] = np.sqrt(n)  # DC component
    for k in range(1, N // 2 + 1):
        phase = rng.uniform(0, 2 * np.pi)
        A[k] = np.exp(1j * phase)
    
    # Inverse FFT and take first n elements
    a = np.fft.irfft(A, N)[:n]
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 1e-30:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def _make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()
