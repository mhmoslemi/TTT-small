FISTA-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing of the max, Nesterov accelerated proximal gradient,
and simplex projection. Multiple restarts and temperature annealing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a, b=None):
    """Full convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
        n = a.size
    else:
        b = np.asarray(b, dtype=np.float64)
        n = a.size  # we want length 2n-1 output
    
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:2 * n - 1]
    return c


def _cross_corr(a, w):
    """Compute h_i = sum_j a_j * w_{i+j} for i=0,...,n-1.
    Using FFT: h = convolve(a_reversed, w)[n-1:2n-1]
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    h_full = np.fft.irfft(fa * fw, N)[:2 * n - 1]
    return h_full[n - 1:2 * n - 1]


def _simplex_proj(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = v.size
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).ravel()
    
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - target) / rho
    rho_max = np.max(rho[cond]) if np.any(cond) else 1
    
    theta = (u[:int(rho_max)].sum() - target) / rho_max
    proj = np.maximum(v - theta, 0.0)
    
    # Ensure exact sum
    s = proj.sum()
    if s > 0:
        proj = proj * (target / s)
    else:
        proj = np.full(n, target / n)
    return proj


def _soft_max_weights(c, beta):
    """Compute soft-max weights over c with temperature beta."""
    c = np.asarray(c, dtype=np.float64)
    m = c.max()
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= m - 1e-10
        cnt = mask.sum()
        w[mask] = 1.0 / cnt
        return w
    
    x = (c - m) / beta
    ex = np.exp(x)
    s = ex.sum()
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    w = ex / s
    return w


def _compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = _convolve_fft(a)
    m = c.max()
    return 2.0 * n * m / (s * s)


def _compute_grad_smoth(a, beta):
    """Compute gradient of beta*log(sum exp(c_k/beta)) where c = a*a."""
    n = a.size
    c = _convolve_fft(a)
    w = _soft_max_weights(c, beta)
    h = _cross_corr(a, w)
    return 2.0 * h, c


def _fista_optimize(a0, deadline, rng, 
                    beta_start=0.5, beta_end=0.01, 
                    n_betas=100, lr=0.05, 
                    restart_every=500, noise_scale=0.1):
    """
    FISTA optimization with decreasing temperature.
    a0: initial point on simplex (sum=1, all >= 0)
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 0)
    a0 = a0 / a0.sum()
    n = a0.size
    
    best_a = a0.copy()
    best_score = _compute_score(a0)
    
    t = 1.0  # Nesterov momentum parameter
    x = a0.copy()
    x_prev = a0.copy()
    
    # Betas from beta_start to beta_end
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    iter_in_beta = 0
    total_iter = 0
    
    for bi, beta in enumerate(betas):
        iter_in_beta = 0
        while time.monotonic() < deadline and iter_in_beta < restart_every:
            iter_in_beta += 1
            total_iter += 1
            
            # Compute gradient at y
            y = x + (t - 1) / (t + 1) * (x - x_prev) if t > 1 else x.copy()
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 0:
                y = y / ys
            else:
                y = x.copy()
            
            grad, c = _compute_grad_smoth(y, beta)
            
            # Gradient step
            x_new = y - lr * grad
            x_new = _simplex_proj(x_new, target=1.0)
            
            # Check improvement
            score = _compute_score(x_new)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = x_new.copy()
            
            # Update momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            
            x_prev = x
            x = x_new
            t = t_new
            
            if total_iter % 1000 == 0:
                print(f"  FISTA: iter={total_iter}, beta={beta:.4f}, score={best_score:.10f}")

            x_noisy = _simplex_proj(x_noisy, target=1.0)
            x = x_noisy
            x_prev = x_noisy
            t = 1.0
    
    return best_a, best_score

Finding the exact optimum of a piecewise-linear function requires checking all breakpoints, which could be O(n) per coordinate, O(n²) total per sweep. For n=1024, that's about 1M operations per sweep, which is fine.

"""
FISTA-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing of the max objective with Nesterov accelerated
proximal gradient on the probability simplex, temperature annealing,
multiple restarts, and a subgradient polish phase.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a, b=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:2 * n - 1]
    return c


def _cross_corr(a, w):
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    h_full = np.fft.irfft(fa * fw, N)[:2 * n - 1]
    return h_full[n - 1:2 * n - 1]


def _simplex_proj(v, target=1.0):
    n = v.size
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).ravel()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssv - target) / rho
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho_max = int(idx[-1] + 1)
    theta = (u[:rho_max].sum() - target) / rho_max
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 0:
        proj = proj * (target / s)
    else:
        proj = np.full(n, target / n)
    return proj


def _soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    m = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= m - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - m) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = _convolve_fft(a)
    m = float(c.max())
    return 2.0 * n * m / (s * s)


def _fista_optimize(a0, deadline, rng, beta_start=0.3, beta_end=0.005,
                    n_betas=200, lr=0.08, restart_every=800,
                    noise_scales=None, use_topk=False, topk_frac=0.05):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 < 1e-30:
        return a0, np.inf
    a0 = a0 / s0
    n = a0.size

    best_a = a0.copy()
    best_score = _compute_score(a0)

    x = a0.copy()
    x_prev = a0.copy()
    t = 1.0

    if noise_scales is None:
        noise_scales = [0.05, 0.1, 0.15, 0.2, 0.25]

    betas = np.geomspace(beta_start, beta_end, n_betas)
    total_iter = 0

    for bi, beta in enumerate(betas):
        iter_count = 0
        no_improve = 0
        while time.monotonic() < deadline and iter_count < restart_every:
            iter_count += 1
            total_iter += 1

            if t > 1.0:
                y = x + (t - 1.0) / (t + 1.0) * (x - x_prev)
            else:
                y = x.copy()
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys > 1e-30:
                y = y / ys
            else:
                y = x.copy()

            c = _convolve_fft(y)
            w = _soft_max_weights(c, beta)

            if use_topk:
                k_top = max(2, int(topk_frac * len(c)))
                top_idx = np.argpartition(c, -k_top)[-k_top:]
                w_top = np.zeros_like(w)
                w_top[top_idx] = w[top_idx]
                w_top = w_top / w_top.sum()
                w = w_top

            h = _cross_corr(y, w)
            grad = 2.0 * h

            x_new = y - lr * grad
            x_new = _simplex_proj(x_new, target=1.0)

            score = _compute_score(x_new)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = x_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            t = t_new
            x_prev = x
            x = x_new

            if no_improve > restart_every // 2:
                break

        if bi < len(betas) - 1 and time.monotonic() < deadline:
            ns = noise_scales[bi % len(noise_scales)]
            noise = ns * rng.normal(size=n)
            x_noisy = best_a + noise
            x_noisy = _simplex_proj(x_noisy, target=1.0)
            x = x_noisy
            x_prev = x_noisy
            t = 1.0

        if total_iter % 5000 == 0:
            print(f"  FISTA: iter={total_iter}, beta={beta:.4f}, score={best_score:.10f}")

    return best_a, best_score


def _subgrad_polish(a0, deadline, rng, lr_start=0.005, decay=0.9995):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()
    n = a0.size

    best_a = a0.copy()
    best_score = _compute_score(a0)

    x = a0.copy()
    x_prev = x.copy()
    t = 1.0
    lr = lr_start
    total_iter = 0

    while time.monotonic() < deadline:
        total_iter += 1
        lr *= decay

        c = _convolve_fft(x)
        k_star = int(np.argmax(c))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * x[j]

        if t > 1.0:
            y = x + (t - 1.0) / (t + 1.0) * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-30:
            y = y / ys
        else:
            y = x.copy()

        c_y = _convolve_fft(y)
        k_star_y = int(np.argmax(c_y))
        grad_y = np.zeros(n)
        for i in range(n):
            j = k_star_y - i
            if 0 <= j < n:
                grad_y[i] = 2.0 * y[j]

        x_new = y - lr * grad_y
        x_new = _simplex_proj(x_new, target=1.0)

        score = _compute_score(x_new)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = x_new.copy()

        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        x_prev = x
        x = x_new

        if total_iter % 2000 == 0:
            print(f"  Subgrad: iter={total_iter}, lr={lr:.6f}, score={best_score:.10f}")

    return best_a, best_score


def _local_search_polish(a0, deadline, rng, n_iter=50000):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()
    n = a0.size

    best_a = a0.copy()
    best_score = _compute_score(a0)
    x = a0.copy()

    T = 0.002
    T_min = 0.0001
    cool = 0.9998
    total = 0

    while time.monotonic() < deadline and total < n_iter:
        total += 1
        ps = int(np.clip(rng.integers(1, max(2, n // 20 + 1)), 1, n // 2))
        start = int(rng.integers(0, n - ps + 1))
        end = start + ps

        scale = T * np.sqrt(ps)
        x_new = x.copy()
        x_new[start:end] += rng.normal(0, scale, ps)
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-30:
            continue
        x_new = x_new / s

        score_new = _compute_score(x_new)
        if not np.isfinite(score_new):
            continue

        delta = score_new - best_score
        if delta < 0:
            x = x_new
            best_a = x_new.copy()
            best_score = score_new
            T = max(T_min, T * 0.995)
        else:
            if T > T_min:
                if rng.random() < np.exp(-delta / T):
                    x = x_new
            T = max(T_min, T * cool)

        if total % 10000 == 0:
            print(f"  Local: iter={total}, T={T:.6f}, score={best_score:.10f}")

    return best_a, best_score


def _make_profile(n, type_id):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if type_id == 0:
        a = np.ones(n)
    elif type_id == 1:
        a = 1.0 + 1.5 * (t - 0.5)
    elif type_id == 2:
        a = 1.0 + 2.0 * (t - 0.5)
    elif type_id == 3:
        a = 1.0 + 1.5 * u - 1.0 * u ** 3
    elif type_id == 4:
        a = 1.0 + 1.8 * u - 2.0 * u ** 3
    elif type_id == 5:
        a = 1.0 + 1.0 * u + 0.5 * u ** 2
    elif type_id == 6:
        a = 1.0 + u ** 2
    elif type_id == 7:
        a = 1.0 + 0.5 * u ** 2 - 0.3 * u ** 4
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        a = np.maximum(p, 0.0)
        s = a.sum()
        return a / s if s > 0 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    a = np.interp(x1, x0, p)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
