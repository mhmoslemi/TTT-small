import numpy as np
import time

def _fft_convolve(a):
    """Efficient convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    # Remove small negative values due to numerical error
    c = np.maximum(c, 0)
    return c

def _score(a):
    """Compute the score 2n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of conv(a,a) using FFT."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip for numerical stability
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # Compute gradient: grad_j = 2 * sum_m w[j+m] * a[m]
    # This is 2 * conv(w, a_rev)[n-1 : 2n-1]
    N = 1
    while N < 2*n - 1 + 2*n - 1:
        N <<= 1
    # Actually, we need conv of w (length 2n-1) and a_rev (length n)
    # Result has length 3n-2
    # We want indices n-1 to 2n-2 (the last n elements)
    N = 1
    while N < 3*n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n-1:2*n-1]  # Wait, this should be h_full[n-1 : n-1+n] = h_full[n-1:2n-1]

def _adam_smooth_max(a, deadline, max_iter=50000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s  # normalize to sum = 1
    
    best_a = a.copy()
    best_score = _score(a)
    
    beta = 0.05 * float(_fft_convolve(a).max())  # initial smoothness
    lr = 0.1
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad, mval = _smooth_max_grad(a, beta)
        
        # Add small regularization to keep a away from 0
        grad += 1e-8 * a / (a + 1e-10)  # or just a small constant
        
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        a_new = a - lr * mh / (np.sqrt(vh) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
            continue
        a_new = a_new / s_new
        
        score = _score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.01, lr * 0.9)
        
        if it % 100 == 0:
            beta = max(1e-6, beta * 0.8)
        
        a = a_new
        
        if no_imp > max(2000, 5*n):
            # Restart
            a = best_a * (1 + 0.1*np.random.randn(n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.1
            beta = 0.05 * float(_fft_convolve(a).max())
            no_imp = 0
    
    return best_a

The LP approach:
- Given current a and M = max(conv(a,a)), find direction d that maximizes sum(d) subject to:
  - conv(a, d)[k] <= M for all k (this is linear in d!)
  - d >= 0
  
- Then blend: a_new = (1-t)*a + t*d/sum(d)

The key insight is that conv(a, d) is linear in d (since a is fixed), so this is a valid LP.

The LP has:
- n variables (d_0, ..., d_{n-1})
- 2n-1 constraints (conv(a,d)[k] <= M for k=0,...,2n-2)
- n non-negativity constraints

For n=512, this is 512 variables and 1023 constraints, which is very manageable for modern LP solvers.

The constraint matrix is:
A[k, j] = a_{k-j} (if 0 <= k-j < n), else 0

This is a banded matrix (each row has at most n nonzeros, each column has at most n nonzeros), so it's sparse.

def _next_pow2(n):
    k = 1
    while k < n:

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)

    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0)

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)

    s = w.sum()
    if s < 1e-30:
