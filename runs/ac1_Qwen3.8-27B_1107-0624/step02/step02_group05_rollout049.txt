def project_simplex(v):
    """Project onto simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Solve LP: max sum(g) s.t. g >= 0, conv(f,g) <= M."""
    n = len(f)

                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
