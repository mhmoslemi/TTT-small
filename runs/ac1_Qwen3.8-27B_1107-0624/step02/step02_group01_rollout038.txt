"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using L-BFGS-B with softmax
smoothing, multi-length search, and adaptive temperature annealing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def get_f_from_z(z):
    z_shifted = z - z.max()
    f = np.exp(z_shifted)
    f = f / f.sum()
    return f


def smooth_obj_grad(z, beta):
    """Compute smoothed objective and gradient wrt z."""
    n = z.size
    f = get_f_from_z(z)
    c = fft_convolve(f)
    c_max = float(c.max())
    # Softmax weights
    x = beta * (c - c_max)
    x = np.clip(x, -100.0, 50.0)
    w = np.exp(x)
    w_sum = float(w.sum())
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        w_sum = 1.0
    w = w / w_sum
    # Smoothed objective: 2n * c_max + 2n/beta * log(sum exp(beta*(c-c_max)))
    log_sum_exp = float(np.log(w_sum))  # log(sum exp(beta*(c-c_max)))
    val = 2.0 * n * c_max + 2.0 * n / beta * log_sum_exp
    # Cross-correlation: h_j = sum_k w_k * f_{k-j}
    L = next_pow2(3 * n)
    W = np.fft.rfft(w, L)
    F = np.fft.rfft(f, L)
    h = np.fft.irfft(W * np.conj(F), L)[:n]
    # Gradient wrt f
    grad_f = 4.0 * n * h
    # Gradient wrt z (softmax reparam)
    mean_fg = float(f @ grad_f)
    grad_z = f * (grad_f - mean_fg)
    return val, grad_z


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quad_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=10):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.1, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def optimize_lbfgsb(z0, deadline, rng, betas=None, max_iter_per_beta=500):
    """Run L-BFGS-B optimization with beta annealing."""
    if betas is None:

    z = z0.copy()
    n = z.size
    best_f = get_f_from_z(z)
    best_score = compute_score(best_f)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        def obj_grad(z_curr):
            return smooth_obj_grad(z_curr, beta)
        
        try:

                obj_grad,
                x0=z,
                jac=True,
                method='L-BFGS-B',
                options={
                    'maxfun': max_iter_per_beta * n,  # total function evals
                    'maxiter': max_iter_per_beta,
                    'ftol': 1e-12,
                    'gtol': 1e-8,
                    'gtol_abs': 1e-10,
                }

            score = compute_score(f)
            if score < best_score:

        except Exception as e:

            # Try a simpler gradient step

    best_score = compute_score(f)

        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()
        
        f = get_f_from_z(z)
        score = compute_score(f)
        
        if score < best_score:
            best_score = score
            best_f = f.copy()
            no_imp = 0
        else:

            z += 0.05 * rng.normal(size=n)

            m = np.zeros(n)

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using L-BFGS-B with softmax smoothing,

import numpy as np

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def get_f_from_z(z):
    z_shifted = z - z.max()
    f = np.exp(z_shifted)
    f = f / f.sum()
    return f


def smooth_obj_grad(z, beta):
    n = z.size
    f = get_f_from_z(z)
    c = fft_convolve(f)
    c_max = float(c.max())
    x = beta * (c - c_max)
    x = np.clip(x, -50.0, 0.0)
    w = np.exp(x)
    w_sum = float(w.sum())
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        w_sum = 1.0
    w = w / w_sum
    log_sum_exp = float(np.log(w_sum))
    val = 2.0 * n * c_max + 2.0 * n / beta * log_sum_exp
    L = next_pow2(3 * n)
    W = np.fft.rfft(w, L)
    F = np.fft.rfft(f, L)
    h = np.fft.irfft(W * np.conj(F), L)[:n]
    grad_f = 4.0 * n * h
    mean_fg = float(f @ grad_f)
    grad_z = f * (grad_f - mean_fg)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros(n)
    return val, grad_z


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quad_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a2, a3):

    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u + a3 * u * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=12):

def interpolate_profile(p0, n_new):

    n0 = p0.size
    if n0 == n_new:

    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None
