    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines adaptive softmax gradient descent, LP direction finding,
subgradient methods, and interpolation across sequence lengths.
Key improvements: aggressive temperature scheduling, interleaved
methods, peak-targeted auxiliary gradients, and diverse restarts.

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_max_conv(a):
    c = fft_convolve(a)
    return float(c.max())


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * (2.0 * t - 1.0))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_gaussian_profile(n, center, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=12):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 1.2) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def compute_gradient_conv(f, w, N_corr):
    """Compute cross-correlation h[i] = sum_k w[k] * f[k-i] via FFT."""
    fw = np.fft.rfft(w, N_corr)
    fa = np.fft.rfft(f, N_corr)
    h = np.fft.irfft(fw * np.conj(fa), N_corr)[:f.size]
    return h


def softmax_grad_optimize(f0, deadline, rng, lr=0.05, max_iter=200000,
                          beta_init_frac=0.08, beta_decay=0.6,
                          restart_threshold=None):
    """Adaptive softmax gradient descent in logit space with Adam."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-12, beta_init_frac * M_val)

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    if restart_threshold is None:
        restart_threshold = max(800, 4 * n)

    N_corr = next_pow2(3 * n)
    step_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute softmax weights
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -200.0, 10.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        # Gradient in z-space
        h = compute_gradient_conv(f, w, N_corr)
        grad_z = 2.0 * f * (h - float(np.dot(f, h)))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        # Adaptive LR: scale inversely with gradient magnitude
        grad_norm = float(np.sqrt(np.mean(grad_z ** 2))) + 1e-12
        adaptive_lr = lr * (0.1 / grad_norm) ** 0.3

        z = z - adaptive_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
            step_count = 0
        else:
            no_imp += 1
            step_count += 1

        # Adaptive beta cooling
        if step_count % 200 == 0:
            beta = max(1e-14, beta * beta_decay)

        # Restarts with increasing aggression
        if no_imp > restart_threshold:
            strength = min(1.5, 0.4 + 0.1 * (no_imp // restart_threshold))
            z = (1.0 - 0.3) * best_z + strength * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.08 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
            step_count = 0

    return best_f, best_M


def subgradient_optimize(f0, deadline, rng, lr_init=0.3, max_iter=100000):
    """Subgradient method with projection onto simplex."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    best_f = f.copy()
    best_M = compute_max_conv(f)
    no_imp = 0
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(f)
        M = float(c.max())

        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Find top values for weighted subgradient
        top_k = min(5, c.size)
        idx = np.argsort(c)[-top_k:]
        w = np.zeros(c.size)
        for i in idx:
            w[i] = 1.0
        w = w / w.sum()

        # Compute subgradient: 2 * sum_k w[k] * a[k-i]
        N_corr = next_pow2(3 * n)
        h = compute_gradient_conv(f, w, N_corr)
        grad = 2.0 * h
        # Project gradient to be orthogonal to the constraint sum(a)=1
        grad = grad - np.mean(grad)

        # Step
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-30:
            f_new = best_f.copy()
        else:
            f_new = f_new / s

        f = f_new

        # Decay LR
        lr = max(0.01, lr * 0.999)

        # Restart if stuck
        if no_imp > 2000:
            f = best_f + 0.05 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 1e-30:
                f = f / s
            else:
                f = best_f.copy()
            lr = lr_init * 0.5
            no_imp = 0
            best_M = compute_max_conv(f)

    return best_f, best_M


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

                if fs < 1e-30:

                    M = best_M

                f = f / fs
                M = compute_max_conv(f)

                    best_f = f.copy()

        g_sum = float(g.sum())

            try_update(p_best, sc)
            print(f"Phase 4 (n={n2}): score={sc:.10f}")

    # Phase 5: Try n=256 (shorter might give better ratio)
    n3 = 256
    if best_seq is not None and best_seq.size != n3 and time.monotonic() < phase4_end:
        p_short = interpolate_profile(best_seq, n3)
        if p_short is not None:
            per = min(phase4_end, time.monotonic() + 50.0)
            p_best, M_best = softmax_grad_optimize(p_short, per, rng, lr=0.06,
                                                   beta_init_frac=0.08)
            sc = compute_score(p_best)
            try_update(p_best, sc)
            print(f"Phase 5 (n={n3}): score={sc:.10f}")

    # Phase 6: Final polish on absolute best
    if best_seq is not None and time.monotonic() < deadline - 10:
        per = min(deadline - 5, time.monotonic() + 40.0)
        n_best = best_seq.size
        if n_best <= 600:
            p_best, M_best = combined_optimize(best_seq, per, rng, max_iter=50000)
        else:
            p_best, M_best = softmax_grad_optimize(best_seq, per, rng, lr=0.02,
                                                   beta_init_frac=0.04)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        print(f"Phase 6 (final): score={sc:.10f}")

    # Final check
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final: score={sc_final:.10f}, n={final.size}")

        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Evaluated: {sc_eval:.10f}")
            except Exception as e:
                print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    print("Using fallback")
    return [float(x) for x in fb]
