"""
Phase-alignment optimization: minimize max autocorrelation by making the spectrum
as flat as possible (e^{-iθt} templates + gradient refinement in logit space).
Multi-start, multi-length, with Adam and cosine annealing.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score_vec(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def make_phase_profile(n, theta):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-1j * theta * t)
    return np.abs(a).copy()


def make_phase_profile_real(n, theta):
    """Use cos(theta*t) shifted to be nonnegative - gives oscillatory structure."""
    t = np.linspace(0.0, 1.0, n)
    a = np.cos(theta * t)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def make_smooth_cosine(n, freq_start, freq_end, n_harm):
    """Sum of cosines with different frequencies, shifted to nonnegative."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * 0.5
    for k in range(n_harm):
        f = freq_start + k * (freq_end - freq_start) / max(1, n_harm - 1)
        a += 0.3 * np.cos(2.0 * np.pi * f * t + 0.5 * k)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, alpha):
    """Parabolic: 1 + alpha*(2t-1)^2, high at edges."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=12):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * 0.8
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.1, 0.6) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def compute_obj_grad(z, N, beta, a, c, w, h, N_corr):
    n = z.size
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    bc = beta * c
    bc_shifted = bc - bc.max()
    exp_bc = np.exp(np.clip(bc_shifted, -100.0, 50.0))
    s_exp = float(exp_bc.sum())
    if s_exp < 1e-30:
        exp_bc = np.zeros_like(c)
        exp_bc[int(np.argmax(c))] = 1.0
        s_exp = 1.0
    w = exp_bc / s_exp
    smooth_max = c.max() + np.log(s_exp) / beta
    obj = 2.0 * n * smooth_max

    N_corr = next_pow2(3 * n)
    fw = np.fft.rfft(w, N_corr)
    fa_corr = np.fft.rfft(a, N_corr)
    h = np.fft.irfft(fw * np.conj(fa_corr), N_corr)[:n]

    sh = float(np.dot(h, a))
    grad_z = 4.0 * n * a * (h - sh)

    cache = (a, c, w, h, N, N_corr)
    return obj, grad_z, cache


def adam_optimize_z(z0, deadline, rng, lr=0.03, beta_start=80.0, beta_end=200.0,
                    max_iter=100000, verbose=False):
    n = z0.size
    z = z0.copy().astype(np.float64)
    z -= z.mean()
    z = np.clip(z, -30.0, 30.0)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    # Initial computation
    obj, grad, cache = compute_obj_grad(z, 0, beta_start, None, None, None, None, 0)
    a, c, w, h, N, N_corr = cache

    best_obj = obj
    best_z = z.copy()
    no_imp = 0
    imp_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        # Cosine annealing for beta
        progress = min(1.0, it / max(1, max_iter * 0.7))
        beta = beta_start + (beta_end - beta_start) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Cosine LR schedule
        lr_cur = lr * (0.5 + 0.5 * np.cos(np.pi * min(1.0, it / max_iter)))
        lr_cur = max(lr_cur, lr * 0.05)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -40.0, 40.0)
        z_new -= z_new.mean()

        # Evaluate
        obj_new, grad_new, cache_new = compute_obj_grad(z_new, 0, beta, None, None, None, None, 0)
        a_new, c_new, w_new, h_new, N_new, N_corr_new = cache_new

        if np.isfinite(obj_new) and obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_z = z_new.copy()
            no_imp = 0
            imp_count += 1
        else:
            no_imp += 1
            # Revert if too bad
            if obj_new > obj * 1.5:
                z_new = z
                grad_new = grad
                obj_new = obj

        z = z_new
        grad = grad_new
        obj = obj_new

        if no_imp > 500:
            # Restart: jump near best
            z = best_z + 0.2 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
            beta = beta_start * 0.8
            obj_new, grad_new, cache_new = compute_obj_grad(z, 0, beta, None, None, None, None, 0)
            a, c, w, h, N, N_corr = cache_new
            obj = obj_new
            grad = grad_new

        if verbose and it % 500 == 0:
            print(f"      Adam iter {it}: obj={obj:.10f}, best={best_obj:.10f}, beta={beta:.1f}")

    return best_z, best_obj


def gradient_descent_a_space(a0, deadline, rng, lr_start=0.01, max_iter=50000):
    """Gradient descent directly in a-space with simplex projection."""
    n = a0.size
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return a0, np.inf
    a = a / s

    obj = 2.0 * n * fft_convolve(a).max()
    best_obj = obj
    best_a = a.copy()

    # Track active set (lags near the max)
    active_set = [int(np.argmax(fft_convolve(a)))]

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(a)
        M = float(c.max())
        obj = 2.0 * n * M

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Update active set: lags where c[k] > 0.8 * M
        active = np.where(c > 0.85 * M)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Subgradient: average gradient over active set
        # grad[k] for c[k]: d(c[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n
        grad = np.zeros(n)
        for k in active[:30]:  # Limit to top 30 active constraints
            lag = int(k)
            for j in range(n):
                m = lag - j
                if 0 <= m < n:
                    grad[j] += 2.0 * a[m]
        grad /= len(active[:30])

        # Projected gradient step
        lr = lr_start / (1.0 + 0.01 * it)
        a_new = a - lr * grad
        # Project onto simplex
        a_new = project_simplex(a_new)
        a = a_new

    return best_a, best_obj


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def make_deterministic_starts(n, rng):
    """Generate diverse deterministic and random starting profiles."""
    starts = []
    # Quadratic profiles (U-shaped)
    for alpha in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        starts.append(make_quadratic_profile(n, alpha))
    # Cubic profiles
    for a1, a3 in ((1.0, -0.5), (1.5, -1.0), (2.0, -1.5), (0.5, 0.3), (1.2, -0.8)):
        starts.append(make_cubic_profile(n, a1, a3))
    # Phase-based
    for theta in (3.0, 5.0, 7.0, 10.0, 15.0, 20.0):
        starts.append(make_phase_profile_real(n, theta))
    # Smooth random
    for _ in range(8):
        starts.append(make_smooth_random(n, rng, n_harm=10))
    # Pure random (smoothed)
    for _ in range(3):
        a = np.abs(rng.normal(size=n)) + 0.1
        # Smooth with moving average
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-12)
        starts.append(a / a.sum())
    return starts


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def z_from_a(a):
    """Convert a (sum=1, positive) to logit z."""
    a = np.maximum(a, 1e-15)
    z = np.log(a)
    z -= z.mean()
    return z


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Evaluate parent and deterministic profiles
    2. Adam optimization in logit space at n=512
    3. Extended optimization and length scaling
    4. Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score_vec(parent)
        try_update(parent, sc)
        print(f"Parent: n={parent.size}, score={sc:.10f}")

    # Phase 1: Quick scan of deterministic profiles at n=512
    n1 = 512
    phase1_end = min(deadline, t_start + 60.0)

    starts_512 = make_deterministic_starts(n1, rng)
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts_512.insert(0, p_int)

    # Evaluate all starts
    scored_starts = []
    for s in starts_512:
        sc = compute_score_vec(s)
        scored_starts.append((sc, s))
    scored_starts.sort(key=lambda x: x[0])
    for sc, s in scored_starts[:5]:
        try_update(s, sc)
    print(f"Phase 1: top 5 starts, best={best_score:.10f}")
    for i, (sc, s) in enumerate(scored_starts[:10]):
        print(f"  start {i}: score={sc:.8f}")

    # Phase 2: Adam optimization at n=512 from top starts
    phase2_end = min(deadline, t_start + 900.0)
    top_k = min(8, len(scored_starts))
    per_start = max(15.0, (phase2_end - t_start) / max(1, top_k))

    candidates_512 = []
    for i in range(top_k):
        if time.monotonic() > phase2_end - 10:
            break
        sc0, a0 = scored_starts[i]
        z0 = z_from_a(a0)
        per = min(phase2_end - 5, time.monotonic() + per_start)
        z_best, obj_best = adam_optimize_z(z0, per, rng, lr=0.04,
                                            beta_start=60.0, beta_end=300.0,
                                            max_iter=80000, verbose=False)
        # Convert back
        a_best = np.exp(z_best - z_best.max())
        a_best = a_best / a_best.sum()
        sc = compute_score_vec(a_best)
        try_update(a_best, sc)
        candidates_512.append((sc, a_best))
        print(f"  Phase 2 start {i}: score={sc:.10f} (t={time.monotonic()-t_start:.1f}s)")

    candidates_512.sort(key=lambda x: x[0])
    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Further refinement on best candidates
    phase3_end = min(deadline, t_start + 1350.0)
    top_k2 = min(4, len(candidates_512))
    per_cand = max(20.0, (phase3_end - t_start) / max(1, top_k2))

    for i in range(top_k2):
        if time.monotonic() > phase3_end - 10:
            break
        sc0, a0 = candidates_512[i]
        z0 = z_from_a(a0)
        # Use smaller learning rate for refinement
        per = min(phase3_end - 5, time.monotonic() + per_cand)
        z_best, obj_best = adam_optimize_z(z0, per, rng, lr=0.015,
                                            beta_start=150.0, beta_end=500.0,
                                            max_iter=60000, verbose=False)
        a_best = np.exp(z_best - z_best.max())
        a_best = a_best / a_best.sum()
        sc = compute_score_vec(a_best)
        try_update(a_best, sc)
        print(f"  Phase 3 cand {i}: score={sc:.10f} (t={time.monotonic()-t_start:.1f}s)")

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Try n=1024 by interpolation
    n2 = 1024
    phase4_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            sc_ext = compute_score_vec(p_ext)
            print(f"  Interpolated to n={n2}: score={sc_ext:.10f}")
            z_ext = z_from_a(p_ext)
            per = min(phase4_end - 10, time.monotonic() + 200.0)
            z_best, obj_best = adam_optimize_z(z_ext, per, rng, lr=0.03,
                                                beta_start=80.0, beta_end=400.0,
                                                max_iter=50000, verbose=True)
            a_best = np.exp(z_best - z_best.max())
            a_best = a_best / a_best.sum()
            sc = compute_score_vec(a_best)
            try_update(a_best, sc)
            print(f"  Phase 4 (n={n2}): score={sc:.10f}")

    # Phase 5: Also try n=768
    n3 = 768
    phase5_end = min(deadline, t_start + 1720.0)
    if best_seq is not None and time.monotonic() < phase5_end - 30:
        p_ext3 = interpolate_profile(best_seq, n3)
        if p_ext3 is not None:
            sc_ext3 = compute_score_vec(p_ext3)
            print(f"  Interpolated to n={n3}: score={sc_ext3:.10f}")
            z_ext3 = z_from_a(p_ext3)
            per = min(phase5_end - 5, time.monotonic() + 80.0)
            z_best3, obj_best3 = adam_optimize_z(z_ext3, per, rng, lr=0.03,
                                                  beta_start=80.0, beta_end=350.0,
                                                  max_iter=30000, verbose=False)
            a_best3 = np.exp(z_best3 - z_best3.max())
            a_best3 = a_best3 / a_best3.sum()
            sc3 = compute_score_vec(a_best3)
            try_update(a_best3, sc3)
            print(f"  Phase 5 (n={n3}): score={sc3:.10f}")

    # Phase 6: Final polish on absolute best
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = best_seq.size
        z_best_final = z_from_a(best_seq)
        per = min(deadline - 3, time.monotonic() + 50.0)
        z_fin, obj_fin = adam_optimize_z(z_best_final, per, rng, lr=0.01,
                                          beta_start=300.0, beta_end=800.0,
                                          max_iter=30000, verbose=False)
        a_fin = np.exp(z_fin - z_fin.max())
        a_fin = a_fin / a_fin.sum()
        sc_fin = compute_score_vec(a_fin)
        try_update(a_fin, sc_fin)
        print(f"After Phase 6 polish: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # Final: ensure valid output
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score_vec(final)
        print(f"\nFinal: score={sc_final:.10f}, n={final.size}")

        # Verify with evaluator if small enough
        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Verified with evaluator: {sc_eval:.10f}")
            except Exception as e:
                print(f"Eval check error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_quadratic_profile(512, 1.0)
    print(f"Fallback used, score={compute_score_vec(fb):.6f}")
    return [float(x) for x in fb]
