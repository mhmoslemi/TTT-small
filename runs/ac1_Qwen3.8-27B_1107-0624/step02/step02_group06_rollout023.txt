M - alpha*M + 2*alpha*t* < M
2*alpha*t* < alpha*M
2*t* < M (for alpha > 0)

So if t* < M/2, any alpha > 0 will decrease the peak (approximately). If M/2 <= t* < M, we need a small enough alpha.

In practice, t* should be significantly less than M if the current sequence is far from optimal. So the LP approach should make good progress.

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find descent directions, with coordinate descent polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.special import expit


def full_convolve(a):
    """Compute linear convolution a*a using FFT for efficiency."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Clean up numerical noise
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    """Compute the evaluation score."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = full_convolve(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))


def build_conv_matrix(a):
    """
    Build the matrix A such that (A @ g)[k] = (a*g)[k] for linear convolution.
    A has shape (2n-1, n).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        # (a*g)[k] = sum_j a[k-j] * g[j] where 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build convolution matrix more efficiently using matrix operations."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # Create index matrix
    k = np.arange(2 * n - 1)[:, None]  # (2n-1, 1)
    j = np.arange(n)[None, :]  # (1, n)
    idx = k - j  # (2n-1, n), A[k,j] = a[idx[k,j]] if 0 <= idx < n
    mask = (idx >= 0) & (idx < n)
    A = np.zeros_like(idx, dtype=np.float64)
    A[mask] = a[idx[mask]]
    return A


def solve_lp(a, M):
    """
    Solve the LP: minimize t
    subject to: (a*g)_k <= t for all k
                g >= 0
                sum(g) = 1
    
    Returns: optimal g, optimal t
    """
    n = a.size
    A_conv = build_conv_matrix_fast(a)  # (2n-1, n)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_conv @ g - t <= 0
    # => A_conv @ g + (-1) * t <= 0
    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, t is free (but will be >= 0)
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None, None
    
    g = result.x[:n]
    t_val = result.x[n]
    g = np.maximum(g, 0)  # Clean up
    s = g.sum()
    if s < 1e-10:
        return None, None
    g = g / s
    return g, t_val

LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions at each step,
combined with coordinate descent polishing and multi-length search.
"""
import numpy as np
import time
from scipy.optimize import linprog


def full_convolve(a):
    n = a.size
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def score(a):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = full_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def conv_matrix(a):
    """Build matrix A where (A @ g) = linear_conv(a, g)."""
    n = a.size
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((2*n-1, n))
    A[mask] = a[idx[mask]]
    return A

def solve_lp_direction(a):
    """
    Solve: min t s.t. (a*g)_k <= t, g >= 0, sum(g) = 1.
    Returns (g, t_val) or (None, None) on failure.
    """
    n = a.size
    A_c = conv_matrix(a)  # (2n-1, n)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.hstack([A_c, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    
    g = np.maximum(res.x[:n], 0)
    s = g.sum()
    if s < 1e-10:
        return None, None
    g /= s
    return g, float(res.x[n])

def lp_optimize(a, deadline, max_iters=10000):
    """Main LP-based optimization loop."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best = a.copy()
    best_sc = score(a)
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g, t_val = solve_lp_direction(a)
        if g is None:
            break
        
        # Line search
        best_alpha = 0.0
        best_trial = best_sc
        for alpha in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            a_t = (1 - alpha) * a + alpha * g
            a_t = np.maximum(a_t, 0)
            st = a_t.sum()
            if st < 1e-10:
                continue
            a_t /= st
            sc_t = score(a_t)
            if sc_t < best_trial:
                best_trial = sc_t
                best_alpha = alpha
        
        if best_alpha == 0.0:
            stall += 1
            if stall > 30:
                break
            continue
        
        a = (1 - best_alpha) * a + best_alpha * g
        a = np.maximum(a, 0)
        a /= a.sum()
        
        sc = score(a)

            best_sc = sc
            best = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
