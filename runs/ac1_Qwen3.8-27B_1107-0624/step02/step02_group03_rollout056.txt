def subset_lp_direction(f, M, K=50, timeout=5.0):
    """
    Solve LP using only the top K constraints (where conv(f,g) is most likely to be binding).
    """
    n = len(f)

    # Find the K most "active" constraints: those where A_full @ f is closest to M

    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def subset_lp_direction(f, M, K=40, timeout=8.0):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)[x]<=M, using top-K constraints."""
    n = len(f)
    A_full = build_conv_matrix(f)
    Af = A_full @ f
    # Pick the K constraints most likely to be active (largest Af values)
    idx = np.argsort(Af)[-K:]
    A = A_full[idx, :]
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def fft_correlation(w, p):
    """Compute h[i] = sum_{j=0}^{n-1} w[i+j] * p[j] for i=0..n-1 using FFT."""
    n = len(p)
    size = 4 * n
    W = np.fft.rfft(w, size)
    P = np.fft.rfft(p, size)
    corr = np.fft.irfft(W * np.conj(P), size)
    return corr[:n]


def adam_polish(p0, deadline, lr=0.05, reg=1e-10, verbose=True, label=""):
    """
    FFT-accelerated Adam gradient descent in logit space.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.15 * max_c)
    start_beta = beta
    iterations_since_best = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Soft argmax weights (FFT-friendly)
        x = (c - max_c) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Gradient via FFT
        h_arr = fft_correlation(w, p0)
        grad_p = 2.0 * h_arr + reg * p0

        # Logit space gradient
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
            iterations_since_best = 0
        else:
            no_imp += 1
            iterations_since_best += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.92)

        # Anneal beta (soft-argmax temperature)
        if it % 80 == 0:
            beta = max(1e-14, beta * 0.65)

        # Restart with perturbation if stuck
        if no_imp > 800:
            rng_r = np.random.default_rng(1000 + it)
            z = 0.85 * best_z + 0.15 * rng_r.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-12, start_beta * 0.5)

        p0 = p

    return best_p, best_score


def lp_polish(f, max_iters=200, t_init=0.35, t_min=0.02, deadline=None,
              K=40, verbose=True, label=""):
    """
    Subset LP-based iterations to reduce max(conv(f,f)).
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M

    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g_raw = subset_lp_direction(f, M, K=K, timeout=5.0)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Try full step first
        g_self_score = 2.0 * n * float(np.convolve(g, g).max())

        if g_self_score < best_score * 0.9999:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
        else:
            f_new = (1.0 - t_step) * f + t_step * g
            f_new = normalize(f_new)
            if f_new is None:
                break
            c_new = np.convolve(f_new, f_new)
            M_new = float(c_new.max())
            score_new = 2.0 * n * M_new

            if score_new < best_score - 1e-14:
                best_f = f_new.copy()
                best_score = score_new
                best_M = M_new
                f = f_new.copy()
                M = M_new
                no_improve = 0
                t_step = max(t_min, t_step * 0.997)
            else:
                no_improve += 1
                f = best_f.copy()
                M = best_M
                if no_improve % 30 == 0:
                    t_step = max(t_min, t_step * 0.7)
                if no_improve > 150:
                    break

        if verbose and it % 50 == 0:
            print(f"    {label} LP iter {it}: score={best_score:.8f}, t={t_step:.4f}")

    return best_f, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_power_profile(n, p=1.0):
    """Create profile f(t) = t^p, normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = t**p + 1e-10
    return a / a.sum()


def make_beta_profile(n, alpha=1.5, beta=1.0):
    """Create beta-like profile f(t) = t^alpha * (1-t)^(beta-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = (t + 1e-10)**alpha * ((1.0 - t) + 1e-10)**(beta - 1.0)
    return a / a.sum()


def make_cos_profile(n):
    """Create cosine ramp profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * t / 2)
    return a / a.sum()


def make_asymmetric_profile(n, eta=1.5, p=1.0):
    """Create asymmetric profile f(t) = (t + eta*(t-0.5))^p."""
    t = np.linspace(0.0, 1.0, n)
    u = t + eta * (t - 0.5)
    a = np.maximum(u, 1e-10)**p
    return a / a.sum()


def make_multi_poly(n, coeffs):
    """Profile from polynomial: a(u) = sum coeffs[k]*u^k, u=2t-1."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.ones(n)
    for k, c in enumerate(coeffs, start=1):
        a += c * (u ** k)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
      1. Multi-start with diverse profiles at n=512
      2. Fast Adam on each to find good local minima
      3. LP refinement on the best
      4. Scale up to n=768, n=1024
      5. Final Adam polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 10.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Try to load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent
                best_score = internal_score(parent)
                print(f"Parent: n={len(parent)}, score={best_score:.10f}")
    except Exception:
        pass

    # === Phase 1: Multi-start at n=512 ===
    n1 = 512
    phase1_deadline = t_start + 250.0

    # Generate diverse candidates
    candidates = []
    
    if best_seq is not None:
        c1 = interpolate_seq(best_seq, n1)
        if c1 is not None:
            candidates.append(c1)

    profiles = [
        make_power_profile(n1, 1.0),
        make_power_profile(n1, 1.3),
        make_power_profile(n1, 1.5),
        make_power_profile(n1, 1.8),
        make_power_profile(n1, 2.0),
        make_beta_profile(n1, 1.5, 1.0),
        make_beta_profile(n1, 2.0, 1.0),
        make_cos_profile(n1),
        make_asymmetric_profile(n1, 1.5, 1.0),
        make_asymmetric_profile(n1, 0.8, 1.5),
        make_multi_poly(n1, [1.5, -0.5]),
        make_multi_poly(n1, [2.0, -1.0, 0.3]),
        make_multi_poly(n1, [1.0, 0.3, -0.2, 0.1]),
    ]
    candidates.extend(profiles)

    # Quick Adam on each candidate (30s each, but we limit total time)
    per_candidate_time = 20.0
    print(f"\nPhase 1: Multi-start at n={n1}, {len(candidates)} candidates")

    for i, cand in enumerate(candidates):
        if time.monotonic() > phase1_deadline - 10:
            break
        sc = internal_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"  Candidate {i}: score={sc:.10f} (new best)")

    # Adam polish the current best
    if best_seq is not None and time.monotonic() < phase1_deadline:
        print(f"  Adam polishing best at n={n1}...")
        p_best, sc_best = adam_polish(best_seq, phase1_deadline, lr=0.06,
                                       verbose=True, label="P1")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 1 best: n={n1}, score={best_score:.10f}")

    # === Phase 2: LP refinement at n=512 ===
    phase2_deadline = t_start + 700.0
    if time.monotonic() < phase2_deadline - 30 and best_seq is not None:
        n2 = 512
        a2 = interpolate_seq(best_seq, n2)
        if a2 is not None:
            print(f"\nPhase 2: LP refinement at n={n2}, score={internal_score(a2):.10f}")
            f_best, sc_best = lp_polish(a2, max_iters=300, t_init=0.35,
                                         deadline=phase2_deadline, K=40,
                                         verbose=True, label="P2")
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  Phase 2 best: score={best_score:.10f}")

            # Quick Adam after LP
            if time.monotonic() < phase2_deadline:
                p_best, sc_best = adam_polish(best_seq, phase2_deadline, lr=0.04,
                                               verbose=True, label="P2a")
                if sc_best < best_score:
                    best_score = sc_best
                    best_seq = p_best
                    print(f"  Post-LP Adam: score={best_score:.10f}")

    # === Phase 3: Try larger n ===
    phase3_deadline = t_start + 1350.0

    for n_try in [768, 1024]:
        if time.monotonic() > phase3_deadline - 60:
            break
        if best_seq is None:
            break

        a_try = interpolate_seq(best_seq, n_try)
        if a_try is None:
            continue

        sc_try = internal_score(a_try)
        print(f"\nPhase 3: Trying n={n_try}, interpolated score={sc_try:.10f}")

        # Adam polish at this size
        remaining_to_phase3 = phase3_deadline - time.monotonic()
        if remaining_to_phase3 < 30:
            break

        p_best, sc_best = adam_polish(a_try, time.monotonic() + min(remaining_to_phase3 * 0.6, 250.0),
                                       lr=0.05, verbose=True, label=f"P3_{n_try}")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 3 n={n_try} best: score={best_score:.10f}")
        else:
            print(f"  Phase 3 n={n_try} did not improve (got {sc_best:.10f})")
            # Keep the smaller n version

    # === Phase 4: Try n=512 again with fresh Adam (different lr) ===
    phase4_deadline = deadline
    if time.monotonic() < phase4_deadline - 30 and best_seq is not None:
        n4 = len(best_seq)
        if n4 != 512:
            a4 = interpolate_seq(best_seq, 512)
            if a4 is not None:
                sc4 = internal_score(a4)
                if sc4 < best_score:
                    best_score = sc4
                    best_seq = a4
                    n4 = 512

        print(f"\nPhase 4: Final Adam at n={n4}, score={internal_score(best_seq):.10f}")
        p_best, sc_best = adam_polish(best_seq, phase4_deadline, lr=0.03,
                                       verbose=True, label="P4")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 4 best: score={best_score:.10f}")

    # === Final: one more pass at the best n ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_final = len(best_seq)
        print(f"\nFinal polish at n={n_final}...")
        p_best, sc_best = adam_polish(best_seq, deadline - 2, lr=0.02,
                                       verbose=False, label="Final")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Final best: score={best_score:.10f}")

    # Validation
    final_score = internal_score(best_seq)
    print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.10f} ===")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_power_profile(512, 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
