```
def gradient_with_weights(w, a, L):
    n=len(a)
    w_full=np.zeros(L)
    Lc=min(len(w), 2*n-1)
    w_full[:Lc]=w[:Lc]
    ar=np.zeros(L)
    if n > 0:
        ar[0]=a[0]
        if n > 1:
            ar[L-n+1:] = a[:0:-1]
    Fw=np.fft.rfft(w_full, L)
    Far=np.fft.rfft(ar, L)
    g=np.fft.irfft(Fw * Far, L)[:n]
```

```
def conv_fft(a):
    n = int(a.size)
```

```python
"""
Multi-resolution search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.

Algorithm:
  * Start from the provided sequence when available, plus several analytic
    random smooth profiles.
  * Use LP direction-finding at moderate lengths (512, 1024).
  * Use FFT-based soft-max / active-set Adam polish in logit space with a
    small L2 flatness penalty.
  * Interpolate to larger lengths (2048, 4096) and polish.
  * Keep a pool of candidates and select the best by the official evaluator
    near the end.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def fft_len(n):
    n = int(max(1, n))
    return 1 << (n - 1).bit_length()


def conv_fft(a):
    a = np.asarray(a, dtype=float).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=float)
    Lc = 2 * n - 1
    L = fft_len(Lc)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:Lc]
    c = np.asarray(c, dtype=float)
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def score(a):
    a = np.asarray(a, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-12:
        return np.inf
    c = conv_fft(a)
    mc = float(c.max()) if c.size else 0.0
    return float(2.0 * a.size * mc / (s * s))


def normalize(a):
    a = np.asarray(a, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return None
    return a / s


def softmax_logits(z):
    z = np.asarray(z, dtype=float).ravel()
    if z.size == 0:
        return np.array([], dtype=float)
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -50.0, 50.0)
    z = z - float(z.max())
    p = np.exp(z)
    s = float(p.sum())
    if s <= 1e-300:
        return np.ones(z.size, dtype=float) / float(z.size)
    return p / s


def make_weights(c, beta, topK):
    Lc = len(c)
    if Lc == 0:
        return np.array([], dtype=float)
    cmax = float(c.max())
    if topK is None:
        topK = 0
    else:
        topK = int(topK)

    if 0 < topK < Lc:
        idx = np.argpartition(c, -topK)[-topK:]
        w = np.zeros(Lc, dtype=float)
        if beta <= 1e-18:
            w[idx] = 1.0 / float(topK)
        else:
            xm = (c[idx] - cmax) / float(beta)
            xm = np.clip(xm, -50.0, 0.0)
            val = np.exp(xm)
            s = float(val.sum())
            if s <= 1e-300:
                w[idx] = 1.0 / float(topK)
            else:
                w[idx] = val / s
        return w

    if beta <= 1e-18:
        w = np.zeros(Lc, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return w

    x = (c - cmax) / float(beta)
    x = np.clip(x, -50.0, 0.0)
    w = np.exp(x)
    s = float(w.sum())
    if s <= 1e-300:
        w = np.zeros(Lc, dtype=float)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / s
    return w


def interpolate_seq(a, n_new):
    a = np.asarray(a, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    if a.size == 0:
        return None
    s = float(a.sum())
    if s <= 1e-300:
        return None
    a = a / s
    n_old = a.size
    if n_old == int(n_new):
        return a.copy()
    if n_old == 1:
        return np.full(int(n_new), 1.0 / int(n_new), dtype=float)

    x_old = np.linspace(0.0, 1.0, n_old, dtype=float)
    x_new = np.linspace(0.0, 1.0, int(n_new), dtype=float)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    return normalize(out + 1e-12)


def smooth_seq(a, sigma):
    a = normalize(a)
    if a is None:
        return None
    sigma = float(sigma)
    if sigma <= 0.0:
        return a.copy()
    n = a.size
    if n < 3:
        return a.copy()
    rad = int(max(1, 3 * sigma + 1))
    rad = int(min(rad, n // 2))
    x = np.arange(-rad, rad + 1, dtype=float)
    k = np.exp(-(x * x) / (2.0 * sigma * sigma))
    k = k / k.sum()
    full = np.convolve(a, k, mode="full")
    start = rad
    out = full[start:start + n]
    return normalize(out)


def make_profile(n, coeffs):
    t = np.linspace(0.0, 1.0, int(n), dtype=float)
    u = 2.0 * t - 1.0
    a = np.ones(int(n), dtype=float)
    for k, c in enumerate(coeffs, start=1):
        a = a + float(c) * (u ** k)
    return normalize(a + 1e-8)


def initial_candidates(seed=42):
    cands = []

    try:
        p = np.asarray(height_sequence_1, dtype=float).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if p.size > 0 and s > 1e-12:
            cands.append(p / s)
    except Exception:
        pass

    rng = np.random.default_rng(seed)

    for n in (512, 1024):
        t = np.linspace(0.0, 1.0, n, dtype=float)
        u = 2.0 * t - 1.0

        for eta in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
            a = np.maximum(1.0 + eta * u, 1e-8)
            cands.append(a / a.sum())

        eps = 1e-3
        for p in (1.0, 2.0):
            a = (1.0 + eps + (1.0 - t) ** p) ** (-0.5)
            cands.append(a / a.sum())
            a = (1.0 + eps + t ** p) ** (-0.5)
            cands.append(a / a.sum())

        for eps in (0.01, 0.05, 0.2):
            a = 1.0 / np.sqrt(t + eps)
            cands.append(a / a.sum())
            a = 1.0 / np.sqrt(1.0 - t + eps)
            cands.append(a / a.sum())

        for _ in range(10):
            deg = int(rng.integers(2, 6))
            coeffs = rng.normal(0.0, 1.0, size=deg) * (0.5 / np.arange(1, deg + 1, dtype=float))
            cands.append(make_profile(n, coeffs.tolist()))

        for _ in range(5):
            mu = float(rng.uniform(0.2, 0.8))
            sig = float(rng.uniform(0.1, 0.4))
            a = np.exp(-((t - mu) ** 2) / (2.0 * sig * sig))
            cands.append(a / a.sum())

    return cands


def consider_candidate(a, best_a, best_score, pool=None):
    a = normalize(a)
    if a is None:
        return best_a, best_score
    sc = score(a)
    if np.isfinite(sc) and sc < best_score:
        best_a = a
        best_score = sc
        if pool is not None and len(pool) < 80:
            pool.append(a.copy())
    return best_a, best_score


def eval_exact(a):
    a = normalize(a)
    if a is None:
        return np.inf
    try:
        val = evaluate_sequence([float(x) for x in a])
        if np.isfinite(val):
            return float(val)
    except Exception:
        pass
    return score(a)


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def extract_lp_solution(x, n):
    if x is None:
        return None
    x = np.asarray(x, dtype=float).ravel()
    if x.size < n:
        return None
    g = x[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def lp_min_t(f, timeout=5.0):
    if linprog is None:
        return None
    f = normalize(f)
    if f is None:
        return None
    n = f.size
    if n < 2 or n > 1500:
        return None

    A = build_conv_matrix(f)
    L = A.shape[0]
    A_ub = np.hstack((A, -np.ones((L, 1), dtype=float)))
    b_ub = np.zeros(L, dtype=float)
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    cobj = np.zeros(n + 1, dtype=float)
    cobj[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    opts = {"time_limit": float(max(0.5, timeout))}
    try:
        res = linprog(
            cobj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=1.0,
            bounds=bounds,
            method="highs",
            options=opts,
        )
    except Exception:
        return None
    return extract_lp_solution(getattr(res, "x", None), n)


def lp_max_sum(f, M, timeout=5.0):
    if linprog is None:
        return None
    f = normalize(f)
    if f is None:
        return None
    n = f.size
    if n < 2 or n > 1500:
        return None
    if not np.isfinite(M) or M < 0.0:
        return None

    A = build_conv_matrix(f)
    L = A.shape[0]
    cobj = -np.ones(n, dtype=float)
    b_ub = np.full(L, float(M), dtype=float)
    bounds = [(0.0, 1.0)] * n

    opts = {"time_limit": float(max(0.5, timeout))}
    try:
        res = linprog(
            cobj,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options=opts,
        )
    except Exception:
        return None
    return extract_lp_solution(getattr(res, "x", None), n)


def line_search_simple(a, b, initial_score, max_eval=8):
    a = normalize(a)
    b = normalize(b)
    if a is None or b is None:
        return None, initial_score, 0.0

    n = a.size
    best_a = a.copy()
    best_s = float(initial_score)
    best_t = 0.0

    if max_eval >= 8:
        ts = (1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04)
    elif max_eval >= 5:
        ts = (1.0, 0.6, 0.3, 0.1)
    else:
        ts = (1.0, 0.4)

    for t in ts:
        tv = float(t)
        x = (1.0 - tv) * a + tv * b
        x = np.maximum(x, 0.0)
        s = float(x.sum())
        if s <= 1e-300:
            continue
        x = x / s
        c = conv_fft(x)
        sc = float(2.0 * n * c.max())
        if sc < best_s - 1e-13:
            best_a = x
            best_s = sc
            best_t = tv

    if 0.02 < best_t < 0.98 and max_eval >= 6:
        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        for _ in range(6):
            mid = 0.5 * (lo + hi)
            x = (1.0 - mid) * a + mid * b
            x = np.maximum(x, 0.0)
            s = float(x.sum())
            if s <= 1e-300:
                lo = mid
                continue
            x = x / s
            c = conv_fft(x)
            sc = float(2.0 * n * c.max())
            if sc < best_s - 1e-14:
                best_a = x
                best_s = sc
                best_t = mid
                hi = mid
            else:
                lo = mid

    return best_a, best_s, best_t


def lp_loop(a, deadline, timeout=5.0, max_iters=2000, verbose=False):
    a = normalize(a)
    if a is None:
        return None, np.inf
    n = a.size
    if n > 1500:
        return a, score(a)

    c = conv_fft(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    it = 0
    no_imp = 0
    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem <= 5.0:
            break
        if no_imp > 12:
            break
        if it >= int(max_iters):
            break

        M = float(c.max())
        use_timeout = min(float(max(0.5, timeout)), max(0.5, 0.25 * rem))

        b = None
        if it % 3 == 2:
            b = lp_max_sum(a, M, use_timeout)
            if b is None:
                b = lp_min_t(a, use_timeout)
        else:
            b = lp_min_t(a, use_timeout)
            if b is None:
                b = lp_max_sum(a, M, use_timeout)

        if b is not None:
            cand, sc, _ = line_search_simple(a, b, best_score, max_eval=6)
            if cand is not None and sc < best_score - 1e-12:
                best_a = cand
                best_score = sc
                a = cand
                no_imp = 0
                c = conv_fft(a)
            else:
                no_imp += 1
        else:
            no_imp += 1

        it += 1
        if verbose and it % 20 == 0:
            print(f"    lp_iter {it}: score={best_score:.8f}")

    return best_a, best_score


def adam_refine(a0, deadline, lr=0.12, beta_ratio=0.2, topk_ratio=0.25,
                flat_mu=0.15, seed=0, verbose=False):
    a = normalize(a0)
    if a is None:
        return a0, np.inf

    n = a.size
    Lc = 2 * n - 1
    L = fft_len(Lc)
    avg_c = 1.0 / float(Lc)

    z = np.log(np.maximum(a, 1e-300))
    z = z - float(np.mean(z))
    best_z = z.copy()
    best_a = a.copy()

    c = conv_fft(a)
    best_score = float(2.0 * n * c.max())

    cmax = float(c.max())
    spread = max(cmax - avg_c, cmax * 0.1, 1e-15)
    beta = float(beta_ratio) * spread
    base_beta = beta

    topK = int(max(16, Lc * topk_ratio)) if topk_ratio > 0 else 0
    if topK > Lc // 2:
        topK = 0
    base_topK = topK

    base_lr = float(lr)
    base_flat = float(flat_mu)
    flat_mu = base_flat

    rng = np.random.default_rng(abs(int(seed)) % (2**31 - 1) or 12345)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=float)
    v = np.zeros(n, dtype=float)
    step = 0
    no_imp = 0
    restarts = 0

    rfft = np.fft.rfft
    irfft = np.fft.irfft

    a_full = np.zeros(L, dtype=float)
    ar = np.zeros(L, dtype=float)
    w_full = np.zeros(L, dtype=float)

    while time.monotonic() < deadline:
        step += 1

        if step % 250 == 0:
            cmax = float(c.max())
            spread = max(cmax - avg_c, 1e-15)
            beta = max(beta * 0.75, spread * 1e-6)
            if base_flat > 0.0:
                flat_mu = max(0.0, flat_mu * 0.9)
            if topK > 0:
                topK = max(16, int(topK * 0.85))

        w_max = make_weights(c, beta, topK)
        if base_flat > 0.0 and flat_mu > 1e-8:
            w_total = w_max + flat_mu * 2.0 * (c - avg_c)
        else:
            w_total = w_max

        w_full.fill(0.0)
        w_full[:Lc] = w_total

        ar.fill(0.0)
        ar[0] = a[0]
        if n > 1:
            start = L - n + 1
            if 0 <= start < L:
                ar[start:] = a[:0:-1]

        gw = rfft(w_full, L)
        ga = rfft(ar, L)
        g = irfft(gw * ga, L)[:n]
        g = 2.0 * np.asarray(g, dtype=float)

        if not np.all(np.isfinite(g)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            step = 0
            a = softmax_logits(z)
            c = conv_fft(a)
            continue

        dot = float(np.dot(a, g))
        gz = a * (g - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            step = 0
            a = softmax_logits(z)
            c = conv_fft(a)
            continue

        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)

        mh = m / (1.0 - b1 ** step)
        vh = v / (1.0 - b2 ** step)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z = z - float(np.mean(z))
        a = softmax_logits(z)

        c = conv_fft(a)
        sc = float(2.0 * n * c.max())

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.02, lr * 0.9)

            if no_imp > 2000 and restarts < 35:
                noise = 0.20 + 0.05 * float(restarts)
                z = best_z.copy() + noise * rng.normal(size=n)
                z = np.clip(z, -40.0, 40.0)
                z = z - float(np.mean(z))
                a = softmax_logits(z)
                m.fill(0.0)
                v.fill(0.0)
                step = 0
                no_imp = 0
                restarts += 1
                lr = base_lr
                c = conv_fft(a)
                cmax = float(c.max())
                spread = max(cmax - avg_c, 1e-15)
                beta = min(base_beta * 2.0, max(0.3 * spread, spread * 1e-6))
                topK = base_topK
                if base_flat > 0.0:
                    flat_mu = min(base_flat, 0.5 * base_flat + 0.05)

        if no_imp > 8000:
            break

        if verbose and step % 1000 == 0:
            print(f"    adam_iter {step}: score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    deadline = t0 + budget

    final_pool = []
    best_a = None
    best_score = np.inf

    candidates = initial_candidates(seed)
    for cand in candidates:
        best_a, best_score = consider_candidate(cand, best_a, best_score, final_pool)

    if best_a is None:
        best_a = np.full(512, 1.0 / 512.0, dtype=float)
        best_score = score(best_a)
        final_pool.append(best_a.copy())

    print(f"t={time.monotonic()-t0:7.1f}s initial best score={best_score:.8f} n={best_a.size}")

    # Phase 1: moderate length 512, LP + Adam.
    if deadline - time.monotonic() > 700.0:
        a = interpolate_seq(best_a, 512)
        if a is not None:
            for sig in (0.0, 2.0, 5.0):
                sm = smooth_seq(a, sig)
                best_a, best_score = consider_candidate(sm, best_a, best_score, final_pool)

            lp_deadline = min(t0 + 220.0, deadline - 650.0)
            if lp_deadline > time.monotonic() + 60.0:
                a_lp, sc_lp = lp_loop(a, lp_deadline, timeout=3.0, max_iters=1000, verbose=False)
                best_a, best_score = consider_candidate(a_lp, best_a, best_score, final_pool)
                print(f"t={time.monotonic()-t0:7.1f}s n=512 LP score={sc_lp:.8f}")

            adam_deadline = min(t0 + 400.0, deadline - 550.0)
            if adam_deadline > time.monotonic() + 30.0:
                a_ad, sc_ad = adam_refine(
                    best_a,
                    adam_deadline,
                    lr=0.12,
                    beta_ratio=0.2,
                    topk_ratio=0.3,
                    flat_mu=0.2,
                    seed=101,
                    verbose=False,
                )
                best_a, best_score = consider_candidate(a_ad, best_a, best_score, final_pool)
                print(f"t={time.monotonic()-t0:7.1f}s n=512 Adam score={sc_ad:.8f}")

    # Phase 2: length 1024, LP + Adam.
    if deadline - time.monotonic() > 500.0:
        a = interpolate_seq(best_a, 1024)
        if a is not None:
            for sig in (0.0, 4.0, 10.0):
                sm = smooth_seq(a, sig)
                best_a, best_score = consider_candidate(sm, best_a, best_score, final_pool)

            lp_deadline = min(t0 + 850.0, deadline - 600.0)
            if lp_deadline > time.monotonic() + 90.0:
                a_lp, sc_lp = lp_loop(a, lp_deadline, timeout=8.0, max_iters=500, verbose=False)
                best_a, best_score = consider_candidate(a_lp, best_a, best_score, final_pool)
                print(f"t={time.monotonic()-t0:7.1f}s n=1024 LP score={sc_lp:.8f}")

            adam_deadline = min(t0 + 1300.0, deadline - 350.0)
            if adam_deadline > time.monotonic() + 60.0:
                a_ad, sc_ad = adam_refine(
                    best_a,
                    adam_deadline,
                    lr=0.12,
                    beta_ratio=0.15,
                    topk_ratio=0.25,
                    flat_mu=0.15,
                    seed=202,
                    verbose=False,
                )
                best_a, best_score = consider_candidate(a_ad, best_a, best_score, final_pool)
                print(f"t={time.monotonic()-t0:7.1f}s n=1024 Adam score={sc_ad:.8f}")

    # Phase 3: length 2048, no LP, Adam only.
    if deadline - time.monotonic() > 350.0:
        a = interpolate_seq(best_a, 2048)
        if a is not None:
            sm = smooth_seq(a, 8.0)
            best_a, best_score = consider_candidate(sm, best_a, best_score, final_pool)

            a = best_a
            d1 = min(t0 + 1550.0, deadline - 300.0)
            if d1 > time.monotonic() + 60.0:
                a_ad, sc_ad = adam_refine(
                    a,
                    d1,
                    lr=0.15,
                    beta_ratio=0.2,
                    topk_ratio=0.25,
                    flat_mu=0.10,
                    seed=303,
                    verbose=False,
                )
                best_a, best_score = consider_candidate(a_ad, best_a, best_score, final_pool)
                print(f"t={time.monotonic()-t0:7.1f}s n=2048 Adam score={sc_ad:.8f}")

    # Phase 4: length 4096, polish.
    if deadline - time.monotonic() > 300.0:
        a = interpolate_seq(best_a, 4096)
        if a is not None:
            sm = smooth_seq(a, 10.0)
            best_a, best_score = consider_candidate(sm, best_a, best_score, final_pool)

            a = best_a
            d1 = deadline - 180.0
            if d1 > time.monotonic() + 80.0:
                a_ad, sc_ad = adam_refine(
                    a,
                    d1,
                    lr=0.10,
                    beta_ratio=0.12,
                    topk_ratio=0.2,
                    flat_mu=0.08,
                    seed=404,
                    verbose=False,
                )
                best_a, best_score = consider_candidate(a_ad, best_a, best_score, final_pool)
                print(f"t={time.monotonic()-t0:7.1f}s n=4096 Adam score={sc_ad:.8f}")

    # Final coarse-fine restarts if time remains.
    if deadline - time.monotonic() > 220.0:
        rng = np.random.default_rng(seed + 7777)
        n_best = best_a.size
        base_n = max(128, n_best // 4)
        sub_deadline = deadline - 160.0
        for i in range(4):
            if time.monotonic() >= sub_deadline:
                break
            a_coarse = interpolate_seq(best_a, base_n)
            if a_coarse is None:
                continue
            z = np.log(np.maximum(a_coarse, 1e-300))
            z = z + 0.5 * rng.normal(size=z.size)
            p_coarse = softmax_logits(z)
            a_pert = interpolate_seq(p_coarse, n_best)
            if a_pert is None:
                continue
            remaining = max(30.0, sub_deadline - time.monotonic())
            chunk = min(120.0, remaining / max(1, 4 - i))
            sub = min(sub_deadline, time.monotonic() + chunk)
            if sub <= time.monotonic() + 20.0:
                continue
            a_ad, sc_ad = adam_refine(
                a_pert,
                sub,
                lr=0.12,
                beta_ratio=0.2,
                topk_ratio=0.2,
                flat_mu=0.10,
                seed=500 + i,
                verbose=False,
            )
            best_a, best_score = consider_candidate(a_ad, best_a, best_score, final_pool)
            print(f"t={time.monotonic()-t0:7.1f}s restart {i} score={sc_ad:.8f}")

    # Exact evaluation of a small final pool.
    best_final_a = normalize(best_a)
    if best_final_a is None:
        best_final_a = np.full(512, 1.0 / 512.0, dtype=float)
    best_final_score = eval_exact(best_final_a)

    if time.monotonic() < deadline - 10.0:
        pool = final_pool[-12:]
        for cand in pool:
            if time.monotonic() > deadline - 8.0:
                break
            cand = normalize(cand)
            if cand is None:
                continue
            sc = eval_exact(cand)
            if np.isfinite(sc) and sc < best_final_score:
                best_final_score = sc
                best_final_a = cand

    out = normalize(best_final_a)
    if out is None:
        out = np.full(512, 1.0 / 512.0, dtype=float)
    out = np.asarray(out, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 1e-300:
        out = np.full(512, 1.0 / 512.0, dtype=float)
    else:
        out = out / s

    print(f"final returned score (official or fallback) = {best_final_score:.8f}, length={out.size}")
    return [float(x) for x in out]
```
