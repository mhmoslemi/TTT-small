"""
Optimizes non-negative sequences to minimize 2n*max(conv(s,s))/(sum(s))^2.
Uses a combination of LP-based direction finding (via scipy linprog), FFT-based
smooth gradient descent, adaptive beta scheduling, and multiple initializations
including structured sequences and the provided height_sequence_1.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def compute_smooth_gradient_fft(s, w):
    """
    Compute sum_k w[k] * 2*s[k-i] for each i (the smooth gradient direction).
    Uses FFT for efficiency. Returns array of length n.
    """
    n = len(s)
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    W = np.fft.rfft(w, N_fft)
    # We want: grad[i] = sum_k w[k] * 2 * s[k-i] (where 0<=k-i<=n-1)
    # This is 2 * cross-correlation of w and s
    # cross_corr[i] = sum_k w[k] * s[k-i]
    term1 = np.fft.irfft(np.conj(S) * W, N_fft)[0:n]
    term2 = np.fft.irfft(S * W, N_fft)[0:n]
    return 2.0 * (term1 + term2)


def gradient_step(s, beta=5.0, eta=0.3):
    """One multiplicative gradient step using smooth-max weighting."""
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = conv.max()
    if max_conv <= 0:
        return s.copy()
    # Soft weighting emphasizing peaks
    shifted = conv - max_conv
    w = np.exp(beta * shifted / max_conv)
    w = w / w.sum()
    grad = compute_smooth_gradient_fft(s, w)
    grad = np.maximum(grad, 0.0)
    grad_max = grad.max()
    if grad_max < 1e-15:
        return s.copy()
    s_new = s * np.exp(-eta * grad / grad_max)
    s_new = np.maximum(s_new, 1e-15)
    s_new = s_new / s_new.sum()
    return s_new


def build_lp_matrix(s):
    """Build the constraint matrix A for the LP. A[k,j] = s[k-j] if valid."""
    n = len(s)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def lp_direction(s, M, time_limit=5.0):
    """Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M for all k, b >= 0."""
    n = len(s)
    A = build_lp_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
        elif res.status == 2:  # time limit
            if res.x is not None:
                return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        mask[cutoff:] = 0.0
        phases = 2 * np.pi * rng.random(len(freqs))
        spec = mask * np.exp(1j * phases)
        s = np.fft.irfft(spec * n, n)
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        candidates.append(s)

    # Pure random
    for _ in range(3):
        s = rng.random(n) + 0.05
        s = s / s.sum()
        candidates.append(s)

    # M-sequence like (for n that's close to 2^k - 1)
    if (n + 1 & n) == 0 or n == 2**int(np.log2(n+1)) - 1:
        # Try a simple pattern
        s = np.zeros(n)
        for i in range(n):
            # Simple LFSR-like pattern
            s[i] = 1.0 if (i * 7 + 3) % 11 < 6 else 0.3
        s = s / s.sum()
        candidates.append(s)

    return candidates


def refined_gradient_polish(s, deadline, rng):
    """Fine-tune with adaptive gradient steps, focusing on peak reduction."""
    n = len(s)
    s = s.copy() / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)

    eta = 0.1
    beta = 10.0

    for it in range(50000):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(s, s)
        max_conv = conv.max()
        if max_conv <= 0:
            break

        # Focus even more on the top peaks
        threshold = max_conv * 0.8
        above = conv >= threshold
        w = np.where(above, np.exp(3.0 * (conv - max_conv) / max_conv), 0.0)
        if w.sum() < 1e-10:
            w = np.exp(3.0 * (conv - max_conv) / max_conv)
        w = w / w.sum()

        grad = compute_smooth_gradient_fft(s, w)
        grad = np.maximum(grad, 0.0)
        grad_max = grad.max()
        if grad_max < 1e-15:
            break

        s_new = s * np.exp(-eta * grad / grad_max)
        s_new = np.maximum(s_new, 1e-12)
        s_new = s_new / s_new.sum()

        score_new = fast_score(s_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_s = s_new.copy()
            s = s_new
            # Slowly increase beta
            if it % 500 == 0:
                beta = min(beta + 0.5, 50.0)
                eta = max(eta * 0.98, 0.02)
        else:
            # Tiny perturbation to escape
            if it % 100 == 0:
                s = s + 0.005 * rng.standard_normal(n)
                s = np.maximum(s, 1e-6)
                s = s / s.sum()
            else:
                break  # If we can't improve, stop

    return best_s, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for non-negative sequence minimizing 2n*max(conv(s,s))/(sum(s))^2.
    Uses LP-based direction finding combined with FFT gradient steps,
    multiple initializations, and adaptive parameter tuning.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0

    global_best_score = float('inf')
    global_best_seq = None

    # Load height_sequence_1 if available
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            print(f"Loaded height_sequence_1, length={h.size}, score={fast_score(h/h.sum()):.6f}")
    except Exception:
        pass

    # Phase 1: Quick screening across n values
    print("\n=== Phase 1: Screening ===")
    n_values = [128, 192, 256, 384, 512]

    phase1_deadline = time.monotonic() + min(300, (deadline - time.monotonic()) * 0.25)

    candidates_for_phase2 = []

    for n in n_values:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 20:
            break
        print(f"\n  n={n}, time_left={remaining:.0f}s")

        inits = generate_initializations(n, rng, hseq)
        # Score and sort
        scored = [(fast_score(s), s) for s in inits]
        scored.sort(key=lambda x: x[0])

        # Try top 3 initializations with brief optimization
        top_k = min(3, len(scored))
        time_per = max(10.0, remaining / (top_k + 1))

        for i in range(top_k):
            if time.monotonic() > phase1_deadline - 5:
                break
            sc, s0 = scored[i]
            if sc >= global_best_score and i > 0:
                break
            print(f"    init {i}: score={sc:.6f}")
            cand_deadline = time.monotonic() + time_per
            use_lp = (n <= 384)
            best_s, best_sc = combined_optimize(s0, cand_deadline, rng,
                                                 use_lp=use_lp, lp_interval=10,
                                                 verbose=False)
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"    ** New best: {best_sc:.6f} (n={n})")
            candidates_for_phase2.append((best_sc, best_s, n))

    # Phase 2: Focused optimization on best candidates
    print(f"\n=== Phase 2: Focused optimization (best so far: {global_best_score:.6f}) ===")
    phase2_deadline = time.monotonic() + min(900, (deadline - time.monotonic()) * 0.5)

    # Sort candidates and pick top ones
    candidates_for_phase2.sort(key=lambda x: x[0])
    top_cands = candidates_for_phase2[:5]

    # Also add the global best explicitly
    if global_best_seq is not None:
        top_cands.insert(0, (global_best_score, global_best_seq.copy(), len(global_best_seq)))

    seen_n = set()
    for sc, s0, n in top_cands:
        if time.monotonic() > phase2_deadline - 60:
            break
        if n in seen_n:
            continue
        seen_n.add(n)
        remaining = phase2_deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"\n  Optimizing n={n}, start_score={sc:.6f}, time_left={remaining:.0f}s")

        use_lp = (n <= 384)
        cand_deadline = time.monotonic() + min(remaining - 30, 300)
        best_s, best_sc = combined_optimize(s0, cand_deadline, rng,
                                             use_lp=use_lp, lp_interval=5,
                                             verbose=True)
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s.copy()
            print(f"  ** Phase 2 new best: {best_sc:.6f} (n={n})")

    # Phase 3: Gradient polish on the best
    print(f"\n=== Phase 3: Gradient polish (best: {global_best_score:.6f}) ===")
    if global_best_seq is not None and time.monotonic() < deadline - 30:
        n = len(global_best_seq)
        polish_deadline = deadline - 20
        best_s, best_sc = refined_gradient_polish(global_best_seq, polish_deadline, rng)
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s.copy()
            print(f"  ** Polish improved: {best_sc:.6f}")

        # Try LP on the polished result if time allows
        if time.monotonic() < deadline - 60 and n <= 384:
            lp_deadline = deadline - 20
            best_s, best_sc = combined_optimize(global_best_seq, lp_deadline, rng,
                                                 use_lp=True, lp_interval=3,
                                                 verbose=True)
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Final LP improved: {best_sc:.6f}")

    # Final verification
    if global_best_seq is None:
        global_best_seq = np.ones(256) / 256.0

    final_score = float(evaluate_sequence(global_best_seq.tolist()))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(global_best_seq)}")

    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result
