```python
def exp_poly_sequence(n, coeffs):
    """a[i] = exp(sum_k coeffs[k] * (i/(n-1))^k), normalized."""
    t = np.linspace(0, 1, n)
    log_a = np.zeros(n)
    for k, c in enumerate(coeffs):
        log_a += c * t**k
    # Normalize: shift so mean is 0, then scale
    log_a -= log_a.max()
    a = np.exp(log_a)
    a /= a.sum()
    return a
```

This is always positive and well-defined. The optimization of coefficients is unconstrained.

**Mirror Descent for our problem:**

The function to minimize is f(a) = max_k (a*a)[k] (convex).
The constraint is a in simplex: a >= 0, sum(a) = 1.

Mirror descent with the negative entropy mirror (which gives multiplicative updates):

where Z_t = sum_j a_t[j] * exp(-eta_t * sg_t[j]).

The step size eta_t can be chosen as 1/sqrt(t) (for subgradient methods) or using a line search.

For our problem, the subgradient is:

where w is a probability distribution on the lags where the maximum is achieved.

This is just 2 times a "shifted and reflected" version of a!

a_new[i] = a[i] / Z (otherwise)

This is a simple and interesting update! It reduces the values of a[i] where a[k*-i] is large (i.e., where the "partner" in the peak autocorrelation is large).

Let me implement this:

```python
def mirror_descent_step(a, k_star, eta):
    n = len(a)
    log_mult = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            log_mult[i] = -eta * 2 * a[j]
    a_new = a * np.exp(log_mult)
    a_new /= a_new.sum()
    return a_new
```

This is O(n) per step (the loop can be vectorized). Very fast!

```python
Combined optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) polynomial parametrization with L-BFGS, (2) mirror descent with
adaptive step size, (3) LP correction steps, (4) multi-resolution search.

import numpy as np
import time
from scipy.optimize import linprog, minimize

def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def ramp_sequence(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()

def exp_poly_sequence(n, coeffs):
    t = np.linspace(0.0, 1.0, n)
    log_a = np.zeros(n)
    for k in range(len(coeffs)):
        log_a += coeffs[k] * (t ** k)
    log_a -= log_a.max()
    a = np.exp(log_a)
    return a / a.sum()

def subgradient(a, w):
    """Compute subgradient of max(conv(a,a)) given weights w on lags."""
    n = len(a)
    sg = np.zeros(n)
    # sg[i] = 2 * sum_j w[i+j] * a[j] for valid i+j
    L = len(w)
    for i in range(n):
        # j ranges from 0 to n-1, and i+j ranges from i to i+n-1
        w_slice = w[i:i+n]
        if len(w_slice) == n:
            sg[i] = 2.0 * np.dot(w_slice, a)
        else:
            # Pad with zeros
            w_full = np.zeros(n)
            w_full[:len(w_slice)] = w_slice
            sg[i] = 2.0 * np.dot(w_full, a)
    return sg

def mirror_descent(a, eta, beta_smooth=0.0):
    """One mirror descent step with optional smoothing."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    # Compute softmax weights on the max
    if beta_smooth > 1e-12:
        x = (c - max_c) / beta_smooth
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s
    else:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    sg = subgradient(a, w)
    log_mult = -eta * sg
    log_mult -= log_mult.max()
    a_new = a * np.exp(log_mult)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy(), w
    a_new /= s
    return a_new, w

def lp_direction(f, M, timeout=10.0):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_refine(f, max_iters=100, deadline=None, verbose=False):
    n = len(f)
    f = normalize(f.copy())
    if f is None:
        return f, np.inf
    M = float(np.convolve(f, f).max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    t_step = 0.4
    no_improve = 0
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        g_raw = lp_direction(f, M, timeout=5.0)
        if g_raw is None:
            break
        g = normalize(g_raw)
        if g is None:
            break
        # Try blend
        for t in [t_step, 0.2, 0.1, 0.05]:
            f_new = (1.0 - t) * f + t * g
            f_new = normalize(f_new)
            if f_new is None:
                continue
            sc = 2.0 * n * float(np.convolve(f_new, f_new).max())
            if sc < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = sc
                f = f_new.copy()
                M = float(np.convolve(f, f).max())
                no_improve = 0
                break
        else:
            no_improve += 1
            t_step = max(0.02, t_step * 0.8)
            if no_improve > 50:
                break
        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}")
    return best_f, best_score

def polynomial_optimize(n, degree, initial_coeffs=None, deadline=None, verbose=True):
    """Optimize exp-polynomial coefficients to minimize score."""
    def objective(coeffs):
        a = exp_poly_sequence(n, coeffs)
        return internal_score(a)
    if initial_coeffs is None:
        # Initialize with ramp: log(ramp) ≈ log(i+1) ≈ t for large n (roughly linear)
        initial_coeffs = np.zeros(degree)
        initial_coeffs[0] = 0.0
        if degree > 1:
            initial_coeffs[1] = 2.0  # Approximation of log(ramp)
    best_x = None
    best_val = np.inf
    # Try multiple starting points
    rng = np.random.default_rng(42)
    starts = [initial_coeffs.copy()]
    for _ in range(5):
        starts.append(rng.normal(0, 1.0, degree) * 0.5)
    for x0 in starts:
        if deadline is not None and time.monotonic() > deadline:
            break
        try:
            res = minimize(objective, x0, method='L-BFGS-B',
                           options={'maxiter': 200, 'maxfun': 500,
                                    'ftol': 1e-12, 'gtol': 1e-10})
            if res.fun < best_val:
                best_val = res.fun
                best_x = res.x.copy()
        except Exception:
            pass
    if best_x is not None:
        if verbose:
            print(f"  Poly opt (deg={degree}): score={best_val:.8f}")
        return exp_poly_sequence(n, best_x), best_val
    return None, np.inf

def mirror_descent_optimize(a0, deadline, eta_init=0.5, beta_init=0.5, verbose=True):
    """Mirror descent optimization with adaptive step size."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * float(np.convolve(a, a).max())
    eta = eta_init
    beta = beta_init
    no_improve = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        a_new, w = mirror_descent(a, eta, beta)
        sc = 2.0 * n * float(np.convolve(a_new, a_new).max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            eta = max(0.01, eta * 0.95)
            if no_improve % 100 == 0:
                beta = max(0.05, beta * 0.8)
            if no_improve > 500:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a = normalize(a + 0.1 * rng.random(n))
                eta = eta_init
                beta = beta_init
                no_improve = 0
        if it % 500 == 0:
            beta = max(0.05, beta * 0.9)
            if verbose:
                print(f"    MD iter {it}: score={best_score:.8f}, eta={eta:.4f}, beta={beta:.4f}")
    return best_a, best_score

def mixed_optimize(a0, deadline, verbose=True):
    """Combined mirror descent + LP refinement."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * float(np.convolve(a, a).max())
    eta = 0.5
    beta = 1.0
    no_improve = 0
    it = 0
    lp_interval = 200
    last_lp = 0
    while time.monotonic() < deadline:
        it += 1
        a_new, w = mirror_descent(a, eta, beta)
        sc = 2.0 * n * float(np.convolve(a_new, a_new).max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            eta = max(0.01, eta * 0.97)
        # Occasional LP correction
        if it - last_lp >= lp_interval and time.monotonic() < deadline - 30:
            M = float(np.convolve(best_a, best_a).max())
            g_raw = lp_direction(best_a, M, timeout=10.0)
            if g_raw is not None:
                g = normalize(g_raw)
                if g is not None:
                    for t in [0.3, 0.2, 0.1, 0.05]:
                        f_new = (1.0 - t) * best_a + t * g
                        f_new = normalize(f_new)
                        if f_new is not None:
                            sc_new = 2.0 * n * float(np.convolve(f_new, f_new).max())
                            if sc_new < best_score - 1e-12:
                                best_score = sc_new
                                best_a = f_new.copy()
                                a = f_new.copy()
                                no_improve = 0
                                break
            last_lp = it
        if it % 500 == 0:
            beta = max(0.05, beta * 0.85)
            if verbose:
                print(f"    Mixed iter {it}: score={best_score:.8f}")
        if no_improve > 1000:
            rng = np.random.default_rng(it)
            a = normalize(best_a + 0.05 * rng.random(n))
            eta = 0.3
            beta = 0.5
            no_improve = 0
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    total_deadline = t_start + budget_s - 30.0
    best_score = np.inf
    best_seq = None
    best_n = 512

    print(f"Starting optimization with budget {budget_s}s")

    # Phase 0: Gather initial candidates
    candidates = []

    # Ramp sequences
    for n in [512, 1024, 2048]:
        r = ramp_sequence(n)
        sc = internal_score(r)
        print(f"  Ramp n={n}: score={sc:.8f}")
        candidates.append((sc, r, n))
        if sc < best_score:
            best_score = sc
            best_seq = r.copy()
            best_n = n

    # Try height_sequence_1
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                sc = internal_score(parent)
                print(f"  Parent: n={len(parent)}, score={sc:.8f}")
                candidates.append((sc, parent, len(parent)))
                if sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                    best_n = len(parent)
    except Exception:
        pass

    # Try quadratic ramp
    for n in [512, 1024]:
        t = np.linspace(0, 1, n)
        a = 1.0 + 3.0 * t  # Steeper ramp
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
        sc = internal_score(a)
        candidates.append((sc, a, n))
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            best_n = n
        # Try power ramp
        a2 = (np.arange(1, n+1, dtype=np.float64)) ** 1.5
        a2 = a2 / a2.sum()
        sc2 = internal_score(a2)
        candidates.append((sc2, a2, n))
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            best_n = n

    print(f"\nInitial best: n={best_n}, score={best_score:.8f}")

    # Phase 1: Polynomial optimization at best_n
    n_opt = best_n
    deadline_p1 = min(total_deadline, t_start + 400.0)
    print(f"\nPhase 1: Polynomial optimization at n={n_opt}")
    for degree in [2, 3, 4, 5, 6]:
        if time.monotonic() > deadline_p1:
            break
        a_poly, sc_poly = polynomial_optimize(n_opt, degree,
                                               deadline=deadline_p1, verbose=True)
        if a_poly is not None and sc_poly < best_score:
            best_score = sc_poly
            best_seq = a_poly.copy()
            best_n = n_opt
            print(f"  New best from poly deg={degree}: {best_score:.8f}")

    # Phase 2: Mirror descent + LP at best_n
    deadline_p2 = min(total_deadline, t_start + 1000.0)
    print(f"\nPhase 2: Mixed MD+LP at n={best_n}, starting score={internal_score(best_seq):.8f}")
    a_mixed, sc_mixed = mixed_optimize(best_seq, deadline_p2, verbose=True)
    if sc_mixed < best_score:
        best_score = sc_mixed
        best_seq = a_mixed.copy()
        print(f"  Mixed result: {best_score:.8f}")

    # Phase 3: Try larger n by interpolation
    deadline_p3 = min(total_deadline, t_start + 1400.0)
    if time.monotonic() < deadline_p3:
        for n_try in [1024, 2048]:
            if n_try <= best_n:
                continue
            # Interpolate best_seq to n_try
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = normalize(a_interp)
            if a_interp is None:
                continue
            sc_interp = internal_score(a_interp)
            print(f"  Interpolated to n={n_try}: score={sc_interp:.8f}")
            if sc_interp < best_score:
                best_score = sc_interp
                best_seq = a_interp.copy()
                best_n = n_try
            # Quick mirror descent on interpolated
            if time.monotonic() < deadline_p3 - 30:
                a_md, sc_md = mirror_descent_optimize(a_interp, deadline_p3 - 10,
                                                       eta_init=0.3, beta_init=0.5)
                if sc_md < best_score:
                    best_score = sc_md
                    best_seq = a_md.copy()
                    best_n = n_try
                    print(f"  MD on n={n_try}: {best_score:.8f}")
            break  # Only try the immediate next size

    # Phase 4: Final LP refinement
    if time.monotonic() < total_deadline - 20:
        print(f"\nPhase 4: Final LP refinement at n={best_n}")
        a_lp, sc_lp = lp_refine(best_seq, max_iters=50,
                                 deadline=total_deadline - 10, verbose=True)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  LP result: {best_score:.8f}")

    # Final validation
    final_seq = np.asarray(best_seq, dtype=np.float64)
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = ramp_sequence(512)
        s = final_seq.sum()
    final_seq = final_seq / s

    final_score = internal_score(final_seq)
    print(f"\nFinal: n={len(final_seq)}, score={final_score:.8f}")

    # Verify with evaluate_sequence if available
    try:
        ev_score = evaluate_sequence([float(x) for x in final_seq])
        print(f"evaluate_sequence score: {ev_score:.8f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in final_seq]
```
