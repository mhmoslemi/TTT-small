The problem is: min max_k (a*a)[k] s.t. sum(a) = 1, a >= 0.

Subgradient method:
1. At each step, compute g = subgradient of max_k (a*a)[k]
2. Take a step: a_new = a - lr * g
3. Project onto the simplex {a >= 0, sum(a) = 1}
4. Average all iterates to get the final answer (Polyak averaging)

The subgradient at `a` is `2 * sum_k w_k * conv_shift(a, k)` where `w_k` is a distribution over the argmax set.

If I use `w_k = 1/(2n-1)` for all k (uniform), the subgradient is `2 * mean of all shifts of a` which is just a constant vector (times 2). That's not useful.

If I use `w_k = 1` only at the argmax k, the subgradient points in the direction of reducing that particular convolution entry.

A better choice: `w_k = exp((a*a)[k] / beta) / sum_j exp((a*a)[j] / beta)` (softmax over the convolution entries).

    k = 1

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

        return np.array([])

    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    if mn < 0.0:

    return c

def _smooth_max_grad(a, beta):
    """Compute gradient of smooth approximation to max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_r = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_r, N)
    grad = 2.0 * h[n-1:2*n-1]

def _project_simplex(v):

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:

    return w / s

def _lp_best_response(f, tl=30):
    """Find g >= 0, sum(g)=1, minimizing max_k (f*g)[k]."""
    n = len(f)
    # Variables: g[0..n-1], t
    # Minimize t
    # Subject to: (f*g)[k] - t <= 0 for all k
    #             sum(g) = 1
    #             g >= 0
    #             t free (or t >= 0)
    
    A = _conv_matrix(f)  # (2n-1) x n
    # Constraint: A @ g - t <= 0
    # Extend A to include -t column: [A | -1]
    A_ext = np.hstack([A, -np.ones((2*n-1, 1))])
    b = np.zeros(2*n-1)
    
    # Objective: [0, 0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Bounds: g[i] >= 0 for i=0..n-1, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    res = linprog(c, A_ub=A_ext, b_ub=b, A_eq=A_eq, b_eq=[1.0],
                  bounds=bounds, method='highs',
                  options={'time_limit': tl})
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M, tl=20):
    n = len(a)
    A = _conv_matrix(a)
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': tl, 'presolve': 'on'})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_conv(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_r = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_r, N)
    grad = 2.0 * h[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a, deadline, max_iter=50000):
    a = _project_simplex(np.asarray(a, dtype=np.float64).copy())
    n = len(a)
    best_a = a.copy()

    beta = max(1e-8, 0.05 * float(_fft_conv(a).max()))
    lr = 0.015
    mom = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a_n = a + mom * v
        grad, _ = _smooth_max_grad(a_n, beta)
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 3)
            v = np.zeros(n)
            continue
        v = mom * v - lr * grad
        a = _project_simplex(a + v)
        sc = 2.0 * n * float(_fft_conv(a).max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

            if no_imp % 200 == 0:

                v = np.zeros(n)
                beta = max(1e-8, 0.05 * float(_fft_conv(a).max()))
                lr = base_lr
                no_imp = 0
        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)
    return best_a


def _adam_polish(a, deadline, max_iter=50000, lr=0.04):

    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)
    z = np.log(a)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_z = z.copy()
    best_score = 2.0 * n * float(_fft_conv(a).max())
    beta = max(1e-8, 0.05 * float(_fft_conv(a).max()))
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(p)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_r = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp_r, N)
        vp = 2.0 * h[n - 1:2 * n - 1] + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            continue
        t += 1
        m = b1 * m + (1 - b1) * gz
        v_arr = b2 * v_arr + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t)
        vh = v_arr / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            continue
        z = np.clip(z, -50, 50)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        sc = 2.0 * n * float(_fft_conv(a).max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.003, lr * 0.95)
        if it % 150 == 0:
            beta = max(1e-10, beta * 0.7)
        if no_imp > max(3000, 4 * n):
            rng2 = np.random.default_rng(12345 + it)
            z = 0.7 * best_z + 0.3 * rng2.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-8, 0.05 * float(_fft_conv(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            lr = base_lr
            no_imp = 0

def _local_search(a, deadline, rng, max_iter=200000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_conv(a).max())
    T = 0.001
    T_min = 1e-6
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        T = max(T_min, T * 0.9999)
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        frac = float(rng.uniform(0.005, 0.05))
        delta = a[i] * frac
        if delta < 1e-15:
            continue
        a_try = a.copy()
        a_try[i] -= delta
        a_try[j] += delta
        if a_try[i] < 0:
            continue
        sc = 2.0 * n * float(_fft_conv(a_try).max())
        dE = sc - best_score
        if dE < 0 or (dE < 0.01 and rng.random() < np.exp(-dE / (T + 1e-15))):
            best_score = sc
            best_a = a_try.copy()
            a = a_try.copy()
    return best_a


def _interpolate(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        return p / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _make_quad(n, c=1.0):
    t = np.linspace(-1, 1, n)
    a = 1.0 + c * t * t
    return a / a.sum()


def _make_linear(n, eta=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma=0.3):
    t = np.linspace(-1, 1, n)
    a = np.exp(-t**2 / (2 * sigma**2))
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            parent = p
            sc = _score(parent)
            if np.isfinite(sc):
                best_score = sc
                best_seq = parent.copy()
                print(f"Parent: n={p.size}, score={sc:.10f}")
    except Exception:
        pass

    lengths = [512, 1024]
    if parent is not None and parent.size not in lengths:
        lengths.insert(0, parent.size)

    results = {}

    for n in lengths:
        if time.monotonic() > deadline - 180:
            break

        # Multiple initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate(parent, n))
            for _ in range(2):
                pert = _interpolate(parent, n) * (1 + 0.15 * rng.normal(size=n))
                pert = np.maximum(pert, 0)
                if pert.sum() > 1e-10:
                    inits.append(pert / pert.sum())
        inits.append(_make_quad(n, 1.2))
        inits.append(_make_linear(n, 2.0))
        inits.append(_make_gaussian(n, 0.35))

        for init_idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 120:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # LP refinement
            if time.monotonic() < deadline - 100:
                lp_dl = min(time.monotonic() + 150, deadline - 100)
                a_lp = _lp_refine(a0, lp_dl, inflate=1.002, blend=0.3)
                sc_lp = _score(a_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                results[f'lp_n{n}_i{init_idx}'] = sc_lp
                print(f"  n={n} init{init_idx}: LP score={sc_lp:.10f}")

                # Fictitious play on LP result
                if time.monotonic() < deadline - 80:
                    fp_dl = min(time.monotonic() + 80, deadline - 80)
                    a_fp = _fictitious_play(a_lp, fp_dl)
                    sc_fp = _score(a_fp)
                    if sc_fp < best_score:
                        best_score = sc_fp
                        best_seq = a_fp.copy()
                    results[f'fp_n{n}_i{init_idx}'] = sc_fp
                    print(f"    FP score={sc_fp:.10f}")

                # Gradient polish
                if time.monotonic() < deadline - 60:
                    gp_dl = min(time.monotonic() + 50, deadline - 60)
                    a_gp = _gradient_polish(a_lp, gp_dl)
                    sc_gp = _score(a_gp)
                    if sc_gp < best_score:
                        best_score = sc_gp
                        best_seq = a_gp.copy()
                    results[f'gp_n{n}_i{init_idx}'] = sc_gp

    # Final combined polish on best
    if best_seq is not None and time.monotonic() < deadline - 40:
        # Adam polish
        ad_dl = min(time.monotonic() + 100, deadline - 40)
        a_ad = _adam_polish(best_seq, ad_dl)
        sc_ad = _score(a_ad)
        if sc_ad < best_score:
            best_score = sc_ad
            best_seq = a_ad.copy()
        print(f"Adam polish: {sc_ad:.10f}")

    # Local search
    if best_seq is not None and time.monotonic() < deadline - 20:
        ls_dl = deadline - 10
        a_ls = _local_search(best_seq, ls_dl, rng)
        sc_ls = _score(a_ls)
        if sc_ls < best_score:
            best_score = sc_ls
            best_seq = a_ls.copy()
        print(f"Local search: {sc_ls:.10f}")

    # Final gradient
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = _gradient_polish(best_seq, deadline - 3)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
        print(f"Final gradient: {sc_fin:.10f}")

    if best_seq is None:
        best_seq = _make_quad(256, 1.0)

    final = np.clip(best_seq, 0, 1000)
    s = float(final.sum())
    if s > 0.01:
        final = final / s
    else:
        final = _make_quad(256, 1.0)

    print(f"FINAL: n={final.size}, score={_score(final):.10f}")
    return [float(x) for x in final]
