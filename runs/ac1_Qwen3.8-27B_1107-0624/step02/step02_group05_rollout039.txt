"""
Multi-phase optimization to minimize 2*n*max(convolve(f,f))/sum(f)^2.

Strategy:
1. Generate diverse starting points including the binomial (flat-convolution) sequence
   and the previous best.
2. Use LP-based iterative optimization (maximize sum of increment g subject to
   convolution bound) to push toward lower max-convolution.
3. Use a max-min LP variant for flatness optimization.
4. Adam polish with adaptive log-sum-exp for smooth gradient descent.
5. Multi-scale search across n=128, 256, 512.
6. Active-set reduction to speed up LPs.

def binomial_sequence(n):
    """Generate the (truncated) binomial sequence c_i = C(2i,i)/4^i, normalized to sum=1."""
    c = np.zeros(n)
    val = 1.0
    for i in range(n):
        if i == 0:
            val = 1.0
        else:
            val *= (2 * i) * (2 * i - 1) / (4 * i * i)
        c[i] = val
    s = c.sum()
    return c / s


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_conv(f):
    """Compute linear convolution of f with itself."""
    return np.convolve(f, f)


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def make_conv_matrix_reduced(f, active_lags):
    """Build reduced conv matrix for only active lags."""
    n = len(f)
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[idx, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp_sum(f, M, active_lags=None):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for active lags (or all)."""
    n = len(f)
    c = -np.ones(n)
    if active_lags is not None and len(active_lags) < 2 * n - 1:
        A_ub = make_conv_matrix_reduced(f, active_lags)
        b_ub = np.full(len(active_lags), M)
    else:
        A_ub = make_conv_matrix(f)
        b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 8.0, 'dual_feasibility_tolerance': 1e-8})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
        elif res.status == 3:  # time limit
            if res.x is not None:
                return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_flatness(f, M):
    """Solve LP: max t s.t. g >= 0, (f*g)[k] <= M, (f*g)[k] >= t for all k."""
    n = len(f)
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = -1.0  # maximize t
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    # Add constraints: (f*g)[k] - t >= 0, i.e., -(f*g)[k] + t <= 0
    n_constraints = 2 * n - 1
    A_flat = np.zeros((n_constraints, n + 1))
    A_flat[:, :n] = -make_conv_matrix(f)
    A_flat[:, n] = 1.0
    b_flat = np.zeros(n_constraints)
    # Combine constraints
    A_ub_all = np.vstack([A_ub, A_flat])
    b_ub_all = np.concatenate([b_ub, b_flat])
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub_all, b_ub=b_ub_all, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 8.0})
        if res.success and res.x is not None:
            return np.maximum(res.x[:n], 0.0)
    except Exception:
        pass
    return None


def get_active_lags(conv, f, threshold=0.95):
    """Get lags where conv is close to max."""
    M = float(conv.max())
    if M < 1e-30:
        return list(range(len(conv)))
    active = [k for k in range(len(conv)) if conv[k] >= threshold * M]
    if len(active) < 5:
        # Always include top 5
        top5 = np.argsort(conv)[-5:]
        active = list(set(active) | set(top5.tolist()))
    return sorted(active)

def soft_max(c, beta):
    """Compute soft-max and its weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def corr_weights_fft(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 using FFT."""
    n = len(p)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def adam_polish(f_init, deadline, rng, lr=0.05, max_iters=10000):
    """Adam optimization in logit space with adaptive log-sum-exp."""
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = np.convolve(f, f)
    M = float(c.max())
    beta = max(1e-12, 0.1 * M)

    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    no_imp = 0
    base_lr = lr

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        L, w = soft_max(c, beta)
        gp = 2.0 * corr_weights_fft(w, f)
        vp = gp + 1e-8 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z_new, -60, 60)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.9)

        # Adaptive beta: decrease over time to sharpen the max approximation
        if adam_t % 200 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > max(500, 4 * n):
            # Restart with perturbation
            z = 0.7 * best_z + 0.4 * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-8, 0.15 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_f, best_M


def make_linear_profile(n, slope=1.0):
    """Linear ramp profile normalized to sum 1."""
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    """Cubic profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_power_profile(n, p=2.0):
    """Power law profile f[i] = (i+1)^p normalized."""
    f = (np.arange(n) + 1) ** p
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_tapered_binomial(n, taper=0.5):
    """Binomial sequence with sinusoidal taper."""
    c = binomial_sequence(n)
    t = np.linspace(0, 1, n)
    taper_fn = 0.5 + 0.5 * np.sin(np.pi * t)
    f = c * taper_fn
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def get_starting_points(n, rng, count=10):
    """Generate diverse starting points."""
    starts = []
    # Binomial and variants
    starts.append(binomial_sequence(n))
    starts.append(make_tapered_binomial(n, 0.3))
    starts.append(make_tapered_binomial(n, 0.7))
    # Power profiles
    for p in [1.5, 2.0, 2.5, 3.0]:
        starts.append(make_power_profile(n, p))
    # Linear and cubic
    starts.append(make_linear_profile(n, 1.0))
    starts.append(make_linear_profile(n, 2.0))
    starts.append(make_cubic_profile(n, 2.0, -1.5))
    # Perturbed binomial
    for _ in range(count):
        b = binomial_sequence(n)
        pert = 0.1 * rng.normal(size=n)
        f = normalize_nonneg(b + pert)
        if f is not None:
            starts.append(f)
    return starts


def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, f)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 0:
        result = result / s
    return result

            print(f"  Final Adam: score={sc:.8f}, best={best_score:.8f}")

    # Also try n=128 if time permits (shorter sequences might have better ratios)
    if time.monotonic() < phase4_end - 80:
        n4 = 128
        print(f"  Trying n={n4}...")
        f_128 = interpolate_profile(best_seq, n4) if best_seq is not None else binomial_sequence(128)
        run_deadline = min(phase4_end - 30, time.monotonic() + 60.0)
        f_best, M_best = lp_optimize(f_128, run_deadline, rng, max_iters=400)
        sc = 2.0 * n4 * M_best
        try_update(f_best, sc)
        print(f"  n=128 result: score={sc:.8f}, best={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = binomial_sequence(256)
        best_score = compute_score(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = binomial_sequence(256)
        best_score = compute_score(best_seq)
    else:
        best_seq = best_seq / s

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = compute_score(best_seq)
    print(f"Final: n={len(best_seq)}, score={final_score:.10f}, best_score={best_score:.10f}")

    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
        if np.isfinite(ev_score):
            final_score = min(final_score, ev_score)
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
