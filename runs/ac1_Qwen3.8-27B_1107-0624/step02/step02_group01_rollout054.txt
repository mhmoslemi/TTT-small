import numpy as np
import time

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def softmax_gradient_optimize(f0, deadline, rng, lr=0.05, max_iter=100000):
    """Optimize using softmax gradient in logit space."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size
    
    z = np.log(f)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-10, 0.1 * M_val)
    
    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    
    N_corr = next_pow2(3 * n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Soft-max weights
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))
        
        # Gradient in logit space
        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]
        
        grad_z = 2.0 * f * (h - L)
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        
        # Adaptive learning rate
        eff_lr = lr * (0.5 ** (no_imp / max(100, n)))
        z = z - eff_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()
        
        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        c = fft_convolve(f)
        M_new = float(c.max())
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive temperature
        if t_adam % 200 == 0:
            beta = max(1e-12, beta * 0.85)
        
        # Restart mechanism
        if no_imp > max(500, 5 * n):
            # Try different restart strategies
            strategy = rng.integers(3)
            if strategy == 0:
                z = 0.8 * best_z + 0.5 * rng.normal(size=n)
            elif strategy == 1:
                f = best_f + 0.05 * rng.normal(size=n)
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                z = np.log(f)
                z -= z.mean()
            else:
                # Mirror/randomize
                f = best_f.copy()
                mask = rng.random(n) < 0.3
                f[mask] = best_f[mask] * rng.uniform(0.5, 1.5, mask.sum())
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                z = np.log(f)
                z -= z.mean()
            
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
    
    return best_f, best_M
