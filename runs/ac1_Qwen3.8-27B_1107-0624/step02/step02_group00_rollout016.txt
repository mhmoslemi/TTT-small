def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    """Solve SDP relaxation for the minimization problem."""
    import cvxpy as cp
    
    X = cp.Variable((n, n), PSD=True)
    M = cp.Variable()
    
    constraints = []
    for k in range(2 * n - 1):
        # (a*a)[k] = sum_i a[i] * a[k-i]
        # = sum_{i,j} a[i]*a[j] * delta_{i+j, k}
        # = trace(D_k * X) where D_k[i,j] = 1 if i+j = k
        D_k = np.zeros((n, n))
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                D_k[i, j] = 1
        constraints.append(cp.trace(D_k @ X) <= M)
    
    # 1^T X 1 = 1 (i.e., (sum a)^2 = 1)
    ones = np.ones(n)
    constraints.append(ones @ X @ ones == 1)
    
    # X is PSD (enforced by PSD=True)
    # Also need X[i,i] >= 0 (implied by PSD)
    
    obj = cp.Minimize(M)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, warm_start=True)
    
    return M.value, X.value

"""
Improved hybrid optimization: LP spreading + projected gradient (Nesterov) + subgradient tracking.
Key improvements: better LP scheduling with adaptive parameters, Nesterov-accelerated projected
gradient with adaptive smooth-max beta, direct subgradient steps for final squeeze,
multiple diverse initializations, and multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=100000, inflate_start=1.001, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _lp_spread(a, M_lp)
        if g is None:
            inflate = min(1.05, inflate * 1.5)
            if inflate > 1.05:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(1.02, inflate * 1.1)
            if no_imp > 400:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate = min(1.01, inflate * 1.05)
            if no_imp > 300:
                a = best_a.copy()
                blend_cur = blend
                inflate = inflate_start
                no_imp = 0

        a = a_new

    return best_a


def _smooth_max_obj_and_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return val, grad


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _projected_gradient_polish(a, deadline, max_iter=50000, lr_init=0.02):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    beta = 0.05 * float(c.max())
    lr = lr_init
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        val, grad = _smooth_max_obj_and_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-10, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.95)

        if it % 150 == 0:
            beta = max(0.005 * float(c.max()), beta * 0.7)

        if no_imp > max(3000, 4 * n):
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.05 * float(_fft_convolve(a).max())
            lr = lr_init
            no_imp = 0

    return best_a


def _subgradient_step(a, step_size):
    n = len(a)
    c = _fft_convolve(a)
    k_star = int(np.argmax(c))

    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    grad = np.zeros(n)
    for i in range(lo, hi + 1):
        grad[i] = a[k_star - i]
    grad *= 2.0
    grad -= float(np.mean(grad))

    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def _subgradient_polish(a, deadline, max_iter=100000, step_init=0.008):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    step = step_init
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _subgradient_step(a, step)
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 500 == 499:
            step *= 0.97
        if no_imp > 15000:
            break

    return best_a


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(np.pi * t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _local_perturb_search(a, deadline, rng, max_iter=5000):
    """Try random perturbations and keep improvements."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        idx = rng.integers(0, n, size=min(8, n // 2))
        a_pert = a.copy()
        a_pert[idx] *= (1.0 + 0.1 * rng.normal(size=len(idx)))
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        c_pert = _fft_convolve(a_pert)
        score_pert = float(2.0 * n * c_pert.max())

        if score_pert < best_score - 1e-13:
            best_score = score_pert
            best_a = a_pert.copy()
            a = a_pert
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 2000:
            break

    return best_a
