"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-9:
        c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f, active_set):
    """Build sparse representation: rows for active_set lags only."""
    n = len(f)
    m = len(active_set)
    A = np.zeros((m, n), dtype=np.float64)
    for idx, k in enumerate(active_set):
        # conv(f, g)[k] = sum_j f[j] * g[k-j] where 0<=j<n, 0<=k-j<n
        # = sum_j g[j] * f[k-j]
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[idx, j] = f[k - j]
    return A


def lp_step_active(f, M, active_set):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for k in active_set, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, active_set)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_set), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize_active(f_init, deadline, rng, top_k_start=50, top_k_end=200):
    """LP-based optimization with growing active set."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()

    n = len(f)

    best_f = f.copy()

    stagnant = 0
    iteration = 0

        M = float(conv_ff.max())

def soft_mx(c, beta):
    """Smooth max with log-sum-exp, returns (value, weights)."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j] via FFT."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def logit_anneal(p0, deadline, rng, lr=0.06, reg=1e-8, beta_start=0.3, beta_end=0.005):
    """Logit-space Adam with annealed soft-max temperature."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = beta_start * maxc

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    total_steps = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0); v.fill(0.0); adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0); v.fill(0.0); adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        total_steps += 1

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 500:
                lr = max(0.01, lr * 0.97)

        # Anneal beta
        if total_steps % 60 == 0:
            beta = max(beta_end * maxc, beta * 0.82)

        # Restart
        if no_imp > max(600, 3 * n):
            noise_scale = 0.3 + 0.5 * rng.random()
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = 0.25 * maxc
            m.fill(0.0); v.fill(0.0); adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score

    old_val = a[i]
    if old_val == 0.0:
        # Try small positive values
        s_arr = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            v1 = a[k - i] if 0 <= k - i < n else 0.0
            v2 = a[i - k] if 0 <= i - k < n else 0.0
            s_arr[k] = v1 + v2
        # h(0) = max(c[k]) where c[k] = conv[ k] when a[i]=0
        # Just use current conv (which already has a[i]=0)
        c_cur = np.convolve(a, a)
        max_c = float(c_cur.max())
        # Try a few small x values
        best_x = 0.0
        best_h = max_c
        for trial_x in [0.01, 0.05, 0.1 * np.mean(a), 0.5 * np.mean(a)]:
            a[i] = trial_x
            h = float(np.convolve(a, a).max())
            if h < best_h:
                best_h = h
                best_x = trial_x
            a[i] = 0.0
        a[i] = best_x
        return best_x

    # Compute s[k] = a[k-i] + a[i-k]
    s_arr = np.full(2 * n - 1, 0.0)
    # a[k-i] is valid when 0 <= k-i < n, i.e., i <= k < i+n
    lo1 = i
    hi1 = min(2 * n - 1, i + n)
    if lo1 < hi1:
        s_arr[lo1:hi1] += a[:min(n, hi1 - i)]
    # a[i-k] is valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
    lo2 = max(0, i - n + 1)
    hi2 = min(2 * n - 1, i + 1)
    if lo2 < hi2:
        s_arr[lo2:hi2] += a[max(0, i - (hi2 - 1)):n - (lo2) if False else n][:hi2 - lo2]
        # Let me redo this more carefully
        for k in range(lo2, hi2):
            idx = i - k
            if 0 <= idx < n:
                s_arr[k] += a[idx]

    # Recompute s_arr properly
    s_arr = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        v = 0.0
        j1 = k - i  # a[k-i]
        if 0 <= j1 < n:
            v += a[j1]
        j2 = i - k  # a[i-k]
        if 0 <= j2 < n:
            v += a[j2]
        s_arr[k] = v

    # c[k] = conv_current[k] - old_val * s[k]
    conv_cur = np.convolve(a, a)
    c_arr = conv_cur - old_val * s_arr

    # Minimize h(x) = max_k (c[k] + x * s[k]) for x >= 0
    # Binary search for where subgradient contains 0
    # h is convex PL, minimum where slope crosses 0

    def h_at(x):
        return float(np.max(c_arr + x * s_arr))

    # Check x=0
    h0 = h_at(0.0)

    # Find upper bound for x: where h starts increasing
    # If all s[k] where c[k]+x*s[k] is max are positive, h is increasing
    # Try x = old_val, then maybe 2*old_val
    h_old = h_at(old_val)

    # Binary search for minimizer in [0, 2*old_val + 0.1*max(a)]
    x_lo = 0.0
    x_hi = 2.0 * old_val + 0.1 * float(np.max(a))

    # Check if minimum is at the boundary
    # At x_hi, check slope
    trial = c_arr + x_hi * s_arr
    max_trial = float(np.max(trial))
    active = trial >= max_trial - 1e-10
    if np.all(s_arr[active] > 1e-10):
        # Still increasing at x_hi, minimum is below
        pass
    elif np.all(s_arr[active] < -1e-10):
        # Still decreasing, minimum is above - but let's just use x_hi region
        x_lo = x_hi * 0.5
        x_hi = x_hi * 1.5
    else:
        # Minimum is in [0, x_hi]
        pass

    # Do ~30 bisection steps
    for _ in range(30):
        x_mid = 0.5 * (x_lo + x_hi)
        trial = c_arr + x_mid * s_arr
        max_val = float(np.max(trial))
        active = trial >= max_val - 1e-10
        min_s = float(np.min(s_arr[active]))
        max_s = float(np.max(s_arr[active]))
        if min_s > 1e-10:
            # Slope positive everywhere active: h increasing, go left
            x_hi = x_mid
        elif max_s < -1e-10:
            # Slope negative: h decreasing, go right
            x_lo = x_mid
        else:
            # 0 in subgradient: this is the minimum
            x_lo = x_mid
            x_hi = x_mid
            break

    x_opt = 0.5 * (x_lo + x_hi)
    return x_opt


def coordinate_descent(a, n_sweeps=5, deadline=None, rng=None):
    """
    Cyclic coordinate descent. For each coordinate i, replace a[i]
    with the value that minimizes max(conv(a,a)).
    """
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0).copy()
    n = a.size
    best_score = compute_score(a)
    best_a = a.copy()

    perm = np.arange(n)
    if rng is not None:
        perm = rng.permutation(n)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        for i in perm:
            old = a[i]
            new = coordinate_descent_1d(a, i)
            a[i] = new
            # Periodic normalization to prevent drift
            s = a.sum()
            if s > 0:
                a = a / s
                # Re-optimize i with new scale
                # (skip for speed, the scale invariance means it's fine)
            sc = compute_score(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()

        if deadline is not None and time.monotonic() > deadline:
            break
        # Randomize order
        if rng is not None:
            perm = rng.permutation(n)

    return best_a, best_score


def make_lopsided_profile(n, peak_pos=0.6, width=0.3):
    """A lopsided Gaussian profile (asymmetric)."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - peak_pos) / width) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_double_lobe(n, p1=0.3, p2=0.7, w1=0.2, w2=0.15):
    """Two-lobe asymmetric profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - p1) / w1) ** 2) + 0.7 * np.exp(-((t - p2) / w2) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_interpolated_profile(p0, n_new):
    """Interpolate profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    return np.maximum(result, 0.0)


def get_diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    # Lopsided Gaussians
    for pp in [0.4, 0.5, 0.6, 0.7]:
        for w in [0.2, 0.3, 0.4]:
            profiles.append(make_lopsided_profile(n, pp, w))
    # Double lobe
    profiles.append(make_double_lobe(n, 0.3, 0.7, 0.2, 0.15))
    profiles.append(make_double_lobe(n, 0.25, 0.65, 0.15, 0.2))
    # Cosine window variants
    t = np.linspace(0.0, 1.0, n)
    for alpha in [0.3, 0.5, 0.7]:
        a = np.cos(np.pi * (t - alpha)) ** 2
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())
    # Polynomial
    for a1, a2 in [(1.0, -0.5), (-0.5, 1.0), (0.5, 0.5), (2.0, -1.5)]:
        a = 1.0 + a1 * t + a2 * (t - 0.5) ** 2
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())
    # Random smooth
    for _ in range(3):
        coeffs = rng.uniform(-0.4, 0.4, 4)
        a = 1.0
        for k in range(1, 5):
            a = a + coeffs[k-1] * np.cos(2 * np.pi * k * t)
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())
    return profiles


def fast_coordinate_descent(a, deadline, rng):
    """
    Fast coordinate descent using vectorized slope computation.
    Key optimization: precompute the structure for all coordinates at once.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0).copy()
    n = a.size
    best_score = compute_score(a)
    best_a = a.copy()
    perm = np.arange(n)

    sweep = 0
    while time.monotonic() < deadline:
        if rng is not None:
            perm = rng.permutation(n)

        improved = False
        for i in perm:
            old = a[i]
            # Compute s[k] = a[k-i] + a[i-k]
            s_arr = np.zeros(2 * n - 1)
            # Vectorized: a[k-i] for k=i..i+n-1 is a[0..n-1]
            # a[i-k] for k=0..i is a[i..0]
            if i + n <= 2 * n - 1:
                s_arr[i:i+n] += a
            if i >= 0:
                s_arr[max(0, i-n+1):i+1] += a[i:max(0, i-n+1)+n-1-i-1:-1] if False else a[min(n, i+1)::-1]

            # Let me just do it simply (n is at most 2048, so 2n-1 ~ 4095, fine)
            s_arr = np.zeros(2 * n - 1)
            j1 = i  # start
            j1_end = min(2*n-1, i+n)
            s_arr[j1:j1_end] += a[:j1_end-i]
            j2_start = max(0, i-n+1)
            j2_end = i + 1
            s_arr[j2_start:j2_end] += a[i-j2_end+1:i+1+1-n+j2_start-j2_start:]  # this is getting messy

            # Just use the simple loop (it's fast enough for n<=2048)
            s_arr = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                v = 0.0
                j = k - i
                if 0 <= j < n:
                    v += a[j]
                j = i - k
                if 0 <= j < n:
                    v += a[j]
                s_arr[k] = v

            conv_cur = np.convolve(a, a)
            c_arr = conv_cur - old * s_arr

            # Minimize max(c + x*s) for x >= 0 via bisection
            x_lo, x_hi = 0.0, old * 3 + 0.5 * float(np.max(a))

            for _ in range(20):
                x_mid = 0.5 * (x_lo + x_hi)
                trial = c_arr + x_mid * s_arr
                mx = float(np.max(trial))
                active = trial >= mx - 1e-10
                ms = s_arr[active]
                if len(ms) == 0:
                    break
                if np.min(ms) > 1e-8:
                    x_hi = x_mid
                elif np.max(ms) < -1e-8:
                    x_lo = x_mid
                else:
                    break

            x_opt = max(0.0, 0.5 * (x_lo + x_hi))
            if abs(x_opt - old) > 1e-10:
                a[i] = x_opt
                s = a.sum()
                if s > 1e-10:
                    a /= s
                    improved = True

        sweep += 1
        if not improved:
            break

        # Check score periodically
        if sweep % 3 == 0:
            sc = compute_score(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: active-set LP -> coordinate descent -> soft-max Adam -> multi-n.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    print(f"Starting. Budget={budget_s}s")

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    if init_seq is not None:
        sc = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc:.10f}")
        update(init_seq, sc)

    # ============ Phase 1: LP at n=512 ============
    n1 = 512
    phase1_end = min(deadline, t_start + 300.0)

    if init_seq is not None and init_seq.size != n1:
        p1 = make_interpolated_profile(init_seq, n1)
    elif init_seq is not None:
        p1 = init_seq.copy()
    else:
        p1 = make_lopsided_profile(n1, 0.55, 0.25)

    update(p1, compute_score(p1))
    print(f"Phase 1: n={n1}, start score={compute_score(p1):.10f}")

    # Main LP run
    lp_end = min(phase1_end, time.monotonic() + 200.0)
    result = lp_optimize_active(p1, lp_end, rng)
    sc = compute_score(result)
    update(result, sc)
    print(f"  After LP: {sc:.10f}")

    # Try a few diverse profiles
    if time.monotonic() < phase1_end - 30:
        profiles = get_diverse_profiles(n1, rng)
        for p in profiles[:6]:
            if time.monotonic() > phase1_end - 5:
                break
            sc_p = compute_score(p)
            if sc_p < best_score + 0.02:
                per_end = min(phase1_end, time.monotonic() + 25.0)
                r = lp_optimize_active(p, per_end, rng)
                sc_r = compute_score(r)
                update(r, sc_r)

    print(f"After Phase 1: best={best_score:.10f}")

    # ============ Phase 2: Coordinate descent at n=512 ============
    phase2_end = min(deadline, t_start + 700.0)
    if best_seq is not None and best_seq.size == n1:
        print(f"Phase 2: Coordinate descent, n={n1}")
        cd_end = min(phase2_end, time.monotonic() + 250.0)
        cd_result, cd_score = fast_coordinate_descent(best_seq, cd_end, rng)
        update(cd_result, cd_score)
        print(f"  After CD: {cd_score:.10f}")

        # Second CD round with fresh random order
        if time.monotonic() < phase2_end - 30:
            cd2_end = min(phase2_end, time.monotonic() + 100.0)
            cd_result2, cd_score2 = fast_coordinate_descent(cd_result, cd2_end, rng)
            update(cd_result2, cd_score2)
            print(f"  After CD2: {cd_score2:.10f}")

    # ============ Phase 3: Soft-max Adam polish ============
    phase3_end = min(deadline, t_start + 1200.0)
    if best_seq is not None:
        print(f"Phase 3: Adam polish, n={best_seq.size}, current={best_score:.10f}")
        n_polish = best_seq.size

        # Multiple Adam rounds with different parameters
        for lr, beta_s in [(0.05, 0.25), (0.03, 0.15), (0.02, 0.1)]:
            if time.monotonic() > phase3_end - 30:
                break
            p0 = best_seq + 0.002 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            per_end = min(phase3_end, time.monotonic() + 150.0)
            r, sc_r = logit_anneal(p0, per_end, rng, lr=lr, beta_start=beta_s)
            update(r, sc_r)
            print(f"  Adam lr={lr}: {sc_r:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Try n=1024 ============
    n2 = 1024
    phase4_end = min(deadline, t_start + 1600.0)
    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p2 = make_interpolated_profile(best_seq, n2)
        sc2 = compute_score(p2)
        update(p2, sc2)
        print(f"Phase 4: n={n2}, interpolated score={sc2:.10f}")

        # LP at n=1024
        if sc2 < best_score + 0.05:
            lp_end2 = min(phase4_end, time.monotonic() + 200.0)
            r = lp_optimize_active(p2, lp_end2, rng, top_k_start=80, top_k_end=300)
            sc_r = compute_score(r)
            update(r, sc_r)
            print(f"  After LP n=1024: {sc_r:.10f}")

            # CD at n=1024
            if time.monotonic() < phase4_end - 30:
                cd_end2 = min(phase4_end, time.monotonic() + 100.0)
                cd_r, cd_s = fast_coordinate_descent(best_seq, cd_end2, rng)
                update(cd_r, cd_s)
                print(f"  After CD n=1024: {cd_s:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ============ Phase 5: Final polish ============
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_f = best_seq.size
        p0 = best_seq + 0.001 * rng.normal(size=n_f)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        per = min(deadline - 5, time.monotonic() + 60.0)
        r, sc_r = logit_anneal(p0, per, rng, lr=0.02, beta_start=0.05, beta_end=0.002)
        update(r, sc_r)
        print(f"Final polish: {sc_r:.10f}")

        # Quick CD
        if time.monotonic() < deadline - 5:
            per_cd = deadline - 5
            cd_r, cd_s = fast_coordinate_descent(best_seq, per_cd, rng)
            update(cd_r, cd_s)
            print(f"Final CD: {cd_s:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_lopsided_profile(512, 0.55, 0.25)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_lopsided_profile(512, 0.55, 0.25)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence: {eval_score:.10f}")
    except Exception as e:
        print(f"eval error: {e}")

    return [float(x) for x in final_seq]
