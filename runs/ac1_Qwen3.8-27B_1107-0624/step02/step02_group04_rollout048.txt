"""
Optimized subgradient descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-K lag subgradients, logit-space Adam with natural gradient,
multi-scale escalation, and structured restarts. Exploits the fact that
for non-negative a, the convolution subgradient at lag k is 2*a reversed/shifted.
"""
import numpy as np
import time
import scipy.optimize as sco


def _fft_convolve(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _subgrad_at_lag(a, k):
    """Subgradient of conv(a,a)[k] w.r.t. a. Returns 2*a[k-j] for valid j."""
    n = len(a)
    g = np.zeros(n)
    # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        g[j_lo:j_hi + 1] = 2.0 * a[k - j_hi:k - j_lo + 1][:][::-1]
    return g


def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_z(z, floor=1e-15):
    z = z - z.max()
    e = np.exp(np.clip(z, -100, 100))
    p = e + floor
    return p / p.sum()


def _topK_subgradient(a, b, K, temperature=0.0):
    """Compute weighted combination of top-K subgradients."""
    n = len(a)
    if temperature > 1e-10:
        # Soft weighting: w_k proportional to exp(b[k]/temperature)
        m = b.max()
        w = np.exp((b - m) / temperature)
        w = w / w.sum()
        # Only use top K
        idx = np.argsort(w)[::-1][:K]
        w = np.zeros_like(b)
        w[idx] = np.exp((b[idx] - m) / temperature)
        w = w / w.sum()
    else:
        idx = np.argsort(b)[::-1][:K]
        w = np.zeros_like(b)
        w[idx] = 1.0 / K

    g = np.zeros(n)
    for k in np.where(w > 1e-10)[0]:
        g += w[k] * _subgrad_at_lag(a, k)
    return g


def _optimize_logit(a0, deadline, rng, lr=0.05, K=5, temp_frac=0.1,
                    beta_decay=0.95, restart_every=2000, noise_scale=0.15):
    """
    Optimize in logit space using top-K subgradient + Adam.
    a = softmax(z), gradient in z-space: a * (g - <g,a>)
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    z = np.log(a + 1e-30)
    z = z - z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    c = _fft_convolve(a)
    maxc = c.max()
    best_a = a.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    temperature = temp_frac * maxc

    while time.monotonic() < deadline:
        t += 1
        c = _fft_convolve(a)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive temperature
        if t % 100 == 0:
            temperature = max(1e-10, temperature * beta_decay)

        # Compute top-K subgradient
        g = _topK_subgradient(a, c, K, temperature)

        # Natural gradient in logit space
        dot_ga = float(np.dot(g, a))
        gz = a * (g - dot_ga)
        gz = gz - gz.mean()  # Project to tangent space

        if not np.all(np.isfinite(gz)):
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()
            m.fill(0); v.fill(0); t = 0
            continue

        # Adam update
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a = _softmax_z(z)

        # Restart
        if no_imp > restart_every:
            noise = noise_scale * rng.standard_normal(n)
            z = np.log(np.maximum(best_a, 1e-15)) + noise
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a = _softmax_z(z)
            m.fill(0); v.fill(0); t = 0
            no_imp = 0
            lr = max(0.01, lr * 1.2)
            temperature = temp_frac * _fft_convolve(a).max()

    return best_a, best_score


def _optimize_direct(a0, deadline, rng, lr=0.02, K=5, momentum=0.9):
    """
    Direct subgradient descent on a (with projection to simplex).
    Uses heavy ball momentum.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        g = _topK_subgradient(a, c, K, 0.0)
        # Project gradient to tangent space of simplex
        g = g - g.mean()

        # Heavy ball
        v = momentum * v - base_lr * g
        a = a + v

        # Project to simplex (sum=1, a>=0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-30:
            a = a / s
        else:
            a = best_a.copy()
            v.fill(0)
            continue

        if no_imp > 500:
            a = best_a + lr * 0.5 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            v.fill(0)
            no_imp = 0
            base_lr = max(0.005, base_lr * 0.8)

    return best_a, best_score


def _scipy_polish(a0, deadline, n_max=512):
    """Use scipy minimize with smoothed objective."""
    n = len(a0)
    if n > n_max:
        return a0, _score(a0)

    def obj(z):
        a = _softmax_z(z)
        c = _fft_convolve(a)
        # Log-sum-exp approximation to max
        LSE_temp = 0.05 * c.max()
        m = c.max()
        lse = m + LSE_temp * np.log(np.sum(np.exp((c - m) / LSE_temp)))
        return 2.0 * n * lse

    def obj_grad(z):
        a = _softmax_z(z)
        c = _fft_convolve(a)
        LSE_temp = 0.05 * c.max()
        m = c.max()
        w = np.exp((c - m) / LSE_temp)
        w = w / w.sum()
        # grad of lse w.r.t. a
        ga = np.zeros(n)
        for k in range(min(20, len(c))):
            if w[k] > 1e-8:
                ga += w[k] * _subgrad_at_lag(a, k)
        # Chain rule to z
        dot = float(np.dot(ga, a))
        gz = a * (ga - dot)
        return 2.0 * n * gz

    z0 = np.log(np.maximum(a0, 1e-15))
    z0 = z0 - z0.mean()

    res = sco.minimize(obj, z0, jac=obj_grad, method='L-BFGS-B',
                       options={'maxiter': 500, 'maxfun': 2000,
                                'gtol': 1e-8, 'maxls': 50})
    if hasattr(res, 'x') and res.x is not None:
        a_new = _softmax_z(res.x)
        sc = _score(a_new)
        if sc < _score(a0):
            return a_new, sc
    return a0, _score(a0)


def _make_profile(n, kind='cosine'):
    t = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'cosine':
        a = 1.0 + 0.8 * np.cos(2 * np.pi * t)
    elif kind == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == 'raised_cos2':
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t) + 0.3 * np.cos(4 * np.pi * t)
    elif kind == 'quadratic':
        u = 2 * t - 1
        a = 1.0 + 0.5 * u**2
    elif kind == 'bell':
        a = np.exp(-2.0 * (t - 0.5)**2)
    elif kind == 'S_shape':
        a = 1.0 + 0.7 * np.sin(np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _interp_to_length(a, n_new):
    n = len(a)
    if n == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new

            current_best = best_seq.copy()

    # --- Phase 4: Scipy polish on best (if small enough) ---
    print("Phase 4: Scipy polish")
    t_phase4_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and len(best_seq) <= 512 and time.monotonic() < t_phase4_end:
        a_sp, sc_sp = _scipy_polish(best_seq, t_phase4_end, n_max=512)
        update(a_sp, sc_sp, "scipy")

    # --- Phase 5: Aggressive multi-restart on best ---
    print("Phase 5: Multi-restart refinement")
    t_phase5_end = deadline
    if best_seq is not None and time.monotonic() < t_phase5_end:
        n_best = len(best_seq)
        remaining = t_phase5_end - time.monotonic()
        n_restarts = min(6, int(remaining // 30))
        per_restart = remaining / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > t_phase5_end:
                break
            per_end = min(t_phase5_end, time.monotonic() + per_restart)
            # Different perturbation strategies
            if i % 3 == 0:
                # Gaussian noise
                z = np.log(np.maximum(best_seq, 1e-15))
                z += 0.1 * (1 + 0.3 * i) * rng.standard_normal(n_best)
                z = z - z.mean()
                a_start = _softmax_z(z)
            elif i % 3 == 1:
                # Flip a random block
                a_start = best_seq.copy()
                blen = max(5, n_best // 20)
                start = rng.integers(0, n_best - blen)
                a_start[start:start+blen] *= 0.5
                a_start = np.maximum(a_start, 0)
                a_start = a_start / a_start.sum()
            else:
                # Smooth with different kernel
                from scipy.ndimage import gaussian_filter1d
                a_start = gaussian_filter1d(best_seq, sigma=min(5, n_best//50))
                a_start = np.maximum(a_start, 0)
                a_start = a_start / a_start.sum()

            a_r, sc_r = _optimize_logit(a_start, per_end, rng, lr=0.03, K=10,
                                         temp_frac=0.04, beta_decay=0.94,
                                         restart_every=2000, noise_scale=0.08)
            update(a_r, sc_r, f"restart_{i}")

    # --- Phase 6: Final fine-tune ---
    print("Phase 6: Final fine-tune")
    if best_seq is not None and time.monotonic() < deadline:
        per_end = deadline
        a_fin, sc_fin = _optimize_logit(best_seq, per_end, rng, lr=0.01, K=15,
                                         temp_frac=0.02, beta_decay=0.97,
                                         restart_every=5000, noise_scale=0.03)
        update(a_fin, sc_fin, "final")

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = final.sum()
        if s > 0.01:
            final = final / s * np.sqrt(2 * len(final))
            # Actually just return as-is; evaluator handles scaling
            final = best_seq.copy()
            final = np.clip(final, 0.0, 1000.0)
            s = final.sum()
            if s <= 0:
                final = _make_profile(256, 'cosine')

        # Verify with evaluate_sequence if available
        try:
            test_score = evaluate_sequence([float(x) for x in final])
            print(f"  evaluate_sequence score: {test_score:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence failed: {e}")

        print(f"FINAL: score={best_score:.10f}, n={len(final)}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, 'cosine')
    print(f"FALLBACK: n=256, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
