import numpy as np

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def direct_convolve(a):
    """O(n^2) direct convolution for exact results."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    c = np.zeros(2 * n - 1, dtype=np.float64)
    for i in range(n):
        c[i:i + n] += a[i] * a
    return c


def fft_convolve(a):
    """FFT-based convolution (fast for larger n)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return direct_convolve(a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

        return np.inf

        return np.inf
    c = fft_convolve(a)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf

        for peak_k in peak_indices[:3]:
            # Find elements that contribute to this peak
            # c[peak_k] = sum_i p[i] * p[peak_k - i]
            contributions = np.zeros(n)
            for i in range(n):
                j = peak_k - i
                if 0 <= j < n:
                    contributions[i] = p[i] * p[j]
            
            # Sort by contribution
            contrib_order = np.argsort(contributions)[::-1]
            
            # Try adjusting top contributors
            for idx in contrib_order[:min(10, n)]:
                orig = p[idx]
                if orig < 1e-10:
                    continue
                # Try scaling down (reduces peak)
                for factor in (0.8, 0.9, 0.7):
                    new_val = orig * factor
                    p[idx] = new_val
                    s = float(p.sum())
                    if s < 1e-10:
                        p[idx] = orig
                        continue
                    c_new = direct_convolve(p)
                    mc_new = float(c_new.max())
                    if mc_new < best_mc - 1e-15:
                        best_mc = mc_new
                        best_p = p.copy()
                        improved = True
                        no_imp = 0
                        break
                    p[idx] = orig
                
                # Try scaling up (might help if it reduces another peak)
                for factor in (1.1, 1.2):
                    new_val = orig * factor
                    p[idx] = new_val / (s + (new_val - orig))  # renormalize
                    # Actually, let me just try without renormalization
                    c_new = direct_convolve(p)
                    mc_new = float(c_new.max())
                    if mc_new < best_mc - 1e-15:
                        # Renormalize
                        p = p / p.sum()
                        best_mc = float(direct_convolve(p).max())
                        best_p = p.copy()
                        improved = True
                        no_imp = 0
                        break
                    p[idx] = orig
        
                if time.monotonic() > deadline:
                    break
            
            if time.monotonic() > deadline:
                break
        
        if not improved:
            no_imp += 1
            if no_imp > 10:
                break
        else:
            p = best_p.copy()
            c = direct_convolve(p)
    
    # Renormalize
    best_p = best_p / best_p.sum()
    return best_p, float(2.0 * n * direct_convolve(best_p).max())

def simulated_annealing(p, deadline, rng):
    """
    Simulated annealing: randomly perturb elements and accept/reject.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    
    c = direct_convolve(p)
    cur_mc = float(c.max())
    best_p = p.copy()
    best_mc = cur_mc
    
    T = 0.01  # Initial "temperature" (relative)
    T_min = 1e-6
    cool_rate = 0.999
    
    while time.monotonic() < deadline:
        # Choose 1-3 random elements to perturb
        n_perturb = int(rng.integers(1, 4))
        indices = rng.choice(n, size=n_perturb, replace=False)
        
        p_new = p.copy()
        for idx in indices:
            factor = 1.0 + rng.normal(0, 0.1)
            p_new[idx] = max(1e-15, p_new[idx] * factor)
        
        p_new = p_new / p_new.sum()
        c_new = direct_convolve(p_new)
        new_mc = float(c_new.max())
        
        delta = new_mc - cur_mc
        if delta < 0 or rng.random() < np.exp(-delta / (T * cur_mc + 1e-15)):
            p = p_new
            cur_mc = new_mc
            if cur_mc < best_mc - 1e-15:
                best_mc = cur_mc
                best_p = p.copy()
        
        T = max(T_min, T * cool_rate)
    
    best_p = best_p / best_p.sum()
    return best_p, float(2.0 * n * direct_convolve(best_p).max())
