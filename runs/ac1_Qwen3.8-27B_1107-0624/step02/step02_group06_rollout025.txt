"""
Optimized search for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multi-phase approach: diverse initialization, soft-max gradient Adam,
targeted peak reduction, and coordinate descent polish.
"""
import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_autocorr(a):
    """Compute autocorrelation using FFT."""
    n = len(a)
    if n <= 32:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Clean up numerical noise
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_autocorr(a)
    return 2.0 * n * c.max() / (s * s)


def true_score(a):
    """Exact score using numpy convolve."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def soft_max_and_weights(c, beta):
    """Compute soft max value and attention weights."""
    mval = c.max()
    if beta < 1e-15:
        mask = c >= mval - 1e-10
        w = np.zeros_like(c)
        w[mask] = 1.0 / mask.sum()
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = ex.sum()
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
        return mval, w
    val = mval + beta * np.log(s)
    w = ex / s
    return val, w


def corr_weights_fft(w, a):
    """Compute the correlation of weights w with sequence a (for gradient)."""
    n = len(a)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1: 2 * n - 1]


def softmax_from_logits(z):
    """Convert logits to probabilities (normalized non-negative)."""
    z = np.clip(z, -50, 50)
    z = z - z.max()
    e = np.exp(z)
    e = e + 1e-15
    return e / e.sum()


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def make_random_sequence(n, rng, dist='uniform'):
    if dist == 'uniform':
        a = rng.uniform(0.1, 1.0, n)
    elif dist == 'exponential':
        a = rng.exponential(1.0, n) + 0.01
    elif dist == 'gamma':
        a = rng.gamma(2.0, 1.0, n) + 0.01
    elif dist == 'bimodal':
        a = np.where(rng.random(n) < 0.5, rng.uniform(0.5, 1.0, n), rng.uniform(0.05, 0.3, n))
    else:
        a = rng.random(n) + 0.01
    return normalize(a)


def make_structured_sequence(n, rng):
    """Generate a structured (smooth but non-uniform) sequence."""
    t = np.linspace(0, 1, n)
    choice = rng.integers(0, 5)
    if choice == 0:
        a = 1.0 + rng.uniform(0.5, 2.0) * np.sin(2 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 2*np.pi))
    elif choice == 1:
        a = 1.0 + rng.uniform(0.5, 2.0) * np.cos(2 * np.pi * rng.uniform(1, 5) * t)
    elif choice == 2:

    elif choice == 3:

    a = np.maximum(a, 0.01)
    return normalize(a)


def soft_max_adam(p0, deadline, rng, lr=0.1, beta_decay=0.85, beta_interval=30):
    """
    Soft-max Adam optimization in logit space.
    Minimizes 2n * max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    p0 = np.asarray(p0, dtype=np.float64)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    # Initialize logits
    z = np.log(p0 + 1e-15)
    z = z - z.mean()

    # Adam state
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    # Initial beta
    c = fft_autocorr(p0)
    beta = max(1e-6, 0.3 * c.max())

    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * c.max()
    no_imp = 0
    step = 0

    while time.monotonic() < deadline:
        # Compute soft max and gradient
        L, w = soft_max_and_weights(c, beta)
        gp = 2.0 * corr_weights_fft(w, p0)
        # Reg
        vp = gp + 1e-7 * p0
        dot = p0 @ vp
        gz = p0 * (vp - dot)
        gz -= gz.mean()

        # Adam update
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m[:] = 0
            v[:] = 0
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m[:] = 0
            v[:] = 0
            t_adam = 0
            continue

        z = np.clip(z, -50, 50)
        z = z - z.mean()
        p = softmax_from_logits(z)
        c = fft_autocorr(p)
        score = 2.0 * n * c.max()

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1

            # Decay beta
            if step % beta_interval == 0:
                beta = max(1e-10, beta * beta_decay)

            # Learning rate adjustment
            if no_imp % 100 == 0 and no_imp > 0:
                lr = max(0.02, lr * 0.95)

            # Restart if stuck
            if no_imp > max(200, 3 * n):
                # Perturb from best
                z = best_z + 0.5 * rng.standard_normal(n)
                z = np.clip(z, -50, 50)
                z = z - z.mean()
                p = softmax_from_logits(z)
                c = fft_autocorr(p)
                beta = max(1e-8, 0.2 * c.max())
                m[:] = 0
                v[:] = 0
                t_adam = 0
                lr = 0.1
                no_imp = 0
        else:
            z = best_z.copy()
            m[:] = 0
            v[:] = 0
            t_adam = 0
            continue

        step += 1

    return best_p, best_score


def targeted_peak_reduction(p, deadline, rng, top_k=5, n_iters=500):
    """
    Directly reduce the top-k peaks of the autocorrelation.
    For each peak position, try to reduce it by adjusting the sequence.
    """
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = 2.0 * n * fft_autocorr(p).max()
    no_imp = 0

    for it in range(n_iters):
        if time.monotonic() > deadline or no_imp > 50:
            break

        c = fft_autocorr(p)
        # Get top-k peak positions
        top_idx = np.argsort(c)[-top_k:][::-1]

        # For each peak position, try to reduce it
        improved = False
        for k in top_idx:
            if time.monotonic() > deadline:
                break
            # The peak at position k involves pairs (i, k-i) for valid i
            # Try adjusting the elements involved in the peak
            # Simple approach: find the element that contributes most to the peak
            # and reduce it slightly
            # conv[k] = sum_i a[i] * a[k-i] (with bounds)
            # The "center" elements contribute most for central peaks

            # Try reducing the element at position max(0, k-(n-1)) 
            # or the element at position (k)//n or similar
            # Better: try all elements and pick the one that most reduces the peak
            
            # For efficiency, try a few random elements
            for _ in range(3):
                i = rng.integers(0, n)
                orig = p[i]
                # Try reducing element i
                factor = np.random.choice([0.9, 0.8, 0.7, 0.5])
                p[i] = orig * factor
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
                newscore = 2.0 * n * fft_autocorr(p).max()
                if newscore < best_score - 1e-12:
                    best_score = newscore
                    best_p = p.copy()
                    improved = True
                    no_imp = 0
                else:
                    p = best_p.copy()

        if improved:
            no_imp = 0
        else:
            no_imp += 1

        # Occasionally add a random perturbation
        if it % 20 == 0:
            idx = rng.integers(0, n)
            p[idx] *= rng.uniform(0.8, 1.2)
            p = np.maximum(p, 1e-12)
            p = p / p.sum()

    return best_p, best_score


def coordinate_descent(p, deadline, rng, n_sweeps=10):
    """Coordinate descent: try adjusting one element at a time."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved_any = False

import numpy as np

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_autocorr(a):
    n = len(a)
    if n <= 32:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_autocorr(a)
    return 2.0 * n * c.max() / (s * s)


def true_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def soft_max_and_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            mask[int(c.argmax())] = True
            cnt = 1
        w = np.zeros_like(c)
        w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
        return mval, w
    val = mval + beta * np.log(s)
    w = ex / s
    return val, w


def corr_weights_fft(w, a):
    n = len(a)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1: 2 * n - 1]


def softmax_from_logits(z):
    z = np.clip(z, -50.0, 50.0)

    return e / e.sum()

def soft_max_adam(p0, deadline, rng, lr=0.10, beta_decay=0.82, beta_interval=25, reg=1e-7):

    z = z - float(z.mean())

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0

    c = fft_autocorr(p0)
    beta = max(1e-7, 0.35 * float(c.max()))

    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * float(c.max())
    no_imp = 0
    step = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_max_and_weights(c, beta)
        gp = 2.0 * corr_weights_fft(w, p0)

        gz = p0 * (vp - dot)
        gz = gz - float(gz.mean())

            z = best_z.copy()
            m_adam[:] = 0
            v_adam[:] = 0
            t_adam = 0
            continue

        t_adam += 1
        m_adam = b1_adam * m_adam + (1.0 - b1_adam) * gz
        v_adam = b2_adam * v_adam + (1.0 - b2_adam) * gz * gz

        vh = v_adam / (1.0 - b2_adam ** t_adam)

            z = best_z.copy()
            m_adam[:] = 0
            v_adam[:] = 0

                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0

        else:
            no_imp = 0

        if rnd % 30 == 29:
            idx = int(rng.integers(0, n))
            best_p[idx] *= float(rng.uniform(0.7, 1.4))
            best_p = np.maximum(best_p, 1e-12)
            best_p = best_p / best_p.sum()
            p = best_p.copy()
            sc = 2.0 * n * float(fft_autocorr(p).max())
            if sc < best_score:
                best_score = sc

    return best_p, best_score


def coordinate_descent(p, deadline, rng, max_sweeps=8):
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = 2.0 * n * float(fft_autocorr(p).max())

    factors = [1.5, 1.3, 1.15, 1.07, 0.93, 0.85, 0.7, 0.5, 2.0, 0.3]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved_any = False
        for i in order:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = float(fft_autocorr(p).max())
            for factor in factors:
                new_val = orig * factor
                if new_val < 1e-12:
                    continue
                p[i] = new_val
                p = p / p.sum()
                sc = float(fft_autocorr(p).max())
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            cur_sc = float(fft_autocorr(p).max())
            if cur_sc < best_score - 1e-13:
                best_score = cur_sc
                best_p = p.copy()
                improved_any = True

        if not improved_any:
            break

    return best_p, best_score
