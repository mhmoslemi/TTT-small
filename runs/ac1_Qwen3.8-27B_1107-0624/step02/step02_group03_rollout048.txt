        g = np.maximum(res.x, 0.0)

def smooth_max_obj_grad(z, n, T):
    """
    Objective and gradient in logit space.
    z: logit vector (length n)
    Returns (objective_value, gradient)
    Objective = T * logsumexp(conv(softmax(z)) / T)  [which approx max(conv)]
    """
    # Compute softmax
    z_shifted = z - z.max()
    ez = np.exp(z_shifted)
    a = ez / ez.sum()
    
    # Compute convolution
    c = np.convolve(a, a)
    
    # Smooth max
    cm = c.max()
    if T < 1e-10:
        obj = cm
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        c_shifted = c - cm
        c_scaled = c_shifted / T
        # Clip for numerical stability
        c_scaled = np.clip(c_scaled, -500, 500)
        exp_c = np.exp(c_scaled)
        log_sum = np.log(exp_c.sum())
        obj = cm + T * log_sum
        w = exp_c / exp_c.sum()
    
    # Gradient of obj w.r.t. a
    # d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i < n
    # grad_a[i] = sum_k w[k] * 2*a[k-i] = 2 * (w cross-correlate a)[i]
    grad_a = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = i + n
        grad_a[i] = 2.0 * np.dot(w[k_start:k_end], a)
    
    # Gradient in logit space
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z

"""

Uses L-BFGS-B with smooth max continuation, LP direction finding,
and multi-resolution search across n=512 and n=1024.

def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=15.0):

    n = len(f)

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:

def compute_grad_wrt_a(a, T, c=None):
    """
    Compute gradient of T*logsumexp(conv(a,T)) w.r.t. a.
    Returns (objective_value, grad_a, w) where w is the softmax weights.
    """
    n = len(a)
    if c is None:
        c = np.convolve(a, a)
    cm = c.max()
    if T < 1e-12:
        obj = cm
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:

        c_shifted = np.clip(c_shifted, -600, 600)
        exp_c = np.exp(c_shifted)
        exp_sum = exp_c.sum()
        if exp_sum < 1e-300:
            obj = cm
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            obj = cm + T * np.log(exp_sum)
            w = exp_c / exp_sum

    # grad_a[i] = 2 * sum_k w[k] * a[k-i] for k in [i, i+n-1]
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = 2.0 * np.dot(w[i:i + n], a)

    return obj, grad_a, w

    v = np.zeros(n)

    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()

    beta = max(1e-12, 0.1 * max_c)

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        if s < 1e-300:

        gz = p0 * (grad_a - dot + reg * p0)

        if not np.all(np.isfinite(gz)):

            m.fill(0)
            v.fill(0)
            it = 0
            continue

        v = b2 * v + (1 - b2) * gz * gz

        z = z - lr * mh / (np.sqrt(vh) + eps)

def make_profile(n, type_id, rng):
    """Generate various initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    if type_id == 0:
        # Linear
        eta = rng.uniform(0.5, 2.5)
        a = 1.0 + eta * u
    elif type_id == 1:
        # Quadratic
        a = 1.0 + rng.uniform(-0.5, 1.5) * u + rng.uniform(0.1, 1.0) * u * u
    elif type_id == 2:
        # Cubic
        a = 1.0 + rng.uniform(-0.5, 1.5) * u + rng.uniform(-0.3, 0.8) * u * u + rng.uniform(-0.2, 0.5) * u ** 3
    elif type_id == 3:
        # Gaussian-like
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.4)

    elif type_id == 4:
        # Two-bump

        c2 = rng.uniform(0.6, 0.8)
        w = rng.uniform(0.05, 0.2)
        a = np.exp(-((t - c1) ** 2) / (2 * w ** 2)) + np.exp(-((t - c2) ** 2) / (2 * w ** 2))
    else:

        a = np.ones(n)
        for k, c_val in enumerate(coeffs, start=1):

        for tid in range(6):
            p = make_profile(n_init, tid, rng)
            sc = internal_score(p)
            if sc < best_init_score:
                best_init_score = sc
                best_init = p
        update_best(best_init)
        print(f"Generated initial: n={len(best_init)}, score={best_score:.8f}")

    # === Phase 1: LP at n=512 ===
    n1 = 512
    deadline_p1 = min(deadline, t_start + 350.0)
    a1 = interpolate_seq(best_seq, n1)
    if a1 is not None and time.monotonic() < deadline_p1:
        print(f"\nPhase 1: LP at n={n1}, initial score={internal_score(a1):.8f}")
        f_best, sc_best = lp_iterate(a1, max_iters=500, t_init=0.4, t_min=0.02,
                                     deadline=deadline_p1, verbose=True, time_per_lp=5.0)
        if f_best is not None:
            update_best(f_best)
        a1 = best_seq if len(best_seq) == n1 else interpolate_seq(best_seq, n1)

    # === Phase 2: L-BFGS at n=512 ===
    deadline_p2 = min(deadline, t_start + 700.0)
    if time.monotonic() < deadline_p2:
        a2 = interpolate_seq(best_seq, n1) if len(best_seq) != n1 else best_seq.copy()
        print(f"\nPhase 2: L-BFGS at n={n1}, initial score={internal_score(a2):.8f}")
        p_best, sc_best = lbgfs_polish(a2, deadline_p2, T_init=0.05, T_min=0.0005,
                                       n_cont=10, max_iter=2000, verbose=True)
        update_best(p_best)

    # === Phase 3: LP + L-BFGS at n=1024 ===
    n3 = 1024
    deadline_p3 = min(deadline, t_start + 1300.0)
    if time.monotonic() < deadline_p3 - 100:
        a3 = interpolate_seq(best_seq, n3)
        print(f"\nPhase 3a: LP at n={n3}, initial score={internal_score(a3):.8f}")
        f_best, sc_best = lp_iterate(a3, max_iters=100, t_init=0.35, t_min=0.02,
                                     deadline=min(deadline_p3 - 200, time.monotonic() + 350),
                                     verbose=True, time_per_lp=10.0)
        if f_best is not None:
            update_best(f_best)

    deadline_p3b = min(deadline, t_start + 1600.0)
    if time.monotonic() < deadline_p3b:
        a3 = interpolate_seq(best_seq, n3)
        print(f"\nPhase 3b: L-BFGS at n={n3}, initial score={internal_score(a3):.8f}")
        p_best, sc_best = lbgfs_polish(a3, deadline_p3b, T_init=0.03, T_min=0.0003,
                                       n_cont=8, max_iter=2000, verbose=True)
        update_best(p_best)

    # === Phase 4: Final Adam polish on best found ===
    if time.monotonic() < deadline - 5:
        polish_n = len(best_seq)
        print(f"\nPhase 4: Adam polish at n={polish_n}, initial score={internal_score(best_seq):.8f}")
        p_best, sc_best = adam_polish(best_seq, deadline - 5, lr=0.015, reg=1e-9, verbose=False)
        update_best(p_best)

    # === Phase 5: Try a few random restarts at n=512 if time remains ===
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_restarts = min(3, int(remaining / 120))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            print(f"\nPhase 5: Restart {r+1}/{n_restarts}")
            # Perturb best solution
            perturb = best_seq + 0.1 * rng.standard_normal(len(best_seq))
            perturb = normalize(perturb)
            if perturb is None:
                continue
            p_best, sc_best = lbgfs_polish(perturb, min(deadline - 30, time.monotonic() + 100),
                                           T_init=0.03, T_min=0.001, n_cont=5, max_iter=1000,
                                           verbose=False)
            update_best(p_best)

    # Final cleanup
    print(f"\n=== FINAL: n={len(best_seq)}, score={internal_score(best_seq):.8f} ===")

    final = np.asarray(best_seq, dtype=np.float64)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        # Fallback
        final = np.ones(512) / 512.0
        s = 1.0
    else:
        final = final / s

    return [float(x) for x in final]
