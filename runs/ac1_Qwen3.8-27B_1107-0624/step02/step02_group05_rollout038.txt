    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def cross_corr_fft(w, f):
    """
    Compute h[i] = sum_j w[i+j] * f[j] for i=0..n-1.
    w has length 2n-1, f has length n.
    """
    n = len(f)
    N = 1
    while N < 3 * n:
        N <<= 1
    fw = np.fft.rfft(w, N)
    ff = np.fft.rfft(f, N)
    h = np.fft.irfft(fw * np.conj(ff), N)
    return h[:n]


def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp(f, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def isotonic_regression(f):
    """Project f onto monotonically non-decreasing sequences (PAVA)."""
    n = len(f)
    result = f.copy()
    block_start = [0] * n
    block_end = [0] * n
    weight = [1.0] * n
    value = f.copy()

    # Pool Adjacent Violators Algorithm
    blocks = list(range(n))
    block_vals = f.copy()
    block_weights = np.ones(n)
    active = [i for i in range(n)]

    # Simpler PAVA implementation
    result = f.astype(np.float64).copy()
    n_blocks = [1] * n
    n_start = list(range(n))

    # Use the standard PAVA
    blocks = []
    for i in range(n):
        blocks.append([f[i], 1])
        while len(blocks) > 1 and blocks[-2][0] > blocks[-1][0]:
            v1, w1 = blocks[-2]
            v2, w2 = blocks[-1]
            blocks.pop()
            blocks.pop()
            blocks.append([(v1 * w1 + v2 * w2) / (w1 + w2), w1 + w2])

    # Reconstruct
    pos = 0
    for val, w in blocks:
        for j in range(int(w)):
            result[pos + j] = val
        pos += int(w)

    return result


def adam_optimize(f_init, deadline, rng, lr=0.05, max_iters=10000,
                  beta_init=0.1, beta_decay=0.995, verbose=False,
                  use_lp=False, lp_interval=100):
    """
    Adam optimization in softmax parameterization with adaptive soft-max.
    """
    n = len(f_init)
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()

    # Logit parameterization
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = np.convolve(f, f)
    M = float(c.max())
    beta = beta_init * M

    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute gradient
        w = soft_max_weights(c, beta)
        h = cross_corr_fft(w, f)
        g = 2.0 * h
        dot = float(np.dot(f, g))
        gz = f * (g - dot)
        gz -= gz.mean()

        # Add small regularization to prevent zeros
        gz += 1e-8 * f

        if not np.all(np.isfinite(gz)):
            # Reset to best
            f = best_f.copy()
            z = np.log(np.maximum(f, 1e-12))
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        # Adam update
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        # Clip and renormalize
        z = np.clip(z, -50, 50)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()

        # Ensure non-negative
        f = np.maximum(f, 0.0)
        s = float(f.sum())
        if s < 1e-30:
            f = best_f.copy()
            continue
        f = f / s

        # Recompute convolution
        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta
        if adam_t % 100 == 0:
            beta = max(M * 0.01, beta * beta_decay)

        # LP step (complementary direction)
        if use_lp and it > 0 and it % lp_interval == 0 and no_imp > 20:
            g_dir = solve_lp(f, M, time_limit=3.0)
            if g_dir is not None:
                g_sum = float(g_dir.sum())
                if g_sum > 1e-10:
                    g_dir = g_dir / g_sum
                    # Try blending
                    for t in [0.3, 0.2, 0.1]:
                        f_try = (1 - t) * f + t * g_dir
                        f_try = normalize_nonneg(f_try)
                        if f_try is not None:
                            M_try = compute_M(f_try)
                            if M_try < M - 1e-14:
                                z = np.log(np.maximum(f_try, 1e-12))
                                z -= z.mean()
                                f = f_try
                                c = np.convolve(f, f)
                                M = float(c.max())
                                m.fill(0)
                                v.fill(0)
                                adam_t = 0
                                break

        # Restart if stuck
        if no_imp > max(500, 5 * n):
            # Perturb in log space
            z = np.log(np.maximum(best_f, 1e-12))
            z -= z.mean()
            z += 0.2 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(M * 0.1, beta * 0.5)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            no_imp = 0

        if verbose and it % 500 == 0:
            print(f"    iter {it}: M={M:.12f}, best_M={best_M:.12f}, beta={beta:.6f}")

    return best_f, best_M


def lp_optimize(f_init, deadline, rng, max_iters=500, verbose=False):
    """LP-based iterative optimization."""
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_M(f_init)

    n = len(f)
    M = compute_M(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = solve_lp(f, M, time_limit=5.0)
        if g is None:
            f = f + 0.1 * rng.normal(size=n)
            f = normalize_nonneg(f)
            if f is None:
                f = best_f.copy()
            M = compute_M(f)
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search
        best_t = 0.0
        f_try = f
        M_try = M
        for t in [0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            f_cand = (1.0 - t) * f + t * g
            f_cand = normalize_nonneg(f_cand)
            if f_cand is None:
                continue
            M_cand = compute_M(f_cand)
            if M_cand < M_try - 1e-14:
                M_try = M_cand
                f_try = f_cand
                best_t = t

        if best_t > 0:
            f = f_try
            M = M_try
            if M < best_M - 1e-14:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 30:
            f = 0.7 * best_f + 0.3 * (best_f + 0.2 * rng.normal(size=n))
            f = normalize_nonneg(f)
            if f is None:
                f = best_f.copy()
            M = compute_M(f)
            no_imp = 0

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: M={M:.12f}, best_M={best_M:.12f}")

    return best_f, best_M


def multi_scale_optimize(f_best, n_target, deadline, rng):
    """
    Multi-scale optimization: optimize at smaller scales, then upsample and refine.
    """
    results = []

    # Try the target size directly
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.6, 600.0)
        run_deadline = time.monotonic() + alloc
        f_init = f_best if len(f_best) == n_target else interpolate(f_best, n_target)
        f_opt, M_opt = adam_optimize(f_init, run_deadline, rng, lr=0.04,
                                     beta_init=0.15, use_lp=True, lp_interval=200)
        results.append((f_opt, M_opt))

    # Try intermediate scales
    n_mid = n_target // 2
    if n_mid >= 32 and time.monotonic() < deadline - 60:
        alloc = min(deadline - time.monotonic() - 30, 200.0)
        if alloc > 30:
            run_deadline = time.monotonic() + alloc
            f_init = f_best if len(f_best) == n_mid else interpolate(f_best, n_mid)
            f_opt, M_opt = adam_optimize(f_init, run_deadline, rng, lr=0.05,
                                         beta_init=0.1, use_lp=True, lp_interval=100)
            results.append((f_opt, M_opt))

    return results


def interpolate(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, f)
    return np.maximum(result, 1e-12)


def get_starting_points(n, rng, count=10):
    """Generate diverse starting points based on analytical insights."""
    starts = []
    # Linear ramp (best analytical solution)
    starts.append(linear_ramp(n))
    # Slight variations
    starts.append(power_ramp(n, 0.9))
    starts.append(power_ramp(n, 1.1))
    starts.append(power_ramp(n, 0.8))
    starts.append(power_ramp(n, 1.2))
    # Quadratic variations
    starts.append(quadratic_ramp(n, 1.0, 0.1))
    starts.append(quadratic_ramp(n, 1.0, 0.3))
    starts.append(quadratic_ramp(n, 0.8, 0.4))
    # Random smooth perturbations of the ramp
    base = linear_ramp(n)
    for _ in range(count):
        theta = rng.uniform(-1, 1, 6)
        t = np.linspace(0, 1, n)
        pert = 1.0
        for k in range(3):
            pert = pert + 0.1 * theta[2*k] * np.cos(2 * np.pi * (k + 1) * t)
            pert = pert + 0.1 * theta[2*k + 1] * np.sin(2 * np.pi * (k + 1) * t)
        f = base * pert
        f = np.maximum(f, 1e-12)
        f = f / f.sum()
        starts.append(f)
    return starts
