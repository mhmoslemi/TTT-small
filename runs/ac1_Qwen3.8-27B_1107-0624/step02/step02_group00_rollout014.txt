import numpy as np
import time
from scipy.optimize import linprog

# --- Utility functions ---

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _lp_direction(a):
    """Solve the LP to find the optimal spreading direction.
    Returns the direction (normalized) and the LP value V.
    """
    n = len(a)
    c = _fft_convolve(a)
    M = float(c.max())
    
    # Try with slightly inflated M
    for inflate in [1.0, 1.0001, 1.001, 1.01, 1.05]:
        M_try = M * inflate
        g = _lp_spread(a, M_try)
        if g is not None:
            V = float(np.sum(g))
            if V > 1.0 + 1e-6:
                return g / V, V, inflate
    return None, 1.0, 1.0

def _line_search_blend(a, g, max_trials=10):
    """Find the best blend factor t in [0,1] for (1-t)*a + t*g."""
    best_t = 0.0
    best_score = _score(a)
    
    # Try a set of blend factors
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new = _normalize(a_new)
        s = _score(a_new)
        if s < best_score:
            best_score = s
            best_t = t

def _subgradient_step(a, lr, top_k=5):
    """Take a subgradient step targeting the top-k peak positions."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(c.max())
    
    # Find top-k peak positions
    k_peaks = np.argsort(c)[::-1][:top_k]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in k_peaks:
        k = int(k)
        # grad[j] += 2 * a[k-j] for valid j
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            indices = np.arange(j_min, j_max + 1)
            k_minus_j = k - indices
            grad[indices] += 2.0 * a[k_minus_j]
    
    grad /= max(1, len(k_peaks))
    
    # Project to simplex with step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    a_new = _normalize(a_new)
    return a_new, grad

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _make_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_exp_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(alpha * (0.5 - t))
    return a / a.sum()

def _make_beta_profile(n, p=2.0, q=1.0):
    t = np.linspace(0.01, 0.99, n)
    a = t ** p * (1.0 - t) ** q
    return a / a.sum()

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    return _normalize(p)

def _subgradient_polish(a, deadline, max_iter=50000):

    best_score = _score(a)

    base_lr = lr
    no_imp = 0
    top_k = 3
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a_new, grad = _subgradient_step(a, lr, top_k)
        s = _score(a_new)

            no_imp = 0

                lr = base_lr
                no_imp = 0
                top_k = min(10, top_k + 2)
        
        if it % 500 == 0:
            # Periodically try with different top_k
            top_k = 1 + (it // 500) % 5
    
    return best_a

"""
Improved LP + Subgradient + Basin Hopping optimization for minimizing
2n*max(conv(a,a))/sum(a)^2.

Key improvements over previous:
1. Line search for blend factor in LP direction
2. Adaptive inflation with progress-based reset
3. Subgradient descent targeting active peaks
4. Basin hopping for diversity
5. Polynomial parameter optimization for global search
6. Better time management
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_spread_sparse(a, M):
    """LP spread using sparse matrix for efficiency."""
    from scipy.sparse import diags, eye
    n = len(a)
    # Build sparse Toeplitz-like matrix
    offsets = list(range(-(n - 1), n))
    diags_list = []
    for off in offsets:
        if off >= 0:
            length = n - off
            val = a[:length]
        else:
            length = n + off
            val = a[-off:]
        diags_list.append(val)
    A = diags(diags_list, offsets, shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

    n = len(a)

    # Find top-k peak positions
    if top_k >= len(c):
        k_peaks = np.arange(len(c))
    else:
        k_peaks = np.argsort(c)[::-1][:top_k]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in k_peaks:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            indices = np.arange(j_min, j_max + 1)
            k_minus_j = k - indices
            grad[indices] += 2.0 * a[k_minus_j]
    
    grad /= max(1, len(k_peaks))
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    a_new = _normalize(a_new)
    return a_new


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(alpha * (0.5 - t))
    return a / a.sum()


def _make_beta_profile(n, p=2.0, q=1.0):
    t = np.linspace(0.01, 0.99, n)
    a = t ** p * (1.0 - t) ** q
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    return _normalize(p)


def _lp_refine(a, deadline, max_iter=50000, use_sparse=False):
    """LP refinement with line search and adaptive inflation."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(a)
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    no_imp = 0
    inflate = 1.0
    blend_candidates = [0.1, 0.2, 0.3, 0.5]
    
    solve_fn = _lp_spread_sparse if use_sparse else _lp_spread
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        
        # Try with current M first, then inflate if needed
        found = False
        for inf in [1.0, 1.0001, 1.001, 1.01, 1.05]:
            M_try = M * inf
            g = solve_fn(a, M_try)
            if g is not None:
                V = float(np.sum(g))
                if V > 1.0 + 1e-6:
                    found = True
                    break
        
        if not found:
            inflate = min(2.0, inflate * 1.5)
            if inflate > 1.5:
                break
            no_imp += 1
            if no_imp > 100:
                break
            continue
        
        g_norm = g / V
        
        # Line search for best blend
        best_t = 0.0
        best_s = _score(a)
        for t in blend_candidates:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            a_new = _normalize(a_new)
            s = _score(a_new)
            if s < best_s:
                best_s = s
                best_t = t
        
        if best_t == 0.0:
            no_imp += 1
            if no_imp % 100 == 0:
                # Try smaller blends
                blend_candidates = [0.02, 0.05, 0.1, 0.15]
            if no_imp > 500:
                # Reset and try with more inflation
                a = best_a.copy()
                no_imp = 0
                inflate = min(1.1, inflate + 0.05)
                blend_candidates = [0.1, 0.2, 0.3, 0.5]
                if inflate > 1.3:
                    break
            continue
        
        a = (1.0 - best_t) * a + best_t * g_norm
        a = np.maximum(a, 0)
        a = _normalize(a)
        
        if best_s < best_score - 1e-12:
            best_score = best_s
            best_a = a.copy()
            no_imp = 0
            if it % 10000 == 0 and it > 0:
                print(f"    LP iter {it}: score={best_score:.10f}")
        else:
            no_imp += 1
    
    return best_a


def _subgradient_polish(a, deadline, max_iter=50000):
    """Subgradient descent with adaptive step size and top-k selection."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(a)
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    lr = 0.005
    base_lr = lr
    no_imp = 0
    top_k = 3
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Vary top_k periodically
        current_top_k = 1 + (it // 200) % 7
        
        a_new = _subgradient_step(a, lr, current_top_k)
        s = _score(a_new)
        
        if s < best_score - 1e-12:
            best_score = s
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            a = a_new
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.0005, lr * 0.92)
            if no_imp > 3000:
                a = best_a.copy()
                lr = base_lr
                no_imp = 0
    
    return best_a


def _adam_smooth_max(a, deadline, max_iter=30000, lr=0.03):
    """Adam optimization using smooth max approximation."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    
    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0
    current_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        
        # Gradient: 2 * (w convolved with reversed a), truncated
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        
        vp = 2.0 * h + 1e-6 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                current_lr = max(0.003, current_lr * 0.95)
        
        if it % 300 == 0:
            beta = max(1e-10, beta * 0.7)
        
        if no_imp > max(5000, 8 * n):
            z = 0.7 * best_z + 0.5 * np.random.default_rng(it).normal(size=n) * 0.1
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            current_lr = lr
            no_imp = 0
    
    return best_a


def _polynomial_optimize(n, deadline, parent=None):
    """Optimize polynomial profile coefficients."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    
    def objective(coeffs):
        a = 1.0
        for i, c in enumerate(coeffs):
            a = a + c * (u ** i)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        return _score(a)
    
    best_x0 = None
    best_obj = np.inf
    
    # Try multiple starting points
    n_coeffs = 6
    starts = []
    if parent is not None:
        p_interp = _interpolate_profile(parent, n)
        starts.append(np.zeros(n_coeffs))
        for c in np.random.default_rng(42).normal(size=(5, n_coeffs)) * 0.5:
            starts.append(c)
    else:
        starts.append(np.zeros(n_coeffs))
        for c in np.random.default_rng(42).normal(size=(5, n_coeffs)) * 0.5:
            starts.append(c)
    
    for x0 in starts:
        if time.monotonic() > deadline:
            break
        res = minimize(objective, x0, method='Nelder-Mead',
                       options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10})
        if res.fun < best_obj:
            best_obj = res.fun
            best_x0 = res.x
    
    if best_x0 is not None:
        a = 1.0
        for i, c in enumerate(best_x0):
            a = a + c * (u ** i)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    return _make_linear_profile(n, 1.5)


def _basin_hop(a, deadline, rng, min_hop_time=15, max_hops=50):
    """Basin hopping with LP re-optimization."""
    best_a = a.copy()
    best_score = _score(a)
    n = len(a)
    
    for hop in range(max_hops):
        remaining = deadline - time.monotonic()
        if remaining < min_hop_time * 2:
            break
        
        # Perturb with varying noise scale
        noise_scale = 0.03 + 0.15 * rng.random()
        a_pert = a * (1.0 + noise_scale * rng.normal(size=n))
        a_pert = np.maximum(a_pert, 0)
        a_pert = _normalize(a_pert)
        
        # Try structured perturbations too
        if hop % 3 == 1:
            # Shift mass from peaks to valleys
            c = _fft_convolve(a_pert)
            shift = 0.02 * rng.random()
            a_pert = a_pert + shift * (a_pert.max() - a_pert)
            a_pert = np.maximum(a_pert, 0)
            a_pert = _normalize(a_pert)
        elif hop % 3 == 2:
            # Random Fourier perturbation
            N = _next_pow2(n)
            fa = np.fft.rfft(a_pert, N)
            fa[1:] *= (1.0 + 0.1 * rng.normal(size=len(fa[1:])))
            a_pert = np.fft.irfft(fa, N)[:n]
            a_pert = np.maximum(a_pert, 0)
            a_pert = _normalize(a_pert)
        
        # Quick LP refinement
        hop_budget = min(remaining * 0.3, 60.0)
        hop_deadline = time.monotonic() + hop_budget
        a_opt = _lp_refine(a_pert, hop_deadline, max_iter=3000)
        s = _score(a_opt)
        
        if s < best_score:
            best_score = s
            best_a = a_opt.copy()
            a = a_opt.copy()
            print(f"    Basin hop {hop}: improved to {s:.10f}")
    
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid LP + Subgradient + Basin Hopping optimization.
    
    Phases:
    1. Initialize from parent + alternatives (0-150s)
    2. Intensive LP refinement (150-800s)
    3. Adam smooth-max polish (800-1100s)
    4. Basin hopping (1100-1400s)
    5. Final combined polish (1400-1770s)
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Phase 1: Multiple initializations
    n_main = 512
    if parent is not None and parent.size in [256, 384, 512, 640, 1024]:
        n_main = parent.size
    
    print(f"Phase 1: Initialization, n_main={n_main}")
    
    candidates = []
    
    if parent is not None:
        p_main = _interpolate_profile(parent, n_main)
        candidates.append(p_main)
        # Perturbed parent
        for _ in range(3):
            p_pert = p_main * (1.0 + 0.05 * rng.normal(size=n_main))
            p_pert = np.maximum(p_pert, 0)
            candidates.append(_normalize(p_pert))
    
    # Standard profiles
    candidates.append(_make_linear_profile(n_main, 1.5))
    candidates.append(_make_linear_profile(n_main, 2.0))
    candidates.append(_make_linear_profile(n_main, 1.0))
    candidates.append(_make_exp_profile(n_main, 1.5))
    candidates.append(_make_exp_profile(n_main, 2.5))
    candidates.append(_make_beta_profile(n_main, 2.0, 1.0))
    candidates.append(_make_beta_profile(n_main, 3.0, 1.5))
    
    # Evaluate all candidates
    for i, a0 in enumerate(candidates):
        sc = _score(a0)
        print(f"  Candidate {i}: score={sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
    
    # Quick LP on top candidates
    scored = sorted(
        [( _score(a), i) for i, a in enumerate(candidates) if _score(a) < np.inf],
        key=lambda x: x[0]
    )
    
    for sc, idx in scored[:4]:
        if time.monotonic() > deadline - 1200:
            break
        a0 = candidates[idx]
        lp_budget = min(80.0, deadline - time.monotonic() - 1200.0)
        if lp_budget < 10:
            break
        lp_end = time.monotonic() + lp_budget
        a_lp = _lp_refine(a0, lp_end, max_iter=5000)
        s = _score(a_lp)
        print(f"  Quick LP on candidate {idx}: {sc:.8f} -> {s:.10f}")
        if s < best_score:
            best_score = s
            best_seq = a_lp.copy()
    
    print(f"Phase 1 done: best_score={best_score:.10f}")
    
    # Phase 2: Intensive LP refinement
    if best_seq is not None and time.monotonic() < deadline - 500:
        lp_budget = min(500.0, deadline - time.monotonic() - 400.0)
        if lp_budget > 30:
            lp_end = time.monotonic() + lp_budget
            print(f"Phase 2: Intensive LP, budget={lp_budget:.0f}s")
            t_phase_start = time.monotonic()
            best_before = best_score
            a_lp = _lp_refine(best_seq, lp_end, max_iter=100000)
            sc_lp = _score(a_lp)
            elapsed = time.monotonic() - t_phase_start
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  Phase 2 result: {sc_lp:.10f} (was {best_before:.10f}) in {elapsed:.0f}s")
    
    # Phase 3: Adam smooth-max polish
    if best_seq is not None and time.monotonic() < deadline - 200:
        ad_budget = min(200.0, deadline - time.monotonic() - 150.0)
        if ad_budget > 20:
            ad_end = time.monotonic() + ad_budget
            print(f"Phase 3: Adam smooth-max, budget={ad_budget:.0f}s")
            a_ad = _adam_smooth_max(best_seq, ad_end, max_iter=20000, lr=0.03)
            sc_ad = _score(a_ad)
            if sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
            print(f"  Phase 3 result: {sc_ad:.10f}")
    
    # Phase 4: Basin hopping
    if best_seq is not None and time.monotonic() < deadline - 100:
        bh_deadline = time.monotonic() + min(200.0, deadline - time.monotonic() - 50.0)
        print(f"Phase 4: Basin hopping, budget={bh_deadline - time.monotonic():.0f}s")
        a_bh = _basin_hop(best_seq, bh_deadline, rng, min_hop_time=15, max_hops=50)
        sc_bh = _score(a_bh)
        if sc_bh < best_score:
            best_score = sc_bh
            best_seq = a_bh.copy()
        print(f"  Phase 4 result: {sc_bh:.10f}")
    
    # Phase 5: Final combined polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"Phase 5: Final polish, budget={remaining:.0f}s")
        
        # Alternate LP and subgradient
        n_rounds = max(1, int(remaining // 60))
        for rnd in range(n_rounds):
            if time.monotonic() > deadline - 15:
                break
            
            # LP step
            lp_budget = min(40.0, (deadline - time.monotonic() - 15) * 0.5)
            if lp_budget > 5:
                lp_end = time.monotonic() + lp_budget
                a_lp = _lp_refine(best_seq, lp_end, max_iter=5000)
                s = _score(a_lp)
                if s < best_score:
                    best_score = s
                    best_seq = a_lp.copy()
                    print(f"    Final LP round {rnd}: {s:.10f}")
            
            # Subgradient step
            sg_budget = min(20.0, (deadline - time.monotonic() - 15) * 0.4)
            if sg_budget > 3:
                sg_end = time.monotonic() + sg_budget
                a_sg = _subgradient_polish(best_seq, sg_end, max_iter=5000)
                s = _score(a_sg)
                if s < best_score:
                    best_score = s
                    best_seq = a_sg.copy()
                    print(f"    Final SG round {rnd}: {s:.10f}")
            
            # Quick Adam step
            ad_budget = min(15.0, (deadline - time.monotonic() - 15) * 0.3)
            if ad_budget > 3:
                ad_end = time.monotonic() + ad_budget
                a_ad = _adam_smooth_max(best_seq, ad_end, max_iter=3000, lr=0.01)
                s = _score(a_ad)
                if s < best_score:
                    best_score = s
                    best_seq = a_ad.copy()
                    print(f"    Final Adam round {rnd}: {s:.10f}")
    
    # Final validation and cleanup
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)
    
    # Evaluate with the authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass
    
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
