"""
Gradient-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smooth max approximation with Adam in logit space,
FFT-based convolution for speed, and multi-start with size progression.
"""

import numpy as np
import time


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    size = 1
    while size < n_out:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n_out]


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), clipped and normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (2.0 * t - 1.0)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def make_cosine_profile(n, amp):
    """Create a cosine-shaped profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(np.pi * t)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def make_quadratic_profile(n, coeff):
    """Create a quadratic profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + coeff * u * u
    a = np.maximum(a, 0.001)
    return a / a.sum()


def make_asymmetric_cubic(n, c1, c2, c3):
    """Create an asymmetric cubic profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + c1 * u + c2 * u * u + c3 * u * u * u
    a = np.maximum(a, 0.001)
    return a / a.sum()


def interpolate_to_size(a, n_new):
    """Interpolate a sequence to a new size."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return None
    return a_new / s

        a1_opt, s1_opt = try_multiple_T(a1, t1_deadline)
        if s1_opt < best_score:
            best_score = s1_opt
            best_seq = a1_opt
            print(f"  Phase 1 result: score={best_score:.8f}")

    # --- Phase 2: Optimize at n=512 ---
    print("\nPhase 2: Optimizing at n=512...")
    n2 = 512
    if time.monotonic() < deadline - 120:
        a2 = interpolate_to_size(best_seq, n2)
        if a2 is not None:
            sc2_init = internal_score(a2)
            print(f"  Initial at n=512: score={sc2_init:.8f}")
            t2_deadline = min(deadline, t_start + 900.0)
            a2_opt, s2_opt = try_multiple_T(a2, t2_deadline)
            if s2_opt < best_score:
                best_score = s2_opt
                best_seq = a2_opt
                print(f"  Phase 2 result: score={best_score:.8f}")

    # --- Phase 3: Optimize at n=1024 ---
    print("\nPhase 3: Optimizing at n=1024...")
    n3 = 1024
    if time.monotonic() < deadline - 120:
        a3 = interpolate_to_size(best_seq, n3)
        if a3 is not None:
            sc3_init = internal_score(a3)
            print(f"  Initial at n=1024: score={sc3_init:.8f}")
            t3_deadline = min(deadline, t_start + 1500.0)
            a3_opt, s3_opt = try_multiple_T(a3, t3_deadline)
            if s3_opt < best_score:
                best_score = s3_opt
                best_seq = a3_opt
                print(f"  Phase 3 result: score={best_score:.8f}")

    # --- Phase 4: Final polish with exact subgradient ---
    print("\nPhase 4: Final subgradient polish...")
    if time.monotonic() < deadline - 10:
        a_polish, s_polish = optimize_exact_subgrad(best_seq, deadline - 5, lr=0.001)
        if s_polish < best_score:
            best_score = s_polish
            best_seq = a_polish
            print(f"  Polish result: score={best_score:.8f}")

    # --- Phase 5: Try a few more restarts if time remains ---
    if time.monotonic() < deadline - 30:
        print("\nPhase 5: Additional restarts...")
        for restart in range(3):
            if time.monotonic() > deadline - 10:
                break
            # Perturb the best sequence
            rng_r = np.random.default_rng(seed + 100 + restart)
            z_best = np.log(np.maximum(best_seq, 1e-15))
            z_best -= z_best.mean()
            z_pert = z_best + 0.05 * rng_r.normal(size=len(best_seq))
            z_pert = np.clip(z_pert, -40, 40)
            a_pert = np.exp(z_pert - z_pert.max())
            a_pert = a_pert / a_pert.sum()

            sc_pert = internal_score(a_pert)
            if sc_pert < best_score:
                best_score = sc_pert
                best_seq = a_pert
                print(f"  Restart {restart}: improved to {best_score:.8f}")

            # Quick optimization from perturbed start
            t_rem = deadline - time.monotonic() - 10
            if t_rem > 30:
                a_r, s_r = optimize_smooth(a_pert, time.monotonic() + t_rem * 0.5,
                                           T_start=50.0, T_end=400.0, lr=0.002)
                if s_r < best_score:
                    best_score = s_r
                    best_seq = a_r
                    print(f"  Restart {restart} optimized: score={best_score:.8f}")

    # --- Final validation ---
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
