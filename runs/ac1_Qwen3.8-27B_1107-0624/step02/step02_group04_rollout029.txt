Coordinate descent + gradient polish for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact 1D coordinate minimization via golden section search,
with random coordinate ordering and multi-restart strategy.
"""
import numpy as np
import time
from scipy.optimize import minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    """Compute full linear convolution a * a using FFT for large n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = _full_convolve(a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def _cd_subobjective(t, base, slope, base_2i, S_base, n):
    """1D objective for coordinate descent subproblem."""
    if t < 0.0:
        return np.inf
    S_t = S_base + t
    if S_t <= 1e-30:
        return np.inf
    if base.size > 0:
        lin_max = float(np.max(base + slope * t))
    else:
        lin_max = 0.0
    quad_val = base_2i + t * t
    max_c = max(lin_max, quad_val)
    if max_c < 0.0:
        max_c = 0.0
    return 2.0 * n * max_c / (S_t * S_t)

    Returns the score after the pass.

        k_max = min(2 * n - 2, i + n - 1)

            ks_lin = ks[mask]

        else:

    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))

        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):

def _cd_subobj(t, base, slope, base_2i, S_base, n):

"""
Coordinate descent with exact 1D subproblem solving + gradient polish
for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses scipy bounded minimization for each coordinate, random ordering,
multi-restart with perturbations, and final gradient-based polish.

from scipy.optimize import minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = _full_convolve(a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def _cd_subobj(t, base, slope, base_2i, c_const, S_base, n):
    if t < 0.0:
        return 1e10
    St = S_base + t
    if St <= 1e-30:
        return 1e10
    if base.size > 0:
        lm = float(np.max(base + slope * t))
    else:
        lm = 0.0
    qv = base_2i + t * t
    mc = lm
    if qv > mc:
        mc = qv
    if c_const > mc:
        mc = c_const
    if mc < 0.0:
        mc = 0.0
    return 2.0 * n * mc / (St * St)

def _gradient_polish(p0, deadline, rng):

    z -= z.mean()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_adam = 0

    p = p0.copy()

            v.fill(0)

def _make_profile(n, kind, eta=1.5, a1=1.0, a2=0.3, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'linear':
        a = 1.0 + eta * (t - 0.5)
    elif kind == 'quad':
        a = 1.0 + a1 * u + a2 * u * u
    elif kind == 'cubic':
        a = 1.0 + a1 * u + a3 * u * u * u
    elif kind == 'cosine':
        a = np.cos(np.pi * t) ** 2 + 0.1
    elif kind == 'triangle':
        a = 1.5 - np.abs(u)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()

    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)
