"""
Multi-stage gradient descent with softmax smoothing for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses FFT-based gradients, Adam optimizer,
adaptive temperature, and multi-start exploration across sequence lengths.
"""

import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    """Full linear convolution via FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 96:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score_norm(a):
    """Score assuming sum(a)=1: returns 2*n*max_conv."""
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    a_n = a / s
    c = fft_convolve(a_n)
    return 2.0 * n * float(c.max())


def softmax_from_z(z):
    """Convert logits to probability vector."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    return e / s


def compute_grad_z(a, beta, N_fft):
    """
    Compute gradient of LSE-smoothed max-conv w.r.t. logit space z.
    Returns (grad_z, smoothed_value).
    """
    n = a.size
    c = fft_convolve(a)
    c_max = float(c.max())
    x = (c - c_max) / beta
    x = np.clip(x, -200.0, 10.0)
    ex = np.exp(x)
    s_ex = float(ex.sum())
    if s_ex < 1e-30:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s_ex = 1.0
    w = ex / s_ex
    L = c_max + beta * float(np.log(s_ex))

    # grad_a[j] = 2 * sum_{k: 0<=k-j<2n-1} w[k] * a[k-j]
    # = 2 * (a * flip(w))[2n-2-j]
    fwr = np.fft.rfft(w[::-1], N_fft)
    fa = np.fft.rfft(a, N_fft)
    conv_aw = np.fft.irfft(fa * fwr, N_fft)
    # grad_a[j] = 2 * conv_aw[2n-2-j]
    grad_a = 2.0 * conv_aw[2 * n - 2 : n - 1 : -1]

    # Convert to logit-space gradient: grad_z[k] = a[k]*(grad_a[k] - <a,grad_a>)
    mean_grad = float(a @ grad_a)
    grad_z = a * (grad_a - mean_grad)

    return grad_z, L


def make_profile_linear(n, eta, rng=None):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.3)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_profile_exp(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(alpha * t)
    return a / a.sum()


def make_profile_quad(n, b1, b2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + b1 * t + b2 * t * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_profile_smooth_random(n, rng, n_harm=12):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Linearly interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def adam_optimize(a0, deadline, rng, lr=0.05, max_iter=100000,
                  beta_init=None, beta_decay=0.9995, verbose=False):
    """
    Adam-based optimization in logit space with adaptive softmax temperature.
    a0: initial sequence (will be normalized to sum=1).
    Returns (best_sequence, best_score).
    """
    n = a0.size
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    c = fft_convolve(a)
    M_val = float(c.max())
    if beta_init is None:
        beta = max(1e-6, 0.1 * M_val)
    else:
        beta = beta_init

    N_fft = next_pow2(3 * n)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a.copy()
    best_score = 2.0 * n * M_val
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad_z, L = compute_grad_z(a, beta, N_fft)

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_a)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        a = softmax_from_z(z)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = fft_convolve(a)
        M_new = float(c.max())
        sc = 2.0 * n * M_new

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

            beta = max(1e-8, beta * beta_decay)

        # Escape mechanism

            a = softmax_from_z(z)

            a = a / a.sum()
            c = fft_convolve(a)
            M_val = float(c.max())
            beta = max(1e-8, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

        if verbose and it % 500 == 0:
            print(f"    Adam iter {it}: score={sc:.10f}, best={best_score:.10f}, beta={beta:.6f}")

    return best_a, best_score


def lr_schedule_optimize(a0, deadline, rng, max_iter=100000, verbose=False):
    """
    Gradient descent with cosine learning rate schedule and adaptive beta.
    Often more stable than Adam for this problem.
    """
    n = a0.size
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    c = fft_convolve(a)
    M_val = float(c.max())
    beta = max(1e-6, 0.05 * M_val)

    N_fft = next_pow2(3 * n)

    best_a = a.copy()
    best_score = 2.0 * n * M_val
    no_imp = 0

    lr_init = 0.1
    lr_min = 0.001
    momentum = 0.9
    v_mom = np.zeros(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Cosine schedule
        progress = min(1.0, it / max(1, max_iter * 0.8))
        lr = lr_min + 0.5 * (lr_init - lr_min) * (1.0 + np.cos(np.pi * progress))

        grad_z, L = compute_grad_z(a, beta, N_fft)

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_a)
            z -= z.mean()
            v_mom.fill(0)
            continue

        # Momentum
        v_mom = momentum * v_mom + (1.0 - momentum) * grad_z
        z = z - lr * v_mom
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        a = softmax_from_z(z)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = fft_convolve(a)
        M_new = float(c.max())
        sc = 2.0 * n * M_new

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Decay beta
        if it % 300 == 0:
            beta = max(1e-8, beta * 0.95)

        # Escape
        if no_imp > max(400, 4 * n):
            z = 0.7 * np.log(best_a) + 0.4 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            a = softmax_from_z(z)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            c = fft_convolve(a)
            beta = max(1e-8, 0.1 * float(c.max()))
            v_mom.fill(0)
            no_imp = 0

        if verbose and it % 500 == 0:
            print(f"    LR iter {it}: score={sc:.10f}, best={best_score:.10f}, lr={lr:.6f}")

    return best_a, best_score

    # Parent interpolated to 512
    if parent is not None:
        p_int = interpolate_profile(parent, n_main)
        if p_int is not None:
            starts_512.append(p_int)

    # Deterministic profiles
    starts_512.append(make_profile_linear(n_main, 0.8))
    starts_512.append(make_profile_linear(n_main, 1.2))
    starts_512.append(make_profile_linear(n_main, 1.8))
    starts_512.append(make_profile_linear(n_main, 2.5))
    starts_512.append(make_profile_exp(n_main, 1.5))
    starts_512.append(make_profile_exp(n_main, 2.5))
    starts_512.append(make_profile_exp(n_main, 4.0))
    starts_512.append(make_profile_quad(n_main, 1.5, -1.0))
    starts_512.append(make_profile_quad(n_main, 2.0, -1.5))
    starts_512.append(make_profile_quad(n_main, -0.5, 2.0))

    # Random smooth profiles
    for _ in range(10):
        starts_512.append(make_profile_smooth_random(n_main, rng, n_harm=10))

    # --- Phase 1: Quick optimization from all starts at n=512 ---
    phase1_budget = min(deadline, t_start + 360.0)
    per_start = max(3.0, (phase1_budget - t_start) / len(starts_512))

    results_512 = []
    for i, p0 in enumerate(starts_512):
        if time.monotonic() > phase1_budget:
            break
        p0 = np.maximum(p0, 1e-15)
        p0 = p0 / p0.sum()

        sc0 = compute_score_norm(p0)
        per = min(phase1_budget, time.monotonic() + per_start)
        p_best, sc_best = adam_optimize(p0, per, rng, lr=0.06, max_iter=30000)
        try_update(p_best, sc_best)
        results_512.append((sc_best, p_best))
        if i < 5 or i % 5 == 0:
            print(f"  P1 start {i:2d}/{len(starts_512)}: init={sc0:.8f}, best={sc_best:.8f}, global={best_score:.8f}")

    results_512.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}, top result={results_512[0][0]:.10f}")

    # --- Phase 2: Deep optimization from top candidates at n=512 ---
    phase2_end = min(deadline, t_start + 800.0)
    top_k = min(4, len(results_512))
    per_deep = max(20.0, (phase2_end - time.monotonic()) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase2_end - 30:
            break
        _, p0 = results_512[i]
        per = min(phase2_end - 10, time.monotonic() + per_deep)
        p_best, sc_best = two_phase_optimize(p0, per, rng, verbose=(i == 0))
        try_update(p_best, sc_best)
        print(f"  P2 deep {i}: score={sc_best:.10f}, global={best_score:.10f}")

    # Also try LR-based optimization on the best
    if best_seq is not None and time.monotonic() < phase2_end - 30:
        per = min(phase2_end - 10, time.monotonic() + 60.0)
        p_best, sc_best = lr_schedule_optimize(best_seq, per, rng, max_iter=80000)
        try_update(p_best, sc_best)
        print(f"  P2 LR: score={sc_best:.10f}, global={best_score:.10f}")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Try n=1024 ---
    n2 = 1024
    phase3_end = min(deadline, t_start + 1300.0)
    if best_seq is not None and time.monotonic() < phase3_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase3_end - 30, time.monotonic() + 120.0)
            p_best, sc_best = two_phase_optimize(p_ext, per, rng, verbose=True)
            try_update(p_best, sc_best)
            print(f"  P3 n=1024: score={sc_best:.10f}, global={best_score:.10f}")

    # Try a couple of fresh starts at n=1024
    if time.monotonic() < phase3_end - 60:
        for eta in (1.0, 2.0):
            if time.monotonic() > phase3_end - 60:
                break
            p0 = make_profile_exp(n2, eta)
            per = min(phase3_end - 30, time.monotonic() + 50.0)
            p_best, sc_best = adam_optimize(p0, per, rng, lr=0.05, max_iter=30000)
            try_update(p_best, sc_best)
            print(f"  P3 n=1024 exp({eta}): score={sc_best:.10f}")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Try n=768 (intermediate) ---
    n3 = 768
    phase4_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p_ext = interpolate_profile(best_seq, n3)
        if p_ext is not None:
            per = min(phase4_end - 30, time.monotonic() + 80.0)
            p_best, sc_best = two_phase_optimize(p_ext, per, rng)
            try_update(p_best, sc_best)
            print(f"  P4 n=768: score={sc_best:.10f}, global={best_score:.10f}")

    # --- Phase 5: Final polishing on the absolute best ---
    phase5_end = min(deadline, t_start + 1750.0)
    if best_seq is not None and time.monotonic() < phase5_end - 10:
        n_best = best_seq.size
        per = min(phase5_end - 5, time.monotonic() + 100.0)

        # Multiple polishing strategies
        c = fft_convolve(best_seq / best_seq.sum())
        M_val = float(c.max())
        p_best, sc_best = adam_optimize(best_seq, per, rng, lr=0.01,
                                        beta_init=max(1e-9, 0.005 * M_val),
                                        beta_decay=0.9999, max_iter=200000)
        try_update(p_best, sc_best)

        # Try LR method too
        if time.monotonic() < phase5_end - 20:
            per2 = min(phase5_end - 5, time.monotonic() + 50.0)
            p_best2, sc_best2 = lr_schedule_optimize(best_seq, per2, rng, max_iter=100000)
            try_update(p_best2, sc_best2)

        print(f"Phase 5 done: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # --- Final evaluation ---
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s < 1e-30:
            final = make_profile_linear(512, 1.5)
        else:
            final = final / s
        sc_final = compute_score_norm(final)
        print(f"Final: score={sc_final:.10f}, n={final.size}")

        # Verify with evaluate_sequence if available
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"Eval check: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_profile_linear(512, 1.5)
    return [float(x) for x in fb]
