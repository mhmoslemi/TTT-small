    # M2[i,k] = w[i-k] if i >= k, else 0
    idx2 = np.arange(n)[:, None] - np.arange(n)[None, :]
    M2 = np.where(idx2 >= 0, w[idx2], 0.0)
    h2 = M2 @ a
    
    return h1 + h2

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def compute_grad(z, tau):

    n = len(a)
