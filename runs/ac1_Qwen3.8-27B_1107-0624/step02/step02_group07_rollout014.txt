**Coordinate descent for min max_k c[k] subject to sum s = 1, s >= 0:**

Step: Pick coordinate i. Fix all s[j] for j ≠ i. Find s[i] >= 0 that minimizes max_k c[k].

c[0] = C0 + s[i]^2 where C0 = sum_{j≠i} s[j]^2
c[k] = A_k + s[i] * B_k for k ≠ 0 (linear in s[i])

We want to find s[i] >= 0 minimizing max(C0 + s[i]^2, max_{k≠0}(A_k + s[i]*B_k)).

Let f(t) = max(C0 + t^2, max_{k≠0}(A_k + t*B_k)) for t >= 0.

f is convex (max of convex functions). We can minimize it with golden section search or ternary search in O(n) per evaluation.

But we need to maintain sum(s) = 1. If we change s[i], the sum changes. Two options:
(a) Don't maintain the constraint; just optimize the score (which is scale-invariant).
(b) Do pairwise updates to maintain the constraint.

For option (a): After updating s[i], renormalize. This is valid because the score is scale-invariant.

For option (b): Pick pair (i, j). Fix s[i] + s[j] = C. Let s[i] = x, s[j] = C - x.
Then optimize over x in [0, C].

I'll go with option (a) for simplicity.

The computational cost per coordinate update:
- Computing A_k and B_k: O(n) if we maintain conv incrementally, or O(n^2) if we recompute.
- Actually, we can maintain conv and update it incrementally: when s[i] changes from old_val to new_val, the change in conv[k] is (new_val - old_val) * (s[k-i] + s[i-k]) + (new_val^2 - old_val^2) * δ_{k,0}.
- Then we need to find the optimal s[i] by searching over t >= 0, evaluating f(t) in O(n) each time.
- With golden section: ~50 evaluations per coordinate.
- Total per sweep: n * 50 * n = 50*n^2. For n=512: ~13 million. Very fast!

def lse_score_and_grad(s, beta):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf'), None
    conv = np.convolve(s, s)
    c_max = conv.max()
    w = np.exp(beta * (conv - c_max))
    w /= w.sum()
    lse_val = c_max + np.log(w.sum()) / beta  # = c_max + log(sum(exp(beta*(conv-c_max))))/beta

    # d lse / d s[i] = sum_k w[k] * d conv[k] / d s[i]
    # d conv[k] / d s[i] = s[k-i] + s[i-k] (with s[m]=0 for m out of [0,n-1])

    # grad[i] = sum_k w[k] * (s[k-i] + s[i-k])
    # = sum_k w[k]*s[k-i] + sum_k w[k]*s[i-k]
    # First term: cross-correlation of w and s
    # Second term: convolution of w and s (reversed)
    
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    W = np.fft.rfft(w, N_fft)
    term1 = np.fft.irfft(np.conj(S) * W, N_fft)[0:n]  # cross-correlation
    term2 = np.fft.irfft(S * W, N_fft)[0:n]  # convolution
    grad_lse = term1 + term2
    
    score = 2.0 * n * lse_val / (total * total)
    grad = 2.0 * n * (grad_lse * total - 2.0 * lse_val) / (total ** 3)
    return score, grad

import numpy as np
from scipy.optimize import minimize, linprog
import time

"""
Hybrid optimizer: coordinate descent + L-BFGS-B with log-sum-exp smoothing
for minimizing 2*n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
Uses multiple initializations, sequence lengths, and alternating between
fast coordinate descent (exact 1D subproblems) and quasi-Newton refinement.
"""


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return float(2.0 * n * np.max(conv) / (total * total))


def lse_score_and_grad(s, beta):
    """Score and gradient using log-sum-exp smoothing of max(conv)."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf'), None
    conv = np.convolve(s, s)
    c_max = conv.max()
    if c_max < 1e-15:
        return 0.0, np.zeros(n)
    w = np.exp(beta * (conv - c_max))
    w_sum = w.sum()
    w = w / w_sum
    lse_val = c_max + np.log(w_sum) / beta

    # Gradient of lse_val w.r.t. s via FFT
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    W = np.fft.rfft(w, N_fft)
    term1 = np.fft.irfft(np.conj(S) * W, N_fft)[0:n]
    term2 = np.fft.irfft(S * W, N_fft)[0:n]
    grad_lse = term1 + term2

    score = 2.0 * n * lse_val / (total * total)
    grad = 2.0 * n * (grad_lse * total - 2.0 * lse_val) / (total ** 3)
    return score, grad


def coord_descent_sweep(s, deadline=None):
    """One sweep of coordinate descent. Returns updated s."""
    n = len(s)
    s = np.maximum(s, 0.0).astype(np.float64)
    total = s.sum()
    if total < 1e-15:
        return s
    s = s / total

    for i in range(n):
        if deadline is not None and time.monotonic() > deadline:
            break

        old = s[i]
        conv = np.convolve(s, s)
        ks = np.arange(2 * n - 1)

        # Compute contribution of s[i] to conv
        # For k where 0 <= k-i < n: contribution is 2*s[i]*s[k-i]
        # Except at k = 2i where it's s[i]^2
        valid = (ks >= i) & (ks < i + n)
        B = np.zeros(2 * n - 1)
        B[valid] = s[ks[valid] - i]

        contribution = 2.0 * old * B
        if 2 * i < 2 * n - 1:
            contribution[2 * i] = old * old

        conv_no_i = conv - contribution

        # Minimize f(t) = max_k (conv_no_i[k] + 2*t*B[k]) with t^2 at k=2i
        # Subject to t >= 0, also considering the scale-invariant score
        # We minimize f(t) / (total + t - old)^2 which equals the score (up to 2n factor)

        t_max = max(old * 4.0 + 2.0 / n, 5.0 / n)

        def f_of_t(t):
            delta = 2.0 * t * B
            if 2 * i < 2 * n - 1:
                delta = delta.copy()
                delta[2 * i] = t * t
            return np.max(conv_no_i + delta)

        # Golden section on t in [0, t_max]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = 0.0, t_max
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        fc = f_of_t(c) / (total + c - old) ** 2
        fd = f_of_t(d) / (total + d - old) ** 2

        for _ in range(40):
            if fc < fd:
                b = d
                d = c
                fd = fc
                c = b - gr * (b - a)
                fc = f_of_t(c) / (total + c - old) ** 2
            else:
                a = c
                c = d
                fc = fd
                d = a + gr * (b - a)
                fd = f_of_t(d) / (total + d - old) ** 2

        t_opt = max(0.0, (a + b) / 2.0)
        s[i] = t_opt
        total = s.sum()
        if total < 1e-15:
            # Fallback
            s[i] = old
            total = s.sum()
        s = s / total

    return s


def coordinate_descent(s, num_sweeps, deadline=None):
    """Run coordinate descent for num_sweeps iterations."""
    n = len(s)
    s = np.maximum(s.copy(), 0.0).astype(np.float64)
    total = s.sum()
    if total < 1e-15:
        return s
    s = s / total

    best_s = s.copy()
    best_score = fast_score(s)
    no_improve = 0
    max_no_improve = 50

    for sweep in range(num_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        s = coord_descent_sweep(s, deadline)
        score = fast_score(s)

        if score < best_score - 1e-14:
            best_score = score
            best_s = s.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            break

    return best_s


def lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M, b >= 0."""
    n = len(s)
    if n > 400:
        return None
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def gradient_descent(s, deadline, rng, use_lp=True):

    n = len(s)

    best_s = s.copy()
    best_score = fast_score(s)

    beta = 8.0
    no_improve = 0
    patience = 200

        conv = np.convolve(s, s)
        max_conv = conv.max()

        w = np.exp(beta * (conv - max_conv) / max_conv)
        w = w / w.sum()

        N_fft = 1
        while N_fft < 4 * n:
            N_fft *= 2
        S = np.fft.rfft(s, N_fft)
        W = np.fft.rfft(w, N_fft)
        term1 = np.fft.irfft(np.conj(S) * W, N_fft)[0:n]
        term2 = np.fft.irfft(S * W, N_fft)[0:n]
        grad = term1 + term2
        grad = np.maximum(grad, 0.0)
        grad_max = grad.max()
        if grad_max < 1e-15:
            break

        s_new = np.maximum(s_new, 1e-15)
        s_new = s_new / s_new.sum()
        score_new = fast_score(s_new)

        if score_new < best_score - 1e-13:

        else:
            no_improve += 1

        if score_new < cur_score:
            s = s_new
        else:
            # Tentative step with smaller eta
            s_try = s * np.exp(-eta * 0.5 * grad / grad_max)
            s_try = np.maximum(s_try, 1e-15)
            s_try = s_try / s_try.sum()
            score_try = fast_score(s_try)
            if score_try < cur_score:
                s = s_try
            # Adjust parameters

                    s = np.maximum(s, 0.01)
                    s = s / s.sum()
                    eta = 0.2
                    beta = 8.0

        # Occasional LP step

            b = lp_direction(s, max_conv * 1.01, time_limit=2.0)

                        best_t = t
                if best_t > 0 and best_ts < best_score:
                    s = (1 - best_t) * s + best_t * b
                    s = np.maximum(s, 0.0) / s.sum()
                    best_score = best_ts
                    best_s = s.copy()

        # Periodically increase beta
        if it % 2000 == 1999:

    # Resized previous best
    if hseq is not None and hseq.size > 5:
        h_r = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, hseq.size), hseq)
        h_r = np.maximum(h_r, 0.0)
        if h_r.sum() > 0.01:
            h_r = h_r / h_r.sum()
            candidates.append(h_r)

    # Symmetric patterns
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0]:
        s = np.cos(alpha * np.pi * x / 2.0) ** 2 + 0.05
        s = s / s.sum()
        candidates.append(s.copy())

    # Two-level
    for frac in [0.3, 0.5, 0.7]:
        k = max(1, int(frac * n))
        for ratio in [0.3, 0.5, 2.0, 3.0]:
            s = np.ones(n)
            s[:k] = ratio
            s = s / s.sum()
            candidates.append(s.copy())

    # Gaussian
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.3, 0.5]:
            s = np.exp(-(x - center) ** 2 / (2 * width ** 2)) + 0.02
            s = s / s.sum()
            candidates.append(s.copy())

    # Random
    for _ in range(4):
        s = rng.random(n) + 0.1
        s = s / s.sum()
        candidates.append(s)

    # Triangular
    s = 1.0 - np.abs(2.0 * x - 1.0)
    s = s + 0.05
    s = s / s.sum()
    candidates.append(s)

    return candidates


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(s,s))/(sum(s))^2.
    Uses coordinate descent (exact 1D subproblems), FFT gradient descent,
    LP-guided steps, and L-BFGS-B refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0

    global_best_score = float('inf')
    global_best_seq = np.ones(256) / 256.0

    # Load previous best
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            h_norm = h / h.sum()
            print(f"Loaded height_sequence_1, n={h.size}, score={fast_score(h_norm):.6f}")
    except Exception:
        pass

    lengths = [192, 256, 320, 384, 448, 512]

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        print(f"\n=== n={n}, remaining={remaining:.0f}s ===")

        candidates = generate_candidates(n, rng, hseq)

        # Score and sort candidates
        scored = []
        for c in candidates:
            c = np.maximum(c, 0.0)
            if c.sum() < 1e-10:
                continue
            c = c / c.sum()
            sc = fast_score(c)
            scored.append((sc, c))
        scored.sort(key=lambda x: x[0])

        # Try top candidates with coordinate descent
        num_cands = min(5, len(scored))
        time_per_cand = max(20.0, (remaining - 120) / num_cands)

        for i in range(num_cands):
            if time.monotonic() > deadline - 60:
                break
            sc0, s0 = scored[i]
            print(f"  Cand {i}: score={sc0:.6f}")

            cand_deadline = time.monotonic() + time_per_cand

            # Phase 1: Coordinate descent
            s_cd = coordinate_descent(s0, num_sweeps=3000, deadline=cand_deadline)
            sc_cd = fast_score(s_cd)

            # Phase 2: Gradient descent if time remains
            if time.monotonic() < deadline - 60:
                s_gd, sc_gd = gradient_descent(s_cd, deadline - 60, rng, use_lp=(n <= 350))
                if sc_gd < sc_cd:
                    s_cd = s_gd
                    sc_cd = sc_gd

            # Phase 3: More coordinate descent
            if time.monotonic() < deadline - 60:
                s_cd = coordinate_descent(s_cd, num_sweeps=3000, deadline=deadline - 60)
                sc_cd = fast_score(s_cd)

            if sc_cd < global_best_score:
                global_best_score = sc_cd
                global_best_seq = s_cd.copy()
                print(f"  ** New best: {sc_cd:.6f} (n={n}, cand={i})")

    # Final refinement on the best found
    if time.monotonic() < deadline - 30:
        print(f"\n=== Final refinement, score={global_best_score:.6f} ===")
        n = len(global_best_seq)
        s_ref = global_best_seq.copy()

        # L-BFGS-B with increasing beta
        for beta in [20.0, 50.0, 100.0, 200.0]:
            if time.monotonic() > deadline - 20:
                break
            try:
                res = minimize(
                    fun=lambda x: lse_score_and_grad(x, beta)[0],
                    x0=s_ref,
                    jac=lambda x: lse_score_and_grad(x, beta)[1],
                    method='L-BFGS-B',
                    bounds=[(0.0, None)] * n,
                    options={'maxiter': 300, 'factr': 1e12, 'gtol': 1e-14}
                )
                s_try = np.maximum(res.x, 0.0)
                if s_try.sum() > 1e-10:
                    s_try = s_try / s_try.sum()
                    sc_try = fast_score(s_try)
                    if sc_try < global_best_score:
                        global_best_score = sc_try
                        global_best_seq = s_try.copy()
                        s_ref = s_try.copy()
                        print(f"  L-BFGS-B (beta={beta}): {sc_try:.6f}")
            except Exception as e:
                print(f"  L-BFGS-B error: {e}")

        # Final coordinate descent polish
        if time.monotonic() < deadline - 10:
            s_final = coordinate_descent(global_best_seq, num_sweeps=5000, deadline=deadline - 5)
            sc_final = fast_score(s_final)
            if sc_final < global_best_score:
                global_best_score = sc_final
                global_best_seq = s_final.copy()
                print(f"  Final CD: {sc_final:.6f}")

    # Verify with evaluator
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_score = float(evaluate_sequence(result))
    print(f"\nFinal verified score: {final_score:.6f}, n={len(result)}")

    return result
