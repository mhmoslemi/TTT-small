"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines LP-based cross-convolution constraint optimization, subgradient Adam,
soft-max Adam with aggressive annealing, and coordinate descent polish.
Exploits mirror symmetry and multi-length search (128-1024).
"""
import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def cross_convolve_fft(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e) / float(e.size)
    return e / s


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def subgrad_of_max_conv(a, k_star):
    """Compute d(conv[k_star])/d(a[j]) = 2*a[k_star - j] if 0<=k_star-j<n else 0."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    return g


def make_profiles(n, rng):
    """Generate diverse initial profiles for length n."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profs = []
    # Linear tilts (various slopes)
    for eta in (-3.0, -2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0, 3.0):
        a = 1.0 + eta * t
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    # Exponential
    for eta in (-4.0, -2.0, -1.0, 1.0, 2.0, 4.0):
        a = np.exp(eta * t)
        profs.append(a / a.sum())
    # Power
    for p in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0, 3.0):
        a = np.power(np.maximum(t, 1e-10), p)
        profs.append(a / a.sum())
    # U-shaped (high at edges)
    for alpha in (0.5, 1.0, 2.0, 3.0):
        a = np.power(t * (1 - t) + 1e-10, -alpha)
        profs.append(a / a.sum())
    # Sigmoid
    for eta in (5.0, 8.0, 12.0, 20.0):
        for center in (0.3, 0.5, 0.7):
            a = 1.0 / (1.0 + np.exp(-eta * (t - center)))
            profs.append(a / a.sum())
    # Cubic
    for a1 in (-2.0, -1.0, 1.0, 2.0):
        for a3 in (-1.0, -0.5, 0.5, 1.0):
            a = 1.0 + a1 * u + a3 * u ** 3
            a = np.maximum(a, 1e-10)
            profs.append(a / a.sum())
    # Random smooth (Fourier)
    for _ in range(20):
        coeffs = rng.normal(0, 1, min(10, n // 4))
        a = 1.0
        for m, cf in enumerate(coeffs, 1):
            a = a + cf * np.cos(2 * np.pi * m * t)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    return profs


def lp_cross_step(a, M, damping=1.0, method='highs'):
    """LP step: find b maximizing sum(b) s.t. cross_conv(a,b) <= M, b >= 0.
    Returns damped update (1-d)*a + d*b/sum(b)."""
    n = len(a)
    if n > 2048:
        return a.copy()
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    c_obj = -np.ones(n)
    b_ub_arr = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub_arr, bounds=bounds, method=method)
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            s = float(b.sum())
            if s > 1e-15:
                b = b / s
                return (1.0 - damping) * a + damping * b
    except Exception:
        pass
    return a.copy()


def soft_max_adam(p0, deadline, rng, lr=0.12, reg=1e-7, beta_init_frac=0.3,
                  use_subgrad_every=10):
    """Adam optimizer in logit space with soft-max + periodic subgradient steps."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    step_count = 0

    while time.monotonic() < deadline:
        step_count += 1
        if step_count % use_subgrad_every == 0:
            # Subgradient step: use exact max
            k_star = int(np.argmax(c))
            g = subgrad_of_max_conv(p, k_star)
            vp = g + reg * p
        else:
            L, w = soft_mx(c, beta)
            gp = 2.0 * corr_weights(w, p)
            vp = gp + reg * p

        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m_adam = b1 * m_adam + (1.0 - b1) * gz
        v_adam = b2 * v_adam + (1.0 - b2) * (gz * gz)
        mh = m_adam / (1.0 - b1 ** adam_t)
        vh = v_adam / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            continue

        # Aggressive beta annealing
        if adam_t % 40 == 0:
            beta = max(1e-15, beta * 0.6)

        if no_imp > max(400, 5 * n):
            # Restart with perturbation
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, beta_init_frac * maxc)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def lp_optimize(p0, deadline, max_iters=5000, damping=1.0):
    """Iterative LP-based optimization: repeatedly solve LP and update."""
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = full_convolve_fft(p)
        M = float(c.max())
        if not np.isfinite(M) or M <= 0:
            break
        p_new = lp_cross_step(p, M, damping=damping)
        if p_new is None or not np.all(np.isfinite(p_new)):
            break
        p_new = np.maximum(p_new, 1e-15)
        s = float(p_new.sum())
        if s < 1e-15:
            break
        p_new = p_new / s
        sc = internal_score(p_new)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_p = p_new.copy()
            p = p_new
            no_imp = 0
        else:
            p = p_new
            no_imp += 1
            if no_imp > 50:
                # Try smaller damping
                p = p0.copy() if no_imp > 200 else p
                damping = max(0.3, damping * 0.8)
                if no_imp > 500:
                    break

    return best_p, best_score


def coordinate_descent_polish(p, deadline, rng, n_sweeps=None, step_frac=0.1):
    """Coordinate descent: try scaling individual elements."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    if n_sweeps is None:
        n_sweeps = max(3, int((deadline - time.monotonic()) * 3))

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved_this_sweep = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            if orig < 1e-14:
                # Try setting it to a small value
                for factor in (1e-3, 0.01):
                    p[i] = factor * float(np.mean(p[np.concatenate([p > 1e-14, [1.0]])]) if np.any(p > 1e-14) else 1.0)
                    s = p.sum()
                    if s > 0.01:
                        p = p / s
                        sc = internal_score(p)
                        if np.isfinite(sc) and sc < best_score:
                            best_score = sc
                            best_p = p.copy()
                            improved_this_sweep = True
                            no_imp = 0
                            orig = p[i]
                            break
                p[i] = orig
                p = p / p.sum()
                continue
            factors = [1.0 + step_frac, 1.0 + 0.5 * step_frac, 1.0 + 2.0 * step_frac,
                       1.0 - step_frac, 1.0 - 0.5 * step_frac, 1.0 - 0.3 * step_frac]
            best_val = orig
            best_sc = best_score
            for factor in factors:
                new_val = orig * factor
                if new_val < 1e-15:
                    continue
                p[i] = new_val
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                p = p / s
                sc = internal_score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score - 1e-14:
                best_score = best_sc
                best_p = p.copy()
                improved_this_sweep = True
                no_imp = 0
            else:
                no_imp += 1
        if improved_this_sweep:
            step_frac = max(0.02, step_frac * 0.92)
        if no_imp > 3 * n:
            break

    return best_p, best_score


def mirror_improve(p):
    """Try using mirror (reverse) and average with mirror."""
    p_rev = p[::-1].copy()
    p_avg = (p + p_rev) / 2.0
    candidates = [p, p_rev, p_avg]
    best = p.copy()
    best_sc = internal_score(p)
    for c in candidates:
        c = np.maximum(c, 1e-15)
        c = c / c.sum()
        sc = internal_score(c)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = c
    return best, best_sc


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))

            except Exception:
                pass

        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s
        print(f"  FINAL: score={final_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = make_profiles(256, rng)[5]
    print(f"  FALLBACK: score={internal_score(fallback):.10f}")
    return [float(x) for x in fallback]
