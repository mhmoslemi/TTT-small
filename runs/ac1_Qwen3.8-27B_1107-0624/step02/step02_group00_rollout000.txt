```python
def _line_search_blend(a, g, n_evals=100):
    """Find t in [0,1] that minimizes 2n*max_conv((1-t)*a + t*g)"""
    # Precompute convolutions
    aa = _fft_convolve(a)
    ag = _fft_convolve(a, g)  # cross convolution
    gg = _fft_convolve(g)
    
    # c_k(t) = (1-t)^2 * aa_k + 2t(1-t) * ag_k + t^2 * gg_k
    #       = (aa_k - 2*ag_k + gg_k) * t^2 + 2*(ag_k - aa_k) * t + aa_k
    
    A = gg - 2*ag + aa  # coefficient of t^2
    B = 2*(ag - aa)     # coefficient of t
    C = aa              # constant term
    
    def f(t):
        return float(np.max(A * t*t + B * t + C))
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2
    a_lo, a_hi = 0.0, 1.0
    b_pt = a_hi - gr * (a_hi - a_lo)
    c_pt = a_lo + gr * (a_hi - a_lo)
    f_b = f(b_pt)
    f_c = f(c_pt)
    
    for _ in range(n_evals):
        if f_b < f_c:
            a_hi = c_pt
            c_pt = b_pt
            f_c = f_b
            b_pt = a_hi - gr * (a_hi - a_lo)
            f_b = f(b_pt)
        else:
            a_lo = b_pt
            b_pt = c_pt
            f_b = f_c
            c_pt = a_lo + gr * (a_hi - a_lo)
            f_c = f(c_pt)
    
    t_opt = (a_lo + a_hi) / 2
    return t_opt
```

def _fft_cross_convolve(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

"""
LP + Golden-Section Line Search optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key improvements over previous approach:
1. LP-based direction finding with exact golden-section line search (vs fixed blend)
2. Multi-scale initialization (optimize at small n, interpolate up)
3. Adaptive stagnation handling with structured perturbations
4. Adam log-space polish as final refinement
5. Multiple sequence lengths tried

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _fft_cross_convolve(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    if n <= 64:
        return np.convolve(a, b)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _golden_section_blend(a, g, n_iter=80):
    """Find optimal t in [0,1] minimizing max_conv((1-t)*a + t*g).
    Both a and g should be normalized to sum=1."""
    aa = _fft_convolve(a)
    ag = _fft_cross_convolve(a, g)
    gg = _fft_convolve(g)

    A_coeff = gg - 2.0 * ag + aa
    B_coeff = 2.0 * (ag - aa)
    C_coeff = aa

    def f_val(t):
        return float(np.max(A_coeff * t * t + B_coeff * t + C_coeff))

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = 1.0
    b_pt = a_hi - gr * (a_hi - a_lo)
    c_pt = a_lo + gr * (a_hi - a_lo)
    f_b = f_val(b_pt)
    f_c = f_val(c_pt)

    for _ in range(n_iter):
        if f_b < f_c:
            a_hi = c_pt
            c_pt = b_pt
            f_c = f_b
            b_pt = a_hi - gr * (a_hi - a_lo)
            f_b = f_val(b_pt)
        else:
            a_lo = b_pt
            b_pt = c_pt
            f_b = f_c
            c_pt = a_lo + gr * (a_hi - a_lo)
            f_c = f_val(c_pt)

    t_opt = (a_lo + a_hi) / 2.0
    return t_opt


def _lp_refine_line_search(a, deadline, max_iter=50000):
    """LP-based refinement with golden section line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    inflate = 1.002
    no_imp = 0
    inflate_base = 1.002

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _lp_spread(a, M_lp)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
        else:
            g = g / sum_g
            t_opt = _golden_section_blend(a, g, n_iter=60)

            if t_opt < 1e-6 or t_opt > 1.0 - 1e-6:
                no_imp += 1
            else:
                a_new = (1.0 - t_opt) * a + t_opt * g
                a_new = np.maximum(a_new, 0.0)
                s_new = float(a_new.sum())
                if s_new < 1e-10:
                    no_imp += 1
                else:
                    a_new = a_new / s_new
                    sc_new = _score(a_new)

                    if sc_new < best_score - 1e-12:
                        best_score = sc_new
                        best_a = a_new.copy()
                        no_imp = 0
                        a = a_new
                        # Slightly reduce inflate on success
                        inflate = max(inflate_base, inflate * 0.99)
                    else:
                        no_imp += 1
                        a = a_new  # still accept, might help later

        if no_imp > 0:
            if no_imp % 50 == 0:
                inflate = min(1.02, inflate * 1.05)
            if no_imp > 500:
                # Perturb and restart
                rng = np.random.default_rng(hash((it, n)) % (2**31))
                a = best_a * (1.0 + 0.05 * rng.normal(size=n))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                inflate = inflate_base
                no_imp = 0
            if no_imp > 5000:
                break

    return best_a


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _adam_logit_polish(a, deadline, max_iter=30000, lr=0.03):
    """Adam optimization in logit space with smooth max annealing."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    best_score = _score(a)

    c = _fft_convolve(a)
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.93)

        if it % 300 == 0:
            beta = max(1e-10, beta * 0.6)

        if no_imp > max(5000, 8 * n):
            rng = np.random.default_rng(it * 7919)
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _make_gaussian_profile(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    sigma = sigma_frac
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _quick_adam_step(a, deadline, max_iter=2000, lr=0.05):
    """Quick Adam polish for initialization."""
    a = np.asarray(a, dtype=np.float64).copy()

        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]

        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
