    if s_new_sum < 1e-15:

    """Compute full convolution of s with itself using FFT for efficiency."""
    n = len(s)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    return np.fft.irfft(np.abs(np.fft.rfft(s, N))**2, N)[:2 * n - 1]

def compute_score(s):
    """Compute 2n * max(conv(s,s)) / (sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = compute_convolution(s)
    return 2.0 * n * conv.max() / (total * total)

def compute_gradient(s, conv, beta):
    """Compute subgradient of max_k conv[k] using soft weighting."""
    n = len(s)
    max_conv = conv.max()
    if max_conv <= 1e-15:
        return np.zeros(n)

    w = np.exp(beta * (conv - max_conv) / max_conv)

    return 2.0 * grad
