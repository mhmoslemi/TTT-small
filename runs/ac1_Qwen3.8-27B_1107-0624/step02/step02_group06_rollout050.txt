"""
Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses tapered window function initialization, targeted peak reduction,

import numpy as np

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

def normalized_score(a):
    """Score assuming sum(a)=1, so just 2*n*max(conv)"""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval)

# Window functions
def hann_window(n):
    w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1))) if n > 1 else np.array([1.0])
    return w / w.sum()

def raised_cosine_window(n, beta=0.5):
    # Generalized raised cosine: 1 in the middle, tapered at edges
    t = np.linspace(0, 1, n)
    w = np.ones(n)
    edge = int(beta * n / 2)
    if edge > 0:
        x = np.arange(edge) / edge
        taper = 0.5 * (1 + np.cos(np.pi * x))
        w[:edge] = taper
        w[-edge:] = taper[::-1]
    return w / w.sum()

def gaussian_window(n, sigma_frac=0.3):
    t = np.linspace(0, 1, n)
    sigma = sigma_frac * 0.5
    w = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    return w / w.sum()

def triangle_window(n):
    t = np.linspace(0, 1, n)
    w = 1 - np.abs(2 * t - 1)
    w = np.maximum(w, 0)
    if w.sum() < 1e-10:
        w = np.ones(n)
    return w / w.sum()

def binomial_window(n):
    from math import comb
    if n <= 1:
        return np.array([1.0])
    k = n - 1
    w = np.array([comb(k, i) for i in range(n)], dtype=float)
    return w / w.sum()

def make_initial_profiles(n):
    """Generate diverse initial profiles, all tapered at edges."""
    profiles = []
    profiles.append(hann_window(n))
    profiles.append(raised_cosine_window(n, 0.3))
    profiles.append(raised_cosine_window(n, 0.5))
    profiles.append(raised_cosine_window(n, 0.7))
    profiles.append(gaussian_window(n, 0.2))
    profiles.append(gaussian_window(n, 0.3))
    profiles.append(gaussian_window(n, 0.5))
    profiles.append(triangle_window(n))
    if n <= 200:
        profiles.append(binomial_window(n))
    # Asymmetric variants
    t = np.linspace(0, 1, n)
    for eta in [1.0, 2.0, 3.0]:
        w = 0.5 * (1 + np.cos(2 * np.pi * t)) * np.exp(eta * (t - 0.5))
        w = np.maximum(w, 1e-10)
        profiles.append(w / w.sum())
    # Double-hump
    w = np.exp(-((t - 0.3)/0.15)**2) + np.exp(-((t - 0.7)/0.15)**2)

        # Find top peaks
        top_k = min(5, n - 1)
        peak_indices = np.argsort(c)[-top_k:][::-1]
        
        # For each peak, compute gradient with respect to a
        # c[k] = sum_i a[i] * a[k-i]
        # dc[k]/da[j] = a[k-j] + a[j] (for valid indices, considering symmetry)
        # Actually: dc[k]/da[j] = sum_i [delta(i,j) * a[k-i] + a[i] * delta(k-i,j)]
        # = a[k-j] + a[j] (when both indices are valid)
        # But we need to be careful with bounds.
        
        total_grad = np.zeros(n)
        for k in peak_indices:

            i_max = min(k, n - 1)
            for i in range(i_min, i_max + 1):
                j = k - i
                if 0 <= j < n:
                    total_grad[i] += a[j]
                    total_grad[j] += a[i]

        total_grad = total_grad - np.mean(total_grad)

# But wait, in the sum c[k] = sum_i a[i]*a[k-i], the term i=j gives a[j]*a[k-j].
# If k-j = j (i.e., k = 2j), this is a[j]^2, and d/d a[j] = 2*a[j]. ✓
# If k-j ≠ j, then in the sum, a[j] appears in two places:
#   - When i=j: term is a[j]*a[k-j], derivative w.r.t. a[j] is a[k-j]
#   - When k-i=j, i.e., i=k-j: term is a[k-j]*a[j], derivative w.r.t. a[j] is a[k-j]
# Total: 2*a[k-j]. ✓
#
# So the formula is: d c[k] / d a[j] = 2 * a[k-j] if 0 <= k-j < n, else 0.
#
# Therefore: d(soft_max)/d a[j] = sum_k w[k] * 2 * a[k-j] (for valid k-j in [0,n-1])
# = 2 * sum_k w[k] * a[k-j] (for 0 <= k-j < n, i.e., j <= k < j+n)
# = 2 * sum_{m=0}^{n-1} w[m+j] * a[m] (where m = k-j, and we need m+j < 2n-1)
#
# This is 2 * (cross-correlation of w and a at lag j).
#
# For efficient computation, we can use:
# Let W = w (length 2n-1), A = a (length n)
# cross_corr[j] = sum_{m=0}^{n-1} W[m+j] * A[m] for m+j < len(W)
#
# This can be computed as: conv(W, A_flipped)[n-1+j] where A_flipped = A[::-1]
# Let me verify: conv(W, A_flipped)[k] = sum_i W[i] * A_flipped[k-i] = sum_i W[i] * A[n-1-k+i]
# Set k = n-1+j: sum_i W[i] * A[n-1-(n-1+j)+i] = sum_i W[i] * A[i-j]
# For this to equal sum_m W[m+j] * A[m], we need i-j = m, i = m+j:
# sum_m W[m+j] * A[m]. ✓
#
# So: grad[j] = 2 * conv(w, a[::-1])[n-1+j]

# Let me rewrite the gradient computation using this insight.

def compute_gradient(a, w):
    """Compute d(softmax_approx)/da where w are the soft-max weights on conv(a,a)."""
    n = a.size
    # grad[j] = 2 * conv(w, a[::-1])[n-1+j]
    Na = next_pow2(len(w) + n - 1)  # w has length 2n-1, a[::-1] has length n
    fw = np.fft.rfft(w, Na)
    fa = np.fft.rfft(a[::-1], Na)
    corr = np.fft.irfft(fw * fa, Na)
    # corr[k] = conv(w, a[::-1])[k] for k in [0, len(w)+n-2]
    # We want corr[n-1+j] for j in [0, n-1]
    grad = 2.0 * corr[n-1:n-1+n]
    return grad

def softmax_gradient_descent_v2(a0, deadline, rng, n_peaks=3, lr=0.5):
    """
    Improved gradient descent using FFT-based gradient computation.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    
    log_a = np.log(a)
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    best_a = a.copy()
    best_score = normalized_score(a)
    no_improve = 0
    base_lr = lr
    current_lr = lr
    
    # Initial beta based on spread of convolution
    c = full_convolve_fft(a)
    c_max = c.max()
    beta = 0.1 * c_max
    
    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        c_max = c.max()
        
        # Soft-max weights
        x = (c - c_max) / beta
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex > 0 and np.isfinite(s_ex):
            w = ex / s_ex
        else:
            w = np.zeros_like(c)
            w[c.argmax()] = 1.0
        
        # Gradient
        g_direct = compute_gradient(a, w)
        
        # Convert to log-space gradient
        # da/dlog_a[j] = a[j]*(1-a[j]) ... wait, for softmax parametrization:
        # a = exp(z) / sum(exp(z))
        # da[i]/dz[j] = a[i]*(delta_ij - a[j])
        # d(obj)/dz[j] = sum_i (d obj/d a[i]) * a[i]*(delta_ij - a[j])
        # = a[j] * (g[j] - sum_i g[i]*a[i])
        g_z = a * (g_direct - np.dot(g_direct, a))
        
        # Add small regularization
        g_z += 1e-10 / (a + 1e-30)  # repulsive force to keep a positive
        
        if not np.all(np.isfinite(g_z)):
            log_a = np.log(best_a)
            m[:] = 0; v[:] = 0; t_adam = 0
            continue
        
        # Adam
        t_adam += 1
        m = b1 * m + (1 - b1) * g_z
        v = b2 * v + (1 - b2) * g_z ** 2
        m_hat = m / (1 - b1 ** t_adam)
        v_hat = v / (1 - b2 ** t_adam)
        log_a = log_a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back
        log_a = np.clip(log_a, -50, 50)
        a_new = np.exp(log_a - log_a.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = normalized_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                current_lr = max(0.05, current_lr * 0.9)
        
        # Adapt beta
        if t_adam % 40 == 0:
            beta = max(1e-8, beta * 0.85)
        
        # Restart
        if no_improve > max(400, 2 * n):
            log_a = np.log(best_a) + 0.5 * rng.standard_normal(n)
            log_a = np.clip(log_a, -50,
50)
            a = np.exp(log_a - log_a.max())
            a = a / a.sum()
            m[:] = 0; v[:] = 0; t_adam = 0
            current_lr = base_lr
            beta = 0.1 * normalized_score(a) / (2 * n)
            no_improve = 0
    
    return best_a, best_score

def slsqp_refine(a0, deadline, max_iter=1000):
    """
    Use scipy SLSQP to refine a solution.
    Objective: max(conv(a,a)) subject to sum(a)=1, a>=0
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel().copy()
    a0 = np.maximum(a0, 1e-10)
    a0 = a0 / a0.sum()
    n = a0.size
    
    if n > 500:
        # Too many parameters for SLSQP; use a reduced approach
        return a0, normalized_score(a0)
    
    def objective(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-10:
            return 1e10
        x = x / s
        c = full_convolve_fft(x)
        return c.max()
    
    def objective_grad(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-10:
            return np.zeros(n)
        x = x / s
        c = full_convolve_fft(x)
        k = c.argmax()
        # d c[k] / d x[j] = 2 * x[k-j] if 0<=k-j<n
        g = np.zeros(n)
        for j in range(n):
            if 0 <= k - j < n:
                g[j] = 2.0 * x[k - j]
        # Account for normalization
        g = g - x * np.dot(g, x)  # project onto simplex tangent
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(1e-15, None)] * n
    
    result = minimize(objective, a0, method='SLSQP', jac=objective_grad,
                     bounds=bounds, constraints=constraints,
                     options={'maxiter': max_iter, 'ftol': 1e-12})
    
    if result.success or np.isfinite(result.fun):
        a_out = np.maximum(result.x, 1e-15)
        a_out = a_out / a_out.sum()
        return a_out, normalized_score(a_out)
    
    return a0, normalized_score(a0)

def coordinate_descent(a0, deadline, rng, max_sweeps=None):
    """
    Coordinate descent: try adjusting one element at a time.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    
    best_a = a.copy()
    best_score = normalized_score(a)
    
    if max_sweeps is None:
        max_sweeps = int((deadline - time.monotonic()) * 3) + 10
    
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            orig = a[i]
            # Try several factors
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                new_val = orig * factor
                if new_val < 1e-15:
                    continue
                a[i] = new_val
                s = a.sum()
                if s < 0.01:
                    a[i] = orig
                    continue
                a[i] /= s  # renormalize
                sc = normalized_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            else:
                # No improvement, restore
                a[i] = orig
                no_improve += 1
                continue
            # Keep the improved value
            no_improve = 0
        
        if not improved and no_improve > n:
            break
    
    return best_a, best_score
