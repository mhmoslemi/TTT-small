"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses coordinate descent, LP direction finding, and gradient-based polish.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.fft import fft, ifft
import time

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = fft(a, N)
    c = ifft(fa * fa, N).real[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

def coord_descent(a, max_passes=1000, deadline=None, verbose=False):
    """Coordinate descent: optimize each a[j] to minimize the score."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    n = a.size
    c = fft_convolve(a)
    S = float(a.sum())
    best_score = compute_score(a)
    best_a = a.copy()
    
    for p in range(max_passes):
        if deadline and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            # Direction: d[k] = a[k-j] + a[j-k] (out of range = 0)
            d = np.zeros(2 * n - 1)
            # a[k-j] valid for k in [j, j+n-1]
            lo, hi = j, min(2*n-2, j+n-1)
            d[lo:hi+1] = a[:hi+1-lo]
            # a[j-k] valid for k in [j-(n-1), j]
            lo2, hi2 = max(0, j-(n-1)), j
            d[lo2:hi2+1] += a[j-lo2:]
            
            if d.max() < 1e-15:
                continue
            
            # Minimize g(delta) = 2n * max(c + delta*d) / (S+delta)^2 over delta >= -a[j]
            # For active constraint at k*, optimal delta = S - 2*c[k*]/d[k*]
            # Let's try the peak first
            k_star = int(np.argmax(c))
            if d[k_star] > 1e-15:
                delta_opt = S - 2.0 * c[k_star] / d[k_star]
            else:
                delta_opt = 0.0
            
            # Clamp: delta >= -a[j], and we want S + delta > 0
            delta_opt = max(delta_opt, -a[j] + 1e-15)
            delta_opt = min(delta_opt, S)  # don't make it too large
            
            # Try this delta and also delta=0 (no change)
            best_delta = 0.0
            best_g = 2.0 * n * c.max() / (S * S)
            
            for delta in [delta_opt, -0.5 * a[j], -0.25 * a[j], -0.75 * a[j], 0.5 * a[j]]:
                delta = max(delta, -a[j] + 1e-15)
                new_S = S + delta
                if new_S < 1e-15:
                    continue
                new_peak = (c + delta * d).max()
                g = 2.0 * n * new_peak / (new_S * new_S)
                if g < best_g:
                    best_g = g
                    best_delta = delta
            
            if best_delta != 0.0:
                a[j] += best_delta
                a[j] = max(a[j], 0.0)
                c += best_delta * d
                S += best_delta
                if S < 1e-15:
                    a[j] -= best_delta
                    c -= best_delta * d
                    S -= best_delta
                improved = True
        
        cur_score = 2.0 * n * c.max() / (S * S)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
        
        if not improved and p > 10:
            break
        if verbose and p % 100 == 0:
            print(f"    CD pass {p}: score={cur_score:.10f}")
    
    return best_a, best_score

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 20})
    if res.success:
        return res.x
    return None

def lp_optimize(f0, deadline, rng, max_iter=5000):
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size
    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        g = lp_direction(f, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 30:
                break
            f = best_f + 0.02 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                M = best_M
                continue
            f = f / fs
            M = float(fft_convolve(f).max())
            if M < best_M:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            continue
        
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 100:
                f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n)) + 1e-10
                f = f / f.sum()
                M = float(fft_convolve(f).max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
            continue
        
        g = g / g_sum
        improved = False
        for t in (0.3, 0.1, 0.5, 0.2, 0.7, 0.05, 0.4):
            f_new = np.maximum((1.0 - t) * f + t * g, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(fft_convolve(f_new).max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break
        
        if not improved:
            no_imp += 1
    
    return best_f, best_M

def gradient_polish(f0, deadline, rng, lr=0.05, max_iter=20000):
    """Adam-based gradient polish in log space."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-10)
    f = f / f.sum()
    n = f.size
    z = np.log(f)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-10, 0.1 * M_val)
    
    best_f = f.copy()
    best_M = M_val
    no_imp = 0
    N_corr = next_pow2(3 * n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Soft-max weights
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))
        
        # Gradient in z-space: grad_z[j] = 2 * f[j] * (h[j] - L)
        # where h[j] = sum_k w[k] * (f[k-j] + f[j-k]) [valid indices]
        # = cross-correlation of w with f, plus the "energy" term
        # h[j] = sum_k w[k] * f[k-j] + sum_k w[k] * f[j-k]
        # Both terms are the same by change of variable (let k' = 2(n-1) - k... no)
        # Actually, sum_k w[k] * f[k-j] is (w * f)[j] in the "full" convolution sense
        # and sum_k w[k] * f[j-k] is (w * f_reversed) ... 
        # Let me just compute h directly using FFT cross-correlation.
        # h[j] = sum_k w[k] * f[|k-j|] ... no that's not right either.
        
        # Let me think again. The convolution c[k] = sum_j f[j]*f[k-j].
        # d(c[k])/d(f[j]) = f[k-j] + f[j-k] (with out-of-range = 0)
        # So d(soft_max(c, beta))/d(f[j]) = sum_k w[k] * (f[k-j] + f[j-k])
        # Let h1[j] = sum_k w[k] * f[k-j] (this is the cross-correlation, specifically (w ⊛ f)[j])
        # Let h2[j] = sum_k w[k] * f[j-k] (this is also a cross-correlation, (w ⊛ f_rev)... )
        # Actually h1[j] = sum_k w[k] * f[k-j]. If we define the "full" cross-correlation:
        # (w ⊛ f)[j] = sum_k w[k+j] * f[k] ... this is getting confusing.
        
        # Let me just compute it directly for small n, or use FFT.
        # h1[j] = sum_k w[k] * f[k-j] for k in [max(0,j), min(2n-2, j+n-1)]
        # This is the cross-correlation of w and f at lag j.
        # Using FFT: ifft(fft(w) * conj(fft(f))) gives the cross-correlation.
        
        fw = fft(w, N_corr)
        ff = fft(f, N_corr)
        # Cross-correlation: ifft(fw * conj(ff)) gives sum_k w[k] * f[k-j] ... 
        # Actually, ifft(F * conj(G))[n] = sum_k f[k] * g[k-n] (circular)
        # So ifft(fw * conj(ff))[j] = sum_k w[k] * f[k-j] (circular, with N_corr size)
        # Since N_corr >= 3n, there's no circular aliasing for the range we care about.
        h1 = ifft(fw * np.conj(ff), N_corr).real[:n]
        
        # h2[j] = sum_k w[k] * f[j-k]
        # = sum_k w[k] * f[-(k-j)] = sum_k w[k] * f_rev[k-j] where f_rev[m] = f[-m mod something]
        # Actually h2[j] = sum_k w[k] * f[j-k]. Let l = j-k, so k = j-l.
        # h2[j] = sum_l w[j-l] * f[l] = (w * f)[j] (linear convolution at position j)
        # Using FFT: ifft(fw * ff)[j] gives the circular convolution.
        h2 = ifft(fw * ff, N_corr).real[:2*n-1]  # full linear conv
        h2 = h2[:n]  # we only need j in [0, n-1]
        
        h = h1 + h2
        grad_z = 2.0 * f * (h - L)
        
        # Also need the gradient of (sum f)^-2 term... 
        # Actually, the full objective is 2n * soft_max / (sum f)^2
        # d/dz[j] = 2n * [d(soft_max)/d(f[j]) * f[j] * (sum f)^-2 + soft_max * d((sum f)^-2)/d(z[j])]
        # d(soft_max)/d(f[j]) = h[j] (computed above, but in f-space)
        # Wait, I computed grad_f = h (the gradient wrt f[j] of the soft_max)
        # Then grad_z[j] = f[j] * grad_f[j] / (sum f)^2 * 2n - 2 * 2n * soft_max * f[j] / (sum f)^3
        # = 2n * f[j] * (grad_f[j] * (sum f) - 2 * soft_max) / (sum f)^3
        
        # Hmm, let me just use the simpler form since we're normalizing f to sum=1.
        # If f is normalized (sum=1), then the objective is 2n * soft_max(f, beta).
        # But the constraint sum(f)=1 makes the gradient more complex.
        
        # Let me just use the simple gradient: grad_z = 2 * f * (h - L)
        # This is the gradient of 2n * soft_max assuming sum(f) is fixed (on the simplex).
        # The projection onto the simplex is handled by the softmax parameterization.
        
        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_f)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= z.mean()
        
        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        c = fft_convolve(f)
        M_new = float(c.max())
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if t_adam % 200 == 0:
            beta = max(1e-12, beta * 0.7)
        
        if no_imp > max(200, 2 * n):
            z = 0.8 * np.log(best_f) + 0.3 * rng.normal(size=n)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
    
    return best_f, best_M

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None

def make_initial_profiles(n, rng):
    """Generate diverse initial profiles."""
    profs = []
    t = np.linspace(0.0, 1.0, n)
    # Linear ramps
    for eta in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0):
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    # Quadratic
    for a2 in (0.5, 1.0, 1.5, 2.0):
        a = 1.0 + a2 * (2*t - 1.0)**2
        profs.append(a / a.sum())
    # Exponential
    for rate in (1.0, 2.0, 3.0, 5.0):
        a = np.exp(rate * (t - 0.5))
        profs.append(a / a.sum())
    # Smooth random
    for _ in range(6):
        a = np.ones(n)
        for k in range(1, 10):
            amp = rng.uniform(0.1, 0.5) / k
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + phase)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    return profs

"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Primary method: fast coordinate descent with analytical 1D subproblem solve.
Secondary: LP direction finding for global moves, gradient polish for refinement.
Exploits multiple sequence lengths and diverse starting points.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = fft(a, N)
    c = ifft(fa * fa, N).real[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))


def score_from_c(c, S, n):
    if S < 1e-30:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lp_direction(f, M, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 15})
    if res.success:
        return res.x
    return None


def lp_optimize(f0, deadline, rng, max_iter=10000):

    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size
    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        g = lp_direction(f, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 20:
                break
            f = best_f + 0.02 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                M = best_M
                continue
            f = f / fs
            M = float(fft_convolve(f).max())
            if M < best_M:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 80:
                f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n)) + 1e-10
                f = f / f.sum()
                M = float(fft_convolve(f).max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
            continue

        g = g / g_sum
        improved = False
        for t in (0.3, 0.1, 0.5, 0.15, 0.4, 0.2, 0.7):
            f_new = np.maximum((1.0 - t) * f + t * g, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(fft_convolve(f_new).max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1

    return best_f, best_M


def gradient_polish(f0, deadline, rng, lr=0.04, max_iter=30000):
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-10)
    f = f / f.sum()
    n = f.size
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-10, 0.08 * M_val)

    best_f = f.copy()
    best_M = M_val
    no_imp = 0
    N_corr = next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        fw = fft(w, N_corr)
        ff = fft(f, N_corr)
        h1 = ifft(fw * np.conj(ff), N_corr).real[:n]
        h2 = ifft(fw * ff, N_corr).real[:n]
        h = h1 + h2

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_f)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam % 150 == 0:
            beta = max(1e-12, beta * 0.65)

        if no_imp > max(200, 2 * n):
            z = 0.8 * np.log(best_f) + 0.3 * rng.normal(size=n)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.08 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

    return best_f, best_M


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def make_profiles(n, rng):
    profs = []
    t = np.linspace(0.0, 1.0, n)
    for eta in (0.8, 1.2, 1.5, 2.0, 2.5, 3.0, 4.0):
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        a = np.exp(rate * (t - 0.5))
        profs.append(a / a.sum())
    for a2 in (1.0, 2.0, 3.0):
        a = 1.0 + a2 * (2.0 * t - 1.0) ** 2
        profs.append(a / a.sum())
    for _ in range(8):
        a = np.ones(n)
        for k in range(1, 12):
            amp = rng.uniform(0.05, 0.6) / k
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + phase)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    return profs


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Load parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}", flush=True)

    n1 = parent.size if parent is not None else 512

    # Phase 1: Aggressive coordinate descent from multiple starts
    print(f"Phase 1: CD from diverse starts, n={n1}", flush=True)
    phase1_end = min(deadline, t_start + 360.0)

    starts = []
    if parent is not None:
        starts.append(parent.copy())
        # Perturbed versions
        for _ in range(4):
            p = parent + 0.05 * rng.normal(size=n1)
            p = np.maximum(p, 0.0)
            starts.append(p / p.sum())
    starts.extend(make_profiles(n1, rng)[:12])

    per_start = max(5.0, (phase1_end - t_start) / max(1, len(starts)))
    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        per = min(phase1_end, time.monotonic() + per_start)
        p_best, sc = coord_descent(p0, max_passes=3000, deadline=per)
        try_update(p_best, sc)
        if i % 4 == 0:
            print(f"  CD start {i}/{len(starts)}: best={best_score:.10f}", flush=True)

    # Greedy CD on best
    if time.monotonic() < phase1_end - 10:
        per = min(phase1_end, time.monotonic() + 60.0)
        p_best, sc = coord_descent_greedy(best_seq, max_passes=5000, deadline=per)
        try_update(p_best, sc)
        print(f"  Greedy CD: best={best_score:.10f}", flush=True)

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s", flush=True)

    # Phase 2: LP for global moves
    print(f"Phase 2: LP optimization", flush=True)
    phase2_end = min(deadline, t_start + 900.0)
    if best_seq is not None:
        n_lp = best_seq.size
        num_lp_rounds = 0
        while time.monotonic() < phase2_end - 20:
            per = min(phase2_end, time.monotonic() + 120.0)
            p_best, M_best = lp_optimize(best_seq, per, rng, max_iter=8000)
            sc = compute_score(p_best)
            try_update(p_best, sc)
            num_lp_rounds += 1
            print(f"  LP round {num_lp_rounds}: score={sc:.10f}, best={best_score:.10f}", flush=True)

            # After LP, do CD to refine
            if time.monotonic() < phase2_end - 30:
                per2 = min(phase2_end, time.monotonic() + 40.0)
                p_best2, sc2 = coord_descent(best_seq, max_passes=2000, deadline=per2)
                try_update(p_best2, sc2)

            if sc >= best_score - 1e-9:
                # Stuck, perturb
                p_pert = best_seq + 0.04 * rng.normal(size=n_lp)
                p_pert = np.maximum(p_pert, 0.0)
                p_pert = p_pert / p_pert.sum()
                per3 = min(phase2_end, time.monotonic() + 30.0)
                p_best3, sc3 = coord_descent(p_pert, max_passes=1000, deadline=per3)
                try_update(p_best3, sc3)

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s", flush=True)

    # Phase 3: Gradient polish
    print(f"Phase 3: Gradient polish", flush=True)
    phase3_end = min(deadline, t_start + 1320.0)
    if best_seq is not None:
        for lr in (0.04, 0.02, 0.01):
            if time.monotonic() > phase3_end - 30:
                break
            per = min(phase3_end, time.monotonic() + 120.0)
            p_best, M_best = gradient_polish(best_seq, per, rng, lr=lr)
            sc = compute_score(p_best)
            try_update(p_best, sc)
            print(f"  GP lr={lr}: score={sc:.10f}", flush=True)
            # CD after each polish
            if time.monotonic() < phase3_end - 20:
                per2 = min(phase3_end, time.monotonic() + 30.0)
                p_best2, sc2 = coord_descent(best_seq, max_passes=2000, deadline=per2)
                try_update(p_best2, sc2)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s", flush=True)

    # Phase 4: Try larger n
    print(f"Phase 4: Larger n", flush=True)
    phase4_end = min(deadline, t_start + 1650.0)
    if best_seq is not None:
        for n_try in (1024, 768, 1536):
            if time.monotonic() > phase4_end - 30:
                break
            if n_try <= best_seq.size:
                continue
            p_ext = interpolate_profile(best_seq, n_try)
            if p_ext is None:
                continue
            # CD first
            per = min(phase4_end, time.monotonic() + 50.0)
            p_best, sc = coord_descent(p_ext, max_passes=1000, deadline=per)
            try_update(p_best, sc)
            print(f"  n={n_try} CD: score={sc:.10f}", flush=True)

            # LP if time allows
            if time.monotonic() < phase4_end - 40 and n_try <= 1024:
                per2 = min(phase4_end, time.monotonic() + 50.0)
                p_best2, M2 = lp_optimize(p_best, per2, rng, max_iter=5000)
                sc2 = compute_score(p_best2)
                try_update(p_best2, sc2)
                # CD after LP
                if time.monotonic() < phase4_end - 15:
                    per3 = min(phase4_end, time.monotonic() + 20.0)
                    p_best3, sc3 = coord_descent(p_best2, max_passes=1500, deadline=per3)
                    try_update(p_best3, sc3)
                print(f"  n={n_try} LP: best={best_score:.10f}", flush=True)

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s", flush=True)

    # Phase 5: Final intensive refinement
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_best = best_seq.size
        per = min(deadline - 3, time.monotonic() + 60.0)
        p_best, sc = coord_descent(best_seq, max_passes=10000, deadline=per)
        try_update(p_best, sc)

        if time.monotonic() < deadline - 3:
            per2 = min(deadline - 2, time.monotonic() + 20.0)
            p_best2, sc2 = coord_descent_greedy(best_seq, max_passes=3000, deadline=per2)
            try_update(p_best2, sc2)

    print(f"Phase 5 done: best={best_score:.10f}", flush=True)

    # Final
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}", flush=True)
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}", flush=True)
        except Exception as e:
            print(f"Eval error: {e}", flush=True)
        return [float(x) for x in final]

    # Fallback
    fb = np.ones(512) / 512.0
    return [float(x) for x in fb]
