```python
import numpy as np
import time

def compute_conv_fft(f):
    """Compute linear convolution of f with itself using FFT."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)[:2*n-1]

def compute_grad_fft(f, w):
    """Compute gradient: grad[i] = 2 * sum_j w[i+j] * f[j]."""
    n = len(f)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    h = np.fft.irfft(W * np.conj(F), N)[:n]
    return 2.0 * h

def softmax_stable(z):
    """Compute softmax with numerical stability."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def compute_score(f):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-30:
        return np.inf
    c = compute_conv_fft(f)
    return 2.0 * n * c.max() / (s * s)

def optimize_log_domain(z_init, deadline, rng, lr=0.1, max_iters=100000):
    """Adam optimization in log-domain with soft-max objective."""
    n = len(z_init)
    z = z_init.copy() - z_init.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    f = softmax_stable(z)
    c = compute_conv_fft(f)
    M = c.max()
    
    # Initial beta based on the spread of c
    beta = 0.5 * M
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute soft-max weights
        c_shift = c - c.max()
        w = np.exp(c_shift / beta)
        w /= w.sum()
        
        # Compute gradient in f-space
        grad_f = compute_grad_fft(f, w)
        
        # Convert to z-space gradient
        grad_z = f * (grad_f - np.dot(f, grad_f))
        
        # Safety check
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t = 0
            continue
        
        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()  # Keep centered for numerical stability
        
        # Compute new f
        f = softmax_stable(z)
        c = compute_conv_fft(f)
        M = c.max()
        
        # Update beta (anneal down)
        beta = max(0.01 * M, beta * 0.999)
        
        # Track best
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Restart if stuck
        if no_imp > 2000:
            z = 0.5 * best_z + 0.5 * rng.normal(0, 0.5, n)
            z -= z.mean()
            f = softmax_stable(z)
            c = compute_conv_fft(f)
            M = c.max()
            beta = 0.5 * M
            m.fill(0); v.fill(0); t = 0
            no_imp = 0
    
    return best_f, best_M
```

Looking at this problem, I need to minimize `2*n*max(convolve(f,f))/sum(f)^2` for non-negative sequences. The key insight is that this is scale-invariant, so I normalize to `sum(f)=1` and minimize `2*n*max(convolve(f,f))`.

The theoretical lower bound is `2n/(2n-1)` (from the fact that the sum of all convolution values equals `sum(f)^2` and there are `2n-1` values). So for n=256, the bound is ~1.002.

My approach uses **gradient descent in log-domain** (softmax parameterization) with:
- FFT-based fast gradient computation
- Adam optimizer with soft-max temperature annealing
- Multiple diverse initializations and restarts
- Multiple sequence lengths (256, 512)
- The parent sequence as a warm start

```python
import numpy as np
import time


def compute_conv_fft(f):
    """Compute linear convolution of f with itself using FFT."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)[:2 * n - 1]


def compute_grad_fft(f, w):
    """Compute grad[i] = 2 * sum_j w[i+j] * f[j] using FFT cross-correlation."""
    n = len(f)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    h = np.fft.irfft(W * np.conj(F), N)[:n]
    return 2.0 * h


def softmax_stable(z):
    """Compute softmax with numerical stability."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def compute_score_direct(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = compute_conv_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def make_linear_profile(n, direction=1):
    """Linear ramp, normalized to sum 1."""
    t = np.arange(n, dtype=np.float64)
    if direction == 1:
        f = 1.0 + t / (n - 1 + 1e-10)
    else:
        f = 1.0 + (n - 1 - t) / (n - 1 + 1e-10)
    return f / f.sum()


def make_cosine_profile(n):
    """Cosine window, normalized to sum 1."""
    t = np.linspace(0, 1, n)
    f = 0.5 * (1.0 + np.cos(np.pi * (2 * t - 1)))
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_gaussian_profile(n, mu=0.5, sigma=0.3):
    """Gaussian bump, normalized to sum 1."""
    t = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    return f / f.sum()


def make_random_smooth(n, rng, n_harmonics=6):
    """Random smooth profile using Fourier series."""
    t = np.linspace(0, 1, n, endpoint=False)
    f = np.ones(n)
    for k in range(1, n_harmonics + 1):
        a = rng.normal(0, 1.0 / k)
        b = rng.normal(0, 1.0 / k)
        f += a * np.cos(2 * np.pi * k * t) + b * np.sin(2 * np.pi * k * t)
    f = np.maximum(f, 1e-6)
    return f / f.sum()


def make_antisymmetric_profile(n, rng):
    """Anti-symmetric profile: high on one side, low on other."""
    t = np.linspace(0, 1, n)
    slope = rng.uniform(0.5, 3.0)
    f = 1.0 + slope * t
    f = np.maximum(f, 1e-6)
    return f / f.sum()


def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def optimize_log_domain(z_init, deadline, rng, n, lr=0.5, max_iters=500000, verbose=False):
    """
    Adam optimization in log-domain (softmax parameterization).
    Minimizes soft_max(convolve(f,f)) with annealed temperature.
    Returns (best_f, best_M).
    """
    z = z_init.copy()
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    f = softmax_stable(z)
    c = compute_conv_fft(f)
    M = float(c.max())

    # Initialize beta: fraction of M for soft-max temperature
    beta_frac = 0.5
    beta = beta_frac * M

    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    no_imp = 0
    patience = 3000

    # Pre-allocate
    w_buf = np.zeros(2 * n - 1)
    grad_f_buf = np.zeros(n)
    grad_z_buf = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute soft-max weights over convolution
        c_shift = c - c.max()
        w_buf[:] = np.exp(c_shift / beta)
        w_sum = float(w_buf.sum())
        if w_sum < 1e-30:
            w_buf.fill(1.0 / (2 * n - 1))
        else:
            w_buf /= w_sum

        # Compute gradient in f-space
        grad_f_buf[:] = compute_grad_fft(f, w_buf)

        # Convert to z-space: grad_z[i] = f[i] * (grad_f[i] - dot(f, grad_f))
        dot_fgrad = float(np.dot(f, grad_f_buf))
        grad_z_buf[:] = f * (grad_f_buf - dot_fgrad)

        # Safety check
        if not np.all(np.isfinite(grad_z_buf)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            f = softmax_stable(z)
            c = compute_conv_fft(f)
            M = float(c.max())
            continue

        # Adam update in z-space
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z_buf
        v = b2 * v + (1.0 - b2) * (grad_z_buf * grad_z_buf)
        m_hat = m / (1.0 - b1 ** t_adam)
        v_hat = v / (1.0 - b2 ** t_adam)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()

        # Numerical guard
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Clip z to avoid extreme values
        z = np.clip(z, -30, 30)
        z -= z.mean()

        # Compute new f and objective
        f = softmax_stable(z)
        c = compute_conv_fft(f)
        M = float(c.max())

        # Anneal beta: decrease over time
        beta_frac = max(0.02, beta_frac * 0.9995)
        beta = beta_frac * M

        # Track best
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Restart if stuck
        if no_imp > patience:
            # Random restart around best solution
            noise_scale = 0.3 + 0.3 * rng.random()
            z = best_z + noise_scale * rng.standard_normal(n)
            z -= z.mean()
            z = np.clip(z, -30, 30)
            f = softmax_stable(z)
            c = compute_conv_fft(f)
            M = float(c.max())
            beta_frac = 0.5
            beta = beta_frac * M
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
            patience = max(500, patience // 2)

        if verbose and it % 10000 == 0:
            score = 2.0 * n * M
            print(f"    iter {it}: M={M:.10f}, score={score:.8f}, beta={beta:.6f}")

    return best_f, best_M


def coordinate_polish(f, deadline, max_sweeps=50):
    """
    Coordinate descent polish: for each coordinate i, find optimal f[i]
    that minimizes max_k (f*f)[k] subject to f[i] >= 0 and sum(f) = 1.
    """
    n = len(f)
    f = f.copy()
    best_M = float(compute_conv_fft(f).max())
    f_best = f.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        for i in range(n):
            if time.monotonic() > deadline:
                break

            # Current value
            fi = f[i]
            # Remove f[i] from convolution
            f[i] = 0.0
            c_rest = compute_conv_fft(f)
            # Add back: (f*f)[k] = c_rest[k] + 2*f[i]*f[k-i] (if valid)
            # We want to find f[i] >= 0 minimizing max_k (c_rest[k] + 2*f[i]*f[k-i])
            # subject to sum(f) = 1, i.e., f[i] = 1 - sum(f[j] for j != i)

            # Actually with the sum constraint, f[i] is determined.
            # So coordinate descent doesn't apply directly.
            # Instead: scale all other elements to make room.
            # This is more complex, so let's use a simpler approach:
            # Just try a few perturbation directions.
            f[i] = fi

            # Simple: try increasing and decreasing f[i] slightly
            # while compensating by scaling other elements
            best_local_M = best_M
            best_local_fi = fi

            for delta in [0.01, 0.005, 0.002, -0.01, -0.005, -0.002]:
                new_fi = fi + delta
                if new_fi < 0:
                    continue
                # Scale others to maintain sum = 1
                others_sum = 1.0 - new_fi
                if others_sum <= 0:
                    continue
                current_others_sum = 1.0 - fi
                if current_others_sum <= 0:
                    continue
                scale = others_sum / current_others_sum

                f_trial = f.copy()
                f_trial[i] = new_fi
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                f_trial[mask] *= scale

                c_trial = compute_conv_fft(f_trial)
                M_trial = float(c_trial.max())
                if M_trial < best_local_M:
                    best_local_M = M_trial
                    best_local_fi = new_fi
                    best_local_scale = scale

            if best_local_fi != fi:
                f[i] = best_local_fi
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                f[mask] *= best_local_scale
                f = np.maximum(f, 0)
                f = f / f.sum()
                M_now = float(compute_conv_fft(f).max())
                if M_now < best_M:
                    best_M = M_now
                    f_best = f.copy()

    return f_best, best_M


def generate_initializations(n, parent, rng, count=12):
    """Generate diverse initializations, return list of z-vectors (log-domain)."""
    inits = []

    # Parent (interpolated)
    if parent is not None:
        p = interpolate_profile(parent, n)
        p = np.maximum(p, 1e-10)
        p = p / p.sum()
        inits.append(np.log(p) - np.log(p).mean())

    # Structured profiles
    profiles = [
        make_linear_profile(n, 1),
        make_linear_profile(n, -1),
        make_cosine_profile(n),
        make_gaussian_profile(n, 0.4, 0.25),
        make_gaussian_profile(n, 0.6, 0.25),
        make_antisymmetric_profile(n, rng),
        make_antisymmetric_profile(n, rng),
    ]

    # Random smooth
    for _ in range(4):
        profiles.append(make_random_smooth(n, rng))

    for p in profiles:
        p = np.maximum(p, 1e-10)
        p = p / p.sum()
        z = np.log(p)
        z -= z.mean()
        inits.append(z)

    return inits

    """
    Multi-phase optimization to minimize 2*n*max(convolve(f,f))/sum(f)^2.
    Uses gradient descent in log-domain (softmax parameterization) with Adam,
    soft-max temperature annealing, multiple restarts, and diverse initializations.
    Tries multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)

    # Get parent sequence if available
    parent = None
    try:

            if p.sum() > 1e-10:
                parent = p / p.sum()
    except Exception:
        pass

    best_seq = None
    best_score = np.inf

    def try_update(f, score):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_seq = f.copy()

    # Try multiple n values
    n_values = [256, 512]

    # Allocate time: give more time to n=256 (proven to work well)
    time_allocation = {}
    remaining = deadline - t_start
    # 60% to n=256, 40% to n=512
    time_allocation[256] = 0.6 * remaining
    time_allocation[512] = 0.4 * remaining

    for n in n_values:
        phase_start = time.monotonic()
        phase_deadline = min(deadline, phase_start + time_allocation.get(n, 300))
        print(f"\n=== n={n}, time budget={time_allocation.get(n, 300):.0f}s ===")

        inits = generate_initializations(n, parent, rng, count=12)

        # Phase 1: Run optimizer on each initialization with time budget
        results = []
        num_inits = len(inits)
        per_init_time = time_allocation.get(n, 300) / max(1, num_inits)

        for idx, z0 in enumerate(inits):
            if time.monotonic() > phase_deadline - 5:
                break
            init_deadline = min(phase_deadline, time.monotonic() + per_init_time * 1.5)
            f_best, M_best = optimize_log_domain(
                z0, init_deadline, rng, n,
                lr=0.3 + 0.2 * rng.random(),
                max_iters=200000
            )
            score = 2.0 * n * M_best
            results.append((score, f_best, z0))
            try_update(f_best, score)

            if (idx + 1) % 3 == 0:
                print(f"  [{n}] init {idx+1}/{num_inits}: score={score:.8f}, best={best_score:.8f}")

        # Phase 2: Continue optimizing the best results
        results.sort(key=lambda x: x[0])
        top_k = min(5, len(results))
        polish_deadline = min(deadline, time.monotonic() + 0.3 * time_allocation.get(n, 300))

        for idx in range(top_k):
            if time.monotonic() > polish_deadline:
                break
            score, f_best, z0 = results[idx]
            # Re-optimize from best found, with fresh Adam state
            z_start = np.log(np.maximum(f_best, 1e-15))
            z_start -= z_start.mean()
            f_polished, M_polished = optimize_log_domain(
                z_start, polish_deadline, rng, n,
                lr=0.1 + 0.1 * rng.random(),
                max_iters=300000
            )
            score_polished = 2.0 * n * M_polished
            try_update(f_polished, score_polished)
            print(f"  [{n}] polish {idx+1}: score={score_polished:.8f}, best={best_score:.8f}")

        # Phase 3: Coordinate polish on best
        if time.monotonic() < deadline - 30 and best_seq is not None and best_seq.size == n:
            f_cp, M_cp = coordinate_polish(best_seq, min(deadline - 10, time.monotonic() + 60))
            score_cp = 2.0 * n * M_cp
            try_update(f_cp, score_cp)
            print(f"  [{n}] coord polish: score={score_cp:.8f}, best={best_score:.8f}")

    print(f"\n=== FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = 2.0

    # Ensure validity

        best_seq = np.ones(256) / 256.0

        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
```
