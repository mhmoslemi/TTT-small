"""
Multi-stage convex-structured optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS on smoothed log-sum-exp objective with pair-coordinate descent.
Uses FFT for efficient convolution and gradient computation.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score_normalized(a):
    """Compute score assuming a is normalized to sum=1."""
    n = a.size
    c = fft_convolve(a)
    return float(2.0 * n * c.max())


def compute_score(a):
    """Compute the actual score for a sequence."""
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def make_linear_profile(n, eta):
    """Create a linear profile: increasing from 1-eta/2 to 1+eta/2."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=10):
    """Create a smooth random profile using Fourier components."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.6) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def logsumexp_smoothing(c, beta):
    """Compute beta * log(sum(exp(c/beta))) for smoothing."""
    x = c / beta
    x = x - x.max()
    return beta * (x.max() + np.log(np.sum(np.exp(x - (x - x.max())))))
    # Simpler:
    # mx = x.max()
    # return beta * (mx + np.log(np.sum(np.exp(x - mx))))


def lbfgs_smoothed(a0, beta_init, deadline, rng, max_iter=2000):
    """
    Optimize the smoothed objective using L-BFGS with softmax parameterization.
    a0: initial sequence (will be normalized)
    beta_init: initial smoothing parameter
    Returns: optimized sequence (normalized to sum=1), best max_conv value
    """
    n = a0.size
    a0 = np.maximum(a0, 1e-15).astype(np.float64)
    a0 = a0 / a0.sum()
    
    z = np.log(a0)
    z = z - z.mean()  # center for numerical stability
    
    beta = beta_init
    best_a = a0.copy()
    best_M = float(fft_convolve(a0).max())
    no_imp = 0
    
    N_corr = next_pow2(3 * n)  # for cross-correlation
    
    def objective_and_grad(z_vec):
        nonlocal beta
        z_vec = z_vec - z_vec.mean()
        a = np.exp(z_vec - z_vec.max())
        a = a / a.sum()
        
        c = fft_convolve(a)
        M = float(c.max())
        
        # Compute weights
        x = (c - M) / beta
        ex = np.exp(np.clip(x, -100, 5))
        w = ex / ex.sum()
        
        # grad_a_i = sum_k w_k * (a_{k-i} + a_{i-k})
        # = 2 * cross-correlation of w and a (circular, padded)
        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(a, N_corr)
        # cross-correlation: sum_k w[k] * a[k-i] = ifft(fw * conj(fa))
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]
        # Also need sum_k w[k] * a[i-k] = ifft(conj(fw) * fa)
        h2 = np.fft.irfft(np.conj(fw) * fa, N_corr)[:n]
        grad_a = h + h2
        
        # Actually, for self-convolution, a_{k-i} + a_{i-k} is symmetric
        # The cross-correlation h[j] = sum_k w[k] * a[k-j]
        # And h2[j] = sum_k w[k] * a[j-k]
        # For 1D signals, these are related: h2[j] = h[-j] in circular sense
        # But since we're taking the first n elements, they might differ
        # Let's just use: grad_a = h + h2
        # Actually, w is the weight on c_k, and dc_k/da_i = a_{k-i} + a_{i-k}
        # sum_k w_k * a_{k-i} = (w * a)[i] in cross-correlation sense = h[i]
        # sum_k w_k * a_{i-k} = (w * a reversed)[i] = h2[i]
        # So grad_a = h + h2 is correct
        
        # Convert to natural gradient (for softmax parameterization)
        S = float(np.dot(a, grad_a))
        grad_z = grad_a - a * S
        
        # Objective: logsumexp(c / beta)
        obj = float(np.sum(np.log(ex)) + 0.0)  # = log(sum(ex)) since we subtracted M/beta... 
        # Actually: logsumexp(c/beta) = M/beta + log(sum(exp((c-M)/beta)))
        # = M/beta + log(sum(ex))
        obj = M / beta + float(np.log(float(ex.sum())))
        
        return obj, grad_z
    
    # L-BFGS optimization loop with adaptive beta
    z_current = z.copy()
    
    for round in range(30):
        if time.monotonic() > deadline:
            break
        
        # Reset optimizer state each round (to handle beta changes)
        res = minimize(objective_and_grad, z_current, method='L-BFGS-B',
                      jac=True, options={'maxiter': max_iter // 30, 'gtol': 1e-8})
        
        z_current = res.x
        z_current = np.clip(z_current, -30, 30)
        z_current = z_current - z_current.mean()
        
        a_new = np.exp(z_current - z_current.max())
        a_new = a_new / a_new.sum()
        M_new = float(fft_convolve(a_new).max())
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decrease beta
        beta = max(1e-6, beta * 0.6)
        
        if no_imp > 10:
            # Basin hop: perturb and restart
            z_current = 0.8 * z_current + 0.5 * rng.normal(size=n)
            z_current = np.clip(z_current, -30, 30)
            z_current = z_current - z_current.mean()
            no_imp = 0
            beta = max(1e-6, best_M * 0.1)
        
        print(f"    L-BFGS round {round}: M={M_new:.12f}, best={best_M:.12f}, beta={beta:.2e}")
    
    return best_a, best_M


def pair_coordinate_descent(a0, deadline, rng, max_iter=50000):
    """
    Pair coordinate descent: at each step, pick a random pair (i,j)
    and find the optimal delta to transfer from j to i.
    """
    n = a0.size
    a = np.maximum(a0, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-30:
        return a0, np.inf
    a = a / s
    
    c = fft_convolve(a)
    best_M = float(c.max())
    best_a = a.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Compute l_k and q_k for transferring delta from j to i
        # c_k(delta) = c_k + l_k * delta + q_k * delta^2
        # where delta > 0 means a_i += delta, a_j -= delta
        
        # l_k = 2*a_{k-i} - 2*a_{k-j} (with bounds checking)
        # But for k = 2i: the quadratic term from a_i^2 means l_k includes 2*a_i
        # and q_k = 1
        # Similarly for k = 2j
        
        # Let's compute l_k directly:
        # dc_k/da_i = a_{k-i} + a_{i-k} (out of bounds = 0)
        # dc_k/da_j = a_{k-j} + a_{j-k}
        # When we increase a_i by delta and decrease a_j by delta:
        # dc_k = delta * dc_k/da_i - delta * dc_k/da_j + O(delta^2)
        # The O(delta^2) terms: delta^2 * [1 if k=2i (from a_i^2 term) + 1 if k=2j (from a_j^2 term with negative sign... wait]
        
        # Actually: (a_i + delta)^2 - a_i^2 = 2*a_i*delta + delta^2
        # (a_j - delta)^2 - a_j^2 = -2*a_j*delta + delta^2
        # Cross terms: a_i * a_{k-i} changes by delta * a_{k-i} (if k-i != i)
        #              a_{k-i} * a_i changes by delta * a_{k-i} (if k-i != i)
        # So for k != 2i: contribution from a_i change = 2*delta * a_{k-i}
        # For k = 2i: contribution = 2*delta*a_i + delta^2
        # Similarly for a_j.
        
        # So: l_k = 2*a_{k-i} - 2*a_{k-j} (with a_x = 0 if x not in [0,n-1])
        # Special: if k = 2i, l_k += 0 (already counted in the 2*a_{k-i} term since k-i=i)
        # Wait: when k=2i, k-i=i, so a_{k-i} = a_i. The formula gives 2*a_i which is correct
        # for the linear part. The delta^2 term is the quadratic part.
        
        # q_k = 1 if k == 2i, 1 if k == 2j, 0 otherwise
        # (The delta^2 terms come from (a_i+delta)^2 and (a_j-delta)^2)
        
        # Compute l_k efficiently:
        # l_k = 2*a_{k-i} - 2*a_{k-j} for all k
        # a_{k-i} is a shifted version of a
        
        l = np.zeros(2 * n - 1)
        # a_{k-i} for k in [0, 2n-2]: valid when 0 <= k-i < n, i.e., i <= k < i+n
        lo_i = i
        hi_i = i + n  # exclusive
        l[lo_i:hi_i] += 2.0 * a
        # a_{k-j} for k in [0, 2n-2]: valid when 0 <= k-j < n, i.e., j <= k < j+n
        lo_j = j
        hi_j = j + n
        l[lo_j:hi_j] -= 2.0 * a
        
        q = np.zeros(2 * n - 1)
        if 0 <= 2 * i < 2 * n - 1:
            q[2 * i] += 1.0
        if 0 <= 2 * j < 2 * n - 1:
            q[2 * j] += 1.0
        
        # Find optimal delta in [-a_i, a_j] (well, [0, a_j] if we only move from j to i)
        # Actually delta can be negative (move from i to j)
        # Constraint: a_i + delta >= 0 => delta >= -a_i
        #             a_j - delta >= 0 => delta <= a_j
        
        lo = -a[i]
        hi = a[j]
        
        if hi - lo < 1e-15:
            continue
        
        # Objective at delta: max_k (c_k + l_k * delta + q_k * delta^2)
        # This is convex in delta (max of convex functions)
        # Use ternary search
        for _ in range(50):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = float(np.max(c + l * m1 + q * m1 * m1))
            f2 = float(np.max(c + l * m2 + q * m2 * m2))
            if f1 <= f2:
                hi = m2
            else:
                lo = m1
        
        delta = (lo + hi) / 2.0
        
        if abs(delta) < 1e-15:
            continue
        
        # Update
        a[i] += delta
        a[j] -= delta
        c = c + l * delta + q * delta * delta
        c = np.maximum(c, 0.0)
        
        M_new = float(c.max())
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a.copy()
    
    return best_a, best_M


def interpolate_profile(p0, n_new):
    """Interpolate profile p0 to length n_new."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return None
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def make_deterministic_starts(n, rng):
    """Create diverse deterministic and random starting points."""
    starts = []
    # Linear profiles
    for eta in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0):
        starts.append(make_linear_profile(n, eta))
    # Smooth random
    for _ in range(12):
        starts.append(make_smooth_random(n, rng))
    # Quadratic profiles
    t = np.linspace(0, 1, n)
    for a2 in (-2.0, -1.5, -1.0, -0.5, 0.5, 1.0):
