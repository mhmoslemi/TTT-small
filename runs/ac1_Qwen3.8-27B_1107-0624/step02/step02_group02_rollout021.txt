def next_pow2(n):
    k = 1

    return max(2, k)


def full_convolve_fft(a):

    a = np.asarray(a, dtype=np.float64).ravel()

    return np.maximum(c, 0.0)

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M, n):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def soft_max_with_weights(c, beta):
    """Compute soft max and weights."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights_fft(w, p):
    """Compute gradient of soft-max via FFT: h[i] = sum_j w[i+j]*p[j]."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-14
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

            adam_t = 0
            continue
        
        # Adaptive beta: decrease to focus on peak
        if adam_t % 40 == 0:
            beta = max(1e-14, beta * 0.85)
        
        # Restart if stuck
        if no_imp > max(500, 3 * n):
            z = 0.85 * best_z + 0.2 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            m_n.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_sine_profile(n, k=1, phase=0.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.3 * np.sin(2.0 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, a1=0.0, a2=1.0):
    t = np.linspace(0.0, 1.0, n)
    arr = 1.0 + a1 * t + a2 * t ** 2
    arr = np.maximum(arr, 1e-12)
    return arr / arr.sum()


def make_random_smooth(n, rng, n_terms=8):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_terms + 1):
        amp = rng.uniform(-0.15, 0.15)
        freq = rng.uniform(0.5, k * 0.8)
        a += amp * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 1e-12)
    return result / result.sum()


def get_starting_profiles(n, rng):
    profiles = []
    for eta in [1.0, 2.0, 3.0]:
        profiles.append(make_linear_profile(n, eta))
    for k in [1, 2, 3]:
        profiles.append(make_cosine_profile(n, k))
    for k in [1, 2]:
        profiles.append(make_sine_profile(n, k, 0.5))
    profiles.append(make_quadratic_profile(n, -0.5, 1.0))
    profiles.append(make_quadratic_profile(n, 1.0, -0.3))
    for _ in range(6):
        profiles.append(make_random_smooth(n, rng))
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining alternating minimization, LP descent,
    and Nesterov-accelerated Adam polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
    
    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass
    
    print(f"Starting search. budget={budget_s}s")
    
    n_main = 512
    
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        if init_seq.size != n_main:
            p_main = interpolate_profile(init_seq, n_main)
        else:
            p_main = init_seq.copy()
        update(p_main, compute_score(p_main))
    else:
        p_main = make_linear_profile(n_main, 2.0)
        update(p_main, compute_score(p_main))
    
    # ============ Phase 1: Alternating Minimization at n=512 ============
    phase1_end = min(deadline, t_start + 300.0)
    if time.monotonic() < phase1_end - 5:
        print(f"Phase 1: AM at n={n_main}, score={best_score:.10f}")
        am_deadline = phase1_end - 5
        result = alternating_minimize(p_main, n_main, am_deadline, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After AM: score={sc:.10f}")
    
    # ============ Phase 2: LP-based refinement at n=512 ============
    phase2_end = min(deadline, t_start + 550.0)
    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        print(f"Phase 2: LP at n={n_main}, score={best_score:.10f}")
        lp_deadline = phase2_end - 10
        result = lp_optimize(best_seq, lp_deadline, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After LP: score={sc:.10f}")
    
    # ============ Phase 3: Nesterov Adam polish at n=512 ============
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        print(f"Phase 3: Adam at n={best_seq.size}, score={best_score:.10f}")
        
        # Multiple rounds with different parameters
        for lr, reg, beta_frac in [(0.06, 1e-8, 0.1), (0.03, 1e-8, 0.05), (0.02, 1e-9, 0.03)]:
            if time.monotonic() > phase3_end - 10:
                break
            per_deadline = min(phase3_end - 10, time.monotonic() + 120.0)
            p0 = best_seq + 0.002 * rng.normal(size=best_seq.size)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            c0 = full_convolve_fft(p0)
            beta_init = beta_frac * float(c0.max())
            result, sc = nesterov_adam_polish(p0, per_deadline, rng, lr=lr, reg=reg, beta_init=beta_init)
            update(result, sc)
            print(f"  Adam lr={lr}: score={sc:.10f}")
    
    # ============ Phase 4: Try n=1024 ============
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end - 30 and best_seq is not None and best_score > 1.505:
        n_large = 1024
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 4: n={n_large}, score={sc:.10f}")
        
        # Quick AM at 1024
        am_dl = min(phase4_end - 30, time.monotonic() + 150.0)
        result = alternating_minimize(p_large, n_large, am_dl, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        print(f"After AM 1024: score={sc:.10f}")
        
        # Quick Adam at 1024
        if time.monotonic() < phase4_end - 20:
            per = min(phase4_end - 20, time.monotonic() + 100.0)
            p0 = best_seq + 0.001 * rng.normal(size=best_seq.size)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = nesterov_adam_polish(p0, per, rng, lr=0.04, reg=1e-8)
            update(result, sc)
            print(f"After Adam 1024: score={sc:.10f}")
    
    # ============ Phase 5: Final fine polish ============
    if time.monotonic() < deadline - 10 and best_seq is not None:
        print(f"Phase 5: Final polish at n={best_seq.size}")
        n_final = best_seq.size
        # Very fine Adam with tiny lr
        for lr in [0.015, 0.01, 0.008]:
            if time.monotonic() > deadline - 10:
                break
            per = min(deadline - 10, time.monotonic() + 40.0)
            p0 = best_seq + 0.0005 * rng.normal(size=n_final)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            c0 = full_convolve_fft(p0)
            beta_init = 0.02 * float(c0.max())
            result, sc = nesterov_adam_polish(p0, per, rng, lr=lr, reg=1e-9, beta_init=beta_init)
            update(result, sc)
            print(f"  Final lr={lr}: score={sc:.10f}")
    
    # ============ Phase 6: Try diverse profiles at 512 if time remains ============
    if time.monotonic() < deadline - 30 and best_score > 1.503:
        print(f"Phase 6: Diverse profiles, current best={best_score:.10f}")
        profiles = get_starting_profiles(512, rng)
        for p in profiles:
            if time.monotonic() > deadline - 30:
                break
            sc_p = compute_score(p)
            if sc_p >= best_score + 0.02:
                continue
            per = min(deadline - 30, time.monotonic() + 40.0)
            result, sc = nesterov_adam_polish(p, per, rng, lr=0.05, reg=1e-8)
            update(result, sc)
        
        # Try LP from best
        if time.monotonic() < deadline - 20:
            per = deadline - 20
            result = lp_optimize(best_seq, per, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
    
    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 2.0)
    
    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 2.0)
    
    final_score = compute_score(final_seq)
    elapsed = time.monotonic() - t_start
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}, elapsed={elapsed:.1f}s")
    
    # Validate
    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
            if eval_score < final_score:
                print(f"(Note: eval score {eval_score:.10f} < compute_score {final_score:.10f})")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
    
    return [float(x) for x in final_seq]
