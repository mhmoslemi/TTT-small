def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 192:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):

        return np.ones_like(a) / a.size

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _smooth_max_and_grad(a, beta):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    # Gradient of smooth_max = w @ conv_matrix(a) = (w * a_flip) correlation
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    val = mval + beta * float(np.log(s))
    return val, grad, c


def _l_bfgs_smooth_max(a, deadline, max_iter=20000):
    """L-BFGS-B optimization in softmax parameterization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(a)
    n = len(a)

    # Parameterize as a = softmax(z)
    a_safe = np.maximum(a, 1e-15)
    z = np.log(a_safe)
    z -= z.max()  # for numerical stability

    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-6, 0.05 * float(c.max()))
    no_imp = 0
    beta_start = beta

    def objective_and_grad(z_vec):
        nonlocal beta
        e = np.exp(z_vec - z_vec.max())
        a_z = e / e.sum()
        val, grad_c, c_c = _smooth_max_and_grad(a_z, beta)
        # Chain rule: d(val)/d(z) = a_z * (grad_c - sum(a_z * grad_c))
        dot = float(np.dot(a_z, grad_c))
        gz = a_z * (grad_c - dot)
        # Subtract mean for gauge invariance
        gz -= gz.mean()
        if not np.all(np.isfinite(gz)):
            return val, np.zeros(n)
        return val, gz

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        val, gz = objective_and_grad(z)
        if not np.all(np.isfinite(gz)):
            break

        res = minimize(objective_and_grad, z, jac=True, method='L-BFGS-B',
                       options={'maxiter': 30, 'ftol': 1e-12, 'gtol': 1e-8})
        if res.success and res.x is not None:
            z_new = res.x
        else:
            # Fallback: simple gradient step
            lr = 0.1 * beta / (np.linalg.norm(gz) + 1e-10)
            z_new = z - lr * gz

        e = np.exp(z_new - z_new.max())
        a_new = e / e.sum()
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        z = z_new

        # Decrementally reduce beta
        if it % 10 == 0:
            beta = max(1e-10, beta * 0.85)

        if no_imp > 50:
            # Reset and try with perturbed beta
            z = best_z + 0.01 * np.random.default_rng(it).normal(size=n)
            beta = max(1e-8, beta_start * 0.5)
            no_imp = 0

    a_final = np.exp(best_z - best_z.max())
    a_final = a_final / a_final.sum()
    return a_final, float(2.0 * n * _fft_convolve(a_final).max())


def _peak_shave(a, deadline, max_iter=10000):
    """Targeted peak shaving: reduce the worst autocorrelation peak."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(a)
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]

        # Compute which elements contribute to this peak
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j]
        # For j in [0, n-1], we need a[peak_idx - j] to be valid
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for j in range(lo, hi + 1):
            other = peak_idx - j
            contrib[j] = a[other]
            contrib[other] += a[j]

        # Reduce elements with high contribution, increase those with low
        total_contrib = contrib.sum()
        if total_contrib < 1e-10:
            break

        # Create update direction
        direction = 1.0 - contrib / (total_contrib / n)
        # Clip to avoid going too negative
        direction = np.clip(direction, -1.0, 1.0)

        # Small step
        step = 0.05
        a_new = a + step * direction
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                step *= 0.8
            if no_imp > 500:
                a = best_a.copy()
                step = 0.05
                no_imp = 0

    return best_a, best_score

def _make_linear_profile(n, eta=1.5):

    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    return _normalize(p)


def _combined_optimize(a0, deadline_remaining, tag=""):
    """Run LP -> Adam -> peak shave -> LP again pipeline."""
    t0 = time.monotonic()
    a = _normalize(a0)
    best_a = a.copy()
    best_score = _score_normalized(a)
    print(f"  [{tag}] Start: score={best_score:.10f}")

    # Phase 1: LP refinement (40% of time)
    lp_deadline = t0 + 0.4 * deadline_remaining
    a_lp, score_lp = _lp_refine_aggressive(a, lp_deadline, max_iter=30000,
                                            inflate=1.0002, blend=0.12)
    if score_lp < best_score:
        best_score = score_lp
        best_a = a_lp.copy()
    print(f"  [{tag}] After LP: score={best_score:.10f}")

    # Phase 2: Adam softmax (30% of time)
    adam_deadline = t0 + 0.7 * deadline_remaining
    a_ad, score_ad = _adam_softmax(best_a, adam_deadline, max_iter=30000, lr=0.03)
    if score_ad < best_score:
        best_score = score_ad
        best_a = a_ad.copy()
    print(f"  [{tag}] After Adam: score={best_score:.10f}")

    # Phase 3: Peak shave (10% of time)
    ps_deadline = t0 + 0.8 * deadline_remaining
    a_ps, score_ps = _peak_shave(best_a, ps_deadline, max_iter=5000)
    if score_ps < best_score:
        best_score = score_ps
        best_a = a_ps.copy()
    print(f"  [{tag}] After PeakShave: score={best_score:.10f}")

    # Phase 4: Final LP (20% of time)
    a_final_lp, score_final = _lp_refine_aggressive(best_a, t0 + deadline_remaining,
                                                     max_iter=20000, inflate=1.0001, blend=0.08)
    if score_final < best_score:
        best_score = score_final
        best_a = a_final_lp.copy()
    print(f"  [{tag}] After Final LP: score={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase, multi-length optimization combining LP spreading,
    Adam in softmax space, peak shaving, and L-BFGS-B.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Try multiple lengths
    # The parent length is most likely good, but try others too
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(parent.size)
    for ln in [384, 448, 512, 576, 640, 768]:
        if ln not in lengths_to_try:
            lengths_to_try.append(ln)
    lengths_to_try.sort()

    print(f"Lengths to try: {lengths_to_try}")

    # Time per length: give more time to parent length
    n_lengths = len(lengths_to_try)
    time_per_length = (deadline - t_start) / n_lengths * 0.8  # leave 20% for final polish

    for idx, n_t in enumerate(lengths_to_try):
        elapsed = time.monotonic() - t_start
        remaining = deadline - t_start - elapsed
        if remaining < 60:
            break

        # Allocate time: parent length gets 1.5x, others get 1x
        weight = 1.5 if (parent is not None and n_t == parent.size) else 1.0
        alloc = min(time_per_length * weight, remaining - 30)
        if alloc < 30:
            alloc = min(60, remaining - 10)
        if alloc < 15:
            break

        # Create initial sequence
        if parent is not None:
            a0 = _interpolate_profile(parent, n_t)
        else:
            a0 = _make_linear_profile(n_t, 1.5)

        sc0 = _score(a0)
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        print(f"\n=== Length {n_t} (remaining={remaining:.0f}s, alloc={alloc:.0f}s) ===")
        print(f"  Initial score: {sc0:.10f}")

        # Run combined optimization
        a_opt, score_opt = _combined_optimize(a0, alloc, tag=f"n={n_t}")

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"  *** NEW BEST: {best_score:.10f} at n={n_t}")

    # Try a few diverse initializations at the best length
    if best_seq is not None and time.monotonic() < deadline - 60:
        best_n = best_seq.size
        print(f"\n=== Diverse inits at n={best_n} ===")

        inits = []
        for eta in [0.8, 1.0, 1.5, 2.0, 2.5, 3.0]:
            inits.append(_make_linear_profile(best_n, eta))
        for alpha in [0.5, 1.0, 1.5, 2.0]:
            inits.append(_make_quadratic_profile(best_n, alpha))
        for a1, a3 in [(1.0, -0.5), (2.0, -1.5), (2.5, -2.0)]:
            inits.append(_make_cubic_profile(best_n, a1, a3))

        # Perturbed parent
        if parent is not None:
            p_int = _interpolate_profile(parent, best_n)
            inits.append(p_int)
            for _ in range(2):
                p_pert = p_int * (1.0 + 0.15 * rng.normal(size=best_n))
                p_pert = np.maximum(p_pert, 0)
                inits.append(_normalize(p_pert))

        time_per_init = min(40.0, (deadline - time.monotonic() - 30) / len(inits))
        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            if time_per_init < 10:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Quick optimization: LP + Adam
            alloc = time_per_init
            a_opt, score_opt = _combined_optimize(a0, alloc, tag=f"init{i}")
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"  *** NEW BEST: {best_score:.10f} (init {i})")

    # Final polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 10:
        final_time = deadline - time.monotonic() - 5
        if final_time > 10:
            print(f"\n=== Final polish (time={final_time:.0f}s) ===")
            a_final, score_final = _lp_refine_aggressive(
                _normalize(best_seq), deadline - 5,
                max_iter=50000, inflate=1.0001, blend=0.05)
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  Final LP: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(512, 1.5)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as ex:
        print(f"  Eval error: {ex}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
