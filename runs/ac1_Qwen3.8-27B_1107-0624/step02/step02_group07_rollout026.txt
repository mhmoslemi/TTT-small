def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    N = 1
    while N < 2 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    conv = np.fft.irfft(S * S, N)[:2 * n - 1]
    return 2.0 * n * np.max(conv) / (total * total)


def full_conv(s):
    """Compute full linear convolution of s with itself."""
    n = len(s)
    N = 1
    while N < 2 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    return np.fft.irfft(S * S, N)[:2 * n - 1]


def compute_gradient(s, conv, sharpness=20.0):
    """Compute weighted gradient focusing on peaks."""
    n = len(s)
    conv_max = np.max(conv)
    if conv_max <= 1e-15:
        return np.zeros(n)

    # Sharp exponential weighting on peaks
    weights = np.exp(sharpness * (conv - conv_max) / conv_max)
    w_sum = np.sum(weights)
    if w_sum < 1e-15:
        return np.zeros(n)
    weights /= w_sum

    # Cross-correlation: grad[j] = sum_k weights[k] * s[k-j] (valid indices)
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.fft.rfft(weights, N)
    S = np.fft.rfft(s, N)
    grad = np.fft.irfft(W * np.conj(S), N)[:n]
    return np.maximum(grad, 0.0)


def multiplicative_step(s, eta, sharpness=20.0):
    """One multiplicative gradient step."""
    n = len(s)
    conv = full_conv(s)
    grad = compute_gradient(s, conv, sharpness)
    gmax = np.max(grad)
    if gmax < 1e-15:
        return s.copy(), conv
    s_new = s * np.exp(-eta * grad / gmax)
    s_new = np.maximum(s_new, 1e-15)
    s_new = s_new / np.sum(s_new)
    return s_new, conv


def coordinate_descent_step(s, conv, max_coords=20):

    n = len(s)

def spectral_shape(s, alpha=0.1):
    """Apply spectral shaping to flatten autocorrelation."""
    n = len(s)
    N = 1
    while N < 2 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    # Reduce peaks in |S|^2 to flatten autocorrelation
    power = np.abs(S) ** 2
    pmax = np.max(power)
    if pmax < 1e-15:
        return s.copy()
    # Gentle compression of dynamic range
    target = pmax ** (1 - alpha) * (power ** alpha)
    # Scale to preserve energy
    scale = np.sqrt(np.sum(power) / np.sum(target))
    new_power = target * scale
    # Apply to magnitude, preserve phase
    phase = np.angle(S)
    S_new = np.sqrt(new_power) * np.exp(1j * phase)
    s_new = np.fft.irfft(S_new, N)[:n]
    s_new = np.maximum(s_new, 1e-10)
    s_new = s_new / np.sum(s_new)
    return s_new


def lp_direction(s, M, time_limit=2.0):
    """LP: maximize sum(b) s.t. conv(s,b)[k] <= M, b >= 0."""
    n = len(s)
    if n > 400:
        return None
    # Build constraint matrix: A[k, j] = s[k-j] if valid, else 0
    # conv(s, b)[k] = sum_j s[j] * b[k-j]... actually np.convolve(s, b)[k] = sum_j s[j]*b[k-j]
    # So A[k, j] = s[k-j] if 0 <= k-j < n, else 0
    # Wait, np.convolve(s, b)[k] = sum_j s[j] * b[k-j] where b is indexed
    # Actually: (s*b)[k] = sum_j s[j] b[k-j]
    # So for fixed k, the coefficient of b[m] is s[k-m] (if 0<=k-m<n)
    # A[k, m] = s[k-m] if 0 <= k-m < n, else 0
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        # b[m] appears in conv[k] for k = m, m+1, ..., m+n-1
        # with coefficient s[k-m]
        for k in range(m, m + n):
            A[k, m] = s[k - m]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            conv = full_conv(s)
            M = np.max(conv)
            b = lp_direction(s, M, time_limit=1.5)
            if b is not None and np.sum(b) > 1e-10:
                b = b / np.sum(b)
                # Line search
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                    cand = (1 - t) * s + t * b
                    cand = np.maximum(cand, 1e-15)
                    cand = cand / np.sum(cand)
                    sc_lp = fast_score(cand)
                    if sc_lp < best_score - 1e-12:
                        best_score = sc_lp
                        best_s = cand.copy()
                        s = cand.copy()
                        no_improve = 0
                        cur_score = sc_lp
                        break

        # Spectral shaping every 200 iterations
        if it % 200 == 0:
            s_spec = spectral_shape(s, alpha=0.05)
            sc_spec = fast_score(s_spec)
            if sc_spec < best_score - 1e-8:
                best_score = sc_spec
                best_s = s_spec.copy()
                s = s_spec.copy()
                no_improve = 0

        # Adaptive parameter tuning
        if no_improve >= patience:
            eta *= 0.7
            sharpness = min(sharpness * 1.3, 60.0)
            no_improve = 0
            if eta < 0.03:
                # Restart with perturbation
                s = s + 0.03 * rng.standard_normal(n)
                s = np.maximum(s, 0.005)
                s = s / np.sum(s)
                eta = 0.4
                sharpness = rng.choice([10.0, 15.0, 25.0, 40.0])
                no_improve = 0

        # Gradually increase sharpness
        if it % 500 == 0:
            sharpness = min(sharpness + 2.0, 50.0)

    return best_s, best_score


def generate_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    s = np.ones(n)
    inits.append(s / s.sum())

    # Linear ramps (various slopes)
    x = np.arange(n) / max(n - 1, 1)
    for slope in [-2, -1.5, -1, -0.5, 0.5, 1, 1.5, 2]:
        s = 1.0 + slope * (2 * x - 1)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Quadratic
    for a in [-2, -1, 1, 2]:
        s = 1.0 + a * (2 * x - 1) ** 2
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Gaussian bumps
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3, 0.4]:
            s = np.exp(-((x - center) ** 2) / (2 * width ** 2))
            s = np.maximum(s, 0.01)
            inits.append(s / s.sum())

    # Two-level steps
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for ratio in [0.3, 0.5, 1.5, 2.0, 3.0]:
            s = np.ones(n)
            k = int(frac * n)
            s[:k] = ratio
            s = np.maximum(s, 0.01)
            inits.append(s / s.sum())

    # Cosine windows
    for order in [1, 2, 3, 4]:
        s = np.cos(np.pi * x * order / 2) ** 2
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Random
    for _ in range(8):
        s = rng.random(n) + 0.05
        inits.append(s / s.sum())

    # Smooth random (low-pass filtered)
    for _ in range(4):
        s = rng.random(n)
        # Simple moving average smoothing
        k = max(3, n // 20)
        kernel = np.ones(k) / k
        s = np.convolve(s, kernel, mode='same')
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Exponential decay
    for rate in [1, 2, 3, 5]:
        s = np.exp(-rate * x)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())
        s2 = np.exp(-rate * (1 - x))
        s2 = np.maximum(s2, 0.01)
        inits.append(s2 / s2.sum())

    return inits


def try_load_height_sequence():
    """Try to load height_sequence_1 if available."""
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            return h
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(s,s))/(sum(s))^2.
    Uses multiplicative updates, coordinate descent, spectral shaping,
    LP guidance, and diverse initialization across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 60.0

    global_best_score = float('inf')
    global_best_seq = np.ones(256)

    # Try to load previous best
    hseq = try_load_height_sequence()
    if hseq is not None:
        print(f"Loaded height_sequence_1, length={hseq.size}")
        sc = fast_score(hseq)
        print(f"  Initial score: {sc:.6f}")
        if sc < global_best_score:
            global_best_score = sc
            global_best_seq = hseq.copy()

    # Sequence lengths to try
    lengths = [128, 192, 256, 384, 512, 640]

    for n in lengths:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        print(f"\n=== n={n}, remaining time: {remaining:.0f}s ===")

        # Generate initializations
        inits = generate_initializations(n, rng)

        # Add rescaled height_sequence_1 if available
        if hseq is not None and len(hseq) != n:
            x_old = np.linspace(0.0, 1.0, len(hseq))
            x_new = np.linspace(0.0, 1.0, n)
            h_r = np.interp(x_new, x_old, hseq)
            h_r = np.maximum(h_r, 0.0)
            if h_r.sum() > 0.01:
                inits.insert(0, h_r / h_r.sum())
        elif hseq is not None and len(hseq) == n:
            inits.insert(0, hseq / hseq.sum())

        # Also add a "smoothed random" based on the best found so far
        if len(global_best_seq) != n:
            x_old = np.linspace(0.0, 1.0, len(global_best_seq))
            x_new = np.linspace(0.0, 1.0, n)
            g_r = np.interp(x_new, x_old, global_best_seq)
            g_r = np.maximum(g_r, 0.0)
            if g_r.sum() > 0.01:
                inits.insert(0, g_r / g_r.sum())

        # Score all initializations and sort
        scored_inits = []
        for s_init in inits:
            s_init = np.maximum(s_init, 0.0)
            s_init = s_init / s_init.sum()
            sc = fast_score(s_init)
            scored_inits.append((sc, s_init))
        scored_inits.sort(key=lambda x: x[0])

        # Select top candidates for optimization
        top_k = min(5, len(scored_inits))
        time_per = remaining / (top_k + 2)  # +2 for restarts

        for i in range(top_k):
            if time.monotonic() > deadline - 30:
                break
            sc0, s0 = scored_inits[i]
            # Adjust time allocation: more time for better candidates
            time_factor = max(0.3, 1.0 - i * 0.15)
            cand_time = min(time_per * time_factor, remaining - 60)
            cand_deadline = time.monotonic() + max(10.0, cand_time)

            use_lp = (n <= 350 and i == 0)
            print(f"  Init {i}: score={sc0:.6f}, time={cand_time:.0f}s, lp={use_lp}")

            best_s, best_sc = optimize_sequence(s0, cand_deadline, rng, use_lp=use_lp)

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** New global best: {best_sc:.6f} (n={n}, init={i})")

        # Random restarts
        restarts_done = 0
        while time.monotonic() < deadline - 60 and restarts_done < 3:
            s_r = rng.random(n) + 0.05
            s_r = s_r / s_r.sum()
            restart_deadline = time.monotonic() + min(30, deadline - time.monotonic() - 30)
            if restart_deadline > time.monotonic() + 5:
                best_s, best_sc = optimize_sequence(s_r, restart_deadline, rng, use_lp=False)
                restarts_done += 1
                if best_sc < global_best_score:
                    global_best_score = best_sc
                    global_best_seq = best_s.copy()
                    print(f"  ** Restart {restarts_done}: new best {best_sc:.6f}")

    # Final refinement on the best found
    if time.monotonic() < deadline - 20:
        print(f"\n=== Final refinement, current best: {global_best_score:.6f} ===")
        n_best = len(global_best_seq)
        refine_deadline = min(time.monotonic() + 60, deadline - 10)
        use_lp = n_best <= 350
        best_s, best_sc = optimize_sequence(global_best_seq, refine_deadline, rng, use_lp=use_lp)
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s.copy()
            print(f"  ** Refined: {best_sc:.6f}")

    # Verify with actual evaluator
    final_seq = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_score = float(evaluate_sequence(final_seq))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(final_seq)}")

    return final_seq
