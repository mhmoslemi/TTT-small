Adaptive smooth-descent with LP acceleration for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient computation, LP-based large moves, and smooth-max gradient polish.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def full_convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
    else:
        N = next_pow2(2 * n - 1)
        fa = rfft(a, N)
        c = irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def smooth_max_grad(a, beta):
    """
    Compute (smooth_max_value, weights, gradient) for smooth max of conv(a,a).
    gradient[m] = 2 * sum_k w[k] * a[k-m] (cross-correlation structure)

    n = a.size
    c = full_convolve_fft(a)

    mval = float(c.max())

    ex = np.exp(x)
    s = float(ex.sum())
    w = ex / s
    val = mval + beta * float(np.log(s))
    # Gradient: g[m] = 2 * sum_k w[k] * a[k-m]
    # = 2 * cross_correlation(a, w)[m]
    # cross_corr[m] = sum_j a[j] * w[j+m]

    cross = irfft(fa_rev * fw, N)

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):

def lp_descent(f, deadline, rng, verbose=False):

                        t *= 0.5
                    else:

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)

def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append((f"lin_{eta}", a / a.sum()))
    
    # Cosine profiles
    for k in [1, 2, 3, 5]:
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * k * t)
        a = np.maximum(a, 1e-12)
        profiles.append((f"cos_{k}", a / a.sum()))
    
    # Power profiles
    for pwr in [0.5, 1.0, 2.0, 3.0]:
        a = t ** pwr
        a = np.maximum(a, 1e-12)
        profiles.append((f"pow_{pwr}", a / a.sum()))
        # Also reversed
        profiles.append((f"rpow_{pwr}", a[::-1] / a.sum()))
    
    # Gaussian-like
    for sigma in [0.2, 0.3, 0.5]:
        center = 0.5
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        profiles.append((f"gauss_{sigma}", a / a.sum()))
    
    # Random smooth
    for i in range(6):
        freqs = rng.uniform(1, 8, 4).astype(int)
        amps = rng.uniform(0.1, 0.4, 4)
        phases = rng.uniform(0, 2 * np.pi, 4)
        a = np.ones(n)
        for f, amp, ph in zip(freqs, amps, phases):
            a += amp * np.cos(2 * np.pi * f * t + ph)
        a = np.maximum(a, 1e-12)
        profiles.append((f"rand_{i}", a / a.sum()))
    
    # Alternating-like (high frequency)
    a = 1.0 + 0.2 * np.cos(2 * np.pi * (n // 4) * t)
    a = np.maximum(a, 1e-12)
    profiles.append(("alt", a / a.sum()))
    
    return profiles

"""
Multi-stage optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based descent with iterative M-tightening, FFT-based smooth-max gradient
polish in logit space, spectral initialization, and multi-length search.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def full_convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
    else:
        N = next_pow2(2 * n - 1)
        fa = rfft(a, N)
        c = irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def smooth_max_and_grad(a, beta):
    n = a.size
    c = full_convolve_fft(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    w = ex / s
    val = mval + beta * float(np.log(s))
    N = next_pow2(n + 2 * n - 1)
    fa_rev = rfft(a[::-1], N)
    fw = rfft(w, N)
    cross = irfft(fa_rev * fw, N)
    grad = 2.0 * cross[n - 1: 2 * n - 1]
    return val, w, grad


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_descent_tighten(f, deadline, rng, verbose=False):

    s = f.sum()
    if s < 1e-10:
        return f.copy()
    f = f / s
    n = len(f)
    conv_f = full_convolve_fft(f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.15
    tighten_eps = 0.005
    stagnant = 0
    A = build_conv_matrix(f)
    iter_count = 0

    while time.monotonic() < deadline:
        M = best_M
        M_target = M * (1.0 - tighten_eps)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_target)
        bounds = [(0.0, None)] * n
        g = None
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
            else:
                tighten_eps *= 2.0
                tighten_eps = min(tighten_eps, 0.1)
                if res is None or not res.success:
                    b_ub = np.full(2 * n - 1, M)
                    res2 = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
                    if res2.success and res2.x is not None:
                        g = np.maximum(res2.x, 0.0)
                        tighten_eps = max(0.001, tighten_eps * 0.5)
        except Exception:
            b_ub = np.full(2 * n - 1, M)
            try:
                res2 = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
                if res2.success and res2.x is not None:
                    g = np.maximum(res2.x, 0.0)
            except Exception:
                pass

        if g is None or g.sum() < 1e-10:
            stagnant += 1
            if stagnant > 50:
                sizable_eps = 0.01 + 0.01 * (stagnant // 50)
                b_ub = np.full(2 * n - 1, M * (1.0 + sizable_eps))
                try:
                    res3 = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
                    if res3.success and res3.x is not None:

                    t = 0.15
                    stagnant = 0
                    A = build_conv_matrix(f)
                    tighten_eps = 0.005
                else:
                    stagnant = 0
            if g is None or g.sum() < 1e-10:
                continue
        else:
            stagnant = 0

        g = g / g.sum()
        f_new = (1 - t) * f + t * g

        iter_count += 1
        if verbose and iter_count % 80 == 0:

def logit_adam_polish(p0, deadline, rng, base_lr=0.05, beta_factor=0.1):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n = p0.size
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_factor * maxc)
    best_p = p0.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    iter_count = 0
    cur_lr = base_lr

    while time.monotonic() < deadline:
        val, w, grad = smooth_max_and_grad(p, beta)
        dot = float(np.dot(p, grad))
        gz = p * (grad - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            z = np.log(np.maximum(best_p, 1e-15))
            z -= float(np.mean(z))
            m.fill(0); v.fill(0); t_adam = 0
            continue
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - cur_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        exp_z = np.exp(z)
        p = exp_z / exp_z.sum()
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 300 and no_imp % 100 == 0:
                cur_lr = max(0.005, cur_lr * 0.93)
        if iter_count % 40 == 0:
            beta = max(1e-14, beta * 0.75)
        if no_imp > max(800, 4 * n):
            z = np.log(np.maximum(best_p, 1e-15))
            z += 0.25 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            exp_z = np.exp(z)
            p = exp_z / exp_z.sum()
            c = full_convolve_fft(p)
            beta = max(1e-12, 0.12 * float(c.max()))
            m.fill(0); v.fill(0); t_adam = 0
            cur_lr = base_lr
            no_imp = 0
        iter_count += 1
        if iter_count % 5000 == 0:
            print(f"      adam iter={iter_count}, best={best_score:.10f}, beta={beta:.2e}")

    return best_p, best_score


def subgradient_polish(p0, deadline, rng, lr_init=0.01):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n = p0.size
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    p = p0.copy()
    best_p = p0.copy()
    c0 = full_convolve_fft(p0)
    best_M = float(c0.max())
    best_score = 2.0 * n * best_M
    lr = lr_init
    iter_count = 0
    no_imp = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(p)
        k_max = int(np.argmax(c))
        M = float(c[k_max])
        if M < best_M - 1e-14:
            best_M = M
            best_score = 2.0 * n * M
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
        grad_k = np.zeros(n)
        for m in range(n):
            j = k_max - m
            if 0 <= j < n:
                grad_k[m] = 2.0 * p[j]
        dot = float(np.dot(p, grad_k))
        grad_tan = grad_k - dot
        p_new = p - lr * grad_tan
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-10:
            lr *= 0.5
            continue
        p_new = p_new / s
        c_new = full_convolve_fft(p_new)
        M_new = float(c_new.max())
        if M_new <= best_M + 1e-12:
            p = p_new
            lr = min(lr * 1.02, lr_init)
        else:
            lr *= 0.7
            if lr < 1e-6:
                lr = 1e-6
        if no_imp > 2000:
            z = np.log(np.maximum(best_p, 1e-15))
            z += 0.1 * rng.normal(size=n)
            z -= float(np.mean(z))
            exp_z = np.exp(np.clip(z, -30, 30))
            p = exp_z / exp_z.sum()
            lr = lr_init * 0.5
            no_imp = 0
        iter_count += 1
        if iter_count % 10000 == 0:
            print(f"      subgrad iter={iter_count}, best={best_score:.10f}")

    return best_p


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def make_profiles(n, rng):
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append((f"lin_{eta}", a / a.sum()))
    for k in [1, 2, 3, 5]:
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * k * t)
        a = np.maximum(a, 1e-12)
        profiles.append((f"cos_{k}", a / a.sum()))
    for pwr in [0.5, 1.0, 2.0]:
        a = t ** pwr
        a = np.maximum(a, 1e-12)
        profiles.append((f"pow_{pwr}", a / a.sum()))
    for sigma in [0.15, 0.25, 0.4]:
        a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
        a = np.maximum(a, 1e-12)
        profiles.append((f"gauss_{sigma}", a / a.sum()))
    for i in range(8):
        freqs = rng.uniform(1, 10, 5).astype(int)
        amps = rng.uniform(0.05, 0.35, 5)
        phases = rng.uniform(0, 2 * np.pi, 5)
        a = np.ones(n)
        for fq, amp, ph in zip(freqs, amps, phases):
            a += amp * np.cos(2 * np.pi * fq * t + ph)
        a = np.maximum(a, 1e-12)
        profiles.append((f"rand_{i}", a / a.sum()))
    # Spectral init: flat-top via GS iteration
    for trial in range(4):
        spec = np.ones(2 * n - 1)
        phases = rng.uniform(0, 2 * np.pi, 2 * n - 1)
        a_spec = spec * np.exp(1j * phases)
        sig = np.fft.irfft(a_spec, 2 * n - 1)[:n]
        a_gs = np.maximum(sig, 1e-12)
        a_gs = a_gs / a_gs.sum()
        profiles.append((f"gs_{trial}", a_gs.copy()))
        # Run a few GS iterations
        for _ in range(20):
            A = np.fft.rfft(a_gs, 2 * n - 1)
            mag = np.abs(A)
            phase = np.angle(A)
            A_flat = np.ones_like(mag) * phase
            a_gs = np.fft.irfft(A_flat, 2 * n - 1)[:n]
            a_gs = np.maximum(a_gs, 1e-12)
            a_gs = a_gs / a_gs.sum()
        profiles.append((f"gs_it_{trial}", a_gs.copy()))
    return profiles

        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        d = min(phase4_end - 10, time.monotonic() + 200.0)
        if time.monotonic() < d:
            result = lp_descent_tighten(p0, d, rng, verbose=True)
            sc = compute_score(result)
            update(result, sc)
            print(f"After extended LP: score={sc:.10f}")
            # Additional polish
            am_d4 = min(phase4_end - 5, time.monotonic() + 40.0)
            if time.monotonic() < am_d4:
                p0 = result + 0.0003 * rng.normal(size=result.size)
                p0 = np.maximum(p0, 1e-12)
                p0 = p0 / p0.sum()
                res_am, _ = logit_adam_polish(p0, am_d4, rng, base_lr=0.02)
                sc = compute_score(res_am)
                update(res_am, sc)
                print(f"After final Adam: score={sc:.10f}")

    # Phase 5: Final micro-polish
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_f = best_seq.size
        p0 = best_seq + 0.0002 * rng.normal(size=n_f)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        d = deadline - 5
        res_am, _ = logit_adam_polish(p0, d, rng, base_lr=0.01)
        sc = compute_score(res_am)
        update(res_am, sc)
        print(f"Final micro-polish: score={sc:.10f}")

    print(f"FINAL best: {best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = np.ones(256) / 256

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(256) / 256
    else:
        final_seq = final_seq / s

    final_score = compute_score(final_seq)
    print(f"CONFIRMED: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"eval error: {e}")

    return [float(x) for x in final_seq]
