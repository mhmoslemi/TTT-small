from scipy.optimize import linprog
from scipy.special import i0
import time
import math

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding, Adam polish with soft-max, 
multi-profile initialization, and multi-length search.

def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            A[k, lo:hi+1] = f[k - np.arange(lo, hi + 1)]
    return A

def convolve_max(f):
    """Compute max of linear convolution of f with itself."""
    c = np.convolve(f, f)
    return float(c.max())

def compute_score_normalized(f):
    """Compute 2*n*M assuming sum(f) = 1."""
    n = len(f)
    M = convolve_max(f)
    return 2.0 * n * M

def compute_score(f):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    M = convolve_max(f)
    return 2.0 * n * M / (s * s)

def normalize(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s

def profile_uniform(n):
    return np.ones(n) / n

def profile_linear(n, slope=1.0):
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    f = 1.0 + slope * t
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def profile_cosine(n):
    t = np.linspace(0, 1, n)
    f = np.cos(np.pi * t / 2.0) ** 2
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def profile_u_shape(n, alpha=1.0):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    f = 1.0 + alpha * u ** 2
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def profile_v_shape(n, alpha=1.0):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    f = 1.0 - alpha * (1 - u ** 2)  # = 1 - alpha + alpha*u^2
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def profile_gaussian(n, sigma=0.3):
    t = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    f = np.maximum(f, 1e-10)

def profile_power(n, p=0.5):
    t = np.linspace(0, 1, n)
    f = t ** p + (1 - t) ** p + 1e-10
    return f / f.sum()

    f = i0(beta * np.sqrt(np.maximum(1 - t**2, 0)))
    f = np.maximum(f, 1e-10)

    t = np.linspace(0, 1, n)
    f = np.ones(n) * 0.5

        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        f += a * np.cos(2 * np.pi * k * t) + b * np.sin(2 * np.pi * k * t)

    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)

def solve_lp_for_direction(f, M, time_limit=3.0):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def soft_max_and_weights(c, beta):
    """Compute soft max and its weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w

def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j] using FFT."""
    n = len(p)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def adam_optimize(f_init, deadline, rng, lr=0.04, max_iters=10000):
    """Adam optimization in logit space with soft-max surrogate."""
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    c = np.convolve(f, f)
    M = float(c.max())
    beta = max(1e-12, 0.1 * M)
    
    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        L, w = soft_max_and_weights(c, beta)
        gp = 2.0 * corr_weights(w, f)
        vp = gp + 1e-8 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z_new)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        z_new = np.clip(z_new, -50, 50)
        z_new -= z_new.mean()
        e = np.exp(z_new)
        f_new = e / e.sum()
        c_new = np.convolve(f_new, f_new)
        M_new = float(c_new.max())
        
        if M_new < best_M - 1e-12:
            z = z_new
            f = f_new
            c = c_new
            M = M_new
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            # Rejected step, still update z slightly for exploration
            z = z_new
            f = f_new
            c = c_new
            M = M_new
            no_imp += 1
        
        if adam_t % 100 == 0:
            beta = max(1e-12, beta * 0.85)
        
        if no_imp > 200:
            lr = max(0.01, lr * 0.9)
        
        if no_imp > max(500, 2 * n):
            # Restart
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.1 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_f, best_M

def subgradient_optimize(f_init, deadline, rng, max_iters=5000):
    """Simple subgradient method on the max function."""
    f = normalize(f_init.copy())
    if f is None:
        return f_init, convolve_max(f_init)
    
    n = len(f)
    best_f = f.copy()
    best_M = convolve_max(f)
    lr = 0.1
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(f, f)
        k_max = int(np.argmax(c))
        
        # Subgradient of max_k (f*f)_k with respect to f
        # d/d f_j [(f*f)_{k_max}] = f_{k_max - j} (for valid j)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_max - j
            if 0 <= idx < n:
                grad[j] = 2 * f[idx]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Step
        step = lr / (it + 1) ** 0.5
        f_new = f - step * grad
        f_new = normalize(f_new)
        if f_new is None:
            continue
        
        M_new = convolve_max(f_new)
        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
        
        f = f_new
        lr = max(0.01, lr * 0.999)
    
    return best_f, best_M

    phase3_results = []
    for i in range(top_k3):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_k3 - i), 50.0)
        if alloc < 5:
            break
        
        sc, n, f = phase2_results[i]
        f_best, M_best = adam_optimize(f, time.monotonic() + alloc, rng, lr=0.04)
        sc_best = 2.0 * n * M_best
        phase3_results.append((sc_best, n, f_best))
        try_update(f_best, sc_best)
        print(f"  Phase 3 Adam {i+1}/{top_k3} (n={n}): score={sc_best:.10f}, best={best_score:.10f}")
    
    phase3_results.sort(key=lambda x: x[0])
    
    # Phase 4: Combined LP+Adam alternation on the very best
    phase4_deadline = deadline
    if best_seq is not None:
        remaining = phase4_deadline - time.monotonic()
        if remaining > 30:
            n_best = len(best_seq)
            # Alternate LP and Adam
            f_current = best_seq.copy()
            M_current = convolve_max(f_current)
            
            for phase in range(4):
                if time.monotonic() > deadline - 5:
                    break
                sub_deadline = min(deadline - 5, time.monotonic() + remaining / 4)
                
                if phase % 2 == 0:
                    f_best, M_best = lp_optimize(f_current, sub_deadline, rng, max_iters=1000)
                else:
                    f_best, M_best = adam_optimize(f_current, sub_deadline, rng, lr=0.03)
                
                sc = 2.0 * n_best * M_best
                try_update(f_best, sc)
                f_current = f_best
                M_current = M_best
                print(f"  Phase 4 round {phase+1}: score={sc:.10f}, best={best_score:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = profile_linear(256, 1.5)
    
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = profile_linear(256, 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s  # normalize to sum 1
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    
    final_score = compute_score(best_seq)
    print(f"Final: n={len(best_seq)}, score={final_score:.10f}")
    
    # Verify with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
        if np.isfinite(ev_score) and ev_score < final_score:
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")
    
    return [float(x) for x in best_seq]
