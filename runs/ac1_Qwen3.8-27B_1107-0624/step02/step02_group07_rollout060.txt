    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0  # coefficient for t

    # conv(s,d)[k] = sum_i s[i]*d[k-i] (with appropriate bounds)

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""

def solve_direction_lp(s):

    Solve LP to find best descent direction.
    minimize t subject to:
        conv(s,d)[k] <= t  for all k
        sum(d) = 1
        d >= 0
    Returns (d, t_opt) or (None, None).

def qp_direction(s, M, delta):
    """Find d closest to s such that conv(s,d)[k] <= M - delta."""
    n = len(s)
    target = M - delta
    
    def objective(d):
        return np.sum((d - s)**2)
    
    def objective_grad(d):
        return 2.0 * (d - s)
    
    num_ineq = 2 * n - 1
    A = np.zeros((num_ineq, n))
    for j in range(n):
        A[j:j+n, j] = s
    
    constraints = [
        {'type': 'ineq', 'fun': lambda d: target - A @ d, 'jac': lambda d: -A},
        {'type': 'eq', 'fun': lambda d: np.sum(d) - 1.0, 'jac': lambda d: np.ones(n)},
    ]
    
    bounds = [(0.0, 3.0/n)] * n
    
    res = minimize(objective, s, jac=objective_grad, 
                  method='SLSQP', bounds=bounds, constraints=constraints,
                  options={'maxiter': 200, 'ftol': 1e-10})
    
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    """
    LP: min t s.t. conv(s,d)[k] <= t, sum(d)=1, 0<=d[i]<=d_max
    Returns (d, t) or (None, None).

        A[j:j+n, j] = s

    c[n] = 1

"""
LP-based iterative optimization for minimizing 2n*max(conv(s,s))/(sum(s))^2.
Solves a linear program at each step to find the optimal descent direction,
using analytical line search and diverse initializations across multiple
sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_score(s):
    """Compute 2n * max(conv(s,s)) / (sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)


def solve_direction_lp(s, d_max):
    """
    Solve LP: minimize t
    subject to: sum_i s[i]*d[k-i] <= t  for k=0..2n-2
                sum(d) = 1
                0 <= d[i] <= d_max
    Returns (d, t) or (None, None).
    """
    n = len(s)
    m = 2 * n - 1
    # Build constraint matrix
    A = np.zeros((m, n + 1))
    for j in range(n):
        A[j:j + n, j] = s
    A[:, n] = -1.0
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, d_max)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': 30, 'presolve': True})
    
    if res.success:
        d = np.maximum(res.x[:n], 0.0)
        return d, float(res.x[n])
    return None, None


def compute_step_size(s, d, M, t):
    """Analytical step size for s_new = (1-eps)*s + eps*d."""
    conv_d = np.convolve(d, d)
    D = conv_d.max()
    a = M - 2.0 * t + D
    b = 2.0 * t - 2.0 * M
    if a > 1e-15:
        eps = max(0.01, -b / (2.0 * a))
    else:
        eps = 0.3
    return min(eps, 0.9)


def optimize_sequence(s, deadline, verbose=False):
    """
    Main LP-based optimizer. Iteratively finds descent directions via LP
    and takes optimally-sized steps.
    """
    n = len(s)
    s = np.maximum(np.asarray(s, dtype=np.float64), 0.0)
    if s.sum() < 1e-15:
        s = np.ones(n)
    s /= s.sum()
    
    best_s = s.copy()
    best_sc = fast_score(s)
    
    rng = np.random.default_rng(2024)
    stuck = 0
    it = 0
    d_max = 5.0 / n  # Allow entries up to 5x average
    
    while time.monotonic() < deadline:
        d, t = solve_direction_lp(s, d_max)
        if d is None:
            # Try with relaxed bound
            d, t = solve_direction_lp(s, d_max=2.0)
            if d is None:
                break
        
        conv_s = np.convolve(s, s)
        M = conv_s.max()
        
        if t < M - 1e-10:
            # Descent direction found
            eps0 = compute_step_size(s, d, M, t)
            
            # Line search
            best_eps = eps0
            best_trial_sc = best_sc
            
            eps_candidates = np.unique(np.concatenate([
                np.linspace(0.01, max(0.05, eps0), 5),
                np.linspace(max(0.05, eps0), min(0.9, eps0 + 0.4), 5),
            ]))
            
            for ep in eps_candidates:
                s_tr = (1.0 - ep) * s + ep * d
                s_tr = np.maximum(s_tr, 0.0)
                tr_sum = s_tr.sum()
                if tr_sum < 1e-12:
                    continue
                s_tr /= tr_sum
                sc = fast_score(s_tr)
                if sc < best_trial_sc:
                    best_trial_sc = sc
                    best_eps = ep
            
            s = (1.0 - best_eps) * s + best_eps * d
            s = np.maximum(s, 0.0)
            s /= s.sum()
            sc = fast_score(s)
            
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_s = s.copy()
                stuck = 0
            else:
                stuck += 1
                s = best_s.copy()  # Rollback
        else:
            stuck += 3
        
        it += 1
        if verbose and it % 50 == 0:
            print(f"      it={it:4d} cur={fast_score(s):.8f} best={best_sc:.8f} stuck={stuck}")
        
        if stuck > 50:
            # Perturb to escape local optimum
            scale = 0.003 + 0.01 * min(stuck / 200.0, 1.0)
            s = s + scale * rng.standard_normal(n)
            s = np.maximum(s, 0.001)
            s /= s.sum()
            stuck = 0
            it = 0  # Reset counter
    
    return best_s, best_sc
