    t_adam = 0

def peak_targeting(f, M, rng, n_peaks=3, strength=0.05):
    """Directly reduce the top peaks by adjusting contributing elements."""
    n = len(f)
    c = fft_convolve(f, f)
    
    # Find top peaks
    peak_indices = np.argsort(c)[-n_peaks:]
    
    for pk in peak_indices:
        # Find which elements contribute to this peak
        # c[pk] = sum_i f[i]*f[pk-i]
        lo = max(0, pk - n + 1)
        hi = min(n - 1, pk)
        # Contribution of f[i] to c[pk] is f[i]*f[pk-i] (for valid pk-i)
        # Gradient w.r.t. f[i] is f[pk-i] (plus f[i] from the other side if pk-2i is valid)
        
        # Reduce elements that have high contribution
        contributions = np.zeros(n)
        for i in range(lo, hi+1):
            j = pk - i
            if 0 <= j < n:
                contributions[i] += f[j]
        
        # Reduce top contributors
        top_k = max(1, n // 10)
        top_idx = np.argsort(contributions)[-top_k:]
        
        # Shift mass from top contributors to bottom non-contributors
        bottom_idx = np.argsort(contributions)[:top_k]
        
        total_shift = strength * f[top_idx].sum()
        per_shift = total_shift / len(top_idx)
        
        for i in top_idx:
            f[i] -= per_shift
        for i in bottom_idx:
            f[i] += per_shift / max(1, len(bottom_idx) / max(1, len(top_idx)))
    
    f = np.maximum(f, 1e-12)
    f = f / f.sum()
    return f

def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)

def try_poly_search(n, deadline, rng):
    """Search over polynomial parameters to find good shapes."""
    best_f = None
    best_score = np.inf
    
    # Try different polynomial degrees and random initial parameters
    for degree in [2, 3, 4, 5]:
        for trial in range(50):
            if time.monotonic() > deadline:
                return best_f, best_score
            
            params = rng.uniform(-3, 3, degree)
            f = parametric_search(params, n)
            sc = compute_score(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
            
            # Local refinement of parameters using Nelder-Mead
            if sc < best_score + 0.1:
                def obj(p):
                    f_p = parametric_search(p, n)
                    return compute_score(f_p)
                
                res = minimize(obj, params, method='Nelder-Mead', 
                             options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-10})
                if res.fun < best_score:
                    best_score = res.fun
                    best_f = parametric_search(res.x, n).copy()
    
    return best_f, best_score
