Hybrid gradient/LP optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive-temperature log-sum-exp gradient descent with periodic LP direction

import numpy as np

from scipy.optimize import linprog
from scipy.signal import fftconvolve as _fftconvolve


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    """Full linear convolution a*a via FFT, clamped non-negative."""
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _compute_score(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = _fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _log_sum_exp(x, beta):
    """Compute log-sum-exp of x/beta, returning (value, weights)."""
    y = x / beta
    y_max = y.max()
    y_shifted = y - y_max
    ex = np.exp(np.clip(y_shifted, -100.0, 50.0))
    s = float(ex.sum())
    if s < 1e-30:
        ex = np.zeros_like(x)
        ex[int(np.argmax(x))] = 1.0
        s = 1.0
    w = ex / s
    val = float(y_max + np.log(s)) * beta
    return val, w


def _gradient_lse(a, beta):

    Returns (loss_value, gradient_array_of_length_n).

    mean_grad = float(np.dot(a, grad))
    return val, grad


def _adam_update(param, grad, state, lr, t):
    """Adam optimizer step. param and state are numpy arrays/dicts."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    if 'm' not in state:
        state['m'] = np.zeros_like(param)
        state['v'] = np.zeros_like(param)
    state['m'] = b1 * state['m'] + (1.0 - b1) * grad
    state['v'] = b2 * state['v'] + (1.0 - b2) * (grad * grad)
    m_hat = state['m'] / (1.0 - b1 ** t)
    v_hat = state['v'] / (1.0 - b2 ** t)
    new_param = param - lr * m_hat / (np.sqrt(v_hat) + eps)
    return new_param

def _make_smooth_random(n, rng, n_harm=6):
    """Generate a random smooth non-negative profile."""

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    a = np.maximum(a, 1e-10)

def _interpolate_to_length(a, n_new):
    """Interpolate sequence to new length, maintaining sum=1."""
    n_old = a.size
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return None
    return a_new / s


def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    starts.append(_make_profile(n1, 'linear', {'eta': 2.5}))
    starts.append(_make_profile(n1, 'cubic', {'a1': 1.2, 'a3': -0.6}))
    starts.append(_make_profile(n1, 'cubic', {'a1': 1.5, 'a3': -0.8}))
    starts.append(_make_profile(n1, 'cubic', {'a1': 2.0, 'a3': -1.0}))
    starts.append(_make_profile(n1, 'parabola'))
    starts.append(_make_profile(n1, 'gaussian', {'sigma': 0.4}))
    
    # Random smooth profiles
    for _ in range(6):
        starts.append(_make_smooth_random(n1, rng))
    
    candidates_512 = []
    per_start = max(3.0, (phase1_end - t_start) / max(1, len(starts)))
    
    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end - 10:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s
        
        per = min(phase1_end - 5, time.monotonic() + per_start)
        p_best, M_best = _gradient_descent_lse(p0, per, rng, lr_init=0.06,
                                                beta_every=80, restart_every=300,
                                                max_iter=50000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates_512.append((sc, p_best))
        if i % 3 == 0:
            print(f"  Phase 1 start {i}/{len(starts)}: best={best_score:.10f}")
    
    candidates_512.sort(key=lambda x: x[0])
    print(f"After Phase 1 (n={n1}): best={best_score:.10f}")
    
    # Phase 2: Deeper gradient + periodic LP on top candidates
    phase2_end = min(deadline, t_start + 1200.0)
    top_k = min(5, len(candidates_512))
    
    for i in range(top_k):
        if time.monotonic() > phase2_end - 30:
            break
        _, p0 = candidates_512[i]
        
        # Hybrid: gradient with periodic LP
        deadline_sub = min(phase2_end - 10, time.monotonic() + 180.0)
        a_curr = p0.copy()
        n = a_curr.size
        N_corr = _next_pow2(2 * n - 1)
        
        M_curr = float(_fft_convolve(a_curr).max())
        best_a_sub = a_curr.copy()
        best_M_sub = M_curr
        
        z = np.log(np.maximum(a_curr, 1e-15)) - np.log(np.maximum(a_curr, 1e-15)).max()
        beta = 0.05 * best_M_sub
        state = {}
        t_adam = 0
        lr = 0.04
        no_imp = 0
        lp_counter = 0
        
        for it in range(200000):
            if time.monotonic() > deadline_sub:
                break
            
            # Every 200 iterations, try LP
            if it % 200 == 0 and lp_counter < 30:
                f_lp, M_lp = _lp_step(best_a_sub, deadline_sub, rng)
                lp_counter += 1
                if f_lp is not None and M_lp < best_M_sub - 1e-10:
                    best_a_sub = f_lp.copy()
                    best_M_sub = M_lp
                    M_curr = M_lp
                    a_curr = f_lp.copy()
                    z = np.log(np.maximum(a_curr, 1e-15)) - np.log(np.maximum(a_curr, 1e-15)).max()
                    beta = 0.05 * best_M_sub
                    state = {}
                    t_adam = 0
                    no_imp = 0
                    continue
            
            # Gradient step
            c = _fft_convolve(a_curr)
            val, w = _log_sum_exp(c, beta)
            fw = np.fft.rfft(w, N_corr)
            fa = np.fft.rfft(a_curr, N_corr)
            h_full = np.fft.irfft(fw * np.conj(fa), N_corr)
            grad_a = h_full[:n]
            mean_g = float(np.dot(a_curr, grad_a))
            grad_z = a_curr * (grad_a - mean_g)
            
            if not np.all(np.isfinite(grad_z)):
                z = np.log(np.maximum(best_a_sub, 1e-15)) - np.log(np.maximum(best_a_sub, 1e-15)).max()
                state = {}
                t_adam = 0
                continue
            
            t_adam += 1
            lr_t = lr / (1.0 + 0.0005 * t_adam)
            z = _adam_update(z, grad_z, state, lr_t, t_adam)
            z = np.clip(z, -60.0, 60.0)
            z -= z.max()
            
            a_curr = np.exp(z)
            a_curr = np.maximum(a_curr, 1e-15)
            a_curr = a_curr / a_curr.sum()
            
            if it % 20 == 0:
                M_new = float(_fft_convolve(a_curr).max())
                if M_new < best_M_sub - 1e-14:
                    best_M_sub = M_new
                    best_a_sub = a_curr.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            
            if t_adam % 100 == 0:
                beta = max(1e-8, beta * 0.93)
            
            if no_imp > 600:
                # Restart with perturbation
                mix = 0.6 * best_a_sub + 0.4 * _make_smooth_random(n, rng)
                a_curr = np.maximum(mix, 1e-10)
                a_curr = a_curr / a_curr.sum()
                z = np.log(a_curr) - np.log(a_curr).max()
                state = {}
                t_adam = 0
                beta = 0.05 * best_M_sub
                no_imp = 0
                lp_counter = min(lp_counter, 20)  # Allow more LP calls
        
        sc = 2.0 * n * best_M_sub
        try_update(best_a_sub, sc)
        print(f"  Phase 2 cand {i}: score={sc:.10f}")
    
    print(f"After Phase 2: best={best_score:.10f}")
    
    # Phase 3: Try n=1024
    n2 = 1024
    phase3_end = min(deadline, t_start + 1620.0)
    
    if best_seq is not None and time.monotonic() < phase3_end - 60:
        p_ext = _interpolate_to_length(best_seq, n2)
        if p_ext is not None:
            per = min(phase3_end - 30, time.monotonic() + 180.0)
            p_best, M_best = _gradient_descent_lse(p_ext, per, rng, lr_init=0.04,
                                                    beta_every=100, restart_every=400,
                                                    max_iter=100000)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"Phase 3 gradient (n={n2}): score={sc:.10f}")
            
            # Follow up with LP
            if time.monotonic() < phase3_end - 20:
                deadline_sub = min(phase3_end - 10, time.monotonic() + 60.0)
                a_curr = p_best.copy()
                best_a_sub = a_curr.copy()
                best_M_sub = M_best
                lp_count = 0
                for it in range(100):
                    if time.monotonic() > deadline_sub or lp_count >= 15:
                        break
                    f_lp, M_lp = _lp_step(best_a_sub, deadline_sub, rng)
                    lp_count += 1
                    if f_lp is not None and M_lp < best_M_sub - 1e-10:
                        best_a_sub = f_lp.copy()
                        best_M_sub = M_lp
                    else:
                        break
                sc = 2.0 * n2 * best_M_sub
                try_update(best_a_sub, sc)
                print(f"Phase 3 LP (n={n2}): score={sc:.10f}")
    
    # Also try random starts at n=1024
    if time.monotonic() < phase3_end - 30:
        for _ in range(3):
            if time.monotonic() > phase3_end - 20:
                break
            p0 = _make_smooth_random(n2, rng, n_harm=8)
            per = min(phase3_end - 15, time.monotonic() + 60.0)
            p_best, M_best = _gradient_descent_lse(p0, per, rng, lr_init=0.03,
                                                    beta_every=120, restart_every=400,
                                                    max_iter=30000)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
    
    print(f"After Phase 3: best={best_score:.10f}")
    
    # Phase 4: Final polish on best
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 90.0)
        
        # Final LP refinement
        a_curr = best_seq.copy()
        best_a_final = a_curr.copy()
        best_M_final = float(_fft_convolve(a_curr).max())
        
        deadline_sub = min(deadline - 5, time.monotonic() + 60.0)
        for it in range(50):
            if time.monotonic() > deadline_sub:
                break
            f_lp, M_lp = _lp_step(best_a_final, deadline_sub, rng)
            if f_lp is not None and M_lp < best_M_final - 1e-10:
                best_a_final = f_lp.copy()
                best_M_final = M_lp
            else:
                # Small gradient nudge
                break
        
        sc = 2.0 * n_best * best_M_final
        try_update(best_a_final, sc)
        
        # Quick gradient polish
        if time.monotonic() < deadline - 5:
            per2 = min(deadline - 2, time.monotonic() + 30.0)
            p_best, M_best = _gradient_descent_lse(best_a_final, per2, rng,
                                                    lr_init=0.02, beta_init=0.1 * best_M_final,
                                                    beta_every=50, restart_every=200,
                                                    max_iter=20000)
            sc = 2.0 * n_best * M_best
            try_update(p_best, sc)
    
    print(f"After Phase 4: best={best_score:.10f}")
    
    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = _compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")
        
        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
            except Exception as e:
                print(f"Evaluation error: {e}")
        
        return [float(x) for x in final]
    
    # Fallback
    fb = _make_profile(512, 'linear', {'eta': 1.5})
    return [float(x) for x in fb]
