def fft_convolve(a, b):
    """FFT-based convolution for non-negative sequences."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()  # next power of 2
    fa = rfft(a, size)
    fb = rfft(b, size)
    c = irfft(fa * fb, size)
    return c[:n]

def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)

        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_direction(f, M, timeout=10.0):

    return None

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

"""
FFT-accelerated multi-resolution optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses correct analytic gradients in log-space with Adam, LP direction finding at
intermediate resolutions, and progressive upsampling from n=512 to n=2048.

import numpy as np
import time
from scipy.optimize import linprog
from numpy.fft import rfft, irfft


def fft_convolve(a, b):
    """FFT-based full convolution of two sequences."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = rfft(a, size)
    fb = rfft(b, size)
    c = irfft(fa * fb, size)
    return c[:n]


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=8.0):
    """Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g) <= M elementwise."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterate(f, max_iters=200, t_init=0.4, t_min=0.02, deadline=None, verbose=True):
    """LP-based iterations to reduce max(conv(f,f)). f normalized to sum 1."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = fft_convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M
    no_improve = 0
    t_step = t_init

        if deadline is not None and time.monotonic() > deadline:
            break

        if g is None:

        g_self_score = 2.0 * n * float(g_self_c.max())

            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
        else:

            f_new = normalize(f_new)

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_u_profile(n, sharpness):
    """U-shaped profile: high at edges, low in middle."""
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = u ** sharpness + 0.02
    return a / a.sum()


def make_quadratic_profile(n, c1, c2):
    """Quadratic profile: 1 + c1*u + c2*u^2, u = 2t-1."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + c1 * u + c2 * u * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def compute_grad_p(w, p):
    """
    Compute gradient of max(conv(p,p)) w.r.t. p, given soft-max weights w.
    grad_p[i] = 2 * sum_j w[i+j] * p[j]
    Uses FFT for efficiency.
    """
    n = len(p)
    p_rev = p[::-1]
    wp_conv = fft_convolve(w, p_rev)
    h1 = wp_conv[n - 1:n - 1 + n]
    grad_p = 2.0 * h1
    return grad_p


def adam_polish_fft(p0, deadline, lr=0.03, reg=1e-8, verbose=True, beta_init=None,
                    restart_every=2000, pop_size=1):
    """
    Adam gradient descent in logit space with FFT-based convolution.
    Uses correct analytic gradient: grad_p[i] = 2 * sum_j w[i+j] * p[j].
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = fft_convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    if beta_init is None:
        beta = max(1e-14, 0.12 * max_c)
    else:
        beta = beta_init

    rng = np.random.default_rng(12345)

    while time.monotonic() < deadline:
        it += 1

        c = fft_convolve(p0, p0)
        max_c = float(c.max())

        # Soft-max weights for differentiating the max
        x = (c - max_c) / beta
        x = np.clip(x, -60.0, 0.0)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Gradient: grad_p[i] = 2 * sum_j w[i+j] * p[j]
        grad_p = compute_grad_p(w, p0) + reg * p0

        # Logit space gradient (projected to sum(z) = 0 subspace)
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            p0 = best_p.copy()
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            p0 = best_p.copy()
            continue
        p = p / p_sum

        c = fft_convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.003, lr * 0.9)

        if it % 200 == 0:
            beta = max(1e-14, beta * 0.65)

        if no_imp > restart_every:
            # Restart with perturbation around best
            z = 0.6 * best_z + 0.8 * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            p0 = best_p.copy()
            beta = max(1e-14, 0.12 * float(fft_convolve(p0, p0).max()))

        p0 = p

        if verbose and it % 1000 == 0:
            print(f"    Adam iter {it}: score={best_score:.8f}, beta={beta:.2e}, lr={lr:.5f}")

    return best_p, best_score


def adam_populate(p_list, deadline, lr=0.04, verbose=True):
    """Run Adam on multiple candidates, track the best."""
    best_p = None
    best_score = np.inf
    for i, p in enumerate(p_list):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        per_candidate_time = remaining / (len(p_list) - i)
        end_time = time.monotonic() + per_candidate_time
        print(f"  Adam candidate {i+1}/{len(p_list)}, time budget={per_candidate_time:.0f}s")
        p_best, s_best = adam_polish_fft(p, end_time, lr=lr, verbose=False)
        if s_best < best_score:
            best_score = s_best
            best_p = p_best
            print(f"    New best: {best_score:.8f}")
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-resolution FFT-accelerated search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
      1. Initialize from parent sequence or diverse analytic profiles at n=512.
      2. LP-based refinement at n=512.
      3. Upsample to n=1024, LP refinement.
      4. Adam gradient polish at n=1024 (FFT-accelerated, correct analytic gradient).
      5. Upsample to n=2048, Adam polish for final improvement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 20.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Try to get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent
                best_score = internal_score(parent)
                print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass

    # Generate diverse initializations at n=512
    n_init = 512
    profiles = []
    if best_seq is not None:
        profiles.append(interpolate_seq(best_seq, n_init))
    profiles.append(make_linear_profile(n_init, 1.5))
    profiles.append(make_linear_profile(n_init, 2.0))
    profiles.append(make_linear_profile(n_init, 1.0))
    profiles.append(make_quadratic_profile(n_init, 1.5, -0.5))
    profiles.append(make_quadratic_profile(n_init, 2.0, -1.0))
    profiles.append(make_u_profile(n_init, 1.5))
    profiles.append(make_u_profile(n_init, 2.0))
    profiles.append(make_quadratic_profile(n_init, 1.0, 0.5))
    profiles.append(make_quadratic_profile(n_init, -1.5, 0.3))

    # Evaluate all profiles and keep the best few
    scored_profiles = []
    for p in profiles:
        if p is not None and p is not False:
            sc = internal_score(p)
            if np.isfinite(sc):
                scored_profiles.append((sc, p))
    scored_profiles.sort(key=lambda x: x[0])

    if scored_profiles:
        sc0, p0 = scored_profiles[0]
        if sc0 < best_score:
            best_score = sc0
            best_seq = p0
        print(f"Best initial profile: score={best_score:.8f}, n={len(best_seq)}")

    if best_seq is None:
        best_seq = make_linear_profile(n_init, 1.5)
        best_score = internal_score(best_seq)
        print(f"Fallback initial: score={best_score:.8f}")

    # Phase 1: LP at n=512
    deadline_p1 = min(deadline, t_start + 350.0)
    if time.monotonic() < deadline_p1 - 30:
        a1 = interpolate_seq(best_seq, 512)
        if a1 is not None:
            print(f"\nPhase 1: LP at n=512, initial={internal_score(a1):.8f}")
            t0 = time.monotonic()
            M_init = float(fft_convolve(a1, a1).max())
            _ = lp_direction(a1, M_init, timeout=5)
            t_lp = time.monotonic() - t0
            print(f"  LP solve time: {t_lp*1000:.0f}ms")

            max_iters_1 = min(300, max(50, int((deadline_p1 - time.monotonic()) * 0.8 / max(t_lp + 0.01, 0.02))))
            f_best, sc_best = lp_iterate(a1, max_iters=max_iters_1, t_init=0.4, t_min=0.02,
                                          deadline=deadline_p1, verbose=True)
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  Phase 1 result: score={best_score:.8f}")

    # Phase 2: LP at n=1024
    deadline_p2 = min(deadline, t_start + 650.0)
    if time.monotonic() < deadline_p2 - 60:
        a2 = interpolate_seq(best_seq, 1024)
        if a2 is not None:
            print(f"\nPhase 2: LP at n=1024, initial={internal_score(a2):.8f}")
            t0 = time.monotonic()
            M_init = float(fft_convolve(a2, a2).max())
            _ = lp_direction(a2, M_init, timeout=15)
            t_lp = time.monotonic() - t0
            print(f"  LP solve time: {t_lp*1000:.0f}ms")

            max_iters_2 = min(150, max(20, int((deadline_p2 - time.monotonic()) * 0.8 / max(t_lp + 0.05, 0.1))))
            f_best, sc_best = lp_iterate(a2, max_iters=max_iters_2, t_init=0.35, t_min=0.02,
                                          deadline=deadline_p2, verbose=True)
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  Phase 2 result: score={best_score:.8f}")

    # Phase 3: Adam at n=1024 (multiple diverse candidates)
    deadline_p3 = min(deadline, t_start + 1200.0)
    if time.monotonic() < deadline_p3 - 30:
        n3 = 1024
        a3 = interpolate_seq(best_seq, n3)

        # Create diverse candidates for population
        candidates = [a3.copy()]
        # Add a few perturbed versions
        for _ in range(2):
            perturbation = rng.normal(0, 0.1, size=n3)
            a_pert = a3 * (1.0 + 0.3 * perturbation)
            a_pert = normalize(a_pert)
            if a_pert is not None:
                candidates.append(a_pert)
        # Add a slightly different shape
        a_alt = interpolate_seq(make_quadratic_profile(n3, 1.8, -0.7), n3)
        if a_alt is not None:
            candidates.append(a_alt)

        print(f"\nPhase 3: Adam at n={n3}, {len(candidates)} candidates, "
              f"initial best={internal_score(a3):.8f}")
        p_best, sc_best = adam_populate(candidates, deadline_p3, lr=0.05, verbose=True)
        if p_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 3 result: score={best_score:.8f}")

    # Phase 4: Upsample to n=2048 and Adam polish
    deadline_p4 = deadline
    if time.monotonic() < deadline_p4 - 30:
        n4 = 2048
        a4 = interpolate_seq(best_seq, n4)
        if a4 is not None:
            print(f"\nPhase 4: Adam at n={n4}, initial={internal_score(a4):.8f}")
            p_best, sc_best = adam_polish_fft(a4, deadline_p4, lr=0.04, verbose=True,
                                               beta_init=0.1 * float(fft_convolve(a4, a4).max()))
            if p_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best
                print(f"  Phase 4 result: score={best_score:.8f}")

    # Phase 5: Final quick polish at current best n
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n5 = len(best_seq)
        print(f"\nPhase 5: Final Adam polish at n={n5}, remaining={remaining:.0f}s")
        p_best, sc_best = adam_polish_fft(best_seq, deadline, lr=0.02, verbose=True,
                                           beta_init=0.05 * float(fft_convolve(best_seq, best_seq).max()))
        if p_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 5 result: score={best_score:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
