import numpy as np

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))

    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:

    else:
        p = p / s

def normalize_nonneg(a):

def make_bump_profile(n, shape='sin', asym=0.0):
    """Create a bump-shaped profile (zero or near-zero at endpoints)."""
    t = np.linspace(0.0, 1.0, n)
    if shape == 'sin':
        a = np.sin(np.pi * t)
    elif shape == 'sin2':
        a = np.sin(np.pi * t) ** 2
    elif shape == 'raised_cos':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif shape == 'gaussian':
        sigma = 0.2
        a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    elif shape == 'gauss_wide':
        sigma = 0.3
        a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    elif shape == 'triangle':
        a = 1.0 - np.abs(2.0 * t - 1.0)
    elif shape == 'parabola':
        a = 1.0 - (2.0 * t - 1.0) ** 2
    elif shape == 'sinc':
        x = np.pi * (t - 0.5)
        a = np.where(np.abs(x) < 1e-10, 1.0, np.sin(x) / x)
        a = np.maximum(a, 0.0)
    elif shape == 'asym_sin':
        # Asymmetric sine bump
        if asym > 0:
            a = np.sin(np.pi * t) * (1.0 + asym * (t - 0.5))
        else:
            a = np.sin(np.pi * (1.0 + asym * t))
        a = np.maximum(a, 0.0)
    else:
        a = np.sin(np.pi * t)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def make_mono_profile(n, p=1.0):
    """Monotonically increasing profile."""
    t = np.linspace(0.0, 1.0, n)
    a = t ** p + 1e-8
    s = float(a.sum())
    return a / s


def make_asym_bump(n, alpha, beta_param):
    """Asymmetric bump: a(t) ~ t^alpha * (1-t)^beta_param."""
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha * (1.0 - t) ** beta_param + 1e-12
    s = float(a.sum())
    return a / s


def make_multi_bump(n, centers, widths, amps):
    """Multi-modal bump profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for c, w, amp in zip(centers, widths, amps):
        a += amp * np.exp(-0.5 * ((t - c) / w) ** 2)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))
