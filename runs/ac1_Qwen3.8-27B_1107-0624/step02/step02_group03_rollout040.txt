            target_beta = max(beta_min, 0.05 * max_c)
        beta = 0.97 * beta + 0.03 * target_beta

        # Soft-argmax weights
        x = (c - max_c) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Gradient in a-space
        grad = _grad_weighted(a, w)

        # Logit-space gradient
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)

        if not np.all(np.isfinite(gz)):
            z = np.log(best_a) - np.log(best_a).mean()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z = z - z.mean()

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = np.log(best_a) - np.log(best_a).mean()
            continue
        p = p / p_sum

        c_new = np.convolve(p, p)
        score = 2.0 * n * float(c_new.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = p.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = p

        # Restart logic with structured perturbation
        if no_imp > restart_every:
            # Use Fourier-mode perturbation for structured exploration
            k_mode = rng.integers(1, max(2, n // 4))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.2)
            t = np.linspace(0, 1, n)
            perturbation = amp * np.cos(2 * np.pi * k_mode * t + phase)
            z_pert = np.log(best_a) + perturbation
            z_pert = z_pert - z_pert.mean()
            z_pert = np.clip(z_pert, -40, 40)
            z = z_pert
            m.fill(0)
            v.fill(0)
            it = 0
            no_imp = 0
            beta = max(beta_min, 0.1 * float(np.convolve(best_a, best_a).max()))
            if verbose:
                print(f"    Restart (mode k={k_mode}): best_score={best_score:.8f}")

        if verbose and it % 500 == 0:
            print(f"    it={it}, score={best_score:.8f}, beta={beta:.6f}, "
                  f"no_imp={no_imp}, flat={flattening}")

    return best_a, best_score


def _make_linear(n, eta):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (2 * t - 1)
    return _normalize(a)


def _make_cosine(n, eta, phase=0.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * np.cos(np.pi * t + phase)
    return _normalize(a)


def _make_quadratic(n, eta1, eta2):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    a = 1.0 + eta1 * u + eta2 * u ** 2
    return _normalize(a)


def _make_sine(n, eta):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * np.sin(np.pi * t)
    return _normalize(a)


def _interpolate(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return _normalize(a_new)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 25.0)
    rng = np.random.default_rng(seed)

    # === Phase 1: Generate diverse initial profiles at n=256 ===
    n_small = 256
    n_large = 512
    print(f"Phase 1: Screening at n={n_small}")

    profiles = []

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                p = _interpolate(parent, n_small)
                if p is not None:
                    profiles.append(('parent', p))
                    print(f"  Parent interpolated: score={_score(p):.8f}")
    except Exception:
        pass

    # Linear profiles with various slopes
    for eta in [0.6, 0.8, 1.0, 1.2, 1.4]:
        p = _make_linear(n_small, eta)
        if p is not None:
            profiles.append((f'lin_{eta}', p))

    # Cosine profiles
    for eta, phase in [(0.4, 0.0), (0.6, 0.0), (0.5, np.pi / 4), (0.7, -np.pi / 4)]:
        p = _make_cosine(n_small, eta, phase)
        if p is not None:
            profiles.append((f'cos_{eta}_{phase:.2f}', p))

    # Quadratic profiles
    for e1, e2 in [(0.8, 0.3), (1.0, -0.2), (0.5, 0.5), (1.2, 0.1), (0.6, -0.3)]:
        p = _make_quadratic(n_small, e1, e2)
        if p is not None:
            profiles.append((f'q_{e1}_{e2}', p))

    # Sine
    p = _make_sine(n_small, 0.4)
    if p is not None:
        profiles.append(('sine_0.4', p))

    # Screen with quick optimization
    screen_end = t_start + 150
    candidates = []
    for i, (name, p) in enumerate(profiles):
        if time.monotonic() > screen_end - 5:
            break
        a_best, s_best = _adam_optimize(
            p, min(time.monotonic() + 10, screen_end - 2),
            lr=0.05, beta_init=0.01, seed=seed + i)
        candidates.append((s_best, name, a_best))
        print(f"  {name}: {s_best:.8f}")

    candidates.sort(key=lambda x: x[0])
    print(f"\nTop 5 at n={n_small}:")
    for s, name, a in candidates[:5]:
        print(f"  {name}: {s:.8f}")

    # === Phase 2: Deep optimization of top candidates at n=256 ===
    print(f"\nPhase 2: Deep optimization at n={n_small}")
    best_score = candidates[0][0]
    best_a_small = candidates[0][2].copy()

    deep_end = t_start + 700
    n_top = min(3, len(candidates))
    for rank in range(n_top):
        s, name, a = candidates[rank]
        remaining = deep_end - time.monotonic()
        if remaining < 30:
            break
        phase_dl = time.monotonic() + remaining / (n_top - rank)
        print(f"\n  [{rank+1}/{n_top}] {name} (score={s:.8f}), time_left={remaining:.0f}s")
        a_opt, s_opt = _adam_optimize(
            a, phase_dl, lr=0.035, beta_init=0.006, beta_min=0.0003,
            restart_every=350, verbose=True, flatten_interval=800,
            seed=seed + 100 + rank)
        if s_opt < best_score:
            best_score = s_opt
            best_a_small = a_opt.copy()
            print(f"  NEW BEST at n={n_small}: {s_opt:.8f}")

    # === Phase 3: Transfer to n=512 and optimize ===
    print(f"\nPhase 3: Optimizing at n={n_large} (best so far: {best_score:.8f})")
    a_large = _interpolate(best_a_small, n_large)
    if a_large is None:
        a_large = _make_linear(n_large, 1.0)
    print(f"  Interpolated initial score: {_score(a_large):.8f}")

    # Also try interpolating the 2nd and 3rd best
    alt_large = []
    for s2, name2, a2 in candidates[1:3]:
        a2_large = _interpolate(a2, n_large)
        if a2_large is not None:
            alt_large.append((a2_large, name2))

    large_end = t_start + 1550
    if time.monotonic() < large_end - 60:
        a_opt, s_opt = _adam_optimize(
            a_large, large_end, lr=0.03, beta_init=0.005, beta_min=0.0002,
            restart_every=500, verbose=True, flatten_interval=1000,
            seed=seed + 200)
        if s_opt < best_score:
            best_score = s_opt
            best_a_large = a_opt.copy()
            print(f"  Improved at n={n_large}: {s_opt:.8f}")
        else:
            best_a_large = a_large.copy()

        # Try alternatives if time remains
        if time.monotonic() < large_end - 120 and len(alt_large) > 0:
            for a_alt, name_alt in alt_large:
                if time.monotonic() > large_end - 60:
                    break
                a_opt_alt, s_opt_alt = _adam_optimize(
                    a_alt, large_end - 45, lr=0.03, beta_init=0.005,
                    beta_min=0.0002, restart_every=400, verbose=False,
                    seed=seed + 300)
                if s_opt_alt < best_score:
                    best_score = s_opt_alt
                    best_a_large = a_opt_alt.copy()
                    print(f"  Alt {name_alt} improved: {s_opt_alt:.8f}")

    # === Phase 4: Final fine-tuning ===
    print(f"\nPhase 4: Final fine-tuning (best: {best_score:.8f})")
    final_a = best_a_large if n_large == len(best_a_large) else _interpolate(best_a_large, n_large)
    if final_a is None:
        final_a = _make_linear(n_large, 1.0)

    if time.monotonic() < deadline - 10:
        a_final, s_final = _adam_optimize(
            final_a, deadline, lr=0.015, beta_init=0.002, beta_min=0.0001,
            restart_every=600, verbose=True, flatten_interval=0,
            seed=seed + 400)
        if s_final < best_score:
            best_score = s_final
            final_a = a_final.copy()

    # === Final validation ===
    print(f"\nFinal result: n={len(final_a)}, score={_score(final_a):.8f}")
    final_a = np.asarray(final_a, dtype=np.float64)
    final_a = np.clip(final_a, 0.0, 1000.0)
    s = float(final_a.sum())
    if s < 0.01:
        final_a = _make_linear(n_large, 1.0)
        s = final_a.sum()
    final_a = final_a / s

    # Cross-validate
    cross_score = _score(final_a)
    print(f"Cross-validated score: {cross_score:.8f}")

    return [float(x) for x in final_a]
