"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based descent, multiplicative weights (mirror descent), 
and Adam polishing with soft-max smoothing.
"""
import numpy as np
import time
from scipy.optimize import linprog

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_step(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def soft_max_vals(c, beta):
    """Compute soft-max values and weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def corr_weights(w, a):
    """Compute h[i] = sum_j w[j] * a[j-i] for valid indices."""
    n = len(a)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    # We need: g_i = sum_k w_k * a[k-i] (with a zero-padded)
    # This is the cross-correlation of w and a
    # = conv(w, a_reversed) shifted appropriately
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    # h[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-(k-j)]
    # We want g_i = sum_k w[k] * a[k-i]
    # Let's think: corr[w,a][i] = sum_k w[k+i] * a[k] ... 
    # Actually, let me just do it directly for clarity with FFT
    # sum_k w[k] * a[k-i] = (w * a_reversed)[n-1+i] if we define things properly
    # Let me just use the direct formula
    # g_i = sum_{k: 0<=k<2n-1, 0<=k-i<n} w[k] * a[k-i]
    # = sum_{j: 0<=j<n} w[i+j] * a[j]  (substituting j = k-i)
    # This is the cross-correlation at lag i
    # Using FFT: corr_i = sum_j w[i+j] * a[j]
    # = <w_shifted_i, a> where w_shifted_i[k] = w[k+i]
    # Alternatively: it's the i-th element of the "anti-convolution" of w and a
    
    # Let me verify with the formula from the code:
    # h = irfft(fft(w) * fft(a[::-1]))
    # h[m] = sum_j w[j] * a_rev[m-j] = sum_j w[j] * a[n-1-m+j]
    # Let n2 = n-1-m, so m = n-1-n2
    # h[n-1-n2] = sum_j w[j] * a[n2+j]
    # So h[n-1-i] = sum_j w[j] * a[i+j]
    # We want g_i = sum_j w[i+j] * a[j] = sum_j w[j'] * a[j'-i] where j' = i+j
    # Hmm, this is different.
    
    # Let me just compute it as: g_i = sum_{j} w[i+j] * a[j]
    # This is the cross-correlation of w and a at lag i (with w being the "longer" signal)
    # Using FFT: if we compute conv(w, a_rev) = sum_j w[j] * a_rev[k-j]
    # conv(w, a_rev)[i+j] includes the term w[j'] * a[i+j-j'] ... this is getting confusing.
    
    # Simplest: just do it via FFT properly
    # We want: g_i = sum_{j=0}^{n-1} w[i+j] * a[j], for i = 0, ..., n-1
    # Define w_ext of length 2n-1 (already is), and a of length n
    # g_i = sum_j a[j] * w[i+j]
    # This is the correlation of a and w at lag i
    # = <a, shift_i(w)> where shift_i(w)[j] = w[i+j]
    # Using FFT: Corr(a, w)[i] = sum_j a[j] * conj(w[j+i])
    # For real signals: = sum_j a[j] * w[j+i]
    # Using FFT: FFT(Corr(a,w)) = FFT(a) * conj(FFT(w)) (with appropriate zero-padding)
    
    # Actually, the simpler formula: 
    # sum_j a[j] * w[j+i] = (a * w_rev)[i + (n-1)] ... no
    
    # Let me just use: g = corr(a, w) where corr[a,w][i] = sum_j a[j] * w[j+i]
    # This equals conv(a, w_rev)[(n-1) + i] ... let me just check dimensions
    
    # conv(a, w_rev) has length n + (2n-1) - 1 = 3n - 2
    # conv(a, w_rev)[k] = sum_j a[j] * w_rev[k-j] = sum_j a[j] * w[(2n-2)-(k-j)]
    # = sum_j a[j] * w[2n-2-k+j]
    # We want sum_j a[j] * w[j+i], so set 2n-2-k = -i, i.e., k = 2n-2+i
    # conv(a, w_rev)[2n-2+i] = sum_j a[j] * w[j+i] ✓
    
    # But 2n-2+i ranges from 2n-2 to 3n-3, which is within [0, 3n-3] so it's valid.
    
    # The current code computes h = irfft(fft(w) * fft(a_rev))
    # h[m] = sum_j w[j] * a_rev[m-j] = sum_j w[j] * a[n-1-(m-j)] = sum_j w[j] * a[n-1-m+j]
    # Set n-1-m = -i => m = n-1+i... wait that gives m = n-1+i which for i in [0,n-1] gives m in [n-1, 2n-2]
    # h[n-1+i] = sum_j w[j] * a[i+j] ... hmm that's sum_j w[j]*a[i+j] not sum_j a[j]*w[j+i]
    
    # These are actually the same! sum_j w[j]*a[i+j] = sum_j' a[j']*w[j'-i] where j' = i+j... no
    # sum_j w[j]*a[i+j]: let j' = i+j, then = sum_{j'=i}^{i+n-1} w[j'-i] * a[j']
    # sum_j a[j]*w[j+i]: let j' = j+i, then = sum_{j'=i}^{i+n-1} a[j'-i] * w[j']
    # These are different (w and a are in different positions)
    
    # OK let me just compute g directly using the formula I derived:
    # g_i = sum_{j=0}^{n-1} a[j] * w[i+j]
    # Use FFT: pad a to length 3n-1, w to length 3n-1
    # g = conv(a, w)[2n-2+0 : 2n-2+n] ... 
    
    # Actually simplest approach: just use the formula from the original code and verify
    # The original code has:
    #   h = irfft(fft(w) * fft(a[::-1]))
    #   return h[n-1 : 2n-1]
    # 
    # h[m] = sum_j w[j] * a_rev[m-j] = sum_j w[j] * a[n-1-(m-j)]
    # h[n-1+i] = sum_j w[j] * a[n-1-(n-1+i-j)] = sum_j w[j] * a[j-i]
    # 
    # So h[n-1+i] = sum_j w[j] * a[j-i] (valid when 0 <= j-i < n)
    # = sum_{j=i}^{i+n-1} w[j] * a[j-i]
    # = sum_{k=0}^{n-1} w[i+k] * a[k]
    # = g_i ✓
    
    return h[n-1:2*n-1]

def mirror_descent_optimize(a0, deadline, rng, eta0=0.1, beta0=None, verbose=True):
    """Mirror descent (multiplicative weights) with soft-max smoothing."""
    a = np.maximum(np.asarray(a0, dtype=np.float64).ravel(), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a0.copy(), compute_score(a0)
    a = a / s
    n = a.size
    
    c = full_convolve_fft(a)
    maxc = float(c.max())
    if beta0 is None:
        beta = 0.1 * maxc
    else:
        beta = beta0
    
    best_a = a.copy()
    best_score = float(2.0 * n * maxc)
    eta = eta0
    no_imp = 0
    it = 0
    
    while time.monotonic() < deadline:
        L, w = soft_max_vals(c, beta)
        g = 2.0 * corr_weights(w, a)
        
        # Multiplicative weights update
        # a_new[i] = a[i] * exp(-eta * g[i]) / Z
        # For stability, center g first
        g_centered = g - float(np.mean(g))
        
        # Clip to avoid overflow
        updates = np.clip(-eta * g_centered, -10.0, 10.0)
        a_new = a * np.exp(updates)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        c_new = full_convolve_fft(a_new)
        maxc_new = float(c_new.max())
        score_new = float(2.0 * n * maxc_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            no_imp = 0
            # Adjust eta: increase if improving
            eta = min(eta * 1.001, 0.5)
        else:
            no_imp += 1
            eta *= 0.99
            # Don't update a
        
        # Anneal beta
        if it % 50 == 0:
            beta = max(1e-6, beta * 0.95)
        
        # Restart if stuck
        if no_imp > 500:
            # Add noise
            noisy = best_a + 0.05 * rng.normal(size=n)
            noisy = np.maximum(noisy, 0.0)
            noisy = noisy / noisy.sum()
            a = noisy
            c = full_convolve_fft(a)
            maxc = float(c.max())
            beta = 0.2 * maxc
            eta = eta0
            no_imp = 0
        
        it += 1
        if verbose and it % 10000 == 0:
            print(f"  MD iter {it}: score={best_score:.10f}, eta={eta:.4f}, beta={beta:.6f}")
    
    return best_a, best_score

def adam_polish(a0, deadline, rng, lr=0.05, beta_sm=0.1, verbose=True):
    """Adam-based polishing in log-probability space."""
    p = np.maximum(np.asarray(a0, dtype=np.float64).ravel(), 1e-15).copy()
    p = p / p.sum()
    n = p.size
    
    z = np.log(p)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = beta_sm * maxc
    
    best_p = p.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    
    while time.monotonic() < deadline:
        L, w = soft_max_vals(c, beta)
        g = 2.0 * corr_weights(w, p)
        # Subtract mean for stability (gradient of log-partition)
        g -= float(np.mean(g))
        
        # Adam in z-space: grad_z = p * (g - mean(g))
        gz = p * g
        
        t += 1
        m = b1 * m + (1-b1) * gz
        v = b2 * v + (1-b2) * gz*gz
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -30.0, 30.0)
        z -= float(np.mean(z))
        
        e = np.exp(z)
        p_new = e / e.sum()
        
        c_new = full_convolve_fft(p_new)
        maxc_new = float(c_new.max())
        score_new = float(2.0 * n * maxc_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_p = p_new.copy()
            p = p_new
            c = c_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.97)
        
        if t % 100 == 0:
            beta = max(1e-8, beta * 0.85)
        
        if no_imp > max(1000, 2*n):
            z = 0.7 * np.log(best_p + 1e-15) + 0.3 * rng.normal(size=n) * 0.5
            z = np.clip(z, -30.0, 30.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            p = e / e.sum()
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = 0.2 * maxc
            m.fill(0)
            v.fill(0)
            t = 0
            lr = 0.05
            no_imp = 0
        
        if verbose and t % 5000 == 0:
            print(f"  Adam iter {t}: score={best_score:.10f}, lr={lr:.4f}, beta={beta:.6f}")
    
    return best_p, best_score

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)

def make_profile(n, kind="linear", **kwargs):
    t = np.linspace(0.0, 1.0, n)
    if kind == "linear":
        eta = kwargs.get("eta", 1.5)
        a = 1.0 + eta * (t - 0.5)
    elif kind == "cosine":
        k = kwargs.get("k", 1)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * t)
    elif kind == "sine_sq":
        k = kwargs.get("k", 1)
        a = 0.5 + 0.5 * np.sin(np.pi * k * t)**2
    elif kind == "quadratic":
        a = 1.0 + kwargs.get("c1", 0.5)*t + kwargs.get("c2", -0.3)*t**2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def lp_optimize_improved(f_init, deadline, rng, verbose=True):
    """Improved LP-based optimization with multiple M values."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)
    
    conv_f = np.convolve(f, f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.2
    stagnant = 0
    iteration = 0
    
    # Cache the matrix (it changes with f, so we rebuild each time)
    A_cache = build_conv_matrix(f)
    
    while time.monotonic() < deadline:
        M = float(np.convolve(f, f).max())
        
        # Try the LP with current M
        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            t *= 0.5
            if t < 0.01:
                stagnant += 10
            stagnant += 1
            if stagnant > 50:
                # Restart
                f = best_f + 0.1 * max(1.0/n, np.std(best_f)) * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                t = 0.2
                stagnant = 0
            continue
        
        g = g / g.sum()
        
        # Try different step sizes
        best_trial_f = None
        best_trial_M = M + 1e-10
        
        for trial_t in [t, t*0.5, t*1.5, t*0.25]:
            f_new = (1 - trial_t) * f + trial_t * g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new = f_new / s_new
            M_new = float(np.convolve(f_new, f_new).max())
            if M_new < best_trial_M:
                best_trial_M = M_new
                best_trial_f = f_new.copy()
        
        if best_trial_f is not None:
            f = best_trial_f
            if best_trial_M < best_M - 1e-13:
                best_M = best_trial_M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
        else:
            t *= 0.5
            stagnant += 1
        
        t = max(0.05, t * 0.999)
        
        if stagnant > 80:
            f = best_f + 0.05 * max(1.0/n, np.std(best_f)) * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            t = 0.2
            stagnant = 0
            if verbose:
                print(f"  LP restart at iter {iteration}, M={best_M:.10f}")
        
        if verbose and iteration % 50 == 0:
            print(f"  LP iter {iteration}: M={best_M:.10f}, t={t:.4f}")
    
    return best_f

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP descent, mirror descent, and Adam polishing.
    Minimizes 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
    
    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass
    
    print(f"Starting search. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)
    
    # Phase 1: LP at n=512
    n1 = 512
    phase1_end = min(deadline, t_start + 350.0)
    
    if init_seq is not None and init_seq.size != n1:
        p1 = interpolate_profile(init_seq, n1)
    elif init_seq is not None:
        p1 = init_seq.copy()
    else:
        p1 = make_profile(n1, "linear", eta=1.5)
    
    sc = compute_score(p1)
    update(p1, sc)
    print(f"Phase 1: LP at n={n1}, start score={sc:.10f}")
    
    result = lp_optimize_improved(p1, phase1_end, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"After Phase 1: score={sc:.10f}")
    
    # Phase 2: Mirror descent at n=512
    phase2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        print(f"Phase 2: Mirror descent at n={best_seq.size}")
        for eta in [0.05, 0.1, 0.2]:
            if time.monotonic() > phase2_end - 5:
                break
            per = min(phase2_end, time.monotonic() + 70.0)
            p_start = best_seq + 0.002 * rng.normal(size=best_seq.size)
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
            result, sc = mirror_descent_optimize(p_start, per, rng, eta0=eta, verbose=True)
            sc = compute_score(result)
            update(result, sc)
            print(f"  MD eta={eta}: score={sc:.10f}")
    
    print(f"After Phase 2: best={best_score:.10f}")
    
    # Phase 3: LP at n=1024
    n2 = 1024
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        p2 = interpolate_profile(best_seq, n2)
        sc = compute_score(p2)
        update(p2, sc)
        print(f"Phase 3: LP at n={n2}, interpolated score={sc:.10f}")
        
        per = min(phase3_end, time.monotonic() + 350.0)
        result = lp_optimize_improved(p2, per, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After Phase 3: score={sc:.10f}")
    
    print(f"After Phase 3: best={best_score:.10f}")
    
    # Phase 4: Mirror descent at n=1024
    phase4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        n4 = best_seq.size
        print(f"Phase 4: Mirror descent at n={n4}")
        for eta in [0.03, 0.05, 0.08]:
            if time.monotonic() > phase4_end - 5:
                break
            per = min(phase4_end, time.monotonic() + 100.0)
            p_start = best_seq + 0.001 * rng.normal(size=n4)
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
            result, sc = mirror_descent_optimize(p_start, per, rng, eta0=eta, verbose=True)
            sc = compute_score(result)
            update(result, sc)
            print(f"  MD eta={eta}: score={sc:.10f}")
    
    print(f"After Phase 4: best={best_score:.10f}")
    
    # Phase 5: Adam polish
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end - 10 and best_seq is not None:
        n5 = best_seq.size

        for lr in [0.05, 0.03, 0.02, 0.01]:

    # Phase 6: Try n=2048 if time and score justify
    n3 = 2048
    phase6_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase6_end - 20 and best_score > 1.503:
        p3 = interpolate_profile(best_seq, n3)
        sc = compute_score(p3)
        update(p3, sc)
        print(f"Phase 6: n={n3}, interpolated score={sc:.10f}")
        per = min(phase6_end, time.monotonic() + 80.0)
        result, sc = mirror_descent_optimize(p3, per, rng, eta0=0.02, verbose=True)

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n, at least 2."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_solve(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def soft_max_weights(c, beta):
    """Compute soft-max value and weights for smoothing max."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, a):

    R = np.fft.irfft(fw * np.conj(fa), N)

            stagnant += 1

        t = max(0.05, t * 0.998)

        if stagnant > 60:
            f = best_f + 0.06 * max(1.0 / n, float(np.std(best_f))) * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            t = 0.25
            stagnant = 0
            if verbose:
                print(f"    LP restart at iter {iteration}, M={best_M:.10f}")

        if verbose and iteration % 30 == 0:
            print(f"    LP iter {iteration}: M={best_M:.10f}, t={t:.4f}")

    return best_f


def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def make_profile(n, kind="linear"):
    """Generate a structured starting profile."""
    t = np.linspace(0.0, 1.0, n)
    if kind == "linear":
        a = 1.0 + 1.5 * (t - 0.5)
    elif kind == "cos1":
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
    elif kind == "cos2":
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
    elif kind == "sin2":
        a = 0.5 + 0.5 * np.sin(np.pi * t) ** 2
    elif kind == "quad":
        a = 1.0 + 0.8 * t - 0.4 * t * t
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def try_combinations(best_a, best_b, deadline, rng):
    """Try convex combinations of two sequences to find better ones."""
    n = len(best_a)
    best_combo = None
    best_combo_score = np.inf

    for alpha in [0.3, 0.5, 0.7, 0.1, 0.9]:
        combo = (1 - alpha) * best_a + alpha * best_b
        combo = np.maximum(combo, 0.0)
        cs = combo.sum()
        if cs < 1e-10:
            continue
        combo = combo / cs
        sc = compute_score(combo)
        if sc < best_combo_score:
            best_combo_score = sc
            best_combo = combo.copy()

    if best_combo is not None and best_combo_score < compute_score(best_a):
        return best_combo, best_combo_score
    return best_a, compute_score(best_a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP descent -> Mirror Descent -> LP at larger n
    -> Adam polish -> combination search.
    Minimizes 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting search. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update_best(init_seq, sc_init)

    # === Phase 1: LP at n=512 ===
    n1 = 512
    phase1_end = min(deadline, t_start + 400.0)

    if init_seq is not None and init_seq.size != n1:
        p1 = interpolate_profile(init_seq, n1)
    elif init_seq is not None:
        p1 = init_seq.copy()
    else:
        p1 = make_profile(n1, "linear")

    sc = compute_score(p1)
    update_best(p1, sc)
    print(f"Phase 1: LP at n={n1}, start={sc:.10f}")

    result = lp_optimize_multi(p1, phase1_end, rng, verbose=True)
    sc = compute_score(result)
    update_best(result, sc)
    print(f"After Phase 1: score={sc:.10f}")

    # === Phase 2: Mirror Descent at n=512 ===
    phase2_end = min(deadline, t_start + 650.0)
    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        md_results = []
        for eta in [0.06, 0.1, 0.15]:
            if time.monotonic() > phase2_end - 5:
                break
            per = min(phase2_end, time.monotonic() + 70.0)
            p_start = best_seq + 0.003 * rng.normal(size=best_seq.size)
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
            result, sc = mirror_descent_optimize(p_start, per, rng, eta0=eta, verbose=True)
            sc = compute_score(result)
            update_best(result, sc)
            md_results.append(result)
            print(f"  MD eta={eta}: score={sc:.10f}")

        # Try combinations
        if len(md_results) >= 2:
            for i in range(len(md_results)):
                for j in range(i + 1, len(md_results)):
                    if time.monotonic() > phase2_end - 3:
                        break
                    combo, sc = try_combinations(best_seq, md_results[j], phase2_end, rng)
                    update_best(combo, sc)

    print(f"After Phase 2: best={best_score:.10f}")

    # === Phase 3: LP at n=1024 ===
    n2 = 1024
    phase3_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        p2 = interpolate_profile(best_seq, n2)
        sc = compute_score(p2)
        update_best(p2, sc)
        print(f"Phase 3: LP at n={n2}, interp score={sc:.10f}")

        per = min(phase3_end, time.monotonic() + 380.0)
        result = lp_optimize_multi(p2, per, rng, verbose=True)
        sc = compute_score(result)
        update_best(result, sc)
        print(f"After Phase 3: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # === Phase 4: Mirror Descent at n=1024 ===
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        n4 = best_seq.size
        print(f"Phase 4: Mirror Descent at n={n4}")
        for eta in [0.04, 0.07, 0.1]:
            if time.monotonic() > phase4_end - 5:
                break
            per = min(phase4_end, time.monotonic() + 100.0)
            p_start = best_seq + 0.002 * rng.normal(size=n4)
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
            result, sc = mirror_descent_optimize(p_start, per, rng, eta0=eta, verbose=True)
            sc = compute_score(result)
            update_best(result, sc)
            print(f"  MD eta={eta}: score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # === Phase 5: Adam Polish ===
    phase5_end = min(deadline, t_start + 1680.0)
    if time.monotonic() < phase5_end - 10 and best_seq is not None:
        n5 = best_seq.size
        print(f"Phase 5: Adam polish at n={n5}")
        for lr in [0.04, 0.02, 0.01, 0.005]:
            if time.monotonic() > phase5_end - 5:
                break
            per = min(phase5_end, time.monotonic() + 60.0)
            p_start = best_seq + 0.0005 * rng.normal(size=n5)
            p_start = np.maximum(p_start, 1e-12)
            p_start = p_start / p_start.sum()
            result, sc = adam_polish(p_start, per, rng, lr=lr, verbose=False)
            sc = compute_score(result)
            update_best(result, sc)
            print(f"  Adam lr={lr}: score={sc:.10f}")

    print(f"After Phase 5: best={best_score:.10f}")

    # === Phase 6: Try n=2048 if beneficial ===
    n3 = 2048
    phase6_end = min(deadline, t_start + 1760.0)
    if time.monotonic() < phase6_end - 20 and best_score > 1.504 and best_seq is not None:
        p3 = interpolate_profile(best_seq, n3)
        sc = compute_score(p3)
        print(f"Phase 6: n={n3}, interp={sc:.10f}")
        if sc < best_score + 0.02:
            update_best(p3, sc)
            per = min(phase6_end, time.monotonic() + 60.0)
            result, sc = mirror_descent_optimize(p3, per, rng, eta0=0.03, verbose=True)
            sc = compute_score(result)
            update_best(result, sc)
            print(f"After Phase 6: score={sc:.10f}")

    print(f"After Phase 6: best={best_score:.10f}")

    # === Phase 7: Final short Adam + local search ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        nf = best_seq.size
        # Very fine Adam
        per = min(deadline - 5, time.monotonic() + 30.0)
        p_start = best_seq + 0.0002 * rng.normal(size=nf)
        p_start = np.maximum(p_start, 1e-12)
        p_start = p_start / p_start.sum()
        result, sc = adam_polish(p_start, per, rng, lr=0.005, verbose=False)
        sc = compute_score(result)
        update_best(result, sc)
        print(f"Final Adam: score={sc:.10f}")

        # Local random search
        if time.monotonic() < deadline - 2:
            for _ in range(50):
                if time.monotonic() > deadline - 2:
                    break
                noise = 0.001 * rng.normal(size=nf)
                candidate = best_seq + noise
                candidate = np.maximum(candidate, 0.0)
                cs = candidate.sum()
                if cs < 0.01:
                    continue
                candidate = candidate / cs
                sc_c = compute_score(candidate)
                update_best(candidate, sc_c)

    # Final validation
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_profile(512, "linear")

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_profile(256, "linear")

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in final_seq]
