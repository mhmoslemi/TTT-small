def smooth_max(c, beta):
    """Compute smoothed max and softmax weights."""
    c_shifted = c - c.max()
    scaled = beta * c_shifted
    # Numerical stability
    max_scaled = scaled.max()
    ex = np.exp(scaled - max_scaled)
    w = ex / ex.sum()
    smax = c.max() + (max_scaled + np.log(ex.sum())) / beta
    return smax, w

def fista_polish(a0, deadline, beta_init=50.0, beta_decay=0.999, lr_init=0.05, verbose=True):
    """
    FISTA (accelerated proximal gradient) for minimizing max(conv(a,a))
    subject to a in simplex (sum=1, a>=0).
    Uses smoothed max for differentiability.
    """
    n = len(a0)
    a = a0.copy()
    a = normalize(a)
    
    c = np.convolve(a, a)
    M0 = float(c.max())
    best_a = a.copy()
    best_score = 2.0 * n * M0
    beta = beta_init
    lr = lr_init
    
    # FISTA variables
    y = a.copy()
    tk = 1.0
    
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # Compute smoothed max and gradient at y
        c_y = np.convolve(y, y)
        _, w = smooth_max(c_y, beta)
        grad = compute_gradient(y, w)
        
        # Gradient of the smoothed objective
        # f_smooth(y) = (1/beta) * log(sum_k exp(beta * (y*y)_k))
        # grad f_smooth = w (softmax weights) applied to grad of (y*y)_k
        # which is exactly what compute_gradient gives us (the 2* factor is correct)
        
        # Actually, let me re-derive:
        # f_smooth = max_c + (1/beta) * log(sum_k exp(beta*(c_k - max_c)))
        # df_smooth / da_i = sum_k w_k * dc_k/da_i = grad[0] from compute_gradient
        # where w_k = softmax(beta * c)_k
        # So the gradient of f_smooth is exactly compute_gradient(a, w)
        
        # Step: y_new = y - lr * grad
        # But we need to project onto the simplex.
        # FISTA: x_new = proj_simplex(y - lr * grad)
        #         y_new = x_new + ((t_k - 1)/(t_{k+1})) * (x_new - x_old)
        
        x_old = a.copy()
        x_new = proj_simplex(y - lr * grad)
        
        # Check improvement
        c_new = np.convolve(x_new, x_new)
        M_new = float(c_new.max())
        score_new = 2.0 * n * M_new
        
        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = x_new + ((tk - 1.0) / t_new) * (x_new - x_old)
        y = np.maximum(y, 0.0)  # Keep non-negative (approximate projection)
        tk = t_new
        
        # Adapt learning rate
        if no_improve % 100 == 0:
            lr = max(0.01, lr * 0.9)
            beta = max(10.0, beta * 0.95)
        
        # Decay beta slowly to get sharper max approximation
        beta = max(5.0, beta * beta_decay)
        
        if no_improve > 500:
            # Reset momentum
            y = best_a.copy()
            tk = 1.0
            lr = lr_init
            no_improve = 0
            # Add small perturbation
            rng = np.random.default_rng(it)
            pert = rng.normal(0, 0.01, n)
            y = proj_simplex(best_a + pert)
        
        a = x_new
    
    return best_a, best_score

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-30:
        return np.ones(n) / n
    
    # Algorithm from Wang & Carra-Hartley (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

`ifft(fft(x, N) * conj(fft(y, N)))` gives `sum_k x[(k+lag) mod N] * y[k]` (circular cross-correlation).

For FISTA to work well, we want:
1. The smoothing to be good enough that the gradient is informative
2. The step size to be appropriate for the smoothed problem

The optimal step size for gradient descent on a L-Lipschitz smooth function is 1/L. For the smoothed max, L depends on beta and the problem structure.

Let me use an adaptive approach:
- Start with moderate beta
- Use backtracking line search or backtracking step size
- Gradually decrease beta

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA with adaptive smoothed-max, LP direction finding, and multi-resolution search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def compute_gradient(a, w):
    """
    Gradient of sum_k w[k]*(a*a)[k] w.r.t. a.
    grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    """
    n = len(a)
    nw = len(w)
    grad = np.zeros(n)
    for i in range(n):
        j_max = min(n, nw - i)
        if j_max > 0:
            grad[i] = 2.0 * np.dot(w[i:i + j_max], a[:j_max])
    return grad


def compute_gradient_fast(a, w):
    """FFT-based gradient: grad[i] = 2 * sum_j w[i+j]*a[j]."""
    n = len(a)
    N = 1
    while N < 2 * n:
        N *= 2
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    # ifft(fft(w)*conj(fft(a)))[i] = sum_j w[(i-j) mod N] * a[j]
    # We want sum_j w[i+j] * a[j]
    # Let b[j] = a[-j mod N] (= a reversed, padded)
    # Then sum_j w[i+j]*a[j] = sum_j w[(i+j) mod N]*a[j]
    # = sum_j w[(i-(-j)) mod N]*a[j]
    # = sum_m w[(i-m) mod N]*a[-m mod N] where m = -j
    # = ifft(fft(w)*conj(fft(a_flip)))[i] where a_flip[m] = a[-m mod N]
    a_flip = np.zeros(N)
    a_flip[0] = a[0]
    if n > 1:
        a_flip[N - n + 1:] = a[1:]
    AF = np.fft.rfft(a_flip, N)
    cross = np.fft.irfft(W * np.conj(AF), N)
    return 2.0 * cross[:n]


def smooth_max_weights(c, beta):
    """Compute softmax weights for smoothed max."""
    c_shifted = c - c.max()
    scaled = beta * c_shifted
    max_scaled = scaled.max()
    ex = np.exp(scaled - max_scaled)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    return w


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_optimize(a0, deadline, beta_init=100.0, beta_min=5.0, beta_decay=0.9995,
                   lr_init=0.08, verbose=True, use_fft_grad=True):
    """
    FISTA for minimizing max(conv(a,a)) subject to a in simplex.
    Uses softmax smoothing for the max.
    """
    n = len(a0)
    a = normalize(a0).copy()
    c = np.convolve(a, a)
    M0 = float(c.max())
    best_a = a.copy()
    best_score = 2.0 * n * M0
    best_M = M0

    beta = beta_init
    lr = lr_init

    y = a.copy()
    tk = 1.0
    no_improve = 0
    it = 0
    t_start = time.monotonic()

    grad_fn = compute_gradient_fast if use_fft_grad else compute_gradient

    while time.monotonic() < deadline:
        it += 1

        c_y = np.convolve(y, y)
        w = smooth_max_weights(c_y, beta)
        grad = grad_fn(y, w)

        # Ensure gradient is finite
        if not np.all(np.isfinite(grad)):
            y = best_a.copy()
            tk = 1.0
            continue

        # FISTA step
        x_old = a.copy()
        x_new = proj_simplex(y - lr * grad)

        # Evaluate
        c_new = np.convolve(x_new, x_new)
        M_new = float(c_new.max())
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = x_new.copy()
            best_M = M_new
            no_improve = 0
        else:
            no_improve += 1

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = x_new + ((tk - 1.0) / t_new) * (x_new - x_old)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 0:
            y = y / y_sum
        tk = t_new

        # Adapt learning rate and beta
        if no_improve % 200 == 0:
            lr = max(0.02, lr * 0.85)
        if no_improve > 1000:
            # Restart
            y = best_a.copy()
            tk = 1.0
            lr = lr_init * 0.5
            no_improve = 0
            rng = np.random.default_rng(it)
            pert = rng.normal(0, 0.02, n)
            y = proj_simplex(best_a + pert)

        # Decay beta
        beta = max(beta_min, beta * beta_decay)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    FISTA iter {it}: score={best_score:.8f}, beta={beta:.1f}, "
                  f"lr={lr:.5f}, no_improve={no_improve}, time={elapsed:.1f}s")

    return best_a, best_score


def lp_direction(f, M, timeout=15.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, conv(f,g)[x] <= M for all x.
    """
    n = len(f)
    # Build constraint matrix: A[i,j] = f[j] for row i (shifted)
    # (f*g)[i] = sum_j f[j] * g[i-j] ... actually np.convolve convention
    # np.convolve(f, g)[k] = sum_j f[j]*g[k-j]
    # So for the constraint (f*g)[k] <= M:
    # sum_j g[k-j] * f[j] <= M
    # For fixed j, g appears at position k-j, i.e., row (j+m) has coefficient f[j] on g[m]
    # A[j+m, m] = f[j] for m=0..n-1, j=0..n-1, row index j+m in [0, 2n-2]

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, :] = f[j] * np.ones(n)
        # Wait, that's not right. Let me think again.
        # np.convolve(f, g)[k] = sum_{j} f[j] * g[k-j] where 0<=j<n, 0<=k-j<n
        # For the coefficient of g[m] in (f*g)[k]: we need k-j=m, so j=k-m.
        # So (f*g)[k] = sum_m g[m] * f[k-m] (where f is defined on [0,n-1])
        # A[k, m] = f[k-m] if 0 <= k-m < n, else 0

    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        # A[k, m] = f[k-m] for k-m in [0,n-1], i.e., k in [m, m+n-1]
        k_min = m
        k_max = min(m + n - 1, 2 * n - 2)
        A[k_min:k_max + 1, m] = f[0:k_max - k_min + 1]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_refine(f, deadline, max_iters=200, verbose=True):
    """LP-based refinement of a normalized sequence."""
    n = len(f)
    f = normalize(f).copy()
    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M
    t_step = 0.4
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=min(20.0, max(5.0, deadline - time.monotonic() - 1.0)))
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Try full step
        g_score = 2.0 * n * float(np.convolve(g, g).max())
        if g_score < best_score * 0.998:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
        else:
            # Blended step
            f_new = normalize((1.0 - t_step) * f + t_step * g)
            if f_new is None:
                break
            M_new = float(np.convolve(f_new, f_new).max())
            score_new = 2.0 * n * M_new
            if score_new < best_score - 1e-12:

                best_score = score_new
                best_M = M_new
                f = f_new.copy()
                M = M_new

                no_improve += 1
                f = best_f.copy()
                M = best_M
                t_step = max(0.05, t_step * 0.8)
                if no_improve > 100:
                    break

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, t={t_step:.3f}")

    return best_f, best_score


def interpolate_seq(a, n_new):
    """Interpolate a normalized sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_profiles(n, rng):
    """Generate diverse initial profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, -0.5, -1.0]:
        a = np.maximum(1.0 + eta * u, 1e-10)
        profiles.append(a / a.sum())

    # Quadratic
    for c1 in [0.5, 1.0, -0.5]:
        for c2 in [0.2, 0.5, -0.3]:
            a = np.maximum(1.0 + c1 * u + c2 * u * u, 1e-10)
            profiles.append(a / a.sum())

    # Gaussian-like
    for sigma in [0.3, 0.5, 0.7, 1.0]:
        a = np.exp(-0.5 * (u / sigma) ** 2)
        profiles.append(a / a.sum())

    # Random smooth profiles
    for _ in range(5):
        coeffs = rng.normal(0, 0.5, 6)
        a = 1.0
        for k, c in enumerate(coeffs, 1):
            a = a + c * (u ** k)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    return profiles

            else:
                print(f"  Phase 6: no improvement (best={best_score:.8f}, got={sc_best:.8f})")

    # --- Final validation ---
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        # Fallback
        n_fb = 256
        t = np.linspace(0, 1, n_fb)
        best_seq = np.maximum(1.0 + 1.5 * (2 * t - 1), 0.01)
        best_seq = best_seq / best_seq.sum()
        s = best_seq.sum()
    best_seq = best_seq / s

    elapsed = time.monotonic() - t_start
    print(f"Total time: {elapsed:.1f}s")
    print(f"Final score: {internal_score(best_seq):.8f}")

    return [float(x) for x in best_seq]
