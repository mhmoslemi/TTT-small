Wait no. If i=j, the term contributes s[k-j] (since d/ds[j] of s[j]*s[k-j] = s[k-j]). If i=k-j (and k-j ≠ j), the term contributes s[k-j] (since d/ds[j] of s[k-j]*s[j] = s[k-j]). So total = 2*s[k-j].

If k = 2j: only i=j = k-j, and the term is s[j]^2, derivative = 2*s[j] = 2*s[k-j]. Same formula!

So the subgradient of conv_k with respect to s[j] is: `2 * s[k-j]` where s[m] = 0 for m < 0 or m >= n.

This is a nice formula. It means the subgradient is just a "flipped and shifted" version of s.

Now, for the subgradient step:
`s_new[j] = s[j] - eta * 2 * s[k*-j]` (where k* is the current peak)

Then project onto the simplex (non-negative, sum = 1).

This is a very simple update! And it directly targets the current peak.

The issue with pure subgradient is oscillation (the peak might jump to a different location after each step). To mitigate this, I can:
1. Use a decreasing step size (like 1/sqrt(t))
2. Average over multiple steps (Polyak averaging)
3. Use a "smoothed" version: average the subgradients over the top M peaks

- Subgradient: g[j] = (1/M) * sum_m 2*s[k_m - j]
- Update: s_new = project(s - eta * g)

With M=1: focus on the single peak (fast but oscillatory)
With M=10: balance between peaks (slower but smoother)
With M=all: equivalent to the uniform gradient (slowest)

I think a good strategy is to start with small M (focus on the peak) and increase M over time (for smoother convergence near the optimum).

Schedule: beta = beta_0 * exp(alpha * t) or beta = beta_0 + alpha * t, capped at beta_max.

Combined with a decreasing step size: eta = eta_0 / (1 + decay * t).

Score = 2n * max_conv / (sum)^2

If I have a "good" sequence of length n with score S_n, and I repeat it to get length 2n, what happens?

If s has length n and I form S = [s, s] (length 2n):
- sum(S) = 2*sum(s)
- conv(S, S) at the center has contributions from conv(s,s) plus cross terms
- The max of conv(S,S) is roughly 2*max(conv(s,s)) (the cross terms add about the same as the self terms)

import numpy as np
import time

"""
Minimizes 2n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
Uses adaptive multiplicative gradient descent with FFT-accelerated computation,
progressive peak focusing (increasing softmax sharpness), multiple diverse
initializations, multiple sequence lengths, and plateau-escaping perturbations.
"""


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2"""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return float(2.0 * n * conv.max() / (total * total))


def gradient_fft(s, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. s using FFT. Returns (n,) array."""
    n = len(s)
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    W = np.fft.rfft(w, N_fft)
    term1 = np.fft.irfft(np.conj(S) * W, N_fft)[0:n]
    term2 = np.fft.irfft(S * W, N_fft)[0:n]
    return term1 + term2


def local_search(s, deadline, rng):
    """
    Adaptive multiplicative gradient descent with progressive peak focusing.
    Returns (best_s, best_score).
    """
    n = len(s)
    s = np.maximum(s, 0.0).copy()
    total = s.sum()
    if total < 1e-15:
        return s, float('inf')
    s = s / total

    best_s = s.copy()
    best_score = fast_score(s)
    start_time = time.monotonic()

    # Adaptive parameters
    eta = 0.4
    beta = 6.0
    no_improve = 0
    patience = 120
    last_improve_it = 0

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        # Compute convolution
        conv = np.convolve(s, s)
        max_conv = conv.max()

        if max_conv < 1e-15:
            break

        # Progressive beta increase: start broad, end focused
        progress = min(1.0, it / 5000.0)
        beta_target = 6.0 + 40.0 * progress
        beta = min(max(beta, beta_target * 0.9), 60.0)

        # Softmax weights
        scaled = (conv - max_conv) / max_conv
        w = np.exp(beta * scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # Gradient via FFT
        grad = gradient_fft(s, w)
        grad = np.maximum(grad, 0.0)
        gmax = grad.max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        # Multiplicative update
        s_new = s * np.exp(-eta * grad)
        s_new = np.maximum(s_new, 1e-12)
        s_new = s_new / s_new.sum()

        score_new = fast_score(s_new)

        # Track improvement
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_s = s_new.copy()
            no_improve = 0
            last_improve_it = it
            eta = min(eta * 1.02, 0.75)
        elif score_new < best_score:
            no_improve += 1
            eta = max(eta * 0.99, 0.1)
        else:
            no_improve += 2
            eta = max(eta * 0.95, 0.05)

        s = s_new

        # Plateau detection and escape
        if no_improve >= patience:
            # Scale perturbation with stagnation duration
            stall = no_improve / patience
            scale = min(0.02 + 0.02 * stall, 0.1)
            s = s + scale * rng.standard_normal(n)
            s = np.maximum(s, 0.001)
            s = s / s.sum()
            eta = 0.35
            no_improve = 0
        elif no_improve >= patience * 2 // 3:
            # Pre-plateau: increase focus
            beta = min(beta * 1.2, 60.0)

        # Periodic small randomization to prevent over-fitting to local structure
        if it % 3000 == 2999:
            k = max(1, n // 64)
            idx = rng.choice(n, size=k, replace=False)
            s[idx] *= rng.uniform(0.6, 1.4, size=k)
            s = np.maximum(s, 0.001)
            s = s / s.sum()

    return best_s, best_score


def local_search_aggressive(s, deadline, rng):
    """
    More aggressive variant: alternates between peak-focused (M=1) and
    smoothed (M=top-5) subgradient steps.
    """
    n = len(s)
    s = np.maximum(s, 0.0).copy()
    total = s.sum()
    if total < 1e-15:
        return s, float('inf')
    s = s / total

    best_s = s.copy()
    best_score = fast_score(s)

    eta = 0.3
    no_improve = 0
    patience = 80

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(s, s)
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Alternate between focused and smoothed gradients
        if it % 3 == 2:
            # Focused: only the single peak
            k_star = np.argmax(conv)
            w = np.zeros(len(conv))
            w[k_star] = 1.0
        else:
            # Smoothed: top-5 peaks with softmax
            top_k = np.argsort(conv)[-5:]
            top_vals = conv[top_k]
            w = np.zeros(len(conv))
            for idx in top_k:
                w[idx] = np.exp(5.0 * (conv[idx] - top_vals.max()) / max_conv)
            w = w / w.sum()

        grad = gradient_fft(s, w)
        grad = np.maximum(grad, 0.0)
        gmax = grad.max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        s_new = s * np.exp(-eta * grad)
        s_new = np.maximum(s_new, 1e-12)
        s_new = s_new / s_new.sum()

        score_new = fast_score(s_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_s = s_new.copy()
            no_improve = 0
            eta = min(eta * 1.03, 0.6)
        else:
            no_improve += 1
            eta = max(eta * 0.97, 0.05)

        s = s_new

        if no_improve >= patience:
            s = s + 0.03 * rng.standard_normal(n)
            s = np.maximum(s, 0.001)
            s = s / s.sum()
            eta = 0.3
            no_improve = 0

    return best_s, best_score


def generate_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Ramps
    for a in [-0.8, -0.4, 0.4, 0.8]:
        s = np.maximum(1.0 + a * (2.0 * x - 1.0), 0.01)
        inits.append(s / s.sum())

    # Exponentials
    for alpha in [-4, -2, -1, 1, 2, 4]:
        s = np.exp(alpha * (x - 0.5))
        inits.append(s / s.sum())

    # Cosine-based
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            s = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * x), 0.01)
            inits.append(s / s.sum())

    # Two-level random
    for _ in range(8):
        s = rng.choice([0.05, 1.0], size=n)
        inits.append(s / s.sum())

    # Pure random
    for _ in range(8):
        s = rng.random(n) + 0.02
        inits.append(s / s.sum())

    # Gaussian bumps
    for center in [0.2, 0.4, 0.5, 0.6, 0.8]:
        for width in [0.08, 0.15, 0.25]:
            s = np.exp(-((x - center) ** 2) / (2 * width ** 2))
            s = np.maximum(s, 0.001)
            inits.append(s / s.sum())

    # Triangular
    for center in [0.3, 0.5, 0.7]:
        s = np.maximum(1.0 - np.abs(x - center) / 0.5, 0.01)
        inits.append(s / s.sum())

    # Sawtooth patterns
    for period in [4, 8, 16]:
        s = np.abs(((x * period) % 1.0) * 2.0 - 1.0) + 0.1
        inits.append(s / s.sum())

    return inits
