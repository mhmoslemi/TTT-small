"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with soft-max smoothing (multi-scale beta annealing), efficient
FFT-based gradient computation, diverse initializations, and coordinate descent polish.
Key innovation: closed-form gradient in logit space using FFT-based weighted correlation.
"""
import numpy as np
import time
from scipy.optimize import minimize as sp_minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_conv(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def softmax_stable(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -500.0, 500.0)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / z.size
    return e / s


def exact_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 exactly."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def smooth_objective_and_grad(z, beta, n):
    """
    Smooth objective 2n*soft_max(conv(softmax(z))) and its gradient w.r.t. z.
    Uses FFT-based weighted correlation for O(n log n) gradient.
    """
    A = softmax_stable(z)
    conv = fft_conv(A)
    # Soft-max value and weights
    x = conv / beta
    x = x - np.max(x)
    e = np.exp(x)
    se = float(e.sum())
    if not np.isfinite(se) or se <= 0:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
        val = 2.0 * n * conv.max()
        return val, np.zeros(n)
    w = e / se
    val = 2.0 * n * beta * np.log(se)
    # v = weighted mean of conv
    v = float(np.dot(w, conv))
    # u[m] = sum_k w[k] * A[k-m] via FFT cross-correlation
    N = next_pow2(3 * n + 1)
    W = np.fft.fft(w, N)
    Apad = np.zeros(N, dtype=np.complex128)
    Apad[:n] = A
    AP = np.fft.fft(Apad)
    u_full = np.fft.ifft(W * np.conj(AP)).real[:n]
    # Gradient: (4n/beta) * A * (u - v)
    grad = (4.0 * n / beta) * A * (u_full - v)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return val, grad


def lbfgs_phase(z0, beta, n, deadline, maxiter=2000):
    """Run L-BFGS-B on the smooth objective. Returns (best_z, best_val)."""
    t_chk = [deadline]

    def time_cb(xk):
        return time.monotonic() > t_chk[0]

    best_z = z0.copy()
    best_val, _ = smooth_objective_and_grad(z0, beta, n)

    def fun(z):
        return smooth_objective_and_grad(z, beta, n)[0]

    def jac(z):
        return smooth_objective_and_grad(z, beta, n)[1]

    try:
        res = sp_minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            callback=time_cb,
            options={'maxiter': maxiter, 'maxcor': 20, 'ftol': 1e-14, 'gtol': 1e-8}
        )
        if res.fun < best_val:
            best_val = float(res.fun)
            best_z = res.x.copy()
    except Exception:
        pass
    return best_z, best_val


def adam_refine(z0, deadline, rng, lr=0.05, beta_init=None, n=512):
    """Adam optimizer in logit space with adaptive beta."""
    z = z0.copy()
    A = softmax_stable(z)
    conv = fft_conv(A)
    maxc = float(conv.max())
    if beta_init is None:
        beta = max(1e-6, 0.1 * maxc)
    else:
        beta = beta_init

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_z = z.copy()
    best_val = 2.0 * n * maxc
    no_imp = 0

    while time.monotonic() < deadline:
        conv = fft_conv(A)
        val, grad = smooth_objective_and_grad(z, beta, n)

        if np.isfinite(val) and val < best_val - 1e-12:
            best_val = val
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 200:
            # Adaptive: reduce beta or restart
            if beta > 1e-5:
                beta *= 0.5
            else:
                z = best_z + 0.3 * rng.normal(size=n)
                z = np.clip(z, -50, 50)
                A = softmax_stable(z)
                m = np.zeros(n)
                v = np.zeros(n)
                t_adam = 0
                no_imp = 0
                lr = max(0.02, lr * 0.8)
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        A = softmax_stable(z)

        if t_adam % 100 == 0:
            beta = max(1e-7, beta * 0.7)

    return best_z, best_val


def coord_descent(a, deadline, rng, n_sweeps=50):
    """Coordinate descent with exact max evaluation."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = exact_score(a)
    no_imp = 0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = a[i]
            best_val_i = orig
            best_sc_i = best_score
            # Try different scale factors
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                new_val = orig * factor
                if new_val < 1e-15:
                    continue
                a[i] = new_val
                ns = a.sum()
                if ns < 0.01:
                    a[i] = orig
                    continue
                sc = exact_score(a)
                if np.isfinite(sc) and sc < best_sc_i:
                    best_sc_i = sc
                    best_val_i = new_val
            a[i] = best_val_i
            if best_sc_i < best_score:
                best_score = best_sc_i
                best_a = a.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        if no_imp > 3 * n:
            break
        if not improved and sweep > 10:
            break

    return best_a, best_score


def make_diverse_profiles(n, count=30):
    """Generate diverse initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    profs = []
    # Uniform
    profs.append(np.ones(n) / n)
    # Linear
    for slope in [-2.0, -1.0, -0.5, 0.5, 1.0, 2.0, 3.0, 5.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 0.01)
        profs.append(a / a.sum())
    # Power laws
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = t**p + 0.01
        profs.append(a / a.sum())
        a = (1.0 - t)**p + 0.01
        profs.append(a / a.sum())
    # Cosine harmonics
    for k in [1, 2, 3, 4, 5, 6]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            a = 1.0 + amp * np.cos(2 * np.pi * k * t)
            a = np.maximum(a, 0.01)
            profs.append(a / a.sum())
            a = 1.0 + amp * np.sin(2 * np.pi * k * t)
            a = np.maximum(a, 0.01)
            profs.append(a / a.sum())
    # Two-hump
    for c1 in [0.2, 0.3, 0.4]:
        for w in [0.1, 0.2, 0.3]:
            a = np.exp(-((t - c1) / w)**2) + np.exp(-((t - (1-c1)) / w)**2) + 0.01
            profs.append(a / a.sum())
    # Gaussian center
    for w in [0.1, 0.15, 0.2, 0.3]:
        a = np.exp(-((t - 0.5) / w)**2) + 0.01
        profs.append(a / a.sum())
    return profs[:count + len(profs) - count] if count < len(profs) else profs
