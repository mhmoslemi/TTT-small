d loss / d a_i = grad_i (as computed above)
d a_i / d z_j = a_i * (delta_{ij} - a_j) (Jacobian of softmax)

So d loss / d z_j = sum_i (d loss / d a_i) * (d a_i / d z_j)
= sum_i grad_i * a_i * (delta_{ij} - a_j)
= grad_j * a_j - a_j * sum_i grad_i * a_i
= a_j * (grad_j - sum_i grad_i * a_i)

Let mean_grad = sum_i grad_i * a_i (the weighted average of the gradient)
Then d loss / d z_j = a_j * (grad_j - mean_grad)

This is nice and simple!

def optimize_sequence(a0, deadline, rng, beta0=500, beta_min=50, n_iter=10000, lr=0.01):

    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    no_improve = 0
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution and loss
        c = fft_convolve(a)
        c_max = c.max()
        ex = np.exp(np.clip(beta * (c - c_max), -100, 50))
        w = ex / ex.sum()
        loss = c_max + np.log(ex.sum()) / beta
        
        # Compute gradient
        a_rev = a[::-1]
        N = next_pow2(3 * n - 1)
        W = np.fft.rfft(w, N)

        grad_a = 2.0 * g
        
        # Convert to z-space gradient
        mean_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - mean_grad)
        
        # Adam update
        t += 1
        m = b1 * m + (1-b1) * grad_z
        v = b2 * v + (1-b2) * grad_z**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back to a
        z -= z.max()  # numerical stability
        a = np.exp(z)
        a = a / a.sum()
        
        # Anneal beta
        if it % 200 == 0:
            beta = max(beta_min, beta * 0.8)
        
        # Check improvement
        s = compute_score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 500:
            z = np.log(best_a) - np.log(best_a).mean()
            z += 0.1 * rng.normal(size=n)
            z -= z.max()
            a = np.exp(z)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            beta = beta0 * 2
            no_improve = 0
    
    return best_a, best_score

"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses softmax-parameterized gradient descent with Adam optimizer, FFT-based
gradients, log-sum-exp smoothing with annealing, and multi-start exploration
across sequence lengths (512, 768, 1024, 1536, 2048).
"""

import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([])
    out_len = n + m - 1
    if out_len <= 256:
        return np.convolve(a, b)
    N = next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:out_len]
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_grad_a(a, w):
    """Compute gradient of smoothed max-conv w.r.t. a.
    grad_i = 2 * sum_{m=0}^{n-1} w[m+i] * a[m]
    """
    n = a.size
    # g[i] = conv(w, a_rev)[n-1+i]
    a_rev = a[::-1]
    out_len = (2 * n - 1) + n - 1  # 3n-2
    N = next_pow2(out_len)
    W = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a_rev, N)
    conv_res = np.fft.irfft(W * Ar, N)[:out_len]
    g = conv_res[n - 1:2 * n - 1]
    return 2.0 * g


def softmax_from_z(z):
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-30:
        return np.ones(z.size) / z.size
    return a / s


def gradient_optimize(a0, deadline, rng, beta_start=500.0, beta_min=30.0,
                      lr=0.02, max_iter=50000, restart_threshold=400):
    """Optimize using softmax-parameterized Adam with annealed log-sum-exp."""
    n = a0.size
    a = np.maximum(a0.copy(), 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    beta = beta_start
    best_a = a.copy()
    best_score = compute_score(a)
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(a)
        c_max = float(c.max())
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -80.0, 50.0)
        ex = np.exp(shifted)
        ex_sum = float(ex.sum())
        if ex_sum < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            ex_sum = 1.0
        w = ex / ex_sum

        grad_a = compute_grad_a(a, w)

        # Convert to z-space: grad_z_j = a_j * (grad_a_j - sum_i grad_a_i * a_i)
        mm = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - mm)

        if not np.all(np.isfinite(grad_z)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= z.mean()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * grad_z
        v_adam = b2 * v_adam + (1.0 - b2) * (grad_z * grad_z)
        m_hat = m_adam / (1.0 - b1 ** t_adam)
        v_hat = v_adam / (1.0 - b2 ** t_adam)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()

        a = softmax_from_z(z)

        # Anneal beta
        if it % 300 == 0 and beta > beta_min:
            beta = max(beta_min, beta * 0.75)

        score = compute_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > restart_threshold:
            # Restart from best with perturbation
            z = np.log(np.maximum(best_a, 1e-12))
            z -= z.mean()
            scale = 0.05 + 0.1 * rng.random()
            z += scale * rng.normal(size=n)
            z -= z.mean()
            a = softmax_from_z(z)
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            beta = min(beta_start * 1.5, max(beta_min * 5, beta * 2))
            no_improve = 0

    return best_a, best_score

def interpolate_seq(a0, n_new):
    """Linearly interpolate a sequence to a new length."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n0 = a0.size
    if n0 == n_new:
        return np.maximum(a0, 0.0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    a1 = np.interp(x1, x0, a0)
    a1 = np.maximum(a1, 0.0)
    s = a1.sum()
    if s < 1e-30:
        return None
    return a1 / s


def make_linear_profile(n, eta):

    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def make_parabolic(n, alpha):

    u = 2.0 * t - 1.0

def make_smooth_random(n, rng, n_harm=10):

    a = np.ones(n)

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def coordinate_polish(a0, deadline, max_rounds=10):
    """Polish by optimizing each coordinate (cyclic coordinate descent)."""
    n = a0.size
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-30:
        return a0
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Fix all except a[i], optimize a[i]
            # a[i] = 1 - sum_{j!=i} a[j]
            rest_sum = a.sum() - a[i]
            if rest_sum >= 1.0:
                continue
            # We need a[i] + rest_sum = 1, so a[i] = 1 - rest_sum
            # But we can redistribute: set a[i] = t, and scale the rest to sum to 1-t
            # Actually, let's just try a few values of a[i]
            best_t = a[i]
            best_local = None

            # Compute current c for reference
            c_base = fft_convolve(a)

            # Try perturbing a[i]
            for frac in (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.3, 0.1):
                t = a[i] * frac
                if t < 0 or t > 0.9:
                    continue
                a_new = a.copy()
                a_new[i] = t
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                sc = compute_score(a_new)
                if best_local is None or sc < best_local:
                    best_local = sc
                    best_t = t

            if best_local is not None and best_local < best_score - 1e-10:
                a[i] = best_t
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a

        p1b, s1b = gradient_optimize_v2(p1, phase1b_deadline, rng,
                                        beta_start=800.0, beta_min=50.0, lr=0.012,
                                        max_iter=100000, restart_threshold=250)
        try_update(p1b, s1b)
        print(f"  Phase 1b (v2) result: {s1b:.10f}")

    # Phase 2: Try different initializations at n=512
    phase2_deadline = min(deadline, t_start + 900.0)
    starts = []
    for eta in (1.0, 1.3, 1.7, 2.0, 2.5, 3.0):
        starts.append(make_linear_profile(n_base, eta))
    for width in (0.5, 0.8, 1.0, 1.3):
        starts.append(make_raised_cosine(n_base, width))
    for alpha in (0.5, 1.0, 1.5, 2.0):
        starts.append(make_parabolic(n_base, alpha))
    for _ in range(6):
        starts.append(make_smooth_random(n_base, rng, n_harm=8))

    per_start = max(3.0, (phase2_deadline - t_start) / max(1, len(starts)))
    for i, p0 in enumerate(starts):
        if time.monotonic() > phase2_deadline:
            break
        per = min(phase2_deadline, time.monotonic() + per_start)
        p_c, s_c = gradient_optimize(p0, per, rng,
                                     beta_start=400.0, beta_min=30.0, lr=0.025,
                                     max_iter=50000, restart_threshold=200)
        try_update(p_c, s_c)
        if i % 4 == 0:
            print(f"  Phase 2 start {i}/{len(starts)}: best={best_score:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Coordinate descent polish
    if best_seq is not None and best_seq.size == n_base:
        if time.monotonic() < deadline - 200:
            cd_deadline = min(deadline, time.monotonic() + 100.0)
            p_cd = coordinate_polish(best_seq, cd_deadline, max_rounds=20)
            sc_cd = compute_score(p_cd)
            try_update(p_cd, sc_cd)
            print(f"Phase 3 (coord descent): {sc_cd:.10f}")

    # Phase 4: Try longer sequences
    for n_long in (1024, 1536, 2048):
        if time.monotonic() > deadline - 120:
            break
        if best_seq is not None:
            p_ext = interpolate_seq(best_seq, n_long)
            if p_ext is not None:
                per = min(deadline - 60, time.monotonic() + 80.0)
                p_l, s_l = gradient_optimize(p_ext, per, rng,
                                             beta_start=400.0, beta_min=20.0,
                                             lr=0.02, max_iter=50000,
                                             restart_threshold=300)
                sc_l = compute_score(p_l)
                try_update(p_l, sc_l)
                print(f"Phase 4 (n={n_long}): {sc_l:.10f}")

    # Phase 5: Final refinement on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        fin_deadline = deadline - 5
        p_f, s_f = gradient_optimize(best_seq, fin_deadline, rng,
                                     beta_start=300.0, beta_min=20.0,
                                     lr=0.01, max_iter=200000,
                                     restart_threshold=500)
        try_update(p_f, s_f)
        print(f"Phase 5 (final, n={n_best}): {s_f:.10f}")

    # Phase 6: Coordinate polish on final best
    if best_seq is not None and time.monotonic() < deadline - 10:
        p_final_cd = coordinate_polish(best_seq, deadline - 3, max_rounds=10)
        sc_final_cd = compute_score(p_final_cd)
        try_update(p_final_cd, sc_final_cd)
        print(f"Phase 6 (final coord): {sc_final_cd:.10f}")

    # Final output
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final: score={sc_final:.10f}, n={final.size}")
        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
