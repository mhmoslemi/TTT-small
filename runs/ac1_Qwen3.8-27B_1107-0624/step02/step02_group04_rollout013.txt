Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2 using 
exponentiated gradient (mirror descent) with FFT-based gradient computation,
multi-scale search, and active-set refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve_fft(a):
    """Compute full linear convolution using FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _gradient_from_conv_weights(a, w):
    """
    Compute gradient of log-sum-exp(beta * conv(a,a)) given soft-max weights w.
    w has length 2n-1 (the soft-max distribution over convolution values).
    Returns gradient of length n.
    """
    n = a.size
    # V = w reversed
    V = w[::-1]
    # convolve a (length n) with V (length 2n-1)
    # result has length n + (2n-1) - 1 = 3n-2
    # gradient[i] = 2 * result[i + n - 1] for i = 0, ..., n-1
    N = _next_pow2(3 * n - 2)
    fa = np.fft.rfft(a, N)
    fV = np.fft.rfft(V, N)
    conv_result = np.fft.irfft(fa * fV, N)
    grad = 2.0 * conv_result[(n-1):(2*n-2)]
    return grad

    a = a / a.sum()

        r = _full_convolve_fft(a)
        # Find top-k values
        top_idx = np.argsort(r)[-top_k:]
        
        # Use a very high beta on the top-k, lower on the rest
        beta_high = 1000.0
        beta_low = 10.0
        
        br = np.empty_like(r)
        br[top_idx] = beta_high * r[top_idx]
        mask_rest = np.ones(len(r), dtype=bool)
        mask_rest[top_idx] = False
        br[mask_rest] = beta_low * r[mask_rest]
        
        br = br - br.max()
        exp_br = np.exp(br)
        s = exp_br.sum()
        w = exp_br / s
        obj = (br.max() + np.log(s)) / 1.0  # This isn't quite right as a single LSE
        # Actually, let's just use the standard LSE with a moderate beta
        # and let the optimizer figure it out
        
        # Simplified: just use LSE with increasing beta
        beta = 50.0 + it * 0.5
        br2 = beta * r
        br2 = br2 - br2.max()
        exp_br2 = np.exp(br2)
        s2 = exp_br2.sum()
        w2 = exp_br2 / s2
        obj2 = r.max() + np.log(s2) / beta  # = LSE(beta*r) / beta
        
        if obj2 < best_obj - 1e-14:
            best_obj = obj2
            best_a = a.copy()
        
        grad = _gradient_from_conv_weights(a, w2)
        
        # Gradient step with backtracking
        eta = 0.1 / (1.0 + it * 0.01)
        for attempt in range(10):
            a_new = a * np.exp(-eta * grad)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            r_new = _full_convolve_fft(a_new)
            if r_new.max() <= r.max() + 1e-10:
                a = a_new
                break
            eta *= 0.5
        else:
            a = a  # keep current
    
    # Final score
    best_a = np.maximum(best_a, 1e-15)
    best_a = best_a / best_a.sum()
    return best_a

def _make_bell_profile(n, width=0.5):
    """Create a bell-shaped profile."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _make_offset_profile(n, offset=0.3):
    """Create an offset (asymmetric) profile."""
    t = np.linspace(0, 1, n)
    center = 0.5 - offset
    width = 0.3
    a = np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _make_pulse_superposition(n, n_pulses, rng):
    """Create a sequence as superposition of Gaussian pulses."""
    a = np.zeros(n)
    for _ in range(n_pulses):
        center = rng.integers(0, n)
        width = rng.uniform(n * 0.02, n * 0.15)
        amp = rng.uniform(0.5, 2.0)
        t = np.arange(n)
        a += amp * np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _lgbfsb_polish(a0, deadline, beta_seq=None):
    """
    Polish using L-BFGS-B with log-sum-exp objective.
    Work in log-space to maintain positivity.
    """
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    if beta_seq is None:
        beta_seq = [10.0, 50.0, 200.0, 1000.0]
    
    z = np.log(a0)  # log-space parameterization
    z -= z.mean()  # center
    
    best_a = a0.copy()
    best_score = _internal_score(a0)
    
    for beta in beta_seq:
        if time.monotonic() > deadline - 5:
            break
        
        def obj_grad(z):
            a = np.exp(z - z.max())
            s = a.sum()
            a = a / s
            r = _full_convolve_fft(a)
            br = beta * (r - r.max())
            exp_br = np.exp(br)
            ls = exp_br.sum()
            obj = r.max() + np.log(ls) / beta
            
            w = exp_br / ls
            grad_a = _gradient_from_conv_weights(a, w)
            # Chain rule: grad_z_i = a_i * (grad_a_i - sum(a_j * grad_a_j))
            dot = np.dot(a, grad_a)
            grad_z = a * (grad_a - dot)
            grad_z -= grad_z.mean()
            
            return obj, grad_z
        
        # Use L-BFGS-B
        result = minimize(obj_grad, z, method='L-BFGS-B',
                         options={'maxiter': 200, 'gtol': 1e-6,
                                  'maxfun': 500})
        
        z_opt = result.x
        a_opt = np.exp(z_opt - z_opt.max())
        a_opt = a_opt / a_opt.sum()
        
        sc = _internal_score(a_opt)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            z = z_opt
    
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses exponentiated gradient, L-BFGS-B polish, active-set refinement,
    and multi-scale escalation.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            if label:
                print(f"  NEW BEST: {best_score:.10f} (n={best_seq.size}) [{label}]")

    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:

            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    print(f"Parent: {'n=' + str(parent.size) if parent is not None else 'not available'}")
    if parent is not None:
        print(f"  Parent score: {_true_score(parent):.10f}")
    
    # --- Phase 0: Generate diverse candidates ---
    candidates = []
    
    # Parent at its native length
    if parent is not None:
        candidates.append(('parent', parent.copy(), parent.size))
    
    # Various profiles at length 1024
    n_base = 1024
    profiles = [
        ('uniform', np.ones(n_base) / n_base),
        ('bell', _make_bell_profile(n_base, 0.5)),
        ('bell_narrow', _make_bell_profile(n_base, 0.3)),
        ('offset_0.2', _make_offset_profile(n_base, 0.2)),
        ('offset_0.3', _make_offset_profile(n_base, 0.3)),
        ('offset_0.1', _make_offset_profile(n_base, 0.1)),
        ('linear', None),  # will fill below
    ]
    
    t = np.linspace(0, 1, n_base)
    lin = np.maximum(1.0 + 2.0 * (t - 0.5), 1e-15)
    profiles[6] = ('linear', lin / lin.sum())
    
    # Anti-symmetric (high on one side, low on other)
    anti = np.maximum(2.0 - 3.0 * t, 1e-15)
    profiles.append(('anti_lin', anti / anti.sum()))
    
    # Quadratic (parabola)
    quad = np.maximum(1.0 - 2.0 * (t - 0.5) ** 2, 1e-15)
    profiles.append(('parabola', quad / quad.sum()))
    
    # Sine
    sine = np.maximum(np.sin(np.pi * t), 1e-15)
    profiles.append(('sine', sine / sine.sum()))
    
    # Cosine bump (centered)
    cos_bump = 0.5 * (1 + np.cos(2 * np.pi * t))
    cos_bump = np.maximum(cos_bump, 1e-15)
    profiles.append(('cos_bump', cos_bump / cos_bump.sum()))
    
    # Random pulse superpositions
    for i in range(5):
        pp = _make_pulse_superposition(n_base, 8 + i * 4, rng)
        profiles.append((f'pulses_{8+i*4}', pp))
    
    # Interpolate parent to 1024 if different length
    if parent is not None and parent.size != n_base:
        x0 = np.linspace(0, 1, parent.size)
        x1 = np.linspace(0, 1, n_base)
        p_interp = np.interp(x1, x0, parent)
        p_interp = np.maximum(p_interp, 1e-15)
        p_interp = p_interp / p_interp.sum()
        profiles.append(('parent_interp_1024', p_interp))
    
    for label, p in profiles:

    print(f"Phase 3: best={best_score:.10f}")
    
    # --- Phase 4: Multi-restart from best ---

        s = final.sum()
        if s > 0.01:

        # Fallback
        final = _make_bell_profile(1024, 0.5)
        print(f"FALLBACK: n=1024, score={_true_score(final):.10f}")
    
    print(f"FINAL: score={best_score:.10f}, n={len(final)}")
    return [float(x) for x in final]
