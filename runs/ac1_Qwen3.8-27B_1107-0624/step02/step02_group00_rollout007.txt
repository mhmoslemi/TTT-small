def _active_set_lp(a, M, active_indices):
    """Solve LP: max sum(b) s.t. (a*b)_j <= M for j in active_indices, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    
    # Build constraint matrix (only for active indices)
    m = len(active_indices)
    A = np.zeros((m, n))
    for i, j in enumerate(active_indices):
        # Row j: A[j,k] = a[k] if 0 <= j-k < n
        k_min = max(0, j - (n - 1))
        k_max = min(n - 1, j)
        for k in range(k_min, k_max + 1):
            A[i, k] = a[k]
    
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
