def solve_cross_conv_lp(f, M_hint=None):
    """Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (t_opt, g_opt)."""
    n = len(f)
    # Variables: [t, g_0, ..., g_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0  # minimize t
    
    # Inequality constraints: (f*g)[k] - t <= 0 for k = 0, ..., 2n-2
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    for k in range(2*n-1):
        A_ub[k, 0] = -1.0  # coefficient of t
        for j in range(n):
            i = k - j  # f[i] * g[j]
            if 0 <= i < n:
                A_ub[k, j+1] = f[i]
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: t is free (or >= 0), g >= 0
    bounds = [(None, None)] + [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        t_opt = res.x[0]
        g_opt = np.maximum(res.x[1:], 0)
        return t_opt, g_opt
    return None, None

    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def solve_cross_conv_lp(f, time_limit=3.0):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (t_opt, g_opt) or (None, None).
    """
    n = len(f)
    # Variables: [t, g_0, ..., g_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0

    # Inequality: (f*g)[k] - t <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        A_ub[k, 0] = -1.0
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j + 1] = f[i]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])

    bounds = [(None, None)] + [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        t_opt = float(res.x[0])
        g_opt = np.maximum(res.x[1:], 0.0)
        return t_opt, g_opt
    return None, None


def solve_max_sum_lp(f, M, time_limit=3.0):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = f[i]

    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def soft_max_and_weights(c, beta):
    """Soft max approximation and version weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j] via FFT."""
    n = len(p)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def adam_steps(f, deadline, rng, lr=0.03, max_iters=3000, verbose=False):
    """Adam optimization in logit space. Returns (best_f, best_score)."""
    n = len(f)
    f = np.maximum(f, 1e-12)
    f = f / f.sum()
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.85, 0.999, 1e-9
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = np.convolve(f, f)
    M = float(c.max())
    score = 2.0 * n * M
    beta = max(1e-10, 0.15 * M)

    best_f = f.copy()
    best_score = score
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        L, w = soft_max_and_weights(c, beta)
        gp = 2.0 * corr_weights(w, f)
        vp = gp + 1e-8 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = np.log(best_f + 1e-12)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(best_f + 1e-12)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -40, 40)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta
        if adam_t % 200 == 0:
            beta = max(1e-12, beta * 0.75)

        # Adaptive lr
        if no_imp > 300:
            lr = max(0.005, lr * 0.9)

        # Restart if stuck too long
        if no_imp > max(500, 2 * n):
            pert = 0.2 + 0.1 * rng.random()
            z = np.log(best_f + 1e-12) + pert * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.15 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = 0.03
            no_imp = 0

        if verbose and it % 500 == 0:
            print(f"    adam it={it}: score={score:.10f}, best={best_score:.10f}")

    return best_f, best_score


def lp_iteration(f, deadline, rng, verbose=False):
    """
    Single LP iteration: find direction g via cross-conv LP, line search, update f.
    Returns (f, score, improved).
    """
    n = len(f)
    M = compute_max_conv(f)
    score = 2.0 * n * M

    # Try cross-conv LP first (better direction)
    t_opt, g = solve_cross_conv_lp(f, time_limit=4.0)

    if g is not None and g.sum() > 1e-15:
        g = g / g.sum()
    else:
        # Fallback: max-sum LP
        g = solve_max_sum_lp(f, M * 1.05, time_limit=4.0)
        if g is not None and g.sum() > 1e-15:
            g = g / g.sum()
        else:
            # Random direction
            g = np.exp(rng.normal(size=n))
            g = g / g.sum()

    # Line search
    best_t = 0.0
    best_f = f.copy()
    best_M = M
    for t in [0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
        f_try = (1.0 - t) * f + t * g
        f_try = normalize(f_try)
        if f_try is None:
            continue
        M_try = compute_max_conv(f_try)
        if M_try < best_M - 1e-14:
            best_M = M_try
            best_f = f_try
            best_t = t

    if best_t > 0:
        return best_f, 2.0 * n * best_M, True
    return f, score, False


def lp_optimize(f_init, deadline, rng, max_iters=500, verbose=False):
    """
    LP-based iterative optimization with stall detection.
    Returns (best_f, best_score).
    """
    f = normalize(f_init.copy())
    if f is None:
        return f_init, compute_score_unnorm(f_init)

    n = len(f)
    best_f = f.copy()
    best_score = 2.0 * n * compute_max_conv(f)
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        f, score, improved = lp_iteration(f, deadline, rng)

        if improved and score < best_score - 1e-13:
            best_score = score
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Stall: try perturbation
        if no_imp > 20:
            perturb = 0.15 + 0.1 * rng.random()
            f = best_f + perturb * rng.normal(size=n)
            f = normalize(f)
            if f is None:
                f = best_f.copy()
            no_imp = 0

        if verbose and it % 50 == 0:
            print(f"    LP it={it}: score={score:.10f}, best={best_score:.10f}")

    return best_f, best_score


def hybrid_optimize(f_init, deadline, rng, verbose=False):
    """
    Hybrid: alternate LP directions and Adam steps.
    Returns (best_f, best_score).
    """
    n = len(f)
    f = normalize(f_init.copy())
    if f is None:
        return f_init, compute_score_unnorm(f_init)

    best_f = f.copy()
    best_score = 2.0 * n * compute_max_conv(f)
    no_imp = 0
    mode = 0  # 0=LP, 1=Adam

    max_total = 2000
    for it in range(max_total):
        if time.monotonic() > deadline:
            break

        if mode == 0:
            # LP step
            f, score, improved = lp_iteration(f, deadline, rng)
        else:
            # Short Adam burst
            sub_deadline = min(deadline, time.monotonic() + 5.0)
            f, score = adam_steps(f, sub_deadline, rng, lr=0.02, max_iters=100)

        if score < best_score - 1e-13:
            best_score = score
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Switch mode periodically
        if it % 10 == 0:
            mode = 1 - mode

        if no_imp > 30:
            perturb = 0.2
            f = best_f + perturb * rng.normal(size=n)
            f = normalize(f)
            if f is None:
                f = best_f.copy()
            no_imp = 0
            mode = 0

    return best_f, best_score


def get_starting_points(n, rng, parent=None):
    """Generate diverse starting points."""
    starts = []
    # Deterministic profiles
    starts.append(make_linear_profile(n, 1.0))
    starts.append(make_linear_profile(n, 2.0))
    starts.append(make_linear_profile(n, 3.0))
    starts.append(make_cubic_profile(n, 2.0, -1.5))
    starts.append(make_cubic_profile(n, 2.5, -2.0))
    starts.append(make_cubic_profile(n, 3.0, -2.5))
    starts.append(make_cosine_profile(n, [1, 2, 3], [0.3, 0.15, 0.08]))
    starts.append(make_cosine_profile(n, [1, 3, 5], [0.25, 0.1, 0.05]))
    starts.append(make_cosine_profile(n, [1, 2, 4], [0.4, 0.2, 0.1]))

    # Custom profiles: "bowl" shapes
    t = np.linspace(0, 1, n)
    for a in [1.5, 2.0, 2.5, 3.0]:
        f = 1.0 + a * t ** 2
        f = np.maximum(f, 1e-12)
        starts.append(f / f.sum())

    # "Edge-weighted" profiles
    for a in [1.0, 2.0, 3.0]:
        f = 1.0 + a * (t ** 3 + (1 - t) ** 3)
        f = np.maximum(f, 1e-12)
        starts.append(f / f.sum())

    # Random smooth profiles
    for _ in range(10):
        theta = rng.uniform(-2, 2, 6)
        f = 1.0
        for k in range(3):
            f = f + 0.25 * theta[2 * k] * np.cos(2 * np.pi * (k + 1) * t)
            f = f + 0.25 * theta[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * t)
        f = np.maximum(f, 1e-6)
        f = f / f.sum()
        starts.append(f)

    # Parent (interpolated)
    if parent is not None and len(parent) != n:
        starts.append(interpolate_to_length(parent, n))
    elif parent is not None:
        starts.append(parent.copy())

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP + Adam hybrid, multi-length search.
    Minimizes 2*n*max(convolve(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
                print(f"Parent: n={len(parent)}, score={2.0*len(parent)*compute_max_conv(parent):.10f}")
    except Exception:
        pass

    # Phase 1: Refine parent at its native length
    if parent is not None:
        n_parent = len(parent)
        print(f"\nPhase 1: Refine parent at n={n_parent}")
        phase1_end = min(deadline, t_start + 400.0)

        # LP optimization
        f_best, sc_best = lp_optimize(parent, phase1_end, rng, max_iters=500, verbose=True)
        try_update(f_best, sc_best)
        print(f"  After LP: {sc_best:.10f}")

        # Adam polish
        if time.monotonic() < deadline - 30:
            phase1b_end = min(deadline, time.monotonic() + 100.0)
            f_best2, sc_best2 = adam_steps(f_best, phase1b_end, rng, lr=0.02, max_iters=3000, verbose=True)
            try_update(f_best2, sc_best2)
            print(f"  After Adam: {sc_best2:.10f}")

        # Hybrid
        if time.monotonic() < deadline - 30:
            phase1c_end = min(deadline, time.monotonic() + 100.0)
            f_best3, sc_best3 = hybrid_optimize(f_best2, phase1c_end, rng, verbose=True)
            try_update(f_best3, sc_best3)
            print(f"  After hybrid: {sc_best3:.10f}")

    # Phase 2: Try other lengths with the best profile shape
    print(f"\nPhase 2: Multi-length search, current best={best_score:.10f}")
    ref_profile = best_seq  # Best profile shape to interpolate

    lengths_to_try = [128, 192, 256, 320, 384, 512]
    if ref_profile is not None and len(ref_profile) not in lengths_to_try:
        lengths_to_try.insert(0, len(ref_profile))

    # Allocate remaining time
    remaining = deadline - time.monotonic()
    per_length = max(30.0, remaining / len(lengths_to_try))

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 30:
            break
        print(f"  Trying n={n_try}, time_alloc={per_length:.0f}s")

        # Get starting profiles for this length
        if ref_profile is not None:
            starts = [interpolate_to_length(ref_profile, n_try)]
        else:
            starts = [make_linear_profile(n_try, 2.0)]

        # Add a few random starts
        for _ in range(3):
            theta = rng.uniform(-2, 2, 6)
            t = np.linspace(0, 1, n_try)
            f = 1.0
            for k in range(3):
                f = f + 0.25 * theta[2 * k] * np.cos(2 * np.pi * (k + 1) * t)
                f = f + 0.25 * theta[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * t)
            f = np.maximum(f, 1e-6)
            starts.append(f / f.sum())

        # Evaluate starts
        for s in starts:
            sc = 2.0 * n_try * compute_max_conv(s)
            try_update(s, sc)

        # Optimize from best start
        best_start = min(starts, key=lambda s: 2.0 * n_try * compute_max_conv(s))
        run_end = min(deadline - 20, time.monotonic() + per_length * 0.5)
        f_best, sc_best = lp_optimize(best_start, run_end, rng, max_iters=300)
        try_update(f_best, sc_best)

        # Polish with Adam
        if time.monotonic() < deadline - 20:
            run_end2 = min(deadline - 20, time.monotonic() + per_length * 0.3)
            f_best2, sc_best2 = adam_steps(f_best, run_end2, rng, lr=0.02, max_iters=2000)
            try_update(f_best2, sc_best2)

        # Final hybrid polish
        if time.monotonic() < deadline - 20:
            run_end3 = min(deadline - 20, time.monotonic() + per_length * 0.2)
            f_best3, sc_best3 = hybrid_optimize(f_best2, run_end3, rng)
            try_update(f_best3, sc_best3)

        print(f"  n={n_try} done, best so far={best_score:.10f}")

    # Phase 3: Intensive refinement of the best solution
    print(f"\nPhase 3: Intensive refinement, best={best_score:.10f}")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        print(f"  Refining at n={n_best}")

        # Multiple short LP+Adam cycles
        f_cur = best_seq.copy()
        for cycle in range(5):
            if time.monotonic() > deadline - 20:
                break
            sub_end = min(deadline - 20, time.monotonic() + 40.0)
            f_cur, sc = lp_optimize(f_cur, sub_end, rng, max_iters=200)
            try_update(f_cur, sc)

            if time.monotonic() > deadline - 20:
                break
            sub_end2 = min(deadline - 20, time.monotonic() + 30.0)
            f_cur, sc = adam_steps(f_cur, sub_end2, rng, lr=0.01, max_iters=2000)
            try_update(f_cur, sc)
            print(f"  Cycle {cycle}: score={sc:.10f}, best={best_score:.10f}")

    # Phase 4: Very fine Adam polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nPhase 4: Fine Adam polish")
        sub_end = deadline - 5
        f_final, sc_final = adam_steps(best_seq, sub_end, rng, lr=0.005, max_iters=10000, verbose=True)
        try_update(f_final, sc_final)
        print(f"  Final Adam: {sc_final:.10f}")

    # Final validation
    print(f"\nFinal: best_score={best_score:.10f}")
    if best_seq is None:
        best_seq = make_linear_profile(256, 2.0)

    # Ensure validity
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 2.0)
        s = float(best_seq.sum())
    best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score_unnorm(best_seq)
    print(f"Validated: n={len(best_seq)}, score={final_score:.10f}")

    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
