"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based iterative refinement followed by L-BFGS polish with soft-argmax smoothing,
combined with multi-start restarts for escaping local minima.

import numpy as np
import time
from scipy.optimize import linprog, minimize


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=10.0):
    """Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterate(f, max_iters=500, t_init=0.5, t_min=0.05, deadline=None, verbose=True):
    """LP-based iterations to reduce max(conv(f,f))."""
    n = len(f)

    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M

    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        if g_score < best_score * 0.999:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
        else:
            # Try blended step

                f_new = (1.0 - t) * f + t * g
                f_new = normalize(f_new)
                if f_new is None:

                M_new = float(np.convolve(f_new, f_new).max())

                    best_score = score_new

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:

    x_old = np.linspace(0.0, 1.0, n_old)

    # conv[k] = sum_{i=lo}^{hi} a[i] * a[k-i]
    
    # d conv[k] / d a[m]:
    # Case 1: m is in [lo, hi] (m appears as the first factor)
    #   Contribution: a[k-m]
    # Case 2: k-m is in [lo, hi] (k-m appears as the first factor, so m appears as the second factor)
    #   Contribution: a[m]
    # If both conditions hold and m != k-m: total = a[k-m] + a[m-k+m] = a[k-m] + a[m] ... wait
    # If k-m is in [lo,hi], then the term is a[k-m] * a[k-(k-m)] = a[k-m]*a[m]. 
    # d/d a[m] of a[k-m]*a[m] = a[k-m] (treating a[k-m] as constant when k-m != m)
    # So case 2 gives a[k-m].
    # Total: if m in [lo,hi] AND k-m in [lo,hi] AND m != k-m: 2*a[k-m]
    # If m in [lo,hi] AND k-m in [lo,hi] AND m == k-m: a[m] (it's a[m]*a[m], derivative is 2*a[m]... wait)
    
    # Actually, a[m]*a[m] has derivative 2*a[m] w.r.t. a[m]. So even in the equal case, it's 2*a[m].
    # But m == k-m means k = 2m, and the term in the sum is a[m]*a[m] which appears only once (at i=m).
    # So the derivative of that single term is 2*a[m].
    
    # In all cases: d conv[k]/d a[m] = 2 * a[k-m] if 0 <= k-m < n AND m is in the valid range for conv[k].
    
    # But actually, the condition "m is in [lo,hi]" is equivalent to "0 <= k-m < n" (since lo = max(0,k-n+1) and hi = min(n-1,k), and m in [lo,hi] iff 0 <= m < n and 0 <= k-m < n).
    # And "k-m is in [lo,hi]" is also equivalent to "0 <= k-m < n and 0 <= k-(k-m) < n" = "0 <= k-m < n and 0 <= m < n".
    # So both conditions are the same: 0 <= k-m < n.
    
    # Therefore: d conv[k]/d a[m] = 2 * a[k-m] if 0 <= k-m < n, else 0.
    
    # This means: grad_a[m] = sum_k w[k] * 2 * a[k-m] (where 0 <= k-m < n)
    # = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]  (substituting j = k-m, k = m+j)
    # = 2 * (w * a)_{reversed correlation}
    
    # More precisely: grad_a[m] = 2 * sum_{j=0}^{n-1} a[j] * w[m+j]
    # where w is defined on [0, 2n-2] and we only sum where m+j is in [0, 2n-2].
    
    # This is a cross-correlation! grad_a = 2 * correlate(w, a) (with appropriate shifts)
    
    # In numpy terms: if we pad w to length 2n-1, then:
    # grad_a[m] = 2 * sum_j a[j] * w[m+j] for j=0..n-1
    # This is like a sliding dot product: for each m, take w[m:m+n] and dot with a.
    
    grad_a = np.zeros(n)
    for m in range(n):
        # grad_a[m] = 2 * sum_{j=0}^{n-1} a[j] * w[m+j]
        # w has length 2n-1, m+j ranges from m to m+n-1
        # We need m+j < 2n-1, i.e., j < 2n-1-m
        max_j = min(n, 2*n-1-m)
        if max_j > 0:
            grad_a[m] = 2.0 * np.dot(a[:max_j], w[m:m+max_j])
    
    # Actually, for m from 0 to n-1, m+j ranges from m to m+n-1.
    # w has indices 0 to 2n-2.
    # m+j <= 2n-2 iff j <= 2n-2-m
    # Since j <= n-1, we need n-1 <= 2n-2-m, i.e., m <= n-1, which is always true.
    # Also m+j >= 0 is always true.
    # And m+j <= 2n-2: m + (n-1) <= 2n-2 iff m <= n-1. Always true.
    # So the full range j=0..n-1 is always valid!
    
    # Wait, that means grad_a[m] = 2 * dot(a, w[m:m+n]) for all m.
    # Let me verify with a small example.
    
    # a = [a0, a1, a2], n=3
    # conv = [a0^2, a0*a1+a1*a0, a0*a2+a1^2+a2*a0, a1*a2+a2*a1, a2^2]
    # = [a0^2, 2*a0*a1, a0*a2+a1^2+a2*a0, 2*a1*a2, a2^2]
    # 5 elements (2n-1 = 5)
    
    # d conv[2]/d a[1] = d(a0*a2 + a1^2 + a2*a0)/d a1 = 2*a1
    # Using formula: 2 * a[2-1] = 2*a1. ✓
    
    # d conv[1]/d a[0] = d(2*a0*a1)/d a0 = 2*a1
    # Using formula: 2 * a[1-0] = 2*a1. ✓
    
    # d conv[2]/d a[0] = d(a0*a2 + a1^2 + a2*a0)/d a0 = a2 + a2 = 2*a2
    # Using formula: 2 * a[2-0] = 2*a2. ✓
    
    # d conv[3]/d a[0] = d(2*a1*a2)/d a0 = 0
    # Using formula: 2 * a[3-0] = 2*a[3] = 0 (out of bounds). ✓
    
    # Great, the formula works!
    
    # So: grad_a[m] = 2 * dot(a, w[m:m+n])
    # This can be computed efficiently as a correlation!
    
    # Using numpy: pad a to length 2n-1 with zeros, then convolve with w
    # Or use the sliding dot product
    
    # Let's redo grad_a computation efficiently:
    grad_a = np.zeros(n)
    for m in range(n):
        grad_a[m] = 2.0 * np.dot(a, w[m:m+n])
    
    return grad_a, smooth_max


def compute_grad_and_obj_logspace(z, beta):
    """
    Compute gradient and objective in logit space.
    z are logits, a = softmax(z).
    Returns (grad_z, obj_value).
    """
    n = len(z)
    a = softmax(z)
    grad_a, smooth_max = compute_grad_smoothed(a, beta)
    
    # Chain rule: d obj/d z[j] = sum_m (d obj/d a[m]) * d a[m]/d z[j]
    # d a[m]/d z[j] = a[m] * (delta[m,j] - a[j])
    # d obj/d z[j] = a[j] * (grad_a[j] - sum_m grad_a[m] * a[m])
    dot_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_val)
    
    obj = 2.0 * n * smooth_max
    
    return grad_z, obj, a


def lbfgs_polish(a_init, deadline, beta_start=0.01, beta_end=0.001, verbose=True):
    """
    L-BFGS polish using soft-argmax smoothed objective.
    Multiple rounds with decreasing beta.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    # Multiple rounds with decreasing beta
    betas = [beta_start, beta_start*0.5, beta_start*0.25, beta_start*0.1, beta_start*0.05, beta_start*0.02, beta_start*0.01]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        if verbose:
            print(f"    L-BFGS round: beta={beta:.6f}, current score={best_score:.8f}")
        
        def objective(z):
            _, obj, _ = compute_grad_and_obj_logspace(z, beta)
            return obj
        
        def gradient(z):
            grad_z, _, _ = compute_grad_and_obj_logspace(z, beta)
            return grad_z
        
        # Run L-BFGS-B
        res = minimize(objective, z, jac=gradient, method='L-BFGS-B',
                       options={'maxiter': 200, 'maxls': 20, 'ftol': 1e-10, 'gtol': 1e-8})
        
        if res.success or res.fun < best_score:
            a_new = softmax(res.x)
            score_new = internal_score(a_new)
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                z = res.x.copy()
                if verbose:
                    print(f"    Improved: score={best_score:.8f}")
    
    return best_a, best_score


def adam_polish_v2(a_init, deadline, lr=0.01, verbose=True):
    """
    Improved Adam gradient descent in logit space.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z = z - z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    
    c = np.convolve(a, a)
    max_c = float(c.max())
    best_a = a.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    beta = max(1e-12, 0.1 * max_c)
    base_lr = lr
    
    while time.monotonic() < deadline:
        it += 1
        a = softmax(z)
        c = np.convolve(a, a)
        max_c = float(c.max())
        
        # Soft-argmax weights
        x = (c - max_c) / beta
        ex = np.exp(x)
        w = ex / ex.sum()
        
        # Gradient
        grad_a = np.zeros(n)
        for m_idx in range(n):
            grad_a[m_idx] = 2.0 * np.dot(a, w[m_idx:m_idx+n])
        
        dot_val = np.dot(a, grad_a)
        gz = a * (grad_a - dot_val)
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue
        
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        a_new = softmax(z)
        c_new = np.convolve(a_new, a_new)
        score = 2.0 * n * float(c_new.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if it % 100 == 0:
            beta = max(1e-14, beta * 0.7)
        
        if no_imp > 500:
            z = 0.9 * best_z + 0.1 * np.random.default_rng(it).normal(size=n) * 0.01
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-12, 0.1 * float(np.convolve(softmax(z), softmax(z)).max()))
        
        if no_imp > 3000:
            break
    
    return best_a, best_score
