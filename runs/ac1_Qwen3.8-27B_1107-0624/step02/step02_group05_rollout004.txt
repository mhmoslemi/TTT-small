Let me now think about a potentially better approach:

**Approach: Projected Gradient Descent with Smooth Surrogate**

Instead of the LP (which is expensive per iteration), use:
- Objective: `L(f) = log-sum-exp(beta * (f*f)) / beta` (smooth approximation of max(f*f))
- Gradient: Can be computed efficiently using FFT
- Constraint: `sum(f) = 1, f >= 0` → use log-parameterization `f = softmax(z)`

The gradient of `L(f)` with respect to `z` (log-domain parameters):
1. Compute `w = softmax(beta * (f*f))` (weights on the convolution coefficients)
2. Compute `g_f = sum_k w[k] * (partial (f*f)[k] / partial f[i])` for each i
   - `(f*f)[k] = sum_j f[j] * f[k-j]`
   - `partial (f*f)[k] / partial f[i] = f[k-i]` (for valid k-i)
   - So `g_f[i] = sum_k w[k] * f[k-i]` (for valid k-i)
   - This is exactly `g_f = w_rev * f` where w_rev is the reversed w, computed via convolution!
3. The gradient in log-domain: `grad_z = f * (g_f - sum(f * g_f))` (for softmax parameterization)

Wait, let me be more careful. If `f = softmax(z)`, then:
`d f[i] / d z[j] = f[i] * (delta_{ij} - f[j])`

The objective in terms of z is `L(z) = log-sum-exp(beta * ((f(z) * f(z))))`.

dL/dz[j] = sum_i (dL/df[i]) * (df[i]/dz[j])
= (sum_i (dL/df[i]) * f[i]) * (-f[j]) + dL/df[j] * f[j]
= f[j] * (dL/df[j] - sum_i (dL/df[i]) * f[i])

Where `dL/df[i] = sum_k w[k] * f[k-i]` (the "correlation" of w with f).

So the gradient is: `grad_z[j] = f[j] * (h[j] - sum_i h[i]*f[i])` where `h = w_rev * f` (convolution of reversed w with f).

This can be computed in O(n log n) using FFT!

- Variables: g[0], ..., g[n-1] (n variables)
- Objective: maximize sum(g)
- Constraints: (f*g)[k] <= M for k=0..2n-2 (2n-1 constraints), g >= 0 (n constraints)

Total: n variables, 3n-1 constraints. For n=256, that's 256 variables and 767 constraints. This is small for modern LP solvers.

The bottleneck is constructing the constraint matrix A (size (2n-1) x n). Each row k has nonzeros at positions where the convolution is defined. For row k, the nonzeros are at columns max(0, k-n+1) to min(n-1, k), and the values are f[k - j] for j in that range.

This matrix is a "left-shifted" version of f in each row. It's a special case of a "convolution matrix" and can be constructed efficiently.

For the LP, I'll use scipy's linprog with HiGHS, which should be fast for this size.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len
import time

"""
Optimized step function search minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based Adam optimization with log-sum-exp surrogate, LP-based
block coordinate descent, multiple sequence lengths, and adaptive restarts.
"""

def fft_convolve(a, b):
    """Linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = next_fast_len(n)
    fa = fft(a, N)
    fb = fft(b, N)
    return np.real(ifft(fa * fb))[:n]

def compute_conv(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    N = next_fast_len(2 * n - 1)
    ff = fft(f, N)
    return np.real(ifft(ff * ff))[:2 * n - 1]

def compute_score(f):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = compute_conv(f)
    return float(2.0 * n * c.max() / (s * s))

def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s

def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A

def solve_lp_for_g(f, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def lp_optimize(f_init, deadline, rng, max_iters=500):
    """LP-based iterative optimization. Returns (best_f, best_score)."""
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_score(f_init)
    
    n = len(f)
    c = compute_conv(f)
    M = float(c.max())
    s = float(f.sum())
    score = 2.0 * n * M / (s * s)
    best_f = f.copy()
    best_score = score
    
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = solve_lp_for_g(f, M, time_limit=3.0)
        if g is None:
            f = normalize_nonneg(f + 0.05 * rng.normal(size=n))
            if f is None:
                f = best_f.copy()
            c = compute_conv(f)
            M = float(c.max())
            continue
        
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Line search
        best_t = 0.0
        best_f_try = f
        best_M_try = M
        for t in [0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            f_try = normalize_nonneg((1.0 - t) * f + t * g)
            if f_try is None:
                continue
            c_try = compute_conv(f_try)
            M_try = float(c_try.max())
            if M_try < best_M_try - 1e-12:
                best_M_try = M_try
                best_f_try = f_try
                best_t = t
        
        if best_t > 0:
            f = best_f_try
            M = best_M_try
            new_score = 2.0 * n * M
            if new_score < best_score - 1e-10:
                best_score = new_score
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        
        if no_imp > 20:
            f = normalize_nonneg(0.7 * best_f + 0.3 * (best_f + 0.2 * rng.normal(size=n)))
            if f is None:
                f = best_f.copy()
            c = compute_conv(f)
            M = float(c.max())
            no_imp = 0
    
    return best_f, best_score

def adam_smooth_optimize(f_init, deadline, rng, lr=0.1, max_iters=10000):
    """Adam optimization with log-sum-exp smooth surrogate in log-domain."""
    f = np.maximum(f_init, 1e-15)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = compute_conv(f)
    M = float(c.max())
    beta = 0.5 * M  # initial temperature
    
    best_f = f.copy()
    best_M = M
    best_score = 2.0 * n * M
    no_imp = 0
    Nfft = next_fast_len(2 * n - 1)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute soft-max weights
        x = (c - M) / beta
        x = np.clip(x, -100, 100)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex
        
        # h = w_rev * f (correlation of w with f, computed via FFT)
        # h[i] = sum_k w[k] * f[k-i] for valid k-i
        # This is the cross-correlation, which equals convolution of f with reversed w
        w_rev = w[::-1]
        h = fft_convolve(f, w_rev)
        h = h[:n]  # take only the first n elements (valid range for our gradient)
        
        # Actually, let me be more careful.
        # (f*f)[k] = sum_i f[i] * f[k-i]
        # d(f*f)[k] / df[i] = f[k-i] (if 0 <= k-i < n)
        # grad_obj[i] = sum_k w[k] * f[k-i] (sum over valid k-i)
        # = sum_k w[k] * f[k-i] where 0 <= k-i < n
        # Let j = k-i, so k = j+i, and we sum over j where 0 <= j < n and 0 <= j+i < 2n-1
        # = sum_j w[j+i] * f[j] for valid j
        # This is (w_shifted * f)[0] where w_shifted[j] = w[j+i]
        # Alternatively: grad_obj = ifft(fft(w, N) * fft(f, N)) reversed or something...
        
        # Actually, the simplest way: grad_obj[i] = sum_k w[k] * f[k-i] for valid k-i
        # = (w ⊛ f)[i] where ⊛ denotes cross-correlation
        # Cross-correlation can be computed as: ifft(conj(fft(w)) * fft(f))
        # But we need to be careful about the definition.
        
        # Let me just compute it directly using the fact that:
        # grad_obj[i] = sum_{k} w[k] * f[k-i] where 0 <= k-i < n (i.e., 0 <= k-i <= n-1)
        # = sum_{j=0}^{n-1} w[j+i] * f[j]
        # This is the cross-correlation of w and f evaluated at lag i.
        
        # Using FFT: corr[w, f][i] = ifft(conj(fft(w, N)) * fft(f, N))[i]
        # Wait, the standard formula: (f ⊛ g)[k] = sum_n f[n+k] * g[n] = ifft(fft(f) * conj(fft(g)))[k]
        # So grad_obj = ifft(fft(w, N) * conj(fft(f, N)))
        
        N = next_fast_len(2 * n - 1)
        fw = fft(w, N)
        ff2 = fft(f, N)
        h = np.real(ifft(fw * np.conj(ff2)))
        h = h[:n]  # This gives grad_obj[i] for i = 0, ..., n-1
        
        # Verify: h[i] = sum_k w[k] * f[k-i] (for valid k-i)
        # Hmm, let me double-check the cross-correlation formula.
        # (x ⊛ y)[n] = sum_k x[k+n] * conj(y[k])
        # In our case: grad_obj[i] = sum_k w[k] * f[k-i]
        # Let m = k-i: sum_m w[m+i] * f[m] = (w ⊛ f)[i] if we define (w ⊛ f)[i] = sum_m w[m+i] * f[m]
        # Standard: (x ⊛ y)[n] = sum_k x[k+n] * y[k] (for real, no conjugation)
        # So (w ⊛ f)[i] = sum_k w[k+i] * f[k] 
        # But we want sum_k w[k] * f[k-i] = sum_m w[m+i] * f[m] (substituting m = k-i)
        # = (w ⊛ f)[i] ✓
        
        # FFT computation: (x ⊛ y)[n] = ifft(fft(x) * conj(fft(y)))[n]
        # So h = ifft(fft(w, N) * conj(fft(f, N)))
        # This is what I computed above. Good.
        
        # Gradient in log-domain for softmax parameterization
        dot = float(np.dot(f, h))
        grad_z = f * (h - dot)
        grad_z -= grad_z.mean()  # for reparametrization invariance
        
        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_f)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        # Adam update
        t_adam += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z_new)):
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        z_new = np.clip(z_new, -50, 50)
        z_new -= z_new.mean()
        e = np.exp(z_new)
        f_new = e / e.sum()
        
        # Compute new M
        c_new = compute_conv(f_new)
        M_new = float(c_new.max())
        
        z = z_new
        f = f_new
        c = c_new
        M = M_new
        
        score_new = 2.0 * n * M
        if score_new < best_score - 1e-10:
            best_score = score_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adapt beta
        if t_adam % 100 == 0:
            beta = max(1e-6, 0.3 * M)
        
        # Restart
        if no_imp > 200:
            z = np.log(np.maximum(best_f, 1e-10))
            z = z + 0.3 * rng.normal(size=n)
            z -= z.mean()
            z = np.clip(z, -50, 50)
            e = np.exp(z)
            f = e / e.sum()
            c = compute_conv(f)
            M = float(c.max())
            beta = 0.3 * M
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    starts = []
    t = np.linspace(0, 1, n)
    
    # Linear profiles
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
        f = 1.0 + slope * t
        f = np.maximum(f, 1e-10)
        starts.append(f / f.sum())
    
    # Quadratic
    f = 1.0 + 2.0 * t - t**2
    f = np.maximum(f, 1e-10)
    starts.append(f / f.sum())
    
    # Sine-based
    for freq in [1, 2, 3]:
        for phase in [0, np.pi/2, np.pi]:
            f = 1.0 + 0.5 * np.sin(2 * np.pi * freq * t + phase)
            f = np.maximum(f, 1e-10)
            starts.append(f / f.sum())
    
    # Random smooth
    for _ in range(10):
        coeffs = rng.normal(size=6)
        f = 1.0
        for k in range(3):
            f = f + coeffs[2*k] * np.cos(2 * np.pi * (k+1) * t)
            f = f + coeffs[2*k+1] * np.sin(2 * np.pi * (k+1) * t)
        f = np.maximum(f, 1e-10)
        starts.append(f / f.sum())
    
    # Bimodal
    f = np.exp(-20*(t-0.1)**2) + np.exp(-20*(t-0.9)**2) + 0.01
    f = np.maximum(f, 1e-10)
    starts.append(f / f.sum())
    
    # Triangular
    f = 1.0 - np.abs(2*t - 1)
    f = np.maximum(f, 1e-10)
    starts.append(f / f.sum())
    
    return starts

def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, f)
    return result
