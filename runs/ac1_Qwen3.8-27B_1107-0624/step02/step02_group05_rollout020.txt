    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

import numpy as np
from scipy.optimize import linprog
import time
import math

"""
Multi-phase optimization for minimizing 2*n*max(conv(f,f))/(sum(f))^2.

Strategy:
- Log-domain Adam optimization with analytic soft-max gradient
- LP-based direction finding for refinement
- Multiple n values, diverse initializations, adaptive time allocation
- Key insight: gradient of soft-max(conv(f,f)) w.r.t. log-domain params is
  2n * f * (h - W) where h is cross-correlation of soft-max weights with f,
  and W is the soft-max value.

def compute_score(f):
    """Compute 2*n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_score_normalized(f):
    """Assume f is normalized (sum=1). Return 2*n*max(conv(f,f))."""
    n = len(f)
    c = np.convolve(f, f)
    return float(2.0 * n * c.max())


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def make_conv_matrix(f):
    """Build matrix A such that A @ g = conv(f, g) (full linear convolution)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp(f, M, time_limit=10.0):
    """LP: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for all k."""
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def get_fft_size(n1, n2):
    N = 1
    while N < n1 + n2 - 1:
        N <<= 1
    return N


def compute_cross_corr(w, f):
    """Compute h[m] = sum_k w[k]*f[k-m] for m=0..n_f-1 using FFT.
    w has length 2*n_f-1, f has length n_f.
    """
    n = len(f)
    N = get_fft_size(len(w), n)
    fw = np.fft.rfft(w, N)
    ff = np.fft.rfft(f, N)
    corr = np.fft.irfft(fw * np.conj(ff), N)
    return corr[:n]


def make_uniform(n):
    return np.ones(n) / n


def make_linear(n, slope=1.0):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic(n, a=1.5, b=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a * u + b * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_quadratic(n, a=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a * u ** 2
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_random_fourier(n, rng, n_modes=6):
    t = np.linspace(0, 1, n)
    f = 1.0
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        f += amp * np.cos(2 * np.pi * k * t + phase)
    f = np.maximum(f, 1e-6)
    return f / f.sum()


def make_random_smooth(n, rng):
    """Random smooth non-negative sequence via random low-pass filtered noise."""
    raw = rng.standard_normal(n)
    # Simple moving average smoothing
    k = max(1, n // 20)
    kernel = np.ones(k) / k
    smoothed = np.convolve(raw, kernel, mode='same')
    f = np.exp(smoothed * 0.5)
    f = np.maximum(f, 1e-6)
    return f / f.sum()


def interpolate_profile(f, n_new):
    """Linearly interpolate to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def optimize_adam_log(f_init, deadline, lr=0.08, verbose=False):
    """
    Adam optimization in log domain with soft-max gradient.
    Minimizes max_k conv(f,f)[k] subject to f in simplex.
    Returns (best_f, best_M) where best_M = max(conv(best_f, best_f)).
    """
    f = np.maximum(f_init, 1e-15)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0

    c = np.convolve(f, f)
    c_max_init = float(c.max())
    beta = 0.1 * c_max_init
    beta_min = 1e-5 * c_max_init
    beta_decay = 0.995

    M = float(c.max())
    best_f = f.copy()
    best_M = M
    no_imp = 0
    iter_count = 0

    while time.monotonic() < deadline:
        t_adam += 1
        iter_count += 1

        # Compute soft-max weights
        c = np.convolve(f, f)
        c_max = float(c.max())
        beta = max(beta_min, beta * beta_decay)

        x = (c - c_max) / beta
        ex = np.exp(x)
        w = ex / ex.sum()

        # W = sum_k w[k] * c[k]
        W = float(np.dot(w, c))

        # h[m] = sum_k w[k] * f[k-m]
        h = compute_cross_corr(w, f)

        # Gradient: grad_z[m] = 2n * f[m] * (h[m] - W)
        grad_z = 2.0 * n * f * (h - W)

        # Center for numerical stability (gradient should already sum to 0)
        grad_z -= grad_z.mean()

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_f) - np.log(best_f).mean()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            no_imp += 10
            continue

        # Adam update
        m_adam = b1 * m_adam + (1 - b1) * grad_z
        v_adam = b2 * v_adam + (1 - b2) * (grad_z * grad_z)
        m_hat = m_adam / (1 - b1 ** t_adam)
        v_hat = v_adam / (1 - b2 ** t_adam)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -60, 60)
        z -= z.mean()

        e = np.exp(z)
        f = e / e.sum()

        # Evaluate
        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-10:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Stall detection
        if no_imp > 200:
            # Reset with perturbation
            z_new = np.log(best_f) - np.log(best_f).mean()
            z_new += 0.2 * np.random.standard_normal(n)
            z_new = np.clip(z_new, -60, 60)
            z_new -= z_new.mean()
            e = np.exp(z_new)
            f = e / e.sum()
            z = z_new
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            no_imp = 0
            beta = 0.1 * best_M
            lr = min(lr * 1.5, 0.3)

        if verbose and iter_count % 200 == 0:
            print(f"    iter={iter_count}, M={M:.10f}, best={best_M:.10f}, beta={beta:.6e}")

    return best_f, best_M

            if time.monotonic() > deadline - 15:
                break
            f_ext = interpolate_profile(best_seq, n_try)
            run_deadline = min(deadline - 10, time.monotonic() + 30.0)
            f_best, M_best = optimize_adaptive(f_ext, run_deadline, lr_init=0.04)
            sc = compute_score(f_best)
            try_update(f_best, sc)
            print(f"  Try n={n_try}: score={sc:.10f}, best={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = make_linear(256, 1.5)
        best_score = compute_score(best_seq)

    # Ensure validity
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear(256, 1.5)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"Final: n={len(best_seq)}, score={final_score:.10f}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in best_seq]
