import numpy as np
from scipy.optimize import linprog
import time
import math


def compute_conv(f):
    """Compute linear convolution of f with itself."""
    return np.convolve(f, f)


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(np.sum(f))
    if s < 1e-30:
        return np.inf
    c = compute_conv(f)
    return float(2.0 * n * np.max(c) / (s * s))


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def fft_autocorrelation(f):
    """Compute linear autocorrelation using FFT."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    ac = np.fft.irfft(F * np.conj(F), N)
    return ac[:2 * n - 1]


def fft_gradient(c, f, beta):
    """
    Compute gradient of soft-max(c, beta) with respect to f.
    Uses FFT for the correlation computation.
    Returns (soft_max_value, weight_vector_for_c, gradient_wr f).
    """
    n = len(f)
    mval = float(np.max(c))
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        L = mval
    else:
        w = ex / s
        L = mval + beta * np.log(s)
    
    # Gradient: dL/df[m] = 2 * sum_k w[k] * f[k-m] (where 0 <= k-m < n)
    # This is 2 * correlation(w, f) at lags 0..n-1
    # correlation(w, f)[m] = sum_k w[k+m] * f[k] for k where both valid
    # Using FFT: pad w and f to size >= 3n-2
    N = 1
    while N < 3 * n:
        N <<= 1
    # We need: grad[m] = sum_{k=m}^{m+n-1} w[k] * f[k-m] for m=0..n-1
    # This is the cross-correlation of w with f, sampled at lags 0 to n-1
    # corr[lag] = sum_k w[k+lag] * f[k]
    # Using FFT: IFFT(FFT(w) * conj(FFT(f))) gives cross-correlation
    fw = np.fft.rfft(w, N)
    ff = np.fft.rfft(f, N)
    corr = np.fft.irfft(fw * np.conj(ff), N)
    grad = 2.0 * corr[:n]
    return L, w, grad


def adam_fft_polish(f_init, deadline, rng, lr=0.05, max_iters=20000, beta_init=None, verbose=False):
    """
    FFT-accelerated Adam optimization in logit space.

    """
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    adam_t = 0
    
    c = fft_autocorrelation(f)
    M = float(np.max(c))
    if beta_init is None:
        beta = max(1e-12, 0.1 * M)
    else:
        beta = beta_init
    
    best_f = f.copy()
    best_M = M

    for it in range(max_iters):

            break

        dot = float(np.dot(f, grad))

            z = np.log(np.maximum(z, 1e-12))

            m_adam.fill(0)
            v_adam.fill(0)
            adam_t = 0
            continue

            m_adam.fill(0)
            v_adam.fill(0)
            adam_t = 0

        z = np.clip(z, -60, 60)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()

        f = f / f.sum()
        
        c = fft_autocorrelation(f)
        M = float(np.max(c))

            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp_for_g(f, M, time_limit=3.0):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c_vec = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine(f_init, deadline, rng, max_iters=500, verbose=False):
    """LP-based iterative refinement."""
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, float(np.max(compute_conv(f_init)))
    n = len(f)
    M = float(np.max(compute_conv(f)))
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = solve_lp_for_g(f, M, time_limit=2.0)
        if g is None:
            f = best_f.copy()
            M = best_M
            break
        
        g_sum = float(np.sum(g))
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        best_t = 0.0
        best_f_try = f
        best_M_try = M
        for t in [0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            f_try = (1.0 - t) * f + t * g
            f_try = normalize_nonneg(f_try)

                best_t = t

            f = best_f_try
            M = best_M_try
            if M < best_M - 1e-13:

def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, f)
    return np.maximum(result, 0.0)

    """
    Multi-phase optimization combining FFT-accelerated Adam, LP refinement,
    and diverse initialization to minimize 2*n*max(convolve(f,f))/sum(f)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)
    
    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
                print(f"Parent sequence loaded: n={p.size}, score={compute_score(parent):.10f}")
                try_update(parent, compute_score(parent))
    except Exception:
        pass
    
    # =========================================================
    # Phase 1: Fast search at n=128 (200 seconds)
    # =========================================================
    n1 = 128
    phase1_end = min(deadline, t_start + 180.0)
    print(f"Phase 1: Fast Adam at n={n1}")
    
    profiles_128 = make_profiles(n1, rng, count=15)
    if parent is not None:
        profiles_128.append(interpolate_profile(parent, n1))
        profiles_128.append(interpolate_profile(parent[::-1], n1))
    
    # Evaluate all profiles
    for p in profiles_128:
        sc = compute_score(p)
        try_update(p, sc)
    print(f"  Best after init: {best_score:.10f}")
    
    results_phase1 = []
    n_profiles = len(profiles_128)
    time_per_profile = (phase1_end - t_start) / max(1, n_profiles)
    
    for i, p in enumerate(profiles_128):
        remaining = phase1_end - time.monotonic()
        if remaining < 2:
            break
        alloc = min(remaining / max(1, n_profiles - i), time_per_profile)
        if alloc < 1:
            alloc = 1
        run_deadline = time.monotonic() + alloc
        
        # Adam with moderate settings
        f_best, M_best = adam_fft_polish(p, run_deadline, rng, lr=0.08,
                                          max_iters=50000, beta_init=None)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results_phase1.append((sc, f_best))
    
    results_phase1.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best n={n1}: {best_score:.10f}")
    
    # =========================================================
    # Phase 2: Refine at n=256 (400 seconds)
    # =========================================================
    n2 = 256
    phase2_end = min(deadline, t_start + 600.0)
    print(f"Phase 2: Adam polish at n={n2}")
    
    top_k2 = min(8, len(results_phase1))
    results_phase2 = []
    for i in range(top_k2):
        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_k2 - i), 80.0)
        if alloc < 5:
            break
        run_deadline = time.monotonic() + alloc
        
        f_ext = interpolate_profile(results_phase1[i][1], n2)
        f_best, M_best = adam_fft_polish(f_ext, run_deadline, rng, lr=0.04,
                                          max_iters=100000, beta_init=None)
        sc = 2.0 * n2 * M_best
        try_update(f_best, sc)
        results_phase2.append((sc, f_best))
        if (i + 1) % 2 == 0:
            print(f"  Phase 2 run {i+1}/{top_k2}: score={sc:.10f}, best={best_score:.10f}")
    
    # Also try parent directly at n=256
    if parent is not None:
        remaining = phase2_end - time.monotonic()
        if remaining > 30:
            run_deadline = time.monotonic() + min(remaining, 100.0)
            f_best, M_best = adam_fft_polish(parent, run_deadline, rng, lr=0.03,
                                              max_iters=100000, beta_init=None)
            sc = 2.0 * n2 * M_best
            try_update(f_best, sc)
            results_phase2.append((sc, f_best))
            print(f"  Parent at n=256: score={sc:.10f}")
    
    results_phase2.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best: {best_score:.10f}")
    
    # =========================================================
    # Phase 3: LP refinement on best solutions (300 seconds)
    # =========================================================
    phase3_end = min(deadline, t_start + 900.0)
    print(f"Phase 3: LP refinement at n={n2}")
    
    top_k3 = min(4, len(results_phase2))
    results_phase3 = []
    for i in range(top_k3):
        remaining = phase3_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_k3 - i), 100.0)
        if alloc < 5:
            break
        run_deadline = time.monotonic() + alloc
        f_best, M_best = lp_refine(results_phase2[i][1], run_deadline, rng, max_iters=200)
        sc = 2.0 * n2 * M_best
        try_update(f_best, sc)
        results_phase3.append((sc, f_best))
        print(f"  LP refine {i+1}/{top_k3}: score={sc:.10f}, best={best_score:.10f}")
    
    # Combine results
    all_results = results_phase2 + results_phase3
    all_results.sort(key=lambda x: x[0])
    print(f"Phase 3 done. Best: {best_score:.10f}")
    
    # =========================================================
    # Phase 4: More Adam with different hyperparameters (300 seconds)
    # =========================================================
    phase4_end = min(deadline, t_start + 1200.0)
    print(f"Phase 4: Extended Adam with varied hyperparameters")
    
    # Try different learning rates and beta schedules
    lr_schedule = [0.02, 0.01, 0.06, 0.03]
    top_k4 = min(3, len(all_results))
    
    for i in range(top_k4):
        for lr_try in lr_schedule[:2]:
            remaining = phase4_end - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining / 2.0, 60.0)
            if alloc < 5:
                break
            run_deadline = time.monotonic() + alloc
            f_best, M_best = adam_fft_polish(all_results[i][1], run_deadline, rng,
                                              lr=lr_try, max_iters=100000, beta_init=None)
            sc = 2.0 * n2 * M_best
            try_update(f_best, sc)
            all_results.append((sc, f_best))
    
    all_results.sort(key=lambda x: x[0])
    print(f"Phase 4 done. Best: {best_score:.10f}")
    
    # =========================================================
    # Phase 5: Try n=512 (300 seconds)
    # =========================================================
    n5 = 512
    phase5_end = min(deadline, t_start + 1550.0)
    print(f"Phase 5: Extension to n={n5}")
    
    top_k5 = min(3, len(all_results))
    for i in range(top_k5):
        remaining = phase5_end - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining / max(1, top_k5 - i), 100.0)
        if alloc < 10:
            break
        f_ext = interpolate_profile(all_results[i][1], n5)
        run_deadline = time.monotonic() + alloc
        f_best, M_best = adam_fft_polish(f_ext, run_deadline, rng, lr=0.02,
                                          max_iters=50000, beta_init=None)
        sc = 2.0 * n5 * M_best
        try_update(f_best, sc)
        print(f"  n=512 run {i+1}: score={sc:.10f}, best={best_score:.10f}")
    
    # =========================================================
    # Phase 6: Final refinement on absolute best (130 seconds)
    # =========================================================
    phase6_end = deadline
    print(f"Phase 6: Final refinement")
    
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = phase6_end - time.monotonic()
        if remaining > 20:
            # Adam with very small lr for fine-tuning
            run_deadline = time.monotonic() + min(remaining * 0.6, 80.0)
            f_best, M_best = adam_fft_polish(best_seq, run_deadline, rng, lr=0.01,
                                              max_iters=50000, beta_init=None)
            sc = 2.0 * n_best * M_best
            try_update(f_best, sc)
            
            # LP on the refined result
            remaining = phase6_end - time.monotonic()
            if remaining > 10:
                f_best2, M_best2 = lp_refine(best_seq, phase6_end, rng, max_iters=100)
                sc2 = 2.0 * n_best * M_best2
                try_update(f_best2, sc2)
    
    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = make_profiles(256, rng, count=1)[0]
    
    best_seq = np.maximum(best_seq, 0.0)
    s = float(np.sum(best_seq))
    if s < 0.01:
        t = np.arange(256, dtype=np.float64)
        best_seq = 1.0 + 1.5 * t / 255.0
        best_seq = best_seq / best_seq.sum()
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    
    final_score = compute_score(best_seq)
    print(f"Final: n={len(best_seq)}, score={final_score:.10f}")
    
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence score: {ev_score:.10f}")
        if np.isfinite(ev_score):
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")
    
    elapsed = time.monotonic() - t_start
    print(f"Total time: {elapsed:.1f}s, final score: {final_score:.10f}")
    
    return [float(x) for x in best_seq]
