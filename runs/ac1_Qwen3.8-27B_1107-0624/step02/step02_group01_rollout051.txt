"""
Multi-scale optimization with gradient-based and LP-based refinement.
Uses continuation (annealing) for the softmax approximation, analytical 
cross-correlation gradients, and LP direction finding at multiple scales.
"""

import numpy as np
import time
from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def cross_corr_weighted(p, a):
    """Compute h_j = sum_k p_k * a_{k-j} for 0 <= k-j <= n-1, j = 0..n-1.
    p has length 2n-1, a has length n."""
    n = a.size
    N = next_pow2(3 * n)
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    h = np.fft.irfft(fp * np.conj(fa), N)[:n]
    return h

def make_profiles(n, rng):
    """Generate diverse initial profiles."""
    profs = []
    # Linear
    for eta in (0.5, 1.0, 1.5, 2.0, 3.0):
        t = np.linspace(0, 1, n)
        a = 1.0 + eta * t
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    # Quadratic
    for c in (0.5, 1.0, 2.0, -0.5):
        t = np.linspace(0, 1, n)
        a = 1.0 + c * (2*t - 1)**2
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    # Cosine
    for freq in (1, 2, 3):

            t = np.linspace(0, 1, n)
            a = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            a = np.maximum(a, 1e-10)

    # Chirp-like
    t = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * (3 * t**2))

def gradient_descent(a0, deadline, rng, T_init=0.05, T_min=0.001, lr_init=0.1, max_iter=50000):
    """Gradient descent with softmax smoothing and annealing."""
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return a0, np.inf
    a = a / s
    n = a.size
    
    c = fft_convolve(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    
    T = T_init * M
    lr = lr_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = fft_convolve(a)
        M = float(c.max())

        grad = 2.0 * h  # This is the gradient of the softmax-smoothed max
        
        # Adam update

        m_adam = b1 * m_adam + (1 - b1) * grad

            no_imp = 0
        else:
            no_imp += 1

            t_adam = 0
            no_imp = 0

def lp_direction(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 20})
    if res.success:
        return res.x
    return None

                M = best_M

            M = float(fft_convolve(a).max())

                best_a = a.copy()
                no_imp = 0

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None

    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def cross_corr_weighted(p, a):
    """Compute h_j = sum_k p_k * a_{k-j} for valid indices, j=0..n-1."""
    n = a.size
    N = next_pow2(3 * n)
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    h = np.fft.irfft(fp * np.conj(fa), N)[:n]
    return h


def make_profiles(n, rng):
    """Generate diverse non-negative profiles normalized to sum=1."""
    profs = []
    t = np.linspace(0, 1, n)
    # Linear profiles
    for eta in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0):
        a = 1.0 + eta * t
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    # Quadratic / parabolic
    for c in (0.5, 1.0, 1.5, 2.0, -1.0, -2.0):
        a = 1.0 + c * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    # Cosine modulated
    for freq in (1, 2, 3, 4):
        for amp in (0.3, 0.5, 0.7, 0.9):
            a = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            a = np.maximum(a, 1e-10)
            profs.append(a / a.sum())
    # Chirp
    for bw in (2, 4, 6):
        a = 1.0 + 0.6 * np.cos(2 * np.pi * bw * t ** 2)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    # Gaussian centered
    for width in (0.15, 0.25, 0.35, 0.5):
        a = np.exp(-((t - 0.5) / width) ** 2)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    # Smooth random
    for _ in range(6):
        coeffs = rng.normal(size=min(n, 12))
        a = np.ones(n)
        for k in range(1, min(n, 12)):
            a += (coeffs[k - 1] / k) * np.cos(2 * np.pi * k * t)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    # Bimodal
    for d in (0.1, 0.2, 0.3):
        a = np.exp(-((t - 0.3) / 0.15) ** 2) + np.exp(-((t - 0.7) / 0.15) ** 2)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    return profs


def interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return None
    p0 = np.maximum(p0, 0.0)
    s = p0.sum()
    if s < 1e-30:
        return None
    p0 = p0 / s
    if n0 == n_new:
        return p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s1 = p1.sum()
    return p1 / s1 if s1 > 1e-30 else None


def gradient_optimize(a0, deadline, rng, T_scale=0.04, lr=0.08, max_iter=50000,
                      use_nesterov=True, annal_rate=0.97, annal_interval=300):
    """
    Gradient descent in log-space with softmax-smoothed max objective.
    Uses Nesterov momentum and geometric annealing of temperature.
    """
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size

    # Log-space parameterization
    z = np.log(a)
    z -= z.mean()

    c = fft_convolve(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    best_z = z.copy()

    T = T_scale * M
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    v_nest = np.zeros(n)
    t_opt = 0
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(a)
        M = float(c.max())

        # Softmax weights
        x = c / T
        x -= x.max()
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            p = ex / s_ex

        # Gradient via cross-correlation: grad_j = 2 * sum_k p_k * a_{k-j}
        h = cross_corr_weighted(p, a)
        grad_a = 2.0 * h

        # Convert to log-space gradient: grad_z = a * (grad_a - mean(a*grad_a))
        mean_grad = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - mean_grad)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            v_nest.fill(0)
            t_opt = 0
            continue

        t_opt += 1
        m_adam = b1 * m_adam + (1 - b1) * grad_z
        v_adam = b2 * v_adam + (1 - b2) * grad_z * grad_z
        mh = m_adam / (1 - b1 ** t_opt)
        vh = v_adam / (1 - b2 ** t_opt)

        if use_nesterov:
            v_nest = b1 * v_nest - lr * mh / (np.sqrt(vh) + eps)
            z_new = z + v_nest
        else:
            z_new = z - lr * mh / (np.sqrt(vh) + eps)

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= z_new.mean()

        a_new = np.exp(z_new - z_new.max())
        s_new = a_new.sum()
        if s_new < 1e-30:
            continue
        a_new = a_new / s_new

        c_new = fft_convolve(a_new)
        M_new = float(c_new.max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        z = z_new
        a = a_new
        M = M_new

        # Anneal
        if t_opt % annal_interval == 0:
            T = max(1e-6 * M, T * annal_rate)

        # Escape
        if no_imp > max(400, 2 * n):
            z = 0.85 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = fft_convolve(a)
            M = float(c.max())
            T = T_scale * M
            m_adam.fill(0)
            v_adam.fill(0)
            v_nest.fill(0)
            t_opt = 0
            no_imp = 0

    return best_a, best_M


def lp_solve_direction(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a0, deadline, rng, max_iter=5000, step_candidates=None):
    """LP-based iterative optimization with adaptive step size."""
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return a0, np.inf
    a = a / s
    n = a.size

    if step_candidates is None:
        step_candidates = (0.3, 0.1, 0.5, 0.15, 0.4, 0.05, 0.25, 0.6, 0.2, 0.08)

    M = float(fft_convolve(a).max())
    best_a = a.copy()
    best_M = M
    no_imp = 0
    best_step = 0.3

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = lp_solve_direction(a, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 200:
                break
            a = best_a + 0.02 * rng.normal(size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-30:
                a = best_a.copy()
                M = best_M
                continue
            a = a / s
            M = float(fft_convolve(a).max())
            if M < best_M:
                best_M = M
                best_a = a.copy()
                no_imp = 0
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-5:
            no_imp += 1
            if no_imp > 300:
                a = 0.9 * best_a + 0.1 * np.abs(rng.normal(size=n)) + 1e-10
                a = a / a.sum()
                M = float(fft_convolve(a).max())
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                no_imp = 0
            continue

        g = g / g_sum
        improved = False

        # Try steps in order of preference (recently successful first)
        ordered_steps = (best_step,) + tuple(s for s in step_candidates if s != best_step)
        for t in ordered_steps:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-30:
                continue
            a_new = a_new / s_new
            M_new = float(fft_convolve(a_new).max())
            if M_new < M - 1e-14:
                a = a_new
                M = M_new
                best_step = t
                improved = True
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp % 150 == 0:
                a = 0.87 * best_a + 0.13 * np.abs(rng.normal(size=n)) + 1e-10
                a = a / a.sum()
                M = float(fft_convolve(a).max())
                no_imp = 0

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search:
    1. Quick gradient optimization at n=256 for shape discovery.
    2. LP + gradient refinement at n=512 (main workhorse).
    3. Interpolation to n=1024 with further refinement.
    4. Final polish on best found.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            sv = pv.sum()
            if sv > 1e-30:
                parent = pv / sv
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent: score={sc:.10f}, n={parent.size}")

    # === Phase 1: Shape discovery at small n (fast) ===
    phase1_end = min(deadline, t_start + 180.0)

    for n_small in (128, 256):
        if time.monotonic() > phase1_end:
            break

        profs = make_profiles(n_small, rng)
        if parent is not None:
            pi = interp_profile(parent, n_small)
            if pi is not None:
                profs.insert(0, pi)

        per = max(3.0, (phase1_end - time.monotonic()) / max(1, len(profs)))
        phase1_results = []

        for i, p0 in enumerate(profs):
            if time.monotonic() > phase1_end:
                break
            t_end = min(phase1_end, time.monotonic() + per)
            a_b, M_b = gradient_optimize(p0, t_end, rng, T_scale=0.06, lr=0.1,
                                         max_iter=5000, annal_rate=0.95)
            sc = 2.0 * n_small * M_b
            try_update(a_b, sc)
            phase1_results.append((sc, n_small, a_b))

        phase1_results.sort(key=lambda x: x[0])
        print(f"Phase 1 (n={n_small}): best={best_score:.10f}")

    # === Phase 2: Main optimization at n=512 ===
    n2 = 512
    phase2_end = min(deadline, t_start + 1080.0)

    # Build diverse starts for n=512
    starts = []
    if parent is not None:
        pi = interp_profile(parent, n2)
        if pi is not None:
            starts.append(pi)
    if best_seq is not None:
        pi = interp_profile(best_seq, n2)
        if pi is not None:
            starts.append(pi)
    # Add interpolated phase-1 results
    for sc, ns, pb in phase1_results[:4]:
        pi = interp_profile(pb, n2)
        if pi is not None:
            starts.append(pi)
    # Add fresh profiles
    starts.extend(make_profiles(n2, rng)[:8])

    # Deduplicate (roughly)
    unique_starts = []
    for s in starts:
        dup = False
        for u in unique_starts:
            if np.linalg.norm(s - u) < 0.05:
                dup = True
                break
        if not dup:
            unique_starts.append(s)
    starts = unique_starts[:12]

    # Gradient pass on all starts
    grad_per = max(5.0, (phase2_end - time.monotonic()) * 0.4 / max(1, len(starts)))
    grad_results = []

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase2_end * 0.45:
            break
        t_end = min(phase2_end * 0.45, time.monotonic() + grad_per)
        a_b, M_b = gradient_optimize(p0, t_end, rng, T_scale=0.03, lr=0.07,
                                     max_iter=15000, annal_rate=0.96)
        sc = 2.0 * n2 * M_b
        try_update(a_b, sc)
        grad_results.append((sc, a_b))

    grad_results.sort(key=lambda x: x[0])
    print(f"Phase 2 gradient (n={n2}): best={best_score:.10f}")

    # LP pass on top gradient results
    lp_per = max(15.0, (phase2_end - time.monotonic()) * 0.45 / max(1, min(5, len(grad_results))))
    lp_end = min(phase2_end * 0.9, time.monotonic() + lp_per * min(5, len(grad_results)))

    for i in range(min(5, len(grad_results))):
        if time.monotonic() > lp_end:
            break
        _, p0 = grad_results[i]
        t_end = min(lp_end, time.monotonic() + lp_per)
        a_b, M_b = lp_optimize(p0, t_end, rng, max_iter=8000)
        sc = 2.0 * n2 * M_b
        try_update(a_b, sc)
        print(f"  Phase 2 LP cand {i}: score={sc:.10f}")

    # Gradient polish after LP
    if best_seq is not None and best_seq.size == n2:
        t_end = min(phase2_end, time.monotonic() + 90.0)
        a_b, M_b = gradient_optimize(best_seq, t_end, rng, T_scale=0.015, lr=0.04,
                                     max_iter=30000, annal_rate=0.97, use_nesterov=True)
        sc = 2.0 * n2 * M_b
        try_update(a_b, sc)
        print(f"Phase 2 final polish (n={n2}): best={best_score:.10f}")

    # Extra LP rounds on best
    if best_seq is not None and best_seq.size == n2:
        t_end = min(phase2_end, time.monotonic() + 120.0)
        a_b, M_b = lp_optimize(best_seq, t_end, rng, max_iter=10000)
        sc = 2.0 * n2 * M_b
        try_update(a_b, sc)
        print(f"Phase 2 extra LP (n={n2}): best={best_score:.10f}")

    # === Phase 3: Scale up to n=1024 ===
    n3 = 1024
    phase3_end = min(deadline, t_start + 1560.0)

    if best_seq is not None and time.monotonic() < phase3_end - 60:
        pi = interp_profile(best_seq, n3)
        if pi is not None:
            # Gradient at n=1024
            t_end = min(phase3_end - 60, time.monotonic() + 80.0)
            a_b, M_b = gradient_optimize(pi, t_end, rng, T_scale=0.025, lr=0.05,
                                         max_iter=15000, annal_rate=0.96)
            sc = 2.0 * n3 * M_b
            try_update(a_b, sc)
            print(f"Phase 3 gradient (n={n3}): score={sc:.10f}, best={best_score:.10f}")

            # LP at n=1024
            if time.monotonic() < phase3_end - 30:
                t_end = min(phase3_end - 30, time.monotonic() + 90.0)
                a_b, M_b = lp_optimize(a_b, t_end, rng, max_iter=4000)
                sc = 2.0 * n3 * M_b
                try_update(a_b, sc)
                print(f"Phase 3 LP (n={n3}): score={sc:.10f}, best={best_score:.10f}")

    # Also try parent directly at n=1024
    if parent is not None and time.monotonic() < phase3_end - 30 and parent.size != n3:
        pi = interp_profile(parent, n3)
        if pi is not None:
            t_end = min(phase3_end - 30, time.monotonic() + 40.0)
            a_b, M_b = gradient_optimize(pi, t_end, rng, T_scale=0.025, lr=0.05,
                                         max_iter=8000)
            sc = 2.0 * n3 * M_b
            try_update(a_b, sc)

    # === Phase 4: Final refinement ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = best_seq.size
        remaining = deadline - time.monotonic()

        # Gradient polish
        t_end = min(deadline - 10, time.monotonic() + remaining * 0.5)
        a_b, M_b = gradient_optimize(best_seq, t_end, rng, T_scale=0.01, lr=0.03,
                                     max_iter=20000, annal_rate=0.98)
        sc = compute_score(a_b)
        try_update(a_b, sc)
        print(f"Phase 4 gradient (n={n_best}): best={best_score:.10f}")

        # LP polish
        if time.monotonic() < deadline - 10:
            t_end = min(deadline - 5, time.monotonic() + remaining * 0.4)
            a_b, M_b = lp_optimize(a_b, t_end, rng, max_iter=5000)
            sc = compute_score(a_b)
            try_update(a_b, sc)
            print(f"Phase 4 LP (n={n_best}): best={best_score:.10f}")

        # One more gradient pass
        if time.monotonic() < deadline - 5:
            t_end = min(deadline - 3, time.monotonic() + 30.0)
            a_b, M_b = gradient_optimize(a_b, t_end, rng, T_scale=0.008, lr=0.02,
                                         max_iter=10000)
            sc = compute_score(a_b)
            try_update(a_b, sc)
            print(f"Phase 4 final gradient: best={best_score:.10f}")

    print(f"=== FINAL: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0} ===")

    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = float(final.sum())
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        if final.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                print(f"Final evaluator score: {sc_eval:.10f}")
            except Exception as e:
                print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_profiles(256, rng)[0]
    return [float(x) for x in fb]
