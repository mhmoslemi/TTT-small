   Minimizes 2n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
   Uses projected subgradient descent with soft-max multi-lag weighting,
   FFT-based subgradient and convolution, adaptive step sizes, momentum,

   def fft_convolve(s):
       """Compute full convolution of s with itself using FFT."""
       n = len(s)
       N = 1
       while N < 2 * n - 1:
           N *= 2
       S = np.fft.rfft(s, N)
       result = np.fft.irfft(S * S, N)
       return result[:2*n-1]
   
   def fft_cross(w, s):
       """Compute cross[j] = sum_m w[j+m]*s[m] for j=0..len(s)-1 using FFT."""
       n = len(s)
       N = 1
       while N < len(w) + n - 1:
           N *= 2
       W = np.fft.rfft(w, N)
       S = np.fft.rfft(s, N)
       result = np.fft.irfft(np.conj(W) * S, N)
       return result[:n]
   
   def project_simplex(v):
       """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
       n = len(v)
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       rho = np.arange(1, n + 1)
       cond = u > (cssv - 1.0) / rho
       if not np.any(cond):
           return np.ones(n) / n
       rho_val = int(np.max(rho[cond]))
       tau = (cssv[rho_val - 1] - 1.0) / rho_val
       return np.maximum(v - tau, 0.0)
   
   def compute_score(s):
       """Compute 2n*max(conv(s,s))/(sum(s))^2."""
       n = len(s)
       S = s.sum()
       if S < 1e-15:
           return float('inf')
       c = fft_convolve(s)
       return 2.0 * n * c.max() / (S * S)
   
   def compute_subgrad(s, c, beta=10.0):
       """Compute weighted subgradient of max autocorrelation."""
       n = len(s)
       M = c.max()
       if M <= 1e-30:
           return np.zeros(n)
       # Soft-max weights over all lags
       w = np.exp(beta * (c - M) / (M + 1e-30))
       W_total = w.sum()
       if W_total < 1e-30:
           k_star = int(np.argmax(c))
           grad = np.zeros(n)
           for j in range(n):
               m = k_star - j
               if 0 <= m < n:
                   grad[j] = 2.0 * s[m]
           return grad
       # Use FFT cross-correlation
       cross = fft_cross(w, s)
       grad = 2.0 * cross / W_total
       return grad

   ```
   s = initial
   for iteration:
       grad = smooth_grad(s)
       # Project gradient onto tangent space of simplex (optional but helps)
       grad_tangent = grad - mean(grad)  # projection onto {sum = 0}
       # Backtracking line search
       step = 1.0
       while True:
           s_try = project_simplex(s - step * grad)
           if smooth_score(s_try) <= smooth_score(s) - 0.5 * step * sum(grad * (s - s_try)):
               break
           step *= 0.5
       s = s_try
   ```
