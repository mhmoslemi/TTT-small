At positions where `conv(a,a)[k] < M`: `conv(a,d)[k] <= (M - conv(a,a)[k]) / (2t)`.

At the position where f*f is maximum (= M):

def _lp_refinement(a, n_iter=10, t_blend=0.1):
    """

    M = float(c.max())
    
    best_a = a_norm.copy()

    for it in range(n_iter):
        # Build LP:
        # maximize sum(b) s.t. b >= 0, conv(a_norm, b)[k] <= M for all k
        # Variables: b_0, ..., b_{n-1}
        c_obj = -np.ones(n)

        A_ub = np.zeros((2*n-1, n))
        for k in range(2*n-1):

            hi = min(n, k + 1)
            for i in range(lo, hi):
                A_ub[k, i] = a_norm[k - i]

        bounds = [(0, None)] * n

                         method='highs', options={'maxiter': 10000})
            if not res.success:
                break
            b = res.x
            b_sum = float(b.sum())
            if b_sum <= 1e-10:
                break
            b = b / b_sum  # normalize to sum=1
        except Exception:
            break
        
        # Blend
        a_new = (1 - t_blend) * a_norm + t_blend * b
        a_new = np.maximum(a_new, 0)
        a_new_sum = float(a_new.sum())
        if a_new_sum <= 1e-10:
            break
        a_new = a_new / a_new_sum

            best_a = a_new.copy()
            a_norm = a_new

            M = M_new
        else:

            t_blend = t_blend * 0.5
            if t_blend < 0.01:

            a_new = (1 - t_blend) * a_norm + t_blend * b

def _simulated_annealing(p0, deadline, rng, max_iter=100000):
    """
    Patch-based simulated annealing for local escape.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    best_p = p0.copy()
    best_score = float(2.0 * n * maxc)

    T = 0.03
    T_min = 0.0005
    cool_rate = 0.9998
    min_patch = 3
    max_patch = min(50, n // 4)

    n_iter = 0
    stall = 0
    
    while time.monotonic() < deadline and n_iter < max_iter:
        n_iter += 1
        ps = int(np.clip(rng.integers(min_patch, max_patch + 1), min_patch, max(n//3, 1)))
        start = int(rng.integers(0, n - ps + 1))
        end = start + ps

        noise_scale = T * 0.3
        p_new = p.copy()
        p_new[start:end] += rng.normal(0, noise_scale, ps)
        p_new = np.maximum(p_new, 1e-12)
        p_new_sum = float(p_new.sum())
        p_new = p_new / p_new_sum

        c_new = _full_convolve_fft(p_new)
        maxc_new = float(c_new.max())
        score_new = float(2.0 * n * maxc_new)

        delta = score_new - best_score
        if delta < -1e-14:
            p = p_new
            best_p = p_new.copy()
            best_score = score_new
            c = c_new
            T = max(T_min, T * 0.995)
            stall = 0
        else:
            if T > T_min:
                accept_prob = np.exp(-delta / T) if delta / T < 50 else 0.0
                if rng.random() < accept_prob:
                    p = p_new
                    c = c_new
            T = max(T_min, T * cool_rate)
            stall += 1
        
        # Restart if stalled
        if stall > 5000:
            T = max(T, 0.02)
            stall = 0

    return best_p, best_score


def _subgradient_descent(p0, deadline, rng, n_steps=5000, step_scale=1.0):
    """
    Projected subgradient method with Nesterov momentum.
    Uses the exact subgradient (argmax of convolution).
    """
    n = p0.size
    p = np.maximum(p0, 1e-12).copy()
    p = p / p.sum()
    
    best_p = p.copy()
    c = _full_convolve_fft(p)
    best_score = float(2.0 * n * c.max())
    
    # Nesterov momentum
    y = p.copy()
    beta = 1.0  # momentum parameter
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at y
        c_y = _full_convolve_fft(y)
        k_star = int(np.argmax(c_y))
        max_c = float(c_y[k_star])
        
        # Subgradient: sg[i] = y[k_star - i] if 0 <= k_star - i <= n-1
        sg = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            sg[i] = y[k_star - i]
        
        # Step size (decreasing)
        step = step_scale / np.sqrt(t)
        
        # Gradient step
        p_new = p - step * sg
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        p_new = _proj_simplex(p_new)
        
        # Nesterov update

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = _full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:

def _soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _corr_weights(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _make_profile(n, ptype='quadratic', **kwargs):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if ptype == 'uniform':
        a = np.ones(n)
    elif ptype == 'linear':
        a = 1.0 + kwargs.get('slope', 1.5) * (t - 0.5)
    elif ptype == 'quadratic':
        a = 1.0 + kwargs.get('a1', 0.8) * u + kwargs.get('a2', 0.3) * u**2
    elif ptype == 'cubic':
        a = 1.0 + kwargs.get('a1', 1.2) * u + kwargs.get('a3', -0.8) * u**3
    elif ptype == 'U_shaped':
        a = 1.0 + kwargs.get('depth', 0.5) * u**2
    elif ptype == 'V_shaped':
        a = 1.0 + kwargs.get('depth', 0.5) * np.abs(u)
    elif ptype == 'cosine':
        a = 1.0 + kwargs.get('amp', 0.3) * np.cos(2.0 * np.pi * t)
    elif ptype == 'cos2':

    elif ptype == 'sigmoid_atan':
        k = kwargs.get('k', 2.0)
        a = 1.0 + kwargs.get('depth', 0.4) * np.tanh(k * u)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s

    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))

def _adam_gd(p0, deadline, rng, lr=0.06, beta_init_frac=0.1,
             beta_decay=0.72, no_imp_mult=4, var_weight=0.0,
             use_adaptive_lr=True, multi_scale=False):
    """Soft-max gradient descent in logit space with Adam."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)
    beta2 = beta * 5.0 if multi_scale else 0.0

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = _soft_max_weights(c, beta)
        gp = 2.0 * _corr_weights(w, p)

        if multi_scale and beta2 > 0:
            _, w2 = _soft_max_weights(c, beta2)
            gp += 0.5 * _corr_weights(w2, p)

        if var_weight > 0.0:
            gp += var_weight * (p - p.mean())

        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)

        if use_adaptive_lr:
            grad_norm = float(np.sqrt(np.mean(gz * gz))) + 1e-15
            eff_lr = lr * min(1.0, 0.15 / grad_norm)
        else:
            eff_lr = lr

        z = z - eff_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 150 == 0:
                    lr = max(0.003, lr * 0.94)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Adaptive beta
        if adam_t % 25 == 0:
            beta = max(1e-14, beta * beta_decay)
            beta2 = beta * 5.0 if multi_scale else 0.0
            # Check effective support
            w_sq_sum = float(np.sum(w * w))
            if w_sq_sum > 1e-15:
                eff_support = 1.0 / w_sq_sum
                if eff_support < 3.0:
                    beta = max(1e-14, beta * 1.8)
                elif eff_support > 100.0:
                    beta = max(1e-14, beta * 0.6)

        # Restart strategy
        no_imp_limit = max(80, no_imp_mult * 50)
        if no_imp > no_imp_limit:
            if best_score < 1.51:
                noise_scale = 0.06
            elif best_score < 1.53:
                noise_scale = 0.12
            elif best_score < 1.56:
                noise_scale = 0.22
            else:
                noise_scale = 0.35
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_refine(a, max_iters=8, initial_blend=0.08):
    """LP-based direction finding and blending."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s <= 1e-10:
        return a, _internal_score(a)
    a = a / s

    c = _full_convolve_fft(a)
    M = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * M)
    t_blend = initial_blend

    for _ in range(max_iters):
        # Build LP matrix: A[k, j] = a[k-j] if 0 <= k-j <= n-1
        A_mat = np.zeros((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                A_mat[k, lo:hi] = a[hi - 1:lo - 1:-1] if hi - 1 >= lo else a[k - lo:k + 1]

        # More careful: A_mat[k,j] = a[k-j] if 0<=k-j<=n-1
        A_mat = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            for k in range(j, j + n):
                A_mat[k, j] = a[k - j]

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'maxiter': 5000})
            if not res.success:
                break
            b_dir = res.x
            b_sum = float(b_dir.sum())
            if b_sum <= 1e-10:
                break
            b_dir = b_dir / b_sum
        except Exception:
            break

        # Try blending
        improved = False
        for tt in [t_blend, t_blend * 0.5, t_blend * 0.25]:
            a_new = (1.0 - tt) * a + tt * b_dir
            a_new = np.maximum(a_new, 0)
            sn = float(a_new.sum())
            if sn <= 1e-10:
                continue
            a_new = a_new / sn
            c_new = _full_convolve_fft(a_new)
            sc_new = float(2.0 * n * c_new.max())
            if sc_new < best_score - 1e-12:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                t_blend = tt
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _subgradient_nesterov(p0, deadline, step_scale=0.5, max_steps=3000):
    """Projected subgradient with Nesterov momentum."""
    n = p0.size
    p = np.maximum(p0, 1e-12).copy()
    p = p / p.sum()
    y = p.copy()

    c0 = _full_convolve_fft(p)
    best_score = float(2.0 * n * c0.max())
    best_p = p.copy()
    t_count = 0

    while time.monotonic() < deadline and t_count < max_steps:
        t_count += 1

        c_y = _full_convolve_fft(y)
        k_star = int(np.argmax(c_y))
        max_c = float(c_y[k_star])

        # Subgradient: sg[i] = y[k_star - i] for valid i
        sg = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo <= hi:
            sg[lo:hi + 1] = y[k_star:hi - lo - 1:-1] if k_star > hi - lo else y[k_star:hi - lo - 1:-1]
            # More carefully: sg[i] = y[k_star - i] for i in [lo, hi]
            indices = np.arange(lo, hi + 1)
            sg[indices] = y[k_star - indices]

        step = step_scale / np.sqrt(float(t_count))
        p_new = p - step * sg
        p_new = _proj_simplex(p_new)

        # Nesterov
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * float(t_count) * float(t_count))) / 2.0
        mu = (beta_t - 1.0) / (beta_t + 1.0)
        y = p_new + mu * (p_new - p)
        p = p_new

        score = float(2.0 * n * max_c)
        if score < best_score - 1e-12:
            best_score = score
            best_p = p_new.copy()

    return best_p, best_score


def _sim_anneal(p0, deadline, rng, max_iter=200000):
    """Patch-based simulated annealing."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    p = p0.copy()
    c = _full_convolve_fft(p)
    best_p = p0.copy()
    best_score = float(2.0 * n * c.max())

    T = 0.02
    T_min = 0.0003
    cool_rate = 0.9997
    min_patch = max(3, n // 100)
    max_patch = max(min_patch + 1, n // 20)
    stall = 0

    for _ in range(max_iter):
        if time.monotonic() > deadline:
            break
        ps = int(np.clip(rng.integers(min_patch, max_patch + 1), min_patch, max(3, n // 3)))
        start = int(rng.integers(0, n - ps + 1))
        end = start + ps

        p_new = p.copy()
        p_new[start:end] += rng.normal(0, T * 0.25, ps)
        p_new = np.maximum(p_new, 1e-12)
        p_new = p_new / p_new.sum()

        c_new = _full_convolve_fft(p_new)
        sc_new = float(2.0 * n * c_new.max())
        delta = sc_new - best_score

        if delta < -1e-14:
            p = p_new
            best_p = p_new.copy()
            best_score = sc_new
            c = c_new
            T = max(T_min, T * 0.995)
            stall = 0
        else:
            if T > T_min:
                ratio = -delta / T
                if ratio < 50:
                    if rng.random() < np.exp(-ratio):
                        p = p_new
                        c = c_new
            T = max(T_min, T * cool_rate)
            stall += 1

        if stall > 8000:
            T = max(T, 0.015)
            stall = 0

    return best_p, best_score

        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                parent_n = parent.size
    except Exception:
        parent = None

    print(f"Parent: n={parent_n} score={_internal_score(parent) if parent is not None else 'N/A'}")
    if parent is not None:
        update_best(parent, _internal_score(parent), "parent")

    # Phase 0: Generate initial candidates
    print("Phase 0: Initial candidates")
    init_n_values = [256, 512, 1024]
    candidates = []

    profile_defs = [
        ('quad_1', 'quadratic', {'a1': 0.8, 'a2': 0.3}),
        ('quad_2', 'quadratic', {'a1': 1.2, 'a2': 0.5}),
        ('quad_3', 'quadratic', {'a1': 0.5, 'a2': 0.8}),
        ('cubic_1', 'cubic', {'a1': 1.0, 'a3': -0.6}),
        ('cubic_2', 'cubic', {'a1': 1.5, 'a3': -1.0}),
        ('lin_1', 'linear', {'slope': 1.5}),
        ('lin_2', 'linear', {'slope': 2.5}),
        ('U_1', 'U_shaped', {'depth': 0.4}),
        ('U_2', 'U_shaped', {'depth': 0.8}),
        ('V_1', 'V_shaped', {'depth': 0.5}),
        ('cos_1', 'cosine', {'amp': 0.25}),
        ('cos2_1', 'cos2', {'amp': 0.15}),
        ('atan_1', 'sigmoid_atan', {'k': 2.0, 'depth': 0.4}),
    ]

    for n in init_n_values:
        for label, ptype, kw in profile_defs:
            p = _make_profile(n, ptype, **kw)
            sc = _internal_score(p)
            if np.isfinite(sc):
                candidates.append((sc, p, n, label))
                update_best(p, sc, f"init {label} n={n}")
            if time.monotonic() > t_start + 30:
                break

    # Add parent at various lengths
    if parent is not None:
        for n in init_n_values:
            if parent_n != n:
                p_interp = _interp_profile(parent, n)
                if p_interp is not None:
                    sc = _internal_score(p_interp)
                    candidates.append((sc, p_interp, n, "parent_interp"))
                    update_best(p_interp, sc, f"parent_interp n={n}")
        if parent_n in init_n_values or parent_n <= 2048:
            candidates.append((_internal_score(parent), parent.copy(), parent_n, "parent"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # Phase 1: Polish top candidates
    print("Phase 1: Polish top candidates")
    phase1_end = min(deadline, t_start + 300.0)
    polished = []
    count = 0
    for sc, p, n, label in candidates:
        if time.monotonic() > phase1_end:
            break
        if count >= 6:
            break
        per_end = min(phase1_end, time.monotonic() + (phase1_end - t_start - 300.0 + 300.0) / max(1, 6 - count))
        per_end = min(phase1_end, time.monotonic() + 45.0)
        p_best, sc_best = _adam_gd(p, per_end, rng, lr=0.07,
                                   beta_init_frac=0.12, beta_decay=0.70,
                                   multi_scale=True)
        update_best(p_best, sc_best, f"polish {label} n={n}")
        polished.append((sc_best, p_best, n, label + "_polished"))
        count += 1

    candidates.extend(polished)
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}")

    # Phase 2: Multi-length escalation
    print("Phase 2: Multi-length escalation")
    phase2_end = min(deadline, t_start + 800.0)
    if best_seq is not None:
        for target_n in [512, 1024, 2048]:
            if time.monotonic() > phase2_end - 60.0:
                break
            if best_seq.size == target_n:
                p_interp = best_seq.copy()
            else:
                p_interp = _interp_profile(best_seq, target_n)
            if p_interp is None:
                continue
            sc_interp = _internal_score(p_interp)
            update_best(p_interp, sc_interp, f"interp n={target_n}")

            # Polish
            per_end = min(phase2_end - 20.0, time.monotonic() + 50.0)
            p_b, sc_b = _adam_gd(p_interp, per_end, rng, lr=0.05,
                                 beta_init_frac=0.08, beta_decay=0.72,
                                 multi_scale=True)
            update_best(p_b, sc_b, f"escalate n={target_n}")

            # LP refinement
            if time.monotonic() < phase2_end - 30.0:
                p_lp, sc_lp = _lp_refine(p_b, max_iters=5, initial_blend=0.06)
                update_best(p_lp, sc_lp, f"LP n={target_n}")

            # Final polish after LP
            if time.monotonic() < phase2_end - 15.0:
                per_end3 = min(phase2_end - 10.0, time.monotonic() + 25.0)
                p_f, sc_f = _adam_gd(p_lp, per_end3, rng, lr=0.04,
                                     beta_init_frac=0.06, beta_decay=0.75)
                update_best(p_f, sc_f, f"post-LP n={target_n}")

    print(f"Phase 2 done: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # Phase 3: Multi-restart with LP kicks
    print("Phase 3: Multi-restart at best length")
    phase3_end = min(deadline, t_start + 1350.0)
    if best_seq is not None:
        n_best = best_seq.size
        n_restarts = 4
        per_restart = min(80.0, (phase3_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase3_end:
                break
            per_end = min(phase3_end, time.monotonic() + per_restart)

            if i == 0:
                p_start = best_seq.copy()
            else:
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                noise = 0.1 * (1.0 + 0.4 * i)
                z = z + noise * rng.normal(size=n_best)
                z = np.clip(z, -50, 50)
                z -= z.mean()
                p_start = _softmax_from_z(z)

            p_try, sc_try = _adam_gd(p_start, per_end, rng, lr=0.05,
                                     beta_init_frac=0.08, beta_decay=0.73,
                                     multi_scale=(i % 2 == 0))
            update_best(p_try, sc_try, f"restart{i} n={n_best}")

            # LP kick after each restart
            if time.monotonic() < phase3_end - 20.0:
                p_lp, sc_lp = _lp_refine(p_try, max_iters=4, initial_blend=0.05)
                update_best(p_lp, sc_lp, f"LP_restart{i}")

        # Subgradient refinement
        if time.monotonic() < phase3_end - 30.0:
            per_sub = min(40.0, phase3_end - time.monotonic() - 10.0)
            p_sub, sc_sub = _subgradient_nesterov(best_seq, time.monotonic() + per_sub,
                                                  step_scale=0.3, max_steps=2000)
            update_best(p_sub, sc_sub, "subgradient")

    print(f"Phase 3 done: best={best_score:.10f}")

    # Phase 4: Simulated annealing
    print("Phase 4: Simulated annealing")
    phase4_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        p_sa, sc_sa = _sim_anneal(best_seq, phase4_end, rng, max_iter=300000)
        update_best(p_sa, sc_sa, "SA")
        print(f"  SA result: {sc_sa:.10f}")

    # Phase 5: Final polish
    print("Phase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline:
        per_final = min(60.0, deadline - time.monotonic() - 5.0)
        if per_final > 10.0:
            p_fin, sc_fin = _adam_gd(best_seq, time.monotonic() + per_final, rng,
                                     lr=0.03, beta_init_frac=0.05, beta_decay=0.80,
                                     multi_scale=True, no_imp_mult=3)
            update_best(p_fin, sc_fin, "final_polish")

        # One more LP
        if time.monotonic() < deadline - 15.0:
            p_final_lp, sc_final_lp = _lp_refine(best_seq, max_iters=5, initial_blend=0.03)
            update_best(p_final_lp, sc_final_lp, "final_LP")

            # Tiny final GD
            if time.monotonic() < deadline - 5.0:
                tiny_end = min(deadline - 2.0, time.monotonic() + 10.0)
                p_tiny, sc_tiny = _adam_gd(p_final_lp, tiny_end, rng, lr=0.02,
                                           beta_init_frac=0.04, beta_decay=0.85)
                update_best(p_tiny, sc_tiny, "tiny_final")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(512, 'quadratic', a1=0.8, a2=0.3)

        # Exact score
        if final_seq.size <= 4096:
            sc_exact = _true_score(final_seq)
            if np.isfinite(sc_exact):
                print(f"Exact score: {sc_exact:.10f}")
                best_score = min(best_score, sc_exact)

        # Try evaluator
        if final_seq.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    best_score = min(best_score, sc_eval)
            except Exception:
                pass

        print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(512, 'quadratic', a1=0.8, a2=0.3)
    print(f"FALLBACK: n=512, score={_internal_score(fallback):.10f}")
    return [float(x) for x in fallback]
