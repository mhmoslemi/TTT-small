```python
def lp_step(f, M, solver):
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g)
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    #              g_j >= 0
    
    # The constraint matrix is n x (2n-1) (convolution matrix)
    # Row k, column j: f[k-j] if 0 <= k-j < n, else 0
    
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    # Compute f*g for all lags
    # (f*g)[k] = sum_{j} f[k-j] * g[j] for valid k-j
    
    # Build the constraint matrix
    # C[k, j] = f[k-j] if 0 <= k-j < n, else 0
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                C[k, j] = f[idx]
    
    objective = cp.Maximize(cp.sum(g))
    constraints = [C @ g <= M]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    return g.value
```

```python
        pass
    return None


def _lp_optimize(f_init, deadline, rng, lr_init=0.3, lp_interval=1):
    f = np.asarray(f_init, dtype=np.float64).ravel().copy()
    f = _normalize(f)
    n = len(f)
    best_f = f.copy()
    best_sc = _score(f)
    it = 0
    no_imp = 0
    lr = lr_init
    while time.monotonic() < deadline:
        it += 1
        if it % lp_interval == 0:
            c = _convolve(f)
            M = float(c.max())
            g = _lp_direction(f, M, time_limit=2.0)
            if g is not None:
                f_new = (1.0 - lr) * f + lr * g
                f_new = _normalize(f_new)
                sc_new = _score(f_new)
                if sc_new < best_sc - 1e-14:
                    best_sc = sc_new
                    best_f = f_new.copy()
                    no_imp = 0
                    f = f_new
                    lr = min(0.5, lr * 1.02)
                else:
                    no_imp += 1
                    if no_imp % 10 == 0:
                        lr = max(0.05, lr * 0.9)
            else:
                no_imp += 1
                lr = max(0.05, lr * 0.95)
        if time.monotonic() < deadline and it % 5 == 0:
            c = _convolve(f)
            k_max = int(np.argmax(c))
            grad = np.zeros(n)
            for i in range(n):
                j = k_max - i
                if 0 <= j < n:
                    grad[i] = f[j] + f[i]
            grad = grad - grad.mean()
            gnorm = float(np.linalg.norm(grad))
            if gnorm > 1e-15:
                lr_g = min(0.1, 0.05 / (gnorm / n))
                f_new = f - lr_g * grad
                f_new = _normalize(f_new)
                sc_new = _score(f_new)
                if sc_new < best_sc - 1e-14:
                    best_sc = sc_new
                    best_f = f_new.copy()
                    no_imp = 0
                    f = f_new
                else:
                    no_imp += 1
        if no_imp > 200:
            f = best_f + 0.05 * (rng.random(n) - 0.5)
            f = _normalize(f)
            no_imp = 0
            lr = lr_init
        if it % 500 == 0:
            print(f"  LP-opt: it={it}, best={best_sc:.10f}, lr={lr:.4f}, no_imp={no_imp}")
    return best_f, best_sc


def _nesterov_subgrad(f_init, deadline, rng, lr_init=0.05, K=1):
    f = np.asarray(f_init, dtype=np.float64).ravel().copy()
    f = _normalize(f)
    n = len(f)
    best_f = f.copy()
    best_sc = _score(f)
    v = np.zeros(n)
    lr = lr_init
    it = 0
    no_imp = 0
    while time.monotonic() < deadline:
        it += 1
        c = _convolve(f)
        if K == 1:
            k_max = int(np.argmax(c))
            grad = np.zeros(n)
            for i in range(n):
                j = k_max - i
                if 0 <= j < n:
                    grad[i] = f[j] + f[i]
        else:
            idx = np.argsort(c)[::-1][:K]
            grad = np.zeros(n)
            for k in idx:
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        grad[i] += f[j] + f[i]
        grad = grad - grad.mean()
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        v = 0.9 * v - lr * grad
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f
        else:
            f_new = f_new / s
        sc_new = _score(f_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_f = f_new.copy()
            f = f_new
            no_imp = 0
        else:
            f = f_new if f_new.sum() > 0.01 else f
            no_imp += 1
            if no_imp % 50 == 0:
                lr *= 0.92
        if no_imp > 300:
            f = _normalize(best_f + 0.1 * rng.normal(size=n))
            v = np.zeros(n)
            no_imp = 0
            lr = lr_init
        if it % 2000 == 0:
            print(f"  NS: it={it}, best={best_sc:.10f}, lr={lr:.6f}")
    return best_f, best_sc


def _subgrad_fine(f_init, deadline, rng, lr_init=0.02):
    f = np.asarray(f_init, dtype=np.float64).ravel().copy()
    f = _normalize(f)
    n = len(f)
    best_f = f.copy()
    best_sc = _score(f)
    lr = lr_init
    no_imp = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = _convolve(f)
        k_max = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = f[j] + f[i]
        grad = grad - grad.mean()
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        step = lr * n / gnorm
        f_new = f - step * grad
        f_new = _normalize(f_new)
        sc_new = _score(f_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_f = f_new.copy()
            f = f_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                lr *= 0.9
        if no_imp > 100:
            break
        if it % 5000 == 0:
            print(f"  SF: it={it}, best={best_sc:.10f}, lr={lr:.8f}")
    return best_f, best_sc


def _pairwise_exchange(f_init, deadline, rng):
    f = np.asarray(f_init, dtype=np.float64).ravel().copy()
    f = _normalize(f)
    n = len(f)
    best_f = f.copy()
    best_sc = _score(f)
    it = 0
    no_imp = 0
    while time.monotonic() < deadline:
        it += 1
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        delta = float(rng.normal(0, 0.02))
        f_new = f.copy()
        f_new[i] += delta
        f_new[j] -= delta
        f_new = _normalize(f_new)
        sc_new = _score(f_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_f = f_new.copy()
            f = f_new
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 500:
            break
        if it % 5000 == 0:
            print(f"  PE: it={it}, best={best_sc:.10f}")
    return best_f, best_sc


def _make_initial(n, seed=0):
    rng = np.random.default_rng(seed)
    profiles = []
    t = np.linspace(0, 1, n)
    profiles.append(np.ones(n) / n)
    for eta in [1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-6)
        profiles.append(a / a.sum())
    for a3 in [-0.5, -1.0, -1.5, -2.0]:
        u = 2.0 * t - 1.0
        a = 1.0 + 1.0 * u + a3 * u ** 3
        a = np.maximum(a, 1e-6)
        profiles.append(a / a.sum())
    for freq in [2, 3, 5, 7]:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 1e-6)
        profiles.append(a / a.sum())
    a = 1.0 + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 1e-6)
    profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    print(f"Parent: {'n=' + str(len(parent)) if parent is not None else 'none'}, "
          f"score={_score(parent):.10f}" if parent is not None else "Parent: none")

    # Target length
    n_target = 1024

    # Phase 0: Gather initial candidates
    candidates = []
    if parent is not None and len(parent) == n_target:
        sc = _score(parent)
        candidates.append((sc, parent.copy(), "parent"))
        update(parent, sc, "parent initial")
    elif parent is not None:
        from scipy.interpolate import interp1d
        x0 = np.linspace(0, 1, len(parent))
        x1 = np.linspace(0, 1, n_target)
        p_interp = np.interp(x1, x0, parent)
        p_interp = _normalize(p_interp)
        sc = _score(p_interp)
        candidates.append((sc, p_interp, "parent_interp"))
        update(p_interp, sc, "parent_interp")

    profiles = _make_initial(n_target, seed=int(seed))
    for i, p in enumerate(profiles):
        sc = _score(p)
        candidates.append((sc, p, f"profile_{i}"))
        update(p, sc, f"init_profile_{i}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: {len(candidates)} candidates, best={best_score:.10f}")

    # Phase 1: LP optimization on top candidates
    phase1_end = min(deadline, t_start + 500.0)
    top_k = min(4, len(candidates))
    for idx in range(top_k):
        if time.monotonic() >= phase1_end - 30:
            break
        sc, p, label = candidates[idx]
        per_end = min(phase1_end, time.monotonic() + (phase1_end - time.monotonic()) / (top_k - idx))
        print(f"  LP-opt on {label} (score={sc:.10f})...")
        p_best, sc_best = _lp_optimize(p, per_end, rng)
        update(p_best, sc_best, f"LP_{label}")
        candidates.append((sc_best, p_best, f"LP_{label}"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}")

    # Phase 2: Nesterov subgradient polish
    phase2_end = min(deadline, t_start + 900.0)
    if best_seq is not None:
        top_k2 = min(3, len(candidates))
        for idx in range(top_k2):
            if time.monotonic() >= phase2_end - 20:
                break
            sc, p, label = candidates[idx]
            per_end = min(phase2_end, time.monotonic() + (phase2_end - time.monotonic()) / (top_k2 - idx))
            print(f"  NS on {label} (score={sc:.10f})...")
            p_best, sc_best = _nesterov_subgrad(p, per_end, rng, lr_init=0.04, K=1)
            update(p_best, sc_best, f"NS_{label}")

    print(f"Phase 2 done: best={best_score:.10f}")

    # Phase 3: Multi-restart with LP
    phase3_end = min(deadline, t_start + 1300.0)
    if best_seq is not None:
        n_restarts = 4
        per_restart = (phase3_end - time.monotonic()) / n_restarts
        for r in range(n_restarts):
            if time.monotonic() >= phase3_end:
                break
            noise = 0.08 + 0.05 * r
            p_perturb = _normalize(best_seq + noise * rng.standard_normal(n_target))
            per_end = min(phase3_end, time.monotonic() + per_restart)
            print(f"  Restart {r}: noise={noise:.3f}")
            p_best, sc_best = _lp_optimize(p_perturb, per_end, rng, lr_init=0.2)
            update(p_best, sc_best, f"restart_{r}")

    print(f"Phase 3 done: best={best_score:.10f}")

    # Phase 4: Fine subgradient + pairwise exchange
    phase4_end = min(deadline, t_start + 1600.0)
    if best_seq is not None:
        p1, sc1 = _subgrad_fine(best_seq, min(phase4_end, time.monotonic() + 150.0), rng)
        update(p1, sc1, "fine_subgrad")
        if time.monotonic() < phase4_end - 10:
            p2, sc2 = _pairwise_exchange(best_seq, phase4_end, rng)
            update(p2, sc2, "pairwise")

    print(f"Phase 4 done: best={best_score:.10f}")

    # Phase 5: Final LP polish
    phase5_end = deadline
    if best_seq is not None and time.monotonic() < phase5_end - 5:
        p_final, sc_final = _lp_optimize(best_seq, phase5_end, rng, lr_init=0.1, lp_interval=2)
        update(p_final, sc_final, "final_LP")

    # Final validation
    if best_seq is not None:
        final = _normalize(best_seq)
        sc_fft = _score(final)
        if len(final) <= 2048:
            sc_exact = _score_exact(final)
            if np.isfinite(sc_exact):
                print(f"Exact score: {sc_exact:.10f}")
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception:
                pass
        print(f"FINAL: n={len(final)}, fft_score={sc_fft:.10f}, best_tracked={best_score:.10f}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(n_target) / n_target
    print("FALLBACK: uniform")
    return [float(x) for x in fallback]
```
