def fft_grad(p, w):
    """Compute gradient of soft_max(convolve(p,p)) using FFT.
    w: softmax weights over convolution values (length 2n-1)
    Returns gradient (length n) = 2 * sum_k w[k] * p[k-i] for each i.
    """
    n = len(p)
    size = 1
    while size < 2 * (2 * n - 1):
        size *= 2
    W = fft(w, size)
    P = fft(p, size)
    corr = ifft(W * np.conj(P)).real
    return 2.0 * corr[:n]

    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:

    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

    z -= np.mean(z)

    M = float(np.convolve(f, f).max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g = lp_direction(f, M, timeout=20, active_frac=active_frac)
        if g is None:
            if no_improve > 50:
                break
            no_improve += 1
            M *= 0.995
            continue

        # Line search over step sizes
        best_F = None
        best_S = np.inf
        for t in [1.0, t_step, t_step * 0.5, t_step * 0.25, t_step * 0.75]:
            f_new = (1.0 - t) * f + t * g
            f_new = normalize(f_new)
            if f_new is None:
                continue
            M_new = float(np.convolve(f_new, f_new).max())
            S = 2.0 * n * M_new
            if S < best_S:
                best_S = S
                best_F = f_new

        if best_F is not None and best_S < best_score - 1e-12:
            f = best_F.copy()
            M = best_S / (2.0 * n)
            best_f = f.copy()
            best_score = best_S
            no_improve = 0
            t_step = max(t_min, t_step * 0.99)
        else:
            no_improve += 1
            t_step = max(t_min, t_step * 0.8)
            if no_improve > 200:
                break

    return best_f, best_score


def fft_grad(p, w):
    """
    Compute gradient of soft_max(convolve(p,p)) w.r.t. p using FFT.
    w: softmax weights over convolution values (length 2n-1).
    Returns gradient (length n) where grad[i] = 2 * sum_k w[k]*p[k-i] (valid range).
    """
    n = len(p)
    size = 1
    while size < 2 * (2 * n - 1):
        size *= 2
    W = fft(w, size)
    P = fft(p, size)
    corr = ifft(W * np.conj(P)).real
    return 2.0 * corr[:n]


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_profile(n, eta, gamma=0.0, delta=0.0):
    """Create profile: a(u) = 1 + eta*u + gamma*u^2 + delta*u^3, u=2t-1."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u + gamma * u * u + delta * u * u * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def adam_polish_fft(p0, deadline, lr=0.05, beta_init=5.0, verbose=False):
    """
    Adam gradient descent in logit space with FFT-based gradient.
    Uses annealed soft-max to approximate the max.
    """
    n = len(p0)
    p = np.maximum(p0, 1e-12)
    p = p / p.sum()

    z = np.log(p)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    beta = beta_init
    best_p = p.copy()
    best_score = 2.0 * n * float(np.convolve(p, p).max())
    no_imp = 0
    base_lr = lr
    restart_count = 0

    while time.monotonic() < deadline:
        it += 1

        c = np.convolve(p, p)
        max_c = float(c.max())

        # Soft-max weights
        x = (c - max_c) / beta
        ex = np.exp(x)
        s_ex = ex.sum()
        w = ex / s_ex

        # Gradient via FFT
        grad = fft_grad(p, w)

        # Logit space gradient (projected onto tangent of simplex)
        dot = float(np.dot(p, grad))
        gz = p * (grad - dot)

        if not np.all(np.isfinite(gz)):
            p = best_p.copy()
            z = np.log(np.maximum(p, 1e-12))
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = best_p.copy()
            z = np.log(best_p)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.003, lr * 0.9)

        # Anneal beta
        if it % 800 == 0:
            beta = min(200.0, beta * 2.5)

        if no_imp > 8000:
            restart_count += 1
            if restart_count > 10:
                break
            rng = np.random.default_rng(it + 9999)
            noise_scale = 0.3 + 0.1 * restart_count
            z_pert = np.log(np.maximum(best_p, 1e-12))
            z_pert -= np.mean(z_pert)
            z = z_pert + noise_scale * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = beta_init
            p = np.exp(z - z.max())
            p = p / p.sum()

        if verbose and it % 10000 == 0:
            print(f"      Adam iter {it}: score={best_score:.8f}, beta={beta:.1f}, restarts={restart_count}")

    return best_p, best_score


def local_search(p, deadline, max_rounds=20000):
    """
    Coordinate descent with simulated annealing acceptance.
    Tries point mutations and small neighborhood changes.
    """
    n = len(p)
    p = normalize(p)
    if p is None:
        return None, np.inf

    best_p = p.copy()
    best_score = 2.0 * n * float(np.convolve(p, p).max())
    current_score = best_score
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)

    temp = 0.001
    no_improve = 0
    step_scale = 0.01

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Choose a random coordinate and a perturbation
        i = rng.integers(0, n)
        delta = rng.normal(0, step_scale) * p.mean()

        p_new = p.copy()
        p_new[i] = max(0.0, p_new[i] + delta)
        s = p_new.sum()
        if s < 1e-10:
            continue
        p_new = p_new / s

        c_new = np.convolve(p_new, p_new)
        score_new = 2.0 * n * float(c_new.max())

        # Metropolis acceptance
        diff = score_new - current_score
        if diff < 0 or rng.random() < np.exp(-diff / max(temp, 1e-10)):
            p = p_new
            current_score = score_new
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_p = p_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Cool down
        if no_improve > 200:
            temp *= 0.9
            step_scale *= 0.95
            no_improve = 0

        if no_improve > 5000:
            break

    return best_p, best_score


def multi_start_lp(n, deadline, n_starts=5):
    """Try multiple random/parametric starts with LP refinement at given n."""
    best_f = None
    best_score = np.inf
    rng = np.random.default_rng(777)

    # Generate diverse initial profiles
    candidates = [
        make_profile(n, 1.0, 0.0),
        make_profile(n, 2.0, -0.5),
        make_profile(n, 1.5, 0.3, -0.2),
        make_profile(n, 0.5, -0.3, 0.1),
        make_profile(n, 2.5, -1.0, 0.3),
    ]
    # Add random smooth profiles
    for _ in range(3):
        coeffs = rng.normal(0, 0.5, size=4)
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + coeffs[0]*u + coeffs[1]*u**2 + coeffs[2]*u**3 + coeffs[3]*u**4
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
        candidates.append(a)

    per_start_time = max(30, (deadline - time.monotonic()) / len(candidates) - 5)

    for ci, cand in enumerate(candidates):
        if time.monotonic() > deadline:
            break
        sc = internal_score(cand)
        dl = min(deadline, time.monotonic() + per_start_time)
        f_best, sc_best = lp_iterate(cand, max_iters=300, t_init=0.4,
                                      deadline=dl, active_frac=0.0)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_f = f_best
        if (ci + 1) % 3 == 0:
            print(f"    Multi-start LP: {ci+1}/{len(candidates)} done, best so far={best_score:.8f}")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. LP at n=128 with multiple parametric starts (300s)
    2. Upscale to n=512 + LP refinement (300s)
    3. FFT Adam gradient descent at n=512 (500s)
    4. Upscale to n=1024 + FFT Adam (400s)
    5. Local search / coordinate descent (100s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(0, budget_s - 25)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent
                best_score = internal_score(parent)
                print(f"Parent loaded: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass

    # Phase 1: LP at n=128
    n1 = 128
    deadline_p1 = min(deadline, t_start + 350)
    print(f"\n=== Phase 1: LP at n={n1} ===")

    if best_seq is not None:
        parent_128 = interpolate_seq(best_seq, n1)
    else:
        parent_128 = make_profile(n1, 1.5, 0.0)

    f_best, sc_best = multi_start_lp(n1, deadline_p1, n_starts=6)
    if parent_128 is not None:
        # Also run LP on parent interpolation
        sc_parent = internal_score(parent_128)
        print(f"  Parent at n={n1}: score={sc_parent:.8f}")
        if sc_parent < best_score:
            f_p, sc_p = lp_iterate(parent_128, max_iters=300, t_init=0.4,
                                    deadline=min(deadline_p1, time.monotonic() + 100))
            if f_p is not None and sc_p < best_score:
                best_score = sc_p
                best_seq = f_p

    if f_best is not None and sc_best < best_score:
        best_score = sc_best
        best_seq = f_best
    print(f"  Phase 1 best: score={best_score:.8f}")

    # Phase 2: Upscale to n=512 + LP
    n2 = 512
    deadline_p2 = min(deadline, t_start + 700)
    print(f"\n=== Phase 2: LP at n={n2} ===")
    if time.monotonic() < deadline_p2 - 60:
        a2 = interpolate_seq(best_seq, n2)
        sc2 = internal_score(a2)
        print(f"  Initial at n={n2}: score={sc2:.8f}")
        f_best, sc_best = lp_iterate(a2, max_iters=150, t_init=0.3,
                                       deadline=deadline_p2 - 30, active_frac=0.3)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best
            print(f"  Phase 2 result: score={best_score:.8f}")
        else:
            print(f"  Phase 2: no improvement (best={best_score:.8f})")

    # Phase 3: FFT Adam at n=512
    n3 = 512
    deadline_p3 = min(deadline, t_start + 1300)
    print(f"\n=== Phase 3: FFT Adam at n={n3} ===")
    if time.monotonic() < deadline_p3 - 30:
        if len(best_seq) != n3:
            best_seq = interpolate_seq(best_seq, n3)
        sc3 = internal_score(best_seq)
        print(f"  Initial: score={sc3:.8f}")
        p_best, sc_best = adam_polish_fft(best_seq, deadline_p3 - 20, lr=0.06,
                                           beta_init=3.0, verbose=True)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 3 result: score={best_score:.8f}")
        else:
            print(f"  Phase 3: no improvement (best={best_score:.8f})")

        # Second Adam pass with different hyperparams
        if time.monotonic() < deadline_p3 - 40:
            p_best2, sc_best2 = adam_polish_fft(best_seq, deadline_p3 - 20,
                                                 lr=0.03, beta_init=10.0, verbose=False)
            if sc_best2 < best_score:
                best_score = sc_best2
                best_seq = p_best2
                print(f"  Phase 3b result: score={best_score:.8f}")

    # Phase 4: Try n=1024
    n4 = 1024
    deadline_p4 = min(deadline, t_start + 1650)
    print(f"\n=== Phase 4: FFT Adam at n={n4} ===")
    if time.monotonic() < deadline_p4 - 60:
        a4 = interpolate_seq(best_seq, n4)
        sc4 = internal_score(a4)
        print(f"  Initial: score={sc4:.8f}")
        p_best, sc_best = adam_polish_fft(a4, deadline_p4 - 20, lr=0.05,
                                           beta_init=3.0, verbose=True)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 4 result: score={best_score:.8f}")
        else:
            print(f"  Phase 4: no improvement (best={best_score:.8f})")

    # Phase 5: Local search
    print(f"\n=== Phase 5: Local search ===")
    if time.monotonic() < deadline - 15:
        sc5 = internal_score(best_seq)
        print(f"  Initial: score={sc5:.8f}")
        p_best, sc_best = local_search(best_seq, deadline - 10, max_rounds=50000)
        if p_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 5 result: score={best_score:.8f}")
        else:
            print(f"  Phase 5: no improvement (best={best_score:.8f})")

    # Final: verify and clean up
    final_seq = np.asarray(best_seq, dtype=np.float64)
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = make_profile(512, 1.5, 0.0)
        s = final_seq.sum()
    final_seq = final_seq / s

    final_score = internal_score(final_seq)
    # Also check with evaluate_sequence
    try:
        eval_score = evaluate_sequence([float(x) for x in final_seq])
        print(f"\nFinal: n={len(final_seq)}, internal_score={final_score:.8f}, eval_score={eval_score:.8f}")
    except Exception as e:
        print(f"\nFinal: n={len(final_seq)}, internal_score={final_score:.8f}, eval failed: {e}")

    return [float(x) for x in final_seq]
