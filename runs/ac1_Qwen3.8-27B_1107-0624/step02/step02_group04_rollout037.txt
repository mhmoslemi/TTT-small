import time
from scipy.optimize import linprog

def convolve_fft(a, b):
    """Fast convolution using FFT."""
    n = len(a)
    m = len(b)
    N = 1
    while N < n + m - 1:

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)

    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = convolve_fft(a, a)
    return 2.0 * n * np.max(c) / (np.sum(a))**2

def solve_lp(f, M):

    # Build the constraint matrix A such that A @ g <= M * ones
    # (f * g)[k] = sum_i f[i] * g[k-i] <= M
    # The matrix A has shape (2n-1, n)
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k)

        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)

    b = np.full(2*n - 1, M)

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

"""
Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weights outer loop with LP inner solves,
combined with gradient polishing and simulated annealing.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# Precompute helpers

def _fft_convolve(a, b):
    """Fast linear convolution."""
    n, m = len(a), len(b)
    N = 1
    while N < n + m - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    """Compute objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _conv_matrix(f):
    """Build sparse convolution matrix A where (A@g)[k] = (f*g)[k]."""
    n = len(f)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _solve_inner_lp(f, M):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = _conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return np.zeros(n)

def _mw_lp_iteration(f, M, t):
    """One MW+LP iteration. Returns updated f."""
    g = _solve_inner_lp(f, M)
    s = g.sum()
    if s < 1e-10:
        return f.copy()
    g = g / s
    f_new = (1 - t) * f + t * g
    return f_new / f_new.sum()

def _gradient_step(f, lr, beta=0.01):
    """Soft-max gradient step to reduce max convolution."""
    n = len(f)
    c = _fft_convolve(f, f)
    # Soft max weights
    beta_val = max(1e-6, beta * c.max())
    w = np.exp((c - c.max()) / beta_val)
    w = w / w.sum()
    # Gradient of soft-max(c) w.r.t. f is 2 * conv(w, f) (by symmetry)
    grad = 2.0 * _fft_convolve(w, f)
    # Project gradient to tangent space of simplex
    grad = grad - grad.sum()
    # Adam-like update
    f_new = f - lr * grad
    f_new = np.maximum(f_new, 1e-15)
    return f_new / f_new.sum()

```python
"""
Optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Strategy: Multiplicative weights with LP inner solves (state-of-the-art game-theoretic
approach), combined with subgradient polishing, Adam refinement, and simulated
annealing for local escape. Tries multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    N = 1
    while N < n + m - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def _convolve_mat_precompute(n):
    """Precompute row/col indices for the convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def _solve_lp(f, M, rows, cols):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    nnz = len(rows)
    vals = np.empty(nnz)
    # A[k,j] = f[k-j]
    idx = rows - cols  # k - j for each entry
    vals = f[idx]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except:
        pass
    return np.ones(n) / n  # fallback


def _subgrad_step(f, lr):
    """One subgradient step on max(conv(f,f))."""
    n = len(f)
    c = _fft_convolve(f, f)
    cmax = c.max()
    # Find all lags at (or near) the max
    threshold = cmax * (1 - 1e-6)
    active = c >= threshold
    # Subgradient: for each i, grad_i = sum_{k active} f[k-i] (valid)
    # This is conv(active_as_signal, f) but active is a 0/1 signal
    # Actually: grad_i = sum_{k: active[k]} f[k-i] where 0 <= k-i < n
    # = (active * f)[i] where * is cross-correlation... 
    # More simply: grad = convolve(active, f) evaluated at the appropriate shifts
    # Let me think: conv(f, f)[k] = sum_j f[j]*f[k-j]
    # d/d f[i] conv(f,f)[k] = f[k-i] + f[k-i] if i != k-i, else f[k-i]
    # Wait: conv(f,f)[k] = sum_j f[j]*f[k-j]
    # d/d f[i]: terms where j=i: f[k-i]; terms where k-j=i, i.e., j=k-i: f[k-i]
    # So if i != k-i (i.e., k != 2i): d/d f[i] = 2*f[k-i]
    # If k == 2i: d/d f[i] = f[k-i] = f[i]
    # So grad_i = sum_{k active} (f[k-i] + f[k-i] if k != 2i, else f[k-i])
    # = sum_{k active, k != 2i} 2*f[k-i] + sum_{k active, k = 2i} f[i]
    # = 2 * sum_{k active} f[k-i] - sum_{k active, k=2i} f[i]
    
    # Simpler: just compute conv(active_float, f) which gives sum_k active[k]*f[k-i] (approximately)
    # Actually conv(active, f)[i] = sum_k active[k] * f[i-k]... no.
    # np.convolve(active, f)[m] = sum_j active[j]*f[m-j]
    # I want grad_i = sum_k active[k] * f[k-i] (with valid indices)
    # = sum_k active[k] * f[k-i]
    # Let l = k-i: = sum_l active[l+i] * f[l]
    # This is the cross-correlation of active and f, evaluated at shift i.
    # = np.correlate(active, f, mode='full')[n-1+i]... this is getting complicated.
    
    # Easiest: just use the FFT convolution
    # grad_i = sum_{k: active} f[k-i] for valid (k-i)
    # This is: for each i, sum over k where both k and k-i are valid
    # = (active_reversed * f)[some_shift]... 
    
    # Let me just do it directly for clarity (will be O(n^2) but n=1024 so 1M ops)
    # Actually, I can use: grad = convolve(f, active) evaluated at the right position
    # conv(f, active)[i] = sum_j f[j] * active[i-j]
    # I want sum_k active[k] * f[k-i]
    # Let me define: corr[i] = sum_k active[k] * f[k-i]
    # = sum_k active[k] * f[k-i]
    # If I reverse f: f_rev[j] = f[n-1-j]
    # Then corr[i] = sum_k active[k] * f[k-i]
    # = sum_k active[k] * f_rev[n-1-(k-i)]
    # = sum_k active[k] * f_rev[n-1-k+i]
    # = conv(active, f_rev)[n-1+i]
    
    # So grad = conv(active, f)[ evaluated at indices ]
    # Actually, let me just compute it as:
    # grad[i] = sum_{k} active[k] * f[k-i] for 0 <= k-i < n
    # This is the cross-correlation. In numpy:
    # np.correlate(active, f, 'full') gives the full cross-correlation
    # corr[m] = sum_j active[j+m] * f[j] for appropriate m
    
    # Let me just do the direct computation for active entries
    # For n=1024, active has at most n entries (usually fewer), and for each we do O(n) work
    # Total: O(|active| * n) which is at most O(n^2) = 1M. Fine.
    
    grad = np.zeros(n)
    active_indices = np.where(active)[0]
    for k in active_indices:
        # grad_i += f[k-i] for 0 <= k-i < n, i.e., 0 <= i <= k and k-i < n
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        # f[k-i] for i in [i_min, i_max]
        # = f[k-i_max : k-i_min+1] reversed
        # k-i_max to k-i_min
        f_slice = f[k - i_max : k - i_min + 1][::-1]
        grad[i_min:i_max+1] += f_slice
    
    # Project to tangent space of simplex
    grad = grad - grad.sum() / n
    # Actually, for the simplex constraint sum(a)=1, the projection is:
    # a_new = a - lr * grad, then project to {a >= 0, sum(a) = 1}
    # Simple projection: clip and normalize
    f_new = f - lr * grad
    f_new = np.maximum(f_new, 1e-15)
    f_new = f_new / f_new.sum()
    return f_new


def _adam_soft_max_polish(f, deadline, rng, lr_init=0.01, beta_init_frac=0.1):
    """Adam in log-space with soft-max objective."""
    n = len(f)
    c0 = _fft_convolve(f, f)
    beta = max(1e-10, beta_init_frac * c0.max())
    
    z = np.log(f + 1e-15)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_f = f.copy()
    best_sc = _score(f)
    no_imp = 0
    beta_decay = 0.85
    
    while time.monotonic() < deadline:
        t += 1
        c = _fft_convolve(f, f)
        cmax = c.max()
        
        # Soft max weights
        x = (c - cmax) / beta
        w = np.exp(x)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(c)
            w[c.argmax()] = 1.0
        else:
            w = w / w_sum
        
        # Gradient of soft-max w.r.t. f
        grad_c = 2.0 * _fft_convolve(w, f)
        
        # Log-space gradient
        dot = (f * grad_c).sum()
        gz = f * (grad_c - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = np.log(best_f + 1e-15)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t = 0
            continue
        
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z -= lr_init * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        e = np.exp(z)
        f = (e + 1e-15) / (e + 1e-15).sum()
        
        sc = _score(f)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decay beta periodically
        if t % 100 == 0:
            beta = max(1e-12, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > 500:
            noise = 0.1 if best_sc < 1.55 else 0.2
            z = np.log(best_f + 1e-15)
            z -= z.mean()
            z += noise * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z)
            f = (e + 1e-15) / (e + 1e-15).sum()
            m.fill(0)
            v.fill(0)
            t = 0
            no_imp = 0
            beta = max(1e-10, beta_init_frac * _score(f).max() if hasattr(_score(f), 'max') else 0.05)
            beta = 0.05  # reset to reasonable value
    
    return best_f, best_sc


def _sim_anneal(f, deadline, rng, T_init=0.01, T_min=0.0005):
    """Patch-based simulated annealing."""
    n = len(f)
    best_f = f.copy()
    best_sc = _score(f)
    T = T_init
    min_patch = max(3, n // 50)
    max_patch = max(min_patch + 1, n // 8)
    n_iter = 0
    
    while time.monotonic() < deadline:
        n_iter += 1
        ps = int(rng.integers(min_patch, max_patch + 1))
        start = int(rng.integers(0, n - ps + 1))
        
        f_new = f.copy()
        f_new[start:start+ps] += rng.normal(0, T * 0.5, ps)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        sc_new = _score(f_new)
        delta = sc_new - best_sc
        
        if delta < 0:
            f = f_new
            best_f = f_new.copy()
            best_sc = sc_new
            T = max(T_min, T * 0.99)
        else:
            if T > T_min and rng.random() < np.exp(-delta / T):
                f = f_new
            T = max(T_min, T * 0.999)
        
        if n_iter % 5000 == 0:
            print(f"    SA: iter={n_iter}, T={T:.6f}, best={best_sc:.10f}")
    
    return best_f, best_sc
