"""
Improved LP + Power-Mean optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key improvements over previous approach:
1. Adaptive LP with line search for blend factor
2. Power-mean gradient as complementary optimization
3. Peak-shaving heuristic between LP iterations
4. Multiple lengths with adaptive time allocation
5. Random restarts to escape local minima

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _power_mean_grad(c, p):
    """Compute gradient of the p-mean of c w.r.t. c."""
    c = np.asarray(c, dtype=np.float64)
    c_safe = np.maximum(c, 1e-15)
    cp = c_safe ** p
    s = float(cp.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    w = (c_safe ** (p - 1)) / s
    return w


def _gradient_power_mean(a, p):
    """Gradient of p-mean of autocorrelation w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    w = _power_mean_grad(c, p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _smooth_max_grad(a, beta):
    """Gradient using smooth max (softmax) approximation."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _peak_shave(a, deadline, max_iter=5000):
    """Direct peak shaving: target the maximum correlation peak specifically."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    lr = 0.005
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_max = int(np.argmax(c))

        # grad[i] = 2 * a[k_max - i] if 0 <= k_max - i <= n-1
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Also target the second and third peaks
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) >= 3:
            threshold = c_sorted[2]
            peak_indices = np.where(c >= threshold * 0.9)[0]
            for kp in peak_indices[:5]:
                for i in range(n):
                    j = kp - i
                    if 0 <= j < n:
                        grad[i] += 0.5 * a[j]

        # Subtract mean to preserve sum
        grad = grad - grad.mean()

        # Line search
        best_t = lr
        best_sc = np.inf
        for t in [lr * 0.5, lr, lr * 2]:
            a_trial = a - t * grad
            a_trial = _normalize(a_trial)
            sc = float(2.0 * n * _fft_convolve(a_trial).max())
            if sc < best_sc:
                best_sc = sc
                best_t = t

        a = _normalize(a - best_t * grad)
        sc = float(2.0 * n * _fft_convolve(a).max())

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.0005, lr * 0.9)
            if no_imp > 1000:
                break

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1=1.0, a2=0.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _adam_polish(a, deadline, max_iter=20000, lr=0.03):
    """Adam optimization in logit space with power-mean objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    p_mean = 4.0
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w = _power_mean_grad(c, p_mean)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fftconvolve(a) if False else _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.95)

        if it % 200 == 0:
            p_mean = min(30.0, p_mean * 1.05)

        if no_imp > max(3000, 5 * n):
            rng = np.random.default_rng(hash(z.tobytes()) % (2**31))
            z = 0.7 * best_z + 0.3 * rng.normal(size=n) * 0.5
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            p_mean = 4.0
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid LP + Gradient + Peak-Shaving optimization.
    
    Phases:
    1. Initial LP refinement on parent (35% time)
    2. Gradient polish with power mean (15% time)
    3. Try multiple lengths with LP (25% time)
    4. Adam polish on best (15% time)
    5. Final combined refinement (10% time)
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is None:
        parent = _make_linear_profile(512, 1.5)
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Generated parent: score={best_score:.10f}")

    n_parent = parent.size

    # Phase 1: LP refinement on parent length
    phase1_end = t_start + (deadline - t_start) * 0.35
    print(f"\nPhase 1: LP refinement on n={n_parent}, budget={phase1_end - time.monotonic():.0f}s")

    a1 = parent.copy()
    a1_lp, s1_lp = _lp_refine_v2(a1, phase1_end, max_iter=50000,
                                  inflate=1.001, blend=0.35, use_line_search=True)
    print(f"  LP score: {s1_lp:.10f}")
    if s1_lp < best_score:
        best_score = s1_lp
        best_seq = a1_lp.copy()

    # Quick peak shave
    ps_end = min(t_start + (deadline - t_start) * 0.40, deadline - 60)
    if time.monotonic() < ps_end:
        a1_ps, s1_ps = _peak_shave(a1_lp, ps_end, max_iter=3000)
        print(f"  Peak shave score: {s1_ps:.10f}")
        if s1_ps < best_score:
            best_score = s1_ps
            best_seq = a1_ps.copy()

    # Phase 2: Gradient polish
    phase2_end = t_start + (deadline - t_start) * 0.55
    print(f"\nPhase 2: Gradient polish, budget={phase2_end - time.monotonic():.0f}s")

    if time.monotonic() < phase2_end:
        a2, s2 = _gradient_polish_v2(best_seq, phase2_end, max_iter=20000,
                                     method='power_mean')
        print(f"  Power-mean gradient score: {s2:.10f}")
        if s2 < best_score:
            best_score = s2
            best_seq = a2.copy()

    # Phase 3: Try other lengths
    phase3_end = t_start + (deadline - t_start) * 0.80
    print(f"\nPhase 3: Multi-length search, budget={phase3_end - time.monotonic():.0f}s")

    other_lengths = []
    for ln in [256, 384, 640, 768, 1024]:
        if ln != n_parent:
            other_lengths.append(ln)

    for n_t in other_lengths:
        if time.monotonic() > phase3_end - 30:
            break

        a_init = _interpolate_profile(best_seq, n_t)
        sc_init = _score(a_init)
        print(f"  n={n_t}: init score={sc_init:.8f}")

        if sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()

        # LP on this length
        lp_end = min(time.monotonic() + 120, phase3_end - 30)
        if lp_end < time.monotonic() + 5:
            continue

        a_lp, s_lp = _lp_refine_v2(a_init, lp_end, max_iter=10000,
                                    inflate=1.001, blend=0.3, use_line_search=True)
        print(f"  n={n_t}: LP score={s_lp:.10f}")
        if s_lp < best_score:
            best_score = s_lp
            best_seq = a_lp.copy()

        # Quick gradient on this length
        if time.monotonic() < phase3_end - 20:
            gp_end = min(time.monotonic() + 40, phase3_end - 20)
            a_gp, s_gp = _gradient_polish_v2(a_lp, gp_end, max_iter=5000,
                                             method='power_mean')
            if s_gp < best_score:
                best_score = s_gp
                best_seq = a_gp.copy()
            print(f"  n={n_t}: grad score={s_gp:.10f}")

    # Phase 4: Adam polish on best
    phase4_end = t_start + (deadline - t_start) * 0.92
    print(f"\nPhase 4: Adam polish, budget={phase4_end - time.monotonic():.0f}s")

    if time.monotonic() < phase4_end:
        a_ad, s_ad = _adam_polish(best_seq, phase4_end, max_iter=20000, lr=0.04)
        print(f"  Adam score: {s_ad:.10f}")
        if s_ad < best_score:
            best_score = s_ad
            best_seq = a_ad.copy()

    # Phase 5: Final LP + gradient refinement
    print(f"\nPhase 5: Final refinement, budget={deadline - time.monotonic():.0f}s")

    if time.monotonic() < deadline - 10:
        # One more LP pass
        lp_end = min(time.monotonic() + 60, deadline - 30)
        if lp_end > time.monotonic() + 5:
            a_final_lp, s_final_lp = _lp_refine_v2(best_seq, lp_end, max_iter=5000,
                                                    inflate=1.0005, blend=0.2,
                                                    use_line_search=True)
            if s_final_lp < best_score:
                best_score = s_final_lp
                best_seq = a_final_lp.copy()
            print(f"  Final LP score: {s_final_lp:.10f}")

    # Final gradient polish
    if time.monotonic() < deadline - 5:
        gp_end = deadline - 5
        a_final_gp, s_final_gp = _gradient_polish_v2(best_seq, gp_end, max_iter=10000,
                                                     method='smooth_max')
        if s_final_gp < best_score:
            best_score = s_final_gp
            best_seq = a_final_gp.copy()
        print(f"  Final gradient score: {s_final_gp:.10f}")

    # Cleanup and return
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
