def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())

    return a / s


def make_power_profile(n, p):

    s = float(a.sum())
    return a / s


def make_ramp_profile(n, asym):
    t = np.linspace(0.0, 1.0, n)
    a = 0.1 + asym * t
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def make_bump_profile(n, center, width, height):
    t = np.linspace(0.0, 1.0, n)
    a = 0.05 + height * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def make_two_block_profile(n, k_frac, ratio):
    k = max(1, int(n * k_frac))
    a = np.zeros(n)
    a[:k] = 1.0
    a[k:] = ratio
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def make_deterministic_profiles(n):
    profs = []
    for eta in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0):
        profs.append(make_ramp_profile(n, eta))
    for eta in (1.0, 2.0, 3.0, 4.0, 6.0, 8.0):
        profs.append(make_exp_profile(n, eta))
    for p in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        profs.append(make_power_profile(n, p))
    for center, width, height in [(0.3, 0.15, 2.0), (0.4, 0.2, 3.0), (0.6, 0.2, 3.0),
                                   (0.5, 0.1, 5.0), (0.7, 0.25, 2.0)]:
        profs.append(make_bump_profile(n, center, width, height))
    for kf, r in [(0.3, 0.3), (0.4, 0.2), (0.5, 0.1), (0.3, 0.5), (0.6, 0.4)]:
        profs.append(make_two_block_profile(n, kf, r))
    return profs


def solve_lp_direction(f, M, delta=1e-10):
    """
    Solve LP: max sum(g) subject to (f*g)_k <= M for all k, g >= delta.
    Returns the normalized direction g/sum(g) or None if LP fails.
    """
    n = f.size
    # Build constraint matrix A: A[k,j] = f[k-j] if 0 <= k-j < n else 0
    # A has shape (2n-1, n)
    # For efficiency, build using the Toeplitz structure
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = f[k - j_end:k - j_start + 1][::-1]

    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(delta, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

    if res.status != 0:
        return None

    g = res.x
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-30:
        return None
    return g / s


def lp_optimize(f0, deadline, max_iters=500, blend=0.7, delta=1e-10):
    """
    LP-based iterative optimization.
    At each step: solve LP to get direction, blend with current f.
    """
    f = np.asarray(f0, dtype=np.float64).ravel().copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    best_f = f.copy()
    best_score = internal_score(f)
    no_improve = 0
    blend_curr = blend

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = full_convolve_fft(f)
        M = float(c.max())
        if not np.isfinite(M) or M <= 0:
            break

        g_dir = solve_lp_direction(f, M, delta=delta)
        if g_dir is None:
            # Fallback: try with larger delta
            g_dir = solve_lp_direction(f, M, delta=1e-8)
            if g_dir is None:
                break

        # Blend
        f_new = (1.0 - blend_curr) * f + blend_curr * g_dir
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        sc_new = internal_score(f_new)

        if np.isfinite(sc_new) and sc_new < best_score - 1e-12:
            f = f_new
            best_f = f.copy()
            best_score = sc_new
            no_improve = 0
        else:
            # Try smaller blend
            for t in (0.5, 0.3, 0.1):
                f_try = (1.0 - t) * f + t * g_dir
                f_try = np.maximum(f_try, 1e-15)
                f_try = f_try / f_try.sum()
                sc_try = internal_score(f_try)
                if np.isfinite(sc_try) and sc_try < best_score - 1e-12:
                    f = f_try
                    best_f = f.copy()
                    best_score = sc_try
                    no_improve = 0
                    blend_curr = t
                    break
            else:
                no_improve += 1
                blend_curr = max(0.1, blend_curr * 0.7)
                if no_improve > 50:
                    break

    return best_f, best_score


def soft_max_adam(p0, deadline, rng, lr=0.10, reg=1e-6, beta_init_frac=0.2):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 60 == 0:
            beta = max(1e-14, beta * 0.8)

        if no_imp > max(200, 3 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def coordinate_descent_polish(p, deadline, rng, step_frac=0.08):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0

    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        improved_this_sweep = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            candidates = [orig * f for f in [1.15, 1.08, 1.04, 0.96, 0.92, 0.85, 1.3, 0.7]]
            best_val = orig
            best_sc = best_score
            for new_val in candidates:
                if new_val < 1e-15 or new_val > 100.0:
                    continue
                p[i] = new_val
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                sc = internal_score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score - 1e-13:
                best_score = best_sc
                best_p = p.copy()
                improved_this_sweep = True
                no_imp = 0
            else:
                no_imp += 1
        if improved_this_sweep:
            step_frac = max(0.01, step_frac * 0.92)
        if no_imp > 3 * n:
            break

    return best_p, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def mirror_descent_step(f, deadline_chunk, n_steps=100, lr_init=0.5):
    """Mirror descent (multiplicative weights) on the simplex."""
    f = np.asarray(f, dtype=np.float64).ravel().copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size
    best_f = f.copy()
    best_score = internal_score(f)
    lr = lr_init

    for step in range(n_steps):
        if time.monotonic() > deadline_chunk:
            break
        c = full_convolve_fft(f)
        M = float(c.max())
        # Gradient of 2*n*M at the max point (use softmax for smoothness)
        beta = 0.1 * M
        L, w = soft_mx(c, beta)
        gp = 2.0 * n * corr_weights(w, f)
        # Mirrored gradient: g_i / f_i
        mg = gp / (f + 1e-15)
        mg -= float(np.mean(mg))
        # Update in log space
        f = f * np.exp(-lr * mg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        sc = internal_score(f)
        if np.isfinite(sc) and sc < best_score:
            best_f = f.copy()
            best_score = sc
        else:
            lr = max(0.01, lr * 0.9)
    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase LP-guided optimization with gradient polishing.
    Uses alternating LP to find descent directions, then polishes with
    softmax-Adam and coordinate descent at multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []

    def local_update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            if label:
                print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} [{label}]")
        if len(candidates) < 25:
            candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
            candidates.sort(key=lambda x: x[0])
        else:
            if float(sc) < candidates[-1][0]:
                candidates[-1] = (float(sc), np.asarray(p, dtype=np.float64).copy())
                candidates.sort(key=lambda x: x[0])

    # Load parent solution
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_nonneg(parent)
    except Exception:
        parent = None

    # Phase 0: Initialization
    print("Phase 0: Initialization")
    if parent is not None:
        sc = internal_score(parent)
        local_update(parent, sc, "parent")
        print(f"  Parent score: {sc:.10f}, n={parent.size}")

    # Quick profiles at n=512
    n_main = 512
    init_profiles = make_deterministic_profiles(n_main)
    for p0 in init_profiles[:10]:
        sc = internal_score(p0)
        local_update(p0, sc, "init")

    print(f"  After init: best={best_score:.10f}")

    # Phase 1: LP optimization from parent and best profiles
    print("Phase 1: LP optimization")
    phase1_end = min(deadline, t_start + 500.0)
    lp_starts = []

    if parent is not None:
        lp_starts.append(parent)
    for p0 in init_profiles[:6]:
        lp_starts.append(p0)

    # Also try at n=256
    n_alt = 256
    for p0 in make_deterministic_profiles(n_alt)[:4]:
        lp_starts.append(p0)

    for si, p0 in enumerate(lp_starts):
        if time.monotonic() > phase1_end:
            break
        per_time = min(phase1_end, time.monotonic() + 30.0)
        f_best, sc = lp_optimize(p0, per_time, max_iters=200, blend=0.6, delta=1e-8)
        sc_full = internal_score(f_best)
        local_update(f_best, sc_full, f"lp_{si}_n{f_best.size}")

        # Follow up with mirror descent
        if time.monotonic() < phase1_end:
            per_time2 = min(phase1_end, time.monotonic() + 10.0)
            f_md, sc_md = mirror_descent_step(f_best, per_time2, n_steps=200, lr_init=0.3)
            sc_md_full = internal_score(f_md)
            local_update(f_md, sc_md_full, f"md_{si}")

    print(f"  After phase 1: best={best_score:.10f}")

    # Phase 2: Softmax-Adam polishing of top candidates
    print("Phase 2: Softmax-Adam polishing")
    phase2_end = min(deadline, t_start + 900.0)
    top_k = min(15, len(candidates))
    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        sc_c, p_c = candidates[i]
        per_time = min(phase2_end, time.monotonic() + 20.0)
        p_best, sc = soft_max_adam(p_c, per_time, rng, lr=0.08, beta_init_frac=0.15)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"adam_{i}")

    print(f"  After phase 2: best={best_score:.10f}")

    # Phase 3: Try larger n (1024)
    print("Phase 3: Larger length optimization (n=1024)")
    phase3_end = min(deadline, t_start + 1300.0)
    n_large = 1024
    if time.monotonic() < phase3_end:
        # Interpolate top candidates to n=1024
        for i in range(min(5, len(candidates))):
            if time.monotonic() > phase3_end:
                break
            sc_c, p_c = candidates[i]
            p_interp = interpolate_profile(p_c, n_large)
            if p_interp is None:
                continue
            # LP at n=1024
            per_time = min(phase3_end, time.monotonic() + 25.0)
            f_best, sc = lp_optimize(p_interp, per_time, max_iters=100, blend=0.5, delta=1e-8)
            sc_full = internal_score(f_best)
            local_update(f_best, sc_full, f"lp_n1024_{i}")
            # Adam polish
            if time.monotonic() < phase3_end:
                per_time2 = min(phase3_end, time.monotonic() + 15.0)
                p_best, sc = soft_max_adam(f_best, per_time2, rng, lr=0.06)
                sc_full = internal_score(p_best)
                local_update(p_best, sc_full, f"adam_n1024_{i}")

        # Also try asymmetric profiles directly at n=1024
        for p0 in make_ramp_profile(n_large, 3.0), make_exp_profile(n_large, 4.0), make_power_profile(n_large, 2.0):
            if time.monotonic() > phase3_end:
                break
            per_time = min(phase3_end, time.monotonic() + 20.0)
            f_best, sc = lp_optimize(p0, per_time, max_iters=100, blend=0.5, delta=1e-8)
            sc_full = internal_score(f_best)
            local_update(f_best, sc_full, f"lp_n1024_det")

    print(f"  After phase 3: best={best_score:.10f}")

    # Phase 4: Extended LP + Adam on absolute best
    print("Phase 4: Extended optimization of top candidates")
    phase4_end = min(deadline, t_start + 1600.0)
    for i in range(min(8, len(candidates))):
        if time.monotonic() > phase4_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size <= 512:
            # LP polish
            per_time = min(phase4_end, time.monotonic() + 20.0)
            f_best, sc = lp_optimize(p_c, per_time, max_iters=150, blend=0.4, delta=1e-9)
            sc_full = internal_score(f_best)
            local_update(f_best, sc_full, f"lp_ext_{i}")
            # Adam
            if time.monotonic() < phase4_end:
                per_time2 = min(phase4_end, time.monotonic() + 15.0)
                p_best, sc = soft_max_adam(f_best, per_time2, rng, lr=0.05, beta_init_frac=0.08)
                sc_full = internal_score(p_best)
                local_update(p_best, sc_full, f"adam_ext_{i}")

    print(f"  After phase 4: best={best_score:.10f}")

    # Phase 5: Coordinate descent polish
    print("Phase 5: Coordinate descent polish")
    phase5_end = min(deadline, t_start + 1750.0)
    for i in range(min(5, len(candidates))):
        if time.monotonic() > phase5_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size <= 1024:
            per_time = min(phase5_end, time.monotonic() + 20.0)
            p_best, sc = coordinate_descent_polish(p_c, per_time, rng, step_frac=0.06)
            sc_full = internal_score(p_best)
            local_update(p_best, sc_full, f"cd_{i}")

    print(f"  After phase 5: best={best_score:.10f}")

    # Phase 6: Final selection with exact scoring
    print("Phase 6: Final selection")
    final_seq = best_seq
    final_score = best_score

    if final_seq is not None:
        if final_seq.size <= 2048:
            sc = true_score(final_seq)
            if np.isfinite(sc):
                final_score = sc

        for sc_c, p_c in candidates:
            if p_c.size <= 2048:
                sc_cx = true_score(p_c)
                if np.isfinite(sc_cx) and sc_cx < final_score:
                    final_score = sc_cx
                    final_seq = p_c

        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                final_score = min(final_score, sc_eval)
        except Exception:
            pass

        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s

        print(f"  FINAL: score={final_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = make_ramp_profile(512, 2.0)
    sc_fb = internal_score(fallback)
    print(f"  FALLBACK: score={sc_fb:.10f}")
    return [float(x) for x in fallback]
