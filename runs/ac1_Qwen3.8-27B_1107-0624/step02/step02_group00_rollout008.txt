The LP is: maximize sum(b) subject to Conv(a) * b <= M (element-wise) and b >= 0.

The dual is: minimize M * sum(y) subject to Conv(a)^T * y >= 1 (element-wise) and y >= 0.

Another idea: **Iterative reweighting**. Instead of solving the full LP, do coordinate descent on the LP:
- For each coordinate b_i, update it to its optimal value given the other coordinates
- This is much faster than a full LP solve

- Start with b = 0
- At each step, find the coordinate b_i that can be increased the most (given current constraints)
- Increase it until a constraint becomes tight
- Repeat

def _score(a):

def _project_simplex(v):

    n = len(v)

        return v

    cssv = np.cumsum(u)

def _make_conv_matrix(a):

        A[j:j + n, j] = a
    return A


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c


def _make_linear_profile(n, eta=1.5):

    a = 1.0 + eta * (t - 0.5)

    return a / a.sum()

    a = np.maximum(a, 1e-12)

    a = 1.0 + a1 * u + a3 * u ** 3

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:

        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:

    return p / s


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _block_swap_optimize(a, deadline, max_iter=5000):
    """Try random block swaps to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(12345)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Random block swap
        block_size = rng.integers(max(2, n // 32), max(3, n // 8))
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if abs(i - j) < block_size:
            continue

        a_new = a.copy()
        a_new[i:i + block_size], a_new[j:j + block_size] = \
            a_new[j:j + block_size].copy(), a_new[i:i + block_size].copy()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_score - 1e-10:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break

    return best_a


def _peak_aware_gradient(a, beta, top_k=5):
    """Gradient that emphasizes the top-k peaks more."""
    n = len(a)
    c = _fft_convolve(a)
    # Combine uniform weight with peak-focused weight
    w_uniform = _smooth_max_weight(c, beta)
    # Focus on top-k peaks
    sorted_idx = np.argsort(c)[::-1]
    w_peak = np.zeros_like(c)
    peak_vals = c[sorted_idx[:top_k]]
    w_peak[sorted_idx[:top_k]] = peak_vals / peak_vals.sum()
    w = 0.7 * w_uniform + 0.3 * w_peak
    w = w / w.sum()

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _peak_aware_polish(a, deadline, max_iter=15000):
    """Gradient polish with peak-aware weighting."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.05 * float(c.max()))
    lr = 0.025
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        grad, maxc = _peak_aware_gradient(a_nest, beta, top_k=5)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.96)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a


def _coordinate_descent_polish(a, deadline, max_iter=50000):
    """Try individual coordinate adjustments."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(777)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Pick a random coordinate to modify
        i = rng.integers(0, n)
        # Try moving mass from i to a random other coordinate
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try several amounts of transfer
        best_local = None
        best_local_score = best_score
        for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
            delta = frac * a[i]
            if delta > a[i]:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_local_score - 1e-12:
                best_local_score = sc
                best_local = a_try.copy()

        if best_local is not None:
            a = best_local
            if best_local_score < best_score - 1e-12:
                best_score = best_local_score
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 5000:
            break

    return best_a


def _full_pipeline_for_length(a_init, n, deadline, phase_budget):
    """Run the full optimization pipeline for a given initial sequence and length."""
    a = _interpolate_profile(a_init, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_linear_profile(n, 1.5)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    # Phase A: LP refinement
    lp_budget = min(phase_budget * 0.4, deadline - time.monotonic() - 60.0)
    if lp_budget > 5:
        lp_end = time.monotonic() + lp_budget
        a_lp = _lp_refine(a, lp_end, max_iter=15000, inflate=1.0015, blend=0.3)
        sc_lp = _score(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()
        print(f"    LP: {sc_lp:.10f}")

    # Phase B: Gradient polish
    gp_budget = min(phase_budget * 0.2, deadline - time.monotonic() - 40.0)
    if gp_budget > 5:
        gp_end = time.monotonic() + gp_budget
        a_gp = _gradient_polish(best_a, gp_end, max_iter=10000)
        sc_gp = _score(a_gp)
        if sc_gp < best_score:
            best_score = sc_gp
            best_a = a_gp.copy()
        print(f"    Gradient: {sc_gp:.10f}")

    # Phase C: Block swap
    bs_budget = min(phase_budget * 0.1, deadline - time.monotonic() - 30.0)
    if bs_budget > 3:
        bs_end = time.monotonic() + bs_budget
        a_bs = _block_swap_optimize(best_a, bs_end, max_iter=3000)
        sc_bs = _score(a_bs)
        if sc_bs < best_score:
            best_score = sc_bs
            best_a = a_bs.copy()
        print(f"    BlockSwap: {sc_bs:.10f}")

    # Phase D: Adam polish
    ad_budget = min(phase_budget * 0.15, deadline - time.monotonic() - 20.0)
    if ad_budget > 5:
        ad_end = time.monotonic() + ad_budget
        a_ad = _logit_adam_polish(best_a, ad_end, max_iter=10000, lr=0.04)
        sc_ad = _score(a_ad)
        if sc_ad < best_score:
            best_score = sc_ad
            best_a = a_ad.copy()
        print(f"    Adam: {sc_ad:.10f}")

    # Phase E: Peak-aware gradient
    pa_budget = min(phase_budget * 0.1, deadline - time.monotonic() - 15.0)
    if pa_budget > 5:
        pa_end = time.monotonic() + pa_budget
        a_pa = _peak_aware_polish(best_a, pa_end, max_iter=8000)
        sc_pa = _score(a_pa)
        if sc_pa < best_score:
            best_score = sc_pa
            best_a = a_pa.copy()
        print(f"    PeakAware: {sc_pa:.10f}")

    # Phase F: Brief LP
    lp2_budget = min(phase_budget * 0.1, deadline - time.monotonic() - 10.0)
    if lp2_budget > 5:
        lp2_end = time.monotonic() + lp2_budget
        a_lp2 = _lp_refine(best_a, lp2_end, max_iter=5000, inflate=1.0008, blend=0.2)
        sc_lp2 = _score(a_lp2)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_a = a_lp2.copy()
        print(f"    LP2: {sc_lp2:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining LP spreading, adaptive gradient,
    block-swap perturbations, peak-aware optimization, and multi-length search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine target lengths - include larger ones for more degrees of freedom
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [384, 512, 640, 768, 1024]:
        if ln not in lengths:
            lengths.append(ln)

    # Sort by expected promise: parent length first, then others
    # Remove duplicates
    lengths = list(dict.fromkeys(lengths))

    print(f"Trying lengths: {lengths}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: For each length, run the full pipeline
    # Allocate time proportional to number of lengths
    num_lengths = len(lengths)
    per_length_budget = (deadline - time.monotonic() - 200.0) / num_lengths

    for n_t in lengths:
        if time.monotonic() > deadline - 150:
            break

        elapsed_frac = (time.monotonic() - t_start) / budget_s
        # Spend less time on later lengths if we're running low
        remaining_frac = max(0.1, 1.0 - elapsed_frac * 1.2)
        phase_budget = per_length_budget * remaining_frac
        phase_budget = max(30.0, min(phase_budget, deadline - time.monotonic() - 100.0))

        if phase_budget < 20:
            continue

        print(f"\nLength {n_t}: budget={phase_budget:.0f}s")

        # Determine initialization
        if parent is not None:
            a_init = _interpolate_profile(parent, n_t)
        else:
            a_init = _make_linear_profile(n_t, 1.5)

        sc_init = _score(a_init)
        if np.isfinite(sc_init) and sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()

        a_best_n, sc_best_n = _full_pipeline_for_length(a_init, n_t, deadline, phase_budget)
        if sc_best_n < best_score:
            best_score = sc_best_n
            best_seq = a_best_n.copy()
            print(f"  ** New best at n={n_t}: {sc_best_n:.10f}")

    # Phase 2: Try diverse initializations at the best length
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = best_seq.size
        print(f"\nPhase 2: Diverse inits at n={n_best}")

        inits = []
        # U-shaped profiles
        for alpha in [1.0, 2.0, 3.0, 5.0]:
            inits.append(_make_u_profile(n_best, alpha))
        # Gaussian mixtures (bimodal)
        inits.append(_make_gaussian_mixture(n_best, [0.1, 0.9], [0.15, 0.15], [1.0, 1.0]))
        inits.append(_make_gaussian_mixture(n_best, [0.05, 0.95], [0.1, 0.1], [1.0, 1.0]))
        inits.append(_make_gaussian_mixture(n_best, [0.2, 0.8], [0.2, 0.2], [1.0, 1.0]))
        # Linear with various slopes
        for eta in [0.8, 1.2, 2.0, 3.0]:
            inits.append(_make_linear_profile(n_best, eta))
        # Cubic
        inits.append(_make_cubic_profile(n_best, 2.0, -1.5))
        inits.append(_make_cubic_profile(n_best, 1.5, -0.8))
        # Random structured
        for _ in range(3):
            a_rand = np.maximum(rng.normal(1.0, 0.5, n_best), 0.0)
            a_rand = a_rand / a_rand.sum()
            inits.append(a_rand)

        # Perturbed best
        for _ in range(3):
            a_pert = best_seq * (1.0 + 0.15 * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            inits.append(a_pert)

        inits_per_budget = max(1, int((deadline - time.monotonic() - 50.0) / 60.0))
        inits = inits[:min(len(inits), inits_per_budget)]

        for a0 in inits:
            if time.monotonic() > deadline - 50:
                break
            sc0 = _score(a0)
            print(f"  Init score={sc0:.8f}")
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Quick LP + gradient
            rem = deadline - time.monotonic() - 40.0
            if rem < 15:
                break
            sub_budget = min(45.0, rem * 0.5)
            a_res, sc_res = _full_pipeline_for_length(a0, n_best, deadline, sub_budget)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** New best: {sc_res:.10f}")

    # Phase 3: Final intensive polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        final_budget = deadline - time.monotonic() - 20.0
        print(f"\nPhase 3: Final polish, budget={final_budget:.0f}s")

        # Sub-phase 3a: LP
        lp_b = min(final_budget * 0.3, 120.0)
        if lp_b > 10:
            a_lp = _lp_refine(best_seq, time.monotonic() + lp_b,
                             max_iter=20000, inflate=1.0008, blend=0.25)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"  Final LP: {sc:.10f}")

        # Sub-phase 3b: Adam
        ad_b = min(final_budget * 0.25, 100.0)
        if ad_b > 10:
            a_ad = _logit_adam_polish(best_seq, time.monotonic() + ad_b,
                                      max_iter=15000, lr=0.03)
            sc = _score(a_ad)
            if sc < best_score:
                best_score = sc
                best_seq = a_ad.copy()
            print(f"  Final Adam: {sc:.10f}")

        # Sub-phase 3c: Peak-aware
        pa_b = min(final_budget * 0.2, 80.0)
        if pa_b > 10:
            a_pa = _peak_aware_polish(best_seq, time.monotonic() + pa_b, max_iter=10000)
            sc = _score(a_pa)
            if sc < best_score:
                best_score = sc
                best_seq = a_pa.copy()
            print(f"  Final PeakAware: {sc:.10f}")

        # Sub-phase 3d: Block swap
        bs_b = min(final_budget * 0.1, 40.0)
        if bs_b > 5:
            a_bs = _block_swap_optimize(best_seq, time.monotonic() + bs_b, max_iter=5000)
            sc = _score(a_bs)
            if sc < best_score:
                best_score = sc
                best_seq = a_bs.copy()
            print(f"  Final BlockSwap: {sc:.10f}")

        # Sub-phase 3e: One more LP
        lp_b2 = min(final_budget * 0.1, 30.0)
        if lp_b2 > 5:
            a_lp2 = _lp_refine(best_seq, time.monotonic() + lp_b2,
                              max_iter=5000, inflate=1.0005, blend=0.15)
            sc = _score(a_lp2)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp2.copy()
            print(f"  Final LP2: {sc:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(512, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
