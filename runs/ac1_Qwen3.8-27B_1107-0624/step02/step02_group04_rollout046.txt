def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):

    if c.size == 0:

def _soft_max_weights(c, beta):

    c = np.asarray(c, dtype=np.float64)

        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * float(np.log(s))

    w[top_idx] = 1.0 / k
    return float(c[top_idx].max()), w

            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new

    return best_a, best_sc


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return np.maximum(p0, 1e-12)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Generate and score diverse initial profiles at multiple lengths.
    Phase 2: Polish top candidates using adaptive logit-space Adam.
    Phase 3: Escalate best to longer lengths and re-optimize.
    Phase 4: Fine-tune with exact subgradient.
    Phase 5: Directed peak reduction.
    Phase 6: Final micro-polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0, budget_s - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={p.size}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if parent_raw.sum() > 0.01:
                parent = parent_raw / parent_raw.sum()
    except Exception:
        parent = None

    print(f"Parent: n={parent.size if parent is not None else 'None'}")

    # === Phase 1: Initial candidates ===
    candidates = []
    n_values = [256, 512, 1024]

    for n in n_values:
        profile_defs = [
            ('hann', 'hann', 1.0),
            ('hamming', 'hamming', 1.0),
            ('blackman', 'blackman', 1.0),
            ('u2_05', 'u2', 0.5),
            ('u2_10', 'u2', 1.0),
            ('u2_15', 'u2', 1.5),
            ('u4_05', 'u4', 0.5),
            ('u4_10', 'u4', 1.0),
            ('bell2_1', 'bell2', 1.0),
            ('bell2_2', 'bell2', 2.0),
            ('bell4_1', 'bell4', 1.0),
            ('lin_05', 'lin', 0.5),
            ('lin_10', 'lin', 1.0),
            ('lin_15', 'lin', 1.5),
            ('cosh_1', 'cosh', 1.0),
            ('cosh_2', 'cosh', 2.0),
            ('sech_2', 'sech', 2.0),
            ('tri', 'tri', 1.0),
            ('raised_cos', 'raised_cos', 1.0),
        ]

        for name, ptype, param in profile_defs:
            p = _make_profile(n, ptype, param)
            sc = _score_normalized(p)
            candidates.append((sc, p, n, name))

        # Parent interpolation
        if parent is not None and parent.size != n:
            p_int = _interpolate_profile(parent, n)
            sc = _score_normalized(p_int)
            candidates.append((sc, p_int, n, 'parent_inter'))
        elif parent is not None and parent.size == n:
            sc = _score_normalized(parent)
            candidates.append((sc, parent.copy(), n, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    for sc, p, n, name in candidates[:5]:
        print(f"    {sc:.10f} n={n} {name}")
        update_best(p, sc, f"init {name} n={n}")

    # === Phase 2: Polish top candidates ===
    phase2_end = min(deadline, t_start + 400.0)
    top_k = 6
    polished = []

    for idx in range(min(top_k, len(candidates))):
        if time.monotonic() > phase2_end - 30:
            break
        sc0, p0, n0, name0 = candidates[idx]
        per_end = min(phase2_end - 10, time.monotonic() + 50.0)
        p_best, sc_best = _optimize(p0, per_end, rng, lr=0.06,
                                     beta_init=0.12, beta_decay=0.96,
                                     k_mode='adaptive')
        update_best(p_best, sc_best, f"polish {name0} n={n0}")
        polished.append((sc_best, p_best, n0, name0 + "_pol"))

    # Also polish parent directly if it's a good starting point
    if parent is not None and parent.size <= 2048:
        per_end = min(phase2_end, time.monotonic() + 60.0)
        p_best, sc_best = _optimize(parent, per_end, rng, lr=0.05,
                                     beta_init=0.10, beta_decay=0.96,
                                     k_mode='adaptive')
        update_best(p_best, sc_best, f"polish parent n={parent.size}")

    print(f"Phase 2: best={best_score:.10f}")

    # === Phase 3: Length escalation ===
    phase3_end = min(deadline, t_start + 1000.0)
    escalation_lengths = [512, 1024, 2048]

    for target_n in escalation_lengths:
        if time.monotonic() > phase3_end - 120:
            break
        if best_seq is None:
            continue

        if best_seq.size == target_n:
            p_curr = best_seq.copy()
        else:
            p_curr = _interpolate_profile(best_seq, target_n)

        sc_interp = _score_normalized(p_curr)
        print(f"  Escalating to n={target_n}: score={sc_interp:.10f}")
        update_best(p_curr, sc_interp, f"interp n={target_n}")

        # Optimize at new length
        per_end_1 = min(phase3_end - 60, time.monotonic() + 80.0)
        p_b1, sc_b1 = _optimize(p_curr, per_end_1, rng, lr=0.05,
                                 beta_init=0.10, beta_decay=0.95,
                                 k_mode='adaptive')
        update_best(p_b1, sc_b1, f"esc_opt1 n={target_n}")

        if time.monotonic() < phase3_end - 40:
            per_end_2 = min(phase3_end - 30, time.monotonic() + 60.0)
            p_b2, sc_b2 = _optimize(p_b1, per_end_2, rng, lr=0.03,
                                     beta_init=0.06, beta_decay=0.96,
                                     k_mode='adaptive')
            update_best(p_b2, sc_b2, f"esc_opt2 n={target_n}")

    print(f"Phase 3: best={best_score:.10f} (n={best_seq.size if best_seq is not None else 0})")

    # === Phase 4: Exact subgradient fine-tuning ===
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        per_end = min(phase4_end, time.monotonic() + 200.0)
        p_fin, sc_fin = _optimize_exact_subgrad(best_seq, per_end, rng,
                                                 lr_init=0.03, lr_decay=0.9995)
        update_best(p_fin, sc_fin, "exact_subgrad")

    # === Phase 5: Directed peak reduction ===
    phase5_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        per_end = min(phase5_end, time.monotonic() + 150.0)
        p_dir, sc_dir = _directed_peak_reduction(best_seq, per_end, rng, n_iters=50000)
        update_best(p_dir, sc_dir, "directed")

    # === Phase 6: Final micro-polish ===
    phase6_end = deadline
    if time.monotonic() < phase6_end and best_seq is not None:
        remaining = phase6_end - time.monotonic()
        if remaining > 10:
            per_end = time.monotonic() + max(10, remaining * 0.7)
            p_micro, sc_micro = _optimize(best_seq, per_end, rng, lr=0.01,
                                          beta_init=0.03, beta_decay=0.98,
                                          k_mode='adaptive')
            update_best(p_micro, sc_micro, "micro_polish")

    # Multi-restart at the very end if time permits
    if time.monotonic() < deadline - 20 and best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        n_restarts = max(1, int(remaining // 40))
        for i in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            per_end = min(deadline - 15, time.monotonic() + min(35, remaining / max(1, n_restarts - i)))
            z = _logit_of(best_seq)
            noise = 0.15 * (1 + 0.3 * i)
            z = z + noise * rng.normal(size=best_seq.size)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_start = _softmax_of(z)
            p_try, sc_try = _optimize(p_start, per_end, rng, lr=0.04,
                                       beta_init=0.08, beta_decay=0.95)
            update_best(p_try, sc_try, f"restart_{i}")

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(256, 'hann')

        # Try exact score
        if final.size <= 2048:
            sc_exact = _true_score(final)
            print(f"Exact score: {sc_exact:.10f}")
            if np.isfinite(sc_exact) and sc_exact < best_score:
                best_score = sc_exact

        # Try evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
            if np.isfinite(sc_eval) and sc_eval < best_score:
                best_score = sc_eval
        except Exception:
            pass

        print(f"FINAL: score={best_score:.10f}, n={final.size}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, 'hann')
    print(f"FALLBACK: n=256")
    return [float(x) for x in fallback]
