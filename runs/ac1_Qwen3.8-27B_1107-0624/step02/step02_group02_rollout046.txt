    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -100, 100))
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def conv_gradient(w, a):
    """Gradient of sum_l w[l]*conv(a,a)[l] w.r.t. a.
    
    conv(a,a)[l] = sum_j a[j]*a[l-j]
    d/d(a[j]) = a[l-j] (if 0 <= l-j < n)
    
    So grad[j] = sum_l w[l] * a[l-j] = (w cross-correlate a)[j]
    """
    n = a.size
    N = next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # Cross-correlation: ifft(conj(fw) * fa)
    h = np.fft.irfft(np.conj(fw) * fa, N)
    return h[:n]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def gradient_optimize(p0, deadline, rng, lr=0.05, beta_factor=0.15,
                      restart_freq=400, verbose=False, label=""):
    """Logit-space gradient descent minimizing max(conv(a,a))."""
    a = np.maximum(p0.copy(), 1e-12)
    a = a / a.sum()
    n = a.size
    z = np.log(a)
    z -= z.mean()

    c = full_convolve_fft(a)
    maxc = float(c.max())
    beta = max(1e-8, beta_factor * maxc)

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    cur_lr = lr

    best_a = a.copy()
    best_score = 2.0 * n * maxc
    no_improve = 0

    while time.monotonic() < deadline:
        t += 1
        val, w = soft_max_and_weights(c, beta)

        # Gradient of soft_max(conv(a,a)) w.r.t. a
        grad_a = conv_gradient(w, a) * 2.0  # factor of 2 from convolution symmetry

        # Gradient in logit space
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        # Regularization
        grad_z += 1e-8 * a

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_a)
            z -= z.mean()
            m1.fill(0.0)
            m2.fill(0.0)
            t = 0
            continue

        # Adam
        b1, b2, eps = 0.9, 0.999, 1e-8
        m1 = b1 * m1 + (1 - b1) * grad_z
        m2 = b2 * m2 + (1 - b2) * grad_z * grad_z
        m1_hat = m1 / (1 - b1 ** t)
        m2_hat = m2 / (1 - b2 ** t)

        z = z - cur_lr * m1_hat / (np.sqrt(m2_hat) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))

        a = softmax_from_z(z)
        c = full_convolve_fft(a)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score) and score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal beta
        if t % 30 == 0:
            beta = max(1e-15, beta * 0.82)

        # Adaptive learning rate
        if no_improve > restart_freq // 3:
            cur_lr = max(0.003, cur_lr * 0.97)

        # Restart with perturbation
        if no_improve > restart_freq:
            # Choose perturbation strategy
            strategy = rng.integers(0, 4)
            if strategy == 0:
                # Small noise in logit space
                z = np.log(best_a) + 0.2 * rng.normal(size=n)
            elif strategy == 1:
                # Medium noise
                z = np.log(best_a) + 0.5 * rng.normal(size=n)
            elif strategy == 2:
                # Blend with a smoothed version
                from scipy.ndimage import uniform_filter1d
                smooth_a = uniform_filter1d(best_a, size=max(3, n // 32))
                smooth_a = np.maximum(smooth_a, 1e-12)
                z = np.log(0.7 * best_a + 0.3 * smooth_a / smooth_a.sum())
            else:
                # Blend with a different profile
                t_profile = np.linspace(0, 1, n)
                alt = 1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(1, 5) * t_profile)
                alt = np.maximum(alt, 1e-12)
                alt = alt / alt.sum()
                z = np.log(0.6 * best_a + 0.4 * alt)

            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            a = softmax_from_z(z)
            c = full_convolve_fft(a)
            maxc = float(c.max())
            beta = max(1e-8, 0.15 * maxc)
            cur_lr = lr
            m1.fill(0.0)
            m2.fill(0.0)
            t = 0
            no_improve = 0

        if verbose and t % 500 == 0:
            print(f"    {label} grad t={t}, score={best_score:.10f}, beta={beta:.2e}, lr={cur_lr:.4f}")

    return best_a, best_score


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M, max_iter=5000):
    """LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={"max_iter": max_iter})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine(f_init, num_steps, rng):
    """Brief LP refinement. Returns improved sequence."""
    f = np.maximum(f_init.copy(), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_M = float(np.convolve(f, f).max())
    t_blend = 0.3

    for step in range(num_steps):
        M = best_M
        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()

        f_new = (1 - t_blend) * f + t_blend * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        M_new = float(np.convolve(f_new, f_new).max())

        if M_new <= best_M + 1e-12:
            f = f_new
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
        else:
            t_blend *= 0.5
            if t_blend < 0.01:
                break

        t_blend = max(0.05, t_blend * 0.98)

    return best_f


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_triangle_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(2 * t - 1)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_parabola_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - (2 * t - 1) ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_bell_profile(n, width=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2 * width ** 2))
    a = a + 0.1 * np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_mixed_profile(n, rng):
    t = np.linspace(0.0, 1.0, n)
    k1 = rng.integers(1, 6)
    k2 = rng.integers(1, 10)
    amp1 = rng.uniform(0.1, 0.5)
    amp2 = rng.uniform(0.05, 0.3)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + amp1 * np.cos(2 * np.pi * k1 * t + phase) + amp2 * np.cos(2 * np.pi * k2 * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Cubic interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x0, p0)
        result = cs(x1)
        result = np.maximum(result, 0.0)
        s = result.sum()
        return result / s if s > 1e-10 else np.ones(n_new) / n_new
    except Exception:
        return np.interp(x1, x0, p0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses gradient descent with soft-max, adaptive restarts, and LP refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting search. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    n_main = 512

    # ============ Phase 1: Gradient optimization at n=512 from init ============
    if init_seq is not None and init_seq.size != n_main:
        p_main = interpolate_profile(init_seq, n_main)
    elif init_seq is not None:
        p_main = init_seq.copy()
    else:
        p_main = make_linear_profile(n_main, 1.5)

    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1 start: n={n_main}, score={sc:.10f}")

    # Main gradient run
    p1_deadline = min(deadline, t_start + 300.0)
    result, sc = gradient_optimize(p_main, p1_deadline, rng, lr=0.06, verbose=True, label="main")
    update(result, sc)
    print(f"After main gradient: score={sc:.10f}")

    # ============ Phase 2: Other starting profiles ============
    p2_end = min(deadline, t_start + 550.0)
    profiles = [
        ("lin1.0", make_linear_profile(n_main, 1.0)),
        ("lin2.0", make_linear_profile(n_main, 2.0)),
        ("lin3.0", make_linear_profile(n_main, 3.0)),
        ("cos1", make_cosine_profile(n_main, 1, 0.3)),
        ("cos2", make_cosine_profile(n_main, 2, 0.2)),
        ("tri", make_triangle_profile(n_main)),
        ("par", make_parabola_profile(n_main)),
        ("bell", make_bell_profile(n_main, 0.35)),
        ("bell2", make_bell_profile(n_main, 0.5)),
    ]
    # Add random profiles
    for _ in range(3):
        profiles.append((f"rand{len(profiles)}", make_mixed_profile(n_main, rng)))

    for name, p in profiles:
        if time.monotonic() > p2_end - 10:
            break
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.08:
            continue
        per_deadline = min(p2_end, time.monotonic() + 25.0)
        result, sc = gradient_optimize(p, per_deadline, rng, lr=0.05, label=f"p2_{name}")
        update(result, sc)
        if sc < best_score + 0.02:
            print(f"  Phase2 {name}: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}, n={best_seq.size if best_seq is not None else '?'}")

    # ============ Phase 3: Continued refinement with different params ============
    p3_end = min(deadline, t_start + 850.0)
    if best_seq is not None:
        # Try with different learning rates and restart strategies
        for lr_try in [0.04, 0.08, 0.03]:
            if time.monotonic() > p3_end - 10:
                break
            p0 = best_seq.copy() + 0.002 * rng.normal(size=best_seq.size)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            per_deadline = min(p3_end, time.monotonic() + 80.0)
            result, sc = gradient_optimize(p0, per_deadline, rng, lr=lr_try,
                                           beta_factor=0.2, restart_freq=300,
                                           label=f"p3_lr{lr_try}")
            update(result, sc)
            if sc < best_score + 0.01:
                print(f"  Phase3 lr={lr_try}: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Try n=1024 ============
    n_large = 1024
    p4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < p4_end - 30 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 4: n={n_large}, interpolated score={sc:.10f}")

        # Gradient optimize at n=1024
        per_deadline = min(p4_end, time.monotonic() + 200.0)
        result, sc = gradient_optimize(p_large, per_deadline, rng, lr=0.04,
                                       beta_factor=0.12, restart_freq=500,
                                       verbose=True, label="n1024")
        update(result, sc)
        print(f"After n=1024 gradient: score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}, n={best_seq.size if best_seq is not None else '?'}")

    # ============ Phase 5: LP refinement on best ============
    p5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < p5_end - 30 and best_seq is not None:
        n_lp = best_seq.size
        if n_lp <= 1024:
            n_lp_steps = 20 if n_lp <= 512 else 10
            lp_deadline = min(p5_end, time.monotonic() + 150.0)

            # Do LP refinement in chunks to respect deadline
            f_cur = best_seq.copy()
            for round_idx in range(n_lp_steps):
                if time.monotonic() > lp_deadline:
                    break
                # Do a few LP steps
                f_cur = f_cur / f_cur.sum()
                M_cur = float(np.convolve(f_cur, f_cur).max())
                g = lp_step(f_cur, M_cur)
                if g is None or g.sum() < 1e-10:
                    break
                g = g / g.sum()
                for blend in [0.3, 0.2, 0.1]:
                    f_new = (1 - blend) * f_cur + blend * g
                    f_new = np.maximum(f_new, 0.0)
                    f_new = f_new / f_new.sum()
                    M_new = float(np.convolve(f_new, f_new).max())
                    if M_new <= M_cur + 1e-12:
                        f_cur = f_new
                        M_cur = M_new
                        break

            sc = compute_score(f_cur)
            update(f_cur, sc)
            print(f"After LP refinement: score={sc:.10f}")

    print(f"After Phase 5: best={best_score:.10f}")

    # ============ Phase 6: Final gradient polish ============
    p6_end = min(deadline, t_start + 1720.0)
    if time.monotonic() < p6_end - 10 and best_seq is not None:
        # Multiple short runs with very small perturbations
        for attempt in range(4):
            if time.monotonic() > p6_end - 5:
                break
            p0 = best_seq.copy()
            noise_scale = 0.001 * (attempt + 1)
            p0 = p0 + noise_scale * rng.normal(size=best_seq.size)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            per_deadline = min(p6_end, time.monotonic() + 40.0)
            result, sc = gradient_optimize(p0, per_deadline, rng, lr=0.02,
                                           beta_factor=0.05, restart_freq=200,
                                           label=f"final{attempt}")
            update(result, sc)
            if sc < best_score + 0.005:
                print(f"  Final polish {attempt}: score={sc:.10f}")

    print(f"After Phase 6: best={best_score:.10f}")

    # ============ Final: try n=768 if beneficial ============
    if time.monotonic() < deadline - 20 and best_seq is not None and best_score > 1.503:
        for n_try in [768, 640]:
            if time.monotonic() > deadline - 20:
                break
            p_try = interpolate_profile(best_seq, n_try)
            sc_try = compute_score(p_try)
            if sc_try < best_score:
                per_deadline = min(deadline, time.monotonic() + 30.0)
                result, sc = gradient_optimize(p_try, per_deadline, rng, lr=0.03,
                                               label=f"n{n_try}")
                update(result, sc)
                print(f"  n={n_try}: score={sc:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
            if eval_score < final_score:
                final_score = eval_score
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
