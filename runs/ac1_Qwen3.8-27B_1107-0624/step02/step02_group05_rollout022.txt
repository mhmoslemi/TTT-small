"""
Hybrid optimization for minimizing 2*n*max(conv(f,f))/sum(f)^2 over non-negative sequences.
Uses projected subgradient descent with exact subgradients, Frank-Wolfe LP steps,
and Adam polishing. Multi-start with diverse profile initializations and multiple lengths.

import numpy as np
from scipy.optimize import linprog
import time
import math


def compute_conv(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)[:2 * n - 1]


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = compute_conv(f)
    return float(2.0 * n * c.max() / (s * s))


def compute_M(f):
    """Compute max of linear convolution of f with itself."""
    return float(compute_conv(f).max())


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s

    max_val = float(conv.max())
    tol = 1e-10 * max_val
    active = np.where(conv >= max_val - tol)[0]
    
    # Subgradient: for each active k, grad[i] = 2*f[k-i] (zero-padded)
    g = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * f[j]
    g /= len(active)
    return g, max_val, active


def projected_subgradient(f, deadline, rng, max_iters=50000, use_momentum=True):
    """
    Projected subgradient descent with momentum.
    Minimizes max_k (f*f)[k] subject to f >= 0, sum(f) = 1.
    """
    n = len(f)
    f = normalize_nonneg(f.copy())
    if f is None:
        return f, np.inf
    
    best_f = f.copy()
    best_M = compute_M(f)
    
    # Momentum buffers
    v = np.zeros(n)
    beta = 0.9
    
    # Learning rate schedule
    lr0 = 0.5
    lr_decay = 0.9995
    
    it = 0
    no_imp = 0
    while time.monotonic() < deadline and it < max_iters:
        conv = compute_conv(f)
        g, M, active = subgradient_of_max_conv(f, conv)
        
        if not np.all(np.isfinite(g)):
            break
        
        lr = lr0 * (lr_decay ** it)
        
        if use_momentum:
            v = beta * v + (1 - beta) * g
            step = v
        else:
            step = g
        
        f_new = f - lr * step
        f_new = normalize_nonneg(f_new)
        if f_new is None:
            f_new = f.copy()
        
        M_new = compute_M(f_new)
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = f_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        f = f_new
        it += 1
        
        # Restart if stuck
        if no_imp > 500:
            f = 0.8 * best_f + 0.2 * normalize_nonneg(best_f + 0.1 * rng.normal(size=n))
            if f is None:
                f = best_f.copy()
            v.fill(0)
            no_imp = 0
            lr0 = max(0.1, lr0 * 0.8)
    
    return best_f, best_M


def make_conv_matrix_rows(f, indices):
    """Build rows of convolution matrix for specific lags."""
    n = len(f)
    A = np.zeros((len(indices), n))
    for row, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def lp_frank_wolfe(f, M, deadline, max_constraints=None):
    """
    Frank-Wolfe step: solve LP to find g maximizing sum(g)
    subject to g >= 0 and (f*g)[k] <= M for all k (or top constraints).
    """
    n = len(f)
    conv = compute_conv(f)
    
    # Use only the most binding constraints for speed
    if max_constraints is not None and 2 * n - 1 > max_constraints:
        threshold = np.sort(conv)[-max_constraints][0]
        indices = np.where(conv >= threshold)[0]
    else:
        indices = np.arange(2 * n - 1)
    
    A_ub = make_conv_matrix_rows(f, indices)
    b_ub = np.full(len(indices), M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_iteration(f, deadline, rng, max_iters=200, use_lp=True, lp_every=10):
    """
    Frank-Wolfe style iteration combining subgradient and LP steps.
    """
    n = len(f)
    f = normalize_nonneg(f.copy())
    if f is None:
        return f, np.inf
    
    best_f = f.copy()
    best_M = compute_M(f)
    no_imp = 0
    
    it = 0
    t_base = 2.0
    while time.monotonic() < deadline and it < max_iters:
        conv = compute_conv(f)
        M = float(conv.max())
        g, M, active = subgradient_of_max_conv(f, conv)
        
        if not np.all(np.isfinite(g)):
            break
        
        if use_lp and (it % lp_every == 0 or no_imp > 50):
            # LP-based FW step
            g_lp = lp_frank_wolfe(f, M, deadline, max_constraints=min(200, 2 * n - 1))
            if g_lp is not None and g_lp.sum() > 1e-15:
                g_use = g_lp / g_lp.sum()
            else:
                g_use = None
        else:
            g_use = None
        
        if g_use is not None:
            # FW step toward g_use
            t = 2.0 / (it + 2)
            f_try = (1 - t) * f + t * g_use
            f_try = normalize_nonneg(f_try)
            if f_try is None:
                f_try = f.copy()
            M_try = compute_M(f_try)
        else:
            # Subgradient step
            lr = t_base / (1 + 0.01 * it)
            f_try = f - lr * g
            f_try = normalize_nonneg(f_try)
            if f_try is None:
                f_try = f.copy()
            M_try = compute_M(f_try)
        
        if M_try < best_M - 1e-12:
            best_M = M_try
            best_f = f_try.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        f = f_try
        it += 1
        
        if no_imp > 100:
            f = 0.7 * best_f + 0.3 * normalize_nonneg(best_f + 0.2 * rng.normal(size=n))
            if f is None:
                f = best_f.copy()
            no_imp = 0
            t_base = max(0.5, t_base * 0.9)
    
    return best_f, best_M


def adam_polish(f_init, deadline, rng, lr=0.02, max_iters=20000):
    """
    Adam optimization in log-space for fine polishing.
    Uses soft-max approximation of the max for smooth gradients.
    """
    f = np.maximum(f_init, 1e-15)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    conv = compute_conv(f)
    M = float(conv.max())
    beta = max(1e-10, 0.1 * M)
    
    best_f = f.copy()
    best_M = M
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Soft-max gradient
        mval = M
        x = (conv - mval) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if s < 1e-30:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            w = ex / s
        
        # Gradient of max_conv w.r.t. f via correlation with w
        # grad[i] = sum_k w[k] * 2*f[k-i]
        # Compute via FFT
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(f[::-1], N)
        grad_full = np.fft.irfft(fw * fp, N)
        grad = 2.0 * grad_full[n - 1:2 * n - 1]
        
        # Add small regularization
        grad += 1e-8 * f
        
        # Convert to log-space gradient
        dot = float(np.dot(f, grad))
        gz = f * (grad - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z if 'best_z' in dir() else np.zeros(n)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = np.zeros(n)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        z = np.clip(z, -30, 30)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        conv = compute_conv(f)
        M = float(conv.max())
        
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.005, lr * 0.95)
        
        if adam_t % 100 == 0:
            beta = max(1e-12, beta * 0.7)
        
        if no_imp > max(200, 2 * n):
            z = np.zeros(n)
            e_init = np.maximum(f_init, 1e-15)
            z = np.log(e_init / e_init.sum())
            z += 0.3 * rng.normal(size=n)
            z = np.clip(z, -30, 30)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            conv = compute_conv(f)
            M = float(conv.max())
            beta = max(1e-10, 0.1 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_f, best_M


def make_linear_profile(n, slope=2.0):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_inv_linear_profile(n, slope=2.0):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * (1 - t / max(n - 1, 1))
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_u_shape(n, sharpness=2.0):
    t = np.linspace(0.0, 1.0, n)
    f = (2.0 * t - 1.0) ** 2 * sharpness + 1.0 / (sharpness + 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_v_shape(n, sharpness=2.0):
    t = np.linspace(0.0, 1.0, n)
    f = np.abs(2.0 * t - 1.0) * sharpness + 1.0 / (sharpness + 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_gaussian_bump(n, mu=0.5, sigma=0.2):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_dual_bump(n, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * (t / width) ** 2) + np.exp(-0.5 * ((t - 1) / width) ** 2) + 0.05
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_fourier_profile(n, rng, num_harms=10, max_amp=0.5):
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    for k in range(1, num_harms + 1):
        amp = max_amp / k * rng.uniform(0.5, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        f = f + amp * np.cos(2 * np.pi * k * t + phase)
    f = np.maximum(f, 1e-6)
    return f / f.sum()


def get_starting_points(n, rng, count=30):
    starts = [
        make_linear_profile(n, 1.0),
        make_linear_profile(n, 2.0),
        make_linear_profile(n, 3.0),
        make_linear_profile(n, 5.0),
        make_inv_linear_profile(n, 1.0),
        make_inv_linear_profile(n, 2.0),
        make_inv_linear_profile(n, 3.0),
        make_cubic_profile(n, 1.0, -0.5),
        make_cubic_profile(n, 1.5, -1.0),
        make_cubic_profile(n, 2.0, -1.5),
        make_cubic_profile(n, 2.5, -2.0),
        make_u_shape(n, 1.0),
        make_u_shape(n, 2.0),
        make_u_shape(n, 4.0),
        make_v_shape(n, 1.0),
        make_v_shape(n, 2.0),
        make_dual_bump(n, 0.1),
        make_dual_bump(n, 0.2),
        make_dual_bump(n, 0.3),
        make_gaussian_bump(n, 0.3, 0.2),
        make_gaussian_bump(n, 0.7, 0.2),
        make_gaussian_bump(n, 0.5, 0.4),
    ]
    for _ in range(count):
        starts.append(make_fourier_profile(n, rng, num_harms=rng.integers(3, 15), max_amp=rng.uniform(0.3, 0.7)))
    return starts


def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid optimization:
    1. Projected subgradient with momentum (fast)
    2. Frank-Wolfe with LP steps (more informed)
    3. Adam polish (fine-tuning)
    4. Multiple lengths and restarts
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)
    
    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
    except Exception:
        pass
    
    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent sequence score: {sc:.10f}")
    
    # === Phase 1: Fast subgradient at n=256 ===
    n1 = 256
    phase1_end = min(deadline, t_start + 300.0)
    print(f"Phase 1: Subgradient at n={n1}, time budget={phase1_end - t_start:.0f}s")
    
    starts = get_starting_points(n1, rng, count=20)
    if parent is not None:
        if len(parent) == n1:
            starts.append(parent.copy())
        else:
            starts.append(interpolate_profile(parent, n1))
    
    # Evaluate all starts
    for s in starts:
        sc = compute_score(s)
        try_update(s, sc)
    print(f"  Best after init: {best_score:.10f}")
    
    results_p1 = []
    # Run subgradient from top starts
    scored_starts = [(compute_score(s), s) for s in starts]
    scored_starts.sort(key=lambda x: x[0])
    
    num_starts = min(10, len(scored_starts))
    per_start_time = (phase1_end - time.monotonic()) / num_starts
    
    for i in range(num_starts):
        remaining = phase1_end - time.monotonic()
        if remaining < 5:
            break
        alloc = min(remaining, per_start_time)
        run_deadline = time.monotonic() + alloc
        f_best, M_best = projected_subgradient(scored_starts[i][1], run_deadline, rng, max_iters=20000)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results_p1.append((sc, f_best))
        if (i + 1) % 3 == 0:
            print(f"  Subgrad {i+1}/{num_starts}: best={best_score:.10f}")
    
    results_p1.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.10f}")
    
    # === Phase 2: FW with LP at n=256 ===
    phase2_end = min(deadline, t_start + 800.0)
    print(f"Phase 2: FW+LP at n={n1}, time budget={phase2_end - time.monotonic():.0f}s")
    
    top2 = min(6, len(results_p1))
    results_p2 = []
    per_start_time2 = (phase2_end - time.monotonic()) / max(1, top2)
    
    for i in range(top2):
        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining, per_start_time2)
        run_deadline = time.monotonic() + alloc
        f_best, M_best = fw_iteration(results_p1[i][1], run_deadline, rng, max_iters=100, use_lp=True, lp_every=5)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results_p2.append((sc, f_best))
        print(f"  FW {i+1}/{top2}: score={sc:.10f}, best={best_score:.10f}")
    
    results_p2.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best: {best_score:.10f}")
    
    # === Phase 3: Adam polish at n=256 ===
    phase3_end = min(deadline, t_start + 1100.0)
    print(f"Phase 3: Adam polish at n={n1}, time budget={phase3_end - time.monotonic():.0f}s")
    
    all_results = results_p1 + results_p2
    all_results.sort(key=lambda x: x[0])
    top3 = min(5, len(all_results))
    results_p3 = []
    per_start_time3 = (phase3_end - time.monotonic()) / max(1, top3)
    
    for i in range(top3):
        remaining = phase3_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining, per_start_time3)
        run_deadline = time.monotonic() + alloc
        f_best, M_best = adam_polish(all_results[i][1], run_deadline, rng, lr=0.015)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results_p3.append((sc, f_best))
        print(f"  Adam {i+1}/{top3}: score={sc:.10f}, best={best_score:.10f}")
    
    results_p3.sort(key=lambda x: x[0])
    print(f"Phase 3 done. Best: {best_score:.10f}")
    
    # === Phase 4: Try n=512 ===
    n4 = 512
    phase4_end = min(deadline, t_start + 1450.0)
    print(f"Phase 4: Extension to n={n4}, time budget={phase4_end - time.monotonic():.0f}s")
    
    all_combined = results_p1 + results_p2 + results_p3
    all_combined.sort(key=lambda x: x[0])
    top4 = min(3, len(all_combined))
    results_p4 = []
    per_start_time4 = (phase4_end - time.monotonic()) / max(1, top4 * 2)
    
    for i in range(top4):
        remaining = phase4_end - time.monotonic()
        if remaining < 15:
            break
        f_ext = interpolate_profile(all_combined[i][1], n4)
        
        # Subgradient at n=512
        alloc = min(remaining, per_start_time4)
        run_deadline = time.monotonic() + alloc
        f_best, M_best = projected_subgradient(f_ext, run_deadline, rng, max_iters=15000)
        sc = 2.0 * n4 * M_best
        try_update(f_best, sc)
        results_p4.append((sc, f_best))
        print(f"  n=512 subgrad {i+1}: score={sc:.10f}, best={best_score:.10f}")
        
        # Adam polish
        remaining = phase4_end - time.monotonic()
        if remaining > 10:
            alloc = min(remaining, per_start_time4)
            run_deadline = time.monotonic() + alloc
            f_best2, M_best2 = adam_polish(f_best, run_deadline, rng, lr=0.01)
            sc2 = 2.0 * n4 * M_best2
            try_update(f_best2, sc2)
            results_p4.append((sc2, f_best2))
            print(f"  n=512 adam {i+1}: score={sc2:.10f}, best={best_score:.10f}")
    
    results_p4.sort(key=lambda x: x[0])
    print(f"Phase 4 done. Best: {best_score:.10f}")
    
    # === Phase 5: Final refinement on best solution ===
    phase5_end = deadline
    if best_seq is not None and time.monotonic() < phase5_end - 30:
        n_best = len(best_seq)
        print(f"Phase 5: Final refinement at n={n_best}, time={phase5_end - time.monotonic():.0f}s")
        
        # Try subgradient
        f_best, M_best = projected_subgradient(best_seq, min(phase5_end - 10, time.monotonic() + (phase5_end - time.monotonic()) * 0.4), rng, max_iters=100000)
        sc = 2.0 * n_best * M_best
        try_update(f_best, sc)
        print(f"  Final subgrad: {sc:.10f}")
        
        # Try Adam
        remaining = phase5_end - time.monotonic()
        if remaining > 20:
            f_best2, M_best2 = adam_polish(best_seq, min(phase5_end - 5, time.monotonic() + remaining * 0.6), rng, lr=0.01)
            sc2 = 2.0 * n_best * M_best2
            try_update(f_best2, sc2)
            print(f"  Final Adam: {sc2:.10f}")
    
    # === Phase 6: Try n=128 (sometimes shorter is better) ===
    if time.monotonic() < deadline - 40:
        n6 = 128
        phase6_end = min(deadline - 20, time.monotonic() + 40.0)
        print(f"Phase 6: Try n={n6}")
        
        # Interpolate best to n=128
        if best_seq is not None:
            f_ext = interpolate_profile(best_seq, n6)
            f_best, M_best = projected_subgradient(f_ext, phase6_end, rng, max_iters=10000)
            sc = 2.0 * n6 * M_best
            try_update(f_best, sc)
            print(f"  n=128: {sc:.10f}, best={best_score:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = make_linear_profile(256, 1.5)
        best_score = compute_score(best_seq)
    
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    
    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}")
    
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence score: {ev_score:.10f}")
        if np.isfinite(ev_score) and ev_score < final_score:
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")
    
    return [float(x) for x in best_seq]
