"""
Step function optimizer: minimizes 2*n*max(conv(a,a))/(sum(a))^2
using multiplicative weight updates, LP spreading, and smooth gradient descent.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0)  # numerical safety
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_grad(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] where c = conv(a,a), w has length 2n-1.
    Returns g of length n where g[j] = sum_k w[k]*a[k-j].
    """
    n = len(a)
    L = 2 * n - 1
    N = _next_pow2(L)
    
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:L] = w
    
    fa = np.fft.rfft(a_pad, N)
    fw = np.fft.rfft(w_pad, N)
    
    # Cross-correlation: X[k] = sum_n a[n] * w[n-k] (circular)
    X = np.fft.irfft(fa * np.conj(fw), N)
    
    # g[j] = X[(N-j) % N]
    g = np.empty(n)
    g[0] = X[0]
    if n > 1:
        g[1:] = X[N-1:N-n:-1]  # X[N-1], X[N-2], ..., X[N-n+1]
    
    return g

"""
Step function optimizer: minimizes 2*n*max(conv(a,a))/(sum(a))^2
using multiplicative weight updates, LP spreading, smooth gradient descent,
and multi-resolution refinement. Key innovations: MPW method (distinct from
previous gradient-only approach), adaptive LP inflation, and length diversity.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
        return np.maximum(c, 0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_grad(a, w):
    """
    Gradient of sum_k w[k]*c[k] where c=conv(a,a).
    g[j] = sum_k w[k] * a[k-j] (a zero-padded outside [0,n-1])
    """
    n = len(a)
    L = 2 * n - 1
    N = _next_pow2(L)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:L] = w
    fa = np.fft.rfft(a_pad, N)
    fw = np.fft.rfft(w_pad, N)
    X = np.fft.irfft(fa * np.conj(fw), N)
    g = np.empty(n)
    g[0] = X[0]
    if n > 1:
        g[1:] = X[N - 1: N - n: -1]
    return g


def _smooth_max(a, beta):
    """Returns (smooth_max_value, weight_vector) for conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a)
    M = float(c.max())
    x = (c - M) / beta
    x = np.clip(x, -60, 0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-15:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    val = M + beta * float(np.log(max(s, 1e-30)))
    return val, w, c


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _mpw_optimize(a, deadline, eta_init=0.8):
    """Multiplicative Weight Update for minimizing max(conv(a,a))."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()

        return a

    best_a = a.copy()

    eta = eta_init
    no_imp = 0
    total_iter = 0

    while time.monotonic() < deadline:
        batch_size = 50
        for _ in range(batch_size):
            if time.monotonic() > deadline:
                break
            c = _fft_convolve(a)
            M = float(c.max())
            k_star = int(np.argmax(c))
            j_lo = max(0, k_star - n + 1)
            j_hi = min(n - 1, k_star)
            subgrad = np.zeros(n)
            subgrad[j_lo:j_hi + 1] = a[j_lo:j_hi + 1]
            a_new = a * np.exp(-eta * subgrad)
            sn = a_new.sum()
            if sn < 1e-15:
                break
            a_new = a_new / sn
            sn2 = float(a_new.sum())
            c_new = _fft_convolve(a_new)

            a = a_new

def _lp_optimize(a, deadline, inflate=1.001, blend=0.3):
    """LP-based spreading optimization."""
    a = np.asarray(a, dtype=np.float64).copy()

    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

        if sum_g <= 1.0 + 1e-8:

            if no_imp > 500:
                break
            continue
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:

        c_new = _fft_convolve(a_new)

def _grad_optimize(a, deadline, beta_init=None):
    """Smooth gradient descent with Adam."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    if beta_init is None:
        beta = max(1e-8, 0.05 * float(c.max()))
    else:
        beta = beta_init
    lr = 0.05
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    no_imp = 0

    while time.monotonic() < deadline:
        for _ in range(20):
            if time.monotonic() > deadline:
                break
            _, w, _ = _smooth_max(a, beta)
            g = _conv_grad(a, w)
            g += 1e-8 * a
            dot = float(np.dot(a, g))
            g_centered = g - dot * a
            t_adam += 1
            m = b1 * m + (1 - b1) * g_centered
            v = b2 * v + (1 - b2) * (g_centered * g_centered)
            mh = m / (1 - b1 ** t_adam)
            vh = v / (1 - b2 ** t_adam)
            a_new = _project_simplex(a - lr * mh / (np.sqrt(vh) + eps))
            a = a_new

        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                lr = max(0.005, lr * 0.95)
            if no_imp > 500:
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t_adam = 0
                beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
                lr = 0.05
                no_imp = 0
        if t_adam % 100 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a2=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_profile(n, rate=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * (t - 0.5))
    return a / a.sum()


def _make_bimodal_profile(n, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.2) / width) ** 2) + np.exp(-((t - 0.8) / width) ** 2)
    return a / a.sum()
