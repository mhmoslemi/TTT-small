        affected = []
        for k in range(k_min, k_max + 1):
            if k == 2 * j:
                affected.append(('quad', k, c[2*j] - old*old))
            else:
                affected.append(('lin', k, c[k] - 2.0*old*a[k-j], 2.0*a[k-j]))
        
        def g(x):

                    val = item[2] + x * x
                else:
                    val = item[2] + item[3] * x
                if val > max_val:
                    max_val = val
            return max_val

        f_hi = g(a_hi)

            a_lo = a_hi
            f_lo = f_hi
            a_hi *= 2.0
            f_hi = g(a_hi)

        c2 = a_lo + (a_hi - a_lo) / phi

                c2 = c1
                f2 = f1

                f1 = g(c1)
            else:
                a_lo = c1
                c1 = c2
                f1 = f2
                c2 = a_lo + (a_hi - a_lo) / phi
                f2 = g(c2)

        if x_opt < 0:
            x_opt = 0.0

Actually, for the 1D subproblem, there's a more efficient approach. The function g(x) = max(T_base, max_k (A_k + B_k * x), C + x^2) where the max is over the affected linear terms.

The minimum of g(x) for x >= 0 occurs at:
- x = 0, or
- A point where two linear terms are equal and both are >= T_base and >= C + x^2, or
- A point where a linear term equals C + x^2 and both are >= T_base.

For the case where two linear terms are equal: A_i + B_i * x = A_j + B_j * x => x = (A_j - A_i) / (B_i - B_j) (if B_i ≠ B_j).

So the candidate x values are:
- x = 0
- All intersection points x = (A_j - A_i) / (B_i - B_j) where B_i ≠ B_j and x >= 0
- All points where A_i + B_i * x = C + x^2 (quadratic equation)

The optimal x is the one among these candidates that minimizes g(x).

But there are O(n^2) intersection points, which is too many for n=512.

A better approach: use the fact that g(x) is convex and use Newton's method or bisection on the subgradient.

The subgradient of g(x) at x is:
- If the max is achieved by a unique linear term A_i + B_i*x: subgradient is {B_i}
- If the max is achieved by multiple linear terms: subgradient is {weighted average of B_i's}
- If the max is achieved by the quadratic: subgradient is {2x}
- If T_base is the max: subgradient is {0}

g(x) is minimized when 0 is in the subgradient. This means:
- If T_base is the max: we need x = 0 (or any x where T_base >= all others)
- If a linear term is the unique max: we need B_i = 0 (unlikely)
- If two linear terms are both max: we need B_i = B_j (unlikely)
- If the quadratic is the unique max: we need 2x = 0, i.e., x = 0
- If a linear and the quadratic are both max: we need B_i = 2x

So in most cases, the optimal x is either 0 or at a point where a linear term equals the quadratic.

This gives us O(n) candidate points to check:
- x = 0
- For each affected k (linear): solve A_k + B_k * x = C + x^2 => x^2 - B_k*x + (C - A_k) = 0
  => x = (B_k ± sqrt(B_k^2 - 4(C - A_k))) / 2

For each valid x >= 0, compute g(x). The minimum g(x) over all candidates is the answer.

This is O(n) per coordinate, making the total O(n^2) per sweep! Much better.

"""
Coordinate descent + LP + Adam hybrid for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses fast exact 1D subproblem solving via golden section, periodic LP escapes,
and Adam logit-space polish. Multiple restarts with perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def optimize_coord(a, j, c, n):
    """Golden section search for optimal a[j]. Returns optimal value."""
    old = a[j]
    k_min = j
    k_max = min(j + n - 1, 2 * n - 2)

    # Unaffected max
    T_base = 0.0
    if k_min > 0:
        T_base = max(T_base, float(c[:k_min].max()))
    if k_max < 2 * n - 2:
        T_base = max(T_base, float(c[k_max + 1:].max()))

    # Precompute linear terms for affected k != 2j
    affected_mask = np.ones(k_max - k_min + 1, dtype=bool)
    idx_2j = 2 * j - k_min
    if 0 <= idx_2j < len(affected_mask):
        affected_mask[idx_2j] = False

    k_indices = np.arange(k_min, k_max + 1)
    k_aff = k_indices[affected_mask]
    A = c[k_aff] - 2.0 * old * a[k_aff - j]
    B = 2.0 * a[k_aff - j]
    C = c[2 * j] - old * old  # quadratic: C + x^2

    def g(x):
        max_val = T_base
        if len(A) > 0:
            lm = float(np.maximum.reduce(A + B * x))
            if lm > max_val:
                max_val = lm
        qv = C + x * x
        if qv > max_val:
            max_val = qv
        return max_val

    # Golden section search
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    a0 = 0.0
    b0 = max(old * 3.0, 0.05)

    f_a0 = g(a0)
    f_b0 = g(b0)
    while f_b0 < f_a0 and b0 < 50.0:
        a0 = b0
        f_a0 = f_b0
        b0 *= 2.0
        f_b0 = g(b0)

    c1 = b0 - (b0 - a0) / phi
    c2 = a0 + (b0 - a0) / phi
    f1 = g(c1)
    f2 = g(c2)

    for _ in range(55):
        if f1 < f2:
            b0 = c2
            c2 = c1
            f2 = f1
            c1 = b0 - (b0 - a0) / phi
            f1 = g(c1)
        else:
            a0 = c1
            c1 = c2
            f1 = f2
            c2 = a0 + (b0 - a0) / phi
            f2 = g(c2)

    x_opt = (a0 + b0) / 2.0
    return max(0.0, x_opt)


def coord_descent_sweep(a, c, n, order=None):
    """One sweep of coordinate descent. Returns (a, c) or (None, None)."""
    if order is None:
        order = np.arange(n)

    for j in order:
        j = int(j)
        x_opt = optimize_coord(a, j, c, n)
        old = a[j]
        delta = x_opt - old
        if abs(delta) > 1e-15:
            a[j] = x_opt
            # Update c incrementally
            k_min = j
            k_max = min(j + n - 1, 2 * n - 2)
            # For k != 2j: c[k] += 2*delta*a[k-j]
            k_indices = np.arange(k_min, k_max + 1)
            other_idx = k_indices - j
            k_indices[2 * j - k_min] = -1  # temporarily exclude
            valid = k_indices >= 0
            if np.any(valid):
                c[k_indices[valid]] += 2.0 * delta * a[other_idx[valid]]
            # For k = 2j: c[2j] += delta*(new+old)
            if 0 <= 2 * j <= 2 * n - 2:
                c[2 * j] += delta * (x_opt + old)

    # Normalize
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None, None
    a = a / s
    c = np.convolve(a, a)
    return a, c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f, g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=15.0):
    """LP: maximize sum(g) s.t. g >= 0, conv(f,g) <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def adam_polish(p0, deadline, lr=0.03, reg=1e-8):
    """Adam gradient descent in logit space."""
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.2 * max_c)

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        x = (c - max_c) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # grad_p[i] = 2 * sum_x w[x] * p[x-i] (correlation)
        h = np.correlate(w, p0, mode='valid')[:n]
        # Actually need: h[i] = sum_{x} w[x] * p0[x - i] where valid
        # Use FFT or direct
        # h[i] = sum_j w[i+j] * p0[j] for j=0..n-1, i=0..n-1
        # This is cross-correlation
        h = np.zeros(n)
        for i in range(n):
            h[i] = np.dot(w[i:i + n], p0)

        grad_p = 2.0 * h + reg * p0
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.01, lr * 0.95)

        if it % 300 == 0:
            beta = max(1e-12, beta * 0.5)

        if no_imp > 3000:
            rng = np.random.default_rng(it)
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            p = np.exp(z - z.max())
            p = p / p.sum()
            beta = max(1e-10, 0.2 * float(np.convolve(p, p).max()))

        p0 = p

    return best_p, best_score


def interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

                no_improve = 0
            else:
                no_improve += 1
                a = local_best_a.copy()
                c = np.convolve(a, a)

            if no_improve > 80:
                # Perturb and try again
                pert = rng.normal(0, 0.01, n)
                a = local_best_a + pert
                a = normalize(a)
                if a is None:
                    break
                c = np.convolve(a, a)
                no_improve = 0
                if time.monotonic() > phase1_deadline - 20:
                    break

            if no_improve > 300:
                break

        # Update global best
        if local_best < best_score:
            best_score = local_best
            best_seq = local_best_a.copy()
            print(f"  Start {si} done: best={best_score:.8f}")

    print(f"  Phase 1 complete: {sweep_count} sweeps, best={best_score:.8f}")

    # === Phase 2: LP-based refinement ===
    print(f"\n=== Phase 2: LP Refinement ===")
    phase2_deadline = min(deadline, t_start + 1550.0)

    if time.monotonic() < phase2_deadline - 30:
        a = best_seq.copy()
        c = np.convolve(a, a)
        M = float(c.max())
        t_step = 0.3
        no_improve = 0

        t0 = time.monotonic()
        g = lp_direction(a, M, timeout=10)
        t_lp = time.monotonic() - t0
        print(f"  LP time: {t_lp*1000:.0f}ms")

        if g is not None:
            for it in range(100):
                if time.monotonic() > phase2_deadline - 20:
                    break
                if t_lp > 30:
                    break

                g = lp_direction(a, M, timeout=15)
                if g is None:
                    break
                g = normalize(g)
                if g is None:
                    break

                # Blend
                a_new = (1 - t_step) * a + t_step * g
                a_new = normalize(a_new)
                if a_new is None:
                    break
                c_new = np.convolve(a_new, a_new)
                score_new = 2.0 * n * float(c_new.max())

                if score_new < best_score - 1e-12:
                    best_score = score_new
                    best_seq = a_new.copy()
                    a = a_new.copy()
                    c = c_new.copy()
                    M = float(c.max())
                    no_improve = 0
                    t_step = max(0.05, t_step * 0.98)
                else:
                    no_improve += 1
                    t_step = max(0.05, t_step * 0.8)
                    if no_improve > 30:
                        break

        print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Adam polish ===
    print(f"\n=== Phase 3: Adam Polish ===")
    if time.monotonic() < deadline - 15:
        print(f"  Starting Adam: score={internal_score(best_seq):.8f}")
        p_best, sc_best = adam_polish(best_seq, deadline - 10, lr=0.025, reg=1e-8)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Adam improved to: {best_score:.8f}")
        else:
            print(f"  Adam no improvement: {sc_best:.8f} vs {best_score:.8f}")

    # === Final: One more round of coordinate descent on the best ===
    if time.monotonic() < deadline - 5:
        a = best_seq.copy()
        c = np.convolve(a, a)
        score = 2.0 * n * float(c.max())
        for sweep in range(200):
            if time.monotonic() > deadline - 3:
                break
            a, c = coord_descent_sweep(a, c, n)
            if a is None:
                break
            score = 2.0 * n * float(c.max())
            if score < best_score:
                best_score = score
                best_seq = a.copy()

    # Final output
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        # Fallback
        n_fb = 256
        t_arr = np.linspace(0, 1, n_fb)
        best_seq = 1.0 + 1.5 * (2.0 * t_arr - 1.0)
        best_seq = np.maximum(best_seq, 0.01)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
