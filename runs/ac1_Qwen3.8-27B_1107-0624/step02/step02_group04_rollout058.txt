"""
LP-iteration optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines linear programming response with gradient polishing.
"""
import numpy as np
import time
from scipy.sparse import lil_matrix, csr_matrix
from scipy.optimize import linprog

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    return np.convolve(a, a)

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return a
    return a / s

def _build_conv_matrix(f):
    """Build sparse matrix C where (C @ g) = conv(f, g)."""
    n = f.size
    m = 2 * n - 1
    C = lil_matrix((m, n))
    for k in range(m):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            C[k, j] = f[k - j]
    return C.tocsr()

def _lp_response(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Return g normalized to sum=1."""
    n = f.size
    C = _build_conv_matrix(f)
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)]*n, 
                  method='highs', options={'presolve': True})
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-30:
            return g / s
    return f.copy()  # fallback

def _profile_uniform(n):
    return np.ones(n) / n

def _profile_linear(n, eta=2.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (t - 0.5)
    return _normalize(a)

def _profile_ramp(n):
    return _normalize(np.linspace(0.01, 3.0, n))

def _profile_asym(n, gamma=2.0):
    """Asymmetric: high on left, decays to right."""
    t = np.linspace(0, 1, n)
    a = gamma ** (1 - t)
    return _normalize(a)

def _profile_centered(n, width_frac=0.6):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5) ** 2) / (2 * (width_frac/2) ** 2))
    return _normalize(a)

def _profile_two_hump(n, sep=0.3):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - (0.5-sep)) ** 2) / 0.01) + np.exp(-((t - (0.5+sep)) ** 2) / 0.01)
    return _normalize(a)

def _lp_iterate(f, deadline, rng, max_iter=500, blend_init=0.5, blend_min=0.05,
                blend_decay=0.98):
    """
    LP iteration: at each step, solve for optimal response g, then blend.
    """
    f = _normalize(f.copy())
    n = f.size
    best_f = f.copy()
    best_score = _score(f)
    
    blend = blend_init
    iter_count = 0
    
    while time.monotonic() < deadline and iter_count < max_iter:
        iter_count += 1
        c = _full_convolve(f)
        M = float(c.max())
        
        # Solve LP
        g = _lp_response(f, M)
        
        if g is None or not np.all(np.isfinite(g)):
            break
        
        # Blend
        f_new = (1 - blend) * f + blend * g
        f_new = _normalize(f_new)
        
        sc = _score(f_new)
        
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        
        f = f_new
        blend = max(blend_min, blend * blend_decay)
        
        # Check convergence
        if np.max(np.abs(f - g)) < 1e-8:
            break
    
    return best_f, best_score

    """

    """
    f = _normalize(f.copy())
    n = f.size
    z = np.log(np.maximum(f, 1e-30))
    z -= z.mean()
    
    best_f = f.copy()
    best_score = _score(f)
    
    m = np.zeros(n)

  = f_j * (grad_j - dot(f, grad))

So: `gz = f * (grad - dot(f, grad))`

And then Adam update in z-space.

```python
"""
LP-iteration and gradient descent for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming response time (primal-dual style) combined with
soft-max gradient polishing for final improvement.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _full_conv(a):
    return np.convolve(a, a)


def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_conv(a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / a.size
    return a / s


def _build_conv_matrix_indices(n):
    """Precompute row/col indices for the convolution matrix of sequences of length n."""
    m = 2 * n - 1
    rows = []
    cols = []
    for k in range(m):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        cnt = j_max - j_min + 1
        rows.extend([k] * cnt)
        cols.extend(range(j_min, j_max + 1))
    return np.array(rows), np.array(cols)


def _build_conv_matrix(f, row_idx, col_idx):
    """Build the sparse convolution matrix using precomputed indices."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    m = 2 * n - 1
    # Value at (k, j) is f[k-j]
    # k = row_idx, j = col_idx
    offsets = row_idx - col_idx  # k - j
    vals = f[offsets]
    return csr_matrix((vals, (row_idx, col_idx)), shape=(m, n))


def _lp_response(f, M, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    n = f.size
    C = _build_conv_matrix(f, row_idx, col_idx)
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = float(g.sum())
        if s > 1e-30:
            return g / s
    return None


def _lp_iterate(f, deadline, row_idx=None, col_idx=None,
                max_iter=1000, blend_init=0.5, blend_min=0.01,
                blend_decay=0.995):
    """LP iteration loop."""
    f = _normalize(f.copy())
    n = f.size
    
    if row_idx is None:
        row_idx, col_idx = _build_conv_matrix_indices(n)
    
    best_f = f.copy()
    best_sc = _score(f)
    
    blend = blend_init
    prev_sc = best_sc
    
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        
        c = _full_conv(f)
        M = float(c.max())
        
        g = _lp_response(f, M, row_idx, col_idx)
        if g is None:
            break
        
        f_new = (1 - blend) * f + blend * g
        f_new = _normalize(f_new)
        
        sc = _score(f_new)
        
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
        
        f = f_new
        blend = max(blend_min, blend * blend_decay)
        
        # Convergence check
        if abs(sc - prev_sc) < 1e-10:
            if it > 50:  # Allow some iterations before declaring convergence
                break
        prev_sc = sc
        
        if it % 50 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, blend={blend:.4f}")
    
    return best_f, best_sc


def _gradient_polish(f, deadline, lr=0.1, beta_init=0.1, max_iter=10000):
    """Gradient descent in logit space with soft-max objective."""
    f = _normalize(f.copy())
    n = f.size
    
    z = np.log(np.maximum(f, 1e-30))
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_f = f.copy()
    best_sc = _score(f)
    no_improve = 0
    
    beta = beta_init * float(_full_conv(f).max())
    
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        
        c = _full_conv(f)
        c_max = float(c.max())
        beta = max(1e-12, beta * 0.999)  # Slow decay
        
        # Soft-max weights
        c_shifted = (c - c_max) / beta
        w = np.exp(c_shifted)
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Gradient: 2 * correlate(w, f, 'full')[:n]
        grad = 2.0 * np.correlate(w, f, mode='full')[:n]
        
        # Logit-space gradient
        dot = float(np.dot(f, grad))
        gz = f * (grad - dot)
        gz -= gz.mean()
        
        # Safety
        if not np.all(np.isfinite(gz)):
            break
        
        # Adam
        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        f = _normalize(np.exp(z - z.max()))
        sc = _score(f)
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Restart with noise
            z = np.log(np.maximum(best_f, 1e-30))
            z -= z.mean()
            z += 0.1 * np.random.default_rng(t).normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            f = _normalize(np.exp(z - z.max()))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = min(0.2, lr * 1.2)
            no_improve = 0
        
        if it % 200 == 0:
            print(f"    Grad iter {it}: score={sc:.10f}, beta={beta:.6f}")
    
    return best_f, best_sc

    return np.ones(n) / n
