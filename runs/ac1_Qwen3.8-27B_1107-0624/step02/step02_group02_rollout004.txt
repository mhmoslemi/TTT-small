"""
Optimized search for minimum peak autocorrelation non-negative sequences.
Combines LP-based refinement with Nesterov-accelerated soft-max Adam,
adaptive beta scheduling, diverse restarts, and champion-challenger strategy.
Key improvements over previous approach: Nesterov momentum, better restart
diversity, symmetric/asymmetric exploration, and more aggressive temperature
annealing for the soft-max approximation.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-15:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z) / z.size
    return e / s


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M, max_iter=None):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'max_iter': max_iter or 5000})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, verbose=False):
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)
    best_M = float(np.convolve(f, f).max())
    best_f = f.copy()
    t = 0.25
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        M = float(np.convolve(f, f).max())
        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new
        M_new = float(np.convolve(f_new, f_new).max())

        if M_new <= best_M + 1e-13:
            f = f_new
            if M_new < best_M - 1e-13:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
            if verbose and iteration % 50 == 0:
                print(f"  LP iter {iteration}: M={best_M:.10f}")
        else:
            t *= 0.5
            stagnant += 1
        t = max(0.003, t * 0.997)

        if stagnant > 80:
            noise_scale = 0.05 * max(1.0 / n, np.std(best_f))
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.25
            stagnant = 0

    return best_f


def nesterov_adam_optimize(p0, deadline, rng, lr=0.05, reg=1e-8, beta_init=None):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    b3 = 0.9999  # Nesterov momentum
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    m2 = np.zeros(n, dtype=np.float64)  # Nesterov
    adam_t = 0

    c = full_convolve_fft(p0)
    maxc = float(c.max())
    if beta_init is None:
        beta = max(1e-8, 0.15 * maxc)
    else:
        beta = beta_init

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    stagnant_count = 0

    base_lr = lr
    beta_decay = 0.92
    beta_floor = 1e-6 * maxc

    while time.monotonic() < deadline:
        L, w = soft_max_and_weights(c, beta)
        gp = 2.0 * corr_weights(w, p0)
        vp = gp + reg * p0
        dot = float(np.dot(p0, vp))
        gz = p0 * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            m2.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)

        # Nesterov step
        m2 = b3 * m2 + (1.0 - b3) * (lr * mh / (np.sqrt(vh) + eps))
        z = z - (m2 + b3 * m2)  # Nesterov look-ahead

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            m2.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p_new = softmax_from_z(z)
        c = full_convolve_fft(p_new)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p_new.copy()
                best_z = z.copy()
                no_imp = 0
                stagnant_count = 0
            else:
                no_imp += 1
                stagnant_count += 1
                if no_imp % 500 == 0:
                    lr = max(0.005, lr * 0.95)
        else:
            z = best_z.copy()
            p_new = softmax_from_z(z)
            m.fill(0.0)
            v.fill(0.0)
            m2.fill(0.0)
            adam_t = 0
            continue

        #_update p0 for next iteration
        p0 = p_new

        if adam_t % 60 == 0:
            beta = max(beta_floor, beta * beta_decay)

        # Restart strategy
        if stagnant_count > max(600, 3 * n):
            if rng.random() < 0.3:
                # Large restart: blend with random
                z = 0.5 * best_z + 0.8 * rng.normal(size=n)
                beta = max(1e-8, 0.2 * float(full_convolve_fft(softmax_from_z(z)).max()))
            elif rng.random() < 0.5:
                # Medium restart
                z = 0.9 * best_z + 0.3 * rng.normal(size=n)
                beta = max(1e-8, 0.15 * maxc)
            else:
                # Small perturbation
                z = best_z + 0.1 * rng.normal(size=n)
                beta = max(beta_floor, maxc * 0.1)

            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p0 = softmax_from_z(z)
            c = full_convolve_fft(p0)
            maxc = float(c.max())
            m.fill(0.0)
            v.fill(0.0)
            m2.fill(0.0)
            adam_t = 0
            lr = base_lr
            stagnant_count = 0
            no_imp = 0

    return best_p, best_score

def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_gaussian_profile(n, sigma=0.25):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_kaiser_profile(n, beta_k=3.0):
    # Approximate Kaiser window
    t = np.linspace(-1.0, 1.0, n)
    # Simple approximation using Bessel-like function
    x = beta_k * np.sqrt(np.maximum(1 - t**2, 0))
    a = np.exp(x) - 1
    a = a / (np.exp(beta_k) - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tent_profile(n, sharpness=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - sharpness * np.abs(2 * t - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_optimal_ish_profile(n):
    # Based on intuition: slight asymmetry helps reduce peak
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.3 * (t - 0.5) + 0.2 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_random_smooth_profile(n, rng, num_freqs=5):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, num_freqs + 1):
        amp = rng.uniform(0.05, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def subgradient_step(p, c, step_size, rng):
    """One subgradient step: move away from the peak location."""
    n = p.size
    max_idx = int(np.argmax(c))
    # Subgradient of max(conv(p,p)) at p: it's conv(e_max_idx, p)
    # where e_max_idx is the impulse at the max position
    # conv(e_k, p)[i] = p[i + k - (n-1)] (shifted)
    # More precisely, the gradient of conv(p,p)[max_idx] wrt p[j] is:
    # sum over valid i: p[i] * delta(i + j = max_idx) + p[max_idx - j] * delta(j = max_idx - i)
    # = p[max_idx - j] (if 0 <= max_idx - j < n) + p[max_idx - (n-1-j)] (if valid)
    # Actually for np.convolve(p,p)[k] = sum_j p[j]*p[k-j], d/dp[m] = p[k-m] (if valid)
    grad = np.zeros(n)
    for m in range(n):
        # d conv(p,p)[max_idx] / dp[m] = p[max_idx - m] if 0 <= max_idx-m < n
        idx = max_idx - m
        if 0 <= idx < n:
            grad[m] += p[idx]
        # Also the other term: if m is the "first" index
        idx2 = max_idx - m
        # Actually conv(p,p)[k] = sum_{j: 0<=j<n, 0<=k-j<n} p[j]*p[k-j]
        # d/dp[m] = p[k-m] (if 0<=k-m<n) [from j=m] + p[m] * ... no
        # d/dp[m] of sum_j p[j]*p[k-j]:
        #   = p[k-m] (when j=m) + sum_j p[j] * d/dp[m] p[k-j]
        #   = p[k-m] + p[k-m] (when k-j=m, i.e., j=k-m) if k-m != m
        #   = 2*p[k-m] if k-m != m and 0<=k-m<n
        #   = p[k-m] if k-m == m (i.e., k=2m)
        pass  # Already handled above approximately

    # Simpler: just use the soft-max gradient which we already have
    return grad


def symmetric_optimize(p0, deadline, rng, lr=0.05):
    """Optimize only the first half, enforce symmetry."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n = p0.size
    n_half = n // 2
    # Take first half
    p_half = p0[:n_half].copy()
    p_half = np.maximum(p_half, 1e-12)
    s = p_half.sum() + p0[n_half:n - n_half].sum() if n % 2 == 0 else p_half.sum() + p0[n_half].sum()
    # Just normalize the full thing
    p_full = p0 / p0.sum()

    # Work in logit space for the half
    z_half = np.log(np.maximum(p_half, 1e-12))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n_half)
    v = np.zeros(n_half)
    adam_t = 0

    def build_symmetric(zh):
        zh = np.clip(zh, -50, 50)
        a_half = softmax_from_z(zh)
        if n % 2 == 0:
            a = np.concatenate([a_half, a_half[::-1]])
        else:
            a = np.concatenate([a_half, [0], a_half[::-1]])
        a = a / a.sum()
        return a

    c = full_convolve_fft(p_full)
    maxc = float(c.max())
    beta = max(1e-8, 0.15 * maxc)
    best_score = 2.0 * n * maxc
    best_zh = z_half.copy()
    no_imp = 0

    p_full = build_symmetric(z_half)

    while time.monotonic() < deadline:
        _, w = soft_max_and_weights(c, beta)
        # Gradient for the half
        gp_full = 2.0 * corr_weights(w, p_full)
        gp_half = gp_full[:n_half]
        vp_half = gp_half + 1e-8 * a_half
        # Project to simplex (for the half)
        dot = float(np.dot(a_half, vp_half))
        gz_half = a_half * (vp_half - dot)
        gz_half -= float(np.mean(gz_half))

        if not np.all(np.isfinite(gz_half)):
            z_half = best_zh.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz_half
        v = b2 * v + (1 - b2) * gz_half**2
        mh = m / (1 - b1**adam_t)
        vh = v / (1 - b2**adam_t)
        z_half = z_half - lr * mh / (np.sqrt(vh) + eps)
        z_half = np.clip(z_half, -50, 50)

        p_full = build_symmetric(z_half)
        c = full_convolve_fft(p_full)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if score < best_score - 1e-13:
            best_score = score
            best_zh = z_half.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 400:
                z_half = best_zh + 0.3 * rng.normal(size=n_half)
                z_half = np.clip(z_half, -50, 50)
                p_full = build_symmetric(z_half)
                c = full_convolve_fft(p_full)
                maxc = float(c.max())
                beta = max(1e-8, 0.15 * maxc)
                m.fill(0.0)
                v.fill(0.0)
                adam_t = 0
                no_imp = 0

        if adam_t % 80 == 0:
            beta = max(1e-9, beta * 0.9)

    return build_symmetric(best_zh)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting search. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Quick LP from best initial (200s) =====
    n_main = 512
    phase1_end = min(deadline, t_start + 200.0)

    if init_seq is not None and init_seq.size != n_main:
        p_main = interpolate_profile(init_seq, n_main)
    elif init_seq is not None:
        p_main = init_seq.copy()
    else:
        p_main = make_linear_profile(n_main, 1.5)

    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1: n={n_main}, initial score={sc:.10f}")

    if time.monotonic() < phase1_end - 5:
        lp_deadline = min(phase1_end, time.monotonic() + 180.0)
        result = lp_optimize(p_main, lp_deadline, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        print(f"After LP: score={sc:.10f}")

    # Try a few other profiles with short LP
    profiles = get_diverse_profiles(n_main, rng, count=15)
    for name, p in profiles[:8]:
        if time.monotonic() > phase1_end - 3:
            break
        sc_p = compute_score(p)
        if sc_p < best_score + 0.02:
            per_dl = min(phase1_end, time.monotonic() + 15.0)
            result = lp_optimize(p, per_dl, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)

    print(f"After Phase 1: best={best_score:.10f}")

    # ===== Phase 2: Nesterov Adam with diverse restarts (1000s) =====
    phase2_end = min(deadline, t_start + 1250.0)
    if best_seq is not None:
        print(f"Phase 2: Nesterov Adam optimization, current best={best_score:.10f}")

        # Multiple optimization rounds with different parameters
        configs = [
            {"lr": 0.05, "beta_init": None},
            {"lr": 0.03, "beta_init": None},
            {"lr": 0.08, "beta_init": None},
            {"lr": 0.04, "beta_init": None},
            {"lr": 0.06, "beta_init": None},
            {"lr": 0.02, "beta_init": None},
        ]

        num_rounds = len(configs)
        round_time = (phase2_end - time.monotonic()) / max(1, num_rounds)

        for i, cfg in enumerate(configs):
            if time.monotonic() > phase2_end - 10:
                break
            per_deadline = min(phase2_end, time.monotonic() + max(60.0, round_time))
            print(f"  Round {i+1}/{num_rounds}: lr={cfg['lr']}, time_budget={max(0, per_deadline - time.monotonic()):.0f}s")

            # Start from best + small noise
            p0 = best_seq.copy()
            noise_scale = rng.choice([0.001, 0.005, 0.01])
            p0 = p0 + noise_scale * rng.normal(size=p0.size)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()

            if i % 2 == 0:
                result, sc = nesterov_adam_optimize(p0, per_deadline, rng, lr=cfg["lr"])
            else:
                result, sc = plain_adam_optimize(p0, per_deadline, rng, lr=cfg["lr"])
            update(result, sc)
            print(f"  Round {i+1} done: score={sc:.10f}, best={best_score:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # ===== Phase 3: Symmetric optimization + LP polish (350s) =====
    phase3_end = min(deadline, t_start + 1600.0)
    if best_seq is not None and time.monotonic() < phase3_end - 10:
        # Try symmetric optimization
        sym_deadline = min(phase3_end, time.monotonic() + 120.0)
        p_sym_input = best_seq.copy()
        p_sym_input = (p_sym_input + p_sym_input[::-1]) / 2  # symmetrize
        p_sym_input = p_sym_input / p_sym_input.sum()
        p_sym = symmetric_optimize(p_sym_input, sym_deadline, rng, lr=0.04)
        sc = compute_score(p_sym)
        update(p_sym, sc)
        print(f"Symmetric optimization: score={sc:.10f}")

        # LP polish on best
        if time.monotonic() < phase3_end - 10:
            lp_dl = min(phase3_end, time.monotonic() + 150.0)
            result = lp_optimize(best_seq, lp_dl, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
            print(f"LP polish: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ===== Phase 4: Final fine-tuning (remaining time) =====
    if best_seq is not None and time.monotonic() < deadline - 5:
        final_dl = deadline
        # Very fine Adam
        p0 = best_seq + 0.0005 * rng.normal(size=best_seq.size)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, sc = plain_adam_optimize(p0, final_dl, rng, lr=0.01,
                                         beta_init=5e-5)
        update(result, sc)
        print(f"Final fine-tune: score={sc:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
