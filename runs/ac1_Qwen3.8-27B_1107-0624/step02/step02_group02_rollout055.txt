    z = np.log(a)
    z -= z.mean()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    beta = beta_init
    
    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        val, grad_a = soft_max_and_grad(c, a, beta)
        
        # Chain rule for softmax
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= grad_z.mean()  # center for stability
        
        t += 1
        m = b1*m + (1-b1)*grad_z
        v = b2*v + (1-b2)*grad_z**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50, 50)
        z_new -= z_new.mean()
        
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()
        
        score = compute_score(a_new)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
            z = z_new
            a = a_new
        else:
            # Revert
            z = z + lr * mh / (np.sqrt(vh) + eps)
        
        # Decay beta
        if t % 50 == 0:
            beta *= 0.95
            beta = max(beta, 1e-4)
    
    return best_a, best_score

def compute_soft_max_grad(c, a, beta):
    """
    Compute soft max of c and gradient w.r.t. a.
    Returns (val, grad) where grad_i = d(val)/d(a_i)
    """
    n = len(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    w = ex / s
    val = mval + beta * np.log(s)
    
    # grad_i = 2 * beta * sum_j w[i+j] * a[j] for i=0,...,n-1
    # This is 2*beta * (a_rev * w)[i+n-1]
    a_rev = a[::-1]
    N = next_pow2((n-1) + (2*n-2) + 1)  # >= 3n-2
    fa = rfft(a_rev, N)
    fw = rfft(w, N)
    conv_full = irfft(fa * fw, N)
    h = conv_full[n-1:2*n-1]
    grad = 2.0 * beta * h
    return val, grad

def softmax_from_z(z):
    z = z - float(z.max())
    e = np.exp(z)
    return e / e.sum()

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(ind[cond])
    theta = cssv[rho-1] / rho
    return np.maximum(v - theta, 0)

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else np.ones(n_new)/n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)

def logit_adam_optimize(a0, deadline, rng, lr_init=0.1, beta_init=None, lr_end=None):
    """
    Optimize a in logit space using Adam with adaptive smooth max.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-12)
    s = a.sum()
    if s < 1e-10:
        return a0, compute_score(a0)
    a = a / s
    
    z = np.log(a)
    z -= z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    if beta_init is None:
        c0 = full_convolve_fft(a)
        beta_init = 0.1 * c0.max()
    beta = beta_init
    
    if lr_end is None:
        lr_end = lr_init * 0.1
    
    best_a = a.copy()
    best_score = compute_score(a)
    no_imp = 0
    max_no_imp = 500
    
    t_start_inner = time.monotonic()
    
    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        val, grad_a = compute_soft_max_grad(c, a, beta)
        
        # Chain rule for softmax: grad_z = a * (grad_a - dot(a, grad_a))
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= grad_z.mean()
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        # Learning rate with cosine annealing
        progress = min(1.0, (t_start_inner - time.monotonic() + (deadline - t_start_inner)) / (deadline - t_start_inner)) if deadline > t_start_inner else 0
        # Simpler: just decay linearly
        lr = lr_init + (lr_end - lr_init) * min(1.0, t / 10000)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z -= z.mean()
        
        a = softmax_from_z(z)
        
        score = 2.0 * n * float(full_convolve_fft(a).max())
        
        if np.isfinite(score) and score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decay beta
        if t % 30 == 0:
            beta = max(1e-5 * n, beta * 0.92)
        
        # Restart if stuck
        if no_imp > max_no_imp:
            # Try different perturbation
            scale = 0.3 / np.sqrt(n)
            z = z + scale * rng.standard_normal(n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            a = softmax_from_z(z)
            beta = max(0.05 * full_convolve_fft(a).max(), 1e-4)
            m.fill(0)
            v.fill(0)
            t = 0
            no_imp = 0
            lr = lr_init
            max_no_imp = int(max_no_imp * 0.8)
            if max_no_imp < 100:
                break
    
    return best_a, best_score

def make_profiles(n, rng):

"""
Gradient-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses logit-space Adam with adaptive smooth-max, exact subgradient steps,
multi-scale search (n=256,512,1024), and diverse restarts.

import numpy as np

from scipy.fft import rfft, irfft


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    n = a.size
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_soft_max_grad(c, a, beta):
    n = len(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    w = ex / s
    val = mval + beta * np.log(s)
    a_rev = a[::-1]
    N = next_pow2(3 * n - 2)
    fa = rfft(a_rev, N)
    fw = rfft(w, N)
    conv_full = irfft(fa * fw, N)
    h = conv_full[n - 1:2 * n - 1]
    grad = 2.0 * beta * h
    return val, grad


def compute_exact_subgrad(c, a):
    n = len(a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def softmax_from_z(z):

    if s < 1e-10:
        return a0, compute_score(a0)
    a = a / s

    z = np.log(a)
    z -= z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    if beta_init is None:
        c0 = full_convolve_fft(a)
        beta_init = 0.08 * c0.max()
    beta = beta_init

    best_a = a.copy()
    best_score = compute_score(a)
    no_imp = 0
    max_no_imp = restart_patience
    t_start_inner = time.monotonic()
    total_budget = max(1.0, deadline - t_start_inner)

    while time.monotonic() < deadline:
        progress = min(1.0, (time.monotonic() - t_start_inner) / total_budget)
        lr = lr_init * (1.0 - 0.7 * progress)
        lr = max(lr, lr_init * 0.05)

        c = full_convolve_fft(a)

        if use_exact_steps and t > 0 and t % exact_every == 0:
            grad_a = compute_exact_subgrad(c, a)
        else:
            _, grad_a = compute_soft_max_grad(c, a, beta)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = np.log(np.maximum(best_a, 1e-12))
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t = 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= float(np.mean(z))
        a = softmax_from_z(z)

        sc = 2.0 * n * float(full_convolve_fft(a).max())

        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t % 20 == 0:
            beta = max(2e-5 * n, beta * 0.9)

        if no_imp > max_no_imp:
            scale = 0.25 / np.sqrt(n)
            z = 0.7 * z + scale * rng.standard_normal(n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            a = softmax_from_z(z)
            c_tmp = full_convolve_fft(a)
            beta = max(0.1 * c_tmp.max(), 1e-4)
            m.fill(0)
            v.fill(0)
            t = 0
            no_imp = 0
            max_no_imp = max(100, int(max_no_imp * 0.7))
            if max_no_imp < 80:
                break

    return best_a, best_score


def subgradient_polish(a0, deadline, rng, step_init=0.1):
    """Subgradient method with averaging for final polish."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0, compute_score(a0)
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    a_avg = a.copy()
    step = step_init
    t = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        a_avg = a_avg * t / (t + 1) + a_new / (t + 1)
        a = a_new
        t += 1
        step = step_init / np.sqrt(t + 1)

        if t % 100 == 0:
            sc_avg = 2.0 * n * float(full_convolve_fft(a_avg).max())
            sc_cur = 2.0 * n * float(full_convolve_fft(a).max())
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()
            elif sc_cur < best_score:
                best_score = sc_cur
                best_a = a.copy()

    sc_final = 2.0 * n * float(full_convolve_fft(a_avg).max())
    if sc_final < best_score:
        best_a = a_avg.copy()
        best_score = sc_final
    return best_a, best_score


def make_profiles(n, rng):
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for k in [1, 2, 3]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * k * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    a = 1.0 - np.abs(2 * t - 1)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 4.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for _ in range(4):
        nc = int(rng.integers(3, 7))
        coeffs = rng.uniform(-0.25, 0.25, nc)
        phases = rng.uniform(0, 2 * np.pi, nc)
        a = np.ones(n)
        for i in range(1, nc):
            a += coeffs[i] * np.cos(2 * np.pi * i * t + phases[i])
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def coordinate_descent(a0, deadline, rng, max_sweeps=200):
    """Block coordinate descent: for each coordinate, find optimal value."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0, compute_score(a0)
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    order = rng.permutation(n)
    sweep = 0

    while time.monotonic() < deadline and sweep < max_sweeps:
        improved = False
        for idx in order:
            c = full_convolve_fft(a)
            a_i = a[idx]
            other_sum = 1.0 - a_i
            if other_sum < 1e-10:
                continue
            # When we change a_i to x, the new sum is 1 (other coords unchanged if we renormalize)
            # Actually let's fix sum=1 and optimize a_i with the constraint that
            # the rest scales to maintain sum=1.
            # Simpler: just try a few candidate values for a_i
            # The optimal a_i minimizes max_k (c_k - a_i*a_{k-i} + x * a_{k-i})
            # = max_k (base_k + x * V_k) where base_k = c_k - a_i * a_{k-i}
            #   V_k = a_{k-i} for valid k-i
            # This is a 1D convex piecewise linear minimization.
            # Use golden section search on x in [0, 1]
            
            # Compute base and V
            V = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                j = k - idx
                if 0 <= j < n:
                    V[k] = a[j]
            base = c - a_i * V
            
            # Golden section search for x in [0, 1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = 1.0
            for _ in range(40):
                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)
                f1 = float(np.max(base + x1 * V))
                f2 = float(np.max(base + x2 * V))
                if f1 < f2:
                    hi = x2
                else:
                    lo = x1
            x_opt = (lo + hi) / 2.0
            
            if abs(x_opt - a_i) > 1e-8:
                a = (1.0 - x_opt) * (a / (1.0 - a_i)) if (1.0 - a_i) > 1e-10 else a
                a[idx] = x_opt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                improved = True
            
            if time.monotonic() > deadline:
                break
        
        if not improved and sweep > 5:
            break
        sweep += 1
        
        if sweep % 10 == 0:
            sc = 2.0 * n * float(full_convolve_fft(a).max())
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
    
    sc = 2.0 * n * float(full_convolve_fft(a).max())
    if sc < best_score:
        best_a = a.copy()
        best_score = sc
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting. Budget: {budget_s}s")
    if init_seq is not None:
        sc = compute_score(init_seq)
        update(init_seq, sc)
        print(f"Initial: n={init_seq.size}, score={sc:.10f}")

    # Phase 1: Primary optimization at n=512
    n1 = 512
    if init_seq is not None and init_seq.size != n1:
        p1 = interpolate_profile(init_seq, n1)
    elif init_seq is not None:
        p1 = init_seq.copy()
    else:
        t = np.linspace(0, 1, n1)
        p1 = (1.0 + 1.5 * (t - 0.5))
        p1 = p1 / p1.sum()

    phase1_end = t_start + 500.0
    print(f"\n=== Phase 1: n={n1} (primary) ===")
    sc = compute_score(p1)
    update(p1, sc)
    print(f"  Start: {sc:.10f}")

    lrs = [0.08, 0.05, 0.12, 0.04, 0.06, 0.1]
    for i in range(len(lrs)):
        if time.monotonic() > phase1_end - 5:
            break
        per_dl = min(phase1_end, time.monotonic() + 80.0)
        p = best_seq.copy() if best_seq is not None else p1
        if i > 0:
            noise = 0.003 * (i % 3 + 1)
            p = p + noise * rng.standard_normal(n1)
            p = np.maximum(p, 1e-12)
            p = p / p.sum()
        result, sc = logit_adam_optimize(p, per_dl, rng, lr_init=lrs[i])
        update(result, sc)
        print(f"  Adam run {i} (lr={lrs[i]}): {sc:.10f}, best={best_score:.10f}")

    # Coordinate descent polish
    if time.monotonic() < phase1_end - 30:
        per_dl = min(phase1_end, time.monotonic() + 60.0)
        result, sc = coordinate_descent(best_seq, per_dl, rng)
        update(result, sc)
        print(f"  Coord descent: {sc:.10f}, best={best_score:.10f}")

    # Subgradient polish
    if time.monotonic() < phase1_end - 20:
        per_dl = min(phase1_end, time.monotonic() + 30.0)
        result, sc = subgradient_polish(best_seq, per_dl, rng, step_init=0.05)
        update(result, sc)
        print(f"  Subgradient: {sc:.10f}, best={best_score:.10f}")

    print(f"Phase 1 done. best={best_score:.10f}")

    # Phase 2: Diverse profiles at n=512
    phase2_end = t_start + 750.0
    print(f"\n=== Phase 2: Diverse starts, n={n1} ===")
    profiles = make_profiles(n1, rng)
    for pi, p in enumerate(profiles):
        if time.monotonic() > phase2_end - 10:
            break
        sc_p = compute_score(p)
        if sc_p > best_score * 1.03:
            continue
        per_dl = min(phase2_end, time.monotonic() + 30.0)
        result, sc = logit_adam_optimize(p, per_dl, rng, lr_init=0.06)
        update(result, sc)
        if sc < best_score + 0.02:
            print(f"  Profile {pi}: {sc:.10f}")

    # Phase 3: Try n=1024
    n2 = 1024
    phase3_end = t_start + 1150.0
    print(f"\n=== Phase 3: n={n2} ===")
    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        p2 = interpolate_profile(best_seq, n2)
        sc = compute_score(p2)
        update(p2, sc)
        print(f"  Interpolated: {sc:.10f}")

        for i in range(3):
            if time.monotonic() > phase3_end - 10:
                break
            per_dl = min(phase3_end, time.monotonic() + 120.0)
            lr = [0.04, 0.03, 0.06][i]
            p = best_seq.copy() if best_seq is not None else p2
            if p.size != n2:
                p = interpolate_profile(p, n2)
            if i > 0:
                p = p + 0.002 * rng.standard_normal(n2)
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            result, sc = logit_adam_optimize(p, per_dl, rng, lr_init=lr)
            update(result, sc)
            print(f"  Run {i} (lr={lr}): {sc:.10f}, best={best_score:.10f}")

        if time.monotonic() < phase3_end - 20:
            per_dl = min(phase3_end, time.monotonic() + 40.0)
            result, sc = coordinate_descent(best_seq, per_dl, rng, max_sweeps=50)
            update(result, sc)
            print(f"  Coord descent: {sc:.10f}, best={best_score:.10f}")

    # Phase 4: Try n=256
    n3 = 256
    phase4_end = t_start + 1350.0
    print(f"\n=== Phase 4: n={n3} ===")
    if time.monotonic() < phase4_end - 20 and best_seq is not None:
        p3 = interpolate_profile(best_seq, n3)
        sc = compute_score(p3)
        update(p3, sc)
        for i in range(3):
            if time.monotonic() > phase4_end - 10:
                break
            per_dl = min(phase4_end, time.monotonic() + 60.0)
            lr = [0.1, 0.06, 0.08][i]
            p = best_seq.copy() if best_seq is not None else p3
            if p.size != n3:
                p = interpolate_profile(p, n3)
            result, sc = logit_adam_optimize(p, per_dl, rng, lr_init=lr)
            update(result, sc)
            print(f"  Run {i}: {sc:.10f}, best={best_score:.10f}")

    # Phase 5: Final intensive polish
    phase5_end = deadline
    if time.monotonic() < phase5_end - 10 and best_seq is not None:
        n_best = best_seq.size
        print(f"\n=== Phase 5: Final polish, n={n_best} ===")

        for i in range(8):
            if time.monotonic() > phase5_end - 5:
                break
            per_dl = min(phase5_end, time.monotonic() + 40.0)
            lr = [0.02, 0.01, 0.03, 0.015, 0.025, 0.012, 0.02, 0.008][i]
            noise = [0.0, 0.001, 0.002, 0.0005, 0.0, 0.001, 0.0005, 0.0][i]
            p = best_seq.copy()
            if noise > 0:
                p = p + noise * rng.standard_normal(n_best)
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            result, sc = logit_adam_optimize(p, per_dl, rng, lr_init=lr,
                                             exact_every=5, restart_patience=200)
            update(result, sc)
            print(f"  Polish {i} (lr={lr}): {sc:.10f}, best={best_score:.10f}")

        # Final coordinate descent
        if time.monotonic() < phase5_end - 30:
            per_dl = min(phase5_end, time.monotonic() + 30.0)
            result, sc = coordinate_descent(best_seq, per_dl, rng, max_sweeps=50)
            update(result, sc)
            print(f"  Final coord descent: {sc:.10f}, best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.maximum(best_seq, 0.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.linspace(0.5, 1.5, 512)
            final_seq = final_seq / final_seq.sum()
    else:
        final_seq = np.linspace(0.5, 1.5, 512)
        final_seq = final_seq / final_seq.sum()

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
