    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):

    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:

    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)

        j_end = min(n-1, k)
