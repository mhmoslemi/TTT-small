"""
Optimization for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines Adam in logit-space with LP-based descent directions and multi-scale search.
Uses soft-max smoothing for the peak, gradient-based polishing, and random restarts.

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max_and_weights(c, beta):
    """Compute soft max and normalize weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(cnt)
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def correlation_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 via FFT."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1].copy(), N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_stable(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -100.0, 100.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e) / float(e.size)
    return e / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b)_k <= M for all k, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def adam_logit_optimize(p0, deadline, rng, lr=0.1, beta_init=None, reg=1e-8):

    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.15 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)

    base_lr = lr
    current_lr = lr
    restart_count = 0

    while time.monotonic() < deadline:

        g_a = 2.0 * correlation_weights(w, p)

            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        m = b1 * m + (1.0 - b1) * g_z

        vh = v / (1.0 - b2 ** adam_t)

            m.fill(0.0)

def lp_refine(f_init, deadline, rng, max_iters=2000):
    """LP-based local refinement."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    best_M = float(full_convolve_fft(f).max())
    best_f = f.copy()
    t = 0.2
    stagnant = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        M = float(full_convolve_fft(f).max())
        if stagnant > 0:
            M = best_M

        g = lp_direction(f, M)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new

        M_new = float(full_convolve_fft(f_new).max())

        if M_new <= best_M + 1e-12:
            f = f_new
            if M_new < best_M - 1e-14:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            it += 1
        else:
            t *= 0.5
            if t < 0.003:
                t = 0.003
            stagnant += 1

        t = max(0.01, t * 0.997)

        if stagnant > 100:
            noise_scale = 0.05 * float(np.std(best_f))
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.2
            stagnant = 0

    return best_f

def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""

            update(result, sc)

    print(f"After Phase 3: best={best_score:.10f}")

    # ========== Phase 4: Try n=1024 ==========
    n_large = 1024
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end - 30 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        print(f"Phase 4: n={n_large}, interpolated score={sc:.10f}")
        update(p_large, sc)

        # Quick LP at large n
        per_dl = min(phase4_end, time.monotonic() + 150.0)
        result_lp_large = lp_refine(p_large, per_dl, rng)
        sc = compute_score(result_lp_large)
        update(result_lp_large, sc)
        print(f"After large LP: score={sc:.10f}")

        # Adam at large n
        if time.monotonic() < phase4_end - 10:
            per_dl = min(phase4_end, time.monotonic() + 100.0)
            result_a, sc = adam_logit_optimize(result_lp_large, per_dl, rng, lr=0.06)
            update(result_a, sc)
            print(f"After large Adam: score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ========== Phase 5: Final fine-tuning ==========
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_final = best_seq.size
        print(f"Phase 5: Final polish at n={n_final}, best={best_score:.10f}")

        per_dl = min(deadline - 5, time.monotonic() + 40.0)
        p0 = best_seq + 0.0005 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()

        # Very fine Adam
        result, sc = adam_logit_optimize(p0, per_dl, rng, lr=0.03, beta_init=0.01)
        update(result, sc)
        print(f"Final fine Adam: score={sc:.10f}")

        # Quick LP polish
        if time.monotonic() < deadline - 3:
            per_dl = min(deadline - 3, time.monotonic() + 15.0)
            result = lp_refine(best_seq, per_dl, rng)
            sc = compute_score(result)
            update(result, sc)

    print(f"After Phase 5: best={best_score:.10f}")

    # ========== Final validation ==========
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_profiles(512, rng)["lin_2.0"]

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_profiles(256, rng)["lin_2.0"]

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not validate: {e}")

    return [float(x) for x in final_seq]
