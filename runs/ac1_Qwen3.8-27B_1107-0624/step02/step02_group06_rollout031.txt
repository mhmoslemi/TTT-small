def solve_lp_flat(f, M, S_target):
    """
    Solve: min t
    s.t. g_i <= t for all i
         conv(f,g)_k <= M for all k
         g >= 0
         sum(g) >= S_target
    """
    n = len(f)
    # Variables: g[0..n-1], t (total n+1 variables)
    # Objective: minimize t => c = [0, 0, ..., 0, 1] (length n+1)
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize t
    
    A_ub = []
    b_ub = []
    
    # Constraint: g_i <= t for all i
    # g_i - t <= 0
    for i in range(n):
        row = np.zeros(n+1)
        row[i] = 1.0
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(0.0)

    A_conv = build_conv_matrix(f)  # (2n-1) x n

        row[:n] = A_conv[k]
        A_ub.append(row)
        b_ub.append(M)
    
    # Constraint: -sum(g) <= -S_target (i.e., sum(g) >= S_target)
    row = np.zeros(n+1)
    row[:n] = -1.0
    A_ub.append(row)

    b_ub = np.array(b_ub)

def coord_ascent(a, max_sweeps=1000, tol=1e-10):
    """
    Monotone coordinate ascent: increase each coordinate to its max feasible value
    given the current M. Monotonically decreases 2nM/S^2.
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    S = a.sum()
    if S < 1e-30:
        return a
    
    conv_full = full_convolve_fft(a)
    M = float(conv_full.max())
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            ai = a[i]
            if ai >= 1000:
                continue
            
            # Compute upper bound on a_i
            ub = 1000.0
            for k in range(2*n-1):
                ki = k - i
                if 0 <= ki < n:
                    if ki == i:
                        # a_i^2 + conv_wo[k] <= M
                        conv_wo_k = conv_full[k] - ai*ai
                        if M > conv_wo_k:

                        else:

                    else:

                            else:
                                ub = 0.0

            if new_ai > ai * (1 + tol):
                # Update a[i] and conv_full
                delta = new_ai - ai
                a[i] = new_ai

                # Update conv_full: conv_full[k] += change

                            conv_full[k] += new_ai*new_ai - ai*ai
                        else:
