def internal_score(a):
    n = len(a)
    s = a.sum()

        return np.inf

    return float(2.0 * n * c.max() / (s * s))


def normalize(a):

    s = a.sum()

def compute_gradient(a, w):
    """
    Compute gradient of max(conv(a,a)) w.r.t. a.
    a has length n, w has length 2n-1 (soft-argmax weights over conv(a,a)).
    Returns grad of shape (n,).
    grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    """
    n = len(a)
    N = 2 * n
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]

    A = np.zeros((2 * n - 1, n))

Uses FFT-based gradient with multi-temperature soft-argmax, LP direction

from scipy.optimize import linprog


def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def compute_grad(a, w):
    """Gradient of max(conv(a,a)) w.r.t. a, using soft-argmax weights w."""
    n = len(a)
    N = 2 * n
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=10.0):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def make_profile_quadratic(n, eta):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u * u

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines FFT-based gradient (multi-temperature soft-argmax with adaptive
temperatures), LP direction finding for local minimum escape, adaptive
restarts, and a final tight polish phase.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def compute_grad(a, w):
    """
    Gradient of max(conv(a,a)) w.r.t. a using soft-argmax weights w.
    grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    Computed via FFT cross-correlation.
    """
    n = len(a)
    N = 4 * n  # Enough to avoid circular correlation artifacts (need >= 3n-2)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=10.0):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g) <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def make_profile_quadratic(n, eta):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u * u
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def make_profile_bimodal(n, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - width) / width) ** 2) + np.exp(-((t - 1.0 + width) / width) ** 2)
    a += 0.01
    return a / a.sum()


def make_profile_sigmoid(n, steepness):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 / (1.0 + np.exp(steepness * u))
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def gradient_optimize(a0, deadline, betas=None, base_lr=0.04, name=""):

    Adam gradient descent in logit space with adaptive multi-temperature soft-argmax.
    a0 should be normalized to sum 1.
    Returns (best_a, best_score) where score = 2n*max(conv(a,a)).

    v = np.zeros(n)
