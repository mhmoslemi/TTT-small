"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses variance minimization as a smooth relaxation, followed by soft-max peak reduction,
with multiple restarts and length exploration.

import numpy as np
import time

def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def grad_sum_c2(a):
    """
    Gradient of sum_k (conv(a,a)[k])^2 with respect to a.
    = 4 * correlate(conv(a,a), a, 'full')[:n]
    """
    n = len(a)
    c = np.convolve(a, a)
    return 4.0 * np.correlate(c, a, 'full')[:n]

def soft_max_grad(c, beta):
    """
    Compute soft-max weights and gradient of soft_max(c, beta) w.r.t. a.
    soft_max(c, beta) = beta * log(sum(exp(c[k]/beta)))
    d(soft_max)/da[i] = sum_k w[k] * dc[k]/da[i] = 2 * sum_k w[k] * a[k-i]
    where w[k] = exp(c[k]/beta) / sum_j exp(c[j]/beta)
    """
    n = len(a)
    c_max = c.max()
    x = (c - c_max) / beta
    ex = np.exp(x)
    w = ex / ex.sum()
    # Gradient w.r.t. a: 2 * sum_k w[k] * a[k-i] for each i
    # = 2 * correlate(w, a, 'full')[:n]
    grad = 2.0 * np.correlate(w, a, 'full')[:n]
    return grad, w

def adam_logit_step(z, grad_z, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """One Adam step in logit space."""
    m = b1 * m + (1 - b1) * grad_z
    v = b2 * v + (1 - b2) * grad_z ** 2
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m, v

def optimize_variance(a, deadline, lr=0.1, verbose=True):
    """
    Minimize sum(c^2) using Adam in logit space.
    a should be normalized (sum=1, non-negative).
    """
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= z.max()  # for numerical stability
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_v = np.sum(np.convolve(a, a)**2)
    no_imp = 0
    
    while time.monotonic() < deadline:
        t += 1
        c = np.convolve(a, a)
        cur_v = np.sum(c**2)
        
        if cur_v < best_v - 1e-12:
            best_v = cur_v
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 5000:
                break
            if no_imp % 1000 == 0:
                lr = max(0.01, lr * 0.9)
        
        # Gradient of sum(c^2) w.r.t. a
        grad_a = grad_sum_c2(a)
        
        # Convert to logit space gradient
        # d(sum(c^2))/dz[i] = d(sum(c^2))/da[i] * da[i]/dz[i]
        # a[i] = exp(z[i] - logsumexp(z)) (approximately, since we use softmax)
        # da[i]/dz[j] = a[i] * (delta(i,j) - a[j])
        # So grad_z[j] = sum_i grad_a[i] * a[i] * (delta(i,j) - a[j])
        # = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])
        
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        
        z, m, v = adam_logit_step(z, grad_z, m, v, t, lr)
        z = np.clip(z, -50, 50)
        
        # Convert back
        z_shift = z - z.max()
        a = np.exp(z_shift)
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
    
    if np.sum(np.convolve(best_a, best_a)**2) < best_v:
        best_v = np.sum(np.convolve(best_a, best_a)**2)
    
    return best_a, best_v

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses variance minimization (smooth) as a relaxation, then soft-max peak reduction,
with multiple restarts, length exploration, and LP-assisted direction finding.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def grad_sum_c2(a):
    """
    Gradient of sum_k (conv(a,a)[k])^2 w.r.t. a.
    = 4 * correlate(conv(a,a), a, 'full')[:n]
    """
    n = len(a)
    c = np.convolve(a, a)
    return 4.0 * np.correlate(c, a, 'full')[:n]


def soft_max_weights(c, beta):
    """Compute soft-max weights w[k] proportional to exp(c[k]/beta)."""
    c_max = c.max()
    x = (c - c_max) / beta
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return ex / s


def soft_max_grad_a(c, a, beta):
    """
    Gradient of soft_max(c, beta) w.r.t. a.
    = 2 * correlate(w, a, 'full')[:n]
    where w are the soft-max weights.
    """
    n = len(a)
    w = soft_max_weights(c, beta)
    grad = 2.0 * np.correlate(w, a, 'full')[:n]
    return grad, w


def to_logit(a):
    """Convert probability vector to logit (log of ratio)."""
    a = np.maximum(a, 1e-15)
    z = np.log(a)
    z -= z.max()
    return z


def from_logit(z):
    """Convert logit to probability vector."""
    z = z - z.max()
    p = np.exp(z)
    s = p.sum()
    if s < 1e-30:
        return None
    return p / s


def logit_grad(grad_a, a):
    """
    Convert gradient w.r.t. a to gradient w.r.t. logit z.
    grad_z[j] = a[j] * (grad_a[j] - dot(a, grad_a))
    """
    dot = np.dot(a, grad_a)
    return a * (grad_a - dot)


def adam_state_init(n):
    """Initialize Adam state."""
    return np.zeros(n), np.zeros(n), 0


def adam_step(z, grad_z, state, lr, b1=0.9, b2=0.999, eps=1e-8):
    """One Adam step. Returns (z_new, state_new)."""
    m, v, t = state
    t += 1
    m = b1 * m + (1 - b1) * grad_z
    v = b2 * v + (1 - b2) * grad_z ** 2
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = np.clip(z_new, -50.0, 50.0)
    return z_new, (m, v, t)


def optimize_variance(a, deadline, lr=0.1, reg=0.0, verbose=False):

    Minimize sum(c^2) + reg*sum(a^2) using Adam in logit space.
    a should be normalized (sum=1, non-negative).
    Returns (best_a, best_sum_c2).
    """

    a = np.maximum(a, 1e-12).astype(np.float64)

    z = to_logit(a)
    state = adam_state_init(n)

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_initial_candidates(n, rng):
    """Generate diverse initial sequences."""
    seqs = []
    t = np.linspace(0.0, 1.0, n)

    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, -1.5, -2.0]:
        u = 2.0 * t - 1.0
        a = 1.0 + eta * u
        a = np.maximum(a, 1e-10)
        seqs.append(a / a.sum())

    # Quadratic
    for c1, c2 in [(0.5, -0.3), (1.0, -0.5), (1.5, -0.8), (-0.5, 0.3),
                   (0.3, -1.0), (2.0, -1.2), (-1.0, 0.8)]:
        u = 2.0 * t - 1.0
        a = 1.0 + c1 * u + c2 * u ** 2
        a = np.maximum(a, 1e-10)
        seqs.append(a / a.sum())

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.5, 2.0, 3.0]:
        a = t ** p + 0.01
        seqs.append(a / a.sum())
        a = (1.0 - t) ** p + 0.01
        seqs.append(a / a.sum())

    # Sine-based
    for freq, amp in [(1, 0.3), (1, 0.5), (2, 0.3), (2, 0.5), (3, 0.3), (1, 0.7)]:

        a = np.maximum(a, 1e-10)
        seqs.append(a / a.sum())

    # Cosine-based
    for freq, amp in [(1, 0.3), (2, 0.3), (1, 0.5)]:
        a = 1.0 + amp * np.cos(freq * np.pi * t)
        a = np.maximum(a, 1e-10)
        seqs.append(a / a.sum())

    # Random smooth (random Fourier)
    for _ in range(8):
        coeffs = rng.normal(size=5) * 0.3

    # Palindromic random
    for _ in range(4):

        h = rng.random(half) + 0.1

def lp_direction(f, M, timeout=10.0):
    """
    Solve LP: maximize sum(g) s.t. g>=0, convolve(f,g)[x] <= M.
    """
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def lp_refine(f, deadline, max_iters=50, verbose=False):
    """LP-based refinement from a starting point."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    t_step = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=min(15.0, max(2.0, deadline - time.monotonic() - 1)))
        if g_raw is None:
            break

        # Try blended step
        for t_try in [t_step, t_step * 0.5, 0.8, 1.0]:
            f_new = (1.0 - t_try) * f + t_try * g_raw
            f_new = normalize(f_new)
            if f_new is None:
                continue
            c_new = np.convolve(f_new, f_new)
            M_new = float(c_new.max())
            score_new = 2.0 * n * M_new

            if score_new < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = score_new
                f = f_new.copy()
                M = M_new
                break

        if verbose and it % 10 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}")

    return best_f, best_score

    phase2_results = []
    for i, cand in enumerate(top_candidates):
        if time.monotonic() > phase2_end - 1:
            break
        dl = min(time.monotonic() + time_per_cand2, phase2_end)

        # First variance pass, then soft-max
        a_var, _ = optimize_variance(cand, min(time.monotonic() + 20, dl), lr=0.08)
        a_sm, sc = optimize_soft_max(a_var, dl, beta_start=0.05, beta_end=0.0005, lr=0.04)
        sc = internal_score(a_sm)
        phase2_results.append((sc, a_sm))
        if sc < best_score:
            best_score = sc
            best_seq = a_sm.copy()
            print(f"  Phase 2 cand {i}: score={sc:.8f}")

    phase2_results.sort(key=lambda x: x[0])
    if phase2_results:
        print(f"Phase 2 best: score={phase2_results[0][0]:.8f}")

    # Phase 3: Combined objective on top (200s)
    phase3_end = t_start + 1200.0
    if time.monotonic() < phase3_end - 10:
        time_per_cand3 = max(10.0, (phase3_end - time.monotonic()) / max(1, top_k))
        print(f"Phase 3: combined obj, ~{time_per_cand3:.1f}s per candidate")
        for i, cand in enumerate(top_candidates[:3]):
            if time.monotonic() > phase3_end - 1:
                break
            dl = min(time.monotonic() + time_per_cand3, phase3_end)
            a_comb, sc = optimize_combined(cand, dl, alpha=0.3, beta_start=0.03,
                                           beta_end=0.0005, lr=0.04)
            sc = internal_score(a_comb)
            if sc < best_score:
                best_score = sc
                best_seq = a_comb.copy()
                print(f"  Phase 3 cand {i}: score={sc:.8f}")

    # Phase 4: Length exploration (150s)
    phase4_end = t_start + 1450.0
    if time.monotonic() < phase4_end - 10:
        print(f"Phase 4: length exploration")
        for n_try in [384, 640, 768, 1024]:
            if time.monotonic() > phase4_end - 15:
                break
            a_interp = interpolate_seq(best_seq, n_try)
            if a_interp is None:
                continue
            time_budget = min(30.0, phase4_end - time.monotonic() - 5)
            if time_budget < 5:
                break
            a_opt, _ = optimize_variance(a_interp, time.monotonic() + time_budget * 0.4, lr=0.08)
            a_opt, sc = optimize_soft_max(a_opt, time.monotonic() + time_budget * 0.6,
                                          beta_start=0.03, beta_end=0.001, lr=0.03)
            sc = internal_score(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Phase 4 n={n_try}: score={sc:.8f}")

    # Phase 5: Final long soft-max polish (remaining time)
    final_deadline = deadline
    if time.monotonic() < final_deadline - 20:
        n_final = len(best_seq)
        print(f"Phase 5: final polish at n={n_final}, remaining={final_deadline - time.monotonic():.0f}s")

        # Multiple short runs with different betas
        remaining = final_deadline - time.monotonic() - 20
        n_runs = max(1, int(remaining // 60))
        run_time = min(80.0, remaining / max(1, n_runs))

        for run_i in range(n_runs):
            if time.monotonic() > final_deadline - 20:
                break
            dl = time.monotonic() + run_time
            beta_s = max(0.01, 0.03 * (0.7 ** run_i))
            beta_e = max(0.0001, 0.001 * (0.5 ** run_i))
            lr_try = 0.03 * (0.9 ** run_i)

            a_opt, sc = optimize_soft_max(best_seq, dl, beta_start=beta_s,
                                          beta_end=beta_e, lr=lr_try)
            sc = internal_score(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Phase 5 run {run_i}: score={sc:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = np.ones(512, dtype=np.float64) / 512.0
        s = 1.0
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
