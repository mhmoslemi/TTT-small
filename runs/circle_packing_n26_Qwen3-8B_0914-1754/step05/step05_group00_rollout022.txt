import cvxpy as cp

    cols = 5

            x = j * 0.2
            y = i * 0.2 + (j % 2) * 0.1
            if i % 2 == 0:
                grid[i, j, 0] = x

            else:

    hex_centers = hex_centers[:26]

    x0_centers = np.array(hex_centers).flatten()

    x0_radii = np.full(n, 0.1)

    optimal_centers = result_positions.x.reshape(n, 2)

    radii = cp.Variable(n)
