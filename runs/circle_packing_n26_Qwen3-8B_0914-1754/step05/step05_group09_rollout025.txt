        radii = x[n * 2:]
        return -np.sum(radii)

    constraints = []

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[n * 2 + i]})

    for i in range(n):
        for j in range(i + 1, n):

                centers = centers_flat.reshape(n, 2)
                radii = x[n * 2:]

                dy = centers[i, 1] - centers[j, 1]
                return np.sqrt(dx**2 + dy**2) - (radii[i] + radii[j])
            constraints.append({'type': 'ineq', 'fun': dist_constraint})

            centers_flat = x[:n * 2]
