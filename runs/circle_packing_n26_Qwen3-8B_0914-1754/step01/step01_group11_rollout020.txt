import scipy.optimize as optimize
import random

def run_packing():

    centers = np.random.rand(26, 2)
    # Create a boundary circle near each corner
    corner_centers = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.9, 0.9], [0.1, 0.9]
    ])
    centers[:4] = corner_centers
    # Create boundary circles along edges

        radii_flat = x[2*26:]

        for i in range(26):

                return 1e10
            if centers_flat[i, 1] < -1e-12 or centers_flat[i, 1] > 1 + 1e-12:
                return 1e10
            if radii_flat[i] < 0:
                return 1e10

        for i in range(26):

                dy = centers_flat[i, 1] - centers_flat[j, 1]
