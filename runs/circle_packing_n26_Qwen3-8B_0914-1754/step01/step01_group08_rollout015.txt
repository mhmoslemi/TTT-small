import scipy.optimize as optimize

def run_packing():

    initial_centers = np.random.rand(num_circles, 2)
    initial_radii = np.full(num_circles, 0.05)

    # Function for handling the optimization
    def objective(x):
        centers_flat = x[:2 * num_circles].reshape(num_circles, 2)
        radii_flat = x[2 * num_circles:]

        # Penalty function
        penalty = 0
        for i in range(num_circles):
            # Check if circle is outside the boundary [0,1]^2
            if centers_flat[i, 0] < -1e-12 or centers_flat[i, 0] > 1 + 1e-12:
                penalty += 1e6
            if centers_flat[i, 1] < -1e-12 or centers_flat[i, 1] > 1 + 1e-12:
                penalty += 1e6

            # Check if radius is negative or too small
            if radii_flat[i] < 0:
                penalty += 1e6 * (-radii_flat[i])
            if radii_flat[i] < 1e-6:
                penalty += 1e6 * (1e-6 - radii_flat[i])

        # Check for overlaps between all pairs of circles
        for i in range(num_circles):

        # Encourage boundary usage by placing circles not too far from edges

        # We scale it to encourage maximization of the total radius
