import numpy as np

def run_packing():

    # The distance between adjacent circles will be twice the radius
    centers_hex = []
    for i in range(3):
        for j in range(3):
            # Hexagonal grid for the center 12 circles
            if i % 2 == 0:
                x = 0.25 + j * 0.3
            else:
                x = 0.25 + j * 0.3 + 0.15
            y = 0.25 + i * 0.3
            centers_hex.append([x, y])
    centers_hex = np.array(centers_hex)

    # Define 14 boundary and corner circles
    # These will be placed near the boundaries and corners to fill the gaps
    centers_boundary = [
        [0.05, 0.5],    # Left edge

        [0.1, 0.7],     # Additional corner cluster
        [0.1, 0.9],     # Additional corner cluster
        [0.3, 0.1],     # Additional corner cluster
    ]
    centers_boundary = np.array(centers_boundary)

    # Combine all centers
    centers = np.vstack((centers_hex, centers_boundary))

    # Compute radii based on distance to boundaries and neighbor circles
    radii = []
    for i in range(centers.shape[0]):
        x, y = centers[i]
        # Distance to nearest boundary
        dist_to_boundary = np.min([x, 1 - x, y, 1 - y])
        # Distance to nearest circle
        dist_to_neighbors = np.inf

                dist_to_neighbors = np.min([dist_to_neighbors, dist])
        # Choose minimum of boundary and neighbor distance to ensure non-overlap
        r = np.min([dist_to_boundary, dist_to_neighbors / 2])
        radii.append(r)

    radii = np.array(radii)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
