def run_packing():

    centers = []
    for r in range(4):
        for c in range(6 + (r % 2)):
            x = c * 0.15 + (r % 2) * 0.075
            y = r * 0.18
            # Clip to the unit square
            x = np.clip(x, 0.01, 0.99)
            y = np.clip(y, 0.01, 0.99)
            centers.append([x, y])

    # Ensure we have 26 circles
    centers = centers[:n]

    # Define the optimization objective
    model = pyo.ConcreteModel()

    # Define variables
    model.radii = pyo.Var(range(n), domain=pyo.NonNegativeReals)
    model.x = pyo.Var(range(n), domain=pyo.Continuous)
    model.y = pyo.Var(range(n), domain=pyo.Continuous)

    # Objective: Maximize the sum of radii
    model.obj = pyo.Objective(
        expr=sum(model.radii[i] for i in range(n)),
        sense=pyo.maximize
    )

    # Constraints for boundaries
    for i in range(n):
        model.bound_left = pyo.Constraint(expr=model.x[i] - model.radii[i] >= 0)
        model.bound_right = pyo.Constraint(expr=1 - model.x[i] - model.radii[i] >= 0)
        model.bound_bottom = pyo.Constraint(expr=model.y[i] - model.radii[i] >= 0)
        model.bound_top = pyo.Constraint(expr=1 - model.y[i] - model.radii[i] >= 0)

    # Constraints for circle-circle distance
    model.dist = pyo.ConstraintList()
    for i in range(n):
        for j in range(i + 1, n):
            model.dist.add(
                pyo.Constraint(
                    expr=np.sqrt((model.x[i] - model.x[j])**2 + (model.y[i] - model.y[j])**2)
                    >= model.radii[i] + model.radii[j] - 1e-10
                )
            )

    # Initial guesses from the centers
    for i in range(n):
        model.x[i].set_value(centers[i][0])
        model.y[i].set_value(centers[i][1])

    # Solve the problem with IPOPT
    solver = SolverFactory("ipopt")
    solver.options["print_timing"] = "yes"
    solver.options["max_iter"] = 500
    solver.options["acceptable_tol"] = 1e-9

    return centers, radii, sum_radii
