from scipy.optimize import minimize

def run_packing():

    # Function to compute radii for given centers
    def compute_radii(centers):
        radii = []
        for i in range(n):
            x, y = centers[i]
            max_radius = 0
            # Distance to the nearest boundary
            dist_to_boundary = np.min([x, 1 - x, y, 1 - y])
            max_radius = dist_to_boundary
            # Distance to the nearest circle
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < max_radius:
                    max_radius = dist
            # Radius is half of the smallest of the two
            max_radius = max_radius / 2
            radii.append(max_radius)
        return np.array(radii)

    # Function to maximize total radius sum
    def total_radius_sum(centers_flat):

            y = row / 5 + (row % 2) * 0.2
