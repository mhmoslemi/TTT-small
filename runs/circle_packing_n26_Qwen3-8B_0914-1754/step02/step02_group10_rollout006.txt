import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26

    def compute_radii(centers):
        radii = []
        for i in range(n):
            x, y = centers[i]
            max_radius = 0

            # Distance to nearest boundary
            dist_to_boundary = min(x, 1 - x, y, 1 - y)
            max_radius = dist_to_boundary

            # Distance to nearest circle
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < max_radius:
                    max_radius = dist

            max_radius /= 2
            radii.append(max_radius)
        return np.array(radii)

    def total_radius_sum(centers_flat):
        centers = centers_flat.reshape((n, 2))
        radii = compute_radii(centers)
        return -np.sum(radii)
