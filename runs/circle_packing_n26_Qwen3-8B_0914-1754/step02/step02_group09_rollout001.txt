import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26

    # Generate an initial hexagonal-like grid inside the unit square
    def generate_hexagonal_layout(n):
        # Create an initial grid where we place circles in a hexagonal pattern
        # Scale to fit in [0,1]^2
        centers = []
        for i in range(n):
            x = i / (n - 1)
            y = (i % 2) / 2
            centers.append([x, y])
        return np.array(centers)

    # Define a function to compute safe radii for given centers
    def compute_radii(centers):
        radii = []
        for i in range(n):
            x, y = centers[i]
            max_radius = 0
            # Distance to the nearest boundary
            dist_to_boundary = min(x, 1 - x, y, 1 - y)
            max_radius = dist_to_boundary
            # Distance to the nearest circle
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < max_radius:
                    max_radius = dist
            max_radius = max_radius / 2
            radii.append(max_radius)
        return np.array(radii)

    # Define a function to maximize the total radius sum
    def total_radius_sum(centers_flat):
        centers = centers_flat.reshape((n, 2))
        radii = compute_radii(centers)
        return -np.sum(radii)  # Negative sum to maximize via minimization

    # Generate initial hexagonal-like layout
    initial_centers = generate_hexagonal_layout(n)

    # Flatten initial guess for optimization
    initial_guess = initial_centers.flatten()

    # Set up optimization
    bounds = [(0, 1) for _ in range(2 * n)]
    result = minimize(
        total_radius_sum,
        initial_guess,
        bounds=bounds,
        method="L-BFGS-B",
        options={"ftol": 1e-10, "maxiter": 1000}
    )

    # Extract final centers and radii
    centers = result.x.reshape((n, 2))
    radii = compute_radii(centers)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
