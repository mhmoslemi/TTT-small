    # Compute radii considering minimal distance to boundary and nearest neighbor
    radii = []
    for i in range(len(centers)):
        x, y = centers[i]
        # Distance to nearest boundary
        dist_to_boundary = np.min([x, 1 - x, y, 1 - y])
