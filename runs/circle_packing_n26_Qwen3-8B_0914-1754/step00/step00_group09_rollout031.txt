import numpy as np

from scipy.optimize import minimize

def compute_forces(centers, radii, n):
    forces = np.zeros((n, 2))
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            if dist < 1e-12:
                continue
            repel = (radii[i] + radii[j]) * (radii[i] + radii[j]) / (dist ** 3)
            forces[i, 0] -= dx * repel
            forces[i, 1] -= dy * repel
            forces[j, 0] += dx * repel
            forces[j, 1] += dy * repel
    return forces

def objective(params, n):

    radii = params[n * 2:]

    for i in range(n):
        x, y = centers[i]

        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            return float('inf')
    
    # Compute forces
    forces = compute_forces(centers, radii, n)
    
    # Compute total repulsion
    total_repel = np.sum(np.abs(forces))
    return total_repel

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
