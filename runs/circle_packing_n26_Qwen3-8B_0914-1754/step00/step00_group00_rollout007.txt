        [0.5, 0.9],
        [0.7, 0.9],

        [0.1, 0.95],
        [0.3, 0.95],
        [0.5, 0.95],
    ])

    anchor = np.array([0.5, 0.5])

    centers = np.vstack([layer1, layer2, layer3, anchor])

    n = centers.shape[0]
    r = cp.Variable(n)
    x = cp.Variable(n)
    y = cp.Variable(n)

    # Objective function: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints:
    # 1. All circles must be within [0, 1] x [0, 1]
    constraints = [
        x >= 0,
        x <= 1,
        y >= 0,
        y <= 1
    ]

    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)

    # 3. All radii must be non-negative
    constraints.append(r >= 0)

    # Solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve()

    # Extract results
    centers_result = np.column_stack([x.value, y.value])
    radii_result = r.value

    # Compute total sum of radii
    sum_radii = np.sum(radii_result)
