import numpy as np
import cvxpy as cp

def run_packing():

    # We'll divide the square into rows and place centers accordingly
    # We'll use 4 rows to ensure circles remain within the square

    # Number of circles = 26
    # Number of rows = 4
    # Number of circles per row varies from 7 to 6

    centers = []
    for row in range(4):
        if row % 2 == 0:
            # Even row: more circles to the left
            num_circles = 7
            offset = 0.15
        else:
            # Odd row: fewer circles to the left
            num_circles = 6
            offset = 0.2

        # Calculate the horizontal spacing between circles
        horizontal_spacing = 1.0 / (num_circles - 1)
        # Vertical spacing between rows
        vertical_spacing = np.sqrt(3) / 2 * horizontal_spacing

        for col in range(num_circles):
            x = col * horizontal_spacing
            y = row * vertical_spacing

            # Apply an offset to shift the circle slightly horizontally
            # This helps avoid alignment with adjacent rows
            x += offset if row % 2 == 1 else -offset

            # Ensure the center is within the square
            x = np.clip(x, 0.0, 1.0)
            y = np.clip(y, 0.0, 1.0)

            centers.append([x, y])

    centers = np.array(centers)
    n = centers.shape[0]

    # Step 2: Initial guess for radii based on spacing
    # Assume we want radii close to (1 / sqrt(2)) / 26 to use the most space
    # Initialize them to a small value, then optimize
    radii = np.full(n, 0.03)

    # Step 3: Optimize using CVXPy to maximize the sum of radii

    # Objective: Maximize sum of radii
    # Constraints: Each circle must not overlap with another
    #              Each circle must not exceed the square boundary

    # Create variables for radii (we'll fix the centers since they're already chosen)
    r = cp.Variable(n)

    # Objective: Maximize the total area (i.e., sum of radii)
    objective = cp.Maximize(cp.sum(r))

    # Define the constraints
    constraints = []

    # 1. Boundary constraint: circles must fit entirely within the unit square
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - r[i] >= -1e-12)
        constraints.append(x + r[i] <= 1.0 + 1e-12)
        constraints.append(y - r[i] >= -1e-12)
        constraints.append(y + r[i] <= 1.0 + 1e-12)

    # 2. Circle-collision constraint: distance between centers ≥ r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraint = dist >= r[i] + r[j] - 1e-12
            constraints.append(constraint)

    # Solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve()

    # Extract the optimized radii
    radii = r.value

    # Step 4: Return the result
    return (centers, radii, np.sum(radii))
