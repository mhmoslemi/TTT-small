import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
from typing import Tuple, List

def pack_in_grid(n: int, square_size: float = 1.0) -> np.ndarray:
    """
    Generate a grid of centers for n circles within the square.
    """
    grid = []
    spacing = np.sqrt(n) / (np.sqrt(2) * square_size)  # Based on hexagonal grid spacing
    for i in range(int(np.sqrt(n)) + 1):
        for j in range(int(np.sqrt(n)) + 1):
            x = i * spacing
            y = j * spacing
            grid.append([x, y])
    grid = np.array(grid, dtype=np.float64)
    # Remove any points that are outside the square
    grid = grid[(grid[:, 0] < square_size) & (grid[:, 1] < square_size)]
    return grid[:n]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def refine_packing(centers, square_size: float = 1.0) -> np.ndarray:

    def constraint(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)

                    return -1
        # Check boundaries

            r = radii[i]

        return 0

    # Add bounds to radii
    bounds = [(0, 1) for _ in range(n)]

    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint}]
    )

    if not result.success:
        raise ValueError("Optimization failed with message: " + result.message)

    return result.x

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    square_size = 1.0
    centers = pack_in_grid(n, square_size)

    sum_radii = np.sum(radii)

    # Validate the packing again to ensure boundaries and non-overlap
    if not validate_packing(centers, radii)[0]:

    return centers, radii, sum_radii
