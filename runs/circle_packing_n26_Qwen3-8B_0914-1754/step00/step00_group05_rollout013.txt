import numpy as np
import cvxpy as cp

def run_packing():
    # Initial guess: 5 rows of 5 circles in a staggered (hexagonal) layout
    centers = np.array([
        [0.25, 0.25], [0.75, 0.25], [0.25, 0.5], [0.75, 0.5], [0.25, 0.75], [0.75, 0.75],
        [0.25, 0.25], [0.75, 0.25], [0.25, 0.5], [0.75, 0.5], [0.25, 0.75], [0.75, 0.75],
        [0.25, 0.25], [0.75, 0.25], [0.25, 0.5], [0.75, 0.5], [0.25, 0.75], [0.75, 0.75],
        [0.25, 0.25], [0.75, 0.25], [0.25, 0.5], [0.75, 0.5], [0.25, 0.75], [0.75, 0.75]
    ])

    # Optimization variables
    radii = cp.Variable(25)  # We'll add one more circle at the center later
    r_center = cp.Variable()  # Additional circle at center

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii) + r_center)

    # Constraints: All circles inside the unit square
    constraints = []
    for i in range(25):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-6)
        constraints.append(x + radii[i] <= 1.0 + 1e-6)
        constraints.append(y - radii[i] >= -1e-6)
        constraints.append(y + radii[i] <= 1.0 + 1e-6)

    # Additional circle at center
    constraints.append(0.5 - r_center >= -1e-6)
    constraints.append(0.5 + r_center <= 1.0 + 1e-6)

    # No overlap between circles
    for i in range(25):
        for j in range(i + 1, 25):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(radii[i] + radii[j] <= dist + 1e-6)

    # No overlap between additional circle and others
    for i in range(25):
        dist = np.sqrt((centers[i, 0] - 0.5)**2 + (centers[i, 1] - 0.5)**2)
        constraints.append(radii[i] + r_center <= dist + 1e-6)

    # Solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS)

    # Check solution
    if not prob.status in [cp.OPTIMAL, cp.SOLUTION_PRESENT]:
        raise ValueError("Optimization did not converge")

    # Extract final centers and radii
    final_centers = np.vstack([centers, [0.5, 0.5]])
    final_radii = np.append(radii.value, r_center.value)
    total_sum = np.sum(final_radii)

    return final_centers, final_radii, total_sum
