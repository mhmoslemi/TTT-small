def run_packing():

    num_circles = 26
    grid_size = 5
    positions = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i / (grid_size - 1)
            y = j / (grid_size - 1)
            positions.append([x, y])
    # Add a central circle
    positions.append([0.5, 0.5])
    centers = np.array(positions)
    
    # Step 2: Initial radii guess based on grid spacing
    def get_initial_radii(centers):

            min_dist_to_neighbor = float('inf')

                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    min_dist_to_neighbor = min(min_dist_to_neighbor, dist)
            radius = min(min_dist_to_boundary, min_dist_to_neighbor / 2)
            radii.append(radius)
        return np.array(radii)

    radii = get_initial_radii(centers)

    # Step 3: Optimization function
    def objective(x, centers, radii):
        # x is a flat array of variables: centers and radii
        # Extract centers and radii from x
        num_centers = centers.shape[0]
        num_radii = len(radii)
        flat_centers = x[:num_centers * 2].reshape(num_centers, 2)
        flat_radii = x[num_centers * 2:].reshape(num_radii)
        
        # Compute sum of radii
        return -np.sum(flat_radii)  # Minimize negative of sum

    def constraints(x, centers, radii):
        num_centers = centers.shape[0]
        num_radii = len(radii)

        # Boundary constraints: x - r >= 0 and x + r <= 1

            r_i = flat_radii[i]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
