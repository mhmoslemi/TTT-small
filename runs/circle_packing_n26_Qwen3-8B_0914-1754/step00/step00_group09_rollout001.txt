import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Define initial positions (grid pattern with some modifications)
    num_circles = 26
    initial_positions = []
    
    # Create a hybrid grid: 4 rows of 6 circles, and 2 rows of 5 circles
    for row_idx in range(4):
        for col_idx in range(6):
            x = col_idx / 5.0
            y = row_idx / 4.0
            initial_positions.append([x, y])
    
    for row_idx in range(2):
        for col_idx in range(5):
            x = col_idx / 4.0
            y = row_idx / 2.0 + 0.5
            initial_positions.append([x, y])
    
    initial_centers = np.array(initial_positions)
    
    # Optimization: maximize the sum of radii given constraints
    def objective(radii):
        return -np.sum(radii)  # Use negative for maximization
    
    def constraint1(radii):
        return np.sum(radii)  # Radius must be non-negative
    
    def constraint2(radii):
        return np.sum(radii)  # To encourage maximizing
    
    bounds = [(0, 1) for _ in range(num_circles)]
    
    result = opt.minimize(
        fun=objective,
        x0=np.full(num_circles, 0.1),
        bounds=bounds,
        constraints=[
            {"type": "ineq", "fun": constraint1},
            {"type": "ineq", "fun": constraint2}
        ]
    )

    centers = initial_centers

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
