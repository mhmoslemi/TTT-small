import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    The algorithm uses a grid to approximate a hexagonal packing.
    """
    # Generate a 5x5 grid of centers
    grid_centers = []
    for i in range(5):
        for j in range(5):
            x = i / 4
            y = j / 4
            grid_centers.append([x, y])
    centers = np.array(grid_centers)
    n = len(centers)

    # Start with all circles of the same radius, and optimize for the total sum
    # Initialize with a uniform radius
    initial_r = 0.1

    # Optimization: maximize the minimum radius, while ensuring all circles are within the square
    # and no two circles overlap.

    # Setup the optimization problem:
    # Objective: maximize the minimal radius (sum of radii)
    # Constraints:
    # - Each circle is inside the square
    # - No overlap between circles

    # Define radius variable
    r = np.full(n, initial_r)

    # Optimization using gradient-based optimization
    def objective(r):
        return -np.sum(r)  # We want to maximize sum(r), so minimize -sum(r)

    def constraint_circle_inside(r, i):
        # Circle i must be inside the unit square
        # x and y are determined by the grid

        y_i = centers[i, 1]

        return (x_i - r_i >= 0) & (x_i + r_i <= 1) & (y_i - r_i >= 0) & (y_i + r_i <= 1)

    def constraint_no_overlap(r, i, j):

        dx = x_i - x_j
        dy = y_i - y_j
