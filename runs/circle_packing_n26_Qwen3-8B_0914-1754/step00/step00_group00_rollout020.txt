import numpy as np
import scipy.optimize as optimize
from scipy.spatial.distance import pdist, squareform
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: 26 circles arranged in a hexagonal grid with varying spacing
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initialize in a staggered hexagonal layout
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)
        centers[i] = [x, y]
        radii[i] = 0.02  # Initial small radii

    def objective(radii):
        return -np.sum(radii)  # We want to maximize radii, so minimize the negative

    # Constraints:
    # 1. Boundaries: all circles must fit inside [0,1]^2
    # 2. No overlaps: distance between centers must be at least sum of radii

    # Bounds on radii
    bounds = [(0, 1) for _ in range(n)]

    # Optimization: this is a difficult constrained optimization problem
    # Using Sequential Least Squares Programming (SLSQP) for bounded optimization
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            # Boundary constraint: center_x - radius >= 0 and center_x + radius <= 1
            # (We'll implement this as a "soft constraint" in the objective)
            # Distance between circles must be at least sum of radii
            {
                'type': 'ineq',
                'fun': lambda r: np.min(
                    np.sqrt(np.sum((centers[i] - centers[j])**2) - r[i] - r[j])
                    for i in range(n)
                    for j in range(i + 1, n)
                )
            }
        ],
        method='SLSQP'
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    radii = result.x
    sum_radii = np.sum(radii)
    # Final validation
    valid, _ = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError("Packing is invalid")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
