import numpy as np
import random

def run_packing():
    # Constants
    NUM_CIRCLES = 26
    MAX_GENERATIONS = 100
    POPULATION_SIZE = 50
    MUTATION_RATE = 0.1
    MUTATION_STRENGTH = 0.1
    NUM_BEST_TO_CARRY_FORWARD = 10

    def fitness_function(centers, radii):
        # Check for out-of-bound circles
        for i in range(NUM_CIRCLES):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return 0.0

        for i in range(NUM_CIRCLES):
            for j in range(i + 1, NUM_CIRCLES):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return 0.0
        # Calculate total radius
        return np.sum(radii)

        def hexagonal_positions(n):
            rows = int(np.ceil(n / 3))
            cols = 3

                    if row * cols + col < n:
                        x = col / (cols - 1) if cols > 1 else 0.5

                        positions.append((x, y))
            return np.array(positions)

        return initial_centers, initial_radii

            if random.random() < MUTATION_RATE:
                centers[i, 0] += np.random.uniform(-MUTATION_STRENGTH, MUTATION_STRENGTH)
