from scipy.optimize import Bounds, minimize

        radii_flat = x[26*2:]

        return -sum_radii(radii)

        bounds.append((0, 1))  # radius
    bounds = Bounds(bounds)

    # This is a nonlinear constraint and can't be directly expressed in scipy.optimize
    # So we use a penalty function to enforce this

    # Define penalty function
    def penalty(x):
        centers_flat = x[:26*2]
        radii_flat = x[26*2:]
        centers = centers_flat.reshape((26, 2))

        for i in range(26):
            x, y = centers[i]
            r = radii[i]
