import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Define initial positions and radii
    initial_centers = np.random.rand(26, 2) * 0.9 + 0.05
    initial_radii = np.ones(26) * 0.05

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint functions
    def boundary_constraint(radii):
        return np.sum((initial_centers - radii[:, np.newaxis]) ** 2) - 1.0 + 1e-8

    def non_overlap_constraints(radii):
        dist_matrix = squareform(pdist(initial_centers))
        return np.sum(dist_matrix - (radii[:, np.newaxis] + radii[np.newaxis, :])) - 1e-8

    # Set up bounds for radii (non-negative)
    bounds = Bounds(0, 1)

    # Optimization setup
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        method='SLSQP',
        constraints=[
            {'type': 'ineq', 'fun': boundary_constraint},
            {'type': 'ineq', 'fun': non_overlap_constraints}
        ]
    )

    centers = initial_centers
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
