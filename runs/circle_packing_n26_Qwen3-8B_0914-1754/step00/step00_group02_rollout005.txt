import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26
    # Initialize centers and radii with random positions
    np.random.seed(42)  # For reproducibility
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1  # Small initial radii

    # We will use CVXPY to perform a constrained optimization to maximize the sum of radii
    # with non-overlapping and boundary constraints
    
    # Define variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n, nonneg=True)

    # Objective: maximize the sum of radii
    obj = cp.Maximize(cp.sum(r))

    # Constraints: boundaries
    for i in range(n):
        x_i = x[i]
        r_i = r[i]
        constraints = [
            x_i[0] - r_i >= 0,
            x_i[0] + r_i <= 1,
            x_i[1] - r_i >= 0,
            x_i[1] + r_i <= 1
        ]
        # Add constraints for each circle
        for constr in constraints:
            obj.constraints.append(constr)

    # Non-overlap constraints between circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt(cp.sum_squares(x[i] - x[j]))
            constraints = [
                dist >= r[i] + r[j]
            ]
            for constr in constraints:
                obj.constraints.append(constr)

    # Solve the problem
    problem = cp.Problem(obj, [])
    problem.solve(solver=cp.SCS, eps=1e-6)

    # Extract results
    centers = x.value
    radii = r.value
    sum_radii = np.sum(radii)

    # Return results
    return (centers, radii, sum_radii)
