import scipy.optimize as optimize

def run_packing():

        return -np.sum(radii)

    def constraint(x):

        radii = x[-n:]

                if dist < radii[i] + radii[j] - 1e-8:
                    return -1  # Violated non-overlap constraint
        # Boundary constraints
        for i in range(n):

        return 0  # All constraints satisfied
