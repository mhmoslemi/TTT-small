import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial configuration of 26 circles in a grid-like arrangement
    # This is a manually designed starting point with equal radii
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.full(num_circles, 0.05)  # Initial guess for radius
    
    # Define a symmetric grid-like configuration (for hexagonal packing)
    # Let's define 6 rows of circles with varying column counts
    row_height = 0.5
    col_width = 0.5
    row_offset = 0.25
    col_offset = 0.25
    row_radii = [3, 4, 4, 4, 3]  # Number of circles in each row
    
    for row_idx, row_count in enumerate(row_radii):
        for col_idx in range(row_count):
            x = col_offset + col_idx * col_width
            y = row_offset + row_idx * row_height
            centers[row_idx * row_count + col_idx] = [x, y]
    
    # Define the objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Negative for minimization in scipy

    # Define constraints: boundary and non-overlap
    def constraint_boundary(x):
        return np.sum((x - radii) ** 2)  # Ensure circles are within the square

    def constraint_overlaps(x):
        dists = np.zeros((num_circles, num_circles))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                x1 = centers[i][0]
                y1 = centers[i][1]
                x2 = centers[j][0]
                y2 = centers[j][1]
                dists[i, j] = np.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)
        return np.min(dists - (x[i] + x[j] for i, j in zip(range(num_circles), range(num_circles))))  # Non-overlapping constraint

    # Set up the optimization problem
    bounds = [(0, 1) for _ in range(num_circles)]  # Radii can't be negative or exceed 1
    constraints = [
        {'type': 'ineq', 'fun': lambda x: np.min(x)},  # Radii must be non-negative
        {'type': 'ineq', 'fun': constraint_boundary},  # Ensure circles are within square
        {'type': 'ineq', 'fun': constraint_overlaps},  # Non-overlapping constraint
    ]

    result = opt.minimize(objective, radii, bounds=bounds, constraints=constraints)
    radii = result.x
    centers = centers  # Centers are fixed as we're optimizing radii

    total_sum = np.sum(radii)

    return centers, radii, total_sum
