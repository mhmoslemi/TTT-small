import numpy as np
from scipy.optimize import minimize
from scipy.optimizeconstraints import ConstraintSet

def run_packing():
    # Define an initial grid for approximate placement
    # Split the square into 2 rows of 13 circles each, like a "row-based" grid
    # But with dynamic spacing to avoid overlap

    initial_centers = []
    initial_radii = []

    # Define row-wise placement
    for i in range(2):
        for j in range(13):
            x = j / 12
            y = 0.5 + i * 0.5
            initial_centers.append([x, y])
            initial_radii.append(0.04)

    initial_centers = np.array(initial_centers)
    initial_radii = np.array(initial_radii)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(radii):
        # Ensure radii are non-negative
        for r in radii:
            if r < 0:
                return False

        # Ensure circles stay within unit square
        for i in range(len(initial_centers)):
            x, y = initial_centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False

        # Ensure no overlap between circles
        for i in range(len(initial_centers)):
            for j in range(i + 1, len(initial_centers)):
                dx = initial_centers[i, 0] - initial_centers[j, 0]
                dy = initial_centers[i, 1] - initial_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False

        return True

    # Optimize with constraints
    result = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        constraints=[ConstraintSet(constraints, initial_radii)],
        bounds=[(0, 0.1) for _ in range(26)]
    )

    centers = initial_centers
    radii = result.x

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
