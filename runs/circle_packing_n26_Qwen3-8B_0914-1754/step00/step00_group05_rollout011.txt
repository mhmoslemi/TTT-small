import numpy as np
import scipy.optimize as opt
import math

def run_packing():
    # Initial grid-based packing of 26 circles (5 rows with 5, 5, 6, 5, 5 circles)
    rows = 5
    cols = 5
    grid_centers = []
    grid_radii = []
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            grid_centers.append([x, y])
            grid_radii.append(0.1)
    
    # Add a corner circle with radius adjusted for boundary
    grid_centers.append([0.0, 0.0])
    grid_radii.append(0.15)

    # Add a gap-filling circle in the middle
    grid_centers.append([0.5, 0.5])
    grid_radii.append(0.1)

    # Optimization: slightly perturb to avoid overlaps
    def objective(radii):
        # We optimize by maximizing the sum of radii (minimizing -sum)
        return -sum(radii)

    def constraints(radii):
        # Ensure non-negative radii
        for r in radii:
            if r < 0:
                return False

        # Ensure all circles are within the unit square
        for i in range(len(grid_centers)):
            x, y = grid_centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False

        # Ensure circles don't overlap
        for i in range(len(grid_centers)):

                x1, y1 = grid_centers[i]
                x2, y2 = grid_centers[j]
                dx = x1 - x2

                dist = math.sqrt(dx**2 + dy**2)
                r1 = radii[i]
                r2 = radii[j]
                if dist < r1 + r2 - 1e-12:
                    return False

        return True

    # Convert grid_centers to numpy array
    centers = np.array(grid_centers)
    radii = np.array(grid_radii)

    # Optimizer: maximize sum of radii
    result = opt.minimize(
        fun=lambda r: -sum(r),
        x0=radii,
        bounds=[(0, 1) for _ in range(len(radii))],
        constraints=[{"type": "ineq", "fun": constraints}]
    )

    final_radii = result.x
