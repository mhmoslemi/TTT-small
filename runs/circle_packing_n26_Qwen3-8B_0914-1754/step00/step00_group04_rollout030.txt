import matplotlib.pyplot as plt
import scipy.optimize as opt

def run_packing():

        n_rows = 5
        n_cols = 5

        for i in range(n_rows):
            for j in range(n_cols):
                x = j / (n_cols - 1)
                y = i / (n_rows - 1)
                centers.append([x, y])

        corners = [(0, 0), (1, 0), (1, 1), (0, 1)]
        centers += corners
        # Add remaining circles in a more sparse manner
        remaining = n_circles - len(centers)
        for _ in range(remaining):

            x = np.random.uniform(0, 1)
            y = np.random.uniform(0, 1)
            centers.append([x, y])
        return np.array(centers)

    # Generate an initial set of positions
    n_circles = 26
    centers = generate_initial_centers(n_circles)

    # Intialize radii to be uniform
    radii = np.ones(n_circles, dtype=np.float64)

    # Optimization to maximize total radius sum with no overlaps
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum => Maximize sum

    def constraints(radii):

        for i in range(n_circles):

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False

    result = opt.minimize(
