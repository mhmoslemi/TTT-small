from scipy.optimize import minimize
from scipy.optimize import Bounds

    n = 26

    # We'll use 5 rows and 5-6 columns to fit 26 circles in a square

    centers = []

        for col in range(5 + (row % 2)):

            centers.append([x, y])

    centers = (centers - np.min(centers, axis=0)) / (np.max(centers, axis=0) - np.min(centers, axis=0))

    def objective(x, radii=radii, centers=centers):

                dy = x[2*i + 1] - x[2*j + 1]

    result = minimize(
