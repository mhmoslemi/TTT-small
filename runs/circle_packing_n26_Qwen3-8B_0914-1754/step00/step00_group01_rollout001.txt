import numpy as np
import random
import math

def run_packing():
    # Hexagonal grid packing (26 circles)
    # 6 rows with 5, 6, 5, 6, 4, 2 circles respectively
    # Spacing between centers: ~2r
    # We start with all radii equal, then optimize

    # Start with hexagonal grid
    centers = []
    radii = []

    # Define initial spacing based on a rough guess
    initial_r = 0.2
    spacing = 2 * initial_r

    # Initialize with a hexagonal grid
    for i in range(6):  # 6 rows
        for j in range(5 if i < 3 else 4):  # 5, 6, 5, 6, 4, 2
            x = j * spacing
            y = i * spacing * math.sqrt(3) / 2
            centers.append([x, y])
            radii.append(initial_r)

    # Normalize
    max_x = max(centers[i][0] for i in range(len(centers)))
    max_y = max(centers[i][1] for i in range(len(centers)))
    normalization_factor = min(1 / max_x, 1 / max_y)
    centers = [[x * normalization_factor, y * normalization_factor] for x, y in centers]

    # Now perform a simulated annealing optimization
    def simulated_annealing(centers, radii, target_sum=2.636, temp=1.0, cooling_rate=0.999, num_steps=10000):

        best_radii = np.array(radii)
        best_sum = sum(best_radii)

            new_centers[idx] = [new_centers[idx][0] + dx, new_centers[idx][1] + dy]

                dist_to_top = 1 - y
                new_r = min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)
                new_radii.append(new_r)

            ok = True

                    best_radii = np.array(new_radii)
                    best_sum = current_sum
