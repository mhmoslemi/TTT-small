def generate_hexagonal_initial(n):
    # Generate hexagonal packing in unit square
    rows = int(np.sqrt(n)) + 1
    cols = int(np.ceil(n / rows))
    spacing = 1.0 / (math.sqrt(3) / 2 * cols + 1e-6)
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(rows):
        for col in range(cols):
            x = col * spacing
            y = row * (math.sqrt(3) / 2) * spacing
            if row % 2 == 1:
                x += spacing / 2
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
    # Normalize to unit square
    centers /= np.max(centers)
    return centers
