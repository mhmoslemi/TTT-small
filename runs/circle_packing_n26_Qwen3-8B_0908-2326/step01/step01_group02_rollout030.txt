import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    # Initial guess: Hexagonal grid with some vertical offset
    num_circles = 26
    rows = 5
    cols = 6
    spacing = 1.0 / (cols - 1)  # Horizontal spacing
    vertical_offset = np.sqrt(3)/2 * spacing  # Vertical offset for hexagonal packing
    centers = np.zeros((num_circles, 2))
    radii = np.ones(num_circles) * 0.05  # Initial guess for radii

                y += vertical_offset / 2
            centers[idx] = [x, y]

    # Add a bit of randomness to avoid poor local minima
    centers += np.random.uniform(-0.03, 0.03, centers.shape)

    def energy_function(x):
        # x is a flattened array of [x_0, y_0, r_0, x_1, y_1, r_1, ..., x_n, y_n, r_n]
        centers_flat = x[:num_circles * 2].reshape((num_circles, 2))
        radii_flat = x[num_circles * 2:].reshape((num_circles,))
        
        energy = 0
        # Penalize out-of-bounds positions
        for i in range(num_circles):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                energy += 1e4  # Strong penalty for out-of-bounds circles
        
        # Penalize overlaps between circles

                dx, dy = centers_flat[i] - centers_flat[j]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j]:
                    energy += 1e4  # Strong penalty for overlaps
        
        # Penalize negative radii
        for r in radii_flat:
            if r < 0:
                energy += 1e4
        
        return energy

    # Define bounds for x, y, and radii
    bounds = []

    # Optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):

            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
