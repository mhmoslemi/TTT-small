import numpy as np
import scipy.optimize as optimize
import math

def run_packing():
    """
    This function generates a packing of 26 circles in a unit square [0,1]×[0,1] 
    to maximize the sum of the radii. The approach minimizes the sum of inverse-square
    radii (a technique used in circle packing optimization) with constraints ensuring
    that circles are non-overlapping and fully contained within the square.

    Returns:
        centers: (26, 2) array of circle centers
        radii: (26,) array of circle radii
        sum_radii: total of radii
    """

    def objective(radii):
        return -np.sum(1.0 / radii)  # Negative because we aim to maximize sum of radii

    def constraints(radii, centers):
        # Enforce that all circles are within the square
        for i in range(len(radii)):
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
                return False
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
                return False

        # Enforce that circles do not overlap
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distance = math.sqrt(dx * dx + dy * dy)
                if distance < radii[i] + radii[j] - 1e-12:
                    return False

        return True

    # Predefined initial guess for centers (based on hexagonal packing)
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = col * 0.5 + 0.25
        y = row * math.sqrt(3) / 2 + 0.25
        centers[i] = [x, y]

    # Define optimization bounds for each circle
    bounds = [(0.0, 1.0) for _ in range(num_circles * 2)]

    # Initial guess for radii
    radii_initial = np.zeros(num_circles)
    for i in range(num_circles):
        # Use a heuristic based on the distance to the nearest boundary
        min_dist = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii_initial[i] = min_dist * 0.9

    # Optimization settings
    options = {
        'disp': False,
        'ftol': 1e-10,
        'maxiter': 1000,
        'method': 'SLSQP'
    }

    # Perform the optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii_initial,
        bounds=bounds,
        constraints=[lambda r: constraints(r, centers)],  # Constraint function
        options=options
    )

    # Extract results
    radii = result.x
    sum_radii = np.sum(radii)

    # Recheck the constraints to ensure the packing is valid
    for i in range(num_circles):
        x, y = centers[i]
        if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or
            y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
            raise ValueError(f"Circle {i} is outside the unit square with radius {radii[i]}")

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            distance = math.sqrt(dx * dx + dy * dy)
            if distance < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap with distance {distance} and radii sum {radii[i] + radii[j]}")

    return centers, radii, sum_radii
