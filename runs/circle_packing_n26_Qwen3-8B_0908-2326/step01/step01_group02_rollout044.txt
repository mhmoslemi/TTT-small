import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():

    n = 26

    num_cols = int(np.ceil(n / num_rows))

    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // num_cols
        col = i % num_cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i, 0] = x
        centers[i, 1] = y
    
    # Normalize the centers to fit within [0,1] x [0,1]
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= np.array([max_x, max_y])
    
    # Initial radii (same for all)
    radii = np.full(n, 0.03)
    
    # Objective function to maximize the sum of radii
    def objective(x):
        # x contains all the variables: centers and radii
        # Reshape to get centers and radii
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:].reshape(n)
        
        # Ensure all radii are non-negative
        if np.any(radii_flat < 0):
            return -np.inf
        
        # Check if all circles are inside the unit square
        for i in range(n):
            x_center, y_center = centers_flat[i]
            r = radii_flat[i]
            if not (0 - 1e-12 <= x_center - r and x_center + r <= 1 + 1e-12 and
                    0 - 1e-12 <= y_center - r and y_center + r <= 1 + 1e-12):
                return -np.inf
        
        # Check for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -np.inf
        
        return -np.sum(radii_flat)  # We minimize the negative of the sum
    
    # Flatten the centers and radii to create a single array of variables
    initial_guess = np.concatenate([centers.flatten(), radii])
