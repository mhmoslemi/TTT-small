import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def hexagonal_packing(n_circles):
        # Generate initial hexagonal packing of n_circles
        rows = int(np.ceil(np.sqrt(n_circles)))
        cols = int(np.ceil(n_circles / rows))
        centers = np.zeros((n_circles, 2))
        for i in range(n_circles):
            row = i // cols
            col = i % cols
            x = 0.5 + (col - (cols // 2)) * (1.0 / (cols - 1)) if cols > 1 else 0.5
            y = 0.5 + (row - (rows // 2)) * (np.sqrt(3) / 2) / (rows - 1) if rows > 1 else 0.5
            centers[i] = np.array([x, y])
        return centers

    # Initial packing
    initial_centers = hexagonal_packing(26)
    # Assume all initial radii are the same for now
    initial_radii = np.full(26, 0.05)

    # Define the objective function to maximize the total area (sum of radii squared)
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define constraints for optimization
    def constraint_overlap(centers, radii):
        def constraint(x):
            nonlocal centers, radii
            for i in range(len(radii)):
                for j in range(i + 1, len(radii)):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    if dist < radii[i] + radii[j] - 1e-8:
                        return -1  # Violation of no overlap
            return 0
        return constraint

    def constraint_boundary(centers, radii):
        def constraint(x):
            nonlocal centers, radii
            for i in range(len(radii)):
                if centers[i, 0] - radii[i] < 0 or centers[i, 0] + radii[i] > 1:
                    return -1  # Violation of boundary
                if centers[i, 1] - radii[i] < 0 or centers[i, 1] + radii[i] > 1:
                    return -1
            return 0
        return constraint

    # Optimization problem setup
    bounds = [(0, 0.5) for _ in range(26)]  # Radii between 0 and 0.5
    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap(initial_centers, initial_radii)},
        {'type': 'ineq', 'fun': constraint_boundary(initial_centers, initial_radii)}
    ]

    result = optimize.minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
