import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid of centers (5x5 grid with one extra circle)
    grid_size = 5
    initial_centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            initial_centers.append([i / (grid_size - 1), j / (grid_size - 1)])
    initial_centers.append([0.5, 0.9])  # One extra circle at the top
    initial_centers = np.array(initial_centers)

    # Initial radii (all equal)
    initial_radii = np.full(len(initial_centers), 1 / (grid_size * grid_size))

    # Objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(x[1:])  # Minimize the negative of the sum of radii

    # Constraints: center and radius for each circle
    def constraint_center_radius(x):
        centers = x[:len(initial_centers) * 2].reshape(-1, 2)
        radii = x[len(initial_centers) * 2:]
        for i in range(len(centers)):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if not (0 <= x_i - r_i and x_i + r_i <= 1 and 0 <= y_i - r_i and y_i + r_i <= 1):
                return -np.inf
        return 0

    # Constraints: non-overlap between any two circles
    def constraint_non_overlap(x):
        centers = x[:len(initial_centers) * 2].reshape(-1, 2)
        radii = x[len(initial_centers) * 2:]
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return 0

    # Combine constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_center_radius},
        {'type': 'ineq', 'fun': constraint_non_overlap}
    ]

    # Initial guess: centers + radii
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])

    # Optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1) for _ in range(len(initial_centers) * 2)] + [(0, 1) for _ in range(len(initial_centers))],

    if not result.success:

    if not valid:
        raise RuntimeError(f"Validation failed: {message}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
