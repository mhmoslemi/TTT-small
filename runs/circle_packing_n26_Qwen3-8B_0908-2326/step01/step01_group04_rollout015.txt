import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n_circles = 26
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)

    # We can manually place the circles in a hexagonal pattern
    # Simplified for demonstration, but in practice, a more thorough grid would be better
    
    # First row (top to bottom, left to right)
    for i in range(n_circles):
        row = i // 5
        col = i % 5
        x = col * 0.25 + 0.125
        y = row * np.sqrt(3) * 0.25 + 0.125
        centers[i] = np.array([x, y])
        radii[i] = 0.05
    
    # Define objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative of sum to maximize

    # Define constraints
    def constraint_center(x, idx):
        return x[2*idx] - 1e-12  # Ensure x-coordinate is not less than 0
    def constraint_center_right(x, idx):
        return 1 + 1e-12 - x[2*idx]  # Ensure x-coordinate is not more than 1
    def constraint_center_top(x, idx):
        return x[2*idx + 1] - 1e-12  # Ensure y-coordinate is not less than 0
    def constraint_center_bottom(x, idx):
        return 1 + 1e-12 - x[2*idx + 1]  # Ensure y-coordinate is not more than 1

    def constraint_boundary(x, radius_idx):
        r = x[2*n_circles + radius_idx]
        x_center = x[2*radius_idx]
        y_center = x[2*radius_idx + 1]
        return x_center - r - 1e-12  # Ensure x - r not less than 0
    def constraint_boundary_right(x, radius_idx):
        r = x[2*n_circles + radius_idx]
        x_center = x[2*radius_idx]
        return 1 + 1e-12 - x_center - r  # Ensure x + r not more than 1
    def constraint_boundary_top(x, radius_idx):
        r = x[2*n_circles + radius_idx]
        y_center = x[2*radius_idx + 1]
        return y_center - r - 1e-12  # Ensure y - r not less than 0
    def constraint_boundary_bottom(x, radius_idx):
        r = x[2*n_circles + radius_idx]
        y_center = x[2*radius_idx + 1]
        return 1 + 1e-12 - y_center - r  # Ensure y + r not more than 1

    # Define constraints for non-overlapping circles
    def constraint_overlap(x, idx1, idx2):
        x1 = x[2*idx1]
        y1 = x[2*idx1 + 1]

        x2 = x[2*idx2]
        y2 = x[2*idx2 + 1]
        r2 = x[2*n_circles + idx2]
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return dist - r1 - r2 + 1e-12  # Ensure distance >= r1 + r2

    x0 = np.concatenate([centers.flatten(), radii])

    # Constraints for center boundaries
    constraints = []
    for i in range(n_circles):
        constraints.append({'type': 'ineq', 'fun': lambda x, idx=i: constraint_center(x, idx)})
        constraints.append({'type': 'ineq', 'fun': lambda x, idx=i: constraint_center_right(x, idx)})
        constraints.append({'type': 'ineq', 'fun': lambda x, idx=i: constraint_center_top(x, idx)})

    bounds = []
