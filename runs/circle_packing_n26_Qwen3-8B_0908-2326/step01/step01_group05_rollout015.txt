```python
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26
    square_size = 1.0
    init_radius = 0.05  # Initial guess for radius

    # Initial grid of circle centers (hexagonal packing)
    # Start with 6 rows
    rows = 6
    cols = 5  # Slightly less than 6 for symmetry and boundary fit
    radii = np.full(n, init_radius)

    # Set up the initial centers in a hexagonal grid
    # Each row is offset by radius * sqrt(3)
    # Horizontal spacing is 2 * radius (for non-overlapping)
    centers = np.zeros((n, 2))

    offset = radius * np.sqrt(3)

                x = j * 2 * radius
                y = i * offset
            else:
                x = (j + 0.5) * 2 * radius
                y = i * offset
            centers[row_idx] = [x, y]

    # Prepare for optimization
    def objective(radii):
        return -np.sum(radii)  # We are maximizing the sum, so minimize the negative

    def constraint_dist(radii, centers):
        # Define the constraints for the circles being within square and not overlapping
        constraints = []
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append((x - r >= -1e-12, x + r <= 1 + 1e-12, y - r >= -1e-12, y + r <= 1 + 1e-12))
        # Non-overlap constraints

            for j in range(i + 1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                dist = np.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)
                constraints.append((dist >= r1 + r2 - 1e-12,))
        return constraints

    # Optimize
    bounds = Bounds(np.zeros(n), np.full(n, square_size / 2.0))  # Max possible radius for each circle
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
