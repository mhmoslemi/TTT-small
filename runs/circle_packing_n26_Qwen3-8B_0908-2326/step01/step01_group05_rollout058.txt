import scipy.optimize as opt

def run_packing():

    # Number of circles
    n = 26

    # Define the grid: rows and columns of centers
    rows = 5
    cols = 6
    # Hexagonal grid: each row is offset by half a column
    # Vertical spacing between rows: sqrt(3)/2
    # Horizontal spacing between columns: 1
    spacing = 1.0 / (cols - 1)
    vertical_spacing = np.sqrt(3) / 2.0 / (rows - 1)

    # Initial centers grid
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * vertical_spacing
            if i % 2 == 1:
                x += spacing / 2
            centers.append([x, y])

    # Convert to numpy array
    centers = np.array(centers)
    # Scale to unit square (already in [0, 1])
    
    # Initial guess for radii: based on spacing
    # For hexagonal packing, the minimal radius is determined by the spacing
    # We can use the minimal spacing between centers to determine an initial radius
    initial_radius = (spacing / 2) * 0.9  # slightly less than half the spacing

    # Create initial radii estimate
    radii = np.full(n, initial_radius)

    # Objective function: maximize the sum of radii
    def objective(x):
        # x is a vector of [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
        flat_x = x.reshape(n, 3)
        centers_flat = flat_x[:, :2]
        radii_flat = flat_x[:, 2]
        sum_radii = np.sum(radii_flat)
        return -sum_radii  # We minimize the negative to maximize

    # Constraints:
    # 1. Boundary constraints (x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1)
    # 2. Distance constraints (distance between centers >= r_i + r_j)
    def constraint_boundary(x):
        flat_x = x.reshape(n, 3)
        centers_flat = flat_x[:, :2]
        radii_flat = flat_x[:, 2]
        for i in range(n):
            x_i, y_i, r_i = flat_x[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        return 0

    def constraint_distance(x):
        flat_x = x.reshape(n, 3)
        centers_flat = flat_x[:, :2]
        radii_flat = flat_x[:, 2]
        for i in range(n):

                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -1
        return 0

        x0=initial_x,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
