```python
import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define initial guesses based on a hexagonal packing in the unit square
    def initial_guess_hexagonal(n):
        rows = int(np.sqrt(n))
        cols = n // rows + 1
        centers = np.zeros((n, 2))
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * (i % 2)
                y = i + 0.5
                if x >= 1 or y >= 1:
                    continue
                centers[i * cols + j] = [x, y]
        return centers

    n = 26
    initial_centers = initial_guess_hexagonal(n)
    initial_radii = np.ones(n) * 0.05  # Small initial radius for all circles

    # Objective: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Constraints: 
    # 1. Circle centers must lie within [0, 1]^2
    # 2. Circle must be fully within the unit square
    # 3. Circles must not overlap
    def constraints(radii):
        centers = initial_centers  # Assume centers are fixed for now
        constraints = []
        # Constraint: circle within square
        for i in range(n):
            def in_square(r):
                x, y = centers[i]
                return (x - r >= 0) and (x + r <= 1) and (y - r >= 0) and (y + r <= 1)
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: in_square(r)})
        # Constraint: no overlap between circles
        for i in range(n):
            for j in range(i + 1, n):
                def no_overlap(r_i, r_j):
                    x1, y1 = centers[i]
                    x2, y2 = centers[j]
                    dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                    return dist >= r_i + r_j - 1e-12
                # We need to define constraints for all radii
                # This can be refined for better performance
                if i == 0:
                    constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: no_overlap(r[i], r[j])})
                else:
                    constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: no_overlap(r[i], r[j])})
        return constraints

    # Optimization problem
    bounds = [(0, 1) for _ in range(n)]
    result = optimize.minimize(
        fun=objective,
        x0=initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints(initial_radii),
        options={'maxiter': 1000, 'ftol': 1e-9}

    if not valid:
        raise ValueError(f"Packing validation failed: {msg}")

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
