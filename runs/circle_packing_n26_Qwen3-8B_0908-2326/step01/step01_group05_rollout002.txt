import scipy.optimize as opt
from typing import Tuple, Any

def validate_packing(centers: np.ndarray, radii: np.ndarray) -> Tuple[bool, str]:

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

    return True, "ok"

def objective_function(params: np.ndarray, n: int, max_radius: float = 1.0) -> float:
    # params has shape (2 * n, ), where each pair is (x, y, r) for each circle
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:].reshape(n)
    
    # Ensure radii are non-negative and bounded
    radii = np.clip(radii, 0, max_radius)

    sum_radii = np.sum(radii)

            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-8:
