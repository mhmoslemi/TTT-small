import numpy as np
from scipy.optimize import minimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Initial grid layout: 5 rows, 5 columns (with one extra circle)
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        centers[i, 0] = col / 4.0 + 0.1
        centers[i, 1] = row / 4.0 + 0.1

    # Objective: maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)

    # Constraints for optimization
    def constraints(radii, centers):
        # Constraint 1: Radii must be non-negative
        non_neg = np.concatenate([radii >= 0])
        
        # Constraint 2: Circles must be within [0,1]²
        boundary = []
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            boundary.append(x - r >= -1e-12)
            boundary.append(x + r <= 1 + 1e-12)
            boundary.append(y - r >= -1e-12)
            boundary.append(y + r <= 1 + 1e-12)
        
        # Constraint 3: Circles must not overlap
        overlap = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                overlap.append(dist >= radii[i] + radii[j] - 1e-12)

        return np.concatenate([non_neg, boundary, overlap])

    # Optimize
    bounds = [(0, 1.0) for _ in range(num_circles)]
    result = minimize(
        objective,
        np.ones(num_circles),
        args=(centers,),
        method='SLSQP',
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints, "args": (centers,)},
        options={'maxiter': 1000, 'disp': False}
    )

    radii = result.x
    centers = centers
    sum_radii = np.sum(radii)

    # Validate the packing (for debugging purposes)
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing failed with message: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
