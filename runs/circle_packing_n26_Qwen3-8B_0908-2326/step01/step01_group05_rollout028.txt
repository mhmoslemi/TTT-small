import numpy as np
import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: standard hexagonal packing
    n = 26
    cols = int(math.sqrt(n))
    rows = (n + cols - 1) // cols
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        col = i % cols
        row = i // cols
        x = 0.5 + (col - cols / 2) * (1.0 / (cols - 1))
        y = 0.5 + row * (1.0 / (rows - 1))
        centers[i] = np.array([x, y])
        radii[i] = 0.04
    
    # Define the objective function to minimize
    def objective(x):
        # x contains the flattened centers and radii
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:].reshape(n)
        
        # Check for NaNs or negative radii
        if np.isnan(centers_flat).any() or np.isnan(radii_flat).any():
            return np.inf
        
        for i in range(n):
            if radii_flat[i] < 0:
                return np.inf
        
        # Objective function: minimize the sum of squared distances between centers
        total_distance = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                total_distance += (dist - (radii_flat[i] + radii_flat[j])) ** 2
        
        # Add constraints: circles must stay within the unit square
        for i in range(n):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                total_distance += 1e6
        
        # Add overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    total_distance += 1e6
        
        return total_distance
    
    # Initial x: flatten centers and radii
    initial_x = np.concatenate([centers.flatten(), radii])
    
    # Bounds: centers are in [0,1] and radii are non-negative
    bounds = []

        bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # center x, y
        bounds.extend([(0.0, 1.0)])              # radius
    bounds = bounds[:2*n + n]  # total of 3n parameters
    
    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=initial_x,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    # Extract centers and radii from result
    centers_result = result.x[:2*n].reshape(n, 2)
    radii_result = result.x[2*n:].reshape(n)
    
    sum_radii = np.sum(radii_result)
    
    return (centers_result, radii_result, sum_radii)
