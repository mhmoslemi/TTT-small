import scipy.optimize as opt
import math

def run_packing():

    n = 26
    # Calculate the number of rows and columns that can fit
    max_rows = int(math.sqrt(n))
    max_cols = int(math.sqrt(n))
    if max_rows * max_cols < n:
        max_cols += 1

    # Estimate the radius for an initial hexagonal grid
    # Assume an initial radius that gives some spacing
    # Adjust this as needed during optimization
    initial_radius = 0.1

        y = row * math.sqrt(3) * initial_radius
        centers[i] = [x, y]

    # Optimize the radius of each circle to maximize the sum of radii
    # Define the optimization problem

    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    def constraints(radii):
        # Constraint: all circles must be within the unit square
        for i in range(n):
            r = radii[i]
            if centers[i, 0] - r < -1e-12 or centers[i, 0] + r > 1 + 1e-12:
                return np.inf

                return np.inf

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
