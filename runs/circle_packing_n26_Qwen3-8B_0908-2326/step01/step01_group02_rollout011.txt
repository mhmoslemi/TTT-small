import numpy as np
from scipy.optimize import minimize
from scipy.optimize.constraints import InequalityConstraint

def generate_initial_config(n_circles):
    # Generate a hexagonal packing in the unit square
    # Start with a hexagonal grid and adjust positions to fit into [0,1]^2
    radius_grid = np.sqrt(3) / 2 * 0.4  # Adjust for spacing
    rows = int(np.ceil(n_circles / 2))
    cols = int(np.ceil(n_circles / rows))

    centers = []
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * (2 * radius_grid)
                y = i * (np.sqrt(3) * radius_grid)

                x = (j + 0.5) * (2 * radius_grid)

            y = np.clip(y, 0, 1)
            centers.append([x, y])

    radii = np.full(n_circles, 0.4)
    
    # Add some circles near the corners to fill gaps

    return centers, radii

def objective_function(x, n_circles):

    centers = x.reshape((n_circles, 2))
    radii = np.full(n_circles, 0.4)
    return -np.sum(radii)  # Maximize sum of radii, so minimize negative of it

def constraint_functions(x, n_circles, centers, radii):
    # Boundary constraint
    def bound_constraint(x, n_circles):
        centers = x.reshape((n_circles, 2))
        return np.array([np.all(centers[:, 0] >= -1e-6), np.all(centers[:, 0] <= 1 + 1e-6),
                         np.all(centers[:, 1] >= -1e-6), np.all(centers[:, 1] <= 1 + 1e-6)])
    
    # Non-overlap constraint
    def non_overlap_constraint(x, n_circles, centers, radii):
        centers = x.reshape((n_circles, 2))
        dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :])**2, axis=2))
        return np.array([dists[i, j] >= radii[i] + radii[j] - 1e-6 for i in range(n_circles) for j in range(i+1, n_circles)]).flatten()
    
    # Combine constraints
    bound_cons = InequalityConstraint(bound_constraint(x, n_circles), lb=-np.inf, ub=0)
    non_overlap_cons = InequalityConstraint(non_overlap_constraint(x, n_circles, centers, radii), lb=-np.inf, ub=0)
    return [bound_cons, non_overlap_cons]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
