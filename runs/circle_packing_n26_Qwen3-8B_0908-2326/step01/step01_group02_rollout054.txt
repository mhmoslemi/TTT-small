import numpy as np
import scipy.optimize as opt
import itertools

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # 5x5 grid with one extra circle in the center
    grid_centers = np.array([[i / 4, j / 4] for i, j in itertools.product(range(5), repeat=2)])
    centers = np.vstack([grid_centers, [0.5, 0.5]])
    
    # Flatten the center and radius parameters
    # Each circle has (x, y, r), so we stack them
    n = centers.shape[0]
    params = np.zeros(n * 3)
    
    # Initial guess: centers stay the same, radii are set to 0.05 initially
    for i in range(n):
        params[i * 3 : (i + 1) * 3] = [centers[i][0], centers[i][1], 0.05]
    
    # Optimization variables: x, y, r for each circle
    def objective(params):
        # Extract centers and radii
        centers = params.reshape(n, 3)[:, :2]
        radii = params.reshape(n, 3)[:, 2]
        # Objective: maximize sum of radii => minimize negative sum
        return -np.sum(radii)
    
    # Box constraints for each circle: 0 <= x - r <= 1, 0 <= y - r <= 1
    # So for each circle, the bounds are:
    # lower: [0, 0, 0] (x, y, r)
    # upper: [1, 1, 1] (x, y, r)
    bounds = [(0.0, 1.0) for _ in range(n * 3)]
    
    # Non-overlap constraints
    def constraint_overlap(params):
        centers = params.reshape(n, 3)[:, :2]
        radii = params.reshape(n, 3)[:, 2]
        violations = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    violations.append(1.0)
        return np.array(violations)
    
    # Feasibility constraint: all pairs must have at least a small distance
    def constraint_feasibility(params):
        centers = params.reshape(n, 3)[:, :2]
        radii = params.reshape(n, 3)[:, 2]
        violations = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    violations.append(1.0)
        return np.array(violations)
    
    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=params,
        method='SLSQP',
