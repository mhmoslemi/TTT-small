import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial guess: hexagonal grid of circles in the center
    # Start with a hexagonal grid of circles in the center and fill outer space
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Hexagonal grid with a central circle
    hex_grid_rows = 5
    hex_grid_cols = 5
    spacing = 0.3  # initial spacing
    offset_y = np.sqrt(3) / 2 * spacing

    for row in range(hex_grid_rows):
        for col in range(hex_grid_cols):
            x = 0.5 - 0.5 + (col - hex_grid_cols // 2) * spacing
            y = 0.5 - 0.5 + (row - hex_grid_rows // 2) * spacing
            if row % 2 == 1:
                y += offset_y
            centers[row * hex_grid_cols + col] = [x, y]
            radii[row * hex_grid_cols + col] = 0.05  # initial radius

    # Create variables for optimization
    x = np.concatenate((centers.flatten(), radii))

    # Objective function to maximize sum of radii
    def objective(x):
        return -x[-num_circles:].sum()  # Negative for minimization

    # Constraints
    def constraints(x):
        centers_opt = x[:num_circles * 2].reshape((num_circles, 2))
        radii_opt = x[-num_circles:]
        constraints = []
        for i in range(num_circles):
            # Boundary constraints
            if centers_opt[i, 0] - radii_opt[i] < -1e-12:
                constraints.append(centers_opt[i, 0] - radii_opt[i] + 1e-12)
            if centers_opt[i, 0] + radii_opt[i] > 1 + 1e-12:
                constraints.append(1 + 1e-12 - centers_opt[i, 0] - radii_opt[i])
            if centers_opt[i, 1] - radii_opt[i] < -1e-12:
                constraints.append(centers_opt[i, 1] - radii_opt[i] + 1e-12)
            if centers_opt[i, 1] + radii_opt[i] > 1 + 1e-12:
                constraints.append(1 + 1e-12 - centers_opt[i, 1] - radii_opt[i])
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j]) ** 2))
                if dist < radii_opt[i] + radii_opt[j] - 1e-12:
                    constraints.append(radii_opt[i] + radii_opt[j] - dist + 1e-12)
        return constraints

    # Optimization
    bounds = []
    for i in range(num_circles):
        bounds.extend([(0, 1) for _ in range(2)])  # centers
        bounds.extend([(0, 0.2) for _ in range(1)])  # radii

    result = optimize.minimize(
        objective,
        x,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},
        bounds=bounds,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}

    # Extract results
    centers_opt = result.x[:num_circles * 2].reshape((num_circles, 2))
    radii_opt = result.x[-num_circles:]

    # Validate the packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(msg)

    sum_radii = radii_opt.sum()
    return centers_opt, radii_opt, sum_radii
