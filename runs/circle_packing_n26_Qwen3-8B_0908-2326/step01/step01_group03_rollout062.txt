```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import time

# Define constants
N_CIRCLES = 26
MAX_ITER = 100
TOLERANCE = 1e-6
INITIAL_RADIUS = 0.05

# Strategy: Use a hybrid approach combining a hexagonal packing near the center
# with optimized placement of circles near the boundaries and corners to maximize the total area.
# Start with a grid-based initialization with some randomness to avoid symmetry.
# Then perform a gradient-based optimization to refine the arrangement.

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialize centers in a grid-like pattern with some randomness
    centers = []
    radii = []
    for i in range(N_CIRCLES):
        row = i // 4
        col = i % 4
        x = col * 0.4 + np.random.rand() * 0.2
        y = row * 0.4 + np.random.rand() * 0.2
        centers.append([x, y])
        radii.append(INITIAL_RADIUS)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Define objective function: maximize sum of radii
    def objective(x):
        # x is a flat array: [x0, y0, r0, x1, y1, r1, ...]
        x_flat = x.reshape(-1, 3)
        centers_opt = x_flat[:, :2]
        radii_opt = x_flat[:, 2]
        return -np.sum(radii_opt)  # Negative for minimization

    # Define constraints:
    # 1. Center within the unit square
    # 2. Radius non-negative
    # 3. No overlap between circles

    def constraint_center_bounds(x):
        x_flat = x.reshape(-1, 3)
        centers_opt = x_flat[:, :2]
        radii_opt = x_flat[:, 2]
        for i in range(N_CIRCLES):
            x, y = centers_opt[i]
            r = radii_opt[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return -1
        return 0

    def constraint_radius_nonnegative(x):
        x_flat = x.reshape(-1, 3)
        radii_opt = x_flat[:, 2]
        for r in radii_opt:
            if r < 0:
                return -1
        return 0

    def constraint_overlap(x):
        x_flat = x.reshape(-1, 3)
        centers_opt = x_flat[:, :2]
        radii_opt = x_flat[:, 2]
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_opt[i] + radii_opt[j] - 1e-8:
                    return -1
        return 0

    # Create bounds: x and y in [0,1], r >= 0
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # r

    # Initial guess
    x0 = np.zeros(3 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x0[i * 3] = centers[i, 0]
        x0[i * 3 + 1] = centers[i, 1]
        x0[i * 3 + 2] = radii[i]

    # Set up constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_center_bounds},
        {'type': 'ineq', 'fun': constraint_radius_nonnegative},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Perform optimization
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': MAX_ITER, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract optimized centers and radii
    optimized_x = result.x.reshape(N_CIRCLES, 3)
    centers_opt = optimized_x[:, :2]
    radii_opt = optimized_x[:, 2]

    # Validate packing
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Packing validation failed: {message}")

    # Compute total sum of radii
    total_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, total_radii
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
