import numpy as np
import cvxpy as cp
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Initial random placement of centers
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1  # Initial small radii

    # Optimization variables
    r = cp.Variable(n)
    x = cp.Variable((n, 2))

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    # Each circle must be inside the unit square
    for i in range(n):
        constraints.append(x[i, 0] - r[i] >= 0)
        constraints.append(x[i, 0] + r[i] <= 1)
        constraints.append(x[i, 1] - r[i] >= 0)
        constraints.append(x[i, 1] + r[i] <= 1)

    # Circles cannot overlap
    for i in range(n):
        for j in range(i + 1, n):
            distance = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(distance >= r[i] + r[j] - 1e-8)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve()

    # Extract the results
    centers_opt = x.value
    radii_opt = r.value

    # Ensure valid results
    if centers_opt is None or radii_opt is None:
        raise ValueError("Optimization failed to converge")

    # Final check for validity
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(msg)

    return centers_opt, radii_opt, np.sum(radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
