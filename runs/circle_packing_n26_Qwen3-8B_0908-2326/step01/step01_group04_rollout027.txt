import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial 5x5 grid-based packing
    num_circles = 26
    grid_size = 5
    spacing = 1.0 / (grid_size - 1)  # Grid step

    # For grid-based centers
    centers = np.zeros((num_circles, 2))
    # First 25 circles in a grid
    for i in range(grid_size):
        for j in range(grid_size):
            idx = i * grid_size + j
            centers[idx, 0] = j * spacing
            centers[idx, 1] = i * spacing
    # 26th circle at the center
    centers[25, 0] = 0.5
    centers[25, 1] = 0.5

    # Initial radii (this will be optimized)
    radii = np.full(num_circles, 0.05)

    # Objective function to maximize sum of radii
    def objective(radii_vector):
        return -np.sum(radii_vector)

    # List to store constraints
    constraints = []

    # Constraint: all radii must be non-negative
    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})

    # Constraint: each circle must be inside the square
    for i in range(num_circles):
        x, y = centers[i]
        def inside_box(r, i=i):
            r_i = r[i]
            return (x - r_i >= -1e-12) and (x + r_i <= 1 + 1e-12) and \
                   (y - r_i >= -1e-12) and (y + r_i <= 1 + 1e-12)
        constraints.append({'type': 'ineq', 'fun': inside_box})

    # Constraint: circles must not overlap
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def no_overlap(r, i=i, j=j):
                r_i = r[i]
                r_j = r[j]
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                return dist - r_i - r_j + 1e-12
            constraints.append({'type': 'ineq', 'fun': no_overlap})

    # Optimization
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=constraints,
        options={'ftol': 1e-12, 'maxiter': 1000}
    )

    # Extract the optimized radii and centers
    optimized_radii = result.x
    sum_radii = np.sum(optimized_radii)

    return centers, optimized_radii, sum_radii
