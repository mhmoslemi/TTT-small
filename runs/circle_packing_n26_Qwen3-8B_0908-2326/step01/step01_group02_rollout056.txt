import numpy as np
import cvxpy as cp

def run_packing():
    # Base hexagonal packing of 24 circles
    num_circles = 26
    num_hex = 24
    num_corners = 2

    # Hexagonal grid of 24 circles (4 rows of hexagons)
    # Rows: 4, Columns: 6
    hex_centers = np.zeros((num_hex, 2))
    hex_radius = cp.Variable(num_hex)
    for i in range(4):
        for j in range(6):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            hex_centers[i * 6 + j] = [x, y]

    # Corner circles
    corner_centers = np.array([
        [0, 0],
        [1, 1]
    ])
    corner_radius = cp.Variable(num_corners)

    # Combine centers and radii
    centers = cp.vstack((hex_centers, corner_centers))
    radii = cp.hstack((hex_radius, corner_radius))

    # Objective: maximize the total radius sum
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # All circles must be inside the unit square
    for i in range(num_circles):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-12)
        constraints.append(x + radii[i] <= 1 + 1e-12)
        constraints.append(y - radii[i] >= -1e-12)
        constraints.append(y + radii[i] <= 1 + 1e-12)

    # No two circles overlap
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # All radii must be non-negative
    for i in range(num_circles):
        constraints.append(radii[i] >= 0)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iter=5000, eps=1e-6)
    except:
        return np.array([]), np.array([]), 0.0

    # Extract results
    centers_np = centers.value
    radii_np = radii.value
    sum_radii = np.sum(radii_np)

    return centers_np, radii_np, sum_radii
