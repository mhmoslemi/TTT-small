import numpy as np
import scipy.optimize as optimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # First pass: hexagonal grid in a square
    rows = int(math.sqrt(num_circles))
    cols = (num_circles + rows - 1) // rows
    
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * math.sqrt(3) / 2
            if x < 0 or x > 1 or y < 0 or y > 1:
                continue
            centers[i * cols + j] = [x, y]
    
    # Resize to fit in unit square
    max_distance = np.max(np.sqrt(np.sum((centers[i] - centers[j])**2 for i in range(num_circles) for j in range(i+1, num_circles))))
    initial_radius = 1 / (2 * max_distance)
    radii[0:num_circles] = initial_radius
    
    # Define optimization variables
    bounds = []
    for _ in range(num_circles):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # radius
    
    # Objective function to maximize the sum of radii
    def objective(x):
        centers_flat = x[:num_circles*2].reshape(num_circles, 2)
        radii_flat = x[num_circles*2:].reshape(num_circles)
        return -np.sum(radii_flat)  # minimize negative sum

    # Constraint function to check non-overlapping and boundary
    def constraints(x):
        centers_flat = x[:num_circles*2].reshape(num_circles, 2)
        radii_flat = x[num_circles*2:].reshape(num_circles)
        
        # Check boundary constraints
        for i in range(num_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or \
               y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf
        
        # Check non-overlapping constraints
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]

                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return np.inf
        
        return 0  # No constraints violated

    # Set up optimization
    result = optimize.minimize(
        objective,
        np.concatenate([centers.flatten(), radii.flatten()]),
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP",
        options={"maxiter": 500, "ftol": 1e-9, "eps": 1e-8}
    )

    centers_opt = result.x[:num_circles*2].reshape(num_circles, 2)

    sum_radii = np.sum(radii_opt)
    
    # Post-validation
    if not validate_packing(centers_opt, radii_opt)[0]:
        raise ValueError("Final packing validation failed")
    
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
