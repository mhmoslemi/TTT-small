import numpy as np
from scipy.optimize import minimize
from scipy.optimize.constraints import Constraint
import math

def run_packing():
    # Initial placement: staggered grid (hexagonal close packing) to start
    num_circles = 26
    grid_rows = 5
    grid_cols = 6
    spacing = 1.0 / (grid_cols - 1)
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Initial estimate for circle radius
    initial_radius = 0.05

    # Place circles in staggered hexagonal grid
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j * spacing
            y = i * spacing * (np.sqrt(3) / 2 if i % 2 == 0 else 0)
            if i % 2 == 1:
                x += spacing / 2
            idx = i * grid_cols + j
            centers[idx] = np.array([x, y])
            radii[idx] = initial_radius

    # Objective function to maximize the sum of radii
    def objective(x):
        return -x  # Minimize negative sum of radii

    # Constraints for boundary and non-overlapping circles
    def constraint_boundary(x):
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = x[i + num_circles]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        return 0

    def constraint_overlap(x):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                r_i = x[i + num_circles]
                r_j = x[j + num_circles]
                if dist < r_i + r_j - 1e-12:
                    return -1
        return 0

    # Flatten centers and radii into a single array for optimization
    x0 = np.concatenate((centers.flatten(), radii))

    # Define optimization problem
    constraints = [
        Constraint(lambda x: constraint_boundary(x), name='boundary'),
        Constraint(lambda x: constraint_overlap(x), name='overlap'),
    ]

    # Use L-BFGS-B optimizer with bounds for radius
    bounds = [(0, 1.0) for _ in range(num_circles * 2)]
    result = minimize(

        x0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        tol=1e-6
