import scipy.optimize as opt

        # We use a hexagonal arrangement with 6 rows for a square
        # Approximate horizontal and vertical spacing
        spacing_x = 2 / np.sqrt(3)
        spacing_y = 1
        centers = np.zeros((n_circles, 2))
        idx = 0
        for row in range(6):
            for col in range(n_circles // 6 + 1):
                if idx >= n_circles:
                    break
                x = col * spacing_x
                y = row * spacing_y if row % 2 == 0 else (row + 0.5) * spacing_y
                centers[idx, 0], centers[idx, 1] = x, y
                idx += 1
        return centers

        centers_opt = x[:26*2].reshape(26, 2)
        radii_opt = x[26*2:].reshape(26)
        return -np.sum(radii_opt)  # Negative for minimization

    def constraints(x: np.ndarray) -> np.ndarray:
        # x has shape (26 * 3,) = (x1, y1, r1, x2, y2, r2, ..., x26, y26, r26)
        centers_opt = x[:26*2].reshape(26, 2)
        radii_opt = x[26*2:].reshape(26)
        # Constraint 1: Radii must be non-negative

        bounds_sq = np.zeros(26)
        for i in range(26):

            r_i = radii_opt[i]
            x_left = x_i - r_i
            x_right = x_i + r_i
            y_bottom = y_i - r_i
            y_top = y_i + r_i
            bounds_sq[i] = max(0, min(1, x_right - x_left, y_top - y_bottom))  # Slack to ensure within square

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
