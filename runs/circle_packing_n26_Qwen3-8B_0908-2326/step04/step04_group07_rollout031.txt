def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    num_generations = 200
    population_size = 100
    cx_prob = 0.8
    mut_prob = 0.2
    elitism = 5

    # Helper function to validate a candidate solution
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    # Hyperparameters for the genetic algorithm
    def fitness(individual):
        centers = individual[::2].reshape(num_circles, 2)
        radii = individual[1::2]
        valid, msg = validate_packing(centers, radii)
        if not valid:
            return -1000,  # Penalize invalid configurations
        sum_radii = np.sum(radii)
        return sum_radii,

    # Fitness and selection
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    # Initialization for the genetic algorithm
    def init_individual():
        centers = np.random.uniform(0.1, 0.9, (num_circles, 2))
        radii = np.random.uniform(0.01, 0.1, num_circles)
        return np.concatenate((centers.flatten(), radii))

    def init_population():
        return [init_individual() for _ in range(population_size)]

    # Crossover and mutation operators
    def cx_two_point(ind1, ind2):
        size = len(ind1)
        cx_point1 = np.random.randint(1, size - 1)
        cx_point2 = np.random.randint(cx_point1, size - 1)
        ind1[cx_point1:cx_point2], ind2[cx_point1:cx_point2] = ind2[cx_point1:cx_point2], ind1[cx_point1:cx_point2]
        return ind1, ind2

        for i in range(len(individual)):
            if np.random.random() < mut_prob:
                individual[i] += np.random.normal(0, 0.05)
                individual[i] = np.clip(individual[i], 0, 1)

   toolbox = base.Toolbox()
