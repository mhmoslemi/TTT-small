            if i < 4:

                positions[i] = [0.1, 0.1] if i == 0 else [0.1, 0.9] if i == 1 else [0.9, 0.9] if i == 2 else [0.9, 0.1]

                col = (i - 8) % 5
                x = 0.3 + col * 0.3
                y = 0.3 + row * 0.3
                positions[i] = [x, y]
        return positions

    centers = initialize_positions(num_circles)

    def get_radii(centers):
        radii = np.zeros(num_circles)
        for i in range(num_circles):

            r = min(x_i, 1 - x_i, y_i, 1 - y_i)
            for j in range(num_circles):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                r = min(r, (dist - 1e-12) / 2)
            radii[i] = r
        return radii

    best_centers = None
