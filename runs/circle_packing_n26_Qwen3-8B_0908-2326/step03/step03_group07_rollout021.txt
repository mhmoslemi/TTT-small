import scipy.optimize as opt
import matplotlib.pyplot as plt

def compute_distance(p1, p2):
    return np.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

def generate_hexagonal_initial_guess(num_circles):

        y = row * spacing + spacing / 2
        centers.append([x, y])
    return np.array(centers)

            repulsion_force += max(0, 1 - dist / (radii_vec[i] + radii_vec[j]))
    return -np.sum(radii_vec) + 0.1 * repulsion_force  # Adjust repulsion coefficient as needed

def constraint_non_overlap(radii_vec, centers):

            dist = compute_distance(centers[i], centers[j])
