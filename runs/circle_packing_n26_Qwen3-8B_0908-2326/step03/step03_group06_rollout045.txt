def generate_initial_config(num_circles, config):
    centers = np.zeros((num_circles, 2))
    if config == 'hexagonal':
        # Generate hexagonal grid
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        spacing = 1.0 / (rows + 1)
        for i in range(num_circles):
            row = i // cols
            col = i % cols
            x = col * spacing + spacing / 2
            y = row * spacing + (spacing / 2) * (1 if row % 2 == 0 else 0)
            centers[i] = [x, y]
    elif config == 'grid':
        # Generate square grid
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        spacing = 1.0 / (rows + 1)
        for i in range(num_circles):
            row = i // cols
            col = i % cols
            x = col * spacing + spacing / 2
            y = row * spacing + spacing / 2
            centers[i] = [x, y]
    elif config == 'random':
        # Generate random positions inside the square
        centers = np.random.rand(num_circles, 2)
        # Ensure all points are inside the square
        centers = np.clip(centers, 0, 1)
    elif config == 'nested':
        # Generate nested circles (similar to hexagonal but centered)
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        spacing = 1.0 / (rows + 1)
        for i in range(num_circles):
            row = i // cols
            col = i % cols
            x = col * spacing + spacing / 2
            y = row * spacing + spacing / 2
            centers[i] = [x, y]
    elif config == 'zigzag':
        # Generate zigzag pattern similar to hexagonal
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        spacing = 1.0 / (rows + 1)
        for i in range(num_circles):
            row = i // cols
            col = i % cols
            x = col * spacing + spacing / 2
            y = row * spacing
            if row % 2 == 1:
                y += spacing / 2
            centers[i] = [x, y]

def validate_packing(centers, radii):

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
