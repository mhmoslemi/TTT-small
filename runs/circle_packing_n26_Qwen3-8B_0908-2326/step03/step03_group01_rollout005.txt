from scipy.optimize import Bounds
from scipy.optimize import NonlinearConstraint

import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # This avoids symmetry and lays the ground for better packing
    centers = np.zeros((num_circles, 2))

        row = i // 5
        col = i % 5

    bounds = Bounds(0, 1)

    # Function to compute radii based on nearest neighbors and boundaries
    def compute_radii(centers_opt):
        radii_opt = np.zeros(num_circles)
        for i in range(num_circles):
            x_i, y_i = centers_opt[i]
            r_max = min(x_i, 1 - x_i, y_i, 1 - y_i)

                    continue

                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                r_max = min(r_max, (dist - eps) / 2)
            radii_opt[i] = r_max
        return radii_opt

    # Objective function to maximize the sum of radii
    def objective(x):
