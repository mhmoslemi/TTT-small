import scipy.optimize as opt
import matplotlib.pyplot as plt
from concurrent.futures import ThreadPoolExecutor

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    max_attempts = 5  # Try with different initial configurations
    best_centers = None
    best_radii = None
    best_sum_radii = 0.0

    def generate_initial_centers(num_circles, variant='grid'):
        if variant == 'grid':
            grid_size = int(np.sqrt(num_circles))
            spacing = 1.0 / (grid_size + 1)
            centers = np.zeros((num_circles, 2))
            for i in range(num_circles):
                row = i // grid_size
                col = i % grid_size
                x = col * spacing + spacing / 2
                y = row * spacing + spacing / 2
                centers[i] = [x, y]
            return centers
        elif variant == 'hex_grid':
            rows = int(np.sqrt(num_circles))
            cols = (num_circles + rows - 1) // rows
            spacing = 1.0 / (rows + 1)
            centers = np.zeros((num_circles, 2))
            for i in range(num_circles):
                row = i // cols
                col = i % cols
                x = col * spacing + spacing / 2

            return centers

            return centers
        else:
            raise ValueError("Unknown initial grid variant")

        return 0
