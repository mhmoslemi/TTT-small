def compute_hexagonal_positions(n, rows, cols):
    positions = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * math.sqrt(3) / 2
        positions[i, 0] = x
        positions[i, 1] = y
    return positions

def scale_to_unit_square(positions):
    x_min, x_max = np.min(positions[:, 0]), np.max(positions[:, 0])
    y_min, y_max = np.min(positions[:, 1]), np.max(positions[:, 1])

    scaled[:, 0] = (positions[:, 0] - x_min) / (x_max - x_min)
    scaled[:, 1] = (positions[:, 1] - y_min) / (y_max - y_min)
    return scaled
