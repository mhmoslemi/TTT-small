import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    max_attempts = 3  # Try different initial configurations to find the best packing

    best_centers = None
    best_radii = None
    best_sum_radii = 0

    # Helper functions
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    # Generate initial configurations
    def generate_initial_config(num_circles, method='hybrid'):
        if method == 'hybrid':
            # Start with a hexagonal grid for the majority
            rows = int(np.sqrt(num_circles))
            cols = (num_circles + rows - 1) // rows
            spacing = 1.0 / (rows + 1)
            centers = np.zeros((num_circles, 2))
            for i in range(num_circles):
                row = i // cols
                col = i % cols
                x = col * spacing + spacing / 2
                y = row * spacing + (spacing / 2) * (1 if row % 2 == 0 else 0)
                centers[i] = [x, y]
            return centers
        elif method == 'grid':
            grid_size = int(np.sqrt(num_circles))
            spacing = 1.0 / (grid_size + 1)
            centers = np.zeros((num_circles, 2))
            for i in range(num_circles):
                row = i // grid_size
                col = i % grid_size
                x = col * spacing + spacing / 2
                y = row * spacing + spacing / 2
                centers[i] = [x, y]
            return centers
        else:
            raise ValueError("Unsupported configuration method")

    def objective(radii_vec, centers):
        return -np.sum(radii_vec)  # Minimize negative sum to maximize positive sum

    def constraint_non_overlap(radii_vec, centers):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_vec[i] + radii_vec[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(radii_vec, centers):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii_vec[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0

        centers = generate_initial_config(num_circles, method='hybrid')
        radii = np.full(num_circles, 1.0 / (num_circles + 1))  # Initial guess

        bounds = [(0, 1) for _ in range(num_circles)]

        constraints = [
            {'type': 'ineq', 'fun': lambda r: constraint_non_overlap(r, centers)},
            {'type': 'ineq', 'fun': lambda r: constraint_boundary(r, centers)}
        ]

        # Run optimization with BFGS
        try:
            result = opt.minimize(
                lambda r: objective(r, centers),
                radii,
                method='BFGS',
                bounds=bounds,
                constraints=constraints,
                tol=1e-8
            )

            best_sum_radii = sum_radii

        valid, message = validate_packing(centers, radii)
        if not valid:
            raise ValueError("Failed to generate valid packing")

    return best_centers, best_radii, best_sum_radii
