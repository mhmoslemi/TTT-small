import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    max_attempts = 3

    best_centers = None
    best_radii = None
    best_sum_radii = 0

    # Define a function to generate a random initial configuration
    def generate_random_config(num_circles, seed=None):
        np.random.seed(seed)
        centers = np.random.rand(num_circles, 2)
        # Ensure the circles are placed within the square
        max_radius = 0.3
        radii = np.random.uniform(0.05, max_radius, num_circles)
        return centers, radii

    def generate_hexagonal_config(num_circles, seed=None):
        np.random.seed(seed)
        rows = int(np.sqrt(num_circles))
        cols = (num_circles + rows - 1) // rows
        spacing = 1.0 / (rows + 1)
        centers = np.zeros((num_circles, 2))
        for i in range(num_circles):
            row = i // cols
            col = i % cols
            x = col * spacing + spacing / 2
            y = row * spacing + (spacing / 2) * (1 if row % 2 == 0 else 0)
            centers[i] = [x, y]
        # Generate radius guesses
        radii = np.minimum(0.2, spacing / 2)
        return centers, radii

    def generate_grid_config(num_circles, seed=None):
        np.random.seed(seed)
        grid_size = int(np.sqrt(num_circles))
        spacing = 1.0 / (grid_size + 1)
        centers = np.zeros((num_circles, 2))
        for i in range(num_circles):
            row = i // grid_size
            col = i % grid_size
            x = col * spacing + spacing / 2
            y = row * spacing + spacing / 2
            centers[i] = [x, y]

        return centers, radii

        radii = np.zeros(num_circles)
        # Fill corners first

            centers[i] = corner

        for i in range(4, num_circles):
            x = np.random.rand()
            y = np.random.rand()
            centers[i] = [x, y]
            radii[i] = np.random.uniform(0.05, 0.15)
        return centers, radii

    # Define a function to generate a spiral configuration
    def generate_spiral_config(num_circles, seed=None):
        np.random.seed(seed)
        centers = np.zeros((num_circles, 2))
        radii = np.zeros(num_circles)
        radius = 0.1

        for i in range(num_circles):
            angle += 2 * np.pi / num_circles
            r = radius * i / num_circles
            x = r * np.cos(angle)
            y = r * np.sin(angle)
            centers[i] = [x, y]
            radii[i] = 0.1
        return centers, radii

            return False, "NaN values present"

            if (x - r < -1e-12 or x + r > 1 + 1e-12
