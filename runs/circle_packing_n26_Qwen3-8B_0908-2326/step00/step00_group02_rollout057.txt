import numpy as np
import scipy.optimize as optimize

def run_packing():

    grid_size = 4  # number of cells in x and y directions
    grid_x, grid_y = np.meshgrid(np.linspace(0.5 / grid_size, 1 - 0.5 / grid_size, grid_size),
                                 np.linspace(0.5 / grid_size, 1 - 0.5 / grid_size, grid_size))
    centers = np.column_stack((grid_x.flatten(), grid_y.flatten()))

    # Convert to optimization variables
    centers_flat = centers.flatten()
    radii_flat = radii.flatten()

    def objective(x):
        return -np.sum(x[1:])  # We minimize the negative sum to maximize the actual sum

    def square_bounds(x):
        return np.array([x[0] - x[1] >= -1e-12,  # x - r >= 0

                         x[2] - x[3] >= -1e-12,  # y - r >= 0
                         x[2] + x[3] <= 1 + 1e-12])  # y + r <= 1

    # Constraint: circles must not overlap
    def non_overlap(x):
        distances = np.zeros(num_circles)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = x[2 * i] - x[2 * j]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):

        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
