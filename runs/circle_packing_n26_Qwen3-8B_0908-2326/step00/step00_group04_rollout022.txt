import numpy as np
import scipy.optimize as optimize
import itertools

def run_packing():
    # Hexagonal packing parameters
    n_circles = 26
    # Number of rows in hexagonal packing
    num_rows = 6
    # Number of circles per row (fewer in the middle for hexagonal packing)
    num_cols = 5
    # Height of a hexagon (distance between rows)
    hex_row_height = np.sqrt(3) / 2
    # Width of a hexagon (distance between adjacent circles)
    hex_col_width = 1

    # Calculate the initial positions and radii
    centers = []
    radii = []

    # Calculate the minimum radius such that all circles fit in the unit square
    # With hexagonal packing in a unit square, we can approximate the radius
    # as the optimal value for the most compact configuration
    min_radius = 0.1
    max_possible_radius = 0.15

    # Generate the initial hexagonal grid
    for row in range(num_rows):
        for col in range(num_cols):
            x = col * hex_col_width
            y = row * hex_row_height
            centers.append([x, y])
            radii.append(min_radius)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define the objective function to maximize the sum of radii
    def objective_function(x):
        # x is the flattened array of centers and radii
        vec = x.reshape((n_circles, 4))  # 4 variables per circle: x, y, r, r
        centers_opt = vec[:, :2]
        radii_opt = vec[:, 2]
        return -np.sum(radii_opt)  # Negative because we are minimizing

    # Define constraints: no circle can go outside the square and no two circles can overlap
    def constraint_outside(x):
        vec = x.reshape((n_circles, 4))
        centers_opt = vec[:, :2]
        radii_opt = vec[:, 2]
        for i in range(n_circles):
            x_i, y_i = centers_opt[i]
            r_i = radii_opt[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return np.inf
        return 0

    def constraint_overlap(x):
        vec = x.reshape((n_circles, 4))
        centers_opt = vec[:, :2]
        radii_opt = vec[:, 2]
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                x_i, y_i = centers_opt[i]
                x_j, y_j = centers_opt[j]
                r_i = radii_opt[i]
                r_j = radii_opt[j]
                dist = np.hypot(x_i - x_j, y_i - y_j)
                if dist < r_i + r_j - 1e-6:
                    return np.inf
        return 0

    initial_guess = np.zeros(n_circles * 4)

        initial_guess[i * 4] = centers[i, 0]

    bounds = []

    # Set up the optimization problem
    cons = [

        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Run the optimization
    result = optimize.minimize(

        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,

    sum_radii = np.sum(best_radii)
    return best_centers, best_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
