import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial grid layout
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Uniformly spaced in a grid
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05  # Initial radius

    idx = 0
    for i in range(rows):

            if idx >= n:

            centers[idx] = [x, y]

    def objective(radii_vec):
        return -np.sum(radii_vec)

    def constraint_center_in_square(radii_vec, centers):

                return -1
        return 0

    def constraint_no_overlap(radii_vec, centers):
        for i in range(n):
