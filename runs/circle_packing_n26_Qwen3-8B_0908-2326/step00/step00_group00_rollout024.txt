import warnings

    num_circles = 26
    grid_size = 5  # Create a grid that fits within the square

        radius_initial[i] = 0.04  # Small initial radius for optimization

    def objective_function(radii):
        return -np.sum(radii)  # Minimize the negative sum to maximize sum

        return 1 + 1e-12 - x - r  # X upper bound
    def constraint_circle_boundary_inequality_3(x, y, r, idx):
        return y - r + 1e-12  # Y lower bound

        return 1 + 1e-12 - y - r  # Y upper bound

    def constraint_circle_overlap(x1, y1, r1, x2, y2, r2):

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]

        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
