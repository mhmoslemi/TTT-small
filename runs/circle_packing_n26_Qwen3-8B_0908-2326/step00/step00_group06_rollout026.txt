import numpy as np
import scipy.optimize as optimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based placement (4x7 grid for 26 circles)
    grid_size = 4
    num_circles = 26
    col_padding = 0.05  # Extra padding for boundary adjustments
    row_padding = 0.05

    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Place circles in a grid pattern with some spacing
    for i in range(num_circles):
        col = i % grid_size
        row = i // grid_size
        x = col * (1 - 2 * col_padding) / (grid_size - 1)
        y = row * (1 - 2 * row_padding) / (grid_size - 1)
        centers[i] = [x, y]

    # Use a simple equal radius approach as a starting point
    initial_radius = 0.05
    radii[:] = initial_radius

    # Define constraints for optimization
    def objective_function(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative sum for maximization

    def constraint_no_overlap(i, j, centers, radii):
        def constraint(radii):
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - (radii[i] + radii[j]) + 1e-12
        return constraint

    def constraint_boundary(i, centers, radii):
        def constraint(radii):
            x, y = centers[i]
            return (x - radii[i]) - (-1e-12)  # Ensure circle does not go out of bounds
        return constraint

    # Initial guess
    initial_guess = radii.copy()

    # Set up optimization problem
    constraints = []
    for i in range(num_circles):
        # Add boundary constraint for circle i
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i, centers, radii),
                            'args': (centers, radii)})
        # Add non-negativity constraint for radius
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i], 'args': (radii, i)})

    for i in range(num_circles):

            # Add non-overlap constraint between circles i and j
            constraints.append({'type': 'ineq', 'fun': constraint_no_overlap(i, j, centers, radii),

    # Use a local optimizer to maximize the sum of radii
    result = optimize.minimize(
        objective_function,
        initial_guess,
        method='SLSQP',

    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
