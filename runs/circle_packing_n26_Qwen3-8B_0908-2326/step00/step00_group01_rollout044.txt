import scipy.optimize as opt

def run_packing():

                if i % 2 == 0:

        return grid

    def objective(x, n):

    def constraint_center_bounds(x, n):

            r = radii[i]
            if x_center - r < -1e-12 or x_center + r > 1 + 1e-12:
                return -1
            if y_center - r < -1e-12 or y_center + r > 1 + 1e-12:

        radii = x[2*n:]

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
