import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    # Initial grid layout: 5 rows and 5 columns with spacing to leave room for radius
    num_circles = 26
    grid_rows = 5
    grid_cols = 6  # 5 x 6 grid = 30 circles, but we'll take only 26
    spacing = 1.0 / (grid_rows + 1)  # Distribute evenly

    # Initial grid centers
    centers = np.zeros((num_circles, 2))
    idx = 0
    for r in range(grid_rows):
        for c in range(grid_cols):
            if idx >= num_circles:
                break
            x = (c + 1) * spacing
            y = (r + 1) * spacing
            centers[idx] = [x, y]
            idx += 1

    # Initial radii: small to start with
    radii = np.full(num_circles, 0.03)

    # Objective function to maximize total radius
    def objective(x):
        # x is flat array: [r0, r1, ..., r25]
        r = x.reshape(-1)
        return -np.sum(r)  # Negative for maximization

    # Constraints: boundaries and non-overlap
    def constraint_boundary(x):
        r = x.reshape(-1)
        for i in range(num_circles):
            x_i, y_i = centers[i]
            if x_i - r[i] < 0 or x_i + r[i] > 1 or y_i - r[i] < 0 or y_i + r[i] > 1:
                return -1  # Violated
        return 0

    def constraint_overlap(x):
        r = x.reshape(-1)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return -1  # Violated
        return 0

    # Optimization bounds: radii >= 0
    bounds = [(0, 1) for _ in range(num_circles)]

    # Optimization
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-6
    )

    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    radii = result.x
    centers = centers[:num_circles]
    total_radius = np.sum(radii)

    # Plot the packing
    plt.figure(figsize=(6, 6))
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        circle = plt.Circle((x, y), r, edgecolor='r', facecolor='none', lw=1)
        plt.gca().add_patch(circle)

    plt.xlim(0, 1)
    plt.ylim(0, 1)
    plt.axis('equal')
    plt.title(f"Packing with Total Radius: {total_radius:.4f}")
    plt.show()

    return centers, radii, total_radius
