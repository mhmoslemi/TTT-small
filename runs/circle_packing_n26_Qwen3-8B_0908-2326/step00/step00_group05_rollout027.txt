import numpy as np
import scipy.optimize as opt

def run_packing():

    n = 26
    num_rows = 5
    num_cols = 5 + 1  # For 5 rows, 6 columns (to fit 26 circles)
    
    # Radius for the first circle as a starting guess
    initial_radius = 0.1
    
    # Initial positions in hexagonal packing
    centers = np.zeros((n, 2))
    radii = np.full(n, initial_radius)
    
    # Hexagonal packing parameters
    hex_row_height = np.sqrt(3) * initial_radius
    hex_col_width = 2 * initial_radius
    
    # Distribute the circles in a hexagonal grid
    for i in range(n):
        row = i // num_cols
        col = i % num_cols
        x = col * hex_col_width
        y = row * hex_row_height
        if row % 2 == 1:
            x += hex_col_width / 2
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = initial_radius

    # Bounds for centers and radii
    center_bounds = [(0, 1) for _ in range(2 * n)]
    radius_bounds = [(0, 1) for _ in range(n)]

    # Objective function: maximize sum of radii
    def objective(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:].reshape(n)
        return -np.sum(radii_flat)  # Minimize negative sum to maximize sum

    # Constraints: no overlap and within unit square
    def constraint1(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:].reshape(n)
        for i in range(n):
            x_center, y_center = centers_flat[i]
            r = radii_flat[i]
            if not (0 <= x_center - r and x_center + r <= 1 and 0 <= y_center - r and y_center + r <= 1):
                return -1e6  # Penalize out-of-bound circles
        return 0

    def constraint2(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:].reshape(n)

                dist = np.sqrt(dx**2 + dy**2)

        return 0

    # Create bounds and constraints
    bounds = opt.Bounds(center_bounds + radius_bounds)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
