from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    max_radius = 1.0  # Initial maximum radius

    for i in range(num_circles):
        row = i // 5
        col = i % 5

        centers[i] = [x, y]
        radii[i] = min(0.15, max_radius - 0.02 * i)  # Gradually reduce radius for outer circles

    # Objective function to maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum (equivalent to maximizing sum)

    # Constraint: circles must be inside the unit square
    def boundary_constraint(radii, centers):
        return np.array([np.all(centers[i] - radii[i] >= 0) and np.all(centers[i] + radii[i] <= 1)
                         for i in range(num_circles)])

    # Constraint: circles must not overlap
    def non_overlap_constraint(radii, centers):
        dists = cdist(centers, centers)
        return np.array([dists[i, j] >= radii[i] + radii[j] - 1e-6
                         for i in range(num_circles) for j in range(i + 1, num_circles)])

    # Optimization
    bounds = [(0, 1.0) for _ in range(num_circles)]  # Bounds for radii

        objective,

        bounds=bounds,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
