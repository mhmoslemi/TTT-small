from scipy.optimize import minimize

def run_packing():

    circle_index = 0

    rows = 5
    cols = 5
    offset = 0.1
    spacing = 0.2

            y = i * spacing + offset
            centers[circle_index] = [x, y]
            circle_index += 1

        return -np.sum(radii)

def validate_packing(centers, radii):

    if np.isnan(centers).any() or np.isnan(radii).any():
