from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    n = 26

    # Initial guess for circle centers (simple grid-like arrangement)
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 6
        col = i % 6
        centers[i, 0] = col / 5
        centers[i, 1] = row / 5

    # Initial guess for radii (equal radii)
    radii = np.full(n, 0.1)

    # Objective function to maximize: sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative because we minimize

    def boundary_constraint(radii):
        return np.array([np.sum((centers[i] - radii[i]) ** 2) for i in range(n)])

        return np.array([radii[i] for i in range(n)])  # Radii must be non-negative

    # Constraint: Circles must not overlap
    def overlap_constraint(radii):
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dists[i, j] = np.sqrt(dx ** 2 + dy ** 2)
                dists[j, i] = dists[i, j]
        return np.array([dists[i, j] - radii[i] - radii[j] for i in range(n) for j in range(n) if i < j])

    # Define bounds for radii (non-negative)

    # Optimization problem

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
