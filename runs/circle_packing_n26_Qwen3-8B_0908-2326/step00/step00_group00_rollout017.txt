import scipy.optimize as optimize

def run_packing():

            y = i * np.sqrt(3) / 2

    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= max_x if max_x > 0 else 1

    radii = np.ones(n) * 0.1

    def objective(r):
