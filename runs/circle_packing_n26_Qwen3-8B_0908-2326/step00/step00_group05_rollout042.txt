import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial configuration: hexagonal grid of 26 circles
    num_circles = 26
    rows = int(np.sqrt(num_circles))
    cols = (num_circles + rows - 1) // rows

    # Step 1: Generate a hexagonal grid layout (with spacing)
    spacing = 0.5  # Initial spacing (can be adjusted later)
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col * spacing
        y = row * (np.sqrt(3)/2) * spacing
        centers[i] = [x, y]

    # Step 2: Optimize the sum of the radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    def constraints(centers, radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Boundary constraints
            constraints.append({'type': 'ineq', 'fun': lambda r: r - 1e-12})
            constraints.append({'type': 'ineq', 'fun': lambda r: 1 - r - 1e-12})
            constraints.append({'type': 'ineq', 'fun': lambda x: x - r - 1e-12})
            constraints.append({'type': 'ineq', 'fun': lambda x: 1 - x - r - 1e-12})
            constraints.append({'type': 'ineq', 'fun': lambda y: y - r - 1e-12})
            constraints.append({'type': 'ineq', 'fun': lambda y: 1 - y - r - 1e-12})
        # Overlap constraints

            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append({'type': 'ineq', 'fun': lambda r: dist - r[i] - r[j] - 1e-12})
        return constraints

    # Initial guess: same for all circles
    initial_radii = np.full(num_circles, 0.05)

    # Optimization settings
    bounds = [(0, 1.0) for _ in range(num_circles)]

    # Run the optimization
    result = opt.minimize(
        fun=objective,
        x0=initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints(centers, initial_radii)
    )

    optimal_radii = result.x

    # Validate the packing
    valid, msg = validate_packing(centers, optimal_radii)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    # Return the results
    return centers, optimal_radii, np.sum(optimal_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
