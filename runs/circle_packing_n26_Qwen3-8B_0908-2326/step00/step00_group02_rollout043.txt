import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal grid
    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    # Hexagonal grid
    centers = np.zeros((num_circles, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 1:
                x = j * np.sqrt(3) / 2
            else:
                x = (j + 0.5) * np.sqrt(3) / 2
            y = i * 1.5
            if idx < num_circles:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
    
    # Scale centers to fit into the unit square
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= max_x if max_x > 0 else 1
    centers /= max_y if max_y > 0 else 1

    # Initial radii (uniform)
    radii = np.ones(num_circles) * 0.1

    # Define objective function: maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint_overlap(radii, centers):
        n = centers.shape[0]
        dist_matrix = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(radii, centers):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -1
        return 0

    # Optimization with bounds
    bounds = [(0, 0.5) for _ in range(num_circles)]  # Max radius of 0.5

    # Initial guess
    initial_guess = radii.copy()

    # Set up optimization problem
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap, 'args': (centers,)},
        {'type': 'ineq', 'fun': constraint_boundary, 'args': (centers,)}
    ]

    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        tol=1e-8

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
