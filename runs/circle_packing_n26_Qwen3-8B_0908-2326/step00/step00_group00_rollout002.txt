import scipy.optimize as opt
import math

def run_packing():

    radii = np.zeros(n)

    # Use hexagonal pattern to ensure better spacing
    num_rows = 6
    num_cols = 5

    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # 1. All circles must be within the unit square
    # 2. No two circles can overlap (distance between centers > sum of radii)

    def constraints(radii):
