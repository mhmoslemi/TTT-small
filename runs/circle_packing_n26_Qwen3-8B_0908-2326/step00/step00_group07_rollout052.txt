import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing in a unit square
    def hexagonal_grid(n):
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))
        centers = np.zeros((n, 2))
        idx = 0
        for row in range(rows):
            for col in range(cols):
                if row % 2 == 0:
                    x = col / (cols - 1)
                else:
                    x = (col + 0.5) / (cols - 1)
                y = row / (rows - 1)
                centers[idx] = [x, y]
                idx += 1
        return centers

    # Initial guess for radii
    def initial_radii(n):
        return np.ones(n) * 0.05  # Start with small radii

    centers = hexagonal_grid(26)
    radii = initial_radii(26)

    # Objective function to maximize the sum of radii
    def objective(radii_flat):
        return -np.sum(radii_flat)  # To be minimized

    # Constraints
    def constraint_boundary(radii_flat):
        for i in range(len(radii_flat)):
            x, y = centers[i]
            r = radii_flat[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1e6
        return 0

    def constraint_overlap(radii_flat):
        for i in range(len(radii_flat)):
            for j in range(i + 1, len(radii_flat)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -1e6
        return 0

    # Optimization
    bounds = [(0, 0.5) for _ in range(26)]
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    result = optimize.minimize(
        objective,
        radii.flatten(),
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 1000}
    )

    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    optimal_radii = result.x
    optimal_centers = centers.copy()
    sum_radii = np.sum(optimal_radii)

    return optimal_centers, optimal_radii, sum_radii
