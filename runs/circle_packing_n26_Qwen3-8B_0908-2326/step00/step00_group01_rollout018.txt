import numpy as np
import scipy.optimize as opt

def run_packing():
    # Define number of circles and initial positions
    n_circles = 26

    # Initial guess for centers (some large circles at the center, others around)
    centers = np.zeros((n_circles, 2))

    # Place 10 large circles in a 4x4 grid near the center
    for i in range(4):
        for j in range(4):
            centers[4 * i + j, 0] = 0.3 + 0.25 * i
            centers[4 * i + j, 1] = 0.3 + 0.25 * j

    # Place 16 small circles around the large ones
    for i in range(16):
        centers[4 * 4 + i, 0] = np.random.uniform(0.1, 0.9)

    max_radius = 0.12
    radii = np.full(n_circles, max_radius)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint functions
    def boundary_constraint(radii):
        return np.sum((centers - radii[:, np.newaxis]) ** 2) - 1

    def non_overlap_constraints(radii):
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Optimization
    bounds = [(0.001, 0.2) for _ in range(n_circles)]  # Avoid zero radius
    constraints = [
        {'type': 'ineq', 'fun': lambda r: boundary_constraint(r)},
        *non_overlap_constraints
    ]

    result = opt.minimize(

        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    # Extract the final centers and radii
    centers_final = centers

    sum_radii = np.sum(radii_final)
