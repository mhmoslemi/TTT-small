import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial grid configuration with hexagonal packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Define grid positions
    rows, cols = 5, 6
    spacing = 1.0 / (cols - 1)

            x = j * spacing
            y = i * spacing
            centers[i * cols + j] = [x, y]

    initial_radius = 0.045  # A reasonable initial guess
    radii[:] = initial_radius

        # x contains all radii (flat array)

        radii = radii_flat.reshape((num_circles,))
        sum_radii = np.sum(radii)
        return -sum_radii  # Minimize negative sum to maximize sum

    # Constraints
    def constraints(x):
        radii_flat = x
        radii = radii_flat.reshape((num_circles,))
        centers_flat = centers.reshape((num_circles, 2))

        for i in range(num_circles):

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
