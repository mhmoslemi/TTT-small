import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n_circles = 26

    # Initial guess: distribute circles in a grid-like pattern
    # We will place them in a hexagonal-like arrangement to maximize efficiency
    # We will also ensure they fit within the unit square

    # First, define a grid of centers
    rows = int(np.ceil(np.sqrt(n_circles)))
    cols = int(np.ceil(n_circles / rows))
    spacing = 1.0 / (cols + 1)  # Adjust to fit in the unit square

    centers = np.zeros((n_circles, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= n_circles:
                break
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2
            if i % 2 == 1:
                x += spacing / 2
            centers[idx] = [x, y]
            idx += 1

    # Generate initial radii equal for all
    radii = np.ones(n_circles) * 0.1  # Start with small radii and optimize

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative of the sum

    # Constraints
    def constraints(radii):
        # Constraint 1: All circles must lie within the unit square
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return np.inf  # Penalize for exceeding boundaries
        # Constraint 2: No two circles overlap
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j]:
                    return np.inf  # Penalize for overlap
        return 0  # Valid configuration

    # Run optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        method='Nelder-Mead',
        bounds=[(0, 1) for _ in range(n_circles)],
        constraints=[{
            'type': 'ineq',
            'fun': constraints
        }],
        options={'xatol': 1e-10, 'maxiter': 1000}
    )

    # Final radii and centers
    radii_opt = result.x
    centers_opt = centers
    sum_radii = np.sum(radii_opt)

    # Validate
    assert validate_packing(centers_opt, radii_opt)[0], "Invalid packing"

    # Return the result
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
