import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n_circles = 26
    # Initial guess: staggered hexagonal grid with some padding
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    # Initial radius for all circles
    initial_radius = 0.05
    radii[:] = initial_radius
    # Number of rows in our hexagonal grid
    rows = 5
    cols_per_row = 6
    # Spacing between centers
    dx = 1.0 / (cols_per_row + 1)
    dy = np.sqrt(3) / (rows + 1)
    # Place the circles in hexagonal grid pattern
    for i in range(rows):
        for j in range(cols_per_row):
            x = dx * (j + 1)
            y = dy * (i + 1)

                x += dx / 2
            centers[i * cols_per_row + j] = [x, y]
    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint1(radii):
        # Constraint: each circle is inside the square
        for i in range(n_circles):
            x, y = centers[i]
            if not (0 <= x - radii[i] and x + radii[i] <= 1 and 0 <= y - radii[i] and y + radii[i] <= 1):
                return -1
        return 0

    def constraint2(radii):
        # Constraint: no two circles overlap
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    cons = [
        {'type': 'ineq', 'fun': constraint1},

    ]

    # Bounding box for radii (non-negative)

        bounds=bounds,
        constraints=cons,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
