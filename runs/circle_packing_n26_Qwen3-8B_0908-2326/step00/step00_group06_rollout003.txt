import numpy as np
import cvxpy as cp

def run_packing():
    # Number of circles
    n = 26
    
    # Initialize centers and radii as variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)
    
    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Constraints
    constraints = []
    
    # Each circle must lie within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints += [
            x >= r,
            x <= 1 - r,
            y >= r,
            y <= 1 - r
        ]
    
    # No two circles may overlap
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = cp.sqrt(dx*dx + dy*dy)
            constraints += [dist >= radii[i] + radii[j]]
    
    # Solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, verbose=False)
    
    # Extract the solution
    centers_val = centers.value
    radii_val = radii.value
    sum_radii = np.sum(radii_val)
    
    return (centers_val, radii_val, sum_radii)
