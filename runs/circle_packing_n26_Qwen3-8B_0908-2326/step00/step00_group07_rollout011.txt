import numpy as np
import random
import copy
from deap import base, creator, tools, algorithms

def run_packing():
    # Define the number of circles and the unit square
    n_circles = 26
    square_size = 1.0

    # Step 1: Generate an initial hexagonal grid layout
    def hexagonal_grid(n, square_size):
        radius = 0.0
        centers = []
        for i in range(n):
            row = i // 2
            col = i % 2
            x = col * 2 * radius + 0.5 * radius
            y = row * (np.sqrt(3) * radius) + 0.5 * radius
            # Adjust to fit in the unit square
            x = x / square_size
            y = y / square_size
            centers.append([x, y])
        return np.array(centers)

    # Initial estimate of radius based on hexagonal grid
    initial_centers = hexagonal_grid(n_circles, square_size)
    max_possible_radius = 0.2  # estimate based on hexagonal packing

    # Function to compute fitness (maximize sum of radii)
    def evaluate(individual):
        centers = individual[:2 * n_circles].reshape((n_circles, 2))
        radii = individual[2 * n_circles:]
        valid, msg = validate_packing(centers, radii)
        if not valid:
            return -np.inf,  # Invalid solution
        return np.sum(radii),  # Maximize sum of radii

    # Step 2: Use a genetic algorithm to optimize the positions and radii
    # Define the genotype: 2*n_circles floats for center coordinates, n_circles floats for radii
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    # Genetic algorithm setup
    toolbox = base.Toolbox()
    toolbox.register("attr_float", random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=2 * n_circles + n_circles)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("mate", tools.cxBlend, alpha=0.1)
    toolbox.register("mutate", tools.mutGaussian, mu=0.0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("evaluate", evaluate)

    # Initialize population and run the algorithm
    population = toolbox.population(n=50)
    algorithms.eaMuPlusLambda(population, toolbox, mu=50, lambda_=50, cxpb=0.5, mutpb=0.1, ngen=200, verbose=False)

    # Extract the best solution
    best_individual = tools.selBest(population, k=1)[0]
    centers = best_individual[:2 * n_circles].reshape((n_circles, 2))
    radii = best_individual[2 * n_circles:]

    # Step 3: Refine the packing locally using simulated annealing
    def simulated_annealing(centers, radii):
        # Simple local search
        for _ in range(100):
            # Perturb positions slightly
            perturb = np.random.uniform(-0.01, 0.01, size=centers.shape)

            valid, msg = validate_packing(centers_new, radii)
            if valid:

        return centers, radii

    # Final validation

            radii = np.full(n_circles, 0.01)
            valid, msg = validate_packing(centers, radii)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
