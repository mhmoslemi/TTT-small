import scipy.optimize as optimize

def run_packing():

    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))

    initial_radii = np.zeros(num_circles)

    def objective(x):
        return -x.sum()  # We minimize the negative sum to maximize the sum

        return x

    def constraint_radii(x):
        return x

    def constraint_bounds(x):
        for i in range(num_circles):
            x_i, y_i, r_i = x[i * 3 : (i + 1) * 3]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:

        return 0

    def constraint_overlap(x):
        for i in range(num_circles):

                x_j, y_j, r_j = x[j * 3 : (j + 1) * 3]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
