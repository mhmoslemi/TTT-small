import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define the number of circles
    n_circles = 26

    # Initial hexagonal packing in a unit square
    # Hexagonal packing:n_circles = 26
    # Rows: sqrt(26) ≈ 5.1 → let's use 5 rows

    # Initial positions
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)

    # First define vertical spacing
    vertical_spacing = 1.0 / (n_circles**0.5)
    horizontal_spacing = (1.0 - 2 * vertical_spacing) / (n_circles**0.5)

    for i in range(n_circles):
        row = i // 5
        col = i % 5
        row_offset = (row - 2) * vertical_spacing
        col_offset = (col - 2) * horizontal_spacing
        centers[i, 0] = 0.5 + col_offset
        centers[i, 1] = 0.5 + row_offset

    # Initial guess for radii based on spacing
    rad = (horizontal_spacing + vertical_spacing) / 2
    radii.fill(rad)

    # Objective function to maximize sum of radii
    def objective(x):
        return -x.sum()  # Minimize negative of sum to maximize

    # Optimization bounds
    bounds = []
    for i in range(n_circles):
        lower = 0.0
        upper = 1.0 - (np.sqrt(2) * rad)  # No overlap with boundaries
        bounds.append((lower, upper))

    # Constraint functions
    def circle_boundary_constraint(x, i):
        x_i, y_i = centers[i]
        r_i = x[i]
        if x_i - r_i < -1e-12 or x_i + r_i > 1.0 + 1e-12:
            return x_i - r_i + 1e-12
        if y_i - r_i < -1e-12 or y_i + r_i > 1.0 + 1e-12:
            return y_i - r_i + 1e-12
        return 0

    def circle_pair_overlap_constraint(x, i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        r_i = x[i]
        r_j = x[j]
        return dist - r_i - r_j + 1e-12

    # Set up constraints for optimization
    constraints = []

    # Boundary constraints for each circle
    for i in range(n_circles):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: circle_boundary_constraint(x, i)})

    # Pairwise overlap constraints between circles
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: circle_pair_overlap_constraint(x, i, j)})

    # Initial guess for the radii
    initial_guess = radii.copy()

    # Run optimization
    result = optimize.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8
    )

    # Extract the optimized centers and radii
    optimized_radii = result.x
    centers = np.atleast_2d(centers).T
    sum_radii = optimized_radii.sum()

    # Validate the packing (if failed, the error will be raised)
    valid, msg = validate_packing(centers, optimized_radii)
    if not valid:
        raise ValueError(msg)

    return centers, optimized_radii, sum_radii
