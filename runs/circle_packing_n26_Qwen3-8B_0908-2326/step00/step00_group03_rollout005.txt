import numpy as np
from scipy.optimize import minimize
import time

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialize centers and radii
    n = 26
    np.random.seed(42)
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1  # Initial radii are small to avoid overlaps

    # Objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint_center_bounds(radii):
        return np.array([0.0 for _ in range(n)])

    def constraint_radius_bounds(radii):
        return np.array([r >= 0 for r in radii])

    def constraint_circle_overlap(radii):
        # Calculate the minimum distance between circle centers and ensure
        # it is at least the sum of radii
        dists = np.zeros(n * (n - 1) // 2)
        i = 0
        for j in range(n):
            for k in range(j + 1, n):
                dx = centers[j, 0] - centers[k, 0]
                dy = centers[j, 1] - centers[k, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                dists[i] = dist - (radii[j] + radii[k])
                i += 1
        return dists

    # Set up the optimization problem
    bounds = [(0.0, 0.5) for _ in range(n)]  # Radii cannot exceed 0.5
    constraints = [
        {'type': 'ineq', 'fun': lambda r: constraint_center_bounds(r)},
        {'type': 'ineq', 'fun': lambda r: constraint_radius_bounds(r)},
        {'type': 'ineq', 'fun': constraint_circle_overlap},
    ]

    # Run the optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'disp': False}
    )

    # Extract results
    optimized_radii = result.x
    sum_radii = np.sum(optimized_radii)
    return centers, optimized_radii, sum_radii
