import numpy as np
import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Calculate number of rows and columns that can fit in the square
    # We'll use a grid with rows and columns that are spaced in a way to achieve hexagonal pattern

    # Initial guess for radii (will be refined)
    radii = np.full(n, 0.05)

    # Initial positions (hexagonal grid)

    def objective(radii):

    def constraint_overlap(radii):

                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-8:

        for i in range(n):
