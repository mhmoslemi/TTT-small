def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # We start with a square and expand to a hexagonal pattern
    n_circles = 26
    centers = []
    radii = []

    for i in range(n_circles):
        row = i // 5  # 5 per row

        centers.append([x, y])
        radii.append(0.05)  # Start with a small radius

    centers = np.array(centers)
    radii = np.array(radii)

    # Define a function to compute the total radius sum
    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(i, j, centers, radii):

            return dist - (radii[i] + radii[j])
        return cons

    # Define nonlinear constraints for non-overlapping circles
    def nonlinear_constraints(x):
        x = np.reshape(x, (n_circles, 2))
        r = radii
        c = []

        # Fix the first point (place it at the bottom left)
        c.append( (x[0, 0] - 0.05, 0.0) )
        c.append( (x[0, 1] - 0.05, 0.0) )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
