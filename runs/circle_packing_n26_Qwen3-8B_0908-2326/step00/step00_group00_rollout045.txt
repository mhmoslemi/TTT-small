import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing with equal radii
    num_circles = 26
    radius = 0.05  # Initial rough estimate
    centers = np.zeros((num_circles, 2))
    r = np.full(num_circles, radius)

    # Hexagonal arrangement with initial hexagonal packing
    for i in range(num_circles):
        col = i % 6
        row = i // 6
        # Offset for hexagonal arrangement
        x = (col + 0.5 * row) * (1 - 2 * radius) / (num_circles ** 0.5)
        y = row * (np.sqrt(3) * radius)
        centers[i, 0] = x
        centers[i, 1] = y

    # Define objective function
    def objective(x):
        # x has shape (2*n, ), reshape to (n, 2)
        centers_flat = x[:2*num_circles].reshape(num_circles, 2)
        radii = x[2*num_circles:]
        # Check boundaries
        for i in range(num_circles):
            if centers_flat[i, 0] < -1e-6 or centers_flat[i, 0] > 1 + 1e-6 or \
               centers_flat[i, 1] < -1e-6 or centers_flat[i, 1] > 1 + 1e-6:
                return np.inf
        # Check overlaps
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-6:
                    return np.inf
        # Objective is to maximize sum of radii, so minimize -sum of radii
        return -np.sum(radii)

    # Define bounds for positions (0 to 1) and non-negative radii
    bounds = []
    for i in range(num_circles):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # radius
    bounds = Bounds(bounds)

    # Initial guess: flattened centers and radii
    x0 = centers.reshape(-1)
    x0 = np.concatenate((x0, r))

    # Use L-BFGS-B optimizer
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-10}
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers_opt = result.x[:2*num_circles].reshape(num_circles, 2)
    radii_opt = result.x[2*num_circles:]

    # Validate the packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise RuntimeError(f"Packing invalid: {msg}")

    return centers_opt, radii_opt, np.sum(radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
