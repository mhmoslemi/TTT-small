import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial placement of centers in a hexagonal grid pattern
    # First few circles placed at corners and midpoints
    centers = np.array([
        [0.0, 0.0],    # corner
        [1.0, 0.0],    # corner
        [0.0, 1.0],    # corner
        [1.0, 1.0],    # corner
        [0.5, 0.0],    # midpoint of bottom
        [0.5, 1.0],    # midpoint of top
        [0.0, 0.5],    # midpoint of left
        [1.0, 0.5],    # midpoint of right
    ])

    # Remaining 18 circles
    for i in range(9, 26):
        row = i // 6
        col = i % 6
        x = col / 6.0
        y = row / 6.0
        centers = np.vstack([centers, [x, y]])

    # Initial radii (start with small values)
    radii = np.full(26, 0.01)

    # Define objective function to maximize sum of radii
    def objective(x):
        # x is a flat array of radii
        return -np.sum(x)  # Negative because scipy minimizes

    # Define constraints
    def constraint_overlap(radii_flat):
        # Check that all circles are within [0,1]x[0,1]
        for i in range(26):
            if radii_flat[i] < 0:
                return -np.inf  # Invalid
            x, y = centers[i]
            if x - radii_flat[i] < -1e-12 or x + radii_flat[i] > 1 + 1e-12:
                return -np.inf
            if y - radii_flat[i] < -1e-12 or y + radii_flat[i] > 1 + 1e-12:
                return -np.inf

        # Check that no two circles overlap
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -np.inf
        return 0.0  # Valid configuration

    # Flatten radii for optimization
    radii_flat = radii.copy().flatten()

    # Set up bounds: radii must be non-negative
    bounds = [(0, 1) for _ in range(26)]

    # Setup for optimization
    cons = {'type': 'ineq', 'fun': constraint_overlap}
    result = opt.minimize(
        fun=objective,
        x0=radii_flat,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6
    )

    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")

    # Reshape back to vector
    radii = result.x.reshape(26)

    centers = centers.reshape(26, 2)

    # Validate the packing
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
