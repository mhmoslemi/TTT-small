from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import constraints

def run_packing():

    n = 26

    # Start with a hexagonal grid of circles within the unit square
    # Each circle has a fixed radius to ensure non-overlapping in the initial guess
    # We'll use a small initial radius and let the optimization expand them

    # Define the number of rows and columns we can fit
    # For simplicity, let's use a 5x6 grid
    rows = 5
    cols = 6
    radius_initial = 0.05  # Small initial radius

    # Generate initial positions (centers)
    def hexagonal_grid(n_rows, n_cols):
        centers = []
        for i in range(n_rows):
            for j in range(n_cols):

                y = i * np.sqrt(3)/2
                if not (x < 0 or x > 1 or y < 0 or y > 1):
                    centers.append([x, y])
        return np.array(centers)

    # Ensure we have exactly 26 circles
    if len(initial_centers) < n:

        additional_centers = []

            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            additional_centers.append([x, y])

        return -np.sum(radii)  # Negative because we use minimization
