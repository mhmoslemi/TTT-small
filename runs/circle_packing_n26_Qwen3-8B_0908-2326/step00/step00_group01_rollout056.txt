import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial guess: grid-based placement
    n_circles = 26
    init_radius = 0.05
    centers = np.zeros((n_circles, 2))
    radii = np.full(n_circles, init_radius)

    cols = int(np.sqrt(n_circles))
    rows = (n_circles + cols - 1) // cols

        col = i % cols
        centers[i, 0] = col / (cols - 1)
        centers[i, 1] = row / (rows - 1)

    def objective(x):

        # [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
        x_flat = x.reshape(n_circles, 3)
        centers_flat = x_flat[:, :2]
        radii_flat = x_flat[:, 2]
        return -np.sum(radii_flat)  # Negative for minimization

    # Define bounds for each circle: (x, y, r)
    bounds = []
    for _ in range(n_circles):
        # x and y must be in [0, 1]
        # r must be >= 0
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((0.0, 1.0))  # r

    constraints = []

        return True

    constraints.append({'type': 'ineq', 'fun': center_in_square})

    for i in range(n_circles):

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
