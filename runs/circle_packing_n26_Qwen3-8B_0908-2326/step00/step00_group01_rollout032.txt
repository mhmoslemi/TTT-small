import numpy as np
import scipy.optimize as optimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial grid layout: 5x5 grid of points
    grid_size = 5
    grid_step = 1.0 / (grid_size - 1)
    centers = np.zeros((grid_size * grid_size, 2))
    for i in range(grid_size):
        for j in range(grid_size):
            centers[i * grid_size + j] = [i * grid_step, j * grid_step]
    
    # Initial radii (start with equal size circles)
    radii = np.full(grid_size * grid_size, 1.0 / (grid_size + 1))
    
    # Optimization parameters
    constraints = []
    bounds = []

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlap(i, j):
        def cons(radii):
            r_i = radii[i]
            r_j = radii[j]
            x_i, y_i = centers[i]
            x_j, y_j = centers[j]
            dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            return dist - r_i - r_j + 1e-12  # Non-overlapping constraint
        return cons

    def constraint_boundary(i):
        def cons(radii):
            r_i = radii[i]
            x_i, y_i = centers[i]
            return (x_i - r_i + 1e-12)  # Ensure x - r >= 0
        return cons

    def constraint_boundary_y(i):
        def cons(radii):
            r_i = radii[i]
            x_i, y_i = centers[i]
            return (y_i - r_i + 1e-12)  # Ensure y - r >= 0
        return cons

    # Set up all constraints
    for i in range(len(centers)):
        bounds.append((0.0, 1.0))  # Radii between 0 and 1
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_y(i)})
        for j in range(i + 1, len(centers)):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})

    # Run optimization
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
