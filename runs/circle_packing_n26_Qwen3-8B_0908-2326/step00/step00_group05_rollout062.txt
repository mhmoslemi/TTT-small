import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    grid = np.linspace(0, 1, int(np.sqrt(num_circles)) + 1)
    centers = []
    for i in range(len(grid) - 1):
        for j in range(len(grid) - 1):
            centers.append([grid[i], grid[j]])
    centers = np.array(centers, dtype=np.float64)
    
    # Initial radii based on grid spacing
    initial_radii = np.ones(num_circles) * 0.05
    
    # Objective function to maximize; we minimize the negative
    def objective(x):
        return -np.sum(x[1:])

    # Constraints
    def constraint_boundary(x):
        # x[0:26] are the centers, x[26:] are the radii
        centers = x[:26].reshape(-1, 2)
        radii = x[26:]
        return np.concatenate([centers[:, 0] - radii - 1e-12, 1 - centers[:, 0] - radii - 1e-12,
                               centers[:, 1] - radii - 1e-12, 1 - centers[:, 1] - radii - 1e-12])

    def constraint_overlap(x):
        centers = x[:26].reshape(-1, 2)
        radii = x[26:]
        dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :]) ** 2, axis=2))
        dists[np.arange(num_circles), np.arange(num_circles)] = np.inf
        return dists - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-12

        (0.0, 1.0) for _ in range(2 * num_circles)  # centers
    ] + [(0.0, 1.0) for _ in range(num_circles)]  # radii

        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Optimize
    result = optimize.minimize(

        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
