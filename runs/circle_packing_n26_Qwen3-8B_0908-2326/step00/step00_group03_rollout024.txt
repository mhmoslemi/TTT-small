import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initial guess: hexagonal packing
    # Compute the max radius for hexagonal packing
    # This is a rough estimate and serves as initial guess
    max_radius = 0.15
    centers = np.zeros((n, 2))
    radii = np.zeros(n) + max_radius

    # Generate a hexagonal grid pattern
    row_count = int(np.sqrt(n))
    col_count = int(np.ceil(n / row_count))
    
    for i in range(row_count):
        for j in range(col_count):
            if i % 2 == 0:
                x = j * (1 + 1.5 * max_radius)
            else:
                x = (j + 0.5) * (1 + 1.5 * max_radius)
            y = i * (1 + np.sqrt(3) * max_radius)
            idx = i * col_count + j
            if idx < n:
                centers[idx, 0] = x
                centers[idx, 1] = y
                radii[idx] = max_radius

    # Use optimization to refine the initial hexagonal packing
    # Define objective function to maximize sum of radii
    def objective(x):
        x_flat = x.reshape(-1, 2)
        return -np.sum(x_flat[:, 2])  # Minimize negative of sum

    # Define bounds for centers and radii
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x-coordinate
        bounds.append((0, 1))  # y-coordinate
        bounds.append((0, 1))  # radius

    # Define constraints for non-overlap
    def non_overlap_constraint(x):
        x_flat = x.reshape(n, 2)
        radii = x_flat[:, 2]
        centers = x_flat[:, :2]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-6:
                    return dist - (radii[i] + radii[j] - 1e-6)
        return 0

    # Define constraints for boundaries
    def boundary_constraint(x):
        x_flat = x.reshape(n, 2)
        radii = x_flat[:, 2]
        centers = x_flat[:, :2]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return 1e6  # Penalize large violation
        return 0

    # Optimization
    initial_guess = np.concatenate([centers.reshape(-1), radii])
    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': non_overlap_constraint},
            {'type': 'ineq', 'fun': boundary_constraint}
        ],
        options={'maxiter': 100, 'disp': False}
    )

    # Extract the best solution
    centers_opt = result.x[:2*n].reshape(n, 2)
    radii_opt = result.x[2*n:]
    sum_radii = np.sum(radii_opt)

    # Validate the packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Packing invalid: {msg}")

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
