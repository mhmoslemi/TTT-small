import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Parameters
    n = 26
    max_radius_baseline = 0.15  # Estimation based on hexagonal packing
    radius_guess = np.full(n, max_radius_baseline)

    # Generate hexagonal packing
    def generate_hexagonal_config(n):
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        centers = []
        y_offset = 0.5 * np.sqrt(3) * max_radius_baseline
        for i in range(rows):
            for j in range(cols):
                x = j * (2 * max_radius_baseline)
                y = i * y_offset
                if i % 2 == 1:
                    x += max_radius_baseline
                centers.append([x, y])
        # Trim to n circles
        centers = np.array(centers)[:n]
        # Normalize to unit square
        centers = centers / np.max(centers) * 1
        return centers

    centers = generate_hexagonal_config(n)

    # Objective function
    def objective(x):

        centers = x[:n * 2].reshape(n, 2)
        radii = x[n * 2:]
        return -np.sum(radii)  # Negative for maximization

    # Constraint to ensure circles stay within [0,1]^2
    def boundary_constraint(x):
        centers = x[:n * 2].reshape(n, 2)
        radii = x[n * 2:]
        return np.concatenate([
            centers[:, 0] - radii,
            1 - centers[:, 0] - radii,
            centers[:, 1] - radii,
            1 - centers[:, 1] - radii
        ])

    # Constraint to prevent overlaps
    def non_overlap_constraint(x):
        centers = x[:n * 2].reshape(n, 2)
        radii = x[n * 2:]
        dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :]) ** 2, axis=2))
        return np.concatenate([dists[i, j] - radii[i] - radii[j] for i in range(n) for j in range(i + 1, n)])

    # Initial guess: centers and radii
    x0 = np.concatenate([centers.flatten(), radius_guess])

    bounds = Bounds([0] * (n * 2 + n), [1] * (n * 2 + n))

        boundary_constraint,
        -np.inf,
        1e-12,

        objective,

        bounds=bounds,

    radii = result.x[n * 2:]

    return centers, radii, sum_radii
