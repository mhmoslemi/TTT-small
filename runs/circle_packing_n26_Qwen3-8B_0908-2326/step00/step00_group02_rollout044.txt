import scipy.optimize as optimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    rows = 6  # Number of rows

    cell_size = 1 / (cols + 1)
    axial_offset = np.sqrt(3) / 2 * cell_size

    centers = np.zeros((n, 2))
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if row % 2 == 0:
                x = col * cell_size + cell_size / 2
            else:
                x = col * cell_size + cell_size / 2 - cell_size / 2
            y = row * axial_offset
            centers[idx] = [x, y]
            idx += 1

    # Initial radii: each circle touches its neighbors
    radii = np.full(n, 0.1)

    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative for minimization

    # Constraints: ensure circles are in unit square and don't overlap
    def constraint_square(x):
        # x is a flattened array of (centers, radii)
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1
        return 0

                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]

                if dist < radii[i] + radii[j] - 1e-8:

    # Optimization bounds: centers must be within [0, 1], radii must be non-negative
    bounds = []
