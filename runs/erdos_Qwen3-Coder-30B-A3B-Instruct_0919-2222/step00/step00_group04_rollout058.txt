import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Discretization parameters
    N = 120
    dx = 2.0 / N
    # Compute shift matrices A_k for k=0 to N-1
    A = []
    for k in range(N):
        A_k = np.zeros((N, N))
        for i in range(N - k):
            A_k[i, i + k] = 1.0
        A.append(A_k)

    # Bisection on tau in [0.35, 0.5]
    low = 0.35
    high = 0.5
    tolerance = 1e-4
    
    best_h = None
    best_C5 = float('inf')
    
    # SDP solver parameters
    solver = 'ECOS'  # Fast and reliable for small to moderate problems
    
    # Some fallback behavior in case the SDP solver fails to converge
    use_lp_fallback = False
    if use_lp_fallback:
        # LP (power spectrum approximation)
        def solve_lp():
            J = 30  # Number of frequencies
            omega = np.linspace(0, np.pi, J)
            s = cp.Variable(J)
            gamma = cp.Variable()
            alpha = cp.Variable()
            
            # Constraints to enforce [0,1] bounds: linear in s and alpha
            h_func = lambda x: alpha + np.sum(s * np.cos(omega * x))
            
            # Sample points for feasibility checks
            x = np.linspace(0, 2, N)
            h_actual = alpha + s @ np.cos(omega @ x.reshape(-1, 1))  # N vector
            
            constraints = [
                s >= 0,
                cp.max(h_actual) <= 1,
                cp.min(h_actual) >= 0,
                cp.sum(h_actual) == N / 2,
                gamma <= 1  # conservative upper bound
            ]
            
            # Autocorrelation is sum of s_j^2 * cos(omega_j * k * dx) for shift k
            # For now let's bound gamma based on a small test
            # We'll just only compute and constrain a sample of shifts
            
            # Baseline approach: add some sampled shifts
            shift_indices = list(range(0, N, 2 if N > 30 else 1))  # fewer constraints for big N
            for k_idx in shift_indices:
                if k_idx == 0:
                    corr = alpha**2 + cp.sum(s**2)
                else:
                    corr = alpha**2 + cp.sum(s * np.cos(omega * k_idx * dx))
                constraints.append(corr >= gamma)
                
            objective = cp.Maximize(gamma)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=solver, verbose=False)
            
            if prob.status == 'optimal':
                return np.array([s.value, alpha.value, gamma.value])
            return None
            
        try:
            soln = solve_lp()
            if soln is not None:
                s_vals, alpha_vals, gamma_val = soln
                # Generate h
                x = np.linspace(0, 2, N)
                h_est = alpha_vals + np.dot(s_vals, np.cos(omega * x.reshape(-1, 1)))
                h_est = np.clip(h_est, 0, 1)
                # Normalize sum to exactly N/2
                h_est = h_est * (N / 2) / np.sum(h_est)
                C5_new = np.max(np.correlate(h_est, 1 - h_est, mode='full')) * dx
                if C5_new < best_C5:
                    best_h = h_est.copy()
                    best_C5 = C5_new
        except:
            pass
            
        if best_h is not None:
            return (best_h.tolist(), best_C5, N)
    
    # entries: N variables h[i], N(N+1)/2 variables in symmetric matrix X
    while high - low > tolerance and (time.time() - start_time) < budget_s - 5:
        tau = (low + high) / 2.
        gamma = 1 - tau

        # Create variables
        h = cp.Variable(N)
        X = cp.Variable((N, N), PSD=True)

        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == N / 2,
            ]
        
        for i in range(N):
            constraints.append(X[i, i] == h[i])

        # autocorrelations trade-off (greater than gamma)
        for k in range(N):
            constraints.append(cp.trace(A[k] @ X) >= gamma)

        # Optimize to a dummy constant scalar (feasibility only)
        prob = cp.Problem(cp.Minimize(0.), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            success = prob.status == 'optimal'
        except:
            success = False

        if success:
            high = tau  # feasible, tighten the upper bound on tau
        else:
            low = tau  # infeasible, raise lower bound on tau

        # Early exit if we have a very good candidate
        if (time.time() - start_time) > budget_s - 5:
            break

    # Extract solution from SDP
    final_tau = (low + high) / 2.
    final_gamma = 1 - final_tau

    h_new = cp.Variable(N)
    X_new = cp.Variable((N, N), PSD=True)

    constraints_new = [
        h_new >= 0,
        h_new <= 1,
        cp.sum(h_new) == N / 2,
    ]

    for i in range(N):
        constraints_new.append(X_new[i, i] == h_new[i])

    # autocorrelations trade-off (greater than final_gamma)
    for k in range(N):
        constraints_new.append(cp.trace(A[k] @ X_new) >= final_gamma)

    prob_new = cp.Problem(cp.Minimize(0.), constraints_new)
    try:
        prob_new.solve(solver=solver, verbose=False)
        if prob_new.status == 'optimal':
            h_vals = h_new.value
            X_vals = X_new.value
            h_final = h_vals
        else:
            h_final = np.ones(N) * 0.5  # Fallback
    except:
        h_final = np.ones(N) * 0.5  # Fallback

    # Reconstructable h via iterative rank-one recovery
    # Eigenvalue / SVD recomposition
    if h_final is not None and np.all(np.isfinite(h_final)):
        # Compute the best rank-1 approximation to matrix
        eigvals, eigvecs = np.linalg.eigh(X_vals)
        max_eval = np.max(eigvals)
        if max_eval > 0:
            v_max = eigvecs[:, np.argmax(eigvals)]
            h_guess = np.sqrt(max_eval) * v_max
        else:
            h_guess = h_final.copy()
    else:
        h_guess = np.ones(N) * 0.5

    # Ensure that values exceed machine precision of zero
    h_guess = np.clip(h_guess, 0, 1)

    # Project onto capped simplex that keeps sum = N/2
    def project_simplex(v, target_sum):
        v = np.array(v)
        n = len(v)
        # Sort in descending order
        sorted_v = np.sort(v)[::-1]
        # Find the threshold
        cumsum = np.cumsum(sorted_v)
        idxs = np.arange(1, n+1)
        theta = (cumsum - target_sum) / idxs
        theta_max = np.max(theta)
        if theta_max <= sorted_v[0]:
            # All values can be reduced below theta_max
            return np.clip(v - theta_max, 0, 1)
        else:
            # We need to find the exact point where the sum becomes target_sum
            best_theta = None
            for i in reversed(range(n)):
                t = (cumsum[i] - target_sum) / (i+1)
                if t >= 0 and (best_theta is None or t < best_theta):
                    best_theta = t
            if best_theta is None:
                best_theta = 0.
            return np.clip(v - best_theta, 0, 1)

    # If h_guess has correct sum, we're likely done
    h_projected = project_simplex(h_guess, N / 2.0)

    # Polish with small gradient steps (if time allows)
    def compute_overlap(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(conv)

    if (time.time() - start_time) < budget_s - 10:
        # Projected gradient descent
        gamma_step = 0.1
        max_it = 2000
        h_current = h_projected.copy()
        for i in range(max_it):
            # Compute max over shifts:
            conv = np.correlate(h_current, 1 - h_current, mode='full')
            idx_max_shift = np.argmax(conv)
            max_value = conv[idx_max_shift]
            # Sub-gradient:
            # (gradient of sum over k of [h_i * (1-h_{i+k})]) wrt h_i
            grad = np.zeros_like(h_current)
            # Clear the gradient except where it is involved in the max term
            # i such that h_i and h_{i+k} contribute to the maximized value
            # and with max_shift index (shifted shift)
            max_shift_len = idx_max_shift - N + 1  # for mode='full', import shift 0:

            for i in range(N):
                if 0 <= i + max_shift_len < N:
                    grad[i] += 1 - h_current[i + max_shift_len]
                if max_shift_len <= i < N:
                    grad[i] -= h_current[i - max_shift_len]

            # Normalize step size (slowly decaying)
            eta = gamma_step / np.sqrt(i + 1)
            h_trial = h_current - eta * grad

            # Clip and re-project
            h_trial = np.clip(h_trial, 0, 1)
            h_current = project_simplex(h_trial, N / 2.0)

            current_C5 = compute_overlap(h_current)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_current.copy()

    if best_h is None:
        best_h = h_projected.copy()
    best_C5 = compute_overlap(best_h)
    
    if (time.time() - start_time) > budget_s - 5:
        pass  # already past the time limit, time to return

    return (best_h.tolist(), best_C5, N)
