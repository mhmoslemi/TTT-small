import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate C5 for a given step function h."""
    dx = 2.0 / len(h)
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Index of the maximum correlation (for full mode, this is at the center)
    max_idx = len(corr) // 2
    c5 = corr[max_idx] * dx
    return c5

def build_mcormick_constraints(h, y, n, s):
    """Helper to build McCormick constraints for one shift s."""
    constraints = []
    for i in range(n):
        # i+s could be out of bounds; handle by checking and skipping
        j = i + s
        if 0 <= j < n:
            # McCormick envelopes
            constraints.extend([
                y[i, s + n - 1] <= h[i],  # index in y for shift s: s + n - 1
                y[i, s + n - 1] <= h[j],
                y[i, s + n - 1] >= h[i] + h[j] - 1,
                y[i, s + n - 1] >= 0
            ])
    return constraints

def solve_relaxed_lp(n, h0=None):
    """Solve LP relaxation with McCormick constraints."""
    h = cp.Variable(n, name='h')
    y = cp.Variable((n, 2*n-1), name='y')  # for shifts from -n+1 to n-1
    t = cp.Variable(name='t')

    # Constraints
    constraints = [
        0 <= h, h <= 1,
        cp.sum(h) == n/2,
        t >= 0
    ]

    # McCormick envelopes
    for s in range(-n+1, n):
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                idx_s = s + n - 1  # y[i, s] maps to y[i, s+n-1]
                constraints.extend([
                    y[i, idx_s] <= h[i],
                    y[i, idx_s] <= h[j],
                    y[i, idx_s] >= h[i] + h[j] - 1,
                    y[i, idx_s] >= 0
                ])

    # Constraint for each shift
    for s in range(-n+1, n):
        idx_s = s + n - 1
        total_y_s = cp.sum(y[:, idx_s])  # Sum over all i for this shift
        constraints.append(total_y_s >= n/2 - t)

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)

    if not prob.status.startswith('optimal'):
        # fallback to h0 if provided
        h_val = h0 if h0 is not None else np.ones(n) * 0.5
    else:
        h_val = h.value

    return h_val, prob.value

def get_binary_from_fractional(h_f, method='threshold'):
    """Convert fractional h to binary"""
    if method == 'threshold':
        # set top n/2 entries to 1, rest to 0
        idxs = np.argsort(h_f)[::-1]
        h_bin = np.zeros_like(h_f)
        h_bin[idxs[:int(len(h_f)/2)]] = 1
        return h_bin
    elif method == 'randomized':
        n = len(h_f)
        num_ones = n // 2
        # sample with replacement (since normally some entries are close to 0)
        # use weighted sampling to produce num_ones 1s
        probs = h_f / np.sum(h_f)
        selected = np.random.choice(n, size=num_ones, p=probs, replace=False)
        h_bin = np.zeros(n)
        h_bin[selected] = 1
        return h_bin

def polish_with_swaps(h, max_iter=300):
    """Apply targeted swaps to reduce the worst correlation."""
    h = h.copy()
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = corr[len(corr)//2]
    best_h = h.copy()
    best_corr = max_corr

    for _ in range(max_iter):
        max_idx = len(corr) // 2
        s_max = max_idx - (len(h) - 1)
        # Recompute correlations for current h
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        s_max = max_idx - (len(h) - 1)
        max_corr = corr[max_idx]
        if max_corr < best_corr:
            best_h = h.copy()
            best_corr = max_corr

        # Find improving swaps
        best_swap_reduction = 0
        best_swap_i = best_swap_j = None

        # Check all possible swaps
        for i in range(len(h)):
            if h[i] == 1:
                j_start = max(0, i - (len(h) - 1))
                j_end = min(len(h), i + len(h))
                for j in range(j_start, j_end):
                    # compute replacement: h_i, h_j and shifted partners
                    h_i = h[i]
                    h_j = h[j]
                    shift = j - i
                    shifted_i = (i + shift) % len(h)
                    shifted_j = (j + shift) % len(h)
                    if (h_i == 1 and h[shifted_i] == 0 and
                        h_j == 0 and h[shifted_j] == 1):
                        delta = 2.0 * dx  # Subtracting two contributions: one from h[i] and another from h[j]
                        if delta > best_swap_reduction:
                            best_swap_reduction = delta
                            best_swap_i = i
                            best_swap_j = j

        if best_swap_i is not None:
            h[best_swap_i], h[best_swap_j] = h[best_swap_j], h[best_swap_i]
        else:
            break  # no more improving swaps

    # Final check
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_max = corr[len(corr)//2]
    return best_h, final_max
