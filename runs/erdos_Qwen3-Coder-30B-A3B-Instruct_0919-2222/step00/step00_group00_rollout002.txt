import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value corresponds to the largest overlap
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Return initial construction as described in problem"""
    n = 45
    h = np.zeros(n)
    # This is a heuristic initial guess - a piecewise constant function
    # with value 1 on a central interval and 0 elsewhere. 
    # We will optimize from this guess
    h[15:30] = 1.0 
    h = h / np.sum(h) * (2.0 / n)  # normalize to integral 1
    return h

def objective(h, n_points):
    """Objective function to minimize - C5 bound"""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Start from the initial construction
    h_init = initial_h_values()
    n_points = len(h_init)
    
    # Optimization
    # We perform a few different strategies to avoid local optima
    
    # Strategy 1: Direct minimize with bounds and constraints
    def obj_h(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Bounds: each value in [0,1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) * dx = 1
    def constraint_sum(h):
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Initial constraint evaluation
    initial_violation = abs(constraint_sum(h_init))
    if initial_violation > 1e-4:
        # Adjust initial solution to be feasible
        h_init = h_init / np.sum(h_init) * (2.0 / n_points)
    
    # Use BFGS for optimization with bounds
    try:
        result = minimize(
            obj_h, 
            h_init.copy(), 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 1000, 'disp': False}
        )
        if result.success:
            final_h = result.x
            final_c5 = obj_h(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
    except Exception as e:
        pass
    
    # Strategy 2: Random restarts to escape local minima
    max_restarts = 5
    restarts = 0
    while restarts < max_restarts and (time.time() - start_time) < budget_s * 0.9:
        # Generate a random feasible starting point
        h_random = np.random.rand(n_points)
        # Normalize to make sure the integral constraint is satisfied
        h_random = h_random / np.sum(h_random) * (2.0 / n_points)
        # Clip to [0, 1]
        h_random = np.clip(h_random, 0, 1)
        
        try:
            result = minimize(
                obj_h, 
                h_random.copy(), 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 500, 'disp': False}
            )
            if result.success:
                final_h = result.x
                final_c5 = obj_h(final_h)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = final_h.copy()
        except Exception as e:
            pass
        restarts += 1

    # If no good solution was found, return the initial solution
    if best_h is None:
        best_h = h_init.copy()
        best_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), float(best_c5), int(n_points))
