import numpy as np
from scipy.optimize import differential_evolution
import time
from math import floor

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h_values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h)
    corr = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # The maximum value among all shifts
    max_val = np.max(corr) * dx
    return max_val

def initial_h_values():
    """Return initial h values - starting from a known construction"""
    n = 70
    h = np.zeros(n)
    # Simple construction: piecewise constant function
    # Split [0,2] into 5 equal intervals
    intervals = 5
    dx = 2.0 / intervals
    # Place 1s in the first 3 intervals, 0s in last 2
    # But adjust for integral constraint
    h[:60] = 1.0
    h[60:] = 0.0
    # Now make integral = 1
    integral = np.sum(h) * (2.0 / n)
    if integral > 0:
        h = h * (1.0 / integral)
    # Ensure each value in [0,1]
    h = np.clip(h, 0, 1)
    return h

def objective_function(h_values, n_points):
    """Compute the C5 constant for given h values"""
    dx = 2.0 / n_points
    # Compute correlation
    corr = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # Maximum value * dx
    return -np.max(corr) * dx  # negative because we're minimizing

def constraint_integral(h_values, n_points):
    """Constraint that integral equals 1"""
    dx = 2.0 / n_points
    integral = np.sum(h_values) * dx
    return integral - 1.0

def run(seed=42, budget_s=600, **kwargs):
    """Main function to run the optimization"""
    # Set seeds for reproducibility
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Parameters
    n_points = 70  # default to good balance of speed and accuracy
    dx = 2.0 / n_points
    
    # Use initial construction
    h_initial = initial_h_values()
    
    # Bounds limits for optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define the constraint for the integral 
    constraints = [{'type': 'eq', 'fun': lambda x: constraint_integral(x, n_points)}]
    
    # Perform the optimization using differential evolution
    result = differential_evolution(
        lambda x: objective_function(x, n_points),
        bounds,
        constraints=constraints,
        seed=seed,
        maxiter=int(100000),  # large but bounded iteration count
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        tol=1e-6,
        disp=False
    )
    
    # Return optimized solution
    h_opt = np.clip(result.x, 0, 1)
    c5_bound = -result.fun
    
    return (h_opt, c5_bound, n_points)

# Check if this run has not exceeded budget - we didn't use time tracking in the inner loops
def optimized_run(seed=42, budget_s=600, **kwargs):
    # Since the differential evolution is meant to be fast per iteration,
    # we expect this to finish within the budget of 600 seconds
    return run(seed, budget_s, **kwargs)

# Our main function
def run(seed=42, budget_s=600, **kwargs):
    return optimized_run(seed, budget_s, **kwargs)
