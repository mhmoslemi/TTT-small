def evaluate_erdos_solution(h_values, n_points):
    # Assuming the evaluation as per problem description
    # d = np.array(h_values)  # Already assumed to be in [0,1]
    # C5 = max(correlate(d, 1-d, mode='full')) * (2.0 / n_points)
    # But let's follow stricter procedure from problem statement
    d = np.array(h_values)
    dx = 2.0 / n_points
    # Compute all overlaps O_k = sum h_i * (1 - h_{i+k})
    # Using correlate with mode='full', we get convolution of h with (1-h)
    # The overlap integral is sum h_i * (1 - h_{i+k}) = (sum h_i) - sum h_i * h_{i+k}
    # Since sum h_i = n_points / 2, then O_k = n_points / 2 - sum h_i * h_{i+k}
    # We must compute sum h_i * h_{i+k} for each k.
    # Full correlation of h with (1-h) is computed:
    conv_full = correlate(d, 1 - d, mode='full')  # Length: 2*n - 1
    total_sum = n_points / 2
    # Max overlap O_k = sum h_i (1 - h_{i+k}) = total_sum - sum(h_i h_{i+k})
    overlaps = total_sum - conv_full  # Apply conversion (h_i h_{i+k} -> 1-h_{i+k} term)
    # Overlaps for full range. k goes from -(n-1) to (n-1)
    # overlap k corresponds to index k + (n-1) in the array
    # We want to consider only non-neg k for max k->infinity. but actually we compute all shifts
    # because for integer h, only effective shifts (k=-n+1,...,n-1) matter.
    # We extract:
    C5 = np.max(overlaps) * dx
    return C5

def find_worst_shift(h_values, n_points):
    d = np.array(h_values)
    total_sum = n_points / 2
    conv_full = correlate(d, 1 - d, mode='full')
    overlaps = total_sum - conv_full  # k goes from -(n-1) to (n-1)
    worst_k_idx = np.argmax(overlaps)
    worst_k = worst_k_idx - (n_points - 1)
    max_overlap = overlaps[worst_k_idx] * (2.0 / n_points)
    return worst_k, max_overlap
