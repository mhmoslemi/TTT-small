import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solution from global variable
    h0 = np.array(initial_h_values)
    n0 = len(h0)
    
    # Upsample to 200 points
    n = 200
    dx = 2.0 / n
    h = np.tile(h0, (n // n0))[:n]
    h = h * (n / n0) / np.sum(h) * (n / 2)
    
    # Optimization parameters
    max_iter = 100000
    alpha0 = 0.1
    tol = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    iter_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation (equivalent to f_k values)
        f = np.correlate(h, 1 - h, mode='full')
        
        # The correlation is of length 2*n-1, so we take indices from n-1 to 2*n-2
        # This corresponds to shifts from 0 to n-1 in positive direction
        # The negative shifts are also present but don't make a difference
        max_idx = np.argmax(f)
        max_f = f[max_idx]
        
        # Compute gradient with respect to h (subgradient at max)
        grad = np.zeros_like(h)
        for k in range(n):
            if k == max_idx - (n - 1):  # One of the worst shifts
                # gradient at h_i = 1 - h_{i+k} for i = 0 to n-1-k, 
                # minus h_{i-k} for i = k to n-1 (where the second factor is h_{i-k})
                # Both terms apply for i in range [k, n-1-k]
                if k == 0:
                    grad += 1 - h
                else:
                    # Partial gradient computation
                    if k < n:
                        grad[:n-k] += 1 - h[k:]
                    if k < n:
                        grad[k:] -= h[:n-k]
        
        # Step size
        step_size = alpha0 / np.sqrt(iteration + 1)
        
        # Update
        h_new = h - step_size * grad
        
        # Project onto box constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum = n/2 constraint
        h_new = h_new * (n / 2) / np.sum(h_new)
        
        h = h_new
        
        # Check for improvement
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early stopping if no improvement for a while
        if iteration - last_improvement > 5000 and iteration > 10000:
            break
            
        iter_count += 1
    
    return best_h.tolist(), best_c5, n
