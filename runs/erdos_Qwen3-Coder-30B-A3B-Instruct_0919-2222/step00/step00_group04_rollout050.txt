import numpy as np
import time

def evaluate_h(h, dx):
    """Compute the overlap integral for given step function h"""
    # Compute correlation of h and (1-h) using full mode
    # np.correlate gives correlation with full mode = 2*n-1 length
    # Result at index k corresponds to shift k - (n-1) 
    correlation = np.correlate(h, 1-h, mode='full')
    # Take max correlation value (excluding zero shift)
    max_correlation = np.max(correlation)
    # The actual integral is max_correlation * dx
    return max_correlation * dx

def project_to_simplex(h, target_sum):
    """Project h onto simplex with sum = target_sum and box constraints [0,1]"""
    n = len(h)
    # Sort, then compute threshold
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    indices = np.arange(1, n+1)
    
    # Find threshold such that sum of max(0, h[i] - tau) = target_sum
    # Find the largest index such that sorted_h[i] >= (target_sum - cumsum[i-1]) / i
    thresholds = (target_sum - cumsum) / indices
    valid_idx = np.where(sorted_h >= thresholds)[0]
    
    if len(valid_idx) == 0:
        tau = target_sum / n
        result = np.maximum(0, h - tau)
    else:
        tau = thresholds[valid_idx[-1]]
        result = np.maximum(0, h - tau)
        
        # Ensure we're correct by adjusting sum
        current_sum = np.sum(result)
        diff = target_sum - current_sum
        if abs(diff) > 1e-10:
            # Add diff to elements that are not at their upper bound
            indices_to_inc = np.where(result < 1)[0]
            if len(indices_to_inc) > 0:
                inc = diff / len(indices_to_inc)
                result[indices_to_inc] = np.minimum(1, result[indices_to_inc] + inc)
            
    return result

def compute_gradient(h, correlation, max_indices, dx):
    """Compute subgradient of max correlation w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    
    for k in max_indices:
        # For shift k, update gradients
        # i from 0 to n-1-k: h_i contributes to k
        if k <= n-1:
            # g[:n-k] += 1 - h[k:] 
            g[:n-k] += 1 - h[k:]
            # g[k:] -= h[:n-k] 
            g[k:] -= h[:n-k]
    
    return g * dx
