import numpy as np
import time
import random

def evaluate_h(h):
    """Evaluate the maximum overlap integral for given step function h."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode='full')
    max_O = O_counts.max()
    return max_O * dx

def max_overlap_counts(h):
    """Return the maximum integer overlap count (O_k) for given h."""
    O_counts = np.correlate(h, 1 - h, mode='full')
    return int(O_counts.max())

def random_balanced_binary(n, rng):
    """Generate a random balanced binary array of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    ones = rng.sample(range(n), n // 2)
    h[ones] = 1
    return h

def binary_anneal(n, time_limit, rng):
    """Perform simulated annealing on binary step function."""
    h = random_balanced_binary(n, rng)
    cur_val = max_overlap_counts(h)
    best_val = cur_val
    best_h = h.copy()
    start_time = time.time()
    
    # Temperature schedules
    max_iters = 50000
    for iter in range(max_iters):
        if time.time() - start_time > time_limit:
            break
            
        T = 0.5 * (1 - (time.time() - start_time) / time_limit) + 0.001
        
        # propose swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        # swap
        h[p], h[q] = 0, 1
        new_val = max_overlap_counts(h)
        delta = new_val - cur_val
        
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
                
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            h[p], h[q] = 1, 0
    
    return best_h, best_val

def project_onto_box_and_sum(v, target):
    """Project vector v onto [0,1] box and sum constraint."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def continuous_descent(h, time_limit, rng):
    """Refine binary solution with continuous optimization."""
    n = len(h)
    h = h.astype(float)
    target_sum = n / 2
    start_time = time.time()
    best_h = h.copy()
    best_obj = evaluate_h(h)
    
    # Simple gradient descent with projection
    max_iters = 10000
    step_size = 0.05
    
    for iter in range(max_iters):
        if time.time() - start_time > time_limit:
            break
            
        O_counts = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(O_counts)
        shift = max_idx - (n - 1)
        
        # compute gradient
        grad = np.zeros(n)
        if shift >= 0:
            left = np.zeros_like(h)
            right = np.zeros_like(h)
            left[shift:] = h[:-shift]
            right[:n-shift] = h[shift:]
            grad = 1 - left - right
        else:
            s = -shift
            left = np.zeros_like(h)
            right = np.zeros_like(h)
            left[:n-s] = h[s:]
            right[s:] = h[:-s]
            grad = 1 - left - right
            
        h_new = h - step_size * grad
        h_new = project_onto_box_and_sum(h_new, target_sum)
        
        obj = evaluate_h(h_new)
        if obj < best_obj:
            best_obj = obj
            best_h = h_new.copy()
            
        h = h_new
        
    return best_h, best_obj

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    rng = random.Random(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # candidate lengths for search
    n_values = [30, 40, 45, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    n_values = [n for n in n_values if n % 2 == 0]
    
    time_per_n = budget_s / len(n_values)
    
    # check initial_h_values if exists
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        c5 = evaluate_h(h0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()
            best_n = len(h0)
    
    for n in n_values:
        if time.time() - start_time > budget_s - 1:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1:
            break
        
        # run binary optimization
        h, val = binary_anneal(n, min(time_per_n, remaining_time), rng)
        c5 = val * 2 / n
        
        # refine with continuous optimization if improvement
        if c5 < best_c5:
            h_cont, c5_refined = continuous_descent(h, min(10, 0.8*remaining_time), rng)
            if c5_refined < c5:
                c5 = c5_refined
                h = h_cont
                
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
            
        if best_c5 < 0.3799:
            break
    
    if best_h is not None:
        return (best_h.tolist(), best_c5, best_n)
    else:
        # fallback to initial construction
        if 'initial_h_values' in globals():
            return (initial_h_values, evaluate_h(np.array(initial_h_values)), len(initial_h_values))
        # generate random balanced if no initial construction
        n = 45 if 45 in n_values else max(n_values)
        h = random_balanced_binary(n, rng)
        return (h.tolist(), evaluate_h(h), n)
