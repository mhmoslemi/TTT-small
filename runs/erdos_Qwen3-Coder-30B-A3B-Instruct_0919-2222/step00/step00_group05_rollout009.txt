import numpy as np
import time
from scipy.special import logsumexp
from collections import defaultdict

# -----------------------------
# Utility functions
# -----------------------------

def project_to_capped_simplex(v, M):
    """
    Project vector v onto the capped simplex:
    { h : 0 <= h_i <= 1, sum h_i = M }
    """
    n = len(v)
    # Clip to [0, 1]
    u = np.clip(v, 0, 1)
    total = np.sum(u)
    if abs(total - M) < 1e-10:
        return u
    if total > M:
        # Need to decrease values
        # Sort indices by v_i in descending order
        sorted_indices = np.argsort(v)[::-1]
        cumsum = 0.0
        theta = 0.0
        for i in range(n):
            idx = sorted_indices[i]
            if u[idx] > theta:
                cumsum += u[idx] - theta
                if cumsum >= total - M:
                    theta = u[idx] - (cumsum - (total - M))
                    break
        # Apply threshold theta
        result = np.clip(v - theta, 0, 1)
    else:
        # Need to increase values
        # Sort indices by v_i in ascending order
        sorted_indices = np.argsort(v)
        cumsum = 0.0
        theta = 0.0
        for i in range(n):
            idx = sorted_indices[i]
            if u[idx] < theta:
                cumsum += theta - u[idx]
                if cumsum >= M - total:
                    theta = u[idx] + (M - total - cumsum)
                    break
        # Apply threshold theta
        result = np.clip(v - theta, 0, 1)
    return result

def build_shift_maps(n):
    """Return list of (i, i+s) pairs for each shift s."""
    maps = {}
    for s in range(-(n-1), n):
        pairs = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                pairs.append((i, j))
        maps[s] = pairs
    return maps

def correlations(h, shift_maps):
    """Compute correlation vector C_s for all shifts s."""
    n = len(h)
    C = np.zeros(2*n - 1)
    for s, pairs in shift_maps.items():
        C[s + (n-1)] = sum(h[i] * (1 - h[j]) for i, j in pairs)
    return C

def smooth_obj_and_grad(h, alpha, shift_maps):
    """Return [Phi_alpha, grad Phi_alpha]"""
    C = correlations(h, shift_maps)
    max_C = np.max(C)
    exp_terms = np.exp(alpha * (C - max_C))
    Z = np.sum(exp_terms)
    w = exp_terms / Z
    # Gradient computation
    grad = np.zeros_like(h)
    for s, pairs in shift_maps.items():
        factor = w[s + (n-1)]
        for i, j in pairs:
            grad[i] += factor * (1 - h[j] - h[i] if i != j else 1 - h[j])  # Diagonal case included in the loop
    # Same formula simplifies with different indexing:
    # Replicate the optional part:
    # Compute ∂C_s/∂h_i = 1 - h_{i+s} - h_{i-s} if those exist, add to components 
    # Let's re-implement a cleaner version
    grad = np.zeros_like(h)
    for s in shift_maps:
        idx_pairs = shift_maps[s]
        w_s = w[s + (n-1)]
        for i, j in idx_pairs:
            grad[i] += w_s * (1 - h[j])
    return np.log(Z) / alpha + max_C / alpha, grad

def run_descent(h0, alpha_schedule, eta0, max_iter, shift_maps, time_limit):
    """Perform projected gradient descent."""
    h = h0.copy()
    elapsed = 0.0
    eta = eta0
    best_h = h.copy()
    best_C = np.max(correlations(h, shift_maps))
    for alpha in alpha_schedule:
        if elapsed > time_limit:
            break
        start_t = time.time()
        for k in range(max_iter):
            if time.time() - start_t > (time_limit - elapsed) * 0.9:
                break
            try:
                val, grad = smooth_obj_and_grad(h, alpha, shift_maps)
            except Exception:
                break
            v = h - eta * grad
            h = project_to_capped_simplex(v, len(h)/2)
            C = np.max(correlations(h, shift_maps))
            if C < best_C:
                best_C = C
                best_h = h.copy()
            eta *= 0.95  # Decay learning rate
        elapsed += time.time() - start_t
    return best_h, best_C

# Main optimization pipeline

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    n = 70
    dx = 2.0 / n
    M = n // 2
    max_iter_per_alpha = 50
    eta0 = 0.1
    alpha_schedule = [0.5, 1, 2, 4, 8, 16, 32]

    # Precompute shift maps once
    shift_maps = build_shift_maps(n)
    
    # Initialize list of seeds
    initials = []
    
    # 1. Use provided initial construction
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initials.append(np.array(initial_h_values))
    else:
        # Fallback to random binary
        h = np.zeros(n)
        h[:M] = 1
        np.random.shuffle(h)
        initials.append(h)
    
    # 2. Add random binary seeds
    for _ in range(4):
        h = np.zeros(n)
        ones_idx = np.random.choice(n, size=M, replace=False)
        h[ones_idx] = 1
        initials.append(h)
    
    # 3. Add random fractional seeds (Dirichlet + projection)
    for _ in range(5):
        v = np.random.dirichlet([1.0]*n)
        h = project_to_capped_simplex(v, M)
        initials.append(h)
    
    #------------------------------
    # Run optimizations
    #------------------------------
    start_time = time.time()
    best_h = None
    best_C = float('inf')
    
    for i, h0 in enumerate(initials):
        if time.time() - start_time > budget_s * 0.9:
            break
        try:
            h_sol, C_sol = run_descent(h0, alpha_schedule, eta0, max_iter_per_alpha, shift_maps, budget_s * 0.9 - (time.time() - start_time))
            if C_sol < best_C:
                best_C = C_sol
                best_h = h_sol.copy()
        except Exception:
            continue  # skip if error
    
    # Post-processing: round best continuous vector to binary
    if best_h is not None:
        # Deterministic rounding: sort and select top M
        sort_idx = np.argsort(best_h)[::-1]
        h_rounded = np.zeros_like(best_h)
        h_rounded[sort_idx[:int(M)]] = 1.0
        # Local swap refinement:
        C_rounded = np.max(correlations(h_rounded, shift_maps))
        best_h = h_rounded
        best_C = C_rounded
        # Minor improvement through swaps on the binary version
        current_C = C_rounded
        swapped = True
        swap_count = 0
        max_swaps = 2000
        while swapped and swap_count < max_swaps:
            swapped = False
            # Compute all correlations
            C_all = correlations(best_h, shift_maps)
            s_best = np.argmax(C_all)
            C_best = C_all[s_best]
            if C_best <= current_C:
                break
            # Find indices that, if swapped, would improve the current worst correlation
            worst_shift = s_best - (n - 1)
            pairs = shift_maps[worst_shift]
            # Check for possible valid swaps
            for (i, j) in pairs:
                if best_h[i] == 1 and best_h[j] == 0:
                    # Fallback for other indices that might be better
                    for k in range(n):
                        # look for 'good' swap points
                        if best_h[k] == 0 and (k + worst_shift) < n and (k + worst_shift) >= 0:
                            if best_h[k + worst_shift] == 1:
                                # Test if swapping i and k improves s_best cumulative
                                # More detailed evaluation — just improve it by direct measurement
                                h_test = best_h.copy()
                                h_test[i], h_test[k] = h_test[k], h_test[i]
                                C_test = np.max(correlations(h_test, shift_maps))
                                if C_test < current_C:
                                    best_h = h_test
                                    current_C = C_test
                                    swapped = True
                                    swap_count += 1
                                    break
                    if swapped:
                        break
    else:
        # Fallback if no optimization worked
        h_fallback = np.zeros(n)
        h_fallback[:M] = 1
        np.random.shuffle(h_fallback)  # random initialization
        best_h = h_fallback
        best_C = np.max(correlations(best_h, shift_maps))
    
    # Final evaluation using np.correlate
    correlation_full = np.correlate(best_h, 1 - best_h, mode='full')
    # Use dx scaling correctly as integral
    C5_eval = np.max(correlation_full) * dx
    
    return (list(best_h), float(C5_eval), int(n))
