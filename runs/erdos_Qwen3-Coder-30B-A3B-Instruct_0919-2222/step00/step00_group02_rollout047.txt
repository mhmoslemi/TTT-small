import numpy as np
import time
import cvxpy as cp
from scipy import fft
from scipy.optimize import minimize_scalar

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given h array."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    corr = np.correlate(h, 1-h, mode='full')
    max_overlap = np.max(corr)
    c5 = max_overlap * dx
    return c5

def initial_h_values():
    """Return the initial construction."""
    # Construct h as the initial solution (n=45)
    n = 45
    h = np.zeros(n)
    # Fill with the standard construction: a balanced step function
    # Start with a simple binary approach but scaled for average=1/2
    # For n=45, place 22.5 ones uniformly, which means 23 ones and 22 zeros
    h[:23] = 1.0
    return h

def solve_lp_relaxation(h0, n, epsilon=1e-6, max_iter=100):
    """Solve the LP relaxation with column generation."""
    # Initialize with a small set of shifts
    K = [-2, -1, 0, 1, 2]
    K = [k for k in K if -n+1 <= k <= n-1]  # Ensure within bounds
    
    # Variable to optimize
    h = cp.Variable(n, bounds=(0, 1))
    t = cp.Variable()
    
    # Optimise over the active set of constraints
    constraints = []
    for k in K:
        # Re-indexing for convolution loop
        shift = np.zeros(n)
        if k >= 0:
            shift[k:] = h[:-k]
        else:
            shift[:k] = h[-k:]
            
        # Linear approximation of h * (1 - h_shifted)
        w_sum = cp.sum(cp.multiply(h, 1 - shift))
        constraints.append(cp.sum(h) == n / 2)
        constraints.append(w_sum <= t)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    old_bound = float('inf')
    for i in range(max_iter):
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000)
        except:
            # Fall back to GLPK if ECOS fails
            try:
                prob.solve(solver=cp.GLPK)
            except:
                return h0, float('inf')
                
        if prob.status != cp.OPTIMAL:
            return h0, float('inf')
            
        h_val = h.value
        t_val = t.value
        # Compute all correlations to find the worst one
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        max_corr = np.max(corr)
        
        if abs(max_corr - t_val) < epsilon and abs(old_bound - t_val) < epsilon:
            break
        old_bound = t_val
        
        # Find the index of the maximum correlation
        max_k = np.argmax(corr) - (n - 1)
        if max_k not in K:
            K.append(max_k)
        if len(K) > 30:
            break  # Safety check
            
    return h_val, t_val

def stochastic_rounding(h_float, n):
    """Round a fractional solution to a binary solution via swaps."""
    h = np.copy(h_float)
    h = np.clip(h, 0, 1)
    h_int = np.round(h).astype(int)
    target_sum = int(n / 2)
    current_sum = np.sum(h_int)
    diff = int(target_sum - current_sum)
    
    if diff == 0:
        return h_int
    
    # Swap rounds
    for _ in range(1000):
        # Look for two non-integral positions
        non_int = np.where((h > 0) & (h < 1))[0]
        if len(non_int) < 2:
            break
        
        pairs = list(zip(non_int[:-1], non_int[1:]))
        if not pairs:
            break
            
        i, j = pairs[0]
        delta = min(h[i], 1 - h[j])
        if delta < 1e-8:
            break
            
        # Choose randomness
        rand = np.random.random()
        if rand > 0.5:
            h[i] -= delta
            h[j] += delta
        else:
            h[i] += delta
            h[j] -= delta
            
    # Final rounding
    h = np.clip(h, 0, 1)
    h_int = np.round(h).astype(int)
    
    # Adjust if necessary to maintain sum
    actual_sum = np.sum(h_int)
    if actual_sum != target_sum:
        diff = int(target_sum - actual_sum)
        # Find indices to modify
        to_change = np.where(h_int == 0)[0]
        change_idx = to_change[:diff] if diff > 0 else np.where(h_int == 1)[0][:abs(diff)]
        for idx in change_idx:
            h_int[idx] = 0 if h_int[idx] == 1 else 1
    
    return h_int

def sturmian_construction(n, alpha_opt=None, tau_opt=None):
    """Produce a Sturmian sequence."""
    if alpha_opt is None:
        alpha_opt = 0.6180339887498949  # Golden ratio
    if tau_opt is None:
        tau_opt = 0.5
    
    h = np.zeros(n, dtype=int)
    for i in range(n):
        val = (i * alpha_opt + tau_opt) % 1
        if val < 0.5:
            h[i] = 1
    return h

def sturmian_search(n, max_iter=500):
    """Search for parameter sets that minimize the bound."""
    best_c5 = float('inf')
    best_h = None
    
    # Start with coarse search
    n_alpha = 200
    n_tau = 100
    
    alphas = np.linspace(0.2, 0.8, n_alpha)
    taus = np.linspace(0.0, 1.0, n_tau, endpoint=False)
    
    min_c5 = float('inf')
    best_alpha = 0.5
    best_tau = 0.5
    
    for alpha in alphas:
        for tau in taus:
            h = np.zeros(n, dtype=int)
            for i in range(n):
                val = (i * alpha + tau) % 1
                if val < 0.5:
                    h[i] = 1
            
            c5 = evaluate_erdos_solution(h)
            if c5 < min_c5:
                min_c5 = c5
                best_alpha = alpha
                best_tau = tau
    
    best_h = sturmian_construction(n, best_alpha, best_tau)
    
    # Use local optimization to further refine
    def sturmian_obj(params):
        alpha, tau = params
        h = np.zeros(n, dtype=int)
        for i in range(n):
            val = (i * alpha + tau) % 1
            if val < 0.5:
                h[i] = 1
        c5 = evaluate_erdos_solution(h)
        return c5
    
    # Golden section optimization
    # Temporarily adjust range for alpha
    try:
        res = minimize_scalar(lambda a: sturmian_obj([a, best_tau]), bounds=(0.2, 0.8), method='bounded', options={'maxiter': 100})
        if res.success:
            best_alpha = res.x
        res2 = minimize_scalar(lambda t: sturmian_obj([best_alpha, t]), bounds=(0.0, 1.0), method='bounded', options={'maxiter': 100})
        if res2.success:
            best_tau = res2.x
    except:
        pass  # Fallback to coarse
    
    # Create best vector
    final_h = np.zeros(n, dtype=int)
    for i in range(n):
        val = (i * best_alpha + best_tau) % 1
        if val < 0.5:
            final_h[i] = 1
    return final_h

def optimize_h(h_start, n_points, max_iter=500):
    """Perform local optimization on a given h vector."""
    best_eval = evaluate_erdos_solution(h_start)
    best_h = h_start.copy()
    
    # Perform local swaps
    for _ in range(max_iter):
        if best_eval <= 0.3807:
            break
            
        # Find worst overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_k = np.argmax(corr) - (n_points - 1)
        max_overlap = np.max(corr)
        
        # Try a swap
        best_swap_change = 0
        best_i, best_j = -1, -1
        
        # Iterate through candidate pairs
        for i in range(n_points):
            for j in range(n_points):
                if i == j:
                    continue
                    
                # Try swaps
                if best_h[i] == 0 or best_h[j] == 1:
                    continue
                    
                if i + worst_k < 0 or i + worst_k >= n_points:
                    continue
                    
                h_test = best_h.copy()
                h_test[i] = 0
                h_test[j] = 1
                
                test_eval = evaluate_erdos_solution(h_test)
                if test_eval < best_eval:
                    change = best_eval - test_eval
                    if change > best_swap_change:
                        best_swap_change = change
                        best_i, best_j = i, j
                        break
        
        if best_i != -1:
            best_h[best_i] = 0
            best_h[best_j] = 1
            best_eval = evaluate_erdos_solution(best_h)
        else:
            break
            
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points_best = 0
    h_best = None
    c5_best = float('inf')
    
    # Try different n values
    test_ns = [30, 45, 60, 90, 120, 150, 200]
    
    # Assign time quotas
    total_time = 0
    for n in test_ns:
        remaining = budget_s - total_time
        if remaining < 5:
            break
        # Quota proportional to n, but capped to avoid over-allocation
        allocated_time = min(remaining * n / sum(test_ns), remaining * 0.2)
        if allocated_time < 10:  # Can't run efficiently
            continue
            
        current_time = time.time()
        
        # Try the LP relaxation with column generation
        h_start = initial_h_values() if n == 45 else np.ones(n) * 0.5
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n / 2) / np.sum(h_start)
        if not (h_start.shape[0] == n):
            h_start = h_start[:n]
            h_start[-1] = 1 if np.sum(h_start) < n / 2 else 0 if np.sum(h_start) > n / 2 else h_start[-1]
        h, t = solve_lp_relaxation(h_start, n, epsilon=1e-5, max_iter=15)
        
        # Convert to binary
        h_bin = stochastic_rounding(h, n)
        c5 = evaluate_erdos_solution(h_bin)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_bin.copy()
            n_points_best = n
            
        # Optional Sturmian construction for large n if we have time
        if n >= 120 and (time.time() - current_time) < (allocated_time - 1):
            h_sturm = sturmian_search(n)
            c5_sturm = evaluate_erdos_solution(h_sturm)
            if c5_sturm < c5_best:
                c5_best = c5_sturm
                h_best = h_sturm.copy()
                n_points_best = n
                
        # Stop once we hit the target
        if c5_best <= 0.3807:
            break
            
        total_time = time.time() - start_time
        if total_time + 5 >= budget_s:
            break
    
    # Final polish if close enough to target
    if h_best is not None:
        if c5_best > 0.3807:
            h_best = optimize_h(h_best, n_points_best, 500)
            c5_best = evaluate_erdos_solution(h_best)
    
    # Return the best result
    if h_best is None:
        h_best = initial_h_values()
        c5_best = evaluate_erdos_solution(h_best)
        n_points_best = len(h_best)
    
    return (h_best, c5_best, n_points_best)
