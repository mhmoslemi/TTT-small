def project_to_simplex(v, target_sum):
    """
    Project vector v to the simplex with sum = target_sum, with bounds [0,1]
    """
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(u)
    # Find the threshold
    rho = np.where(u > (target_sum - cumsum) / np.arange(1, n+1))[0]
    if len(rho) == 0:
        # All elements are too large, set to target_sum/n
        tau = target_sum / n
        return np.clip(v - tau, 0, 1)
    else:
        # Compute the index
        rho = rho[-1] + 1
        tau = (target_sum - cumsum[rho-1]) / rho
        # Apply threshold
        result = np.clip(v - tau, 0, 1)
        return result

def compute_C5_bound(h, dx):
    """Compute the C5 bound using correlation"""
    # Compute correlation: h * (1 - h) with full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum value times dx
    return np.max(corr) * dx

def compute_gradient(h, dx):
    """Compute subgradient of max correlation w.r.t. h"""
    # Compute correlation for all shifts (with full mode)
    corr = np.correlate(h, 1 - h, mode='full')
    # Find indices where max occurs
    max_val = np.max(corr)
    max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    # Compute gradient vector
    n = len(h)
    gradient = np.zeros(n)
    
    # For each maximizing shift, add contributions to gradient
    for k_idx in max_indices:
        # Convert index back to shift value (range -n+1 to n-1)
        shift = k_idx - (n - 1)
        
        # For shift k, gradient contributions:
        # Element i contributes [1 - h[i+k]] if valid
        # Element i contributes [-h[i-k]] if valid
        if shift >= 0:
            # Shift is forward (k > 0)
            # Element i in range [0, n-1-shift] contributes 1-h[i+k]
            # Element j in range [shift, n-1] contributes -h[j-shift] 
            gradient[:n-shift] += 1 - h[shift:]
            gradient[shift:] -= h[:n-shift]
        else:
            # Shift is backward
            # Element i in range [0, n+shift-1] contributes 1-h[i-shift] 
            # Element j in range [-shift, n-1] contributes -h[j+shift]
            shift_abs = -shift
            gradient[:n-shift_abs] += 1 - h[shift_abs:]
            gradient[shift_abs:] -= h[:n-shift_abs]
    
    # Normalize by dx since each term in corr has dx factor
    gradient *= dx
    return gradient
