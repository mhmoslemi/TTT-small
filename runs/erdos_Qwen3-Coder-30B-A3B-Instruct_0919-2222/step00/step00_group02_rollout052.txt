def evaluate_overlap(h):
    """Evaluate maximum overlap for h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = corr.max()
    return max_overlap * dx

def compute_max_overlap(h):
    """Compute max overlap using numpy correlation (vectorized)"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max()

def get_worst_shift(a):
    """Get shift index of maximum correlation"""
    n = len(a)
    corr = np.correlate(a, 1 - a, mode="full")
    return corr.argmax() - (n - 1), corr.max()

def sample_01(h_f):
    """Convert fractional vector to binary (0/1) with same mean"""
    n = len(h_f)
    s = np.sum(h_f)
    h_b = np.zeros(n, dtype=int)
    # Create actual 0/1 vector with target mean s
    h_b[:int(s)] = 1
    # Shuffle for randomization to avoid biased sampling
    np.random.shuffle(h_b)
    # Adjust if needed due to integer cutting
    total = np.sum(h_b)
    if total > s:
        # Change some 1s to 0s
        indices1 = np.where(h_b == 1)[0]
        to_change = int(total - s)
        change_idx = np.random.choice(indices1, to_change, replace=False)
        h_b[change_idx] = 0
    elif total < s:
        # Change some 0s to 1s
        indices0 = np.where(h_b == 0)[0]
        to_change = int(s - total)
        change_idx = np.random.choice(indices0, to_change, replace=False)
        h_b[change_idx] = 1
    return h_b

def random_balanced_vector(n):
    """Generate random balanced binary vector of 0s and 1s"""
    h_b = np.zeros(n, dtype=int)
    half = n // 2
    h_b[:half] = 1
    np.random.shuffle(h_b)
    return h_b

def random_balanced_swap(a):
    """Perform a random balanced swap: (1,0) -> (0,1)"""
    n = len(a)
    ones = np.where(a == 1)[0]
    zeros = np.where(a == 0)[0]
    if len(ones) < 1 or len(zeros) < 1:
        return a.copy()
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    a = a.copy()
    a[i] = 0
    a[j] = 1
    return a

def improve_binary_descent(a, max_iter=200, patience=50):
    """Worst-shift-based improvement of binary vector"""
    n = len(a)
    corr = np.correlate(a, 1 - a, mode="full")
    for i in range(max_iter):
        s = corr.argmax() - (n - 1)
        max_cnt = corr.max()
        # Get mismatch sets
        M1 = []
        M0 = []
        if s >= 0:
            for i in range(n - s):
                if a[i] == 1 and a[i + s] == 0:
                    M1.append(i)
                if a[i] == 0 and a[i + s] == 1:
                    M0.append(i)
        else:
            s_neg = -s
            for i in range(n - s_neg):
                if a[i] == 1 and a[i - s_neg] == 0:
                    M1.append(i)
                if a[i] == 0 and a[i - s_neg] == 1:
                    M0.append(i)
        if len(M1) > 0 and len(M0) > 0:
            idx1 = np.random.choice(M1)
            idx0 = np.random.choice(M0)
            # Update corr by flipping
            a_new = a.copy()
            a_new[idx1] = 0
            a_new[idx0] = 1
            corr_new = np.correlate(a_new, 1 - a_new, mode="full")
            # Accept if improvement or weak
            if corr_new.max() <= max_cnt:
                a = a_new
                corr = corr_new
        else:
            # Random balanced swap if no mismatches
            a = random_balanced_swap(a)
            corr = np.correlate(a, 1 - a, mode="full")
    return a, corr.max()

def simulate_annealing(a, max_iter=5000):
    """Local simulated annealing with exponential cooling"""
    n = len(a)
    max_cnt = compute_max_overlap(a)
    temp_start = 0.5
    temp_end = 1e-4
    temp = temp_start
    alpha = np.log(temp_end / temp_start) / max_iter
    best_a = a.copy()
    best_cnt = max_cnt
    for i in range(max_iter):
        new_a = random_balanced_swap(a)
        new_cnt = compute_max_overlap(new_a)
        delta = new_cnt - max_cnt
        if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
            a = new_a
            max_cnt = new_cnt
        if max_cnt < best_cnt:
            best_a = a.copy()
            best_cnt = max_cnt
        temp = temp_start * np.exp(alpha * i)
    return best_a, best_cnt

def continuous_refinement(h0, max_time_seconds):
    """Apply projected softmax gradient descent to refine"""
    start_time = time.time()
    n = len(h0)
    dx = 2.0 / n
    h = h0.astype(float)
    max_alpha = 80
    steps_per_alpha = 200
    lr0 = 0.2

    # Function to compute overlaps and costs
    def get_overlaps(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return corr

    # Function to compute gradient of softmax approximation
    def compute_grad(h_vec, alpha=1.0):
        corr = get_overlaps(h_vec)
        max_o = corr.max()
        exp_corr = np.exp(alpha * (corr - max_o))
        exp_sum = np.sum(exp_corr)
        weights = exp_corr / exp_sum
        grad = np.zeros(n)
        # Compute contribution from each shift
        for shift_idx, weight in enumerate(weights):
            s = shift_idx - (n - 1)
            if s >= 0:
                # For s >= 0, subscripts map to [0:n] with upward shift: i -> i+s
                grad[:n - s] += weight * (1 - h_vec[s:(n-s)])
                grad[s:(n-s)] -= weight * h_vec[:n - s]
            else:
                # s < 0, so i -> i+s means we shift backwards
                grad[-s:] += weight * (1 - h_vec[:-s])
                grad[:-s] -= weight * h_vec[s:]
        return grad

    def project_to_box_and_mean(h_vec):
        """Project to box [0,1] and mean constraint ∫h = 1"""
        h_vec = np.clip(h_vec, 0, 1)
        # Waterfill to project onto ∫h = 1
        # Directly impose sum_h = 1
        h_vec = h_vec * (n_points / np.sum(h_vec))
        return h_vec

    h = project_to_box_and_mean(h)

    for alpha in [1, 5, 10, 20, 40, 80]:
        if time.time() - start_time > max_time_seconds - 10:
            break
        for i in range(steps_per_alpha):
            if time.time() - start_time > max_time_seconds - 10:
                break
            grad = compute_grad(h, alpha)
            lr = lr0 / np.sqrt(i + 1)
            h = h - lr * grad
            h = project_to_box_and_mean(h)
        c = compute_max_overlap(h) * dx
        if i == 0 and alpha == 1:
            initial_h = h.copy()
    c = compute_max_overlap(h) * dx
    return h, c
