import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    # Compute correlation of h with (1-h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Only keep the relevant part (max k values)
    # The full correlation has length 2*n - 1
    # We want the max for k=1 to n-1 (corresponding to indices n-1 to 2*n-2)
    max_corr = np.max(correlation[n-1:])
    return max_corr * dx

def initial_h_values(n_points):
    # Start with a simple construction
    # A symmetric step function
    h = np.ones(n_points)
    # Divide into 3 steps: 0, 1/2, 1
    segment_length = n_points // 3
    h[:segment_length] = 0.0
    h[segment_length:2*segment_length] = 1.0
    h[2*segment_length:] = 0.0
    
    # Adjust to satisfy integral constraint
    integral = np.sum(h) * 2.0 / n_points
    if integral > 0:
        h = h / integral * 0.5
    else:
        h = np.ones_like(h) * 0.5 / n_points
    
    return np.clip(h, 0, 1)

def objective(h, n_points, dx):
    # Normalize h to satisfy constraint
    integral = np.sum(h) * dx
    if integral > 0:
        h = h / integral * 0.5  # scale to make integral = 1
    h = np.clip(h, 0, 1)
    
    # Evaluate C5
    c5 = evaluate_erdos_solution(h, dx)
    return c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70  # Starting smaller than 1k for faster optimization
    dx = 2.0 / n_points
    
    # Start with the initial construction
    h_current = initial_h_values(n_points)
    
    # We'll optimize with different methods and try to improve
    def objective_wrapper(h):
        return objective(h, n_points, dx)
    
    # Use scipy.optimize.minimize to refine our solution
    # We first try a local optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Store best result found
    best_c5 = float('inf')
    best_h = None
    
    # Try with differential evolution (global optimizer)
    try:
        de_result = differential_evolution(
            objective_wrapper,
            bounds,
            maxiter=50,
            popsize=10,
            seed=seed,
            updating='deferred',
            disp=False
        )
        if not de_result.success:
            raise Exception("Differential evolution failed")
        
        h_de = np.clip(de_result.x, 0, 1)
        integral_de = np.sum(h_de) * dx
        h_de = h_de / integral_de * 0.5
        c5_de = evaluate_erdos_solution(h_de, dx)
        
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = np.copy(h_de)
            
    except Exception as e:
        pass  # Continue with other methods
    
    if best_h is None:
        # Fallback to original h
        h_norm = np.copy(h_current)
        integral = np.sum(h_norm) * dx
        h_norm = h_norm / integral * 0.5
        best_c5 = evaluate_erdos_solution(h_norm, dx)
        best_h = h_norm
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, best_c5, n_points)
    
    # Perform a few steps of a more fine-grained optimization
    try:
        # Use scipy.optimize.minimize
        options = {'maxiter': 200, 'disp': False}
        result = minimize(
            objective_wrapper,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options=options
        )
        if result.success:
            h_final = np.clip(result.x, 0, 1)
            integral_final = np.sum(h_final) * dx
            h_final = h_final / integral_final * 0.5
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
    except Exception as e:
        pass  # Continue with current best
    
    return (best_h, best_c5, n_points)
