import numpy as np
import cvxpy as cp
import time
from scipy import correlate

def evaluate_h(h_values, n_points):
    """Evaluate C5 for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # The valid range of shifts is from -(n-1) to (n-1)
    # For shift s, corr[n-1+s] gives the value
    max_corr = np.max(corr)
    return max_corr * dx

def build_mcormick_constraints(h, y, n, s):
    """Build McCormick constraints for variable y[i,s] = h[i]*h[i+s]"""
    constraints = []
    for i in range(n):
        # Determine the valid range for i+s
        if 0 <= i + s < n:
            # y[i,s] <= h[i]
            constraints.append(y[i, s] <= h[i])
            # y[i,s] <= h[i+s]
            constraints.append(y[i, s] <= h[i + s])
            # y[i,s] >= h[i] + h[i+s] - 1
            constraints.append(y[i, s] >= h[i] + h[i + s] - 1)
            # y[i,s] >= 0
            constraints.append(y[i, s] >= 0)
    return constraints
