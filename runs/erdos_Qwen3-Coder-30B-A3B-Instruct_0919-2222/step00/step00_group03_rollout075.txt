import numpy as np
from scipy.optimize import minimize
from scipy import signal
import time
import math

def evaluate_c5(h):
    """Evaluate C5 for a given h using full convolution."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = signal.correlate(h, 1 - h, mode="full")
    # The maximum value of the correlation scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Return the initial construction for h."""
    # A simple smoothed step function with n=70 points
    n = 70
    h = np.zeros(n)
    # Define step points
    x = np.linspace(0, 2, n)
    # Use a piecewise linear approximation to a step function
    for i in range(n):
        if x[i] <= 0.5:
            h[i] = 0
        elif x[i] <= 1.0:
            h[i] = 2 * (x[i] - 0.5)  # Linear rise
        elif x[i] <= 1.5:
            h[i] = 1  # Plateau
        else:  # x[i] > 1.5
            h[i] = 2 * (2.0 - x[i])  # Linear fall
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Define the optimization function
    def objective(h):
        return evaluate_c5(h)
    
    # Define bounds for h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(70)]
    
    # Define constraint that integral equals 1
    def constraint_eq(h):
        return np.sum(h) * (2.0 / len(h)) - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Get initial guess
    h0 = initial_h_values()
    
    # Use scipy optimization with L-BFGS-B method
    try:
        result = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 10000})
        h_opt = result.x
        c5_bound = objective(h_opt)
        n_points = len(h_opt)
    except Exception as e:
        # Fallback to initial guess
        h_opt = h0
        c5_bound = evaluate_c5(h0)
        n_points = len(h0)
    
    # Ensure we return the best result within budget
    elapsed = time.time() - start_time
    if elapsed < budget_s - 1:
        # Try different initialization strategies if time allows
        # Try random perturbations of the initial solution
        for _ in range(5):
            if elapsed >= budget_s - 1:
                break
            h_perturbed = h_opt + np.random.normal(0, 0.05, len(h_opt))
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Enforce constraint again
            scale_factor = 1.0 / (np.sum(h_perturbed) * (2.0 / len(h_perturbed)))
            h_perturbed = h_perturbed * scale_factor
            c5_try = evaluate_c5(h_perturbed)
            if c5_try < c5_bound:
                c5_bound = c5_try
                h_opt = h_perturbed
    else:
        pass

    return (h_opt, c5_bound, n_points)
