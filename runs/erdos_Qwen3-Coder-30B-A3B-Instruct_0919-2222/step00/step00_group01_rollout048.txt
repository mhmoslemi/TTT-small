import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h):
    """Evaluate C5 bound for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum of the overlap integral
    max_overlap = np.max(corr)
    return max_overlap * dx
