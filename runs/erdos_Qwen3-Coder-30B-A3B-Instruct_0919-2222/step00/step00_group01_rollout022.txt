import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap integral is the maximum of the correlation
    # scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    start_time = time.time()
    end_time = start_time + budget_s * 0.99  # Leave a margin
    
    # Parameters for the optimization
    n_points = 501  # odd number for symmetric shape
    dx = 2.0 / n_points
    
    # Generate initial shapes and find the best
    best_h = None
    best_c5 = float('inf')
    
    # Triangle shape
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2  # midpoint sampling
    triangle_h = np.maximum(0, 1 - np.abs(x - 1))
    triangle_h = triangle_h * (n_points / 2) / np.sum(triangle_h)  # normalize to sum = n_points/2
    triangle_h = np.clip(triangle_h, 0, 1)
    c5_triangle = evaluate_erdos_solution(triangle_h)
    if c5_triangle < best_c5:
        best_c5 = c5_triangle
        best_h = triangle_h.copy()
    
    # Cosine squared shape
    cosine_h = np.cos(np.pi / 2 * (x - 1)) ** 2
    cosine_h = cosine_h * (n_points / 2) / np.sum(cosine_h)
    cosine_h = np.clip(cosine_h, 0, 1)
    c5_cosine = evaluate_erdos_solution(cosine_h)
    if c5_cosine < best_c5:
        best_c5 = c5_cosine
        best_h = cosine_h.copy()
    
    # Try a few powers of cosine
    for p in [0.5, 1.5, 2.5, 3.5]:
        temp_h = np.maximum(0, np.cos(np.pi / 2 * (x - 1)) ** p)
        temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
        temp_h = np.clip(temp_h, 0, 1)
        temp_c5 = evaluate_erdos_solution(temp_h)
        if temp_c5 < best_c5:
            best_c5 = temp_c5
            best_h = temp_h.copy()
    
    # Parametric polynomial shapes
    for a in [1, 2, 3, 4]:
        poly_h = np.maximum(0, 1 - np.abs(x - 1) ** a)
        poly_h = poly_h * (n_points / 2) / np.sum(poly_h)
        poly_h = np.clip(poly_h, 0, 1)
        c5_poly = evaluate_erdos_solution(poly_h)
        if c5_poly < best_c5:
            best_c5 = c5_poly
            best_h = poly_h.copy()
    
    # Hill climbing from the best shape found
    if best_h is not None:
        h = best_h.copy()
        current_c5 = best_c5
        max_iters = 300000  # maybe too many, but perform time checks
        iter_count = 0
        
        # Base hill-climbing parameters
        epsilon_base = 0.05
        max_epsilon = 0.05
        min_epsilon = 1e-6
        epsilon_decay = 0.9999
        
        while iter_count < max_iters and time.time() < end_time:
            iter_count += 1
            
            # Choose two random indices
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            
            # Compute maximum delta we can move
            delta_max = min(h[i], 1 - h[j], max_epsilon)
            
            if delta_max <= 0:
                continue
                
            # Select a move amount
            delta = np.random.random() * delta_max
            
            # Compute new values
            h_new = h.copy()
            h_new[i] = h[i] - delta
            h_new[j] = h[j] + delta
            
            # Ensure bounds
            h_new = np.clip(h_new, 0, 1)
            
            # Evaluate the new C5
            new_c5 = evaluate_erdos_solution(h_new)
            
            # Accept if improvement
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                # Reset epsilon for restart
                epsilon = epsilon_base
            else:
                # Accept with decreasing probability
                prob = np.exp(-(new_c5 - current_c5) / (max_epsilon * 0.02))
                if np.random.random() < prob:
                    h = h_new
                    current_c5 = new_c5
            
            # Decrease epsilon gradually
            epsilon_base = max(min_epsilon, epsilon_base * epsilon_decay)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Ensure the final best_h meets the sum constraint and is within bounds
    if best_h is not None:
        sum_h = np.sum(best_h)
        factor = (n_points / 2) / sum_h
        best_h = best_h * factor
        best_h = np.clip(best_h, 0, 1)
        # Validate final sum
        final_sum = np.sum(best_h)
        # Correct small deviation
        best_h = best_h * (n_points / 2) / final_sum
        best_h = np.clip(best_h, 0, 1)
        # Final evaluation
        final_c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to uniform
        uniform_h = np.ones(n_points) * 0.5
        final_c5 = evaluate_erdos_solution(uniform_h)
        return (uniform_h.tolist(), final_c5, n_points)
