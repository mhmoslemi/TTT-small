import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Try different resolutions
    candidates = [150, 200, 250]
    
    for n in candidates:
        if time.time() - start_time > budget_s - 5:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
            
        try:
            # Create LP problem
            prob = pulp.LpProblem("ErdosMinOverlapLP", pulp.LpMinimize)
            
            # Variables
            h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Continuous')
            T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
            
            # Constraint: sum h_i = n/2
            prob += pulp.lpSum([h[i] for i in range(n)]) == n/2
            
            # Product constraints and shift constraints
            for d in range(1, n):  # Skip d=0
                y_vars = []
                for i in range(n-d):
                    y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
                    y_vars.append(y)
                    
                    # McCormick envelope constraints
                    prob += y <= h[i]
                    prob += y <= h[i+d]
                    prob += y >= h[i] + h[i+d] - 1
                
                # Shift constraint: sum_y >= n/2 - T
                prob += pulp.lpSum(y_vars) >= n/2 - T
            
            # Objective: minimize T
            prob += T
            
            # Solve
            solver = pulp.PULP_CBC_CMD(msg=0, timeLimit=remaining_time - 1)
            prob.solve(solver)
            
            # Check solution
            if pulp.LpStatus[prob.status] in ['Optimal', 'Feasible']:
                h_vals = np.array([pulp.value(h[i]) for i in range(n)], dtype=float)
                result_c5 = evaluate_erdos_solution(h_vals)
                
                if result_c5 < best_c5:
                    best_c5 = result_c5
                    best_h = h_vals.copy()
                    best_n = n
                    
        except Exception as e:
            continue
            
        if best_c5 < 0.3807:
            break
            
    # Fallback to initial solution if no better found
    if best_h is None:
        h_vals = np.array(initial_h_values, dtype=float)
        best_c5 = evaluate_erdos_solution(h_vals)
        best_h = h_vals.copy()
        best_n = len(h_vals)
        
    return (best_h, best_c5, best_n)
