import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    end_time = start_time + budget_s * 0.99
    
    # Parameters
    n_points = 501  # Odd number for better symmetry
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Start with initial shape: cosine^2
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2.0  # Midpoints
    h_initial = np.cos(np.pi/2 * (x - 1)) ** 2
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    h_initial = h_initial * (target_sum / sum_h)
    h_initial = np.clip(h_initial, 0, 1)
    
    # Evaluate initial solution
    def evaluate(h):
        # Using the provided evaluation function
        correlation = np.correlate(h, 1 - h, mode='full')
        C5 = np.max(correlation) * dx
        return C5

    h_best = h_initial.copy()
    c5_best = evaluate(h_best)
    
    # Simulated annealing parameters
    temp_start = 0.02
    temp_end = 0.0001
    cooling_rate = 0.99995
    max_iter = 100000
    
    iteration = 0
    temp = temp_start
    
    # Keep track of the best found so far
    h_best_global = h_best.copy()
    c5_best_global = c5_best
    
    while time.time() < end_time and iteration < max_iter:
        # Random move
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Do not move an element with itself
        if i == j:
            iteration += 1
            continue
            
        # Compute max_delta
        max_delta = min(h_best[i], 1 - h_best[j])
        if max_delta < 1e-8:
            iteration += 1
            continue
            
        # Choose delta
        delta = np.random.uniform(0, max_delta * 0.1)  # Small change
        
        # Propose new h
        h_new = h_best.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Ensure stay within bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Correction to maintain sum
        sum_new = np.sum(h_new)
        diff = sum_new - target_sum
        if abs(diff) > 1e-10:
            # Redistribute difference to some random positions
            # Pick one position to subtract
            k = np.random.randint(0, n_points)
            delta_k = min(h_new[k], abs(diff))
            h_new[k] -= delta_k
            # Distribute the rest
            remaining = abs(diff) - delta_k
            if remaining > 1e-10:
                # Distribute among other random positions
                ids = np.random.permutation(n_points)
                for idx in ids:
                    if remaining <= 0:
                        break
                    if idx == k:
                        continue
                    take = min(h_new[idx], remaining)
                    h_new[idx] -= take
                    remaining -= take
            h_new = np.clip(h_new, 0, 1)
            
        # Evaluate new solution
        c5_new = evaluate(h_new)
        
        # Accept or reject based on temperature
        if c5_new < c5_best or np.random.rand() < np.exp(-(c5_new - c5_best) / temp):
            h_best = h_new
            c5_best = c5_new
            
            # Update global best
            if c5_best < c5_best_global:
                h_best_global = h_best.copy()
                c5_best_global = c5_best
                
        # Anneal temperature
        temp = max(temp_end, temp * cooling_rate)
        
        iteration += 1
    
    # Final check
    final_c5 = evaluate(h_best_global)
    
    return (h_best_global.tolist(), final_c5, n_points)
