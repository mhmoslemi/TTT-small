import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx=2.0):
    """Evaluate the maximum overlap integral for discretized h"""
    # compute all correlations for shifts from -n+1 to n-1
    corr = correlate(h, 1 - h, mode="full") * dx
    # returns the maximum across all shifts
    return np.max(corr)

def build_mcormick_constraints(h, y_vars, n, s):
    """Generate McCormick constraints for variable pair h[i], h[i+s]"""
    constraints = []
    for i in range(n):
        if i + s < 0 or i + s >= n:
            continue
        j = i + s
        # y_ij >= h[i] + h[j] - 1
        constraints.append(y_vars[i, s + n - 1] >= h[i] + h[j] - 1)
        # y_ij >= 0
        constraints.append(y_vars[i, s + n - 1] >= 0)
        # y_ij <= h[i]
        constraints.append(y_vars[i, s + n - 1] <= h[i])
        # y_ij <= h[j]
        constraints.append(y_vars[i, s + n - 1] <= h[j])
    return constraints

    t = cp.Variable()
    objective = cp.Minimize(t)

    # Constraints
    constraints = []
    constraints.extend([0 <= h[i] for i in range(n)])
    constraints.extend([h[i] <= 1 for i in range(n)])
    constraints.append(cp.sum(h) == mass_target)
    constraints.append(t >= 0)

    # Add McCormick constraints for all valid pairs
    for s in range(-n + 1, n):
        if s == 0:
            continue  # skip when s=0 (not relevant to correlations)

            if i + s < 0 or i + s >= n:
                continue

            constraints.append(y_vars[i, s + n - 1] >= h[i] + h[j] - 1)
            # y_{i,s} >= 0
            constraints.append(y_vars[i, s + n - 1] >= 0)
            # y_{i,s} <= h[i]
            constraints.append(y_vars[i, s + n - 1] <= h[i])
            # y_{i,s} <= h[j]
            constraints.append(y_vars[i, s + n - 1] <= h[j])

    for s in range(-n + 1, n):

    prob = cp.Problem(objective, constraints)
