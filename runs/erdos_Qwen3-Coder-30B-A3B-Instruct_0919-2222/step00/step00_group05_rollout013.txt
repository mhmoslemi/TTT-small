def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h function."""

    # Compute cross-correlation of h with (1 - h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx
    conv *= dx
    # Return maximum value
    return np.max(conv)

def project_onto_constraints(h, dx):
    """Project h onto the feasible set."""
    # Fix sum constraint (integral = 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / h_sum) * (2.0 / dx)
    # Clamp all values to [0, 1]
    h = np.clip(h, 0.0, 1.0)
    return h

def generate_random_feasible(n_points, dx):
    """Generate a random feasible solution."""
    h = np.random.rand(n_points)
    h = project_onto_constraints(h, dx)
    return h

    """Perform a single swap move."""
    h_new = h.copy()

    return project_onto_constraints(h_new, dx)

def single_shift_move(h, dx, i, delta):
