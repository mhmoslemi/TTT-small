import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    N = 120
    dx = 2.0 / N
    A = []
    for k in range(N):
        A_k = np.zeros((N, N))
        for i in range(N - k):
            A_k[i, i + k] = 1.0
            A_k[i + k, i] = 1.0  # symmetric
        A.append(A_k)
    
    # Bisection on τ to find smallest feasible value
    low, high = 0.35, 0.5
    target_precision = 1e-4
    best_h = None
    best_c5 = float('inf')
    
    while high - low > target_precision and (time.time() - start_time < budget_s - 5):
        tau = (low + high) / 2.0
        gamma = 1 - tau
        
        h = cp.Variable(N)
        X = cp.Variable((N, N), symmetric=True)
        
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == N / 2.0,
            X >> 0,
        ]
        
        for i in range(N):
            constraints.append(X[i, i] == h[i])
            
        for k in range(N):
            constraints.append(cp.trace(A[k] @ X) >= gamma)
        
        prob = cp.Problem(cp.Maximize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_val = h.value.copy()
                X_val = X.value.copy()
                # Extract using leading eigenvalue
                eigenvals, eigenvecs = np.linalg.eigh(X_val)
                max_eigenval = eigenvals[-1]
                max_eigenvec = eigenvecs[:, -1]
                h_candidate = np.sqrt(max_eigenval) * max_eigenvec
                
                # Clip and project
                h_candidate = np.clip(h_candidate, 0, 1)
                objective = lambda theta: np.sum(np.maximum(0, np.minimum(1, h_candidate - theta))) - N/2
                l, r = 0, 1
                for _ in range(30):
                    theta = (l + r) / 2
                    val = objective(theta)
                    if val < 0:
                        r = theta
                    else:
                        l = theta
                theta = (l + r) / 2
                h_projected = np.maximum(0, np.minimum(1, h_candidate - theta))
                
                # Verify mass constraint
                assert abs(np.sum(h_projected) - N/2) < 1e-10
                
                # Polishing with projected gradient
                def compute_c5(h_array):
                    return np.max(correlate(h_array, 1 - h_array, mode='full')) * dx
                
                current_c5 = compute_c5(h_projected)
                best_c5 = min(best_c5, current_c5)
                if current_c5 < best_c5:
                    best_h = h_projected.copy()
                
                # Slight polishing via projected gradient
                max_iter = 500
                alpha_0 = 0.1
                original_c5 = current_c5
                
                def projected_gradient_step(h, k_star):
                    # Assumes largest negative term dominates
                    grad = np.zeros_like(h)
                    if k_star < len(h):
                        grad[:len(h)-k_star] += (1 - h[k_star:])
                        grad[k_star:] -= h[:len(h)-k_star]
                    # Project onto capped simplex
                    h -= alpha_0 * grad
                    # Project onto [0,1] and re-normalize to sum N/2
                    h = np.clip(h, 0, 1)
                    
                    # Adjust to hit exact mass
                    current_sum = np.sum(h)
                    diff = N/2 - current_sum
                    if abs(diff) > 1e-10:
                        # Add evenly to all entries with shrinkage step if needed
                        h += diff / len(h)
                        h = np.clip(h, 0, 1)
                        # Normalize final sum if overshoot
                        actual_sum = np.sum(h)
                        h *= (N/2) / actual_sum
                    return h
                
                k_max = None
                for _ in range(max_iter):
                    if time.time() - start_time > budget_s - 5:
                        break
                    c5_vec = correlate(h_projected, 1 - h_projected, mode='full')
                    k_max_idx = np.argmin(c5_vec)
                    k_max = k_max_idx - (len(h_projected) - 1)
                    h_projected = projected_gradient_step(h_projected, k_max)
                    new_c5 = compute_c5(h_projected)
                    if new_c5 < current_c5 - 1e-6:
                        current_c5 = new_c5
                        best_c5 = min(best_c5, current_c5)
                        if current_c5 < best_c5:
                            best_h = h_projected.copy()
                        if new_c5 < 0.3807:
                            break  # early termination
                    else:
                        # Stopped improving or oscillating, stop polishing
                        break
                        
                # Update bisection bounds
                if abs(current_c5 - tau) < 1e-4:
                    break  # convergence within tolerance
                if current_c5 < tau:
                    high = tau
                else:
                    low = tau
            else:
                low = tau
        except Exception:
            low = tau
    
    # Final evaluation even if incomplete
    if best_h is None:
        # Simple fallback
        h_test = np.ones(N) * 0.5
        best_c5 = np.max(correlate(h_test, 1 - h_test, mode='full')) * dx
        best_h = h_test

    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    final_h *= (N/2) / final_sum  # Normalize

    c5_eval = np.max(correlate(final_h, 1 - final_h, mode='full')) * dx
    return (final_h.tolist(), float(c5_eval), N)
