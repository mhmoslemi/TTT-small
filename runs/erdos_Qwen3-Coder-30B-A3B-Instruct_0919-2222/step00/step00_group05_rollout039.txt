import numpy as np
import time
from scipy.special import logsumexp

# Precomputed shift maps
def precompute_shift_maps(n):
    shift_maps = {}
    for s in range(-(n-1), n):
        pairs = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                pairs.append((i, j))
        shift_maps[s] = pairs
    return shift_maps

# Euclidean projection onto capped simplex
def project_to_capped_simplex(v, M):
    # v: array of length n
    # M: target sum (e.g., n/2)
    u = np.clip(v, 0, 1)
    current_sum = np.sum(u)
    if abs(current_sum - M) < 1e-10:
        return u
    elif current_sum > M:
        # Need to decrease values
        # Sort in descending order
        sorted_indices = np.argsort(u)[::-1]
        # Find threshold
        cumsum = 0.0
        for i, idx in enumerate(sorted_indices):
            if cumsum + (u[idx] - 1) > M:
                break
            cumsum += (u[idx] - 1)
        theta = (cumsum - M) / (len(sorted_indices) - i) if i < len(sorted_indices) else 0
        # Actually we need a more precise algorithm
        # Let's compute residuals and find right threshold
        u_sorted = u[sorted_indices]
        remaining = M
        for i in range(len(u_sorted)):
            if u_sorted[i] <= remaining / (len(u_sorted) - i):
                remaining -= u_sorted[i]
            else:
                tmp = remaining / (len(u_sorted) - i)
                arr = np.copy(u_sorted)
                for j in range(i, len(arr)):
                    arr[j] = tmp
                result = np.zeros_like(u)
                result[sorted_indices] = arr
                return result
    else:
        # Need to increase values
        # Sort in ascending order
        sorted_indices = np.argsort(u)
        cumsum = 0.0
        for i, idx in enumerate(sorted_indices):
            if cumsum + (1 - u[idx]) < M - current_sum:
                cumsum += (1 - u[idx])
            else:
                break
        theta = (M - current_sum - cumsum) / (len(sorted_indices) - i) if i < len(sorted_indices) else 0
        # Actually more correctly based on current method
        # Use the known standard algorithm
    # Use the iterative method (best for numerical stability):
    def proj_simplex(v, s):
        # projects v onto standard simplex (sum = s)
        # using algorithm from https://stanford.edu/~jduchi/projects/DuchiShSiCh08.pdf
        u = np.sort(v)[::-1]
        rho = 0.0
        for i in range(len(u)):
            rho += u[i]
            if (rho - s) / (i + 1) < u[i]:
                break
        return np.clip(v - (rho - s) / (i + 1), 0, v)
    
    # Project v onto threshold s with 0 <= h_i <= 1
    # Found in https://math.stackexchange.com/a/2560979/399839
    # Simple axis-aligned projection: first project into box, then into the hyperplane
    # When not in box, project to the boundary or midpoint of [0,1],
    # then iterate or just clip and adjust (exact math indeed)
    # More tractable: Collate to find threshold as in standard algorithm
    
    # Simpler: due to the exact structure here, we adopt direct Tailor:
    # https://citeseerx.ist.psu.edu/viewdoc/download?doi=10.1.1.105.2175&rep=rep1&type=pdf
    # or standard Refernce:
    # https://www.cs.cmu.edu/~suvrit/teaching/notes/taylorprojection.pdf
    
    # Let's implement a clean implement:
    # Sort values in descending order
    z = np.argsort(v)[::-1]
    y = v[z]
    
    # Initial sum
    sj = 0.0
    for i in range(len(y)):
        sj += y[i]
        # Compute threshold sigma
        sigma = (sj - M) / (i+1)
        if sigma >= y[i]:
            break
    else:
        # All elements too small
        sigma = (sj - M) / len(y)
    # Now construct the result by applying the threshold
    result = np.zeros_like(v)
    for i in range(len(z)):
        idx = z[i]
        if v[idx] <= sigma:
            result[idx] = 0
        else:
            result[idx] = v[idx] - sigma
    
    # Ensure sum is correct (numerical balance)
    diff = np.sum(result) - M
    result += diff / len(v)  # Equal redistribution to satisfy constraint
    return np.clip(result, 0.0, 1.0)

# Compute all correlations C_s(h)
def correlations(h, shift_maps):
    n = len(h)
    corrs = []
    for s in range(-(n-1), n):
        C_s = 0.0
        for i, j in shift_maps[s]:
            C_s += h[i] * (1 - h[j])
        corrs.append(C_s)
    return np.array(corrs)

# Smooth approximation and its gradient
def smooth_obj_and_grad(h, alpha, shift_maps):
    n = len(h)
    corrs = correlations(h, shift_maps)
    max_corr = np.max(corrs)
    # Use log-sum-exp trick for numerical stability
    shifted = alpha * (corrs - max_corr)
    # Numerical trick inside logsumexp
    log_sum_exp = logsumexp(shifted, axis=0)
    obj = (log_sum_exp + max_corr * alpha) / alpha
    # Compute gradient
    # w_s = exp(alpha * (C_s - max_corr)) / sum exp(alpha * (C_r - max_corr))
    # Here already numerically stable enough
    w_s = np.exp(shifted) / np.exp(log_sum_exp)
    grad = np.zeros_like(h)
    for i in range(n):
        for s in range(-(n-1), n):
            pairs = shift_maps[s]
            for idx1, idx2 in pairs:
                if idx1 == i:
                    # derivative of h_idx1 (1 - h_idx2)
                    # of C_s = 1 - h_idx2 - h_idx1 (but are we in backdiff?)
                    # Actually C_s = sum h_i (1 - h_{i+s})
                    # So ∂C_s/∂h_i = 1 - h_{i+s} - h_{i-s} if both indices present
                    if (s == 0):
                        grad[i] += w_s[s + n - 1] * (1 - h[i])  # Since the below are indexed shifted in reverse aliasing manner
                    else:
                        if s > 0:
                            offset_i_s = i + s
                            if 0 <= offset_i_s < n:
                                grad[i] -= w_s[s + n - 1] * h[offset_i_s]
                        else:  # s < 0
                            offset_i_s = i + s
                            if 0 <= offset_i_s < n:
                                grad[i] -= w_s[s + n - 1] * h[offset_i_s]
                        # Add this term once: if s = i + s - i or s = s,
                        # But rule is ∂C_s/∂h_i = 1 - h_{i+s} - h_{i-s}
                        grad[i] += w_s[s + n - 1]
    return obj, grad

# Run single descent step for fixed α
def descent_one_alpha(h0, alpha, eta0, max_iter, shift_maps):
    h = h0.copy()
    eta = eta0
    for k in range(max_iter):
        obj, g = smooth_obj_and_grad(h, alpha, shift_maps)
        # Update in direction of -grad
        h = h - eta * g
        h = project_to_capped_simplex(h, len(h) / 2)
    return h, obj

# Gradient descent continuation scheme
def run_continuation(h_start, alpha_schedule, initial_eta, shift_maps, time_limit):
    # Reuse worst-case computing from vanilla
    start_time = time.time()
    current_h = h_start.copy()
    best_obj = np.inf
    last_time = time.time()
    
    for i, alpha in enumerate(alpha_schedule):
        if (time.time() - start_time) / (time_limit) > 0.95:
            break
        last_time = time.time()
        # Run few descent steps
        # First fix eta
        current_eta = initial_eta / 2**(i // 3)  # decrease step size with α
        max_iter_current = min(50, max(10, int(200 * (1 - i / len(alpha_schedule)))))
        h_new, obj_new = descent_one_alpha(current_h, alpha, current_eta, max_iter_current, shift_maps)
        current_h = h_new
        
        # Re-evaluate true max correlation for monitoring  
        corrs = correlations(current_h, shift_maps)
        true_max = np.max(corrs)
        if true_max < best_obj:
            best_obj = true_max
            # Keep the current_h for best cubic polynomial
    return current_h, best_obj

# Create random binary initialisations
def random_binary(n, M):
    idxs = np.random.permutation(n)[:int(M)]
    h = np.zeros(n)
    h[idxs] = 1
    return h

# Create random fractional vectors from Dirichlet(1)
def random_fractional(n, M):
    h = np.random.dirichlet([1]*n)
    h = project_to_capped_simplex(h, M)
    return h

# Local targeted swaps to improve binary vectors
def swap_local_search(h_bin, shift_maps, time_limit):
    # h_bin in [0,1] but represents step function
    n = len(h_bin)
    start_time = time.time()
    while time.time() - start_time < time_limit:
        corrs = correlations(h_bin, shift_maps)
        idx_max = np.argmax(corrs)
        max_val = corrs[idx_max]
        if max_val <= 0.3808:
            break  # found value we want
        # Try swaps to improve it
        # For shift idx_max, we can manipulate
        # pairs (i,j) such that h[i] * (1-h[j]) contributes to C[idx_max]
        # i -> j where h[i]=1, h[j]=0  ('carry' weight)
        # But if h[i]=1 and h[j]=1 => (1-h[j])=0, so contribution zero.
        # Find pairs where h[i]=1 and h[j]=0, and if j is shifted by idx_max from i
        # then setting h[i]=0 and h[j]=1 reduces the correlation
        i_bad_idx = -1
        j_good_idx = -1
        # Find where i is set to 1 but shifted gives 0
        for i, j in shift_maps[idx_max]:
            if h_bin[i] == 1 and h_bin[j] == 0:
                i_bad_idx = i
                j_good_idx = j 
                break
        if i_bad_idx == -1:
            # search for any place to swap and improve that worst shift
            for i in range(n):
                for j in range(n):
                    # look at indices that would affect the worst shift
                    if j - i == idx_max and h_bin[i] == 1 and h_bin[j] == 0:
                        i_bad_idx = i  
                        j_good_idx = j
                        break
                if i_bad_idx != -1:
                    break

        if i_bad_idx != -1:
            # perform a swap attempt to potentially reduce max_corr [c]
            old_h = h_bin.copy()
            h_bin[i_bad_idx] = 0
            h_bin[j_good_idx] = 1
            # recompute with a full correlation
            corrs = correlations(h_bin, shift_maps)
            new_max = np.max(corrs)
            if new_max < max_val:
                continue  # Use new h_bin
            else:
                h_bin[i_bad_idx] = 1
                h_bin[j_good_idx] = 0
        else:
            break # Graphically poor choice, no swaps beneficial
    return h_bin
