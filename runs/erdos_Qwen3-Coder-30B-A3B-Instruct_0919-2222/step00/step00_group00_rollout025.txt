def project_onto_feasible(y, target_sum):
    """Projection onto [0,1]^n intersected with sum = target_sum using water-filling"""
    y = np.clip(y, 0, 1)
    delta = np.sum(y) - target_sum
    if abs(delta) < 1e-12:
        return y
    
    y_sorted = np.sort(y)[::-1]
    free_indices = np.where((y > 0) & (y < 1))[0]
    free_count = len(free_indices)
    
    # Compute how many points to adjust
    while free_count > 0 and abs(delta) > 1e-12:
        # Try to adjust all free variables at once
        if delta > 0:
            # Need to decrease
            adjust = min(delta / free_count, np.min(y[free_indices]) if len(free_indices) > 0 else float('inf'))
            if adjust <= 0:
                break
            for i in free_indices:
                y[i] -= adjust
            delta -= adjust * free_count
        else:
            # Need to increase
            adjust = -delta / free_count
            for i in free_indices:
                max_inc = 1 - y[i]
                if max_inc < adjust:
                    break
                y[i] += adjust
            delta += adjust * free_count
            
        # Re-evaluate free indices
        y_sorted = np.sort(y)[::-1]
        free_indices = np.where((y > 0) & (y < 1))[0]
        free_count = len(free_indices)
    
    if abs(delta) < 1e-12:
        return y
    elif delta > 0:
        for i in range(len(y)):
            if y[i] > 0:
                y[i] = max(0, y[i] - delta)
                delta = 0
                break
        return y
    else:
        for i in range(len(y)):
            if y[i] < 1:
                delta = -delta
                y[i] += min(delta, 1 - y[i])
                delta = 0
                break
        return y

def smooth_objective_and_weights(c, alpha):
    """Compute smoothed ln-sum-exp value and weights for gradient."""
    c_max = np.max(c)
    exp_term = np.exp(alpha * (c - c_max))
    weight = exp_term / np.sum(exp_term)
    smooth_val = (1./alpha) * (np.log(np.sum(exp_term)) + c_max)
    return smooth_val, weight

def gradient_h(h, weights):

    # Use more efficient prefix sums for ts stability
    
    n = len(h)
    # Forward contrib: sum_{k=0}^{n-1-i} weights[k] * (1 - h[i+k])  
    # This requires conv of weights and 1-h for diagonal terms
    p1 = np.pad(1-h, (0, n), constant_values=0)
    p2 = np.pad(weights, (0, n-1), constant_values=0)
    conv1 = np.convolve(p2, p1)[n-1:2*n-1]  # offset!
    
    # Like reverse conv but shifted
    p3 = np.pad(h, (0, n), constant_values=0)
    p4 = np.pad(weights[::-1], (0, n-1), constant_values=0)
    conv2 = np.convolve(p4, p3)[n-1:2*n-1]
    grad = conv1 - conv2
    
    return grad[:n]

def continuous_optimization(h, n, t0, budget_s):
    """Run projected smooth gradient descent with momentum and adaptive α."""
    dx = 2.0 / n
    eta = 0.5 * dx
    beta = 0.9
    alpha = 5.0
    max_alpha = 200.0
    stagnation_count = 0
    
    h_prev = h.copy()
    best_h_cont = h.copy()
    best_c5_cont = float('inf')
    
    max_iter = 10000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - t0 < budget_s - safety_margin:
        iter_count += 1
        if iter_count % 100 == 0:
            if time.time() - t0 > budget_s - safety_margin:
                break
            
        # Compute correlation
        c = compute_correlation(h)
        
        # Smooth surrogate
        smooth_val, weight = smooth_objective_and_weights(c, alpha)
        
        # Compute gradient
        grad = gradient_h(h, weight)
        
        # Take momentum step
        h_trial = h - eta * (grad + beta * (h - h_prev))
        h_prev = h.copy()
        
        # Project
        h_trial = project_onto_feasible(h_trial, n / 2)
        h = h_trial.copy()
        
        # New correlation after projection
        new_c = compute_correlation(h)
        new_val = np.max(new_c) * dx
        
        if new_val < best_c5_cont:
            best_c5_cont = new_val
            best_h_cont = h.copy()
        
        # Update α if improvement was made
        if smooth_val < max_alpha and iter_count % 20 == 0:
            alpha *= 1.2
            alpha = min(alpha, max_alpha)
        
        # Early stop if stale
        if iter_count % 500 == 0:
            if abs(c.min() - np.max(c)) < 1e-6:
                stagnation_count += 1
                if stagnation_count > 5:
                    break
            else:
                stagnation_count = 0

    return best_h_cont

def round_to_binary(h_float, n):
    """Round the continuous solution to binary given exactly n/2 ones."""
    h_bin = np.zeros(n)
    idxs = np.argsort(h_float)[::-1]
    # Set exactly n/2 highest values to 1
    count = 0
    for i in range(n):
        if h_float[idxs[i]] > 0:
            h_bin[idxs[i]] = 1
            count += 1
            if count >= n//2:
                break
    return h_bin

def local_binary_search(h, n, t0, budget_s):
    """Improve the binary solution with swaps."""
    # Compute full correlation
    c = compute_correlation(h)
    c_max = np.max(c)
    off = n - 1
    best_h = h.copy()
    best_c_max = c_max
    
    # Define sets for ONES and ZEROS
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    # Compute approximate delta for faster search or use full double loop
    swap_count = 0
    max_swaps = 100000  # Conservative limit that should be within 600s
    
    while swap_count < max_swaps and time.time() - t0 < budget_s - safety_margin:
        # Find the worst shift
        s_best = np.argmax(c)
        a = []  # Candidates for set A (ones that helped)
        b = []  # Candidates for set B (zeros that could help)
        for i in ones:
            j = i + s_best
            if j < n and h[j] == 0:
                a.append(i)
        for j in zeros:
            i = j - s_best
            if i >= 0 and h[i] == 1:
                b.append(j)
        
        if a and b:
            i = np.random.choice(a)
            j = np.random.choice(b)
            
            # Swap i and j - this directly updates correlation
            hnew = h.copy()
            hnew[i], hnew[j] = hnew[j], hnew[i]
            new_c = compute_correlation(hnew)
            new_max = np.max(new_c)
            
            if new_max <= c_max:
                h = hnew
                c = new_c
                c_max = new_max
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                if new_max < best_c_max:
                    best_c_max = new_max
                    best_h = h.copy()
        else:
            # Fall back to random swap if no matching pair found
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                
                # Swap and recompute
                hnew = h.copy()
                hnew[i], hnew[j] = hnew[j], hnew[i]
                new_c = compute_correlation(hnew)
                new_max = np.max(new_c)
                
                # Accept if non-worsening or accept rarely if worsening with prob
                if new_max <= c_max:
                    h = hnew
                    c = new_c
                    c_max = new_max
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    if new_max < best_c_max:
                        best_c_max = new_max
                        best_h = h.copy()
        
        swap_count += 1
        
        #收敛判断
        if swap_count % 5000 == 0:
            if np.max(c) < best_c_max + 1e-6:
                break
                
    return best_h, best_c_max * (2.0 / n)
