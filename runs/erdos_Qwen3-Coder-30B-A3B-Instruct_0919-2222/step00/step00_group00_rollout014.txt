import numpy as np
import time
import random
from math import log, exp

def compute_correlation(h):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    return c

def smooth_objective_and_weights(c, alpha):
    c_max = np.max(c)
    if c_max == -np.inf:
        c_max = 0
    exp_c = np.exp(alpha * (c - c_max))
    exp_sum = np.sum(exp_c)
    weights = exp_c / exp_sum
    smooth_val = (1 / alpha) * (log(exp_sum) + c_max)
    return smooth_val, weights

def grad_h(h, weights):

    # Compute prefix sums for efficient gradient calculation
    prefix_1mh = np.cumsum(1 - h[::-1])[::-1]

    g = np.zeros(n)
    # Forward contributions: (1 - h[i+s]) terms
    # This can be computed as 1-h[i+s] * I for s=0,...,n-1-i

    # The direct sum is for i fixed, computing partial sums forward
    # What we want: g[i] = sum_{s=0}^{n-1-i} w_{s} (1 - h[i+s]) - sum_{s=i}^{n-1} w_{-s} h[i-s]
    # We'll use convolution
    g_direct = np.convolve(weights[:n], 1 - h, mode='valid') # indices shift in [0,n-1] but needs better handling
    # Better approach: better loop over shifts to avoid transversal issues
    
    # We compute in O(n) using prefix sums:
    # g[i] = forward_sum[i] - backward_sum[i]
    # where forward_sum[i] = sum_{j=i}^{n-1} (1-h[j])
    # and backward_sum[i] = sum_{j=0}^{i} h[j] 
    # for the contribution of each shift:
    # It's complex - simpler is as convolution with appropriate indexing
    
    # Direct summation with prefix: 
    # Each shift contributes term j with weight w_{j-i}
    # So g[i] = sum_{j=0}^{i-1} w_{i-j}(1-h[j]) + sum_{j=i+1}^{n-1} w_{j-i}(1-h[j])
    #  = sum_{j=0}^{i-1} w_{i-j}(1-h[j]) + sum_{j=i+1}^{n-1} w_{j-i}(1-h[j])
    # = sum_{j=0}^{i-1} w_{i-j}(1-h[j]) + sum_{k=0}^{n-2-i} w_{-k-1}(1-h_{n-1-k})
    # This suggests reverse/backward sums must be handled. Simpler:
    # For each i, compute the weighted sum:
    # w_{i+shift}(1 - h[i+shift]) for different shifts (where valid)
    # and w_{i-shift} h[i-shift] for different shifts (where valid)
    # Let's use simple loop for clarity
    
    forward_sum = np.zeros(n) # sum_{j=i}^{n-1} (1-h[j])

    cum1mh = np.flip(np.cumsum(np.flip(1 - h)))
    cumh = np.cumsum(h)

    forward_sum = np.concatenate([cum1mh[1:], np.array([0])])
    backward_sum = cumh
    
    # compute inner products for all shifts at once
    # We need: sum_{position=0..n-1} weights_shift[position] * [{1-h}[pos] or h[pos]]
    g = np.zeros(n)
    
    # Manually sum for clarity:
    # Let's use numpy convolutions on slices in loops if needed,
    # For simplicity do the basic gradient computation by hand
    
    # g[i] = sum_{s} w_s [ (1 - h[i+s]) I[i+s < n] - h[i-s] I[i-s >= 0]]
    # Consider the back-convolution way for speed:
    # s from -n+1 to n-1 (but properly offset). 
    # Use a combination of correlation operations
    # g = (weights_o * 1-h) + (weights_o * h) with the right indexing
    # We'll use a loop for correctness here
    
    g = np.zeros(n)
    for i in range(n):
        grad_i = 0
        for s in range (-n+1, n):
            shift_idx = i + s
            if shift_idx < 0: continue
            if shift_idx >= n: continue
            # contribution: w_s * (1 - h[shift_idx]) 
            grad_i += weights[s+n-1] * (1 - h[shift_idx])
            # also, same term with negative sign if s > 0 
            # contribution: - w_{-s} h[i-s] if i-s >= 0
            if s > 0:
                shift_neg = i - s
                if shift_neg >= 0:
                    grad_i -= weights[-s + (n-1)] * h[shift_neg]
        g[i] = grad_i
    return g

def project_onto_feasible(y, target_sum):
    n = len(y)
    # Clip to [0,1]
    y = np.clip(y, 0, 1)
    current_sum = np.sum(y)
    delta = current_sum - target_sum
    
    if abs(delta) < 1e-15: 
        return y
    # Adjust elements that are in (0, 1)
    entries = [(i, y[i]) for i in range(n)]
    entries.sort(key=lambda x: x[1])
    
    # While difference is large, adjust entries inside (0,1)
    # Using water-filling: reduce or increase entries equally    
    # We have them sorted by value ascending
    # Now reduce or increase more elements
    if delta > 0:
        # want to reduce: if delta > 0, need to decrease entries
        # So decrease |delta| by distribute among elements in (0,1)
        i = 0
        left_sum = 0
        # Find all free positions (in (0,1))
        free_indices = [k for k in range(n) if 0 < y[k] < 1]
        # Sort by value increasing  
        free_indices.sort(key=lambda x: y[x])
        idxs_in = free_indices
        # We process from lowest value to highest so we improve the equality
        if len(idxs_in) == 0: return y  # All at bounds (or all 0 or all 1)
        
        # Total excess (amount to remove): delta
        excess = delta
        if excess > 0:
            # We need to reduce current_sum, distribute excess
            # Adjust the 'leftmost' free elements (lowest) to remove excess
            # If all elements in (0,1) need to be reduced below 0, better handle this!
            for k in idxs_in:
                # Ensure this doesn't underflow or overflow  
                # Let target_k = y[k] - excess/(total num_free)
                # However we process until we reach minimum
                new_y_k = y[k] - excess / len(idxs_in)
                if new_y_k <= 0:
                    # we have to remove y[k] from free list,
                    # adjust excess including term for that element
                    excess += y[k]
                    # Remove element and continue  
                    idxs_in.remove(k)
                    continue
                else:
                    y[k] = new_y_k
                    excess = 0
                    break
            else:
                # If we end above, some penalty applied above may have been
                # obtained by trial-and-error. Fall through to return
                pass
    else:
        # Add 'delta' = deficit = negative of excess
        diff = -delta
        # We increase interior elements by diff/total_free such that all go to 1.0
        # Find all elements that can be raised
        free_indices = [k for k in range(n) if 0 < y[k] < 1]
        # Increase all equally until we get to 1 or we run out of diff 
        # We may be increasing all:
        total_free = len(free_indices)
        if total_free == 0: return y
        to_add = diff / total_free
        for k in free_indices:
            new_val = y[k] + to_add
            y[k] = min(new_val, 1)
            diff -= to_add
    return y

def swap_update(c, h, i, j, offset):
    # update c due to swap of h[i] and h[j]
    # original shift i+j -> c[shift + offset]
    # delta update in contribution to shift s:
    # s0 = i - j -> old was h[i] * (1-h[j]) = 1*0 = 0, new: 0*1 = 0
    # for shift k = i-l:
    # In old: h[l], h[k] = h[i-l] = h[i-l]
    # update: c[shift+off] += ...
    n = len(h)
    c_old = c.copy()
    # for 's' in [-n+1, n-1], compute new contribution to c[s]
    # but since every update involves the same index pairs, we can compute directly
    
    # at i, delta is - (1 - h[i]) if previously was h[i] = 1
    # the delta at shift s for i:
    for l in range(n): # l since i-l = s
        s = i - l
        if s < 0:
            shift_idx = s + offset   # s from -n+1 to n-1 -> [0, 2n-2]
        else:
            shift_idx = s + offset
            
        # Val to be incremented due to change at shift i (from l)
        if s == 0:  # shift 0 wrt i
            val = 1 - h[i]
        else:
            val = 1 - h[l]
            
        if l+shift_idx < len(c): c[l+shift_idx] += (1 - h[j] ) - (1 - h[i]) if l == i else 0
        # Looks messy originally...
        # explain part also for change around j

    # Simpler way:
    # Rebuild the correlation under change
    if i >= j:
        i, j = j, i
    # changing h[i] from 1 to 0, h[j] from 0 to 1:
    # For shift `p = k - i`:
    # old continuous prob:
    # h[i] * (1 - h[i + p]) = 1 * (1 - h[i + p])
    # new: h[j] * (1 - h[j + p]) = 1 * (1 - h[j + p])
    # so total change is (1 - h[i + p]) - (1 - h[j + p]) = h[j + p] - h[i + p] 
    # but h[i] was 1 (going to 0)
    # and h[j] is 0 (going to 1)
    # These changes happen for all p such that i + p is in [0,n) and j + p is in [0,n)
    # i + p >= 0 => p >= -i
    # i + p < n => p < n - i
    # j + p >= 0 => p >= -j
    # j + p < n => p < n - j
    # So p in [max(-i, -j), min(n-i, n-j))  
    # Actually, changes in linear form:
    # For each shift is it the effect of shift i on any j as if j moves in direction -> ops is (1-h[i]) vs (1-h[j])
    # Both ways can be read using correlations 
    # i at 1 -> old contribution to corr = (1 - h[i+p])
    # j at 0 -> old corr is (1 - h[j+p]) 
    # New: i at 0 -> contribution (1 - h[i+p])
    # New: j at 1 -> contribution (1 - h[j+p])
    # We compute difference resulting from the swap:
    # New - Old: (1-h[j+p]) - (1-h[i+p]) = h[i+p] - h[j+p] 
    # For shift p:
    # Del = sum_{p in range}[ h[i+p]  - h[j+p] ]  
    # But the shifts only valid when i+p and j+p are valid indices
    
    # Use: previous contribution of i is 1 - h[i+p] for p in [0, n-i), 
    #      j's is 1 - h[j+p] for p in [0, n-j)
    #      after, i's is 1 - h[i+p], j's is 1 - h[j+p]
    #      So the delta term must be (1 - h[j+p] ) - (1 - h[i+p]) = h[i+p] - h[j+p]
    #        for all useful shifts p where i+p is in [0,n) and j+p is in [0,n)
    
    old_changes = np.zeros(len(c)) 
    for p in range(0, n):
        if i + p < n and j + p < n:
            # old contribution of i: h[i] = 1, so h[i+p] = 1 * (1-h[i+p])
            # contribution from j: h[j] = 0, so 0 * (1-h[j+p])
            # after swap: contribution from i = 0 * (1-h[i+p]) = 0 
            #              contribution from j = 1 * (1-h[j+p]) = (1-h[j+p])
            # intermediate = (1-h[j+p]) - (1-h[i+p]) = h[i+p] - h[j+p]
            pdelta = h[i] * (1 - h[i + p]) - h[j] * (1 - h[j + p])
            old_changes[p + offset] += pdelta  # TODO: The index offset may need more thought repeatedly
            # Because we talk about correlation at multiple inds.
            # If we store correlation as [shift 0, ..., shift 2*n - 2]  
            # shift from -n+1 to n-1 using multipliers
            # shift s corresponds to index s + offset where offset = n - 1
    # Compute newly changes made by delta
    # i pushes increases by h[i] (which is 1) minus old should be 
    # It looks simpler in terms of correction to specific points
    # Final version:
    l_start = max(-i, -j)   # min after shift should be >= 0
    l_end = min(n-i, n-j)   # max after shift should be < n
    for l in range(l_start, l_end):  
        c[l+offset] = c[l+offset] - (1 - h[i + l]) + (1 - h[j + l])
    return c

def targeted_swap(h, c, n, offset):
    # compute c and enforce since might be outdated
    # and we know victim is where max shift is
    # identify worst shift
    max_shift = np.argmax(c)
    max_shift_val = c[max_shift]
    # For that shift s, sort indices by h[i]=1. 
    # Then find in the sorted list the one for this shift
    max_s = max_shift - offset  # map from index back to actual shift
    ONES = np.where(h == 1)[0]
    ZEROS = np.where(h == 0)[0]
    
    if len(ONES) == 0 or len(ZEROS) == 0:
        return h, c  # no points to swap
    
    A = []  # indices where h[i] == 1 and h[i + shift] = 0 
    B = []  # indices where h[i] == 0 and h[i + shift] = 1
    
    # Scan ONES
    for i in ONES:
        if 0 <= i + max_s < n and h[i + max_s] == 0:
            A.append(i)
    # Scan ZEROS  
    for i in ZEROS:
        if 0 <= i + max_s < n and h[i + max_s] == 1:
            B.append(i)
    # if A and B are not empty, swap a random pair to decrease the shift value
    if A and B:
        i = random.choice(A)
        j = random.choice(B)
        # before swapping, store old r (which becomes observation)
        orig_c_s = c[max_shift]
        # Do the actual swap
        h[i], h[j] = h[j], h[i]
        # and update c  
        # Only one shift changes due to this swap
        # If s is shift of index i <-> j (reciprocal), then shift s is affected:
        # https://math.stackexchange.com/questions/865886/what-is-the-change-in-cross-correlation-function-when-two-points-are-swapped 
        # Realize!
        # We compute delta changes in the correlation vector for a swap:
        # When i and j swap, the correlation c[s] for shifts s = i-j  and s=j-i change:
        # We only compute delta that matters for chain of changes.
        # Simple loop updates for current c for now:
        # Just recompute entire correlation vector
        c = compute_correlation(h)
        # Or override specific shift:
        # delta_c_s = h[i] * (1 - h[i + max_s]) - h[j] * (1 - h[j + max_s]) 
        # Since we're swapping 1 meaning old contribution = h[j] * (1 - h[j + max_s])
        #                        new contribution = h[i] * (1 - h[i + max_s])  
        # Let's just temporary recompute correctly:
        old = h[i] * (1 - h[i + max_s]) + h[j] * (1 - h[j + max_s])
        new = h[j] * (1 - h[j + max_s]) + h[i] * (1 - h[i + max_s])
        # old vs new diff is (h[j] - h[i]) * (1 - h[j + max_s])  +  (h[i] - h[j])*(1 - h[i + max_s])
        # = (h[j] - h[i]) [1 - h[j + max_s]] + (h[i] - h[j]) [1 - h[i + max_s]]  
        
        # Manually update just the affected in case fix if needed
        
        # Actually save weighted sum update trick instead:
        # By large scale corr update, just force to recompute for simplicity
        c = compute_correlation(h)
        return h, c
    
    # Else - random swap
    # Also, we could fall back to exactly this case when not all A, B
    i = random.choice(ONES)
    j = random.choice(ZEROS)
    # mimic swapping using temporary replacement of values
    h[i], h[j] = h[j], h[i] 
    # recompute full which is valid
    c = compute_correlation(h)
    # Validate return is clean 
    return h, c

def upscale(h, new_n):
    old_n = len(h)
    new_h = np.zeros(new_n) 
    ratio = new_n / old_n
    for i in range(old_n):
        start = int(i * ratio)
        end = int((i+1)*ratio)
        for j in range(start, end):
            new_h[j] = h[i]
    # apply same time-similarity stretching, further ensure exact average
    return new_h

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)
    
    # Discretization sizes to try
    ns = [80, 120, 180, 240]
    
    best_h = None
    best_val = float('inf')
    best_n = None
    
    safety_margin = 2.0
    
    for n in ns:
        if time.time() - start_time > budget_s - safety_margin:
            break
            
        print(f"Optimizing for n={n}")
        
        # Initial feasible point: random then project
        h0 = np.random.rand(n)
        h0 = project_onto_feasible(h0, n / 2)
        
        # Continuous optimization
        h = h0.copy()
        alpha = 5.0
        eta = 0.5 * (2.0 / n)
        beta = 0.9
        v_prev = 0
        best_cont_val = float('inf')
        
        max_iter = min(10000, int(budget_s * 200 / n)) if n < 100 else 5000
        t0 = time.time()
        
        for k_iter in range(max_iter):
            if time.time() - start_time > budget_s - safety_margin:
                break
                
            c = compute_correlation(h)
            smooth_val, weights = smooth_objective_and_weights(c, alpha) 
            
            g = grad_h(h, weights) 
            v = beta * v_prev + (1 - beta) * g
            
            # Gradient descent step with momentum
            h_trial = h - eta * v
            h_trial = project_onto_feasible(h_trial, n / 2)
            
            c_trial = compute_correlation(h_trial)
            smooth_val_trial, _ = smooth_objective_and_weights(c_trial, alpha)
            
            # Accept step?
            if smooth_val_trial <= smooth_val:
                h = h_trial
                v_prev = v
            else:
                # Try smaller step-size
                eta *= 0.5
                if eta < 1e-10: # too small, should not happen often
                    break
                h_trial = h - eta * v
                h_trial = project_onto_feasible(h_trial, n / 2)
                try:
                    # Recompute again if previous smooth val matches 
                    c_trial2 = compute_correlation(h_trial)
                    smooth_val_trial2, _ = smooth_objective_and_weights(c_trial2, alpha)
                    if smooth_val_trial2 <= smooth_val:
                        h = h_trial
                        v_prev = v
                except:
                    pass
                
            # Update alpha slowly
            if k_iter % 100 == 0 and alpha < 200:
                alpha *= 1.2
                
            # Evaluate true objective
            true_val = np.max(c) * (2.0 / n)
            if true_val < best_cont_val:
                best_cont_val = true_val 
                
            # Record best even if not the continuous-chosen one
            if true_val < best_val:
                best_val = true_val
                best_h = h.copy()
                best_n = n
                                
            if k_iter % 100 == 0 and (time.time() - t0) > 0.05:
                t0 = time.time()
                
        # Finish continuous stage early if time needed
        if time.time() - start_time > budget_s - safety_margin:
            break

        # Rounding phase
        idxs = np.argsort(h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idxs[:n//2]] = 1
        
        # Polishing phase (binary local search)
        c = compute_correlation(h_bin)
        t0 = time.time()
        for _ in range(100000):
            if time.time() - start_time > budget_s - safety_margin:
                break
            h_bin, c = targeted_swap(h_bin, c, n, n-1)
            max_shift = np.argmax(c)
            if c[max_shift] <= 0:
                break
                
        # Final value check
        final_val = np.max(c) * (2.0 / n)
        if final_val < best_val:
            best_val = final_val
            best_h = h_bin.copy()
            best_n = n
            
    # Return best solution
    return best_h.tolist(), best_val, best_n
