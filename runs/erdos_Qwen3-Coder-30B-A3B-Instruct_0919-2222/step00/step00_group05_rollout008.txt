import numpy as np
import time
from scipy.special import softmax

# Precomputed shift maps for n=70
def precompute_shifts(n):
    shift_maps = {}
    for s in range(-(n-1), n):
        indices = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                indices.append((i, j))
        shift_maps[s] = indices
    return shift_maps

# Project a vector onto the capped simplex: 0 <= h_i <= 1, sum h_i = M
def project_to_capped_simplex(v, M):
    u = np.clip(v, 0, 1)
    s = np.sum(u)
    if abs(s - M) < 1e-10:
        return u
    # Sort in descending order
    sorted_u = np.sort(u)[::-1]
    # Find threshold
    cumsum = np.cumsum(sorted_u)
    threshold = (cumsum - M) / np.arange(1, len(sorted_u) + 1)
    # Take maximum threshold such that cumulative sum exceeds M
    idx = np.where(sorted_u > threshold)[0]
    if len(idx) == 0:
        theta = 0
    else:
        theta = threshold[idx[-1]]
    # Apply threshold
    result = np.clip(v - theta, 0, 1)
    # Ensure sum is exactly M
    diff = M - np.sum(result)
    # Adjust by assigning difference to largest values
    indices = np.argsort(v)[::-1]
    for i in indices:
        if diff == 0:
            break
        if result[i] < 1:
            take = min(1 - result[i], diff)
            result[i] += take
            diff -= take
    return result

# Compute correlation vector for shift s
def correlations(h, shift_maps):
    n = len(h)
    Cs = np.zeros(2*n - 1)
    for s in shift_maps:
        corr = 0
        for i, j in shift_maps[s]:
            corr += h[i] * (1 - h[j])
        # Map to index -(n-1) to n-1 -> 0 to 2n-2
        Cs[s + (n - 1)] = corr
    return Cs

# Smooth maximum and its gradient with respect to h
def smooth_obj_and_grad(h, alpha, shift_maps):
    Cs = correlations(h, shift_maps)
    max_C = np.max(Cs)
    # Numerically stable softmax
    exp_Cs = np.exp(alpha * (Cs - max_C))
    w = exp_Cs / np.sum(exp_Cs)
    # Gradient
    grad = np.zeros_like(h)
    for s in shift_maps:
        idxs = shift_maps[s]
        for i, j in idxs:
            grad[i] += w[s + (n-1)] * (1 - h[j] - h[i if s < 0 else j])
    return np.sum(w * Cs), grad

# Projected gradient descent for fixed alpha
def descent_one_alpha(h0, alpha, eta0, max_iter, shift_maps, time_limit):
    start_time = time.time()
    h = h0.copy()
    eta = eta0
    for t in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        # Evaluate the smoothed surrogate
        val, grad = smooth_obj_and_grad(h, alpha, shift_maps)
        # Update
        h_new = h - eta * grad
        # Project
        h = project_to_capped_simplex(h_new, n/2)
        # Possibly backtrack if target decreases
        # For simplicity, we do just fixed step size, unless early stopping later
        if t % 10 == 0:
            eta *= 0.95  # decay
    return h

# Run full continuation from single seed
def run_continuation(h0, alpha_schedule, eta0, shift_maps, time_limit):
    # Precompute grid
    n_points = len(h0)
    remaining_time = time_limit
    h = h0.copy()
    for alpha in alpha_schedule:
        if remaining_time < 10:
            break
        remaining_time -= 3 # coarse estimate of the inner loop duration
        max_iter = min(100, max(20, 1000//alpha)) # adaptive iterations
        h = descent_one_alpha(h, alpha, eta0, max_iter, shift_maps, remaining_time)
        if time.time() - start_time > time_limit:
            break
    return h

# Generate a random binary vector with fixed sum
def random_binary(n, M):
    h = np.zeros(n)
    pos = np.random.choice(n, int(M), replace=False)
    h[pos] = 1
    return h

# Generate a random fractional vector and project it
def random_fractional(n, M):
    a = np.random.dirichlet([1.0] * n)
    return project_to_capped_simplex(a, M)

# Evaluate C5 with exact correlation evaluation
def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all shifts: full convolution (1-h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

# Binary rounding + local swap refinement
def enhance_binary(h_cont):
    n = len(h_cont)
    # Deterministic rounding
    idxs = np.argsort(h_cont)[::-1]
    b = np.zeros(n)
    for i in range(int(n/2)):
        b[idxs[i]] = 1

    # Local swaps to improve
    max_corr = evaluate(b)
    # Run a number of swaps
    for _ in range(1000):
        max_s = -1
        max_val = -1
        Cs = correlations(b, shift_maps)
        for s in range(-(n-1), n):
            if Cs[s + (n-1)] > max_val:
                max_val = Cs[s + (n-1)]
                max_s = s
        if max_s == -1:
            break
        i = -1
        # Find leftmost 1
        for k in range(n):
            if b[k] == 1:
                i = k
                break
        if i == -1:
            break
        # Try to find a swap to make trial b adjustable
        improved = False
        for j in range(n):
            # Swap dropping 1 at i and setting 0 at j
            if b[i] == 1 and b[j] == 0:
                # See if it improves the shift at max_s
                k1 = i + max_s
                k2 = j + max_s
                # check bounds:
                if 0 <= k1 < n and 0 <= k2 < n:
                    # whether it improves max_s correlation:
                    a1 = i
                    a2 = j
                    if (b[k1] == 0 and b[k2] == 1 and 
                        a1 >= 0 and a2 >= 0):
                        # Perform swap
                        b[i] = 0
                        b[j] = 1
                        new_max = evaluate(b)
                        if new_max < max_corr:
                            max_corr = new_max
                            improved = True
                        else:
                            b[i] = 1
                            b[j] = 0
        if not improved:
            break
    return b
