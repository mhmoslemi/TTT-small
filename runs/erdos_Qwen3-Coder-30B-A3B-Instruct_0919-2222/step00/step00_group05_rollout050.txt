import time
import random
from scipy import correlate

def evaluate_erdos_solution(h):
    # Compute correlation using full mode
    corr = correlate(h, 1 - h, mode='full')
    # Multiply by dx to get integral scaling factor
    dx = 2.0 / len(h)
    corr *= dx
    # Return maximum correlation value
    return np.max(corr)
