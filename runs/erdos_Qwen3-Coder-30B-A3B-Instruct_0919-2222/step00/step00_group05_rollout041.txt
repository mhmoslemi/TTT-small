import numpy as np
import time
import random
from scipy import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap problem constant C5 for given step function h."""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h) using full mode (zero-padded)
    corr = correlate(h, 1 - h, mode='full') * dx
    # Return the maximum of the correlation over all shifts
    return np.max(corr)

    no_improvement_count = 0
    max_no_improvement = 50000

    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        # Annealing temperature
        temperature = max(temp_min, temp_start * (cool_down_factor ** (iteration // iter_per_temp)))
        
        # Choose random indices to swap values
        i = random.randint(0, n_points - 1)
        j = random.randint(0, n_points - 1)
        
        # Only proceed if different indices
        if i == j:
            continue
            
        # Determine change amount (small value)
        delta = random.uniform(0.001, 0.05)
        # Ensure movement within [0,1] bounds
        delta = min(delta, h[i], 1 - h[j])
        
        # Make a copy of current h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # Ensure bounds
        h_new[i] = max(0, min(1, h_new[i]))
        h_new[j] = max(0, min(1, h_new[j]))
        
        # Rescale to restore constraint sum(h) = target_sum
        new_sum = np.sum(h_new)
        if abs(new_sum - target_sum) > 1e-10:
            scaling_factor = target_sum / new_sum
            for k in range(n_points):
                h_new[k] *= scaling_factor
                h_new[k] = max(0, min(1, h_new[k]))

        # Evaluate new solution
        current_c5 = evaluate_erdos_solution(h_new)
        
        # Accept or reject based on simulated annealing criteria
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        elif random.random() < np.exp(-(current_c5 - best_c5) / temperature):
            # Accept the worse solution with some probability

        if iteration % 50000 == 0:

            # Rescale to meet sum constraint

            h = np.clip(h, 0, 1)
