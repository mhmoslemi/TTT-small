import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations with mode="full"
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum (over all shifts)
    max_corr = np.max(corr)
    return max_corr * dx

def build_mcormick_constraints(h, y, s, i, n):
    """Build McCormick constraints for y[i,s] = h[i] * h[i+s]"""
    # Ensure indices are within bounds
    if i + s < 0 or i + s >= n:
        return []
    # y[i,s] <= h[i]
    c1 = y[i, s + n - 1] <= h[i]
    # y[i,s] <= h[i+s]
    c2 = y[i, s + n - 1] <= h[i + s]
    # y[i,s] >= h[i] + h[i+s] - 1
    c3 = y[i, s + n - 1] >= h[i] + h[i + s] - 1
    # y[i,s] >= 0
    c4 = y[i, s + n - 1] >= 0
    return [c1, c2, c3, c4]

def solve_relaxed_lp(h, y, t, n, M):
    """Solve the relaxed linear program"""
    # Constraints
    constraints = []
    # Box constraints
    for i in range(n):
        constraints.append(h[i] >= 0)
        constraints.append(h[i] <= 1)
    # Mass constraint
    mass_constraint = cp.sum(h) == M
    constraints.append(mass_constraint)
    # t >= 0
    constraints.append(t >= 0)
    # McCormick constraints for each shift
    for s in range(1 - n, n):
        for i in range(n):
            if 0 <= i + s < n:
                c = build_mcormick_constraints(h, y, s, i, n)
                constraints.extend(c)
    # Shift bounds: sum(y[i,s]) >= M - t
    for s in range(1 - n, n):
        total = 0
        for i in range(n):
            if 0 <= i + s < n:
                total += y[i, s + n - 1]
        constraints.append(total >= M - t)
    # Objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        return True
    except:
        return False

def deterministic_round(h_star, M):
    """Deterministic rounding of fractional h to integer."""
    h = h_star.copy()
    idxs = np.argsort(-h)  # sort indices by descending value
    h[:] = 0
    h[idxs[:int(M)]] = 1
    return h

def random_round(h_star, M, n_samples=200):
    """Random rounding of fractional h to integer."""
    h_samples = []
    for _ in range(n_samples):
        # Sample indices to set to 1
        probs = np.clip(h_star, 0, 1)
        probs = probs / np.sum(probs)
        selected = np.random.choice(n, int(M), p=probs, replace=False)
        h = np.zeros(n)
        h[selected] = 1
        h_samples.append(h)
    return h_samples

def targeted_swap_polish(h, max_time_s=120):
    """Apply targeted swap improvements to reduce max correlation."""
    start_time = time.time()
    h = h.copy()
    n = len(h)
    
    def compute_corr(h, dx=2.0 / n):
        corr = correlate(h, 1 - h, mode="full")
        return corr * dx
    
    corr = compute_corr(h)
    max_corr = np.max(corr)
    chosen_shift = np.argmax(corr)
    
    swapped = True
    while swapped and time.time() - start_time < max_time_s:
        swapped = False
        best_corr = max_corr
        best_i, best_j = None, None
        
        # Find a pair that would reduce max correlation
        for i in range(n):
            if h[i] == 0:
                continue
            # For the current worst shift
            s = chosen_shift - (n - 1)
            j = (i + s) % n
            # If this makes shift i an invalid connection
            # and j is a valid connection on the other side
            if h[j] == 1:
                # However, this is not a complete swap check
                # Let's just evaluate a single swap
                h_test = h.copy()
                h_test[i] = 0
                h_test[j] = 1
                corr_test = compute_corr(h_test)
                new_max_corr = np.max(corr_test)
                if new_max_corr < best_corr:
                    best_corr = new_max_corr
                    best_i, best_j = i, j
        
        if best_i is not None:
            h[best_i] = 0
            h[best_j] = 1
            max_corr = best_corr
            corr = compute_corr(h)
            chosen_shift = np.argmax(corr)
            swapped = True
    
    return h, max_corr
