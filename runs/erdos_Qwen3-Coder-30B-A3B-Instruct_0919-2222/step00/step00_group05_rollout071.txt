def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h"""
    # Convert to numpy array if needed
    h = np.array(h)
    # Compute correlation of h with (1-h) using full convolution mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Compute max over all possible shifts
    c5 = np.max(corr) * (2.0 / len(h))
    return c5
