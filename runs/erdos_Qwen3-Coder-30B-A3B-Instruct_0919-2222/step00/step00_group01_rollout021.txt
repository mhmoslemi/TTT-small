        if not isinstance(h, np.ndarray):
            h = np.array(h)
        h = np.maximum(0, np.minimum(1, h))  # Clamp to [0,1]
        # Compute correlation using np.correlate (full mode)
        conv = np.correlate(h, 1 - h, mode='full')
        # Scale the convolution by dx
        conv = conv * dx
        # Return max value, which is the C5 constant
        return np.max(conv)
    
    # Objective function for optimization to minimize C5
    def objective(h):
        return evaluate_C5(h)
    
    # Bounds for each h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) * dx = 1
    def constraint_sum(h):
        return np.sum(h) * dx - 1
    
    # Use differential evolution as it's good for global optimization
    # with constraints, and is suitable for non-smooth objectives
    result = differential_evolution(
        objective,
        bounds,
        constraints={'fun': constraint_sum, 'type': 'eq'},
        seed=seed,
        maxiter=int(budget_s / 5) if budget_s > 10 else 100,
        disp=False,
        polish=False  # To speed up, avoid local optimization if too slow
    )

        final_h = np.clip(result.x, 0, 1)
        final_c5 = evaluate_C5(final_h)
        return (final_h, final_c5, n_points)
