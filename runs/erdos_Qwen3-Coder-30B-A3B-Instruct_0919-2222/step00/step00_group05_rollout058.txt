# Helper functions
def project_to_capped_simplex(v, M):
    """Project vector v onto capped simplex {h: 0 <= h_i <= 1, sum h_i = M}"""
    u = np.clip(v, 0, 1)
    s = np.sum(u)
    if abs(s - M) < 1e-10:
        return u
    # Sort in descending order
    sorted_indices = np.argsort(u)[::-1]
    sorted_u = u[sorted_indices]
    
    # Find threshold
    threshold = 0.0
    cumsum = 0.0
    for i, val in enumerate(sorted_u):
        cumsum += val
        if cumsum - (i+1) * val > M:
            # Clip at threshold
            threshold = (cumsum - M) / (i + 1)
            break
    else:
        # All should be clipped
        threshold = cumsum / len(sorted_u)
    
    result = np.zeros_like(v)
    for i in range(len(sorted_indices)):
        result[sorted_indices[i]] = max(0, min(1, v[sorted_indices[i]] - threshold))
    return result

def compute_correlations(h, shift_maps):
    """Compute correlations C_s for all shifts"""
    n = len(h)
    correlations = []
    for s, indices in shift_maps.items():
        corr = 0.0
        for i, j in indices:
            corr += h[i] * (1 - h[j])
        correlations.append(corr)
    return np.array(correlations)

def smooth_obj_and_grad(h, alpha, shift_maps):
    """Compute smooth approximation and its gradient"""
    corr = compute_correlations(h, shift_maps)
    max_corr = np.max(corr)
    # Numerically stable softmax
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    w = w / w_sum
    # Gradient
    grad = np.zeros_like(h)
    
    # Compute ∂C_s/∂h_i = 1 - h_{i+s} - h_{i-s}
    for s, indices in shift_maps.items():
        for i, j in indices:
            # Add in direction of s
            # compute contribution to ∂Φ/∂h_i
            if i < len(h):
                grad[i] += w[s + len(shift_maps) - 1] * (1 - h[j])  # (1-h_{i+s})
            if j < len(h):
                grad[i] -= w[s + len(shift_maps) - 1] * h[i]  # -h_i (in the product term)
        
    # Compute result
    smooth_val = (1/alpha) * (np.log(w_sum) + max_corr * alpha)
    return smooth_val, grad

def run_descent(h0, alpha_schedule, eta0, max_inner_iter, shift_maps, time_limit):
    """Run gradient descent for the entire alpha schedule starting from h0"""
    h = h0.copy()
    start_time = time.time()
    best_h = None
    best_corr = float('inf')
    
    # Initialize alpha schedule indices
    current_alpha_idx = 0
    alpha_max_idx = len(alpha_schedule) - 1
    eta = eta0
    
    for alpha_idx in range(len(alpha_schedule)):
        alpha = alpha_schedule[alpha_idx]
        for _ in range(max_inner_iter):
            if time.time() - start_time > time_limit - 0.2:
                break
            # Compute smooth objective and gradient
            obj_val, grad = smooth_obj_and_grad(h, alpha, shift_maps)
            # Update
            v = h - eta * grad
            # Project
            h = project_to_capped_simplex(v, len(h)/2)
            # Check
            corr = compute_correlations(h, shift_maps)
            current_max = np.max(corr)
            if current_max < best_corr:
                best_corr = current_max
                best_h = h.copy()
            # Update eta
            if alpha > 8: eta *= 0.99                
        if time.time() - start_time > time_limit - 0.5:
            break
    return best_h, best_corr

def is_valid_candidate(h):
    """Check that h is a valid candidate"""
    return np.all((h >= 0) & (h <= 1)) and abs(np.sum(h) - 35) < 1e-6

# Pre-computing shift maps for each shift index
def make_shift_maps(n):
    shift_maps = {}
    for s in range(-(n-1), n):
        indices = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                indices.append((i, j))
        shift_maps[s] = indices
    return shift_maps

def random_binary(n, M):
    """Generate a random binary vector with exactly M ones"""
    indices = sorted(random.sample(range(n), M))
    h = np.zeros(n)
    for idx in indices:
        h[idx] = 1.0
    return h

def random_fractional(n, M):
    """Generate a random fractional vector projected onto simplex"""
    v = np.random.dirichlet([1.0]*n)
    return project_to_capped_simplex(v, M)

def deterministic_rounding(h):
    """Deterministic rounding to binary vector"""
    # Sort indices by h values (descending)
    indices = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    for i in range(int(len(h)/2)):
        h_bin[indices[i]] = 1.0
    return h_bin

def compute_max_correlation(h):
    """Compute the actual maximum correlation using numpy correlate"""
    h = np.array(h)
    n = len(h)
    correlations = []
    # This corresponds to what happens in full mode
    conv = np.correlate(h, 1 - h, mode='full')
    # The "maximum" entry in the convolution vector corresponds to the max C_s
    max_corr = np.max(conv)
    return max_corr

def swap_local_search(h_bin, max_iter, time_budget):
    """Improve binary vector with swaps to decrease max correlation"""
    start_time = time.time()
    n = len(h_bin)
    current_corr = compute_max_correlation(h_bin)  # float value
    if current_corr <= 0.3807:
        return h_bin[:]
    for _ in range(max_iter):
        if time.time() - start_time > time_budget - 0.5:
            break
        # Compute all correlations
        correlations = np.correlate(h_bin, 1 - h_bin, mode='full')
        max_idx = np.argmax(correlations)
        max_corr = correlations[max_idx]
        if max_corr <= 0.3807:
            break
        
        # Shift index (map full convolution index to shift)
        shift = max_idx - n + 1  # since mode='full' gives lags from -(n-1) to (n-1)
        
        # Try to find good swap
        best_swap_improvement = 0
        best_i, best_j = -1, -1
        
        for i in range(n):
            j = (i + shift) % n
            # Check if this i, j produces an improvement
            if h_bin[i] == 1 and h_bin[j] == 0:  # i has a 1, j has a 0: we can swap
                # Try swapping i to 0, j to 1, and compute new correlations
                h_trial = h_bin.copy()
                h_trial[i] = 0
                h_trial[j] = 1
                
                corr_trial = compute_max_correlation(h_trial)
                improvement = max_corr - corr_trial
                if improvement > best_swap_improvement:
                    best_swap_improvement = improvement
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h_bin[best_i] = 0
            h_bin[best_j] = 1
            current_corr = compute_max_correlation(h_bin)
        else:
            # No improving swap found, stop
            break
    return h_bin
