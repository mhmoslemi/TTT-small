import numpy as np
from scipy.optimize import differential_evolution
from scipy import fft
import time

def project_to_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum x = target_sum"""
    v = np.array(v)
    v_sorted = np.sort(v)[::-1]
    n = len(v)
    tau = 0.0
    cumsum = 0.0
    for i in range(n):
        cumsum += v_sorted[i]
        threshold = (cumsum - target_sum) / (i + 1)
        if threshold <= 1:
            tau = threshold
        else:
            break
    x = np.maximum(0, np.minimum(1, v - tau))
    return x

def compute_autocorrelation(h, dx):
    """
    Compute the autocorrelation G(k) = ∫ h(x) h(x+k) dx, then f(k) = 1 - G(k).
    Return max(f) * dx for the final C5 bound.
    """
    N = len(h)
    H = fft.rfft(h)
    K = fft.rfft(1.0 - h)
    corr = fft.irfft(H * np.conj(K), n=2*N-1)
    # Remove the constant term and leading zero padding issues
    f_vals = 1.0 - corr[N-1:]
    max_f = np.max(f_vals)
    return max_f * dx

def evaluate_objective_block_params(u, v, M):
    # Reconstruct widths and heights
    w = 2 * np.exp(u) / np.sum(np.exp(u))
    a = 1.0 / (1.0 + np.exp(-v))
    # Compute integral constraint penalty
    integral = np.dot(a, w)
    pen = 1e3 * abs(integral - 1.0)
    # Block based computation of autocorrelation
    # Compute all interval overlaps - O(M^2 * N) where N = 200 is resolution
    N = 200
    dx = 2.0 / N
    s = np.concatenate([[0], np.cumsum(w)])  # segment boundaries
    # Construct piecewise constant function
    h_vals = np.zeros(N)
    for i in range(M):
        start = int(s[i] / dx)
        end = int(s[i+1] / dx)
        h_vals[start:end] = a[i]
    # Evaluate autocorrelation
    g = compute_autocorrelation(h_vals, dx)
    obj = g + pen
    return obj

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    
    # Parameters
    M = 12  # number of blocks
    N = 200  # resolution for final grid
    
    # Time management
    start_time = time.time()
    
    # Phase 1: Differential Evolution global search
    bounds = [(-5, 5)] * (2 * M)
    options = {'popsize': 15 * (2 * M), 'maxiter': 50}
    
    def objective(z):
        u = z[:M]
        v = z[M:]
        return evaluate_objective_block_params(u, v, M)
    
    remaining_time = budget_s * 0.2
    result_de = differential_evolution(
        objective, bounds, seed=seed,
        maxiter=20,  # limit iterations
        disp=False, polish=False
    )
    
    # Extract best DE solution
    if not result_de.success:
        print("DE failed, using random solution")
        u_init = rng.uniform(-5, 5, M)
        v_init = rng.uniform(-5, 5, M)
    else:
        u_init = result_de.x[:M]
        v_init = result_de.x[M:]
    
    # Phase 2: Local refinements
    best_h = None
    best_c5 = float('inf')
    
    # Refine initial solution
    u = u_init.copy()
    v = v_init.copy()
    
    # Reconstruct final scaled function
    w = 2 * np.exp(u) / np.sum(np.exp(u))
    a = 1.0 / (1.0 + np.exp(-v))
    
    integral = np.dot(a, w)
    a = a / integral
    
    # Ensure no overshoot in height
    a = np.clip(a, 0, 1)
    
    # Sample to fine grid
    dx = 2.0 / N
    s = np.concatenate([[0], np.cumsum(w)])
    h_vals = np.zeros(N)
    for i in range(M):
        start = int(s[i] / dx)
        end = int(s[i+1] / dx)
        if start >= N:
            continue
        end = min(end, N)
        if start < end:
            h_vals[start:end] = a[i]
    
    # Project to feasible simplex
    h_vals = project_to_capped_simplex(h_vals, N / 2)
    
    # Initial evaluation
    c5_bound = compute_autocorrelation(h_vals, dx)
    
    # Subgradient descent
    max_iter = 5000
    alpha0 = 0.2
    for t in range(max_iter):
        # Compute correlation
        H = fft.rfft(h_vals)
        K = fft.rfft(1.0 - h_vals)
        corr = fft.irfft(H * np.conj(K), n=2*N-1)
        corr = corr[N-1:]
        
        # Find argmax
        max_idx = np.argmax(corr)
        shift = max_idx
        
        # Compute gradient
        g = np.zeros(N)
        # Forward difference only for simplicity
        g[:-1] += (1 - h_vals[1:])  # part (1-h_{i+k})
        g[1:] -= h_vals[:-1]       # part h_{i-k}
        
        # Clamp to interval [0, 1]
        h_vals = np.clip(h_vals - alpha0 / np.sqrt(t+1) * g, 0, 1)
        # Project onto simplex
        h_vals = project_to_capped_simplex(h_vals, N / 2)
        
        # Evaluate every few iterations
        if t % 100 == 0:
            c5 = compute_autocorrelation(h_vals, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_vals.copy()
                
    # Double-check best
    if best_h is None:
        best_h = h_vals
        best_c5 = compute_autocorrelation(best_h, dx)
    
    # Final evaluate
    final_c5 = compute_autocorrelation(best_h, dx)
    
    return (best_h.tolist(), final_c5, N)
