    """Evaluate the Erdős constant for given step function h."""
    n = len(h)

    correlation = np.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (shifts from -1 to 1)
    max_correlation = np.max(correlation[n-1:2*n-1])
    return max_correlation * dx
