import numpy as np
import time

global initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    safety_margin = 2.0
    
    # Define discretizations to try, increasing resolution
    ns = [80, 120, 180, 240]
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    for i, n in enumerate(ns):
        if time.time() - t0 > budget_s - safety_margin:
            break
            
        # Generate initial point
        h0 = np.random.rand(n)
        h0 = project_onto_feasible(h0, n/2)
        
        # Continuous optimization phase
        h_cont, val_cont = optimize_continuous(h0, n, budget_s - (time.time() - t0), t0)
        
        # Round to binary
        h_bin = round_to_binary(h_cont, n)
        
        # Local combinatorial polishing
        h_polished, val_polished = polish_binary(h_bin, n, budget_s - (time.time() - t0), t0)
        
        # Keep the best
        if val_polished < best_c5:
            best_h = h_polished.copy()
            best_c5 = val_polished
            best_n = n
            
        # Possibly upscale for the next iteration (basic upsampling)
        if i < len(ns) - 1 and time.time() - t0 < budget_s - safety_margin:
            h_upscaled = upscale_binary(h_polished, ns[i+1])
            # Use h_upscaled as next initial point
        
    if best_h is None:
        # Fallback to initial solution
        h0 = np.array(initial_h_values)
        h0 = project_onto_feasible(h0, len(h0)/2)
        # Just evaluate the initial solution
        # Correlation computation
        corr = np.correlate(h0, 1-h0, mode='full')
        c5 = np.max(corr) * (2.0 / len(h0))
        return h0.tolist(), c5, len(h0)
    
    return best_h.tolist(), best_c5, best_n

# Helper function: Project a vector onto the feasible set (box [0,1] and sum = target).
def project_onto_feasible(y, target_sum):
    y = np.clip(y, 0, 1)
    total = np.sum(y)
    diff = total - target_sum
    
    if abs(diff) < 1e-12:
        return y
    if diff > 0:
        # Reduce values in interior to meet sum
        idx_free = np.where((y > 0) & (y < 1))[0]
        if len(idx_free) == 0:
            # All are at bounds, adjust all equally to target
            shift = diff / len(y)
            y -= shift
            # Re-project to 0,1
            y = np.clip(y, 0, 1)
        else:
            # Sequential reduction
            while abs(diff) > 1e-12:
                delta = diff / len(idx_free)
                for i in idx_free:
                    new = y[i] - delta
                    if new < 0:
                        # Remove from free set
                        y[i] = 0
                        diff += y[i]
                        idx_free = idx_free[idx_free != i]
                    elif new > 1:
                        y[i] = 1
                        diff -= (1 - y[i])
                        idx_free = idx_free[idx_free != i]
                    else:
                        y[i] = new
        # Ensure total is correct
        y += (target_sum - np.sum(y)) / len(y)
    else:  # diff < 0
        # Increase values in interior
        idx_free = np.where((y > 0) & (y < 1))[0]
        if len(idx_free) == 0:
            y += (target_sum - np.sum(y)) / len(y)
        else:
            while abs(diff) > 1e-12:
                delta = diff / len(idx_free)
                for i in idx_free:
                    new = y[i] + delta
                    if new > 1:
                        y[i] = 1
                        diff -= (1 - y[i])
                        idx_free = idx_free[idx_free != i]
                    elif new < 0:
                        y[i] = 0
                        diff += y[i]
                        idx_free = idx_free[idx_free != i]
                    else:
                        y[i] = new
    return y

# Compute correlation (with proper convolution)
def compute_correlation(h):
    corr = np.correlate(h, 1-h, mode='full')
    return corr

# Smoothed objective using log-sum-exp with stability trick
def smoothed_objective_and_weights(c, alpha):
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    Z = np.sum(exp_c)
    weights = exp_c / Z
    F_alpha = (1/alpha) * (np.log(Z) + c_max)
    return F_alpha, weights

# Compute gradient with respect to h of the surrogate
def grad_h(h, weights, n):
    # Forward and backward sums via convolution (O(n log n) but O(n) for earlier)
    # Using two Python loops for clarity and simplicity (assumes n <= 300)
    grad = np.zeros(n)
    # For each shift k:
    # change h[i] affects contribution to c_k by:
    #  (1 - h[i+k]) if i+k < n, -h[i-k] if i-k >= 0
    # But we'll do a slight variant for efficiency (both parts together):
    
    ones = np.ones(n)
    w = weights
    corr1 = np.convolve(1 - h, ones, 'full')  # Convolve (1-h) with ones
    corr2 = np.convolve(h, ones, 'full') 
    # Each element corr1[i] = sum_{j=i}^{n-1} (1 - h[j])
    # But we need: for shift s, when updating h[i] we want
    #   (1 - h[i+s]) if i+s < n; -h[i-s] if i-s >= 0
    
    # So for each shift, model:
    # weighted sum of (1-h_{i+s}) - h_{i-s} across shifts
    # We do this using indices manually
    for k, w_k in enumerate(w):
        # shift index: k = shift_val + (n-1)
        # k in [0, 2*n-2]
        shift_val = k - (n - 1)
        if shift_val >= 0:
            for i in range(n):
                # (1 - h[i + shift_val]) if i+shift_val < n
                if i + shift_val < n:
                    grad[i] += w_k * (1 - h[i + shift_val])
        else:
            # shift_val < 0 = -(-shift_val)
            for i in range(n):
                # - h[i - (-shift_val)] if i - (-shift_val) >= 0
                if i - (-shift_val) >= 0:
                    grad[i] -= w_k * h[i - (-shift_val)]
    return grad

# Optimizing continuous h via projected gradient descent
def optimize_continuous(h0, n, remaining_time, t_start):
    # Ensure initial is feasible
    h = h0.copy()
    h = project_onto_feasible(h, n/2)
    
    dx = 2.0 / n

    alpha = 5.0
    max_alpha = 200.0
    alpha_increase_interval = 200
    eta = 0.5 * dx
    beta = 0.9
    gamma = 0.5
    tol = 1e-6
    max_iter = 10000
    h_prev = h0.copy()
    
    best_val = float('inf')
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - t_start > remaining_time - 1.0:
            break
        if it % 100 == 0:
            corr = compute_correlation(h)
            c5 = np.max(corr) * dx
            if c5 < best_val:
                best_val = c5
                best_h = h.copy()
        
        # Compute correlation
        c = compute_correlation(h)
        F_alpha, w = smoothed_objective_and_weights(c, alpha)
        
        # Compute gradient
        g = grad_h(h, w, n)
        
        # Momentum step
        v = beta * (h - h_prev) + g
        h_new = h - eta * v
        
        # Project to feasible set
        h_new = project_onto_feasible(h_new, n/2)
        
        # Accept or reject with backtracking line search
        c_new = compute_correlation(h_new)
        F_alpha_new, _ = smoothed_objective_and_weights(c_new, alpha)
        
        if F_alpha_new <= F_alpha:
            h = h_new
            h_prev = h
            alpha = min(alpha * 1.2, max_alpha)
        else:
            eta *= gamma
            continue  # Re-try step with smaller eta
            
    # Final evaluation of best solution
    corr_final = compute_correlation(best_h)
    c5_final = np.max(corr_final) * dx
    return best_h, c5_final

# Round continuous h to binary (set first n/2 ones, rest zeros)
def round_to_binary(h, n):
    # Get sorted indices descending
    idxs = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    # Set top n//2 to 1
    h_bin[idxs[:n//2]] = 1
    # Just to be safe, enforce sum exactly
    total = np.sum(h_bin)
    excess = total - n/2
    if abs(excess) > 1e-12:
        # adjust using np.where necessary
        if excess > 0:
            # find some 1s to make 0
            ones_idxs = np.where(h_bin == 1)[0]
            offset = min(len(ones_idxs), int(excess))
            if offset > 0:
                for i in range(offset):
                    h_bin[ones_idxs[i]] = 0
        else:  # excess < 0
            zs = np.where(h_bin == 0)[0]
            offset = min(len(zs), int(-excess))
            for i in range(offset):
                h_bin[zs[i]] = 1
    return h_bin

        # for s = p - i (and p - j)
        #   tmp = delta * (1 - h[p]) for i add, delta * h[p] for j remove
        #   corr_new[shift] += tmp where shift = s + off
        
        # To keep few dive-ins:
        for q in range(n):
            # Update for removing i
            if q >= i:
                shift = q - i
                delta = -1 * (1 - h_bin[q])
                corr_new[shift] += delta

                shift = j - q
                delta = 1 * h_bin[q]

        if time.time() - t_start > remaining_time - 1.0:
            break
        attempt_count += 1
        # Need worst shift:
        c = compute_correlation(h_bin)
        worst_shift = np.argmax(c)
        max_c = c[worst_shift]
        if max_c <= best_val:  # No more improvements
            break
            
        # Find valid i,j pair for swap
        # i in ONES, j in ZEROS st c[worst_shift] = sum over q  terms of i,j

        mask_i = np.zeros(n, dtype=bool)

            candidates_i = [x for x in ONES if x + worst_shift < n and h_bin[x + worst_shift] == 0]
            candidates_j = [x for x in ZEROS if x - worst_shift >= 0 and h_bin[x - worst_shift] == 1]
        else:
            # shift values [n, 2n-2]
            shift_val = worst_shift - (n - 1)

            # j in ZEROS, if j + shift_val<n and h[j+shift_val]=0
            candidates_i = [x for x in ONES if x - shift_val >= 0 and h_bin[x - shift_val] == 1]
            candidates_j = [x for x in ZEROS if x + shift_val < n and h_bin[x + shift_val] == 0]

        # Pick one at random
        i = np.random.choice(candidates_i)
        j = np.random.choice(candidates_j)
        
        # Perform update
        h_bin[i] = 0
        h_bin[j] = 1
        ONES.remove(i)

# Upscale binary solution
def upscale_binary(h_old, n_new):
    h_up = np.repeat(h_old, 2)  # Fold each point
    if len(h_up) > n_new:
        h_up = h_up[:n_new]
    elif len(h_up) < n_new:
        h_up = np.pad(h_up, (0, n_new - len(h_up)), mode='constant')
    return h_up
