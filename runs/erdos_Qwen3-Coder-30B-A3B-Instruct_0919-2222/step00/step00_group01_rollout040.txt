import numpy as np
from scipy.optimize import minimize
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # since ∫ h(x) dx = 1 and dx = 2/n, sum should be n/2
    
    # Initialize with a simple construction: tent function
    h = np.zeros(n_points)
    mid = n_points // 2
    half_width = n_points // 3
    for i in range(n_points):
        if i < mid - half_width or i > mid + half_width:
            h[i] = 0
        elif i < mid:
            h[i] = (i - (mid - half_width)) / half_width
        else:
            h[i] = (mid + half_width - i) / half_width
    
    # Normalize to integral = 1
    h_sum = np.sum(h) * dx
    h = h * (target_sum / h_sum)
    
    # Optimization loop
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)[1]
    
    # Local search with random perturbations
    max_iter = 100000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make a small random change
        h_new = h.copy()
        idx1 = random.randint(0, n_points - 1)
        idx2 = random.randint(0, n_points - 1)
        delta = random.uniform(-0.1, 0.1)
        
        # Apply change
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to maintain sum constraint
        h_sum = np.sum(h_new) * dx
        h_new = h_new * (target_sum / h_sum)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new)[1]
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h):
    h = np.array(h)
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full') * dx
    c5 = max(conv)
    return (h, c5)

# Initial value for comparison
initial_h_values = [0.5] * 200 # Default uniform

# Function to handle the final optimization
def optimize_h():
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h_values)
    h_sum = np.sum(h) * dx
    h = h * (target_sum / h_sum)
    
    # Local search improvement
    max_iter = 50000
    best_c5 = evaluate_erdos_solution(h)[1]
    best_h = h.copy()

    for i in range(max_iter):
        # Another small random change
        h_new = h.copy()
        idx1 = random.randint(0, n_points - 1)
        idx2 = random.randint(0, n_points - 1)
        delta = random.uniform(-0.05, 0.05)
        
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new) * dx
        h_new = h_new * (target_sum / h_sum)
        
        new_c5 = evaluate_erdos_solution(h_new)[1]
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new.copy()

    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
