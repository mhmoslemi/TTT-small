import numpy as np
import time
import random

def evaluate_h(h):
    """Evaluate the maximum overlap integral for given h."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode='full')
    return np.max(O_counts) * dx

def max_overlap_counts(h):
    """Return the maximum overlap count."""
    O_counts = np.correlate(h, 1 - h, mode='full')
    return np.max(O_counts)

def random_balanced_binary(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    ones = rng.sample(range(n), n // 2)
    h[ones] = 1
    return h

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto box [0,1] with sum constraint."""
    v = np.array(v)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def binary_anneal(n, time_limit, rng):
    """Perform simulated annealing on binary vector."""
    h = random_balanced_binary(n, rng)
    cur_val = max_overlap_counts(h)
    best_val = cur_val
    best_h = h.copy()
    start_time = time.time()
    max_iters = 100000
    target_max = int(np.floor(0.3807 * n / 2))
    
    for iter in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
            
        T = 0.5 * (1 - elapsed / time_limit) + 0.001
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        h[p], h[q] = 0, 1
        new_val = max_overlap_counts(h)
        delta = new_val - cur_val
        
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
                
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val <= target_max:
                    break
        else:
            h[p], h[q] = 1, 0
            
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    rng = random.Random(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    n_values = [30, 40, 45, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    
    for n in n_values:
        if time.time() - start_time > budget_s - 10:
            break
            
        time_per_n = min(100, (budget_s - (time.time() - start_time)) / (len(n_values) - n_values.index(n)))
        if time_per_n < 1:
            continue
            
        h, val = binary_anneal(n, time_per_n, rng)
        c5 = val * 2.0 / n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
            
        if best_c5 <= 0.379:
            break
    
    # Do final refinement with continuous method if needed
    if best_h is not None and best_c5 > 0.379:
        try:
            best_h_float = best_h.astype(float)
            best_h_float = project_onto_box_and_sum(best_h_float, best_n / 2)
            h_refined, val = binary_anneal(best_n, 50, rng)
            c5_refined = val * 2.0 / best_n
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined
        except:
            pass
    
    return (best_h, best_c5, best_n)
