import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    return max_val * dx

def evaluate_integer(a):
    n = len(a)
    corr = correlate(a, 1 - a, mode='full')
    return corr.max()

def compute_mismatch_sets(h, k):
    n = len(h)
    M1 = []
    M0 = []
    if k >= 0:
        for i in range(n - k):
            if h[i] == 1 and h[i + k] == 0:
                M1.append(i)
            elif h[i] == 0 and h[i + k] == 1:
                M0.append(i)
    else:
        k = -k
        for i in range(n - k):
            if h[i] == 1 and h[i + k] == 0:
                M1.append(i)
            elif h[i] == 0 and h[i + k] == 1:
                M0.append(i)
    return M1, M0

def binary_descent_step(h, k, M1, M0):
    if not M1 or not M0:
        i = np.random.choice(np.where(h == 1)[0])
        j = np.random.choice(np.where(h == 0)[0])
        return i, j
    i = np.random.choice(M1)
    j = np.random.choice(M0)
    return i, j

def binary_descent(a, max_iters=2000):
    n = len(a)
    best_a = a.copy()
    best_val = evaluate_integer(best_a)
    for _ in range(max_iters):
        corr = correlate(best_a, 1 - best_a, mode='full')
        max_k = corr.argmax() - (n - 1)
        max_val = corr.max()
        if max_val < best_val:
            best_val = max_val
            best_a = best_a.copy()
        M1, M0 = compute_mismatch_sets(best_a, max_k)
        if not M1 or not M0:
            i = np.random.choice(np.where(best_a == 1)[0])
            j = np.random.choice(np.where(best_a == 0)[0])
        else:
            i = np.random.choice(M1)
            j = np.random.choice(M0)
        best_a[i], best_a[j] = best_a[j], best_a[i]
        corr_new = correlate(best_a, 1 - best_a, mode='full')
        if corr_new.max() <= max_val:
            continue
        else:
            best_a[i], best_a[j] = best_a[j], best_a[i]
    return best_a, best_val

def simulated_annealing(a, max_iters=5000):
    n = len(a)
    best_a = a.copy()
    best_val = evaluate_integer(best_a)
    T0 = 0.5
    Tf = 1e-4
    T = T0
    decay = (Tf / T0) ** (1.0 / max_iters)
    for _ in range(max_iters):
        i = np.random.choice(np.where(best_a == 1)[0])
        j = np.random.choice(np.where(best_a == 0)[0])
        old_val = evaluate_integer(best_a)
        best_a[i], best_a[j] = best_a[j], best_a[i]
        new_val = evaluate_integer(best_a)
        delta = new_val - old_val
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            if new_val < best_val:
                best_val = new_val
                best_a = best_a.copy()
        else:
            best_a[i], best_a[j] = best_a[j], best_a[i]
        T *= decay
    return best_a, best_val

def binary_search(n, time_budget):
    start_time = time.time()
    best_h = None
    best_val = float('inf')
    num_restarts = 0
    while time.time() - start_time < time_budget and num_restarts < 30:
        a = np.zeros(n, dtype=int)
        ones = np.random.choice(n, size=n // 2, replace=False)
        a[ones] = 1
        a, val = binary_descent(a, 2 * n)
        if time.time() - start_time > time_budget - 1:
            break
        a, val = simulated_annealing(a, 5 * n * int(np.log(n)))
        if val < best_val:
            best_val = val
            best_h = a.copy()
        num_restarts += 1
    if best_h is None:
        best_h = np.zeros(n, dtype=int)
        ones = np.random.choice(n, size=n // 2, replace=False)
        best_h[ones] = 1
        best_h, best_val = binary_descent(best_h, 2 * n)
    return best_h, best_val

def softmax_projection(h, target_sum):
    def waterfill(v, target_sum):
        v = np.maximum(v, 0)
        v = np.minimum(v, 1)
        lmin = v.min() - 1
        lmax = v.max() + 1
        for _ in range(40):
            lam = (lmin + lmax) / 2
            z = np.clip(v - lam, 0, 1)
            s = np.sum(z)
            if s > target_sum:
                lmin = lam
            else:
                lmax = lam
        return np.clip(v - lmax, 0, 1)
    return waterfill(h, target_sum)

def continuous_refine(h_start, time_budget):
    start_time = time.time()
    best_h = h_start.copy()
    n = len(best_h)
    target_sum = n / 2.0
    best_val = evaluate(best_h)
    for alpha in [1, 5, 10, 20, 40, 80]:
        if time.time() - start_time > time_budget - 5:
            break
        t_steps = 200
        eta = 0.2
        h = best_h.copy()
        for t in range(t_steps):
            if time.time() - start_time > time_budget - 5:
                break
            corr = correlate(h, 1 - h, mode='full')
            shift_vals = corr.copy()
            max_o = shift_vals.max()
            shift_vals = shift_vals - max_o
            exps = np.exp(shift_vals * alpha)
            p = exps / exps.sum()
            g = np.zeros_like(h)
            for k in range(-n + 1, n):
                if k >= 0:
                    g += p[k + n - 1] * (1 - np.r_[h[k:], np.zeros(k)]) * (h[:-k] if k else h[:])
                else:
                    k = -k
                    g += p[k + n - 1] * (1 - np.r_[np.zeros(k), h[:-k]]) * (h[k:] if k else h[:])
            h += eta / np.sqrt(t + 1) * g
            h = softmax_projection(h, target_sum)
            val = evaluate(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    return best_h, best_val
