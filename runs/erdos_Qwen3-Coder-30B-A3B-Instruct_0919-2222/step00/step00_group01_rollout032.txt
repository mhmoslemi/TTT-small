import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap problem constant C5 for given h."""
    # Ensure h is a numpy array
    h = np.array(h)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation between h and (1-h)
    c = correlate(h, 1 - h, mode='full')
    # Compute C5 as maximum of correlation scaled by dx
    c5 = max(c) * dx
    return c5

def initial_h_values():

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    end_time = start_time + budget_s * 0.99  # Leave margin

    # Parameters
    n_points = 501  # Odd number for symmetry
    dx = 2.0 / n_points
    
    # Create initial shape (Start with triangle shape)
    x = np.arange(n_points) * dx + dx/2  # Midpoint sampling
    h = np.maximum(0, 1 - np.abs(x - 1))  # Triangle shape
    
    # Normalize
    sum_h = np.sum(h)
    h = h * (n_points/2) / sum_h
    
    # Initial C5
    best_c5 = evaluate_erdos_solution(h)
    
    # Keep track of best
    best_h = h.copy()

    # Hill climbing parameters
    iterations = 0
    max_iter = 30000
    
    while iterations < max_iter and time.time() < end_time:
        # Random move
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Determine change direction
        if i == j:
            continue
            
        # Determine amount to move (based on relative values)
        hi = h[i]
        hj = h[j]
        delta_max = min(hi, 1 - hj, 0.1)  # Up to 0.1
        delta = np.random.random() * delta_max
        
        # New values
        new_h = h.copy()
        new_h[i] = hi - delta
        new_h[j] = hj + delta
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = new_h
            best_c5 = new_c5
            best_h = new_h.copy()
        
        iterations += 1
        
        # Occasionally increase iteration limit a bit to not incur timeout
        if iterations % 2000 == 0 and time.time() > end_time - 10:
            break

    # Final Evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
