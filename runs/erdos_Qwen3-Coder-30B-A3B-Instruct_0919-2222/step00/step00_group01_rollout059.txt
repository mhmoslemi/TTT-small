import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation of h with (1-h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the overlap integral
    max_overlap = np.max(correlation) * dx
    return max_overlap

def initial_h_values():
    """Initial construction for h(x) on [0,2] with n=47 points."""
    n = 47
    h = np.zeros(n)
    dx = 2.0 / n
    # Start with a simple construction
    for i in range(n):
        x = i * dx
        if 0 <= x < 1:
            h[i] = 1.0
        elif 1 <= x < 2:
            h[i] = 0.0
    # Normalize to sum to 1
    h = h * (1.0 / np.sum(h) * 2.0 / dx)
    return h

def run(seed=42, budget_s=600):
    """Run optimization to find a better h to minimize C5."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up the optimization
    n_points = 47  # Use the provided initial points
    dx = 2.0 / n_points
    
    # Initialize with the given construction
    h = initial_h_values()
    h = h.copy()
    
    # Normalize to sum to 1
    h = h * (2.0 / (np.sum(h) * dx))
    
    # Function to minimize C5
    def objective(h_vec):
        h = h_vec.copy()
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        h = h * (2.0 / (np.sum(h) * dx))  # Normalize
        return evaluate_erdos_solution(h)
    
    # Constraint for integral = 1
    def constraint_integral(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    # Bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Create constraint dictionary
    constraint = {'type': 'eq', 'fun': constraint_integral}
    
    # Initial guess
    h0 = h.copy()
    
    # Optimization with gradient-based method
    options = {'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
    
    # Use scipy.optimize.minimize
    result = minimize(objective, h0, method='SLSQP', bounds=bounds, 
                      constraints=constraint, options=options)
    
    # Extract best solution and ensure it meets constraint
    h_opt = result.x.copy()
    h_opt = np.clip(h_opt, 0, 1)
    
    # Normalize to integral 1
    h_opt = h_opt * (2.0 / (np.sum(h_opt) * dx))
    
    # Compute final C5 bound
    final_c5 = evaluate_erdos_solution(h_opt)
    
    # Try several local optimizations with random restarts
    best_h = h_opt.copy()
    best_c5 = final_c5
    
    num_restarts = 20
    for i in range(num_restarts):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Random perturbation of solution
        perturb = np.random.normal(0, 0.01, n_points)
        h_pert = h_opt + perturb
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (2.0 / (np.sum(h_pert) * dx))  # Normalize
        
        # Run local optimization from perturbed solution
        result_pert = minimize(objective, h_pert, method='SLSQP', bounds=bounds, 
                               constraints=constraint, options=options)
        
        h_pert_result = result_pert.x.copy()
        h_pert_result = np.clip(h_pert_result, 0, 1)
        h_pert_result = h_pert_result * (2.0 / (np.sum(h_pert_result) * dx))
        c5_pert = evaluate_erdos_solution(h_pert_result)
        
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert_result.copy()
    
    # Return best solution found
    return (best_h.tolist(), best_c5, n_points)
