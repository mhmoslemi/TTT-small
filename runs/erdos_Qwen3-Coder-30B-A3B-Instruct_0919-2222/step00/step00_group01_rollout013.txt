import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    # Evaluate the overlap integral using the provided method
    # np.correlate with full mode, then multiply by dx
    n_points = len(h)
    dx = 2.0 / n_points
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    end_time = start_time + budget_s * 0.99  # Leave margin
    
    # Use reasonable-sized number of points
    n_points = 501  # Odd number for better domain symmetry
    dx = 2.0 / n_points
    
    # Initialize with a reference shape (cosine^2)
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2  # Midpoints for better accuracy
    
    # Initialize with a cos^2 shape
    h = np.cos(np.pi * (x - 1) / 2) ** 2
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum  # Normalize to sum = n_points/2
    
    # Ensure all values in [0,1]
    h = np.clip(h, 0, 1)
    
    current_best_h = h.copy()
    current_best_c5 = evaluate_erdos_solution(h)
    
    # Algorithm: Hill climbing with simulated annealing
    # Start with high temperature
    temp = 0.02
    min_temp = 1e-6
    cooling_rate = 0.99995
    
    # Initial steps to find a good shape
    steps = 0
    max_steps = 300000  # Limit iterations but within time budget
    step_limit = max_steps
    while time.time() < end_time and steps < step_limit:
        # Randomly select indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        if i == j:
            continue
            
        # Compute maximum move allowed to stay in [0,1]
        max_move = min(h[i], 1 - h[j])
        if max_move <= 0:
            continue
            
        # Move a fraction of max_move with randomness
        delta = np.random.random() * min(0.1, max_move)
        
        # Check if moving hi to hj will decrease the overlap integral
        # We compute approximate change to correlation at shifts involving i,j
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Ensure within bounds
        h_new = np.clip(h_new, 0, 1)
        
        if np.sum(h_new) > n_points / 2 + 1e-8 or np.sum(h_new) < n_points / 2 - 1e-8:
            # Rescale to keep sum exactly n_points/2
            h_new *= (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
        
        new_c5 = evaluate_erdos_solution(h_new)
        
        # Accept if better or with temperature probability
        if new_c5 < current_best_c5:
            h = h_new
            current_best_h = h.copy()
            current_best_c5 = new_c5
            # Reset temperature if improvement
            temp = 0.02
        else:
            # Accept with probability based on temperature
            if np.random.random() < np.exp(-(new_c5 - current_best_c5) / temp):
                h = h_new
                current_best_h = h.copy()
                current_best_c5 = new_c5
                temp = 0.02
        
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
        steps += 1
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(current_best_h)
    return (current_best_h.tolist(), final_c5, n_points)
