def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)

    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum of these integrals is our C_5
    c5 = np.max(conv) * dx
    return c5
