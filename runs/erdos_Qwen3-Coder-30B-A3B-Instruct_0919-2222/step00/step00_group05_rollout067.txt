import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial values
    global initial_h_values
    try:
        h_init = np.array(initial_h_values)
    except:
        h_init = np.array([0.5]*70)
    n_points = len(h_init)
    
    # Ensure correct mass
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    y = cp.Variable((n_points, 2*n_points-1))  # y[i, s] where s = -n+1 to n-1

    # Constraints for mass
    mass_constraint = cp.sum(h) == n_points/2
    bounds = [0 <= h, h <= 1]
    
    # McCormick envelopes for y[i, s]
    constraints = bounds + [t >= 0]
    for i in range(n_points):
        for s in range(1-n_points, n_points):
            if 0 <= i+s < n_points:
                y_idx = (i, s + n_points - 1)  # Adjust s to be 0-based for indexing
                constraints += [
                    y[y_idx] <= h[i],
                    y[y_idx] <= h[i+s],
                    y[y_idx] >= h[i] + h[i+s] - 1,
                    y[y_idx] >= 0
                ]

    # Compute correlations for all shifts and enforce the inequality
    for s in range(1-n_points, n_points):
        if s == 0:
            continue  # skip 0 correlation
        corr_sum = cp.sum([y[(i, s + n_points - 1)] for i in range(n_points) if 0 <= i+s < n_points])
        constraints += [corr_sum >= n_points/2 - t]
    
    # Objective
    objective = cp.Minimize(t)
    
    # Formulate and solve LP
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000)
        h_star = h.value
        if h_star is None:
            # Fallback to initial solution
            h_star = h_init
    except:
        h_star = h_init
    
    # Rounding strategies
    candidates = []
    
    # 1. Binary exact if possible
    h_rounded = np.round(h_star)
    if np.allclose(h_rounded, h_star) or np.all(np.abs(h_rounded - h_star) < 1e-8):
        # It's binary already or very close to it. Evaluate directly.
        candidates.append((h_rounded, evaluate_erdos_solution(h_rounded)))
    else:
        # Deterministic rounding - choose the indices of largest values to set to 1
        num_ones = int(n_points / 2)
        sorted_indices = np.argsort(-h_star)
        h_det = np.zeros(n_points)
        h_det[sorted_indices[:num_ones]] = 1
        candidates.append((h_det, evaluate_erdos_solution(h_det)))
        
        # Randomized rounding - sample multiple times
        num_samples = 200
        for _ in range(num_samples):
            idx = np.random.choice(n_points, size=num_ones, replace=False, p=h_star / np.sum(h_star))
            h_rand = np.zeros(n_points)
            h_rand[idx] = 1
            candidates.append((h_rand, evaluate_erdos_solution(h_rand)))
    
    # Greedy polishing
    def polish_vector(h_vec):
        if time.time() - start_time > budget_s * 0.95:
            return h_vec, evaluate_erdos_solution(h_vec)
        corr_arr = correlate(h_vec, 1 - h_vec, mode='full')
        max_corr_idx = np.argmax(corr_arr)
        corr_val = corr_arr[max_corr_idx]
        max_lag = max_corr_idx - (len(h_vec) - 1)
        
        # Initial guess for corr
        best_h = h_vec.copy()
        best_val = corr_val
        counter = 0
        num_iterations = 5000  # Prevent infinite loop
        while counter < num_iterations and time.time() - start_time < budget_s * 0.9:
            updated = False
            # Scan for both source and target indexes that can improve.
            for i in range(len(h_vec)):
                if h_vec[i] == 1:
                    # Find a mismatched element for this move
                    j = (i + max_lag) % len(h_vec)
                    if h_vec[j] == 0:
                        # Move to improve by swapping these
                        # Find target
                        for k in range(len(h_vec)):
                            if k != i and h_vec[k] == 0:
                                # Swapping
                                new_h = h_vec.copy()
                                new_h[i] = 0
                                new_h[k] = 1
                                # Recalculate specific correlation
                                temp_corr = correlate(new_h, 1 - new_h, mode='full')
                                new_corr = temp_corr[max_corr_idx]  # Reform from max_lag
                                if new_corr < best_val:
                                    best_val = new_corr
                                    best_h = new_h.copy()
                                    updated = True
                                    break
            if not updated:
                break
            counter += 1
        return best_h, evaluate_erdos_solution(best_h)
    
    # Evaluate candidates
    best_candidate = None
    best_c5 = np.inf
    for h_cand, c5_val in candidates:
        polished_h, polished_c5 = polish_vector(h_cand)
        if polished_c5 < best_c5:
            best_c5 = polished_c5
            best_candidate = polished_h
    
    # Try to solve as MIP if we're far from target and the LP step was fractional
    if best_c5 > 0.381 and not np.allclose(h_star, np.round(h_star)) and time.time() - start_time < budget_s * 0.8:
        # Solve MIP
        h_int = cp.Variable(n_points, integer=True)
        t_int = cp.Variable()
        y_int = cp.Variable((n_points, 2*n_points-1))

        # Constraints
        mass_constraint_int = cp.sum(h_int) == n_points/2
        bounds_int = [0 <= h_int, h_int <= 1]
        
        constraints_int = bounds_int + [t_int >= 0]
        for i in range(n_points):
            for s in range(1-n_points, n_points):
                if 0 <= i+s < n_points:
                    y_idx = (i, s + n_points - 1)
                    constraints_int += [
                        y_int[y_idx] <= h_int[i],
                        y_int[y_idx] <= h_int[i+s],
                        y_int[y_idx] >= h_int[i] + h_int[i+s] - 1,
                        y_int[y_idx] >= 0
                    ]

        for s in range(1-n_points, n_points):
            if s == 0:
                continue
            corr_sum_int = cp.sum([y_int[(i, s + n_points - 1)] for i in range(n_points) if 0 <= i+s < n_points])
            constraints_int += [corr_sum_int >= n_points/2 - t_int]
        
        # Objective
        objective_mip = cp.Minimize(t_int)
        prob_mip = cp.Problem(objective_mip, constraints_int)
        
        try:
            prob_mip.solve(solver=cp.CBC, verbose=False, max_time=250)
            h_mip = h_int.value
            if h_mip is not None:
                # Polish MIP integer solution
                h_mip, c5_mip = polish_vector(h_mip)
                if c5_mip < best_c5:
                    best_c5 = c5_mip
                    best_candidate = h_mip
        except:
            pass
    
    return (best_candidate.tolist(), float(best_c5), n_points)
