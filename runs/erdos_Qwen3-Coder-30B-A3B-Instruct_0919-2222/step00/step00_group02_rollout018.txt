import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from math import floor

def evaluate_erdos_solution(h, n_points=45):
    # Compute the overlap integral for the given step function h
    dx = 2.0 / n_points
    # Correlate h with (1-h) using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value and multiply by dx
    C5 = np.max(correlation) * dx
    return C5

def initial_h_values(n_points=45):
    # Initial construction from the paper
    h = np.zeros(n_points)
    # The construction is a piecewise constant function
    # We define intervals: [0, 0.5), [0.5, 1.5), [1.5, 2]
    # Interval lengths: 0.5, 1.0, 0.5
    # So we take h=1 on [0.5, 1.5] which has length 1.0
    # and h=0.5 on [0, 0.5) and [1.5, 2] which have combined length 1.0
    # Let the assignments be:
    # h = 1 on [0.5, 1.5] (n=22 points)
    # h = 0.5 on [0, 0.5) and [1.5, 2] (n=11 points on each interval)
    # But we want sum(h) * dx = 1, so sum(h) = n_points / 2 = 22.5
    # We have 22*1 + 11*0.5 + 11*0.5 = 22 + 5.5 + 5.5 = 33
    # So we normalize: h[i] = values[i] / 33 * 22.5
    dx = 2.0 / n_points
    # Calculate indices mapping to intervals
    mid_start = int(0.5 / dx)
    mid_end = int(1.5 / dx)
    left_end = int(0.5 / dx)
    right_start = int(1.5 / dx)
    
    # Manual assignment of values
    h = np.zeros(n_points)
    # Leave h = 0 on [0, 0.5) - 9 points  
    # Assign h = 0.5 on [0.5, 1.5) which is (1.0/dx) = 22.5 points -> 22 points
    # Assign h = 1 on [1.5, 2] which is 9 points
    h[9:31] = 0.5  # indices 9 through 30  
    h[31:] = 1.0   # 31 to end
    
    # Now normalize so that sum(h)*dx = 1
    normalization_factor = np.sum(h) * dx / 1.0
    h /= normalization_factor
    
    return h

def objective(h_values, n_points=45):
    # Make sure h_values respect bounds
    h = np.array(h_values)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Ensure the integral constraint
    dx = 2.0 / n_points
    integral = np.sum(h) * dx
    # If we want integral = 1, so we use the actual constraint:
    # We don't actually adjust h_values unless they're out of bounds,
    # Instead we use the direct function (we can also adjust h to make sum = 1)
    h = h * (1.0 / integral) if integral != 0 else h
    return evaluate_erdos_solution(h, n_points)

def constraint_func(h_values):
    # Constraint that sum(h)*dx = 1 (equivalent to sum(h) = n_points/2)
    n_points = 45
    dx = 2.0 / n_points
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    
    # Start with the initial construction
    h_current = initial_h_values(n_points)
    
    # Define bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint to ensure the integral equals 1.0
    constraint_eq = {'type': 'eq', 'fun': constraint_func}
    
    best_C5 = evaluate_erdos_solution(h_current, n_points)
    best_h = h_current.copy()
    
    # We run differential evolution with restarts to explore better solutions
    # Best for finding a decent local minimum with global search
    
    # Optimization with different starting points
    iterations = 0
    while time.time() - start_time < budget_s - 1:  # Keep 1 second buffer
        try:
            # Run differential evolution
            res = differential_evolution(
                lambda x: objective(x, n_points), 
                bounds, 
                maxiter=100,
                popsize=15,
                strategy='best1bin',
                seed=seed + iterations,
                disp=False,
                tol=1e-6
            )
            if not res.success:
                break
            
            h_opt = np.clip(res.x, 0, 1)  # Ensure bounds
            # Normalize to sum(h)*dx = 1
            integral = np.sum(h_opt) * dx
            if integral != 0:
                h_opt = h_opt * (1.0 / integral)
                
            C5_opt = evaluate_erdos_solution(h_opt, n_points)
            
            if C5_opt < best_C5:
                best_C5 = C5_opt
                best_h = h_opt.copy()
                
        except Exception as e:
            # Silently fail if any error occurs
            pass
        
        iterations += 1
        
        # Early termination if we've gone too long
        if time.time() - start_time > budget_s - 2:
            break
        
        if best_C5 <= 0.379:
            break
    
    # Ensure final h has correct normalization
    integral = np.sum(best_h) * dx
    if integral != 0:
        best_h = best_h * (1.0 / integral)
    
    return (best_h.tolist(), best_C5, n_points)
