import numpy as np
import time
from scipy.special import softmax

def project_to_capped_simplex(v, M):
    """Project vector v onto capped simplex: 0 <= h_i <= 1 and sum h_i = M."""
    u = np.clip(v, 0, 1)
    s = np.sum(u)
    if abs(s - M) < 1e-10:
        return u
    elif s > M:
        # Sort in descending order to find threshold
        sorted_u = np.sort(u)[::-1]
        cumsum = np.cumsum(sorted_u)
        # Find index where threshold is located
        indices = np.arange(len(sorted_u)) + 1
        threshold = (cumsum[-1] - M) / len(sorted_u)
        # Note: This logic needs slight refinement; using a more direct binary search
        # or iterative method works better numerically.  More careful approach:
        # Try to find a theta such that sum(min(1, max(0, v[i] - theta))) = M
        # Use simple bisection method
        low, high = 0.0, np.max(v)
        tol = 1e-10
        for _ in range(100):
            theta = (low + high) / 2
            h = np.clip(v - theta, 0, 1)
            if np.sum(h) > M:
                low = theta
            else:
                high = theta
            if abs(np.sum(h) - M) < 1e-8:
                break
        return np.clip(v - low, 0, 1)
    else:
        # s < M
        low, high = 0.0, np.min(v) + 1.0
        tol = 1e-10
        for _ in range(100):
            theta = (low + high) / 2
            h = np.clip(v + theta, 0, 1)
            if np.sum(h) < M:
                high = theta
            else:
                low = theta
            if abs(np.sum(h) - M) < 1e-8:
                break
        return np.clip(v + low, 0, 1)

def correlations(h, shift_maps):
    """Compute all correlation values C_s(h) = sum_i h_i * (1 - h_{i+s}) for valid shifts."""
    n = len(h)
    C = np.zeros(2*n - 1)
    for s in range(-n+1, n):
        val = 0.0
        for i, j in shift_maps[s + n - 1]:
            val += h[i] * (1 - h[j])
        C[s + n - 1] = val
    return C

def smooth_obj_and_grad(h, alpha, shift_maps):
    """Compute smooth approximation Φ_α(h) and its gradient."""
    C_vals = correlations(h, shift_maps)
    max_C = np.max(C_vals)
    if max_C > 100:  # Avoid overflow
        exp_C = np.exp(alpha * (C_vals - max_C))
    else:
        exp_C = np.exp(alpha * C_vals)
    w = exp_C / np.sum(exp_C)
    
    # Now compute gradient ∂Φ_α / ∂h_i = sum_s w_s * (1 - h_{i+s} - h_{i-s})
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for s in range(-len(h)+1, len(h)):
            if s == 0:
                continue
            j = i + s
            if 0 <= j < len(h):
                grad[i] += w[s + len(h) - 1] * (-h[j])  # -h_{i+s}
            # Check i-s (backwards shift)
            k = i - s
            if 0 <= k < len(h):
                grad[i] += w[s + len(h) - 1] * (-h[k])  # -h_{i-s}
                
    # Add the 1 term:
    for i in range(len(h)):
        for s in range(-len(h)+1, len(h)):
            if s == 0:
                continue
            j = i + s
            if 0 <= j < len(h):
                grad[i] += w[s + len(h) - 1]
    
    # Final objective:
    obj_val = (1./alpha) * np.log(np.sum(exp_C))
    return obj_val, grad

def run_descent(h0, alpha_schedule, eta0, max_iter_per_alpha, shift_maps, time_budget):
    """Perform gradient descent for a given α schedule."""
    start_time = time.time()
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    eta = eta0
    
    for j, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_budget * 0.97:
            break
        h_prev = h.copy()
        for _ in range(max_iter_per_alpha):
            if time.time() - start_time > time_budget * 0.97:
                break
            obj_val, grad = smooth_obj_and_grad(h, alpha, shift_maps)
            v = h - eta * grad
            h = project_to_capped_simplex(v, len(h)/2)
            # Evaluate true C5
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        eta *= 0.9
    return best_h, best_c5

def evaluate(h):
    """Use numpy correlate to compute actual max correlation."""
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c)

def generate_random_binary(n, M):
    """Return random binary vector with exact M ones."""
    idx = np.random.permutation(n)[:M]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def generate_random_fractional(n, M):
    """Return vector uniformly drawn from Dirichlet(1.0, ..., 1.0) and projected."""
    alpha = np.ones(n)
    sample = np.random.dirichlet(alpha)
    return project_to_capped_simplex(sample, M)

def swap_local_search(h_bin, max_iter=1000):
    """Improve binary h by swapping elements that reduce worst correlation."""
    c_orig = np.correlate(h_bin, 1 - h_bin, mode='full')
    s_max = np.argmax(c_orig)
    c_max = c_orig[s_max]
    
    for _ in range(max_iter):
        # Find worst shift
        s_max = np.argmax(c_orig) - (len(h_bin) - 1)
        if c_max <= 0.3807:
            break
        # Try swapping some bits that improve that shift
        neighbors = []
        # Find candidate indices
        for i in range(len(h_bin)):
            if h_bin[i] == 1:
                for j in range(len(h_bin)):
                    if h_bin[j] == 0 and h_bin[(j - s_max) % len(h_bin)] == 1:
                        neighbors.append((i, j))
                        break
        if not neighbors:
            break
        # Pick a random improving swap from candidates
        k = np.random.randint(len(neighbors))
        i, j = neighbors[k]
        if (i - s_max) % len(h_bin) >= 0 and (i - s_max) % len(h_bin) < len(h_bin):
            # Swap
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            c_orig = np.correlate(h_bin, 1 - h_bin, mode='full')
            c_max = np.max(c_orig)
        else:
            continue
    return h_bin
