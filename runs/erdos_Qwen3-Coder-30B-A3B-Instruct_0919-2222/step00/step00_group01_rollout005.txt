def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for a given h"""
    # Compute correlation of h with (1-h)
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value and scale by dx
    max_correlation = np.max(correlation)
    return max_correlation * dx
