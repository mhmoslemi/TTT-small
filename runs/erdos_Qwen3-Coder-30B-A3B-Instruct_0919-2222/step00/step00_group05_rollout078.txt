import numpy as np
import time
from scipy.optimize import minimize
import random


def evaluate(h):
    """Evaluate the actual C5 value using numpy correlate"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def project_to_capped_simplex(v, M):
    """Project a vector v onto the capped simplex {x: 0 <= x_i <= 1, sum x_i = M}"""
    n = len(v)
    # Clip to [0, 1]
    u = np.clip(v, 0, 1)
    s = np.sum(u)
    if abs(s - M) < 1e-10:
        return u
    # Sort v in descending order
    idx_sorted = np.argsort(v)[::-1]
    # Compute cumulative sums
    u_sorted = u[idx_sorted]
    cumsum = np.cumsum(u_sorted)
    # Find threshold
    theta = 0
    for i in range(n):
        if cumsum[i] <= M - (n - i - 1):
            continue
        else:
            theta = (cumsum[i] - M) / (i + 1)
            break
    else:
        theta = (cumsum[n - 1] - M) / n
    # Apply threshold
    h = np.zeros_like(v)
    for i in range(n):
        h[idx_sorted[i]] = max(0, min(1, v[idx_sorted[i]] - theta))
    return h


def precompute_shifts(n):
    """Precompute valid index pairs for each shift"""
    shift_maps = {}
    for s in range(-n + 1, n):
        indices = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                indices.append((i, j))
        shift_maps[s] = indices
    return shift_maps


def correlations(h, shift_maps):
    """Compute all correlation values C_s(h) for given shift maps"""
    n = len(h)
    cs = np.zeros(2 * n - 1)
    for s, pairs in shift_maps.items():
        c = 0
        for i, j in pairs:
            c += h[i] * (1 - h[j])
        # store at correct index in full array
        cs[s + n - 1] = c
    return cs


def smooth_obj_and_grad(h, alpha, shift_maps):
    """Compute the smooth surrogate and its gradient"""
    cs = correlations(h, shift_maps)
    # Get max of correlations for numerical stability
    max_c = np.max(cs)
    if max_c == -np.inf:
        # If all are -inf then all are 0
        exp_cs = np.ones_like(cs)
    else:
        # Exponentiate
        exp_cs = np.exp(alpha * (cs - max_c))
    sum_exp = np.sum(exp_cs)
    # Soft max weights
    weights = exp_cs / sum_exp
    # Compute gradient
    grad = np.zeros_like(h)
    for s, pairs in shift_maps.items():
        for i, j in pairs:
            # ∂C_s/∂h_i = 1 - h_{i+s} - h_{i-s}
            # s = j - i so i + s = j, and i - s = i - (j-i) = 2*i - j
            grad[i] += weights[s + (len(shift_maps) - 1)] * (1 - h[j] - h[i - s])
    # The gradient of Φ_α = (1/α) * log sum
    # That's weighted gradient of log sum = weighted gradient of sum / sum = weighted gradient
    # Let's be more careful:
    # d/dh_i log sum = (1/sum * sum_j df_j/dh_i ) / (sum) = (1/sum) sum_j (df_j/dh_i) * exp(f_j) / sum
    # This is not easy to compute by hand, so we compute full d/dh_i logic properly.
    # Φ_α(h) = (1/α) * log sum exp(α C_s(h))
    # ∂Φ_α / ∂ h_i = (1/α) * (1/sum exp) * sum_s ∂ exp(α C_s)/∂ h_i
    # = (1/α) * (1/sum exp) * α * exp(α C_s) * ∂C_s/h_i = (1/sum exp) * sum_s exp(α C_s) * ∂C_s/∂h_i
    # = sum_s w_s * ∂C_s/∂h_i
    # ∂C_s/∂h_i = 1 - h_{i+s} - h_{i-s} if both indices are in [0,n)
    # i+s and i-s indices [-n, 2*n-2] skip out-of-range (which contribute 0)
    # Now, for each s, for indices i,j in pairs, we add w_s to i's gradient and -w_s * h_j in case i=j
    # No, we're double counting.
    # For each contribution ∂C_s/∂h_i = 1 - h_{i+s} - h_{i-s}
    # That means for i we look at all s such that i+s is valid index. We get - w_s * h_{i+s}.
    # Also from the term 1 we contribute w_s
    # - w_s * h_{i-s} if  i-s is valid
    # We can compute total gradient in one loop like:
    grad = np.zeros_like(h)
    for s, pairs in shift_maps.items():
        for i, j in pairs:
            grad[i] += weights[s + n - 1]  # because full size is 2*n-1 but index for shift s is at s+n-1
            # h_{i+s}, and h_{i-s}
            # s = j - i => i + s = j. 
            # But we want i - s = i - (j - i) = 2*i - j
    # Reset grad
    grad = np.zeros_like(h)
    for s, pairs in shift_maps.items():
        w_s = weights[s + n - 1]
        for i, j in pairs:
            grad[i] += w_s
            if i + s >= 0 and i + s < n:
                grad[i] -= w_s * h[i + s]
            if i - s >= 0 and i - s < n:
                grad[i] -= w_s * h[i - s]
    return np.log(sum_exp) / alpha, grad


def descent_one_alpha(h0, alpha, eta0, max_iter, shift_maps, time_limit):
    """Run projected gradient descent at fixed alpha; early return if full time is used"""
    h = h0.copy()
    eta = eta0
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > time_limit:
            return h
        phi_val, g = smooth_obj_and_grad(h, alpha, shift_maps)
        v = h - eta * g
        h = project_to_capped_simplex(v, len(h) / 2)
        # Optionally adapt step size
        # Check if smooth value decreases - for now, just decrease eta if not enough improvement
    return h


def run_descent(h0, alpha_schedule, eta0, max_iter, shift_maps, time_limit):
    """Run the entire continuation"""
    h = h0.copy()
    start_time = time.time()
    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit:
            break
        time_slice = 0.9 * time_limit * 0.1  # Just a slice of time for this alpha phase
        h = descent_one_alpha(h, alpha, eta0, max_iter, shift_maps, time_slice)
    return h


def random_binary(n, M):
    """Generate a random binary vector with exactly M ones"""
    positions = random.sample(range(n), M)
    h = np.zeros(n)
    for pos in positions:
        h[pos] = 1
    return h


def random_fractional(n, M):
    """Generate a random fractional vector with exact mass M"""
    h = np.random.dirichlet(np.ones(n))
    # Now project to capped simplex
    return project_to_capped_simplex(h, M)


def swap_local_search(h_bin, time_limit):
    """Perform targeted swaps to improve discrete C5"""
    n = len(h_bin)
    # Evaluate current C5
    corr = np.correlate(h_bin, 1 - h_bin, mode="full")
    s_max = np.argmax(corr)
    max_c = corr[s_max]
    # For shift s_max, find locations where correlation is large
    # Find an index i such that h_bin[i] = 1 and h_bin[i+s_max] = 0
    # Consider the starting point at indices such that both i and i+s_max are in bounds
    # Find the index i that contributes to this value - but we did not compute the actual C_s
    # We simulate: C_s(h) = sum of h_i * (1 - h_{i+s})
    # Find one i such that h_i = 1 and h_{i+s} = 0 that contributes to max_c  
    start_time = time.time()
    max_iter = 10000
    iter_count = 0
    last_improvement = time.time()
    while time.time() - start_time < time_limit and iter_count < max_iter and time.time() - last_improvement < 10:
        # Compute s_max
        corr = np.correlate(h_bin, 1 - h_bin, mode="full")
        s_max = np.argmax(corr)
        max_c = corr[s_max]
        # Find an improving swap
        improvement = False
        # Iterate over all valid i
        for i in range(n):
            if h_bin[i] == 1:
                j = i + s_max
                if 0 <= j < n and h_bin[j] == 0:
                    # Potential swap: i=1, i+s_max=0
                    # Try swapping i and j
                    h_temp = h_bin.copy()
                    h_temp[i] = 0
                    h_temp[j] = 1
                    # Evaluate new C5
                    corr_new = np.correlate(h_temp, 1 - h_temp, mode="full")
                    max_c_new = np.max(corr_new)
                    # Better if it reduces max_c
                    if max_c_new - max_c < -1e-10:  # We must be strictly improving
                        h_bin[i] = 0
                        h_bin[j] = 1
                        max_c = max_c_new
                        improvement = True
                        last_improvement = time.time()
                        break
        if not improvement:
            break
        iter_count += 1
    return h_bin
