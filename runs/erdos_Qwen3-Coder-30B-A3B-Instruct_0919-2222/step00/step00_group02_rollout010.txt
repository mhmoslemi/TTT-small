def random_balanced_binary(n, rng):
    """Generate a random binary array with exactly n/2 ones"""

    return h

def project_onto_box_and_sum(v, target):
    """Project v onto [0,1] box and sum constraint via binary search"""
    # Ensure we find λ such that sum of clip(v - λ, 0, 1) equals target
    lo = np.min(v) - 1.0

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)
