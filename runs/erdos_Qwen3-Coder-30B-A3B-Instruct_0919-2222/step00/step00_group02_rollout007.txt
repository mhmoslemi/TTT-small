import numpy as np

def evaluate_erdos_solution(h):
    """ Evaluate the C5 constant for a given step function h """
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    correlations = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(correlations)
    return max_overlap * dx

                constraints += [w_vars[(i, k)] <= h[i+k]]

def swap_rounding(h_frac, n):
    h = np.copy(h_frac)
    # Swap round to make binary
    sums = np.sum(h)
    while True:
        # Find any fractional elements
        indices = np.where((h > 0) & (h < 1))[0]
        if len(indices) == 0:
            break
            
        # Hardcoded deterministic pair selection for safety
        i = indices[0]
        j = indices[-1] if len(indices) > 1 else indices[-1] - 1
        if j < 0 or j >= n or abs(h[i] - h[j]) > 1e-10:
            # Greedy: pick pair with same fractional part or closest
            ratios = h % 1
            min_diff = float('inf')
            best_i, best_j = i, j
            for iidx in indices:
                for jidx in indices:
                    if iidx != jidx:
                        diff = abs(ratios[iidx] - ratios[jidx])
                        if diff < min_diff:
                            min_diff = diff
                            best_i, best_j = iidx, jidx
            i, j = best_i, best_j            
        
        if h[i] >= 0.5 and h[j] <= 0.5:
            delta = min(h[i], 1 - h[j])
            if delta > 1e-8:
                h[i], h[j] = h[i] - delta, h[j] + delta
        elif h[i] < 0.5 and h[j] > 0.5:
            delta = min(1 - h[i], h[j])
            if delta > 1e-8:
                h[i], h[j] = h[i] + delta, h[j] - delta
        else:
            # Both too large or too small -> avoid invalid swap
            break
            
        if abs(np.sum(h) - n/2) < 1e-10:
            break
    return h

def sturmian_construction(n):
    # Define sensible search space for alpha and tau
    tau_range = np.linspace(0, 1, 100)
    alpha_range = np.linspace(0.2, 0.8, 200)
    best_bound = float('inf')
    best_h = None
    
    info = {'alpha': 0, 'tau': 0} 
    
    for alpha in alpha_range:
        for tau in tau_range:
            h = np.zeros(n)
            for i in range(n):
                frac = (i * alpha + tau) % 1
                h[i] = 1 if frac < 0.5 else 0
            c5 = evaluate_erdos_solution(h)
            if c5 < best_bound:
                best_bound = c5
                best_h = h.copy()
                info['alpha'] = alpha
                info['tau'] = tau
    return best_h, best_bound
