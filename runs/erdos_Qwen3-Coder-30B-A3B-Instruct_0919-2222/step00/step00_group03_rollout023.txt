import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The full correlation is of length 2*n - 1
    # We want the maximum of the integrals for shifts k in [0, 2]
    # Each integral corresponds to a shift from -1 to 1, step size dx
    max_val = 0.0
    # For each valid shift (corresponding to k in [0, 2])
    # We examine the corresponding correlation value
    for i in range(len(correlation)):
        # The lag in the correlation corresponds to shift k
        # k = i*dx - (n-1)*dx = i*dx - n*dx + dx = (i - n + 1)*dx
        # We care about shifts from 0 to 2
        # i ranges from 0 to 2*n - 2
        k = (i - (n - 1)) * dx
        if 0 <= k <= 2:  # Only valid shifts in [0, 2]
            val = correlation[i] * dx
            max_val = max(max_val, val)
    return max_val
