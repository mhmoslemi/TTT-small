import numpy as np
import time
import random

# Global variable provided
initial_h_values = None

def compute_correlation(h):
    """Compute correlation of h with (1-h) using full convolution"""
    return np.correlate(h, 1 - h, mode="full")

def max_overlap(h):
    """Compute the maximum overlap integral"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    return np.max(corr) * dx

def apply_delta(c, h, idx, delta):
    """Apply delta change to h at index idx and update correlation c"""
    n = len(h)
    offset = n - 1
    # First term: for j from idx to n-1
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # Second term: for j from 0 to idx
    for j in range(0, idx + 1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(i, j, c, h):
    """Perform a swap between indices i and j, updating h and c"""
    # Save old values
    old_i = h[i]
    old_j = h[j]
    
    # Apply delta changes
    delta_i = 0 - old_i
    delta_j = 1 - old_j
    
    # Update correlation
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def targeted_swap(c, h):
    """Try a targeted swap based on max correlation shift"""
    n = len(h)
    offset = n - 1
    max_idx = np.argmax(c)
    s = max_idx - offset  # Get actual shift
    
    # Find candidate indices for i (h[i]=1) and j (h[j]=0) that contribute to this shift
    candidates_i = []
    candidates_j = []
    
    for i in range(n):
        if h[i] == 1:
            j = i + s
            if 0 <= j < n and h[j] == 0:
                candidates_i.append(i)
        elif h[i] == 0:
            j = i + s
            if 0 <= j < n and h[j] == 1:
                candidates_j.append(i)
    
    if candidates_i and candidates_j:
        i = random.choice(candidates_i)
        j = random.choice(candidates_j)
        return (i, j)
    return None

def hill_climb(h, n_iter, time_left):
    """Perform hill climbing optimization"""
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h)
    current_max = np.max(c) * dx
    best_h = h.copy()
    best_max = current_max
    best_c = c.copy()
    
    # Maintain sets of indices
    ones_idx = [i for i in range(n) if h[i] == 1]
    zeros_idx = [i for i in range(n) if h[i] == 0]
    
    for _ in range(n_iter):
        if time_left() <= 0:
            break
            
        # Try targeted swap first
        swap_indices = targeted_swap(c, h)
        if swap_indices is not None:
            i, j = swap_indices
            # Apply swap
            swap(i, j, c, h)
            # Update the tracking indices
            ones_idx.remove(i)
            ones_idx.append(j)
            zeros_idx.remove(j)
            zeros_idx.append(i)
        else:
            # Fallback to random swap
            i = random.choice(ones_idx)
            j = random.choice(zeros_idx)
            swap(i, j, c, h)
            # Update tracking indices
            ones_idx.remove(i)
            ones_idx.append(j)
            zeros_idx.remove(j)
            zeros_idx.append(i)
            
        # Update maximum correlation
        current_max = np.max(c) * dx
        if current_max < best_max:
            best_h = h.copy()
            best_max = current_max
            best_c = c.copy()
            
    return best_h, best_max, best_c

def interpolate_h(initial_h, new_n):
    """Linearly interpolate h from old size to new size"""
    old_n = len(initial_h)
    x_old = np.linspace(0, 2, old_n)
    x_new = np.linspace(0, 2, new_n)
    h_new = np.interp(x_new, x_old, initial_h)
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    return h_new

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Time manager
    def time_left():
        return max(0, budget_s - (time.time() - start_time))
    
    # Rolling window for multiple restarts
    n_values = [60, 80, 100, 120, 150]
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    for n in n_values:
        if time_left() < 20:
            break
            
        # Initialize h
        if n <= len(initial_h_values):
            h = interpolate_h(initial_h_values, n)
        else:
            # Upsample initial if needed
            h = interpolate_h(initial_h_values, n)
            
        # Normalize to target sum
        target_sum = n / 2
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        
        # Ensure exact sum by adjusting one element
        diff = target_sum - np.sum(h)
        # Adjust one element (at random if needed)
        idx = np.random.randint(0, len(h))
        h[idx] = min(1, max(0, h[idx] + diff))
        
        # Convert to binary for swap operations
        h_bin = (h > 0.5).astype(int)
        # Adjust to exact sum
        actual_sum = np.sum(h_bin)
        if actual_sum < target_sum:
            # Add more ones
            ones_indices = np.where(h_bin == 0)[0]
            to_change = int(target_sum - actual_sum)
            for i in range(min(to_change, len(ones_indices))):
                h_bin[ones_indices[i]] = 1
        elif actual_sum > target_sum:
            # Remove some ones
            ones_indices = np.where(h_bin == 1)[0]
            to_change = int(actual_sum - target_sum)
            for i in range(min(to_change, len(ones_indices))):
                h_bin[ones_indices[i]] = 0
        
        h = h_bin.astype(float)
        
        # Run hill climbing
        iterations = min(50000, int(time_left() * 100))
        h_opt, c5_bound, _ = hill_climb(h.copy(), iterations, time_left)
        
        # Check if we found a better solution
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
            best_n = n
            
        # Early exit if we achieved target
        if c5_bound <= 0.379:
            break
            
        # Check remaining time
        if time_left() < 5:
            break
    
    if best_h is None:
        # Fallback to initial solution if nothing was found
        h = interpolate_h(initial_h_values, 100)
        h_bin = (h > 0.5).astype(int)
        actual_sum = np.sum(h_bin)
        target_sum = 50  # n/2 for n=100
        diff = target_sum - actual_sum
        if diff > 0:
            ones_idx = np.where(h_bin == 0)[0]
            for i in range(diff):
                h_bin[ones_idx[i]] = 1
        elif diff < 0:
            ones_idx = np.where(h_bin == 1)[0]
            for i in range(-diff):
                h_bin[ones_idx[i]] = 0
        best_h = h_bin.astype(float)
        best_n = 100
        best_c5 = max_overlap(best_h)
        
    return best_h.tolist(), best_c5, best_n
