def evaluate_erdos_solution(h_values):
    """Evaluate C5 for given h_values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute all correlations (including full convolution mode)
    conv = correlate(h_values, 1 - h_values, mode="full")
    # We need to consider only the valid ranges
    # The sum is always valid, but the shifted correlations (shifts k in [-n+1, n-1])
    # For a shift k, we need h[i] * (1-h[i+k]) for valid i
    # which means i in [0, n-1-k] if k >= 0, or [0-k, n-1] if k < 0
    # The collection of valid correlations have lengths:
    #    for k in [-n+1,..,n-1]: length = n - abs(k)
    # But for the explicitly computed convolution, we only care about the middle part
    # Let's extract max one, we'll use the fact that convolution center is at index (2*n-2)
    # The result is an array of length (2*n-1)
    indices = np.arange(len(conv))
    # For k=0: we take from index (n-1) to (n-1) -> this is the shift k=0
    # But in sum for each k, we do offset for i such that it's valid with h[i+k]
    # Let's compute properly...
    # effectively for shift k, valid indices to multiply are for i=0 to n-1-k if k >= 0,
    # or i=-k to n-1 if k < 0. The max shifts dimension makes the distance from 
    # the center correlation window.
    # The English way:
    # For k=0,  we compute sum over i of h[i] * (1 - h[i]) = sum(h * (1-h))
    # For shift k: 0 <= i <= n-1-k (for k >= 0)
    # For shift k: -k <= i <= n-1 (for k < 0)
    # Instead of direct indexing... let's stick to the shifted and correlate directly.
    # Build for each shift k, compute sum(h[i] * (1-h[i+k]))
    o_k = np.zeros(2*n-1)
    for k in range(-n+1, n):
        # compute sum over i of h[i] * (1 - h[i+k])
        if k >= 0:
            o_k[n-1+k] = np.sum(h_values[:n-k] * (1 - h_values[k:]))
        else:
            o_k[n-1+k] = np.sum(h_values[-k:] * (1 - h_values[:-k]))
    # remove a few terms from each end to align with curve version:
    o_k = o_k[n-1:]
    c5 = np.max(o_k)
    return c5

def evaluate_convolution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # We compute the overlap = sum of all h(i) * (1 - h(i+k)) over k
    # Normal correlate gives the full shifted sums, so we split that page into shifts
    full_conv = correlate(h_values, 1 - h_values, mode="full")
    # The center of the correlation is at n-1
    # The shift k corresponds to index n-1 + k
    max_overlap = 0.0
    for k in range(-n+1, n):
        idx = n-1 + k
        # Adjust for shift k
        val = full_conv[idx]
        # But we only sum valid combinations
        # i must be valid such that h[i] and h[i+k] both exist
        # i: [0, n-1]  
        # i+k: [0, n-1]
        # thus i should be [max(0, -k), min(n-1, n-1-k)] = [max(0, -k), n-1-k]
        # if k<0, then -k > 0 so max(0, -k) = -k, valid i range [max(0, -k), n-1-k]
        # if k > 0, then n-1-k < n-1, all i within [0, n-1-k] work
        # if k == 0, range is [0, n-1]
        if k >= 0:
            valid_i_start = 0
            valid_i_end = n - k
        else:  # k < 0
            valid_i_start = -k  # = max(0, -k)
            valid_i_end = n  # = n-1
        # The convolution gives the same value for the range but is also a closed form identity
        # So we know that the computed sum full_conv[idx] is indeed the sum:
        # sum_{i=valid_start}^{valid_end-1} h[i] * (1 - h[i+k])
        max_overlap = max(max_overlap, val)
    return max_overlap

def convex_relaxation(h0, n_points, dx):
    #Initial z scores for the LP problem
    # set up
    n = n_points
    K = {0, -1, 1, -2, 2}
    def compute_Ok(h, k_idx):
        start = max(0, k_idx)
        end = min(n, n - k_idx)
        if k_idx >= 0:
            valid_i = np.arange(0, n - k_idx)
        else:
            valid_i = np.arange(-k_idx, n)
        return np.sum(h[valid_i] * (1 - h[valid_i + k_idx]))
    # After solving the LP, we get fractional h^star
    h_star = h0.copy()  # fractionals, iff needed
    # we can also get this from solve_LLP, etc.
    # Generate relaxation for a few basic cases:
    h = cp.Variable(n, name="h")
    t = cp.Variable(1, name="t")
    w = {}
    # w[i,k] = h[i]*h[i+k]
    exprs = []
    Ec = []
    # Precompute pairs
    for k in range(-n+1, n):
        for i in range(n):
            i_plus_k = i+k
            if i_plus_k < 0 or i_plus_k >= n:
                continue
            key = (i,k)
            w[key] = cp.Variable(name=f"w_{i}_{k}")
            exprs.append(w[key] <= h[i])
            exprs.append(w[key] <= h[i_plus_k])
            exprs.append(w[key] >= h[i] + h[i_plus_k] - 1)
            exprs.append(w[key] >= 0)        
    # The objective is to minimize t
    obj = cp.Minimize(t)
    # Add constraints from K only 
    l_exprs = []
    for k_idx in K:
        # compute sum h[i] (1 - h[i+k]) 
        # = sum h[i] - sum h[i]*h[i+k] = sum h - sum w[i,k]
        start = max(0, k_idx)
        end = min(n, n - k_idx)
        if k_idx >= 0:
            valid_i = np.arange(0, n - k_idx)
        else:
            valid_i = np.arange(-k_idx, n)
        if len(valid_i) > 0:
            if k_idx == 0:
                o_k = sum(h[valid_i]) - 0  # Special case
            else:
                o_k = sum(h[valid_i]) - sum([w[(i,k_idx)] for i in valid_i])
            l_exprs.append(o_k <= t)
    # Finally, add overall constraints
    l_exprs.append(sum(h) == n/2)
    l_exprs.extend(exprs)
    prob = cp.Problem(obj, l_exprs)
    prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000)
    if prob.status == "optimal":
        h_final = h.value
        t_value = t.value
        hh = h_final.copy()
        return hh, t_value
    else:
        # fallback to basic case or h0
        return h0, compute_Ok(h0, 0)

def round_step(h_frac, seed=42):
    rng = np.random.default_rng(seed)
    h_bin = h_frac.copy()
    n = len(h_bin)
    hall = list(np.where((h_bin > 0) & (h_bin < 1))[0])  # indices that are fractional
    while len(hall) > 0:
        # Find indices with identical fractional part
        frac_parts = h_bin[hall]
        unique_fracs, inverse_indices = np.unique(frac_parts, return_inverse=True)
        paired = []
        for frac in unique_fracs:
            matching = np.where(frac_parts == frac)[0]
            if len(matching) >= 2:
                # choose a pair
                i0, i1 = rng.choice(matching, 2, replace=False)
                paired.append((hall[i0], hall[i1]))
        if len(paired) == 0:
            # fallback to any random pair
            i = rng.choice(hall)
            j = rng.choice(hall)
            while i == j:
                j = rng.choice(hall)
            paired.append((i,j))
        # Perform one swap
        u, v = paired[0]
        delta = min(h_bin[u], 1 - h_bin[v])
        if delta == 0:
            # remove from hall and continue
            hall.remove(u)
            if u != v:
                hall.remove(v)
            continue
        if rng.random() < 0.5:
            h_bin[u] -= delta
            h_bin[v] += delta
        else:
            h_bin[u] += delta
            h_bin[v] -= delta
        # remove completed points
        if 0 < h_bin[u] < 1:
            pass
        else:
            try:
                hall.remove(u)
            except:
                pass
        if 0 < h_bin[v] < 1:
            pass
        else:
            try:
                hall.remove(v)
            except:
                pass
        # Ensure sum is preserved
        if abs(sum(h_bin) - n/2) >= 1e-10:
            print("warning: sum broke in rounding", abs(sum(h_bin) - n/2))
    return h_bin

def local_improve(h0, max_iter):
    h_opt = h0.copy()
    cur_score = evaluate_convolution(h_opt)
    count = 0
    n = len(h_opt)
    while count < max_iter:
        improvements_made = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                h_trial = h_opt.copy()
                h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
                trial_score = evaluate_convolution(h_trial)
                if trial_score < cur_score:
                    h_opt = h_trial
                    cur_score = trial_score
                    improvements_made = True
                    break
            if improvements_made:
                break
        if not improvements_made:
            break
        count += 1
    return h_opt

def sturmian_construction(n):
    # Use phi = (sqrt(5) - 1)/2  ~ 0.618 which is known to yield balanced sequences
    phi = (np.sqrt(5) - 1) / 2
    # Phase shift parameter: try to find a good one
    best_score = float('inf')
    best_h = None
    for tau in np.linspace(0, 1, 100):
        h_temp = np.array([1 if ((i * phi + tau) % 1 < 0.5) else 0 for i in range(n)])
        score = evaluate_convolution(h_temp)
        if score < best_score:
            best_score = score
            best_h = h_temp
    # Fine-tune with golden section search
    left = 0.0
    right = 1.0
    tolerance = 1e-8
    while (right - left) > tolerance:
        # Golden ratio splits
        phi_g = (3 - np.sqrt(5)) / 2
        x1 = left + phi_g * (right - left)
        x2 = right - phi_g * (right - left)
        h1 = np.array([1 if ((i * phi + x1) % 1 < 0.5) else 0 for i in range(n)])
        h2 = np.array([1 if ((i * phi + x2) % 1 < 0.5) else 0 for i in range(n)])
        s1 = evaluate_convolution(h1)
        s2 = evaluate_convolution(h2)
        if s1 < s2:
            right = x2
        else:
            left = x1
        if s1 < best_score:
            best_score = s1
            best_h = h1
        if s2 < best_score:
            best_score = s2
            best_h = h2
    # Full polish
    # Try local swaps (or better, do a deterministic one?)
    return best_h, best_score
