import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value corresponds to the desired C5
    c5 = np.max(correlation) * dx
    return c5

def initial_h_values():
    """Provide an initial construction for h."""
    n = 70
    h = np.ones(n) * 0.5
    h[:n//2] = 1.0
    h[n//2:] = 0.0
    return h

def objective(h):
    """Objective function to minimize (negative of C5)."""
    return -evaluate_erdos_solution(h)

def constraint_eq(h):
    """Equality constraint: integral of h over [0,2] equals 1."""
    n = len(h)
    dx = 2.0 / n
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial solution
    h0 = initial_h_values()
    n = len(h0)
    dx = 2.0 / n
    
    # Define bounds for each variable (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n)]
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use differential evolution for global optimization
    result = differential_evolution(
        func=objective,
        bounds=bounds,
        constraints=cons,
        seed=seed,
        maxiter=int(budget_s / 2),  # Roughly half the budget for the solver
        popsize=15,
        disp=False
    )
    
    # If we haven't finished all time, try more refinement
    current_time = time.time()
    if current_time - start_time < budget_s - 5:  # Leave some margin
        # Try local optimization with the result
        h_refined = result.x
        res_refined = minimize(
            fun=objective,
            x0=h_refined,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int((budget_s - (current_time - start_time)) * 20)},
            tol=1e-8
        )
        if res_refined.success and -res_refined.fun < -result.fun:
            solution = res_refined.x
            c5 = -res_refined.fun
        else:
            solution = result.x
            c5 = -result.fun
    else:
        solution = result.x
        c5 = -result.fun
    
    # Ensure the solution respects constraints
    h_result = np.clip(solution, 0, 1)
    if abs(np.sum(h_result) * dx - 1.0) > 1e-6:
        # Adjust to satisfy the integral constraint
        avg_val = 1.0 / dx / n
        h_result = np.clip(h_result, 0, 1)
        sum_h = np.sum(h_result)
        if sum_h > 0:
            h_result = h_result * (1.0/dx/n) / sum_h
    
    # Final check
    final_c5 = evaluate_erdos_solution(h_result)
    
    return (h_result, final_c5, n)
