def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h"""

    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value corresponds to the maximum overlap
    max_overlap = np.max(correlation)
    return max_overlap * dx
