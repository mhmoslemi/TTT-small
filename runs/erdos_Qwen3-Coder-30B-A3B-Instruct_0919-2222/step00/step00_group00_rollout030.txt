import numpy as np
import time
import random

def compute_correlation(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap(h, dx):
    c = compute_correlation(h)
    return c.max() * dx

def apply_delta(c, h, idx, delta):
    n = len(h)
    offset = n - 1
    # first term: for j from idx to n-1:
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # second term: for j from 0 to idx:
    for j in range(0, idx+1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(i, j, c, h):
    delta_i = 0 - h[i]
    delta_j = 1 - h[j]
    
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def targeted_swap(c, h):
    n = len(h)
    offset = n - 1
    max_idx = np.argmax(c)
    s = max_idx - offset  # the shift corresponding to maximum overlap
    
    # Find candidate swaps based on the maximum shift
    candidates_i = []
    candidates_j = []
    
    for i in range(n):
        j = i + s
        if 0 <= j < n:
            if h[i] == 1 and h[j] == 0:
                candidates_i.append(i)
            if h[i] == 0 and h[j] == 1:
                candidates_j.append(i)
    
    if candidates_i and candidates_j:
        i = random.choice(candidates_i)
        j = random.choice(candidates_j)
        return (i, j)
    
    return None

def hill_climb(h, n_iter, time_left):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h)
    current_max = c.max() * dx
    
    best_h = h.copy()
    best_max = current_max
    
    # Keep track of positions with h[i] = 1 and h[i] = 0
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    
    iterations_since_improvement = 0
    max_iterations_since_improvement = 20000  # Avoid infinite loop
    
    for iteration in range(n_iter):
        if time_left() <= 1:
            break
            
        # Targeted swap based on maximum overlap shift
        swap_indices = targeted_swap(c, h)
        
        if swap_indices is not None:
            i, j = swap_indices
            old_i = h[i]
            old_j = h[j]
            
            swap(i, j, c, h)
            
            # Update sets
            ones.remove(i)
            zeros.remove(j)
            ones.add(j)
            zeros.add(i)
            
            new_max = c.max() * dx
            
            if new_max < current_max:
                current_max = new_max
                iterations_since_improvement = 0
                if current_max < best_max:
                    best_max = current_max
                    best_h = h.copy()
            else:
                # Revert
                swap_indices = (j, i)  # revert swap correctly
                h[i] = old_i
                h[j] = old_j
                # Update c to reflect reverted action manually  
                # Using indices for precision
                apply_delta(c, h, i, -delta_i if 'delta_i' in locals() else 0)
                swap(i, j, c, h)
        else:
            # Random swap to escape local minima
            if len(ones) > 0 and len(zeros) > 0:
                i = random.choice(list(ones))
                j = random.choice(list(zeros))
                old_i = h[i]
                old_j = h[j]
                swap(i, j, c, h)
                ones.remove(i)
                zeros.remove(j)
                ones.add(j)
                zeros.add(i)
    
                new_max = c.max() * dx
                
                if new_max < current_max:
                    current_max = new_max
                    iterations_since_improvement = 0
                    if current_max < best_max:
                        best_max = current_max
                        best_h = h.copy()
                else:
                    # Revert
                    h[i] = old_i
                    h[j] = old_j
                    # Reapply manually like swap revert
                    apply_delta(c, h, i, -delta_i if 'delta_i' in locals() else 0)
                    swap(i, j, c, h)

            iterations_since_improvement += 1
            if iterations_since_improvement > max_iterations_since_improvement:
                break
    
    return (best_h, best_max)

def run(seed=42, budget_s=600):
    # Set the random seed
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    
    # Determine the best n based on known solution length
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
    else:
        n0 = 45 # Default if no initial_h_values

    # Try different grid sizes
    n_values = [int(n0/2*2), 60, 80, 100, 120, 140, 160]
    n_values = [n for n in n_values if n <= 200]  # Cap the n values to search
    
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Process each n size
    for n in n_values:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate an initial step function - use binary starting for speed
        # Here we just reproduce the known initial distribution approximating n=45
        if n == 45:
            h = np.array(initial_h_values)
        elif n <= 100:
            # Linear interpolation of initial solution if necessary
            x_old = np.linspace(0, 2, n0)
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_old, initial_h_values)
        else:
            # Larger sizes just random binary with fixed sum
            h = np.zeros(n)
            ones_indices = random.sample(range(n), n//2)
            h[ones_indices] = 1.0

        # Limit number of optimization steps based on time remaining
        time_left = lambda: budget_s - (time.time() - start_time)
        remaining_sec = time_left()
        
        if remaining_sec <= 1:
            continue
            
        # Initialize binary if necessary
        if n <= 100:
            # Ensure binary start
            h = (h > 0.5).astype(float)
        else:
            # Ensure it's in [0,1]
            h = np.clip(h, 0, 1)
            
        # Normalize to sum to n/2 exactly
        current_sum = h.sum()
        if abs(current_sum - n/2) > 1e-6:
            diff = (n/2 - current_sum)/n
            h += diff
            h = np.clip(h, 0, 1)
            
        # Run hill climbing
        best_h_from_n, best_c5_from_n = hill_climb(h, 30000, time_left)
        
        if best_c5_from_n < best_c5:
            best_c5 = best_c5_from_n
            best_h = best_h_from_n.copy()
            best_n = n
            
        if best_c5 <= 0.379:
            break  # Stop early if we're good
    
    if best_h is None:
        # Recover from complete failure
        h = np.zeros(120)
        ones_indices = random.sample(range(120), 60)
        h[ones_indices] = 1.0
        best_h, best_c5 = hill_climb(h, 50000, lambda: budget_s - (time.time() - start_time))

    return (list(best_h), best_c5, best_n if best_n else 120)
