import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap problem constant C5 for given step function h."""
    # Ensure h is a numpy array
    h = np.array(h)
    # Compute correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode="full")
    # Find maximum correlation scaled by dx (which is 2/len(h))
    dx = 2.0 / len(h)
    return np.max(corr) * dx

def initial_h_values():
    """Return the initial construction."""
    n_points = 47
    h = np.zeros(n_points)
    # Set first half to 1, second half to 0 (approximately)
    h[:n_points//2] = 1  
    # Scale to sum = n_points/2 exactly
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_h_values(n_points, shape_type='cos2', p=2.0, a=1.0):
    """Generate h values for given shape."""
    x = np.linspace(0, 2, n_points, endpoint=False) + 1.0/n_points
    if shape_type == 'triangle':
        h = np.maximum(0, 1 - np.abs(x - 1))
    elif shape_type == 'cos2':
        h = np.cos(np.pi/2 * (x - 1))**2
    elif shape_type == 'cos_power':
        h = np.cos(np.pi/2 * (x - 1))**p
    elif shape_type == 'poly':
        h = np.maximum(0, 1 - np.abs(x - 1)**a)
    else:
        # Default to uniform
        h = np.ones(n_points) * 0.5
    # Normalize to sum = n_points/2
    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-10:
        h = h * (n_points / 2) / h_sum
    # Clip to [0,1] range
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function to find improved upper bound for C5."""
    np.random.seed(seed)
    start_time = time.time()
    end_time = start_time + budget_s * 0.99
    
    # Parameters for shape search
    n_points = 501  # Make it odd for symmetry
    dx = 2.0 / n_points
    
    # Search over parameter space
    best_h = None
    best_c5 = float('inf')
    
    # Shape types to try
    shapes = [
        ('triangle', 0, 0),
        ('cos2', 0, 0),
        ('cos_power', 0.5, 0),
        ('cos_power', 1.0, 0),
        ('cos_power', 2.0, 0),
        ('cos_power', 4.0, 0),
        ('cos_power', 8.0, 0),
        ('poly', 1, 0),
        ('poly', 2, 0),
        ('poly', 3, 0)
    ]
    
    # Evaluate all parameter combinations
    for shape_type, p, a in shapes:
        try:
            h = compute_h_values(n_points, shape_type, p, a)
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except Exception:
            continue
            
        if time.time() > end_time:
            break
    
    # Local search refinement
    if best_h is not None and time.time() < end_time:
        # Simple hill climbing
        current_h = best_h.copy()
        current_c5 = best_c5
        
        # Use a budget of 10k iterations with time checking
        max_iters = 10000
        iter_count = 0
        
        while iter_count < max_iters and time.time() < end_time:
            # Choose two random indices
            i, j = np.random.randint(0, n_points, 2)
            if i == j:
                continue
                
            # Compute maximum changes allowed
            delta_max = min(current_h[i], 1 - current_h[j])
            if delta_max <= 0:
                continue
                
            # Choose random step size
            delta = np.random.random() * min(delta_max, 0.05)
            
            # Update values (try to move from i to j)
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            
            # Ensure bounds
            if new_h[i] < 0 or new_h[j] > 1:
                continue
                
            # Recompute C5 for the new h
            new_c5 = evaluate_erdos_solution(new_h)
            
            # Accept improvement or accept with some probability (simple SA)
            if new_c5 < current_c5 or np.random.random() < 0.001:
                current_h = new_h
                current_c5 = new_c5
            
            # Update best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            
            iter_count += 1
    
    # Make sure we are within the required constraint
    if best_h is not None:
        # If needed, recompute to ensure precision
        final_c5 = evaluate_erdos_solution(best_h)
        return list(best_h), final_c5, n_points
    
    # If no improvement found, return the initial solution
    h0 = initial_h_values()
    c5_0 = evaluate_erdos_solution(h0)
    return list(h0), c5_0, len(h0)
