import numpy as np
import pulp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial solution as starting point if available
    try:
        h0 = initial_h_values
        n = len(h0)
        best_h = h0.copy()
        best_c5 = evaluate_erdos_solution(h0)
    except:
        # Default setup
        n = 200
        best_h = np.array([0.5] * n)
        best_c5 = 0.5  # initial estimate
    
    # Try different n values in descending order to find good solutions
    n_values = [200, 180, 160, 140, 120, 100, 90, 80, 70]
    best_c5_so_far = best_c5
    
    for n_trial in n_values:
        current_time = time.time()
        elapsed = current_time - start_time
        remaining_time = budget_s - elapsed
        
        if remaining_time < 20:
            break
            
        try:
            # Solve the MILP problem
            prob = pulp.LpProblem("ErdosMinOverlap", pulp.LpMinimize)
            
            # Variables
            h = pulp.LpVariable.dicts('h', list(range(n_trial)), lowBound=0, upBound=1, cat='Binary')
            T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
            
            # Constraint: sum(h) = n/2
            prob += pulp.lpSum([h[i] for i in range(n_trial)]) == n_trial / 2
            
            # For each shift d = 1 to n-1:
            for d in range(1, n_trial):
                # Create y variables for this shift (consistent indexing)
                y_vars = []
                for i in range(n_trial - d):
                    y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
                    y_vars.append(y)
                    # McCormick constraints:
                    prob += y <= h[i]
                    prob += y <= h[i+d]
                    prob += y >= h[i] + h[i+d] - 1
                # Constraint for the shift: sum of products >= n/2 - T
                prob += pulp.lpSum(y_vars) >= n_trial / 2 - T
            
            # Objective: minimize T
            prob += T
            
            # Solve with time limit
            solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=remaining_time-5)
            prob.solve(solver)
            
            # Retrieve solution and compute bound
            if pulp.LpStatus[prob.status] in ['Optimal', 'Feasible']:
                h_vals = np.array([pulp.value(h[i]) for i in range(n_trial)], dtype=float)
                t_val = pulp.value(T)
                c5_bound = t_val * (2.0 / n_trial)
                if c5_bound < best_c5_so_far:
                    best_h = h_vals
                    best_c5_so_far = c5_bound
        except:
            continue  # Skip if fails
            
    # Final evaluation
    try:
        final_bound = evaluate_erdos_solution(best_h)
        return best_h, final_bound, len(best_h)
    except:
        # Fallback evaluation if we couldn't compute final evaluation
        return best_h, best_c5_so_far, len(best_h)
