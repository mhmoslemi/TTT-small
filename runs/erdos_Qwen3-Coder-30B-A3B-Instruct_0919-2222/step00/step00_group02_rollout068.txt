import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from math import floor

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all possible shifts
    # This is equivalent to convolving h with (1-h) and taking the maximum
    conv = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def initial_h_values():
    """Provide initial construction with n=45 samples"""
    n = 45
    h = np.zeros(n)
    dx = 2.0 / n
    # Simple construction based on initial guess from literature
    # Place a step function with height 1 in the middle
    half = n // 2
    start = n // 4
    end = 3 * n // 4
    h[start:end] = 1.0
    # Scale so that integral is 1
    h *= 2.0 / (end - start)
    return h

def objective(h, n_points):
    """Objective function to minimize C_5"""
    # Rescale h to satisfy integral constraint
    dx = 2.0 / n_points
    h_scaled = h * (1.0 / (np.sum(h) * dx))
    h_scaled = np.clip(h_scaled, 0, 1)
    return evaluate_erdos_solution(h_scaled)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h_init = initial_h_values()
    n_points = len(h_init)
    
    # Parameters for optimization
    max_iter = 500
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for global optimization
    de_result = differential_evolution(
        lambda h: objective(h, n_points),
        bounds,
        maxiter=max_iter,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False,
        tol=1e-6
    )
    
    # Final refinement with local optimization
    best_h = np.clip(de_result.x, 0, 1)
    # Ensure integral constraint
    dx = 2.0 / n_points
    best_h = best_h * (1.0 / (np.sum(best_h) * dx))
    best_h = np.clip(best_h, 0, 1)
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Verify it's the best so far or within time
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        pass  # Return what we have
    
    # Refine further with L-BFGS if still time
    if elapsed < budget_s - 5:
        # Add small extra refinement
        def obj_to_minimize(h):
            h = np.clip(h, 0, 1)
            dx = 2.0 / len(h)
            h_scaled = h * (1.0 / (np.sum(h) * dx))
            h_scaled = np.clip(h_scaled, 0, 1)
            return evaluate_erdos_solution(h_scaled)

        # Use scipy minimize for better quality
        res = minimize(
            obj_to_minimize,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 50}
        )
        if res.success:
            sol = np.clip(res.x, 0, 1)
            sol = sol * (1.0 / (np.sum(sol) * dx))
            sol = np.clip(sol, 0, 1)
            new_c5 = evaluate_erdos_solution(sol)
            if new_c5 < final_c5:
                best_h = sol
                final_c5 = new_c5
    
    # Return the best solution found
    return (best_h, final_c5, n_points)
