import numpy as np
import cvxpy as cp
import time
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    # Discretisation sizes to try
    n_list = [80, 120, 160]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_list:
        if time.time() - t_start > budget_s - 5:
            break
            
        # Limit the shifts we consider initially
        K0 = min(n // 4, 50)
        
        # Initial solution using heuristic
        h = np.zeros(n)
        h[:n//2] = 1
        h = h / np.sum(h) * (n/2)
        
        # Try different starting points using small random swaps
        for _ in range(50):
            if time.time() - t_start > budget_s - 5:
                break
            i, j = np.random.randint(0, n, size=2)
            if i != j and not (h[i] == 0 and h[j] == 1):
                h[i], h[j] = h[j], h[i]
        
        # Binary search on t with lazy constraint loop
        h_bin = (h > 0.5).astype(int)
        h_bin = h_bin / np.sum(h_bin) * (n/2)
        h_bin = h_bin.astype(int)
        
        # Compute initial overlap
        c_full = np.correlate(h_bin, 1 - h_bin, mode='full')
        current_max = np.max(c_full)
        c5_bound = current_max / n * 2.0  # dx = 2/n
        
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_bin.copy()
            best_n = n
            break
        
        # Use MILP approach with lazy constraints
        t_end = time.time() + (budget_s - (time.time() - t_start)) * 0.8
        while time.time() < t_end:
            try:
                # Build MADP variables
                x = cp.Variable(n, boolean=True)
                t = cp.Variable()
                
                # Initial set of active shifts
                K = list(range(1, min(K0 + 1, n)))
                
                # Collect all the constraints
                constraints = [cp.sum(x) == n // 2]  # mass constraint
                constraints.append(x[0] == 1)  # symmetry breaking
                for i in range(n//2 - 1):
                    constraints.append(x[i] >= x[i+1])  # non-increasing prefix
                
                # Add McCormick constraints for each active shift k
                m_y = {}
                m_c = {}
                
                for k in K:
                    # For shift k, create y variables
                    y_vars = []
                    for i in range(n - k):
                        y = cp.Variable(boolean=True)
                        y_vars.append(y)
                        m_y[(i,k)] = y
                        
                    # Add McCormick constraints
                    for i in range(n - k):
                        constraints.extend([
                            y_vars[i] <= x[i],
                            y_vars[i] <= x[i+k],
                            y_vars[i] >= x[i] + x[i+k] - 1
                        ])
                    
                    # Overlap constraint
                    c_k = cp.sum(x[:n-k]) - cp.sum(y_vars)
                    m_c[k] = c_k
                    constraints.append(c_k <= t)
                
                # Define the problem
                prob = cp.Problem(cp.Minimize(t), constraints)
                
                # Solve the integer program
                prob.solve(solver=cp.CBC, verbose=False, time_limit=5)
                
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    x_opt = np.round(x.value).astype(int)
                else:
                    break  # Solver failed
                
                # Check if we found a better solution
                max_c_full = np.max(np.correlate(x_opt, 1 - x_opt, mode='full'))
                c5_new = max_c_full / n * 2.0
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = x_opt
                    best_n = n
                
                # Check for violating shifts (find the current max and full correlation)
                full_corr = np.correlate(x_opt, 1 - x_opt, mode='full')
                violated = []
                
                for k in range(1, n):
                    c_k = full_corr[n-1+k] if k <= n-1 else full_corr[n-1-k]
                    if c_k > t.value + 1e-6:
                        violated.append(k)
                
                # If violation found, add the most violating shift
                if violated:
                    k_max = max(violated, key=lambda k: full_corr[n-1+k] if k <= n-1 else full_corr[n-1-k])
                    if k_max not in K:
                        K.append(k_max)
                        if len(K) >= n // 2:
                            break  # Protection against excessive shifts
                else:
                    break  # All satisfied
                    
            except Exception:
                break  # If anything fails, try next n

    # Final polishing phase
    if best_h is not None and time.time() - t_start < budget_s - 5:
        poly_h = best_h.astype(float)
        c_old = np.correlate(poly_h, 1 - poly_h, mode='full')
        max_old = np.max(c_old)
        while True:
            max_idx = np.argmax(c_old)
            # Determine which index contributes to this overlap
            left = max_idx
            right = max_idx - (len(c_old) // 2)
            if right < 0:
                right += len(c_old)
            
            # Create A and B sets
            A = []  # indices where on gives 1, off gives 0 (contributes to max)
            B = []  # indices where on gives 0, off gives 1

            # Try changing one value to improve max
            swapped = False
            for i in range(len(poly_h)):
                new_idx = (i + max_idx - (len(c_old) // 2)) % len(poly_h)
                if poly_h[i] > 0.5 and poly_h[new_idx] < 0.5:
                    A.append(i)
                if poly_h[i] < 0.5 and poly_h[new_idx] > 0.5:
                    B.append(i)
            
            if A and B:
                i = A[0]
                j = B[0]
                if i != j:
                    # Perform swap
                    poly_h[i], poly_h[j] = poly_h[j], poly_h[i]
                    swapped = True
            
            if not swapped:
                break
            
            c_new = np.correlate(poly_h, 1 - poly_h, mode='full')
            max_val = np.max(c_new)
            if max_val < max_old:
                max_old = max_val
                c_old = c_new
            else:
                break
        
        # Final check
        final_c = np.correlate(poly_h, 1 - poly_h, mode='full')
        final_c5 = np.max(final_c) / len(poly_h) * 2.0
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = poly_h.astype(int)
    
    return (best_h.astype(float).tolist(), best_c5, best_n)
