def evaluate(h):
    """Evaluate the maximum overlap integral for discretized h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_cnt = corr.max()
    c5 = max_cnt * dx
    return c5

def continuous_refine(h0, time_budget):
    """Refine an integer solution to fractional using softmax gradient descent."""
    start_time = time.time()
    n = len(h0)
    h = h0.astype(float)
    # Normalize to sum to n/2
    h *= (n / 2) / np.sum(h)
    best_h = h.copy()
    best_max_cnt = np.correlate(h, 1-h, mode='full').max()

    # Softmax surrogate parameters
    alphas = [1, 5, 10, 20, 40, 80]
    max_steps = [200] * len(alphas)
    for alpha, t_steps in zip(alphas, max_steps):
        if time.time() - start_time > time_budget - 0.1:
            break
        eta0 = 0.2
        for t in range(t_steps):
            if time.time() - start_time > time_budget - 0.1:
                break

            # Compute all o_k
            o = np.zeros(2*n - 1)
            for k in range(1-n, n):
                s = 0
                if k >= 0:
                    s = np.sum(h[:n-k] * (1-h[k:]))
                else:
                    s = np.sum(h[-k:] * (1-h[:n+k]))
                o[k+n-1] = s
            # Subtract max for numerical stability
            o -= np.max(o)
            exp_o = np.exp(alpha * o)
            Z = np.sum(exp_o)
            p = exp_o / Z

            # Compute gradient
            grad = np.zeros(n)
            for k in range(1-n, n):
                if k >= 0:
                    grad[:n-k] += p[k+n-1] * (1 - h[k:])
                else:
                    grad[-k:] += p[k+n-1] * (1 - h[:n+k])

            eta = eta0 / np.sqrt(t+1)
            h_new = h - eta * grad
            # Projection onto [0,1]^n and sum = n/2
            h_new = np.clip(h_new, 0, 1)
            mu = np.sum(h_new) - n/2
            h_new -= mu / n
            h_new = np.clip(h_new, 0, 1)
            h = h_new.copy()
            
            # Check the actual max overlap
            corr = np.correlate(h, 1-h, mode='full')
            max_cnt = corr.max()
            if max_cnt < best_max_cnt:
                best_max_cnt = max_cnt
                best_h = h.copy()
    return best_h, best_max_cnt
