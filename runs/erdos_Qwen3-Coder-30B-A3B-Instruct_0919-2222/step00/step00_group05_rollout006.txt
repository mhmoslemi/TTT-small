import numpy as np
import cvxpy as cp
import time
from scipy import correlate

def evaluate_erdos_solution(h_values, dx=2.0/70):
    """Evaluate C5 for given h_values."""
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def build_mcormick_constraints(h, y, n, shifts):
    """Helper to build McCormick constraints for all pairs."""
    constraints = []
    for s in shifts:
        for i in range(n):
            if 0 <= i + s < n:
                # y[i,s] <= h[i]
                constraints.append(y[i, s + n - 1] <= h[i])
                # y[i,s] <= h[i + s]
                constraints.append(y[i, s + n - 1] <= h[i + s])
                # y[i,s] >= h[i] + h[i + s] - 1
                constraints.append(y[i, s + n - 1] >= h[i] + h[i + s] - 1)
                # y[i,s] >= 0
                constraints.append(y[i, s + n - 1] >= 0)
    return constraints

    prob = cp.Problem(objective, constraints)

        h_round = np.zeros_like(h_star)

            ones = np.random.choice(n, int(mass), replace=False, p=prob_h / np.sum(prob_h))
