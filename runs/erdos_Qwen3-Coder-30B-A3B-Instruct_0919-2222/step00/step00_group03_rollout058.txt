import numpy as np
from scipy.optimize import minimize
from scipy.sparse import diags
from scipy.sparse.linalg import spsolve
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value gives us the C5 constant
    c5 = np.max(correlation) * dx
    return c5

def initial_h_values(n_points):
    """Generate initial h values based on the problem setup"""
    # Use the initial construction as provided
    # For this problem, we'll scale the initial to have integral 1
    h = np.array([0.0]*n_points)
    mid = n_points // 2
    # Simple middle spike construction
    if n_points >= 10:
        h[mid-2:mid+3] = 0.5
    else:
        h[n_points//2] = 1.0
    # Normalize to ensure integral = 1
    integral = np.sum(h) * (2.0 / n_points)
    h = h / integral * 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Starting parameters
    n_points = 70  # Using reasonable number of points
    
    # Starting point construction
    h = initial_h_values(n_points).copy()
    
    # Optimize with Nelder-Mead using bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vals):
        # Normalize to ensure integral = 1
        dx = 2.0 / n_points
        h = np.array(h_vals)
        integral = np.sum(h) * dx
        if integral > 0:
            h = h / integral * 1.0
            
        # Compute C5
        c5 = evaluate_erdos_solution(h)
        return c5
    
    def constraint_eq(h_vals):
        dx = 2.0 / n_points
        return np.sum(h_vals) * dx - 1.0
    
    # Optimization setup
    cons = {'type': 'eq', 'fun': constraint_eq}
    method = 'L-BFGS-B'  # Robust method for bound constrained problems
    tol = 1e-8  # Tight tolerance
    
    # Initialize to current best
    current_h = h.copy()
    current_c5 = objective(current_h)
    
    # Try several initial points with different pattern to escape local optima
    for it in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation of solution vector
        h_rand = initial_h_values(n_points) + np.random.normal(0, 0.02, n_points)
        h_rand = np.clip(h_rand, 0, 1)
        
        # Normalize
        dx = 2.0 / n_points
        integral = np.sum(h_rand) * dx
        if integral > 0:
            h_rand = h_rand / integral * 1.0
        
        # Optimize
        try:
            result = minimize(objective, h_rand, method=method, bounds=bounds, 
                             constraints=cons, options={'ftol': tol, 'gtol': tol})
            if result.success:
                h_opt = result.x
                c5_opt = objective(h_opt)
                if c5_opt < current_c5:
                    current_c5 = c5_opt
                    current_h = h_opt.copy()
        except Exception:
            continue
    
    # Final refinement using a more refined local search
    if time.time() - start_time < budget_s * 0.98:
        # Try with L-BFGS-B another time with current best as start
        try:
            result = minimize(objective, current_h, method=method, bounds=bounds, 
                             constraints=cons, options={'ftol': 1e-10, 'gtol': 1e-10})
            if result.success:
                h_opt = result.x
                c5_opt = objective(h_opt)
                if c5_opt < current_c5:
                    current_c5 = c5_opt
                    current_h = h_opt.copy()
        except Exception:
            pass
    
    # Final check
    dx = 2.0 / n_points
    integral = np.sum(current_h) * dx
    current_h = current_h / integral * 1.0
    final_c5 = evaluate_erdos_solution(current_h)
    
    # Make sure the computed solution respects constraints
    h_final = np.clip(current_h, 0, 1)
    integral_final = np.sum(h_final) * dx
    h_final = h_final / integral_final * 1.0
    
    return (h_final, final_c5, n_points)
