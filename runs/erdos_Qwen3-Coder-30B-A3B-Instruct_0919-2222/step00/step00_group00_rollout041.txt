import numpy as np
import time
import random

def compute_correlation(h):
    """Compute full correlation of h with (1-h)"""
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    """Compute the C5 constant (maximum overlap integral) for given h"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    return corr.max() * dx

def apply_delta(c, h, idx, delta):
    """Apply delta change to h[idx] and update correlation c in place"""
    n = len(h)
    offset = n - 1
    
    # First term: for j from idx to n-1
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    
    # Second term: for j from 0 to idx
    for j in range(0, idx+1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(h, c, i, j):
    """Swap values at indices i and j and update correlation"""
    old_i, old_j = h[i], h[j]
    delta_i = 0 - old_i
    delta_j = 1 - old_j
    
    # Apply delta at i
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    
    # Apply delta at j
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def find_targeted_swap(c, h):
    """Find a swap that would reduce the maximum overlap"""
    n = len(h)
    offset = n - 1
    max_idx = np.argmax(c)
    max_shift = max_idx - offset  # Convert to actual shift value
    
    # Only consider large shifts whose values affect the maximum
    # For large positive or negative shifts, skip as they rarely affect maximum
    if max_shift == 0:
        return None
    
    # Try to find candidates in A and B sets for current max shift
    A = []  # Positions where h[i] = 1 and h[i + max_shift] = 0
    B = []  # Positions where h[j] = 0 and h[j + max_shift] = 1
    
    for i in range(n):
        if i + max_shift < 0 or i + max_shift >= n:
            continue
        if h[i] == 1 and h[i + max_shift] == 0:
            A.append(i)
        if h[i] == 0 and h[i + max_shift] == 1:
            B.append(i)          
    
    if len(A) > 0 and len(B) > 0:
        return A[random.randint(0, len(A)-1)], B[random.randint(0, len(B)-1)]
    else:
        return None

def hill_climb(h, n_iter, time_limit):
    """Perform hill climb optimization"""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    c = compute_correlation(h)
    current_c5 = c.max() * dx
    best_c5 = current_c5
    start_time = time.time()
    
    # Keep lists of indices for 0s and 1s
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] < 0.5]
    
    for _ in range(n_iter):
        if time.time() - start_time > time_limit - 1:
            break
            
        # Try targeted swap to reduce max
        target_swap = find_targeted_swap(c, h)
        if target_swap:
            i, j = target_swap
        else:
            # Fallback to random swap
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[random.randint(0, len(ones)-1)]
                j = zeros[random.randint(0, len(zeros)-1)]
            else:
                continue
                
        # Apply swap
        old_c5 = current_c5
        swap(h, c, i, j)
        
        # Update lists
        if h[i] == 0 and i in ones:
            ones.remove(i)
            zeros.append(i)
        elif h[i] == 1 and i in zeros:
            zeros.remove(i)
            ones.append(i)
            
        if h[j] == 0 and j in ones:
            ones.remove(j)
            zeros.append(j)
        elif h[j] == 1 and j in zeros:
            zeros.remove(j)
            ones.append(j)
            
        current_c5 = c.max() * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5

def upsample_h(initial_h_values, n):
    """Upsample initial_h_values to match n"""
    n_old = len(initial_h_values)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n
    x_old = np.arange(0, 2, dx_old)
    x_new = np.arange(0, 2, dx_new)
    h_new = np.interp(x_new, x_old, initial_h_values)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    """Run the optimization to find a better C5 bound"""
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    global initial_h_values
    
    # Define candidate n sizes
    n_values = [60, 80, 100, 120, 150]
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    max_iter_per_n = 50000
    
    # Trial with multiple n values to find best local minimum
    for n in n_values:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Initialize h: either upsample if needed, otherwise random binary
        dx = 2.0 / n
        h = np.random.rand(n)  # Initialize uniformly random first
        
        # Try to scale to correct average
        current_sum = h.sum()
        target_sum = n / 2.0
        if abs(current_sum - target_sum) > 1e-5:
            scale_factor = target_sum / current_sum
            h *= scale_factor
            h = np.clip(h, 0, 1)
        
        # Binary sparsification if needed to ensure sum constraint
        real_sum = h.sum()
        if abs(real_sum - target_sum) > 1e-5:
            diff = target_sum - real_sum
            # Find the index to increase or decrease
            indices = np.argsort(h)
            idx = indices[0] if diff < 0 else indices[-1]
            h[idx] += diff
            h = np.clip(h, 0, 1)
        
        h, c5 = hill_climb(h, max_iter_per_n, budget_s - (time.time() - start_time))
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        
        if best_c5 < 0.379:  # Early exit target
            break
    
    # Return best solution found
    if best_h is not None:
        return (best_h.tolist(), best_c5, best_n)
    else:
        return (np.zeros(100).tolist(), 0.381, 100)  # Fallback
