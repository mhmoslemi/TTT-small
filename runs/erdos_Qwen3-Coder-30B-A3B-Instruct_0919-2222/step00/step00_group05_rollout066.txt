import numpy as np
import cvxpy as cp
import time
from scipy import signal

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    correlations = np.correlate(h, 1 - h, mode="full")
    c5 = max(correlations) * dx
    return c5

def get_mcormick_constraints(h, y, n, s, i):
    # Introduce constraints for y[i,s] <= h[i], y[i,s] <= h[i+s], etc.
    if i + s < n and i + s >= 0:
        # y[i,s] <= h[i]
        yield y[i, s] <= h[i]
        # y[i,s] <= h[i+s]
        yield y[i, s] <= h[i+s]
        # y[i,s] >= h[i] + h[i+s] - 1
        yield y[i, s] >= h[i] + h[i+s] - 1
        # y[i,s] >= 0
        yield y[i, s] >= 0
    else:
        # Ignoring out-of-bound pairs as they are not used.
        pass

def build_and_solve_lp(h_vals, n_points):
    # Reset h_vals to be a full array (n_points) for vectorized operations
    h = cp.Variable(n_points)
    # Use 2D array for y variables, rows index i, cols index s
    y = cp.Variable((n_points, 2 * n_points - 1))
    t = cp.Variable()

    # Objective
    objective = cp.Minimize(t)

    # Constraints
    constraints = []

    # Box constraints on h
    constraints += [h[i] >= 0 for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]

    # Mass constraint
    constraints += [cp.sum(h) == n_points / 2]

    # t >= 0
    constraints += [t >= 0]

    # McCormick constraints: for each i, s, the modified variable y[i,s] must
    # satisfy all the constraints for the bilinear term h[i] * h[i+s]
    # In the two-d structure, index j=s+n_points-1 maps s to column 0..2*n-2
    for s in range(-n_points + 1, n_points):
        for i in range(n_points):
            if 0 <= i + s < n_points:
                idx = s + n_points - 1  # maps s to index in y array
                #y[i, s] <= h[i], y[i, s] <= h[i+s], y[i, s] >= h[i]+h[i+s]-1, y[i, s] >= 0
                constraints.append(y[i, idx] <= h[i])
                constraints.append(y[i, idx] <= h[i+s])
                constraints.append(y[i, idx] >= h[i] + h[i+s] - 1)
                constraints.append(y[i, idx] >= 0)

    # Constraint on Q_s
    for s in range(-n_points + 1, n_points):
        idx = s + n_points - 1
        Q_s = cp.sum(y[:, idx])
        upper_bound = n_points / 2 - t
        constraints.append(Q_s >= upper_bound)

    # Construct the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        raise RuntimeError(f"Could not solve LP; status: {prob.status}")

    result_h = h.value
    return result_h, t.value

def flip_random(h_vals, n_points):
    """sample a binary solution with exactly n_points/2 ones, using h_vals as probabilities"""
    # Normalize
    probs = np.array(h_vals)
    probs = probs / np.sum(probs)
    # Do replacement sampling
    ones_indices = np.random.choice(n_points, size=int(n_points / 2), replace=False, p=probs)
    h = np.zeros(n_points)
    h[ones_indices] = 1
    return h

def round_deterministic(h_vals, n_points):
    """Round to n/2 1's with the largest values"""
    h = np.array(h_vals)
    # Sort indices
    sorted_indices = np.argsort(h)[::-1]
    h_new = np.zeros(n_points)
    h_new[sorted_indices[:int(n_points / 2)]] = 1
    return h_new

def targeted_swap(h_vals, n_points):

    h = np.array(h_vals, dtype='int')

    for _ in range(max_iterations):
        # Recompute all correlations
        correlations = np.correlate(h, 1 - h, mode="full") * dx
        s_max = np.argmax(correlations)
        # The shift that gives maximum correlation (correlations at index (n-1) + s_max = (n-1) + (s_max - (n-1)) = s_max)
        # We want: s_max = max_offset + s_max in (rdx1 = 1, 2,... 2*n-2)
        # Actually, make sure that all entries in corr are indexed by shifting from -n+1 to n-1 
        s_max_actual = s_max - (n_points - 1) # s_max was retrieved from 0..(2n-2)
        corr_full = correlations  
        
        # Find best move
        best_move = (None, None)
        best_delta = 0
        h_backup = h.copy()
        for i in range(n_points):
            if h[i] == 1: # it's a '1'
                j = (i + s_max_actual) % n_points
                if h[j] == 0: # this can contribute
                    # Try to find a good adjustment target
                    for k in range(n_points):
                        if h[k] == 0:
                            l = (k + s_max_actual) % n_points
                            if h[l] == 1:
                                # Potential swap
                                # Update both correlations
                                # Do not compute the whole, just impact on the worst.
                                # The change in correlation when swapping h[i] and h[k]:
                                delta = 0
                                # foreach i in indices contributes to the max shift:
                                # current: h[i]*(1-h[i+s])  
                                # after swap: h[k]*(1-h[k+s])
                                # contributions we evaluate for those indices:
                                # for i only: 0 -> 1; for k: 1 -> 0 
                                # For the shift s_max:
                                # original i contributes: h[i]*(1-h[i+s_max]) = 1*(1-0) = 1 (since h[i]=1 and h[i+s_max]=0)
                                # after swap: h[k]*(1-h[k+s_max]) = 0*(1-1) = 0
                                # original k contributes: h[k]*(1-h[k+s_max]) = 0*(1-1) = 0
                                # after swap: h[i]*(1-h[i+s_max]) = 1*(1-0) = 1
                                # so corr_s_max changes by -1 (we set h[i] from 1 to 0 and h[k] from 0 to 1)
                                # But for other indices i', shift s_max:
                                # before swap: h[i']*(1-h[i'+s_max])  
                                # after swap: h[i']*(1-h[i'+s_max])  -> same if h[i'] unchanged and i'=i, i' != k
                                # This is simplified, correctly done below:
                                # Before swap: 
                                #     At positions i: h[i]*1 = 1; c_s_max += 1 
                                #     At positions k: h[k]*0 = 0; c_s_max += 0 
                                # After swap:  
                                #     At positions i: h[i]*0 = 0; c_s_max -= 1
                                #     At positions k: h[k]*1 = 1; c_s_max += 1
                                # net change in c_s_max is: -1+1=0.
                                # But also at the "shifted" version:
                                #   i + s_max should be 0 before swap and 1 after (looking for h[i + s_max] = 0 -> 1) 
                                #   k + s_max should be 1 before swap and 0 after (h[k + s_max] = 1 -> 0)
                                #   before shuffle: 
                                #       One 1 at position i+s_max and 0 at position k+s_max
                                #   after shuffle:
                                #       One 0 at position i+s_max and 1 at position k+s_max 
                                #   So the change
                                #       Total: corr_s_max += -1 + 1 = 0.
                                # The same reasoning leads to: All other shifts should remain same, 
                                # so it's impossible just by swapping these points to decrease the worst shift.
                                # But let’s recompute and assert:
                                # Simpler Method:
                                # alternative approach is to compare full correlation vectors after each swap (if feasible).
                                h_copy = h.copy()
                                # Swap h[i], h[k]
                                h_copy[i] = 0
                                h_copy[k] = 1
                                corr_new = np.correlate(h_copy, 1 - h_copy, mode="full") * dx
                                corr_old = np.correlate(h, 1 - h, mode="full") * dx
                                delta = corr_old[s_max] - corr_new[s_max]
                                if delta > best_delta:
                                    best_delta = delta
                                    best_move = (i, k)
        if best_delta <= 0:
            break
        else:
            # Apply best move
            i, k = best_move
            h[i] = 0
            h[k] = 1
    return h
