import numpy as np
from scipy.optimize import differential_evolution
import time

def project_to_capped_simplex(v, target_sum):
    """
    Euclidean projection of v onto {x : 0 <= x_i <= 1, sum x_i = target_sum}
    """
    v = np.array(v)
    n = len(v)
    if target_sum <= 0:
        return np.zeros_like(v)
    if target_sum >= n:
        return np.ones_like(v)
    
    # Sort in descending order
    sorted_idx = np.argsort(-v)
    sorted_v = v[sorted_idx]
    
    # Water-filling algorithm
    cumsum = np.cumsum(sorted_v)
    threshold_idx = -1
    for i in range(n):
        if i == n-1:
            # All elements should be at threshold
            if sorted_v[i] >= target_sum / n:
                threshold_idx = i
                break
        else:
            # Check if we can set elements up to i at a common value
            threshold = (target_sum - cumsum[i]) / (n - i - 1)
            if sorted_v[i+1] <= threshold <= sorted_v[i]:
                threshold_idx = i
                break
    if threshold_idx == -1:
        # All elements >= target_sum / n or <= target_sum / n
        threshold = target_sum / n
        threshold_idx = n - 1
    if threshold_idx == -1:
        threshold = target_sum / n
    else:
        threshold = (target_sum - cumsum[threshold_idx]) / (n - threshold_idx - 1)
    # With diminished boundary
    threshold = max(0, min(1, threshold))
    result = np.zeros_like(v)
    for i in range(n):
        if i <= threshold_idx:
            result[sorted_idx[i]] = 1.0
        else:
            result[sorted_idx[i]] = max(0, min(1, v[sorted_idx[i]] - threshold))
    # May slightly violate sum due to numerical issues, so renormalize
    current_sum = np.sum(result)
    if not np.isclose(current_sum, target_sum, atol=1e-8):
        # Bin search a correction
        safety = 1e-12
        result = np.clip(result, 0, 1)
        diff = current_sum - target_sum
        if abs(diff) < safety:
            return result
        # Try to push some up, some down
        tmp = np.copy(result)
        # Increase 1's to compensate for missing weight
        # We decrement diff (i.e. grow) adapted from others
        # Simple procedure: just modulate based on ones vs non-ones
        indices_0 = np.where(result < 1 - safety)[0]
        indices_1 = np.where(result > safety)[0]
        if len(indices_0) > 0 and len(indices_1) > 0:
            delta = diff / len(indices_0)
            for idx in indices_0:
                # Increase by delta, but cap at 1
                tmp[idx] = min(1.0, tmp[idx] + delta)
            for idx in indices_1:
                tmp[idx] = max(0.0, tmp[idx] - delta)
        # Renormalize
        tmp_sum = np.sum(tmp)
        if abs(tmp_sum) > safety:
            scale = target_sum / tmp_sum
            tmp = tmp * scale
        result = tmp
        result = np.clip(result, 0, 1)
    return result

def compute_overlaps(a, w, dtype='float64'):
    """
    Compute G(k) = \int h(x) h(x+k) dx for each k
    a are the heights (len M)
    w are the widths (len M) (sum(w) = 2.0)
    Returns: vector of length 2*N - 1 (kernel from -N+1 to N-1)
    """
    M = len(a)
    N = 200  # output size; ensures resolution of evaluation
    dx = 2.0 / N
    # compute s_i = cumsum(w)
    s = np.cumsum(w)
    s = np.concatenate([[0], s])
    # Shift points from [0, 2] to [0, N] to discretize:
    #   We'll evaluate G(k) on several discrete k values.
    #   Since domain is [0,2], k can range from -2 to 2.
    #   But since the function is periodic, it suffices to consider shifts
    #   in [0, 2) by padding.
    K = 200  # How many shift samples in [0,2)
    shift_points = np.linspace(0, 2, K, endpoint=False)
    # Offset for the discrete indices
    result = np.zeros(2 * N - 1)
    # Discretization error will not significantly affect accuracy,
    # especially with high-resolution shape.
    for k_idx, k in enumerate(shift_points):
        # compute G(k)
        total = 0.0
        for i in range(M):
            for j in range(M):
                start_i = s[i]
                end_i = s[i+1]
                start_j = s[j]
                end_j = s[j+1]
                # Simulate periodic shift:
                ext_start_j = start_j - k
                ext_end_j = end_j - k
                overlap = max(0.0, min(end_i, ext_end_j) - max(start_i, ext_start_j))
                if overlap > 0:
                    total += a[i] * a[j] * overlap
        # map the k to the full-size correlation array:
        # relative shift index is k / dx = k * N / 2
        abs_idx = int(k * N / 2)
        if 0 <= abs_idx < len(result):
            result[abs_idx] += total / K
    return result

def eval_single_point_block(h):
    """
    Evaluate the full objective from h as a step function
    h is list of dtype float (N samples), h[i] in [0, 1], sum h * dx = 1
    Compute C5 = max_k \int h(x) (1 - h(x+k)) dx
    """
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - np.array(h), mode='full')
    corr = corr * dx  # rescale to reflect measure dx
    assert len(corr) == 2*N - 1
    # Only look at lags from 0 to 2*N-2
    # But we want to observe all positive lags
    # Since h is viewed on [0, 2], and we're periodic on [0, 2],
    # It is represented as a circular function in C^([0,2]), for which
    # the kernel looks like
    #   valid: lag index [0, 2*N-2]
    #   lag k = -N+1+shift, shift from 0 to 2*N-2
    # But we can still take maximum over valid ranges or k:
    # we are panel vers definition that's run on [0,2], process as though k ∈ [0, 2)
    # But offset by 1, so shift always [1, N]
    # Rather, best to just compute maximum directly:
    corr = corr[1:N]  # take only non-negative lags for [0,2)
    # okay, now start from index 1, go N indices ahead.
    # Now observe all combinations:
    # For any valid lag k (in [0,2) or [0, N*dx), that's (0..N-1) in terms of index)
    # We want max_{k} ∫ h(x) (1 - h(x+k)) dx = max_k (1 - ∫ h(x) h(x+k) dx)
    # Assume integral fixed at 1
    max_val = 0.0
    for i in range(len(corr)):
        # corr[i] corresponds to integration over lag i corresponding to k = i*dx
        # But actual k can be negative, so use symmetric stuff:
        # We can include shifts from 0 to N.
        # So we consider the actual double-side periodic window.
        # The relevant intervals are periodic expansion on [0,2], which maps
        # to [-N+1, N-1] in full correlation indices (i.e., index = lag + (N-1)
        # Valid range of indices of interest = full convolution = [0, 2*N-2]
        # We want max\int h(x) * (1 - h(x+l)) dx = max (1 - \int h(x) h(x+l) dx).
        # Actually, the periodic function syntax makes
        #   shift by 0: h(x) times h(x) -> 1
        #   shift by 1: h(x) times h(x+1) -> large
        #   shift by N/2: h(x) times h(x+N/2) -> fairly small
        # So valid max range for k is [0, N], not symmetric
        # But we are allowed to only consider 0 to N-1 index here,
        # i = 0 to N-1 in terms of lag index.
        # Lag i is mapped to k = i*dx for i = 0 to N-1, i.e., lag from 0 to 2-dx.
        # But they are defined through [0, N] where each lag is in [0, 2]
        # Define     l = 0 to N-1
        #            k = l * dx = l * (2/N)
        # Now evaluate the given integration at this lag
        # The main thing:
        # Coherent part of    ∫ h(x) * (1 - h(x+k)) dx for k = l * dx
        # Is the same as    ∫ h(x) * h(x+k) dx = ∫ h(x) h(x+k) dx  
        # And we subtract it.
        # But in the sum, we have 2*N-1 terms, which includes from 'negative lags'
        # But those may not be strictly necessary for describing full correlation behavior
        # Recall that the support is [0,2], so shift may be placed onto Gaussian filter
        # structure or equivalent. As part of full convolution, indices should be used.
        # In general manner:
        # We have a periodic sequence, z_n = h(x_n) where x_n = 2 n/N.
        # Correlation at lag k is:
        # C[k] = Σ h_n h_{n+k} for integer k and periodic wrap
        # We are here computing the full correlation. So if we did it with FFT based on signal,
        # it's 2*N - 1 positions, as given by the general FFT of x with itself.
        # Try to rederive and understand: we look over full circular convolution.
        # for shift l, we integrate h(x) * (1 - h(x+l)) dx
        # i.e. 1 - integral of h(x) h(x+l) dx
        # So:
        # C[l] = integral of h(x) h(x+l) dx * dx
        # Want to find the l (in [0, 2]) where 1 - C[l] is max.
        # Therefore we minimize C[l] (i.e., maximize integral of h(x) h(x+l)).
        # So we must fix valid maximum:
        # For l=0 to N-1, Get the lag 0 to N-1.
        # Here we have full periodic sequence h is [h[0], ..., h[N-1]]
        # We pad 0 at left and right
        # Period index shifted up from zero, this is computed via circular to linear
        # Lag=0, 1, ..., N-1 are normative
        # The k must be within [0, 2).
        # Let's maintain index as [0 to 2*N-2]
        # We compute maximum over l=0,...,N-1 for: 1 - C[l]
        # Would look like:
        # c = np.correlate(h, 1-h, mode='full')
        # But mode might not work due to wrap.
        # So compute the actual integration again.
        # c = corr
        # max_{k} = max_{l} (1 - c[l]) = 1 - min(c[l])
        max_val = max(max_val, corr[i])
    return 1. - max_val

def eval_objective(params):
    """
    Global DE objective:
    params = [u1,...,uM, v1,...,vM] where
    w_i = 2 * softmax(u_i)
    a_i = sigmoid(v_i)
    Returns the objective with penalty for integral constraint.  
    """
    λ = 1000.0
    M = 12
    try:
        if len(params) != 2 * M:
            # Invalid
            return 1e10
        u = params[:M]
        v = params[M:]
    except Exception:
        return 1e10

    w = np.exp(u) / np.sum(np.exp(u)) * 2.0  # normalization to sum 2
    a = 1.0 / (1.0 + np.exp(-np.array(v)))  # sigmoid to [0,1]
    if np.any(a < 0) or np.any(a > 1):
        return 1e10
    if np.any(w < 0):
        return 1e10

    # Evaluate integral
    integral = np.dot(a, w)
    penalty = λ * abs(integral - 1.0)

    # Evaluate min{G(k)} for several k values
    # very expensive, so we'll do it less directly – simpler check:
    # compute or estimate overlap via block model
    try:
        # approximate with a bit of precomputed stuff
        corr = compute_overlaps(a, w)  # this gives discrete map of G(k)
        # get range [0,N) since we only care about positive lags
        # note that it returns full. This is hard: full convolution from -N to N
        corr = corr[:len(corr)//2 + 1]
        eval_val = np.min(corr)
    except Exception:
        eval_val = 1e10

    return eval_val + penalty

def eval_h_function(h_values):
    """
    Evaluate the true C5 bound for solution h (given as list)
    """
    dx = 2.0 / len(h_values)
    C = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    C = C * dx
    max_val = np.max(C[:len(C)//2 + 1])
    return 1.0 - max_val

    )
    if time.time() - t_start > 0.7 * budget_s:
        return h0.tolist(), eval_h_function(h0), len(h0)
    
    # Optional secondary run
    best_params = res_de.x.reshape(-1)
    # Rescale via integral constrain.
    u = best_params[:M]
    v = best_params[M:]
    w_all = np.exp(u) / np.sum(np.exp(u)) * 2.0  # normalization to sum 2
    a_all = 1.0 / (1.0 + np.exp(-np.array(v)))  # sigmoid to [0,1]
    integral = np.dot(a_all, w_all)
    a_all = a_all / integral
    # project if too large heights.
    a_all = np.clip(a_all, 0, 1)
    # Now sample
    s = np.cumsum(w_all)
    s = np.concatenate([[0], s])
    h = np.zeros(N)
    for i in range(N):
        x = i * dx
        for j in range(M):
            if s[j] <= x < s[j+1]:
                h[i] = a_all[j]
                break
    
    # Subgradient polish: with the grid h above
    # Apply projection also
    final_h = h.copy()
    
    Δ = 0.05  # for small finite diff
    α0 = 0.2 # original α
    max_iter = min(8000, (budget_s - (time.time() - t_start)) // 0.01)
    best = float('inf')
    h_best = None
    
    for iter_num in range(max_iter):
        if time.time() - t_start > 0.95 * budget_s:
            break
        α = α0 / np.sqrt(iter_num + 1 + 100)
        
        # Compute 1-h, return in same size as 1-h
        corr_kernel = np.correlate(final_h, 1.0 - final_h, mode='full')
        corr_kernel = corr_kernel * dx  # preserve units
        # Now slice out positives only, up to around N points
        corr_kernel = corr_kernel[:len(corr_kernel)//2 + 1]  # from lag=0 to N
        max_ind = np.argmax(corr_kernel)
        # whoops, we did: `1 - integral of h(x) h(x+l) dx`
        # so max of `corr_kernel` is the max over lags of the *overlap*
        # So 1 - the max value is the min of the overlap function,
        # This leads to the correct `C5 = max_k integral h(x) (1-h(x+k)) dx`
        c5 = 1.0 - corr_kernel[max_ind]
        
        if c5 < best:
            best = c5
            h_best = final_h.copy()
        
        # Gradient elements: derivative of C_k w.r.t. h_i
        # ∂C_k/∂h_i = (1 - h_{i+k}) - h_{i-k}  for valid i.
        # For each active lag k*, we backpropagate its gradient:
        # G = sum over k* of G_k = ∫ h(x) (1-h(x+k)) dx := 1 - ∫ h(x) h(x+k) dx
        # We want dG/dh_i = - d/dh_i integral of h(x) h(x+k) = - [sum over j of  h_j · δ_{j+i kiss k}] = - h_{i+k} or - h_{i-k}]
        # but not just -h_{i+k} but - h_{i+k} for i + k < N.
        # Same idea but more carefully:
        g = np.zeros_like(final_h)
        # Threshold to prevent loop blowup, just in case
        actual_k = max_ind
        # k in [0, N) range where k * dx = actual_k * dx
        k = actual_k
        # += for i in [0,...,N - k)
        # but shifting from i to i+k, apply 1 - h[i+k] 
        # so h[i] got a linear effect from h[i+k]
        # and vice versa as derivative correspondences exist.
        # If h_i variable impacts values at later indices, their contribution
        # is 1 - h_{i+k}. And we're at fixed k.
        for i in range(N - k):
            # i to i+k, total contribution to G by the change of h_i
            # Similar to the forward pass thinking:
            # If we vary h_i,
            #   h_i contributes to terms      u = h_i * (1 - h_{i+k})
            # so the derivative is 1 - h_{i+k}  
            # To compute dC_k/dh_i
            # we vary h_i -> costs some dC_k/dh_i term (if++) 
            # It affects value at i+k = i+k term of integral.
            # Gradient update based only on most active
            if i + k < N:
                g[i] += 1 - final_h[i+k]
            if i - k >= 0:
                # need additional this time when inner prev link works 
                # activated by shift variable i being impacted by h[i-k] which also contributes
                g[i] -= final_h[i-k]
        
        # Update h:
        final_h = final_h - α * g

        # Project
        final_h = np.clip(final_h, 0, 1)
        # Project sum to N/2 via capped simplex
        x = project_to_capped_simplex(final_h, N / 2)
        final_h = x

    # Return best partner seen either by search or locally
    if h_best is not None:
        return h_best.tolist(), best, N
    else:
        # fall back to simple discretization
        h0 = h0[:N]
        return h0.tolist(), eval_h_function(h0), len(h0)
