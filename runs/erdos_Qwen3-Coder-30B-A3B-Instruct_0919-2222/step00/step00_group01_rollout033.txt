import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    # Evaluate using the specified correlation method
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute full correlation between h and (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum correlation scaled by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    end_time = start_time + budget_s * 0.99
    
    # Use a moderate resolution
    n_points = 401  # Odd number for better symmetry
    dx = 2.0 / n_points
    
    # Initialize with a good baseline: cosine^2 shape
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    h = np.cos(np.pi/2 * (x - 1)) ** 2
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h  # Normalize to sum = n_points / 2
    h = np.clip(h, 0, 1)  # Ensure within [0,1]
    
    # Initial evaluation
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Hill climbing loop
    iteration = 0
    max_iter = 100000  # Safe upper bound
    
    while iteration < max_iter and time.time() < end_time:
        # Randomly select two different indices
        i, j = np.random.choice(n_points, size=2, replace=False)
        
        # Compute maximum allowable delta to move mass
        delta_max = min(h[i], 1 - h[j])
        # Select a random delta
        delta = np.random.random() * delta_max * 0.1  # Small moves
        
        # Update the values
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Ensure constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to maintain sum constraint
        sum_new = np.sum(h_new)
        h_new = h_new * (n_points / 2) / sum_new
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept if better
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            
        iteration += 1
    
    # Final evaluation to ensure correctness
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return best solution within budget
    return (best_h.tolist(), final_c5, n_points)
