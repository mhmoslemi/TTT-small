import random

    np.random.seed(seed)
    
    def evaluate(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full') * dx
        # Return the maximum correlation value
        return np.max(corr)
    
    def objective(h):
        return evaluate(h)
    
    n_points = 200  # Using a reasonable number of points
    
    # Starting point: initial_h_values if available, else uniform distribution
    if 'initial_h_values' in globals():

        # Replicate to match required n_points

        elif len(h_start) > n_points:
            h_start = h_start[:n_points]
    else:
        # Default: uniform distribution
        h_start = np.full(n_points, 0.5)
    
    # Function to ensure sum constraint
    def enforce_sum_constraint(h):
        h = np.array(h)
        current_sum = np.sum(h) * (2.0 / len(h))  # dx = 2/n, so actual sum = current_sum*dx*n/2 = current_sum
        scale = (n_points / 2) / current_sum  # Target sum is n_points/2 because dx = 2/n_points
        h = h * scale
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        return h
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    # Simple local search
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        # Create a neighbor by perturbing a few random elements
        h_trial = best_h.copy()
        n_changes = random.randint(1, 10)
        indices = random.sample(range(n_points), n_changes)
        for idx in indices:
            delta = random.uniform(-0.1, 0.1)
            h_trial[idx] += delta
        # Enforce constraints
        h_trial = enforce_sum_constraint(h_trial)
        h_trial = np.clip(h_trial, 0, 1)
        
        c5_trial = evaluate(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial.copy()

    # Final check
