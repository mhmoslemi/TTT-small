import numpy as np
import time
from scipy.signal import correlate
import math

def evaluate_h(h):
    """Evaluate overlap integral for given h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = corr.max()
    return max_overlap * dx

def update_correlation(corr, h, i, j, s):
    """Fast update of correlation for binary swap at positions i,j with shift s"""
    # Assume s >= 0
    # Adjust indices to handle wrap-around
    if i + s < len(h):
        corr[s + len(h) - 1] -= 1  # Original h[i] = 1, h[i+s] = 0
    if j + s < len(h):
        corr[s + len(h) - 1] += 1  # Swapped h[j] = 1, h[j+s] = 0
    return corr

def binary_local_search(h, max_time_s):
    n = len(h)
    start_time = time.time()
    best_h = h.copy()
    best_count = np.correlate(h, 1 - h, mode="full").max()
    
    def run_descents():
        nonlocal best_h, best_count
        h_current = h.copy()
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        max_count = corr.max()
        
        # Track which indices are mismatched for each shift
        # For binary vectors, we can quickly compute mismatch sets
        shift_index = np.argmax(corr)
        shift_k = shift_index - (n - 1)
        
        max_iters = 2 * n
        no_improvement_iter = 0
        iter_no = 0

        while iter_no < max_iters and time.time() - start_time < max_time_s - 0.1:
            iter_no += 1
            # Find actual mismatched positions
            if shift_k >= 0:
                M1 = np.where((h_current[:-shift_k] == 1) & (h_current[shift_k:] == 0))[0]
                M0 = np.where((h_current[:-shift_k] == 0) & (h_current[shift_k:] == 1))[0]
            else:
                # shift_k < 0 implies a negative shift
                M1 = np.where((h_current[-shift_k:] == 1) & (h_current[:-shift_k] == 0))[0]
                M0 = np.where((h_current[-shift_k:] == 0) & (h_current[:-shift_k] == 1))[0]
            
            if len(M1) > 0 and len(M0) > 0:
                # Pick a random swap (i,j)
                i = np.random.choice(M1)
                j = np.random.choice(M0)
                # Do swap
                h_current[i], h_current[j] = h_current[j], h_current[i]
                # Update correlation
                if shift_k >= 0:
                    corr[shift_k + n - 1] += 2
                else:
                    corr[-shift_k + n - 1] += 2

                # Re-evaluate correlation after swap
                corr_new = np.correlate(h_current, 1 - h_current, mode="full")
                new_max = corr_new.max()
                if new_max <= max_count:
                    max_count = new_max
                    corr = corr_new
                    shift_index = np.argmax(corr)
                    shift_k = shift_index - (n - 1)
                    no_improvement_iter = 0
                    if new_max < best_count:
                        best_count = new_max
                        best_h = h_current.copy()
                else:
                    # Revert swap if not beneficial
                    h_current[i], h_current[j] = h_current[j], h_current[i]
                    if shift_k >= 0:
                        corr[shift_k + n - 1] -= 2
                    else:
                        corr[-shift_k + n - 1] -= 2
            else:
                # Fall back to random balanced swap
                ones = np.where(h_current == 1)[0]
                zeros = np.where(h_current == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = np.random.choice(ones)
                    j = np.random.choice(zeros)
                    h_current[i], h_current[j] = h_current[j], h_current[i]
                    corr_new = np.correlate(h_current, 1 - h_current, mode="full")
                    new_max = corr_new.max()
                    if new_max <= max_count:
                        max_count = new_max
                        corr = corr_new
                        shift_index = np.argmax(corr)
                        shift_k = shift_index - (n - 1)
                        no_improvement_iter = 0
                        if new_max < best_count:
                            best_count = new_max
                            best_h = h_current.copy()
                    else:
                        # Revert swap
                        h_current[i], h_current[j] = h_current[j], h_current[i]
                        if shift_k >= 0:
                            corr[shift_k + n - 1] -= 2
                        else:
                            corr[-shift_k + n - 1] -= 2
                no_improvement_iter += 1
                if no_improvement_iter > n:
                    break
        return best_h, best_count

    # Do five restarts
    for _ in range(5):
        _ = run_descents()
        if time.time() - start_time > max_time_s - 0.1:
            break

    return best_h, best_count

def simulated_annealing_binary(h, T_max=0.5, T_min=1e-4, max_iter=5000):
    """Run simulated annealing starting from binary h"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    current_count = corr.max()
    best_count = current_count
    best_h = h.copy()
    
    T = T_max
    alpha = (T_min / T_max) ** (1 / max_iter)
    for _ in range(max_iter):
        # Generate a random balanced swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Swap i and j
        h[i], h[j] = h[j], h[i]
        corr_new = np.correlate(h, 1 - h, mode="full")
        new_count = corr_new.max()
        delta = new_count - current_count
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            current_count = new_count
            corr = corr_new
            if current_count < best_count:
                best_count = current_count
                best_h = h.copy()
        else:
            # Revert the swap
            h[i], h[j] = h[j], h[i]
            corr = corr_new  # it will be old_corr since old was restored via revert
        T *= alpha
        
    return best_h, best_count

def proj_l2(v, target_sum):
    """Water-filling projection onto ball"""
    v = np.array(v)
    
    def Phi(lambda_val):
        r = np.clip(v - lambda_val, a_min=0, a_max=1)
        return np.sum(r) - target_sum
    
    # Binary search
    a = np.min(v) - 1
    b = np.max(v) + 1
    eps = 1e-8
    for _ in range(40):
        mid = (a + b) / 2
        phi = Phi(mid)
        if phi > 0:
            a = mid
        else:
            b = mid
    lambda_ = (a + b) / 2

    result = np.clip(v - lambda_, a_min=0, a_max=1)
    return result

def soft_max_gradient_step(h, corr, alpha, lr=0.2):
    """Update h via gradient descent on soft_max"""
    # Generate gradient of each shift
    n = len(h)
    grad_total = np.zeros_like(h)
    
    for k in range(-n + 1, n):
        # compute \partial o_k(h) / \partial h_i
        shift_ind = k + n - 1
        grad_k = np.zeros_like(h)
        # source of 1: (1 - h_{i+k})
        if k >= 0:  # Positive shift
            # front is i=0 to n-k-1
            # back is h_{i+k}
            grad_k[:-k] = h[k:] * -1
        else:  # Negative shift
            grad_k[-k:] = h[:-k] * -1
        # add the 1 at i
        grad_k += 1
        # the gradient for shift k
        grad_total += np.exp(alpha * corr[shift_ind]) * grad_k
    
    # Normalize by the sum of exps
    grad_total /= np.sum(np.exp(alpha * corr))
    # Gradient step
    h_new = h - lr * grad_total
    # Project onto feasible set (box + sum)
    target_sum = n / 2
    h_proj = proj_l2(h_new, target_sum)
    return h_proj

def continuous_refinement(h_binary, max_time_s):
    """Refine binary h to fractional"""
    n = len(h_binary)
    h = h_binary.astype(float)
    start_time = time.time()

    # Use at most 5000 steps
    steps_per_lr = 500
    
    alpha_schedule = [1, 5, 10, 20, 40, 80]
    
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    best_count = corr.max()
    
    for alpha in alpha_schedule:
        num_steps = int(200 * (alpha / 10.0))
        lr = 0.2 / np.sqrt(alpha)
        for _ in range(num_steps):
            if time.time() - start_time > max_time_s - 0.1:
                break
            # Compute gradients wrt alpha
            h = soft_max_gradient_step(h, corr, alpha, lr)
            corr = np.correlate(h, 1 - h, mode="full")
            new_count = corr.max()
            if new_count < best_count:
                best_count = new_count
                best_h = h.copy()
                  
    return best_h, best_count

def generate_random_balanced_vector(length):
    """Generate a balanced binary vector of given length."""
    half = length // 2
    arr = [1] * half + [0] * (length - half)
    np.random.shuffle(arr)
    return np.array(arr)
