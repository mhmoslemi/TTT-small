def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap constant for given step function h."""

    corr = correlate(h, 1 - h, mode='full')
    # Take the maximum correlation value (scaled by dx)
    c5 = np.max(corr) * dx
    return c5
