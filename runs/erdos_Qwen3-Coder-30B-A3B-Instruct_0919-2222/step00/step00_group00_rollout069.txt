```python
import time
import numpy as np
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    def solve_milp_lazy(n, K0=None, max_iters=20):
        dx = 2.0 / n
        if K0 is None:
            K0 = max(1, n // 4)
        
        # Variables
        x = cp.Variable(n, boolean=True)
        t = cp.Variable()
        
        # Symmetry breaking constraints
        constraints = [
            x[0] == 1,
            cp.diff(x[:n//2]) <= 0,  # Non-increasing prefix
            cp.sum(x) == n // 2
        ]
        
        # For each k in active set K
        y_vars = {}
        c_constraints = []
        
        K = list(range(1, K0 + 1))
        for k in K:
            y_vars[k] = cp.Variable(n - k, boolean=True)
            # McCormick envelopes
            for i in range(n - k):
                constraints.extend([
                    y_vars[k][i] <= x[i],
                    y_vars[k][i] <= x[i + k],
                    y_vars[k][i] >= x[i] + x[i + k] - 1
                ])
            # Overlap constraint
            c_k = cp.sum(x[:n-k]) - cp.sum(y_vars[k])
            c_constraints.append(c_k <= t)
        
        constraints.extend(c_constraints)
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=budget_s/2)
        
        if prob.status in [cp.OPTIMAL, cp.OPTAINED]:
            x_sol = np.round(x.value).astype(int)
            t_opt = t.value
        else:
            # Fallback to incumbent
            x_sol = np.round(x.value).astype(int)
            t_opt = t.value
            
        # Lazy constraint loop
        iter_count = 0
        while iter_count < max_iters:
            # Compute full correlation
            x_full = x_sol
            corr = correlate(x_full, 1 - x_full, mode='full')
            max_corr = np.max(corr)
            
            if max_corr <= t_opt + 1e-6:
                break
            
            # Find violating shift
            violating_k = np.argmax(corr) - (len(corr) // 2)
            if violating_k not in K:
                # Add new shift
                K.append(violating_k)
                # Add constraints for new k
                k = violating_k
                y_vars[k] = cp.Variable(n - abs(k), boolean=True)
                # McCormick
                start = 0 if k > 0 else abs(k)
                end = n - abs(k) if k > 0 else n
                for i in range(start, end):
                    constraints.extend([
                        y_vars[k][i - start] <= x[i],
                        y_vars[k][i - start] <= x[i + k],
                        y_vars[k][i - start] >= x[i] + x[i + k] - 1
                    ])
                # Overlap
                c_k = cp.sum(x[start:end]) - cp.sum(y_vars[k])
                c_constraints.append(c_k <= t)
                constraints.extend(c_constraints)
                # Re-solve
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.CBC, verbose=False, max_seconds=budget_s/4)
                if prob.status in [cp.OPTIMAL, cp.OPTAINED]:
                    t_opt = t.value
            else:
                break
            iter_count += 1
            if time.time() - t_start > budget_s - 5:
                break
                
        return x_sol, t_opt * dx, n

    def search_pattern(n_points):
        h_best, c5_best, n_best = solve_milp_lazy(n_points)
        return h_best, c5_best, n_points

    # Run for various grid sizes
    best_sol = None
    best_c5 = float('inf')
    budget_per_trial = budget_s // 4
    
    for n in [80, 120, 160]:
        if time.time() - t_start > budget_s - 5:
            break
        h, c5, n_pts = search_pattern(n)
        if c5 < best_c5:
            best_c5 = c5
            best_sol = (h, c5, n_pts)

    # Optional polishing
    if best_sol is not None:
        h, c5, n = best_sol
        h_polished = h.copy()
        corr = correlate(h_polished, 1 - h_polished, mode='full')
        max_corr = np.max(corr)
        
        while True:
            if time.time() - t_start > budget_s - 5:
                break
            k_max = np.argmax(corr) - (len(corr) // 2)
            c_max = max_corr
            if c_max <= 0:
                break
            
            ones = np.where(h_polished == 1)[0]
            zeros = np.where(h_polished == 0)[0]
            
            # Find an improving swap between 1 and 0
            improvement_found = False
            for i in ones:
                if i + k_max < len(h_polished) and h_polished[i+k_max] == 0:
                    for j in zeros:
                        if j + k_max < len(h_polished) and h_polished[j+k_max] == 1:
                            # Perform swap
                            h_polished[i] = 0
                            h_polished[j] = 1
                            # Update correlation incrementally
                            corr = correlate(h_polished, 1 - h_polished, mode='full')
                            max_corr = np.max(corr)
                            improvement_found = True
                            break
                    if improvement_found:
                        break
            if not improvement_found:
                break
        # Re-evaluate
        corr = correlate(h_polished, 1 - h_polished, mode='full')
        max_corr = np.max(corr)
        c5_final = max_corr * (2.0 / len(h_polished))
        return (h_polished, c5_final, len(h_polished))

    # Fallback: return best previously found
    if best_sol is not None:
        return best_sol
    else:
        return (np.ones(80), 0.5, 80)
```
