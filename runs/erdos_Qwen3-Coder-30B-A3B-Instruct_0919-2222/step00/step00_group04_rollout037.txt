import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project_to_capped_simplex(v, target_sum):
    """Projection onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum."""
    v = np.asarray(v)
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    # Find threshold
    tmp = (cumsum - target_sum) / np.arange(1, len(v) + 1)
    idx = np.argmax(sorted_v >= tmp)
    if idx == 0:
        tau = tmp[0]
    else:
        tau = tmp[idx - 1]
    result = np.maximum(0, np.minimum(1, v - tau))
    return result

def compute_overlap_on_blocks(a, w, N, dx):
    """Compute the correlation vector for a piecewise-constant function."""
    # Compute block boundaries
    s = np.cumsum([0] + w.tolist())
    n_blocks = len(a)

    # For each shift, compute G(k)
    # Shifts: 0 to 2*2 = 4, step dx
    shifts = np.arange(0, 2 * N) * dx
    G = np.zeros_like(shifts)

    # For each shift k
    for idx_k, k in enumerate(shifts):
        g = 0.0
        for i in range(n_blocks):
            for j in range(n_blocks):
                s_i = s[i]
                s_ip1 = s[i+1]
                s_j = s[j]
                s_jp1 = s[j+1]
                # Compute overlap of [s_i, s_ip1) and [s_j + k, s_jp1 + k)
                left_overlap = max(s_i, s_j + k)
                right_overlap = min(s_ip1, s_jp1 + k)
                if right_overlap > left_overlap:
                    length_overlap = right_overlap - left_overlap
                    g += a[i] * a[j] * length_overlap
        G[idx_k] = g
    return G

def evaluate_fitness(params, M, N, dx):
    # Extract s, before scaling
    u = params[:M]
    v = params[M:]
    # Rescale to ensure widths sum to 2 (softmax)
    w = 2 * np.exp(u) / np.sum(np.exp(u))
    # Rescale to ensure a in [0,1] (sigmoid)
    a = 1 / (1 + np.exp(-v))
    # Check constraint
    total_mass = np.sum(a * w)
    penalty = 1e3 * abs(total_mass - 1.0)
    # Compute overlap
    G = compute_overlap_on_blocks(a, w, N, dx)
    f = 1.0 - G  # Transform to overlap measure
    obj = np.max(f)
    return obj + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    # Parameters
    M = 12  # number of blocks
    N = 200  # fine grid size
    dx = 2.0 / N
    
    # Start timing
    start_t = time.time()
    
    # Initial guess and bounds
    bounds = [(-5, 5)] * (2 * M)  # (u, v) params
    
    # Run global optimization using DE
    try:
        res = differential_evolution(
            evaluate_fitness,
            bounds,
            args=(M, N, dx),
            seed=seed,
            maxiter=20,  # fewer iterations to save time
            popsize=15 * (2 * M),
            polish=False,
            updating='deferred',
            disp=False
        )
        u_opt = res.x[:M]
        v_opt = res.x[M:]
        w = 2 * np.exp(u_opt) / np.sum(np.exp(u_opt))
        a = 1 / (1 + np.exp(-v_opt))
    except Exception:
        # Fall back to initial construction if DE fails
        a = np.array([0.5] * M)
        w = np.array([2.0 / M] * M)

    # Interpolate to fine grid
    s = np.cumsum([0] + w.tolist())
    h = np.zeros(N)
    for i in range(N):
        x = i * dx
        # Find corresponding block
        for j in range(len(w)):
            if s[j] <= x < s[j+1]:
                h[i] = a[j]
                break
        else:
            # x == 2.0, impose boundary condition
            h[i] = a[-1]
    
    # Enforce sum constraint
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        # Re-scale
        h = h * (1.0 / current_sum)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
    
    # Get number of points
    n_points = N

    # Compute the actual C5 bound on this vector
    start_eval = time.time()
    best_h = h.copy()
    c5_bound = 0.0

    # Sub-gradient descent refinement
    max_iter = 4000
    alpha0 = 0.2
    for it in range(max_iter):
        if time.time() - start_t > budget_s - 5:
            break
            
        # Compute correlations
        corr = correlate(best_h, 1.0 - best_h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        # Compute gradient
        shift = max_k - (len(best_h) - 1)
        g = np.zeros_like(best_h)
        for i in range(len(best_h)):
            if i < len(best_h) - abs(shift):
                if shift >= 0:
                    g[i] += (1.0 - best_h[i + shift])
                else:
                    g[i] -= best_h[i - shift]
        
        # Subgradient step
        alpha = alpha0 / np.sqrt(it + 1)
        updated_h = best_h - alpha * g
        # Clamp and project
        updated_h = np.clip(updated_h, 0, 1)
        updated_h = project_to_capped_simplex(updated_h, n_points / 2.0)
        best_h = updated_h

        # Evaluate bound
        corr2 = correlate(best_h, 1.0 - best_h, mode='full')
        new_bound = np.max(corr2) * (dx)
        if new_bound < c5_bound or c5_bound == 0:
            c5_bound = new_bound

    return (best_h.tolist(), c5_bound, n_points)
