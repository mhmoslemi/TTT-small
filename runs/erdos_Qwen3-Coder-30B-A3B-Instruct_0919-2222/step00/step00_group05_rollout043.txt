import time
import random
from scipy import signal

def evaluate_h(h):
    """Evaluate C5 for step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = signal.correlate(h, 1 - h, mode="full")
    # Extract the relevant part and scale appropriately
    max_corr = np.max(corr) * dx
    return max_corr

def project_onto_constraints(h):
    """Ensure that h values are in [0, 1] and integral equals 1."""
    h = np.clip(h, 0, 1)
    # Normalize to sum to 1
    dx = 2.0 / len(h)
    scale_factor = 1.0 / np.sum(h) / dx
    h *= scale_factor
    return h

def generate_random_feasible(n_points):
    """Generate a random feasible solution."""
    h = np.random.rand(n_points)
    h = project_onto_constraints(h)
    return h

def perturb_h(h, step_size=0.05):
    """Small random perturbation of h."""
    h_perturbed = h.copy()

    n = len(h)

    indices = random.sample(range(n), num_perturbs)
    for i in indices:
        delta = random.uniform(-step_size, step_size)
        h_perturbed[i] += delta
    return project_onto_constraints(h_perturbed)
