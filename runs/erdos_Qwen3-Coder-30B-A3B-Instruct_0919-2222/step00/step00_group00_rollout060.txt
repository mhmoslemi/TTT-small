def compute_correlation(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap(h, dx):
    c = compute_correlation(h)
    return c.max() * dx

def apply_delta(c, h, idx, delta):
    n = len(h)
    offset = n - 1
    # First term: for j from idx to n-1
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # Second term: for j from 0 to idx
    for j in range(0, idx+1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(i, j, c, h):
    # First, apply delta_i = -1 to h[i]
    delta_i = -h[i]
    apply_delta(c, h, i, delta_i)
    h[i] = 0  # Set to 0
    
    # Then apply delta_j = +1 to h[j]
    delta_j = 1 - h[j]
    apply_delta(c, h, j, delta_j)
    h[j] = 1  # Set to 1

def try_targeted_swap(c, h, dx):

    max_idx = np.argmax(c)
    max_shift = max_idx - (len(h) - 1)
    
    # Find candidates for a targeted swap
    n = len(h)
    offset = n - 1
    
    # Collect candidate indices for i and j
    candidates_i = []
    candidates_j = []
    
    # Find 1s where shifted position is 0 
    for i in range(n):
        if h[i] == 1:
            shifted_idx = i + max_shift
            if 0 <= shifted_idx < n and h[shifted_idx] == 0:
                candidates_i.append(i)
    
    # Find 0s where shifted position is 1
    for j in range(n):
        if h[j] == 0:
            shifted_idx = j + max_shift
            if 0 <= shifted_idx < n and h[shifted_idx] == 1:
                candidates_j.append(j)
    
    # If both non-empty, randomly pair one from each
    if candidates_i and candidates_j:
        i = random.choice(candidates_i)
        j = random.choice(candidates_j)
        return i, j
    return None, None

def hill_climb(h, n_iter, timeout, dx):
    n = len(h)
    c = compute_correlation(h)
    current_max = c.max() * dx
    best_h = h.copy()
    best_max = current_max

    # Convert to list of indices for fast lookups
    ones = [i for i in range(n) if h[i] == 1]
    zeros = [i for i in range(n) if h[i] == 0]
    
    start_time = time.time()
    
    for iteration in range(n_iter):
        if time.time() - start_time > timeout:
            break
            
        # Try targeted swap
        i, j = try_targeted_swap(c, h, dx)
        
        # If no targeted swap found, do random swap
        if i is None:
            if ones and zeros:
                i = random.choice(ones)
                j = random.choice(zeros)
            else:
                # Fallback: random indices
                i, j = random.randint(0, n-1), random.randint(0, n-1)
                # Ensure they have different values
                while h[i] == h[j]:
                    i = random.randint(0, n-1)
                    j = random.randint(0, n-1)
                        
        # Store before swapping
        old_c = c.copy()
        
        # Perform swap
        old_i_val = h[i]
        old_j_val = h[j]
        
        # Apply swaps
        swap(i, j, c, h)
        
        # Update sets
        if old_i_val == 1 and h[i] == 0: # Removed from ones
            ones.remove(i)
            zeros.append(i)
        elif old_i_val == 0 and h[i] == 1: # Added to ones
            zeros.remove(i)
            ones.append(i)
            
        if old_j_val == 1 and h[j] == 0: # Removed from ones
            ones.remove(j)
            zeros.append(j)
        elif old_j_val == 0 and h[j] == 1: # Added to ones
            zeros.remove(j)
            ones.append(j)
        
        # Compute new max
        new_max = c.max() * dx
        
        # Accept improvement, or accept anyway for diversity if not too bad
        if new_max < current_max or random.random() < 0.1:
            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            # Revert the change
            c[:] = old_c
            h[i] = old_i_val
            h[j] = old_j_val
            
            # Update sets back
            if old_i_val == 1 and h[i] == 0:
                ones.remove(i)
                zeros.append(i)
            elif old_i_val == 0 and h[i] == 1:
                zeros.remove(i)
                ones.append(i)
                
            if old_j_val == 1 and h[j] == 0:
                ones.remove(j)
                zeros.append(j)
            elif old_j_val == 0 and h[j] == 1:
                zeros.remove(j)
                ones.append(j)
    
    return best_h, best_max
