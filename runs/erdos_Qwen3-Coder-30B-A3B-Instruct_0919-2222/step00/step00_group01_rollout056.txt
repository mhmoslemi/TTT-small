import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given step function h"""
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value is the one we want
    max_corr = np.max(corr)
    return max_corr * dx
