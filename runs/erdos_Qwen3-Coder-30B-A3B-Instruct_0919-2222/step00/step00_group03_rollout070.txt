import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def solve_erdos_milp(n, time_limit):
    start_time = time.time()
    
    # Create the problem
    prob = pulp.LpProblem("ErdosMinOverlapMILP", pulp.LpMinimize)
    
    # Variables
    h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
    T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
    
    # Constraint: sum of h equals n/2
    prob += pulp.lpSum([h[i] for i in range(n)]) == n/2
    
    # Product constraints and shift constraints
    for d in range(n):  # For all shifts 0 to n-1
        # Create list of y variables for this shift
        y_vars = []
        for i in range(n-d):
            # y_i_d = h_i * h_{i+d}
            y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
            y_vars.append(y)
            
            # McCormick envelope constraints for y = h_i * h_{i+d}
            prob += y <= h[i]
            prob += y <= h[i+d]
            prob += y >= h[i] + h[i+d] - 1
            
        # Shift constraint: sum of products plus T >= n/2
        prob += pulp.lpSum(y_vars) + T >= n/2

    # Objective: minimize T
    prob += T
    
    # Solve with time limit
    solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit)
    prob.solve(solver)
    
    # Check if optimal solution found
    status = pulp.LpStatus[prob.status]
    if status != 'Optimal':
        return None, None
    
    # Retrieve solution
    h_vals = np.array([pulp.value(h[i]) for i in range(n)], dtype=float)
    t_val = pulp.value(T)
    
    return h_vals, t_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Try different resolutions
    resolutions = [100, 120, 150, 200]
    
    for n in resolutions:
        if time.time() - start_time > budget_s - 5:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 1:
            break
            
        try:
            h_vals, t_val = solve_erdos_milp(n, remaining_time)
            if h_vals is not None:
                # Compute actual C5 bound using evaluation
                actual_c5 = evaluate_erdos_solution(h_vals)
                if actual_c5 < best_c5:
                    best_c5 = actual_c5
                    best_h = h_vals.copy()
                    best_n = n
                    
                if best_c5 < 0.3807:
                    break  # Early termination if target achieved
                    
        except Exception as e:
            # Continue to next resolution if this one fails
            continue
    
    # Fallback to initial solution if no better was found
    if best_h is None:
        h_vals = np.array(initial_h_values)
        best_c5 = evaluate_erdos_solution(h_vals)
        best_h = h_vals.copy()
        best_n = len(h_vals)
    
    return best_h, best_c5, best_n
