import time
import numpy as np
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, ECOS_BB, CBC, Optimal, Infeasible


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Grid sizes to try
    grid_sizes = [80, 120, 160]
    
    for n in grid_sizes:
        if time.time() - t_start > budget_s - 5:
            break
            
        remaining_time = budget_s - (time.time() - t_start)
        if remaining_time < 30:
            break
            
        dx = 2.0 / n
        # Initial shift set -- start with small shifts
        K0 = max(1, min(n // 4, 30))  # Reasonable initial set
        K = list(range(1, K0 + 1))
        max_shift = n - 1
        
        # The actual optimisation variables
        x = Variable(n, boolean=True)
        t = Variable()
        y_vars = {}  # y[i, k] for each i,k that allows x[i+k] to exist
        y = {}       # Maps k to list of variables per shift
        
        for k in K:
            y[k] = [Variable(boolean=True) for i in range(n - k)]
        
        # Constraints
        constraints = []
        
        # Mass constraint
        constraints.append(cp.sum(x) == n / 2)
        
        # Symmetry-breaking
        constraints.append(x[0] == 1)
        
        # Non-increasing prefix (helps symmetry)
        for i in range(n // 2 - 1):
            constraints.append(x[i] >= x[i + 1])
        
        # McCormick envelopes
        for k in K:
            for i in range(n - k):
                # y[i,k] = x[i] * x[i+k]
                y_var = y[k][i]
                constraints.append(y_var <= x[i])
                constraints.append(y_var <= x[i + k])
                constraints.append(y_var >= x[i] + x[i + k] - 1)
        
        # Overlap bounds
        for k in K:
            c_k = cp.sum(x) - cp.sum(y[k])  # Always n/2 - sum of y
            constraints.append(c_k <= t)
        
        # Objective
        prob = Problem(Minimize(t), constraints)
        
        # Lazy constraint loop
        violations = []
        iteration = 0
        max_iterations = 20  # Prevent infinite loop
        
        while True:
            if time.time() - t_start > budget_s - 2:
                break
            iteration += 1
            if iteration > max_iterations:
                break
                
            try:
                # Solve
                prob.solve(solver=CBC, verbose=False, max_seconds=remaining_time/3)
                
                # Check if infeasible
                if prob.status not in [Optimal, 'optimal']:
                    break
                    
                t_opt = prob.value
                x_opt = [1 if x[i].value > 0.5 else 0 for i in range(n)]
                
                # Full correlation check (for violation detection)
                h_full = np.array(x_opt)
                # Mode 'full' computes all shifts circularly (infinite period)
                # Our constraint ensures 0 padding for non-circularity
                corr_full = np.correlate(h_full, 1 - h_full, mode='full')
                max_c_full = np.max(corr_full)
                
                # Found a commitable solution
                if max_c_full <= t_opt + 1e-6:
                    c5_bound = t_opt * dx
                    if c5_bound < best_c5:
                        best_c5 = c5_bound
                        best_h = h_full.copy()
                        best_n = n
                    break
                else:
                    # Add violated shifts
                    k_violated = np.argmax(corr_full)
                    if k_violated != 0 and k_violated not in K:
                        K.append(k_violated)
                        y[k_violated] = [Variable(boolean=True) for i in range(n - k_violated)]
                        
                        # Add the McCormick constraints for new k
                        for i in range(n - k_violated):
                            y_var = y[k_violated][i]
                            constraints.append(y_var <= x[i])
                            constraints.append(y_var <= x[i + k_violated])
                            constraints.append(y_var >= x[i] + x[i + k_violated] - 1)
                        
                        # Add the max constraint for this k
                        c_k = cp.sum(x) - cp.sum(y[k_violated])
                        constraints.append(c_k <= t)
                        
                        prob = Problem(Minimize(t), constraints)
                        
                        # Need to re-solve
                        continue
            
            except Exception:
                break
        
    # Final polishing
    if best_h is not None:
        h_vals = best_h
        c5_bound = best_c5
        n_points = best_n
        if n_points is None:
            return None, None, None
            
        h_vals = h_vals.astype(float)
        dx = 2.0 / n_points
        
        # Perform one more polishing step (in-place modification)
        # (resource constrained, so only once per valid result)
        max_iter = 100
        for _ in range(max_iter):
            corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            k_max = np.argmax(corr)
            c_max = corr[k_max]
            
            if c_max < 1e-10:
                break
            
            # Positions where 1 contributes to max
            ones = np.where(h_vals == 1)[0]
            zeros = np.where(h_vals == 0)[0]
            
            # Find swap pair
            pairs = []
            for i in ones:
                i2 = (i + k_max) % len(h_vals)
                if h_vals[i2] == 0:
                    pairs.append((i, i2))
            
            pairs2 = []
            for j in zeros:
                j2 = (j + k_max) % len(h_vals)
                if h_vals[j2] == 1:
                    pairs2.append((j, j2))
            
            if not pairs or not pairs2:
                break
                
            i, j = pairs[0]
            # Perform swap
            h_vals[i] = 0
            h_vals[j] = 1
            
            # Update correlation values_dirty
            # For efficiency, update only the max shift directly
            
        # Final valid corr max check (faster)
        corr_final = np.correlate(h_vals, 1 - h_vals, mode='full')
        refined_c5 = max(corr_final) * dx
        
        if refined_c5 < c5_bound:
            c5_bound = refined_c5
            
        # Return final result
        return (h_vals, c5_bound, n_points)
    
    # Could not find any solution
    return (np.array([]), float('inf'), 0)
