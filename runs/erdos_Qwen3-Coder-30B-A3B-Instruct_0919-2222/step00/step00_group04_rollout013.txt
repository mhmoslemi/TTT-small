import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap
    max_overlap = np.max(correlation)
    return max_overlap * dx
