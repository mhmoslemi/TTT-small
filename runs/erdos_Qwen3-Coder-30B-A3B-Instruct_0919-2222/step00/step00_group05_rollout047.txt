import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the maximum overlap integral for a given step function h."""
    # The evaluator uses full correlation mode, so we do the same
    # We compute all correlations, from shift -n+1 to n-1
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    # corr is of length 2*n - 1, corresponding to shifts from -n+1 to n-1
    # We want to find max over all shifts
    max_corr = np.max(corr)
    return max_corr * dx

def build_mcormick_constraints(h, y, n, s):
    """Build McCormick constraints for the bilinear term h_i * h_{i+s}."""
    constraints = []
    for i in range(n):
        if i+s < 0 or i+s >= n:
            continue  # Skip out-of-bounds indices
        constraints.append(y[i, s] <= h[i])
        constraints.append(y[i, s] <= h[i+s])
        constraints.append(y[i, s] >= h[i] + h[i+s] - 1)
        constraints.append(y[i, s] >= 0)
    return constraints

def solve_nonlinear_programming(h_vals, n, M):
    """Solve the LP relaxation of the problem."""
    # Variables
    h = cp.Variable(n)
    y = cp.Variable((n, 2*n - 1))
    t = cp.Variable()

    # Objective
    objective = cp.Minimize(t)

    # Constraints
    constraints = [
        0 <= h, h <= 1,
        cp.sum(h) == M,
        t >= 0,
    ]

    # McCormick constraints for all valid (i, s) pairs.
    for s in range(-n+1, n):
        for i in range(n):
            if i+s < 0 or i+s >= n:
                continue  # Skip out of bounds
            constraints.append(y[i, s+n-1] <= h[i])
            constraints.append(y[i, s+n-1] <= h[i+s])
            constraints.append(y[i, s+n-1] >= h[i] + h[i+s] - 1)
            constraints.append(y[i, s+n-1] >= 0)

    # Build each constraint Q_s >= (n/2) - t
    for shift in range(-n+1, n):
        relevant_y = [y[i, shift+n-1] for i in range(n) if 0 <= i+shift < n]
        Q_s = cp.sum(relevant_y)
        constraints.append(Q_s >= M - t)

    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver='ECOS', verbose=False)

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h.value, y.value
    else:
        raise RuntimeError("LP solver failed to produce a solution")

def deterministic_round(h_vals):
    """Rounding method: set top M values to 1 and rest to 0."""
    indices = np.argsort(h_vals)[::-1]  # Sort descending
    h = np.zeros_like(h_vals)
    h[indices[:len(h_vals)//2]] = 1
    return h

def randomized_round(h_vals, n, M, num_samples=200):
    """Generate binary vectors using sampling with replacement, keeping those with the right mass."""
    # Sample without replacement, favouring higher h_vals
    samples = []
    for _ in range(num_samples):
        # Using np.random.choice with probabilities
        prob = h_vals / np.sum(h_vals)  # Normalize
        selected_indices = np.random.choice(n, size=M, replace=False, p=prob)
        h = np.zeros(n)
        h[selected_indices] = 1
        samples.append(h)
    return samples

def targeted_swap_local_search(h):
    """Apply targeted improvement by swapping elements to reduce max correlation."""
    n = len(h)
    dx = 2.0 / n

    def compute_all_correlations(h):
        corr = correlate(h, 1 - h, mode='full')
        return corr * dx

    def compute_max_corr_index(corr_vals):
        return np.argmax(corr_vals)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    max_iter = 1000
    iter_count = 0
    while iter_count < max_iter:
        corr_vals = compute_all_correlations(best_h)
        max_corr_idx = compute_max_corr_index(corr_vals)

        # Search for a swap that reduces the maximum correlation
        s = max_corr_idx - (n - 1)  # Convert full index to shift
        candidates_i = np.where((best_h == 1) & (np.roll(best_h, -s) == 0))[0]
        candidates_j = np.where((best_h == 0) & (np.roll(best_h, -s) == 1))[0]

        if len(candidates_i) == 0 or len(candidates_j) == 0:
            break  # No more improving swaps

        best_improvement = 0
        best_i, best_j = -1, -1
        for i in candidates_i:
            for j in candidates_j:
                # Simulate swap
                temp_h = best_h.copy()
                temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
                new_c5 = evaluate_c5(temp_h)
                improvement = best_c5 - new_c5
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i, best_j = i, j

        if best_improvement <= 1e-10:
            break  # No beneficial swap found

        # Apply the best swap
        best_h[best_i], best_h[best_j] = best_h[best_j], best_h[best_i]
        best_c5 = best_c5 - best_improvement
        iter_count += 1

    return best_h, best_c5
