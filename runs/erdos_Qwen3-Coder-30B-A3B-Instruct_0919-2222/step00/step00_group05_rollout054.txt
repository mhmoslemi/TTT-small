import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

def evaluate_erdos_solution(h_values):
    """Compute C5 bound from h_values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    # This is equivalent to max of correlate(h, 1-h, mode="full")
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The maximum value is at the center (shift k=0)
    # But we need to check all valid shifts
    max_val = 0
    for i in range(len(corr)):
        shift = i - (len(h_values) - 1)
        if abs(shift) <= 2:  # Valid shifts
            if corr[i] > max_val:
                max_val = corr[i]
    return max_val * dx

def initial_h_values():
    """Initial construction from paper"""
    n = 70
    h = np.zeros(n)
    dx = 2.0 / n
    # Simple construction: piecewise constant
    # Try a construction that gives a small overlap
    h[10:30] = 1.0  # First segment
    h[35:55] = 1.0  # Second segment
    h[60:65] = 0.5  # Third segment
    # Normalize
    h = h * (n / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction
    n_points = 70
    h = initial_h_values()
    h = np.clip(h, 0, 1)
    
    # Normalize to integral = 1
    dx = 2.0 / n_points
    current_integral = np.sum(h) * dx
    h = h * (1.0 / current_integral)
    
    # Objective function to minimize C5
    def objective(h_candidate):
        h_norm = h_candidate / np.sum(h_candidate) * (n_points / 2) 
        h_norm = np.clip(h_norm, 0, 1)
        return evaluate_erdos_solution(h_norm)
    
    # Constraints
    def constraint_integral(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    def constraint_bounds(h_vec):
        return np.min(h_vec)  # Should be >= 0
    
    def constraint_bounds2(h_vec):
        return 1.0 - np.max(h_vec)  # Should be >= 0
    
    # Create bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [
        {'type': 'eq', 'fun': constraint_integral},
        {'type': 'ineq', 'fun': constraint_bounds},
        {'type': 'ineq', 'fun': constraint_bounds2}
    ]
    
    # Try to optimize 
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Some heuristics to perturb better
    # 1. Try small random jumps and gradient-like steps
    step_size = 0.01
    
    for i in range(20000):  # Limited iterations to avoid long runtime
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Try perturbation
        h_trial = best_h.copy()
        j = np.random.randint(0, n_points)
        perturb = np.random.uniform(-step_size, step_size)
        h_trial[j] += perturb
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        trial_c5 = objective(h_trial)
        
        if trial_c5 < best_c5:
            best_h = h_trial
            best_c5 = trial_c5
            
        # Gradient-like movement with small steps
        h_trial_grad = best_h.copy()
        for j in range(n_points):
            v = np.random.uniform(-0.005, 0.005)
            h_trial_grad[j] = h_trial_grad[j] + v
        h_trial_grad = np.clip(h_trial_grad, 0, 1)
        h_trial_grad = h_trial_grad * (n_points / 2) / np.sum(h_trial_grad)
        
        trial_c5_grad = objective(h_trial_grad)
        if trial_c5_grad < best_c5:
            best_h = h_trial_grad
            best_c5 = trial_c5_grad
            
    return best_h, best_c5, n_points
