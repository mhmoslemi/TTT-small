```python
import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    best_bound = float('inf')
    best_h = None
    best_n = 0

    # Try different resolutions
    n_values = [100, 120, 150, 200]
    
    for n in n_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
            
        try:
            # Solve MILP with binary h variables and continuous y
            prob = pulp.LpProblem("ErdosMinOverlap", pulp.LpMinimize)
            
            # Variables
            h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
            T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
            
            # Constraint: sum of h equals n/2
            prob += pulp.lpSum([h[i] for i in range(n)]) == n/2
            
            # Product constraints and shift constraints
            for d in range(1, n):  # skip d=0 because it's trivial
                y_vars = []
                for i in range(n - d):
                    y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
                    y_vars.append(y)
                    
                    # McCormick envelope constraints
                    prob += y <= h[i]
                    prob += y <= h[i+d]
                    prob += y >= h[i] + h[i+d] - 1
                
                # Shift constraint: sum of y >= n/2 - T
                prob += pulp.lpSum(y_vars) >= n/2 - T
                
            # Objective
            prob += T
            
            # Solve
            solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=int(remaining_time * 0.9))
            prob.solve(solver)
            
            if pulp.LpStatus[prob.status] in ['Optimal', 'Not Solved']:
                h_vals = np.array([float(pulp.value(h[i])) for i in range(n)])
                t_val = float(pulp.value(T))
                c5_bound = t_val * (2.0 / n)
                
                # Verify with evaluation function
                computed_bound = evaluate_erdos_solution(h_vals)
                
                if computed_bound < best_bound:
                    best_bound = computed_bound
                    best_h = h_vals.copy()
                    best_n = n
                    
                # Early exit if we've achieved our target
                if best_bound <= 0.379:
                    break
                    
        except Exception as e:
            continue
    
    # fallback to initial solution if nothing better found
    if best_h is None:
        h_vals = np.array(initial_h_values)
        best_bound = evaluate_erdos_solution(h_vals)
        best_h = h_vals.copy()
        best_n = len(h_vals)
    
    return (best_h, best_bound, best_n)
```
