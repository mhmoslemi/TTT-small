import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given step function h"""
    # Compute the overlap integral for all valid shifts
    corr = correlate(h, 1 - h, mode='full')
    # The full correlation has length 2*n - 1
    # We want the indices corresponding to shifts k in [0,2]
    # The shift k corresponds to index n-1+k/dx
    max_val = 0
    n = len(h)
    for i in range(len(corr)):
        # Convert correlation index to actual shift
        shift = (i - (n - 1)) * dx
        if 0 <= shift <= 2:
            max_val = max(max_val, corr[i] * dx)
    return max_val

def objective(h, dx, initial_h=None):
    """Objective function to minimize"""

        # Regularize towards initial construction
        reg = 0.01 * np.sum((h - initial_h)**2)
    else:
        reg = 0
    return evaluate_h(h, dx) + reg

def constraint_eq(h, dx):
    """Equality constraint: integral of h(x) dx = 1"""
    return np.sum(h) * dx - 1.0
