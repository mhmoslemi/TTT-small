import numpy as np

from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate C5 for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    correlations = correlate(h, 1-h, mode="full")
    # The maximum is over all shifts k
    max_corr = np.max(correlations)
    return max_corr * dx
