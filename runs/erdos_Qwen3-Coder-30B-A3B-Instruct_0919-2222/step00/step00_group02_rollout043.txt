import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_h(h):
    # h is a numpy array of length n
    # returns (max_overlap, corresponding shift)
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1-h, mode='full')
    max_k = corrs.argmax()
    max_overlap = corrs[max_k]
    shift = max_k - (n - 1)
    return max_overlap, shift

def evaluate_h_float(h):
    # h is a numpy array of length n, in [0, 1]
    # returns (max_overlap, shift) using np.correlate

    shift = max_k - (n - 1)
    return max_overlap, shift

def generate_balanced_binary(n):
    # generate a balanced binary vector (exactly n/2 ones and n/2 zeros)
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    np.random.shuffle(h)
    return h

def get_mismatch_sets(a, k):
    n = len(a)
    if k >= 0:
        # a[i] = 1, a[i+k] = 0
        M1 = np.where((a == 1) & (np.roll(a, -k) == 0))[0]
        # a[i] = 0, a[i+k] = 1
        M0 = np.where((a == 0) & (np.roll(a, -k) == 1))[0]
    else:
        # a[i] = 1, a[i-k] = 0 (i.e. a[i-k] == 0)
        M1 = np.where((a == 1) & (np.roll(a, -k) == 0))[0]
        M0 = np.where((a == 0) & (np.roll(a, -k) == 1))[0]
    return M1, M0

def swap_indices(a, i, j):
    a[i], a[j] = a[j], a[i]

def worst_shift_swaps(a, max_iter):
    n = len(a)
    for _ in range(max_iter):
        max_overlap, shift = evaluate_h(a)
        # get mismatch sets
        M1, M0 = get_mismatch_sets(a, shift)
        if len(M1) > 0 and len(M0) > 0:
            i = np.random.choice(M1)
            j = np.random.choice(M0)
            # try swap
            old_overlap, _ = evaluate_h(a)
            swap_indices(a, i, j)
            new_overlap, _ = evaluate_h(a)
            if new_overlap <= old_overlap:
                # accept
                pass
            else:
                # rollback
                swap_indices(a, i, j)
        else:
            # fallback
            ones = np.where(a==1)[0]
            zeros = np.where(a==0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                swap_indices(a, i, j)
    return a

def simulated_annealing(a0, max_iter):
    n = len(a0)
    a = a0.copy()
    temperature = 0.5
    min_temp = 1e-4
    decay = (min_temp / temperature) ** (1.0 / max_iter)
    best_a = a.copy()
    best_val, _ = evaluate_h(best_a)
    for t in range(max_iter):
        # random swap
        ones = np.where(a==1)[0]
        zeros = np.where(a==0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            old_val, _ = evaluate_h(a)
            swap_indices(a, i, j)
            new_val, _ = evaluate_h(a)
            delta = new_val - old_val
            if delta <= 0 or np.random.rand() < np.exp(-delta / temperature):
                # accept
                if new_val < best_val:
                    best_val = new_val
                    best_a = a.copy()
            else:
                # rollback
                swap_indices(a, i, j)
        temperature *= decay
    return best_a, best_val

def projected_softmax_gradient_descent(h_init, max_iter, alpha_schedule, step_size=0.2):
    h = h_init.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    # weight for projection
    target_sum = n / 2.0
    # run the schedule
    for alpha in alpha_schedule:
        for t in range(max_iter):
            # compute overlaps for each shift
            corrs = np.correlate(h, 1 - h, mode='full')
            # normalize and compute soft max weights
            max_o = np.max(corrs)
            corrs_shifted = corrs - max_o
            exps = np.exp(alpha * corrs_shifted)
            exps_sum = np.sum(exps)
            if exps_sum == 0:
                # small fallback
                weights = np.ones_like(exps)
            else:
                weights = exps / exps_sum
            # compute gradient
            grad = np.zeros(n)
            for k in range(-n + 1, n):
                if k >= 0:
                    # shift right: h[i] * (1 - h[i+k]) -> grad: 1 - (h[i+k] + 0?)
                    id_i = np.arange(n)
                    id_k = id_i + k
                    mask = (id_k < n)
                    right_contrib = np.zeros(n)
                    right_contrib[mask] = 1.0 - h[id_k[mask]]
                else:
                    id_i = np.arange(n)
                    id_k = id_i - k
                    mask = (id_k >= 0)
                    right_contrib = np.zeros(n)
                    right_contrib[mask] = 1.0 - h[id_k[mask]]
                grad += weights[k + n - 1] * (1.0 - right_contrib)
            # gradient step
            h_new = h - step_size * grad
            # project
            # note: h_new < 0 or h_new > 1 are likely, but we should also project onto sum = n/2
            # use water-filling
            v = h_new.copy()
            def phi(lam):
                S = np.sum(np.clip(v - lam, 0, 1))
                return S - target_sum
            # binary search
            lo = np.min(v) - 1.0
            hi = np.max(v) + 1.0
            for _ in range(40):
                mid = (lo + hi) / 2.0
                if phi(mid) > 0:
                    lo = mid
                else:
                    hi = mid
            lambda_val = (lo + hi) / 2.0
            h = np.clip(v - lambda_val, 0, 1)
    return h
