import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1-h, mode='full')
    # Take the maximum correlation value
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Return initial construction for h"""
    n_points = 42
    h = np.zeros(n_points)
    # Simple initial construction: piecewise constant function
    # We'll use a construction that alternates between 0 and 1 in a way that sums to 1 over [0,2]
    # This is a simple construction that we'll optimize from
    half = n_points // 2
    h[:half] = 1.0
    h[half:] = 0.0
    return h

def objective(h):
    """Minimize the maximum overlap integral"""
    return evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved C5 bound"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction
    h_init = initial_h_values()
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Normalize h_init to have correct integral
    norm_factor = np.sum(h_init) * dx
    if abs(norm_factor - 1.0) > 1e-10:
        h_init = h_init * (1.0 / norm_factor)
    
    # Optimization using differential evolution
    # We apply bounds and normalization constraints
    def bounded_objective(h):
        # Clip h to [0,1]
        h_clipped = np.clip(h, 0, 1)
        # Ensure integral equals 1
        integral = np.sum(h_clipped) * dx
        if abs(integral - 1.0) > 1e-10:
            h_clipped = h_clipped * (1.0 / integral)
        return evaluate_c5(h_clipped)
    
    # Run optimization for 80% of the budget
    remaining_budget = budget_s * 0.8
    opt_result = differential_evolution(
        bounded_objective,
        bounds,
        maxiter=1000,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        disp=False
    )
    
    if not opt_result.success:
        h_opt = h_init
    else:
        h_opt = np.clip(opt_result.x, 0, 1)
        integral = np.sum(h_opt) * dx
        if abs(integral - 1.0) > 1e-10:
            h_opt = h_opt * (1.0 / integral)
    
    # Evaluate final result
    c5_bound = evaluate_c5(h_opt)
    
    # Finetune the best solution using local optimization if time permits
    if time.time() - start_time < budget_s * 0.9:
        # Try local optimization to improve the solution
        from scipy.optimize import minimize
        def obj_to_minimize(h):
            h_clip = np.clip(h, 0, 1)
            integral = np.sum(h_clip) * dx
            if abs(integral - 1.0) > 1e-10:
                h_clip = h_clip * (1.0 / integral)
            return evaluate_c5(h_clip)
        
        # Use a small perturbation around current solution
        h_start = h_opt.copy()
        try:
            res = minimize(
                obj_to_minimize,
                h_start,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6},
                tol=1e-8
            )
            if res.success:
                # Check if we got a better solution
                h_final = np.clip(res.x, 0, 1)
                integral = np.sum(h_final) * dx
                if abs(integral - 1.0) > 1e-10:
                    h_final = h_final * (1.0 / integral)
                c5_test = evaluate_c5(h_final)
                if c5_test < c5_bound:
                    h_opt = h_final
                    c5_bound = c5_test
        except:
            pass
    
    # Final check for feasibility
    h_final = np.clip(h_opt, 0, 1)
    integral = np.sum(h_final) * dx
    if abs(integral - 1.0) > 1e-10:
        h_final = h_final * (1.0 / integral)
        
    return (h_final, c5_bound, n_points)
