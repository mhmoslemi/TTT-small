import time
import numpy as np
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, OR, sum as cp_sum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    # Eligible grid sizes (to try coarse → fine resolution)
    n_values = [80, 120, 160]  # There are other n values; we keep them to fewer required
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_values:
        if time.time() - t_start > budget_s - 2:
            break
            
        # Initialize with a warm start if possible
        current_h = initial_h_values if 'initial_h_values' in globals() and n == 80 else None
        
        # Solve MILP with lazy constraint loop
        h_solution, c5_solution = solve_lazy_milp(n, t_start, budget_s, current_h)
        
        if h_solution is not None and c5_solution < best_c5:
            best_c5 = c5_solution
            best_h = h_solution
            best_n = n
            
            if best_c5 <= 0.379:
                break  # Exit early if we reach the target
    
    if best_h is None:
        # Fallback to initial solution if nothing better found
        try:
            h_init = np.array(initial_h_values)
            n = len(h_init)
            dx = 2.0 / n
            c5_init = max(np.correlate(h_init, 1-h_init, mode="full")) * dx
            return (h_init, c5_init, n)
        except:
            pass
        # Use the best one we can get
        return (np.zeros(80), 1.0, 80)
        
    return (best_h, best_c5, best_n)


def solve_lazy_milp(n, start_t, budget_s, warm_start=None):
    dx = 2.0 / n
    K0 = max(1, n // 4)
    K = list(range(1, K0 + 1))
    
    # Warm start
    if warm_start is not None:
        x_init = warm_start.copy()
        x_init = np.array(x_init)
        if len(x_init) != n:
            # Interpolate with upsampling
            x_interp = []
            for i in range(n):
                x_interp.append(warm_start[i // 2] if i // 2 < len(warm_start) else 0)
            x_init = np.array(x_interp)
    else:
        # Default: start with a heuristic step function
        x_init = np.zeros(n)
        x_init[:n//2] = 1
    
    # Try to initialize based on previous solution
    def build_milp_model(x_start):
        x = Variable(n, boolean=True)
        t = Variable()
        y = {}
        c_sum = cp_sum(x)
        constraints = [c_sum == n // 2]
        constraints.append(x[0] == 1)  # Symmetry breaking
        
        # Non-increasing prefix up to median
        for i in range(n // 2 - 1):
            constraints.append(x[i] >= x[i+1])
            
        for k in K:
            y[k] = Variable(n - k, boolean=True)
            y_sum = cp_sum(y[k])
            x_sum = cp_sum(x[:n-k])
            constraints.append(x_sum - y_sum <= t)
            # McCormick constraints
            for i in range(n - k):
                constraints.append(y[k][i] <= x[i])
                constraints.append(y[k][i] <= x[i+k])
                constraints.append(y[k][i] >= x[i] + x[i+k] - 1)
        
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        return prob, x, y, t

    x_approx = np.array(x_init)
    best_t = float('inf')
    best_x = None
    
    max_iterations = 10
    iteration = 0
    while iteration < max_iterations:
        if time.time() - start_t > budget_s - 2:
            break
            
        iteration += 1
        try:
            prob, x_vars, y_vars, t_var = build_milp_model(x_approx)
            
            # Try solving
            prob.solve(solver='CBC', verbose=False, max_seconds=30)
            
            if prob.status in ['optimal', 'optimal_inaccurate']:
                t_val = t_var.value
                x_sol = np.array([x_vars[i].value for i in range(n)]).astype(int)
                
                if t_val is not None and not np.isnan(t_val):
                    # Evaluate all overlaps
                    c_full = np.correlate(x_sol, 1 - x_sol, mode="full")
                    c_max = max(c_full)
                    c_max_scaled = c_max * dx
                    
                    if c_max_scaled < best_t:
                        best_t = c_max_scaled
                        best_x = x_sol.copy()
                        
                    # Check if violated any other shifts
                    all_k = list(range(1, n))  # All shift values
                    violated_k_list = []
                    for k in all_k:
                        if k not in K:
                            c_k = sum(x_sol[i] * (1 - x_sol[i + k]) 
                                    for i in range(n - k))
                            if c_k > t_val:
                                violated_k_list.append(k)
                                
                    if len(violated_k_list) > 0:
                        # Add violated shifts to K
                        K.extend(violated_k_list[:min(5, len(violated_k_list))])
                        K = sorted(list(set(K)))  # ensure uniqueness and sort
                        # Early exit if too many violated
                        if len(K) > 200:
                            break
                        continue  # Rebuild model
                    else:
                        # Done with this size
                        break
            else:
                # Try again with updated x as warm start
                pass
        except Exception as e:
            print("error solving MILP:", e)
            break
            
        # Even if fails, go to polishing
        break

    # polishing step
    if best_x is not None:
        best_x = polish_x(best_x, n)
        c_full = np.correlate(best_x, 1 - best_x, mode="full")
        c_max = max(c_full)
        polished_c5 = c_max * dx
        return best_x, polished_c5
    else:
        # fallback if no valid MILP result
        x = np.array(x_init)
        x[0] = 1
        if x.sum() > n // 2:
            # Need to flip some 1's to 0's to get sum = n // 2
            ones = np.where(x == 1)[0]
            for idx in ones[n//2:]:
                x[idx] = 0
        c_full = np.correlate(x, 1 - x, mode="full")
        c_max = max(c_full)
        return x, c_max * dx


def polish_x(x_start, n):
    """Apply deterministic local swap improvements"""
    x = x_start.copy()
    max_iter = 50
    for _ in range(max_iter):
        c_full = np.correlate(x, 1 - x, mode="full")
        k_star = np.argmax(c_full)
        if k_star >= n or c_full[k_star] <= 0:
            break

        A = []
        B = []
        for i in range(n):
            if x[i] == 1 and k_star + i < n and x[k_star + i] == 0:
                A.append(i)
            if x[i] == 0 and k_star + i < n and x[k_star + i] == 1:
                B.append(i)
                
        if len(A) == 0 or len(B) == 0:
            break
            
        i = A[0]
        j = B[0]
        x[i] = 0
        x[j] = 1
        # After swap, c_k_star decreases by 1. So we updated!

    return x
