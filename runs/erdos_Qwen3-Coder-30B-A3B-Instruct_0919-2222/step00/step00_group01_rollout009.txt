def evaluate_c5(h):
    """Evaluate the C5 bound for given step function h"""

    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum correlation (scaled by dx) is our C5 bound
    c5 = np.max(corr) * dx
    return c5
