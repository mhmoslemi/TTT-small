import numpy as np

def compute_correlation(h):
    """Compute full correlation of h with (1-h)"""
    return np.correlate(h, 1 - h, mode='full')

def max_overlap(h, dx):
    """Compute maximum overlap integral for step function h"""
    c = compute_correlation(h)
    return c.max() * dx

def apply_delta(c, h, idx, delta):
    """Apply delta change to h[idx] and update correlation c"""
    n = len(h)
    offset = n - 1
    # First term: h[i] * (1 - h[i+s]) for all valid s where i+s is in range
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # Second term: h[i-s] * (1 - h[i]) for all valid s where i-s is in range
    for j in range(0, idx + 1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(c, h, i, j):
    """Perform swap operation on h and update correlation accordingly"""
    # Save old values
    old_i = h[i]
    old_j = h[j]
    # Apply swaps
    delta_i = 0 - old_i  # new_val - old_val
    delta_j = 1 - old_j  # new_val - old_val
    # Update correlation
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def targeted_swap(c, h):
    """Find best targeted swap for current correlation"""
    n = len(h)
    offset = n - 1
    # Find maximizing shift
    max_idx = np.argmax(c)
    s = max_idx - offset  # actual shift
    # Collect candidates
    A = []  # indices where h[i]=1 and h[i+s]=0
    B = []  # indices where h[i]=0 and h[i+s]=1
    for i in range(n):
        if i + s >= 0 and i + s < n:
            if h[i] == 1 and h[i + s] == 0:
                A.append(i)
            elif h[i] == 0 and h[i + s] == 1:
                B.append(i)
    # Try to do a swap
    if A and B:
        i = random.choice(A)
        j = random.choice(B)
        return i, j
    return None, None

def hill_climb(h, n_iter, time_remaining, dx):
    """Optimize h using hill climbing"""
    n = len(h)
    c = compute_correlation(h)
    best_val = max_overlap(h, dx)
    best_h = h.copy()
    
    # Track indices with 0 and 1
    ones = [i for i in range(n) if h[i] == 1]
    zeros = [i for i in range(n) if h[i] == 0]

    for iteration in range(n_iter):
        if time_remaining() < 2.0:
            break
            
        # Try targeted swap first
        i, j = targeted_swap(c, h)
        success = False
        if i is not None:
            # Perform swap
            swap(c, h, i, j)
            # Update indices
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            success = True
        else:
            # Fallback: random swap
            i = random.choice(ones)
            j = random.choice(zeros)
            swap(c, h, i, j)
            # Update indices
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            success = True
            
        if success:
            new_val = max_overlap(h, dx)
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
    
    return best_h, best_val
