import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for a discrete h function."""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def build_mcormick_constraints(h, y, s, n):
    """Add McCormick inequalities for product h_i * h_{i+s}."""
    constraints = []
    for i in range(n):
        if 0 <= i + s < n:
            # McCormick inequalities y_{i,s} <= h_i, h_{i+s}
            constraints.append(y[i, s + n - 1] <= h[i])
            constraints.append(y[i, s + n - 1] <= h[i + s])
            # y_{i,s} >= h_i + h_{i+s} - 1
            constraints.append(y[i, s + n - 1] >= h[i] + h[i + s] - 1)
            # y_{i,s} >= 0
            constraints.append(y[i, s + n - 1] >= 0)
    return constraints

def solve_relaxed_lp(h_vals, y_vals, t, s_list, n, M):
    """Solve the LP relaxation."""
    # Variables: h, y, t
    h = cp.Variable(n)
    y = cp.Variable((n, 2 * n - 1))  # y[i, s] for s in [-n+1, ..., n-1]
    t = cp.Variable()

    # Constraints
    constraints = []
    # Bound constraints
    constraints += [0 <= h, h <= 1]
    # Mass constraint
    constraints.append(cp.sum(h) == M)
    # t >= 0
    constraints.append(t >= 0)

    # Add McCormick constraints for each shift
    for s in s_list:
        constraints += build_mcormick_constraints(h, y, s, n)

    # Shift-dependent constraints: sum over i of y[i, s] >= M - t
    for s in s_list:
        total = cp.sum([y[i, s + n - 1] for i in range(n) if 0 <= i + s < n])
        constraints.append(total >= M - t)

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_star = h.value
        return h_star
    else:
        return None

def deterministic_round(h_star, M):
    """Round fractional h_star to a binary solution."""
    h = h_star.copy()
    idx = np.argsort(h)[::-1]
    h.fill(0)
    h[idx[:int(M)]] = 1
    return h

def random_round(h_star, M, num_samples=200):
    """Randomly sample binary vectors from the fractional solution."""
    samples = []
    for _ in range(num_samples):
        p = h_star
        # Sample indices with prob proportional to p (without replacement)
        idx = np.random.choice(len(p), size=int(M), p=p/p.sum(), replace=False)
        h = np.zeros_like(h_star)
        h[idx] = 1
        samples.append(h)
    return samples

def polish_candidate(h, n, dx, time_limit=150):
    """Apply targeted swap polishing to improve local candidate."""
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    start_time = time.time()
    while time.time() - start_time < time_limit:
        # Compute correlation for each shift
        corr = correlate(best_h, 1 - best_h, mode="full")
        max_shift = np.argmax(corr)
        max_corr = corr[max_shift]
        
        if max_corr <= best_c5:
            break
            
        # Find swap candidates
        s = max_shift - (n - 1)  # convert index to shift value
        
        best_swap = None
        best_delta = 0
        for i in range(n):
            # Check source condition: h[i] = 1 and h[i+s] = 0 (if within bounds)
            if best_h[i] == 1:
                j = i + s
                if 0 <= j < n and best_h[j] == 0:
                    # Check target condition: h[j+s] = 1 and h[j] = 0
                    # but to make swap beneficial, look for positions that decrease the max
                    j2 = j + s
                    if 0 <= j2 < n and best_h[j2] == 1:
                        # Swap h[i] and h[j]: this potentially reduces the max
                        delta = -1  # Approximate change owed?
                        if delta < best_delta:
                            best_delta = delta
                            best_swap = (i, j)
        
        if best_swap is None:
            break
            
        i, j = best_swap
        # Perform the swap
        best_h[i] = 0
        best_h[j] = 1
        
        # Recompute the entire correlation to see effect
        new_corr = correlate(best_h, 1 - best_h, mode="full")
        new_max = np.max(new_corr) * dx
        if new_max >= best_c5:
            # Undo swap
            best_h[i] = 1
            best_h[j] = 0
        else:
            best_c5 = new_max
            
    return best_h, best_c5

def solve_mip(h_vals, y_vals, t, s_list, n, M):
    """Solve the MIP with binary constraints."""
    # Redefine variables as binary
    h = cp.Variable(n, integer=True)
    y = cp.Variable((n, 2 * n - 1))  # y[i, s]
    t = cp.Variable()

    constraints = []
    constraints += [0 <= h, h <= 1]
    constraints.append(cp.sum(h) == M)
    constraints.append(t >= 0)

    # Add McCormick constraints
    for s in s_list:
        constraints += build_mcormick_constraints(h, y, s, n)

    # Shift constraints
    for s in s_list:
        total = cp.sum([y[i, s + n - 1] for i in range(n) if 0 <= i + s < n])
        constraints.append(total >= M - t)

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    prob.solve(solver=cp.CBC, verbose=False, max_time=250)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h.value
    else:
        return None
