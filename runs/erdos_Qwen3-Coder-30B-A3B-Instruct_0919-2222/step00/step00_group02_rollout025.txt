import numpy as np
from scipy.optimize import minimize
from scipy import signal
import time
import math

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant for given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    corr = signal.correlate(h_values, 1 - np.array(h_values), mode='full')
    # The maximum value gives us C_5
    C5 = np.max(corr) * dx
    return C5

def initial_h_values():
    """Return an initial construction with 45 samples."""
    n = 45
    h = np.zeros(n)
    # Simple construction: step function at center
    mid = n // 2
    h[mid-1:mid+1] = 1.0
    # Normalize to integrate to 1
    h = h * 2.0 / n
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Get initial solution
    n_points = 45
    h = initial_h_values()
    dx = 2.0 / n_points
    
    # Normalize
    h_sum = np.sum(h)
    h = h * (1.0 / h_sum) * (n_points / 2)
    
    # Convert to optimization variables (0 to 1)
    # We will optimize the actual values directly
    h_opt = h.copy()
    
    def objective(h_vals):
        # h_vals is array of values to be optimized
        # Apply constraints if needed
        h_vec = np.array(h_vals)
        # Ensure h_vals are in [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Normalize based on sum constraint
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        # Compute C5
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        h_vec = np.array(h_vals)
        return np.sum(h_vec) - n_points / 2
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds (0 to 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    x0 = h_opt.copy()
    
    # Use L-BFGS-B with bounds and constraints
    best_h = h_opt.copy()
    best_C5 = evaluate_erdos_solution(best_h)
    
    # For better exploration, try multiple starts
    # Also use heuristic to get convergence faster

    start_time = time.time()
    
    # Just run one iteration of L-BFGS-B
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   constraints=cons, options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10})
    
    if res.success:
        h_opt = np.clip(res.x, 0, 1)
        # Normalize again
        h_sum = np.sum(h_opt)
        if h_sum > 0:
            h_opt = h_opt * (n_points / 2) / h_sum
        C5 = objective(h_opt)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h_opt.copy()
    
    # Perform more optimization with random search for better results
    # Generate a few random solutions and refine one that's better
    
    for _ in range(100):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate a random initial guess
        test_h = np.random.rand(n_points)
        # Normalize to make integral = 1
        h_sum = np.sum(test_h)
        if h_sum > 0:
            test_h = test_h * (n_points / 2) / h_sum
        
        # Optimize starting from this initial point
        res = minimize(objective, test_h, method='L-BFGS-B', bounds=bounds, 
                       constraints=cons, options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10})
        
        if res.success:
            h_test = np.clip(res.x, 0, 1)
            # Normalize again
            h_sum = np.sum(h_test)
            if h_sum > 0:
                h_test = h_test * (n_points / 2) / h_sum
            C5_test = objective(h_test)
            if C5_test < best_C5:
                best_C5 = C5_test
                best_h = h_test.copy()
        else:
            # If optimization fails, still try the test_h after ad-hoc clipping and normalization
            h_test = np.clip(test_h, 0, 1)
            h_sum = np.sum(h_test)
            if h_sum > 0:
                h_test = h_test * (n_points / 2) / h_sum
            C5_test = objective(h_test)
            if C5_test < best_C5:
                best_C5 = C5_test    
                best_h = h_test.copy()

    # Final verification
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_C5, n_points)
