import time
import random
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx=2.0 / len(h)):
    """Evaluate the Erdős minimum overlap constant C5 for given step function h."""
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Maximize over all shifts
    max_corr = np.max(corr) * dx
    return max_corr
