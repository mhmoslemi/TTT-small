def evaluate_erdos_solution(h_values, n_points):
    """Evaluate the Erdős overlap bound for given step function h"""
    dx = 2.0 / n_points
    # Compute correlation using np.correlate with full mode (zero padded)
    correlation = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # The result has length 2*n-1 where lags range from -(n-1) to (n-1)
    # The peak is at index n-1 (lag = 0)
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_to_simplex(v, target_sum):
    """
    Projects vector v onto the simplex with sum = target_sum and bounds [0,1]
    Uses the standard algorithm for projection onto a simplex with box constraints
    """
    n = len(v)
    # sort in descending order
    sorted_v = np.sort(v)[::-1]
    # compute cumulative sums
    cumsum = np.cumsum(sorted_v)
    # find the threshold
    idx = np.arange(1, n + 1)
    # compute the threshold that satisfies the sum constraint
    # we want to find tau such that sum(max(0, min(1, v[i] - tau))) = target_sum
    # after sorting v, we find the largest k such that the sum with threshold
    # tau = (cumsum[k] - target_sum) / k is satisfied
    # compute the threshold tau
    temp = (cumsum - target_sum) / idx
    # Find the largest k for which the threshold yields a valid hull
    # the valid case is where all elements in the first k elements are >= tau
    # but we want to cover all elements
    # get indices where threshold works for all elements up to k
    # Reconstruct the valid projection
    # Need to check: descending order of values
    # For each element, we compute final value as max(0, min(1, v[i] - tau))
    # Let me implement correctly:
    # V = sort(v, descending)
    # Let S_k = sum_{i=1}^k V_i, we compute for each k
    # tau_k = (S_k - target_sum) / k
    # We want validitas so that tau_k should be such that max(0, v_i - tau_k) = v_i - tau_k >= 0
    # So v_i >= tau_k for all i <= k. But v_i = V_k, so we need V_k >= tau_k = (S_k - target_sum) / k
    # This is redundant given that v_i >= v_{i+1}, so we need to find largest k such that
    # tau_k >= V_k, i.e., (S_k - target_sum) / k >= V_k => S_k - target_sum >= k * V_k => S_k - k*V_k >= target_sum
    mask = np.arange(1, n + 1) * sorted_v - cumsum >= target_sum
    if np.any(mask):
        k = np.argmax(mask)
        if k == 0:
            k = 0  # which means we take entire range
        else:
            k = k - 1  # since argmax gives first True, but we want the last False -> first True
    else:
        k = n - 1
    # k is now: largest such that condition is satisfied
    
    # If condition not satisfied for any k, it means no such valid k exists.
    # But in theory there should be a solution
    if k < 0:
        k = 0
    
    tau = (cumsum[k] - target_sum) / (k + 1)
    if tau < 0:
        # If no solution to apply threshold properly, return valid projection
        result = np.clip(v - tau, 0, 1)
        # adjust sum to target via row-wise correction
        diff = target_sum - np.sum(result)
        if abs(diff) > 1e-10:
            # Add to smallest elements
            sorted_indices = np.argsort(result)  # smallest first
            # Add diff to smallest items
            for i in reversed(sorted_indices):  # from smallest to largest
                if abs(diff) < 1e-10:
                    break
                add = min(1.0 - result[i], diff)
                if add > 0:
                    result[i] += add
                    diff -= add
        return result
    else:
        new_v = np.clip(v - tau, 0, 1)
        # correct sum
        diff = target_sum - np.sum(new_v)
        if abs(diff) > 1e-10 and len(new_v) > 0:
            # Add diff to smallest (closest to zero in value)
            sorted_indices = np.argsort(new_v)
            for i in reversed(sorted_indices):  # from smallest to largest
                if abs(diff) < 1e-10:
                    break
                add = min(1.0 - new_v[i], diff)
                if add > 0:
                    new_v[i] += add
                    diff -= add
        return new_v

def compute_gradient(h_values, correlation):
    """Compute the subgradient for the maximum correlation term"""
    n = len(h_values)
    dx = 2.0 / n
    # Find the maximum correlation value and its indices
    max_val = np.max(correlation)
    worst_indices = np.where(np.abs(correlation - max_val) < 1e-8)[0]
    
    # Already scaled by dx, so max_val = actual value * dx;
    # to get the true max correlation without dx scaling, we do max_val / dx
    # But for optimization, not scaling changes optimal h, only the constant factor
    # Since we will when evaluating use actual dx later, it’s industry; let’s keep it normalized
    
    # Now compute subgradient vector (see derivation)
    subgrad = np.zeros_like(h_values)
    
    # correlation is indexed from -(n-1) to (n-1); index i corresponds to shift (i - (n-1))
    # Our target function is sum h_i * (1 - h_{i+k}) = sum (h_i - h_i h_{i+k}) = sum h_i - sum h_i h_{i+k}
    # The value is slightly less than discretization due to the full correlation computation mode 
    # we have correlation[i] = sum_{j=0}^{n-1} h[j] * (1- h[j + i - (n-1)])
    # So i in range [0, 2(n-1)-1];
    # i = 0 corresponds to lag = (n-1) - 0 = n-1  => lag = n-1 (which shifts 0 elements)
    # The leftmost (i=0): 0th term, lag = n-1
    # Lag k = from-(n-1) to (n-1), so for correlation index i, the actual k = i - (n-1)
    # Thus we want indices where abs(correlation[i] - max_val) is small
    # And for k = i - (n-1), k can range from -(n-1) to (n-1)
    # The effective range of j goes from 0 to n-1, 
    # and i-k (the shifted accumulator axis) goes from 0 to n-1 too.
    # We diffuse the impacts
    for i in worst_indices:
        k = i - (n-1)
        # k = shift amount: -n+1 <= k <= n-1
        # the effective indices according to correlation definition
        # It is different than in direct correlation implementation
        # Let's take heterogeneous look: for implicit inner product:
        # correlation indices i are for shifts k = i - (n-1) 
        # also correlation = 0 for no valid elements, including shift beyond domain
        # For instance k = -1, shift 1 leftwards on j from 0 to n-1 to 1 to n.
        # So it’s considered in terms of: h[j] * (1 - h[j + k]), if j+k valid
        # This means deterministically for kثم, h[j] (from prev) and h[j+k], where for positive k the end j+k must be at most n-1.
        # For k >= 0: 1 to n-k
        # For k < 0: 0 to n+k-1 and j used over 0 to n-1-k, 
        # We overlap:  j+k >= 0 AND j <= n-1 => j >= -k, j <= n-1 
        # Thus j over range [-k, n-1] AND [0, n-1-k] => max(0, -k) to min(n-1, n-1-k) = 
        # = max(0, -k) to n-1-min(0, k) = max(0, -k) to min(n-1, n-1-k)
        # Case by case: 
        #   k >= 0: [0, n-1-k] => valid indices for j since j <= n-1-k => all j in [0, n-1-k]
        #   k < 0: [k, n-1] so intersection with 0 to n-1-k-1 is [max(0, k), min(n-1, n-1-k-1)] = max(0, k) to n-1-k
        # But no confusion since what assumption is made?
        # Ok, abuse case: let's take 0th element of correlation: index 0 => k to abs k  = (0 - n+1) = -n+1
        # We consider correlation(i) = sum_{j = 0}^{n-1} h[j] * (1 - h[j + i - (n-1)] )
        # For i = n-1: h[j] * (1 - h[j + i - (n-1)]) = h[j] * (1 - h[j + i - (n-1)] )
        # i = n-1 => k = 0, that is: h[j] * (1 - h[j]) = h[j] - h[j]^2
        # k = 1 => shift up 1, index j to j+1; correlation(n) = sum_{j=0}^{n-1} h[j] * (1 - h[j+1])
        # k = -1 => shift down 1 (positive shift right below): 
        # ... h[j] * (1 - h[j-1]) ... 
        # TODO we need a canonical way to compute this discretely
        
        # Easier: let's more directly convert from computation:
        # Let's reuse the helper function corr:
        # For large headers having indexing logic from our own side
        
        # Specific form for shift k: correlation(vec,vec,k) is convolution with different mode
        # Better Idris compute proper indices:
        if k >= 0:  
            # k positive: sum {j=0 to n-1-k} h[j] * (1-h[j+k])
            # This means for index j: term h[j], one term  h[j+k] 
            # So g is indexed like:
            # g[j] += 1 - h[j+k] 
            # g[j+k] -= h[j] 
            # Check ranges are valid:
            # For term j: h[j] * (1-h[j+k]), valid if (j+k) < n (i.e., j <= n-1-k)  
            # For term (j+k) version h[j]: must have j in [0, n-1-k]
            # So: index full = j from 0 to n-1-k inclusive, size is n-k (at least n-k), 
            # Add:
            # g[0:(n-k)] += (1 - h[k:])
            # g[k:] -= h[0:(n-k)]
            end2 = n - k
            g_1 = np.zeros_like(h_values)
            g_1[:end2] += 1 - h_values[k:]
            g_1[k:] -= h_values[:end2]
            
        else:  # k < 0
            # k = -|k|, sum over j where 0 <= j+k <= n-1 =>  k <= j <= n-1-k
            # i in [0, n-1] so j + k is also valid to have 0 <= j <= n-1 with 0 <= j+k <= n-1
            # j + k >= 0 means j >= -k, which is now j >= |k|
            # j <= n-1-k 
            # so valid j: [max(0, -k), min(n-1, n-1-k)]
            # This gives valid first j: max(0, |k|) to n-1-|k|
            # Since k < 0:  e.g., k = -2, j ∈ [2, n-1-2] 
            # The terms affect:
            # g[j] += (1-h[j+k]) 
            # Here for k = -2, j in [2, n-3], terms h[j]* (1 - h[j-2])
            # g[j] += (1 - h[j-(-2)]) = (1 - h[j+k' for k'=-2])
            # But we know this is g[j+k] - h[j], so g[j] += 1 - h[j+k] and g[j+k] -= h[j]
            # With k negative: 
            # in range [max(0, -k), n-1]
            # for sum terms h[j] * (1-h[j + k]) = sum_{j=0}^{n-1} 
            # the shift is l.k. for l in [0, n-1] then h[j] present where h[l] = h[j-k]
            # That is for shift k < 0:
            #         for each term h[l] * (1-h[l-k]) 
            # sum over l. But l-k should in [0, n-1] too. 
            # To ensure l-k >= 0 => l >= k.
            # l-k <= n-1 => l <= n-1+k.  
            # l >= 0 and l <= n-1+k. 
            # (including k<0)
            # Here l = j but treated as h[i], not shifted
            # More elucidating (original notation): h[j] * (1 - h[j+k]), terms h[j] for j such that j+k in valid range [0,n-1], so j in [0,n-1], j+k in [0,n-1]
            # That is, j+ k >= 0 => j >= -k = -min(0,k)  and j+k <= n-1 => j <= n-1-k = n-1+|k| 
            # but also j in [0,n-1], take all
            # valid j is: max(0, -k) to min(n-1, n-1-k) = max(0, -k) to n-1-k
            # For k=-1, this gives j in [1, n-2] 
            # We know for shift k in correlation index: 
            # Realization (pair indices) such that terms h[left]* (1 - h[left + k])
            # The contribution for j to gradient from k shift:
            # One term h[j] 
            # Another term h[j+k]
            
            # Preference: let's iterate through only valid l (j) and add corresponding induced terms in gradient  
            left_start = max(0, -k)
            left_end = min(n-1, n-1-k)  
            if left_start < left_end: # check they are a valid range
                # case of k < 0: 
                g_1 = np.zeros_like(h_values)
                g_1[left_start:left_end] += 1 - h_values[left_start+k:left_end+k]
                g_1[left_start+k:left_end] -= h_values[left_start:left_end]
            else:
                g_1 = np.zeros_like(h_values)
            # model correctly term in [0, n-1]
        subgrad += g_1  
    # normalize to start with:
    subgrad = subgrad / len(worst_indices)
    return subgrad

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Optimizer parameters:
    max_iter = 100000
    step_size = 0.002
    t0 = time.time()
    
    for it in range(max_iter):
        # Check if budget exceeded (except for final iteration completed)
        elapsed = time.time() - t0
        if elapsed >= budget_s * 0.95:
            break
        
        # compute the correlation: this is done by convolution with support range
        corr = np.correlate(h, 1-h, mode='full')
        
        # evaluate current bound
        c5 = np.max(corr) * dx
        
        # update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # compute gradient for the max
        grad = compute_gradient(h, corr)
        # Step
        h_new = h - step_size * grad
        # project
        h = project_to_simplex(h_new, target_sum)
        
        # occasionally adjust using random noise
        if it % 1000 == 0:
            # Add small noise to push out of local minima?
            noise = np.random.random(h.shape) * 0.0001
            # Re-project after noise too
            h = project_to_simplex(h + noise, target_sum)
    
    # final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
