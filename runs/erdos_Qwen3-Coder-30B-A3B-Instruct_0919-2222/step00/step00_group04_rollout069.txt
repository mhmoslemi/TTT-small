import cvxpy as cp
import numpy as np
from scipy.optimize import minimize
from math import sqrt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    remaining_time = budget_s

    N = 120  # Points in [0,2]
    dx = 2.0 / N
    max_shifts = N  # We consider all k up to N-1

    # Precompute shift matrices for all shifts
    A_k = []
    for k in range(N):
        A = np.zeros((N, N))
        # Copy ones from i to i+k for i=0,...,N-1-k
        for i in range(N - k):
            A[i, i + k] = 1
        A_k.append(A)

    # Convert the problem to SDP:
    # For each shift k, we'll have the constraint tr(A_k * X) >= gamma
    # where X = h * h^T (subject to X_ij = h_i * h_j)

    h = cp.Variable(N)
    X = cp.Variable((N, N), symmetric=True)

    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == h,  # diagonal elements agree with h
        h >= 0,
        h <= 1,
        cp.sum(h) == N / 2,
    ]

    # At this point we do a bisection on gamma.
    # Let's start with interval
    tau_low = 0.35
    tau_high = 0.5

    best_c5 = 1.0
    best_h = None

    def check_feasibility(tau):
        gamma = 1 - tau
        cons = constraints.copy()
        for A in A_k:
            cons.append(cp.trace(A @ X) >= gamma)
        prob = cp.Problem(cp.Minimize(0), cons)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return prob.status in ["optimal", "optimal_inaccurate"]
        except:
            return False

    # Bisection loop
    iter_count = 0
    while tau_high - tau_low > 1e-4 and iter_count < 20 and time.time() - start_time < budget_s * 0.9:
        iter_count += 1
        tau = (tau_low + tau_high) / 2
        if check_feasibility(tau):
            tau_high = tau
        else:
            tau_low = tau

    tau_final = (tau_low + tau_high) / 2
    gamma_final = 1 - tau_final

    # Re-solve for the final gamma using a strictly feasible constraint
    final_constraints = constraints.copy()
    for A in A_k:
        final_constraints.append(cp.trace(A @ X) >= gamma_final)
    prob = cp.Problem(cp.Minimize(0), final_constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h.value
            X_sol = X.value
        else:
            h_sol = None
    except:
        h_sol = None

    if h_sol is None:
        # Fallback to simpler direct construction
        h_sol = np.ones(N) * 0.5
        h_sol[30:90] = 1.0  # step function

    # Now extract a rank-1 solution (simple from eigenvalue decomposition)
    try:
        w, v = np.linalg.eigh(X_sol)
        rank1_approx = sqrt(w[-1]) * v[:, -1]  # Leading eigenpair
        # Clip and project
        clipped = np.clip(rank1_approx, 0, 1)
        # Ensure it sums to N/2 by rescaling
        current_sum = np.sum(clipped)
        scaled = clipped * (N / 2) / current_sum if current_sum != 0 else clipped

        # Project onto the feasible space: capped simplex
        def project_to_simplex(v, target_sum=N / 2):
            # Waterfill
            v_sorted = np.sort(v)[::-1]
            cumsum = np.cumsum(v_sorted)
            n = len(v_sorted)
            max_k = -1
            for k in range(n):
                if (cumsum[k] - (k+1) * v_sorted[k]) <= target_sum:
                    max_k = k
            if max_k == -1:
                return np.zeros_like(v)
            theta = (cumsum[max_k] - target_sum) / (max_k + 1)
            proj = np.maximum(v - theta, 0)
            # Further ensure exact sum
            final_sum = np.sum(proj)
            adj = (target_sum - final_sum) / len(proj)
            return proj + adj

        h_candidate = project_to_simplex(scaled)
    except:
        h_candidate = np.clip(h_sol, 0, 1)
        # Project
        def project_to_simplex(v, target_sum=N / 2):
            v_sorted = np.sort(v)[::-1]
            cumsum = np.cumsum(v_sorted)
            n = len(v_sorted)
            max_k = -1
            for k in range(n):
                if (cumsum[k] - (k+1) * v_sorted[k]) <= target_sum:
                    max_k = k
            if max_k == -1:
                return np.zeros_like(v)
            theta = (cumsum[max_k] - target_sum) / (max_k + 1)
            proj = np.maximum(v - theta, 0)
            return proj
        h_candidate = project_to_simplex(h_candidate)

    # Polishing via projected gradient descent
    def evaluate_c5(h_vec):
        # Do correlations via numpy correlate
        c = np.correlate(h_vec, 1-h_vec, mode="full")
        return max(c * dx)

    def compute_subgradient(h_vec, k_star):
        # Compute gradient for max index k_star
        n = len(h_vec)
        grad = np.zeros_like(h_vec)
        for i in range(n):
            if i + k_star < n:
                grad[i] += (1 - h_vec[i + k_star])
            if i - k_star >= 0:
                grad[i] -= h_vec[i - k_star]
        return grad

    def project_simplex(v, target_sum=N / 2):
        v_sorted = np.sort(v)[::-1]
        cumsum = np.cumsum(v_sorted)
        n = len(v_sorted)
        max_k = -1
        for k in range(n):
            if (cumsum[k] - (k+1) * v_sorted[k]) <= target_sum:
                max_k = k
        if max_k == -1:
            return np.zeros_like(v)
        theta = (cumsum[max_k] - target_sum) / (max_k + 1)
        proj = np.maximum(v - theta, 0)
        return proj

    h_polished = h_candidate.copy()
    alpha0 = 0.01
    max_iter = 500

    for t in range(max_iter):
        if time.time() - start_time > budget_s * 0.99:
            break
        c5_vals = np.correlate(h_polished, 1-h_polished, mode="full") * dx
        k_star = np.argmax(c5_vals)
        c5 = c5_vals[k_star]
        grad = compute_subgradient(h_polished, k_star)
        alpha = alpha0 / sqrt(t + 1)
        h_new = h_polished - alpha * grad
        # Project
        h_projected = project_simplex(h_new)
        h_polished = h_projected

    c5_bound = evaluate_c5(h_polished)
    return (h_polished.tolist(), c5_bound, N)
