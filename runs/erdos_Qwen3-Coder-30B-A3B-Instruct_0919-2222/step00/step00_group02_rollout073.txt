import numpy as np

def evaluate_overlap(h):
    """Compute the maximum overlap integral for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_overlap = corr.max()
    return max_overlap * dx

def get_mismatched_pairs(a, s):
    """Get mismatched indices for shift s in binary vector a."""
    n = len(a)
    if s >= 0:
        M1 = np.where((a == 1) & (np.roll(a, -s) == 0))[0]
        M0 = np.where((a == 0) & (np.roll(a, -s) == 1))[0]
    else:
        M1 = np.where((a == 1) & (np.roll(a, -s) == 0))[0]
        M0 = np.where((a == 0) & (np.roll(a, -s) == 1))[0]
    return M1, M0

def improve_binary(a, max_iter=2000):
    """Improve binary vector a via local swaps on worst shift."""
    n = len(a)
    a = a.copy()
    corr = np.correlate(a, 1-a, mode='full')
    best_val = corr.max()
    no_improvement_count = 0
    
    for _ in range(max_iter):
        s = corr.argmax() - (n-1)
        M1, M0 = get_mismatched_pairs(a, s)
        
        if len(M1) > 0 and len(M0) > 0:
            i = np.random.choice(M1)
            j = np.random.choice(M0)
            # Perform the swap
            a[i], a[j] = a[j], a[i]
            corr = np.correlate(a, 1-a, mode='full')
            new_val = corr.max()
            if new_val <= best_val:
                best_val = new_val
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                if no_improvement_count >= n:
                    # Random balanced swap fallback
                    ones = np.where(a == 1)[0]
                    zeros = np.where(a == 0)[0]
                    if len(ones) > 0 and len(zeros) > 0:
                        i = np.random.choice(ones)
                        j = np.random.choice(zeros)
                        a[i], a[j] = a[j], a[i]
                        corr = np.correlate(a, 1-a, mode='full')
                        best_val = corr.max()
                        no_improvement_count = 0
        else:
            # Random balanced swap fallback
            ones = np.where(a == 1)[0]
            zeros = np.where(a == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                a[i], a[j] = a[j], a[i]
                corr = np.correlate(a, 1-a, mode='full')
                best_val = corr.max()
                no_improvement_count = 0
                
        if no_improvement_count >= 2*n:
            break
            
    return a, best_val

def simulated_annealing_binary(a, max_iter=5000):
    """Run simulated annealing to escape local minima."""
    n = len(a)
    a = a.copy()
    corr = np.correlate(a, 1-a, mode='full')
    best_val = corr.max()
    best_a = a.copy()
    T0 = 0.5
    Tf = 1e-4
    T = T0
    factor = (Tf / T0) ** (1.0 / max_iter)
    
    for i in range(max_iter):
        # Random balanced swap
        ones = np.where(a == 1)[0]
        zeros = np.where(a == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i_idx = np.random.choice(ones)
            j_idx = np.random.choice(zeros)
            a[i_idx], a[j_idx] = a[j_idx], a[i_idx]
            corr = np.correlate(a, 1-a, mode='full')
            new_val = corr.max()
            delta = new_val - best_val
            if delta <= 0:
                best_val = new_val
                best_a = a.copy()
            else:
                if np.random.rand() < np.exp(-delta / T):
                    best_val = new_val
                    best_a = a.copy()
                else:
                    a[i_idx], a[j_idx] = a[j_idx], a[i_idx]  # swap back
        T *= factor
    
    return best_a, best_val

def optimize_fractional(h, alpha_vals=[1,5,10,20,40,80], steps_per_alpha=200, eta_0=0.2):
    """Refine fractional solution with smooth maximum gradient descent."""
    n = len(h)
    h = h.copy().astype(float)
    h = np.clip(h, 0, 1)
    dx = 2.0 / n
    
    def compute_overlaps(h):
        overlaps = np.array([np.sum(h * np.roll(1-h, k)) for k in range(-n+1, n)])
        return overlaps
    
    def project_onto_box_and_sum(h, target_sum):
        v = h.copy()
        low = 0
        high = 1
        def phi(lam):
            return np.sum(np.clip(v - lam, 0, 1))
        # Binary search for lambda
        tol = 1e-12
        iters = 0
        while low < high and iters < 80:
            mid = (low + high) / 2
            val = phi(mid)
            if abs(val - target_sum) < tol:
                break
            elif val > target_sum:
                low = mid
            else:
                high = mid
            iters += 1
        lam = (low + high) / 2
        h_proj = np.clip(v - lam, 0, 1)
        return h_proj
    
    prev_best = compute_overlaps(h).max()
    
    for alpha in alpha_vals:
        step_size = eta_0 / np.sqrt(1)
        for _ in range(steps_per_alpha):
            overlaps = compute_overlaps(h)
            exp_o = np.exp(alpha * (overlaps - overlaps.max()))
            weights = exp_o / np.sum(exp_o)
            
            grad = np.zeros_like(h)
            # Compute gradients for each shift k
            for k in range(-n+1, n):
                if k >= 0:
                    grad += weights[k+n-1] * (1 - np.roll(h, k))
                else:
                    grad += weights[k+n-1] * (1 - np.roll(h, k))
            
            h = h - step_size * grad
            h = project_onto_box_and_sum(h, n/2)
            
            # Check current max overlap
            curr_overlaps = compute_overlaps(h)
            current_max = curr_overlaps.max()
            if current_max < prev_best:
                prev_best = current_max
                h_best = h.copy()
    
    return h_best, prev_best

def binary_search(n, budget_s):
    """Perform binary local search with restarts."""
    best_val = np.inf
    best_h = None
    t_start = time.time()
    
    max_restarts = max(1, int(budget_s / 10))
    
    for _ in range(max_restarts):
        if time.time() - t_start > budget_s - 0.1:
            break
        # Random balanced starting vector
        h = np.zeros(n)
        ones = np.random.permutation(n)[:n//2]
        h[ones] = 1
        h = h.astype(int)
        
        h, val = improve_binary(h, max_iter=1000)
        h, val = simulated_annealing_binary(h, max_iter=2000)
        
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val
