from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value of the correlation is the C5 bound
    c5 = np.max(corr) * dx
    return c5
