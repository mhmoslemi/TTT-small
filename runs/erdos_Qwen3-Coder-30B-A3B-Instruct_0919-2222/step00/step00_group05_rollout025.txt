import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    h = np.array(h)
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations using numpy's correlate
    corr = np.correlate(h, 1 - h, mode='full')
    # Only take the relevant part (the "full" correlation has length 2*n-1)
    # The middle part corresponds to k from 0 to 2
    mid_idx = n - 1
    # We select correlations for k from 0 to 2, which is 
    # indices [mid_idx, mid_idx+1, ..., mid_idx+n-1] 
    # But that's wrong, we want to map k values in [0,2] 
    # and k = i*dx where i runs from 0 to n-1 
    # The k values in our context are these:
    # k values = [0*dx, 1*dx, 2*dx, ..., (n-1)*dx] 
    # but we map i back to k = i*dx = i * dx
    # Our correlation function has offset, but we just pick the maximum
    c5 = np.max(corr[mid_idx:])
    return c5 * dx

def initial_h_values():
    # From initial construction (70 points)
    n = 70
    h = np.zeros(n)
    dx = 2.0 / n
    # We'll construct an initial step function
    # Divide [0, 2] into intervals where the function 
    # alternates between 0 and some value to achieve integral 1
    # Let's make it symmetric and test a simple approach
    # Split into 4 parts of equal size: [0,0.5), [0.5,1), [1,1.5), [1.5,2)
    # Use a simple pattern for illustration
    
    # Let's find a special case, reuse a known construction
    # Sample different approaches
    
    # Approach 1: sawtooth-like
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    # Wave pattern
    h[x < 1.0] = 1.0
    h[x >= 1.0] = 0.0
    
    # Another pattern: simplest symmetric step function that integrates to 1
    h = np.zeros(n)
    midpoint = n // 2
    h[0:midpoint] = 2.0
    h[midpoint:n] = 0.0
    
    # Normalize to integral of 1
    h = h * (1.0 / (np.sum(h) * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initial construction - but also try others
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization loop
    max_iter = 30000
    iter_count = 0
    patience = 5000
    patience_count = 0
    
    while time.time() - start_time < budget_s - 2 and iter_count < max_iter:
        iter_count += 1
        
        # Random perturbation on h
        h_new = best_h.copy()
        # Pick a random point to perturb
        i = np.random.randint(0, n_points)
        # Add a small random change
        delta = np.random.uniform(-0.1, 0.1)
        h_new[i] += delta
        # Project to bounds
        h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Enforce the constraint with Lagrange multiplier or re-normalization
        integral = np.sum(h_new) * dx
        if abs(integral - 1.0) > 1e-10:
            scale = 1.0 / integral
            h_new = h_new * scale
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        # Sometimes try a more radical change if stuck
        if patience_count > patience // 2 and np.random.rand() < 0.2:
            h_new = best_h.copy()
            # Try modifying 5 random points
            indices = np.random.choice(n_points, 5, replace=False)
            for i in indices:
                h_new[i] = np.random.rand()
                h_new[i] = np.clip(h_new[i], 0, 1)
            integral = np.sum(h_new) * dx
            h_new = h_new * (1.0 / integral)
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                patience_count = 0

        # Early stopping if we got good enough
        if best_c5 < 0.379:
            break
    
    # Final evaluate
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Round h to reasonable precision if needed
    best_h = [round(x, 8) for x in best_h]
    
    return (best_h, final_c5, n_points)
