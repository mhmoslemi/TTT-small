import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.special import erf
import time
import math

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant for given h values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # Only consider the valid shifts (k from 0 to 2)
    # The valid range starts at n_points-1 and has length n_points
    valid_correlation = correlation[n_points-1:]
    max_overlap = np.max(valid_correlation) * dx
    return max_overlap

def initial_h_values():
    """Return initial construction for h values (45 points as mentioned)."""
    n_points = 45
    # Create an initial step function
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    # We'll try a two-step function
    # First, a step from 0 to 1
    # Place a step somewhere around x = 0.5 to 1.0
    step_position = 0.75  # middle of interval
    step_index = int(step_position / dx)
    
    # For simplicity, make a simple step function:
    # 0 for first half, 1 for second half, but scaled to sum to 1
    h[:step_index] = 0.0
    h[step_index:] = 1.0
    
    # Scale to have integral equal to 1
    h = h * (1.0 / (np.sum(h) * dx))
    
    return h

def objective(h_values, dx):
    """Compute the maximum overlap integral."""
    # Create the correlation
    correlation = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # Extract the relevant part (valid shifts from 0 to 2)
    # The full convolution is length 2*n - 1, 
    # middle index is n-1, we want from n-1 to 2n-2
    valid_correlation = correlation[len(h_values)-1:]
    max_overlap = np.max(valid_correlation) * dx
    return max_overlap

def constraint_sum(h_values, n_points, dx):
    """Ensure that integral is 1."""
    return np.sum(h_values) * dx - 1.0

def constraint_bounds(h_values):
    """Ensure bounds on h values."""
    violations = []
    violations.extend(h_values < 0)   # lower bound
    violations.extend(h_values > 1)   # upper bound
    return np.sum(violations)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize with the initial construction
    h = initial_h_values()
    if len(h) != n_points:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)

    # Optimization bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Store the best solution found
    best_h = None
    best_c5 = float('inf')
    
    # Use a combination of search strategies
    # First try optimization with a few restarts
    for _ in range(10):
        # Random perturbation to break symmetry
        h_perturb = h.copy()
        noise_magnitude = 0.05
        h_perturb += np.random.normal(0, noise_magnitude, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (1.0 / (np.sum(h_perturb) * dx))
        
        # Use differential evolution for global search on smaller number of variables
        try:
            result = differential_evolution(
                lambda x: objective(np.clip(x, 0, 1), dx),
                bounds, 
                maxiter=200,
                popsize=15,
                seed=seed,
                disp=False
            )
            
            if not result.success:
                continue
                
            h_opt = np.clip(result.x, 0, 1)
            h_opt = h_opt * (1.0 / (np.sum(h_opt) * dx))
            c5_val = objective(h_opt, dx)
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                
        except Exception:
            pass
            
        # If time is running out, stop early
        if time.time() - start_time > 0.9 * budget_s:
            break
    
    # Try local optimization if we have a good solution
    if best_h is not None:
        try:
            # Use smaller optimization for faster convergence
            bounds_small = [(0, 1) for _ in range(n_points)]
            result = minimize(
                lambda x: objective(np.clip(x, 0, 1), dx),
                best_h,
                method='L-BFGS-B',
                bounds=bounds_small,
                options={'maxiter': 200, 'ftol': 1e-9},
                callback=None
            )
            
            if result.success:
                h_final = np.clip(result.x, 0, 1)
                h_final = h_final * (1.0 / (np.sum(h_final) * dx))
                c5_final = objective(h_final, dx)
                
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
                    
        except Exception:
            pass
    
    # Final check
    if best_h is None:
        # Fall back to coarse grid search
        best_h = initial_h_values()
        if len(best_h) != n_points:
            best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
            
        best_c5 = evaluate_erdos_solution(best_h)
        
        # Try refining it a bit
        h_try = best_h.copy()
        for i in range(n_points):
            if i > 0 and i < n_points-1:
                # Try small adjustments
                h_try[i] = np.clip(h_try[i] + 0.05 * np.random.randn(), 0, 1)
                h_try = h_try * (1.0 / (np.sum(h_try) * dx))
                c5_try = evaluate_erdos_solution(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                
    return (best_h, best_c5, n_points)
