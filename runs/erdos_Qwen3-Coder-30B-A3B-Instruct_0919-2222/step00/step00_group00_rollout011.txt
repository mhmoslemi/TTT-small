def compute_correlation(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap(h):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h)
    return c.max() * dx

def apply_delta(c, h, idx, delta):
    n = len(h)
    offset = n - 1
    # first term: for j from idx to n-1:
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # second term: for j from 0 to idx:
    for j in range(0, idx+1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(i, j, c, h):
    # First apply delta_i = -1 (h[i] -> 0)
    delta_i = -h[i]
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    
    # Then apply delta_j = +1 (h[j] -> 1)
    delta_j = 1 - h[j]
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def get_ones_zeros(h):
    ones = [i for i in range(len(h)) if h[i] > 0.5]
    zeros = [i for i in range(len(h)) if h[i] < 0.5]
    return ones, zeros

def targeted_swap(c, h):
    n = len(h)
    dx = 2.0 / n
    offset = n - 1
    
    max_idx = np.argmax(c)
    max_shift = max_idx - offset
    
    ones, zeros = get_ones_zeros(h)
    
    # Find candidates for swap to reduce maximum overlap at max_shift
    # for i in ones: if h[i + max_shift] == 0, and h[i] == 1
    # for j in zeros: if h[j + max_shift] == 1, and h[j] == 0
    
    candidates_i = []
    candidates_j = []
    
    # Check for positions where we can reduce overlap at shift max_shift
    for i in ones:
        if 0 <= i + max_shift < n:
            if h[i + max_shift] == 0:
                candidates_i.append(i)
    
    for j in zeros:
        if 0 <= j + max_shift < n:
            if h[j + max_shift] == 1:
                candidates_j.append(j)
    
    if candidates_i and candidates_j:
        i = random.choice(candidates_i)
        j = random.choice(candidates_j)
        return i, j
    return None, None

def hill_climb(h, n_iter, time_left):
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    
    c = compute_correlation(h)
    current_max = c.max() * dx
    best_h = h.copy()
    best_max = current_max
    
    # Track indices with values 0 and 1
    ones, zeros = get_ones_zeros(h)
    
    iteration = 0
    for i in range(n_iter):
        if time.time() - start_time > time_left:
            break
            
        # Try targeted swap for max shift first
        i_swap, j_swap = targeted_swap(c, h)
        
        if i_swap is not None:
            swap(i_swap, j_swap, c, h)
            # Update sets
            ones = [idx for idx in ones if idx != i_swap]
            zeros = [idx for idx in zeros if idx != j_swap]
            ones.append(j_swap)
            zeros.append(i_swap)
        else:
            # Fallback to random swap
            if len(ones) > 0 and len(zeros) > 0:
                i_swap = random.choice(ones)
                j_swap = random.choice(zeros)
                swap(i_swap, j_swap, c, h)
                # Update sets
                ones.remove(i_swap)
                zeros.remove(j_swap)
                ones.append(j_swap)
                zeros.append(i_swap)
                
        # Recompute max
        current_max = c.max() * dx
        if current_max < best_max:
            best_max = current_max
            best_h = h.copy()
            
        iteration += 1
        
    return best_h, best_max
