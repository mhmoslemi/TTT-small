def evaluate_erdos_solution(h, n_points):
    """Evaluate the C5 bound for a given step function h."""
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The correlation is of length 2*n_points - 1
    # The valid shifts are from -1 to 1, so we look at the middle part
    # or the full correlation since it's circular
    max_corr = np.max(correlation) * dx
    return max_corr
