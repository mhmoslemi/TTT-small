import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def solve_erdos_milp(n, time_limit):
    """Solve the Erdos minimum overlap problem using MILP."""
    prob = pulp.LpProblem("ErdosMinOverlapMILP", pulp.LpMinimize)
    
    # Variables
    h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
    T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
    
    # Constraint sum hi = n/2
    prob += pulp.lpSum([h[i] for i in range(n)]) == n/2

    # Product constraints and shift constraints
    for d in range(n):  # d from 0 to n-1
        # create list of y variables for this shift
        y_vars = []
        for i in range(n-d):
            y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
            y_vars.append(y)
            # McCormick envelope constraints for y = h_i * h_{i+d}
            prob += y <= h[i]
            prob += y <= h[i+d]
            prob += y >= h[i] + h[i+d] - 1
        # shift constraint: sum y >= n/2 - T
        prob += pulp.lpSum(y_vars) >= n/2 - T
    
    # objective: minimize T
    prob += T
    
    # solve
    solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit)
    prob.solve(solver)
    
    # check status
    status = pulp.LpStatus[prob.status]
    
    # get solution vector
    h_vals = np.array([pulp.value(h[i]) for i in range(n)], dtype=float)
    t_val = pulp.value(T)
    
    return h_vals, t_val

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    start_time = time.time()
    
    best_c5_bound = float('inf')
    best_h_values = None
    best_n_points = None
    
    # Prefer smaller n for faster computation
    candidate_ns = [100, 120, 140, 160, 180, 200]
    
    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 5:
            break
            
        try:
            h_vals, t_val = solve_erdos_milp(n, min(remaining_time, 300))
            
            # Calculate the actual C5 bound using evaluation function
            c5_bound = evaluate_erdos_solution(h_vals)
            
            if c5_bound < best_c5_bound:
                best_c5_bound = c5_bound
                best_h_values = h_vals.copy()
                best_n_points = n
                
        except Exception as e:
            # If something fails, continue
            continue
            
        # If we find a good enough solution, stop early
        if best_c5_bound <= 0.3807:
            break
    
    # If nothing worked well, try the initial solution
    if best_h_values is None:
        try:
            h_vals = np.array(initial_h_values)
            c5_bound = evaluate_erdos_solution(h_vals)
            best_c5_bound = c5_bound
            best_h_values = h_vals.copy()
            best_n_points = len(initial_h_values)
        except:
            # fallback to zeros
            best_h_values = np.zeros(100)
            best_n_points = 100
            best_c5_bound = 0.5

    return (best_h_values, best_c5_bound, best_n_points)
