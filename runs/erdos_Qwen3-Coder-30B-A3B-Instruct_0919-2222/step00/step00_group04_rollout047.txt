import time
import numpy as np
from scipy.optimize import differential_evolution
from scipy import fft

def project_to_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum x = target_sum"""
    v = np.asarray(v)
    n = len(v)
    if n == 0:
        return np.array([])
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    # Find threshold
    cumsum = np.cumsum(sorted_v)
    threshold = (cumsum - target_sum) / np.arange(1, n + 1)
    # Find the largest index where sorted_v[i] > threshold[i]
    idx = np.searchsorted(-threshold, -sorted_v, side="right")
    if idx == 0:
        tau = sorted_v[0] - target_sum / n
    else:
        tau = threshold[idx - 1]
    # Compute projection
    x = np.clip(v - tau, 0, 1)
    # Ensure sum is correct by correcting the last element
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        x[-1] += diff
    return x

def compute_autocorrelation_bounds(a, w, dx=1.0):
    """Compute min and max of G(k) = ∫ h(x) h(x + k) dx over all k."""
    M = len(a)
    if M == 0:
        return 0.0, 0.0
    # Compute overlap matrix
    # We discretize the k axis by dx, i.e., we sample on [0,2] with step dx
    # But for accurate calculation, compute the exact overlap, smarter
    # Actually, we want overlaps under shift. Discretize the "shifts" - take integer mag shifts from 0 to 2/dx
    # But let's just compute all shifts that matter and find max f(k) = 1 - G(k)
    # The integral constraint ∑ a_i w_i = 1
    
    # Get intervals [s_{i-1}, s_i)
    s = np.concatenate([[0], np.cumsum(w)])
    total_overlap = 0.0
    n_shifts = 200  # For small N evaluations, can do full evaluation
    # For accuracy we compute G for all shifts k ∈ [-2, 2]
    # But let's avoid too much computation by focusing on discrete shifts
    # Use exact computation of all overlaps
    
    overlaps = []
    for k in range(1 - n_shifts, n_shifts):
        k_cont = float(k) * 2.0 / n_shifts  # from -2 to +2
        # Compute G for shift k
        G = 0.0
        for i in range(M):
            for j in range(M):
                # Overlap: interval I_i = [s_i_prev, s_i) and interval I_j shifted = [s_j_prev + k, s_j + k)
                s_iprev, s_i = s[i], s[i+1]
                s_jprev, s_j = s[j], s[j+1]
                overlap_start = max(s_iprev, s_jprev + k_cont)
                overlap_end = min(s_i, s_j + k_cont)
                if overlap_start < overlap_end:
                    length = overlap_end - overlap_start
                    G += a[i] * a[j] * length
        overlaps.append(1 - G)  # f(k) = 1 - G(k)
    
    max_f = max(overlaps)
    return max_f

def objective_block_de(params, M=12):
    """Evaluate C5 bound using the block representation"""
    u = params[:M]
    v = params[M:]
    
    # Transform to ensure constraints
    w = 2.0 * np.exp(u) / np.sum(np.exp(u))
    a = 1.0 / (1.0 + np.exp(-v))
    
    if np.any(a < 0) or np.any(a > 1):
        return 1e10  # penalize infeasible points
    
    # Compute the integral check
    integral = np.sum(a * w)
    
    # If integral != 1 or violated, enforce with penalty
    penalty = 1e3 * abs(integral - 1.0) if abs(integral - 1.0) > 1e-5 else 0.0
    
    # The detailed computation of overlap for the sampled pattern:
    # We now evaluate the function for 100 points
    N = 200  # grid size for final evaluation
    dx = 2.0 / N
    h = np.zeros(N)
    
    s = np.concatenate([[0], np.cumsum(w)])
    for i in range(N):
        x = i * dx
        # Find correct index
        idx = np.searchsorted(s, x, side='right') - 1
        if idx < 0:
            idx = 0
        if idx >= M:
            idx = M - 1
        h[i] = a[idx]
    
    # Compute correlation using fft
    corr = fft.irfft(fft.rfft(h) * np.conj(fft.rfft(1.0 - h)), n=2*N-1)
    max_corr = np.max(corr)
    return max_corr + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters for block method
    M = 12
    bounds = [(-5.0, 5.0)] * (2 * M)
    
    def optimize_swarm():
        result = differential_evolution(
            objective_block_de,
            bounds,
            seed=seed,
            popsize=15,
            maxiter=50,
            disp=False
        )
        if not result.success:
            return None
        return result.x

    # Phase 1: Global search using DE
    best_params = None
    try:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 10.0
        if remaining <= 10.0:
            # Just initialize from known
            u_init = np.random.normal(0.0, 0.2, M)
            v_init = np.random.normal(0.0, 0.2, M)
            best_params = np.concatenate([u_init, v_init])
        else:
            best_params = optimize_swarm()
            if best_params is None:
                # fallback to initial
                u_init = np.random.normal(0.0, 0.2, M)
                v_init = np.random.normal(0.0, 0.2, M)
                best_params = np.concatenate([u_init, v_init])
    except Exception:
        # fallback to initial
        u_init = np.random.normal(0.0, 0.2, M)
        v_init = np.random.normal(0.0, 0.2, M)
        best_params = np.concatenate([u_init, v_init])

    # Phase 2: Construct the actual function with projected simplex
    u = best_params[:M]
    v = best_params[M:]
    
    w = 2.0 * np.exp(u) / np.sum(np.exp(u))
    a = 1.0 / (1.0 + np.exp(-v))
    
    # Ensure integral = 1 or adjust a[i]
    integral = np.sum(a * w)
    if abs(integral - 1.0) > 1e-5:
        # Rescale a to make integral = 1
        a = a / integral
    
    # Construct h on a fine grid
    N = 200
    dx = 2.0 / N
    h = np.zeros(N)
    s = np.concatenate([[0], np.cumsum(w)])
    
    for i in range(N):
        x = i * dx
        idx = np.searchsorted(s, x, side='right') - 1
        if idx < 0:
            idx = 0
        if idx >= M:
            idx = M - 1
        h[i] = a[idx]
    
    # Phase 3: Refinement with projected subgradient descent
    
    max_iter = 4000
    alpha0 = 0.2
    best_h = h.copy()
    best_c5 = None
    
    def compute_max_overlap(h_values):
        # Same eval as in objective
        shifted = 1.0 - h_values
        corr = fft.irfft(fft.rfft(h_values) * np.conj(fft.rfft(shifted)), n=2*N-1)
        return np.max(corr)
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        # Compute the current C5 (max correlation)
        current_corr = compute_max_overlap(h)
        # Actual gradient computation
        # Look for k such that corr[k] = current_corr
        idxs = np.where(np.abs(fft.irfft(fft.rfft(h) * np.conj(fft.rfft(1.0 - h)), n=2*N-1) - current_corr) < 1e-10)[0]
        if len(idxs) == 0:
            k = np.argmax(fft.irfft(fft.rfft(h) * np.conj(fft.rfft(1.0 - h)), n=2*N-1))
            
        # Take just one J=k to use gradient descent
        corr = fft.irfft(fft.rfft(h) * np.conj(fft.rfft(1.0 - h)), n=2*N-1)
        k = np.argmax(corr)
        # Gradient according to the formula:
        g = np.zeros(N)
        for i in range(N):
            if i + k < N:
                g[i] += (1.0 - h[i + k])  # count the + part
            if i >= k:
                g[i] -= h[i - k]  # count the - part
        # Step
        alpha = alpha0 / np.sqrt(t + 1)
        h = h - alpha * g
        # Projecting
        h = project_to_capped_simplex(h, N / 2.0)
        # Store best
        new_c5 = compute_max_overlap(h)
        if best_c5 is None or new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    # Final validation
    actual_c5 = compute_max_overlap(best_h)
    actual_c5_bound = actual_c5 * (2.0 / N)
    
    return (best_h.tolist(), actual_c5_bound, N)
