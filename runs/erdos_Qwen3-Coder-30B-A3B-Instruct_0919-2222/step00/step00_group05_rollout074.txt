    """Evaluate the C5 bound for given h values."""

        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral C5 is the maximum value of this correlation
    c5 = np.max(corr) * dx
    return c5
