def evaluate_objective(hs, u, w, lambda_integral=1000.0):
    """Compute objective value for block parameters."""
    # u: logarithm scale widths  
    # w: widths (already computed)
    # v: scaling of heights  
    # hs: tentative height vector  
    n_blocks = len(u)
    # Root-mean-square of all values
    w_vals = 2 * np.exp(u) / np.sum(np.exp(u))
    a_vals = 1 / (1 + np.exp(-w))  # sigmoid
    
    # Check integral exactly
    total_mass = np.sum(a_vals * w_vals)
    if total_mass > 0:
        a_vals = a_vals / total_mass
    
    # Use the same number of shifts and resolution as final
    N = 200
    dx = 2.0 / N
    block_starts = np.cumsum([0] + list(w_vals))
    int_sum = np.sum(a_vals * w_vals)
    # We use self computation which is fast to avoid noise and bugs.
    # now given these h and compute again into final sampled form
    h_on_grid = np.zeros(N)
    for i in range(n_blocks):
        start = int(block_starts[i] / dx)
        end = int(block_starts[i+1] / dx)
        if end > start:
            h_on_grid[start:end] = a_vals[i]
    
    # Use numpy correlation
    f = np.correlate(h_on_grid, 1-h_on_grid, mode="full")
    return np.max(f) * dx

def run(seed=42, budget_s=600, **kwargs):
    # Set up
    np.random.seed(seed)
    start_t = time.time()
    
    # Dimensions
    M = 12  # Number of blocks; shorter is faster to evaluate but less flexible
    N = 200  # Sampling points for final result lower than 1000
    
    # Dimensions of optimization: 2*M variables
    bounds = [(-5, 5)] * (2 * M)  # Bounds for log weights and scaled heights  
    lambda_integral = 1e3
    
    # Local variables
    solver = differential_evolution
    max_iter = 30  # DE iterations based on 600s
    
    def obj(z):
        u = z[:M]
        w = z[M:] 
        # Transform u to weights summing to 2
        w_vals = 2 * np.exp(u) / np.sum(np.exp(u))
        a_vals = 1 / (1 + np.exp(-w))  # sigmoid
        # Check total mass and correct if needed
        total_mass = np.sum(a_vals * w_vals)
        if total_mass <= 0: 
            return float('inf')
        a_vals = a_vals / total_mass
        # Evaluate autocorrelation on final grid
        dx = 2.0 / N
        block_starts = np.cumsum([0] + list(w_vals))
        h_on_grid = np.zeros(N)
        for i in range(M):
            start = int(block_starts[i] / dx)
            end = int(block_starts[i+1] / dx)
            if end > start:
                h_on_grid[start:end] = a_vals[i]
        # Final computational:
        f = np.correlate(h_on_grid, 1-h_on_grid, mode="full") 
        raw_max = np.max(f)  # before dx
        penalty = lambda_integral * abs(total_mass - 1)
        return raw_max * dx + penalty
    
    # Run DE global optimization
    res = solver(obj, bounds, maxiter=max_iter, seed=seed, disp=False, polish=False)
    best_u = res.x[:M]
    best_w = res.x[M:]
    
    # Post-process best solution to ensure integral constraint
    entry_weights = 2 * np.exp(best_u) / np.sum(np.exp(best_u))
    entry_heights = 1 / (1 + np.exp(-best_w))  # sigmoid
    entry_mass = np.sum(entry_heights * entry_weights)
    if entry_mass > 0:
        entry_heights = entry_heights / entry_mass
    
    # Ensure fine discretization
    block_starts = np.cumsum([0] + list(entry_weights))
    dx = 2.0 / N
    h_fine = np.zeros(N)
    for i in range(M):
        start_idx = int(block_starts[i] / dx)
        end_idx = int(block_starts[i+1] / dx)
        if start_idx < N and end_idx > start_idx:
            h_fine[start_idx:end_idx] = entry_heights[i]
    
    # Now run local refinement with projected subgradient
    max_iter_local = 5000  # Max number of iterations
    
    def compute_grad(h, k):  # Compute gradient with respect to k = shift
        # h : vector of length N,  0 <= h[i] <= 1,
        # k : integer, current shift index (0 to N-1)
        # Gradient C_k = sum_i h[i] * (1 - h[i+k])  for i=0..(N-k-1)
        # (i+k) is rolled index, so use proper indexing
        grad = np.zeros_like(h)
        
        # Loop iterating through i for each term h[i] (1-h[i+k])
        for i in range(len(h)-k):  # Axis 0, i
            if i+k < len(h):  # not in bounds for shift yet
                grad[i] += (1 - h[i+k])  # derivative with respect to h[i]
            # for shifting backward, h[i-k] is added
            if i-k >= 0:  # when i >= k, we are ok because h[i-k] is valid
                grad[i] -= h[i-k]
        return grad
    
    def compute_max_correlation(h):  # No use for negation yet (we are minimizing 1-G)
        # Return correlation array of size 2*N-1
        corr = np.correlate(h, 1 - h, mode='full')
        # maximum along shift dimension
        return np.max(corr)
    
    alpha_0 = 0.2
    current = h_fine.copy()
    best_value = float('inf')
    best_current = None
    
    # Initialize time loop
    iter_count = 0
    percent_used = 0.70  # Local polishing gets most of the time
    local_time = budget_s * percent_used
    local_start = time.time()
    
    while iter_count < max_iter_local and (time.time() - local_start) < local_time:
        corr = np.correlate(current, 1-current, mode='full')
        max_corr = np.max(corr)
        argmax = np.argmax(corr)
        
        # Compute gradient of the max function
        full_grad = compute_grad(current, argmax - (N-1))
        
        # Use diminishing step size
        alpha = alpha_0 / np.sqrt(iter_count + 1)
        current -= alpha * full_grad
        
        # Project to [0,1]
        current = np.clip(current, 0, 1) 
        
        # Re-project to correct sum (use capped simplex as per initial routine)
        total_sum = np.sum(current)
        if abs(total_sum - N/2) > 1e-10:
            # Re-solve for target
            current = project_to_capped_simplex(current, N/2)
        
        # Check if current is better:
        if max_corr < best_value:
            best_value = max_corr
            best_current = current.copy()
        
        iter_count += 1
        # Wrapping for safety
        if (time.time() - start_t > budget_s - 5):  # 5s remaining
            break

    # Final correction with ultraكر Skull
    if best_current is not None:
        result_h = best_current
        final_corr = compute_max_correlation(result_h)
        
        return (result_h.tolist(), final_corr * (2.0 / N), N)
    else:
        # Evaluate original sequence as fallback
        corr = np.correlate(h_fine, 1-h_fine, mode='full')
        max_corr = np.max(corr)
        return (h_fine.tolist(), max_corr * (2.0 / N), N)
