def evaluate_erdos_solution(h):
    """Evaluate the C5 constant from a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlaps for all shifts using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(overlap)
    return max_overlap * dx

def solve_lp_cg(n, end_time):
    dx = 2.0 / n
    # Start with a small set of constraints to avoid too large LPs
    K = {0, -1, 1, -2, 2}  # Seed shifts
    MAX_K = 30  # Cap on number of constraints
    
    # Initialize variables
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Constraints on h
    constraints = [
        cp.sum(h) == n / 2,
        h >= 0,
        h <= 1
    ]
    
    # Initialize auxiliary variables and McCormick envelopes
    # w[i,k] = h[i] * h[i+k]
    w = {}
    # For each shift k in K, generate the constraint
    for k in K:
        if k >= 0:
            # i ranges from 0 to n-1-k
            for i in range(n - k):
                w[(i, k)] = cp.Variable()
                constraints += [
                    w[(i, k)] >= h[i] + h[i + k] - 1,
                    w[(i, k)] <= h[i],
                    w[(i, k)] <= h[i + k],
                    w[(i, k)] >= 0
                ]
        else:
            # k < 0
            for i in range(-k, n):
                w[(i, k)] = cp.Variable()
                constraints += [
                    w[(i, k)] >= h[i] + h[i + k] - 1,
                    w[(i, k)] <= h[i],
                    w[(i, k)] <= h[i + k],
                    w[(i, k)] >= 0
                ]
    
    # Add initial constraints
    for k in K:
        row = []
        if k >= 0:
            for i in range(n - k):
                row.append(w[(i, k)])
            sum_w = cp.sum(row)
            constraints.append(n / 2 - sum_w <= t)
        else:
            for i in range(-k, n):
                row.append(w[(i, k)])
            sum_w = cp.sum(row)
            constraints.append(n / 2 - sum_w <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    # Iteratively strengthen constraints
    while time.time() < end_time and len(K) <= MAX_K:
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
            if prob.status != cp.OPTIMAL:
                break
            h_sol = h.value
            t_sol = t.value
            
            # Compute all overlaps (O_k) to find worst
            O_all = []
            for k in range(-n + 1, n):
                shift = np.zeros_like(h_sol)
                if k >= 0:
                    shift[k:] = h_sol[:-k]
                    if len(shift) != len(h_sol):
                        shift = shift[:len(h_sol)]
                else:
                    shift[:k] = h_sol[-k:]
                    if len(shift) != len(h_sol):
                        shift = shift[-len(h_sol):]
                o = np.sum(h_sol * (1 - shift))
                O_all.append(o)
            
            max_o = max(O_all)
            max_k = O_all.index(max_o)
            
            if max_o <= t_sol + 1e-6:
                # All constraints are satisfied, break
                break
                
            # Add constraint for worst shift
            K.add(max_k - n + 1)
            if len(K) > MAX_K:
                # To maintain constraint count for solvers
                # or shrunk set of constraints from including more
                break
                
            # Update constraints dynamically
            # All constraints with current k
            row = []
            for i in range(n - abs(max_k)):
                if max_k >= 0:
                    row.append(w[(i, max_k)])
                else:
                    row.append(w[(i, max_k)])
            sum_w = cp.sum(row)
            constraints.append(n / 2 - sum_w <= t)
        
        except Exception as e:
            break

    return h.value, t.value

def solve_sturmian(n, end_time):
    # Based on Sturmian words: a_i = 1 if {i * alpha + tau} < 0.5, 0 otherwise
    # This is an ordered grid search
    best_bound = float('inf')
    best_h = None
    
    # Grid search over alpha, tau
    alpha_min, alpha_max = 0.2, 0.8
    M = 200
    L = 100
    
    alpha_step = (alpha_max - alpha_min) / (M - 1)
    tau_step = 1.0 / L
    
    for i in range(M):
        if time.time() > end_time:
            break
        alpha = alpha_min + (i * alpha_step)
        for j in range(L):
            if time.time() > end_time:
                break
            tau = j * tau_step
            
            # Build binary sequence
            h = np.zeros(n)
            for idx in range(n):
                frac = (idx * alpha + tau) % 1
                h[idx] = 1 if frac < 0.5 else 0
            
            # Compute overlap
            overlap = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(overlap)
            
            # Adjust bound
            c5 = max_overlap * 2.0 / n
            
            if c5 < best_bound:
                best_bound = c5
                best_h = h.copy()
    
    # Local refinement if needed (a few local swaps)
    if best_h is not None:
        for i in range(min(5000, int(0.1 * n ** 2))):
            if time.time() > end_time:
                break
            # Find worst shift
            overlap = np.correlate(best_h, 1 - best_h, mode='full')
            k_best = np.argmax(overlap)
            max_o = overlap[k_best]
            
            # Try swap
            if max_o <= 2.0 / n:
                break
            else:
                # Swap two random conflicting indices
                # Find some conflicting index pair
                # For simplicity, greedy swap
                for idx in range(n):
                    if best_h[idx] == 1:
                        k = k_best - (n - 1) + idx
                        # Ensure k is valid and h[idx] != h[k]
                        if 0 <= k < n and k != idx and best_h[idx] != best_h[k]:
                            # Actually swap
                            old = best_h[idx]
                            best_h[idx] = 1 - best_h[idx]
                            best_h[k] = 1 - best_h[k]
                            break
            if i % 100 == 0:
                overlap = np.correlate(best_h, 1 - best_h, mode='full')
                max_o = np.max(overlap)
                c5 = max_o * 2.0 / n
                if c5 < best_bound:
                    best_bound = c5
                if c5 <= 0.3807:
                    break
                
    return best_h, best_bound
