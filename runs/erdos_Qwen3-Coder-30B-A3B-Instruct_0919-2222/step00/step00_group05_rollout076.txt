import time
import numpy as np
import cvxpy as cp
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    correlations = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(correlations)
    return max_correlation * dx

def build_mcormick_constraints(h, y, n_points):
    """Build McCormick envelope constraints for y[i,s] = h[i]*h[i+s]"""
    constraints = []
    # Ensure sum of h is n/2
    constraints.append(cp.sum(h) == n_points / 2)
    # Box constraints on h
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # For each shift s and index i
    for s in range(-n_points+1, n_points):
        for i in range(n_points):
            if 0 <= i+s < n_points:
                y_i_s = y[i, s + n_points - 1]  # Shift to fit into [0, 2*n-2] range
                # McCormick envelope constraints for y[i,s]
                constraints.append(y_i_s <= h[i])
                constraints.append(y_i_s <= h[i+s])
                constraints.append(y_i_s >= h[i] + h[i+s] - 1)
                constraints.append(y_i_s >= 0)
    return constraints

def generate_candidate_from_relaxed_solution(h_relaxed, n_points):
    """Generate binary candidate from relaxed solution."""
    h_round = np.zeros(n_points)
    # Assign 1s to the 'n/2' largest values
    indices = np.argsort(h_relaxed)[::-1][:int(n_points/2)]
    h_round[indices] = 1
    return h_round.tolist()

def reduce_correlation_via_swap(h, correlations):
    """Improve solution via targeted swaps for the worst shift."""
    n_points = len(h)
    # Compute shifts that contribute to correlations
    max_idx = np.argmax(correlations)
    s_max = max_idx - (n_points - 1)
    # Calculate actual max correlation value
    current_corr = correlations[max_idx]

    # Perform swaps until no improvement or time-out
    start_time = time.time()
    while time.time() - start_time < 5:  # Short loop
        swapped = False
        for i in range(n_points):
            if h[i] == 1 and h[(i + s_max) % n_points] == 0:
                # Look for a better target to swap with
                for j in range(n_points):
                    if h[j] == 0 and h[(j + s_max) % n_points] == 1:
                        # Perform the swap
                        h[i], h[j] = 0, 1
                        # Recalculate after swap
                        new_corr = np.max(np.correlate(h, 1 - h, mode='full'))
                        if new_corr < current_corr:
                            current_corr = new_corr
                            swapped = True
                        else:
                            # Rollback if no gain
                            h[i], h[j] = 1, 0
                        break
                if swapped:
                    break
        if not swapped:
            break
    return h, current_corr
