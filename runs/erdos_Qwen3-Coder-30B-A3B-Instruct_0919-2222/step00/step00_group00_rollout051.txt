def compute_correlation(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap(h):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h)
    return c.max() * dx

def apply_delta(c, h, idx, delta):
    """Update correlation after changing h[idx] by delta."""
    n = len(h)
    offset = n - 1
    # First term: h[i] * (1 - h[i+s]) for all valid i, s such that i+s = j
    # where j ranges from idx to n-1, and s = j - idx
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # Second term: h[i-s] * (1 - h[i]) for all valid i, s such that i-s = j
    # where j ranges from 0 to idx, and s = idx - j
    for j in range(0, idx+1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(i, j, c, h):
    """Swap h[i] and h[j] and update correlation c accordingly."""
    delta_i = 0 - h[i]
    delta_j = 1 - h[j]
    # Apply delta_i to index i
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    # Apply delta_j to index j
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def targeted_swap(c, h):
    """Find an optimal targeted swap based on current maximum correlation."""
    n = len(h)
    offset = n - 1

    s = max_idx - offset  # s = shift approximately
    if abs(s) <= 1:

        candidate_shifts = [s, 0, 1, -1] if s != 0 else [0, 1, -1]
        for shift in candidate_shifts:
            if shift == 0:
                continue
            max_idx = np.argmax(c)
            s = max_idx - offset
            if abs(s) <= 1:  # Skip if not significantly necessary
                continue
            # Get valid set of indices for such shift
            valid_indices_i = []
            valid_indices_j = []
            for i in range(n):
                j = i + s
                if 0 <= j < n:
                    if h[i] == 1 and h[j] == 0:
                        valid_indices_i.append(i)
                    elif h[i] == 0 and h[j] == 1:
                        valid_indices_j.append(i)
            if valid_indices_i and valid_indices_j:
                i = random.choice(valid_indices_i)
                j = random.choice(valid_indices_j)
                return i, j
    else:
        # General case: arbitrary swap
        valid_indices_i = [i for i in range(n) if h[i] == 1]
        valid_indices_j = [i for i in range(n) if h[i] == 0]
        if not valid_indices_i or not valid_indices_j:
            return None, None
        i = random.choice(valid_indices_i)
        j = random.choice(valid_indices_j)
        return i, j
    return None, None

def hill_climb(h, n_iter, time_left):
    """Perform iterative local search on h to minimize max overlap."""
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h).copy()
    current_val = c.max() * dx
    best_h = h.copy()
    best_val = current_val
    # Sets of ones and zeros for efficient swaps
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] <= 0.5]

    # Main loop
    for _ in range(n_iter):
        if time_left() <= 0.1:
            break
        # Try targeted swap first
        i, j = targeted_swap(c, h)
        if i is not None and j is not None:
            # Perform swap
            swap(i, j, c, h)
            # Update sets
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            # Fallback random swap
            if ones and zeros:
                i = random.choice(ones)
                j = random.choice(zeros)
                # Perform swap
                swap(i, j, c, h)
                # Update sets
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
        # Refetch objective
        new_max = c.max()
        current_val = new_max * dx
        if current_val < best_val:
            best_val = current_val
            best_h = h.copy()
        # Early stopping
        if best_val <= 0.379:
            break

    return best_h, best_val
