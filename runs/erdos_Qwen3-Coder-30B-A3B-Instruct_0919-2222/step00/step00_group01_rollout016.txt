import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts k
    correlation = np.correlate(h, 1 - h, mode="full")
    # The correlation gives us values for shifts from -n+1 to n-1
    # We want the maximum of the positive shifts (k = 1 to n-1)
    # But actually we want the maximum over all k from 1 to n-1
    # The maximum is at the center of the full correlation
    max_corr = np.max(correlation[n-1:])  # shifts k=1 to n-1
    return max_corr * dx
