import numpy as np
import cvxpy as cp
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value among all k
    max_overlap = np.max(correlation) * dx
    return max_overlap

def initial_h_values():
    """Initial construction of h from the problem description"""
    n = 45
    h = np.zeros(n)
    # Simple construction: piecewise constant with some structure
    # We'll make a simple two-piece construction 
    h[:n//2] = 0.5
    h[n//2:] = 0.0
    return h

def create_h_from_vars(vars, n_points):
    """Convert optimization variables to valid h values"""
    # vars are unconstrained but we need h[i] in [0,1]
    # Apply sigmoid transform to map to [0,1]
    h = 1 / (1 + np.exp(-vars))
    # Normalize so that sum(h)*dx = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    return np.clip(h, 0, 1)

def optimize_h(n_points, max_time_s):
    """Optimize h to minimize C_5"""
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Try multiple random initializations to avoid local optima
    for trial in range(10):
        if time.time() - start_time > max_time_s - 1:
            break
            
        # Initialize with a randomized vector
        np.random.seed(trial * 1000)
        h0 = np.random.rand(n_points)
        # Normalize to correct integral
        dx = 2.0 / n_points
        h0 = h0 * (n_points / 2) / np.sum(h0)
        
        # Optimization objective function
        def objective(vars):
            h = create_h_from_vars(vars, n_points)
            return evaluate_erdos_solution(h)
        
        # Bounds for optimization variables (unconstrained)
        bounds = [(-10, 10) for _ in range(n_points)]
        
        # Use differential evolution for robust global optimization
        try:
            result = differential_evolution(
                objective, 
                bounds, 
                maxiter=100, 
                popsize=15, 
                seed=trial,
                tol=1e-6,
                recombination=0.7,
                mutation=(0.5, 1)
            )
            if not result.success:
                continue
                
            h_opt = create_h_from_vars(result.x, n_points)
            current_c5 = evaluate_erdos_solution(h_opt)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
                
        except Exception:
            continue
            
        if time.time() - start_time > max_time_s - 1:
            break
            
    # If no optimization worked, use initial construction
    if best_h is None:
        best_h = initial_h_values()
        best_c5 = evaluate_erdos_solution(best_h)
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Use 100 points as a good balance of resolution and speed
    n_points = 100
    start_time = time.time()
    
    h_opt, c5_bound = optimize_h(n_points, budget_s)
    
    # Make sure we don't exceed time
    elapsed = time.time() - start_time
    if elapsed > budget_s - 0.1:
        # Use a simpler construction to meet time requirements
        h_opt = initial_h_values()
        c5_bound = evaluate_erdos_solution(h_opt)
    
    return (h_opt.tolist(), float(c5_bound), n_points)
