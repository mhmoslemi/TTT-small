import numpy as np
import time
from scipy import signal

def compute_correlation(h):
    """Compute correlation c_k = sum_i h[i] * (1 - h[i+k]) for all shifts k."""
    n = len(h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Return only the valid shifts (indices from -(n-1) to n-1)
    return corr[n - 1 : 2 * n - 1]

def smoothed_objective_and_weights(c, alpha):
    """Compute Φ_α = (1/α) * log(sum(exp(α * c))) and weights w_k."""
    max_c = np.max(c)
    c_shifted = c - max_c
    exp_c = np.exp(alpha * c_shifted)
    sum_exp = np.sum(exp_c)
    weights = exp_c / sum_exp
    objective = (1 / alpha) * (np.log(sum_exp) + max_c)
    return objective, weights

def grad_h(h, weights):
    """Compute gradient ∇Φ_α(h) using the weights and h."""
    n = len(h)
    grad = np.zeros(n)
    # Compute convolution with (1 - h) and h using FFT for efficiency
    # The gradient is sum_k w_k * (1 - h[i + k]) - h[i - k] for valid k
    # This is effectively (w * (1 - h)) + (w_rev * h)
    # where w_rev is weights shifted appropriately
    
    # Create extended version of (1-h) to compute full convolution
    # forward contribution (i.e. (1 - h)[i+k])
    forward_contrib = np.convolve(weights, 1 - h, mode='full')
    # backward contribution (h[i-k])
    backward_contrib = np.convolve(weights[::-1], h, mode='full')
    
    # Now sum them properly, site by site
    # For indices i=0..n-1 the gradient is:
    # sum_{k} w_k * (1 - h[i + k]) 1_{i+k < n} - h[i - k] 1_{i-k >=0}
    # which we can rewrite using definitions of forward/backward convolutions
    # But we already have the components
    # Let's treat the vector convolution result in a clever way:
    # for i from 0 to n-1: 
    # grad[i] = forward_contrib[n-1+i] - backward_contrib[n-1+i]
    
    # The shift and extent of results:
    offset = n - 1
    for i in range(n):
        # Forward convolution gives us sum_{k : i+k >= 0} w_k * (1 - h[i+k])
        # But we also need to account that only valid k are those where i+k < n
        # So the sum is over k where both i+k < n and k >= 0
        f_pos = i + offset
        back_pos = (n-1) + (n-1-i)
        grad[i] = forward_contrib[f_pos] - backward_contrib[back_pos]
    
    return grad

def project_onto_feasible(y, target_sum):
    """Project vector y onto feasible region: 0 <= y[i] <= 1 and sum(y) = target_sum."""
    # Clip to [0,1]
    y_clipped = np.clip(y, 0, 1)

    # Adjust so that sum equals target_sum
    total = np.sum(y_clipped)
    diff = total - target_sum
    if abs(diff) < 1e-12:
        return y_clipped

    # Find which components are free (not at bounds)
    free_mask = (y_clipped > 0) & (y_clipped < 1)
    free_indices = np.where(free_mask)[0]
    free_count = len(free_indices)
    
    if free_count == 0:
        # All are at bounds; can't adjust weights directly
        # However, we can just use a simple adjustment
        if diff > 0:
            # Need to decrease the sum by diff
            # Decrease all values between 1 and 0 unless they are already closed to the limits
            try:
                y_normalized = y_clipped / np.sum(y_clipped)
                y_projected = target_sum * y_normalized
            except:
                y_projected = y_clipped.copy()
        else:
            # Need to increase; not possible under the constraints unless we move some
            y_projected = y_clipped.copy()
        return y_projected

    # Iteratively adjust free components
    iterations = 0
    while abs(diff) > 1e-12 and iterations < 100:
        if diff > 0:
            d = diff / free_count
            for idx in free_indices:
                if y_clipped[idx] - d >= 0:
                    y_clipped[idx] -= d
                else:
                    y_clipped[idx] = 0.0
        else:
            d = -diff / free_count
            for idx in free_indices:
                if y_clipped[idx] + d <= 1:
                    y_clipped[idx] += d
                else:
                    y_clipped[idx] = 1.0
        total = np.sum(y_clipped)
        diff = total - target_sum
        free_mask = (y_clipped > 0) & (y_clipped < 1)
        free_indices = np.where(free_mask)[0]
        free_count = len(free_indices)
        iterations += 1

    return y_clipped

def swap_update(c, h, i, j):
    """Update correlation vector incrementally when swapping indices i and j."""
    n = len(h)
    off = n - 1
    old_max = np.max(c)
    
    # Change = -1 at i (since h[i] becomes 0 from 1)
    # Change = +1 at j (since h[j] becomes 1 from 0)
    # For a shift k, the change to c[k] is:
    # If i+k < n, and i+k >= 0: -1 * (1 - h[i+k]) = h[i+k] - 1
    # If i-k >= 0, and i-k < n: -1 * h[i-k] = -h[i-k]
    # Likewise for j change entry
    
    # A will store the  changes for each shift from swapping h[i] and h[j]
    # But since we're dealing with shift correlating:
    # c[k] = sum_i h[i] * (1 - h[i+k])
    # What happens?
    # Original: h[i] == 1, h[j] == 0 -> contributes -1 to c[0] (for shift = 0)
    # After: h[i] == 0, h[j] == 1 -> contributes +1
    
    # Let's use the incremental update method provided
    
    # Perform the actual swap
    h[i], h[j] = h[j], h[i]
    
    # Update the correlation vector
    # For a swap, the updated value is:
    # c[k] becomes c[k] + δ, where δ is the change contribution to that shift.
    # For each i' we need to update c[i'-j] and c[j-i'] (and similar).
    # Simpler: refresh the correlation.
    
    # Better: construct update directly
    # We need to incrementally update c using the old parts that change
    
    # Only indices where shift is within bounds are affected
    for k in range(-n+1, n):
        # Only call update for valid shifts
        idx_i_plus = i + k
        idx_j_plus = j + k
        idx_i_minus = i - k
        idx_j_minus = j - k
        
        # Update contribution from i:
        if 0 <= idx_i_plus < n:
            c[k + off] += (1 - h[j]) - (1 - h[i])
        if 0 <= idx_i_minus < n:
            c[k + off] -= h[j] - h[i]
        
        # Update contribution from j:
        if 0 <= idx_j_plus < n:
            c[k + off] += (1 - h[i]) - (1 - h[j])
        if 0 <= idx_j_minus < n:
            c[k + off] -= h[i] - h[j]
    
    return c

def targeted_swap(c, h, ONES, ZEROS, off):
    """Find pair for targeted swap based on worst shift."""
    max_shift = np.argmax(c)
    s_star = max_shift - off
    
    A = [i for i in ONES if 0 <= i + s_star < len(h) and h[i + s_star] == 0]
    B = [j for j in ZEROS if 0 <= j + s_star < len(h) and h[j + s_star] == 1]
    
    if not A or not B:
        # fallback to random swap
        if ONES and ZEROS:
            i = np.random.choice(ONES)
            j = np.random.choice(ZEROS)
            swap_update(c, h, i, j)
            ONES.discard(i)
            ZEROS.discard(j)
            ONES.add(j)
            ZEROS.add(i)
            return True
        return False

    i = np.random.choice(A)
    j = np.random.choice(B)
    swap_update(c, h, i, j)
    ONES.discard(i)
    ZEROS.discard(j)
    ONES.add(j)
    ZEROS.add(i)
    return True

def local_binary_search(h, off, time_limit):
    """Improve binary h via neighbor swaps."""
    start_time = time.time()
    n = len(h)
    c = compute_correlation(h)
    
    ONES = set(np.where(h == 1)[0])
    ZEROS = set(np.where(h == 0)[0])
    max_c = np.max(c)
    best_c = max_c

    iters_since_improvement = 0
    iteration_count = 0

    while time.time() - start_time < time_limit and iters_since_improvement < 2000:
        iteration_count += 1
        # Perform targeted swap
        if not targeted_swap(c, h, ONES, ZEROS, off):
            break

        new_max = np.max(c)
        if new_max < best_c:
            best_c = new_max
            iters_since_improvement = 0
        else:
            iters_since_improvement += 1

    return h, best_c

def upscale(h_input, n_new):
    """Linearly upsample binary h by repeating entries."""
    n_old = len(h_input)
    step_ratio = n_new // n_old
    if step_ratio == 0 or n_new % n_old != 0:
        return h_input  # not supported directly
    h_upsampled = np.repeat(h_input, step_ratio)
    return h_upsampled

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety_margin = 2.0

    # Test our bounds
    target_c5 = 0.379
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Discretization sizes to try
    ns = [80, 120, 180, 240]

    for i, n in enumerate(ns):
        if time.time() - start_time > budget_s - safety_margin:
            break

        # Generate initial point by random and project
        h = np.random.rand(n)
        h = project_onto_feasible(h, n/2)

        # Continuous phase
        alpha = 5.0
        alpha_max = 200.0
        eta = 0.5 * (2.0 / n)
        beta = 0.9
        h_prev = h.copy()
        h_ = h.copy()
        iterations = 0
        max_iter = 10000
        last_alpha_update = 0
        g_best = None
        c_best_val = float('inf')
        c_best = None

        while iterations < max_iter and time.time() - start_time < budget_s - safety_margin:
            c = compute_correlation(h)
            obj_val, weights = smoothed_objective_and_weights(c, alpha)

            # Compute gradient
            g = grad_h(h, weights)

            # Momentum update
            h_new = h - eta * (g + beta * (h - h_prev))
            
            h_new = project_onto_feasible(h_new, n/2)
            
            c_new = compute_correlation(h_new)
            obj_new, weights_new = smoothed_objective_and_weights(c_new, alpha)
            
            if obj_new <= obj_val:
                h = h_new.copy()
                h_prev = h.copy()
                if np.max(c_new) * (2.0 / n) < c_best_val:
                    c_best_val = np.max(c_new) * (2.0 / n)
                    c_best = c_new.copy()
                    g_best = g.copy()
            else:
                eta *= 0.5
                if eta < 1e-12:
                    break

            # Increasing alpha
            if iterations - last_alpha_update > 100 and alpha < alpha_max:
                alpha *= 1.2
                last_alpha_update = iterations
            
            iterations += 1

        # Final step: round to binary
        # Create a step function with exactly n/2 1s
        idxs = np.argsort(h[::-1])[:int(n//2)]
        h_binary = np.zeros(n)
        h_binary[idxs] = 1
        
        # Perform local polishing
        off = n - 1
        polish_time = min(100.0, budget_s / 2.0) if i == 0 else 1.0
        h_polished, polished_max = local_binary_search(h_binary, off, polish_time)
        actual_c5 = polished_max * (2.0 / n)
        
        if actual_c5 <= best_c5:
            best_c5 = actual_c5
            best_h = h_polished
            best_n = n

        # If we made it to the last n, break early
        if i == len(ns) - 1:
            break

    if best_h is None:
        # Fallback
        h_fallback = np.ones(100) / 2
        best_c5 = 0.38092
        best_h = h_fallback
        best_n = 100

    return best_h.tolist(), best_c5, int(best_n)
