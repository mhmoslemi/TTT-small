import numpy as np
import time
import random
from scipy.optimize import minimize_scalar
import math

# Constants

def evaluate_overlap(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max()

def evaluate_c5(h):
    return 2 * evaluate_overlap(h) / len(h)

def get_mismatch_sets(h, s):
    n = len(h)
    if s >= 0:
        M1 = [i for i in range(n-s) if h[i] == 1 and h[i+s] == 0]
        M0 = [j for j in range(s, n) if h[j] == 0 and h[j-s] == 1]
    else:
        s = -s
        M1 = [i for i in range(s, n) if h[i] == 1 and h[i-s] == 0]
        M0 = [j for j in range(n-s) if h[j] == 0 and h[j+s] == 1]
    return M1, M0

def binary_worst_shift_swap(h, max_steps=2000):
    n = len(h)
    h = h.astype(int).copy()
    current_max = evaluate_overlap(h)
    for _ in range(max_steps):
        corr = np.correlate(h, 1 - h, mode='full')
        max_cnt = corr.max()
        s = corr.argmax() - (n - 1)
        if max_cnt < current_max:
            current_max = max_cnt
        if max_cnt == current_max:
            M1, M0 = get_mismatch_sets(h, s)
            if len(M1) > 0 and len(M0) > 0:
                i = random.choice(M1)
                j = random.choice(M0)
                h[i], h[j] = h[j], h[i]
                new_max = evaluate_overlap(h)
                if new_max <= current_max:
                    current_max = new_max
                else:
                    h[i], h[j] = h[j], h[i]
            else:
                # fallback to random swap
                ones = [i for i in range(n) if h[i] == 1]
                zeros = [j for j in range(n) if h[j] == 0]
                if ones and zeros:
                    i = random.choice(ones)
                    j = random.choice(zeros)
                    h[i], h[j] = h[j], h[i]
                    new_max = evaluate_overlap(h)
                    if new_max <= current_max:
                        current_max = new_max
                    else:
                        h[i], h[j] = h[j], h[i]
        else:
            return h.copy(), current_max
    return h.copy(), current_max

def simulated_annealing_binary(h, max_iters=5 * 100 * math.log(100)):
    n = len(h)
    h = h.astype(int).copy()
    T0, Tf = 0.5, 1e-4
    current_max = evaluate_overlap(h)
    best_h = h.copy()
    best_max = current_max
    T = T0
    T_decay = (Tf / T0) ** (1.0 / max_iters)
    for _ in range(max_iters):
        # Propose random swap
        ones = [i for i in range(n) if h[i] == 1]
        zeros = [j for j in range(n) if h[j] == 0]
        if ones and zeros:
            i = random.choice(ones)
            j = random.choice(zeros)
            h[i], h[j] = h[j], h[i]
            new_max = evaluate_overlap(h)
            delta = new_max - current_max
            if delta <= 0 or random.random() < math.exp(-delta / T):
                current_max = new_max
            else:
                h[i], h[j] = h[j], h[i]
            if current_max < best_max:
                best_h = h.copy()
                best_max = current_max
        T *= T_decay
    return best_h, best_max

def binary_search(n, time_budget):
    start_t = time.time()
    max_iters = 5  # number of restarts
    best_h = None
    best_val = float('inf')

    for _ in range(max_iters):
        if time.time() - start_t > time_budget - 1:
            break
        # make balanced binary vector
        h = np.zeros(n, dtype=int)
        ones_idx = random.sample(range(n), n//2)
        h[ones_idx] = 1
        # deterministic phase
        h, val = binary_worst_shift_swap(h, 2 * n)
        if val < best_val:
            best_h = h.copy()
            best_val = val
        # simulated annealing
        h, val = simulated_annealing_binary(h, 5 * n * math.log(n))
        if val < best_val:
            best_h = h.copy()
            best_val = val
    return best_h, best_val

def project_onto_box_and_sum(h, target_sum):
    # water filling
    v = h.copy()
    def phi(lam):
        return np.sum(np.clip(v - lam, 0, 1))
    
    low, high = np.min(v) - 1, np.max(v) + 1
    while abs(high - low) > 1e-10:
        mid = (low + high) / 2
        if phi(mid) > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    return np.clip(v - lam, 0, 1)

def soft_max_gradient_descent(h, alpha_schedule=[1, 5, 10, 20, 40, 80], steps_per_alpha=200, eta0=0.2):
    n = len(h)
    h = h.astype(float)
    h = project_onto_box_and_sum(h, n/2)
    best_val = evaluate_overlap(h)
    best_h = h.copy()

    for alpha in alpha_schedule:
        t = 0
        while t < steps_per_alpha:
            corr = np.correlate(h, 1 - h, mode='full')
            all_o = corr.copy()
            max_o = np.max(all_o)
            # subtract for numerical stability
            exps = np.exp(alpha * (all_o - max_o))
            p = exps / np.sum(exps)
            grad = np.zeros(n)
            for i in range(n):
                for k in range(-n+1, n):
                    if 0 <= i + k < n and 0 <= i < n:
                        if k >= 0:
                            grad[i] += p[k+n-1] * (1 - h[i + k])
                        else:
                            grad[i] += p[k+n-1] * (1 - h[i - k])
            grad = grad / n
            t += 1
            eta = eta0 / np.sqrt(t + 1)
            h -= eta * grad
            h = project_onto_box_and_sum(h, n/2)
            val = evaluate_overlap(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    return best_h, best_val

def run(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    start_t = time.time()

    for i, n in enumerate(N):
        if time.time() - start_t > budget_s - 10:
            break
        # estimate time to run
        if i == len(N) - 1:
            remaining = budget_s - (time.time() - start_t)
            T_n = remaining * 0.9  # leave margin
        else:
            # do not split budget too granularly; are you sure?
            T_n = 50.0  # suitably long for final refinement
        try:
            h, val = binary_search(n, T_n)
            c5 = 2 * val / n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            # also run fractional refinement on final size if possible
            # only do this if we're in the final computational frame of the program
            if i == len(N) - 1 and (time.time() - start_t) < budget_s * 0.9:
                h_frac, val_frac = soft_max_gradient_descent(h.copy())
                c5_frac = 2 * val_frac / n
                if c5_frac < best_c5:
                    best_c5 = c5_frac
                    best_h = h_frac.copy()
            if best_c5 <= TARGET_BOUND:
                break
        except Exception as e:
            print(f"Exception at n={n}: {e}")
            continue

    # if we haven't improved, return initial
    if best_h is None:
        best_h = INITIAL_H
        best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, len(best_h))
