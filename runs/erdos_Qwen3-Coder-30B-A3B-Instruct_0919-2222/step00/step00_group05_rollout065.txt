def evaluate_erdos_solution(h):
    """Evaluate the C_5 bound for a given h function"""
    n_points = len(h)

    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The max value of the overlap integral
    c5 = np.max(corr) * dx
    return c5
