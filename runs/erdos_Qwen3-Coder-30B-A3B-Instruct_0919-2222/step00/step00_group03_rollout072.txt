def evaluate_erdos_solution(h):

    correlation = np.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (overlapping parts)
    # This is the maximum over all shifts k (from -1 to 1 in terms of step sizes)
    max_overlap = np.max(correlation) * dx
    return max_overlap
