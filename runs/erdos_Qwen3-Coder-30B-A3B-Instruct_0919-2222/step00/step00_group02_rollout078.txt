```python
import numpy as np
from scipy.optimize import minimize
from scipy import optimize
import time
import math

def evaluate_erdos_solution(h, n_points):
    """Evaluate the C5 constant for given step function h"""
    dx = 2.0 / n_points
    # Compute correlations
    corr = np.correlate(h, 1-h, mode="full") * dx
    # The maximum overlap is the maximum value in correlation
    c5 = np.max(corr)
    return c5

def initial_h_values(n_points):
    """Return initial construction of h"""
    # Initial construction from the problem
    h = np.ones(n_points) * 0.5
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80  # Reasonable compromise between accuracy and speed
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # We need sum(h) * dx = 1, so sum(h) = n_points/2
    
    # Initial guess - start with a reasonable construction
    h0 = initial_h_values(n_points)
    h0 = h0.copy()
    
    # Use scipy.optimize with SLSQP for constrained optimization
    def objective(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_sum(h):
        # Ensure sum(h) = target_sum
        return np.sum(h) - target_sum
    
    # Bounds for each h[i] between 0 and 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization using L-BFGS-B to reduce gradient evaluation costs
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0, n_points)
    
    # For better exploration, try multiple random starts
    max_iterations = 500
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        try:
            # Randomize the initial point a bit
            h_start = h0 + np.random.normal(0, 0.05, n_points)
            h_start = np.clip(h_start, 0, 1)
            # Renormalize to maintain the sum constraint
            current_sum = np.sum(h_start)
            h_start = h_start * target_sum / current_sum
            
            # Optimize with constraints
            result = minimize(
                objective, 
                h_start, 
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if result.success:
                c5_current = evaluate_erdos_solution(result.x, n_points)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = result.x.copy()
                    
        except Exception as e:
            # If optimization fails, keep current best
            continue
            
        # Early stopping if we're close to target
        if best_c5 < 0.3795:
            break
    
    # Final evaluation with optimization to get more precise result
    if time.time() - start_time < budget_s - 1:
        try:
            # Perform one final fine-tune with a more complex method
            # Trying different regularization to avoid stagnation
            h_final = best_h.copy()
            h_final = np.clip(h_final, 0, 1)
            old_c5 = best_c5
            
            # Run a few more local optimizations
            for _ in range(20):
                if time.time() - start_time > budget_s - 1:
                    break
                # Small perturbations
                pert = np.random.normal(0, 0.01, n_points)
                h_test = h_final + pert
                h_test = np.clip(h_test, 0, 1)
                # Renormalize
                h_test = h_test * target_sum / np.sum(h_test)
                c5_test = evaluate_erdos_solution(h_test, n_points)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()
                    
        except:
            pass
    
    # Ensure final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
```
