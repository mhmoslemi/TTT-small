import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    # h is an array of values on [0,2] with sum(h)*dx = 1
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the correlation integral
    corr = correlate(h, 1 - h, mode="full")
    # The full convolution has 2*n_points - 1 elements
    c5 = np.max(corr) * dx
    return c5

def compute_c5_bound(h):
    """Compute the C5 bound for a given h."""
    n_points = len(h)
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def setup_initial_h(n_points):
    """Setup the initial h values, trying several methods."""
    # Use the initial_h_values if available, else try a few standard constructions
    try:
        # Try using the already imported initial_h_values
        h = np.array(initial_h_values)
        if len(h) < n_points:
            h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=(0, 0))
        elif len(h) > n_points:
            h = h[:n_points]
    except:
        # Fallback to a simple construction
        h = np.ones(n_points) * 0.5
    return h

def obj_func(h_vars, n_points):
    """Objective function to minimize."""
    # Convert the variables to h
    h = np.array(h_vars)
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    # Normalize for the constraint sum(h)*dx = 1
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points / 2.0) / sum_h
    # Compute C5
    c5 = compute_c5_bound(h)
    return c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 45  # Use less than 1k as preferred
    
    # Set up time tracking
    start_time = time.time()
    
    # Initial guess - either from initial_h_values or simple construction
    h0 = setup_initial_h(n_points)
    h0 = np.clip(h0, 0, 1)
    
    # Normalize h0 to sum to n_points/2 (so integral is 1)
    dx = 2.0 / n_points
    current_sum = np.sum(h0)
    if current_sum > 0:
        h0 = h0 * (n_points/2.0) / current_sum
    
    # Optimizing
    # Use differential evolution with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    result = differential_evolution(
        obj_func, 
        bounds, 
        args=(n_points,),
        seed=seed,
        maxiter=1000,
        popsize=15,
        atol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        disp=False
    )
    
    best_h = np.array(result.x)
    # Ensure bounds
    best_h = np.clip(best_h, 0, 1)
    # Normalize
    current_sum = np.sum(best_h)
    if current_sum > 0:
        best_h = best_h * (n_points/2.0) / current_sum
    best_c5 = compute_c5_bound(best_h)
    
    # Refine with local optimization if time allows
    time_left = budget_s - (time.time() - start_time)
    if time_left > 10:
        try:
            result_local = minimize(
                obj_func,
                best_h,
                args=(n_points,),
                method='L-BFGS-B',

            if current_sum > 0:

            refined_c5 = compute_c5_bound(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
