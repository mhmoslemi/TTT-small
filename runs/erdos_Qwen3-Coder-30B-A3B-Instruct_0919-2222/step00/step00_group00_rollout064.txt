def compute_correlation(h):
    """Computes correlation integral c_k = sum_i h[i] * (1 - h[i+k]) for all k."""
    n = len(h)
    h1 = np.array(h)
    h2 = np.ones_like(h1)
    h2[1:] = 1 - h1[:-1]  # one-shifted version of (1 - h)
    c = np.correlate(h1, h2, mode="full")
    return c

def log_sum_exp_stable(x):
    """Numerically stable computation of log(sum(exp(x)))"""
    x_max = np.max(x)
    return x_max + log(np.sum(np.exp(x - x_max)))

def smoothed_objective_and_weights(c, alpha):
    """Computes Φ_α = (1/α)*log Σ_k exp(α c_k) and weight vector w_k."""
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    exp_sum = np.sum(exp_c)
    w = exp_c / exp_sum
    phi_alpha = (1/alpha) * (log(exp_sum) + c_max)
    return phi_alpha, w

def grad_h(h, weights):
    """Gradient of the smoothed objective using the convolution formula."""
    n = len(h)
    w = weights
    c = np.correlate(h, 1 - np.array(h), mode="full")
    # Use the assumption that gradient derivative is a weighted convolution.
    # The true derivative formula from the strategy needs careful handling of three components:
    # - For the point h[i], we have contributions from all shifts k such that (i+k) is valid and (i-k) is valid.
    # For a more precise and efficient gradient vector, we'll compute directly from c derivative.

    # Explicit computation (O(n^2)) is used for accuracy, as n is small enough.
    grad = np.zeros_like(h)
    for i in range(n):
        contrib = 0.0
        for k in range(-n+1, n):
            if i + k >= 0 and i + k < n:
                contrib += w[k+n-1] * (1 - h[i + k])
            if i - k >= 0 and i - k < n:
                contrib -= w[k+n-1] * h[i - k]
        grad[i] = contrib
    return grad

def project_onto_feasible(y, target_sum):
    """Projects vector y onto the set {x : 0 ≤ x_i ≤ 1, sum x_i = target_sum}."""
    y = np.clip(y, 0, 1)
    current_sum = np.sum(y)
    diff = target_sum - current_sum
    if abs(diff) < 1e-12:
        return y
    # Sort by value (descending) to deal with free coordinates efficiently
    sorted_idx = np.argsort(y)[::-1]
    # Each new sum adjustment is applied to all indices from above up to where it becomes 1 or 0
    # touched_indices = []  # existence to prevent overflow
    total_free = 0
    for i in sorted_idx:
        if y[i] < 1e-12 or y[i] > 1 - 1e-12:
            continue
        total_free += 1
    if total_free == 0:
        return y
    avg_change = diff / total_free
    
    updated_y = y.copy()
    for i in sorted_idx:
        if updated_y[i] < 1e-12 or updated_y[i] > 1 - 1e-12:
            continue
        if abs(avg_change) < 1e-16:
            break
        new_val = updated_y[i] + avg_change
        if new_val < 0:
            fraction = (updated_y[i] / -avg_change)
            # Re-distribute among rest
            for j in sorted_idx:
                if j == i or updated_y[j] == 0:
                    continue
                # Adjust by fraction
                try:
                    if avg_change > 0:
                        change = min(avg_change * fraction, updated_y[j] - 0)
                    else:
                        change = max(avg_change * fraction, 1 - updated_y[j])
                    updated_y[j] -= change
                    updated_y[i] += change
                except:
                    continue
            break
        elif new_val > 1:
            fraction = ((1 - updated_y[i]) / avg_change)
            for j in sorted_idx:
                if j == i or updated_y[j] == 1:
                    continue
                try:
                    change = min(avg_change * fraction, 1 - updated_y[j])
                    updated_y[j] += change
                    updated_y[i] -= change
                except:
                    continue
            break
        else:
            updated_y[i] = new_val
    return updated_y

def swap_update(c, h, i, j):
    """Update correlation c when swapping positions i and j in a binary vector h."""
    off = len(c) // 2
    n = len(h)
    delta = h[i] - h[j]
    for q in range(0, n):
        shift = q - i
        if i + shift >= 0 and i + shift < n:
            c[shift + off] += delta * (1 - h[q]) if h[i + shift] != h[q] else 0
    for q in range(0, n):
        shift = q - j
        if j + shift >= 0 and j + shift < n:
            c[shift + off] -= delta * (1 - h[q]) if h[j + shift] != h[q] else 0
    h[i], h[j] = h[j], h[i]
    return c, h

def targeted_swap(c, h, s_max, ones, zeros, off):
    """Performs targeted swap to reduce c[s_max] by 1."""
    delta = -1 if h[s_max] == 1 else 1
    for i in range(len(zeros)):
        if h[zeros[i]] == 0 and (zeros[i] + s_max >= 0 and zeros[i] + s_max < len(h)):
            if h[zeros[i] + s_max] == 1:
                for j in range(len(ones)):
                    if h[ones[j]] == 1 and (ones[j] + s_max >= 0 and ones[j] + s_max < len(h)):
                        if h[ones[j] + s_max] == 0:
                            idx_i = zeros[i]
                            idx_j = ones[j]
                            c, h = swap_update(c, h, idx_i, idx_j)
                            return True, c, h
    return False, c, h

def local_binary_search(h, c, time_budget_s, off=0):
    """Performs a local search on the binary version to reduce max correlation."""
    n = len(h)
    ones = [i for i in range(n) if h[i] == 1]
    zeros = [i for i in range(n) if h[i] == 0]
    start_time = time.time()
    best_h = list(h)
    best_c = np.array(c)
    s_max = np.argmax(best_c)
    swaps_executed = 0
    max_swaps = 10000

    while time.time() - start_time < time_budget_s * 0.9 and swaps_executed < max_swaps:
        # Call targeted swap first if possible
        additional, temp_c, temp_h = targeted_swap(best_c, best_h, s_max, ones, zeros, off)
        if additional:
            swaps_executed += 1
            # Update one and zero set
            ones = [i for i in range(n) if best_h[i] == 1]
            zeros = [i for i in range(n) if best_h[i] == 0]
            s_max = np.argmax(temp_c)
            best_c = temp_c
            continue
        # Otherwise perform a random valid swap
        if not ones or not zeros:
            break
        try:
            i_rand = random.choice(ones)
            j_rand = random.choice(zeros)
            c, h = swap_update(best_c, best_h, i_rand, j_rand)
            ones = [i for i in range(n) if h[i] == 1]
            zeros = [i for i in range(n) if h[i] == 0]
            s_max = np.argmax(c)
            swaps_executed += 1
        except Exception:
            break
    return np.array(best_c), list(best_h)

def upscale_h(h_old, n_new):
    """Upscales h_old to a new grid of size n_new by repeat."""
    n_old = len(h_old)
    h_new = []
    for i in range(n_new):
        idx = i * n_old // n_new
        h_new.append(h_old[idx])
    return h_new

def run(seed=42, budget_s=600, **kwargs):

            h *= ratio
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            # Adjust the sum more carefully
            diff = target_sum - current_sum
            if abs(diff) > 1e-10 and n > 4:
                # Adjust random component(s)
                i = random.randint(0, n-1)
                h[i] += diff
                h[i] = max(0.0, min(1.0, h[i]))
        # Smoothed gradient descent with momentum
        alpha = 5.0
        eta = 0.5 * dx
        beta = 0.9
        c_prev = None
        h_vel = np.zeros_like(h)
        h_best = None
        max_c5_so_far = float('inf')

        max_iter = 30000
        iter_count = 0
        while iter_count < max_iter and time.time() - t_start < budget_s - 2:
            # c = np.correlate(h, 1 - np.array(h), mode="full")
            c = compute_correlation(h)
            phi_alpha, w = smoothed_objective_and_weights(c, alpha)
            g = grad_h(h, w)
            h_vel = beta * h_vel + (1 - beta) * g
            h_new = h - eta * h_vel
            h_new = project_onto_feasible(h_new, target_sum)
            # Re-evaluate c
            c_new = compute_correlation(h_new)
            phi_new, _ = smoothed_objective_and_weights(c_new, alpha)
            
            if phi_new <= phi_alpha:
                h = h_new
                if np.max(c_new) * dx < max_c5_so_far:
                    max_c5_so_far = np.max(c_new) * dx
                    h_best = h.copy()
                if alpha < 200 and iter_count % 100 == 0:
                    alpha *= 1.1
            else:
                eta *= 0.5
            iter_count += 1
        if h_best is not None:
            h_rounded = np.array(h_best)
            h_bin = np.zeros_like(h_best)
            sorted_indices = np.argsort(h_rounded)[::-1]
            for i in range(int(n / 2)):
                h_bin[sorted_indices[i]] = 1.0
            # Re-validate the integral
            sum_bin = np.sum(h_bin)
            if abs(sum_bin - n / 2) > 1e-6:
                # Try to adjust
                sorted_indices = np.argsort(h_bin)[::-1]
                delta = int(n / 2) - int(sum_bin)
                if delta > 0:
                    for i in reversed(range(len(h_bin))):
                        if h_bin[i] == 0:
                            h_bin[i] = 1
                            delta -= 1
                            if delta == 0:
                                break
                else:
                    for i in reversed(range(len(h_bin))):
                        if h_bin[i] == 1:
                            h_bin[i] = 0
                            delta += 1
                            if delta == 0:
                                break
            # Binary polishing
            c_final = np.correlate(h_bin, np.array([1] * len(h_bin)[::-1]), mode="full")
            off = len(c_final) // 2
            s_final = np.argmax(c_final)
            current_time_left = budget_s - (time.time() - t_start)
            if current_time_left > 1:
                try:
                    c_final, h_final_polished = local_binary_search(h_bin, c_final,

                    c5_final = np.max(c_final) * dx
                    if c5_final < best_c5:

                        best_h = list(h_final_polished)

                    pass
            else:

                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = list(h_bin)
                    best_n = n

        # Now upscale and retry with larger grid if allowed
        if n < ns[-1] and time.time() - t_start < budget_s - 2:
            next_n = ns[ns.index(n) + 1]
            if next_n <= 500:

                    h = np.array(h_upscaled)
                    h = np.clip(h, 0, 1)
                    # Re-normalize
                    current_sum = np.sum(h)
                    if abs(current_sum - next_n/2) > 1e-10:
                        ratio = (next_n/2) / current_sum
                        h *= ratio
                        h = np.clip(h, 0, 1)
                except:
                    pass
    # Final switch back to best solution in case needed
    if best_h is None:
        return initial_h_values[:80], 0.38092, 80
