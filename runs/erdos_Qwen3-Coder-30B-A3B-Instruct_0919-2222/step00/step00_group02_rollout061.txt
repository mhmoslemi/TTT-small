import numpy as np
import time
import random

def evaluate_h(h):
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode='full')
    max_val = O_counts.max()
    return max_val * dx

def max_overlap_counts(h):
    O_counts = np.correlate(h, 1 - h, mode='full')
    return O_counts.max()

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=int)
    ones = rng.sample(range(n), n // 2)
    h[ones] = 1
    return h

def project_onto_box_and_sum(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def binary_anneal(n, time_limit, rng):
    h = random_balanced_binary(n, rng)
    cur_val = max_overlap_counts(h)
    best_val = cur_val
    best_h = h.copy()

    start_time = time.time()
    max_iters = 100000
    for iter in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break

        T = 0.5 * (1 - elapsed / time_limit) + 0.001
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = rng.choice(ones)
        q = rng.choice(zeros)
        h[p], h[q] = 0, 1
        new_val = max_overlap_counts(h)
        delta = new_val - cur_val
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            h[p], h[q] = 1, 0
    return best_h, best_val
