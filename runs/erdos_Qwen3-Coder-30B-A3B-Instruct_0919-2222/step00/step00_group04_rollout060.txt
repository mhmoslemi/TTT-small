import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for the given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using np.correlate with mode='full'
    # This computes sum_i h[i] * (1 - h[i+k]) for all k
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take maximum of all shifts (correlation after zero-padding gives
    # correlation for shifts -n+1 to n-1)
    max_correlation = np.max(correlation)
    # Multiply by dx to get actual integral bound
    c5_bound = max_correlation * dx
    return c5_bound

def project_to_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum and h_i in [0,1]"""
    n = len(h)
    # Sort h in descending order
    sorted_h = np.sort(h)[::-1]
    # Find the threshold tau such that 
    # sum_i max(0, min(1, h_i - tau)) = target_sum
    # Use simple bisection search (binary search)
    low, high = 0.0, 1.0
    tol = 1e-10
    while high - low > tol:
        tau = (low + high) / 2
        projected = np.maximum(0, np.minimum(1, sorted_h - tau))
        if np.sum(projected) > target_sum:
            low = tau
        else:
            high = tau
    tau = (low + high) / 2
    # Apply tau to sorted values and reconstruct h
    new_h = np.maximum(0, np.minimum(1, h - tau))
    # Adjust to match exact target_sum
    current_sum = np.sum(new_h)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Distribute adjustment: add proportionally to values (in ascending order)
        sorted_indices = np.argsort(new_h)
        inc = diff / n
        for i in sorted_indices:
            new_h[i] = max(0, min(1, new_h[i] + inc))
            # If we exceed sum target, then do rest in careful way
            new_sum = np.sum(new_h)
            if abs(new_sum - target_sum) < 1e-10:
                break
    return new_h

def compute_gradient(h, max_indices):
    """Compute subgradient of max correlation with respect to h"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in max_indices:
        # For shift k:
        # Part 1: indices where h[i] contributes (i + k < n)
        if k >= 0:
            # Add derivative: 1 - h[i+k] for i in range(n-k)
            g[:n-k] += 1 - h[k:]
        else:
            # k < 0, so indices where h[i] = h[i-k] contributes
            # Add derivative: -h[i-k] for i in range(-k, n) 
            g[-k:] -= h[:-k]
    return g
