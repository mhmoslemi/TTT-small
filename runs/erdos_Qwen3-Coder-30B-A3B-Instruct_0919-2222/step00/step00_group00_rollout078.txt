import time
import numpy as np
import cvxpy as cp
from itertools import product

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    # Discretization sizes to try
    n_list = [80, 120, 160]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Helper function to evaluate full overlap
    def evaluate_full_overlap(h):
        h = np.array(h)
        # Compute full correlation (mode='full')
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr)
    
    # Helper function to solve MILP
    def solve_milp(n, K0=None, time_limit=None):
        # Fix n to be even
        if n % 2 != 0:
            n += 1
            
        dx = 2.0 / n
        half_n = n // 2
        
        # If no K0 given, set to reasonable default
        if K0 is None:
            K0 = min(n // 4, 30)
        
        # Binary variables x[i]
        x = cp.Variable(n, boolean=True)
        
        # Continuous variable t to bound maximum overlap
        t = cp.Variable()
        
        # Constraints for x to ensure sum is exactly half
        constraints = [cp.sum(x) == half_n]
        
        # Symmetry breaking (x[0] = 1, prefix monotonicity)
        constraints.append(x[0] == 1)
        for i in range(n // 2 - 1):
            constraints.append(x[i] >= x[i + 1])
        
        # Initialize active shifts set
        K = list(range(1, K0 + 1))
        
        # Enforce those shifts initially
        # y[i,k] = x[i] * x[i+k] 
        y = {}
        
        # Add the minimum required constraints
        full_constraints = constraints[:]
        
        for k in K:
            # Create y variables for current shift k
            y_list = [cp.Variable(boolean=True) for _ in range(n - k)]
            y[k] = y_list
            
            for i in range(n - k):
                # McCormick envelopes for y[i,k] = x[i] * x[i+k]
                full_constraints.append(y_list[i] <= x[i])
                full_constraints.append(y_list[i] <= x[i + k])
                full_constraints.append(y_list[i] >= x[i] + x[i + k] - 1)
                full_constraints.append(y_list[i] >= 0)
            
            # The overlap constraint for shift k
            # c_k = sum x[i] - sum y[i,k] <= t
            # Since sum x[i] = half_n, c_k = half_n - sum y[i,k]
            full_constraints.append(half_n - cp.sum(y_list) <= t)
        
        # Objective: minimize t  
        prob = cp.Problem(cp.Minimize(t), full_constraints)
        
        try:
            # Add solver parameter if needed
            if time_limit is not None:
                prob.solve(solver=cp.CBC, time_limit=time_limit)
            else:
                prob.solve(solver=cp.CBC)
        except Exception as e:
            return None, None, None
            
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            return None, None, None
            
        x_opt = x.value
        t_opt = t.value
        
        # Now check the actual max_overlap over ALL k
        x_opt_bool = np.array(x_opt, dtype=int)
        x_opt_float = x_opt_bool.astype(float)
        
        max_check = evaluate_full_overlap(x_opt_float)
        if max_check <= t_opt + 1e-6:
            return x_opt_float, t_opt * dx, n
        
        # If not full valid, try lazy constraint loop
        # Continue adding violated shifts
        max_k = n - 1
        max_iter = 10
        iter_count = 0
        while iter_count < max_iter:
            # Recheck validity
            x_opt_bool = np.array(x_opt, dtype=int)
            x_opt_float = x_opt_bool.astype(float)
            
            max_overlap = evaluate_full_overlap(x_opt_float)
            
            if max_overlap <= t_opt + 1e-6:
                return x_opt_float, t_opt * dx, n
                
            violating_k = -1
            max_cutoff = 0
            for k in range(1, max_k):
                # Note: full correlation includes all shifts, so k should be within range
                corr = np.correlate(x_opt_float, 1 - x_opt_float, mode="full")
                indices = np.arange(len(corr))
                offset = len(corr) // 2   # Because mode='full'
                actual_k = indices[offset + k] if offset + k < len(corr) else 0
                actual_k_neg = indices[offset - k] if offset - k >= 0 else 0
                
                # For computational simplicity, get correlation directly at shift k
                # Only compute correlations seen in numba kernel
                corr_at_k = corr[n - 1 + k] if (n - 1 + k) < len(corr) else corr[n - 1 - k]
                
                if corr_at_k > t_opt + 1e-6:
                    if corr_at_k > max_cutoff:
                        max_cutoff = corr_at_k
                        violating_k = k
                        
            if violating_k == -1:
                return x_opt_float, t_opt * dx, n
                
            # Add to constraint set
            if not violating_k in K:
                K.append(violating_k)
                # Add new y variables and constraints
                y_list = [cp.Variable(boolean=True) for _ in range(n - violating_k)]
                y[violating_k] = y_list
                
                for i in range(n - violating_k):
                    full_constraints.append(y_list[i] <= x[i])
                    full_constraints.append(y_list[i] <= x[i + violating_k])
                    full_constraints.append(y_list[i] >= x[i] + x[i + violating_k] - 1)
                    full_constraints.append(y_list[i] >= 0)
                
                full_constraints.append(half_n - cp.sum(y[violating_k]) <= t)
                
                # Re-solve
                try:
                    prob = cp.Problem(cp.Minimize(t), full_constraints)
                    prob.solve(solver=cp.CBC, time_limit=time_limit)
                except:
                    return x_opt_float, t_opt * dx, n
                    
                if prob.status not in ["optimal", "optimal_inaccurate"]:
                    return x_opt_float, t_opt * dx, n
                    
                x_opt = x.value
                t_opt = t.value
                iter_count += 1
            else:
                break
                
        return x_opt_float, t_opt * dx, n
    
    # Helper for post-processing to polish optimal solution  
    def polish_solution(h_vals, c5_bound):
        # Since we are just going to make sure there's no improvement possible via swaps
        # make small swaps back and forth to try improving
        h_vals = np.array(h_vals)
        best_h = h_vals.copy()
        best_val = evaluate_full_overlap(best_h)
        
        # Try random swaps that decrease the max_overlap
        num_swaps = 0
        max_swaps = 3000
        while num_swaps < max_swaps:
            if time.time() - t_start > budget_s - 1.5: # save 1.5 seconds safety margin
                break
                
            idx_on = np.where(h_vals == 1)[0]
            idx_off = np.where(h_vals == 0)[0]
            
            if len(idx_on) == 0 or len(idx_off) == 0:
                break
                
            i = np.random.choice(idx_on)
            j = np.random.choice(idx_off)
            
            if np.random.rand() > 0.5:
                k = i
                l = j
            else:
                l = i
                k = j
                
            # Temporarily swap
            h_tmp = h_vals.copy()
            h_tmp[k], h_tmp[l] = h_tmp[l], h_tmp[k]
            
            # Check if new maximum overlaps are less than or equal to current optimal 
            max_new = evaluate_full_overlap(h_tmp)
            max_old = evaluate_full_overlap(h_vals)
            
            new_val = max_new
            
            if new_val <= best_val:  # got a better arrangement
                best_h = h_tmp
                best_val = new_val
                h_vals = h_tmp.copy()
                num_swaps = 0  # restart counter after a valid change
            else:
                num_swaps += 1
                
        return list(best_h), best_val
    
    # Start with small n
    for n in n_list:
        if time.time() - t_start > budget_s - 1:
            break
            
        time_left = budget_s - (time.time() - t_start)
        if time_left < 2:
            break
            
        if n < 60:
            solver_time = min(30, time_left / 3)
        else:
            solver_time = min(60, time_left / 3)
            
        h_vals, c5_bound, n_used = solve_milp(n, K0=n//5, time_limit=solver_time)
        
        if c5_bound is None:
            continue
            
        # Try to polish it
        try:
            h_polished, c5_polished = polish_solution(h_vals, c5_bound)
        except:
            continue
            
        if c5_polished < best_c5:
            best_c5 = c5_polished
            best_h = h_polished
            best_n = n_used
            
    if best_h is None:
        # fallback to initial construction
        from initial_h_values import initial_h_values
        h_init = np.array(initial_h_values)
        dx = 2.0 / len(h_init)
        c5_init = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
        
        return list(h_init), c5_init, len(h_init)
        
    return best_h, best_c5, best_n
