def evaluate_h(h):
    # h is a numpy array of length n_points
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute full convolution of h with (1-h)
    conv = correlate(h, 1 - h, mode='full')
    # The valid range is from -1 to 1
    # We need all k in [0,2] such that h(x) overlaps with 1-h(x+k)
    # For k >= 0, the overlap is from 0 to 2-k, so we consider indices from n_points-1 to 2*n_points-2
    # For k <= 0, the overlap is from -k to 2 which we map appropriately.
    # Since the condition is integral x in [0,2], x+k in [0,2], 
    # So k is in [-x, 2-x] where x is in [0,2], thus k in [-2, 2]
    # We compute for all valid partial integrals
    # But a better approach: we compute all correlations from -dx to 2*dx
    # The full convolution gives us correlation from -n_points+1 to n_points-1
    # We map these into k values from -2 to 2
    # But in practice, we take max of correlations at positive k values from 0 to 2
    # Since k = m * dx = m * (2/n_points) where m ranges from -n_points+1 to n_points-1
    
    # The key insight is to translate indices in the full correlation to actual k values
    # Full correlation has length 2*n_points - 1
    # Index 0 to n_points-1 correspond to k values from -2 to 0 
    # Index n_points to 2*n_points-2 correspond to k values from 0 to 2
    # Only consider k >= 0 because k < 0 can be mirrored
    # But by symmetry, max is achieved for k in [0,2]
    # So we take indices from n_points-1 to 2*n_points - 2
    max_corr = np.max(conv[n_points-1:])
    return max_corr * dx
