import time
import numpy as np
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, ECOS_BB, CBC, Constant

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    # Evaluation functions
    def evaluate_h(h_vals):
        # h_vals is a list of 0/1 values; returns max correlation
        n = len(h_vals)
        h = np.array(h_vals)
        # Create correlation vector (full mode)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr)
    
    def evaluate_h_scaled(h_vals, dx):
        # scale max correlation by dx
        return evaluate_h(h_vals) * dx

    n_values = [80, 120, 160]  # Or try increasing to 200
    best_c5_bound = float('inf')
    best_h_vals = None
    best_n = 0

    solvers = [CBC, ECOS_BB]
    
    # For each discretization size
    for n in n_values:
        if time.time() - t_start > budget_s - 5:
            break
            
        dx = 2.0 / n
        K0 = max(1, min(n//4, 50))  # initial shift set size
        
        # Lazy constraint MILP loop
        # Start solving with small set of shifts
        K = list(range(1, K0+1))
        while K:
            # Check time budget
            if time.time() - t_start > budget_s - 5:
                break
                
            # Define variables
            x = Variable(n, boolean=True)
            t = Variable()
            
            # Constraints
            constrs = []
            
            # Mass constraint
            constrs.append(cp.sum(x) == n/2)
            
            # Symmetry breaking
            constrs.append(x[0] == 1)
            for i in range(n//2 - 1):
                constrs.append(x[i] >= x[i+1])
            
            # Bilinear variables (lazy) and constraints
            y = {}
            c_k_exprs = []  # For adding shift constraints
            
            for k in K:
                if k >= n:
                    continue
                # Variable for y[i,k] for i = 0 to n-k-1
                y_k = []
                for i in range(n - k):
                    y[i,k] = Variable(boolean=True)
                    y_k.append(y[i,k])
                # McCormick constraints: y[i,k] <= x[i], x[i+k], y[i,k] >= x[i] + x[i+k] - 1
                for i in range(n-k):
                    constrs.append(y[i,k] <= x[i])
                    constrs.append(y[i,k] <= x[i+k])
                    constrs.append(y[i,k] >= x[i] + x[i+k] - 1)
                
                # c_k = sum(x[i]) - sum(y[i,k]) for i=0..n-k-1
                expr = cp.sum(x[:n-k]) - cp.sum(y_k)
                c_k_exprs.append(expr)
            
            # Pointer to k indices used
            k_indices_used = []
            # add the new constraints
            for i, k in enumerate(K):
                if k >= n:
                    continue
                # circular shift will not be needed
                constrs.append(c_k_exprs[i] <= t)
                k_indices_used.append(k)
            
            # Objective
            prob = Problem(Minimize(t), constrs)
            
            # Try different solvers
            solver_chosen = None
            for solver in solvers:
                try:
                    prob.solve(solver=solver, verbose=False)
                    if prob.status in ["optimal", "optimal_inaccurate"]:
                        solver_chosen = solver
                        break
                except:
                    continue
                    
            if solver_chosen is None:
                # fallback to heuristics or skip
                break
                
            # Early termination if infeasible
            if prob.status not in ["optimal", "optimal_inaccurate"]:
                continue
                
            t_opt = t.value
            x_sol = np.round(x.value).astype(int).tolist()
            
            # Rebuild correlation with all shifts to test feasibility
            # This is the full test
            c_full = evaluate_h(x_sol)
            
            # If all shifts bound by t_opt, we have a valid solution
            if c_full <= t_opt + 1e-6:
                h_vals = x_sol
                c5_bound = t_opt * dx
                if c5_bound < best_c5_bound:
                    best_c5_bound = c5_bound
                    best_h_vals = h_vals.copy()
                    best_n = n
                break  # Found feasible solution for this n
                    
            # Identify violated shifts and add to K
            # Compute full correlation vector
            h_u = np.array(x_sol)
            corr_full = np.correlate(h_u, 1 - h_u, mode="full")
            max_corr = np.max(corr_full)
            corr_indices = np.argsort(corr_full)[::-1]
            
            # Add more violations
            added = 0
            for k in corr_indices:
                if k == 0:
                    continue  # self-correlation
                # Add the worst violation
                if k not in K:
                    K.append(k)
                    added += 1
                    if added >= 5 or len(K) > 100:
                        break  # prevent infinite growth
                        
        # End of lazy-constraint for current n
        # Try polishing more if possible
        if best_h_vals is None:
            continue
            
        h_polish = np.array(best_h_vals)
        current_max = evaluate_h(h_polish)
        
        # Small deterministic polishing
        if time.time() - t_start < budget_s - 10:
            done = False
            while not done and time.time() - t_start < budget_s - 5:
                # Compute full correlation
                corr_full = np.correlate(h_polish, 1 - h_polish, mode="full")
                max_k = np.argmax(corr_full)
                max_val = corr_full[max_k]
                if max_val <= current_max:
                    done = True
                    break
                    
                # Find a pair (i, j) to swap
                ones = np.where(h_polish == 1)[0]
                zeros = np.where(h_polish == 0)[0]
                
                # For shift k, which ones are flipped from 1 to 0?
                good_i = [i for i in ones if i + max_k < len(h_polish) and h_polish[i + max_k] == 0]
                good_j = [j for j in zeros if j + max_k < len(h_polish) and h_polish[j + max_k] == 1]
                
                if len(good_i) == 0 or len(good_j) == 0:
                    break  # no swap available
                
                # Perform contiguous swap
                i = good_i[0]
                j = good_j[0]
                
                h_polish[i] = 0
                h_polish[j] = 1
                
                # Update our full correlation (can be slow but we do it rarely)
                corr_full = np.correlate(h_polish, 1 - h_polish, mode="full")
                new_max = np.max(corr_full)
                if new_max < max_val:
                    # Accept the swap
                    current_max = new_max
                else:
                    # Revert
                    h_polish[i] = 1
                    h_polish[j] = 0
                    
            # Final update of best
            final_c = evaluate_h(h_polish)
            final_c5 = final_c * dx
            if final_c5 < best_c5_bound:
                best_c5_bound = final_c5
                best_h_vals = h_polish.tolist()
