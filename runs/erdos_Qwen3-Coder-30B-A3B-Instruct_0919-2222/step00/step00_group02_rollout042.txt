def compute_overlap(h):
    """Compute all overlap counts for each shift k."""
    n = len(h)
    dx = 2.0 / n
    # Use full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = corr.max()*dx
    return corr, max_val

def random_balanced_vector(n):
    """Generate a balanced binary vector with exactly n//2 ones."""
    h = np.zeros(n, dtype=int)
    ones = np.random.choice(n, n//2, replace=False)
    h[ones] = 1
    return h

def get_mismatched_sets(a, shift):
    """Get sets of mismatched positions for a given shift."""
    n = len(a)
    M1, M0 = [], []
    if shift >= 0:
        for i in range(n):
            if i + shift < n:
                if a[i] == 1 and a[i + shift] == 0:
                    M1.append(i)
                elif a[i] == 0 and a[i + shift] == 1:
                    M0.append(i)
    else:
        for i in range(n):
            if i - shift >= 0:
                if a[i] == 1 and a[i - shift] == 0:
                    M1.append(i)
                elif a[i] == 0 and a[i - shift] == 1:
                    M0.append(i)
    return M1, M0

def optimize_shifted_binary(a, max_iter=2000):
    """Perform local search focusing on the current worst overlap."""
    n = len(a)
    corr = np.correlate(a, 1 - a, mode="full")
    best_val = corr.max()
    best_a = a.copy()
    no_improvement_count = 0
    counter = 0
    while counter < max_iter and no_improvement_count < n:
        corr = np.correlate(best_a, 1 - best_a, mode="full")
        max_val = corr.max()
        s = corr.argmax() - (n - 1)
        M1, M0 = get_mismatched_sets(best_a, s)
        if len(M1) > 0 and len(M0) > 0:
            i, j = random.choice(M1), random.choice(M0)
            # Perform swap
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
            new_corr = np.correlate(new_a, 1 - new_a, mode="full")
            new_max = new_corr.max()
            if new_max <= max_val:
                best_a = new_a
                best_val = new_max
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            # Random balanced swap fallback
            pos1 = np.where(best_a == 1)[0]
            pos0 = np.where(best_a == 0)[0]
            if len(pos1) > 0 and len(pos0) > 0:
                i, j = random.choice(pos1), random.choice(pos0)
                new_a = best_a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]
                new_corr = np.correlate(new_a, 1 - new_a, mode="full")
                new_max = new_corr.max()
                if new_max <= max_val:
                    best_a = new_a
                    best_val = new_max
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                break
        counter += 1
    return best_a, best_val

def simulated_annealing_binary(best_a, max_iter=5000):
    """Run simulated annealing to escape plateaus."""
    n = len(best_a)
    corr = np.correlate(best_a, 1 - best_a, mode="full")
    best_val = corr.max()
    current_a = best_a.copy()
    current_val = best_val
    T0 = 0.5
    Tf = 1e-4
    alpha = (Tf / T0) ** (1.0 / max_iter)
    T = T0
    counter = 0
    while counter < max_iter:
        # Propose a random balanced swap
        pos1 = np.where(current_a == 1)[0]
        pos0 = np.where(current_a == 0)[0]
        if len(pos1) > 0 and len(pos0) > 0:
            i, j = random.choice(pos1), random.choice(pos0)
            new_a = current_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
            new_corr = np.correlate(new_a, 1 - new_a, mode="full")
            new_max = new_corr.max()
            delta = new_max - current_val
            if delta <= 0 or random.random() < np.exp(-delta / T):
                current_a = new_a
                current_val = new_max
                if current_val < best_val:
                    best_a = current_a.copy()
                    best_val = current_val
        T *= alpha
        counter += 1
    return best_a, best_val

def multi_start_optimize(n, max_time_s):
    """Multi-start optimization of binary sequences."""
    start_time = time.time()
    best_h = None
    best_val = float('inf')
    # Only do several restarts if we have enough time to spare
    max_restarts = min(int(max_time_s / 10), 5)
    for _ in range(max_restarts):
        if time.time() - start_time > max_time_s - 1:
            break
        h = random_balanced_vector(n)
        h, val = optimize_shifted_binary(h, max_iter=2000)
        h, val = simulated_annealing_binary(h, max_iter=5000)
        if val < best_val:
            best_h = h
            best_val = val
    return best_h, best_val

def project_to_box_and_sum(h):
    # Project to [0, 1] and sum = n/2
    # Water-filling step
    v = h.copy()
    sum_v = v.sum()
    # Manual binsearch for lambda
    l_min = min(v) - 1
    l_max = max(v) + 1
    tol = 1e-12
    max_iter = 40
    for _ in range(max_iter):
        l_mid = (l_min + l_max) / 2.0
        f = (v - l_mid).clip(0, 1).sum() - len(h) / 2
        if f > 0:
            l_min = l_mid
        else:
            l_max = l_mid
    l_final = (l_min + l_max) / 2.0
    h_proj = (v - l_final).clip(0, 1)
    return h_proj

def compute_f_alpha(h, alpha):
    """Compute surrogate function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    # Add large offset to numerical stability for very negative val
    corr = corr + 1e6  # this might not be directly safe; see if removing if works
    # Add offset to avoid numerical errors
    shift = np.arange(-(n - 1), n)
    o = corr
    max_o = o.max()
    exp_o = np.exp(alpha * (o - max_o))
    sum_exp = exp_o.sum()
    if sum_exp == 0:
        f = 0  # fallback
    else:
        # Softmax
        f = (1.0 / alpha) * np.log(sum_exp) + max_o
    return f, corr

def gradient_of_o_k(h, k):
    """Compute gradient of overlap_k."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    # Sparse gradient
    if k >= 0:
        # pad to make sure we stay in bounds
        if k < n:
            # Positions that are used in overlap k:
            # h_i * (1 - h_{i+k})
            g[:n-k] += 1.0
            g[k:] -= h[:n-k]  # this way, we get (1-h_{i+k})
    else:
        if -k < n:
            g[:n+k] += 1.0
            g[-k:] -= h[k-n:]  # similar direction
    return g * dx

def gradient_soft_max(h, corr):
    """Compute gradient of soft max surrogate function."""
    n = len(h)
    p = np.exp(corr)
    p = p / p.sum()
    grad = np.zeros(n)
    for k in range(-n + 1, n):
        pk = p[k + n - 1]
        grad += pk * gradient_of_o_k(h, k)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    global_best_h = None
    global_best_c = float('inf')
    global_best_n = 0

        if time.time() - start_time > budget_s - 5:
            break
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1:
            break
        t_alloc = min(remaining_time * n / sum(N), remaining_time * 0.8)
        best_h_n, val_n = multi_start_optimize(n, t_alloc)

            continue
        c_n = 2 * val_n / n
        if c_n < global_best_c:
            global_best_c = c_n
            global_best_h = best_h_n.copy()
            global_best_n = n
        if global_best_c < TARGET_BOUND:
            break
        
    if global_best_h is None:
        # Either failed or time was too short.
        # Again, simulate a valid initial solution
        try:
            h0 = initial_h_values()
            h0 = h0 / h0.sum() * (len(h0)/2)
            best_corr = np.correlate(h0, 1 - h0, mode="full")

            global_best_h = h0
    best_h = global_best_h
    best_c = global_best_c
    return best_h, best_c, len(best_h)
