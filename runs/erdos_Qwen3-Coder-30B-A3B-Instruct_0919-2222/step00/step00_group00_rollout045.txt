import numpy as np

def compute_correlation(h):
    """Compute correlation c_k = sum_i h[i] * (1 - h[i+k]) for all valid shifts."""
    n = len(h)
    # Using numpy correlate mode="full"
    # Returns 2*n-1 values (shifts from -(n-1) to n-1)
    corr = np.correlate(h, 1-h, mode="full")
    return corr

def smoothed_objective_and_weights(c, alpha):
    """Compute Φ_α = (1/α)*log(sum(exp(α*c))) and weights w_k = exp(α*c) / sum(exp(α*c))."""
    c_max = np.max(c)
    # Stable log-sum-exp
    exp_c = np.exp(alpha * (c - c_max))
    Z = np.sum(exp_c)
    if Z == 0:  # This shouldn't happen with finite values
        weights = np.ones_like(exp_c) / len(exp_c)
    else:
        weights = exp_c / Z
    # Phi_alpha = (1/alpha)*log(Z) + c_max
    # But to avoid issue if using mu = c_max, use regular computation.
    phi_alpha = np.log(Z) / alpha + c_max
    return phi_alpha, weights

def grad_h(h, weights):

    # Influence of each shift contributes to gradient
    # We compute one by one form using loop if slow, otherwise using conv.
    ones = np.ones(n)
    ones_rev = ones[::-1]
    # Compute (1 - h) convolution with first set of weights
    # Then compute h convolution with shifted weights.
    # However, more efficient direct means:
    # Let forward[i] = sum_{j >= i} (1 - h[j]) = prefix1[n-1] - prefix1[i-1]
    # Let backward[i] = sum_{j <= i} h[j] = prefix2[i]
    # Then (∇c_k)_i = forward[i+k] - backward[i-k] (i-k valid)
    
    forward = np.cumsum(1 - h[::-1])[::-1]   # directly reverse cumsum to make i-> sum_{j=i}^{n-1} (1-h[j])
    backward = np.cumsum(h)
    
    # But using more direct mechanism based on brute-force inspired from discrete derivation
    # Use two separate computations of full gradients, then sum up weighted ones:
    grad = np.zeros_like(h)
    w_mid = len(weights) // 2  # location of shift 0 in weights
    n_valid = len(h)
    for i in range(len(h)):
        grad_i = 0
        for k in range(-n_valid + 1, n_valid):
            if k == 0:
                continue
            shifted_idx = i + k
            if shifted_idx < 0 or shifted_idx >= n_valid:
                continue
            w_k = weights[w_mid + k]
            if k >= 0:
                grad_i += w_k * (1 - h[shifted_idx])
            else:
                grad_i -= w_k * h[shifted_idx]
        grad[i] = grad_i
    return grad

def project_onto_feasible(y, target_sum):
    """Project y onto feasible set { x : 0 <= x <= 1 and sum(x) = target_sum }. 
    Uses water-filling algorithm."""
    n = len(y)
    y = np.clip(y, 0, 1)
    # Compute deficiency
    current_sum = np.sum(y)
    diff = current_sum - target_sum
    if abs(diff) < 1e-12:
        return y.copy()

    # Adjust x by modifying intermediate values to make sum equal to target
    # Sort indices by value (descending)
    idx_sorted = np.argsort(y)[::-1]
    y_copy = y.copy()
    
    # Simulate water-filling:
    # The surplus or deficit in total sum needs to be distributed
    # While the difference is large enough:
    # increase/decrease those values that have 0 or 1 in the difference
    # cheaper than full projection
    
    # Find how many are not fully used (not touching 1 or 0) and also how many are exactly at bounds
    free = np.where((y_copy > 0) & (y_copy < 1))[0]
    free_count = len(free)
    while abs(diff) > 1e-10 and free_count > 0:
        # Determine how much to shift per free element
        delta = diff / free_count
        # Propagate amounts
        for i in free:
            prev_val = y_copy[i]
            new_val = prev_val - delta
            if new_val <= 1e-12:
                # Touched the boundary, no longer free
                y_copy[i] = 0
                diff += prev_val
                free = free[free != i]
                free_count -= 1
            elif new_val >= 1 - 1e-12:
                y_copy[i] = 1
                diff -= (1 - prev_val)
                free = free[free != i]
                free_count -= 1
            else:
                y_copy[i] = new_val
                diff -= delta
            if abs(diff) < 1e-10:
                break
                
    return y_copy

def targeted_swap(h, c, ones, zeros, off):
    """Find a good swap (an i from ones, j from zeros) that reduce worst shift by 1."""
    # find worst shift:
    s_max = np.argmax(c)
    # Compute A (indexOf 1s contributing to shift s_max)
    A = [i for i in ones if i+s_max < len(h) and h[i+s_max] == 0]
    B = [j for j in zeros if j + s_max < len(h) and h[j+s_max] == 1]
    
    if A and B:
        i = A[np.random.randint(len(A))]
        j = B[np.random.randint(len(B))]
        return i, j
    else:
        # fallback to random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        return i, j

def local_binary_search(h, time_left, t0):
    """Local combinatorial search refining h."""
    # Using correlation as computed ISO for now:
    # We copy simplest tracking
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    # Use full correlation of h
    corr = compute_correlation(h)
    old_max = max(corr)
    st_time = time.time()
    max_iters = 10000
    iter_count = 0
    T = 1.0 # Temperature for annealing late on
    while iter_count < max_iters:
        # Enough time:
        if time.time() - st_time > time_left:
            break
        # Select a swap using targeted strategy
        i, j = targeted_swap(h, corr, ones, zeros, len(corr)//2)
        # Hypothetical delta:
        old_contrib_i = 0 if i + np.argmax(corr) >= len(h) or h[i + np.argmax(corr)] == 0 else 1
        old_contrib_j = 0 if j + np.argmax(corr) >= len(h) or h[j + np.argmax(corr)] == 1 else 1
        old_diff = old_contrib_i - old_contrib_j
        # Apply swap temporarily
        h[i], h[j] = 0, 1
        # Recompute with updated correlations 
        # This may be needed for several sub-domain: we keep it
        corr_new = swap_update(corr, h, i, j, len(corr)//2)
        new_max = max(corr_new)
        if new_max < old_max:
            old_max = new_max
            # Update sets
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            corr = corr_new
        elif new_max == old_max:  # Allow this with probability exp(-Δ/T)
            if np.random.random() < np.exp(-(new_max - old_max) / T):
                old_max = new_max
                corr = corr_new
                # Update sets
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
        else:  # worse
            # Backtrack
            h[i], h[j] = 1, 0
        iter_count += 1
        if iter_count % 1000 == 0:
            T *= 0.99
    return h, old_max

def upscale_binary(h, new_size):
    """Linear upscaling of binary h to new_size."""
    n = len(h)
    new_h = np.zeros(new_size)
    ratio = new_size / n
    for i in range(n):
        start = int(i * ratio)
        end = int((i + 1) * ratio)
        for j in range(start, end):
            new_h[j] = h[i]
    return new_h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    t0 = time.time()
    
    # Discretizations to try (smaller ones for speed)
    ns = [80, 120, 180]
    best_found = None
    best_c5 = float('inf')
    best_n = None
    
    # Work on ns in order, skip oversized ones if time is tight
    safety_margin = 2.0
    for n in ns:
        if time.time() - t0 > budget_s - safety_margin:
            break
        time_remaining = budget_s - (time.time() - t0) - safety_margin
        if time_remaining <= 0:
            break
        # Initialize
        h = np.random.rand(n)
        h = project_onto_feasible(h, n / 2.0)
        dx = 2.0 / n
        # Continuous optimization:
        iter_count = 0
        max_iter = min(10000, int(50000 / n))  # Stability with n
        alpha = 5.0
        eta = 0.5 * dx  # Learning rate
        beta = 0.9  # Momentum
        h_prev = h.copy()
        phi_prev = float('inf')
        c_max_history = []
        for iteration in range(max_iter):
            if time.time() - t0 > budget_s - safety_margin:
                break
            # Compute correlation
            c = compute_correlation(h)
            # Smoothed objective
            phi_alpha, weights = smoothed_objective_and_weights(c, alpha)
            # Compute gradient
            g = grad_h(h, weights)
            # Momentum step
            h_tent = h - eta * (g + beta * (h - h_prev))
            # Project
            h_proj = project_onto_feasible(h_tent, n / 2.0)
            # Backtracking line search
            c_step = compute_correlation(h_proj)
            phi_step, _ = smoothed_objective_and_weights(c_step, alpha)
            tries = 0
            while phi_step > phi_alpha and tries < 5:
                eta *= 0.5
                h_tent = h - eta * (g + beta * (h - h_prev))
                h_proj = project_onto_feasible(h_tent, n / 2.0)
                c_step = compute_correlation(h_proj)
                phi_step, _ = smoothed_objective_and_weights(c_step, alpha)
                tries += 1
            # Update tracking
            h = h_proj
            h_prev = h.copy()
            # Update alpha if improvement
            if abs(phi_alpha - phi_prev) > 1e-6:
                alpha *= 1.2
            phi_prev = phi_alpha
            c_max = max(c)
            c_max_history.append(c_max)
            if len(c_max_history) > 50 and np.std(c_max_history[-50:]) < 1e-8:
                # Potential stagnation: stop early
                break
            iter_count += 1
        # Final evaluation for continuous h
        c_final = compute_correlation(h)
        actual_c5 = max(c_final) * dx
        if actual_c5 < best_c5:
            best_c5 = actual_c5
            best_n = n
            best_h = h.copy()
    # Discretization complete — round final h to binary
    # Binary refinement:
    comp_h = np.copy(best_h)
    # Sort indices by values descending
    indices_sorted = np.argsort(comp_h)[::-1]
    # Set top half indices to 1, rest to 0 to maintain sum
    num_ones = int(len(comp_h) / 2)
    comp_h.fill(0)
    comp_h[indices_sorted[:num_ones]] = 1
    
    # Local "accidental" binary search on rounded
    time_left = budget_s - (time.time() - t0) - safety_margin
    final_h, final_c5 = local_binary_search(comp_h, time_left, t0)
    actual_final_c5 = max(compute_correlation(final_h)) * dx
    
    if actual_final_c5 < best_c5:
        best_c5 = actual_final_c5
        best_h = final_h
    return best_h.tolist(), best_c5, len(best_h)
