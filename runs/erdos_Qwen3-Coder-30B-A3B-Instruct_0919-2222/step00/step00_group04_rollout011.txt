import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The max_k corresponds to max of the correlation array
    max_val = np.max(corr)
    return max_val * dx

def project_to_simplex(v, target_sum):
    """
    Project vector v onto the simplex with sum = target_sum and bounds [0,1].
    """
    n = len(v)
    # Make sure vector sum equals target_sum by scaling
    current_sum = np.sum(v)
    if abs(current_sum) > 1e-10:
        v = v * target_sum / current_sum
    else:
        v = np.full(n, target_sum / n)
    
    # Project onto [0,1] box and then correct sum
    v = np.clip(v, 0, 1)
    
    # Use simple iterative projection: if sum > target, reduce larger values by same amount
    while True:
        diff = np.sum(v) - target_sum
        if abs(diff) < 1e-6:
            break
        if diff > 0:
            # Reduce some elements
            indices = np.argsort(v)
            # Reduce smallest elements
            for i in reversed(indices):
                if v[i] > 0:
                    reduction = min(v[i], diff)
                    v[i] -= reduction
                    diff -= reduction
                    if diff <= 0:
                        break
        else:
            # Increase some elements
            indices = np.argsort(-v)
            for i in reversed(indices):
                if v[i] < 1:
                    increase = min(1 - v[i], -diff)
                    v[i] += increase
                    diff += increase
                    if diff >= 0:
                        break
    return v

def compute_gradient(h, dx):

    Compute the subgradient of the max correlation function,
    which corresponds to the worst shift k.
    """

    corr = np.correlate(h, 1 - h, mode='full')
    # Find the max value and corresponding indices
    max_corr = np.max(corr)

    # For discrete case, drop the last elements (negative shifts)
    # and represent indices properly to both be within [0, n-1] for real shift.
    # The correlation corresponds to integral in lag.
    # Since we have correlation of length 2*n-1, index i corresponds to shift (i - (n-1)).
    # e.g., index n-1 maps to shift 0.
    # So we keep full full correlation even though issue is valid.
    # To get the worst shifts, we filter by which indices match max
    
    # For any given shift index in [0, 2n-2], compute gradient contribution
    grad = np.zeros(n)
    
    # Process each worst shift index
    for min_ind in max_indices:
        shift = min_ind - (n - 1)  # The shift value
        if shift >= 0:
            # Overlap [0, n-1-shift] : h[i] (1-h[i+shift])
            # Gradient element i: 
            #   contributions from h[i] * (1-h[i+shift]) term 
            #   = 1 - h[i+shift] 
            #   , and also from h[i-shift] * h[i] term  
            #   = -h[i-shift]
            # an earlier wrong interpretation: we must note right conventions:
            # For index i in [0, n-1-shift], the first term contributes (1 - h[i+shift])
            # When h[i-shift] exists as second term: contributes -h[i-shift] to grad[i]
            # we sum over all such indices that appear under this shift.
            # The contribution from shift k is from i in [0, n-1-k] to (1 - h[i+k])
            # and also the contribution from terms of the form h[j] h[j+k] where j to make i = j+k
            
            # Index i where h[i] contributes:
            # to curvature 1 - h[i+shift] when 0 <= i <= n-1-shift
            # and to negative h[i-shift] when shift <= i <= n-1
            for i in range(n - shift):
                grad[i] += 1 - h[i + shift]
            # From h[i-shift] * h[i] terms where i >= shift and we compute -h[i-shift]
            for i in range(shift, n):
                if i-shift >= 0:  # avoid underflow
                    grad[i] -= h[i-shift]
        else:
            # Similar logic but shift is negative
            # Shift value is negative, so effective sum is for i in [0, n+shift-1]
            shift = -shift  # Now positive
            for i in range(n - shift):
                grad[i] += 1 - h[i + shift]
            for i in range(shift, n):
                if i-shift >= 0:
                    grad[i] -= h[i-shift]
    
    return grad
