def evaluate_erdos_solution(h, dx):
    # Compute the overlap integral C5
    n = len(h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Only keep the relevant part of the convolution
    # The contribution of h(x) * (1 - h(x+k)) where k ranges from -n+1 to n-1
    # But we only care about valid overlaps, which are when both h(x) and h(x+k) are defined
    # So conv[n-1:2*n-1] gives us the values for k from 0 to 2*n-2
    # But we want to skip first and last few correlations that don't make sense

    return np.max(conv) * dx
