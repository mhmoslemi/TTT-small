import numpy as np
import time
import random

def evaluate_h(h):
    """Evaluate the overlap integral C5 for a given h."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode="full")
    max_O_count = O_counts.max()
    return max_O_count * dx

def max_overlap_counts(h):
    """Return the maximum overlap count."""
    O_counts = np.correlate(h, 1 - h, mode="full")
    return O_counts.max()

def random_balanced_binary(n, rng):
    """Generate a random balanced binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    ones_indices = rng.sample(range(n), n // 2)
    h[ones_indices] = 1
    return h

def project_onto_box_and_sum(v, target):
    """Project vector v onto box [0,1] and sum constraint."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def binary_anneal(n, time_limit, rng):
    """Run simulated annealing on a binary vector."""
    h = random_balanced_binary(n, rng)
    cur_val = max_overlap_counts(h)
    best_val = cur_val
    best_h = h.copy()
    start_time = time.time()
    
    # precompute early
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() - start_time < time_limit:
        T = 0.5 * (1 - (time.time() - start_time) / time_limit) + 0.001
        
        # propose swap
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        # swap
        h[p], h[q] = 0, 1
        new_val = max_overlap_counts(h)
        delta = new_val - cur_val
        
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # revert
            h[p], h[q] = 1, 0
            
        # update lists efficiently
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    # Set seeds
    np.random.seed(seed)
    random.seed(seed)
    rng = random.Random(seed)
    
    # Nuemric targets
    target_c5 = 0.3807
    
    # List of n values to test
    n_values = [30, 40, 45, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    start_time = time.time()
    for n in n_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        remaining_time = budget_s - (time.time() - start_time) - 5
        if remaining_time <= 0:
            break
            
        # warm up for time budgeting
        try:
            h, max_O = binary_anneal(n, remaining_time, rng)
            c5 = max_O * 2.0 / n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n
        except Exception as e:
            continue
            
        if best_c5 <= target_c5:
            break
    
    return (best_h, best_c5, best_n)
