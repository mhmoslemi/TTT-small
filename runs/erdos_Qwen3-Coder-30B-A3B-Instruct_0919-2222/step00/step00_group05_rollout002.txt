import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import math

def evaluate(h, dx):
    """Evaluate the maximum overlap integral for the given step function h."""
    # Compute the full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum correlation over all shifts (independent of k, since we're in discrete setting)
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, dx):
    """Project h to make sure it's in [0,1] and integrates to 1."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to make sum = 1/dx
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    else:
        h = np.ones_like(h) * (1.0 / (len(h) * dx))
    return h

def generate_random_feasible(n_points, dx):
    """Generate a random feasible h vector."""
    h = np.random.rand(n_points)
    h = project_to_feasible(h, dx)
    return h

def swap_moves(h, dx, num_moves=10):
    """Generate neighbor solutions by swapping values in h."""
    h_new = h.copy()
    for _ in range(num_moves):
        i, j = np.random.randint(0, len(h), size=2)
        h_new[i], h_new[j] = h_new[j], h_new[i]
    return project_to_feasible(h_new, dx)

def local_search_step(h, dx, max_iters=100):
    """Perform a local search around current h."""
    current_h = h.copy()
    current_c = evaluate(current_h, dx)
    num_points = len(h)
    
    for _ in range(max_iters):
        # Try various moves
        # Option 1: Small perturbation
        h_perturb = current_h.copy()
        idx = np.random.randint(0, num_points)
        delta = np.random.normal(0, 0.01)
        h_perturb[idx] += delta
        h_perturb = project_to_feasible(h_perturb, dx)
        c_perturb = evaluate(h_perturb, dx)
        
        # Option 2: Swapping
        h_swap = swap_moves(current_h, dx, num_moves=1)
        c_swap = evaluate(h_swap, dx)
        
        # Accept any improvement
        best_h, best_c = current_h, current_c
        if c_perturb < current_c:
            best_h, best_c = h_perturb, c_perturb
        if c_swap < current_c:
            best_h, best_c = h_swap, c_swap
            
        if best_c < current_c:
            current_h, current_c = best_h, best_c
        else:
            break
        
    return current_h, current_c
