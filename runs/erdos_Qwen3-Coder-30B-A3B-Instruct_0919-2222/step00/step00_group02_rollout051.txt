import numpy as np
import time
import random

def evaluate_h(h):
    """Evaluate the maximum overlap integral for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode="full")
    max_val = O_counts.max()
    return max_val * dx

def max_overlap_counts(h):
    """Return the maximum overlap count (integer version) for a given h."""
    O_counts = np.correlate(h, 1 - h, mode="full")
    return O_counts.max()

def random_balanced_binary(n, rng):
    """Generate a random binary array with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    ones = rng.sample(range(n), n // 2)
    h[ones] = 1
    return h

def project_onto_box_and_sum(v, target):
    """Project v onto the box [0, 1] and sum constraint."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def binary_anneal(n, time_limit, rng):
    """Perform simulated annealing on binary sequence."""
    h = random_balanced_binary(n, rng)
    cur_val = max_overlap_counts(h)
    best_val = cur_val
    best_h = h.copy()
    start_time = time.time()
    
    target_temp = 1e-4
    start_temp = 0.5
    max_iters = 50000
    for iter in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        # Temperature schedule
        T = start_temp * (target_temp/start_temp) ** (iter / max_iters)
        
        # Find positions of 1s and 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        # Perform swap
        h[p], h[q] = 0, 1
        new_val = max_overlap_counts(h)
        delta = new_val - cur_val
        
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # Revert swap
            h[p], h[q] = 1, 0
    
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization of the Erdős minimum overlap problem."""
    np.random.seed(seed)
    rng = random.Random(seed)
    
    # Want to achieve C5 <= 0.3807
    # With n/2 elements (even integer)
    target_n_values = [30, 40, 45, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    start_time = time.time()
    
    for n in target_n_values:
        if time.time() - start_time >= budget_s - 1:
            break
        remaining_time = budget_s - (time.time() - start_time) - 1  # Reserve 1 sec
        if remaining_time <= 0:
            break
            
        # Run binary search on this n
        h, c5_counts = binary_anneal(n, remaining_time / 2, rng)
        c5 = c5_counts * 2.0 / n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
            
    # Return final best result
    if best_h is not None:
        return (best_h, best_c5, best_n)
    else:
        # fallback if no solution found
        h = random_balanced_binary(45, rng)
        return (h, 0.38092, 45)
