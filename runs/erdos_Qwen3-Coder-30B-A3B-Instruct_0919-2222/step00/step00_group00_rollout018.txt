```python
import time
import numpy as np
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, mul_elemwise, 0, 1
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_result = None
    
    # Test different sizes
    sizes = [80, 120, 160]
    
    # Define a few key helper functions
    def compute_full_correlation(h, dx):
        # h is a list or array of 0/1 values
        n = len(h)
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1 - h_arr, mode="full") * dx
        return corr
    
    def get_shifts_for_n(n):
        # Use a subset of shifts, starting from 1 and going up to n//4 roughly
        return list(range(1, min(n//4 + 1, n)))
    
    def build_milp_model(x, y, k_indices, shift_data, n, dx):
        # x: binary variables shaped (n,)
        # y: list of variables shaped (n-k,) for each active k
        # k_indices: list of active shifts
        # shift_data: contains the shift structures
        
        # In this implementation, we'll sum over the active shifts
        # slightly different approach than earlier version but still similar # goal
        
        # Constraint: sum x[i] = n/2
        mass_constraint = cp_sum(x) == n / 2

        # For each active shift k in k_indices, add the constraint
        constraints = [mass_constraint]
        
        t = Variable()  # scalar variable representing max overlap
        
        for k, k_idx in enumerate(k_indices):
            # indices where shift k makes sense
            if k_idx > 0:
                effective_n = n - k_idx
                if effective_n <= 0:
                    continue
                # x[i] * x[i+k] becomes y[i]
                # But in practice we need to compute the full constraint per shift
                corr_expr = cp_sum(x[:effective_n]) - cp_sum(y[(k_idx, slice(None))])
                constraints.append(corr_expr <= t)
            
        return constraints, t
    
    # Pre-compute shift structure efficiently
    full_shifts = []
    
    def initial_h_from_ref(n):
        # Use the provided reference h if possible (e.g., can be 45-point initial starts)  
        h_init = [0] * n
        # Sample from high-performing symmetric pattern to encode likely good pulse-like patterns
        # e.g., a 1-block followed by 0-block
        for i in range(n // 2):
            h_init[i] = 1
        return h_init

    def solve_lazy_milp(n, dx, max_time_remaining, step_limit=10):
        # returns best h (list of 0/1), C5 value (scalar), solve time
        h_best = None
        c5_best = float('inf')
        elapsed_time = 0.0

        # Assume cvxpy with a MILP solver will be used
        # Each solver has different access patterns
        # Using CBC or ECOS_BB if available; fallback to ECOS if not
        
        # Strategy:
        # - Initial small set of shifts
        # - Add violating shifts dynamically via lazy loop
        # - At each step, verify full correlation and update best solution
        # - Store best feasible result across all runs
        
        # Initial variables
        # Variable x[i] for i = 0..n-1 (binary)
        x = Variable(n, boolean=True)
        
        # Some sequences are very symmetric in form – 
        # so we assume indices are well-ordered unless exploit 'auto-cyclic'
        
        # Initial set of active shifts to manage size, introduce some smaller shifts first
        K0 = min(n // 4 + 1, n)
        active_shifts = list(range(1, K0))
        
        # Let's track auxiliary variables and their constraints
        # We'll rebuild constraints explicitly for active shifts
        # For each i and k in active shifts, we create y variables to store x[i]*x[i+k]
        # Each y[i,k] corresponds to i where both i and i+k are valid
        
        y_vars = {}
        y_vars_by_k = {}  # structure: {k: [y_vars_for_k]}
        
        # Temp parameters for trial
        # Just one global target max_thres variable t = max_k sum x - sum y_k
        t = Variable()
        
        start_time = time.time()
        best_h = None
        best_t = float('inf')
        iter_count = 0
        
        # Lazy constraint strategy
        while True:
            if iter_count > step_limit:
                break    
            
            elapsed = time.time() - start_time
            if elapsed > max_time_remaining * 0.97:  # leave some buffer
                break
                
            iter_count += 1
            
            try:
                # Set up linearized model for active shifts
                # Rebuild x-based variables & expressions
                
                # Prepare integer constraints from scratch for active shifts
                # Recall that after construction, x[i]*x[i+k] = y[i,k]  
                # y[i,k] is introduced for i=0 to n-k-1
                
                # In our approach we introduce y and track efficiently
                
                #---

                # Clear any old auxiliary variables per iteration
                aux_vars = [] # for all y_ij variables needed
                
                if iter_count == 1:
                    # initial setup
                    constraints = [cp_sum(x) == n / 2]  # user-supplied
                    # Add McCormick constraints for ALL y for each active shift
                    y_vars = {}
                    y_vars_by_k = {}
                    k_list_used = active_shifts[:]
                    for k in k_list_used:
                        # For each shift k, set up y[i,k] from i=0 to n-k-1
                        # Define y_vars for this shift
                        y_vs = [Variable(boolean=True) for _ in range(n - k)]
                        y_vars_by_k[k] = y_vs
                        
                        # generate constraint per y element
                        for i in range(n - k):
                            # y[i,k] <= x[i],   y[i,k] <= x[i+k]
                            constraints.append(y_vs[i] <= x[i])
                            constraints.append(y_vs[i] <= x[i+k])
                            # y[i,k] >= x[i] + x[i+k] - 1
                            constraints.append(y_vs[i] >= x[i] + x[i+k] - 1)
                    
                    # Now add shift-specific max constraints
                    for k in k_list_used:
                        # The overlap for shift k is sum[x] - sum[y]
                        # Which is same as sum(x[:n-k]) - sum(y[:n-k])
                        expr = cp_sum(x[:n-k]) - cp_sum(y_vars_by_k[k])
                        constraints.append(expr <= t)
                else:
                    constraints = [] 
                    constraints.append(cp_sum(x) == n / 2)
                    
                    # Only reuse keep half of y structures or add new ones together
                    # but ignore previous constraints to avoid overloading solver
                    
                # Clear old
                y_vars.clear()
                y_vars_by_k.clear()
                
                # Rebuild the right constraints
                active_k = list(range(1, K0))
                
                y_vars_by_k = {}
                for k in active_k:
                    if k >= n:
                        continue
                    y_vs = [Variable(boolean=True) for _ in range(n - k)]
                    y_vars_by_k[k] = y_vs
                    
                    for i in range(n - k):
                        # Constraints
                        constraints.append(y_vs[i] <= x[i])
                        constraints.append(y_vs[i] <= x[i+k])
                        constraints.append(y_vs[i] >= x[i] + x[i+k] - 1)

                # Finalize shift constraints with all active k
                for k in active_k:
                    if k >= n:
                        continue
                    expr = cp_sum(x[:n-k]) - cp_sum(y_vars_by_k[k])
                    constraints.append(expr <= t)
                
                # Minimize t -- should be an integer min which will force the actual max
                objective = Minimize(t)
                prob = Problem(objective, constraints)
                
                solver_name = None
                
                # Fallback detection of SUITABLE solver, assume CBC available  
                try:
                    prob.solve(solver=cp.CBC, verbose=False)
                    solver_name = "CBC"
                except:
                    try:
                        prob.solve(solver=cp.ECOS_BB, verbose=False)
                        solver_name = "ECOS_BB"
                    except:
                        # At very minimum, fallback to trivial step or exit?
                        break
                
                if prob.status == 'optimal':  
                    t_opt = t.value
                    x_opt = [int(round(v.value)) for v in x]
                    
                    # Evaluate full correlation and check if this t_opt is saturated
                    h_full = np.array(x_opt)
                    corr_full = np.correlate(h_full, 1-h_full, mode='full')
                    c5_candidate = max(corr_full) * dx
                    
                    assert abs(c5_candidate - t_opt * dx) < 1e-6, "Mismatch between t and actual max corr"
                    
                    # Update best if better
                    if c5_candidate < c5_best:
                        c5_best = c5_candidate
                        h_best = x_opt.copy()
                        
                    # --- verify constraint relaxations by checking all full correlations
                    # could be skipped in later iteration if it already fits the full set
                    
                    # Now, finite-verify correction  
                    corr_full = np.correlate(np.array(x_opt), 1-np.array(x_opt), mode='full')
                    all_cors = corr_full
                    full_max = max(all_cors)
                    if full_max > t_opt:
                        # Add some tighter shift: we might be missing one reducing tightness
                        violating_shifts = np.where(all_cors > t_opt)[0] - len(all_cors)//2
                        if len(violating_shifts) > 0:
                            # Add min violation shift; usually one shift more than needed
                            k_add = max(violating_shifts) + len(all_cors)//2 if max(violating_shifts) < 0 else min(violating_shifts) + len(all_cors)//2
                            if k_add not in active_shifts:
                                active_shifts.append(k_add)
                                if len(active_shifts) > n // 2:
                                    # stop adding rows beyond reasonable size
                                    break
                                    
                    else:
                        # converged and no violation seen, so stop
                        break  
                else:
                    # Solver failed, fallback or stop
                    break
                    
            except Exception as e:
                if 'SolverError' in str(type(e)):
                    break
                elif elapsed > 10:  
                    break # slight wallclock timer, gives up if long run
                else:
                    pass # ignore for now
                
        # Final polishing step if possible
        if h_best is not None and time.time() - t_start < budget_s - 5:
            # Do deterministic polishing
            h_opt = np.array(h_best)
            corr_full = np.correlate(h_opt, 1-h_opt, mode='full')
            max_k = np.argmax(corr_full)
            max_v = corr_full[max_k]
            
            # try improvements through swaps:
            # find all 1s at i such that i+k is 0 in h and
            # find 0s at j such that j+k is 1
            k_opt = max_k - len(corr_full) // 2
            if k_opt < 0:
                k_opt = len(corr_full) // 2 - max_k
            
            try:
                indices_ones = np.where(h_opt == 1)[0]
                indices_zeros = np.where(h_opt == 0)[0]
                
                count = 0
                while count < 10:
                    count += 1
                    found_improving_swap = False
                    for i in indices_ones:
                        if i + k_opt < len(h_opt) and h_opt[i + k_opt] == 0:
                            for j in indices_zeros:
                                if j + k_opt < len(h_opt) and h_opt[j + k_opt] == 1:
                                    # perform swap
                                    h_opt[i] = 0
                                    h_opt[j] = 1
                                    found_improving_swap = True
                                    break
                        if found_improving_swap:
                            break
                    if not found_improving_swap:
                        break
                        
                    corr_full = np.correlate(h_opt, 1-h_opt, mode='full')
                    if max(corr_full) <= max_v - 1e-5:  # small numerical tolerance
                        max_v = max(corr_full)
                    else:
                        h_opt[i] = 1
                        h_opt[j] = 0  # revert
                        break
                        
            except:
                pass
            
            h_best = h_opt.tolist()
            corr_full = np.correlate(h_opt, 1-h_opt, mode='full')
            max_v = max(corr_full)*dx
            if max_v < c5_best:
                c5_best = max_v
                
        return h_best, c5_best
    
    for size in sizes:
        if time.time() - t_start > budget_s - 5:
            break
        dx = 2.0 / size
        
        # run and solve just once per size on a single run until timeout
        # starting from previous best used coordinates or default
        
        remaining_time = budget_s - (time.time() - t_start)
        if remaining_time < 10:
            break
            
        try:
            h_val, c5_val = solve_lazy_milp(size, dx, remaining_time)
            
            if h_val is not None and c5_val < float('inf'):
                if best_result is None or c5_val < best_result[1]:
                    best_result = (h_val, c5_val, size)
                
        except Exception as e:
            continue
    
    if best_result is None:
        # fallback to very default strategy
        n_final = 80
        h_final = initial_h_from_ref(n_final)
        dx_final = 2.0 / n_final
        corr = np.correlate(np.array(h_final), 1-np.array(h_final), mode='full')
        c5_final = max(corr) * dx_final
        best_result = (h_final, c5_final, n_final)
        
    return best_result
```
