import numpy as np
from scipy.optimize import differential_evolution
import time

def project_to_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum"""
    v = np.array(v)
    sorted_v = np.sort(v)[::-1]
    n = len(v)
    # Water-filling algorithm
    tau = 0.0
    cumsum = 0.0
    for i in range(n):
        cumsum += sorted_v[i]
        threshold = (cumsum - target_sum) / (i + 1)
        if threshold <= 1.0:
            tau = threshold
        else:
            break
    x = np.clip(v - tau, 0.0, 1.0)
    return x

def compute_overlap_from_blocks(a, w, N=200):
    """Compute overlap for block representation on a uniform grid of N points."""
    # Construct cumulative widths to find intervals
    s = np.concatenate([[0], np.cumsum(w)])
    if not np.isclose(s[-1], 2.0):
        s = s * 2.0 / s[-1]  # Normalize to [0, 2]

    # Build h on uniform grid
    dx = 2.0 / N
    h = np.zeros(N)
    for i in range(len(w)):
        start = int(np.round(s[i] / dx))
        end = int(np.round(s[i+1] / dx))
        h[start:end] = a[i]

    return h

def block_to_h(a, w, N=200):
    """Sample block structure h at N points."""
    dx = 2.0 / N
    s = np.concatenate([[0], np.cumsum(w)])
    if not np.isclose(s[-1], 2.0):
        s = s * 2.0 / s[-1]
    
    h = np.zeros(N)
    for i in range(len(w)):
        start_idx = int(np.round(s[i] / dx))
        end_idx = int(np.round(s[i+1] / dx))
        h[start_idx:end_idx] = a[i]
    return h

def evaluate_fitness(params, M=12, lambda_penalty=1000.0):
    """Evaluate the fitness for given block parameters."""
    u = params[:M]
    v = params[M:]
    w = 2.0 * np.exp(u) / np.sum(np.exp(u))
    a = 1.0 / (1.0 + np.exp(-v))

    # Ensure inclusion of integral constraint in float arithmetic
    integral_check = np.sum(a * w)
    penalty = lambda_penalty * abs(integral_check - 1.0)

    # Compute the overlap function via block interactions
    # Precompute interval data
    s = np.concatenate([[0], np.cumsum(w)])
    if not np.isclose(s[-1], 2.0):
        s = s * 2.0 / s[-1]
    
    N = 200  # match with final sampling
    dx = 2.0 / N
    num_shifts = 2 * N - 1  # full correlation length
    
    # Compute G(k) for all k
    G = np.zeros(num_shifts)
    
    for i in range(M):
        start_i = int(np.round(s[i] / dx))
        end_i = int(np.round(s[i+1] / dx))
        a_i = a[i]
        for j in range(M):
            start_j = int(np.round(s[j] / dx))
            end_j = int(np.round(s[j+1] / dx))
            a_j = a[j]
            # Evaluate overlap for shift k
            for k_shift in range(num_shifts):
                k_idx = k_shift - (N - 1)
                start_overlap = max(start_i, start_j + k_idx)
                end_overlap = min(end_i, end_j + k_idx) 
                if end_overlap > start_overlap:
                    overlap = (end_overlap - start_overlap) * dx
                    # Populate the correct index in the correlation array
                    idx = k_shift
                    G[idx] += a_i * a_j * overlap
    
    # Compute f(k) = 1 - G(k)
    f = 1.0 - G
    # f value used as objective; actual max and penalty are combined
    obj_val = np.max(f) + penalty
    return obj_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    M = 12  # number of blocks
    total_time = 0.0
    best_solution = None
    best_fitness = np.inf

    # Phase 1: Global optimization
    bounds = [(-5.0, 5.0)] * (2 * M)
    DE_kwargs = {
        'bounds': bounds,
        'strategy': 'best1bin',
        'maxiter': 50,
        'popsize': 15 * (2*M),
        'tol': 1e-6,
        'mutation': (0.5, 1.0),
        'recombination': 0.7,
        'seed': seed,
        'workers': 1
    }
    
    de_result = differential_evolution(evaluate_fitness, bounds, **DE_kwargs)
    best_evolved = de_result.x
    best_fitness = evaluate_fitness(best_evolved, M=M)
    
    # Phase 2: Multiple runs of local refinement from perturbed start
    n_runs = 3
    N_points = 200  # final grid size
    dx = 2.0 / N_points
    sqrt2 = np.sqrt(2.0)
    for _ in range(n_runs):
        if time.time() - start_time > budget_s - 5:
            break
        # small random perturbation
        perturbed = best_evolved + np.random.normal(0, 0.02, best_evolved.shape)
        updated = differential_evolution(evaluate_fitness, bounds, 
                                        x0=perturbed, **DE_kwargs).x
        val = evaluate_fitness(updated, M=M)
        if val < best_fitness:
            best_fitness = val
            best_evolved = updated

    # Phase 3: Convert to high-resolution function and refine with subgradients
    u = best_evolved[:M]
    v = best_evolved[M:]
    w = 2.0 * np.exp(u) / np.sum(np.exp(u))
    a = 1.0 / (1.0 + np.exp(-v))
    
    # Ensure integral is exact
    integral = np.sum(a * w)
    a = a / integral

    # Final discretization using uniform grid
    h = block_to_h(a, w, N_points)
    
    # Refinement with projected subgradient descent
    def compute_subgradient(h, k_best):
        """Compute subgradient for C_k where k is the worst shift."""
        # Define left and right shifts manually to target specific k
        C_k = np.sum(h[:-k_best] * (1 - h[k_best:]))
        # Gradient of C_k w.r.t h[i]
        g = np.zeros_like(h)
        # For C_k = sum h[i]*(1-h[i+k])
        # Derivative is (1 - h[i+k] - h[i])
        for i in range(len(h) - k_best):
            g[i] += 1 - h[i + k_best]
        for i in range(k_best, len(h)):
            g[i] += -h[i - k_best]
        return g
    
    # Keep a rotating best
    best_h = h.copy()
    best_c5 = np.inf
    
    # The optimized value is worst-overlap; Maximizing min autocorrelation in frequency domain
    # is equivalent to minimizing sup O_k, where O_k = h*(1-h)[k]
    # Which equals minimizing max_k (1 - G(k)) where G(k) = h*h[k]
    # So we minimize max_k (1 - integral h(x) h(x+k) dx)
    # Hence maximize min_k integral h(x) h(x+k) dx

    # compute full correlation now
    corr_f = np.correlate(h, 1.0 - h, mode='full')
    corr_f = corr_f * dx # account for dx
    current_c5 = np.max(corr_f)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()

    # Projected gradient descent for small improvements
    alpha = 0.2
    iter_limit = 5000
    for it in range(iter_limit):
        if time.time() - start_time > budget_s - 5:
            break

        # compute correlations
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        best_k_idx = np.argmax(corr_full)
        # find that shift value (not index for corr_full)
        shift = best_k_idx - (len(h) - 1)

        # compute subgradient
        g = np.zeros(len(h))
        for i in range(len(h) - abs(shift)):
            if shift > 0:
                g[i] += (1 - h[i + shift])
            elif shift < 0:
                g[i] += -h[i - shift]
            else:
                g[i] += (1 - 2 * h[i])

        # step and project
        h_new = h - alpha * g
        # clamp to [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)
        # project to capped simplex (mean of 0.5)
        target_sum = len(h) * 0.5
        h_new = project_to_capped_simplex(h_new, target_sum)
        
        # update
        h = h_new
        alpha *= 0.999

        # recompute
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        c5_local = np.max(corr_full)
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h.copy()

    return (best_h.tolist(), best_c5, N_points)
