import numpy as np
import time
import random

def evaluate_h(h):
    """Evaluate the maximum overlap integral for the given function h."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode='full')
    max_val = O_counts.max()
    return max_val * dx

def max_overlap_counts(h):
    """Evaluate the maximum overlap count for the given function h."""
    n = len(h)
    O_counts = np.correlate(h, 1 - h, mode='full')
    return O_counts.max()

def random_balanced_binary(n, rng):
    """Generate a random balanced binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    rng.shuffle(h)
    return h

def project_onto_box_and_sum(v, target):
    """Project vector v onto the box [0,1] and sum constraint."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def binary_anneal(n, time_limit, rng, target_max):
    """Simulated annealing for binary h."""
    h = random_balanced_binary(n, rng)
    cur_val = max_overlap_counts(h)
    best_val = cur_val
    best_h = h.copy()
    start_time = time.time()
    
    max_iters = 200000  # upper bound on iterations
    for iter in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
            
        T = 0.5 * (1 - elapsed / time_limit) + 0.001
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        # swap
        h[p], h[q] = 0, 1
        new_val = max_overlap_counts(h)
        delta = new_val - cur_val
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val <= target_max:
                    break
        else:
            # revert
            h[p], h[q] = 1, 0
    
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    rng = random.Random(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Define candidate n values
    n_values = [30, 40, 45, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    n_values = [n for n in n_values if n % 2 == 0]
    
    # Precompute candidate values for efficient lookup
    target_maxes = {}
    for n in n_values:
        target_maxes[n] = int(np.floor(0.3807 * n / 2))
    
    for n in n_values:
        if time.time() - start_time > budget_s - 1:
            break
        time_left = budget_s - (time.time() - start_time)
        if time_left <= 1:
            break
        time_per_n = time_left / (len(n_values) - n_values.index(n))
        target_max = target_maxes[n]
        
        # Run local search
        h, val = binary_anneal(n, time_per_n, rng, target_max)
        c5 = val * 2.0 / n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
            
        if best_c5 <= 0.3807:
            break
    
    if best_h is None:
        n = 45  # default
        h = random_balanced_binary(n, rng)
        best_h = h.copy()
        best_c5 = evaluate_h(best_h)
        best_n = n

    # Ensure output format
    h_values = best_h.astype(float).tolist() if best_h is not None else []
    c5_bound = best_c5 if best_c5 is not None else 1.0
    n_points = best_n if best_n is not None else 45

    return h_values, c5_bound, n_points
