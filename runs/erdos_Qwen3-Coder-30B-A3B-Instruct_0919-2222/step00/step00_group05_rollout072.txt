import numpy as np
import time
import random
from scipy import signal

def evaluate(h, dx):
    # Compute the overlap integral C5 = max_k int h(x)(1 - h(x+k)) dx
    # Using full convolution
    corr = signal.correlate(h, 1 - h, mode="full")
    # The max index corresponds to k = 0, 1, ..., 2*dx*(len(h)-1)
    # We want max of the valid k values: k from 0 to 2*dx*(len(h)-1)
    # But since full mode, need to extract middle part for overlap
    mid = len(corr) // 2
    half_len = len(h) - 1
    corr_middle = corr[mid - half_len : mid + half_len + 1]
    max_val = np.max(corr_middle) * dx
    return max_val

def project_to_constraints(h, dx):
    # Normalize so that sum(h) * dx = 1
    # i.e., sum(h) = 1/dx
    scale = 1 / dx / len(h)
    h = h * scale
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    return h

def generate_random_feasible(n_points, dx):
    # Generate random feasible h
    h = np.random.rand(n_points)
    h = project_to_constraints(h, dx)
    return h

def local_search_move(h, dx, T=0.1, iterations=100):
    # Perform random local search on h
    h_best = h.copy()
    c_best = evaluate(h, dx)
    for _ in range(iterations):
        h_new = h_best.copy()
        # Propose small random change
        i = random.randint(0, len(h_new)-1)
        delta = random.uniform(-0.1, 0.1)
        h_new[i] += delta
        h_new = project_to_constraints(h_new, dx)
        c_new = evaluate(h_new, dx)
        if c_new < c_best:
            h_best = h_new
            c_best = c_new
    return h_best, c_best

def simulated_annealing_step(h, dx, T):
    # Random perturbation with temperature-based step size
    h_new = h.copy()
    i = random.randint(0, len(h_new)-1)
    delta = random.uniform(-T, T)
    h_new[i] += delta
    h_new = project_to_constraints(h_new, dx)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 100  # reasonable balance for optimization time
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    elapsed = 0
    start_time = time.time()
    
    # Start with initial_h_values if available, otherwise random
    try:
        h_start = initial_h_values
        if len(h_start) != n_points:
            h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    except:
        h_start = generate_random_feasible(n_points, dx)
    
    h_current = h_start.copy()
    c_current = evaluate(h_current, dx)
    
    if c_current < best_c5:
        best_h = h_current.copy()
        best_c5 = c_current

    # Main optimization loop
    T = 0.5  # Starting temperature
    iteration = 0
    T_min = 1e-8

    while elapsed < budget_s * 0.98:
        elapsed = time.time() - start_time
        T = max(T_min, T * 0.9999)  # Gradual cooling
        
        # Random move: Local search + SA
        if random.random() < 0.7:
            # Local search step
            h_trial, c_trial = local_search_move(h_current, dx, T, 50)
        else:
            # Simulated Annealing
            h_trial = simulated_annealing_step(h_current, dx, T)
            c_trial = evaluate(h_trial, dx)

        # Accept or reject
        if c_trial < c_current or random.random() < np.exp(-(c_trial - c_current) / max(T, 1e-10)):
            h_current = h_trial
            c_current = c_trial

        # Update best
        if c_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c_current

        iteration += 1

    # Final evaluation of best solution
    final_c5 = evaluate(best_h, dx) if best_h is not None else float('inf')
    return (best_h.tolist(), final_c5, len(best_h))
