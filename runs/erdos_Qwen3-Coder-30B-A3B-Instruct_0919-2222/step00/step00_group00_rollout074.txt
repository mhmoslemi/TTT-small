            if time.time() - t0 > budget_s - safety_margin:

def correlation(h):
    # Compute ∫ h(x)(1 - h(x+k)) dx for all shifts k
    # Returns vector of length 2*n-1, shift values from -(n-1) to n-1
    c = np.correlate(h, 1 - h, mode='full')
    return c

def smoothed_objective_and_weights(c, alpha):
    # Compute Φ_α = (1/α)·log Σ_k exp(α·c_k)
    # And weight vector w_k = exp(α·c_k) / Σ_j exp(α·c_j) 
    
    c_max = np.max(c)
    if alpha > 100: # Cap alpha to prevent numerical issues
        # For very large alpha, rely on numerical max
        # But also force smooth version with explicit threshold
        exp_c = np.exp(alpha * (c - c_max))
        exp_c = np.clip(exp_c, 1e-20, 1e10)  # Clip extreme values
    else:
        exp_c = np.exp(alpha * (c - c_max))
    
    sum_exp_c = np.sum(exp_c)
    if sum_exp_c == 0:
        weights = np.ones_like(c) / len(c)
    else:
        weights = exp_c / sum_exp_c
    
    smooth_val = (1./alpha) * (np.log(sum_exp_c) + c_max) if alpha > 0 else np.max(c)
    return smooth_val, weights

def gradient_h(h, weights):
    # Given weights w_k for each k, compute ∇Φ_α(h)
    # ∂c_k/∂h[i] = (1 – h[i+k])·1_{i+k∈[0,n)} – h[i–k]·1_{i–k∈[0,n)}
    # So ∇Φ_α(h)_i = Σ_k w_k · [(1 – h[i+k]) I[i+k<n] - h[i-k] I[i-k>=0]]
    
    # Use linear convolution approach: use scipy or numpy
    # But we note a trick: this gradient involves summing products of shifted arrays
    # For numerical reasons, one can break into forward/backward convolutions
    
    n = len(h)
    g = np.zeros_like(h)
    
    # A direct loop (O(n^2)) is ok for small n
    # Instead, use the FFT-based formulation in parts
    # But first verify correctness
    
    # Assume weights are sorted to correspond to shifts from -(n-1) to n-1  
    # Check this definition matches
    # c_k = sum_i h_i (1 - h_{i+k})
    # ∂c_k/∂h_i = (1 - h_{i+k}) 1_{i+k<n} - h_{i-k} 1_{i-k>=0}
    
    # Since both -h[i-k] and (1-h[i+k]) contribution can be easily computed via convolutions
    # Derive the effective shape of number of shifts
    
    # Create extended arrays:
    # positive shifts from 0 to n-1
    # negative shifts from -(n-1) to 0
    
    w_extended = weights
    # Use scipy.fftpack.convolve
    # But for safety, just compute manually:
    
    # Do full O(n^2) compute:
    for i in range(n):
        g_i = 0
        for k in range(-(n-1), n):
            if i+k >=0 and i+k < n:
                term = (1 - h[i+k]) * weights[n-1+k]  # weight for shift k
            else:
                term = 0
                
            if i-k >= 0 and i-k < n:
                term -= h[i-k] * weights[n-1+k]  # same weight although sign may differ
            else:
                term -= 0
                
            g_i += term
            
        g[i] = g_i
    return g

def project_onto_feasible(y, target_sum):
    # Project a vector y onto the intersection of [0,1]^n and { x | sum(x) = target_sum}
    # Water-filling algorithm
    
    y = np.clip(y, 0, 1)
    total = np.sum(y)
    if abs(total - target_sum) < 1e-12:
        return y
    
    excess = total - target_sum
    if abs(excess) < 1e-12:
        return y
    
    # Find where y is in interior (0,1) and partition
    free = np.where((y > 0) & (y < 1))[0]
    while len(free) > 0 and abs(excess) > 1e-12:
        free_y = y[free]
        mid = np.mean(free_y)  # this works due to facility of inline median calc
        # we can compute b = mean of y values in that ranges above threshold
        scale = np.sum(free_y > mid)  # count of elements greater than mid
        if scale == 0:
            break
        adjust = excess / scale
        
        # Remove elements that are about to cross boundaries
        update = np.zeros_like(y)
        for i in range(len(free)):
            if y[free[i]] + adjust >= 1:
                update[free[i]] = 1 - y[free[i]]  # stays at 1
                excess += update[free[i]]
            elif y[free[i]] + adjust <= 0:
                update[free[i]] = -y[free[i]]  # stays at 0
                excess += update[free[i]]
            else:
                update[free[i]] = adjust
                excess -= adjust
        # Apply:
        y += update
        y = np.clip(y, 0, 1)
        # update free set
        free = np.where((y > 0) & (y < 1))[0]
    return y

def gradient_descent_step(h, c, weights, alpha, eta, beta, h_prev, h_prev2):
    # Gradient step with momentum
    n = len(h)
    g = gradient_h(h, weights)
    v = beta * (h - h_prev) + (1 - beta) * g
    
    # Tentative update with momentum
    g_step = h - eta * v
    
    # Project
    h_new = project_onto_feasible(g_step, n/2)
    
    # Evaluate new objective
    c_new = correlation(h_new)  
    smooth_new, weights_new = smoothed_objective_and_weights(c_new, alpha)
    
    # Accept if better
    return h_new, smooth_new, c_new, weights_new

def continuous_phase(h0, n, budget_s, t0):
    # Run full continuous phase with momentum

    max_iter = 10000
    eta = 0.5 * 2.0 / n  # Normalized step size
    beta = 0.9  # momentum 
    alpha = 5.0  # curvature parameter for smooth surrogate
    alpha_step = 1.2  # increase alpha every N
    N_acc = 100  # how often to increase alpha
    
    h = h0.copy()
    h_prev = h.copy()
    h_prev2 = h.copy()
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iter_count = 0
    time_check_interval = 100
    
    for iteration in range(max_iter):
        # Periodic time check
        if iteration % time_check_interval == 0:
            if time.time() - t0 > budget_s - 2.0:
                break
                
        c = correlation(h)
        smooth_val, weights = smoothed_objective_and_weights(c, alpha)
        
        h_new, smooth_new, c_new, weights_new = gradient_descent_step(
            h, c, weights, alpha, eta, beta, h_prev, h_prev2)
        
        # Update momentum
        h_prev2 = h_prev.copy()
        h_prev = h.copy()
        h = h_new.copy()
        
        # Accept step if it improves smoothed surrogate
        # Note we could do stronger line search here with backtracking
        
        # Update best solution in terms of actual max
        if iteration % 10 == 0:
            c_actual = correlation(h)
            current_c5 = np.max(c_actual) * 2.0 / n
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Increase alpha for tighter surrogate
        if iteration % N_acc == 0 and alpha < 200:
            alpha *= alpha_step
        elif alpha >= 200:
            if smooth_new > smooth_val and smooth_new - smooth_val < 1e-6:
                break
                
        iter_count += 1
        
    return best_h, best_c5  

def round_to_binary(h, n):
    # Get `n/2` indices with the highest values and set them to 1
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    h_bin[idx[:int(n//2)]] = 1
    return h_bin

def swap_update(c, h, i, j, off, n):
    # Update correlation values given swapping h[i] with h[j] 
    # We keep track of changes in c due to swaps
    
    # Swap values
    h[i], h[j] = h[j], h[i]
    
    # Update c via simplified incremental update
    # Only necessary shifts are re-computed
    for p in range(n):  # flip the two changed points
        # For each p, we update c_{i-p} and c_{j-p} if relevant
        if i-p >= 0:
            k = i-p
            # check old contributions (due to i starting at 1, now at 0)
            # and new contributions (0,1 are removed, 1,0 added)
            delta = - h[k] * (1 - h[i])
            c[p + off] += delta
        if j-p >= 0:
            k = j-p
            delta = - h[k] * (1 - h[j])
            c[p + off] += delta
        
        # recompute after swap
        for p in [i, j]:
            # NB: here we need to recompute influence on all shifts
            # Replace the point-wisely 
            # But simpler: just run full corr again after swap
            return correlation(h)
    return c
        
def binary_search(h_start, n, budget_s, t0):
    h = h_start.copy()
    c = correlation(h)
    c5 = np.max(c) * 2.0 / n
    best_h = h.copy()
    best_c5 = c5
    
    # Index sets
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Data structure for faster lookup
    pos_ones = set(ones)
    pos_zeros = set(zeros)
    
    # Starting point: use the "target" in algorithm below
    max_shift = np.argmax(c)
    max_c = c[max_shift]
    
    # Algorithm: targeted swap
    # Pick a worst shift, identify alterations
    max_iter = 100000  # May be reduced if we are running short
    
    iter_check = 0  # To reduce time checks
    
    while time.time() - t0 < budget_s - 2.0 and max_iter > 0:
        max_shift = np.argmax(c)
        max_c = c[max_shift]
        
        # If improvement of only marginal, call it enough
        if max_c <= 1.0:
            break
            
        # Targeted swap improvements
        
        # Find potential swaps for the problematic shift
        # Determine which i,j pairs contribute the error
        i_set = set()
        j_set = set()
        off = n - 1  # offset for full correlation
        
        for i in ones:
            # h[i] = 1, expects h[i+s] = 0 to make c[s] positive
            # For shift max_shift, we check h[i + max_shift]
            if i + max_shift < n:
                if not h[i + max_shift]:
                    i_set.add(i)
        for j in zeros:
            # h[j] = 0, expects h[j + max_shift] = 1 to make c[s] positive
            # For shift max_shift, we check h[j + max_shift]
            if j + max_shift < n:
                if h[j + max_shift]:
                    j_set.add(j)
                    
        # If there's no valid pair, do random swap
        if not i_set or not j_set:
            # Try random pairs  
            i = np.random.choice(list(ones))
            j = np.random.choice(list(zeros))
        else:
            # Use move that improves the max_shift specifically
            i = np.random.choice(list(i_set))
            j = np.random.choice(list(j_set))
            
        # Perform the move
        h[i] = 0
        h[j] = 1
        # Update set structures
        pos_ones.remove(i)
        pos_ones.add(j)
        pos_zeros.remove(j)
        pos_zeros.add(i)
        
        # Track current state
        c = correlation(h) 
        current_c5 = np.max(c) * 2.0 / n
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        max_iter -= 1
        iter_check += 1
        if iter_check % 1000 == 0:
            if time.time() - t0 > budget_s - 2.0:
                break
                
    return best_h, best_c5
