def project_feasible(y, target_sum):
    # Clip to bounds
    y = np.clip(y, 0, 1)
    # Water filling algorithm
    total = np.sum(y)
    if abs(total - target_sum) < 1e-12:
        return y
    # Correct original here by not allowing truncation bias
    if total < target_sum:
        # Need to add more mass
        diff = target_sum - total
        # Sort by values decreasing
        sorted_indices = np.argsort(y)[::-1]
        for i in sorted_indices:
            if diff <= 0:
                break
            add = min(1 - y[i], diff)
            y[i] += add
            diff -= add
    else:
        # Need to remove mass
        diff = total - target_sum
        # Sort by values increasing (but prefer interior values first)
        sorted_indices = np.argsort(y)
        for i in sorted_indices:
            if diff <= 0:
                break
            sub = min(y[i], diff)
            y[i] -= sub
            diff -= sub
    return y

def compute_correlation(h):
    # compute correlation sum = sum_{i,j} h[i] * (1-h[i+j]) where i+j is valid index
    # equivalently: sum_{i,k} h[i] * (1 - h[i+k]), using a convolution
    n = len(h)
    if n == 0:
        return np.array([])
    # Following np.correlate approach, so
    # c[k] = sum_{i} h[i] * (1 - h[i-k])
    # But since i-k is treated, we slide in the right direction
    # Easier: compute correlate(h, (1-h), mode='full') where (1-h) is reversed
    # Because correlate(i,j) = \sum_{k} i[k] * j[i-k] for all valid k
    # So                  c[k] = sum_{i} h[i] * (1 - h[i-k]) = sum_{i} h[i] * (1 - h[i-k])
    # This amounts to correlate(h, 1 - h, mode='full')
    # Perform computation
    c = signal.correlate(h, 1 - h, mode='full')
    return c

def compute_smoothed_objective(c, alpha):
    cmax = np.max(c)
    exp_c = np.exp(alpha * (c - cmax))
    sum_exp = np.sum(exp_c)
    if sum_exp == 0:
        return cmax
    return (1 / alpha) * (log(sum_exp) + cmax)

def optimize_continuous(n, h0, t0, remaining_time_s):
    h = h0.copy()
    alpha = 5.0
    max_iter = 10000
    ada = 0
    min_delta = 1e-8
    best_h_local = h.copy()
    best_c5_local = float('inf')
    momentum_factor = 0.9
    step_size = 0.5 * (2.0 / n)
    prev_h = h.copy()
    prev_g = np.zeros_like(h)
    for iteration in range(max_iter):
        if time.time() - t0 > remaining_time_s - safety_margin:
            break
        c = compute_correlation(h)
        c5 = np.max(c) * (2.0 / n)

        if c5 < best_c5_local:
            best_c5_local = c5
            best_h_local = h.copy()

        # Smoothed objective
        alpha = min(alpha * 1.2, 200)
        if alpha >= 200:
            # Force max use
            smoothed_obj = np.max(c)
        else:
            # Use smooth surrogate
            cmax = np.max(c)
            exp_coeffs = np.exp(alpha * (c - cmax))
            exp_sum = np.sum(exp_coeffs)
            if exp_sum == 0:
                smoothed_obj = cmax
            else:
                smoothed_obj = (1 / alpha) * (np.log(exp_sum) + cmax)
        # Compute weights
        w = exp_coeffs / exp_sum
        g = compute_gradient(h, c, w)
        # Momentum term
        m = momentum_factor * g + (1 - momentum_factor) * prev_g
        h_new = h - step_size * m
        h_new = project_feasible(h_new, n / 2)
        # Accept or reject based on decreased surrogate
        c_new = compute_correlation(h_new)
        c5_new = np.max(c_new) * (2.0 / n)
        if alpha >= 200:
            smoothed_new = np.max(c_new)
        else:
            cmax_new = np.max(c_new)
            exp_coeffs_new = np.exp(alpha * (c_new - cmax_new))
            exp_sum_new = np.sum(exp_coeffs_new)
            if exp_sum_new == 0:
                smoothed_new = cmax_new
            else:
                smoothed_new = (1 / alpha) * (np.log(exp_sum_new) + cmax_new)
        if smoothed_new <= smoothed_obj + min_delta:
            h = h_new
            prev_g = m
            prev_h = h_new.copy()
            ada += 1
        else:
            step_size *= 0.5
            if step_size < 1e-10:
                step_size = 1e-6
        if ada > 500:
            alpha = 5.0
            ada = 0
        if np.linalg.norm(h - prev_h) < 1e-6:
            break
    return best_h_local, best_c5_local

def round_to_binary(h):
    n = len(h)
    m = n // 2
    # Sort indices by decreasing h values
    sorted_indices = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    for i in range(m):
        h_bin[sorted_indices[i]] = 1.0
    return h_bin

def delta_update_correlation(c, h, i, j, off):
    # Return dict: k -> change in c[k] when swap i,j (h[i]=1 h[j]=0) -> (h[i]=0, h[j]=1)
    # Let correlation be computed as c[shift] = sum_i h[i] * (1 - h[i+shift]),  i+shift < n
    # For index k = shift + (n -1)
    # For i, now equal 0: subtract contributions from i into jumps (-1)
    # For j, now equal 1: add contributions from j into jumps (+1)
    # form:
    # pd = { shift => h[t] * (1-h[t+shift]) } <- quantify one step for all is
    pd = {}
    # Current h[i] = 1, h[j] = 0
    # i -> 0      Old 1 - h[i+shift] for valid shifts
    # j -> 1      New 1 - h[j+shift] for valid shifts
    # So change per shift k is: h[i] * (1-h[i+k]) - h[j] * (1-h[j+k])
    # Since h[i] = 1, h[j] = 0:
    # change = (1 - h[i+k]) - 0 = 1 - h[i+k]  if valid
    # unchanged = 1 - h[j+k]   if valid
    # total delta = (1-h[i+k]) - (1-h[j+k]) = h[j+k] - h[i+k]

    delta = 0
    for ki in range(-len(h)+1, len(h)):
        k = ki + (len(h)-1)
        # For each k that you currently process, add delta
        j_k = j + ki  # previously h[j+k]
        i_k = i + ki  # previously h[i+k]

        # validity of both:
        check = 0
        if 0 <= j_k < len(h):
            check += 1
        if 0 <= i_k < len(h):
            check += 1

        if check < 2:
            # skip
            continue

        # Add contributions:
        if i_k in range(len(h)) and j_k in range(len(h)):
            delta = h[j_k] - h[i_k]
            if delta != 0:
                pd[k] = delta
    # pd now has all the changes to each shift value.
    return pd
