def evaluate(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral using convolution
    # We want max_k int h(x) * (1 - h(x+k)) dx
    # This equals max_k (int h(x)*1 dx - int h(x)*h(x+k) dx)
    # = max_k (int h(x) dx - int h(x)*h(x+k) dx)
    # = max_k (1 - int h(x)*h(x+k) dx)
    
    # Return max of all correlations where we want to minimize it
    correlation = signal.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    # C5 = 1 - min_correlation, where min_correlation comes from maximizing the overlap
    # In our formulation, max_correlation has the form of -int h(x)*h(x+k) dx
    # So if we subtract from 1, we get for our desired form:
    # C5 = 1 - min_correlation = max_k int h(x) * (1-h(x+k)) dx
    # But our actual negative correlation is over the full array
    # conv(h, 1-h) has length 2*n-1 with indices from -(n-1) to n-1
    # We want to look at indices from 0 to n-1 in the shifted arrays to capture the overlap
    # index n-1, n, ..., 2*n-2 (length n)
    # These represent k=0, -1, ..., -(n-1) (shifts relative to center)
    # The overlap integral corresponds to k = 0, 1, ..., n-1 from original shift convention
    # Correlation at index n-1 is k=0
    # Correlation at index n is k=1 ... up to index 2*n-2 which gives k=n-1
    # So we take indices from n-1 to 2*n-2 to capture shifts 0 to n-1
    actual_overlap = correlation[n-1:]  # corresponds to k=0 to k=n-1 with positive shift
    return np.max(actual_overlap) * dx

def project_onto_constraints(h):
    """Project h onto feasible set: [0,1] and int h(x) dx = 1"""
    n = len(h)
    dx = 2.0 / n
    integral = np.sum(h) * dx
    # Scale to unit integral
    if abs(integral) > 1e-10:
        h = h / integral * 1.0
    # Clip to bounds [0, 1]
    h = np.clip(h, 0, 1)
    return h

def generate_random_feasible(n_points):
    """Generate a random feasible h with integral = 1"""
    h = np.random.rand(n_points)
    return project_onto_constraints(h)

def make_move(h, n_points, temperature=1.0):
    """Make a move to perturb h"""
    h_new = h.copy()
    # Sample a move type:
    move_type = random.choice(["lerp", "swap", "delta"])
    if move_type == "lerp":
        # Linear interpolation between two points
        i1 = random.randint(0, n_points - 1)
        i2 = random.randint(0, n_points - 1)
        ratio = random.random()
        new_value = h[i1] * (1 - ratio) + h[i2] * ratio
        if not (0 <= new_value <= 1):
            new_value = max(0, min(1, new_value))
        h_new[i1] = new_value
        h_new[i2] = new_value
    elif move_type == "swap":
        # Swap two values
        i1 = random.randint(0, n_points - 1)
        i2 = random.randint(0, n_points - 1)
        h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
    else: # delta
        # Add small random value
        i = random.randint(0, n_points - 1)
        delta = random.gauss(0, temperature / 10)
        new_value = h_new[i] + delta
        h_new[i] = max(0, min(1, new_value))
    
    return project_onto_constraints(h_new)
