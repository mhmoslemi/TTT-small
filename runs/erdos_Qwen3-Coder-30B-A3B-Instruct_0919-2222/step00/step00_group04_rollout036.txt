import numpy as np
from scipy.optimize import differential_evolution
from scipy.fft import rfft, irfft
import time

def project_to_capped_simplex(v, target_sum):
    # Project vector v onto the capped simplex: 0 <= x <= 1, sum x = target_sum
    v = np.array(v)
    n = len(v)
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    # Find the threshold
    tmp_sum = 0.0
    threshold = 0.0
    for i in range(n):
        tmp_sum += sorted_v[i]
        if tmp_sum + (i + 1) * (1.0 - sorted_v[i]) >= target_sum:
            threshold = sorted_v[i] - (tmp_sum - target_sum) / (i + 1)
            break
    else:
        threshold = sorted_v[-1] - (tmp_sum - target_sum) / n
    # Apply threshold
    proj = np.maximum(0, np.minimum(1, v - threshold))
    # Scale to match target_sum
    scale = target_sum / np.sum(proj)
    proj *= scale
    return proj

def evaluate_overlap(h, dx):
    # Computes C5 = max_k ∫ h(x)(1-h(x+k)) dx
    # Using FFT for efficiency, correlate h with 1-h
    H = rfft(h)
    K = rfft(1.0 - h)
    corr = irfft(H * np.conj(K), n=2*len(h)-1)
    # The first (len(h)-1) elements correspond to negative lags,
    # the middle element to zero lag, and the last (len(h)-1) to positive lags.
    # We are interested in max over all lags
    # Note: Apply same scaling as the correlation calculation.
    # corr contains correlation values for lags from -(len(h)-1) to (len(h)-1)
    # max_k corresponding to max over positive and negative lags
    return np.max(corr)

def reparameterize_variables(u, v):
    # Transform u, v to widths and heights
    # widths = 2 * softmax(u) to ensure sum = 2
    w = 2 * np.exp(u) / np.sum(np.exp(u))
    # heights = sigmoid(v) to ensure [0,1]
    a = 1.0 / (1.0 + np.exp(-v))
    return w, a

def evaluate_block_fitting(a, w):
    # Evaluate fitness for block structure: maximize min correlation
    M = len(a)
    # Compute the autocorrelation G(k) for all shifts k
    # This requires O(M^2 * N) where N is number of shift samples
    # For a precise handling, we instead approximate with a fine grid
    # But instead let's just ensure a valid way to compute objective
    # Let's create block structure on a grid of size N = 200, then compute
    # max of f(k) = 1 - G(k) (which is the overlap measure we want to minimize)
    N = 200
    dx = 2.0 / N
    # Build step function from block sizes and heights
    h_grid = np.zeros(N)
    s = [0.0]
    for i in range(M):
        s.append(s[-1] + w[i])
    for i in range(N):
        # find which block i-th grid point belongs to
        # s starts with 0, s_0 = 0, s_1 = w_1, ..., s_M = 2
        for j in range(M):
            if s[j] <= i*dx and i*dx < s[j+1]:
                h_grid[i] = a[j]
                break
    # Evaluate overlap
    c5 = evaluate_overlap(h_grid, dx)
    # With integral penalty to ensure condition ∫ h = 1
    integral_check = np.sum(h_grid) * dx - 1.0
    penalty = 1000.0 * abs(integral_check)
    return c5 + penalty

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    M = 12  # Number of blocks
    N = 200  # Number of fine grid points
    dx = 2.0 / N
    lambda_penalty = 1000.0  # Penalty weight for integral constraint

    # Define bounds for u and v parameters
    bounds = [(-5, 5)] * (2 * M)

    # Objective function for differential evolution
    def objective(z):
        u = z[:M]
        v = z[M:]
        w, a = reparameterize_variables(u, v)
        # integral check
        integral = np.dot(a, w)
        integral_penalty = lambda_penalty * abs(integral - 1.0)
        # Evaluate using the smooth approach for now but not exactly correct
        # but DE uses only multimodal to give starting point, so it's okay.
        # For accurate measure, we should construct h and call evaluate_overlap
        
        # But let's separate to method for DE and post-optimize:
        # Before using evaluate_block_fitting, we calculate what happens when we have
        # h on the full grid — construct full grid function and evaluate overlap
        # Use a dedicated function to do this properly.
        # However, for now, compute as best proxy for objective:
        # compute overlap on full grid using reparameterized a, w
        h_full = np.zeros(N)
        points = np.cumsum([0] + w.tolist())
        for i in range(N):
            xi = i * dx
            for j in range(M):
                if points[j] <= xi < points[j+1]:
                    h_full[i] = a[j]
                    break
        
        # Short circuit for DE objective since we want better starting point 
        # let's validate correctness using block construction + correlation
        # Evaluate using complete process
        corr = evaluate_overlap(h_full, dx)
        # Maximize minimal correlation, so minimize 1 - min_corr i.e. maximize min_corr
        # We actually minimize max of (1 - G(k)) over all shifts k 
        # i.e. minimize max_k (1 - G(k))
        # which is same as maximize min_k G(k) => minimize max_k (1 - G(k))
        return corr + integral_penalty

    best_obj = np.inf
    best_h = None

    # Phase 1: Differential Evolution
    print("Starting DE optimization...")
    de_start = time.time()
    try:
        res = differential_evolution(objective, bounds, seed=seed,
                                     maxiter=50,
                                     popsize=15 * (2*M), 
                                     workers=1,
                                     disp=False)
        print("DE completed. Best DE objective:", res.fun)
    except Exception as e:
        print("DE failed:", e)
        pass
    
    # For use without De solution, let's set up a fallback initialization nearby
    u0 = np.random.normal(0, 0.5, M)
    v0 = np.random.normal(0, 0.5, M)
    init_params = np.concatenate([u0, v0])
    
    final_u = u0.copy()
    final_v = v0.copy()
    
    # If DE succeeded, use its result
    if 'res' in locals() and res.success:
        final_u = res.x[:M]
        final_v = res.x[M:]
    
    # Phase 2: Refinement with projected gradient descent
    print(f"Starting gradient descent refinement")
    # Rebuild h from current parameters
    w, a = reparameterize_variables(final_u, final_v)
    integral = np.dot(a, w)
    if abs(integral - 1.0) > 1e-10:
        # Rescale heights
        a = a / integral
        # Sanity check
        if np.max(a) > 1.0:
            a = np.clip(a, 0.0, 1.0)
            # recompute integral
            integral = np.dot(a, w)
            # recompute a to make integral = target
            if abs(integral - 1.0) > 1e-10:
                a = a * (1.0 / integral)
                a = np.clip(a, 0.0, 1.0)
    
    # Build the step function from block info
    h_prev = np.zeros(N)
    s = [0.0]
    for i in range(M):
        s.append(s[-1] + w[i])
    for i in range(N):
        xi = i * dx
        for j in range(M):
            if s[j] <= xi < s[j+1]:
                h_prev[i] = a[j]
                break
    
    # Normalize sum to N / 2
    sum_h = np.sum(h_prev)
    h_prev = h_prev * (N / 2) / sum_h
    
    # Now refine with projected subgradient descent
    iter_max = 8000
    alpha0 = 0.2
    best_h = h_prev.copy()
    best_c5 = evaluate_overlap(h_prev, dx)
    
    for t in range(iter_max):
        if time.time() - start_time > budget_s - 5:
                break

        # evaluate correlation matrix and find max shift
        corr = evaluate_overlap(h_prev, dx)
        # The max_k of (1 - G(k)) => which we are minimizing
        # Hence corr is G_opt = max_k G(k) for all shifts, here we just
        # require minimization => minimize (1 - max_k G(k)) 
        # i.e., maximize min_k G(k)
        # so objective = corr only.

        # Gradient approach - more robust constructor here
        # Let's use known formula
        # let's recomputed h directly, based on the auto-correlation
        # since we have the explicit method via np.correlate
        # c5 = max_k (1 - ∫ h(x) h(x+k) dx) 
        # But it's inaccurate with finite grid.
        
        # Let’s recompute c5 iterate by broadcasting cause we're now 
        # evaluating exact correlation (FFT based): 
        C = np.correlate(h_prev, 1 - h_prev, mode='full') 
        # The first (len(h)-1) elements are negative lags
        # middle element 0
        # last (len(h)-1) elements positive lags
        c5_val = np.max(C)

        # Store best found so far
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_prev.copy()

        # Compute gradient: this is sample wise gradient of objective
        # which is max_k C_k over shifts k, where C_k = sum h_i (1 - h_{i + k})
        # Gradient of C_k w.r.t. h_i = (1 - h_{i + k}) if i < (N-k),  
        #                         = - h_{i - k} if i >= k,  
        #                        = 0 otherwise
        # So we construct this efficiently using shifts. If we want the sub-gradient
        # for max function, we choose average of all gradients at active points.
        # But for subsampling clarity, just take w.r.t. to largest contribution.
        # Find largest contributing k
        max_idx = np.argmax(C)
        shift_k = max_idx - (N - 1)  # Uncentering
        
        # Take sub-gradient for shift_k
        g = np.zeros(N)
        # Forward indices: (for padding, low indices)
        for i in range(N - abs(shift_k)):
            if shift_k > 0:
                g[i] += (1 - h_prev[i + shift_k])
            else:
                g[i] -= h_prev[i - shift_k]
        # apply steps
        alpha = alpha0 / np.sqrt(t + 1)
        h_new = h_prev - alpha * g
        # Project
        h_new = project_to_capped_simplex(h_new, N / 2)
        h_prev = h_new.copy()
    
    # Final evaluation for return
    final_c5 = evaluate_overlap(best_h, dx)
    final_scaled_c5 = final_c5 * (2.0 / N)

    return (best_h.tolist(), final_scaled_c5, N)
