import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute the full convolution of h with (1-h)
    conv = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # The convolution result has length 2*n - 1
    # The maximum value gives us the C5 constant
    c5 = np.max(conv) * dx
    return c5

def initial_h_values():
    """Generate initial step function."""
    n = 47
    h = np.zeros(n)
    # Use piecewise constant construction that's known to work well
    # Split into 3 pieces: first half is 1, second half is 0, middle is linear
    # Actually use a better known construction
    h[0:16] = 1.0  # First 16 points = 1
    h[16:31] = 0.0  # Next 15 points = 0
    h[31:] = 1.0  # Last 16 points = 1
    # But normalize so integral is 1
    h = h * (2.0 / np.sum(h))  # scale to make integral = 1
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Get initial construction
    h_start = initial_h_values()
    n_points = len(h_start)
    dx = 2.0 / n_points
    
    # Need sum(h) * dx = 1, so sum(h) = n_points / 2
    target_sum = n_points / 2
    
    def objective(h):
        # Rescale h to make sum = target_sum
        h_scaled = h * target_sum / np.sum(h)
        # Compute the correlation
        corr = np.correlate(h_scaled, 1 - h_scaled, mode="full")
        max_corr = np.max(corr) * dx
        return max_corr
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add constraint object
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Try to do better using a few local search strategies
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Let's try a few different starting points and local optimizations
    # Strategy 1: Gradient-based optimization
    try:
        # First, try to refine with scipy.optimize.minimize
        result = minimize(objective, h_start, method='SLSQP', bounds=bounds, constraints=constraints, 
                          options={'maxiter': 100, 'ftol': 1e-6})
        if result.success:
            h_new = result.x
            if np.sum(h_new) > 0:
                h_new = h_new * target_sum / np.sum(h_new)  # re-normalize
                c5_new = evaluate_erdos_solution(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
    except Exception as e:
        pass  # Continue, we already have a valid start
    
    # Strategy 2: Small random perturbations around best solution
    # This is a simple local search loop
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Max attempts to best this point
    max_attempts = 10000
    attempts = 0
    last_improvement = 0
    
    while attempts < max_attempts and time.time() - start_time < budget_s - 1:
        attempts += 1
        # Make a small change to current_h
        h_new = current_h.copy()
        # Make one small change
        i = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        h_new[i] += delta
        # Clip to [0, 1] 
        h_new[i] = max(0, min(1, h_new[i]))
        
        # Re-normalize so the sum constraint is satisfied
        h_new = h_new * target_sum / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept if better or accept with some probability (simulated annealing)
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                last_improvement = attempts
        
        # Occasionally restart from the current best
        if attempts % 100 == 0 and attempts > last_improvement + 500:
            current_h = best_h.copy()
    
    # Return best solution found
    return (best_h, best_c5, n_points)
