def evaluate_erdos_solution(h):
    """Evaluates the C5 bound for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations for all shifts
    c = np.correlate(h, 1 - h, mode="full")
    # Maximum value across all shifts
    max_corr = np.max(c)
    return max_corr * dx

def compute_correlation(h):
    """Compute full correlation vector c_k"""
    n = len(h)
    return np.correlate(h, 1 - h, mode="full")

def smoothed_objective_and_weights(c, alpha):
    """Compute smooth approximation and weights"""
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    Z = np.sum(exp_c)
    w = exp_c / Z
    smoothed = c_max + np.log(Z) / alpha
    return smoothed, w

def grad_h(h, weights, n):
    """Compute gradient vector for the smoothed objective"""
    # Weights are precomputed and sum to 1
    # w_s = exp(alpha * c_s) / Z
    # ∂ c_s / ∂ h_i = (1 - h_{i+s}) * I[i+s<n] - h_{i-s} * I[i-s>=0]
    
    # For finite size support computation, represented as cross-correlations:
    # Let p be a vector of ones: p = [1, 1, ..., 1] (length n)
    # We need: g_i = sum_s w_s * (1 - h_{i+s}) * I[i+s<n] - h_{i-s} * I[i-s>=0]
    # This can be viewed as a weighted convolution:
    # Forward part: w_rev * (1-h)  (reversed weights)
    # Backward part: w * h  (weights in original order)
    
    # To avoid explicit loop, it's faster/finer to use FFT convolution
    # But for now we'll compute manually with proper chunking
    
    # Forward component
    one_minus_h = 1 - h
    # For each index i, compute ∑s w_s · (1 - h[i+s]) for valid s >= 0
    # We can think of computing: np.convolve(w_rev, 1-h) but need to truncate
    # Let's do loops taking care of indices:
    g = np.zeros_like(h)
    
    for i in range(n):
        # Backward coupled terms: 
        # w_s * h_{i-s} for s > 0 (i.e. i-s < i -> i-s >= 0)
        # handled below for i-s >= 0
        # Forward coupled terms:
        # w_s * (1 - h[i+s])  for s >= 0
        # i+s needs to be < n
        for s in range(n):
            if i + s < n:
                g[i] += weights[s] * one_minus_h[i + s]
            else:
                break
                
        # for backward terms:
        # we want h_{i-s} for s > 0 and i-s >= 0, i.e. s <= i
        for s in range(1, i+1):
            # s from 1 to i
            # i-s >= 0 => s <= i
            g[i] -= weights[s] * h[i - s]
    return g

def project_onto_feasible(y, target_sum):
    """Project y onto [0,1]^n with sum = target_sum. Use water-filling."""
    n = len(y)
    # Clip to [0,1]
    y_clipped = np.clip(y, 0., 1.)
    total = np.sum(y_clipped)
    delta = total - target_sum
    if abs(delta) < 1e-12:
        return y_clipped.copy()
    # Find free indices: those in (0,1)
    free = np.where((y_clipped > 0) & (y_clipped < 1))[0]
    if len(free) == 0:
        # all are either 0 or 1
        y_out = y_clipped.copy()
        y_out[0] += delta
        return y_out
    # Ensure freeness in relatively stable way by resorting
    # But just sort indices by values descending, and adjust in order
    # Apply delta equally to free indices if possible
    if delta > 0:
        # Reduce values in free indices
        free_sorted_idx = np.argsort(y_clipped[free])[::-1]  # decreasing
        free_idx_remaining = free[free_sorted_idx]
        reduced = 0.0
        while reduced < delta and len(free_idx_remaining) > 0:
            # remove zero or one element from free_idx_remaining and make it pay for delta
            if len(free_idx_remaining) == 0:
                break
            idx = free_idx_remaining[0]
            available = y_clipped[idx]  # amount that can be reduced from this index
            if available <= delta - reduced:
                reduced += available
                y_clipped[idx] = 0.
                free_idx_remaining = free_idx_remaining[1:]
                if abs(reduced - delta) < 1e-10:
                    break
            else:
                # can subtract only some. We're done after this
                y_clipped[idx] -= (delta - reduced)
                reduced = delta
                break
    elif delta < 0:
        # Increase values only in free indices
        free_sorted_idx = np.argsort(y_clipped[free])  # ascending
        free_idx_remaining = free[free_sorted_idx]
        added = 0.0
        while added < -delta and len(free_idx_remaining) > 0:
            idx = free_idx_remaining[0]
            available = 1. - y_clipped[idx]  # amount that can be increased
            if available <= -delta - added:
                added += available
                y_clipped[idx] = 1.
                free_idx_remaining = free_idx_remaining[1:]
                if abs(added + delta) < 1e-10:
                    break
            else:
                y_clipped[idx] += (-delta - added)
                added = -delta
                break
    return y_clipped

def swap_update(c, h, i, j, n, off=0):
    """In-place update of correlation vector c after swapping two elements h[i] and h[j]"""
    # Change affects indices that are shifted positions of i,j
    # Curr policy: accidental duplication, will recompute precise c later
    # But let's compute fully
    # For every position p (0..n-1) and shift s:
    # c[s + off] += (1 - h[p]) * delta  if p = i
    # c[s + off] -= h[p] * delta         if p = j 
    # delta = 1 where j switched from 0 to 1 and i from 1 to 0
    delta = 1 if h[i] == 1 and h[j] == 0 else -1
    if delta > 0:  # i to 0 (was 1), j to 1 (was 0)
        # at shift s:
        for k in range(n):
            shift = k - i
            if i + shift >= 0 and i + shift < n:
                c[shift + off] -= (1 - h[k]) * 1
            shift = j - k
            if j - shift >= 0 and j - shift < n:
                c[shift + off] += h[k] * 1
    else:  # i to 1 (was 0), j to 0 (was 1)
        for k in range(n):
            shift = k - i
            if i + shift >= 0 and i + shift < n:
                c[shift + off] += (1 - h[k]) * 1
            shift = j - k
            if j - shift >= 0 and j - shift < n:
                c[shift + off] -= h[k] * 1
    # Flip h[i] and h[j]
    h[i], h[j] = h[j], h[i]

def targeted_swap(h, c, n, off=0):
    """Perform an efficient one-step swap that reduces the worst shift by 1"""
    # Find worst shift idx
    max_idx = np.argmax(c)
    s = max_idx - off
    # Find indices that contribute to c[s]
    A = []
    B = []
    for i in range(n):
        if h[i] == 1:
            j = i + s
            if 0 <= j < n and h[j] == 0:
                A.append(i)
        elif h[i] == 0:
            j = i + s
            if 0 <= j < n and h[j] == 1:
                B.append(i)
    if A and B:
        # Pick a random pair (this will reduce c[s] by 1)
        import random
        i = random.choice(A)
        j = random.choice(B)
        # Update c
        # We know the direct formula for these swaps
        delta = 1
        # Reconstruct d(individual terms affected by swap):
        # Original contributions:
        # A[i]: 1-h[j+s] ... 
        # B[j]: h[j+s] ...
        # After swap:
        # A[i]: 0-h[j+s] ...
        # B[j]: 1-h[j+s] ...
        # Delta = -1 for what is removed from A-surface, +1 for B-surface
        # loop across all shifts t to roughly simulate what c[t] will be:
        # Instead run one swap update that is efficient: only update c directly
        # 1. compute the new c from swapping
        delta = 0  # Actually the difference in the effect on max_shift
        # We know that both contribute and we're just toggling them
        # Original c[s] is increased by 1 because one prev zero gives 1 (1-0=1), and sim 
        # Through triangle counting (think sliding window, safer way to compute):
        # Since we are only changing two values, expand back by a complete transposition step
        # Process change (assume it's simultaneous update)
        # Could more carefully compute many terms one by one but performance cost vs tightness...
        # Just modify c from indices around the swap location, process directly at an index one:
        # Since it's a specific optimization instead of generic shuffle:
        # systematic way to compute the change in direct shift values
        # Or if allowed to cheat, just optimize and rerun solve!
        # [simplified.module]
        # More accurate updating on ex aequo scenario:
        # Typical observation is both change contribs skip early and never meet.
        # Name current contributions to c[s] which are 1 here. All we care:
        old_val = c[max_idx]
        temp_h = h.copy()
        temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
        new_c = compute_correlation(temp_h)
        # Fixed correct model
        return i, j, new_c
    return -1, -1, c

def local_binary_search(h, c, max_corr, remaining_budget):
    """Refine binary solution via swapping."""
    n = len(h)
    off = n - 1
    prev_max = max_corr
    best_h = h.copy()
    best_c = c.copy()
    best_corr = max_corr
    idx_ones = np.where(h == 1)[0]
    idx_zeros = np.where(h == 0)[0]
    # Only do approximate local search to save time
    iter_count = 0
    # MAX_ITER_PER_PASS = 200000 // n
    MAX_ITER_PER_PASS = 30000 // n  #scale down iter per Achtung
    T_THRESHOLD = 1.0
    t0 = time.time()
    while iter_count < MAX_ITER_PER_PASS and time.time() - t0 < remaining_budget * 0.5:
        # Try to improve a worsened or random swap (deterministic pass)
        picked_i = idx_ones[np.random.randint(len(idx_ones))]
        picked_j = idx_zeros[np.random.randint(len(idx_zeros))]
        # Re-compute c _after_ swap refinement
        temp_h = h.copy()
        temp_h[picked_i], temp_h[picked_j] = temp_h[picked_j], temp_h[picked_i]
        temp_c = compute_correlation(temp_h)
        new_max = np.max(temp_c)
        # Accept only if reduced
        if new_max < prev_max:
            # accept
            h = temp_h
            c = temp_c
            prev_max = new_max
            if prev_max < best_corr:
                best_h = h.copy()
                best_c = c.copy()
                best_corr = prev_max
        iter_count += 1
    return best_h, best_c, best_corr

def init_h(n, seed):
    "Initial point via randomized feasible projection"
    np.random.seed(seed)
    h = np.random.uniform(0, 1, n)
    h = project_onto_feasible(h, n / 2.0)
    return h

def upscale_h(h_coarse):
    """Linear upscale of h array to double size"""
    n_coarse = len(h_coarse)
    n_fine = n_coarse * 2
    h_fine = np.zeros(n_fine)
    for i in range(n_coarse):
        h_fine[2*i] = h_coarse[i]
        h_fine[2*i+1] = h_coarse[i]
    return h_fine

def optimize_n(h0, n, t0, budget_total):
    """Run continuous optimization on a coarse level."""
    dx = 2.0 / n
    fail_safe_percent = 0.98
    # limit iteration based on size and time if needed
    iterations = min(10000, int(2e4 * (200 / n)))
    if iterations < 500:
        iterations = 500
    # Tuning
    alpha = 5.0
    alpha_mult = 1.15
    max_alpha = 200.0
    eta = 0.5 * dx
    beta = 0.85
    learn_gamma = 0.92
    strong_stop_alpha = max_alpha * 0.98
    best_h = h0.copy()
    best_val = float('inf')
    h_prev = h0.copy()
    # project h0 onto feasible sets (hypothetical)
    h = h0.copy()
    # Project h0 to meet constraints
    h = project_onto_feasible(h, n/2.0)

    for step in range(iterations):
        t = time.time() - t0
        if t > budget_total * fail_safe_percent:
            break
            
        # Compute correlation
        c = compute_correlation(h)
        # Smooth objective (LSE)
        smoothed_val, weights = smoothed_objective_and_weights(c, alpha)
        # Compute gradient
        g = grad_h(h, weights, n)
        
        # Update with momentum
        h_try = h - eta * (g + beta * (h - h_prev))
        
        # Project onto feasible set
        h_prox = project_onto_feasible(h_try, n / 2.0)
        # Trial try
        c_try = compute_correlation(h_prox)
        smoothed_try, _ = smoothed_objective_and_weights(c_try, alpha)
        
        # Accept if decrease or backtracking line search
        if smoothed_try <= smoothed_val:
            h = h_prox
            h_prev = h_try
            if smoothed_try < best_val:
                best_val = smoothed_try
                best_h = h.copy()
        else:
            # Try second order line search
            tried = 0
            while tried < 5:
                eta *= 0.5
                h_try = h - eta * (g + beta * (h - h_prev))
                h_try = project_onto_feasible(h_try, n / 2.0)
                c_try = compute_correlation(h_try)
                smoothed_try, _ = smoothed_objective_and_weights(c_try, alpha)
                if smoothed_try <= smoothed_val:
                    h = h_try
                    h_prev = h_try
                    # TODO SURE how to recover gradient? Only one more improvement step
                    break
                tried += 1
            if tried == 5:
                h = h_prev
        
        # If gap small in target, do it 'worse may help' trials
        if alpha < strong_stop_alpha and ( smoothed_val - best_val ) < 1e-6:
            alpha *= alpha_mult
        else:
            if step % 10 == 0:
                alpha = max(alpha, 5.0)  # Not rewarding noiseless loop hike
    return best_h, best_val

def binary_from_cont(h):
    """Round continuous h to a binary step function"""
    n = len(h)
    h_sorted = sorted(enumerate(h), key=lambda x: x[1], reverse=True)
    h_bin = np.zeros(n)
    # lump numbers of largest indices
    for i in range(n//2):
        idx, val = h_sorted[i]
        h_bin[idx] = 1
    return h_bin

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    ns = [80, 120, 180]  # letting total web budget be bigger than current solution
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    for n in ns:
        relay_time = time.time() - t0
        remaining = budget_s - relay_time - 3.0  # padding for cross-stage calcs
        if remaining < 10.0:
            break
        # Initial start with scaled
        h = init_h(n, seed + n)
        h_opt, _ = optimize_n(h, n, t0, remaining)
        h_bin = binary_from_cont(h_opt)

        # Refinement - Polish binary solution
        c = compute_correlation(h_bin)
        corr_max = np.max(c)
        h_best, c_best, corr_best = local_binary_search(h_bin, c, corr_max, remaining)
        
        if corr_best < best_c5:
            best_h = h_best.copy()
            best_c5 = corr_best
            best_n = n
            
        # Early termination
        if best_c5 <= 0.3790:
            break
            
    # Final return
    if best_h is None:
        # fallback to known solution
        h = np.array(initial_h_values)
        h = project_onto_feasible(h, len(h) / 2.0)
        best_h = h
        best_c5 = evaluate_erdos_solution(h)
        best_n = len(h)
    
    return best_h.tolist(), best_c5, best_n
