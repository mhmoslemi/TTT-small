def evaluate_C5(h):
    """Evaluate C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part (k from -1 to 1)
    # The full correlation is of length 2*n-1, we want the middle part
    mid_idx = n - 1
    # We are interested in k from 0 to 1, which corresponds to lags from 0 to n-1
    # But we want max of all k from 0 to 1
    # The overlap for k is at lag = k/dx 
    # So for integer positions, k is in range [0, 1] 
    # Let's compute all valid overlaps 
    overlaps = []
    for k in range(n):
        # For k = 0, 1, 2, ..., n-1, does k/dx = k*2/n 
        if k < n:
            # Corresponding lag is k (in sample steps)
            # This is the overlap for shift k/dx = k*2/n along [0,2]
            if k <= n-1:
                overlaps.append(corr[mid_idx + k] * dx)
    # but we're more interested in k going from 0 to 1 explicitly
    # and the condition is for any k in [0,1], we compute the continuous integral
    # So a better approach for continuous integral
    # Evaluate all positive shifts from 0 to 1 to cover the range
    ranges = []
    for i in range(n):
        # Just evaluate the correlation
        # The first term of full correl is for 0-shift
        # The peak value over k from [0,1] is according to our constraint on integral
        # Actually, let's recompute carefully
        # Given shift k (as continuous value in [0,1]), we want continuous integral
        # That uses convolution (as done in FFT form or similarly)  
        # But we can approximate it by: integral_h(x) * (1 - h(x+k))  
        # So we can estimate for discrete version:
        # For each x from 0 to 2-dx, think shift k = x/dx if we're at sample i
        # Let's go the direct route
        # We want max_k \int h(x) (1 - h(x+k)) dx, for k from 0 to 1
        # We compute for discrete case: each shift k = i * dx for i from 0 to n-1
        correlations = []
        for i in range(n):
            # Compute shift by i samples (i * dx) 
            # So we compute the overlap integral for this shift
            # We compute integral_h(x) * (1 - h(x + i*dx)) 
            # But we have discrete samples:
            # integral is approx sum_h(i) * (1 - h[i+k]) for shift k
            if i < len(h):
                c = corr[mid_idx + i] * dx
                correlations.append(c)
        if correlations:
            ranges.append(max(correlations))
    return max(ranges) if ranges else 0.0

def track_evaluate(h):
    """Evaluate C5 using full convolution style approximation."""
    n = len(h)
    dx = 2.0 / n
    # full correlation: h conv (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract correlations for shifting to get all k in [0,1]  
    max_val = 0.0
    # For each shift k from 0 to 1 
    # This corresponds to shifting by i from 0 to n-1, an integer number 
    # of sample positions (spacing dx)
    for shift in range(n):
        # Central overlap: i shifts, corresponds to k = i * dx
        if shift < len(corr):
            val = corr[n - 1 + shift] * dx
            # Shift includes both ways in [0,1] yields more comprehensive coverage 
            max_val = max(max_val, val)
    return max_val

def compute_C5(h):
    """Computation of C5 based on discrete evaluations."""
    n = len(h)
    dx = 2.0 / n 
    # Defining atkinson states we use x and x+k means discrete x position
    # but let's properly compute the full correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The full correlation has length 2*n-1
    # Compute positive overlaps by actual integration estimate:
    # For each shift, integrate by sampling
    overlaps = []
    for i in range(2*n - 1):
        # shift index i corresponds to:
        if i >= n-1:  
            # actual shift = d_shift = i - (n-1) samples 
            d_shift = i - (n-1) 
            # shift = d_shift * dx = d_shift * 2/n  
            # and now compute the integral (can be negative since it's h(x)(1-h(x+k)) )
            # but we want the worst case, so it's: \int h(x)(1-h(x+k)) dx  
            # We can compute it via reverting the constraint:
            # Since only positive contribution matters for overlap,  
            # the worst case can be seen in the correlation.  
            if i < 2*n - 1:
                overlaps.append(corr[i] * dx)
    return max(overlaps) if overlaps else 0.0
