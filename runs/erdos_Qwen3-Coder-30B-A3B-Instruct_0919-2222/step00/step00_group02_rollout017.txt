import numpy as np

def evaluate_h(h, dx):
    n = len(h)
    # compute all overlaps O_k(h)
    # O_k = sum h_i * (1 - h_{i+k}) for all valid i
    # which is (sum h_i) - sum h_i * h_{i+k} = 0.5*n - sum h_i * h_{i+k}
    # but we compute the correlation sum and do the calculation on the full convolution
    # To avoid confusion and match the problem definition, we compute
    # overlap_k = int h(x) * (1 - h(x+k)) dx
    # with a circular convolution or using np.correlate
    
    overlap = np.correlate(h, 1 - h, mode='full')
    # we only need the middle part where k = 0 to 2*n-2 but we take the max over all (-n+1, ..., n-1)
    # more accurately, for subset:
    # This gives us the full correlation [C_{-n+1}, ..., C_{n-1}] with size 2*n-1
    return np.max(overlap) * dx

def McCormick_hull(h_i, h_j):
    # returns [w_lower, w_upper] such that w >= h_i * h_j and w <= h_i * h_j in [0,1]^2
    # Given h_i, h_j in [0,1], w = h_i * h_j has bounds:
    #    w >= max(0, h_i + h_j - 1)
    #    w <= min(h_i, h_j)
    w_lower = max(0, h_i + h_j - 1)
    w_upper = min(h_i, h_j)
    return w_lower, w_upper

def solve_relaxed_LP(h0, K, n, dx):
    # h: n variables
    # w_{i,k}: auxiliary vars, defined for all i,k with i+k in [0,n-1], and k in K
    # t: scalar

    # Maybe better to solve the max over all shifts vs having K subset for initial solve?
    # For now, stay close to initial idea
    # Let's define auxiliary variables w_{i,k}
    # Search over K = all shifts such that i+k in [0,n-1] for valid i

    # Build mapping from i,k pairs to column index
    # w start from 0 to len(K)*n

    variables = []
    idx_w = {}
    w = []
    for i in range(n):
        for k in K:
            if 0 <= i+k < n:
                idx_w[(i,k)] = len(w)
                w.append(cp.Variable())

    # variable list for h
    h = cp.Variable(n)

    # variable t
    t = cp.Variable()

    # constraints
    constraints = []

    # sum constraint
    constraints.append(cp.sum(h) == n/2)

    # box constraints
    for i in range(n):
        constraints.append(0 <= h[i])
        constraints.append(h[i] <= 1)

    # McCormick envelope
    for i in range(n):
        for k in K:
            if 0 <= i+k < n:
                j = i+k
                w_var = w[idx_w[(i,k)]]
                constraints.append(w_var >= h[i] + h[j] - 1)
                constraints.append(w_var >= 0)
                constraints.append(w_var <= h[i])
                constraints.append(w_var <= h[j])

    # overlap (linearized)
    for k in K:
        # build sum_{i} h_i * (1 - h_{i+k}) = sum h_i - sum h_i h_{i+k}
        # = 0.5*n - sum_{i, i+k in [0,n-1]} h_i h_{i+k}
        # for each k, sum over i such that i+k in [0,n-1]
        # which is i in [max(0, -k), min(n-1, n-1-k)]  = [max(0, -k), min(n-1, n-1-k)]
        # So we compute this sum and write: sum h_i - sum_{i: 0 <= i+k < n} h_i h_{i+k} <= t
        # i in [max(0, -k), min(n-1, n-1-k)]
        i_start = max(0, -k)
        i_stop = min(n-1, n-1-k) + 1
        overlap_k = 0
        for i in range(i_start, i_stop):
            j = i + k
            overlap_k += h[i] - w[idx_w[(i,k)]]
        constraints.append(overlap_k <= t)

    # Make objective
    objective = cp.Minimize(t)

    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return h.value, t.value, True
    except:
        pass
    return None, None, False

def swap_rounding(hfrac, n):
    # Given fractional h vector with sum=0.5*n, turn into binary vector
    # using a swap rounding method that is dependent
    # This will result in sum(h)==0.5*n (exact)
    hbin = hfrac.copy()
    
    # Normal rounding to allow easy tracking
    loot = 0  # will be - mean leftover
    # make all integral to match the sum constraint exactly
    # try to make it homogeneous through swapping
    # Pairwise dependent rounding
    
    # we have to maintain integral sum
    hbin = np.round(hbin).astype(float)
    diff = np.sum(hbin) - n/2  # we want diff = 0
    while abs(diff) > 1e-9:
        # Find first i with hbin[i] < hfrac[i] (not integral)
        # and first j with hbin[j] > hfrac[j] (not integral)
        i = -1
        j = -1
        for k in range(n):
            if hfrac[k] == hbin[k]: continue
            if hfrac[k] > hbin[k]:
                i = k
                break
        for k in range(n):
            if hfrac[k] == hbin[k]: continue
            if hfrac[k] < hbin[k]:
                j = k
                break
        if i == -1 or j == -1:
            # try a different approach - difference is small
            # broadcast to closest integral
            hbin = np.round(hfrac).astype(float)
            # restore sum by adjustments
            diff = np.sum(hbin) - n/2
            if diff == 0:
                break
            # make changes until balance is restored
            if diff > 0:
                for k in range(n):
                    if hbin[k] > hfrac[k]:
                        hbin[k] -= diff
                        break
            else:
                for k in range(n):
                    if hbin[k] < hfrac[k]:
                        hbin[k] -= diff
                        break
            break
        # they both active
        # Since hbin is closest to hfrac
        # Now swap Rounding
        delta = min(hfrac[i]-hbin[i], hbin[j]-hfrac[j])
        
        # with 50% chance swap to increase hbin[i], decrease hbin[j] or vice versa
        if np.random.rand() < 0.5:
            hbin[i] += delta
            hbin[j] -= delta
        else:
            hbin[i] -= delta
            hbin[j] += delta
            
        diff = np.sum(hbin) - n/2
    
    # Ensure that the sum is n/2 (within floating point error)
    diff = np.sum(hbin) - n/2
    if abs(diff) > 1e-9:
        # just adjust one component
        for k in range(n):
            if 0 < hbin[k] < 1:
                hbin[k] -= diff
                break
    
    # clip everything back to [0,1]
    hbin = np.clip(hbin, 0, 1)
    return hbin

def column_generation(n, dx, max_iter=100, eps=1e-6):
    K = set([0, -1, 1, -2, 2])
    h = np.array([0.5] * n)  # Start with uniform
    best_t = np.inf

    # Warm start K with known good shifts
    for i in range(n):
        for k in [-2,-1,0,1,2]:
            if 0 <= i+k < n:
                K.add(k)
    
    K_list = list(K)
    iter_count = 0
    while iter_count < max_iter:
        h0 = h.copy()
        # Solve relaxed LP with current K_list
        h_sol, t_val, success = solve_relaxed_LP(h0, K_list, n, dx)
        if not success:
            break
        # Compute overlaps for ALL shifts to find worst shift
        # But we can also restrict it to a reasonable subset
        h_sol = h_sol.clip(0, 1)
        # compute full correlation for testing candidate shifts
        num_shifts = 2 * n - 1
        all_overlaps = np.correlate(h_sol, 1 - h_sol, mode='full')
        # the valid k values are [-n+1, ..., n-1]
        max_k = np.argmax(all_overlaps)
        # k from max_k = [0:n-1, n, 2*n-2]
        # add 1 diff to get actual k
        k_worst = max_k - (n - 1)
        # Also ensure consistency of k: should be in [-n+1,n-1]
        overlap_worst = all_overlaps[max_k]

        if overlap_worst <= t_val + eps:
            # Found optimum within tolerance, stop
            h = h_sol
            break
        
        # Add worst shift
        K.add(k_worst)
        K_list = list(K)
        iter_count += 1
        if iter_count >= 100:
            break
    # end loop
    h = h.clip(0, 1)
    return h, t_val

def sturmian_construction(n, max_iter=100):
    # Generate sturmian sequence a_i(alpha, tau) where a_i = 1 if {i*alpha + tau} < 0.5
    # For simplicity, split search via coarse grid in (alpha, tau) space.
    # Use golden-section search near best coarse grid.

    alpha_min = 0.2
    alpha_max = 0.8
    # rough grid of coarse search
    num_alpha = 200
    num_tau = 100
    alphas = np.linspace(alpha_min, alpha_max, num_alpha)
    taus = np.linspace(0, 1, num_tau, endpoint=False)
    
    best_bound = np.inf
    best_alpha = 0.5
    best_tau = 0.0

    for alpha in alphas[::2]:
        for tau in taus:
            h = []
            for i in range(n):
                val = (i * alpha + tau) % 1
                h.append(int(val < 0.5))
            h = np.array(h)
            if np.sum(h) != n/2:
                continue
            bound = evaluate_h(h, 2.0 / n)
            if bound < best_bound:
                best_bound = bound
                best_alpha = alpha
                best_tau = tau
    # refine around best with golden section
    g = (1 + 5**0.5) / 2
    a, b = 0.2, 0.8
    # greedy search in both variables
    def refine_alpha():
        nonlocal a, b
        while abs(b-a) > 1e-8:
            left = a + (b - a) / g
            right = b - (b - a) / g
            h1 = []
            for i in range(n):
                val = (i * left + best_tau) % 1
                h1.append(int(val < 0.5))
            h1 = np.array(h1)
            o1 = evaluate_h(h1, 2.0 / n)
            h2 = []
            for i in range(n):
                val = (i * right + best_tau) % 1
                h2.append(int(val < 0.5))
            h2 = np.array(h2)
            o2 = evaluate_h(h2, 2.0 / n)
            if o1 < o2:
                b = right
                best_alpha = left
            else:
                a = left
                best_alpha = right
    # refine_alpha()

    # implementing basic version to meet deadline
    h = []
    for i in range(n):
        val = (i * best_alpha + best_tau) % 1
        h.append(int(val < 0.5))
    h = np.array(h)
    if np.sum(h) != n/2:
        return None, np.inf

    # Now do local swap improvement (like a fast GS search)
    max_swaps = min(5000, int(0.1 * n ** 2))
    best_eval = evaluate_h(h, 2.0 / n)
    for _ in range(max_swaps):
        k_max = -1
        idx = -1
        for i in range(n):
            # We're computing the shift k = 0 ... n-1 (that is C_{-n+1} to C_{n-1} in overlap evaluation)
            new_h = h.copy()
            if h[i] == 0:
                new_h[i] = 1
            else:
                new_h[i] = 0
            # compute all overlaps for this new vector
            # AND/OR update a subset efficiently
            # Effectively compute overlap as O_k = (sum h_i) - sum h_i h_{i+k}
            # so it depends on: new h_i and previous overlap_k
            # compute overlaps efficiently:
            eval_new = evaluate_h(new_h, 2.0 / n)
            if eval_new < best_eval:
                best_eval = eval_new
                h = new_h
                break
        else:
            break
    # final evaluation
    final_bound = evaluate_h(h, 2.0 / n)
    return h, final_bound
