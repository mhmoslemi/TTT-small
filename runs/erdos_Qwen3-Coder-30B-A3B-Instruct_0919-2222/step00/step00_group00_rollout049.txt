import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    # Discretization sizes to try
    n_list = [80, 120, 160]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_list:
        if time.time() - t_start > budget_s - 5:
            break
            
        dx = 2.0 / n
        K0 = max(1, n // 4)  # Initial active shifts: only small shifts to start
        K = list(range(1, K0 + 1))
        
        # Reset best solution for this n
        best_t_for_n = float('inf')
        best_x_for_n = None
        
        # Lazy constraint loop
        max_iterations = 10
        for iteration in range(max_iterations):
            if time.time() - t_start > budget_s - 2:
                break
                
            # Build the MILP
            x = cp.Variable(n, boolean=True)
            t = cp.Variable()
            
            # Constraints for x (mass constraint)
            constraints = [cp.sum(x) == n // 2]
            
            # Symmetry breaking for faster solving
            constraints.append(x[0] == 1)
            for i in range(n // 2 - 1):
                constraints.append(x[i] >= x[i+1])
            
            # Create auxiliary variables and terms for current active shifts
            y = {}
            c_terms = []
            
            # Set up y variables and linearization constraints
            for k in K:
                y[k] = cp.Variable(n - k, boolean=True)
                for i in range(n - k):
                    constraints.append(y[k][i] <= x[i])
                    constraints.append(y[k][i] <= x[i + k])
                    constraints.append(y[k][i] >= x[i] + x[i + k] - 1)
                # Compute correlation term for shift k
                c_k = cp.sum(x[:n-k]) - cp.sum(y[k])
                c_terms.append(c_k)
                constraints.append(c_k <= t)
            
            # Objective: minimize t
            prob = cp.Problem(cp.Minimize(t), constraints)
            
            # Get timeout estimate for this solve
            remaining = budget_s - (time.time() - t_start)
            if remaining < 2:
                break
                
            # Solve the MILP
            try:
                prob.solve(solver=cp.CBC, verbose=False, time_limit=remaining - 1)
            except Exception:
                break
                
            if prob.status != cp.OPTIMAL:
                break
                
            x_sol = np.round(x.value).astype(int)
            t_opt = t.value
            
            # Check full correlation
            c_full = correlate(x_sol, 1 - x_sol, mode="full")
            if c_full.max() <= t_opt + 1e-6:
                # Feasible solution at this n
                if t_opt < best_t_for_n:
                    best_t_for_n = t_opt
                    best_x_for_n = x_sol.copy()
                break
            else:
                # Find violated shifts and add them to K
                k_violated = []
                for k in range(1, n):
                    if c_full[n - 1 + k] > t_opt:
                        k_violated.append(k)
                if not k_violated:
                    break
                for k in k_violated:
                    if k not in K:
                        K.append(k)
                        if len(K) > n + 10:
                            break  # Too many shifts, just break out
                if len(K) > 2 * n:
                    break  # Hard limit to prevent exponential growth
            
        # After final MILP, polish with deterministic swaps
        if best_x_for_n is not None:
            h = best_x_for_n.astype(float)
            c_full = correlate(h, 1 - h, mode="full")
            while True:
                if time.time() - t_start > budget_s - 1:
                    break
                max_k = np.argmax(c_full)
                if c_full[max_k] <= best_t_for_n + 1e-6:
                    break
                # Find swap pair
                k = max_k - (n - 1)
                # List 1s and 0s' indices that affect this shift
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                # Collect elements where we can swap
                swap_candidates_1 = []
                swap_candidates_0 = []
                for i in ones:
                    if i + k < n and h[i + k] == 0:
                        swap_candidates_1.append(i)
                for j in zeros:
                    if j + k < n and h[j + k] == 1:
                        swap_candidates_0.append(j)
                if not swap_candidates_1 or not swap_candidates_0:
                    break
                # Perform a single swap
                i = swap_candidates_1[0]
                j = swap_candidates_0[0]
                h[i] = 0
                h[j] = 1
                # Recompute c_full incrementally
                actions = []
                for shift in range(1, n):
                    if j + shift < n:
                        actions.append((shift, j, 1))
                    if i + shift < n:
                        actions.append((shift, i, -1))
                for shift, idx, delta in actions:
                    c_full[n - 1 + shift] -= delta
                # This is approximate; for correctness recompute full

                best_h = h.copy()
                best_n = n
