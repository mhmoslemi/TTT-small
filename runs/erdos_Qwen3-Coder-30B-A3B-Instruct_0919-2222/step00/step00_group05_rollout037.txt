import cvxpy as cp
import numpy as np
import time
import random
from scipy import correlate

def evaluate_c5(h_values):
    """Evaluate the C5 value of a given step function h."""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute all correlations (full mode) and find the maximum
    correlations = correlate(h, 1 - h, mode="full") * dx
    return np.max(correlations)

def build_mcormick_constraints(h_vars, y_vars, s, i, n):
    """Build Mc-Cormick constraints for the product h[i] * h[i+s]"""
    if i + s < 0 or i + s >= n:
        return []
    return [
        y_vars[i, s] <= h_vars[i],
        y_vars[i, s] <= h_vars[i + s],
        y_vars[i, s] >= h_vars[i] + h_vars[i + s] - 1,
        y_vars[i, s] >= 0
    ]

def solve_relaxed_lp(h_init, n, mass):
    """Solve the LP relaxation of the problem"""
    # Variables: h, y, t
    h = cp.Variable(n)
    y = cp.Variable((n, 2*n-1), nonneg=True)
    t = cp.Variable()  # This represents the upper bound on correlations
    
    # Constraints
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == mass,
        t >= 0
    ]
    
    # McCormick constraints for bilinear terms
    for i in range(n):
        for s in range(-n+1, n):
            if 0 <= i + s < n:
                constraints += [
                    y[i, s+n-1] <= h[i],
                    y[i, s+n-1] <= h[i+s],
                    y[i, s+n-1] >= h[i] + h[i+s] - 1,
                    y[i, s+n-1] >= 0
                ]
    
    # Constraints for each correlation: sum of y terms >= (n/2 - t)
    for s in range(-n+1, n):
        if s >= 0:
            terms = [y[i, s+n-1] for i in range(n-s)]
        else:
            terms = [y[i, s+n-1] for i in range(-s, n)]
        constraints.append(cp.sum(terms) >= mass - t)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Form and solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    
    return h.value, y.value, t.value

def deterministic_rounding(h_lp, n_points, target_mass):
    """Rounds h_lp to a binary vector with exact mass"""
    h_rounded = np.zeros(n_points)
    h_sorted_idx = np.argsort(h_lp)[::-1]
    for i in range(int(target_mass)):
        h_rounded[h_sorted_idx[i]] = 1
    return h_rounded

def randomized_rounding(h_lp, n_points, target_mass, n_samples=200):
    """Generate binary vectors by sampling from probabilities in h_lp"""
    samples = []
    h_prob = np.array(h_lp)
    for _ in range(n_samples):
        # Sample indices with probability proportional to h_prob
        idxs = np.random.choice(n_points, int(target_mass), p=h_prob/h_prob.sum(), replace=False)
        h_sample = np.zeros(n_points)
        h_sample[idxs] = 1
        samples.append(h_sample)
    return samples

def polish_candidate(h_candidate, n_points, target_mass, max_time, start_time):
    """Apply local search (targeted swap) on the candidate"""
    h_current = np.array(h_candidate)
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current)
    
    elapsed = time.time() - start_time
    max_elapsed = max_time - elapsed
    
    while elapsed < max_elapsed:
        # Compute correlations
        corr = correlate(h_current, 1 - h_current, mode="full")
        max_shift = np.argmax(corr)
        max_corr = corr[max_shift]
        
        if max_corr <= best_c5:
            break
            
        # Find good swaps
        best_gain = 0
        best_i, best_j = None, None
        
        # Go through potential swaps for the currently worst shift
        s = max_shift - n_points + 1  # maps index to shift
        for i in range(n_points):
            if h_current[i] != 1:
                continue
            j = (i + s) % n_points
            if h_current[j] == 1:
                continue
            # Estimate impact on max correlations by swapping
            # This is simplified but effective
            gain = -2  # guaranteed to reduce by at least 2 if properly selected
            if h_current[(i - s) % n_points] == 0:
                gain += 1
            if h_current[(j + s) % n_points] == 1:
                gain += 1
            if gain > best_gain:
                best_gain = gain
                best_i, best_j = i, j
        
        if best_i is None or best_gain >= 0:
            break
            
        # Execute swap
        h_current[best_i] = 0
        h_current[best_j] = 1
        cur_c5 = evaluate_c5(h_current)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_current.copy()
        
        elapsed = time.time() - start_time
    
    return best_h, best_c5
