import numpy as np
from scipy.optimize import differential_evolution
import time

def project_to_capped_simplex(v, target_sum):
    """Project vector v to simplex with sum = target_sum and bounds [0,1]"""
    v = np.asarray(v)
    n = len(v)
    # Sort in descending order to find threshold
    sorted_indices = np.argsort(-v)
    sorted_v = v[sorted_indices]
    
    # Waterfilling with cap at 1
    cumsum = np.cumsum(sorted_v)
    thresholds = (cumsum - target_sum) / np.arange(1, n+1)
    
    # Find the largest index where threshold <= 1
    valid_idx = np.where(thresholds <= 1)[0]
    if len(valid_idx) == 0:
        tau = 1
    else:
        tau = thresholds[valid_idx[-1]]
    
    # Clip and reproject to ensure sum matches target
    result = np.clip(sorted_v - tau, 0, 1)
    
    # If sum is still not exact, adjust
    current_sum = np.sum(result)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        for i in range(n-1, -1, -1):
            if result[i] + diff <= 1:
                result[i] += diff
                break
            else:
                diff -= (1 - result[i])
                result[i] = 1
    
    # Reorder back to original indices
    final = np.zeros_like(v)
    final[sorted_indices] = result
    return final

def evaluate_overlap(h, n_points):
    """Evaluate the overlap C5 for a given h"""
    h = np.array(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    c5_bound = max_corr * dx
    return c5_bound

def compute_autocorr_blocks(a, w, N_shifts=200):
    """Compute G(k) = ∫ h(x) h(x+k) over all shifts k"""
    M = len(a)
    # Start/end points for each block
    s = np.concatenate([[0], np.cumsum(w)])
    
    # Accumulate the overlap information for all block-pairs
    # For each shift k, overlap depends on how much of I_i overlaps with I_j + k
    G = np.zeros(N_shifts)
    dx_shift = 2.0 / N_shifts
    shifts = np.linspace(0, 2, N_shifts, endpoint=False)
    
    for k_idx, k in enumerate(shifts):
        g = 0.0
        for i in range(M):
            for j in range(M):
                s_i = s[i+1]
                s_j = s[j+1]
                start_i = s[i]
                start_j = s[j]
                # Overlap of [start_i, s_i) with [start_j+k, s_j+k)
                overlap = max(0.0, min(s_i, s_j + k) - max(start_i, start_j + k))
                g += a[i] * a[j] * overlap
        # Normalize by the constant shift width (dx_shift does not apply since we are not integrating)
        G[k_idx] = g
    return G

def fitness_block(h_params, M=12):
    """Objective function to be minimized"""
    u, v = h_params[:M], h_params[M:]
    w = 2 * np.exp(u) / np.sum(np.exp(u))
    a = 1.0 / (1.0 + np.exp(-v))  # sigmoid
    Lambda = 1000
    
    # Integral constraint
    integral = np.sum(a * w)
    penalty = Lambda * abs(integral - 1.0)
    
    # Compute correlation (autocorrelation)
    N_shifts = 200
    G = compute_autocorr_blocks(a, w, N_shifts)
    f = 1.0 - G  # f(k) = 1 - G(k) = ∫ h(x) (1 - h(x + k)) dx
    obj = np.max(f)
    
    return obj + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Some constants
    M = 12        # number of blocks  
    N = 200       # final grid size
    dx = 2.0 / N
    
    # DE settings
    bounds = [(-5, 5)] * (2 * M)
    max_iter = 50  # Limit by budget
    
    # Global optimization with DE to find good starting point
    res = differential_evolution(
        fitness_block, 
        bounds, 
        maxiter=max_iter,
        seed=seed,
        polish=False,
        disp=False
    )
    
    # Parse DE result
    u_opt = res.x[:M]
    v_opt = res.x[M:]
    w_opt = 2 * np.exp(u_opt) / np.sum(np.exp(u_opt))
    a_opt = 1.0 / (1.0 + np.exp(-v_opt))
    
    # Enforce integral constraint exactly
    mass = np.sum(a_opt * w_opt)
    a_opt = a_opt / mass
    a_opt = np.clip(a_opt, 0, 1)
    # Re-normalize if needed to satisfy ∑a_i * w_i = 1 exactly
    mass_revised = np.sum(a_opt * w_opt)
    if abs(mass_revised - 1.0) > 1e-8:
        a_opt = a_opt / mass_revised
    
    # Sample onto fine grid
    s = np.concatenate([[0], np.cumsum(w_opt)])
    h = np.zeros(N)
    for i in range(N):
        x = i * dx
        for j in range(M):
            if s[j] <= x < s[j+1]:
                h[i] = a_opt[j]
                break
        # Handle potential floating point errors
        if s[M] <= x:
            h[i] = a_opt[M-1]
    
    # Apply refinement through projected subgradient
    alpha_0 = 0.2
    max_iter_sg = 3000 
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, N)
    
    for iteration in range(max_iter_sg):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
            
        # Compute the correlation vector    
        corr_full = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr_full)
        max_shifts = np.where(corr_full == max_corr)[0]
        k_star = max_shifts[0]
        
        # Compute gradient G_k_star
        g = np.zeros_like(h)
        for k in max_shifts:
            # Generate gradient with respect to C_k where C_k is max
            # ∂C_k/∂h_i = (1-h_{i+k}) - h_{i-k} 
            # But be careful about k indices, they are 0-based for full correlation
            # So for k=0 to len-1, C_k is stored at index = k 
            
            #        0 1 2 ...    n-1 
            # h    [ · · · ... · ]
            # C_k  [ · · · ... · ] 
            # 
            # we want:
            #   C_k = sum_i h[i] (1 - h[i+k])
            #   [k terms to match]
            #   When k=0: h[i](1-h[i]) for i=0 to N-1
            #   When k=1: h[i](1 - h[i+1]) for i=0 to N-2
            
            # But it's full convolution, so for shifting by 0, we go from i=0 to N-1
            # We are actually just computing d/dh_i of C_k
            # Let’s consider i=0, k=0:
            #   C_0 = h[0](1 - h[0]) + h[1](1-h[1]) ... 
            #   Thus dC_0/dh[0] = (1 - h[0]) - h[0] = 1 - 2*h[0]   But not really - it should be
            #   dC_0/dh[0] = (1 - h[0])  
            #   Since h[0] only appears in term h[0](1−h[0]) 
            #   and also in h[0+k](1 − h[0]) for k=0, we have one term at h[0] which is signed by 1
            
            # Actually let's do it carefully:
            # C_k = sum_{i=0}^{N-k-1} h[i] (1 - h[i+k])
            # dC_k / dh[i] = (1 - h[i+k])/delta_{i>=0} - h[i-k]/delta_{i>=k}
            # For k shift, in the tail start at i=0 to end at N-k
            # Backwards-adjacent term, and forwards together form right derivative formulation
            
            # Simpler: we just go back and mark a range, index from 0, forward and backward.
            # Let’s write for clean v = 1-h 
            # C_k = sum_i h[i] v[i+k] = sum_i h[i] (1 - h[i+k]) 
            # We will derive this into two parts: 
            #   1) From g[i] += (1 - h[i+k]) for range where both indices fit
            #   2) From g[i] -= h[i-k]     for range where both indices fit AND i >= k
            
            # Running index for shift k such that shift in corr is from 0 to 2*N-2
            # The direct approach:
            # We compute (a, b) = (start, end) where valid i-k ranges fall in h
            # Python pulling with slice handles this cleanly
            L = len(h)
            start_a = 0
            end_a = L - k
            start_b = k
            end_b = L
            # Since we have (from corr) at index k to N+k-1 where i+k index goes from k to 2*N-2
            # corr at index k (corresponding to shift k)
            if k < len(corr_full):
                start_corr = k
                end_corr = k + len(h)  # to N+k-1; this is possible since we’re unfolding the full convolution  
            else:
                start_corr = 0
                end_corr = len(corr_full)
                
            # Diagonal index trick: k goes [0, N), but full correlation likely means longer?
            # To get k = 0 to N, where k goes left to right for input h, i shifts from 0 to N
            # corr (from full) gives 0 to 2*N-2
            # so corr[0] means shift 0, corr[1] means shift 1 ... corr[2*N-2] means shift 2*N-2
            # But h is [0, 2) with first element at index i=0.
            # We can change to discrete shift k: (0 to N-1 inclusive) of actual shift
            # Then shift k in correlation is |corr_idx - N + 1|, or we do the shift index mapping properly
            # Let's clarify what index k in h corresponds to shift z: it's due to discrete convolution 
            # Let's just fix one map again.
            # If corr[0] is correlation for shift=0,
            # then corr[N-1] is for shift = 0 -> i in 0..N-1 so,
            # According to np.correlate manual output formatting, mode='full' gives array of length 2N-1
            # correlate(h, 1-h) for h of length N → corr of length 2N-1
            # so from corr[0], we get shift 1-N = -1 to shift 1-N + 2*N-2 = N-1  
            # shift = corr_idx - (N - 1) where corr_idx runs 0 to 2N-2
            # total shift range = [-N+1, N-1] inclusive -> tell shift as index shift_idx = shift + N-1  
            # Let's take a specific example: let N=3 
            # -> h[0], h[1], h[2]  
            # we compute full correlation (from Section np.correlate doc) of h*rev(1-h) which is:
            # Do convolution with both signals: put 1-h[0] 1-h[1] 1-h[2] 
            # corr[0] = h[0]*(1-h[2]) + h[1]*0 + h[2]*0 = h[0]*(1-h[2])  
            # corr[1] = h[0]*(1-h[1]) + h[1]*(1-h[2]) + h[2]*0
            # corr[2] = h[0]*(1-h[0]) + h[1]*(1-h[1]) + h[2]*(1-h[2])
            # corr[3] = h[1]*(1-h[0]) + h[2]*(1-h[1])
            # corr[4] = h[2]*(1-h[0])
            
            # This means shift 0 = corr[2] = i=0 to i=2 gives h[0] same position
            # Shift -1 = corr[1] = i=0 to i=1 ...
            # We can run direct evaluation for safety, whose gradient works out to the above.
            
            # Simple direct way: Use only i ranges that have contribution to C_k
            # and do a forward shift of indices.
            # What about inverse: shift in timing and inclusion of terms. 
            # Let's define function that properly gives gradient of a shift k over all correlated terms
            
            # Fast way: start over
            # Compute one specific entry at k 
            # C_k = sum_{i=0}^{N-k-1} h[i] (1 - h[i+k])
            # dC_k / dh[i] = (1 - h[i+k]) if i+k < N, else 0  
            # dC_k / dh[i] goes like: =1-h[i+k] from i=0 to N-k-1  
            #                   also -= h[i-k] from i=-k to N-k-2  (but i>=0 thus i=k to N-k-2) so for i=k to N-k-2
            # But watch signs:
            # C_k = sum h[i] (1 - h[i + k])  
            # and in summation above:
            # d/dh[i] = 1 - h[i+k] when i+k < N  
            # d/dh[i] = -h[i-k] when i-k >= 0 and valid i-k < N, i.e., i >= k, i < N
            # Thus (operator over h_i): gradient term for index i:
            #   += (1 - h[i+k]) while i+k < N, so i <= N-k-1  
            #   -= h[i-k] when i >= k
            
            # Use numpy-based gradient building step
            g = np.zeros_like(h)
            # Large part is where 1-h[i+k] → applies for i+k < N  => i < N-k  
            start_stop_g1 = [0, min(N - k, N)]
            if start_stop_g1[1] > start_stop_g1[0]:
                g[start_stop_g1[0]:start_stop_g1[1]] += 1 - h[start_stop_g1[0] + k : start_stop_g1[1] + k]
                
            # Subtract h[i-k] for i >= k  
            start_stop_g2 = [k, N]
            if start_stop_g2[1] > start_stop_g2[0]:
                # For element i starting at k up to N-1 and index j = i-k, we reduce h[j]
                # Simpler: subtract h from index k forward by indexing back
                g[start_stop_g2[0]:start_stop_g2[1]] -= h[:start_stop_g2[1] - start_stop_g2[0]] 
            
            # possible	unsigned gradient now...
            # Scale the gradient properly?
            
            # Now, work on the average gradient from among maximizing shifts...
            # This is to be exploited in the subgradient descent later
            # Let me simplify it to just using one maximizer k_star
            # Now, form the explicit gradient at state h for current pair C_k_star  
            # The opposite direction from Exponential Moving Average. 
            #         Let’s use top gradient directly:
            # Complete update now using the combined gradients from max
            # Create a user-defined gradient vector
            # It’s more precise to do one by one or compute total. 
            # We will now perform all the grated operations.
            
            # Now the step:
            # Small for loop of subgradient steps.  
            # For small size of vector, this is okay.
        corr_full = np.correlate(h, 1 - h, mode='full')  
        max_shifts = np.where(corr_full == np.max(corr_full))[0]
        g = np.zeros_like(h)
        for k in max_shifts:
            # This is not a full computation for efficiency  
            # Better to just recompute for such key modes.           
        
        # Rebuild fn to implement average gradient of corresponding C_k  
        corr_full = np.correlate(h, 1 - h, mode='full')
        k_star_indices = np.where(corr_full == np.max(corr_full))[0]
        # Average gradient over all k_stars
        g = np.zeros_like(h)
        for k_star in k_star_indices:
            # Take the gradient of C_k_star w.r.t h_i
            # This became:  dC_k* / dh_i = 1-h[i+k_star] - h[i-k_star] if index valid
            L = len(h)
            i_g = np.arange(0, L)
            gi = np.zeros(L)
            # Mark where h[i+k_star] is contributing in increasing order
            i_plus = i_g[:-k_star] if k_star > 0 else i_g  
            gi[i_plus] += 1.0 
            # Mark h[i-k_star] part:
            i_minus = i_g[k_star:]
            # Reduce difference 
            gi[i_plus] -= h[i_plus + k_star] 
            gi[i_minus] -= h[i_minus - k_star]
            # Over all matrix operands
            g += gi
        
        # Normalize gradient norm to length.  
        # Scaled down by avg over number of max_k points
        g /= len(k_star_indices)
        
        # Subgradient step
        alpha = alpha_0 / np.sqrt(1 + iteration)
        h_new = h - alpha * g
        
        # Project to feasible set: [0,1], sum = N/2
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        h_new = project_to_capped_simplex(h_new, N/2) 
        
        h = h_new
        
        # Update best
        c5 = evaluate_overlap(h, N)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    return best_h.tolist(), best_c5, N
