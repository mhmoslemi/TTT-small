import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def solve_erdos_milp(n, time_limit):
    prob = pulp.LpProblem("ErdosMinOverlapMILP", pulp.LpMinimize)
    
    # Variables
    h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
    T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
    
    # Constraint sum h_i = n/2
    prob += pulp.lpSum([h[i] for i in range(n)]) == n/2
    
    # Product constraints and shift constraints
    for d in range(1, n):  # skip d=0 to avoid redundancy
        y_vars = []
        for i in range(n-d):
            y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
            y_vars.append(y)
            
            # McCormick envelope constraints for y = h_i * h_{i+d}
            prob += y <= h[i]
            prob += y <= h[i+d]
            prob += y >= h[i] + h[i+d] - 1
            
        # Shift constraint for this shift d: sum_y + T >= n/2
        prob += pulp.lpSum(y_vars) + T >= n/2
    
    # Objective: minimize T
    prob += T
    
    # Solve with time limit
    solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit)
    prob.solve(solver)
    
    # Extract solution
    h_vals = np.array([pulp.value(h[i]) for i in range(n)], dtype=float)
    t_val = pulp.value(T)
    
    return h_vals, t_val

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Test different resolutions
    n_candidates = [100, 120, 150, 180, 200]
    
    for n in n_candidates:
        if time.time() - start_time >= budget_s - 5:
            break

        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
            
        try:
            h_vals, t_val = solve_erdos_milp(n, int(remaining_time * 0.9))
            c5_bound = t_val * (2.0 / n)
            
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_vals.copy()
                best_n = n
                
        except Exception as e:
            continue
            
        # Early termination if we're getting close to target
        if best_c5 <= 0.379:
            break
    
    # Final evaluation with numpy array
    if best_h is not None:
        final_c5 = evaluate_erdos_solution(best_h)
        return (best_h, final_c5, best_n)
    else:
        # Fallback to initial solution
        h_initial = np.array(initial_h_values)
        c5_initial = evaluate_erdos_solution(h_initial)
        return (h_initial, c5_initial, len(initial_h_values))
