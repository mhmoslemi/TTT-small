import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist
import time
import math

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdos constant C5 for given h values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlation = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # The maximum of these values is C5
    c5 = np.max(correlation) * dx
    return c5

def initial_h_values():
    """Return the initial construction for h values"""
    # Using a simple initial construction - a step function
    n = 45
    h = np.zeros(n)
    # Place a step function that satisfies the constraint sum(h)*dx = 1
    # This is a simple approximation to get started
    h[:22] = 1.0  # First 22 points set to 1
    h[22:] = 0.0  # Rest to 0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction - simple step function
    h_init = initial_h_values()
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    # Refine the initial solution with optimization
    def objective(h):
        # Compute overlap integral
        correlation = np.correlate(h, 1 - np.array(h), mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    def constraint_func(h):
        # Integral constraint: sum(h) * dx = 1, i.e., sum(h) = n_points / 2
        return np.sum(h) - n_points / 2.0
    
    # Set up bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial evaluation
    current_c5 = objective(h_init)
    best_h = h_init.copy()
    best_c5 = current_c5
    
    # Use scipy.optimize.minimize with SLSQP method for constrained optimization
    # Convert constraints to the format scipy expects
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Optimization loop - start with random perturbations
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbations to avoid local minima
        perturbation = np.random.normal(0, 0.02, n_points)
        h_perturbed = h_init + perturbation
        # Clip to [0, 1]
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Apply constraint
        sum_h = np.sum(h_perturbed)
        h_perturbed = h_perturbed * (n_points / 2.0) / sum_h
        
        # Evaluate
        c5_new = objective(h_perturbed)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_perturbed.copy()
            # Print progress occasionally
            if iteration % 1000 == 0:
                print(f"Iter: {iteration}, C5: {best_c5:.6f}")
    
    # Final refinement with scipy minimize
    res = minimize(objective, 
                   best_h.copy(), 
                   method='SLSQP', 
                   bounds=bounds,
                   constraints=cons,
                   options={'maxiter': 1000, 'ftol': 1e-8})
    
    if res.success:
        best_h = res.x
        best_c5 = objective(best_h)
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
