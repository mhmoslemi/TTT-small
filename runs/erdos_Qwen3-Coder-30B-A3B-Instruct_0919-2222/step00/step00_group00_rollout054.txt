import numpy as np
import time
from scipy import signal

def compute_correlation(h):
    """Compute the correlation vector c_k = Σ_i h[i] (1 - h[i+k]) for all shifts k."""
    n = len(h)
    # Use numpy's correlate with mode='full'
    # This computes a vector c of size 2*n-1, where c[k+n-1] = c_k.
    c = signal.correlate(h, 1 - h, mode='full')
    return c

def smoothed_objective_and_weights(c, alpha):
    """Compute Φ_α = (1/α) log Σ exp(α c_k) and weights w_k."""
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(alpha * c_shifted)
    exp_sum = np.sum(exp_c)
    w = exp_c / exp_sum
    phi_alpha = (1.0 / alpha) * (np.log(exp_sum) + c_max)
    return phi_alpha, w

def grad_h(h, weights):
    """Compute ∇Φ_α(h) where weights come from the current c."""
    n = len(h)
    # Compute convolution of weights with (1-h) and h
    # w is 2n-1 long; shift indices go from -(n-1) to n-1
    # we want weighted sum over terms (1-h_{i+s}) 1_{i+s<n} and -h_{i-s} 1_{i-s>=0}
    # To compute efficiently, we'll compute via: 
    # forward part = weight convolve (1-h) 
    # backward part = weight convolve h  
    # Then g_i = forward[i] - backward[i] 
    
    # Extended weight vector centered appropriately
    # weights[0] corresponds to shift -(n-1)
    # weights[-1] corresponds to shift n-1
    
    # The forward part: Σ_s w_s (1 - h_{i+s})
    # This equals -convolution(weights_rev, h) + [sum(weights)*1]*n 
    # Or compute flexibly using numpy correlate (advanced)
    
    # Simple approach with direct computation for efficiency:
    # For each shift k in [-n+1, n-1] we extend weights to include all shifts
    # but weighted via the actual shift index, i.e., in the shifted arrangement
    
    # weights vector corresponds to shifts -n+1 to n-1. So to index
    # if shift is k then weights[k + n - 1] = respective weight 
    
    # But it's more direct to compute gradients over all shifts k of index
    # use forward and backward contributions
    
    c = compute_correlation(h)
    # Compute aggregated gradient efficiently via discrete convolution
    # But the gradient is probed directly via computation: for k=-n+1 to n-1
    # gradient of c_k is (1 - h_{i+k}) 1_{i+k < n} - h_{i-k} 1_{i-k >= 0}
    # So let's consider now each i in [0,n)
    
    # A matrix-free gradient computation:
    # We can use prefix sums instead for O(n) per component
    
    # The approach:
    # Compute forward sums of (1-h), and backward sums of h 
    # then exploit the structure:
    # For every i, g_i = forward_sum[i] - backward_sum[i]
    
    # forward[i] = sum_{j >= i} (1 - h[j])
    # backward[i] = sum_{j <= i} h[j]
    # result = (1 - h[i]) + forward[i+1] - backward[i-1]
    
    
    # Prefix sums
    forward_sum = np.cumsum(np.flip(1 - h))
    forward_sum = np.flip(forward_sum)
    forward_sum = np.append(forward_sum, 0)  # handle boundary condition
    
    backward_sum = np.cumsum(h)
    
    # Hypothesizing that:
    # m = len(c) = 2*n - 1 => c is indexed from -(n-1) to n-1
    # we want g_i = {sum over shift k} w_k [ (1-h_{i+k}) 1_{i+k<n} - h_{i-k} 1_{i-k>=0} ]  
    # This could make g_i a function of shifts with fixed i, so:
    # debug: compare correlation computation and expectation  
    
    # Simpler to compute:
    # g = weights * correlation gradients directly
    # To compute the expected gradient at i: 
    # for any s ranging from -(n-1) to (n-1):
    #    weight s contributes to g_i  
    #    via: w[s+n-1] * [ (1-h_{i+s}) - h_{i-s} (only s such that i+s valid and i-s valid) ]
    # i.e. a bit more endian cloze sum directly.
    # We can vectorize individual g_i into full array computation.
    
    # Let's start plain fast method case in O(n^2) with linear indexing:
    g = np.zeros(n)
    for i in range(n):
        for s in range(-(n-1), n):
            if s + i >= 0 and i - s >=0 and i - s < n:
                g[i] += weights[s + n - 1] * ( (1 - h[i+s]) - h[i-s] )
    
    return g

def project_onto_feasible(y, target_sum):
    """Project onto the feasible set: 0 <= h <= 1 and sum(h) = target."""
    assert len(y) > 0
    n = len(y)
    # Clip to [0,1]
    h = np.clip(y, 0, 1)
    current_sum = np.sum(h)
    diff = current_sum - target_sum
    if abs(diff) < 1e-12:
        return h
    # Relaxation via water filling:
    # Adjust components that are free (inside (0,1))
    free = np.where((h > 0) & (h < 1))[0]
    if len(free) == 0:
        # All at bounds, so we need to shift and fix to target
        if current_sum > target_sum:
            # Want to make some values zero. Take the ones that will lose least.
            # Set to zero everything with highest values
            order = np.argsort(h)[::-1]
            diff = current_sum - target_sum
            h = np.copy(y)
            for j in order:
                if diff <= 0:
                    break
                val = h[j]
                h[j] = 0
                diff -= val
        else:
            # current_sum < target_sum
            # Set to 1 where h[j] < 1 (maximize deltas)
            h = np.copy(y)
            h = np.clip(h, 0, 1)
            free = np.where((h > 0) & (h < 1))[0]
            if len(free) > 0:
                # Add excess mass to free components
                excess = target_sum - current_sum
                # divide among free components 
                add = excess / len(free)
                h[free] += add
                h = np.clip(h, 0, 1)
        return h
    
    # More elaborate waterfilling:
    # Sort free indices by value descending
    free_values = h[free]
    sort_idx = np.argsort(free_values)[::-1]
    free_sorted = free[sort_idx]
    
    # Gradually adjust bar height to match target  
    # We need to adjust all free components by the same amount but due to bounds,
    # we gradually constrain   
    while not np.isclose(current_sum, target_sum, atol=1e-12):
        # Compute how much we can adjust:
        # If we want to drain or fill a set S, we compute how much change
        # In terms of water level.
        # decrease at i means exitS = free_indices[i], change = h[i] - 0
        # increase = 1 - h[i]...
        # Instead:
        # image of the values in the end:
        # We will assign values starting from top down.
        # Algo: maintain stack of <= (n free)
        # But in the water filling setting...
        # Start with lowest free component
        delta_sum = 0
        
        # find smallest value of free[idx[i]] such that delta from current sum <= target - sum
        # So we look for current boundary point where we can fill up and still not exceed
        
        # If     i are boundary indices where (h[i] is free)
        # and h[i] <= 1 and we raise to h[i] + d(i), we increase sum by d(i)
        # Find minimal d such that sum += d(i) across balancing indices 
        # becomes exactly target.
        
        # strategy: 
        # work backwards, guess a cumulative shift value (from bottom to top) 
        # and compute total delta. Stop when we reach or exceed target
        
        # May be overcomplex: just recompute total difference on   
        #  
        
        # Simple: until delta is exhausted fold smaller size components same amount
        # i.e. keep delta = pose, and loop:
        # take lowest one (say index j) with low h[j] (after trimming to [0,1])
        # compute how much it can rise: 1 - h[j]
        # decide: adjust next bigger free components or stop.
        # For simplicity: sort free/decrease order and just 
        # shift each one up until adjacent cutoff
        
        if current_sum > target_sum:
            # Must reduce
            # Find best components: those with low values to set to 0  
            # Or reduce free components
            # Efficient to reduce all
            # Determine how much to take out
            excess = current_sum - target_sum
            if excess < 1e-12:
                return h
                
            # Adjust lower components first, or add probability to preserve routine
            # Sort free by val ascending to see which ones to reduce first
            # Simplification: take all with lowest values and reduce
            free_h = h[free]
            # Place k = len(free) items in new array for simple use — let's iterate
            # Better way:
            
            # Alternately find amount we can subtract:
            if len(free) == 0:
                pass
            else:
                # use greedy to get above target
                # Stable sort for tie breaking 
                idx_sorted = np.argsort(h[free])
                energy = excess
                for idx in idx_sorted:
                    if energy <= 0:
                        break
                    i = free[idx]
                    can_reduce = h[i]
                    reduce = min(can_reduce, energy)
                    h[i] -= reduce
                    energy -= reduce
            current_sum -= excess
        else:
            # Must increase low free ones
            # The others cannot be more than 1 but can increase  
            # (we treat box bounds as pathologically fixed, improve to make black box bounce)
            needed = target_sum - current_sum
            energy = needed
            # Could allow facilitation from top criterion but potentially helps
            # try: sort by value descending then add
            if len(free) > 0:
                free_zip = list(zip(free, h[free]))
                free_zip.sort(key=lambda x: x[1], reverse=True) # Top first to give to those willing
                for i, _ in free_zip:
                    if energy <= 0:
                        break
                    # may grow
                    max_growth = 1 - h[i]
                    to_add = min(max_growth, energy)
                    h[i] += to_add
                    energy -= to_add
            current_sum += needed
    
    return h

def local_binary_search(h_bin, c_orig, best_val, time_left):
    """Polish a binary (`h_bin`) solution with efficient swaps.""" 
    # Import utils now
    import heapq
    
    n = len(h_bin)
    h = h_bin.copy()
    c = c_orig.copy()
    best_c = best_val
    # To manage time, we rely on the outside loop, just have heads-up and limit of swaps
    
    ONES = np.where(h == 1)[0]
    ZEROS = np.where(h == 0)[0] 
    
    off = n - 1
    
    def update_c_swap(i, j, delta):
        # Incrementally update correlation vector c 
        # input: candidate swap of h[i]=1 with h[j]=0 which will become h[i]=0 h[j]=1
        # delta = 1 (for i -> 0 and j -> 1)
        # we apply delta changes at i and j
        delta_c = delta  # now delta * 1
        
        # Using the derivative rule:
        # c: result will be a shifted convolution (like during gradient)
        # We modify it by:
        # change at i contributes to c[s] for s in [i .. n-1]  
        # change at j contributes to c[s] for s in [0, j]
        # but exactly:
        # s = i-j : shift = s+off (idx in c)
        # s = j-i : shift = s+off (idx in c)

        # Which means we compute c += (should add it once for each s)
        for q in range(i, n):
            s = q - i 
            idx = s + off
            # remember c was from corr(h, 1-h)
            c[idx] += delta * (1 - h[q])
        for q in range(0, j + 1):
            s = i - q 
            idx = s + off
            # c s = i-q = i-q corresponds to shift q-i.
            c[idx] -= delta * h[q]
    
    iter_count = 0
    max_iter = 1000000  # Just a stopper to avoid runaway
    while time_left > 0.1 and iter_count < max_iter:
        iter_count += 1
        # Determine the worst shift
        s_max = np.argmax(c)
        c_max_here = c[s_max]
        
        if c_max_here <= best_c:
            break  # No improvement
    
        # Find candidates for swap
        # A = {i in ONES | h[i+s_max] == 0 and i+s_max < n}
        # B = {j in ZEROS | h[j+s_max] == 1 and j+s_max < n}  
        A = []
        B = []
        
        for i in ONES:
            j = i + s_max
            if j < n and h[j] == 0:
                A.append(i)
        for j in ZEROS:
            i = j - s_max
            if i >= 0 and h[i] == 1:
                B.append(j)
    
        # Perform a targeting swap only when enough candidates:
        if len(A) > 0 and len(B) > 0:
            i = np.random.choice(A)
            j = np.random.choice(B)

            # point update            
            h[i] = 0
            h[j] = 1
            delta = 1
            update_c_swap(i, j, delta)
            best_c = np.max(c)
        else:
            # fallback to random
            i = np.random.choice(ONES)
            j = np.random.choice(ZEROS)
            # Minor performance keel of full correlation update;OK for debugging   
            h[i] = 0
            h[j] = 1
            # additional sub-iterative update
            delta = 1
            update_c_swap(i, j, delta)
            best_c = np.max(c)

    return h, best_c        


def upscale_binary(h, old_n):
    """Upscale binary (step) function by linear interpolation."""
    new_n = 2 * old_n
    h_new = np.zeros(new_n)
    for i in range(old_n):
        h_new[2*i] = h[i]
        h_new[2*i+1] = h[i]  # replicate
    return h_new
