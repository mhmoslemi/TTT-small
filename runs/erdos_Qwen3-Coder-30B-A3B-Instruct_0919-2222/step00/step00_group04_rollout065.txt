import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    correlations = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # The maximum value is what we want
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Return initial h values as described in the problem"""
    n_points = 42
    h = np.ones(n_points) * 0.5
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 40  # Slightly smaller for faster optimization
    dx = 2.0 / n_points
    h = initial_h_values()
    h = h / np.sum(h) * (n_points / 2)  # Normalize so integral = 1
    
    # Optimization using scipy minimize with bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]
    
    # Create objective function
    def objective(h_vals):
        # Compute the overlap integral C_5
        correlations = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_overlap = np.max(correlations)
        return max_overlap * dx
    
    # Optimization using L-BFGS-B
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 1000})
    
    # Ensure the solution satisfies constraints exactly
    if result.success:
        h_opt = np.array(result.x)
        h_opt = h_opt / np.sum(h_opt) * (n_points / 2)  # Re-normalize
        # Evaluate the final result
        c5_bound = evaluate_erdos_solution(h_opt)
        return (h_opt, c5_bound, n_points)
    else:
        # If optimization fails, fallback to initial solution
        c5_bound = evaluate_erdos_solution(h)
        return (h, c5_bound, n_points)
