    """Evaluate the C5 constant for a given step function h."""

    # Compute the overlap integral for all shifts
    correlation = signal.correlate(h, 1 - h, mode='full')
    # The maximum value is the C5 bound
    c5 = np.max(correlation) * dx
    return c5
