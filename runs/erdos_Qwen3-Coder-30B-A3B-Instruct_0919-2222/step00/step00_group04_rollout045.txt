import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h_values):
    # Compute the maximum overlap integral
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h_values, 1 - np.array(h_values), mode='full')
    # The maximum value in the full correlation is the desired C5 bound
    # We need to scale by dx
    c5_value = np.max(corr) * dx
    return c5_value

def objective_function(h_values):
    # Minimize negative of C5 to maximize the minimum overlap
    return -evaluate_c5(h_values)

def constraint_sum(h_values):
    # Constraint that integral equals 1
    n = len(h_values)
    dx = 2.0 / n
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize with the provided initial construction
    h_start = np.array(initial_h_values)
    n_points = len(h_start)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints dictionary for scipy optimization
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Store best solution and value
    best_solution = None
    best_c5 = float('inf')
    
    # Timer
    start_time = time.time()
    
    # Perform optimization
    try:
        result = differential_evolution(
            objective_function,
            bounds,
            constraints=constraints,
            seed=seed,
            maxiter=budget_s*10  # Estimate max iterations based on time
        )
        
        if not result.success:
            print("Optimization did not succeed. Using initial solution.")
            h = h_start.copy()
        else:
            h = result.x
            
        c5_value = evaluate_c5(h)
        
        # Final check of constraints
        integral_check = np.sum(h) * (2.0 / n_points)
        if abs(integral_check - 1.0) > 1e-6:
            print("Integral constraint not met. Using initial solution.")
            h = h_start.copy()
            c5_value = evaluate_c5(h)
        
        return (h.tolist(), c5_value, n_points)
        
    except Exception as e:
        print("Error during optimization, using initial solution.")
        h = h_start.copy()
        c5_value = evaluate_c5(h)
        return (h.tolist(), c5_value, n_points)
