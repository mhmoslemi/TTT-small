import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap problem constant C5 for given step function h."""
    # Ensure h is a numpy array
    h = np.array(h)
    # Compute overlap integral using full convolution
    c = correlate(h, 1 - h, mode="full")
    # The factor 2/N comes from dx = 2/N and the scaling in correlation
    c5 = np.max(c) * (2.0 / len(h))
    return c5

def initial_h_values():
    """Initial construction from the problem statement."""
    # A reasonable initial guess: 101 points for discretization of [0, 2]
    n = 101
    x = np.linspace(0, 2, n, endpoint=False)
    # Try a triangular shape: h(x) = max(0, 1 - |x-1|) [bounded to [0,1]]
    h = np.maximum(0, 1 - np.abs(x - 1))
    # Normalize so that sum(h) = n / 2 = 50.5
    sum_h = np.sum(h)
    h = h * (n / 2) / sum_h
    return h

def run(seed=42, budget_s=600, **kwargs):
    """
    Find an improved upper bound for the Erdős minimum overlap problem constant C₅.
    
    Returns: (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)

    # Set maximum time to avoid exceeding budget

    timeout = start_time + budget_s * 0.99  # Leave some margin
    
    # Use a reasonable n_points based on target
    n_points = 511  # Odd number for better symmetry
    dx = 2.0 / n_points
    
    # Initialize from the known initial construction if available
    h = initial_h_values() if n_points == 101 else np.full(n_points, 0.5)
    
    # Or use a cosine-based shape
    x = np.linspace(0, 2, n_points, endpoint=False) + dx / 2  # midpoint sampling
    # Try cosine^2 shape for better smoothness
    h = np.cos(np.pi * (x - 1) / 2) ** 2
    h = np.clip(h, 0, 1)
    # Ensure sum equals n_points / 2
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h
    
    # Evaluate initial C5
    current_c5 = evaluate_erdos_solution(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = current_c5
    
    # Optimization loop parameters
    max_iter = 200000
    epsilon_initial = 0.02
    epsilon_min = 1e-5
    decay_rate = 0.99995
    
    for i in range(max_iter):
        if time.time() > timeout:
            break
            
        # Random generation of a move
        i1 = random.randint(0, n_points - 1)
        i2 = random.randint(0, n_points - 1)
        
        if i1 == i2:
            continue
            
        # Compute max allowable change (to stay in [0,1])
        delta_max = min(h[i1], 1 - h[i2])
        if delta_max <= 1e-10:
            continue
            
        # Apply random move
        epsilon = epsilon_initial * random.random()
        epsilon = min(epsilon, delta_max)
        h[i1] -= epsilon
        h[i2] += epsilon
        
        # Update sum to exactly n_points / 2 using balancing
        sum_h = np.sum(h)
        adjustment = (n_points / 2) - sum_h
        if abs(adjustment) > 1e-10:
            # Distribute adjustment among indices to keep all in [0,1]
            indices_to_adjust = np.argsort(h)  # Least values first, so increase them
            adjust_idx = indices_to_adjust[0]
            if h[adjust_idx] + adjustment <= 1:
                h[adjust_idx] += adjustment
                h[i1] -= epsilon  # if we adjusted other, revert corresponding update on i1
                h[i2] -= epsilon
                h[i1] += adjustment
                h[i2] -= adjustment
                h[adjust_idx] += adjustment
            else:
                h[adjust_idx] = 1
                h[i1] -= epsilon
                h[i2] += epsilon
                h[i1] += (1 - h[adjust_idx])  # adjust i1 so that it balances queue of adjustment
                # If adjustment was small we approximate, not fully enforce
        # Recompute
        new_c5 = evaluate_erdos_solution(h)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

            # Currently using deterministic descent based on efficiency

        # Reduced epsilon over iterations

    # Cast result to Python list as expected
    h_values = best_h.tolist()
    c5_bound = best_c5
