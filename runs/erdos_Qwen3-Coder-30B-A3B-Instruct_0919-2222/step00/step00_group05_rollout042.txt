import numpy as np
import random
import time
from scipy import signal

def evaluate(h, dx):
    # compute the overlap integral C_5 = max_k int h(x)(1 - h(x+k)) dx
    # using correlation: convolve h with (1-h)
    h1 = 1 - h
    corr = signal.correlate(h, h1, mode='full')
    # scale by dx for continuous approximation
    corr *= dx
    # return the maximum correlation value
    return np.max(corr)

def project_onto_constraints(h, dx):
    # Project h onto the constraint that sum(h)*dx = 1
    # and h[i] in [0,1]
    total = np.sum(h) * dx
    if abs(total - 1.0) > 1e-10:
        # Scale h to meet the integral constraint
        scale = 1.0 / total
        h = h * scale
    # Clamp values to [0,1]  
    h = np.clip(h, 0, 1)
    return h

def generate_random_feasible(n_points, dx):
    # generate a random feasible h
    h = np.random.rand(n_points)
    h = project_onto_constraints(h, dx)
    return h
