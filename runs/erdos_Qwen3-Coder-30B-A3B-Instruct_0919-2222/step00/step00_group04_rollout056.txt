import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project_to_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 ≤ x ≤ 1, sum x = target_sum"""
    v = np.array(v)
    sorted_v = np.sort(v)[::-1]
    n = len(v)
    cumsum = np.cumsum(sorted_v)
    # Find threshold
    tau = (cumsum - target_sum) / np.arange(1, n+1)
    idx = np.where(sorted_v > tau)[0]
    if len(idx) == 0:
        threshold = sorted_v[-1]
    else:
        threshold = tau[idx[-1]]
    result = np.clip(sorted_v - threshold, 0, 1)
    return result

def compute_overlap_from_blocks(a, w, N=200):
    """Compute C5 from piecewise constant block representation on grid N."""
    # Ensure w sums to 2
    w = np.array(w)
    a = np.array(a)
    w = 2 * w / np.sum(w)
    # Build the piecewise constant function on grid
    dx = 2 / N
    h = np.zeros(N)
    s = np.concatenate([[0], np.cumsum(w)])
    for i in range(len(w)):
        start = int(s[i] / dx)
        end = int(s[i+1] / dx)
        h[start:end] = a[i]
    return h

def evaluate_fitness(params, M, lambda_penalty=1000):
    """Evaluate objective for block representation"""
    u = params[:M]
    v = params[M:]
    # Transform to valid widths and heights
    w = 2 * np.exp(u) / np.sum(np.exp(u))
    a = 1 / (1 + np.exp(-v))
    # Check integral constraint
    integral = np.sum(a * w)
    penalty = lambda_penalty * abs(integral - 1)
    # Sample to fine grid
    N = 200
    dx = 2 / N
    h = compute_overlap_from_blocks(a, w, N)
    # Compute correlation
    C = correlate(h, 1 - h, mode='full')
    max_corr = np.max(C)
    return max_corr + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    M = 12  # Number of blocks
    bounds = [(-5, 5)] * (2 * M)
    lambda_penalty = 1000
    
    # Global search with Differential Evolution
    de_result = differential_evolution(
        lambda x: evaluate_fitness(x, M, lambda_penalty),
        bounds,
        seed=seed,
        maxiter=30,
        popsize=15 * 2 * M,
        polish=False,
        init='latinhypercube'
    )
    
    # Extract best block solution
    u_best = de_result.x[:M]
    v_best = de_result.x[M:]
    w_best = 2 * np.exp(u_best) / np.sum(np.exp(u_best))
    a_best = 1 / (1 + np.exp(-v_best))
    
    # Re-normalize to ensure integral = 1
    integral = np.sum(a_best * w_best)
    a_best = a_best / integral
    a_best = np.clip(a_best, 0, 1)
    
    # Final sample on grid
    N = 200
    dx = 2 / N
    h = compute_overlap_from_blocks(a_best, w_best, N)
    
    # Local polishing with projected gradient descent
    max_iter = 4000
    alpha0 = 0.2
    h_history = [h.copy()]
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        # Compute correlations
        C = correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(C)
        max_corr = C[max_corr_idx]
        
        # Subgradient for the max shift
        k = max_corr_idx - (N - 1)
        # Compute gradient for C_k
        g = np.zeros_like(h)
        if k >= 0:
            g[:N-k] += (1 - h[k:])
        if k <= 0:
            g[-k:] -= h[:-k]
        
        # Update step
        alpha = alpha0 / np.sqrt(t + 1)
        h -= alpha * g
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        # Project onto simplex
        target_sum = N / 2
        h = project_to_capped_simplex(h, target_sum)
        
        if t % 100 == 0:
            h_history.append(h.copy())
    
    # Evaluate final solution
    final_C = correlate(h, 1 - h, mode='full')
    final_max_corr = np.max(final_C)
    c5_bound = final_max_corr * dx
    
    # Return best seen so far
    return (h.tolist(), c5_bound, N)
