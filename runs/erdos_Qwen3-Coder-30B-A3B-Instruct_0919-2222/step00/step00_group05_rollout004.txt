def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation-like integral
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the one we want:
    # max_k integral of h(x)(1 - h(x+k)) dx
    c5 = np.max(corr) * dx
    return c5
