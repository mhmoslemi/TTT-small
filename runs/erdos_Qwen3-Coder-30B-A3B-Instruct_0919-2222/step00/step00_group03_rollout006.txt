import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_C5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts k
    # Using numpy's correlate with mode='full'
    correlation = np.correlate(h, 1 - h, mode='full')
    # The full correlation has 2*n - 1 elements.
    # The valid shifts are from -n+1 to n-1.
    # We focus on shifts from 0 to n-1 for k = 0 to n-1 (corresponding to shifts)
    # But we need to compute max_k of integral h(x)(1-h(x+k)) over k.
    # So ignoring the negative shifts for now to compute our estimate
    # Actually, we should check all possible shifts in [0, 2]
    # Here we compute all correlations and take max
    # but we're limited to only valid shifts that fit within [0,2] domain
    max_val = 0.0
    for k in range(n):
        # shift k corresponds to correlation[k+n-1] only for meaningful overlaps
        val = correlation[k + n - 1] * dx
        if val > max_val:
            max_val = val
    return max_val

def initial_h_values(n_points):
    """Produce a good initial construction."""
    # A simple construction, can be changed based on better ideas
    # One approach uses a piecewise constant function with values near 0.5
    h = np.full(n_points, 0.5)
    # Ensure that sum(h) * dx = 1 => sum(h) = n_points / 2
    # So exact current sum = 0.5 * n_points, which is correct already.
    # This is a good starting point for optimization.
    return h

def objective(h):
    """Objective function to minimize. We maximize over k, so we minimize the max."""
    # Current implementation assumes we compute the full correlation matrix
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') 
    # Cut from -n+1 to n-1, but we only care about k in [0, n-1] corresponding to shifts in [0, 2]
    # The correlations correspond to shifts k=0 to 2n-2, so shift=0 to n-1 are valid for k=0 to n-1
    valid_shifts = corr[n-1:]  # Remove the first n-1 elements corresponding to negative shifts
    # Return negative of the maximum value (since we minimize)
    return -np.max(valid_shifts * dx)

def optimize_h(n_points, max_time_s):
    """Optimize the step function h using differential evolution."""
    # Time tracking
    start_time = time.time()
    
    # Initialize the function
    h_init = initial_h_values(n_points)
    h0 = h_init.copy()
    
    # Set up bounds for h values
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Important constraints:
    # The sum should be such that ∫₀² h(x) dx = 1
    # So sum(h) * dx = 1 => sum(h) = n_points / 2
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    # Convert to scipy constraint
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective
    def obj(x):
        return -evaluate_C5(x)  # Minimize negative of max overlap for better handling
    
    # Run optimization with time limits
    result = differential_evolution(
        func=obj,
        bounds=bounds,
        constraints=cons,
        maxiter=int(max_time_s / 0.01),  # Rough estimate
        popsize=10,
        seed=42,
        disp=False,
        polish=True
    )
    
    # Return best result
    final_h = result.x
    final_c5 = -result.fun
    
    return final_h, final_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    # Set number of points. We're aiming for something not too large.
    n_points = 70
    np.random.seed(seed)
    
    # Track time
    start_time = time.time()
    
    # Initial strategy: use the initial_h_values from imported module as starting point,
    # then refine using objective function.
    
    # Extract the initial h
    h = initial_h_values(n_points)
    best_h = h.copy()
    best_c5 = evaluate_C5(h)
    
    # Start optimization loop
    remaining_time = budget_s 
    iteration = 0
    while remaining_time > 5:
        iteration += 1
        try:
            # always run with the latest seed
            h_new, c5_new, _ = optimize_h(n_points, min(remaining_time, 10))
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception as e:
            # When there's an issue, just continue with the best solution
            pass
        
        remaining_time = budget_s - (time.time() - start_time)
    
    # Final evaluation
    final_c5 = evaluate_C5(best_h)
    
    # Check target
    if final_c5 < 0.3807:
        pass  # Might be good enough for the task
    
    # Final return
    return best_h, final_c5, n_points
