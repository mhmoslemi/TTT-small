def evaluate_overlap(h):
    """Evaluate all overlaps for a step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    overlaps = np.correlate(h, 1-h, mode='full')
    # Extract the relevant portion (shifts from -n+1 to n-1)
    overlaps = overlaps[n-1:2*n-1]
    # Convert to the actual integral values
    return max(overlaps) * dx

def rounding_step(h_frac, n_points):
    """Perform pairwise dependent rounding."""
    h = np.array(h_frac, dtype=float)
    while True:
        # Find indices with non-integral values
        non_integral = np.where((h > 0) & (h < 1))[0]
        if len(non_integral) < 2:
            break
        # Pick first two, fix their fractional part difference
        i1 = non_integral[0]
        i2 = non_integral[1]
        # Swap if sum <= 1
        d1 = h[i1]
        d2 = h[i2]

        # If both Fequal, arbitrary pick
        delta = min(d1, 1 - d2)
        if np.random.rand() > 0.5:
            h[i1] -= delta
            h[i2] += delta
        else:
            h[i1] += delta
            h[i2] -= delta
        # Ensure sum constraint
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-6:
            break
    return h
