import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def solve_erdos_milp(n, time_limit):
    prob = pulp.LpProblem("ErdosMinOverlapMILP", pulp.LpMinimize)
    
    # Variables
    h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
    T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
    
    # Constraint: sum h_i = n/2
    prob += pulp.lpSum([h[i] for i in range(n)]) == n/2

    # Product constraints and shift constraints
    for d in range(1, n):  # skip d=0
        # create list of y variables for this shift
        y_vars = []
        for i in range(n-d):
            y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
            y_vars.append(y)
            # constraints:
            prob += y <= h[i]
            prob += y <= h[i+d]
            prob += y >= h[i] + h[i+d] - 1
        # shift constraint: sum y >= n/2 - T
        prob += pulp.lpSum(y_vars) >= n/2 - T
    
    # Objective
    prob += T
    
    # Solve
    solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit)
    prob.solve(solver)
    
    # Check status
    status = pulp.LpStatus[prob.status]
    if status != 'Optimal':
        return None, None
    
    # Get solution vector
    h_vals = np.array([int(pulp.value(h[i])) for i in range(n)], dtype=float)
    t_val = pulp.value(T) * (2.0 / n)  # C5 bound = T*dx
    
    return h_vals, t_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    n_points = 0
    
    # Try different values of n
    n_values = [200, 180, 160, 140, 120, 100, 80]
    
    for n in n_values:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
            
        time_per_solve = min(remaining_time / 2, 300)  # Max 5 minutes per solve
        
        h_vals, c5_bound = solve_erdos_milp(n, time_per_solve)
        
        if h_vals is not None and c5_bound is not None:
            # Final verification with evaluate_erdos_solution
            computed_c5 = evaluate_erdos_solution(h_vals)
            
            if computed_c5 < best_c5:
                best_c5 = computed_c5
                best_h = h_vals.copy()
                n_points = n
                
            # Early exit if we already beat the target
            if best_c5 <= 0.3807:
                break
    
    # If no valid solution found, use initial construction
    if best_h is None:
        initial_n = len(initial_h_values)
        best_h = np.array(initial_h_values)  
        best_c5 = evaluate_erdos_solution(best_h)
        n_points = initial_n
    
    return best_h, best_c5, n_points
