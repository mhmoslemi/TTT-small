def evaluate_erdos_solution(h):
    """Evaluate the overlap constant C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations using full convolution
    h_full = np.concatenate([h, np.zeros(n - 1)])
    h_rev = np.concatenate([np.zeros(n - 1), h])
    conv = np.correlate(h_full, h_rev, mode='full')
    overlaps = conv[n-1:]  # O_k = sum h_i h_{i+k} for k=0 to n-1
    max_overlap = overlaps.max()
    # actual overlap integral is (sum h_i * (1-h_{i+k})) = (sum h_i) - (sum h_i h_{i+k}) = n*dx - overlaps
    # Since the integral is 2*dx * sum(h(i)(1-h(i+k))), for our normalization we use 2 times: 
    # the value of max_overlap, or rather the maximal value in (n*dx - overlaps) = n*dx - min(overlaps)
    integral_value = 1.0 - (max_overlap / n)  # This becomes the actual overlap measure scaled to [0,1] unit interval
    return 2.0 * integral_value


def generate_sturmian(n, alpha, tau):
    """Generate a Sturmian sequence of length n with slope alpha and phase tau."""
    a = np.array([1 if (i * alpha + tau) % 1 < 0.5 else 0 for i in range(n)])
    return a


def sturmian_search(n, seed=42):
    """Search for best Sturmian sequence parameters."""
    np.random.seed(seed)
    best_bound = float('inf')
    best_alpha = 0.5
    best_tau = 0.0
    
    # Grid search
    M = 200
    L = 100
    alpha_step = 0.6 / M
    tau_step = 1.0 / L
    
    for i in range(M):
        alpha = 0.2 + i * alpha_step
        for j in range(L):
            tau = j * tau_step
            h = generate_sturmian(n, alpha, tau)
            c5 = evaluate_erdos_solution(h)
            if c5 < best_bound:
                best_bound = c5
                best_alpha = alpha
                best_tau = tau
    
    # Refinement
    def obj_alpha(alpha):
        h = generate_sturmian(n, alpha, best_tau)
        return evaluate_erdos_solution(h)
    
    def obj_tau(tau):
        h = generate_sturmian(n, best_alpha, tau)
        return evaluate_erdos_solution(h)
    
    # Golden section search refinement on alpha
    res_alpha = minimize_scalar(obj_alpha, bounds=(0.2, 0.8), method='golden', options={'maxiter': 1000})
    if res_alpha.fun < best_bound:
        best_bound = res_alpha.fun
        best_alpha = res_alpha.x
    
    # Golden section search refinement on tau
    res_tau = minimize_scalar(obj_tau, bounds=(0.0, 1.0), method='golden', options={'maxiter': 1000})
    if res_tau.fun < best_bound:
        best_bound = res_tau.fun
        best_tau = res_tau.x
    
    # Final construction
    best_h = generate_sturmian(n, best_alpha, best_tau)
    
    return best_h, best_bound


def rounding_swap(h, tol=1e-10):
    """Round a fractional vector to binary, preserving sum."""
    h_rounded = np.round(h).astype(int)
    h_float = h.astype(float)
    total_sum = np.sum(h_float)
    goal_sum = int(round(total_sum))
    diff_sum = goal_sum - np.sum(h_rounded)
    
    # Ensure that diff_sum is 0 (i.e., the desired sum is already achieved)
    # If not, we must correct by making swaps
    
    # Simple restoration if necessary
    indices = np.where((h_float > 0) & (h_float < 1))[0]
    while diff_sum != 0 and len(indices) >= 2:
        i1, i2 = indices[0], indices[1]
        if diff_sum > 0:  # Round up some entries
            if h_float[i1] < 1 - h_float[i2]:
                h_rounded[i1] += 1
                diff_sum -= 1
            else:
                h_rounded[i2] += 1
                diff_sum -= 1
        else:
            if h_float[i1] > h_float[i2]:
                h_rounded[i1] -= 1
                diff_sum += 1
            else:
                h_rounded[i2] -= 1
                diff_sum += 1
        indices = np.where((h_float > 0) & (h_float < 1))[0]

    return h_rounded
