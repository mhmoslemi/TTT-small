def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)

    # Using convolution via numpy
    conv = np.correlate(h, 1 - h, mode='full')

    # But we only want the first n elements corresponding to k from 0 to 1
    # Actually, we want the maximum over all possible k (shifts)
    # In the convolution result: index n-1 corresponds to k=0, 
    # index n corresponds to k=dx, etc. 
    # So we want maximum among indices from n-1 to 2*n-2
    # All we actually need is max over all valid k values
    # Correct way to compute the integral for shift k:
    # We can compute it efficiently by defining shifts from 0 to 2
    # and evaluate at discretized points.
    # Our approach:
    # For each k=0 to 2 with step dx, compute:

    # Let's do it correctly: compute for each possible shift k from 0 to 2
    # But since we have finite samples, we simulate and get full range.
    # The said convolution is needed to compute maximum integral correctly
    
    # Let's just take the maximum of the full convolution result (correct values)
    max_val = np.max(conv) * dx
    return max_val
