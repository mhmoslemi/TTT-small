def evaluate_h(h):
    """Evaluate the overlap integral for a given h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # correlate h with (1-h) in full mode
    conv = signal.correlate(h, 1 - h, mode='full')
    # only take the valid overlaps (from k=-1 to k=1)
    # this gives n_points - 1 overlap integrals
    # the entry at index n_points - 1 corresponds to k=0, 
    # and entries at n_points - 1 + k and n_points - 1 - k correspond to k and -k
    # So we want the maximum of the last n_points - 1 entries
    # Since the kernel is flipped and symmetric, we only care about entries from
    # index n_points - 1 to 2*n_points - 2
    # But actually, we want entries where k goes from -1 to 1 (i.e., all entries except the first one)
    # since from the definition, we consider only the shift k that are in the interval [-1,1]
    # Let's compute which entries we need
    
    # full convolution of length 2*n_points - 1
    # the shift k = j - n_points + 1
    # where j runs from 0 to 2*n_points - 2
    # so k ranges from - n_points + 1 to n_points - 1
    # so we want k = -1, 0, 1 
    # means j = n_points - 2, n_points - 1, n_points
    # entries 0 to 2*n_points-2 
    # the corresponding shifts are from -n_points+1 to n_points-1
    
    # We want the maximum overlap among k = -1, 0, 1
    max_overlap = 0.0
    conv_val = conv * dx   # This applies dx scaling
    # map indices to shifts j -> k = j - n_points + 1
    # valid indexes for k in [-1, 0, 1]:

    # k = 0 corresponds to j = n_points - 1  
    # k = 1 corresponds to j = n_points

            conv_val[n_points - 2],  # Corresponds to k = -1
            conv_val[n_points - 1],  # Corresponds to k = 0
            conv_val[n_points]        # Corresponds to k = 1
        )
    return max_overlap
