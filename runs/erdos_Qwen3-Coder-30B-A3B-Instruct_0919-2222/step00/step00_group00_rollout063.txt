import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h_values):
    """Evaluate the C5 constant for given h values."""
    # Normalize so that sum(h) * dx = 1
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the overlap integral for all possible shifts k
    # This involves computing correlate(h, 1-h, mode="full") 
    # and finding the maximum value
    h = np.array(h_values)
    h_shifted = 1 - h
    # mode="full" gives us all overlapping regions
    corr = correlate(h, h_shifted, mode='full')
    # The correlation is computed for lags from -(n-1) to (n-1)
    # Our range is from 0 to 2, which corresponds to lag from 0 to 2* (n/2)
    # But since h is defined on [0,2], we only want lags in [0,2] equivalent to
    # the full range is n points long, so we use the appropriate segment
    # We're interested in the part that corresponds to the valid range [0,2]
    # Lag goes from -n+1 to n-1, where 0 corresponds to no shift
    # So we look at indices from n-1 to 2*(n-1), but we want the part from 0 to n-1:
    # actually, let's compute it carefully:
    # for lags from 0 to n+1 (in the full correlation): we want the first n values after the center
    # Actually for the range [0,2], we don't shift to the left. 
    # The maximum should be something over 0 to 2. So we'll compute from the full correlation
    # and look at relevant lags corresponding to intervals [0,2] 
    # If dx = 2.0/n and we align h on grid points, the 
    # mesh points are 0, dx, 2*dx, ..., (n-1)*dx
    # Integrate h(x) * (1-h(x+k)) over [0,2]
    # Dyadic grid: Let's simplify.
    # clamp the output to the overlap parts, (part of the full correlation that corresponds to our region)
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Construct intial h_values for optimization."""
    n = 45
    # uniform distribution, slightly adapted to be a valid initial h
    h0 = np.ones(n) * 0.5
    # If we don't need to force sum(h) * dx = 1, which we do,
    # we divide by n/2 * 2 to get sum = 1 with dx = 2/n
    # Let's be more careful and construct properly.
    return h0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial solution as starting point
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    def objective(h):
        # Normalize to have integral = 1
        scale = np.sum(h) * dx
        if scale > 0:
            h = h / scale * (0.5 * n_points)  # Correct so that sum(h)*dx = 1  
        # clip to [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_c5(h)
    
    def constraint_sum(h):
        # Constraint: integral = 1 
        return np.sum(h) * dx - 1.0
    
    # Wrap arguments for optimization
    def _objective(h):
        return objective(h)
    
    # Bounds for each h[i] in [0,1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # We explicitly keep the constraint that sum(h)*dx = 1
    # To make sure that it remains satisfied after optimization
    print(f"Starting optimization with {n_points} points")
    
    # Use differential evolution as it works well with the constraints
    # but still manage the integral conservation manually in objective
    best_c5 = float('inf')
    best_h = None
    n_evals = 0
    
    # First, evaluate the initial solution
    c5_initial = objective(h0)
    print(f"Initial C5: {c5_initial}")
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = h0.copy()
    
    # Now, perform optimization
    # Use differential evolution with bounds and a fixed number of iterations to keep under budget
    
    # Custom crude optimization loop to do multiple cycles without dill
    max_iterations = 1000  # <= budget_s * 50
    for i in range(max_iterations):
        try:
            if time.time() - start_time > budget_s - 1:
                break
            # Differential evolution with a small popsize
            # We'll call it manually with a small number of operations to be efficient
            # state needs to be managed inside the loop (avoid saving all history)
            sol = differential_evolution(
                func=_objective,
                bounds=bounds,
                seed=seed + i,
                maxiter=10,
                popsize=5,
                disp=False
            )
            c5 = sol.fun
            h = np.clip(sol.x, 0, 1)
            n_evals += 1
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                print(f"Improved C5: {best_c5} at iteration {i}")
                
        except Exception as e:
            print(f"Optimization error at iter {i}: {e}")
            break

    # Save the best solution so far
    if best_h is not None:
        # Ensure it has integral 1
        scale = np.sum(best_h) * dx
        if scale != 0:
            best_h = best_h / scale * (0.5 * n_points)  # Normalize correctly
        # Final clip
        best_h = np.clip(best_h, 0, 1)
        if abs(np.sum(best_h) * dx - 1.0) > 1e-4:
            print(f"Integrity warning: integral is not exactly 1, but {np.sum(best_h) * dx}")
        # Final evaluation
        final_c5 = evaluate_c5(best_h)
        print(f"Final C5: {final_c5}")
    else:
        print("No improvement found.")
        final_c5 = c5_initial
        best_h = h0
    
    return (best_h.tolist(), final_c5, n_points)
