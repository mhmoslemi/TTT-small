def evaluate_erdos_solution(h, dx=2.0/len(h)):
    """Evaluate the C5 bound for a given step function h."""
    if len(h) < 2:
        return float('inf')
    # Compute correlation with (1-h) using full mode
    corr = np.correlate(h, 1-h, mode='full')
    # Only use the valid shift range
    n = len(h)
    # The maximum is over all shifts; np.correlate includes full range
    # but we want the maximum integral value over valid shifts
    # The valid shifts for full correlation are from -(n-1) to (n-1)
    # However, to match continuous domain [0,2], let's just compute the max
    max_val = np.max(corr) * dx
    return max_val

def gradient_from_max_shift(h, dx=2.0/len(h)):
    """Gradient computation for optimization using subgradient."""
    n = len(h)
    # Compute all correlation values
    corr = np.correlate(h, 1 - h, mode='full')
    # corr[k] corresponds to shift index (k - (n-1))
    max_corr = np.max(corr)
    # Find all shift indices that give maximum correlation
    max_indices = np.where(np.abs(corr - max_corr) < 1e-10)[0]
    # Create gradient vector
    grad = np.zeros(n)
    # Process all worst-case shifts  
    for k_idx in max_indices:
        shift_offset = k_idx - (n - 1)  # Convert index to actual shift
        if shift_offset >= 0:
            # Positive shift: h[i] * (1 - h[i + shift_offset])
            # Partial indices: i from 0 to n-1-shift_offset  
            start_idx = 0
            end_idx = n - shift_offset
            # Gradient wrt h[i] is (1 - h[i + shift_offset])
            if end_idx - start_idx > 0:
                grad[start_idx:end_idx] += 1 - h[shift_offset:end_idx+shift_offset]
        else:
            # Negative shift: h[i] * (1 - h[i + shift_offset])  
            # here shift_offset is negative
            # So h[i-shift_offset] appears in product (h[i] * (1-h[i-shift_offset])
            # for i = |shift_offset| to n-1
            shift_abs = -shift_offset
            start_idx = shift_abs
            end_idx = n
            if end_idx - start_idx > 0:
                grad[start_idx:end_idx] -= h[:end_idx-shift_abs]
    # Normalize: since we're minimizing max(corr) * dx, we ignore dx for gradient
    # Also assume there is a projection/scale factor, like 1 for now
    return grad

def optimize_step_function(h, dx=2.0/len(h), max_iter=10000, alpha=0.01, early_stop=100):
    """Optimize a step function using subgradient descent."""
    n = len(h)
    target_sum = n / 2.0
    
    best_h = h.copy()
    best_bound = evaluate_erdos_solution(best_h, dx)
    
    t0 = time.time()
    
    for i in range(max_iter):
        if time.time() - t0 > 595: # Leave some safety margin
            break
        # Compute gradient
        grad = gradient_from_max_shift(best_h, dx) 
        # Update step size
        step_size = alpha / math.sqrt(i + 1)
        # Update h
        new_h = best_h - step_size * grad
        
        # Project to valid simplex
        projected_h = project_to_simplex(new_h, target_sum)
        
        # Compute new bound
        new_bound = evaluate_erdos_solution(projected_h, dx)
        
        if new_bound < best_bound:
            best_h = projected_h
            best_bound = new_bound
            
    return best_h, best_bound

def upsample_pattern(original_h, n):
    """Upsample a pattern to a longer array with same proportion."""
    old_len = len(original_h)
    t = n // old_len
    r = n % old_len
    # Repeat t times
    repeated = np.tile(original_h, t)
    # Add first r elements
    if r > 0:
        repeated = np.concatenate([repeated, original_h[:r]])
    # Normalize sum to n/2
    scale = (n / 2) / np.sum(repeated)
    return repeated * scale
