def evaluate_h(h):
    """Evaluate C_5 for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    # Using convolution to compute all correlations at once
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral for each k
    overlaps = conv * dx
    # C_5 is the maximum of these
    return np.max(overlaps)
