import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project_to_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum x = target_sum"""
    v = np.asarray(v)
    n = len(v)
    if n == 0:
        return np.array([])
    if np.isclose(np.sum(v), target_sum):
        return np.clip(v, 0, 1)
        
    # Sort values in descending order
    sorted_v = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_v)
    # Find threshold
    indices = np.arange(1, n+1)
    condition = sorted_v - (target_sum - cumsum) / indices > 0
    if not np.any(condition):
        # All projected values would be 0
        return np.zeros(n)
    k = np.argmax(condition)
    if k == 0:
        theta = sorted_v[0] - target_sum / n
    else:
        theta = (target_sum - cumsum[k-1]) / k
    # Generate projection
    x = np.maximum(0, np.minimum(1, v - theta))
    return x

def compute_block_overlap(a, w, dx, N_shifts=200):
    """Compute G(k) = ∫ h(x)*h(x+k) dx for k in [-N_shifts/2, N_shifts/2]"""
    M = len(a)
    # Compute interval boundaries
    s = np.concatenate([[0], np.cumsum(w)])
    # Samples of k: from -1 to 1 with step 2/N_shifts
    k_vals = np.linspace(-1, 1, N_shifts)
    # Samples of x on [0,2] with step dx (N grid points)
    x_vals = np.linspace(0, 2, int(2/dx), endpoint=False)
    # For each k, compute the integral ∫ h(x) h(x+k) dx
    G = np.zeros(N_shifts)
    for i_k, k in enumerate(k_vals):
        # Shift right by k (k can be positive or negative)
        # Interval overlap: For each i,j, compute overlap of [s[i-1], s[i]) with [s[j-1]+k, s[j]+k)
        corr = 0.0
        for i in range(M):
            for j in range(M):
                left_i = s[i]
                right_i = s[i+1]
                left_j = s[j] + k
                right_j = s[j+1] + k
                overlap = max(0, min(right_i, right_j) - max(left_i, left_j))
                if overlap > 0:
                    corr += a[i] * a[j] * overlap
        G[i_k] = corr
    return G, k_vals

def objective_with_penalty(params, M, lambda_penalty=2000):
    """
    Compute max(1 - G(k)) with penalty for integral constraint
    params: [u1,...,uM, v1,...,vM] where wi = 2*softmax(ui), ai = sigmoid(vi)
    """
    u = params[:M]
    v = params[M:]
    # Transform to constraints
    w = 2 * np.exp(u) / np.sum(np.exp(u))
    a = 1 / (1 + np.exp(-v))
    # Integral constraint ∑ a_i w_i = 1
    integral = np.sum(a * w)
    penalty = lambda_penalty * abs(integral - 1)
    
    # Compute G(k) for shift values using fast computation
    # Using N=200 (for compatibility) and compute the overlap properly
    dx = 2.0 / 200
    G, _ = compute_block_overlap(a, w, dx, N_shifts=200)
    f_values = 1 - G
    obj = np.max(f_values)
    total_obj = obj + penalty
    return total_obj

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    M = 12  # Number of blocks
    N = 200  # Final grid points
    dx = 2.0 / N
    
    # Constraint bounds for DE
    bounds = [(-5, 5)] * (2 * M)
    
    # Global DE optimization
    de_start = time.time()
    res = differential_evolution(
        objective_with_penalty,
        bounds,
        args=(M,),
        seed=seed,
        maxiter=20,
        popsize=15 * 2 * M,
        strategy='best1bin',
        polish=False,
        disp=False
    )
    if time.time() - start_time > budget_s * 0.9:
        # Fallback to specific cut, use context-aware conservative refinement
        pass
    
    # Extract best solution from DE
    u_opt = res.x[:M]
    v_opt = res.x[M:]
    w_opt = 2 * np.exp(u_opt) / np.sum(np.exp(u_opt))
    a_opt = 1 / (1 + np.exp(-v_opt))
    
    # Adjust to satisfy the exact integral constraint
    integral = np.sum(a_opt * w_opt)
    a_opt = a_opt / integral
    # Clamp between 0 and 1 if needed, and renormalize
    a_opt = np.clip(a_opt, 0, 1)
    integral_clamped = np.sum(a_opt * w_opt)
    a_opt = a_opt * (1.0 / integral_clamped)
    
    # Reconstruct h on a uniform grid
    h = np.zeros(N)
    s = np.concatenate([[0], np.cumsum(w_opt)])
    for j in range(M):
        start_idx = int(s[j] / dx)
        end_idx = int(s[j+1] / dx)
        if end_idx > start_idx:
            h[start_idx:end_idx] = a_opt[j]
    
    # Make sure the sum is correct
    h_sum = np.sum(h) * dx
    h = h * (0.5 / h_sum)
    
    # Final polishing with projected subgradient descent
    max_steps = 3000
    alpha_0 = 0.2
    current_h = h.copy()
    best_h = h.copy()
    best_c5 = 0.0
    step_count = 0
    while time.time() - start_time < budget_s - 5 and step_count < max_steps:
        # Compute full correlation with 1-h
        correlation = correlate(current_h, 1 - current_h, mode='full')
        # Compute shift values from correlation indices
        shift_len = len(correlation)
        k_vals = np.arange(shift_len) - (shift_len - 1) // 2
        # Evaluate max shift
        max_idx = np.argmax(correlation)
        max_corr = correlation[max_idx]
        
        # Compute gradient for the max corr shift
        k_opt = k_vals[max_idx]
        # Coordinates of the gradient w.r.t. h[i]
        g = np.zeros_like(current_h)
        if k_opt >= 0:
            g[:len(current_h)-k_opt] += (1 - current_h[k_opt:])
        else:
            k_opt_abs = -k_opt
            g[k_opt_abs:] += (1 - current_h[:len(current_h)-k_opt_abs])
        if k_opt <= 0:
            g[:len(current_h)+k_opt] -= (current_h[:k_opt])
        else:
            g[k_opt:] -= (current_h[:-k_opt])
        
        # Determine optimal step size
        alpha_t = alpha_0 / np.sqrt(step_count + 1)
        
        # Update
        new_h = current_h - alpha_t * g
        
        # Project onto feasible set: [0,1], sum = 0.5
        # Clamp
        new_h = np.clip(new_h, 0, 1)
        # Project to correct sum
        new_h = project_to_capped_simplex(new_h, N / 2)
        
        current_h = new_h
        step_count += 1
        
        # Update best if needed
        corr_new = correlate(current_h, 1 - current_h, mode='full')
        if np.max(corr_new) > best_c5:
            best_c5 = np.max(corr_new)
            best_h = current_h.copy()
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    max_corr_final = np.max(corr_final)
    c5_bound = max_corr_final * (2.0 / N)
    
    return(best_h.tolist(), c5_bound, N)
