import numpy as np

from scipy import signal

def compute_correlation(h):
    """Compute correlation c_k = sum h[i] * (1 - h[i+k]) for all valid shifts."""
    n = len(h)
    # Convolve h with (1 - h) using full mode
    c = np.correlate(h, 1 - h, mode='full')
    # c has length 2*n-1. Index k goes from -(n-1) to n-1.
    # We want the valid parts corresponding to actual shifts.
    return c

def smoothed_objective_and_weights(c, alpha):
    """Compute smooth log-sum-exp surrogate and weights."""
    c_max = np.max(c)
    if c_max == -np.inf:
        c_max = 0
    exp_c = np.exp(alpha * (c - c_max))
    Z = np.sum(exp_c)
    if Z == 0:
        w = np.ones_like(c) / len(c)
    else:
        w = exp_c / Z
    F_smooth = (1.0 / alpha) * (np.log(Z) + c_max)
    return F_smooth, w

def project_onto_feasible(y, target_sum):
    """Project vector y onto [0,1]^n and sum = target_sum using water-filling."""
    n = len(y)
    # Clip to [0,1]
    y_clipped = np.clip(y, 0, 1)
    current_sum = np.sum(y_clipped)
    if np.isclose(current_sum, target_sum, atol=1e-12):
        return y_clipped.copy()
    
    # Determine free nodes (those in (0,1))
    free_indices = np.where((y_clipped > 0) & (y_clipped < 1))[0]
    free_count = len(free_indices)
    
    delta = current_sum - target_sum
    
    iters = 0
    while abs(delta) > 1e-12 and iters < n * 2 and free_count > 0:
        adjust = delta / free_count
        # Adjust free variables
        for i in free_indices:
            # Check in bounds
            new_val = y_clipped[i] - adjust
            if new_val <= 0:
                # Move to boundary
                new_val = 0
                # Remove from free
                free_count -= 1
                free_indices = free_indices[free_indices != i]
            elif new_val >= 1:
                new_val = 1
                free_count -= 1
                free_indices = free_indices[free_indices != i]
            y_clipped[i] = new_val
        
        # Recompute delta
        delta = np.sum(y_clipped) - target_sum
        iters += 1

    # Final adjustment - redistribute
    if abs(delta) > 1e-12:
        # Distribute evenly among free to correct
        if delta > 0:
            # Reduce all free to hit target - spread excess
            adjust = delta / free_count
            for i in free_indices:
                new_val = max(0, y_clipped[i] - adjust)
                y_clipped[i] = new_val
        else:
            # Increase. Add to all free
            adjust = -delta / free_count
            for i in free_indices:
                new_val = min(1, y_clipped[i] + adjust)
                y_clipped[i] = new_val

    return y_clipped.copy()

def swap_update(c, h, i, j, off):
    """Incremental update correlation values c due to swapping h[i] and h[j]."""
    # Note: c assumes h is binary (0 or 1)
    # and length 2*n - 1.
    # c[off] = 0 (k=0) after init.
    # We compute delta contributions for shifts involving (i, j).
    
    # i is in ONES -> h[i] = 1, and j in ZEROS -> h[j] = 0.
    # After swap: h'[i] = 0, h'[j] = 1
    
    # For shift k:
    # c[k] += db[k] where db[k] = h'[i] * (1 - h'[i+k]) - h[i] * (1 - h[i+k])
    # But we update multiple indices: all positions i and i+k such that they are valid
    # and the effect needs to be computed.
    # That is:  
    # i+k exists? then add (h'[i+k] - h[i+k]) = (0 - 0) if we're not affecting...
    
    # More carefully:
    # we are swapping h[i]=1 and h[j]=0.
    # What shift contributions are affected?
    # For index p:
    # Original: c[k] += h[p] * (1 - h[p+k]) for all valid k 
    # Let delta(p,k) be change in c[k] due to change in h[p] by value 1.
    
    # delta(h[i]) = 1, in comments we say:
    # If h[i] changed from 0 to 1,  or 1 to 0,  
    # Then delta c[k] = h'[i] * (1 - h'[i+k]) - h[i] * (1 - h[i+k]) = (1 - h'[i+k]) - (1 - h[i+k])
    #                              = -h'[i+k] + h[i+k]
    # So change at forward shift i+k = -h'[i+k] + h[i+k]
    # So change is -1 (if h'[i+k] = 0, h[i+k] = 1)
    
    # Change = /* for shift index */,
    # since we now have h'[i] = 0, and want to exclude original contribution
    
    # Do entire array update:
    n = (len(c) + 1)//2
    delta_c = np.zeros_like(c)
    
    # For i and j:
    for p in (i, j):
        # compute all valid shifts
        # Update all c[shift] where there is a contribution from p
        # p contributes to c[k] when there is factor h[p] * (1 - h[p+k])
        # i contributes to ax[k] = h[p] * (1 - h[p+k]) where p+k < n
        # For shift k = i - p, we have (i-k) = p, so need factor h[p]*(1-h[p-k])
        # But here p is the shift source, not p+k
        # Change in c when p changes:
        # Original: h[p] * (1 - h[p+k]) -> c at shift k
        # If h[p] changes from 1 to 0:
        # New term is 0 -> difference is - h[p] * (1 - h[p+k]) = - (1 - h[p+k])  
        # Because h[p] = 1
        # Thus update c[k] by - (1 - h[p+k])
        # For p = i → h[i]=1, so we add - (1 - h[i+k]) to c[i+k]
        # For p = j → h[j]=0, so we add 0 to c[j+k]
        # But the total effect *on* c is due to h'[i] = 0 
        # Thus effectively subtracts i+* = h[i+k] (1 - h[i+*]) for all k.
        # Actually clarify:
        # Contribution: h[p] * (1 - h[p+k]) = h[p] * (1 - h[p+k])
        # if h[p] = 1:  Contribution = (1 - h[p+k])  
        #              = -h[p+k] + 1  
        #              = -h[p+k] + 1
    
        # Conduct precise update for shift k:

        # All moves must be inline with h[p] changing from 1 to 0 or vice versa
        # For example if p = i=1 (h[i] = 1), and we swap with j=2:
        # We want to compute change in c due to both elements 
        # if h[p] changes from 1 to 0 (small delta)
        # Start with d[i] = - (1 - h[i+k]) = 1 - h[i+k]
        if p == i:
            for k in range(-n+1, n):
                if 0 <= p + k < n:
                    delta_c[k + off] -= 1 - h[p + k]
        if p == j:
            for k in range(-n+1, n):
                if 0 <= p + k < n:
                    delta_c[k + off] += 1 - h[p + k]

    # Update in-place
    c += delta_c
    return c

def targeted_swap(h, c, off, time_left):
    """Perform a batch of targeted improve swaps."""
    n = len(h)
    ONES = np.where(h == 1)[0]
    ZEROS = np.where(h == 0)[0]
    
    if len(ONES) == 0 or len(ZEROS) == 0:
        return None, None
        
    s_star_idx = np.argmax(c)
    s_star = s_star_idx - off  # shift corresponding to max c
    
    # Collect candidate pairs: 
    # (i in ONES such that h[i+s_star] = 0) 
    # (j in ZEROS such that h[j+s_star] = 1)
    A = []
    B = []
    
    for i in ONES:
        if i + s_star >= 0 and i + s_star < n:
            if h[i + s_star] == 0:
                A.append(i)
    if len(A) == 0:
        # Fall back to any swap
        for i in ONES:
            for j in ZEROS:
                if j + s_star >= 0 and j + s_star < n:
                    if h[j + s_star] == 1:
                        A.append(i)
                        B.append(j)
                        break
            if len(A) > 0:
                break
        if len(A) == 0:
            return h, c  # no valid swap
    
    # Coerce to list
    A = list(A)
    B = []
    for j in ZEROS:
        if j + s_star >= 0 and j + s_star < n:
            if h[j + s_star] == 1:
                B.append(j)
    
    if len(B) == 0:
        A = []
        # Find valid pair
        B = []
        for i in ONES:
            for j in ZEROS:
                if i + s_star >= 0 and i + s_star < n:
                    if h[i + s_star] == 0:
                        A.append(i)
                        break
            if len(A) > 0:
                break
        if len(A) == 0:
            return h, c
    
    # Allow more robust way to find new pair for short runs
    if len(A) == 0 or len(B) == 0:  
        # If all iteration fails for fast cases, just swap random diff
        for i in ONES:
            for j in ZEROS:
                if i + s_star >= 0 and i + s_star < n:
                    if h[i + s_star] == 0:  # feed into 0
                        A.append(i)
                        B.append(j)
                        break
            if len(A) > 0:
                break

    to_swap_i = A[0]
    to_swap_j = B[0] if len(B) > 0 else ZEROS[0]  # fallback

    # Perform the swap
    h[to_swap_i], h[to_swap_j] = 0, 1
    # Update correlation delta
    
    c = swap_update(c, h, to_swap_i, to_swap_j, off)

    return h, c

def local_binary_search(h, c, best_val, time_left):
    """Run combinatorial refinement on binary h."""
    n = len(h)
    off = n - 1

    # Clone
    h = h.copy()
    c = c.copy()

    def compute_c_once():
        return compute_correlation(h)

    max_time = 600
    # Allow up to 10^5 swaps with adaptive loop
    for iter_count in range(100000):
        if time.time() - max_time > time_left:
            break
        
        old_max = np.max(c)
        h_old = h.copy()
        c_old = c.copy()
        # Perform targeted improvement swap
        h, c = targeted_swap(h, c, off, time_left - (time.time() - max_time))
        
        new_max = np.max(c)
        if new_max < old_max:
            # Accept improvement
            if new_max < best_val:
                best_val = new_max
        elif new_max > old_max:
            # Reject worsening
            h = h_old 
            c = c_old
        else:
            # Not changed, continue.
            pass
        
        # Occasionally accept same value with low prob
        if iter_count % 1000 == 0 and new_max == old_max and not np.array_equal(h, h_old):
            # maybe accept a small worsening?
            accept_prob = 0.01 * (1.0 - (iter_count / 100000.0))
            if np.random.random() < accept_prob:
                pass  # keep change
                
    return h, c

def upscale_binary(h_old, n_new):
    """
    Upsample binary solution using linear interpolation by repeating each value.
    Assumes h_old is fundamental, and we interpolate to n_new satisfying 2^k for now.
    """
    n_old = len(h_old)
    ratio = n_new // n_old
    if ratio <= 1:
        return h_old.copy()
    # This is a minimal naive upscale: repeat values
    h_up = np.repeat(h_old, ratio)
    if len(h_up) > n_new:
        h_up = h_up[:n_new]
    elif len(h_up) < n_new:
        h_up = np.pad(h_up, (0, n_new - len(h_up)), mode='constant', constant_values=0)
        
    return h_up

def run(seed=42, budget_s=600, **kwargs):
    """Run the optimization process for Erdős minimum overlap constant C5."""
    np.random.seed(seed)
    start_time = time.time()

    # List of n points to test
    ns = [80, 120, 180, 240]
    
    best_h = None
    best_val = np.inf
    best_n = None

    for it_n, n in enumerate(ns):
        dt = time.time() - start_time
        if dt > budget_s - 2:  # Safety margin
            break
            
        # Random initial solution
        h0 = np.random.rand(n)
        h0 = project_onto_feasible(h0, n / 2)
        
        current_h = h0.copy()
        current_alpha = 5.0
        current_eta = 0.5 * 2 / n  
        momentum = 0.9
        last_h = h0.copy()
        history = []

        # Max iterations and early stopping
        max_iter = 20000
        min_iter = 1000
        iter_count = 0

        # Break loop when alpha becomes too large, converge reliably, or time out
        max_alpha = 200
        conv_tol = 1e-6
        last_F_smooth = np.inf
        stable_count = 0

        # Adaptive decay
        decay_factor = 0.5
        while True:
            dt = time.time() - start_time
            if dt > budget_s - 2:
                break
                    
            if iter_count > max_iter:
                break
                
            # Compute correlation
            c = compute_correlation(current_h)
            F_smooth, weights = smoothed_objective_and_weights(c, current_alpha)

            # Compute gradient
            grad = grad_h(current_h, weights)
            
            # Momentum update
            v = momentum * (current_h - last_h) - current_eta * grad
            h_test = current_h + v
            
            # Prox
            h_proj = project_onto_feasible(h_test, n / 2)
            
            # Evaluate new objective
            c_test = compute_correlation(h_proj)
            F_smooth_test, weights_test = smoothed_objective_and_weights(c_test, current_alpha)

            # Backtracking line search
            retry_count = 0
            while retry_count < 3 and F_smooth_test > F_smooth:
                current_eta *= decay_factor
                h_test = current_h + v
                h_proj = project_onto_feasible(h_test, n / 2)
                c_test = compute_correlation(h_proj)
                F_smooth_test, weights_test = smoothed_objective_and_weights(c_test, current_alpha)
                retry_count += 1
            
            # Accept or reject and update
            if F_smooth_test <= F_smooth:
                current_h = h_proj.copy()
                F_smooth = F_smooth_test
                last_h = h_test.copy()
                stable_count = 0
            else:
                current_eta *= decay_factor

            # Update alpha \to increase tightness of surrogate
            if iter_count % 100 == 0 and current_alpha < max_alpha:
                current_alpha *= 1.2
            
            # Stop if converging
            rel_change = abs(F_smooth - last_F_smooth) / (abs(F_smooth) + 1e-12)
            last_F_smooth = F_smooth
            if rel_change < conv_tol:
                stable_count += 1
                if stable_count > 10 and iter_count > min_iter:
                    break
            
            iter_count += 1

        # Round to binary
        h_rounded = np.zeros(n)
        indices = np.argsort(current_h)[::-1]
        threshold = n // 2
        h_rounded[indices[:threshold]] = 1

        # Ensure exact sum
        h_rounded = project_onto_feasible(h_rounded, n / 2)

        # Final polishing
        c_final = compute_correlation(h_rounded)
        val_c5 = np.max(c_final) * 2 / n  # 2 is interval length

        # Perform binary local search if needed and have time
        t_remaining = budget_s - (time.time() - start_time)
        if t_remaining > 1.0:
            # Run local stochastic swaps
            h_final, c_final = local_binary_search(h_rounded, c_final, val_c5, t_remaining)
            # Final evaluation
            val_final = np.max(c_final) * 2 / n
        else:
            h_final = h_rounded
            val_final = val_c5

        if val_final < best_val:
            best_val = val_final
            best_h = h_final.copy()  
            best_n = n

        if best_val <= 0.379:
            break  # early exit needed

    # Return final best
    return best_h.tolist(), best_val, best_n
