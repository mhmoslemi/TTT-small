import numpy as np
import time
import random

def compute_correlation(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap(h, dx):
    c = compute_correlation(h)
    return c.max() * dx

def apply_delta(c, h, idx, delta):
    n = len(h)
    offset = n - 1
    # first term: for j from idx to n-1:
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # second term: for j from 0 to idx:
    for j in range(0, idx+1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(c, h, i, j):
    delta_i = 0 - h[i]
    delta_j = 1 - h[j]
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def get_candidate_indices(h):
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    return ones, zeros

def targeted_swap(c, h, ones, zeros):
    max_idx = np.argmax(c)
    max_shift = max_idx - (len(h) - 1)
    n = len(h)
    if max_shift == 0:
        return None
    a = []
    b = []
    for i in ones:
        j = i + max_shift
        if 0 <= j < n and h[j] == 0:
            a.append(i)
    for j in zeros:
        i = j - max_shift
        if 0 <= i < n and h[i] == 1:
            b.append(j)
    if a and b:
        i = random.choice(a)
        j = random.choice(b)
        return i, j
    return None

def hill_climb(h, n_iter, start_time, budget_s, ones=None, zeros=None):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h)
    current_max = c.max() * dx
    best_h = h.copy()
    best_max = current_max
    best_c = c.copy()
    
    if ones is None or zeros is None:
        ones, zeros = get_candidate_indices(h)
    
    iter_count = 0
    
    while iter_count < n_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        
        if iter_count % 1000 == 0:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.95:
                break

        # Try targeted swap first
        swap_indices = targeted_swap(c, h, ones, zeros)
        if swap_indices:
            i, j = swap_indices
            swap(c, h, i, j)
            current_max = c.max() * dx
            if current_max < best_max:
                best_h = h.copy()
                best_max = current_max
                best_c = c.copy()
            # Update index sets
            if h[i] == 0:
                ones = np.setdiff1d(ones, [i])
                zeros = np.append(zeros, [i])
            if h[j] == 1:
                zeros = np.setdiff1d(zeros, [j])
                ones = np.append(ones, [j])
        else:
            # Fall back to random swap
            if len(ones) > 0 and len(zeros) > 0:
                i = random.choice(ones)
                j = random.choice(zeros)
                swap(c, h, i, j)
                current_max = c.max() * dx
                if current_max < best_max:
                    best_h = h.copy()
                    best_max = current_max
                    best_c = c.copy()
                # Update index sets
                if h[i] == 0:
                    ones = np.setdiff1d(ones, [i])
                    zeros = np.append(zeros, [i])
                if h[j] == 1:
                    zeros = np.setdiff1d(zeros, [j])
                    ones = np.append(ones, [j])
        iter_count += 1

    return best_h, best_max, best_c

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    
    # Define possible n values to try
    n_values = [80, 100, 120, 140]
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    for target_n in n_values:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.9:
            break
            
        dx = 2.0 / target_n
        h = np.zeros(target_n)
        
        # Initialize h to a binary pattern with exactly m = n/2 ones
        m = target_n // 2
        ones_indices = np.random.choice(target_n, size=m, replace=False)
        h[ones_indices] = 1
        
        # Run hill climb
        iter_limit = min(max(10000, int(50000 * budget_s / 600)), 100000)
        try:
            h_best, c5, c_best = hill_climb(h, iter_limit, start_time, budget_s)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_best.copy()
                best_n = target_n
        except:
            pass
            
        if best_c5 <= 0.379:
            break
    
    return (best_h.tolist(), best_c5, best_n)
