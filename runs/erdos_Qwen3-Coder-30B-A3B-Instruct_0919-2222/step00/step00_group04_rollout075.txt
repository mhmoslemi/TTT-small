        return max_corr * dx
    
    def objective(h):
        return evaluate(h)
        
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: integral = 1
    def constraint_integral(h):
        return np.sum(h) * dx - 1.0
    
    constraints = [{'type': 'eq', 'fun': constraint_integral}]
    
    # Optimization parameters
    method = 'SLSQP'
    
    best_h = h0.copy()
    best_value = evaluate(h0)
    start_time = time.time()
    
    # Use a simple local search by random perturbations
    # track current best
    current_h = h0.copy()
    current_value = evaluate(h0)
    
    # perform optimization 
    epoch = 0
    while time.time() - start_time < budget_s - 1:
        epoch += 1
        # make a small random change
        new_h = current_h.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.05)
        new_h[idx] = np.clip(new_h[idx] + delta, 0, 1)
        
        # rescale to maintain integral constraint
        old_sum = np.sum(current_h)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * (old_sum / new_sum)
            
        # enforce integral constraint strictly
        scale_factor = 1.0 / new_sum * (n_points / 2.0)
        new_h = new_h * scale_factor
        
        new_value = evaluate(new_h)
        if new_value < current_value:
            current_h = new_h.copy()
            current_value = new_value
            
        # update global best
        if current_value < best_value:
            best_h = current_h.copy()
            best_value = current_value
    
    # Final refinement with scipy
    # We use a much more limited optimization to avoid spending too much time
    # just try L-BFGS-B with the current best solution
    optimization_result = minimize(objective, best_h, bounds=bounds, 
                                   constraints=constraints, method='L-BFGS-B', 
                                   options={'maxiter': 200, 'ftol': 1e-8})
    
    if optimization_result.success:
        refined_h = optimization_result.x
        final_value = evaluate(refined_h)
        if final_value < best_value:
            best_h = refined_h
            best_value = final_value
    
    return (best_h, best_value, n_points)
