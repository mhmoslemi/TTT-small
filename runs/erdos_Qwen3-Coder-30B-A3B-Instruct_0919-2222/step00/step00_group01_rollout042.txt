import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap problem constant C5 for given step function h"""
    # Discretization
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Compute correlation using full mode (equivalent to periodic convolution)
    # h * (1 - h) with full overlap
    corr = signal.correlate(h, 1 - h, mode='full')
    
    # Max of correlation scaled by dx
    c5 = np.max(corr) * dx
    
    return c5

def initial_h_values():
    # Load from global (predefined here for completeness)
    n = 47
    # Initialize with known good values (approximation)
    dx = 2.0 / n
    x = np.arange(n) * dx + dx/2
    # Modified triangular-like shape from literature studies
    h = np.maximum(0, 1 - np.abs(x - 1))
    # Normalize sum to exactly n/2  
    h = h * (n/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    end_time = start_time + budget_s * 0.99  # Leave margin
    
    # Parameters
    n_points = 501  # For resolution while staying under 1000
    dx = 2.0 / n_points
    
    # Initialize: start with best known shape from literature with small variation
    x = (np.arange(n_points) + 0.5) * dx  # Midpoint positions
    
    # Start with smooth cosine-squared shape, then optimize further
    h = np.cos(np.pi/2 * (x - 1)) ** 2
    # Ensure values in [0,1] 
    h = np.clip(h, 0, 1)
    # Normalize to exact sum = n_points/2  
    h = h * (n_points/2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Evaluate current best
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Parameters for optimization
    max_iter = 200000
    iter_count = 0
    epsilon = 0.05
    
    random_moves_per_iter = 400  # Per iteration
    
    while iter_count < max_iter and time.time() < end_time:
        # Perform multiple random moves per iteration
        for _ in range(random_moves_per_iter):
            # Randomly choose 2 indices to move mass
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            
            # Skip if same index
            if i == j:
                continue
                
            # Compute max delta we can move from i to j
            max_delta = min(h[i], 1 - h[j])
            if max_delta <= 0:
                continue
                
            # Propose move
            delta = np.random.random() * epsilon * max_delta
            
            # Create new h
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            
            # Ensure bounds
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize to sum = n_points/2
            h_new = h_new * (n_points/2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
            # Evaluate
            new_c5 = evaluate_erdos_solution(h_new)
            
            # Accept improvement
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
            
            iter_count += 1
                
        # Gradually decrease epsilon
        epsilon = max(0.001, epsilon * 0.99995)
        
        if iter_count % 2000 == 0:
            if time.time() > end_time:
                break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
