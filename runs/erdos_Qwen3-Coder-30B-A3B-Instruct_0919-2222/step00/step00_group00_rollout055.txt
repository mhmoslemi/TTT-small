def compute_correlation(h):
    return signal.correlate(h, 1 - h, mode='full')

def compute_smoothed_objective(c, alpha):
    max_c = np.max(c)
    shifted = c - max_c
    exp_shifted = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_shifted)
    return (1.0 / alpha) * np.log(sum_exp) + max_c

def compute_weights(c, alpha):
    max_c = np.max(c)
    shifted = c - max_c
    exp_shifted = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    return w

def compute_gradient(h, w):
    n = len(h)
    grad = np.zeros_like(h)
    # Precompute 1 - h and h
    one_minus_h = 1 - h
    # Forward contribution
    forward = np.convolve(one_minus_h, w, mode='full')
    # Backward contribution
    backward = np.convolve(h, w, mode='full')
    
    i_left = (n-1) - 0
    i_right = (n-1) + (n-1)
    
    start_idx = 0
    weight_idx_start = start_idx
    weight_idx_end = min(len(w), i_left + 1)
    for i in range(n):
        j_start = max(0, i - (n-1))
        j_end = min(n-1, i + (n-1))
        # Do for each contribution from w
        # current forward contribution = w shifted by (i, j)
        # need to compute sum over k of w[k] * (1 - h[i+k]) 
        # if we build this manually, it's brutal...
        # So combined correction:
        # but let's fix that IndexError from convolution porting
        # If you prefer correct indexing:
        # The convolution approach uses full mode, which includes indexes from -n+1 to 2*n-2
        # we want loop over i from 0 to n-1 and calculate the contribute from each shift k
        offset = n - 1
        # i-th element in forward convolves w with (1-h) at i+shift and the weight w[shift]
        # forward will be [0 ... 2*n-2], and we need sum for shift k (k from 0 to 2*n-2) at location i
        # Need to get the contribution of weights to the part corresponding to one_minus_h
        # Index in convolution result: result[i + offset] has the same meaning as sum_{j=0}^{n-1} (1 - h[i+(j-0)] ) * w[j]
        # So 
        # For i fixed: take weights w_{i+k} * (1 - h[i+k]) for valid k
        # Convolution result at index j in 1-n: inline computation useful
        # The match for dependencies: 
        #  Forward: start from 0, go to 2*n-2. For a fixed shift k from 0 to n-1, for each i in [0, min(n-1, n-1-k)]
        # But this is forward shift of k, so: sum_{j=0}^{n-1-k} w[k] (1-h[i+j+k]) = w[k] * sum_{j=0}^{n-1-k} (1-h[i+j+k])
        # We will do this by processing the convolution of w * (1-h).
        # Result convوى (1-h) gives value at index j - shifted by (n-1), then offset by that.
        # Let us compute convolution additional indices before hardcoding
        # within loop we take
        # shift is from (n-1) ... (2*n-2) - (n-1) = (0 ... n-1) 
        # So for some fixed i (i=0 to n-1), we take all the convolution contribs:
        # For step k from 0 to n-1, we take w[k] * (1 - h[i+k]) 
        # That means, we are summing for indices (0 to n-1) how many amount of w[shift] * (1 - h[i + shift])
        # So we reuse forward_convolve = w * (1 - h)
        # index: can iterate over forward_convolve indices.
        # The real f[i] = sum_{k=0}^{n-1} w[k] * (1-h[i+k]) for valid range of k (i+k<n)
        # Let the shift index be from 0 to 2*n-2, we translate it to proper phys index by adding (n-1)
        # index in forward result = shift_idx - (n-1) = (i + k)  
        # such that i+k = shift_idx - (n-1)  => k = shift_idx - (n-1) - i  
        # We know shift_idx ranges from 0 to 2*n-2, subtract (n-1) for i+k
        # With this logic, we can get all forward terms can be indexed via forward_convolve 
        # for one i,put up to (n-1) terms, in a separate loop over k
        # Perhaps a bit confusing below. 
        # Let's just compute this the direct sum
        # Let's proceed as given the convolution result is in correct order,  
        # we need hook up with appropriate index  

        # Alternatively, we already have the full convolution, but shift it by (n-1) to match the shift index in our op
        #    forward_convolve = np.convolve(w[0], 1 - h) i.e., Do [w] * (1-h)
        # But we want to index correlation value i (where our i-th element of h)*k 
        # e.g., if k = 0, then [0,0, ... index i], if k=1, the (0,0, ..., index i+1) ...
        # So in result "forward" at position k (converting index convention), we have 
        # sum over j w[j] * (1-h[i + j] )   
        # That's OR, conv matched by should only take part from position (0 to min(n-1, range))
        # A simpler approach: avoid numpy particular indexing, material: compute gradients
        # build convolution using a manual loop:
        # we compute the contribution of each 1-h[i+k]*w[k] that is valid (exist) for k and i
        # Expected: the gradient is to be computed at point i
        g = 0
        # since right is i + (n-1) terms, index j is 0 to n-1 => k is 0 to n-1
        # For each k and i, the term is w[k] * (1 - h[i+k]) if i+k < n
        for k in range(0, n):
            if i + k < n:
                g += w[k] * one_minus_h[i + k]
        # match with backward which is  sum over k of  w[k] * h[i-k]  
        # we have shift k from 0 to n-1 => i-k from 0 to n-1 (so k from i to n-1)  
        # An i-k loop over k where we advance from i to n-1  
        for k in range(0, n):
            if i - k >= 0:
                g -= w[k] * h[i - k]
        grad[i] = g

    return grad

def project_onto_simplex(y, target_sum):
    # assume y has no NaN, inf
    # Project onto the set {x in R^n | 0 <= x[i] <= 1, sum x[i] = target_sum} 
    n = len(y)
    low = 0.0
    high = 1.0
    
    # Clip
    clipped = np.clip(y, low, high)
    
    # Compute diff from target
    diff = np.sum(clipped) - target_sum
    if abs(diff) < 1e-10:
        return clipped.copy()
        
    # Waterfilling
    # sort in descending order - process largest first, decreasing
    sorted_idx = np.argsort(clipped)[::-1]  # indices of sorted in descending order
    sorted_vals = clipped[sorted_idx]
    
    # Determine how many we can influence
    # Start at all values (if sum > target, want to decrease values, below if we want increase)
    # For values below upper bound (1.0), we can set equal to lower bound
    # Start from the highest clipped value
    # Total mass is sum.
    # Now process each desc, compute how much mass is left after setting rights to bounds
    total = 0.0
    for i in range(n):
        total += sorted_vals[i]
        if total == target_sum:
            break
        if total > target_sum: 
            # Previous one was the last valid we can set all below to be equal row
            # the surplus amount was (total - target_sum) 
            # We have values from index i to n-1 which will be adjusted equally
            left = target_sum - (total - sorted_vals[i])
            if i < n:
                # distributed mean to batch
                mean_adjust = left/(n-i)
                for j in range(i, n):
                    clipped[sorted_idx[j]] = mean_adjust
                return clipped
        # don't do anything that makes life simpler
    
    # Check one last pass final accrual
    return clipped

def compute_c_sums(h):
    # Compute sum of (1-h) from i to end, and sum of h from 0 to i  
    # For gradient computation all these access in loop 
    n = len(h)
    forward = np.flip(np.cumsum(np.flip(1-h)))
    backward = np.cumsum(h)
    
    return forward, backward

def stable_log_sum_exp(x):
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    return np.log(np.sum(exp_x)) + max_x

def local_binary_search(h_bin, c, best_val, time_left):
    start = time.time()
    n = len(h_bin)
    # Sets for performance
    ones = set(i for i, val in enumerate(h_bin) if val > 0.5)
    zeros = set(i for i, val in enumerate(h_bin) if val < 0.5)
    c_max_idx = np.argmax(c)
    # Do not mutate input directly
    
    # We will compare a number of swap operations and retain the ones improving the max.
    # Reuse variance to avoid allocating memory
    max_c = c[c_max_idx]
    n_iter = 0
    no_improvement = 0
    # Could also implement simulated-annealing scheme
    while time.time() - start < time_left - 1:
        full_iter = len(ones) * len(zeros)
        if full_iter == 0: break
        # During iteration process get random batch of swaps or a few quick ones that work
        via_fast = True
        if via_fast:
            # Target worst shift:
            # Grab mismatch pairs that can reduce it
            mismatched_ones = set()
            mismatched_zeros = set()
            for i in ones:
                if i + c_max_idx < n and h_bin[i + c_max_idx] == 0:
                    mismatched_ones.add(i)
                if i - c_max_idx >= 0 and h_bin[i - c_max_idx] == 0:
                    mismatched_ones.add(i)
            
            for j in zeros:
                if j + c_max_idx < n and h_bin[j + c_max_idx] == 1:
                    mismatched_zeros.add(j)
                if j - c_max_idx >= 0 and h_bin[j - c_max_idx] == 1:
                    mismatched_zeros.add(j)
                
            try:
                # Pick from ones that contribute 1 to max shift, and zeros that flip to 1  
                i = random.choice(list(mismatched_ones))
                j = random.choice(list(mismatched_zeros))
                # Perform swap change:
                # constants needed for update:
                old = c[c_max_idx]
                delta = 0
                # Naively: 
                # Full function change is small
                for q in range(0, n):
                    if q == i or q == j: continue
                    shift = q - i
                    if 0 <= shift + n - 1 < 2*n-1:
                        delta -= h_bin[q] * (1 - h_bin[q + shift])
                    if q == i:
                        delta += 1 * (1 - h_bin[q + shift])
                        continue
                    shift = i - q
                    if 0 <= shift + n - 1 < 2*n-1:
                        delta -= h_bin[q] * (1 - h_bin[q - shift])
                    if q == i:
                        delta += 1 * (1 - h_bin[q - shift])
                        continue
                        
                # Perform swap (set back to h_bin)
                h_tmp = h_bin.copy()
                h_tmp[i] = 0
                h_tmp[j] = 1
                new_c = compute_correlation(h_tmp)
                # Update sets
                ones.discard(i)
                ones.add(j)
                zeros.discard(j)
                zeros.add(i)
                
                # Check new max
                new_max_c = np.max(new_c)
                new_val = new_max_c * 2.0 / n
                # Accept if not worse, or accept sometimes otherwise
                if new_max_c <= old: 
                    # Better locally
                    h_bin = h_tmp
                    c = new_c
                    max_c = new_max_c
                    ones.discard(i)
                    ones.add(j)
                    zeros.discard(j)
                    zeros.add(i)
                    no_improvement = 0
                else: 
                    # Might be a bad acquisition and also reject
                    # Try some small chance: AA similar to simulated annealing via exponential probabilities.
                    # accept a worsening swap 1% chance or decreasing likelihood.
                    if random.uniform(0, 1) < 0.01:
                        h_bin = h_tmp
                        c = new_c
                        max_c = new_max_c
                        # Update sets? from above?
                        ones.discard(i)
                        ones.add(j)
                        zeros.discard(j)
                        zeros.add(i)
                        no_improvement = 0
                    else: 
                        no_improvement += 1
            except Exception:
                pass  # fallback to random unpairs
        if random.uniform(0, 1) < 0.1 or no_improvement > 5:
            # Random local swap to move around
            if len(ones) == 0 or len(zeros) == 0: break
            i = random.choice(list(ones))
            j = random.choice(list(zeros))
            h_tmp = h_bin.copy()
            h_tmp[i] = 0
            h_tmp[j] = 1
            new_c = compute_correlation(h_tmp)
            new_val = np.max(new_c) * 2.0 / n
            if new_val <= best_val:
                h_bin = h_tmp
                ones.discard(i)
                ones.add(j)
                zeros.discard(j)
                zeros.add(i)
                c = new_c
                max_c = np.max(new_c)
                no_improvement = 0
                best_val = new_val
            else:
                no_improvement += 1
        if no_improvement > 5:
            # Add global check 1 time
            # generalized idea that this fails, and all you have left is property of change of local space
            break
    
    return h_bin, np.max(c) * 2.0 / n, best_val

def upscale_and_retain(h, n_new):
    """ Upsample h from n_new // 2 to n_new by linear interpolation of binary values """
    h_new = np.zeros(n_new)
    step = n_new // len(h)
    if step == 0:
        step = 1
    for i, val in enumerate(h):
        for j in range(i * step, min((i + 1) * step, n_new)):
            h_new[j] = val
    return h_new

def run(seed=42, budget_s=600):
    # Initialize time
    t0 = time.time()
    random.seed(seed)
    np.random.seed(seed)
    
    # Initialize variables
    best_h = None
    best_val = float('inf')
    n_best = None
    
    # Grid: smaller, faster to converge, then upscale.
    # By design choice to keep it within 1000 points; go only to 240
    ns = [80, 120, 180, 240]
    
    # Strategy: start from initial_h_values
    initial = initial_h_values
    
    # Iterate 
    for i, n in enumerate(ns):
        if time.time() - t0 > budget_s - 2:
            break
        print(f"Working on n={n}")
        # Branch to first run on coarse set:
        # Initial matrix: For resolution n compute using b-spline interpolation of the coarse point h, or use from initial_ductive ones if provided.
        if n == ns[0]:
            h = initial[:n].copy()
            if len(h) < n:
                step = n // len(h)
                h_up = np.zeros(n)
                for i, ele in enumerate(h):
                    for j in range(step):
                        if i * step + j < n:
                            h_up[step * i + j] = ele
                h = h_up
        else:
            # Interpolate the previous best binary solution:
            h = upscale_and_retain(best_h, n)
        
        # Phase 1: Gradient descent with projected updates
        # Small buffer time to not run out
        time_left = budget_s - (time.time() - t0) - 2
        if time_left <= 2:
            break
        max_iter = 10000
        alpha = 5.0
        alpha_increase_iter = 500
        momentum = 0.9
        learning_rate = 0.5 * 2.0 / n
        prev_h = h.copy()
        prev_grad = np.zeros_like(h)
        c = compute_correlation(h)
        max_c_val = np.max(c)
        prev_obj = compute_smoothed_objective(c, alpha)
        improvement = 0.01
        
        # Main loop:
        for iter in range(max_iter):
            # Time check
            if time.time() - t0 > budget_s - 2:
                break
            if iter % 500 == 0:
                print(f"  Iteration {iter}, alpha={alpha}")
                
            c = compute_correlation(h)
            w = compute_weights(c, alpha)
            grad = compute_gradient(h, w)
            # Combined with momentum:
            new_h = h - learning_rate * (grad + momentum * prev_grad)
            
            # Project ont s: box and sum constraints
            new_h = project_onto_simplex(new_h, n / 2)
            # Force exact
            new_h = np.clip(new_h, 0, 1)
            new_h_sum = np.sum(new_h)
            if abs(new_h_sum - n/2) > 1e-10:
                # Adjust to exact
                diff = new_h_sum - (n / 2.0)
                # We adjust a few components to fix whole sum
                # Use 2 finest components (or adjust all, but slow), make insertions small
                # Simple: only one
                idx = np.argmax(new_h[0:10]) # Most positive first
                compensation = (n / 2.0) - np.sum(new_h)
                new_h[idx] += compensation / (n - idx) # Not overwrite exacts...
            
            # Objective check
            new_c = compute_correlation(new_h)
            new_obj = compute_smoothed_objective(new_c, alpha)
            # Try update only if objective actually decreased (reduced original evaluation object  as possible)
            if new_obj <= prev_obj:
                h = new_h.copy()
                prev_grad = grad.copy()
                prev_obj = new_obj
                old_alpha = alpha
                # Increase alpha every few steps, i.e., tighten to max
                if iter > 0 and iter % alpha_increase_iter == 0:
                    alpha *= 1.2
                if alpha > 200:
                    alpha = 200  # Stop increasing if too much
            else: 
                # Decrease step if failed updates
                learning_rate *= 0.5
                if learning_rate < 1e-10:
                    learning_rate = 1e-10  # Reset if under threshold
                    
            # Check for numerical gains or stops fitting
            if abs(prev_obj - new_obj) < 1e-6:
                improvement += 1
                if improvement > 20:
                    break  # Early stop
            else:
                improvement = 0
            
        # Phase 2: Binary rounding to step functions
        # set top n/2 largest to 1, rest to 0
        h_sorted_idx = np.argsort(h)[::-1]
        h_default = np.zeros_like(h)
        for i in range(len(h)//2):
            h_default[h_sorted_idx[i]] = 1.0
        h_bin = h_default
        
        # Full objective
        corr_full = compute_correlation(h_bin)
        obj_full = np.max(corr_full) * 2.0 / len(h_bin)
        
        if obj_full < best_val:
            best_val = obj_full
            best_h = h_bin
            n_best = len(h_bin)
            
        # Phase 3: Local polishing: Swap-based refinement on binary grid
        # Note: compute time
        local_time_left = budget_s - (time.time() - t0) - 2
        if local_time_left > 1:
            best_h, val_final, _ = local_binary_search(h_bin, corr_full, obj_full, local_time_left)
            if val_final < best_val:
                best_val = val_final
                n_best = len(best_h)
    
    return best_h.tolist(), best_val, n_best
