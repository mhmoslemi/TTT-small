import numpy as np
import time
from scipy import optimize

def evaluate(h, dx):
    # Compute the correlation of h with (1-h) using FFT for efficiency
    # h is assumed to be normalized
    h = np.array(h)
    h1 = h
    h2 = 1 - h
    # full convolution
    correlation = np.correlate(h1, h2, mode="full")
    # For a step function, max index is len(h1) + len(h2) - 2 = 2*n - 2
    max_corr = np.max(correlation)
    return max_corr * dx

def project_onto_constraints(h, dx):
    # Project h onto [0,1] and satisfy ∫ h = 1
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    if abs(total) < 1e-10:
        h = np.ones_like(h) / (len(h) * dx)
    else:
        h = h * (1.0 / total)
    return h

def generate_random_feasible(n_points, dx):
    # Generate a random vector and project
    h = np.random.rand(n_points)
    h = project_onto_constraints(h, dx)
    return h

def take_step(h, dx, step_size=0.02):
    # Make a small random step
    h = h.copy()
    i = np.random.randint(len(h))
    delta = np.random.uniform(-step_size, step_size)
    h[i] += delta
    return project_onto_constraints(h, dx)

def simulated_annealing(n_points, dx, initial_h, max_time_s):
    best_h = initial_h.copy()
    best_value = evaluate(best_h, dx)
    current_h = initial_h.copy()
    current_value = best_value
    temperature = 0.05
    cooling_rate = 0.9999
    start_time = time.time()
    step_count = 0

    while time.time() - start_time < max_time_s:
        # Generate neighbor by perturbing one point
        new_h = take_step(current_h, dx)
        new_value = evaluate(new_h, dx)
        delta = new_value - current_value

        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            current_h = new_h
            current_value = new_value
            if current_value < best_value:
                best_h = current_h.copy()
                best_value = current_value
        temperature *= cooling_rate
        step_count += 1
        if step_count % 5000 == 0:
            if time.time() - start_time > max_time_s * 0.95:
                break
    return best_h, best_value
