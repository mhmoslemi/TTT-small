import numpy as np
import time
import random

def evaluate_erdos_solution(h, dx=2.0 / len(h)):
    """
    Computes the maximum overlap integral C5 for given step function h
    """
    # Compute correlation using np.correlate with mode='full'
    correlation = np.correlate(h, 1 - h, mode='full')
    # Scale by dx
    correlation *= dx
    # Return maximum value
    return np.max(correlation)

    """
    Optimizes a step function h: [0,2] -> [0,1] to minimize C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    """
    # Initialize random seed
    random.seed(seed)
    np.random.seed(seed)

        # Generate random feasible solution
        h = np.random.uniform(0, 1, n_points)
        # Adjust to make sum = n_points / 2
        h *= (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)

    # Ensure that sum is exactly n_points / 2
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-10:
        # Adjust values to make the sum exactly n_points/2
        diff = (n_points / 2) - h_sum
        if n_points > 1:
            h[0] += diff
            h = np.clip(h, 0, 1)

    # Initialize parameters
    best_h = h.copy()
    dx = 2.0 / n_points
    current_val = evaluate_erdos_solution(h, dx)
    best_val = current_val

    start_time = time.time()
    iteration = 0
    T = 0.05
    cooling_rate = 0.99995
    min_temp = 1e-8
    step_size = 0.02
    max_iterations = 1000000

    # Simulated annealing loop
    while time.time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1

        # Random move: swap a small amount of mass between two indices
        i = random.randint(0, n_points - 1)
        j = random.randint(0, n_points - 1)
        
        # Ensure i != j and choose a small amount to move
        if i == j:
            j = (j + 1) % n_points
            
        # Determine the maximum amount to shift
        delta = min(h[i], 1 - h[j], step_size)
        if delta < 1e-10:
            continue
            
        # Apply move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)

        # Evaluate new solution
        new_val = evaluate_erdos_solution(h_new, dx)

        # Accept or reject with simulated annealing probability
        if new_val < current_val or random.random() < np.exp(-(new_val - current_val) / max(T, 1e-10)):
            h = h_new
            current_val = new_val

        # Update best solution
        if current_val < best_val:
            best_h = h.copy()
            best_val = current_val

        # Decrease temperature
        T = max(T * cooling_rate, min_temp)

        # Periodic restart to avoid local optima
        if iteration % 5000 == 0 and random.random() < 0.1:
            h = np.random.uniform(0, 1, n_points)
            h *= (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            current_val = evaluate_erdos_solution(h, dx)
            T = 0.05  # Reset temperature

    # Evaluate final best solution
    final_evaluation = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), final_evaluation, n_points)
