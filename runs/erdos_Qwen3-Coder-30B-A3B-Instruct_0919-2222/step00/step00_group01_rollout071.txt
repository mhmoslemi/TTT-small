import numpy as np
from scipy.optimize import differential_evolution
import time

def compute_c5(h):
    """Computes C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(conv)

def run(seed=42, budget_s=600, **kwargs):

    # Use 200 points for better resolution while staying under 1000
    n_points = 200
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

            h = np.interp(x_new, x_orig, h)
    else:
        # Default initialization: uniform distribution
        h = np.ones(n_points) * 0.5
    
    # Scale to meet integral constraint
    h = h * (1.0 / np.sum(h)) * n_points * dx
    
    # Optimization function
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    # Bounds for each h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try to find a better solution using differential evolution 
    start_time = time.time()
    
    # Try solving with fewer variables first (local search)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Local search with small perturbations
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # More iterations to search
    for iter_num in range(2000):

    if time.time() - start_time < budget_s - 0.1:
