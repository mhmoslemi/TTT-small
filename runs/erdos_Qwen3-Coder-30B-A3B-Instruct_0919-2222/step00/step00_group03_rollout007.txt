import numpy as np

from scipy.signal import correlate

# Initial solution
initial_h_values = None

def evaluate_h(h, dx):
    """Evaluate the C5 constant for a given h"""
    # Compute the overlap integral for all possible shifts
    # Using full mode correlation to get all shifts
    corr = correlate(h, 1-h, mode='full')
    # Take only the relevant part (shifts from -1 to 1)
    mid = len(corr) // 2
    valid_corr = corr[mid-1:mid+2]  # we need k = -1, 0, 1
    return np.max(valid_corr) * dx

def evaluate_h_full(h, dx):
    """Evaluate the C5 constant for a given h (full version)"""
    # Compute the overlap integral for all possible shifts
    corr = correlate(h, 1-h, mode='full')
    # The correlation result is indexed from -n+1 to n-1
    # We are interested in k from -1 to 1 (shifts)
    mid = len(corr) // 2
    # Get values for k=-1, 0, 1
    corr_vals = []
    corr_vals.append(corr[mid-1])  # k=-1
    corr_vals.append(corr[mid])    # k=0
    corr_vals.append(corr[mid+1])  # k=1
    return np.max(corr_vals) * dx

def objective(h, dx):
    """Objective to minimize"""
    return evaluate_h_full(h, dx)

def constraint_eq(h, dx, target_integral=1):
    """Constraint that integral equals 1"""
    return np.sum(h) * dx - target_integral

def constraint_bounds(h, bound_type):
    """Constraint bounds"""
    return h if bound_type == 'lower' else 1 - h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Start with initial values if available
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        # Start with a step function approximation
        h = np.full(n_points, 0.5)
    
    best_h = h.copy()
    best_c5 = evaluate_h_full(h, dx)
    
    # Define constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: constraint_eq(h, dx)}]
    
    # Additional heuristic for optimization
    def optimize_step(h_current):
        # Local search around current point
        h_opt = h_current.copy()
        best_val = evaluate_h_full(h_current, dx)
        
        # Try small variations using random perturbations
        for iter_ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            # Make a small random change
            i = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.02, 0.02)
            new_h = h_opt.copy()
            new_h[i] = max(0, min(1, new_h[i] + delta))
            
            # Project back to constraint
            current_integral = np.sum(new_h) * dx
            correction_factor = 1.0 / current_integral if current_integral > 0 else 1.0
            new_h = np.clip(new_h * correction_factor, 0, 1)
            
            new_val = evaluate_h_full(new_h, dx)
            if new_val < best_val:
                best_val = new_val
                h_opt = new_h.copy()
        
        return h_opt, best_val
    
    # Main optimization loop
    for iter_count in range(100000):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Try gradient-based or local search approaches
        h_new, val_new = optimize_step(best_h)
        
        if val_new < best_c5:
            best_h = h_new.copy()
            best_c5 = val_new
            
            # Occasionally check if we're achieving the target
            if best_c5 < 0.379:
                break
    
    # Final evaluation
    actual_c5 = evaluate_h_full(best_h, dx)
    
    return (best_h.tolist(), actual_c5, n_points)
