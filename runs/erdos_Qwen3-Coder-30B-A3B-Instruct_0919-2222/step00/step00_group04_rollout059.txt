import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from math import floor, ceil
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120
    dx = 2.0 / N

    # Build shift matrices A_k
    A = []
    for k in range(N):
        A_k = np.zeros((N, N))
        # Fill in k-th super-diagonal
        for i in range(N - k):
            A_k[i, i + k] = 1
        A.append(A_k)

    # SDP Solver Parameters
    # For robustness, use ECOS driver, which is faster for medium size problems
    solver = cp.ECOS

    # Bisection parameters
    low = 0.35
    high = 0.5
    tol = 1e-4
    max_iter = 20

    best_c5 = high
    best_h = None

    def feasible(γ):
        # Define variables
        h = cp.Variable(N)
        X = cp.Variable((N, N), symmetric=True)

        # Constraints
        constraints = [
            cp.diag(X) == h,  # Element-wise diagonal constraint
            X >> 0,           # PSD constraint
            0 <= h, h <= 1,   # Box constraints
            cp.sum(h) == N / 2  # Mass constraint
        ]
        # Correlation constraints
        for k in range(N):
            constraints.append(cp.trace(A[k] @ X) >= γ)

        # Feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=solver, verbose=False)
        return prob.status == cp.OPTIMAL

    # Binary search
    iterations = 0
    while (high - low > tol) and (time.time() - start_time < budget_s * 0.9):
        iterations += 1
        gamma = (low + high) / 2.0
        if feasible(gamma):
            high = gamma
        else:
            low = gamma
    converged_gamma = (low + high) / 2.0
    c5_bound = 1.0 - converged_gamma
    print("Final SDP Bisection: γ =", converged_gamma, "C5 =", c5_bound, "in", iterations, "steps.")

    # Now retrieve solution
    h_opt = cp.Variable(N)
    X_opt = cp.Variable((N, N), symmetric=True)

    constraints = [
        cp.diag(X_opt) == h_opt,
        X_opt >> 0,
        0 <= h_opt, h_opt <= 1,
        cp.sum(h_opt) == N / 2
    ]

    for k in range(N):
        constraints.append(cp.trace(A[k] @ X_opt) >= converged_gamma)

    prob_final = cp.Problem(cp.Minimize(0), constraints)
    prob_final.solve(solver=solver, verbose=False)

    if prob_final.status != cp.OPTIMAL:
        print("SDP final was infeasible, falling back to LP")
        # Use simpler LP variant
        # We note that the time budget is tight: save it for LP instead
        # Construct via Linear Programming on power spectrum
        # Constants and setup
        J = 30  # Number of frequencies (upto pi for Nyquist)
        freqs = [j * np.pi / 2.0 for j in range(J)]
        # cos(k * omega) is periodic with period 2
        shift_indices = np.array(range(N))  # x = i * dx
        shift_vals = shift_indices * dx

        # Build feature matrix for cos(j*omega*x) for each x
        phi = np.zeros((N, J))
        for i in range(N):
            for j in range(J):
                phi[i, j] = np.cos(freqs[j] * shift_vals[i])

        # Also add a constant term (for mean)
        phi = np.hstack([np.ones((N, 1)), phi])

        # LP formulation
        s = cp.Variable(J + 1)  # s[0] is the constant term, s[1:] are coeffs
        gamma = cp.Variable()  # min correlation lower bound

        # The reconstructed h(x) is a linear combination of these terms
        # h_vector = s[0] * ones + phi @ s[1:]  => h = s[0] + Phi * s[1:]
        # Boxes: 0 <= h_i <= 1 for all i
        h = s[0] + phi @ s[1:]
        # Optimise min G_k
        # We have G_k = s[0]^2 + sum_j s_j^2 * cos(omega_j * k)
        G = []
        for k in range(N):
            corr = s[0]**2 + sum(s[i]**2 * np.cos(freqs[i-1] * k * dx) for i in range(1, J+1))
            G.append(corr)

        # We want all entries in [G_min, G_max]
        constraints_lp = [
            h >= 0,
            h <= 1,
            cp.sum(h) == N/2,
            gamma <= cp.hstack(G)
        ]
        # Solve the LP minimising gamma
        prob_lp = cp.Problem(cp.Maximize(gamma), constraints_lp)
        try:
            prob_lp.solve(solver=cp.SCS, verbose=False)
            if prob_lp.status == cp.OPTIMAL:
                h_lp = h.value
                best_h = h_lp
                c5_bound = 1.0 - min(G).value
                print("LP formulation successful, final C5 =", c5_bound)
            else:
                # If that didn't work, fallback to empty baseline
                best_h = np.ones(N) * 0.5
                c5_bound = 0.5
        except Exception:
            best_h = np.ones(N) * 0.5
            c5_bound = 0.5
    else:
        h_lp = h_opt.value
        X_lp = X_opt.value

        # Rank-1 recovery using leading eigenvalue
        w, v = np.linalg.eigh(X_lp)
        # Select the largest eigenvalue and eigenvector
        idx = np.argsort(w)[::-1]
        v_max = v[:, idx[0]]
        val_max = w[idx[0]]
        if val_max < 0: val_max = 0

        # Build relaxation initially as sqrt(lam)*v
        h_candidate = np.sqrt(val_max) * v_max

        # Subset constraint
        h_candidate = np.clip(h_candidate, 0, 1)

        # Project onto simplex: waterfilling or custom projection
        # Need sum(h) == N/2
        def project_simplex(h_vec, target_sum=N/2):
            h_vec = np.copy(h_vec)
            # Bring it within [0,1]
            h_vec = np.clip(h_vec, 0, 1)
            current_sum = np.sum(h_vec)
            if abs(current_sum - target_sum) < 1e-12:
                return h_vec
            # Water filling manually
            sort_idx = np.argsort(h_vec)[::-1]
            h_sorted = h_vec[sort_idx]

            # Find threshold
            cumulative = np.cumsum(h_sorted)
            thresholds = (cumulative - target_sum) / np.arange(1, len(h_sorted) + 1)

            # Find largest threshold such that h[i] > threshold for all i.
            valid = thresholds < h_sorted
            if not valid.any():
                return np.ones_like(h_vec) * target_sum / len(h_vec)
            k = np.argmax(valid) + 1
            theta = thresholds[k - 1] if k != 0 else 0
            result = np.maximum(h_vec - theta, 0)
            return result

        final_h = project_simplex(h_candidate)
        best_h = final_h
        # Evaluate C5
        c5_bound = max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    # Optionally refine the solution with a short projected gradient
    def compute_objective(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode="full")
        return 1 - np.max(correlation) * dx

    def project_simplex_step(h_orig, target_sum=N/2):
        h_new = np.clip(h_orig, 0, 1)
        diff = np.sum(h_new) - target_sum

        # waterfilling-like projection to maintain target sum
        while abs(diff) > 1e-6:
            if diff > 0:
                # Decrease values where they exceed target
                # Find largest values to decrease
                sorted_idx = np.argsort(h_new)
                if h_new[sorted_idx[0]] < 1e-8:
                    break  # Already bounded to 0
                step = min(diff, h_new[sorted_idx[0]])
                h_new[sorted_idx[0]] -= step
                diff -= step
            else:
                # Increase values
                sorted_idx = np.argsort(h_new)[::-1]
                if h_new[sorted_idx[0]] > 1.0 - 1e-8:
                    break  # Already bounded to 1
                step = min(-diff, 1.0 - h_new[sorted_idx[0]])
                h_new[sorted_idx[0]] += step
                diff += step
        return h_new

    # Modified subgradient method
    rate = 0.01
    iters = 2000
    for t in range(iters):
        if time.time() - start_time > budget_s * 0.95:
            break
        corr = np.correlate(best_h, 1 - best_h, mode="full") * dx
        best_shift = np.argmax(corr)
        grad = np.zeros_like(best_h)
        for i in range(len(best_h)):
            if i + best_shift < len(best_h):
                grad[i] += (1 - best_h[i + best_shift])
            if i >= best_shift:
                grad[i] -= best_h[i - best_shift]
        # Update step with learning rate
        updated = best_h - rate * grad
        best_h = project_simplex_step(updated)
        rate *= 0.9999

    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode="full") * dx
    c5_bound = 1.0 - np.min(final_corr)

    return (best_h.tolist(), c5_bound, N)
