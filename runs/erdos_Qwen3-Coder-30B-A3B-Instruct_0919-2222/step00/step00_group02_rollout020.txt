import numpy as np
import time
import random

def evaluate_h(h):
    """Evaluates the maximum overlap integral for the given step function."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode='full')
    max_val = O_counts.max()
    return max_val * dx

def max_overlap_counts(h):
    """Returns integer maximum overlap count"""
    O_counts = np.correlate(h, 1 - h, mode='full')
    return int(O_counts.max())

def random_balanced_binary(n, rng):
    """Generate a random balanced binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    rng.shuffle(h)
    return h

def binary_anneal(n, time_limit, rng):
    """Run simulated annealing on binary vector to minimize max overlap."""
    def compute_max_overlap(h):
        O_counts = np.correlate(h, 1 - h, mode='full')
        max_val = O_counts.max()
        return max_val

    h = random_balanced_binary(n, rng)
    cur_val = compute_max_overlap(h)
    best_val = cur_val
    best_h = h.copy()
    
    start_time = time.time()
    max_iters = 50000  # arbitrary high value for an upper bound
    
    # Use exponential cooling schedule
    T_start = 1.0
    T_end = 1e-4
    alpha = (T_end / T_start) ** (1.0 / max_iters)
    
    for iter in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
            
        T = T_start * (alpha ** iter)
        
        # propose swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        # swap
        h[p], h[q] = 0, 1
        new_val = compute_max_overlap(h)
        delta = new_val - cur_val
        
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
                
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # revert
            h[p], h[q] = 1, 0

    return best_h, best_val

def project_onto_box_and_sum(v, target_sum):
    """Project vector onto the intersection of box constraints [0,1] and sum constraint."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def continuous_descent(h, time_limit, rng):
    """Refine fractional solution via gradient descent and projection."""
    # convert to float
    h = h.astype(float)
    n = len(h)
    target_sum = n / 2
    
    start_time = time.time()
    best_h = h.copy()
    best_obj = evaluate_h(h)
    
    step_size = 0.1
    max_iters = 10000
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
            
        O_counts = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(O_counts)
        shift = max_idx - (n - 1)
        
        # compute gradient for the maximum shift
        grad = np.zeros(n)
        
        # compute for positive shifts
        if shift >= 0:
            left = np.zeros_like(h)
            right = np.zeros_like(h)
            if shift < n:
                left[shift:] = h[:-shift]
            if shift < n:
                right[:n-shift] = h[shift:]
            grad = 1 - left - right
        else:
            # shift negative
            s = -shift
            left = np.zeros_like(h)
            right = np.zeros_like(h)
            if s < n:
                left[:n-s] = h[s:]
            if s < n:
                right[s:] = h[:-s]
            grad = 1 - left - right
            
        # gradient descent step
        h_new = h - step_size * grad
        # project onto known feasible set: box [0,1] and sum constraint
        h_new = project_onto_box_and_sum(h_new, target_sum)
        
        # evaluate objective
        obj = evaluate_h(h_new)
        if obj < best_obj:
            best_obj = obj
            best_h = h_new.copy()
            
        h = h_new.copy()
        
    return best_h, best_obj
