import numpy as np
from cvxpy import Variable, Minimize, Problem, PSD, sum_entries, trace, norm
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120          # Number of grid points
    max_iter_bisection = 25
    tolerance = 1e-4
    final_refine_steps = 2000
    dx = 2.0 / N
    
    # Create shift matrices A_k for all valid shifts k=0..N-1
    A = [np.zeros((N, N)) for _ in range(N)]
    for k in range(N):
        for i in range(N-k):
            A[k][i, i+k] = 1.0
    
    # Cost function to evaluate C5 from h
    def evaluate_c5(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    # SDP-based bisection
    low, high = 0.35, 0.5
    best_c5 = high
    best_h = None

    for _ in range(max_iter_bisection):
        if time.time() - start_time > budget_s - 10:
            break
        tau = (low + high) / 2.0
        gamma = 1.0 - tau  # We want to find minimum autocorrelation ≥ gamma
        
        # Model for feasibility check
        h_var = Variable(N)
        X = Variable((N, N), PSD=True)
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            sum_entries(h_var) == N / 2.0,
            X.diag() == h_var,
        ]
        
        for k in range(N):
            constraints.append(trace(A[k] @ X) >= gamma)
        
        # Set objective to None for feasibility check
        prob = Problem(Minimize(0), constraints)
        
        try:
            prob.solve(solver='ECOS', verbose=False)
            if prob.status == 'optimal':
                h_sol = h_var.value
                X_sol = X.value
                
                if X_sol is not None:
                    # Extract h from dominant eigenvector
                    evals, evecs = np.linalg.eigh(X_sol)
                    v = evecs[:, -1]  # leading eigenvector (index -1)
                    h_candidate = np.sqrt(evals[-1]) * v
                    
                    # Clip and project onto simplex
                    h_proj = np.clip(h_candidate, 0, 1)
                    target_sum = N / 2.0
                    current_sum = np.sum(h_proj)
                    if abs(current_sum - target_sum) > 1e-10:
                        # Water-filling projection
                        sorted_h = np.sort(h_proj)[::-1]
                        cumsum = np.cumsum(sorted_h)
                        threshold_idx = np.argmax((cumsum - target_sum) <= 0)
                        if threshold_idx == 0:
                            theta = (cumsum[0] - target_sum) / N
                            h_final = np.maximum(0, h_proj - theta)
                        else:
                            theta = (cumsum[threshold_idx - 1] - target_sum) / (N - threshold_idx)
                            h_final = np.maximum(0, h_proj - theta)
                        h_final = np.clip(h_final, 0, 1)
                        # Renormalize
                        h_final *= (target_sum / np.sum(h_final))
                        h_final = np.clip(h_final, 0, 1)
                    else:
                        h_final = h_proj
                    
                    c5_candidate = evaluate_c5(h_final)
                    if c5_candidate < best_c5:
                        best_c5 = c5_candidate
                        best_h = h_final.copy()
                    
                    if c5_candidate <= 0.3807 and c5_candidate < tau:
                        high = tau
                    else:
                        low = tau
                else:
                    low = tau
            else:
                low = tau
        except Exception:
            low = tau

    if best_h is None:
        # Fallback to some heuristic if SDP fails
        h_init = np.zeros(N)
        half = N // 2
        h_init[:half] = 1.0
        h_init = h_init / np.sum(h_init) * (N / 2)
        best_h = h_init
    
    # Final refinement via projected subgradient descent
    best_h = best_h.copy()
    def project_simplex(arr, target_sum):
        arr = np.clip(arr, 0, 1)
        current_sum = np.sum(arr)
        diff = target_sum - current_sum
        if abs(diff) < 1e-10:
            return arr
        if diff > 0:
            # Add to components in order of increasing value
            sort_idx = np.argsort(arr)
            for i in range(N):
                idx = sort_idx[i]
                delta = min(diff, 1.0 - arr[idx])
                if delta <= 0:
                    break
                arr[idx] += delta
                diff -= delta
                if diff <= 0:
                    break
        elif diff < 0:
            # Subtract from components in order of decreasing value
            sort_idx = np.argsort(arr)[::-1]
            for i in range(N):
                idx = sort_idx[i]
                delta = min(-diff, arr[idx])
                if delta <= 0:
                    break
                arr[idx] -= delta
                diff += delta
                if diff >= 0:
                    break
        return np.clip(arr, 0, 1)
    
    def compute_c5_grad(h, k_star):
        grad = np.zeros(N)
        # k_star shift
        if k_star < N:
            # d/dh_i (sum h_i * (1-h_{i+k_star})) 
            for i in range(N - k_star):
                if i+k_star < N:
                    grad[i] += (1 - h[i+k_star])
                if i >= k_star:
                    grad[i] -= h[i-k_star]
        return grad

    for t in range(final_refine_steps):
        if time.time() - start_time > budget_s - 2:
            break
        c5_val = evaluate_c5(best_h)
        # Find worst shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_star = np.argmax(corr)
        k_val = corr[k_star]
        # Compute gradient with respect to h for this k
        grad = compute_c5_grad(best_h, k_star)
        # Step size with sqrt decay
        alpha = 0.1 / np.sqrt(t + 1)
        new_h = best_h - alpha * grad
        # Project onto feasible set [0,1] and mass constraint
        new_h = project_simplex(new_h, N/2.)
        # Newton-type update with small momentum or just use new_h
        best_h = new_h

    # Final check with exact correlation
    c5_final = evaluate_c5(best_h)
    return (best_h.tolist(), c5_final, N)
