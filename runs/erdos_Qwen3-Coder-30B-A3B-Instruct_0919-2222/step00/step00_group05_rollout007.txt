import time
import numpy as np
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    M = n / 2  # target mass

    # Check if initial solution already beats the target
    initial_c5 = max(correlate(initial_h_values, 1 - initial_h_values, mode='full')) * dx
    if initial_c5 <= 0.3807:
        return (list(initial_h_values), initial_c5, n)

    # Step 1: Formulate and solve LP relaxation
    print(f"Solving LP for n={n}...")
    # Variables: h, y, t
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Binary matrix for shifts
    y = cp.Variable((n, 2 * n - 1))
    
    # Constraints
    constraints = [0 <= h, h <= 1, cp.sum(h) == M, t >= 0]
    
    for s in range(1 - n, n):  # shift index
        if s >= 0:
            i_range = range(n - s)
        else:
            i_range = range(-s, n)
        # Create McCormick envelope constraints
        for i in i_range:
            i_s = i + s
            y[i, s + n - 1] = cp.Variable()
            constraints += [
                y[i, s + n - 1] <= h[i],
                y[i, s + n - 1] <= h[i_s],
                y[i, s + n - 1] >= h[i] + h[i_s] - 1,
                y[i, s + n - 1] >= 0
            ]
    
    # Shift constraints
    for s in range(1 - n, n):
        if s >= 0:
            i_range = range(n - s)
        else:
            i_range = range(-s, n)
        s_idx = s + n - 1
        Q_s = cp.sum(y[i, s_idx] for i in i_range)
        constraints.append(Q_s >= M - t)
    
    # Solve LP
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True)
    except:
        print("LP solver failed.")
        return (list(initial_h_values), initial_c5, n)
    print(f"LP solved in {time.time() - start_time:.2f}s.")

    # Step 2: Extract the solution and process candidates
    h_star = h.value

    # Check if solution is binary
    h_rounded = np.round(h_star).astype(int)
    if np.allclose(h_star, h_rounded, atol=1e-8):
        candidates = [h_rounded]
        print("LP solution is binary.")
    else:
        # Deterministic rounding (largest values to 1)
        sorted_indices = np.argsort(h_star)[::-1]
        h_det = np.zeros_like(h_star)
        h_det[sorted_indices[:int(M)]] = 1
        candidates = [h_det]

        # Randomized rounding with 200 samples
        for _ in range(200):
            h_sample = np.random.binomial(1, h_star).astype(float)
            if abs(np.sum(h_sample) - M) < 1e-6:
                candidates.append(h_sample)

    # Step 3: Local polishing of candidates
    def compute_C5(h_vec):
        return max(correlate(h_vec, 1 - h_vec, mode='full')) * dx

    best_h = None
    best_c5 = float('inf')
    
    for h_cand in candidates:
        # Apply greedy swap polishing logic (not full aide)
        # A minimal heuristic based on swapping for the worst shift:
        times_up = 0
        while times_up < 100 and time.time() - start_time < budget_s - 30:
            corr_full = correlate(h_cand, 1 - h_cand, mode='full')
            max_s = np.argmax(corr_full)
            max_correlation = corr_full[max_s] * dx
            if max_correlation <= best_c5: break
            
            idx_s = max_s - (n - 1)
            best_i = -1
            best_j = -1
            best_gain = 0
            for i in range(n):
                # valid shift
                j = i + idx_s
                if j < 0 or j >= n:
                    continue
                # Check if it's beneficial to swap
                if h_cand[i] == 1 and h_cand[j] == 0:
                    # Try to find a good target:
                    for k in range(n):
                        if k != i:
                            l = k + idx_s
                            if l >= 0 and l < n and h_cand[k] == 0 and h_cand[l] == 1:
                                # If swapping makes a move from bad to good
                                gain = 1  # current naive gain
                                if gain > best_gain:
                                    best_gain = gain
                                    best_i, best_j = i, k
                                    break
            if best_i >= 0 and best_j >= 0:
                h_cand[best_i], h_cand[best_j] = h_cand[best_j], h_cand[best_i]
                times_up += 1
            else:
                break
        cur_c5 = compute_C5(h_cand)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_cand.copy()

    if best_c5 <= 0.3807:
        return (list(best_h), best_c5, n)

    # MIP refinement if needed
    if len(candidates) > 1 or not np.allclose(h_star, h_rounded, atol=1e-8):
        print("Attempting MIP refinement...")
        try:
            h_bin = cp.Variable(n, integer=True)
            t_mip = cp.Variable()
            
            y_mip = cp.Variable((n, 2 * n - 1))
            constraints_mip = [0 <= h_bin, h_bin <= 1, cp.sum(h_bin) == M, t_mip >= 0]
            
            for s in range(1 - n, n):
                if s >= 0:
                    i_range = range(n - s)
                else:
                    i_range = range(-s, n)
                for i in i_range:
                    i_s = i + s
                    y_mip[i, s + n - 1] = cp.Variable()
                    constraints_mip += [
                        y_mip[i, s + n - 1] <= h_bin[i],
                        y_mip[i, s + n - 1] <= h_bin[i_s],
                        y_mip[i, s + n - 1] >= h_bin[i] + h_bin[i_s] - 1,
                        y_mip[i, s + n - 1] >= 0
                    ]
            
            for s in range(1 - n, n):
                if s >= 0:
                    i_range = range(n - s)
                else:
                    i_range = range(-s, n)
                s_idx = s + n - 1
                Q_s = cp.sum(y_mip[i, s_idx] for i in i_range)
                constraints_mip.append(Q_s >= M - t_mip)
            
            prob_mip = cp.Problem(cp.Minimize(t_mip), constraints_mip)
            remaining_time = budget_s - (time.time() - start_time)
            if remaining_time > 10:
                prob_mip.solve(solver=cp.CBC, max_seconds=remaining_time - 5)
                if h_bin.value is not None:
                    h_got = h_bin.value
                    final_c5 = compute_C5(h_got)
                    if final_c5 < best_c5:
                        best_h = h_got
                        best_c5 = final_c5
        except:
            pass

    if best_c5 <= 0.3807:
        return (list(best_h), best_c5, n)
    else:
        #Fallback: return the best one
        return (list(best_h), best_c5, n)
