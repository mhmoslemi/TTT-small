import numpy as np
from scipy.optimize import differential_evolution
import time
from math import floor

def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    # np.correlate with mode='full' gives us the correlation at all lags
    # We're looking for max of int h(x)(1-h(x+k)) dx for all k
    # Since h is defined on [0,2], the maximum lag k we need to consider 
    # is up to 2, so the full correlation will include relevant values
    correlation = np.correlate(h, 1 - h, mode="full")
    # The valid range in the full correlation for non-zero overlaps
    # is from k=0 (shift 0) to k=2 with the current spacing
    # For no shift (k=0), we get the full correlation
    # The maximum over all k is the maximum of the correlation
    #   correlation indices [0,...,n-1] correspond to k=[n-1,...,0] 
    #   (negative shifts, data shifted right)
    #   correlation indices [n,...,2*n-2] correspond to k=[0,...,n-1] 
    #   (positive shifts, data shifted left)
    # We want max of k=0,...,n-1 (shift from left)
    max_val = np.max(correlation[n_points-1:])
    return max_val * dx

def evaluate_erdos_solution(h_values):
    """Evaluate the solution provided by the system"""
    return evaluate_h(h_values)

def initial_h_values():
    """Provide a reasonable initial construction to start from."""
    # Generate a piecewise constant function to serve as starting point
    # This is a hand-crafted approximation based on previous known good values
    n_points = 45
    h = np.ones(n_points) * 0.5
    dx = 2.0 / n_points  # width of each interval
    
    # Make first 20 intervals have value 0.7 and last 25 intervals have value 0.3
    # to satisfy the constraint that integral equals 1.0
    # sum(h)*dx = 1 => sum(h) = n/2
    # sum(h) = 20*0.7 + 25*0.3 = 14 + 7.5 = 21.5
    # so make h_scaled = h * (n_points/2)/sum(h)
    h[:20] = 0.7
    h[20:] = 0.3
    scaling = (n_points/2)/np.sum(h)
    h = h * scaling
    
    return h.tolist()

def prepare_constraints(n_points):
    """Prepare constraint functions for scipy optimization."""
    dx = 2.0 / n_points
    
    def integral_constraint(x):
        return np.sum(x) * dx - 1.0
    
    def bounds():
        return [(0, 1) for _ in range(n_points)]
    
    return integral_constraint, bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    max_time = time.time() + budget_s 
    best_h = None
    best_c5 = float('inf')
    
    n_points = 45  # Based on initial_h_values
    
    # Start with good initial solution from the problem
    current_h = np.array(initial_h_values())
    initial_c5 = evaluate_h(current_h)
    
    if initial_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = initial_c5
        
    # Use differential evolution for global optimization
    # This is more effective for non-smooth functions like ours
    # The bounds are [0,1] for all variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Integrate the constraint into the objective function penalizing violations
    def objective(x):
        # add penalty if integral != 1
        integral_error = abs(np.sum(x) * (2.0 / n_points) - 1.0)
        penalty = 1e10 * integral_error if integral_error > 1e-3 else 0.0
        return evaluate_h(x) + penalty
    
    # Run optimization with limit on samples
    try:
        # Use a custom limit on iterations to provide control over time
        # Initialize with the best known solution
        start = time.time()
        result = differential_evolution(
            objective,
            bounds,
            seed=seed,
            maxiter=10000,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.7,
            atol=1e-6,
            callback=lambda x, convergence: time.time() > max_time or False,
            disp=False
        )
        
        # Extract the final solution
        optimized_h = result.x
        computed_c5 = evaluate_h(optimized_h)
        
        if computed_c5 < best_c5:
            best_h = optimized_h.copy()
            best_c5 = computed_c5
            
    except Exception as e:
        pass  # Fall back to the initial solution if DE fails
    
    final_h = best_h if best_h is not None else current_h 
    return (final_h.tolist(), best_c5, n_points)
