import numpy as np
import time
import random

# Global variable provided
initial_h_values = None

def compute_correlation(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap(h, dx):
    c = compute_correlation(h)
    return c.max() * dx

def apply_delta(c, h, idx, delta):
    n = len(h)
    offset = n - 1
    # First term: for j from idx to n-1
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # Second term: for j from 0 to idx
    for j in range(0, idx+1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(i, j, c, h):
    delta_i = 0 - h[i]
    delta_j = 1 - h[j]
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def targeted_swap(c, h, n):
    max_idx = np.argmax(c)
    s = max_idx - (n - 1)
    # Determine sets of suitable swap candidates
    A = []
    B = []
    for i in range(n):
        if h[i] > 0.5:  # treat as 1
            j = i + s
            if 0 <= j < n and h[j] < 0.5:
                A.append(i)
        else:  # h[i] < 0.5
            j = i + s
            if 0 <= j < n and h[j] > 0.5:
                B.append(i)
    if A and B:
        i = random.choice(A)
        j = random.choice(B)
        swap(i, j, c, h)
        return True
    return False

def hill_climb(h, n_iter, time_left, dx):
    n = len(h)
    c = compute_correlation(h)
    best_h = h.copy()
    best_val = max_overlap(h, dx)
    iter_count = 0
    start_time = time.time()
    
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] < 0.5]
    
    for iteration in range(n_iter):
        if time.time() - start_time > time_left - 0.1:
            break
            
        if random.random() < 0.7 or not targeted_swap(c, h, n):
            if ones and zeros:
                i = random.choice(ones)
                j = random.choice(zeros)
                delta_i = 0 - h[i]
                delta_j = 1 - h[j]
                apply_delta(c, h, i, delta_i)
                h[i] = 0
                apply_delta(c, h, j, delta_j)
                h[j] = 1
                
                # Update sets
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
        
        val = max_overlap(h, dx)
        if val < best_val:
            best_h = h.copy()
            best_val = val
            
        iter_count += 1
    
    return best_h, best_val

def run(seed=42, budget_s=600):
    # Set random seeds
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points = 0
    
    n_values = [120, 150, 180]
    
    for n in n_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        dx = 2.0 / n
        h = np.zeros(n)
        target_sum = n / 2.0
        
        # Initialize as binary with target sum
        ones_indices = np.random.choice(n, int(target_sum), replace=False)
        for idx in ones_indices:
            h[idx] = 1.0
        
        # Ensure sum equals target_sum
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if diff != 0:
            # Adjust any index that can be changed to ensure sum
            # For binary, this is only possible if we have fractional adjustment
            # In practice, this modification should be very rare for large n
            pass  # Assume random choice has good approximation
        
        # Run hill climber
        time_left = budget_s - (time.time() - start_time)
        if time_left <= 1:
            break
            
        h_final, c5_val = hill_climb(h, 20000, time_left, dx)
        
        if c5_val < best_c5:
            best_h = h_final
            best_c5 = c5_val
            n_points = n
    
    if best_h is not None:
        return list(best_h), best_c5, n_points
    # Fallback to initial if none found
    if initial_h_values is not None and len(initial_h_values) > 0:
        dx = 2.0 / len(initial_h_values)
        return list(initial_h_values), max_overlap(initial_h_values, dx), len(initial_h_values)
    else:
        # Create default binary solution
        n = 120
        dx = 2.0 / n
        h = np.zeros(n)
        ones = np.random.choice(n, n//2, replace=False)
        for i in ones:
            h[i] = 1.0
        return list(h), max_overlap(h, dx), n
