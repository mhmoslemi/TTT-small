import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap problem constant C₅ for given step function h."""
    # Ensure h is a numpy array
    h = np.array(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx
    dx = 2.0 / len(h)
    corr_scaled = corr * dx
    # Return maximum value
    return np.max(corr_scaled)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 501  # Use odd number for better centering
    dx = 2.0 / n_points
    x = np.arange(n_points) * dx + dx / 2  # Midpoint positions
    
    # Define shape functions
    def triangle_shape(x):
        return np.maximum(0, 1 - np.abs(x - 1))
    
    def cosine_power_shape(x, p):
        return np.cos(np.pi / 2 * (x - 1)) ** p
    
    def polynomial_shape(x, a):
        return np.maximum(0, 1 - np.abs(x - 1) ** a)
    
    # Generate candidate shapes
    candidates = []
    
    # Triangle
    h_tri = triangle_shape(x)
    norm_factor = (n_points / 2) / np.sum(h_tri)
    h_tri *= norm_factor
    candidates.append(("triangle", h_tri))
    
    # Cosine shapes with various powers
    powers = [0.5, 1, 2, 4, 8]
    for p in powers:
        h_cos = cosine_power_shape(x, p)
        norm_factor = (n_points / 2) / np.sum(h_cos)
        h_cos *= norm_factor
        candidates.append((f"cos^{p}", h_cos))
    
    # Polynomial shapes
    exponents = [1, 2, 3]
    for a in exponents:
        h_poly = polynomial_shape(x, a)
        norm_factor = (n_points / 2) / np.sum(h_poly)
        h_poly *= norm_factor
        candidates.append((f"poly^{a}", h_poly))
    
    # Evaluate all candidates
    best_h = None
    best_c5 = float('inf')
    best_name = ""
    
    for name, h in candidates:
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_name = name
    
    # Time budget for local refinement
    start_time = time.time()
    end_time = start_time + budget_s * 0.95
    
    # Local refinement via hill climbing
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Optimization parameters
    max_iterations = 100000
    temperature = 0.01
    cooling_rate = 0.99995
    
    for iteration in range(max_iterations):
        if time.time() > end_time:
            break
            
        # Choose two random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
            
        # Compute maximum allowable change
        delta_max = min(current_h[i], 1 - current_h[j])
        if delta_max <= 1e-10:
            continue
            
        # Propose small change
        delta = np.random.random() * delta_max * 0.1
        
        # Apply change
        new_h = current_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
        elif np.random.random() < np.exp(-(new_c5 - current_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
        # Update best if needed
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Cooling
        temperature = max(1e-8, temperature * cooling_rate)
        
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
