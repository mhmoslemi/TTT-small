import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value corresponds to the maximum overlap
    max_corr = np.max(corr)
    return max_corr * dx

    def objective(h):

        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return evaluate_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Required to run optimization
    def optimize_with_sampling():
        nonlocal best_h, best_c5
        # Use a simple local optimization approach 
        h_current = h_init.copy()
        h_current = h_current / np.sum(h_current) * (n_points / 2)
        
        iteration = 0

            iteration += 1
            # Try differential evolution on a subsample of the variables
            # This is a compromise between exploration and being fast
            indices = np.random.choice(n_points, size=min(n_points//3, 15), replace=False)
            bounds_slice = [bounds[i] for i in indices]
            
            def objective_slice(h_slice):
                h_copy = h_current.copy()
                for i, idx in enumerate(indices):
                    h_copy[idx] = h_slice[i]
                return objective(h_copy)
            
            # Use differential evolution on the selected subset
            try:
                res = differential_evolution(
                    objective_slice,
                    bounds_slice,
                    maxiter=20,
                    popsize=5,

                    mutation=(0.5, 1.0),
