import numpy as np
from scipy.optimize import differential_evolution
from scipy import fft
import time

def project_to_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum x = target_sum"""
    v = np.array(v)
    sorted_indices = np.argsort(v)[::-1]
    sorted_v = v[sorted_indices]
    
    # Water-filling algorithm
    n = len(v)
    cumsum = np.cumsum(sorted_v)
    thresholds = (cumsum - target_sum) / np.arange(1, n+1)
    
    # Find threshold
    pos = np.searchsorted(thresholds, sorted_v, side='right') - 1
    pos = np.maximum(pos, 0)
    tau = thresholds[pos]
    
    # Compute projection
    result = np.maximum(0, np.minimum(1, v - tau))
    
    # Adjust for sum mismatch
    current_sum = np.sum(result)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        # Distribute diff to elements with value close to 1
        sorted_idx = np.argsort(result)[::-1]
        for i in sorted_idx:
            if diff == 0: break
            can_add = 1 - result[i]
            add = min(can_add, diff)
            result[i] += add
            diff -= add
        # Same for elements near 0
        sorted_idx = np.argsort(result)
        for i in sorted_idx:
            if diff == 0: break
            can_subtract = result[i]
            subtract = min(can_subtract, -diff)
            result[i] -= subtract
            diff += subtract
    
    return result

def compute_autocorrelation_blocks(w, a, N=200):
    """
    w: widths (sum to 2)
    a: heights (in [0,1])
    N: number of grid points for final sampling
    """
    # Compute interval boundaries
    s = np.cumsum(w)
    s = np.insert(s, 0, 0)
    
    # Sample on grid
    dx = 2.0 / N
    h = np.zeros(N)
    for i in range(len(w)):
        start = int(s[i] / dx)
        end = int(s[i+1] / dx)
        if end > start:
            h[start:end] = a[i]
    
    # Compute full autocorrelation using FFT
    H = fft.rfft(h)
    K = fft.rfft(1.0 - h)
    corr = fft.irfft(H * np.conj(K), n=2*N-1)
    return corr[:N]  # Return only non-negative lags

def evaluate_fitness(z, M=12, penalty_weight=1000):
    # Extract w and a
    u = z[:M]
    v = z[M:]
    
    # Transform to actual values
    w = 2 * np.exp(u) / np.sum(np.exp(u))  # softmax
    a = 1 / (1 + np.exp(-v))  # sigmoid
    
    # Integral constraint
    integral_mass = np.sum(a * w)
    penalty = penalty_weight * abs(integral_mass - 1.0)
    
    # Compute correlation
    corr = compute_autocorrelation_blocks(w, a)
    max_corr = np.max(corr)
    
    return max_corr + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    
    # Parameters
    M = 12  # Number of blocks
    N = 200  # Grid resolution
    dx = 2.0 / N
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Global search with DE
    bounds = [(-5,5)] * (2 * M)
    
    def obj(z):
        return evaluate_fitness(z, M=M)
    
    de_time = budget_s * 0.2
    de_start = time.time()
    
    try:
        res = differential_evolution(
            obj,
            bounds,
            seed=seed,
            maxiter=int(50 * (de_time / 10)),
            polish=False,
            popsize=15 * (2 * M)
        )
        
        # Extract best DE solution
        u_best = res.x[:M]
        v_best = res.x[M:]
        w_best = 2 * np.exp(u_best) / np.sum(np.exp(u_best))
        a_best = 1 / (1 + np.exp(-v_best))
        
        # Normalize integral
        total_mass = np.sum(a_best * w_best)
        if total_mass > 0:
            a_best = a_best / total_mass
        
        # Ensure all heights are in [0,1]
        a_best = np.clip(a_best, 0, 1)
        
        # Sample on grid
        s = np.cumsum(w_best)
        s = np.insert(s, 0, 0)
        h_sampled = np.zeros(N)
        for i in range(len(w_best)):
            start = int(s[i] / dx)
            end = int(s[i+1] / dx)
            if end > start:
                h_sampled[start:end] = a_best[i]
        
        # Final residual penalty for integral in case of numerical error
        integral_check = np.sum(h_sampled) * dx
        penalty_final = 1000 * abs(integral_check - 1.0)
        corr = compute_autocorrelation_blocks(w_best, a_best, N)
        c5 = np.max(corr) + penalty_final
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sampled.copy()
            
    except Exception as e:
        pass
    
    # Phase 2: Local improvement with projected sub-gradient
    direction = 'backward'
    
    refine_time = budget_s * 0.7
    refine_start = time.time()
    
    num_refinements = 5
    alpha_0 = 0.2
    
    for ref in range(num_refinements):
        if time.time() - refine_start > refine_time - 5:
            break
            
        h_ref = best_h.copy()
        if ref > 0:
            noise = rng.normal(0, 0.02, N)
            h_ref = np.clip(h_ref + noise, 0, 1)
            h_ref = project_to_capped_simplex(h_ref, N/2)
            
        max_iter = 1000 + int(3000 * (budget_s - time.time() + start_time) / 600)
        
        for t in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute correlation with FFT
            H = fft.rfft(h_ref)
            K = fft.rfft(1.0 - h_ref)
            corr = fft.irfft(H * np.conj(K), n=2*N-1)
            corr = corr[:N]
            
            # Find maximizing shift
            max_corr = np.max(corr)
            max_indices = np.where(np.abs(corr - max_corr) < 1e-10)[0]
            k_star = max_indices[0]  # Take one
            
            # Compute gradient for shift k_star
            g = np.zeros(N)
            if k_star >= 0:
                g[:N-k_star] += (1 - h_ref[k_star:])
            if k_star <= N-1:
                g[k_star:] -= h_ref[:-k_star] if k_star > 0 else h_ref
            
            # Update
            alpha = alpha_0 / np.sqrt(t+1)
            h_ref = h_ref - alpha * g
            
            # Project
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = project_to_capped_simplex(h_ref, N/2)
            
            # Update best if improved
            corr_new = compute_autocorrelation_blocks(w_best, a_best, N) if ref == 0 else corr[:N]
            new_c5 = np.max(corr_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_ref.copy()
    
    # Phase 3: Final tuning
    final_time = budget_s - time.time() + start_time
    if final_time > 5:
        try:
            for _ in range(500):
                if time.time() - start_time > budget_s - 5:
                    break
                
                H = fft.rfft(best_h)
                K = fft.rfft(1.0 - best_h)
                corr = fft.irfft(H * np.conj(K), n=2*N-1)
                corr = corr[:N]
                max_corr = np.max(corr)
                max_indices = np.where(np.abs(corr - max_corr) < 1e-10)[0]
                k_star = max_indices[0]
                g = np.zeros(N)
                if k_star >= 0:
                    g[:N-k_star] += (1 - best_h[k_star:])
                if k_star <= N-1:
                    g[k_star:] -= best_h[:-k_star] if k_star > 0 else best_h
                best_h = best_h - 0.01 * g
                best_h = np.clip(best_h, 0, 1)
                best_h = project_to_capped_simplex(best_h, N/2)
                
                corr_new = compute_autocorrelation_blocks(w_best, a_best, N) if 'w_best' in locals() else corr[:N]
                new_c5 = np.max(corr_new)
                if new_c5 < best_c5:
                    best_c5 = new_c5
        except:
            pass
    
    # Ensure final integral
    final_sum = np.sum(best_h) * dx
    if abs(final_sum - 1.0) > 1e-6:
        best_h = best_h * (1.0/final_sum) * (N/2)
    best_h = project_to_capped_simplex(best_h, N/2)
    
    return (best_h.tolist(), float(best_c5 * dx), N)
