# Precomputed map of shifts for each index
def precompute_shifts(n):
    shift_maps = []
    for s in range(-n+1, n):
        valid_pairs = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                valid_pairs.append((i, j))
        shift_maps.append(valid_pairs)
    return shift_maps

def correlations(h, shift_maps):
    """Compute all C_s values for given h."""
    n = len(h)
    C = np.zeros(2*n - 1)
    for s, pairs in enumerate(shift_maps):
        if len(pairs) == 0:
            continue
        cs = 0.0
        for i, j in pairs:
            cs += h[i] * (1 - h[j])
        C[s + n - 1] = cs
    return C

def smooth_obj_and_grad(h, alpha, shift_maps):
    """Return smooth approximation and its gradient."""
    C = correlations(h, shift_maps)
    max_C = np.max(C)
    if max_C == -np.inf:
        max_C = 0
    exp_C = np.exp(alpha * (C - max_C))
    Z = np.sum(exp_C)
    w = exp_C / Z
    grad = np.zeros_like(h)
    for s, pairs in enumerate(shift_maps):
        if len(pairs) == 0:
            continue
        w_s = w[s]
        for i, j in pairs:
            grad[i] += w_s * (1 - h[j] - h[i] if i != j else 1 - h[j])
    smooth_val = (1.0 / alpha) * (np.log(Z) + max_C)
    return smooth_val, grad

def project_to_capped_simplex(v, M):
    """Projection onto the capped simplex: [0,1]^n intersect with sum = M."""
    n = len(v)
    u = np.clip(v, 0, 1)
    sum_u = np.sum(u)
    if abs(sum_u - M) < 1e-10:
        return u
    # Sort v descending, then compute the threshold
    sorted_indices = np.argsort(-v)
    total = 0
    for i in range(n):
        idx = sorted_indices[i]
        total += v[idx]
        if total >= M + (i + 1) * (1 - v[idx]):
            theta = (total - M) / (i + 1)
            break
    else:
        theta = np.max([0, (sum_u - M) / n])
    h = np.clip(v - theta, 0, 1)
    return h

def run_descent(h0, α_schedule, η0, shift_maps, max_inner_iter, time_limit):
    start = time.time()
    h = h0.copy()
    η = η0
    for α in α_schedule:
        if time.time() - start > time_limit:
            break
        for _ in range(max_inner_iter):
            if time.time() - start > time_limit:
                break
            _, grad = smooth_obj_and_grad(h, α, shift_maps)
            v = h - η * grad
            h = project_to_capped_simplex(v, len(h)/2)
            η *= 0.95
    return h

def random_binary(n, M):
    indices = np.random.choice(n, size=M, replace=False)
    h = np.zeros(n)
    h[indices] = 1
    return h

def random_fractional(n, M):
    # Dirichlet with α=1 gives uniform over probability simplex
    v = np.random.dirichlet([1]*n) * M
    return project_to_capped_simplex(v, M)

def swap_local_search(h_bin, shift_maps, max_iter, time_limit):
    start = time.time()
    h = h_bin.copy()
    C_base = correlations(h, shift_maps)
    max_corr = np.max(C_base)
    max_s = np.argmax(C_base)
    idx = max_s - len(shift_maps) + 1

    for _ in range(max_iter):
        if time.time() - start > time_limit:
            break
        # Try to improve max_corr by swapping indices
        max_s = np.argmax(C_base)
        idx = max_s - len(shift_maps) + 1
        # Find source, target in a few steps
        pairs = shift_maps[max_s + len(shift_maps) - 1]
        if not pairs:
            continue
        swapped = False
        for i, j in pairs:
            # if h[i] == 1 and h[j] == 0, then we can swap them if beneficial
            # Compute change in max_s
            old = C_base[max_s]
            if h[i] == 1 and h[j] == 0:
                # Simulate swap by changing values temporarily
                h[i], h[j] = 0, 1
                C_test = correlations(h, shift_maps)
                new_max = np.max(C_test)
                if new_max < max_corr:
                    C_base = C_test
                    max_corr = new_max
                    swapped = True
                else:
                    h[i], h[j] = 1, 0
                if swapped:
                    break
        if not swapped:
            break
    return h
