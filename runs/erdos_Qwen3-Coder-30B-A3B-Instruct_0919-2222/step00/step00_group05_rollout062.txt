```python
import numpy as np
import time
import random
from scipy import signal
from scipy.optimize import minimize
import math

def initial_h_values(n_points=70):
    # Initial construction from the problem
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Simulate some good initial pattern
    for i in range(n_points):
        x = i * dx
        if 0.2 <= x <= 0.8:
            h[i] = 1.0
        elif 1.2 <= x <= 1.8:
            h[i] = 1.0
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def evaluate(h, dx):
    # Compute overlap integral using correlation
    corr = signal.correlate(h, 1 - h, mode="full")
    corr = corr * dx
    return np.max(corr)

def project_onto_constraints(h, dx):
    # Ensure h[i] is in [0,1] and sum(h)*dx = 1
    h = np.clip(h, 0, 1)
    h = h * (1.0 / np.sum(h)) * (1.0 / dx)
    return h

def generate_random_feasible_h(n_points, dx):
    # Generate a random feasible h
    h = np.random.rand(n_points)
    h = project_onto_constraints(h, dx)
    return h

def perturb_h(h, dx, temperature):
    # Apply a small perturbation
    h_new = h.copy()
    i = random.randint(0, len(h) - 1)
    delta = random.uniform(-temperature, temperature)
    h_new[i] += delta
    return project_onto_constraints(h_new, dx)

def perturb_h_2(h, dx, temperature):
    # Try to improve by swapping values
    h_new = h.copy()
    i, j = random.sample(range(len(h)), 2)
    delta = random.uniform(-temperature, temperature)
    h_new[i] += delta
    h_new[j] -= delta
    return project_onto_constraints(h_new, dx)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    elapsed = 0.0
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    restarts = 0
    max_restarts = 100
    
    # Initial construction
    current_h = initial_h_values(n_points)
    current_c5 = evaluate(current_h, dx)
    
    if current_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = current_c5
    
    # Annealing parameters
    temp_start = 0.1
    temp_end = 0.0001
    cooling_rate = 0.9995
    
    # Allow for city block search with simulated annealing approach
    T = temp_start
    iteration = 0
    max_iter = 50000
    last_update = time.time()
    
    while elapsed < budget_s and iteration < max_iter and restarts < max_restarts:
        iteration += 1
        
        # Reset temperature occasionally
        if iteration % 5000 == 0:
            T = max(T * cooling_rate, temp_end)
        
        # Choose perturbation type
        if random.random() < 0.7:
            proposed_h = perturb_h(current_h, dx, T)
        else:
            proposed_h = perturb_h_2(current_h, dx, T)
        
        proposed_c5 = evaluate(proposed_h, dx)
        
        # Accept or reject
        if proposed_c5 < current_c5 or random.random() < math.exp(-(proposed_c5 - current_c5) / T):
            current_h = proposed_h
            current_c5 = proposed_c5
    
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        elapsed = time.time() - start_time
        
        # Occasionally check if we should restart due to stagnation
        if elapsed > 0.1 and elapsed > restarts * 10:
            current_h = generate_random_feasible_h(n_points, dx)
            current_c5 = evaluate(current_h, dx)
            T = temp_start
            restarts += 1
            iteration = 0  # reset iteration for new loop
        
    if best_h is None:
        # fallback
        best_h = initial_h_values(n_points)
        best_c5 = evaluate(best_h, dx)
    
    return (best_h, best_c5, n_points)
```
