def evaluate_erdos_solution(h_vals):
    n = len(h_vals)
    dx = 2.0 / n
    # Compute the correlation of h with (1-h), which gives us the overlap integrals
    # We need to do this for all possible shifts k from 0 to 2
    corr = np.correlate(h_vals, 1 - np.array(h_vals), mode='full')
    # The first n-1 elements are the correlations for k > 0, second half gives k < 0
    # The maximum correlation is what we want to minimize
    max_over = np.max(corr)
    return max_over * dx
