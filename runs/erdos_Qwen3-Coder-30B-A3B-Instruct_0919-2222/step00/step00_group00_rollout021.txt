import numpy as np
import time
import random

# Predefined initial solution
initial_h_values = None

def compute_correlation(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap(h, dx):
    c = compute_correlation(h)
    return c.max() * dx

def apply_delta(c, h, idx, delta):
    n = len(h)
    offset = n - 1
    # First term contribution
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # Second term contribution
    for j in range(0, idx + 1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(i, j, c, h):
    delta_i = 0 - h[i]
    delta_j = 1 - h[j]
    
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def targeted_swap(c, h):
    n = len(h)
    offset = n - 1
    max_idx = np.argmax(c)
    max_shift = max_idx - offset
    s = max_shift
    
    # Find candidates for swapping
    ones = [i for i in range(n) if h[i] == 1]
    zeros = [i for i in range(n) if h[i] == 0]
    
    # Candidates for i (has 1, s-shift points to 0)
    candidates_i = []
    # Candidates for j (has 0, s-shift points to 1)
    candidates_j = []
    
    for i in ones:
        if 0 <= i + s < n and h[i + s] == 0:
            candidates_i.append(i)
    
    for j in zeros:
        if 0 <= j + s < n and h[j + s] == 1:
            candidates_j.append(j)
    
    if candidates_i and candidates_j:
        i = random.choice(candidates_i)
        j = random.choice(candidates_j)
        return (i, j)
    return None

def hill_climb(h, n_iter, time_left, dx):
    n = len(h)
    c = compute_correlation(h)
    
    best_h = h.copy()
    best_val = max_overlap(h, dx)
    ones = [i for i in range(n) if h[i] == 1]
    zeros = [i for i in range(n) if h[i] == 0]
    
    iterations = 0
    while iterations < n_iter and time_left() > 0.1:
        if random.random() < 0.7:  # 70% targeted swaps
            pair = targeted_swap(c, h)
            if pair:
                i, j = pair
                swap(i, j, c, h)
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
        else:
            # Random swap
            i = random.choice(ones)
            j = random.choice(zeros)
            swap(i, j, c, h)
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        
        current_val = max_overlap(h, dx)
        if current_val < best_val:
            best_h = h.copy()
            best_val = current_val
        
        iterations += 1
        
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)
    
    if initial_h_values is None:
        # Create initial binary representation (n=45)
        n0 = 45
        h0 = np.zeros(n0)
        ones_indices = random.sample(range(n0), n0//2)
        for idx in ones_indices:
            h0[idx] = 1
        h0 = h0.astype(float)
    else:
        n0 = len(initial_h_values)
        h0 = initial_h_values.copy()
    
    n_values = [60, 80, 100, 120, 140, 160]
    best_h = None
    best_c5 = float('inf')
    
    for n in n_values:
        if time_left() < 5:
            break
            
        dx = 2.0 / n
        h = np.zeros(n)
        
        # Interpolate initial value if needed
        if n <= n0:
            h = h0[:n].copy()
        else:
            # Upsample with linear interpolation
            x0 = np.linspace(0, 2, n0)
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x0, h0)
        
        # Ensure sum = n/2
        current_sum = np.sum(h)
        scale = (n / 2) / current_sum
        h *= scale
        h = np.clip(h, 0, 1)  # Ensure values in [0,1]
        
        # Ensure exact sum
        diff = (n / 2) - np.sum(h)
        for i in range(n):
            if abs(diff) < 1e-8:
                break
            if h[i] < 1 and diff > 0:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif h[i] > 0 and diff < 0:
                subtract = min(-diff, h[i])
                h[i] -= subtract
                diff += subtract
        
        h = h.astype(float)
        
        # Deterministic binary construction to improve the interpolation results
        binary_h = np.zeros(n)
        ones_indices = random.sample(range(n), n//2)
        for idx in ones_indices:
            binary_h[idx] = 1
        binary_h = binary_h.astype(float)
        
        # Try hill climbing from both initial solutions
        for h_start in [h, binary_h]:
            # For different numbers of iterations wrt budget and n
            if n <= 100:
                attempts = min(100000, int(time_left() * 1000))
            elif n <= 120:
                attempts = min(80000, int(time_left() * 800))
            else:
                attempts = min(50000, int(time_left() * 500))
                
            if time_left() < 1:
                break
            
            final_h, final_c5 = hill_climb(h_start, attempts, time_left, dx)
            
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
        
    if best_h is None:
        # Fallback if nothing worked
        best_h = np.full(100, 0.5)
        best_c5 = 0.5
        
    return (list(best_h), best_c5, len(best_h))
