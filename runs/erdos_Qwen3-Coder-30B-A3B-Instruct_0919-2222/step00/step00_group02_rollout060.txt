import numpy as np
import time
import random

def evaluate_h(h):
    """Evaluate the maximum overlap integral for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode='full')
    return O_counts.max() * dx

def max_overlap_counts(h):
    """Return the maximum overlap count."""
    O_counts = np.correlate(h, 1 - h, mode='full')
    return O_counts.max()

def random_balanced_binary(n, rng):
    """Generate a random balanced binary vector of length n."""
    h = np.zeros(n, dtype=int)
    ones = rng.sample(range(n), n // 2)
    h[ones] = 1
    return h

def project_onto_box_and_sum(v, target):
    """Project vector v onto [0,1] and sum = target using water-filling."""
    v = np.array(v, dtype=float)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def binary_anneal(n, time_limit, rng):
    """Perform simulated annealing on binary vectors."""
    h = random_balanced_binary(n, rng)
    cur_val = max_overlap_counts(h)
    best_val = cur_val
    best_h = h.copy()
    start_time = time.time()

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    iter = 0
    while time.time() - start_time < time_limit:
        iter += 1
        T = 0.5 * (1.0 - (time.time() - start_time) / time_limit) + 0.001
        
        # Choose random positions
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        # Swap values
        h[p], h[q] = 0, 1
        
        new_val = max_overlap_counts(h)
        delta = new_val - cur_val
        
        accept = False
        if delta <= 0:
            accept = True
        else:

                accept = True
        
        if accept:

                best_val = cur_val
                best_h = h.copy()
        else:
            # Revert
            h[p], h[q] = 1, 0
            
        # Update indices
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    return best_h, best_val
