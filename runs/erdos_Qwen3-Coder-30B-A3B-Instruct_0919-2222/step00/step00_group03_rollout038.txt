import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts k
    conv = np.correlate(h, 1 - h, mode="full")
    # The convolution result has length 2*n - 1
    # The relevant values are from index n-1 to 2*n-2
    # These correspond to shifts from 0 to 2
    overlap_values = conv[n-1:] * dx
    return np.max(overlap_values)

def initial_h_values():
    """Initial construction - piecewise constant function"""
    n = 70
    dx = 2.0 / n
    h = np.zeros(n)
    # Start with a simple design:
    # Make a step function that is 1 on a central portion and 0 elsewhere
    # But rescale to make the integral = 1
    mid_start = n // 3
    mid_end = 2 * n // 3
    h[mid_start:mid_end] = 1.0
    # Scale it so that the integral equals 1
    h *= 1.0 / (np.sum(h) * dx)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Starting point 
    h = initial_h_values()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Set up a simple optimization with gradient-like steps
    # We'll do gradient descent on the objective (with constraints)
    # Just track projected gradient steps in the feasible region
    
    # Refine using more direct search
    # Perform a few iterations of coordinate descent
    max_iter = 10000
    beta = 0.1  # learning rate
    tol = 1e-6
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Save current solution
        current_c5 = evaluate_erdos_solution(h)
        
        # Try small perturbations to see if we can improve
        for i in range(n):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Sample around h[i] to see if we can decrease the objective
            # We'll try several small changes
            h_backup = h[i]
            h_plus = min(h[i] + 0.01, 1.0)
            h_minus = max(h[i] - 0.01, 0.0)
            
            if h[i] != h_plus:
                h[i] = h_plus
                c5_plus = evaluate_erdos_solution(h)
            else:
                c5_plus = current_c5
                
            if h[i] != h_minus:
                h[i] = h_minus
                c5_minus = evaluate_erdos_solution(h)
            else:
                c5_minus = current_c5
            
            # Revert to the backup
            h[i] = h_backup
            
            # Pick the best, if either was better
            if c5_plus < c5_minus and c5_plus < current_c5:
                h[i] = h_plus
                current_c5 = c5_plus
            elif c5_minus < c5_plus and c5_minus < current_c5:
                h[i] = h_minus
                current_c5 = c5_minus
                
            # Ensure the integral constraint is still satisfied
            integral_err = np.sum(h) * dx - 1.0
            if abs(integral_err) > 1e-6:
                # Adjust all values to maintain integral constraint
                adjustment = integral_err / n
                h -= adjustment
                h = np.clip(h, 0, 1)
        
        # Also try to slightly shift values in an attempt to increase diversity
        # This may help escape local optima
        if iter_count % 50 == 0 and iter_count > 0:
            # Slightly adjust all values 
            noise = np.random.normal(0, 0.005, n)
            h += noise
            h = np.clip(h, 0, 1)
            # Maintain integral
            integral_err = np.sum(h) * dx - 1.0
            h -= integral_err / n
            
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    return (best_h.tolist(), best_c5, n)
