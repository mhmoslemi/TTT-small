import time
import numpy as np
import cvxpy as cp
from cvxpy import CBC, ECOS_BB

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = None

    # Candidate sizes to try (smaller number of points is faster to optimize)
    sizes = [80, 120, 160]
    
    # Safety margin for remaining time
    safety_margin = 5.0
    
    for n in sizes:
        if time.time() - t_start > budget_s - safety_margin:
            break
        
        dx = 2.0 / n
        K0 = max(1, n // 4)  # initial set of shifts to consider
        
        # Build coarse solution from previous solutions if possible, else use random
        x_init = np.zeros(n, dtype=int)
        target_sum = n // 2
        ones_indices = np.random.choice(n, target_sum, replace=False)
        x_init[ones_indices] = 1
        
        # Try upsampling from previous solution if possible
        if best_h is not None and len(best_h) * 2 == n:
            # Upsample existing solution
            upsampled = np.repeat(best_h, 2)
            x_init = upsampled[:n]
        
        # Apply symmetry breaking and monotonicity constraints
        x_init[0] = 1
        if n > 1:
            x_init[n-1] = 0
        # Enforce non-increasing prefix
        for i in range(n-1):
            if x_init[i] < x_init[i+1]:
                x_init[i+1] = x_init[i]
        
        # Lazy constraint loop
        current_x = x_init.copy()
        K = list(range(1, min(K0 + 1, n)))  # initial shift set
        t_best = None
        
        while True:
            # Break early if close to end of time
            if time.time() - t_start > budget_s - safety_margin:
                break
                
            try:
                # Create problem
                x = cp.Variable(n, boolean=True)
                t = cp.Variable()
                
                # Initialize constraints
                constraints = [cp.sum(x) == n // 2]
                constraints.append(x[0] == 1)
                if n > 1:
                    constraints.append(x[n-1] == 0)
                
                # Enforce prefix monotonicity
                for i in range(n-1):
                    if i < n//2 - 1:
                        constraints.append(x[i] >= x[i+1])
                
                # Create auxiliary variables and linearize the overlaps
                y_vars = {}  # store by (i,k) key
                
                for k in K:
                    for i in range(n - k):
                        y_var = cp.Variable(boolean=True)
                        y_vars[(i,k)] = y_var
                        constraints.extend([
                            y_var <= x[i],
                            y_var <= x[i+k],
                            y_var >= x[i] + x[i+k] - 1
                        ])
                
                # Compute the overlap terms and add constraints
                for k in K:
                    first_term = cp.sum(x)
                    second_term = cp.sum([y_vars[(i,k)] for i in range(n-k)])
                    constraints.append(first_term - second_term <= t)
                
                # Objective
                prob = cp.Problem(cp.Minimize(t), constraints)
                
                # Solve
                prob.solve(solver=CBC, verbose=False, max_seconds=30)
                
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    t_best = t.value
                    current_x = x.value.round().astype(int)
                else:
                    t_best = None
                    print(f"MILP solution failed for n={n}, K={len(K)}")
                    break
                
                # Full validation check
                c_full = np.correlate(current_x, 1 - current_x, mode="full")
                max_c_full = np.max(c_full)
                
                if max_c_full <= t_best + 1e-6:
                    # Solution is good enough
                    break
                else:
                    # Find which shift violates
                    k_violated = np.argmax(c_full)
                    if k_violated not in K:
                        K.append(k_violated)
                        K.sort()  # Keep sorted
                    else:
                        # Try to include more
                        if len(K) < n - 1:
                            K.append(min(n-1, k_violated + 1))
                            K.sort()
                        else:
                            break
                            
            except Exception as e:
                print(f"Exception during optimization for n={n}: {e}")
                break
    
        if t_best is not None:
            c5_bound = t_best * dx
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = current_x.copy()
                best_n = n
        
        # Local post-processing
        if best_h is not None and time.time() - t_start < budget_s - safety_margin:
            # First compute full correlation
            c_full = np.correlate(best_h, 1 - best_h, mode="full")
            k_star = np.argmax(c_full)
            c_max = c_full[k_star]
            
            while True:
                # Find violating pairs
                A = []  # 1s that would benefit deletion
                B = []  # 0s that would benefit addition
                for i in range(len(best_h)):
                    j = i + k_star
                    if j < len(best_h):
                        if best_h[i] == 1 and best_h[j] == 0:
                            A.append(i)
                        elif best_h[i] == 0 and best_h[j] == 1:
                            B.append(i)
                
                if not A or not B:
                    break
                
                # Perform one swap
                i_swap = A[0]
                j_swap = B[0]
                temp = best_h[i_swap]
                best_h[i_swap] = best_h[j_swap]
                best_h[j_swap] = temp
                
                # Update correlation incrementally (lower bound for t)
                c_full = np.correlate(best_h, 1 - best_h, mode="full")
                new_c_max = np.max(c_full)
                
                if new_c_max >= c_max:
                    # Reverse the swap if it caused a problem
                    best_h[i_swap] = best_h[j_swap]
                    best_h[j_swap] = temp
                    break
                else:
                    c_max = new_c_max
                    
    if best_h is not None:
        h_vals = best_h.astype(float)
        c5_bound = best_c5
        n_points = best_n
