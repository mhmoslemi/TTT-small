import numpy as np
from scipy.optimize import minimize
import time

def evaluate_C5(h):
    # h is a numpy array of length n_points
    n_points = len(h)
    dx = 2.0 / n_points
    # compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1-h, mode="full")
    # only take the relevant part for k=0 to k=2 (i.e. indices 2*n_points-2 to n_points-1)
    # actually, we want the max of all correlations which can be retrieved by using "full" mode
    max_corr = np.max(corr)
    return max_corr * dx
