import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation integral
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value among all shifts
    max_correlation = np.max(correlation)
    return max_correlation * dx

def initial_h_values():
    """Return initial construction"""
    n = 47
    h = np.ones(n)
    # Set some values to 0 to make the integral constraint feasible
    h[:15] = 0
    h[15:32] = 1
    h[32:] = 0
    # Normalize to satisfy constraint
    h = h * 2.0 / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1, i.e., sum(h) = n_points / 2
    target_sum = n_points / 2
    
    # Objective function to minimize
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Constraint function
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the constraint dictionary
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Initialize best solution
    best_h = h0.copy()
    best_c5 = objective(best_h)
    
    # Annealing with different perturbations
    num_iterations = 10000
    for iteration in range(num_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate a new candidate solution by perturbing current one
        h_new = best_h.copy()
        
        # Randomly perturb some elements
        num_perturb = max(1, n_points // 20)
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        for idx in indices:
            # Add small random perturbation
            delta = np.random.normal(0, 0.05)
            h_new[idx] += delta
            
        # Ensure h_new remains within bound
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum
            
        # Evaluate
        c5_new = objective(h_new)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Occasionally try larger moves with fixed pattern that helps
        if iteration % 100 == 0 and iteration > 0:
            # Try a pattern that might improve the overlap
            h_pattern = h0.copy()
            # Shift by some random amount
            shift = np.random.randint(0, n_points//4)
            h_pattern = np.roll(h_pattern, shift)
            # Soften boundary
            h_pattern = np.clip(h_pattern + np.random.normal(0, 0.01, n_points), 0, 1)
            
            # Renormalize
            current_sum = np.sum(h_pattern)
            if current_sum > 0:
                h_pattern = h_pattern * target_sum / current_sum
                
            c5_pattern = objective(h_pattern)
            if c5_pattern < best_c5:
                best_h = h_pattern.copy()
                best_c5 = c5_pattern
    
    return (best_h.tolist(), best_c5, n_points)
