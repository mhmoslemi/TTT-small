import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h_vals):
    """Evaluate the C5 constant for given h values."""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_vals)
    h = np.array(h_vals)
    integral = np.sum(h) * dx
    h = h / integral
    
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum value in the correlation
    # but we need to adjust for the fact that the correlation is 
    # computed over the full range
    c5 = np.max(corr) * dx
    return c5
