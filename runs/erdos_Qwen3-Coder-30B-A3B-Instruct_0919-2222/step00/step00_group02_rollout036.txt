import cvxpy as cp
import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h_conj = 1 - h
    correlations = correlate(h, h_conj, mode='full')
    max_overlap = np.max(correlations)
    c5 = max_overlap * dx
    return c5

def solve_relaxation(h0, n_points, max_iter=1000, tol=1e-6):
    dx = 2.0 / n_points
    # initialize with small set of shifts
    K = set([0, -1, 1, -2, 2])
    h = cp.Variable(n_points)
    t = cp.Variable(1)
    # bounds on h
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
    # build constraints from current K
    objectives = []
    w_vars = {}
    for k in K:
        # indices where i+k is valid
        valid_i = np.arange(max(0, -k), min(n_points, n_points - k))
        if len(valid_i) == 0:
            continue
        # create auxiliary variable for products
        w = cp.Variable(len(valid_i))
        w_vars[k] = w
        # McCormick envelopes
        for i, ii in enumerate(valid_i):
            i_k = ii + k
            constraints += [
                w[i] >= h[ii] + h[i_k] - 1,
                w[i] <= h[ii],
                w[i] <= h[i_k],
                w[i] >= 0
            ]
        # add objective constraint
        obj = cp.sum(h[valid_i]) - cp.sum(w)
        objectives.append(obj)
    # constraint that all overlap goals are less than t
    for o in objectives:
        constraints.append(o <= t)
    # minimize t
    problem = cp.Problem(cp.Minimize(t), constraints)
    start_time = time.time()
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > 595:
            break
        try:
            problem.solve(solver=cp.ECOS, verbose=False, max_iters=5000)
        except:
            break
        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            break
        # compute all overlaps exactly
        h_val = h.value
        if h_val is None:
            break
        # recompute the correlations with current h
        h_conj = 1 - h_val
        correlations = correlate(h_val, h_conj, mode='full')
        argmax_k = np.argmax(correlations)
        max_overlap = correlations[argmax_k]
        actual_t = max_overlap * dx
        if actual_t <= t.value + tol:
            break
        # add the new constraint
        K.add(argmax_k - (n_points - 1))
        # some safety measure
        if len(K) > 30:
            break
        # recreate constraints with new K
        new_constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        new_objectives = []
        new_w_vars = {}
        for k in K:
            valid_i = np.arange(max(0, -k), min(n_points, n_points - k))
            if len(valid_i) == 0:
                continue
            w = cp.Variable(len(valid_i))
            new_w_vars[k] = w
            for i, ii in enumerate(valid_i):
                i_k = ii + k
                new_constraints += [
                    w[i] >= h[ii] + h[i_k] - 1,
                    w[i] <= h[ii],
                    w[i] <= h[i_k],
                    w[i] >= 0
                ]
            obj = cp.sum(h[valid_i]) - cp.sum(w)
            new_objectives.append(obj)
        for o in new_objectives:
            new_constraints.append(o <= t)
        problem = cp.Problem(cp.Minimize(t), new_constraints)
    if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h0.copy(), 1
    h_val = h.value
    return h_val, t.value * dx

def rounding(h_fractional, n_points):
    h_binary = np.zeros(n_points)
    h_copy = h_fractional.copy()
    # Ensure the exact sum
    target_sum = n_points / 2.0
    current_sum = np.sum(h_copy)
    diff = current_sum - target_sum
    # Adjust for precision errors
    h_copy = np.clip(h_copy, 0, 1)
    # perform swap rounding
    while True:
        existing = np.where((h_copy > 0) & (h_copy < 1))[0]
        if len(existing) < 2:
            break
        # pick two at random
        idxs = np.random.choice(existing, 2, replace=False)
        i, j = idxs[0], idxs[1]
        if abs(h_copy[i] - h_copy[j]) < 1e-10:
            # if same fractional part, swap
            d = min(h_copy[i], 1 - h_copy[j])
            if np.random.rand() < 0.5:
                h_copy[i], h_copy[j] = h_copy[i] - d, h_copy[j] + d
            else:
                h_copy[i], h_copy[j] = h_copy[i] + d, h_copy[j] - d
        else:
            # regular swap
            d = min(h_copy[i], 1 - h_copy[j])
            if np.random.rand() < 0.5:
                h_copy[i], h_copy[j] = h_copy[i] - d, h_copy[j] + d
            else:
                h_copy[i], h_copy[j] = h_copy[i] + d, h_copy[j] - d
        if abs(np.sum(h_copy) - target_sum) < 1e-12:
            break
    h_binary = np.round(h_copy)
    return h_binary

def local_improve(h_binary, n_points, max_iter=500):
    h_current = h_binary.copy()
    c5_current = evaluate_c5(h_current, n_points)
    for _ in range(max_iter):
        if c5_current <= 0.3807:
            break
        # find worst shift
        h_conj = 1 - h_current
        correlations = correlate(h_current, h_conj, mode='full')
        argmax_k = np.argmax(correlations)
        max_k = argmax_k - (n_points - 1)
        # get indices for this shift
        i_indices, j_indices = [], []
        for i in range(n_points):
            j = i + max_k
            if 0 <= j < n_points:
                i_indices.append(i)
                j_indices.append(j)
        if len(i_indices) == 0:
            continue
        # try each pair
        best_delta = 0
        swap_i, swap_j = -1, -1
        for i, j in zip(i_indices, j_indices):
            if h_current[i] != h_current[j]:
                continue
            if h_current[i] == 1:
                # try to make h[i]=0, h[j]=1, but only if it helps
                new_h = h_current.copy()
                new_h[i] = 0
                new_h[j] = 1
                new_c5 = evaluate_c5(new_h, n_points)
                delta = c5_current - new_c5
                if delta > best_delta:
                    best_delta = delta
                    swap_i, swap_j = i, j
            else:
                # try to make h[i]=1, h[j]=0
                new_h = h_current.copy()
                new_h[i] = 1
                new_h[j] = 0
                new_c5 = evaluate_c5(new_h, n_points)
                delta = c5_current - new_c5
                if delta > best_delta:
                    best_delta = delta
                    swap_i, swap_j = i, j
        if swap_i != -1 and best_delta > 0:
            h_current[swap_i] = 1 - h_current[swap_i]
            h_current[swap_j] = 1 - h_current[swap_j]
            c5_current = evaluate_c5(h_current, n_points)
        else:
            break
    return h_current, c5_current

def sturmian(alpha, tau, n_points):
    h = np.zeros(n_points, dtype=int)
    for i in range(n_points):
        frac = (i * alpha + tau) % 1.0
        h[i] = 1 if frac < 0.5 else 0
    return h

def sturmian_optimize(n_points):
    best_c5 = 1.0
    best_h = None
    # search parameters (hardcoded)
    M = 200
    L = 100
    d_alpha = 0.003
    d_tau = 0.01
    a_range = np.arange(0.2, 0.8, d_alpha)
    t_range = np.arange(0, 1, d_tau)
    best_a = 0.5
    best_t = 0.0
    for a in a_range:
        for t in t_range:
            h = sturmian(a, t, n_points)
            c5 = evaluate_c5(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_a = a
                best_t = t
    # refine
    for _ in range(10000):
        a_delta = 1e-8 / 2.0
        if np.random.rand() < 0.5:
            a_test = best_a + np.random.uniform(-a_delta, a_delta)
        else:
            a_test = best_a - np.random.uniform(-a_delta, a_delta)
        t_test = best_t + np.random.uniform(-1e-6, 1e-6)
        if a_test < 0.2 or a_test > 0.8:
            continue
        h_test = sturmian(a_test, t_test, n_points)
        c5_test = evaluate_c5(h_test, n_points)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_a, best_t = a_test, t_test
    # final swap improvement
    h = sturmian(best_a, best_t, n_points)
    h, c5_final = local_improve(h, n_points, max_iter=300)
    return h, c5_final
