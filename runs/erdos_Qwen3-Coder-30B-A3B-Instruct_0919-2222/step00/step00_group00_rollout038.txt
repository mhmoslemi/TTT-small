import time
import numpy as np
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, GBB, ECOS_BB, CBC, GLPK, SOLVERS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_c5_bound = float('inf')
    best_h_values = None
    best_n_points = None
    allowed_n = [80, 120, 160]  # Small sizes for tractability
    max_k_initial = 30  # Limit initial active shifts, remove if this is ineffective
    safety_margin = 2.0  # s to preserve before breaking
    
    def evaluate_full_correlation(h):
        # Compute the full correlation fast
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr)
    
    def solve_milp_lazy(n, k_max, h_warm_start=None):
        dx = 2.0 / n
        # binary variables for h
        h_vars = Variable(n, boolean=True)
        # variable for the bound
        t = Variable()
        # set of k to include now
        active_shifts = list(range(1, min(k_max, n)))
        
        # auxiliary binary variables y[i,k]
        y_vars = {}
        constraints = []
        
        # mass constraint
        constraints.append(cp.sum(h_vars) == n // 2)
        # symmetry breaking
        constraints.append(h_vars[0] == 1)
        
        # non-increasing prefix
        for i in range(n // 2 - 1):
            constraints.append(h_vars[i] >= h_vars[i + 1])
        
        # set up shifts and variables
        y_vars = {}
        for k in active_shifts:
            y_vars[k] = Variable(n - k, boolean=True)
            for i in range(n - k):
                # McCormick linearization
                constraints.append(y_vars[k][i] <= h_vars[i])
                constraints.append(y_vars[k][i] <= h_vars[i + k])
                constraints.append(y_vars[k][i] >= h_vars[i] + h_vars[i + k] - 1)
        
        # Lagrangian and constraints for each shift
        for k in active_shifts:
            # s = sum(h[i]) - sum(y[i,k]) <= t
            expr = cp.sum(h_vars[:n-k]) - cp.sum(y_vars[k])
            constraints.append(expr <= t)
        
        # objective to minimize t
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        
        # solver check
        solver_available = False
        try:
            # Try CBC first (good open source solver)
            sol = prob.solve(solver=CBC, verbose=False, max_seconds=30)
            solver_available = True
        except:
            pass
        if not solver_available:
            try:
                sol = prob.solve(solver=ECOS_BB, verbose=False, max_seconds=30)
                solver_available = True
            except:
                pass
        
        if not solver_available:
            # fallback to fast solver, even if not guaranteed integer feasibility
            try:
                sol = prob.solve(solver=GLPK, verbose=False, max_seconds=30)
            except:
                return None, None
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h_vars.value.astype(int)
            t_sol = t.value
            return h_sol, t_sol
        # rely on incumbent if timed out or infeasible in some way
        if h_vars.value is not None:
            h_sol = h_vars.value.astype(int)
            t_sol = t.value if t.value is not None else float('inf')
            return h_sol, t_sol
        return None, None
    
    # Polishing step
    def polish_solution(h_sol, t_sol, n_points):
        dx = 2.0 / n_points
        # Get full correlation
        corr = np.correlate(h_sol, 1 - h_sol, mode="full")
        k_max_corr = np.argmax(corr)
        max_corr_value = corr[k_max_corr]
        # ensure we have valid probs for numerical stability
        if max_corr_value <= t_sol + 1e-5:
            return h_sol, t_sol
        # Try to improve only by local swaps if needed
        max_corr_value = int(max_corr_value)
        swaps = 0
        max_swaps = 50  # Limit attempts to avoid infinite loop
        while max_corr_value > t_sol and swaps < max_swaps:
            # Find pairs to swap (i,j) such that increasing shift by 1
            # Ensure validity
            ones = np.where(h_sol == 1)[0]
            zeros = np.where(h_sol == 0)[0]
            # Check build validity
            if len(ones) < 1 or len(zeros) < 1:
                break
            # build A (1's that lose a 1 if changed) and B (0's that become 1)
            A = []
            B = []
            for i in ones:
                j = i + k_max_corr
                if j < len(h_sol) and h_sol[j] == 0:
                    A.append(i)
            for j in zeros:
                i = j - k_max_corr
                if i >= 0 and h_sol[i] == 1:
                    B.append(j)
            # If we can swap
            if A and B:
                # try to swap one pair
                i = A[0]
                j = B[0]
                h_new = h_sol.copy()
                h_new[i] = 0
                h_new[j] = 1
                # compute new correlation
                corr_new = np.correlate(h_new, 1 - h_new, mode="full")
                k_max_new = np.argmax(corr_new)
                new_corr_val = corr_new[k_max_new]
                # improve if beneficial
                if new_corr_val <= t_sol:
                    h_sol = h_new
                    max_corr_value = new_corr_val
                    break
                elif new_corr_val < max_corr_value:
                    h_sol = h_new
                    max_corr_value = new_corr_val
                else:
                    swaps += 1
            else:
                break
            swaps += 1
        return h_sol, max_corr_value * dx

    # Run over multiple discretization levels
    for n in allowed_n:
        if time.time() - t_start > budget_s - safety_margin:
            break
        # try warm start
        h_prev = None
        if best_h_values is not None and len(best_h_values) * 2 == n:
            # upsample from previous best solution
            h_prev = np.repeat(best_h_values, 2)
            
        # try to solve for a given n
        # start with smaller k_max to reduce problem size
        k_max_search = min(max_k_initial, n // 2)
        
        t_start_iter = time.time()
        # Repeatedly allow more shifts until no extra shifts needed
        all_shifts_stable = False
        last_h = None
        last_t = None
        last_k_max = 0

        while not all_shifts_stable and time.time() - t_start < budget_s - safety_margin:
            h_sol, t_sol = solve_milp_lazy(n, k_max_search, h_prev)
            if h_sol is None or t_sol is None:
                break
            
            corr_full = evaluate_full_correlation(h_sol)
            max_k = len(h_sol)
            max_iterations = 10
            iterations = 0
            added_shifts = []

            while corr_full > t_sol and iterations < max_iterations:
                if time.time() - t_start > budget_s - safety_margin:
                    break
                # find where the violation is and add to active shifts
                corr = np.correlate(h_sol, 1 - h_sol, mode="full")
                k_violation = np.argmax(corr)
                if k_violation not in added_shifts:
                    added_shifts.append(k_violation)
                    # re-solve with this shift added
                    k_max_search = max(k_max_search, k_violation + 1)
                    iterations += 1
                    # solve again
                    h_sol, t_sol = solve_milp_lazy(n, k_max_search, h_prev)
                    if h_sol is None or t_sol is None:
                        break
                    corr_full = evaluate_full_correlation(h_sol)
                
            # polish if needed
            h_sol, c5_bound = polish_solution(h_sol, t_sol, n)
            dx = 2.0 / n
            c5_bound /= dx  # restore unscaled
                
            # Set the polishing offered solutions
            if c5_bound < best_c5_bound:
                best_c5_bound = c5_bound
                best_h_values = h_sol.astype(float)
                best_n_points = n

            # early termination
            if time.time() - t_start_iter > 100:
                break
