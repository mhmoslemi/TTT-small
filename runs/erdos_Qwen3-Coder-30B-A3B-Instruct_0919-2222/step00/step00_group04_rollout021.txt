import numpy as np
import time
from scipy.fft import fft, ifft
import math

def compute_bound(h, dx):
    # Compute correlation: h * (1 - h) using FFT for efficiency
    H = fft(h)
    K = fft(1 - h)
    correlation = ifft(H * np.conj(K)).real
    # The correlation array is of length 2*n-1, mode='full'
    # We're interested only in positive shifts (excluding the zero shift)
    # But since correlation is centered, the maximum over lag gives us the answer
    max_val = np.max(correlation)
    return max_val * dx

def compute_gradient(h, dx, worst_shifts):
    n = len(h)
    grad = np.zeros(n)
    
    # For each worst shift k
    for k in worst_shifts:
        # Add contribution of term (1 - h[i+k]) for valid indices
        if k >= 0:
            # i ranges from 0 to n-1-k
            i_range = slice(0, n - k)
            grad[i_range] += 1 - h[k:]
        else:
            # i ranges from -k to n-1
            i_range = slice(-k, n)
            grad[i_range] -= h[:n+k]
    
    return grad

def project_simplex(v, m):
    # Project vector v onto the simplex with sum = m and v_i in [0,1]
    # Using sorting algorithm
    n = len(v)
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    
    # Find threshold
    tmp_sum = 0.0
    rho = 0
    for i in range(n):
        tmp_sum += sorted_v[i]
        if tmp_sum > m:
            break
        rho += 1
    
    # Compute threshold
    if rho == 0:
        theta = (tmp_sum - m) / (n - rho)
    else:
        theta = (tmp_sum - m) / rho
    
    # Apply threshold
    result = np.maximum(v - theta, 0)
    result = np.minimum(result, 1)
    
    # Adjust to ensure sum = m
    diff = m - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute difference to elements that are not at bound
        # This should be done using soft projection or naive approach
        # For robustness, we just re-adjust by adding to smallest elements
        # Or ensure that sum is correct by rescaling the vector
        # Simple idea: adjust linearly or just force sum = m  
        result = np.clip(result + diff / n, 0, 1)
        if abs(np.sum(result) - m) > 1e-10:
            # Use more precise rescaling
            # Scale to meet target sum
            if np.sum(result) > 1e-10:
                result = result * (m / np.sum(result))
            else:
                # all zero, set uniform
                result.fill(m / n)
    
    # Final check
    if abs(np.sum(result) - m) > 1e-10 or np.min(result) < -1e-10:
        # Maybe too much precision loss. Exact computation:
        # Use simplex projection more robustly.
        # For simplicity: 
        result = np.clip(v, 0, 1)
        curr_sum = np.sum(result)
        diff = m - curr_sum
        if abs(diff) > 1e-10:
            idx = np.argsort(result)
            added = 0.0
            # Add diff to lowest elements
            for i in idx:
                if abs(added) >= abs(diff):
                    break
                to_add = min(1.0 - result[i], -diff)
                if to_add > 1e-10:
                    result[i] += to_add
                    added += to_add
            # Maybe overshoot, correct the rest
            if abs(np.sum(result) - m) > 1e-10:
                # Just use uniform rescaling
                result = np.clip(result * (m / np.sum(result)), 0, 1)
    return result
