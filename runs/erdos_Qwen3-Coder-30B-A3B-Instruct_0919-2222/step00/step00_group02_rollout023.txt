import numpy as np
import time
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # Problem parameters
    N = [30, 36, 40, 45, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    max_restarts = 5
    initial_h = initial_h_values
    best_h_global = None
    best_c_global = float('inf')
    start_time = time.time()

    def compute_corr(h):
        return np.correlate(h, 1-h, mode='full')

    def find_worst_shift(corr):
        # index in full correlation array
        max_idx = np.argmax(corr)
        # map back to actual shift k
        k = max_idx - (len(h) - 1)
        return k, corr[max_idx]

    def get_mismatched_pairs(h, k):
        n = len(h)
        if k >= 0:
            M1 = np.where((h[:n-k] == 1) & (h[k:] == 0))[0]
            M0 = np.where((h[:n-k] == 0) & (h[k:] == 1))[0]
        else:
            k = -k
            M1 = np.where((h[k:] == 1) & (h[:-k] == 0))[0]
            M0 = np.where((h[k:] == 0) & (h[:-k] == 1))[0]
        return M1, M0

    def binary_local_improvement(h, max_iters=2000, max_stall=1000):
        h = h.copy().astype(bool)
        corr = compute_corr(h)
        max_cnt = int(corr.max())
        total_time = time.time() - start_time
        max_time = budget_s * 0.98
        if total_time > max_time:
            return h, max_cnt
        for _ in range(max_iters):
            if time.time() - start_time > max_time:
                break
            k, count = find_worst_shift(corr)
            if count < max_cnt:
                max_cnt = count
                corr = compute_corr(h)
            if count == max_cnt:
                M1, M0 = get_mismatched_pairs(h, k)
                if len(M1) > 0 and len(M0) > 0:
                    i, j = np.random.choice(M1), np.random.choice(M0)
                    h[i], h[j] = h[j], h[i]
                    corr = compute_corr(h)
                    new_count = int(corr.max())
                    if new_count <= max_cnt:
                        max_cnt = new_count
                    else:
                        h[i], h[j] = h[j], h[i]  # Revert if no improvement
                else:
                    # random balanced swap if no valid pairs
                    ones = np.where(h)[0]
                    zeros = np.where(~h)[0]
                    if len(ones) > 0 and len(zeros) > 0:
                        i, j = np.random.choice(ones), np.random.choice(zeros)
                        h[i], h[j] = h[j], h[i]
                        corr = compute_corr(h)
                        new_count = int(corr.max())
                        if new_count <= max_cnt:
                            max_cnt = new_count
                        else:
                            h[i], h[j] = h[j], h[i]

        return h.astype(float), max_cnt

    def simulated_annealing_escape(h, max_iters=5000):
        h = h.copy()
        corr = compute_corr(h)
        max_cnt = corr.max()
        temp = 0.5
        temp_final = 1e-4
        alpha = np.log(temp_final / temp) / max_iters
        for i in range(max_iters):
            if time.time() - start_time > budget_s * 0.98:
                break
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i_new, j_new = np.random.choice(ones), np.random.choice(zeros)
            orig = h[i_new]
            h[i_new] = 0
            h[j_new] = 1
            corr_new = compute_corr(h)
            new_count = corr_new.max()
            delta = new_count - max_cnt
            if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
                max_cnt = new_count
                corr = corr_new
            else:
                h[i_new] = orig
                h[j_new] = 1 - orig
            temp *= np.exp(alpha)
        return h, int(max_cnt)

    for n in N:
        if time.time() - start_time > budget_s * 0.95:
            break
        if n == N[-1] and (time.time() - start_time < budget_s - 30):
            # Allow more time for final refinement
            max_time_for_n = budget_s - (time.time() - start_time) - 10
        else:
            max_time_for_n = budget_s * 0.9 / len(N)
        best_h_n = None
        best_count_n = float('inf')
        tried = 0

        while tried < max_restarts:
            if time.time() - start_time > budget_s * 0.98:
                break
            h = np.zeros(n)
            # initialize with current best if available, otherwise initial h
            if best_h_global is not None and len(best_h_global) == n:
                h = best_h_global.copy()
            else:
                h.fill(0)
                if n >= len(initial_h):
                    h[:len(initial_h)] = initial_h
                else:
                    h[:n] = initial_h[:n]
            # Force exact average
            h = h.astype(float)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h / h_sum * n / 2
            else:
                h = np.full(n, 0.5)

            # uniform balance at start
            n_ones = n // 2
            indices = np.random.choice(n, size=n_ones, replace=False)
            h[indices] = 1.0
            h = np.clip(h, 0, 1)

            # Step 1: deterministic local search
            h_local, count_local = binary_local_improvement(h, int(max_time_for_n/2), 500)
            h_local, count_local = simulated_annealing_escape(h_local, int(max_time_for_n * 0.2))

            if count_local < best_count_n:
                best_count_n = count_local
                best_h_n = h_local.copy()

            tried += 1

        if best_h_n is not None:
            c_n = 2 * best_count_n / n
            if c_n < best_c_global:
                best_c_global = c_n
                best_h_global = best_h_n.copy()

        # Early stop
        if best_c_global < 0.3807:
            break

    if best_h_global is None:
        best_h_global = initial_h_values
        corr = np.correlate(best_h_global, 1 - best_h_global, mode="full")
        best_c_global = 2 * np.max(corr) / len(best_h_global)

    return (best_h_global, best_c_global, len(best_h_global))
