def project_to_capped_simplex(v, target_sum):

    # (target_sum - sum_{i=0}^{j-1} sorted_v[i]) / j <= sorted_v[j]
    # Work from back to front to find the right number of non-zero elements
    # If we want final sum to be target_sum and current total is cumsum[j],

    # We'll use the correct method, water-filling.

def evaluate_solution(h_values):
    """Compute C5 for given h, assumed to be defined on [0,2] with 2/N dx."""
    h = np.array(h_values)
    N = len(h)
    dx = 2.0 / N
    # do full correlation
    corr = correlate(h, 1 - h, mode='full')
    # Cs for each shift, oscillating but lags go from -N+1 to N-1
    max_corr = np.max(corr)
    return max_corr * dx


def compute_overlap_function(a, w):

    # look at computing G(k) = integrals of h(x) * h(x+k) over all real x in [0, 2]
    # - Make discrete approximate: `x_j = j * 2/N = j*dx`
    # - M pairs of blocks give overlaps under shift k = j*dx (should map at most to -N+1 to N-1)

    # For this:
    # Let's compute r = correlation between h and (1-h) in O(N^2) with exact computation

def evaluate_block_fitness(x, M, N_fine=200, lambda_integral=1000):
    # x contains the float weights w_i (M elements) and a_i (M elements) with u_i, v_i being raw mapped to weight and sigmoid height
    # x vectors are: [u1, ..., uM, v1, ..., vM]
    u_vals = x[0:M]
    v_vals = x[M:2*M]
    
    # reparameterization using softmax and sigmoid to obtain w_i and a_i
    w = 2 * (np.exp(u_vals) / np.sum(np.exp(u_vals)))
    a = 1 / (1 + np.exp(-v_vals))
    
    # Check if integrality condition is met (for valid h) and add penalty if not.
    total_mass = np.dot(a, w)
    penalty = lambda_integral * abs(total_mass - 1.0)
    
    try:
        obj_value = compute_overlap_function(a, w)
    except Exception as e:
        # print("block eval error:", e)
        obj_value = 1e6  # substitute with large value if invalid computation
    
    return obj_value + penalty

        t = iter + 1
        alpha = 0.2 / np.sqrt(t)
        
        # Update
        h = h - alpha * g[:] 
        
        # Project - Euclidean projection onto the capped simplex
        sum_h = np.sum(h)
        h_projected = project_to_capped_simplex(h, sum_h) # But need to fix when the mass changes
        
        # compute correction factor for sum = N/2? manually enforce sum = N/2 identity via scaling
        h = h_projected
        
        # update best
        corr_full = correlate(h, 1-h, mode='full')
        corr_max = np.max(corr_full)
        if iter == 0 or corr_max < c5_bound:
            c5_bound = corr_max
            h_best = h.copy()
    
    # finalize
    c5_final = evaluate_solution(h_best)
    return (h_best.tolist(), c5_final, N)
