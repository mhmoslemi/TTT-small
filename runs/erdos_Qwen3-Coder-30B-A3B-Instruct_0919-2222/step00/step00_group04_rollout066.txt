def project_to_capped_simplex(v, target_sum):
    """Projects vector v onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum"""
    v = np.array(v)
    n = len(v)
    # Sort in descending order
    sorted_idx = np.argsort(-v)
    sorted_v = v[sorted_idx]
    
    # Find threshold tau
    tau = 0.0
    cumsum = 0.0
    for i in range(n):
        cumsum += sorted_v[i]
        if cumsum - (i + 1) * sorted_v[i] >= target_sum:
            continue
        else:
            tau = (cumsum - target_sum) / (i + 1)
            break
    else:
        # If no such i was found, all values are below target_sum / n, we still compute the last one
        tau = cumsum / n - target_sum / n
    
    # Apply threshold
    result = np.zeros(n)
    for i in range(n):
        result[sorted_idx[i]] = np.clip(v[sorted_idx[i]] - tau, 0, 1)
    
    # Adjust sum to exactly target_sum by scaling
    current_sum = np.sum(result)
    if current_sum > 0:
        result *= target_sum / current_sum
    
    return result

def compute_policy_overlap(h):
    """
    Compute the correct C5 using convolution
    """
    # h is sampled on [0,2] at N points; dx = 2/N
    N = len(h)
    dx = 2.0 / N

    # Use the np.correlate for full mode
    # This computes all valid autococorr values for all lags
    corr_full = correlate(h, 1-h, mode='full')
    # Lags from -N+1 to N-1, middle is about 0.
    # corresponding positions: j = 0 to 2*N-2
    # For the indices corresponding to positive shifts from 0 to N-1,
    # the corresp. lag is varying from -N+1 to N-1.
    # So:
    # for lag = - (N-1) to N-1 inclusive,
    #      corresponding indices are from 0 to 2*N-2 inclusive.
    # This gives lag = i - (N-1) where i from 0 to 2*N-2
    # Lag is from - (N-1) to (N-1)
    # Use the reverse approach to get C5 = max for the lags > 0.
    # Only take the positive lags.
    # Skip the first N-1 (negative lags), take from idx N-1 to 2*N-2
    max_corr = max(corr_full[N-1:])

    # Adjust for dx scaling
    return max_corr * dx
