import numpy as np
import time
from scipy.special import softmax
from math import log

def evaluate_overlap(h):
    """Evaluate the overlap integral for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def binary_search(n, budget_s):
    """Perform binary local search and simulated annealing for a fixed n."""
    np.random.seed(42)
    start_time = time.time()
    best_h = None
    best_val = float('inf')
    
    restarts = min(5, int(budget_s / 20)) if budget_s > 20 else 1
    restarts = max(1, restarts)
    
    for _ in range(restarts):
        if time.time() - start_time > budget_s - 2:
            break
        # Create a balanced binary vector
        h = np.zeros(n, dtype=int)
        ones = np.random.permutation(n)[:n//2]
        h[ones] = 1
        h = h.astype(float)
        
        max_cnt = float('inf')
        max_cnt_last_improvement = 0
        iterations_no_improvement = 0
        
        for _ in range(2 * n):
            if time.time() - start_time > budget_s - 2:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_cnt_old = max_cnt
            max_cnt = corr.max()
            s = corr.argmax() - (n - 1)
            
            # Find mismatched indices
            M1, M0 = [], []
            if s >= 0:
                M1 = np.where((h == 1) & (np.roll(h, -s) == 0))[0]
                M0 = np.where((h == 0) & (np.roll(h, -s) == 1))[0]
            else:
                M1 = np.where((h == 1) & (np.roll(h, -s) == 0))[0]
                M0 = np.where((h == 0) & (np.roll(h, -s) == 1))[0]
                
            if len(M1) > 0 and len(M0) > 0:
                i, j = np.random.choice(M1), np.random.choice(M0)
                h[i], h[j] = h[j], h[i]
            else:
                pick1 = np.where(h == 1)[0]
                pick0 = np.where(h == 0)[0]
                if len(pick1) > 0 and len(pick0) > 0:
                    i, j = np.random.choice(pick1), np.random.choice(pick0)
                    h[i], h[j] = h[j], h[i]
            
            new_corr = np.correlate(h, 1 - h, mode='full')
            new_max = new_corr.max()
            if new_max <= max_cnt:
                max_cnt = new_max
                iterations_no_improvement = 0
            else:
                iterations_no_improvement += 1
                if iterations_no_improvement > n:
                    break
            if new_max < best_val:
                best_val = new_max
                best_h = h.copy()
        
        # Simulated Annealing
        if best_h is not None and time.time() - start_time < budget_s - 2:
            h = best_h.copy()
            T0 = 0.5
            Tf = 1e-4
            max_iters = 5 * n * log(n) if n > 1 else 100
            if max_iters > 10000:
                max_iters = 10000
            T = T0
            alpha = (Tf / T0) ** (1.0 / max_iters)
            
            for _ in range(int(max_iters)):
                if time.time() - start_time > budget_s - 2:
                    break
                pick1 = np.where(h == 1)[0]
                pick0 = np.where(h == 0)[0]
                i, j = np.random.choice(pick1), np.random.choice(pick0)
                old_val = np.correlate(h, 1 - h, mode='full').max()
                h[i], h[j] = h[j], h[i]
                new_val = np.correlate(h, 1 - h, mode='full').max()
                delta = new_val - old_val
                if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                    if new_val < best_val:
                        best_val = new_val
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[j], h[i]
                T *= alpha
                
        # If no improvement from this restart, possibly update from the final state
        corr = np.correlate(h, 1 - h, mode='full')
        cval = corr.max()
        if cval < best_val:
            best_val = cval
            best_h = h.copy()

    return best_h, best_val

def continuous_refine(init_h, budget_s):
    """Refine the binary solution to a fractional solution."""
    start_time = time.time()
    
    # Soft-max method with increasing α
    h = init_h.copy()
    
    # Projected gradient descent on smooth surrogate
    for alpha in [1, 5, 10, 20, 40, 80]:
        if time.time() - start_time > budget_s - 2:
            break
        T_alpha = 200  # Gradient steps per alpha
        eta0 = 0.2
        for t in range(T_alpha):
            if time.time() - start_time > budget_s - 2:
                break
            # Get current overlap values
            corr_val = np.correlate(h, 1 - h, mode='full')
            o = corr_val
            # Softmax splitting
            max_o = o.max()
            exp_o = np.exp(alpha * (o - max_o))
            p = exp_o / exp_o.sum()
            # Compute gradient
            grad = np.zeros_like(h)
            for i in range(len(h)):
                grad[i] += 1 - h[(i + 1) % len(h)] if i + 1 < len(h) else 0
                grad[i] += 1 - h[(i - 1) % len(h)] if i - 1 >= 0 else 0
                # Add gradient contribution for each shift k
                for k, weight in enumerate(p):
                    if k < len(o) and k >= 0:
                        j = i + k - (len(h) - 1)
                        if 0 <= j < len(h):
                            grad[i] += weight * (1 - h[j])
                        if 0 <= j < len(h):
                            grad[i] += weight * (1 - h[j])
            # Apply gradient update
            eta = eta0 / np.sqrt(t + 1)
            h -= eta * grad
            # Project onto feasible set
            # Water-filling type projection
            def project_h(x):
                sum_x = x.sum()
                target = len(x) / 2
                if abs(sum_x - target) < 1e-10:
                    return x
                t = 0.0
                lo, hi = np.min(x) - 1, np.max(x) + 1
                for _ in range(40):
                    t = (lo + hi) / 2
                    clamped = np.clip(x - t, 0, 1)
                    if clamped.sum() < target:
                        hi = t
                    else:
                        lo = t
                return np.clip(x - t, 0, 1)
            
            h = project_h(h)
        
        # Compute true max for final check
        corr = np.correlate(h, 1 - h, mode='full')
        if corr.max() < 0.3807 * len(h) / 2:
            break  # Great bound, stop early
    
    corr = np.correlate(h, 1 - h, mode='full')
    return h, corr.max()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    best_h = None
    best_c = float('inf')
    start_time = time.time()
    
    # Candidate lengths
    N = [30,36,40,45,50,60,70,80,90,100,120,150,180,200]
    total_size = sum(N)
    best_global = None
    
    for n in N:
        time_budget = (n / total_size) * (budget_s - 40)
        if time.time() - start_time + time_budget > budget_s - 5:
            time_budget = budget_s - (time.time() - start_time) - 5
            if time_budget < 0:
                break
        if time_budget <= 0:
            break
            
        h, val = binary_search(n, time_budget)
        c = 2 * val / n
        if c < best_c:
            best_c = c
            best_h = h.copy()
            
        # If we beat the target early, stop
        if best_c < 0.3807:
            break
    
    # Optional final refinement for large n
    if best_h is not None:
        if len(best_h) >= 150:
            if time.time() - start_time < budget_s - 50:
                try:
                    h_frac, cnt_frac = continuous_refine(best_h, budget_s - (time.time() - start_time) - 20)
                    c_frac = 2 * cnt_frac / len(h_frac)
                    if c_frac < best_c:
                        best_c = c_frac
                        best_h = h_frac.copy()
                except Exception:
                    pass

    return (best_h, best_c, len(best_h))
