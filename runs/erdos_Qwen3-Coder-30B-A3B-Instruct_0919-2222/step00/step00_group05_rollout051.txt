from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap constant for a given step function h."""

    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode='full')
    # Return maximum correlation value multiplied by dx
    return np.max(corr) * dx
