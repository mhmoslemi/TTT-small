import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Time tracking
    start_time = time.time()
    end_time = start_time + budget_s * 0.99
    
    # Parameters
    n_points = 401  # Odd number for symmetry
    dx = 2.0 / n_points
    
    # Evaluate function
    def evaluate(h):
        # Since np.correlate with mode='full' will produce 2*n_points-1 values, 
        # we compute it once and take max * dx
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Generate initial shapes
    def create_triangle_shape(n):
        x = np.linspace(0, 2, n, endpoint=False) + dx/2  # Midpoints
        h = np.maximum(0, 1 - np.abs(x - 1))
        return h

    def create_cosine_shape(n, p):
        x = np.linspace(0, 2, n, endpoint=False) + dx/2
        h = np.cos(np.pi/2 * (x - 1))**p
        return h

    def create_polynomial_shape(n, a):
        x = np.linspace(0, 2, n, endpoint=False) + dx/2
        h = np.maximum(0, 1 - np.abs(x - 1)**a)
        return h

    # Normalize so integral = 1 (i.e., sum = n/2)
    def normalize(h):
        sum_h = np.sum(h)
        if abs(sum_h - n_points/2) > 1e-10:
            factor = (n_points/2) / sum_h
            h = h * factor
            # Clamp to [0,1] to avoid overshoot due to scaling
            h = np.clip(h, 0, 1)
        return h

    # Find best parameterized shape
    best_h = None
    best_c5 = float('inf')
    
    # Try triangle
    h = create_triangle_shape(n_points)
    h = normalize(h)
    c5 = evaluate(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Try cosine shapes with various exponents
    for p in [0.5, 1, 2, 4, 8]:
        h = create_cosine_shape(n_points, p)
        h = normalize(h)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    # Try polynomial shapes
    for a in [1, 2, 3]:
        h = create_polynomial_shape(n_points, a)
        h = normalize(h)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Start hill climbing from best shape
    h = best_h.copy()
    current_c5 = best_c5
    best_h = h.copy()
    best_c5 = current_c5
    
    step_size = 0.05
    max_steps = 200000  # Cap number of steps to keep within time budget
    step = 0
    
    while time.time() < end_time and step < max_steps:
        # Random pair selection
        i = random.randint(0, n_points - 1)
        j = random.randint(0, n_points - 1)
        
        if i == j:
            continue
            
        # Compute max delta
        max_up = min(1 - h[j], h[i])
        if max_up <= 0:
            continue
            
        delta = random.random() * min(max_up, step_size)
        
        # Move mass from i to j
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Ensure constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to ensure sum constraint
        sum_new = np.sum(h_new)
        factor = (n_points/2) / sum_new
        h_new = h_new * factor
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate
        c5_new = evaluate(h_new)
        
        # Accept or reject
        if c5_new < current_c5:
            h = h_new.copy()
            current_c5 = c5_new 
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        else:
            # Occasionally accept worse moves with small probability
            if random.random() < 0.01:
                h = h_new.copy()
                current_c5 = c5_new
        
        step += 1
        
    return list(best_h), best_c5, n_points
