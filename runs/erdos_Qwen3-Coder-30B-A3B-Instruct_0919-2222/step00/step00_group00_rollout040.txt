import numpy as np
import time
import random

# Pre-imported global variables
# initial_h_values: numpy array of shape (45,)

def compute_correlation(h):
    """Compute correlation between h and (1-h) using numpy correlate."""
    return np.correlate(h, 1 - h, mode='full')

def apply_delta(c, h, idx, delta):
    """Efficiently update correlation when changing h[idx] by delta."""
    n = len(h)
    offset = n - 1
    # First term contribution
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # Second term contribution
    for j in range(0, idx + 1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(i, j, c, h):
    """Swap elements i and j in h and update correlation accordingly."""
    old_i, old_j = h[i], h[j]
    delta_i = 0 - old_i
    delta_j = 1 - old_j
    
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def max_overlap(c, dx):
    """Return maximum overlap value."""
    return c.max() * dx

def hill_climb(h, n_iter, time_left, best_val):
    """Run hill climbing optimization."""
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h)
    current_val = max_overlap(c, dx)
    best_h = h.copy()
    best_val = current_val
    
    # Track indices of ones and zeros
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] < 0.5]
    
    iter_count = 0
    max_iter = n_iter
    
    while iter_count < max_iter and time_left() > 1:
        iter_count += 1
        
        # Find the shift with maximum correlation
        max_idx = np.argmax(c)
        max_shift = max_idx - (n - 1)
        
        # Try to find a good targeted swap
        # Collect candidates for a targeted swap around max_shift
        candidates_i, candidates_j = [], []
        
        # Find indices i where h[i] == 1 and h[i + max_shift] == 0, if defined
        for i in ones:
            if 0 <= i + max_shift < n and h[i + max_shift] == 0:
                candidates_i.append(i)
        
        # Find indices j where h[j] == 0 and h[j + max_shift] == 1, if defined
        for j in zeros:
            if 0 <= j + max_shift < n and h[j + max_shift] == 1:
                candidates_j.append(j)
        
        # If can find candidates, do targeted swap
        if candidates_i and candidates_j:
            i = random.choice(candidates_i)
            j = random.choice(candidates_j)
        else:
            # Otherwise do simple random swap
            if ones and zeros:
                i = random.choice(ones)
                j = random.choice(zeros)
            else:
                continue
                
        # Save current state to be able to revert
        old_c = c.copy()
        old_h = h.copy()
        old_h_i = h[i]
        old_h_j = h[j]
        
        # Perform the swap
        swap(i, j, c, h)
        
        # Compute new value
        new_val = max_overlap(c, dx)
        
        # Accept if it's better or if we're allowing worse
        if new_val < current_val:
            current_val = new_val
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # Revert
            c[:] = old_c
            h[:] = old_h
            ones = [i for i in range(n) if h[i] > 0.5]
            zeros = [i for i in range(n) if h[i] < 0.5]
    
    return best_h, best_val
