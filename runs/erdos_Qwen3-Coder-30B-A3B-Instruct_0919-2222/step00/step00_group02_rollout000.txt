import numpy as np
import time
import random

def evaluate_h(h):
    """Compute the maximum overlap integral C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation: O_k = sum_i h[i] * (1 - h[i+k])
    O_counts = np.correlate(h, 1 - h, mode="full")
    max_O = O_counts.max()
    return max_O * dx

def max_overlap_counts(h):
    """Compute the maximum overlap count (integer) for given h."""
    n = len(h)
    O_counts = np.correlate(h, 1 - h, mode="full")
    return int(O_counts.max())

def random_balanced_binary(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    ones = rng.sample(range(n), n // 2)
    h[ones] = 1
    return h

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto [0,1] box constraint with sum = target_sum."""
    v = np.array(v, dtype=float)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0

    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def continuous_descent(h0, time_limit):
    """Refine a binary solution using gradient descent with projection."""
    n = len(h0)
    h = h0.astype(float)
    best_h = h.copy()
    best_obj = evaluate_h(h)
    
    start_time = time.time()
    step_size = 0.1
    
    while time.time() - start_time < time_limit:
        O_counts = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(O_counts)
        shift = max_idx - (n - 1)

        # Compute gradient
        grad = np.zeros(n)
        if shift >= 0:
            left = np.zeros(n)
            right = np.zeros(n)
            left[shift:] = h[:-shift]
            right[:n-shift] = h[shift:]
            grad = 1 - left - right
        else:
            s = -shift
            left = np.zeros(n)
            right = np.zeros(n)
            left[:n-s] = h[s:]
            right[s:] = h[:-s]
            grad = 1 - left - right

        # Gradient descent step
        h_new = h - step_size * grad
        h_new = project_onto_box_and_sum(h_new, target_sum=n/2)

        # Evaluate new objective
        new_obj = evaluate_h(h_new)
        if new_obj < best_obj:
            best_obj = new_obj
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_obj

def binary_anneal(n, time_limit, rng, target_max=None):
    """Perform simulated annealing on binary step function."""
    h = random_balanced_binary(n, rng)
    cur_val = max_overlap_counts(h)
    best_val = cur_val
    best_h = h.copy()
    start_time = time.time()
    
    initial_temp = 1.0
    final_temp = 1e-4

    # Precompute indices for faster access
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    iter_count = 0
    while time.time() - start_time < time_limit:
        iter_count += 1
        elapsed = time.time() - start_time
        T = initial_temp * (final_temp / initial_temp) ** (elapsed / time_limit)
        
        # Choose two positions of opposite bits
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        # Swap bits
        h[p], h[q] = 0, 1
        new_val = max_overlap_counts(h)
        delta = new_val - cur_val
        
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
                
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            if target_max is not None and cur_val <= target_max:
                break
        else:
            h[p], h[q] = 1, 0
        
        # Update indices for next iteration
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
    return best_h, best_val

    candidate_n = [n for n in candidate_n if n % 2 == 0]
    
    # Allocate time for each n
    remaining_time = budget_s
    elapsed_time = 0
    
    for n in candidate_n:
        if elapsed_time >= budget_s:
            break
        time_per_n = remaining_time / (len(candidate_n) - candidate_n.index(n))
        time_for_this_n = min(time_per_n, remaining_time)
        if time_for_this_n < 1:
            break
            
        # Compute target maximum overlap count for our goal
        target_max = int(np.floor(0.3807 * n / 2))
        
        # Run several restarts for this n
        best_local_h = None
        lowest_local_max = float('inf')
        
        # Run 5 restarts per n
        restarts = min(5, int(100000 / n)) if n < 100 else 1
        
        for _ in range(restarts):
            if elapsed_time >= budget_s:
                break
                
            h, local_max = binary_anneal(n, time_for_this_n * 0.9, rng, target_max)
            if local_max < lowest_local_max:
                lowest_local_max = local_max
                best_local_h = h.copy()
                
            elapsed_time = time.time() - start_time
            remaining_time = budget_s - elapsed_time
            if remaining_time < 1:
                break
                
        if best_local_h is not None:
            # Refine using continuous descent
            refine_time = min(100, remaining_time / 3)
            if refine_time > 1:
                refined_h, refined_c5 = continuous_descent(best_local_h, refine_time)
                if refined_c5 < evaluate_h(best_local_h):
                    best_h = refined_h
                    best_c5 = refined_c5
                else:
                    best_h = best_local_h
                    best_c5 = evaluate_h(best_local_h)
            else:
                best_h = best_local_h
                best_c5 = evaluate_h(best_local_h)
                
        # Update for next iteration
        elapsed_time = time.time() - start_time
        remaining_time = budget_s - elapsed_time
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, len(best_h))
