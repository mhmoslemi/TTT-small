import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 constant for a given h vector."""
    dx = 2.0 / len(h)
    # Compute all correlations for shifts from -len(h)+1 to len(h)-1
    correlations = correlate(h, 1 - h, mode="full")
    # Normalize by dx
    correlations *= dx
    # Return the maximum value
    return np.max(correlations)

def build_mcormick_constraints(h, y, n, s):
    """Build McCormick constraints for one shift s"""
    constraints = []
    for i in range(n):
        if 0 <= i + s < n:
            # y[i, s] <= h[i]
            constraints.append(y[i, s] <= h[i])
            # y[i, s] <= h[i+s]
            constraints.append(y[i, s] <= h[i + s])
            # y[i, s] >= h[i] + h[i+s] - 1
            constraints.append(y[i, s] >= h[i] + h[i + s] - 1)
            # y[i, s] >= 0
            constraints.append(y[i, s] >= 0)
    return constraints

def solve_relaxed_lp(n, target_mass):
    """Solve the LP relaxation for the given n"""
    # Create variables: h vector and y matrix for each shift s
    h = cp.Variable(n)
    y = cp.Variable((n, 2*n - 1))
    
    # Objective is to minimize t
    t = cp.Variable()
    
    # Constraints:
    constraints = []
    
    # Box constraints on h
    constraints.extend([h >= 0, h <= 1])
    
    # Mass constraint
    constraints.append(cp.sum(h) == target_mass)
    
    # Ensure t >= 0
    constraints.append(t >= 0)
    
    # McCormick envelopes for each shift and each valid index pair
    for s in range(1 - n, n):  # shifts from -n+1 to n-1
        for i in range(n):
            if 0 <= i + s < n:
                # Add the McCormick constraints
                constraints.append(y[i, s + n - 1] <= h[i])  # map s to index [0, 2*n-2]
                constraints.append(y[i, s + n - 1] <= h[i + s])
                constraints.append(y[i, s + n - 1] >= h[i] + h[i + s] - 1)
                constraints.append(y[i, s + n - 1] >= 0)
    
    # The constraint sum_y >= (target_mass - t) for each s
    for s in range(1 - n, n):
        shift_sum = 0
        for i in range(n):
            if 0 <= i + s < n:
                shift_sum += y[i, s + n - 1]
        constraints.append(shift_sum >= target_mass - t)
    
    # Solve the problem
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    prob.solve(solver=cp.ECOS, verbose=False)
    
    return h.value, t.value

def threshold_round(h_vals, n, target_mass):
    """Threshold rounding to get binary h"""
    # Sort indices by their values descending
    sorted_indices = np.argsort(h_vals)[::-1]
    h_binary = np.zeros(n)
    # Set the top target_mass entries to 1
    for i in range(int(target_mass)):
        h_binary[sorted_indices[i]] = 1
    return h_binary

def randomized_round(h_vals, n, target_mass, num_samples=200):
    """Randomized rounding"""
    samples = []
    prob = h_vals.copy()
    prob = np.clip(prob, 0, 1)
    prob = prob / np.sum(prob)
    for _ in range(num_samples):
        # Sample using probabilities
        indices = np.random.choice(n, size=int(target_mass), replace=False, p=prob)
        h_binary = np.zeros(n)
        h_binary[indices] = 1
        samples.append(h_binary)
    return samples

def targeted_swap_polish(h_binary, max_time=150):
    """Improve binary h by swapping indices that reduce maximum correlation"""
    start_time = time.time()
    best_h = h_binary.copy()
    best_C5 = evaluate_h(best_h)
    
    while time.time() - start_time < max_time:
        corr = correlate(best_h, 1 - best_h, mode="full")
        max_shift = np.argmax(corr)
        max_corr = corr[max_shift]
        
        if max_corr <= best_C5:
            # No improvement observed
            break
            
        # Determine the corresponding shift
        shift = max_shift - (len(best_h) - 1)
        best_h_updated = best_h.copy()
        found_improvement = False
        best_swap = None
        max_reduction = 0
        
        # Try to swap pairs that would reduce the max correlation
        for i in range(len(best_h)):
            # Check if i is a suitable source index (i can be part of the maximum correlation)
            if best_h[i] == 1:
                # Check if there's a valid target index j
                j = i - shift
                if 0 <= j < len(best_h):
                    if best_h[j] == 0 and best_h[i + shift] == 0:
                        # Trying this swap might help
                        # Compute what happens if we swap positions i and j
                        C5_before = max_corr
                        
                        # Compute the effect of swapping
                        # Set h[i] = 0
                        temp_h = best_h.copy()
                        temp_h[i] = 0
                        temp_h[j] = 1
                        
                        # Compute the new maximum correlation
                        new_corr = correlate(temp_h, 1 - temp_h, mode="full")
                        new_max_corr = np.max(new_corr)
                        
                        reduction = C5_before - new_max_corr
                        
                        if reduction > max_reduction:
                            max_reduction = reduction
                            best_swap = (i, j)
                            
        # Apply the swap if improving
        if best_swap and max_reduction > 0:
            i, j = best_swap
            best_h_updated[i] = 0
            best_h_updated[j] = 1
            best_h = best_h_updated
            best_C5 = evaluate_h(best_h)
            found_improvement = True
        else:
            break
            
    return best_h, best_C5
