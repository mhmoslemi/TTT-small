import numpy as np
from scipy.optimize import differential_evolution

def project_to_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex with sum = target_sum"""
    v = np.array(v)
    n = len(v)
    if np.isclose(np.sum(v), target_sum):
        return v.copy()
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    # Find threshold using water filling
    # sum of clipped values = sum_i max(0, v_i - tau)
    # k indices where v_i >= tau -> kn = sum_i 1_{v_i>=tau} -> kn*tau = sum_{i=0}^{kn-1} v_i
    # so that target_sum = sum_i v_i - sum_{i=0}^{kn-1} (v_i - tau) = sum_{i=0}^{kn-1} (v_i - v_i + tau) + (n-kn)*tau = kn*tau + (n-kn)*tau = target_sum
    # Thus, kn = [(sum_{i=0}^{kn-1} v_i) - (n-kn)*target_sum] / (target_sum - tau)
    # Let’s compute from end
    # for each k from 1 to n, compute sum_{i=0}^{k-1}, and check if this matches the projection
    cumsum = np.cumsum(sorted_v)
    # we seek minimum k such that cumsum[k-1] - (k-1)*target_sum >= target_sum
    # which gives us the threshold
    # iterate to find the correct prefix index
    # actual projection formula:
    # v[i] = max(0, min(1, v[i] - theta))
    # where theta is chosen such that sum = target_sum
    # projected[u] = max(0, min(1, v[u] - t))
    # sum = sum_i max(0, min(1, v_i - t)) = sum_{v_i > t} (v_i - t) + sum_{v_i <= t} 0
    # Let’s say first k elements are such that v_i > t, then sum = sum_{i=0}^{k-1} (v_i - t)
    # = cumsum[k-1] - k*t  = target_sum  =>  t = (cumsum[k-1] - target_sum)/k
    # so we compute for k from 1 to n
    # but only take valid k where cumsum[k-1] >= target_sum + (k-1)*target_sum = K*target_sum
    # If we did not find a match, it means target_sum = sum, then we just cap at 1
    k_opt = 0
    temp_sum = float('inf')
    for k in range(1, n + 1):
        prev_sum = cumsum[k-1]
        if k > 0:
            threshold = (prev_sum - target_sum) / k
        else:
            threshold = 0
        # compute projected sum
        projected_sum = np.sum(np.maximum(0, np.minimum(1, sorted_v - threshold)))
        # or direct calculation:
        # for i in range(k):
        #    if sorted_v[i] - threshold > 0:
        #        projected_sum += (sorted_v[i] - threshold)
        #    else:
        #        break
        # alternatively:
        # compute sum of ((v_i-threshold)+, (v_i-threshold)+...) 
        # plus sum over all other terms (all at 0 due to threshold)
        # where (v_i - threshold)+ = max(0, v_i - threshold)
        # find largest k such that (sum 0 to k-1 v_i - k*th) = target_sum, so (sum of k elements) - k*th = target_sum
        # i.e. sum_{i=0}^{k-1} v_i - k*th = target_sum  =>  th = (sum - target_sum)/k
        # So we find such k that sum_{i=0}^{k-1} v_i >= target_sum
        # then returns the th
        if projected_sum <= target_sum:
            # continue
            pass
        else:
            # check if this k (then cut should be chosen)
            # correct the error: project sum is > target_sum
            # we want the largest one where it is <= target_sum
            pass
        # Let us rather compute valid vals in a less clever order
        if prev_sum - k * threshold < target_sum:
            continue
        if prev_sum - k * threshold > target_sum:
            break
        # this is valid, let's check that projected sum is at target
        # but to also ensure it's valid, need full computation:
        # points where v_i > threshold, so threshold = (cumsum[k-1] - target_sum)/k
        # projected = [v_i - threshold if v_i > threshold else 0 for i in range(n)] 
        # For vectorized:
        projected = np.maximum(0, sorted_v - threshold)
        final_sum = np.sum(projected)
        # to be exact with integral, also cap to 1
        projected = np.minimum(1, projected) # still same sum
        final_sum = np.sum(projected)
        if abs(final_sum - target_sum) < 1e-10:
            k_opt = k
            break
    # Now procedure still incomplete
    # When there is exact match, we compute the proper threshold
    if k_opt == 0:
        # short-circuit one big final loop
        # look for possible k satisfying condition
        # most likely it's the last one with cumsum[n-1] >= target_sum 
        for k in range(n, 0, -1):
            if k > 0:
                # from the proposal:
                # pick k from big to small
                # the threshold is (cumsum[k-1] - target_sum) / k
                if k == 1:
                    threshold = cumsum[k-1] - target_sum
                else:
                    threshold = (cumsum[k-1] - target_sum) / k
                projected = np.maximum(0, sorted_v - threshold)
                projected = np.minimum(1, projected)
                final_sum = np.sum(projected)
                if abs(final_sum - target_sum) < 1e-10:
                    k_opt = k
                    break
        # fall back to full version
        # check cumsum and set the appropriate threshold
        if k_opt == 0:
            # This is already indicative of failure:
            # What we want is a valid minimal k: so perhaps we start from the smallest to get a nearest good approx
            for index in range(n):
                k = index + 1
                if k == 1:
                    threshold = cumsum[index] - target_sum
                else:
                    threshold = (cumsum[index] - target_sum) / k
                projected = np.maximum(0, sorted_v - threshold)
                projected = np.minimum(1, projected)
                final_sum = np.sum(projected)
                if final_sum >= target_sum:
                    k_opt = k
                    break
    
    if k_opt == 0: # fallback
        # what we do when we can't find it correctly
        threshold = (np.sum(sorted_v) - target_sum) / n
        result = np.maximum(0, sorted_v - threshold)
        # overly simple correction to reach target but not clip too much
        result = np.minimum(1, result)
        supp = np.sum(result)
        if abs(supp - target_sum) > 1e-15:
            diff = target_sum - supp
            indices = np.argsort(result)
            for i in reversed(range(n)):
                if diff > 0 and result[indices[i]] < 1:
                    delta = min(diff, 1 - result[indices[i]])
                    result[indices[i]] += delta
                    diff -= delta
                if abs(diff) < 1e-15:
                    break
        return result

    # k_opt is known. Recompute threshold that sets sum correct
    threshold = (cumsum[k_opt - 1] - target_sum) / k_opt
    # use full risky version:
    projected = np.maximum(0, sorted_v - threshold)
    projected = np.minimum(1, projected)
    proj_sum = np.sum(projected)
    # ensure sum is exact
    if abs(proj_sum - target_sum) > 1e-8:
        # try perturbation
        indices = np.argsort(projected)[::-1]
        cur_sum = np.sum(projected)
        diff = target_sum - cur_sum
        for i in range(n):
            if diff == 0:
                break
            idx = indices[i]
            val = projected[idx]
            if val < 1 and diff > 0:
                peak = 1 - val
                incr = min(peak, diff)
                projected[idx] += incr
                diff -= incr
            elif val > 0 and diff < 0:
                decr = min(val, -diff)
                projected[idx] -= decr
                diff += decr
    # un-sort to apply back to original order
    back_indices = np.argsort(v)[::-1]
    result = np.empty_like(projected)
    result[back_indices] = projected
    return result

def compute_autocorrelation(h, n_shifts):
    # Use fast correlation (RFFT)
    H = np.fft.rfft(h)
    K = np.fft.rfft(1.0 - h)
    corr = np.fft.irfft(H * np.conj(K), n=2*len(h)-1)
    return corr[1:len(h)+1]

def compute_Obj(h, w, a, M, N=200):
    # M: number of blocks
    # N: length of the final grid
    # Compute the objective as max_k [1 - ∫ h(x) h(x+k) dx]
    # using the piecewise constant block function with params
    # h_i = a_i for point belongs to interval [s_{i-1}, s_i)  
    # interval weights w_i must sum to 2  
    # mass = sum a_i * w_i  = 1.0 expected
    
    # create intervals
    s = np.concatenate([[0], np.cumsum(w)])
    # Let's compute correlation over a more fine grid for the constraint
    # 2 indices to restrict the shift happens in the window [0,2]
    if N > len(w)*10:
        steps = N // len(w) + 1
    else:
        steps = 1
    N_corr = (N+1) * steps # they then downsample too
    dx = 2.0 / (N_corr - 1)
    x = np.linspace(0, 2, N_corr)
    # compute H on grid by interpolation:
    # h(x) = a_i for x in [s_{i-1}, s_i)
    h_on_grid = np.zeros(N_corr)
    for i in range(M):
        start = int(np.round(s[i]/dx))
        end = int(np.round(s[i+1]/dx)) + 1
        h_on_grid[start:end] = a[i]
    
    h_on_grid = h_on_grid[:N_corr]  # Ensure matching length
    # compute precise G(k) = ∫ h(x) h(x+k) dx 
    # On the full grid, we don't use very dependent FFT directly on h_on_grid, because we will measure only the integral inside [0,2] and easy to get shift overlap issues.
    # Let's instead pretend it's periodic w.r.t [0,2], then listing the overlaps manually
    # Compute overlaps assuming it's periodic and generic shift k 
    # We use inner loop O(M^2) procedure  
    # For each shift k we compute dx * sum_{i,j} a_i * a_j * overlap of (I_i, I_j + k)
    # But some parts of I_i may extend beyond 2, or I_j+k may extend beyond 2
    # So when shift by k, interval I_j becomes [s_{j-1} + k, s_j + k]
    # We take overlap in [0,2], so:
    # overlap = max(0, min(s_i, min(s_j+k, 2)) - max(s_{i-1}, max(s_{j-1}+k,0)))
    # But for integer k samples, we can do it more efficiently.
    max_shift = 200
    shifts = np.linspace(0, 2, max_shift)  
    n_shifts = len(shifts)
    g_vals = []
    # for each shift, compute exact integral over [0,2] of h(x) * h(x+k)
    for shift in shifts:
        overlap_sum = 0.0
        for i in range(M):
            s_i = s[i]
            s_i1 = s[i+1]
            a_i = a[i]
            for j in range(M):
                s_j = s[j]
                s_j1 = s[j+1]
                a_j = a[j]
                # overlap under absolute shift
                start_overlap = max(s_i, s_j + shift)
                end_overlap = min(s_i1, s_j1 + shift)
                # mod 2 shift on periodic, due to lose in info, drop it for now.
                # we will wrap using reflection or padding type method: periodic boundary + trunc.
                if start_overlap < end_overlap:
                    overlap_length = end_overlap - start_overlap
                    overlap_sum += a_i * a_j * overlap_length
                # Periodic 재끼는게 좋을수도 있지만, 그것이 결과 관련해 특별히 중요하지는 않을 수도 있음. 우선 고정 boundary.
                # one could simply handle the case w. wrapping angle if total interval wraps. But we'll use simple.
                # Better yet, when we define interval on [0,2] with s_i, the boundaries must be in [0,2].
                # But not sample on the grid more than the domain, though.
                # Actually, shift by k of [0,2] is [k,2+k]. But we take [0,2] intersection.
                # shift k: interval (s_{j-1}, s_j) -> (s_{j-1}+k, s_j+k) = (start_k = s_{j-1}+k, s_j+k)  
                #     -> overlap = [max(s_i, start_k), min(s_i1, s_j+k)]  
                #     = [max(s_i, s_{j-1} + k), min(s_i1, s_j + k)] 
        g_vals.append(overlap_sum)
    
    # Now invert it to get f(k) = 1 - G(k)
    g_vals = np.array(g_vals)
    f_vals = 1.0 - g_vals
    max_f = np.max(f_vals)
    
    # Return over N shift grid
    # actually let's extract each shift
    # we used linspace of shift from 0 to 2 with 200 steps 
    # so we sample on that or a different refinement
    # lets go even simpler and just compute Gk over reasonable #slices for step 0 to 2
    # with N = 200 fix 
    # Reflect  
    # Each shift k is at step = shift / (2/N)  
    shift_grid = np.arange(N) * 2.0/N
    # Or we can just use numbers manually. Our plotting shows 200 is safest if shift is not too subtle.
    # Use FFT-based computation on the vectors itself for fast conv, but we properly sample H(x) k scratch, or use cosine FFT
    # You think it's very accelerable 
    # Better: recompute acorrelation  
    # Following routine in precomputed is fine.
    corr_full = compute_autocorrelation(h_on_grid, max_shift)
    # No shft is MODULO 2 anyway, so we adjust it to be in [-2,2] -> then take valid part
    # it says 1<index<2N-1  
    # Get meaningful 0 to 2 (or some lag)  
    # shift index lags 0..N-1 the G(x) result with lags lags of the autcorrelation.
    #_corr = signal.correlate(h_on_grid, h_on_grid, mode='full')  
    # full gives length 2*N-1. For all shifts 0 -> N-shift - 1, it's in the middle
    # shift bins (N and next) gives D(k=1) @ lag = 1, etc.
    # permute to start at zero lag
    # lag 0 is when same vector is perfectly positioned, i.e. 0 by definition
    # this vector lookup is fine since it's much cheaper
    corr_vals = corr_full[N-1:]  # this should correctly provide shift = 0, going to k= length-1
    # make it mirror left as well to include negative shift ?
    # shift 0, -> 0-index in reverse 
    # We want lag in ] -2 , 2 ]  
    # But more precise: the first value is zero shift from full
    # yes just take k = [0... N-1] (the -N+1 to N-1 irrelevant zeros were not included
    corr_vals = corr_vals[:N]
    g_vals2 = corr_vals  # this is not yet for h(x)*(1-h(x))
    f_vals2 = 1 - corr_vals  
    return np.max(f_vals2)

def compute_objective(a, w, M, lambda_penalty=1000.0, n_grid=200):
    '''Return the full objective including the penalty'''
    mass = np.sum(np.array(a) * np.array(w))
    penalty = lambda_penalty * abs(mass - 1.0)
    if abs(mass - 1) > 1e-6:
        # Let's handle it directly by adjusting a_i
        a = np.array(a) / mass
    # Recompute integration using exact function
    # On the unit interval, let's reproduce blocked values as a full function h(x)
    # with piecewise constant [0,2] with 2 * w_i sum to 2?
    dx = 2. / n_grid
    # Are these h_i defined on interval or values?
    # Our method is to generate the interpolatory h vector for correlation
    # Generate a function:
    h_fine = np.zeros(n_grid)
    s = np.concatenate([[0], np.cumsum(w)])
    # Assumes s_i < 2    (or last value s_M = 2)
    # But w summing already ensures it  
    for i in range(M):
        start_idx = int(np.round(s[i] / dx))
        end_idx = int(np.round(s[i+1] / dx))
        if end_idx > n_grid:
            end_idx = n_grid - 1
        if i < len(a):
            h_fine[start_idx:end_idx+1] = a[i]
        else:
            # safety: maybe M was larger than len(a) produced 
            pass
            
    # now we compute exact autocorrelation of this fine discretization and extract one-parameter max
    corr = signal.correlate(h_fine, 1.0 - h_fine, mode='full')
    # take middle part with length N
    l = len(corr)
    mid = l // 2
    corr_start = mid - len(h_fine) + 1  # shift from -N to +N, reduce to just valid shift 0 to N-1
    corr_shifted = corr[corr_start:]
    # or find a slice where all shifts are [0,...,N-1]
    corr_valid = corr_shifted[:len(h_fine)]
    # (since correlation on input is c_0, c_1, ..., c_N-1, c_N, c_{N-1}, ..., c_1)  
    # corr_valid = corr[mid - len(h_fine) + 1: mid + len(h_fine) - 1]  # should cover 0 to 2N - 2 but we want short version  
    f_vals = 1.0 - corr_valid
    objective_value = np.max(f_vals) + penalty 
    return objective_value

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # initiate predefined
    M = 12
    # variables: u_1...u_M (weights non-neg), v_1...v_M (heights)
    bounds = [(-5, 5)] * (2 * M)
    
    def objective(vars):
        u = vars[:M]
        v = vars[M:]
        w = 2.0 * np.exp(np.array(u)) / np.sum(np.exp(np.array(u)))  # softmax
        a = 1.0 / (1.0 + np.exp(-np.array(v)))  # sigmoid
        return compute_objective(a.tolist(), w.tolist(), M)
    
    # Run DE
    result = differential_evolution(
        objective, 
        bounds, 
        seed=seed, 
        popsize=15 * 2 * M,
        maxiter=int(200 *(budget_s/600)), 
        polish=False
    )
    
    # Post-process DE result
    u_best = result.x[:M]
    v_best = result.x[M:]
    w_best = 2.0 * np.exp(u_best) / np.sum(np.exp(u_best))  # Softmax of u
    a_best = 1.0 / (1.0 + np.exp(-v_best))  # Sigmoid of v
    if abs(np.sum(np.array(a_best) * np.array(w_best)) - 1.0) > 1e-4:
        # Rescale a to sum to 1
        mass = np.sum(np.array(a_best) * np.array(w_best))
        a_best = np.array(a_best) / mass
        # There may be overflow into > 1, clip to 1 then re-normalize soft
        if np.any(a_best > 1):
            old_sum = np.sum(a_best)
            a_best[np.where(a_best > 1)] = 1.0
            new_sum = np.sum(a_best)
            a_best *= old_sum / new_sum
    
    a_best = np.clip(a_best, 0, 1)
    # Sample it onto a fine grid (n_points)
    n_points = 200
    dx = 2.0 / n_points
    h_sampled = np.zeros(n_points)
    s = np.concatenate([[0], np.cumsum(w_best)])
    
    for i in range(M):
        start_idx = max(0, int(np.round(s[i] / dx)))
        end_idx = min(n_points, int(np.round(s[i+1] / dx)))
        h_sampled[start_idx:end_idx+1] = a_best[i]
    
    # Refine with Projected Subgradient
    h = h_sampled.copy()
    h = project_to_capped_simplex(h, n_points / 2)
    max_iter = int(10000 * (budget_s/600))  
    alpha0 = 0.2
    best_f = np.inf
    best_h = h.copy()
    
    corr_full = compute_autocorrelation(h, n_points)
    left_end = n_points - 1
    corr = corr_full[left_end:] 
    corr = corr[:len(h)]
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute shifted correlation to get overlap of shifts
        # F(k) = sum h_i {1 - h_{i+k}}, k=0...N-1
        # Uses fft or another formula? 
        # Now compute improvements with gradient computed w.r.t the worst lag
        
        # Let us use FFT based to get all correlate, then find max
        corr_full = compute_autocorrelation(h, len(h))
        # just use the right part 
        shift = len(h) - 1
        corr = corr_full[shift:]
        corr = corr[:len(h)]  # this is consistent with convention of right shifting over -i to +i
        
        # argmax over possible shift, get angle for worst
        f_values = 1 - corr  # f(k) = 1 - G(k)
        max_k = np.argmax(f_values)
        max_f_current = f_values[max_k]
        if max_f_current < best_f:
            best_f = max_f_current
            best_h = h.copy()
            
        # Sub-gradient for shift max_k
        # loss is sum over i : h_i * (1 - h_{i+k})
        # gradient w.r.t. h_i (when i+k is in valid range) is (1 - h_{i+k})  
        # when i-k is valid: gradient w.r.t. h_i is (subtractive term) - h_{i-k}
        # This means: for an index i, look at the G_i(k) = h_i * (1 - h_{i+k}) within list
        # and this contributes to gradient as  
        # grad[i] = 1 - h[i+k]  (if i+k < length)  
        #         = - h[i-k]     (if i-k is valid)
        # now build residual precisely 
        # Overlap function is effectively:
        # update with gradient to reduce f_max
        # we know what shift max_k was:
        # This is direct functionalization without vectorization (but is clear)
        
        # approximate effect of shifting all elements on index 0, in one shift operation
        gradient = np.zeros_like(h)  
        
        # Evaluation only
            
        for i in range(len(h)):
            if i + max_k < len(h):
                gradient[i] += 1 - h[i + max_k]  
            if i - max_k >= 0:
                gradient[i] -= h[i - max_k] 
        
        # Step
        alpha = alpha0 / np.sqrt(t + 1)
        h = h - alpha * gradient
            
        # Project 
        h = project_to_capped_simplex(h, n_points / 2)
        
    # Final validation step
    possible_c = np.correlate(best_h, 1 - best_h, mode='full')
    best_corr = possible_c[len(best_h) - 1:]
    best_f = 1 - best_corr
    final_c5 = np.max(best_f) * (2.0 / n_points)
    
    return (best_h.tolist(), final_c5, n_points)
