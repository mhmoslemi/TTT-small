import numpy as np
from scipy.optimize import differential_evolution
from scipy import signal
import time


def project_to_capped_simplex(v, target_sum):
    """
    Euclidean projection onto the set {x: 0 <= x_i <= 1, sum x_i = target_sum}
    """
    v = np.asarray(v)
    n = len(v)
    if n == 0:
        return np.array([])
    # Sort in descending order
    sorted_idx = np.argsort(-v)
    sorted_v = v[sorted_idx]
    # Water-filling approach
    cumsum = np.cumsum(sorted_v)
    # Find threshold
    tau = (cumsum - target_sum) / np.arange(1, n+1)
    # Choose largest tau such that sum of projections remains >= target_sum
    # Since cumulative sum is decreasing, we want the largest index where
    # projected_v = max(0, sorted_v - tau) have sum >= target_sum
    # A bit tricky - compute index of last element with positive projection
    for i in reversed(range(n)):
        projected = np.maximum(0, sorted_v - tau[i])
        if np.sum(projected) >= target_sum:
            break
    # Use actual threshold
    threshold = tau[i]
    projected = np.maximum(0, v - threshold)
    # Adjust to exact sum
    current_sum = np.sum(projected)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Add correction to largest elements
        sorted_proj_idx = np.argsort(-projected)
        for idx in sorted_proj_idx:
            if diff <= 1e-10:
                break
            add = min(diff, 1.0 - projected[idx])
            projected[idx] += add
            diff -= add
    # Reorder back
    result = np.empty_like(projected)
    result[sorted_idx] = projected
    return result


def compute_autocorrelation_blocks(a, w, N_shifts=200):
    """
    Compute G(k) = ∫ h(x) h(x+k) dx
    where h is piecewise constant with blocks of heights a_i and widths w_i.
    Return shape (N_shifts,)
    """
    M = len(a)
    if M < 1:
        return np.zeros(N_shifts)
    
    # Precompute interval boundaries
    s = np.concatenate([[0], np.cumsum(w)])
    # Precompute interval overlaps
    # For all pairs of intervals and all shifts, compute G(k)
    # Because we have fixed shifts and not really k in [0,2), we expect
    # all shifts in range [0,2] with resolution 2/N_shifts.
    # Let's compute for discrete shifts from 0 to 2 in steps of 2/N_shifts.
    # Alternatively, interpret this as computing the same as for shift 0, 1, 2, ..., N_shifts-1
    # But we want this for shift k in [0,2]. So work with more shifted versions.

    # We do it properly: for any shift k in continuous [0,2], we compute overlap
    # We use a coarse grid for computing it, but still evaluate the full discrete
    # The suggestion is to work with discrte grid of shifts
    shifts = np.linspace(0, 2, N_shifts, endpoint=False)
    G_vals = np.zeros(N_shifts)

    # For each shift, we compute ∫ h(x) h(x+k) dx
    # We use the formulation based on overlaps of intervals.
    # For all block pairs (i,j) and shift k, the overlap contributes a_i * a_j * overlap(I_i, I_j+k)
    # I_i = [s[i], s[i+1])
    # I_j + k = [s[j] + k, s[j+1] + k)
    # overlap = max(0, min(s[i+1], s[j+1]+k) - max(s[i], s[j]+k))

    for idx, k in enumerate(shifts):
        total = 0.0
        for i in range(M):
            for j in range(M):
                # Use periods
                s_i = s[i]
                s_im1 = s[i-1] if i > 0 else 0.0
                s_j = s[j]
                s_jm1 = s[j-1] if j > 0 else 0.0
                overlap = max(0, min(s_i, s_j + k) - max(s_im1, s_jm1 + k))
                total += a[i] * a[j] * overlap
        G_vals[idx] = total
    return G_vals


def compute_objective_blocks(a, w, penalty_weight=1000.0, N_shifts=200):
    """
    Compute objective F = max_k (1 - G(k)) + penalty
    """
    G = compute_autocorrelation_blocks(a, w, N_shifts)
    f = 1.0 - G  # f(k) = 1 - G(k)
    obj = np.max(f)
    integral = np.sum(a * w)
    pen = penalty_weight * abs(integral - 1.0)
    return obj + pen


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    M = 12  # Number of blocks
    penalty_weight = 1000.0
    N = 200  # Final grid points
    dx = 2.0 / N
    
    # DE parameters
    bounds = [(-5, 5)] * (2 * M)  # u_i and v_i
    popsize = 15 * (2 * M)
    max_generations = 30  # Will adjust in loop based on time
    
    # Global search using DE
    remaining_time = budget_s * 0.18
    result_de = differential_evolution(
        func=lambda z: compute_objective_blocks(
            a=np.array([1.0 / (1.0 + np.exp(-z[M + i])) for i in range(M)]),
            w=np.array([2.0 * np.exp(z[i]) / np.sum([np.exp(z[j]) for j in range(M)]) for i in range(M)]),
            penalty_weight=penalty_weight,
            N_shifts=200
        ),
        bounds=bounds,
        seed=seed,
        sampling_method="LHS",
        mutation=(0.5, 1.0),
        recombination=0.7,
        maxiter=int(max_generations),  # Limited by number of generations
        polish=False,
        disp=False
    )

    # Extract best solution from DE
    z = result_de.x
    a = np.array([1.0 / (1.0 + np.exp(-z[M + i])) for i in range(M)])
    w = np.array([2.0 * np.exp(z[i]) / np.sum([np.exp(z[j]) for j in range(M)]) for i in range(M)])

    # Renormalize heights to match integral constraint
    mass = np.sum(a * w)
    if abs(mass - 1.0) > 1e-7:
        a = a / mass
    # Clip to [0,1]
    a = np.clip(a, 0, 1)
    # Re-apply constraint: if ajusted heights exceed 1, redistribute
    # Simple: just do re-projection, normal way.
    a = project_to_capped_simplex(a, 1.0)
    
    # Sample on uniform grid
    s = np.concatenate([[0], np.cumsum(w)])
    # Create h on uniform x grid
    h = np.zeros(N)
    x_grid = np.linspace(0, 2, N, endpoint=False)
    h_idx = 0
    for i in range(N):
        x = x_grid[i]
        # Find which interval x belongs to
        # Find smallest k such that x < s[k+1]
        # so hsegment = s[k], s[k+1]
        for k in range(len(s)-1):
            if s[k] <= x < s[k+1]:
                h[i] = a[k]
                break
        else:
            # Handle edge case or upper-bound
            if s[-1] <= x:
                h[i] = a[-1]
    # Final check
    np.testing.assert_allclose(np.sum(h) * dx, 1.0, rtol=1e-5)

    # Local polishing using projected subgradient descent
    max_iterations = int(6000)  # Simulated budget
    current_time = time.time()
    elapsed_s = current_time - start_time
    remaining_time = budget_s - elapsed_s
    iter_time_per = remaining_time / max_iterations * 0.7  # Estimated time per iter
    step_size = 0.2
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Add a small regularizer to avoid numerical issue
    reg_term = 1e-10
    
    for t in range(max_iterations):
        current_time = time.time()
        if current_time - start_time > budget_s - 5:
            break
            
        # Compute correlation
        # Use FFT to compute np.correlate of h and 1-h (the pattern)
        # Result: correlation = full correlation of length 2*N -1
        # We only care about shifts from 0 to N-1 (we'll need full len to get it all)
        # The idea is that correlation of h and (1-h) = C_k via index k (positive lags)
        # Take real part and normalize
        # h use [0, N-1] range for indexing
        # 1-h acts as kernel
        # Re-structuring:
        # h is vector [h[0], ..., h[N-1]]  (N elements)
        # output will be vec corr of len 2*N -1
        # corr[k] = sum h[i] * (1 - h[i+k]) for i = 0..(N-1-k)
        # So we are extending the sequence to wrap
        # Even more clever: use full convolution: h conv (1-h) as FFT of prod (H and 1-H*)
        H = np.fft.rfft(h)
        # Allops distance
        K = np.fft.rfft(1.0 - h)
        # Full correlation (length 2*N -1)
        corr_full = np.fft.irfft(H * np.conj(K), n=2*N - 1)
        # Take center part... The central k range covers both positive and negative shifts.
        # But to get real shifts like in standard situation:
        # shifts = 0,1,...,N-1
        # Here we assume that at correct=N values (positive lags), we compute such that
        # these are the coefficients of the overlap function we care about.
        # For efficiency and clean code, just take first N values (since "positive shifts" are the first).
        # corr values now len 2N-1
        # For inspection:
        # corr[0], corr[1], ..., corr[N-1] correspond to lags 0,1,...,N-1 (each i=0 to N-1-k overk)
        # Just reindex as: 
        correlation = corr_full[:N]  # Will verify that 0 to N-1 is what we're looking for
        
        # Overlap values (1 - correlation)
        overlap_vals = 1.0 - correlation
        max_overlap = np.max(overlap_vals)
        # And also resort to how many time is registered.
        # Final evaluation of c5: due to 1, we can take maximum of 1 - G(k) = max_k f(k)
        # Leave the max without dx, apply final dx separately
        
        # Compute subgradient (for max_k f(k))
        # Key insight: at k* such that f(k*) = max_k f(k), we compute the gradient of each f_k
        # The subgradient of max_k f_k is the average over all such k*...
        # We handle non-differentiability, but let’s suppose we select all k* which give F = maximal value.
        # Let K_active = set of k such that f(k) = max f(k)
        
        max_f = np.max(overlap_vals)
        K_active = np.nonzero(np.abs(overlap_vals - max_f) < 1e-10)[0]
        num_active = len(K_active)
        if num_active < 1:
            K_active = [0]  # fallback
            num_active = 1

        # Build gradient g (one for each element h[i])
        g = np.zeros_like(h)
        for k in K_active:
            # This is for a fixed 0 <= k <= N-1
            # Let us get gradient of f_k = 1 - (h conv 1-h)_k
            # Or, let's get the derivative of h conv (1-h) with respect to h
            # That is just (1-h) conv on left and h conv on right
            # But simplified: let's compute d/dh_i of C_k = Σ_i h[i] * (1-h[i+k])
            # Since h[i+k] from 0 to N-1-k, i starts from 0 to N-k-1
            # We kind of ask d/dh_i of Σ_i h[i] * (1-h[i+k])
            # So: 1 - h[i+k] for i <= N-1-k and -h[i-k] for i >= k
            # Totalized in h conv 1-h _k, the deriv is matrix inv. to known result:
            
            # or compute manually:
            # derivative of
            # C_k = \sum_{i=0}^{N-1-k} h_i (1-h_{i+k})
            # wrt h_j:
            # = \sum_{i=0}^{N-1-k} 1_{i=j}(1-h_{j+k}) + h_i(-1_{i+k=j}) \delta_{i+j}
            # = 1_{j+k < N}(1-h_{j+k}) - 1_{j >= k} h_{j-k}
            
            if k < 0 or k >= N:
                continue
            # Compute gradient vector g, such that g[i] = \partial C_k / \partial h_i
            # Use dense form:
            for i in range(N):
                if i <= N - 1 - k:
                    g[i] -= (1.0 - h[i + k])
                if i >= k:
                    g[i] -= h[i - k]
        
        # Normalize
        g = g / num_active
        # Update step with diminishing step-size
        alpha = step_size / np.sqrt(t + 1)
        h_next = h - alpha * g
        # Project onto feasible set [0,1], sum=1
        h_next = np.clip(h_next, 0, 1)
        current_sum = np.sum(h_next)
        # Adjust sum
        h_next = h_next * (1.0 / current_sum)  # Only throw in overshoot, no adjustment for clipped-under

        # Alternatively: better projection (we had slight drift due to inaccuracy, correct err)
        # Rescale to target = 1.0
        # 1) let v = h_next; 2) use capped simplex as before, 
        # but also enforce unit integral.
        # Simpler: just scale positively
        scale_factor = 1.0 / np.sum(h_next)
        h_new = h_next * scale_factor
        h_new = project_to_capped_simplex(h_new, 1.0)  # enforce sum exactly to 1.0
        # Now check again
        h = h_new

        # Evaluate C5 bound
        H = np.fft.rfft(h)
        K = np.fft.rfft(1.0 - h)
        corr_full = np.fft.irfft(H * np.conj(K), n=2*N - 1)
        correlation = corr_full[:N]  # shift 0 to N-1, expression is already correct
        overlap_vals = 1.0 - correlation
        c5_bound = np.max(overlap_vals)

        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
    
    # Final cleaning 
    # Use np.correlate for precise evaluation
    H = np.fft.rfft(best_h)
    K = np.fft.rfft(1.0 - best_h)
    corr_full = np.fft.irfft(H * np.conj(K), n=2*N - 1)
    correlation = corr_full[:N]
    overlap_vals = 1.0 - correlation
    c5_bound_precise = np.max(overlap_vals) * dx  # Correct final scaling

    # Return
    return (best_h.tolist(), c5_bound_precise, N)
