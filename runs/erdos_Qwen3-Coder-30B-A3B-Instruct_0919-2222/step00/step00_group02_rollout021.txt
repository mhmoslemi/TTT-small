import numpy as np
import time
import random

def evaluate_h(h):
    """Evaluate the maximum overlap integral for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode="full")
    max_O_counts = O_counts.max()
    return max_O_counts * dx

def max_overlap_counts(h):
    """Return max overlap count (integer) for given h."""
    O_counts = np.correlate(h, 1 - h, mode="full")
    return O_counts.max()

def random_balanced_binary(n, rng):
    """Generate a random balanced binary array with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    ones = rng.sample(range(n), n // 2)
    h[ones] = 1
    return h

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto box [0,1] and sum constraint."""
    v = v.copy()
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def binary_anneal(n, time_limit, rng):
    """Simulated annealing for binary h."""
    h = random_balanced_binary(n, rng)
    cur_val = max_overlap_counts(h)
    best_val = cur_val
    best_h = h.copy()
    start_time = time.time()
    
    while time.time() - start_time < time_limit:
        T = max(1e-6, 0.5 * (1 - (time.time() - start_time) / time_limit) + 1e-4)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        h[p], h[q] = 0, 1
        new_val = max_overlap_counts(h)
        delta = new_val - cur_val
        
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            h[p], h[q] = 1, 0
    
    return best_h, best_val
