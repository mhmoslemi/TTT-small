import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def solve_erdos_milp(n, time_limit):
    prob = pulp.LpProblem("ErdosMinOverlapMILP", pulp.LpMinimize)
    
    # Variables
    h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
    T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
    
    # Constraint: sum h_i = n/2
    prob += pulp.lpSum([h[i] for i in range(n)]) == n/2

    # Product constraints and shift constraints
    for d in range(1, n):  # skip d=0 to reduce variables
        y_vars = []
        for i in range(n-d):
            y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
            y_vars.append(y)
            # McCormick envelope constraints
            prob += y <= h[i]
            prob += y <= h[i+d]
            prob += y >= h[i] + h[i+d] - 1
        # Shift constraint
        prob += pulp.lpSum(y_vars) >= n/2 - T
    
    # Objective
    prob += T
    
    # Solve
    solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit)
    prob.solve(solver)
    
    status = pulp.LpStatus[prob.status]
    if status != 'Optimal' and status != 'Not Solved':
        return None, None
    
    # Get solution
    h_vals = np.array([pulp.value(h[i]) for i in range(n)], dtype=float)
    t_val = pulp.value(T)
    
    # Compute actual C5 bound
    c5_bound = evaluate_erdos_solution(h_vals)
    
    return h_vals, c5_bound

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Try different resolutions
    ns = [200, 180, 160, 140, 120, 100, 80, 60]
    
    # Start with the initial construction
    try:
        initial_h = np.array(initial_h_values)
        initial_c5 = evaluate_erdos_solution(initial_h)
        if initial_c5 < best_c5:
            best_h = initial_h
            best_c5 = initial_c5
            best_n = len(initial_h)
    except:
        pass

    for n in ns:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 5:
            break
            
        # Try solving the MILP for current n
        try:
            h_sol, c5_sol = solve_erdos_milp(n, min(300, remaining_time))
            if h_sol is not None and c5_sol is not None and c5_sol < best_c5:
                best_h = h_sol
                best_c5 = c5_sol
                best_n = n
        except:
            continue
            
        # If we found a good solution early, check if we have enough time to try others
        if time.time() - start_time > budget_s * 0.8:
            break
            
    if best_h is None:
        # Fallback to initial solution if nothing better found
        try:
            initial_h = np.array(initial_h_values)
            initial_c5 = evaluate_erdos_solution(initial_h)
            best_h = initial_h
            best_c5 = initial_c5
            best_n = len(initial_h)
        except:
            # Should not happen
            best_h = np.ones(70) * 0.5
            best_c5 = 0.38092
            best_n = 70
    
    return (best_h, best_c5, best_n)
