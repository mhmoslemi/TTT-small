from scipy.optimize import differential_evolution
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the overlap integral
    c5 = np.max(corr) * dx
    return c5

def objective(h):
    """Objective function to minimize."""
    return evaluate_h(h)
