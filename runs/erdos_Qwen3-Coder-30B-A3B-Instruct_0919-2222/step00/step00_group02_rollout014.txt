def evaluate_h(h_values):

    n_points = len(h)

    # Compute all correlations for different k values
    corr = np.correlate(h, 1 - h, mode="full")
    # The first (n_points-1) elements are the k values we want
    # corresponding to shifts from -n_points+1 to n_points-1
    # But we only care about k from 0 to 2
    # Since h is defined on [0,2], and we're looking at h(x) * (1 - h(x+k))
    # We are fixing k to be in [0,2], so we need the corr indices corresponding
    # to the valid k values
    # The correlation indices go from -n_points+1 to n_points-1
    # So we want corr[n_points-1:] = k=0 up to k=n_points-1 (which gives maximum k = 2)
    # But we also have the symmetric k=-k part, so for k=0 we get index n_points-1, k=1 index n_points,
    # k=2 index n_points+1, etc.
    # Actually let me approach this more carefully:
    # The actual k values we want are computed as k = (indices - (n_points-1)) * dx
    # So we want to take corr from n_points-1 to n_points-1 + number_of_valid_k_values
    # Check how many valid k values are there when k is in [0,2] and dx=2/n_points
    # max k = 2 -> max index = n_points-1 + (2/dx) = n_points-1 + n_points = 2*n_points-1
    # min k = 0 -> min index = n_points-1 + 0 = n_points-1
    # Number of valid k indices = 2*n_points - (n_points-1) = n_points+1
    # This is wrong for the general but correct for the specific case:
    # Since dx = 2/n_points, and we want k from 0 to 2, that means k values
    # in the correlation array correspond to:
    # indices = 0 to 2*n_points-2 in the correlation array
    # For k=0: index n_points-1
    # For k=2: index n_points-1 + 2/dx = n_points-1 + n_points = 2*n_points-1
    # But we can't access correlation over k=2, we need k values only.
    # Our wanted k values:
    # k = 0, 1*dx, 2*dx, ..., n_points*dx = 2
    # Notation: We take all k values between 0 and 2, in step dx.
    # So we want indices:
    # k=0: index n_points-1
    # k=dx: index n_points-1+1
    # k=2*dx: index n_points-1+2
    # k=2: index n_points-1+n_points = 2*n_points-1
    # So the values we want are corr[n_points-1:n_points-1+n_points] = corr[n_points-1:2*n_points-1]
    # let's double check:
    # indices: -n_points+1 to n_points-1 (total 2*n_points-1 elements)
    # k values: (-n_points+1)*dx to (n_points-1)*dx
    # For our k range of 0 to 2, we want only those where 0 <= k <= 2
    # If dx = 2/n_points:
    # We want k = 0, dx, 2*dx, ..., n_points*dx = 2
    # Indices = (0/dx + n_points-1) = n_points-1 to 2*n_points-1
    # So we take corr[n_points-1:2*n_points-1]
    # But since k can be >= 2 we want max_corr = max(corr[n_points-1:2*n_points-1])
    max_corr = max(corr[n_points-1:2*n_points-1])
    return max_corr * dx
