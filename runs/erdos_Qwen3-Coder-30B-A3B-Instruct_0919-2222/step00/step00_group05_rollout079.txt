import numpy as np
import time
import random
from scipy.special import logsumexp

# Precomputed shift maps: for each shift s, list of (i, i+s) pairs
def precompute_shifts(n):
    shift_maps = {}
    for s in range(1-n, n):
        pairs = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                pairs.append((i, j))
        shift_maps[s] = pairs
    return shift_maps

def correlations(h, shift_maps):
    n = len(h)
    C = np.zeros(2*n - 1)
    for s, pairs in shift_maps.items():
        C[s + n - 1] = sum(h[i] * (1 - h[j]) for i, j in pairs)
    return C

def smooth_obj_and_grad(h, alpha, shift_maps):
    C = correlations(h, shift_maps)
    log_C = C - np.max(C)  # for numerical stability
    log_sum_exp = logsumexp(alpha * log_C)
    Phi_alpha = log_sum_exp / alpha
    
    # compute softmax weights for each shift
    w = np.exp(alpha * log_C - log_sum_exp)
    
    # compute gradient: g_i = sum_s w_s (1 - h_{i+s} - h_{i-s})
    g = np.zeros_like(h)
    for s, pairs in shift_maps.items():
        for i, j in pairs:
            # for shift s, this correlation contributes to grad of h_i as (1 - h_j)
            g[i] += w[s + (len(h)-1)] * (1 - h[j])
            # add term from h_{i-s} when i-s is valid  
            if s < 0:
                k = i + (-s)  # i - s = k => i = k + s
                if 0 <= k < len(h):
                    g[i] += w[s + (len(h)-1)] * (-h[k])
            elif s > 0:
                k = i - s  
                if 0 <= k < len(h):
                    g[i] += w[s + (len(h)-1)] * (-h[k])
    
    return Phi_alpha, g

def project_to_capped_simplex(v, M):
    # v: array of size n  
    # M: target sum
    n = len(v)
    v = np.clip(v, 0, 1)
    total = np.sum(v)
    
    if abs(total - M) < 1e-10:
        return v
    
    # create sorted indices in descending order
    idx = np.argsort(v)[::-1]
    v_sorted = v[idx]
    
    # find where to threshold
    s = 0.
    for j in range(n):
        if s + v_sorted[j] <= M:
            s += v_sorted[j]
        else:
            threshold = float(M - s) / (n - j)
            break
    else:
        threshold = float(M - s) / n
    
    # apply threshold
    h = np.zeros_like(v)
    for i, j in enumerate(idx):
        h[j] = max(0, min(1, v[j] - threshold))
    return h

def run_descent(h0, alpha_schedule, eta0=0.1, max_iter_per_alpha=50, shift_maps=None, time_limit=None):
    if shift_maps is None:
        raise ValueError("shift_maps must be provided")

    h = h0.copy()
    start_time = time.time()
    eta = eta0
    best_h = h.copy()
    best_C = None
        
    for i, alpha in enumerate(alpha_schedule):
        for k in range(max_iter_per_alpha):
            if time_limit and time.time() - start_time > time_limit - 2:
                break
            Phi_alpha, g = smooth_obj_and_grad(h, alpha, shift_maps)
            v = h - eta * g
            h = project_to_capped_simplex(v, len(h) / 2)
            
            if i == len(alpha_schedule) - 1:
                # evaluate true max C_s in the last phase
                C_vals = correlations(h, shift_maps)
                current_max = np.max(C_vals)
                if best_C is None or current_max < best_C:
                    best_C = current_max
                    best_h = h.copy()
            else:
                # just track progress internally
                pass
                
        eta *= 0.9  # decaying step size
    
    return best_h, best_C

def generate_initial_seeds(initial_h, n):
    seeds = [np.array(initial_h)]
    
    # Generate 5 binary random starts
    for _ in range(5):
        indices = random.sample(range(n), n//2)
        h = np.zeros(n)
        for i in indices:
            h[i] = 1.0
        seeds.append(h)
    
    # Generate 5 fractional starts using Dirichlet + projection
    for _ in range(5):
        h = np.random.dirichlet([1.0] * n)
        h = project_to_capped_simplex(h, n/2)
        seeds.append(h)
    
    return seeds

def swap_local_search(h_bin, time_budget_secs, shift_maps):
    start_time = time.time()
    # Compute all correlations 
    C_vals = correlations(h_bin, shift_maps)
    best_C_val = np.max(C_vals)
    current_h = h_bin.copy()
    
    iter = 0
    max_iter = 10000
    while iter < max_iter and (time.time() - start_time) < time_budget_secs:
        s_max = np.argmax(C_vals)
        # Look for swap minimizing C_s_max
        current_C_s = C_vals[s_max]
        if current_C_s <= best_C_val:
            break
        
        found_better = False
        for i in range(len(current_h)):
            if current_h[i] == 1:
                j = i + s_max
                if 0 <= j < len(current_h) and current_h[j] == 0:
                    # Try swapping i and j
                    flipped = current_h.copy()
                    flipped[i] = 0
                    flipped[j] = 1
                    
                    # Evaluate how this swaps correlation at s_max
                    flipped_C_vals = correlations(flipped, shift_maps)
                    new_max = np.max(flipped_C_vals)
                    
                    if new_max < best_C_val:
                        current_h = flipped
                        best_C_val = new_max
                        C_vals = flipped_C_vals
                        found_better = True
                        break
                        
        if not found_better:
            # find another bad shift
            best_C_val = np.max(C_vals)
            s_max = np.argmax(C_vals)
            # should only try once for speed, or just break earlier
            break
        iter += 1
        
    return current_h, best_C_val
