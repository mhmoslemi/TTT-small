# Predefined initial construction
initial_h_values = None

def evaluate_h(h_values, n_points):
    """Evaluate C5 for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the convolution gives C5
    max_val = np.max(conv) * dx
    return max_val

        h_current = np.ones(n_points) * 0.5
