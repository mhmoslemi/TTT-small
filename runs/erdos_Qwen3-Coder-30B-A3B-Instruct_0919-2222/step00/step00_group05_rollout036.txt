import time
import numpy as np
import cvxpy as cp
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h_values, 1 - np.array(h_values), mode='full') * dx
    # The correlation length is 2*n - 1
    # We take the middle part (the 'valid' overlaps)
    start = n - 1
    end = 2 * n - 2
    correlations = corr[start:end]
    return np.max(correlations)

def build_mcconnick_constraints(h, y, n, max_shift):
    """Creates McCormick constraints for y[i,s] = h[i] * h[i+s]"""
    constraints = []
    for s in range(-max_shift, max_shift+1):
        for i in range(n):
            if 0 <= i+s < n:
                # y[i,s] <= h[i]
                constraints.append(y[i, s + max_shift] <= h[i])
                # y[i,s] <= h[i+s]
                constraints.append(y[i, s + max_shift] <= h[i+s])
                # y[i,s] >= h[i] + h[i+s] - 1
                constraints.append(y[i, s + max_shift] >= h[i] + h[i+s] - 1)
                # y[i,s] >= 0
                constraints.append(y[i, s + max_shift] >= 0)
    return constraints

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Use a default 70-point construction if not provided
        n_points = 70
        h0 = np.array([1.0 if i < n_points//2 else 0.0 for i in range(n_points)])
        h0 = h0 / np.sum(h0) * n_points / 2
    n = len(h0)
    dx = 2.0 / n
    mass = n / 2.0

    print(f"Starting optimization: n={n}, mass={mass}, time={time.time() - start_time:.2f}s")
    
    # Stage 0: LP Relaxation
    print("Stage 0: Solving LP relaxation")
    t = cp.Variable()
    h = cp.Variable(n)
    y = cp.Variable((n, 2*n-1)) # indices for shifts [-n+1, n-1]
    max_shift = n - 1
    constraints = [0 <= h, h <= 1, cp.sum(h) == mass, t >= 0]
    
    # McCormick envelope constraints
    for s in range(-max_shift, max_shift+1):
        for i in range(n):
            if 0 <= i+s < n:
                idx = s + max_shift
                # y[i, idx] <= h[i]
                constraints.append(y[i, idx] <= h[i])
                # y[i, idx] <= h[i+s]
                constraints.append(y[i, idx] <= h[i+s])
                # y[i, idx] >= h[i] + h[i+s] - 1
                constraints.append(y[i, idx] >= h[i] + h[i+s] - 1)
                # y[i, idx] >= 0
                constraints.append(y[i, idx] >= 0)
    
    # Each shift s has its correlation constraint
    for s in range(-max_shift, max_shift+1):
        shift_sum = 0
        for i in range(n):
            if 0 <= i+s < n:
                shift_sum += y[i, s + max_shift]
        constraints.append(shift_sum >= mass - t)
    
    obj = cp.Minimize(t)

    # Use ECOS for speed, with warm start
    prob = cp.Problem(obj, constraints)
    try:
        # Set max time for solving (percentage of total budget)
        solve_time = min(30, budget_s * 0.05)
        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-6, reltol=1e-6)
    except Exception as e:
        print(f"LP solve failed: {e}")
        pass

    h_opt = h.value if h.value is not None else h0
    t_opt = t.value if t.value is not None else float('inf')
    
    print(f"LP solved in {time.time() - start_time:.2f}s, t_opt = {t_opt}")
    
    # Determine candidates - binary or fraction with rounding heuristics
    candidates = []
    
    # Check if we already have integer solution
    h_rounded_int = np.round(h_opt)
    h_rounded_int = np.clip(h_rounded_int, 0, 1)
    if np.allclose(np.sum(h_rounded_int), mass):
        # Valid binary
        candidates.append((h_rounded_int.copy(), evaluate_erdos_solution(h_rounded_int)))
    
    # Deterministic rounding: take top mass entries as 1
    h_sorted = np.argsort(h_opt)[::-1]
    h_determ = np.zeros(n)
    h_determ[h_sorted[:int(mass)]] = 1
    candidates.append((h_determ, evaluate_erdos_solution(h_determ)))
    
    # Randomized rounding with 200 samples
    if not np.allclose(h_opt, h_rounded_int):
        num_samples = 200
        for _ in range(num_samples):
            sample = np.zeros(n)
            indices = np.random.choice(n, size=int(mass), replace=False, p=h_opt/np.sum(h_opt))
            sample[indices] = 1
            c5_val = evaluate_erdos_solution(sample)
            candidates.append((sample, c5_val))
    
    # Now apply greedy swap polishing to all candidates (if we have time)
    best_candidate = None
    best_c5 = float('inf')
    
    def polish_candidate(h_candidate):
        h_c = np.array(h_candidate)
        while True:
            # Compute all correlations
            corr = correlate(h_c, 1 - h_c, mode='full') * dx
            c_max = np.max(corr)
            argmax_c = np.argmax(corr)
            # Get the index of the max correlation
            shift_idx = argmax_c - (n - 1)
            # Find pairs to swap
            found_improvement = False
            opt_diff = 0
            best_i, best_j = -1, -1
            for i in range(n):
                # Check if i contributes to the max correlation
                if h_c[i] == 1 and h_c[(i + shift_idx) % n] == 0:
                    # Try swapping with some j that creates an improvement
                    for j in range(n):
                        if h_c[j] == 0 and h_c[(j + shift_idx) % n] == 1:
                            # Try hypothetical swap
                            h_new = h_c.copy()
                            h_new[i] = 0
                            h_new[j] = 1
                            corr_new = correlate(h_new, 1 - h_new, mode='full') * dx
                            c_max_new = np.max(corr_new)
                            # Check how much we reduce the max
                            diff = c_max - c_max_new
                            if diff > opt_diff:
                                opt_diff = diff
                                best_i, best_j = i, j
            if opt_diff > 0:
                # Perform the swap
                h_c[best_i] = 0
                h_c[best_j] = 1
                found_improvement = True
            else:
                break
        return h_c
    
    for h_cand, c5_val in candidates:
        if time.time() - start_time > 0.9 * budget_s:
            break
        h_polished = polish_candidate(h_cand)
        c5_polished = evaluate_erdos_solution(h_polished)
        if c5_polished < best_c5:
            best_c5 = c5_polished
            best_candidate = h_polished
    
    # If the current best is worse than 0.381 and fractional, try to solve MIP
    if best_c5 > 0.381 and not np.allclose(np.round(h_opt), h_opt):
        print("Trying MIP refinement")
        # Try solving the MIP with tight tolerance
        h_bin = cp.Variable(n, integer=True)
        y_bin = cp.Variable((n, 2*n-1), integer=True)
        obj_mip = cp.Minimize(t)
        constraints_mip = [0 <= h_bin, h_bin <= 1, cp.sum(h_bin) == mass, t >= 0]
        
        # Same McCormick constraints
        for s in range(-max_shift, max_shift+1):
            for i in range(n):
                if 0 <= i+s < n:
                    idx = s + max_shift
                    # y[i, idx] <= h[i]
                    constraints_mip.append(y_bin[i, idx] <= h_bin[i])
                    # y[i, idx] <= h[i+s]
                    constraints_mip.append(y_bin[i, idx] <= h_bin[i+s])
                    # y[i, idx] >= h[i] + h[i+s] - 1
                    constraints_mip.append(y_bin[i, idx] >= h_bin[i] + h_bin[i+s] - 1)
                    # y[i, idx] >= 0
                    constraints_mip.append(y_bin[i, idx] >= 0)
    
        # Each shift s has its correlation constraint
        for s in range(-max_shift, max_shift+1):
            shift_sum = 0
            for i in range(n):
                if 0 <= i+s < n:
                    shift_sum += y_bin[i, s + max_shift]
            constraints_mip.append(shift_sum >= mass - t)
        prob_mip = cp.Problem(obj_mip, constraints_mip)
        try:
            # Solve in reasonably tight time
            prob_mip.solve(solver=cp.CBC, max_time=250)
            if h_bin.value is not None:
                h_mip = np.array(h_bin.value)
                if np.allclose(np.sum(h_mip), mass):
                    h_polished_mip = polish_candidate(h_mip)
                    c5_mip = evaluate_erdos_solution(h_polished_mip)
                    if c5_mip < best_c5:
                        best_c5 = c5_mip
                        best_candidate = h_polished_mip
        except Exception as e:
            print(f"MIP failed: {e}")
    
    if best_candidate is None:
        best_candidate = h_opt
    
    final_c5 = evaluate_erdos_solution(best_candidate)
    print(f"Final C5 = {final_c5}, time = {time.time() - start_time:.2f}s")
    return list(best_candidate), final_c5, n
