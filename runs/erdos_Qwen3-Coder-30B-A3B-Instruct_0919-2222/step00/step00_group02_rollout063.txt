def evaluate_solution(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = corr.max()
    return 2 * max_overlap / n

def generate_balanced_binary(n):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1
    np.random.shuffle(h)
    return h

def compute_mismatch_sets(a, shift):
    """Find mismatched indices for a given shift."""
    n = len(a)
    M1 = []
    M0 = []
    if shift >= 0:
        for i in range(n - shift):
            if a[i] == 1 and a[i + shift] == 0:
                M1.append(i)
            elif a[i] == 0 and a[i + shift] == 1:
                M0.append(i)
        # Extension for negative shifts
        for i in range(shift, n):
            if a[i] == 1 and a[i - shift] == 0:
                M1.append(i)
            elif a[i] == 0 and a[i - shift] == 1:
                M0.append(i)
    else:
        for i in range(-shift, n):
            if a[i] == 1 and a[i + shift] == 0:
                M1.append(i)
            elif a[i] == 0 and a[i + shift] == 1:
                M0.append(i)
        for i in range(n + shift):
            if a[i] == 1 and a[i - shift] == 0:
                M1.append(i)
            elif a[i] == 0 and a[i - shift] == 1:
                M0.append(i)
    return M1, M0

def binary_local_search(h0, max_time):
    start = time.time()
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = np.correlate(h, 1 - h, mode='full').max()
    
    # Build a list of available shifts for efficient sampling
    all_shifts = list(range(-n+1, n))
    
    while time.time() - start < max_time:
        corr = np.correlate(h, 1 - h, mode='full')
        max_cnt = corr.max()
        worst_k = corr.argmax() - (n - 1)
        
        # Bad shift, reconstruct all mismatched sets
        M1, M0 = compute_mismatch_sets(h, worst_k)
        
        if len(M1) > 0 and len(M0) > 0:
            i = np.random.choice(M1)
            j = np.random.choice(M0)
        else:
            # Fallback to random balanced swap if there are no usable mismatches
            idx1 = np.where(h == 1)[0]
            idx0 = np.where(h == 0)[0]
            if len(idx1) > 0 and len(idx0) > 0:
                i = np.random.choice(idx1)
                j = np.random.choice(idx0)
            else:
                break
        # Perform swap
        h[i], h[j] = h[j], h[i]
        
        # Re-evaluate and accept if better
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = new_corr.max()
        if new_max <= max_cnt:
            max_cnt = new_max
        else:
            h[i], h[j] = h[j], h[i]  # rollback
        if max_cnt < best_val:
            best_val = max_cnt
            best_h = h.copy()
    
    return best_h, best_val

def simulated_annealing(h0, max_time, steps=5000):
    start = time.time()
    n = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    current_energy = corr.max()
    best_h = h.copy()
    best_e = current_energy
    
    temp_start = 0.5
    temp_end = 1e-4
    alpha = (temp_end / temp_start) ** (1.0 / steps)
    temp = temp_start
    
    t = 0
    while t < steps and time.time() - start < max_time:
        idx1 = np.where(h == 1)[0]
        idx0 = np.where(h == 0)[0]
        if len(idx1) > 0 and len(idx0) > 0:
            i = np.random.choice(idx1)
            j = np.random.choice(idx0)
            h[i], h[j] = h[j], h[i]
            corr = np.correlate(h, 1 - h, mode='full')
            new_energy = corr.max()
            delta = new_energy - current_energy
            
            if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
                current_energy = new_energy
                if current_energy < best_e:
                    best_e = current_energy
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]  # rollback
            temp *= alpha
        t += 1
    return best_h, best_e

def continuous_refine(h0, max_time):
    start = time.time()
    n = len(h0)
    h = h0.astype(float).copy()
    min_val = 0.0
    max_val = 1.0
    alpha_values = [1.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    max_iter_per_alpha = 200
    eta_0 = 0.2
    
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max()
    
    # Set initial state
    best_h = h.copy()
    best_val = compute_overlap(h)
    
    def proj_box(h):
        return np.clip(h, min_val, max_val)
    
    def proj_sum(h, target_sum):
        total = np.sum(h)
        diff = target_sum - total
        if abs(diff) < 1e-10:
            return h
        # Simple projection using Lagrange multiplier; easy to do with water filling
        # But for simplicity, just rescale
        if abs(total) > 1e-12:
            h_new = h + diff * (h / total)
        else:
            h_new = h + diff / n  # fallback
        return h_new
    
    def gradient_of_shift(h, shift):
        m = len(h)
        grad = np.zeros_like(h)
        if shift >= 0:
            s = slice(shift, None)
            e = slice(None, m - shift)
            grad[s] -= h[e]  # h_{i+k}
            s = slice(None, m - shift)
            e = slice(shift, None)
            grad[s] += h[e]  # h_{i-k}
        else:
            s = slice(-shift, None)
            e = slice(None, m + shift)
            grad[s] -= h[e]
            s = slice(None, m + shift)
            e = slice(-shift, None)
            grad[s] += h[e]
        return grad
    
    def soft_max_gradient(h, alpha):
        """Returns the soft-max gradient for the overlap function."""
        corr = np.correlate(h, 1 - h, mode='full')
        max_o = np.max(corr)
        # Subtract max to avoid overflow
        exps = np.exp(alpha * (corr - max_o))
        weights = exps / np.sum(exps)
        grad = np.zeros_like(h)
        for i in range(-n + 1, n):
            g = gradient_of_shift(h, i)
            grad += weights[i + n - 1] * g
        return grad
    
    def project_to_feasible(h):
        h = proj_box(h)
        target = n / 2
        h = proj_sum(h, target)
        return h
    
    for alpha in alpha_values:
        if time.time() - start > max_time:
            break
        for t in range(max_iter_per_alpha):
            if time.time() - start > max_time:
                break
            eta = eta_0 / np.sqrt(t + 1)
            grad = soft_max_gradient(h, alpha)
            h = h - eta * grad
            h = project_to_feasible(h)
            val = compute_overlap(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    
    return best_h, best_val
