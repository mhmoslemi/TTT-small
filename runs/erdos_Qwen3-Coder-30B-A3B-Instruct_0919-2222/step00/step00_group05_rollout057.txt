def evaluate_solution(h):
    # Calculate C5 for a discrete solution, using np.correlate
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - np.array(h), mode='full')
    return np.max(corr) * dx

def round_solution(h_star, n, M):
    # Hill climbing near the LP result
    h_star = np.array(h_star)
    rounded = np.zeros(n)
    # Simply use deterministic rounding to the closest integer vector respecting the sum constraint
    idx = np.argsort(-h_star)
    ones_indices = idx[:M]
    rounded[ones_indices] = 1
    return rounded

def random_rounding(h_star, n, M, trials=200):
    # Create binary vectors via sampling, only sample those that have mass = M
    candidates = []
    for _ in range(trials):
        probs = np.array(h_star)
        # Normalize probabilities to sum to 1
        if np.sum(probs) > 0:
            probs = probs / np.sum(probs)
        indices = np.random.choice(n, size=M, replace=False, p=probs)
        h = np.zeros(n)
        h[indices] = 1
        candidates.append(h)
    return candidates

def targeted_swap_polish(h_init, max_iter=1000):

    # Create matrix for all valid index-shift pairs

    # 1. 0 <= h_i <= 1
    # 2. sum h == M

    constraints.extend([h[i] >= 0 for i in range(n)])
    constraints.extend([h[i] <= 1 for i in range(n)])

    # We'll use a dictionary of constraints grouped by s

    # Half of the constraints are operational.
    all_y = {}
    all_y_cons = []

    for s in range(-n+1, n):
        if s >= 0:
            valid_pairs = [(i, s) for i in range(n - s)]
        else:
            valid_pairs = [(i, s) for i in range(-s, n)]

        if len(valid_pairs) > 0:
            # Set up a row of y_{i,s} as a list of variables needed
            s_y = [cp.Variable(name=f'y_{i},{s}') for (i, _) in valid_pairs]
            all_y[(s, 'pos')] = s_y
            all_y_cons.extend(s_y) # used later for computing sum_eqs

    # This part is tricky to implement directly within cvxpy without extra constructs

        constraints.append(s_sum <= M)  # upper bound from rules of thumb, OR берется вернее

        c5 = evaluate_solution(h_sol)
        print(f"Binary solution from LP: {c5}")

        # Fall back to rounding approaches
        # First deterministic rounding
        h_rounded = round_solution(h_star, n, M)
        h_rounded_eval = evaluate_solution(h_rounded)
        print(f"Deterministic rounded solution: {h_rounded_eval}")

        h_randoms = random_rounding(h_star, n, M, trials=200)
        best_random_c5 = float('inf')

        print(f"Polishing final discrete solution with random round...")
