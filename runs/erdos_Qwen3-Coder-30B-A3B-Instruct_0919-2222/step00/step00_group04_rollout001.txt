def compute_c5_bound(h, dx):
    """Compute the C5 bound using correlation."""
    # Compute correlation using np.correlate with mode='full'
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum of the correlation (excluding the middle index)
    max_corr = np.max(corr)
    return max_corr * dx

def project_simplex(h, target_sum):
    """Project h onto the simplex: sum(h) = target_sum, h_i in [0,1]."""
    # Sort h in descending order
    sorted_h = np.sort(h)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_h)
    # Find the threshold
    tmp = (cumsum - target_sum) / np.arange(1, len(sorted_h) + 1)
    idx = np.where(tmp >= sorted_h)[0]
    if len(idx) == 0:
        tau = 0
    else:
        tau = tmp[idx[-1]]
    # Compute projection
    projected_h = np.maximum(h - tau, 0)
    # Renormalize
    scale = target_sum / np.sum(projected_h)
    return projected_h * scale

                # For shift k = 0, we get additional contribution
                g[:n] += (1 - h[:n])
            else:
                # For non-zero shifts
                g[:n-k] += 1 - h[k:]
                g[k:] -= h[:n-k]
    return g * dx
