import numpy as np
import time
from scipy.signal import correlate

def evaluate(h):
    """Evaluate overlap integral for step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1-h, mode='full')
    max_val = corr.max()
    return max_val * dx

def count_mismatches(a, s):
    """Find mismatched pairs for shift s in binary vector a"""
    n = len(a)
    if s >= 0:
        M1 = np.where((a == 1) & (np.roll(a, -s) == 0))[0]
        M0 = np.where((a == 0) & (np.roll(a, -s) == 1))[0]
    else:
        M1 = np.where((a == 1) & (np.roll(a, -s) == 0))[0]
        M0 = np.where((a == 0) & (np.roll(a, -s) == 1))[0]
    return M1, M0

def binary_optimize(h0, max_time, T0=0.5, Tf=1e-4, max_iter=5000):
    """Run deterministic descent + SA for binary vector"""
    n = len(h0)
    best_h = h0.copy()
    corr = correlate(best_h, 1-best_h, mode='full')
    max_cnt = corr.max()
    best_cnt = max_cnt
    
    # deterministic descent phase
    iter_count = 0
    no_improvement_count = 0
    for _ in range(2 * n):
        if iter_count >= 2 * n:
            break
        s = corr.argmax() - (n-1)
        M1, M0 = count_mismatches(best_h, s)
        if len(M1) > 0 and len(M0) > 0:
            i, j = np.random.choice(M1), np.random.choice(M0)
            best_h[i], best_h[j] = best_h[j], best_h[i]
            corr = correlate(best_h, 1-best_h, mode='full')
            new_max = corr.max()
            if new_max <= max_cnt:
                max_cnt = new_max
                no_improvement_count = 0
                if new_max < best_cnt:
                    best_h = best_h.copy()
                    best_cnt = new_max
            else:
                no_improvement_count += 1
                best_h[i], best_h[j] = best_h[j], best_h[i]
        else:
            # random balanced swap
            i = np.random.choice(np.where(best_h == 1)[0])
            j = np.random.choice(np.where(best_h == 0)[0])
            best_h[i], best_h[j] = best_h[j], best_h[i]
            corr = correlate(best_h, 1-best_h, mode='full')
            new_max = corr.max()
            if new_max <= max_cnt:
                max_cnt = new_max
                no_improvement_count = 0
                if new_max < best_cnt:
                    best_h = best_h.copy()
                    best_cnt = new_max
            else:
                no_improvement_count += 1
                best_h[i], best_h[j] = best_h[j], best_h[i]
        iter_count += 1
        if no_improvement_count > n:
            break
    # annealing
    h_current = best_h.copy()
    corr_current = correlate(h_current, 1-h_current, mode='full')
    current_max = corr_current.max()
    if current_max < best_cnt:
        best_h = h_current.copy()
        best_cnt = current_max
    temp = T0
    temp_delta = (Tf / T0)**(1.0 / max_iter)
    for _ in range(max_iter):
        s = np.random.randint(-n+1, n)
        M1, M0 = count_mismatches(h_current, s)
        if len(M1) > 0 and len(M0) > 0:
            i, j = np.random.choice(M1), np.random.choice(M0)
            h_current[i], h_current[j] = h_current[j], h_current[i]
            corr_new = correlate(h_current, 1-h_current, mode='full')
            new_max = corr_new.max()
            delta = new_max - current_max
            if delta <= 0 or np.exp(-delta/temp) > np.random.rand():
                current_max = new_max
                if current_max < best_cnt:
                    best_h = h_current.copy()
                    best_cnt = current_max
            else:
                h_current[i], h_current[j] = h_current[j], h_current[i]
        temp *= temp_delta
    return best_h, best_cnt

def init_h(n):
    """Initial construction from initial_h_values vector"""
    h = np.array(initial_h_values)
    if len(h) > n:
        h = h[:n]
    elif len(h) < n:
        h = np.pad(h, (0, n - len(h)), mode='constant', constant_values=h[-1])
    return np.array(h)

def convert_to_binary(h):
    """Convert continuous h to binary with correct sum"""
    n = len(h)
    threshold = 0.5
    binary = (h >= threshold).astype(int)
    diff = n//2 - np.sum(binary)
    if diff > 0:
        zeros = np.where(binary == 0)[0]
        for i in range(diff):
            if i < len(zeros):
                binary[zeros[i]] = 1
    elif diff < 0:
        ones = np.where(binary == 1)[0]
        for i in range(-diff):
            if i < len(ones):
                binary[ones[i]] = 0
    return binary

def project_to_box_and_sum(v, target_sum):
    """Project to [0,1]^n and enforce sum = target_sum using water-filling"""
    n = len(v)
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    
    if abs(current_sum - target_sum) < 1e-12:
        return v
    
    def theta(v, l):
        return np.sum(np.clip(v - l, 0, 1))
    
    low, high = np.min(v) - 1, np.max(v) + 1
    for _ in range(40):
        mid = (low + high) / 2.0
        proj_sum = theta(v, mid)
        if proj_sum < target_sum:
            high = mid
        else:
            low = mid
    
    l = (low + high) / 2.0
    result = np.clip(v - l, 0, 1)
    # Adjust for floating point error
    current_sum = np.sum(result)
    scale = target_sum / current_sum
    if abs(scale - 1.0) > 1e-10:
        result = result * scale
    
    return result

def continuous_refine(h_init, budget_s):
    """Refine binary h to fractional using projected gradient with soft max surrogate"""
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    alpha_schedule = [1.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    max_iter = 200
    eta_0 = 0.2
    
    best_h = h.copy()
    corr = correlate(h, 1-h, mode='full')
    best_val = corr.max()
    
    since_last = 0
    for alpha in alpha_schedule:
        if since_last > 5:
            break
        eta = eta_0 / np.sqrt(1e-2 + 1)
        for t in range(max_iter):
            # compute per-shift gradient efficiently
            grad = np.zeros(n)
            shift_vals = []
            for k in range(1-n, n):
                if k >= 0:
                    h_shift = np.concatenate((np.zeros(k), h[:-k]))
                else:
                    h_shift = np.concatenate((h[-k:], np.zeros(-k)))
                o_k = np.sum(h * (1 - h_shift))
                shift_vals.append(o_k)
                grad += (1 - h_shift) * h
                
            # Compute softmax normalizer and gradient
            m = max(shift_vals)
            exp_vals = np.exp(alpha * (np.array(shift_vals) - m))
            p = exp_vals / np.sum(exp_vals)
            grad_sm = np.sum(
                np.array([1 - h_shift_k for k in range(1-n, n)]) 
                * np.expand_dims(p, axis=1), axis=0
            ) * h
            h -= eta * grad_sm
            h = project_to_box_and_sum(h, n/2)
            new_corr = correlate(h, 1-h, mode='full')
            new_val = new_corr.max()
            
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                since_last = 0
            else:
                since_last = since_last + 1
            
            if t % 50 == 0:
                corr_current = correlate(h, 1-h, mode='full')
                current_val = corr_current.max()
                if current_val < best_val:
                    best_val = current_val
                    best_h = h.copy()
            
            eta = eta * 0.99
        if time.time() > time.time() + budget_s - 1:
            break
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # candidate sizes to test
    sizes = [30, 36, 40, 45, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    total_time = sum(sizes)  
    remaining_time = budget_s
    
    for n in sizes:
        if time.time() - start_time > remaining_time - 5:
            break
        t_for_size = (n / total_time) * (budget_s - 30)
        if t_for_size < 1:
            continue
        remaining_time -= t_for_size
        
        # initialize with warm start or from initial_h_values
        h0 = init_h(n)
        
        # multiple restarts to get good local optima
        num_starts = min(5, int(t_for_size / 20))
        if num_starts < 1:
            num_starts = 1
        for _ in range(num_starts):
            start = time.time()
            if time.time() - start_time > remaining_time - 5:
                break
            
            # binary refinement
            h_bin = convert_to_binary(h0)
            h_final, val = binary_optimize(h_bin, t_for_size)
            
            # eval final value
            if time.time() - start_time > remaining_time - 5:
                break
            
            corr = correlate(h_final, 1-h_final, mode='full')
            val = corr.max()
            c5 = val * 2.0 / n
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()
            
            if c5 <= 0.3807:
                return (best_h, best_c5, len(best_h))
            
            # try continuous refinement for last n
            if n == sizes[-1] and time.time() - start_time < remaining_time - 30:
                h_frac, val_frac = continuous_refine(h_final, remaining_time - 30)
                corr_f = correlate(h_frac, 1-h_frac, mode='full')
                c5_f = corr_f.max() * 2.0 / n
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_frac.copy()
                    if c5_f <= 0.3807:
                        return (best_h, best_c5, len(best_h))
        
        # early termination if target achieved
        if best_c5 <= 0.3807:
            break

    if best_h is None:
        h0 = init_h(45)
        h_bin = convert_to_binary(h0)
        best_h, best_val = binary_optimize(h_bin, 10)
        corr = correlate(best_h, 1-best_h, mode='full')
        best_c5 = corr.max() * 2.0 / len(best_h)
    
    return (best_h, best_c5, len(best_h))
