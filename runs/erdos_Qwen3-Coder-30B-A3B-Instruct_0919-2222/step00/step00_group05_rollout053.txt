def evaluate(h_vals, dx):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    n = len(h_vals)
    corr = correlate(h_vals, 1 - h_vals, mode='full')
    # The full correlation includes negative lags, we need max over all lags
    # The maximum corresponds to the kernel size from 0 to 2
    # But since it's shifted, we just need to take the max of the correlation values
    # which is already computed with mode='full'
    # The slicing accounts for full correlation range properly
    c5 = np.max(corr) * dx
    return c5

def project_onto_constraints(h_vals, dx):
    """Project h_vals onto constraints: h in [0,1], sum(h) * dx = 1"""
    h_vals = np.clip(h_vals, 0, 1)
    # Normalize sum to exactly match constraint
    current_sum = np.sum(h_vals) * dx
    if current_sum > 0:
        h_vals = h_vals / current_sum * dx * len(h_vals)
    h_vals = np.clip(h_vals, 0, 1)
    return h_vals

def generate_random_feasible(n_points, dx):
    """Generate a random feasible solution"""
    h_vals = np.random.rand(n_points)
    h_vals = project_onto_constraints(h_vals, dx)
    return h_vals

def simple_local_search(h_vals, dx, iterations=1000):
    """Perform simple local search"""
    h_best = h_vals.copy()
    c5_best = evaluate(h_best, dx)

        if i % 100 == 0:
            # Re-evaluate the best to maintain history
            c5_best = evaluate(h_best, dx)
        # Try a perturbation
        h_new = h_best.copy()
        idx = np.random.randint(n)

        delta = np.random.normal(0, 0.05)
        h_new[idx] += delta
        h_new = project_onto_constraints(h_new, dx)
        c5_new = evaluate(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    return h_best, c5_best

    h_best = h_vals.copy()
    c5_best = c5_current
    
    temp = temp_start

        if i % 1000 == 0:
            temp *= cool_rate
        # Perturb
        h_new = h_current.copy()
        idx = np.random.randint(len(h_vals))
        delta = np.random.normal(0, 0.05)
        h_new[idx] += delta
        h_new = project_onto_constraints(h_new, dx)
        c5_new = evaluate(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / temp):
            h_current = h_new
            c5_current = c5_new
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
    
    return h_best, c5_best
