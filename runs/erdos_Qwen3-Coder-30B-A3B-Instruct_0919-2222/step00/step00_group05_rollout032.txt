import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def evaluate(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute the correlation integral
    corr = correlate(h, 1 - h, mode='full')
    # Only consider relevant part (the max over all k)
    max_val = max(corr[len(h)-1:])
    return max_val * dx

def project_onto_constraints(h, dx):
    """Project h onto feasible region."""
    # Ensure h[i] in [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to ensure integral = 1
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * 1.0
    return h

def generate_random_feasible(n_points, dx):
    """Generate a random feasible h."""
    h = np.random.rand(n_points)
    h = project_onto_constraints(h, dx)
    return h

def single_swap_move(h, dx, max_tries=200):
    """Perform a single swap move."""
    h_new = h.copy()
    i = random.randint(0, len(h)-1)
    j = random.randint(0, len(h)-1)
    if i != j:
        # Try to swap two elements and re-project
        h_new[i], h_new[j] = h_new[j], h_new[i]
        h_new = project_onto_constraints(h_new, dx)
    return h_new

def multiple_swap_moves(h, dx, n_moves=5):
    """Perform multiple swap moves."""
    h_new = h.copy()
    for _ in range(n_moves):
        h_new = single_swap_move(h_new, dx)
    return h_new

def run(seed=42, budget_s=600, n_points=100):
    """Run optimization to find minimal C_5."""
    # Set up timing
    start_time = time.time()
    # Set seed for reproducibility
    np.random.seed(seed)
    random.seed(seed)
    # Discretization
    dx = 2.0 / n_points
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    # Initial solution
    if 'initial_h_values' in globals():
        current_h = initial_h_values.copy()
        current_h = project_onto_constraints(current_h, dx)
    else:
        current_h = generate_random_feasible(n_points, dx)
    # Main optimization loop
    iteration = 0
    keep_going = True
    while keep_going:
        iteration += 1
        # Evaluate current solution
        current_c5 = evaluate(current_h, dx)
        # Update best solution if needed
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        # Early termination if time is up
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            keep_going = False
            break
        # Perform search moves with some randomness
        if random.random() < 0.3:
            # Try multiple random swaps
            new_h = multiple_swap_moves(current_h, dx)
        else:
            # Single swap
            new_h = single_swap_move(current_h, dx)
        # Accepting moves based on improvement
        new_c5 = evaluate(new_h, dx)
        if new_c5 < current_c5 or random.random() < 0.01:
            current_h = new_h
        # Occasionally try random restarts
        if iteration % 100 == 0 and random.random() < 0.1:
            current_h = generate_random_feasible(n_points, dx)
        # Adaptive termination
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s * 0.95:
            keep_going = False
    if best_h is None:
        best_h = current_h
    return (best_h.tolist(), best_c5, n_points)
