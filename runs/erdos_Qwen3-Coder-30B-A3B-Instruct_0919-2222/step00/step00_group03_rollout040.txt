import numpy as np
import pulp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial solution as starting point
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(initial_h)
    
    # Try larger sizes to improve bound
    sizes_to_try = [120, 150, 180, 200]
    for n in sizes_to_try:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create a new problem
        prob = pulp.LpProblem("ErdosMinOverlapMILP", pulp.LpMinimize)
        
        # Variables: h binary, T continuous
        h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
        T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
        
        # Constraint: sum h_i = n/2
        prob += pulp.lpSum([h[i] for i in range(n)]) == n/2
        
        # Product constraints and shift constraints
        for d in range(1, n):  # skip d=0 as it's trivial
            if time.time() - start_time > budget_s - 5:
                break
                
            y_vars = []
            # Create y variables for positions 0 to n-d-1
            for i in range(n - d):
                y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
                y_vars.append(y)
                # Product linearization constraints
                prob += y <= h[i]
                prob += y <= h[i+d]
                prob += y >= h[i] + h[i+d] - 1
            
            # Shift constraint: sum_y >= n/2 - T
            prob += pulp.lpSum(y_vars) >= n/2 - T
        
        # Objective: minimize T
        prob += T
        
        # Solve with time limit
        solver_time = min(30, budget_s - (time.time() - start_time) - 5)
        if solver_time <= 0:
            break
            
        solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=solver_time)
        prob.solve(solver)
        
        # Check status
        status = pulp.LpStatus[prob.status]
        if status != 'Optimal':
            continue
            
        # Extract solution
        h_vals = np.array([float(pulp.value(h[i])) for i in range(n)])
        t_val = float(pulp.value(T))
        c5_bound = t_val * (2.0 / n)
        
        # Verify true bound with evaluate function
        true_c5 = evaluate_erdos_solution(h_vals)
        
        # Prefer better bounds
        if true_c5 < best_c5:
            best_c5 = true_c5
            best_h = h_vals.copy()
            
        # Early exit if a good enough solution is found
        if best_c5 <= 0.379:
            break
            
    return (best_h, best_c5, len(best_h))
