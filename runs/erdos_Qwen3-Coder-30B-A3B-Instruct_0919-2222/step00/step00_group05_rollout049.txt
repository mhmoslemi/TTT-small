import numpy as np
import time
import math
from scipy.special import logsumexp

# Precomputed shift maps for fast correlation calculation
def precompute_shifts(n):
    shift_maps = {}
    for s in range(-n+1, n):
        indices = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                indices.append((i, j))
        shift_maps[s] = indices
    return shift_maps

# Euclidean projection onto the capped simplex {h: 0≤h≤1, sum h = M}
def project_to_capped_simplex(v, M):
    n = len(v)
    # Clip to [0,1]
    u = np.clip(v, 0, 1)
    total = np.sum(u)
    if abs(total - M) < 1e-10:
        return u.copy()
    # Now sort with indices to determine threshold theta
    indices = np.argsort(u)[::-1]
    sorted_u = u[indices]
    # Find the threshold where cumulative sum hits M
    cumsum = 0.0
    for i in range(n):
        if i > 0:
            cumsum += sorted_u[i-1]
        if (n - i) * sorted_u[i] + cumsum >= M:
            theta = (M - cumsum) / (n - i)
            break
    else:
        theta = min(1.0, (M - cumsum) / n)
    # Apply thresholding
    h = np.zeros(n)
    for i in range(n):
        idx = indices[i]
        h[idx] = max(0.0, min(1.0, v[idx] - theta))
    return h

# Compute all correlations C_s(h) = sum h_i (1 - h_{i+s})
def correlations(h, shift_maps):
    n = len(h)
    C = np.zeros(2*n - 1)
    for s, indices in shift_maps.items():
        if s >= 0:
            C[s] = sum(h[i] * (1 - h[j]) for i, j in indices)
        else:
            C[s + n - 1] = sum(h[i] * (1 - h[j]) for i, j in indices)
    return C

# Smooth approximation Phi_alpha(h) with gradient
def smooth_obj_and_grad(h, alpha, shift_maps):
    C = correlations(h, shift_maps)
    # Log-sum-exp with safe computation
    max_C = np.max(C)
    scaled_C = C - max_C
    logZ = logsumexp(alpha * scaled_C)
    Phi_alpha = logZ / alpha + max_C

    # Gradient calculation
    w = np.exp(alpha * scaled_C)
    w /= np.sum(w)
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for s, indices in shift_maps.items():
            # Compute contribution of C_s to d/dh_i
            for ii, jj in indices:
                if ii == i:
                    grad[i] += w[s + len(C)//2] * (1 - h[jj] - (h[ii - s] if ii - s >= 0 and ii - s < len(h) else 0))
    return Phi_alpha, grad

# Run gradient descent from h0 for a fixed alpha, returning final h
def run_descent(h0, alpha, eta0, max_iter, shift_maps, time_limit):
    h = h0.copy()
    eta = eta0
    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        Phi_alpha, grad = smooth_obj_and_grad(h, alpha, shift_maps)
        v = h - eta * grad
        h = project_to_capped_simplex(v, len(h) / 2)
        # Optional adaptive step size based on actual improvement
        # For efficiency, we keep fixed and decay eta
        eta = 0.9 * eta
    return h

# Continuation solver over alpha schedule
def run_continuation(h_start, alpha_schedule, eta0, shift_maps, time_limit):
    h = h_start.copy()
    start_time = time.time()
    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit:
            break
        remaining_time = time_limit - (time.time() - start_time)
        inner_iter = min(100, max(10, int(remaining_time / 10)))
        h = run_descent(h, alpha, eta0, inner_iter, shift_maps, remaining_time)
    return h

# Swap improvement
def swap_local_search(h, shift_maps, max_iter=5000):
    n = len(h)
    h = h.copy()
    C = correlations(h, shift_maps)
    best_c5 = np.max(C)
    for _ in range(max_iter):
        # Find worst shift
        s_max = np.argmax(C)
        if C[s_max] < 1e-8:  # Zero max implies done
            break
        
        # Find suitable swap pair
        if s_max < len(C) // 2:
            s = s_max
        else:
            s = s_max - len(C) // 2
        indices = shift_maps[s]
        best_improvement = 0
        best_i, best_j = -1, -1
        
        for i, j in indices:
            if h[i] == 1 and h[j] == 0:
                # Try to swap h[i][j]
                # Note: This affects multiple C_{s'} for different s'
                # but we focus only on improving C_{s_max}
                new_h = h.copy()
                new_h[i], new_h[j] = 0, 1
                new_C = correlations(new_h, shift_maps)
                improvement = C[s_max] - new_C[s_max]
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i, best_j = i, j
                    
        if best_improvement > 1e-10:
            h[best_i], h[best_j] = 0, 1
            C = correlations(h, shift_maps)
            new_c5 = np.max(C)
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            break
    return h, best_c5

# Main optimization function
def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    n = 70
    M = n // 2
    # Precompute shift maps
    shift_maps = precompute_shifts(n)
    # Create seeds
    seeds = generate_seeds(n, M, seed)
    best_c5 = float('inf')
    best_h = None
    
    # Alpha schedule
    alpha_schedule = [0.5, 1, 2, 4, 8, 16, 32]
    eta0 = 0.1
    
    # Limit to ~90% of the budget for continuous optimization
    inner_budget = budget_s * 0.9
    time_per_seed = inner_budget / len(seeds)
    
    for i, seed_h in enumerate(seeds):
        if time.time() - start_time > budget_s * 0.95:
            break
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:
            break
        h_opt = run_continuation(seed_h, alpha_schedule, eta0, shift_maps, time_per_seed)
        # Compute true C5 for this continuous vector
        C = correlations(h_opt, shift_maps)
        true_c5 = np.max(C)
        if true_c5 < best_c5:
            best_c5 = true_c5
            best_h = h_opt.copy()
            
    # Final binary rounding and polish
    if best_h is not None:
        # Deterministic rounding: pick top M ones
        sorted_indices = np.argsort(best_h)[::-1]
        h_binary = np.zeros(n)
        h_binary[sorted_indices[:M]] = 1.0
        # Local swap polish
        h_binary, refined_c5 = swap_local_search(h_binary, shift_maps, max_iter=1000)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = h_binary
    else:
        # Fallback to initial vector if nothing found
        try:
            best_h = np.array([0.42857142857142855]*70)
            best_c5 = 0.43
        except:
            return None, float('inf'), n

    # Evaluate final C5 using np.correlate as in task
    true_c5 = np.max(np.correlate(best_h, 1-best_h, mode='full'))
    dx = 2.0 / n
    true_c5 *= dx
    
    return list(best_h), true_c5, n
