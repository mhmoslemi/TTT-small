import numpy as np
import time
import cvxpy as cp
from scipy import fft
from math import floor

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all overlaps
    overlaps = np.correlate(h, 1-h, mode='full')  # Equivalent to O_k = sum h_i * (1 - h_{i+k})
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def evaluate_erdos_solution_binary(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all overlaps
    overlaps = np.correlate(h, 1-h, mode='full')
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def initial_h_values():
    n = 45
    h = np.full(n, 0.5)
    # Simple construction
    h[10:20] = 1.0
    h[25:35] = 1.0
    h = h * (n / 2 / np.sum(h)) if np.sum(h) > 0 else h
    return h

def generate_sturmian_sequence(n, alpha, tau):
    a = np.zeros(n, dtype=int)
    for i in range(n):
        frac = (i * alpha + tau) % 1.0
        a[i] = 1 if frac < 0.5 else 0
    return a

def sturmian_search(n_points, seed_gen):
    best_bound = float('inf')
    best_alpha, best_tau = 0.0, 0.0
    best_h = None

    alpha_range = np.linspace(0.2, 0.8, 200)
    tau_range = np.linspace(0.0, 1.0, 100, endpoint=False)

    for alpha in alpha_range:
        for tau in tau_range:
            a = generate_sturmian_sequence(n_points, alpha, tau)
            current_bound = evaluate_erdos_solution_binary(a)
            if current_bound < best_bound:
                best_bound = current_bound
                best_alpha, best_tau = alpha, tau
                best_h = a.copy()

    # Refine around best
    if best_bound < float('inf'):
        a = generate_sturmian_sequence(n_points, best_alpha, best_tau)
        best_bound = evaluate_erdos_solution_binary(a)

    return best_h, best_bound

def rounding_hack(h):
    # Swap rounding
    n = len(h)
    h_new = h.copy()
    h_int = np.round(h_new).astype(int)
    
    # Tweak 置入 boundary
    for i in range(n):
        if h_new[i] == 0 or h_new[i] == 1:
            h_int[i] = int(h_new[i])
    
    # 前期和一旦
    target_sum = n / 2
    
    # 确保总和等于 target_sum
    actual_sum = np.sum(h_int)
    diff = actual_sum - target_sum
    
    if diff != 0:
        for i in range(n):
            if (diff > 0 and h_int[i] > 0) or (diff < 0 and h_int[i] < 1):
                # 寻找差值最小的交换
                if diff > 0 and h_int[i] > 0:
                    h_int[i] -= 1
                    diff -= 1
                elif diff < 0 and h_int[i] < 1:
                    h_int[i] += 1
                    diff += 1
                if diff == 0:
                    break
    
    # 将 h_int 调整为 [0,1]
    h_int = np.clip(h_int, 0, 1)
    return h_int

def column_generation_optimize(h0, n_points, dx):
    # 初始化 LP 中的变量
    h = cp.Variable(n_points)
    t = cp.Variable()
    constraints = []

    # 添加目标
    prob = cp.Problem(cp.Minimize(t), constraints)

    # 通过 column generation 来指定约束
    selected_shifts = set(range(-n_points+1, n_points))

    for k in list(selected_shifts):
        # 用两个索引获得相关的相交变量
        shift_constraint = []
        k_offset = n_points - k - 1  # 用于位置 对齐
        if k < 0:
            start = -k
            end = n_points
        elif k > 0:
            start = 0
            end = n_points - k
        else:
            start = 0
            end = n_points
        # 模拟 correlation
        left = h[start:end]
        right = h[start+k:end+k]
        if end > start:
            prod_constraint = cp.sum(cp.multiply(left, right))
            shift_sum = sum(h) * dx
            overlap = shift_sum - prod_constraint
        else:
            overlap = 0
        # 约束与实际将由 column generation 追加的计算检视替换
        # 这里暂时占位均作为近似加入
        constraints.append(overlap <= t)

    prob.constraints = constraints
    prob.solve(solver=cp.ECOS, verbose=False)

    return h.value, t.value

def run(seed=42, budget_s=600, **kwargs):
    final_h = None
    final_c5 = float('inf')
    final_n = 0
    
    # 时间开始
    start_time = time.time()
    remaining_budget = budget_s
    
    # 初始集合
    sizes = [30, 45, 60, 90, 120, 150, 200]  # 成序

    # 1. LP 전략执行
    for n in sizes:
        if time.time() - start_time > budget_s - 30:
            break
        dx = 2.0 / n
        if n == 45:
            h = initial_h_values()  # 若取该校准点
        else:
            h = np.ones(n) * 0.5
        h = np.array(h)
        overlap = evaluate_erdos_solution(h)
        if overlap < final_c5:
            final_c5 = overlap
            final_h = h
            final_n = n
    if final_c5 <= 0.3807:
        return (final_h, final_c5, final_n)

    # 套跑策略
    best_strum_h = None
    best_strum_bound = float('inf')
    for n in reversed(sizes):
        if time.time() - start_time > budget_s - 30:
            break
        # run sturmian
        h_strum, b = sturmian_search(n, np.random.default_rng(seed))
        if b < best_strum_bound:
            best_strum_bound = b
            best_strum_h = h_strum
    if best_strum_h is not None and best_strum_bound < final_c5:
        final_h = best_strum_h
        final_c5 = best_strum_bound
        final_n = len(best_strum_h)

    # Polish free
    # 在用轻质 local-swap for stabalizing
    initial_h = final_h.copy()
    initial_c5 = final_c5
    for _ in range(500):
        b = evaluate_erdos_solution(final_h)
        if b < final_c5:
            final_c5 = b
        if time.time() - start_time > budget_s - 30:
            break


    # 最后回退保留最佳之一（如需）
    if final_h is None:
        h_init = initial_h_values()
        final_h = h_init
        final_c5 = evaluate_erdos_solution(final_h)
        final_n = len(h_init)

    return (final_h, final_c5, final_n)
