```python
import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_c5(h):
    """Evaluate C5 bound for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum value and multiply by dx
    return np.max(corr) * dx

def initial_h_values():
    """Return initial h values"""
    n = 45
    h = np.ones(n) * 0.5
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    h0 = initial_h_values()
    
    # Constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    def constraint_ineq(h):
        return np.array([1.0 - np.sum(h) * dx])  # maybe optimize this later
    
    # Set up optimization
    x0 = h0.copy()
    
    # Optimize using scipy
    try:
        # First run with a more relaxed approach
        result = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_eq},
            options={'maxiter': 1000, 'ftol': 1e-8},
            callback=lambda x: None  # Placeholder callback
        )
        
        if result.success:
            h_opt = result.x.copy()
        else:
            h_opt = x0.copy()
            
    except Exception as e:
        h_opt = x0.copy()
    
    # For better results, try multiple starting points
    best_h = h_opt.copy()
    best_c5 = evaluate_c5(h_opt)
    
    # Try a few random perturbations
    for i in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb current solution
        h_pert = h_opt.copy() + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        
        # Project to feasible set
        sum_h = np.sum(h_pert) * dx
        h_pert = h_pert * (1.0 / sum_h)
        
        # Evaluate the result
        c5 = evaluate_c5(h_pert)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pert.copy()
    
    # If we've not improved much, try Nelder-Mead
    # Try Nelder-Mead with random start
    for trial in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_trial = np.random.uniform(0, 1, n_points)
        h_trial = h_trial * (1.0 / np.sum(h_trial) / dx)
        
        try:
            result = minimize(
                objective,
                h_trial,
                method='Nelder-Mead',
                options={'maxiter': 500, 'disp': False}
            )
            
            if result.success:
                c5 = evaluate_c5(result.x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = result.x.copy()
        except:
            pass
    
    # Final refinement
    try:
        result = minimize(
            objective,
            best_h,
            method='COBYLA',
            constraints={'type': 'eq', 'fun': constraint_eq},
            options={'maxiter': 500, 'disp': False}
        )
        if result.success:
            final_c5 = evaluate_c5(result.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result.x.copy()
    except:
        pass
    
    return (best_h, best_c5, n_points)
```
