import numpy as np

from scipy.signal import correlate
import time
import random

def evaluate(h, dx):
    """Compute the maximum overlap integral for given step function h."""
    h = np.array(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_constraints(h, dx):
    """Project h onto feasible region: 0 <= h[i] <= 1 and sum(h)*dx = 1."""
    h = np.array(h)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum * dx - 1.0) > 1e-10:
        scale = 1.0 / (h_sum * dx)
        h = h * scale
    return h.tolist()

def generate_random_feasible(n, dx):
    """Generate a random feasible h vector."""
    h = np.random.rand(n)
    h = project_onto_constraints(h, dx)
    return h

            current_c5 = new_c5
