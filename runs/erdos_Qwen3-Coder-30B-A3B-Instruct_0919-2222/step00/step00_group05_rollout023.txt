def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    # Compute the correlation integral
    conv = signal.correlate(h, 1 - h, mode='full')
    # The maximum value is the C5 bound
    return np.max(conv) * dx

def project_to_constraints(h, dx):
    """Project h to satisfy ∫h(x)dx = 1 and 0 ≤ h ≤ 1"""
    # Ensure h values in [0,1]
    h = np.clip(h, 0, 1)
    # Rescale h to have integral = 1
    integral = np.sum(h) * dx
    if integral > 0:
        h = h / integral * 1.0
    return h

    h = project_to_constraints(h, dx)

    """Create a neighbor by swapping two values"""

    h_new[i], h_new[j] = h_new[j], h_new[i]
    return project_to_constraints(h_new, dx)
