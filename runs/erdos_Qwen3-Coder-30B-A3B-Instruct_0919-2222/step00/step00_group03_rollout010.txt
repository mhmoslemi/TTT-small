import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def solve_erdos_milp(n, time_limit):
    """Solve the Erdos minimum overlap problem via MILP."""
    prob = pulp.LpProblem("ErdosMinOverlapMILP", pulp.LpMinimize)
    
    # Variables
    h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
    T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
    
    # Constraint: sum h_i = n/2
    prob += pulp.lpSum([h[i] for i in range(n)]) == n/2

    # Product constraints and shift constraints
    for d in range(1, n):  # Skip d=0 to reduce redundancy
        # Create y variables for this shift
        y_vars = []
        for i in range(n-d):
            y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
            y_vars.append(y)
            # McCormick envelope constraints
            prob += y <= h[i]
            prob += y <= h[i+d]
            prob += y >= h[i] + h[i+d] - 1
        # Shift constraint: sum_y >= n/2 - T
        prob += pulp.lpSum(y_vars) >= n/2 - T

    # Objective
    prob += T
    
    # Solve
    solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit)
    prob.solve(solver)
    
    status = pulp.LpStatus[prob.status]
    if status not in ['Optimal', 'Feasible']:
        return None, None
    
    # Get solution
    h_vals = np.array([pulp.value(h[i]) for i in range(n)], dtype=float)
    t_val = pulp.value(T)
    c5_bound = t_val * (2.0 / n)
    
    return h_vals, c5_bound

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    n_points = 0
    
    # Try different resolutions
    candidates = [200, 180, 160, 140]
    
    for n in candidates:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 5:
            break
            
        h_vals, c5_bound = solve_erdos_milp(n, int(remaining_time * 0.9))
        
        if h_vals is not None and c5_bound is not None:
            # Evaluate using the official method
            actual_c5 = evaluate_erdos_solution(h_vals)
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h = h_vals.copy()
                n_points = n
                
            # Early stop if we get close to target
            if actual_c5 <= 0.3807:
                break
    
    # Fallback to initial solution if nothing better found
    if best_h is None:
        best_h = initial_h_values.copy()
        best_c5 = evaluate_erdos_solution(best_h)
        n_points = len(best_h)
    
    return best_h, best_c5, n_points
