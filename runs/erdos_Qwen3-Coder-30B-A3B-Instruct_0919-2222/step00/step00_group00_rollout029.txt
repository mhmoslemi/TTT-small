# Helper functions
def build_milp_model(x, y_vars, shifts, n, t):
    """
    Construct the MILP model with given variables and constraints
    """
    constraints = [
        cp.sum(x) == n // 2,
        x >= 0,
        x <= 1,
        x[0] == 1  # symmetry breaking
    ]

    for k in shifts:
        # y_vars[k] is a list of y[i,k] for i = 0..n-k-1
        for i in range(n - k):
            # McCormick envelopes
            constraints.extend([
                y_vars[k][i] <= x[i],
                y_vars[k][i] <= x[i + k],
                y_vars[k][i] >= x[i] + x[i + k] - 1,
                y_vars[k][i] >= 0,
                y_vars[k][i] <= 1
            ])
        
        # c[k] = sum(x) - sum(y[k]) <= t
        # Since sum(x) = n/2, it's (n/2) - sum(y[k]) <= t => sum(y[k]) >= (n/2) - t
        constraints.append(cp.sum(y_vars[k]) <= n//2 - t)
    
    return constraints

def evaluate_full_correlation(h_vals):
    # Evaluate with full correlation - corresponds to c_k for all k
    n = len(h_vals)
    h = np.array(h_vals)
    return np.correlate(h, 1 - h, mode="full")

def polish_solution(h_vals, max_shifts, dx):
    """
    Perform a deterministic local search to improve solution
    """
    h = np.array(h_vals)
    n = len(h)
    # Ensure integer conversion since this is a binary vector
    h = np.round(h).astype(int)
    
    max_iter = 100  # Limit number of swaps
    for _ in range(max_iter):
        corr = evaluate_full_correlation(h)
        k_star = np.argmax(corr)
        c_k_star = corr[k_star]
        if c_k_star <= 0: break  # No overlap to reduce
        
        # Early exit if we have enough overlap reduction
        if k_star >= n:
            k_star -= n  # Adjust for full correlation
        else:
            k_star = n - 1 - k_star  # Adjust for convolution indexing in numpy
            k_star = n - 1 - k_star  # Conversion to positive shift index

        # Find A and B
        A = []
        B = []
        for i in range(n):
            if h[i] == 1 and i + k_star < n and h[i + k_star] == 0:
                A.append(i)
            elif h[i] == 0 and i + k_star < n and h[i + k_star] == 1:
                B.append(i)
        
        if not A or not B:
            break  # no valid swap exists
            
        # Perform the swap
        a = A[0]
        b = B[0]
        h[a], h[b] = h[b], h[a]
        # Recompute correlation incrementally for this shift
        corr[k_star] -= 1  # This is an overapproximation, but better heuristics would be more complex
        if corr[k_star] <= 0:
            break
            
    # Return improved solution
    return h.astype(float)
