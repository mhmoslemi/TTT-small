    def compute_correlation(h):
        # Compute full correlation c_k = sum_i h[i] * (1 - h[i+k])
        # Using np.correlate with mode='full' gives shifted correlations
        # We compute as sum h[i] * (1 - h[i+k]) for k in [-n+1, n-1]
        # Output shape: 2*n-1
        n = len(h)
        one_minus_h = 1 - np.array(h)
        corr = np.correlate(h, one_minus_h, mode='full')
        return corr
    
    def smoothed_objective_and_weights(c, alpha):
        # Compute Phi_alpha = (1/alpha) * log(sum(exp(alpha * c)))
        # and weights w_k = exp(alpha * c_k) / sum(exp(alpha * c))
        max_c = np.max(c)
        shifted_c = c - max_c
        exp_c = np.exp(alpha * shifted_c)
        sum_exp = np.sum(exp_c)
        weights = exp_c / sum_exp
        phi_alpha = (1 / alpha) * (log(sum_exp) + max_c)
        return phi_alpha, weights
    
    def grad_h(h, weights):
        # Compute gradient of Phi_alpha w.r.t. h
        # ∇Phi_alpha[i] = sum_k w_k * ( (1-h[i+k]) * 1[i+k<n] - h[i-k] * 1[i-k>=0] )
        n = len(h)
        # Compute forward and backward sums efficiently
        one_minus_h = 1 - np.array(h)
        forward_sum = np.concatenate([np.cumsum(one_minus_h[::-1])[::-1], [0]])
        backward_sum = np.concatenate([[0], np.cumsum(h)[:-1]])
        # Gradient vector g
        g = np.zeros(n)
        # Compute as convolutions later.
        # Efficient way: sum over k
        # but for now use explicit loops
        for i in range(n):
            for k in range(-n+1, n):
                if k >= 0 and i+k < n:  # forward shift
                    g[i] += weights[k+n-1] * (1 - h[i+k])
                if k <= 0 and i-k >= 0:  # backward shift
                    g[i] -= weights[k+n-1] * h[i-k]
        return g
    
    def project_onto_feasible(y, target_sum):

        # This is the water-filling algorithm
        # Clipping
        y = np.clip(y, 0, 1)
        # Adjust sum
        current_sum = np.sum(y)
        diff = target_sum - current_sum
        if abs(diff) < 1e-12:
            return y
        # This is a simplified version assuming we're working with infeasible but projected y
        # The full water-filling requires a specific algorithm, but this should suffice for small case
        if diff > 0:
            # Distribute excess sum
            free = np.where((y > 0) & (y < 1))[0]
            if len(free) > 0:
                delta = diff / len(free)
                y[free] += delta
                # Clamp to [0,1]
                y = np.clip(y, 0, 1)
        elif diff < 0:
            free = np.where((y > 0) & (y < 1))[0]
            if len(free) > 0:
                delta = -diff / len(free)
                y[free] -= delta
                y = np.clip(y, 0, 1)
        else:
            pass  # No change needed
        return y

def evaluate_erdos_solution(h):
