import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def solve_erdos_milp(n, time_limit):
    # Create problem
    prob = pulp.LpProblem("ErdosMinOverlapMILP", pulp.LpMinimize)
    
    # Variables
    h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
    T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
    
    # Constraint: sum h_i = n/2
    prob += pulp.lpSum([h[i] for i in range(n)]) == n/2
    
    # Product constraints and shift constraints
    for d in range(1, n):  # skip d=0
        # Create y variables for this shift
        y_vars = []
        for i in range(n-d):
            y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
            y_vars.append(y)
            # McCormick constraints
            prob += y <= h[i]
            prob += y <= h[i+d]
            prob += y >= h[i] + h[i+d] - 1
        # Shift constraint: sum y >= n/2 - T
        prob += pulp.lpSum(y_vars) >= n/2 - T
    
    # Objective
    prob += T
    
    # Solve
    solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit)
    prob.solve(solver)
    
    # Check status
    status = pulp.LpStatus[prob.status]
    if status != 'Optimal':
        return None, None
    
    # Get solution
    h_vals = np.array([pulp.value(h[i]) for i in range(n)], dtype=float)
    t_val = pulp.value(T)
    
    return h_vals, t_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Try different n values starting from larger ones to get better bounds quickly
    n_values = [250, 200, 180, 150, 120, 100, 90, 80, 70]
    
    for n in n_values:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
            
        try:
            h_vals, t_val = solve_erdos_milp(n, int(remaining_time * 0.8))
            if h_vals is not None and t_val is not None:
                # Compute actual c5 bound using evaluate_erdos_solution
                dx = 2.0 / n
                c5_bound = t_val * dx
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_vals.copy()
                    best_n = n
                    
                # Early exit if we've beaten our target
                if best_c5 < 0.3807:
                    break
                    
        except Exception as e:
            continue
    
    # Fallback to initial solution if nothing better found
    if best_h is None:
        h_vals = np.array(initial_h_values, dtype=float)
        c5_bound = evaluate_erdos_solution(h_vals)
        best_h = h_vals
        best_c5 = c5_bound
        best_n = len(h_vals)
    
    return best_h, best_c5, best_n
