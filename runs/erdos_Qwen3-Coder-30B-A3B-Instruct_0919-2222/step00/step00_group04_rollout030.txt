    def compute_C5(h_vec):
        # Compute correlation: correlation of h with (1 - h) using full mode
        # This produces array of length 2*n-1 with shifts from -(n-1) to (n-1)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')

        # However, for optimization, we can promote the full correlation instead,
        # and then extract the relevant latest maximum from it.
        # The most appropriate approach: max_k of integral = max_k of correlation * dx
        # The correlation is 2n-1 long where middle element corresponds to lag=0
        # Our goal is to compute max_k of the integral amounts
        # Since lag l in correlation array index corresponds to shift l
        # and correlation at lag = index (n-1 + l) = sum_i h[i] (1-h[i+l])
        # For lag l, we compute from valid indices i where both i and i+l indices exist,
        # i.e., i in 0..n-1, i+l in 0..n-1.

    def projection_simplex(h_vec, total_sum):
        # Project onto the set {x: 0 <= x[i] <= 1, sum x[i] = total_sum}
        # Sort values in descending order and find threshold
        n = len(h_vec)
        sorted_indices = np.argsort(h_vec)[::-1]
        sorted_values = h_vec[sorted_indices]

        # where t_i =max(0, sorted_values[i] - threshold)

        # So recompute each x = max(0, sorted_values_i - t)
        # Sum = sum_i max(0, sorted_values[i] - t) = \sum_{i} (sorted_value[i] - t ) might be < 0
        # Thus set t such that sum = total_sum.
        # If sorted_values are in descending order, partial sums can be computed
        # Let prefix_sums be cumulative sums
        prefix_sums = np.cumsum(sorted_values)
        
        # Find largest i such that sorted_values[i] - (prefix_sums[i] - sorted_values[i]) / (i+1) >= 0
        # But simpler:
        # Find largest i such that sum = total_sum:
        # sum_i max(0, sorted_values_i - t) = sum_{i=0}^m (sorted_values_i - t)

        # Equivalent to t = (prefix_sums[m] - total_sum)/(m+1)
        # where the threshold is optimized so that sum = total_sum

        # This is the "water-filling" method.
        # We're trying to solve: find t such that:
        # sum_i max(0, h[i] - t) = total_sum
        # Or sum_{i: h[i] - t > 0} (h[i] - t) = total_sum  
        # Or check j where h[i] > t for i = 0..j (assuming t decreases), 
        # say h[0] >= h[1] >= ... >= h[j] > t >= h[j+1] >= ..., 
        # then (h[0] - t) + ... + (h[j] - t) = total_sum
        # i.e. (h[0] + ... + h[j]) - (j+1)*t = total_sum
        # We know h is ordered descendingly
        # prefix_sums[j] - (j+1)*t = total_sum
        # => t = (prefix_sums[j] - total_sum)/(j+1)

        # Walk downsorted items to find correct threshold
        # Easy version: Try each one, and compute sum
        
        best_sum = 0.0
        best_i = 0
        for i in range(n):
            # if former prefix is > total_sum, try some compromise
            if i > 0 and prefix_sums[i - 1] > total_sum:
                # This is the case where we want to "fill only first i-1 ones"
                # Correctly use i-1: If we use first i elements and distribute such
                # that sum is total_sum, then since we don't want to let any be

                # involving moved 'out'

            # Actually construct the full algorithm:
            # Find largest i such that we can use take first i elements
            # and pad rest to 0 to fill exactly to total_sum.
            # Or if there's a fraction, then some threshold t
            if i == 0:
                if sorted_values[i] >= total_sum:
                    t = 0.0
                else:
                    t = sorted_values[i] - total_sum
            if i > 0:
                # Do we have enough to give to the first i elements? Check prefix.
                # If prefix_sums[i] > total_sum, we go to more
                # we need to decide where we 'drop' the condition that
                # current cutoff is t = (prefix_sums[i] - total_sum) / (i+1)
                if prefix_sums[i] >= total_sum:
                    # Check if i'th element supports just it
                    part_sum = prefix_sums[i]

                    t = (part_sum - total_sum) / (i+1)
                    if t >= 0:

                        # The best thing is to pass this through all indices, since it 
                        # may not be exact, we use complex logic (could just project at end)

                        # Which enforces threshold t = (prefix[i] - total_sum) / (i+1)
                        # Keep moving down until feasible t > 0  
                        if t >= 0:

                            best_i = i
                            break
                        else:

    def compute_gradient(h_vec):
        # Given h_vec, compute gradient of max correlation * dx
        # correlations for all shifts including negative and positive
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take maximum correlation and its indices
        max_corr = np.max(corr)
        max_indices = np.where(np.abs(corr - max_corr) < 1e-10)[0] # indices where max
        # We still do compute exact subgradient for discretized shifts.
        # Let's be careful: set corr = corr * dx (to preserve final sign, correct for machine precision)

        # This is subgradient: let's take just worst single shift and take gradients

        # Moreover, 0 <= i+k <= n-1 for i <= n-1-k. Thus normal range.

        # + (1 - h_{i+k}) * i from 0 to n-1-k

        if k >= 0:

            #

            g[:n-k] += 1 - h_vec[k:]
            g[k:] -= h_vec[:n-k]
        else:
            # Shift k < 0
            # i+k ranges over [k, n-1+k], and we must ensure valid indices [0, n-1]
            # So i from 0 to n-1, i+k from k to n-1+k, valid is when:
            # i from 0 to n-1, i+k >= 0 => i >= -k, i+k <= n-1 => i <= n-1-k  
            # So valid range over i is max(0, -k) to min(n-1, n-1-k) = n-1-k.

            # If k is negative: i >= abs(k) and i <= n-1-k

            # So i from k_abs to n-1-k, if n-1-k >= k_abs

                # This range exists. 
                g[k_abs:n-k] += 1-h_vec[:n-k-abs(k)]
                # Also i now ranges from k_abs to n-1, so h_{i+k} = h[i+k] exists

                # derivative w.r.t h_i is -(h_{i+k})  
                # within same range of i. Index i+k 
                g[k_abs:] -= h_vec[:n-k-abs(k)]
