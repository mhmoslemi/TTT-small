import numpy as np

def run(seed=42, budget_s=600, **kwargs):

    # But let's just solve the MILP directly
    
    def solve_milp(n, time_limit):
        prob = pulp.LpProblem("ErdosMinOverlap", pulp.LpMinimize)
        
        # Variables
        h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
        T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
        
        # Constraint: sum(h_i) = n/2
        prob += pulp.lpSum([h[i] for i in range(n)]) == n/2
        
        # Product constraints and shift constraints
        for d in range(1, n):  # Skip d=0 since it's trivial
            # Create y variables for this shift
            y_vars = []
            for i in range(n-d):
                # Note: y_i_d = h_i * h_{i+d}  -- integer constraints make this exact
                y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
                y_vars.append(y)
                # McCormick envelope constraints:
                prob += y <= h[i]
                prob += y <= h[i+d]
                prob += y >= h[i] + h[i+d] - 1
            # Shift constraint: sum y >= n/2 - T
            # This corresponds to sum_i h_i h_{i+d} <= n/2 - (n/2 - T) = T
            # But looking at c_d = sum_i h_i (1-h_{i+d}) = n/2 - sum_i h_i h_{i+d}
            # So sum_i h_i h_{i+d} = n/2 - c_d
            # Hence c_d = n/2 - sum_i h_i h_{i+d}
            # Constraint c_d <= T which gives n/2 - c_d >= n/2 - T, thus c_d <= T
            # Also c_d = n/2 - sum_i h_i h_{i+d}, so sum_i h_i h_{i+d} <= n/2 - T
            # Then sum_i y_i_d >= n/2 - T (since y_i_d = h_i h_{i+d})
            prob += pulp.lpSum(y_vars) >= n/2 - T
        
        # Objective
        prob += T
        
        # Solve
        solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit)
        prob.solve(solver)
        
        # Report status
        status = pulp.LpStatus[prob.status]
        if status != 'Optimal':
            return None, None, None
            
        # Extract solution
        h_vals = np.array([pulp.value(h[i]) for i in range(n)], dtype=float)
