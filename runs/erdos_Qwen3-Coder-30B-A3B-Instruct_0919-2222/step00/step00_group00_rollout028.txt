import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Try several grid sizes
    grid_sizes = [80, 120, 160]
    
    for n in grid_sizes:
        if time.time() - t_start > budget_s - 5:
            break
            
        dx = 2.0 / n
        # Initial search with moderate shift set
        K0 = max(1, n // 4)
        K = list(range(1, K0 + 1))
        
        # Binary variables
        x = cp.Variable(n, boolean=True)
        
        # Create y variables for each active shift
        y_vars = {}
        for k in K:
            y_vars[k] = cp.Variable(n - k, boolean=True)
        
        # Scalar variable for max overlap
        t = cp.Variable()
        
        # Constraints
        constrs = []
        constrs.append(cp.sum(x) == n // 2)  # mass constraint
        
        # Symmetry breaking constraints
        constrs.append(x[0] == 1)
        
        # Non-increasing prefix (up to median)
        for i in range(n // 2 - 1):
            constrs.append(x[i] >= x[i+1])
        
        # McCormick linearization for each active shift
        for k in K:
            y = y_vars[k]
            for i in range(n - k):
                # y[i] <= x[i], y[i] <= x[i+k], y[i] >= x[i] + x[i+k] - 1
                constrs.append(y[i] <= x[i])
                constrs.append(y[i] <= x[i+k])
                constrs.append(y[i] >= x[i] + x[i+k] - 1)
        
        # Constraint to bound all valid overlaps
        for k in K:
            y = y_vars[k]
            total_ones = cp.sum(x)
            overlap_term = total_ones - cp.sum(y)
            constrs.append(overlap_term <= t)
        
        # Solver options
        solver = cp.CBC
        max_time = min(30, budget_s / 2)
        
        # Objective
        obj = cp.Minimize(t)
        
        # Solve MILP
        prob = cp.Problem(obj, constrs)
        prob.solve(solver=solver, timelimit=max_time, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            sol = x.value
            sol_t = t.value
            
            # Validate against full set of shifts
            # First compute the full correlation
            h = np.array(sol, dtype=float)
            
            # Full correlation (auto-correlation) with 1-h
            corr_full = np.correlate(h, 1-h, mode='full')
            max_corr = np.max(corr_full)
            
            # If we have tight bound, add more shifts
            if max_corr <= sol_t + 1e-4:
                # This solution is valid for the full problem
                c5 = sol_t * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n
            else:
                # Expand K and re-solve
                K_full = list(range(1, n))
                all_violations = []
                for k in K_full:
                    # Compute actual correlation
                    c_k = np.sum(h * np.concatenate([h[k:], np.zeros(k)]))
                    if c_k > sol_t:
                        all_violations.append(k)
                
                # Optimize using larger set
                if all_violations:
                    for k in all_violations:
                        if k not in K:
                            K.append(k)
                            y_vars[k] = cp.Variable(n - k, boolean=True)
                    
                    # Reset constraints for revised K
                    new_constrs = [cp.sum(x) == n // 2, x[0] == 1]
                    for i in range(n // 2 - 1):
                        new_constrs.append(x[i] >= x[i+1])
                    
                    # Re-linearize all terms
                    for k in K:
                        y = y_vars[k]
                        new_constrs.append(y[0:(n - k)] <= x[0:(n - k)])
                        new_constrs.append(y[0:(n - k)] <= x[k:n])
                        for i in range(n - k):
                            new_constrs.append(y[i] >= x[i] + x[i+k] - 1)
                    
                    #.Constraint on new max
                    for k in K:
                        y = y_vars[k]
                        total_ones = cp.sum(x)
                        overlap_term = total_ones - cp.sum(y)
                        new_constrs.append(overlap_term <= t)
                    
                    prob = cp.Problem(obj, new_constrs)
                    prob.solve(solver=solver, timelimit=max_time, verbose=False)
                    
                    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                        sol = x.value
                        sol_t = t.value
                        
                        # Validate full
                        h = np.array(sol, dtype=float)
                        corr_full = np.correlate(h, 1-h, mode='full')
                        max_corr = np.max(corr_full)
                        
                        c5 = sol_t * dx
                        if max_corr <= sol_t + 1e-4 and c5 < best_c5:
                            best_c5 = c5
                            best_h = h.copy()
                            best_n = n

        best_h = h_vals
        best_c5 = c5_bound
        best_n = len(h_vals)
    
    # Final polishing
    if time.time() - t_start < budget_s - 10 and best_h is not None:
        h_copy = best_h.copy()
        corr0 = np.correlate(h_copy, 1 - h_copy, mode='full')
        c5_orig = np.max(corr0)
        c5_orig_val = c5_orig * 2.0 / len(h_copy)
        
        # Simple local search
        success_count = 0

            corr = np.correlate(h_copy, 1 - h_copy, mode='full')
            max_k = np.argmax(corr)
            max_val = corr[max_k]
            
            # Look for good positions to swap
            ones_pos = np.where(h_copy == 1)[0]
            zero_pos = np.where(h_copy == 0)[0]
            
            if len(ones_pos) > 0 and len(zero_pos) > 0:
                # Identify good pairs that reduce overlap for the worst shift
                k_start = max_k - (len(h_copy) - 1) // 2
                k = max_k
                # The overlap for shift k is sum(h[i] * (1 - h[i+k]))
                # So if h[i]=1 and h[i+k]=0, it contributes 1 to the term
                # We want to remove a 1, add a 0 at positions that are involved in this comparison
                good_swaps = []
                for i in ones_pos:
                    if i + k < len(h_copy) and h_copy[i+k] == 0:
                        # edge case check
                        good_swaps.append((i, i+k))
                
                # Look for good don't-delete positions
                for j in zero_pos:
                    if j+k < len(h_copy) and h_copy[j+k] == 1:
                        good_swaps.append((j+k, j))
                
                if good_swaps:
                    i, j = good_swaps[0]  # Pick one to swap
                    h_copy[i], h_copy[j] = h_copy[j], h_copy[i]
                    success_count += 1
                else:
                    break  # No more improvement
            else:
                break  # No swaps possible

    if best_h is not None:

        corr_full = np.correlate(final_h, 1 - final_h, mode='full')
        final_max = np.max(corr_full)
        final_c5 = final_max * 2.0 / len(final_h)

        c5_bound = np.max(corr) * 2.0 / len(h_vals)
        return (h_vals, c5_bound, len(h_vals))
