import numpy as np
import time
from scipy.special import logsumexp

def evaluate(h):
    """Evaluates the true C5 bound using numpy correlate."""
    n = len(h)
    dx = 2.0 / n
    # correlation with 1-h
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_to_capped_simplex(v, M):
    """Project vector v onto the capped simplex: 0 <= h_i <= 1 and sum h_i = M."""
    # Clip to [0, 1]
    u = np.clip(v, 0, 1)
    s = np.sum(u)
    if np.isclose(s, M):
        return u
    elif s > M:
        # Need to decrease values
        # Find threshold theta such that sum of min(1, max(0, v_i - theta)) = M
        # Sort in decreasing order
        idx = np.argsort(-u)
        cumsum = 0
        threshold_idx = 0
        for i in range(len(u)):
            cumsum += u[idx[i]]
            if cumsum > M:
                threshold_idx = i
                break
        theta = (cumsum - M) / (len(u) - threshold_idx)
        result = np.clip(v - theta, 0, 1)
        return result
    else:
        # Need to increase values
        # Find threshold theta such that ...
        # Sort in increasing order
        idx = np.argsort(u)
        cumsum = 0
        threshold_idx = 0
        for i in range(len(u)):
            cumsum += u[idx[i]]
            if cumsum < M:
                threshold_idx = i
            else:
                break
        theta = (M - cumsum) / (threshold_idx + 1)
        result = np.clip(v + theta, 0, 1)
        return result

def correlations(h, shift_maps):
    """Compute all C_s(h) values for all shifts."""
    n = len(h)
    C = np.zeros(2*n - 1)
    # For shift s = -(n-1) to n-1
    for s, (left_indices, right_indices) in enumerate(shift_maps):
        c = 0.0
        for i, s_idx in enumerate(left_indices):
            c += h[s_idx] * (1.0 - h[right_indices[i]])
        C[s] = c
    return C

def smooth_obj_and_grad(h, alpha, shift_maps):
    """Compute smooth approximation and its gradient."""
    C = correlations(h, shift_maps)
    # Add small value to prevent overflow
    C_max = np.max(C)
    exp_C = np.exp(alpha * (C - C_max))
    Z = np.sum(exp_C)
    w = exp_C / Z
    # Gradient computation
    grad = np.zeros_like(h)
    for s, (left_indices, right_indices) in enumerate(shift_maps):
        # Go through all pairs for this shift s
        term = 1.0
        for i, j in enumerate(left_indices):
            # In our indexing convention, shift of s is s from i to i+s
            # s = shift, so i maps to i+s in right_indices[i]
            if right_indices[i] < len(h):
                term -= h[right_indices[i]]
            if right_indices[i] >= 0:  # for the other direction
                term -= h[right_indices[i]]
            grad[j] += w[s] * term
    return np.log(Z)/alpha + C_max, grad

def run_descent(h0, alpha_schedule, eta0, max_iter, shift_maps, time_limit):
    """Run projected gradient descent with continuation."""
    h = h0.copy()
    eta = eta0
    best_obj = np.inf
    best_h = h.copy()
    start_time = time.time()

    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit * 0.95:
            break
        # Inner loop with fixed alpha
        old_obj = np.inf
        for k in range(max_iter):
            if time.time() - start_time > time_limit * 0.95:
                break
            obj, grad = smooth_obj_and_grad(h, alpha, shift_maps)
            if obj < old_obj:
                old_obj = obj
            else:
                # Maybe decrease step size slightly if oscillating
                pass
            # Update
            h_new = h - eta * grad
            h = project_to_capped_simplex(h_new, len(h)/2)
            # Early stopping based on improvement.
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
            # Decay step size
            eta *= 0.95
    return best_h, best_obj

def get_shift_maps(n):
    """Precompute valid index pairs for each shift."""
    shift_maps = []
    # shift s from -(n-1) to n-1
    for s in range(-(n-1), n):
        left_indices = []
        right_indices = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                left_indices.append(i)
                right_indices.append(j)
        shift_maps.append((left_indices, right_indices))
    return shift_maps

def random_binary(n, M):
    """Generate random binary vector with exactly M ones."""
    idx = np.random.choice(n, M, replace=False)
    h = np.zeros(n)
    h[idx] = 1
    return h

def random_fractional(n, M):
    """Sample from Dirichlet(1) and project to simplex."""
    alpha = np.ones(n)  # Dirichlet alpha=1 is uniform
    v = np.random.dirichlet(alpha)
    return project_to_capped_simplex(v, M)

    best_h = h_bin.copy()
    best_c5 = evaluate(best_h)
    for _ in range(max_iter):
        if time.time() > time.time():
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_s = np.argmax(corr)
        max_corr = corr[max_s]
        # Convert full conv index to shift:
        # max_s = n-1 + shift, so shift = max_s - (n-1)
        shift = max_s - (n - 1)
        # Find source index with h[i]=1 and h[i+shift]=0
        src_indices = np.where((best_h == 1) & (np.roll(best_h, -shift) == 0))[0]
        # Find target index with h[j]=0 and h[j+shift]=1
        tgt_indices = np.where((best_h == 0) & (np.roll(best_h, -shift) == 1))[0]
        if len(src_indices) == 0 or len(tgt_indices) == 0:
            continue
        i = src_indices[0]
        j = tgt_indices[0]
        # Compute how much this changes max_corr
        # Check what we are changing
        old_contrib = 0.0
        if 0 <= i + shift < n:
            old_contrib += best_h[i] * (1 - best_h[i + shift])
        if 0 <= j + shift < n:
            old_contrib += best_h[j] * (1 - best_h[j + shift])
