import numpy as np
import cvxpy as cp
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate C_5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using full convolution
    conv = np.correlate(h, 1 - h, mode="full")
    # The overlap for shift k is at index n-1+k
    max_overlap = 0
    for k in range(-n + 1, n):
        idx = n - 1 + k
        overlap = conv[idx] * dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def initial_h_values():
    """Initial construction."""
    n = 45
    h = np.zeros(n)
    h[:23] = 1
    return h

def mcormick_relaxation_solver(h_start, n, target_bound, time_budget_s):
    """Use Column Generation + McCormick envelopes for LP relaxation."""
    start_t = time.time()
    dx = 2.0 / n
    # Initialize with a small set of shifts
    K = [0, -1, 1, -2, 2]
    max_iter = 100
    tolerance = 1e-6

    # Create symbolic variables
    h_vars = cp.Variable(n)
    t = cp.Variable()
    w_vars = {}

    # Parameters to track best solution so far
    best_h = h_start.copy()
    best_t = float('inf')
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_t
        if elapsed > time_budget_s - 2:
            break
            
        # Create LP formulation
        constraints = []
        
        # First, add fixed bound constraints for the shifts in K
        for k in K:
            if k not in w_vars:
                w_vars[k] = cp.Variable(n, nonneg=True)
            
        # Add McCormick constraints
        for k in K:
            for i in range(n):
                if 0 <= i + k < n:
                    # w_{i,k} >= h_i + h_{i+k} - 1
                    constraints.append(w_vars[k][i] >= h_vars[i] + h_vars[i + k] - 1)
                    # w_{i,k} <= h_i
                    constraints.append(w_vars[k][i] <= h_vars[i])
                    # w_{i,k} <= h_{i+k}
                    constraints.append(w_vars[k][i] <= h_vars[i + k])
                    # w_{i,k} >= 0
                    constraints.append(w_vars[k][i] >= 0)

        # Add average constraint
        constraints.append(cp.sum(h_vars) == n / 2)

        # Add box constraints
        for i in range(n):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)

        # Add actual objective constraint: Sum_i h_i - Sum_i w_{i,k} <= t
        for k in K:
            expr = cp.sum(h_vars) - cp.sum(w_vars[k])
            constraints.append(expr <= t)
            
        # Define optimization problem
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
        except:
            break

        if prob.status != cp.OPTIMAL:
            break
            
        t_val = t.value
        h_sol = h_vars.value
        
        # Check if we're done
        if t_val < best_t:
            best_t = t_val
            best_h = h_sol.copy()
        
        # Compute all overlaps
        all_overlaps = {}
        for k in range(-n + 1, n):
            idx = n - 1 + k
            overlap = n / 2 - np.sum([h_sol[i] * h_sol[min(max(i + k, 0), n - 1)] for i in range(n)])
            all_overlaps[k] = overlap * dx
        
        max_k = max(all_overlaps, key=lambda k: all_overlaps[k])
        max_overlap = all_overlaps[max_k]
        if max_overlap <= t_val + tolerance:
            break
        K.append(max_k)
        if len(K) > 30:
            # Cap the number of shifts
            # Drop some shifts if we're exceeding this limit
            # Evaluate only a sample of the most important ones
            # For now just stop adding new ones
            break
            
    if best_t < target_bound:
        best_h = best_h.clip(0, 1)
        return best_h, best_t * 2 / n
    else:
        return None, target_bound  # Return default

def swap_rounding(h_frac, n):
    """Apply swap-based rounding to turn fractional h into binary."""
    h_binary = h_frac.copy()
    
    # Ensure sum is exactly n/2
    total = np.sum(h_binary)
    if abs(total - n / 2) > 1e-10:
        # Adjust
        adjust = (n / 2 - total) / np.sum(h_binary != 0) if np.sum(h_binary != 0) > 0 else 0
        h_binary += adjust
        h_binary = h_binary.clip(0, 1)
        
    # Ensure it's binary
    num_iter = 0
    max_iter = 2 * n
    while num_iter < max_iter:
        # Find first fractional entry
        i = -1
        for idx in range(n):
            if 0 < h_binary[idx] < 1:
                i = idx
                break
                
        if i == -1:
            break
            
        j = -1
        for idx in range(n):
            if 0 < h_binary[idx] < 1 and idx != i and abs(h_binary[idx] - h_binary[i]) < 1e-10:
                j = idx
                break
                
        if j == -1:
            # Otherwise, pick any other fractional entry
            for idx in range(n):
                if 0 < h_binary[idx] < 1 and idx != i:
                    j = idx
                    break
                    
        if j == -1:
            break
            
        delta = min(h_binary[i], 1 - h_binary[j])
        if np.random.rand() < 0.5:
            h_binary[i] -= delta
            h_binary[j] += delta
        else:
            h_binary[i] += delta
            h_binary[j] -= delta
            
        h_binary[i] = max(0, min(1, h_binary[i]))
        h_binary[j] = max(0, min(1, h_binary[j]))
        num_iter += 1

    return h_binary
    
def sturmian_construction(n):
    """Theoretical construction given by Sturmian sequences with Beatty words."""
    def sturmian_sequence(alpha, tau, n):
        sequence = np.zeros(n, dtype=int)
        for i in range(n):
            val = (i * alpha + tau) % 1
            sequence[i] = 1 if val < 0.5 else 0
        return sequence

    # Grid search for best alpha and tau
    best_bound = float('inf')
    best_h = None
    alpha_grid = np.linspace(0.2, 0.8, 200)
    tau_grid = np.linspace(0.0, 1.0, 100)  # We want [0, 1)
    best_alpha = 0.5
    best_tau = 0.0
    
    # Precompute for fewer evaluations
    for ai, alpha in enumerate(alpha_grid):
        if ai % 10 != 0:
            continue
        for ti, tau in enumerate(tau_grid):
            if ti % 5 != 0:
                continue
            a = sturmian_sequence(alpha, tau, n)
            res = evaluate_erdos_solution(a)
            if res < best_bound:
                best_bound = res
                best_alpha = alpha
                best_tau = tau

    else:
        # Grant rough grid resolution for final choice
        pass

    # Refinement process
    refine_shape = [np.linspace(best_alpha - 0.01, best_alpha + 0.01, 20),
                    np.linspace(best_tau - 0.01, best_tau + 0.01, 20)]

    for _ in range(5):  # Refinement loop
        best_val = float('inf')
        for alpha_val in refine_shape[0]:
            for tau_val in refine_shape[1]:
                a = sturmian_sequence(alpha_val, tau_val, n)
                res = evaluate_erdos_solution(a)
                if res < best_val:
                    best_val = res
                    best_h = a.copy()
                    best_alpha, best_tau = alpha_val, tau_val

        # Narrow the range for next refinement
        refine_shape = [
            np.linspace(max(0.1, best_alpha - 0.001), min(0.9, best_alpha + 0.001), 20),
            np.linspace(max(0.0, best_tau - 0.001), min(1.0, best_tau + 0.001), 20)
        ]
        if best_val < best_bound:
            best_bound = best_val
        
    return best_h, best_bound

def solve_worst_shift_swap(initial_h, max_iter=5000, n=None):
    """Local improvement through swapping worst shifts."""
    if n is None:
        n = len(initial_h)
    h = initial_h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_bound = float('inf')
    
    for i in range(max_iter):
        conv = np.correlate(h, 1 - h, mode="full")
        max_k = 0
        max_overlap = -float('inf')
        
        for k in range(-n + 1, n):
            idx = n - 1 + k
            overlap = conv[idx] * dx
            if overlap > max_overlap:
                max_overlap = overlap
                max_k = k
        
        # If we're already fine
        if max_overlap < best_bound:
            best_bound = max_overlap
            best_h = h.copy()
            
        # If no improvement possible
        if max_overlap == 0:
            break
            
        # Try to fix the worst shift by swapping
        # Find all possible swaps that involve shift k
        max_k_idx = n - 1 + max_k
        adjusted = False
        for i in range(n):
            j = (i + max_k) % n
            if h[i] == 1 and h[j] == 0:
                # Swap h[i] and h[j]
                h[i], h[j] = 0, 1
                adjusted = True
                break
            elif h[i] == 0 and h[j] == 1:
                # Swap h[i] and h[j]
                h[i], h[j] = 1, 0
                adjusted = True
                break
                
        if not adjusted:
            # Something went wrong or we just don't want to alter anymore
            break
            
    return best_h, best_bound

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    target = 0.3807
    
    candidates = []
    
    # Sizes to explore
    sizes = [30, 45, 60, 90, 120, 150, 200]
    max_size = 200
    
    for n in sizes:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        
        h0 = initial_h_values()
        h0 = h0[:n] if len(h0) > n else np.pad(h0, (0, n - len(h0)), constant_values=0)
        best_h = h0.copy()
        best_c5 = float('inf')
        
        # Try large size via shifting analog of Sturmian construction 
        if n == max_size:
            h_s, c5_s = sturmian_construction(n)
            if h_s is not None:
                h_s, c5_s = solve_worst_shift_swap(h_s, max_iter=min(500, n), n=n)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                    
        # Use convex relaxation + column generation for small-medium sizes
        if n <= 150:
            try:
                # Run LP relaxation with column generation
                h_lp, c5_lp = mcormick_relaxation_solver(best_h, n, target, remaining / 3)
                if h_lp is not None:
                    h_lp = swap_rounding(h_lp, n)
                    c5_lp = evaluate_erdos_solution(h_lp)
                    if c5_lp < best_c5:
                        best_c5 = c5_lp
                        best_h = h_lp.copy()
            except Exception as e:
                pass

        # Do a local population of the stochastic manifolds if needed
        if time.time() - start_time > budget_s - 2:
            break
        
        # Final swap tightening only if needed
        if (time.time() - start_time) < budget_s - 10:
            if best_c5 > 0.39:
                try:
                    h_final, c5_final = solve_worst_shift_swap(best_h, max_iter=min(300, 2 * n), n=n)
                    if c5_final < best_c5:
                        best_c5 = c5_final
                        best_h = h_final.copy()
                except Exception as e:
                    pass
                
        candidates.append((best_h.copy(), best_c5, n))

    # Now select the best result from the done candidates
    overall_best = None
    overall_bound = float('inf')
    for h, c, n in candidates:
        if c < overall_bound:
            overall_bound = c
            overall_best = (h.copy(), c, n)

    if overall_best is None:
        # Fallback to initial vector
        h0 = initial_h_values()
        h0 = h0[:max_size] if len(h0) > max_size else np.pad(h0, (0, max_size - len(h0)), constant_values=0)
        c5 = evaluate_erdos_solution(h0)
        overall_best = (h0, c5, max_size)

    return overall_best[0], overall_best[1], overall_best[2]
