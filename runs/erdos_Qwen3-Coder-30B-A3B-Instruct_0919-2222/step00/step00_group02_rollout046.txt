import numpy as np
import time
import cvxpy as cp
from scipy import signal
from math import floor, ceil

def evaluate_erdos_solution(h_values):
    """Evaluate the actual C5 value for given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute all overlaps O_k = sum_i h_i * (1 - h_{i+k})
    # We use correlation in reverse mode and adjust for the 1-h term
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def initial_h_values():
    """Initial construction: 45 samples"""
    n = 45
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1 if (i < 22) else 0
    h[22] = 0.5  # Adjust to achieve sum = 22.5
    return h

def compute_overlaps(h):
    """Compute all overlaps O_k for a given h (for verification etc)"""
    n = len(h)
    overlaps = []
    for k in range(-n+1, n):
        conv = signal.correlate(h, 1 - np.roll(h, k), mode="valid")
        overlaps.append(conv[0] if len(conv) > 0 else 0)
    return overlaps

def compute_max_overlap(h):
    """Compute max overlap value for a given h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def round_fractional_vector(h, seed=42):
    """Round a fractional vector to binary while preserving sum."""
    np.random.seed(seed)
    h = h.copy()
    n = len(h)
    s = np.sum(h)
    target_sum = s  # We want sum = target_sum

    while True:
        # Count fractional entries
        frac_indices = [i for i in range(n) if 0 < h[i] < 1]
        if not frac_indices:
            break

        # Pick a pair of fractional entries with same fractional part
        # Or just pick any two
        i = frac_indices[np.random.randint(len(frac_indices))]
        candidates = [j for j in frac_indices if j != i]
        if not candidates:
            # If only one fractional entry left, move it to nearest integer
            h[i] = round(h[i])
            continue
        j = candidates[np.random.randint(len(candidates))]

        # Determine delta
        delta1 = h[i]
        delta2 = 1 - h[j]
        delta = min(delta1, delta2)

        # Perform swap with 50/50 probability
        if np.random.rand() < 0.5:
            h[i] -= delta
            h[j] += delta
        else:
            h[i] += delta
            h[j] -= delta

        # Ensure no value goes out of [0,1]
        h[i] = np.clip(h[i], 0, 1)
        h[j] = np.clip(h[j], 0, 1)
        
        # Ensure sum is exact
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            continue  # Sum is fine
        # Adjust the last entry
        h[-1] += (target_sum - current_sum)
        h[-1] = np.clip(h[-1], 0, 1)

    return np.round(h).astype(int)

def solve_lp_column_generation(h0, n, max_time=590):
    """Solve relaxed LP using column generation approach"""
    start_time = time.time()
    
    # Initialize K with few shifts (brief initial evaluation)
    K = [0, -1, 1, -2, 2]
    h = h0.copy()
    dx = 2.0 / n

    while True:
        if time.time() - start_time > max_time:
            break
            
        # Constraint generation (generate constraints only for current K)
        constraints = []
        t = cp.Variable()
        
        # Variables for products
        w = {}
        for i in range(n):
            for k in K:
                if 0 <= i + k < n:
                    w[(i,k)] = cp.Variable()
        
        # McCormick envelopes for bilinear constraints
        # We do this explicitly in loop below but always active
        # We try a different approach: just add constraints for k in K
        
        # Objective
        objective = cp.Minimize(t)
        
        # Formulate constraints
        c_list = []
        for k in K:
            s = 0.0
            for i in range(n):
                if 0 <= i + k < n:
                    s += w[(i,k)]
            c_expr = 0.5 - s <= t
            c_list.append(c_expr)
            
        # Add bounds for w
        for i in range(n):
            for k in K:
                if 0 <= i + k < n:
                    w[(i,k)].bounds = 0, h[i]  # Already constraint w <= h[i]
                    w[(i,k)].bounds = 0, h[i+k]  # w <= h[i+k]
                    c = w[(i,k)] >= h[i] + h[i+k] - 1  # w >= h[i] + h[i+k] - 1
                    c_list.append(c)
        
        # Add constraints for lower bound on w too (meaningful McCormick bounds)
        # Actually we use direct variable bounds during the computation,
        # but shorter version using explicit McCormick
        for i in range(n):
            for k in K:
                if 0 <= i + k < n:
                    # w_ij <= h[i], w_ij <= h[i+k], w_ij >= h[i] + h[i+k] - 1
                    c1 = w[(i,k)] <= h[i]
                    c2 = w[(i,k)] <= h[i+k]
                    c3 = w[(i,k)] >= h[i] + h[i+k] - 1
                    c_list.append(c1)
                    c_list.append(c2)
                    c_list.append(c3)
        
        # Note: solver cannot directly handle w(i,k) as bilinear already
        # So instead compute full overlaps and solve LP in smaller scope

        # Do this step: solve constraint LP over permutations of K
        # Do dual parsing
        # Let's use a more direct simulation - fix a simpler approach

        # Reset variables to avoid naming conflict
        constraints = []
        t_var = cp.Variable()

        w_vars = {}
        Wh = [cp.Variable() for _ in range(n * len(K))]

        idx = 0
        for i in range(n):
            for k in K:
                if 0 <= i + k < n:
                    w_vars[(i, k)] = Wh[idx]
                    idx += 1

        # Constraints based on w_vars and sums for overlaps
        all_constraints = []
        for j, k in enumerate(K):
            s = 0.0
            for i in range(n):
                if 0 <= i + k < n:
                    s += w_vars[(i, k)]
            # O_k = 0.5 - sum(w)
            # Constraint: 0.5 - sum(w) <= t
            # So: sum(w) >= 0.5 - t
            # i.e., sum(w) - 0.5 >= -t => -sum(w) + 0.5 >= t
            # That is: t >= 0.5 - sum(w)
            # Therefore: sum(w) <= 0.5 - t  =>  sum(w) + t <= 0.5
            
            c1 = sum(w_vars[(i, k)] for i in range(n) if 0 <= i+k < n) <= (0.5 - t)  # equivalent to above
            all_constraints.append(c1)

        # Add actual bounds
        for i in range(n):
            for k in K:
                if 0 <= i + k < n:
                    all_constraints.append(w_vars[(i, k)] >= h[i] + h[i+k] - 1)
                    all_constraints.append(w_vars[(i, k)] <= h[i])
                    all_constraints.append(w_vars[(i, k)] <= h[i+k])

        # Final problem  
        problem = cp.Problem(objective, all_constraints + [
            cp.sum(h) == 0.5 * n,
            *[0 <= h[i] <= 1 for i in range(n)],
            t >= 0
        ])

        # Try different solvers
        try:
            if problem.is_dcp:
                problem.solve(solver=cp.ECOS, max_iters=5000)
            else:
                problem.solve(solver=cp.GLPK, max_iters=5000)
        except Exception as e:
            print("LP failed trying rescaling: ", e)
            break

        # After solving, get LHS value for all k and check if wrong is still violated
        # Set up all overlaps for possible k, compute hook manually
        
        # This approach is slow, approach alternative
        # We'll compute correlations over all k-values with current h
        
        # So instead of weakening LP, if failure, we proceed to compute full overlaps and add constraint if needed
        # So compute full overlaps O_k(h) for all shifts and find worst
        # Token straddle to avoid confusion remains

        # Compute full O_k(h) using direct correlation
        # O_k = sum([h[i] * (1 - h[i+k]) for i in range(n)])
        corr_vals = compute_overlaps_for_all_k(h)
        # Find max k index as k_star
        max_idx = max(range(len(corr_vals)), key=lambda i: corr_vals[i])
        k_star = max_idx - (len(corr_vals) - 1) // 2  # Convert index to shift k
        O_k_star = corr_vals[max_idx]

        # Check if max overlap is below t + EPS (but use small epsilon)
        if O_k_star <= t.value + 1e-6:
            break

        # Add constraint
        if len(K) < 30:  # Limit on our K set size
            K.append(k_star)
        else:
            if True:  # This is fallback mode if not enough space
                break  # Primitive error handling

    return h, max(O_k_star, 0) * dx

def compute_overlaps_for_all_k(h):
    """returns overlap vector O_k for all k """
    n = len(h)
    overlaps = []
    for k in range(-n+1, n):
        corr = signal.correlate(h, 1 - np.roll(h, k), mode="valid")
        overlaps.append(corr[0])
    return overlaps

def sturmian_construction(n, seed=42):
    """Construct using Beatty/Sturmian sequence"""
    np.random.seed(seed)
    best_bound = 1e9
    best_h = None
    best_alpha, best_tau = None, None
    step_alpha = 0.003
    step_tau = 0.01

    # Coarse grid search
    alphas = np.arange(0.2, 0.8, step_alpha)
    taus = np.arange(0., 1., step_tau)

    for alpha in alphas:
        for tau in taus:
            h = np.zeros(n)
            for i in range(n):
                f_part = (i * alpha + tau) % 1
                h[i] = 1 if f_part < 0.5 else 0

            # Check if we have correct sum
            count = np.sum(h)
            if abs(count - n / 2) < 0.5:
                # Compute overlap
                overlap = compute_max_overlap(h)
                if overlap < best_bound:
                    best_bound = overlap
                    best_h = h.copy()
                    best_alpha, best_tau = alpha, tau

    # If not good enough, refine with golden ratio search in subregion
    if best_bound > 0.381 or best_h is None:
        return None, best_bound

    # Local golden-section refinement
    def objective(alpha_tau):
        alpha, tau = alpha_tau
        h = np.zeros(n)
        for i in range(n):
            f_part = (i * alpha + tau) % 1
            h[i] = 1 if f_part < 0.5 else 0
        return compute_max_overlap(h)

    # Use relevant segment from prior grid and refine further modestly
    refine_alpha_range = [0.3, 0.7]
    refine_tau_range = [0., 1.]

    # Simple bisection to find alpha/tau
    for _ in range(5):
        alpha = best_alpha
        tau = best_tau
        # Refine
        max_val = 1e9
        best_A, best_T = alpha, tau
        for a in np.linspace(refine_alpha_range[0], refine_alpha_range[1], 10):
            for t in np.linspace(refine_tau_range[0], refine_tau_range[1], 10):
                h = np.zeros(n)
                for i in range(n):
                    f_part = (i * a + t) % 1
                    h[i] = 1 if f_part < 0.5 else 0
                o = compute_max_overlap(h)
                if o < max_val:
                    max_val = o
                    best_A = a
                    best_T = t
        best_alpha, best_tau = best_A, best_T
        if max_val < best_bound:
            best_bound = max_val

    # Final construction
    final_h = np.zeros(n)
    for i in range(n):
        f_part = (i * best_alpha + best_tau) % 1
        final_h[i] = 1 if f_part < 0.5 else 0
    # Adjust to ensure sum = n//2
    s = np.sum(final_h)
    if abs(s - n / 2) > 0.5:
        # Try nearest change
        k = int((s - n / 2))
        if k > 0:
            # Find and flip k ones to zeros
            ones = np.where(final_h == 1)[0]
            for i in range(k):
                if i < len(ones):
                    final_h[ones[i]] = 0
        elif k < 0:
            zeros = np.where(final_h == 0)[0]
            for i in range(-k):
                if i < len(zeros):
                    final_h[zeros[i]] = 1

    # Local polishing swap
    max_iter = min(5000, int(n * n * 0.1))
    best_final_h = final_h.copy()
    current_bound = compute_max_overlap(best_final_h)
    iter_count = 0

    while iter_count < max_iter:
        if time.time() > time.time() + 580:  # 600 total limit, leave 20 for sanity check
            break
        # Get index of worst shift
        overlaps = compute_overlaps_for_all_k(best_final_h)
        max_idx = max(range(len(overlaps)), key=lambda i: overlaps[i])
        worst_k = max_idx - (len(overlaps) - 1) // 2
        # Move nearest 1 to nearest 0 and see difference
        # Find all pairs that differ by -1 in the table when shifted 
        max_o = overlaps[max_idx]
        # Flip one 1 to 0 and one 0 to 1
        ones = np.where(best_final_h == 1)[0]
        zeros = np.where(best_final_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        indices = []
        for i in ones:
            for j in zeros:
                # Swap i and j
                test_h = best_final_h.copy()
                test_h[i] = 0
                test_h[j] = 1
                test_o = compute_max_overlap(test_h)
                if test_o < max_o:
                    indices.append((i, j, max_o - test_o))  # store diff

        if indices:
            # Do best gain path
            indices.sort(key=lambda x: x[2])
            i, j, gain = indices[0]
            best_final_h[i] = 0
            best_final_h[j] = 1
            current_bound = compute_max_overlap(best_final_h)
        else:
            break  # No improvement left
        iter_count += 1

    return best_final_h, current_bound

def solve_for_size(n, seed=42, time_budget=600):
    """Solve for a given n"""
    dx = 2.0 / n
    start_time = time.time()
    time_left = lambda: time_budget - (time.time() - start_time)

    # Initial guess using pre-existing vector
    h = initial_h_values()
    h = h[:n] if len(h) > n else np.concatenate([h, np.zeros(n - len(h))])
    # Rescale to target sum = n/2
    old_sum = np.sum(h)
    if old_sum != 0:
        h *= (n / 2) / old_sum

    # LP Column Generation
    if time_left() < 10:
        return None, 1e9
    h_rounded, lp_bound = solve_lp_column_generation(h, n, time_budget * 0.9)
    if h_rounded is not None and time_left() > 30:
        h_rounded = round_fractional_vector(h_rounded, seed=seed)
        lp_bound = compute_max_overlap(h_rounded)

    # Sturmian construction - reserved for large sizes or fast convergence
    if time_left() < 30:
        sturm_h, sturm_bound = None, 1e9
    else:
        sturm_h, sturm_bound = sturmian_construction(n, seed=seed)

    # Choose better between two methods
    final_h = h_rounded if lp_bound < sturm_bound else (sturm_h if sturm_h is not None else h_rounded)
    final_bound = min(lp_bound, sturm_bound)
    if final_h is not None and time_left() > 5:
        final_bound = compute_max_overlap(final_h)

    return final_h, final_bound

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    time_budget = budget_s

    sizes = [30, 45, 60, 90, 120, 150, 200]  # sequence to iterate through
    best_solution = None
    best_c5 = 1e9
    best_n = 30

    start_time = time.time()
    remaining_time = time_budget

    for n in sizes:
        if time.time() - start_time > time_budget - 10:
            break  # leave 10s margin
        solver_budget = min(remaining_time * 0.8, 250)
        h, bound = solve_for_size(n, seed, solver_budget)
        if bound < best_c5:
            best_c5 = bound
            best_solution = h
            best_n = n
        remaining_time = time_budget - (time.time() - start_time)

    # Use best if no good found, otherwise compress to final h and return computed
    if best_solution is not None:
        h_final = best_solution
        c5 = compute_max_overlap(h_final)
        n_final = best_n
        return (h_final, c5, n_final)
    else:
        # Fallback to initial
        h_initial = initial_h_values()
        c5_initial = compute_max_overlap(h_initial)
        return (h_initial, c5_initial, len(h_initial))
