import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import math

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using full convolution
    overlaps = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def initial_h_values():
    """Return the initial construction with n=45 points."""
    n = 45
    h = np.zeros(n)
    # Simple construction based on the idea that we want to 'spread out'
    # the function to minimize overlaps
    # Start from a scaled version of the reference solution that works
    h[:15] = 0.0
    h[15:30] = 1.0
    h[30:] = 0.0
    return h

def solve_lp_column_generation(h0, n, target_bound, max_time_s):
    """Run LP column generation with McCormick relaxations."""
    start_t = time.time()
    dx = 2.0 / n
    # Set up starting shifts
    K = set([0, -2, -1, 0, 1, 2])
    max_k = n - 1
    max_shifts = max_k * 2 + 1

    solver_opts = {"max_iters": 5000}
    
    def get_shifts(used_shifts):
        shifts = list(used_shifts)
        # Ensure we have a reasonable initial set
        if len(shifts) < 5:
            shifts = [0, -2, -1, 1, 2]
        return shifts

    # Iteratively add constraints
    while True:
        if time.time() - start_t > max_time_s - 5:
            break
            
        shifts = get_shifts(K)
        
        # Variables
        h_vars = cp.Variable(n, name="h")
        t_var = cp.Variable(name="t")
        
        # Constraints
        constraints = []
        
        # Box constraints
        constraints += [h_vars >= 0]
        constraints += [h_vars <= 1]
        
        # Constraint: sum(h) = n/2
        constraints += [cp.sum(h_vars) == n / 2]
        
        # McCormick relaxations for each k in shifts
        w_vars_dict = {}
        for k in shifts:
            w_vars = cp.Variable(n, name=f"w_{k}")
            w_vars_dict[k] = w_vars
            # Setup McCormick envelopes  
            for i in range(n):
                if i + k < 0 or i + k >= n:
                    continue
                constraints += [w_vars[i] >= 0]
                constraints += [w_vars[i] >= h_vars[i] + h_vars[i+k] - 1]
                constraints += [w_vars[i] <= h_vars[i]]
                constraints += [w_vars[i] <= h_vars[i+k]]
        
        # For each k, the constraint: sum(h_i) - sum(w_{i,k}) <= t
        for k in shifts:
            w_vars = w_vars_dict[k]
            relevant_indices = [i for i in range(n) if 0 <= i+k < n]
            def compute_overlaps(h_vals):
                # Direct calculation via correlations
                if k >= 0:
                    # shift > 0
                    shifted = np.concatenate([np.zeros(k), h_vals[:-k]])
                else:
                    # shift < 0
                    k = -k
                    shifted = np.concatenate([h_vals[k:], np.zeros(k)])
                return np.sum(h_vals * shifted)
            # But we want: h_sum - sum(w) <= t
            # So only add constraint that is actively used
            # Another benefit of bc onset of cuts is we directly see
            # current recommended constraint k with max g_i
            overlap_val = cp.sum(h_vars) - cp.sum(w_vars[relevant_indices])
            constraints += [overlap_val <= t_var]
        
        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, **solver_opts)
        except Exception as e:
            break
            
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            break
            
        # 1. Get current fractional h
        h_frac = h_vars.value
        
        if h_frac is None:
            break
            
        # 2. Compute all overlaps
        # Using FFT when possible for speed
        full_corr = correlate(h_frac, 1 - h_frac, mode="full")
        max_overlap_idx = np.argmax(full_corr)
        max_overlap = full_corr[max_overlap_idx]
        
        # Determine k corresponding to this max overlap
        # full_corr indices x=0,1,...,2n-2 correspond to shifts -n+1,...,n-1
        shift_index = max_overlap_idx - (n - 1)
        if shift_index in K:
            # Done
            break
            
        # Accept new constraint
        K.add(shift_index)
        
        if max_overlap <= t_var.value + 1e-6:
            break
            
        # Check stopping condition on bound.
        if t_var.value * dx <= target_bound:
            break

    # Final evaluation of fractional h that satisfied the algorithm
    h_fin = h_vars.value
    if h_fin is None:
        h_fin = h0
        
    return h_fin, t_var.value if t_var.value is not None else np.inf

def swap_rounding(h_frac, n):
    """Apply binary swap rounding to make h integer-valued in [0,1] with sum n/2."""
    h = h_frac.copy()
    h = np.clip(h, 0, 1)
    needed = n / 2 - np.sum(h)
    
    while abs(needed) > 1e-10:
        # Find two fractional values at least one noninteger
        fractional_indices = np.where((h > 0) & (h < 1))[0]
        if len(fractional_indices) < 2:
            # If only one non-integer, just increment/decrement it
            for i in range(n):
                if abs(h[i] - 0.5) < 1e-8:
                    continue
                if needed > 0:
                    h[i] = min(1.0, h[i] + needed)
                elif needed < 0:
                    h[i] = max(0.0, h[i] + needed)
                needed = 0.0
                break
        else:
            # Pick two fractional elements to round toward integer
            # If they have same fractional part, round them together.
            diff = h - np.floor(h)  # fractional part
            index_pairs = list(zip(np.where(diff>0)[0], np.where((1-diff)>0)[0]))
            if not index_pairs:
                index_pairs = [(i, j) for i in range(n) for j in range(n) if (i<j)]
            
            # Pick a pair with same fractional parts, if possible
            pair = None
            for i, j in index_pairs:
                rounded_i = round(h[i])
                rounded_j = round(h[j])
                if abs(diff[i] - (1 - diff[j])) < 1e-10:
                    pair = (i, j)
                    break
                    
            if not pair:
                pair = index_pairs[0]
            i, j = pair
            
            # Pull values toward integers with probability 0.5 for each direction
            delta = min(h[i], 1 - h[j])
            if np.random.rand() < 0.5:
                h[i] -= delta
                h[j] += delta
            else:
                h[i] += delta
                h[j] -= delta
                
            needed = n / 2 - np.sum(h)
            
            if abs(needed) < 1e-10:
                break
                
    return h

def stochastic_swap_improvement(h, max_iter=500):
    """Improvement phase: avoid increasing overlap."""
    h = h.copy()
    dx = 2.0 / len(h)
    original_c5 = evaluate_erdos_solution(h)
    
    for _ in range(max_iter):
        best_change = 0
        best_i = -1
        best_j = -1
        # Try move to improve  
        new_h = h.copy()
        for i in range(len(h)):
            for j in range(i + 1, len(h)):
                # Try moving a unit from h[i] to h[j] if possible.
                if h[i] >= 1 and h[j] < 1:
                    new_h[i] -= 1
                    new_h[j] += 1
                    new_c5 = evaluate_erdos_solution(new_h)
                    if new_c5 < original_c5 - 1e-8:
                        c = original_c5 - new_c5
                        if c > best_change:
                            best_change = c
                            best_i = i
                            best_j = j
                            h = new_h.copy()
                            original_c5 = new_c5
                    else:
                        new_h[i] += 1
                        new_h[j] -= 1
        
        if best_i < 0:
            break
            
    return h

def fitness_sturmian(alpha, tau, n):
    """Evaluate a Sturmian sequence fitness."""
    h = np.array([1 if ((i * alpha + tau) % 1) < 0.5 else 0 for i in range(n)])
    return evaluate_erdos_solution(h)

def sturmian_search(n, max_time_s):
    """Parameter tune Sturmian construction to find good binaries."""
    start_t = time.time()
    
    # Heuristic starting values
    alpha_list = np.linspace(0.2, 0.8, 200)
    tau_list = np.linspace(0, 1, 100)
    
    best_c5 = np.inf
    best_alpha = 0.0
    best_tau = 0.0
    best_h = None
    
    grid_evals = 0
    
    for alpha in alpha_list:
        for tau in tau_list:
            if time.time() - start_t > max_time_s - 20:
                break
            grid_evals += 1
            h = np.array([1 if ((i * alpha + tau) % 1) < 0.5 else 0 for i in range(n)])
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_alpha = alpha
                best_tau = tau
                best_h = h.copy()
    
    if best_h is None:
        return None, np.inf
        
    # Local golden-section refinement for best alpha
    lo_a, hi_a = max(0.2, best_alpha - 0.03), min(0.8, best_alpha + 0.03)
    resolution = 1e-8
    
    # Golden ratio method is for minimization of unimodal functions
    phi = (1 + math.sqrt(5)) / 2
    resphi = 2 - phi
    
    def eval_alpha(alpha_val):
        h = np.array([1 if ((i * alpha_val + best_tau) % 1) < 0.5 else 0 for i in range(n)])
        return evaluate_erdos_solution(h)
        
    a, b = lo_a, hi_a
    resx = None
    for _ in range(20):
        # Recursive check for early exit
        if time.time() - start_t > max_time_s - 10:
            break
        # Ensure a modular range  
        x1 = a + resphi * (b - a)
        x2 = b - resphi * (b - a)
        
        if eval_alpha(x1) < eval_alpha(x2):
            b = x2
            resx = x1
        else:
            a = x1
            resx = x2
            
    if resx is not None:
        best_alpha = resx
    
    # Now with best alpha, refine tau too
    lo_t, hi_t = 0, 1
    for _ in range(20):
        if time.time() - start_t > max_time_s - 10:
            break
        x1 = lo_t + resphi * (hi_t - lo_t)
        x2 = hi_t - resphi * (hi_t - lo_t)
        
        h1 = np.array([1 if ((i * best_alpha + x1) % 1) < 0.5 else 0 for i in range(n)])
        h2 = np.array([1 if ((i * best_alpha + x2) % 1) < 0.5 else 0 for i in range(n)])
        
        if evaluate_erdos_solution(h1) < evaluate_erdos_solution(h2):
            hi_t = x2
        else:
            lo_t = x1
            
    # After refinement, make the final solution
    h_final = np.array([1 if ((i * best_alpha + hi_t) % 1) < 0.5 else 0 for i in range(n)])
    
    # Some local polishing
    h_final = stochastic_swap_improvement(h_final, max_iter=min(500, n * 50))
    
    return h_final, evaluate_erdos_solution(h_final)
    
def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    dx = 2.0 / 45
    n_points = 45
    h0 = initial_h_values()

    # Define set of n sizes to process in order of preference
    candidates = [30, 45, 60, 90, 120, 150, 200]
    
    best_result = None
    best_bound = np.inf
    
    # Ensure enough time for final polishing
    remaining_time = budget_s - 30
    time_per_n = {}
    
    for n in candidates:
        # Distribute time inversely proportional to size
        time_per_n[n] = max(20, min(60, remaining_time * n / sum(candidates)))
        if time_per_n[n] <= 0:
            break
    
    # Phase 1: LP + Column Generation
    for n in sorted(time_per_n.keys()):
        if time.time() - time.time() + remaining_time + 20 > 600:
            break
        start_time = time.time()
        try:
            # Solve using LP with column generation
            h_frac, t_opt = solve_lp_column_generation(h0, n, 0.3807, time_per_n[n])
            c5_bound = t_opt * 2.0 / n
            
            if c5_bound < 0.3807:
                h_rounded = swap_rounding(h_frac, n)
                actual_c5 = evaluate_erdos_solution(h_rounded)
                if actual_c5 < best_bound:
                    best_bound = actual_c5
                    best_result = (h_rounded, actual_c5, n)
        except:
            continue
            
    # Phase 2: sturmian search on largest n with time left
    if len(time_per_n) > 0:
        largest_n = max(time_per_n.keys())
        if time.time() - start_time < 600 - 5:
            try:
                h_sturmian, c5_sturmian = sturmian_search(largest_n, 600 - (time.time() - start_time) - 20)
                if h_sturmian is not None and c5_sturmian < best_bound:
                    best_bound = c5_sturmian
                    best_result = (h_sturmian, c5_sturmian, largest_n)
            except:
                pass
    
    # If no result found, return the initial construction
    if best_result is None:
        c5_initial = evaluate_erdos_solution(h0)
        return (h0, c5_initial, 45)
        
    final_h, final_c5, final_n = best_result
    
    return (final_h, final_c5, final_n)
