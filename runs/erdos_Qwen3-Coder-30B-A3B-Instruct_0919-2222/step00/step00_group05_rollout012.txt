def evaluate(h):
    """Evaluate the overlap integral using correlation"""
    dx = 2.0 / len(h)
    # Correlation is computed as convolving h with (1-h)
    # We do not want to flip h for this because h(x) * (1 - h(x+k)) is what is being maximized
    # So maximum of correlation <- h*(1-h) over all k gives C5
    h_windowed = np.array(h)
    one_minus_h = 1 - h_windowed
    corr = signal.correlate(h_windowed, one_minus_h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def project_onto_constraints(h):
    """Ensure the sum constraint is met and values are in [0,1]"""
    n = len(h)
    dx = 2.0 / n
    # Clamp values to [0,1]
    h_clamped = np.clip(h, 0, 1)
    # Adjust to satisfy integral constraint: sum(h)*dx = 1
    current_sum = np.sum(h_clamped) * dx
    h_scaled = h_clamped * (1.0 / current_sum)
    return np.clip(h_scaled, 0, 1)
