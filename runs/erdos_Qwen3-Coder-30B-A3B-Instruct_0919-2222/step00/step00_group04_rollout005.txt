import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from math import sqrt

def evaluate_erdos_solution(h, n_points=42):
    """Evaluate the Erdős constant for given step function h"""
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts k
    # h is a step function with n_points samples
    # We compute the maximum of the correlation integral
    correlation = np.correlate(h, 1 - h, mode='full')
    # The integral is taken over [0,2], so we need to multiply by dx
    # But we need to take the maximum of the integral across k values
    # The overlap integral is ∫h(x)(1-h(x+k))dx
    # We have correlation values for lags from -n_points+1 to n_points-1
    # The maximum value of correlation corresponds to the highest overlap 
    # when we compute the integral properly
    max_corr = np.max(correlation)
    return max_corr * dx

def objective(h_vals, n_points=42):
    """Objective function to minimize - negative of the max overlap"""
    dx = 2.0 / n_points
    correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
    return -np.max(correlation) * dx

def constraint_eq(h_vals, n_points=42):
    """Equality constraint: integral of h must be 1"""
    dx = 2.0 / n_points
    return np.sum(h_vals) * dx - 1.0

def constraint_bounds(h_vals, n_points=42):
    """Bound constraint: each h_i should be in [0,1]"""
    # We don't use these directly as scipy can handle bounds
    # But we return void for now
    return 0.0

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    
    n_points = 42
    dx = 2.0 / n_points
    
    # Start with initial construction, if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)  # Scale to meet integral constraint
    else:
        # Default initialization  
        h0 = np.ones(n_points) * 0.5
        
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Need to include equality constraint for ∫h(x)dx = 1  
    # This means sum(h)*dx = 1, so sum(h) = n_points/2
    
    # Define the constraint for the integral
    def eq_constraint(x):
        return np.sum(x) - n_points / 2.0
    
    constraints = {'type': 'eq', 'fun': eq_constraint}
    
    # Start timing
    start_time = time.time()
    
    # Method 1: Try differential evolution on the full problem 
    # This is more robust for discrete and multimodal problems
    try:
        # For a global search with fewer samples
        result = differential_evolution(
            objective, 
            bounds, 
            args=(n_points,),
            constraints=constraints,
            seed=seed,
            maxiter=max(500, budget_s // 2),  # 500 iterations for a safe try
            popsize=15,  # Smaller popsize for faster search
            init='latinhypercube',
            disp=False
        )
        
        if not result.success:
            # Fallback to local optimization if global search fails
            result = minimize(
                objective,
                result.x if 'result' in locals() else h0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-8},
                tol=1e-8
            )
            
    except Exception as e:
        # Fallback to a simpler approach
        result = minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-8},
            tol=1e-8
        )
    
    # Extract h_values and check if we have a better solution
    h_final = result.x if hasattr(result, 'x') else h0
    
    # Make sure the solution is within bounds
    h_final = np.clip(h_final, 0, 1)
    
    # Enforce the equality constraint exactly
    current_sum = np.sum(h_final)
    if abs(current_sum - n_points / 2.0) > 1e-6:
        # Adjust to fulfill the constraint
        scale_factor = (n_points / 2.0) / current_sum
        h_final = h_final * scale_factor
        if not np.all((h_final >= 0) & (h_final <= 1)):
            # Project back onto constraints
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    
    # Evaluate final solution
    c5_bound = -objective(h_final, n_points)
    
    # Safety: Make sure the constraints are met
    integral_check = np.sum(h_final) * dx
    if abs(integral_check - 1.0) > 1e-6:
        # Compensate again if needed
        h_final = h_final * (1.0 / integral_check)
    
    c5_bound = evaluate_erdos_solution(h_final, n_points)
    
    # Early termination strategy
    if time.time() - start_time > budget_s - 1:  # Wait a bit for final evaluation
        pass
        
    return h_final.tolist(), c5_bound, n_points
