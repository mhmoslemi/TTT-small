import numpy as np
import time
from scipy.signal import correlate
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 501  # odd number for better symmetry
    dx = 2.0 / n_points
    target_time = time.time() + budget_s * 0.99
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    
    # Generate triangular shape
    x = np.arange(n_points) * dx + dx / 2  # midpoint positions
    h_triangle = np.maximum(0, 1 - np.abs(x - 1))
    
    # Normalize triangle to integral = 1 (sum = n_points/2)
    h_triangle = h_triangle * (n_points / 2) / np.sum(h_triangle)
    
    # Evaluate
    c5_tri = evaluate_erdos_solution(h_triangle)
    if c5_tri < best_c5:
        best_c5 = c5_tri
        best_h = h_triangle.copy()
    
    # Generate cosine^2 shape
    h_cos2 = np.cos(np.pi / 2 * (x - 1)) ** 2
    
    # Normalize
    h_cos2 = h_cos2 * (n_points / 2) / np.sum(h_cos2)
    
    # Evaluate
    c5_cos2 = evaluate_erdos_solution(h_cos2)
    if c5_cos2 < best_c5:
        best_c5 = c5_cos2
        best_h = h_cos2.copy()
    
    # Try cosine with different powers
    for p in [0.5, 1.5, 3, 6]:
        h_cos_p = np.cos(np.pi / 2 * (x - 1)) ** p
        # Normalize
        h_cos_p = h_cos_p * (n_points / 2) / np.sum(h_cos_p)
        # Evaluate
        c5_cos_p = evaluate_erdos_solution(h_cos_p)
        if c5_cos_p < best_c5:
            best_c5 = c5_cos_p
            best_h = h_cos_p.copy()
    
    # Now run local optimization from best shape found so far
    h = best_h.copy()
    current_c5 = best_c5
    
    # Optimize using simulated annealing
    temperature = 0.01
    min_temp = 1e-6
    cooling_rate = 0.99995
    iteration = 0
    max_iterations = 50000
    
    while time.time() < target_time and iteration < max_iterations:
        # Random move
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        if i == j:
            continue
            
        # Compute max allowable move
        delta_max = min(h[i], 1 - h[j])
        if delta_max <= 0:
            continue
            
        # Make move
        delta = np.random.random() * delta_max * 0.1
        
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h_new)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
        else:
            # Accept with probability
            if np.random.random() < np.exp(-(new_c5 - current_c5) / temperature):
                h = h_new
                current_c5 = new_c5
                
        # Cool down
        temperature = max(min_temp, temperature * cooling_rate)
        iteration += 1
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
