def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum correlation value times dx
    max_overlap = np.max(corr)
    return max_overlap * dx
