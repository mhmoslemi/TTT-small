import numpy as np
import time
import random

def evaluate_h(h):
    """Evaluate the maximum overlap integral C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode='full')
    max_O = O_counts.max()
    return max_O * dx

def max_overlap_counts(h):
    """Return the maximum overlap count (integer) without dx scaling."""
    O_counts = np.correlate(h, 1 - h, mode='full')
    return O_counts.max()

def random_balanced_binary(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    ones = rng.sample(range(n), n // 2)
    h[ones] = 1
    return h

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto box [0,1] and sum constraint."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    rng = random.Random(seed)
    
    # Predefined list of n values to try
    n_values = [30, 40, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    start_time = time.time()
    
    for n in n_values:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute target for this n
        target_max = int(np.floor(0.3807 * n / 2))
        
        # Try multiple restarts for this n
        max_restarts = max(1, min(10, int(100000 / (n * n))))
        for restart in range(max_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Initialize binary vector
            h = random_balanced_binary(n, rng)
            cur_val = max_overlap_counts(h)
            best_so_far = cur_val
            best_h_so_far = h.copy()
            
            # Simulated Annealing loop
            max_iters = min(50000, 1000000 // n)
            temp_start = 1.0
            temp_end = 0.001
            
            for it in range(max_iters):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                # Compute temperature (exponential decay)
                fraction = float(it) / max_iters
                temp = temp_start * (temp_end / temp_start) ** fraction
                
                # Propose swap
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    
                    # Swap
                    h[i], h[j] = 0, 1
                    new_val = max_overlap_counts(h)
                    delta = new_val - cur_val
                    
                    # Accept or reject
                    accept = False
                    if delta <= 0:
                        accept = True
                    else:
                        if rng.random() < np.exp(-delta / temp):
                            accept = True
                    
                    if accept:
                        cur_val = new_val
                        if cur_val < best_so_far:
                            best_so_far = cur_val
                            best_h_so_far = h.copy()
                        if cur_val <= target_max:
                            break
                    else:
                        # Revert swap
                        h[i], h[j] = 1, 0
                else:
                    # If no valid swap, just swap random elements
                    i = rng.randint(0, n - 1)
                    j = rng.randint(0, n - 1)
                    if h[i] == h[j]:
                        continue
                    h[i], h[j] = h[j], h[i]
                    new_val = max_overlap_counts(h)
                    delta = new_val - cur_val
                    if delta <= 0:
                        cur_val = new_val
                        if cur_val < best_so_far:
                            best_so_far = cur_val
                            best_h_so_far = h.copy()
                    else:
                        h[i], h[j] = h[j], h[i]  # revert
                    
            # Keep best found so far
            if best_so_far < best_c5:
                best_c5 = best_so_far
                best_h = best_h_so_far.copy()
                best_n = n
                
            # Check if we've achieved the goal
            if best_c5 <= 0.3807:
                break
                
    if best_h is not None:
        # Refine with continuous optimization
        h_cont = best_h.astype(float)
        best_c5_cont = evaluate_h(h_cont)
        
        # Simple gradient descent refinement with projection
        max_iter = 10000
        step_size = 0.1
        iter_count = 0
        start_refinement = time.time()
        
        while iter_count < max_iter and (time.time() - start_time < budget_s * 0.95):
            if time.time() - start_refinement > 10:
                break
                
            # Compute gradients for max O
            O_counts = np.correlate(h_cont, 1 - h_cont, mode='full')
            max_idx = np.argmax(O_counts)
            shift = max_idx - (len(h_cont) - 1)
            
            # Compute gradient
            grad = np.zeros_like(h_cont)
            if shift >= 0:
                if shift < len(h_cont):
                    grad[:-shift] -= h_cont[shift:]
                if shift < len(h_cont):
                    grad[shift:] -= h_cont[:-shift]
            else:
                s = -shift
                if s < len(h_cont):
                    grad[s:] -= h_cont[:-s]
                if s < len(h_cont):
                    grad[:-s] -= h_cont[s:]
            grad += 1  # Add 1 for each position
            
            # Step and project
            h_new = h_cont - step_size * grad
            h_new = project_onto_box_and_sum(h_new, len(h_new) / 2)
            
            new_c5 = evaluate_h(h_new)
            if new_c5 < best_c5_cont:
                best_c5_cont = new_c5
                h_cont = h_new.copy()
                
            iter_count += 1
            
        # Final evaluation
        final_h = h_cont if best_c5_cont < best_c5 else best_h
        final_c5 = min(best_c5_cont, best_c5)
        return (final_h, final_c5, len(final_h))
        
    else:
        return (np.array([0.5] * 45), 0.3809, 45)
