import numpy as np
import time
from scipy.special import softmax
import math

def evaluate_overlap(h):
    """Evaluates the maximum overlap integral for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlate
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = corr.max()
    return max_overlap * dx

def binary_search(n, time_budget_s):
    """Performs a binary local search for a given n."""
    np.random.seed(42)
    best_h = None
    best_val = float('inf')
    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > time_budget_s - 1:
            break
        # Create random balanced binary vector
        h = np.zeros(n, dtype=int)
        ones = np.random.permutation(n)[:n//2]
        h[ones] = 1

        # Perform deterministic local search
        iterations = 0
        max_iter = min(10000, 2 * n)
        while iterations < max_iter and time.time() - start_time < time_budget_s - 1:
            # Compute overlaps
            corr = np.correlate(h, 1 - h, mode="full")
            max_cnt = corr.max()
            if max_cnt < best_val:
                best_val = max_cnt
                best_h = h.copy()

            # Find worst shift
            worst_shift_idx = corr.argmax()
            worst_shift = worst_shift_idx - (n - 1)
            # Compute mismatched indices
            M1 = []
            M0 = []
            if worst_shift >= 0:
                for i in range(n - worst_shift):
                    if h[i] == 1 and h[i + worst_shift] == 0:
                        M1.append(i)
                    if h[i] == 0 and h[i + worst_shift] == 1:
                        M0.append(i)
            else:
                for i in range(-worst_shift, n):
                    if h[i] == 1 and h[i + worst_shift] == 0:
                        M1.append(i)
                    if h[i] == 0 and h[i + worst_shift] == 1:
                        M0.append(i)
            # If both mismatch sets are non-empty, attempt a swap
            if M1 and M0:
                i = np.random.choice(M1)
                j = np.random.choice(M0)
                # Perform the swap
                h[i], h[j] = h[j], h[i]
                iterations += 1
            else:
                # Random balanced swap
                ones_indices = np.where(h == 1)[0]
                zeros_indices = np.where(h == 0)[0]
                if ones_indices.size > 0 and zeros_indices.size > 0:
                    i = np.random.choice(ones_indices)
                    j = np.random.choice(zeros_indices)
                    h[i], h[j] = h[j], h[i]
                    iterations += 1
                else:
                    break

        # Simulated annealing restarting from best binary vector
        if best_h is not None:
            h = best_h.copy()
            curr_max = np.correlate(h, 1 - h, mode="full").max()
            T0 = 0.5
            Tf = 1e-4
            T = T0
            steps = 0
            max_steps = min(5 * n * math.log(n), 50000)
            while steps < max_steps and time.time() - start_time < time_budget_s - 1:
                # Propose a swap
                ones_indices = np.where(h == 1)[0]
                zeros_indices = np.where(h == 0)[0]
                if ones_indices.size > 0 and zeros_indices.size > 0:
                    i = np.random.choice(ones_indices)
                    j = np.random.choice(zeros_indices)
                    # Swap
                    h[i], h[j] = h[j], h[i]
                    new_max = np.correlate(h, 1 - h, mode="full").max()
                    delta = new_max - curr_max
                    # Accept or reject
                    if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                        curr_max = new_max
                        if curr_max < best_val:
                            best_val = curr_max
                            best_h = h.copy()
                    else:
                        h[i], h[j] = h[j], h[i]
                steps += 1
                T = Tf + (T0 - Tf) * (1 - steps / max_steps)

    return best_h, best_val

            break

        eta = 0.2
        for t in range(num_steps):
            if time.time() - start_time > time_budget_s - 5:
                break

            grad = np.zeros(n)
            corr = np.correlate(h, 1 - h, mode="full")
            # Normalize to avoid overflow
            max_o = corr.max()
            exps = np.exp(alpha * (corr - max_o))
            weights = exps / exps.sum()
            for k in range(-n + 1, n):
                # Holds grad_k[i] = partial / partial h_i o_k(h)
                grad_k = np.zeros(n)
                if k >= 0:
                    grad_k[k:n] = 1 - h[:n-k]
                else:
                    grad_k[:n+k] = 1 - h[-k:]
                grad += weights[k + n - 1] * grad_k
            # Do gradient step
            h -= eta / np.sqrt(t + 1) * grad
            # Project onto [0,1] and sum constraint
            h_projected = np.clip(h, 0, 1)
            # Water filling to satisfy sum constraint
            sum_diff = np.sum(h_projected) - n / 2

            # Which means we set y_i = x_i + lambda if x_i > 0 or x_i < 1
            # And then solve sum(x_i + lambda * (x_i > 0) * (x_i < 1)) = n/2
            # by trying to find lambda such that sum(x_i + lambda * (0 < x_i < 1)) = n/2.
            # Instead we still set x_i = clip(y_i - lambda, 0, 1), so solve sum(clip(y - lambda, 0, 1)) = n/2
            y = h_projected
            def phi(lam):
                # def clip(x, a, b): return max(a, min(b, x))
                return np.sum(np.clip(y - lam, 0, 1))- n / 2

            # Binary search lambda such that phi(lambda) = 0
            # Assuming the sequence is decreasing, and is bounded independently.
            lo = np.min(y) - 1
            hi = np.max(y) + 1
            for _ in range(40):
                mid = (lo + hi) / 2
                ph = phi(mid)
                if ph < 0:
                    hi = mid
                else:
                    lo = mid

            lambda_opt = (lo + hi) / 2
            h = np.clip(y - lambda_opt, 0, 1)

            # Re-evaluate max_overlap
            corr = np.correlate(h, 1 - h, mode="full")
            new_val = corr.max()
            if new_val < curr_val:
                curr_val = new_val
            # Reduce learning rate for later steps
            eta *= 0.999

    return h, curr_val
