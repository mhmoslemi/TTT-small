def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value of the overlap integral
    max_overlap = np.max(corr) * dx
    return max_overlap
