import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum of these values
    max_overlap = np.max(correlation) * dx
    return max_overlap

def initial_h_values():
    # Return the initial construction from the problem description
    n_points = 42
    h = np.ones(n_points) * 0.5
    # Simple construction - a step function that's 1 for first half and 0 for second half
    h[:n_points//2] = 1.0
    return h

def objective(h):
    """Objective function to be minimized"""
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Get the initial construction
    h_init = initial_h_values()
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint function: sum(h) * dx = 1 => sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Use differential evolution for global optimization
    result = differential_evolution(objective, bounds, seed=seed, 
                                   maxiter=int(budget_s/10), 
                                   popsize=15, 
                                   mutation=(0.5, 1), 
                                   recombination=0.7,
                                   disp=False)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure constraint is satisfied (adjust slightly if needed)
    current_sum = np.sum(best_h)
    scaling_factor = (n_points / 2.0) / current_sum
    if abs(current_sum - n_points / 2.0) > 1e-6:
        best_h = best_h * scaling_factor
    
    # Clamp values to [0,1]
    best_h = np.clip(best_h, 0, 1)
    
    return (best_h, best_c5, n_points)
