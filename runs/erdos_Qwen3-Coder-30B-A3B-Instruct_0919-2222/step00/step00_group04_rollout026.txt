import numpy as np
from scipy.optimize import differential_evolution
import time

def project_to_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum x = target_sum"""
    v = np.asarray(v)
    n = len(v)
    if n == 0:
        return np.array([])
    if np.isclose(np.sum(v), target_sum):
        return np.clip(v, 0, 1)
    
    v_sorted = np.sort(v)[::-1]
    v_cumsum = np.cumsum(v_sorted)
    idx = np.argmax(v_sorted * np.arange(1, n+1) - (v_cumsum - target_sum)) 
    if idx == 0:
        if v_sorted[0] > target_sum:
            threshold = target_sum
        else:
            threshold = v_sorted[0]
    else:
        threshold = (v_cumsum[idx-1] - target_sum) / idx
    x = np.clip(v - threshold, 0, 1)
    return x

def compute_autocorrelation_overlaps(w, a, N=200):
    """Compute the autocorrelation G(k) for all shift k for the piecewise constant function"""
    # compute interval boundaries
    s = np.concatenate([[0], np.cumsum(w)])
    m = len(w)
    # discretize shifts to same resolution as final grid
    dx = 2.0 / N
    shift_idx = np.arange(N) * 2 / N  # shift values in [0,2]
    G = np.zeros(N)
    
    # use dense loops to compute overlap of intervals
    for i in range(m):
        for j in range(m):
            s_i, s_i1 = s[i], s[i+1]
            s_j, s_j1 = s[j], s[j+1]
            # iterate over all shifts
            for kidx in range(N):
                k = shift_idx[kidx]
                overlap = max(0, min(s_i1, s_j1 + k) - max(s_i, s_j + k))
                G[kidx] += a[i] * a[j] * overlap
    
    return G

def evaluate_fitness(vars, M, penalty_weight=1000):
    # unpack variables
    u = vars[:M]
    v = vars[M:]
    
    # map to widths and heights
    w = 2 * np.exp(u) / np.sum(np.exp(u))  # softmax to ensure sum=2
    a = 1 / (1 + np.exp(-v))  # sigmoid to ensure [0,1]
    
    # integral check
    mass = np.sum(a * w)
    integral_penalty = penalty_weight * abs(mass - 1)
    
    # cannot use overlaps in birds - but this is so that DE finds something plausible
    # instead, we return (a reasonable proxy) from a similar computation:
    # for the fitness: we estimate the worst overlap based on a few sampled steps
    # we go with a simplistic estimate using n=200 points like the final function
    N = 200
    G = compute_autocorrelation_overlaps(w, a, N)
    G_min = np.min(G)
    obj = 1 - G_min
    return obj + integral_penalty


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    M = 12  # number of piecewise blocks
    penalty_weight = 1000.0
    
    # initial bounds for DE
    bounds = [(-5, 5)] * (2 * M)
    
    # DE optimization
    de_result = differential_evolution(
        func=evaluate_fitness, 
        bounds=bounds,
        args=(M, penalty_weight),
        seed=seed,
        maxiter=30,
        popsize=15*2*M,
        polish=False,
        seed=seed,
        callback=lambda x, convergence: None
    )
    
    # Extract best solution from DE
    u_best = de_result.x[:M]
    v_best = de_result.x[M:]
    
    # convert back to widths and heights
    w_best = 2 * np.exp(u_best) / np.sum(np.exp(u_best))
    a_best = 1 / (1 + np.exp(-v_best))
    
    # adjust for integral
    mass = np.sum(a_best * w_best)
    if mass > 0:
        a_best = a_best / mass
    a_best = np.clip(a_best, 0, 1)
    # renormalize
    a_best = a_best / np.sum(a_best * w_best) * 1.0
    
    # Sample onto fine grid
    N = 200
    h_vals = []
    dx = 2.0 / N
    s = np.concatenate([[0], np.cumsum(w_best)])
    for j in range(N):
        x = j * dx
        for i in range(len(w_best)):
            if s[i] <= x < s[i+1]:
                h_vals.append(a_best[i])
                break
        else:
            h_vals.append(a_best[-1])
    
    # Validate integral
    h = np.array(h_vals)
    integral_check = np.sum(h) * dx
    if abs(integral_check - 1.0) > 1e-6:
        # Re-project to correct sum
        h = project_to_capped_simplex(h, N / 2)
    
    # Now fine tune with subgradient method
    max_iter = 4000
    alpha0 = 0.2
    best_h = h.copy()
    best_max_corr = float('inf')
    
    # subgradient method
    for t in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr = corr[max_corr_idx]
        if max_corr < best_max_corr:
            best_max_corr = max_corr
            best_h = h.copy()
        
        # find worst shift
        k = max_corr_idx - (len(h) - 1)
        # gradient component for this k
        g = np.zeros_like(h)
        if k == 0:
            g[:-1] = 1 - h[1:]
            g[1:] -= h[:-1]
        elif k > 0:
            # k shift = shift right: h_i -> h_{i+k}
            g[k:] = 1 - h[1+k:]
            g[:k] = 0
        else:
            # shift left
            g[:k] = 1 - h[1:k]  # this logic need filling...
            g[k:] -= h[:k]
        
        # step
        alpha = alpha0 / np.sqrt(t + 1)
        h = h - alpha * g
        # project
        h = project_to_capped_simplex(h, N/2)
    
    # final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_max_corr = np.max(final_corr)
    final_c5 = final_max_corr * (2.0 / N)
    
    return (best_h.tolist(), final_c5, N)
