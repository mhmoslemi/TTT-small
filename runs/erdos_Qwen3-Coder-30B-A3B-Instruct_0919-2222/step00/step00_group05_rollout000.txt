def evaluate_erdos_solution(h):
    """Evaluate the Erdos minimum overlap constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n

    corr = np.correlate(h, 1 - h, mode="full")
    # Max across all shifts (excluding trivial high shifts where it equals 1)
    max_corr = np.max(corr)
    return max_corr * dx
