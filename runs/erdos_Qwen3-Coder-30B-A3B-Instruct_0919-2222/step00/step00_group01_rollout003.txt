    triangle_c5 = evaluate(triangle_h)
    if triangle_c5 < best_c5:
        best_h = triangle_h.copy()
        best_c5 = triangle_c5
    
    # Cos^2 shape
    cos2_h = np.cos(np.pi/2 * (x - 1))**2
    cos2_h *= (n_points / 2) / np.sum(cos2_h)
    cos2_c5 = evaluate(cos2_h)
    if cos2_c5 < best_c5:
        best_h = cos2_h.copy()
        best_c5 = cos2_c5
    
    # Cos^4 shape
    cos4_h = np.cos(np.pi/2 * (x - 1))**4
    cos4_h *= (n_points / 2) / np.sum(cos4_h)
    cos4_c5 = evaluate(cos4_h)
    if cos4_c5 < best_c5:
        best_h = cos4_h.copy()
        best_c5 = cos4_c5
    
    # Polynomial shapes
    for a in [1.5, 2.0, 3.0]:
        poly_h = np.maximum(0, 1 - np.abs(x - 1)**a)
        poly_h *= (n_points / 2) / np.sum(poly_h)
        poly_c5 = evaluate(poly_h)
        if poly_c5 < best_c5:
            best_h = poly_h.copy()
            best_c5 = poly_c5
    
    # Hill climbing
    iteration_limit = 200000  # To allow for time checks
    epsilon_base = 0.05
    epsilon = epsilon_base
    
    for i in range(iteration_limit):
        if time.time() > end_time:
            break
            
        # Random pair move
        i1 = np.random.randint(0, n_points)
        i2 = np.random.randint(0, n_points)

        if delta_max <= 0:

            best_h *= (n_points / 2) / np.sum(best_h)
    
    # Final evaluation to ensure accuracy
    final_c5 = evaluate(best_h)
    
    return list(best_h), final_c5, n_points
