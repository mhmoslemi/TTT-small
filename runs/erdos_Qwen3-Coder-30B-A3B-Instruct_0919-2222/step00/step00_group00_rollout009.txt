import time
import numpy as np
import cvxpy as cp
from cvxpy import ECOS_BB, CBC
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_c5 = float('inf')
    best_h_vals = None
    best_n = None

    # Discretization sizes to try
    n_values = [80, 120, 160]
    n_values = [n for n in n_values if n <= 180]  # Avoid very large models
    
    # Helper to compute full overlap with zero padding
    def full_overlap(h):
        # Length of the full correlation
        full_len = 2*len(h) - 1
        corr = np.correlate(h, 1-h, mode='full')
        return corr

    # Tolerance for lazy constraint detection
    tol = 1e-6
    
    for n in n_values:
        if time.time() - t_start > budget_s - 10:
            break
            
        dx = 2.0 / n
        K0 = max(1, min(n//4, 30))  # Initial set of shifts
        K_active = list(range(1, K0 + 1))  # Active shifts to consider
        K_active = sorted(K_active)
        K_active_set = set(K_active)

        # Global variables
        x = cp.Variable(n, boolean=True)
        t = cp.Variable()

        # Symmetry breaking: first element is 1, enforced by solver
        # But we explicitly do this in the MILP setup
        
        # Mass constraint for exact half
        mass_cons = cp.constraints.eq(cp.sum(x), n // 2)
        
        # Variables for each shift's auxiliary product terms
        y_vars = {}
        constraint_list = []

        # Build initial constraints for first set of k
        y_list = []
        for k in K_active:
            y_shift = cp.Variable(n - k, boolean=True)
            y_vars[k] = y_shift
            y_list.append(y_shift)
            
            # McCormick constraints
            for i in range(n - k):
                # y[i,k] <= x[i]
                c1 = y_shift[i] <= x[i]
                # y[i,k] <= x[i+k]
                c2 = y_shift[i] <= x[i+k]
                # y[i,k] >= x[i] + x[i+k] - 1
                c3 = y_shift[i] >= x[i] + x[i+k] - 1
                constraint_list.extend([c1, c2, c3])
            
            # Overlap bound constraint
            c_k = cp.sum(x) - cp.sum(y_shift)
            c_bound = c_k <= t
            constraint_list.append(c_bound)
            
        # Symmetry break: x[0] = 1 (this will be done in model)
        sym_cons = [x[0] == 1]
        
        # Non-increasing order for first half
        # x[0] >= x[1] >= x[2] >= ... >= x[n//2 - 1]
        for i in range(n // 2 - 1):
            sym_cons.append(x[i] >= x[i+1])
            
        # Full model to be built:
        # obj: minimize t
        # constraints: explicit mass, symmetry, shifts, McCormick
        
        # Solve iteratively with lazy cuts
        max_iter = 30  # Prevent infinite loop if stuck
        best_t_this_n = float('inf')
        best_x_this_n = None
        iteration = 0

        while iteration < max_iter:
            iter_start = time.time()
            if iter_start - t_start > budget_s - 5:
                break

            # Rebuild constraints for current K_active
            cons = list(constraint_list)
            cons.extend([mass_cons])
            cons.extend(sym_cons)

            # Create the problem
            prob = cp.Problem(cp.Minimize(t), cons)
            
            try:
                prob.solve(solver=CBC, verbose=False, max_seconds=30)
            except Exception:
                try:
                    prob.solve(solver=ECOS_BB, verbose=False, max_iter=5000)
                except Exception:
                    break

            if prob.status not in ["optimal", "optimal_inaccurate"]:
                break

            t_opt = t.value
            x_opt = x.value
            if x_opt is None:
                break
            x_opt = np.round(x_opt).astype(int).tolist()
            
            # Evaluate all shifts to see if we have violation
            h_opt = np.array(x_opt)
            full_corr = full_overlap(h_opt)
            max_corr = np.max(full_corr)
            max_shift_idx = np.argmax(full_corr)
            
            max_shift = max_shift_idx - (n-1)  # convert back to shift indices
            if max_shift < 0:
                max_shift = -max_shift
            max_corr_val = max_corr
            
            # Reject the solution if infeasible
            good_solution = True
            if max_corr_val > t_opt + tol:
                good_solution = False
                # Add new shift to active set
                if max_shift not in K_active_set:
                    K_active.append(max_shift)
                    K_active = sorted(K_active)  # Keep sorted for stable indexing
                    K_active_set.add(max_shift)
                    
                    # Add new y and constraints
                    y_new = cp.Variable(n - max_shift, boolean=True)
                    y_vars[max_shift] = y_new
                    
                    for i in range(n - max_shift):
                        c1 = y_new[i] <= x[i]
                        c2 = y_new[i] <= x[i + max_shift]
                        c3 = y_new[i] >= x[i] + x[i + max_shift] - 1
                        constraint_list.extend([c1, c2, c3])
                    
                    # Overlap constraint for k = max_shift
                    c_k_new = cp.sum(x) - cp.sum(y_new)
                    c_bound_new = c_k_new <= t
                    constraint_list.append(c_bound_new)
            
            if good_solution and t_opt < best_t_this_n:
                best_t_this_n = t_opt
                best_x_this_n = x_opt
            
            iteration += 1
            
            # One extra small optimization after convergence
            if not good_solution or iteration > 10:
                break

        if best_x_this_n is not None:
            # Post-process with deterministic swaps to reduce C5 bound
            h = np.array(best_x_this_n)
            corr_full = full_overlap(h)
            max_over = np.max(corr_full)

            # Try a few simple local swaps to reduce max overlap
            # Create indices of 1s and 0s
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            # Try improving swaps
            swaps_made = 0
            while True:
                if swaps_made > 100:
                    break
                swap_made = False
                k_best = np.argmax(corr_full)
                if k_best == 0:
                    k_best = 0  # fixed-shift
                support_shift_k = np.where((h[:n-k_best] == 1) & (h[k_best:] == 0))[0]
                conflict_shift_k = np.where((h[:n-k_best] == 0) & (h[k_best:] == 1))[0]
                
                if len(support_shift_k) > 0 and len(conflict_shift_k) > 0:
                    # Pick first element of each
                    i = support_shift_k[0]
                    j = conflict_shift_k[0]
                    # Perform the swap
                    h[i] = 0  # remove one
                    h[j] = 1  # add one
                    corr_full = full_overlap(h)
                    max_over = np.max(corr_full)
                    swaps_made += 1
                    swap_made = True
                else:
                    break

            # Final evaluation
            corr_full = full_overlap(h)
            max_over_final = np.max(corr_full)
            c5_bound = max_over_final * dx
            
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h_vals = [float(val) for val in h]
                best_n = n
                
    if best_h_vals is None:
        # Fallback to initial construction
        from initial_h_values import h0
        n_init = len(h0)
        dx = 2.0 / n_init
        h = np.array(h0)
        corr_full = full_overlap(h)
        max_over = np.max(corr_full)
        c5_bound = max_over * dx
        return (h0, c5_bound, n_init)
    
    return (best_h_vals, best_c5, best_n)
