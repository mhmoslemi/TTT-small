import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős minimum overlap problem constant C5 for given h."""
    h = np.array(h_values)
    # Compute correlation using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum and scale by dx
    dx = 2.0 / len(h)
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 501  # Odd number to have a center
    dx = 2.0 / n_points
    
    # Start time tracking
    start_time = time.time()
    end_time = start_time + budget_s * 0.99  # Leave margin for final processing
    
    # Define shape functions
    def triangle_shape(x):
        return np.maximum(0, 1 - np.abs(x - 1))
    
    def cosine_shape(x, p=2):
        return np.clip(np.cos(np.pi / 2 * (x - 1)) ** p, 0, 1)
    
    def polynomial_shape(x, a=2):
        return np.clip(1 - np.abs(x - 1) ** a, 0, 1)
    
    # Try different initial shapes
    best_h = None
    best_C5 = float('inf')
    
    # Try triangle shape
    x = np.linspace(0, 2, n_points, endpoint=False) + dx / 2
    h_tri = triangle_shape(x)
    h_tri = h_tri * (n_points / 2) / np.sum(h_tri)  # normalize to sum = n_points / 2
    h_tri = np.clip(h_tri, 0, 1)
    C5_tri = evaluate_erdos_solution(h_tri)
    if C5_tri < best_C5:
        best_C5 = C5_tri
        best_h = h_tri.copy()
    
    # Try cosine shapes
    for p in [0.5, 1, 2, 4, 8]:
        h_cos = cosine_shape(x, p)
        h_cos = h_cos * (n_points / 2) / np.sum(h_cos)  # normalize to sum = n_points / 2
        h_cos = np.clip(h_cos, 0, 1)
        C5_cos = evaluate_erdos_solution(h_cos)
        if C5_cos < best_C5:
            best_C5 = C5_cos
            best_h = h_cos.copy()
    
    # Try polynomial shapes
    for a in [1, 2, 3]:
        h_poly = polynomial_shape(x, a)
        h_poly = h_poly * (n_points / 2) / np.sum(h_poly)  # normalize to sum = n_points / 2
        h_poly = np.clip(h_poly, 0, 1)
        C5_poly = evaluate_erdos_solution(h_poly)
        if C5_poly < best_C5:
            best_C5 = C5_poly
            best_h = h_poly.copy()
    
    # Start hill climbing from the best shape found
    h = best_h.copy()
    current_C5 = best_C5
    
    # Parameters for hill climbing
    max_iterations = 200000  # Adjust based on remaining time
    iteration = 0
    epsilon = 0.05  # Initial epsilon
    min_epsilon = 1e-5
    
    # Precompute indices for efficiency in computation
    indices = list(range(n_points))
    
    # Main hill climbing loop
    while elapsed_time := (time.time() - start_time) < (budget_s * 0.95):
        iteration += 1
        
        # Choose two indices at random
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        if i == j:
            continue
            
        # Determine the max allowable delta to keep both values in [0, 1]
        delta_max = min(h[i], 1 - h[j], epsilon)
        
        if delta_max <= 0:
            continue
            
        # Propose delta
        delta = np.random.uniform(0, delta_max)
        
        # Compute new values
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(h_new)
        
        if new_C5 < current_C5:
            h = h_new
            current_C5 = new_C5
        else:
            # Occasionally accept worse solutions (simulated annealing)
            if np.random.rand() < np.exp(-(new_C5 - current_C5) / (0.01 * (0.99 ** iteration))):
                h = h_new
                current_C5 = new_C5
                
        # Gradually reduce epsilon
        epsilon = max(min_epsilon, epsilon * 0.99999)
    
    # Final check of C5
    final_C5 = evaluate_erdos_solution(h)
    
    return (h.tolist(), final_C5, n_points)
