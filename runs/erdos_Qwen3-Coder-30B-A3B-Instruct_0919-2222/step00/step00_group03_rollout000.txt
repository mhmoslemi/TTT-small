import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def solve_erdos_milp(n, time_limit):
    prob = pulp.LpProblem("ErdosMinOverlapMILP", pulp.LpMinimize)
    
    # Variables: h_i binary, T continuous
    h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
    T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
    
    # Constraint sum h_i = n/2
    prob += pulp.lpSum([h[i] for i in range(n)]) == n/2

    # Product constraints and shift constraints
    for d in range(n):  # Shifts from 0 to n-1
        # Create list of y variables for this shift
        y_vars = []
        for i in range(n-d):
            y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
            y_vars.append(y)
            # McCormick envelope constraints
            prob += y <= h[i]
            prob += y <= h[i+d]
            prob += y >= h[i] + h[i+d] - 1
        # Shift constraint: sum_y >= n/2 - T
        prob += pulp.lpSum(y_vars) >= n/2 - T
    
    # Objective: minimize T
    prob += T
    
    # Solve
    solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit)
    prob.solve(solver)
    
    # Check solution status
    status = pulp.LpStatus[prob.status]
    if status not in ['Optimal', 'Not Solved', 'Feasible']:
        return None, None
    
    # Collect solution
    h_vals = np.array([float(pulp.value(h[i])) for i in range(n)])
    t_val = pulp.value(T)
    
    return h_vals, t_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Try n values in descending order
    n_values = [120, 100, 150, 80, 200]
    
    for n in n_values:
        if time.time() - start_time >= budget_s - 5:
            break
            
        time_left = budget_s - (time.time() - start_time)
        if time_left < 10:
            break
            
        # Solve MILP for this n
        h_vals, t_val = solve_erdos_milp(n, min(300, time_left - 5))
        
        if h_vals is not None:
            # Compute actual C5 bound
            c5_bound = evaluate_erdos_solution(h_vals)
            
            if c5_bound < best_c5:
                best_h = h_vals.copy()
                best_c5 = c5_bound
                best_n = n
                
    # If we could not produce a solution, use initial
    if best_h is None:
        h_vals = initial_h_values.copy()
        c5_bound = evaluate_erdos_solution(h_vals)
        best_h = h_vals
        best_c5 = c5_bound
        best_n = len(h_vals)

    return best_h, best_c5, best_n
