from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 200 points for good resolution within budget
    n_points = 200
    dx = 2.0 / n_points

    if len(initial_h) != n_points:

        initial_h = np.interp(x_new, x_orig, initial_h)
    
    def evaluate_c5(h):
        # Convert to numpy array
        h = np.array(h)
        # Compute convolution of h with (1-h) using full mode
        conv = np.correlate(h, 1 - h, mode='full')
        # Normalize by dx
        conv *= dx
        # Return maximum value
        return np.max(conv)
    
    # Objective function to minimize
    def objective(h):
        return evaluate_c5(h)
        
    # Bounds for each h_i: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) * dx = 1, which means sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Use differential evolution for global search
    # Set up constraint dictionary
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Track time
    start_time = time.time()
    
    # Run optimization with time limit
    best_result = None
    best_c5 = float('inf')
    
    while time.time() - start_time < budget_s - 5:  # Leave 5s for final evaluation
        try:
            # Perform differential evolution optimization
            result = differential_evolution(
                objective, 
                bounds, 
                constraints=constraints,
                seed=seed,
                maxiter=100,  # Reduce iterations for performance
                popsize=15,   # Keep population size reasonable
                mutation=(0.5, 1),
                recombination=0.7,
                disp=False,
                polish=True
            )

                current_c5 = evaluate_c5(result.x)

    current_sum = np.sum(final_h) * dx
