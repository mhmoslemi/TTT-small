import numpy as np
import time
from scipy.special import logsumexp

def evaluate(h):
    """Evaluate the true maximum overlap C5 using numpy correlate."""
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr)

def project_to_capped_simplex(v, M):
    """Project vector v onto the capped simplex: 0 <= h_i <= 1, sum h_i = M."""
    u = np.clip(v, 0, 1)
    s = np.sum(u)
    if abs(s - M) < 1e-10:
        return u
    # Sort v descending to find threshold
    sort_idx = np.argsort(-v)
    sorted_v = v[sort_idx]
    sorted_u = u[sort_idx]
    # Sweep to find threshold theta
    cumsum = 0.0
    for i in range(len(sorted_v)):
        cumsum += sorted_u[i]
        if i == len(sorted_v) - 1 or cumsum + (sorted_v[i+1] - sorted_v[i]) * (i+1) > M:
            theta = (cumsum - M) / (i+1)
            h = np.zeros_like(v)
            h[sort_idx] = np.clip(sorted_v - theta, 0, 1)
            return h

def precompute_shifts(n):
    """Precompute valid indices for each shift s."""
    shift_maps = {}
    for s in range(-n+1, n):
        valid_pairs = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                valid_pairs.append((i, j))
        shift_maps[s] = np.array(valid_pairs)
    return shift_maps

def correlations(h, shift_maps):
    """Compute all correlation values C_s for integer shifts s."""
    n = len(h)
    C = np.zeros(2*n - 1)
    for s, pairs in shift_maps.items():
        if len(pairs) == 0:
            continue
        # Compute sum h_i * (1 - h_{i+s}) for all valid i
        i_vals = pairs[:, 0]
        j_vals = pairs[:, 1]
        C[s + n - 1] = np.sum(h[i_vals] * (1 - h[j_vals]))
    return C

def smooth_obj_and_grad(h, alpha, shift_maps):
    """Compute smooth surrogate Φ_α and its gradients."""
    C = correlations(h, shift_maps)
    max_C = np.max(C)
    C_centered = C - max_C
    log_sum_exp = logsumexp(alpha * C_centered)
    Phi_alpha = log_sum_exp / alpha
    # Compute weights w_s = exp(α*(C_s - max_C)) / sum exp(...)
    w = np.exp(alpha * C_centered) / np.exp(log_sum_exp)  # * exp(max_C) / sum = w_s normalized
    # Compute gradients ∂Phi_α / ∂h_i = sum_s w_s (1 - h_{i+s} - h_{i-s})
    grad = np.zeros_like(h)
    for s, pairs in shift_maps.items():
        if len(pairs) == 0:
            continue
        i_vals = pairs[:, 0]
        j_vals = pairs[:, 1]
        if s >= 0:
            # effect of h_{i+s} terms
            grad[i_vals] += w[s + len(h) - 1] * (-h[j_vals])
        else:
            # effect of h_{i+s} where i+s < 0; we don't have h_{i+s} so ignore it
            grad[i_vals] += w[s + len(h) - 1] * (-h[j_vals])
        # Add 1 to each term due to term (1 - h_{i+s}) or (1 - h_{i-s})
        grad[i_vals] += w[s + len(h) - 1]
    return Phi_alpha, grad

def run_descent(h0, alpha_schedule, eta0, max_iter, shift_maps, time_limit):
    """Run projected gradient descent with continuation on alpha."""
    h = np.copy(h0)
    t0 = time.time()
    eta = eta0
    
    for alpha in alpha_schedule:
        if time.time() - t0 > time_limit:
            break
        for _ in range(max_iter):
            if time.time() - t0 > time_limit:
                break
            # Compute smooth function and gradient
            val, grad = smooth_obj_and_grad(h, alpha, shift_maps)
            # Gradient step
            v = h - eta * grad
            # Project
            h = project_to_capped_simplex(v, len(h)/2)
            # Adaptive step-size based on actual improvement
            current_max = evaluate(h)
            if current_max < 0.3807:  # early stop condition
                return h, current_max
    return h, evaluate(h)

def random_binary(n, M):
    """Generate a random binary vector with exactly M ones."""
    idx = np.random.choice(n, M, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def random_fractional(n, M):
    """Generate a random fractional vector (Dirichlet) and project to capped simplex."""
    alpha_vec = np.ones(n)
    h = np.random.dirichlet(alpha_vec)
    return project_to_capped_simplex(h, M)

def swap_local_search(h_bin, max_iter, time_limit):
    """Perform targeted swap improvements on a binary vector."""
    t0 = time.time()
    best_h = np.copy(h_bin)
    best_c5 = evaluate(best_h)
    if best_c5 < 0.3807:
        return best_h, best_c5
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    worst_s = np.argmax(corr)
    for _ in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        # Find a suitable swap
        # Find i such that h_i = 1 and h_{i+worst_s} = 0
        # Find j such that h_j = 0 and h_{j+worst_s} = 1
        np.random.shuffle(np.arange(len(best_h)))
        for i in range(len(best_h)):
            if best_h[i] != 1:
                continue
            j = i - worst_s  # pos of h_{i+worst_s} in array, specifically looking for that element
            if 0 <= j < len(best_h) and best_h[j] == 0:
                k = (i + worst_s) % len(best_h)
                if 0 <= k < len(best_h) and best_h[k] == 1:
                    # Check if the value is actually 0 at pos j (or else it's out range > n-1 or < 0)
                    j1 = i - worst_s
                    if 0 <= j1 < len(best_h) and best_h[j1] == 0:
                        # Do the swap: h[i], h[j] = h[j], h[i]
                        temp_h = np.copy(best_h)
                        temp_h[i] = 0
                        temp_h[j] = 1
                        c5_new = evaluate(temp_h)
                        if c5_new < best_c5:
                            best_h = temp_h
                            best_c5 = c5_new
                            # Update worst shift
                            corr = np.correlate(best_h, 1 - best_h, mode='full')
                            worst_s = np.argmax(corr)
                continue
        # Recompute the maximum correlation shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_s = np.argmax(corr)
        if best_c5 < 0.3807:
            break
    return best_h, best_c5
