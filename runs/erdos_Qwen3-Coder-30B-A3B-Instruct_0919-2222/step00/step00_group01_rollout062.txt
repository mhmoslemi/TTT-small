```python
import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap problem constant C5 for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation with 1-h using 'full' mode
    correlation = correlate(h, 1 - h, mode="full")
    # The maximum correlation value scaled by dx
    return np.max(correlation) * dx

def initial_h_values():
    """Return initial construction for the optimization."""
    # Create an initial step function with sum = n/2
    n_points = 47
    h = np.zeros(n_points)
    # Set first half to 1, second half to 0
    h[:n_points//2] = 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Initialize
    np.random.seed(seed)
    start_time = time.time()
    end_time = start_time + budget_s * 0.99  # Leave margin
    
    # Initial setup
    n_points = 151  # Odd number for better centering
    dx = 2.0 / n_points
    h = np.ones(n_points) * 0.5  # Uniform initial condition
    
    # Evaluate initial solution
    current_c5 = evaluate_erdos_solution(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Types of shapes to try first to find a good starting point
    shape_functions = [
        lambda x: np.maximum(0, 1 - np.abs(x - 1)),  # triangular
        lambda x: np.cos(np.pi/2 * (x - 1))**2,     # cosine squared
        lambda x: np.maximum(0, 1 - np.abs(x - 1)**2), # quadratic
        lambda x: np.cos(np.pi/2 * (x - 1))**4,     # cosine^4
    ]

    # Try these shapes
    for i, shape_func in enumerate(shape_functions):
        x = (np.arange(n_points) + 0.5) * dx
        h_candidate = shape_func(x)
        
        # Normalize to sum = n/2
        sum_h = np.sum(h_candidate)
        if not np.isclose(sum_h, 0):
            scale = (n_points / 2) / sum_h
            h_candidate *= scale
            
        # Clip to [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        c5_candidate = evaluate_erdos_solution(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
    
    h = best_h.copy()
    current_c5 = best_c5
    
    # Local search loop
    iteration = 0
    max_iterations = 100000
    epsilon = 0.02
    epsilon_min = 1e-6
    decay_rate = 0.9999
    temp = 0.01  # initial temperature
    
    while iteration < max_iterations and time.time() < end_time:
        # Random index pair
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        if i == j:
            j = (j + 1) % n_points
            
        # Compute max allowable delta
        max_delta = min(h[i], 1 - h[j], epsilon)
        if max_delta <= 0:
            iteration += 1
            continue
            
        # propose change
        delta = np.random.rand() * max_delta
        
        # new h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject the move
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        elif np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
            h = h_new
            current_c5 = c5_new
        else:
            pass  # Reject
        
        # update epsilon and temp
        epsilon = max(epsilon_min, epsilon * decay_rate)
        temp *= 0.99999
        
        iteration += 1
        
    return (list(best_h), best_c5, n_points)
```
