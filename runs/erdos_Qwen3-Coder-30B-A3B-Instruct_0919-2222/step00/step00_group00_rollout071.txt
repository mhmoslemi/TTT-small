```python
import numpy as np
import time
import random

# Predefined initial solution
initial_h_values = None

def compute_correlation(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap(h, dx):
    c = compute_correlation(h)
    return c.max() * dx

def apply_delta(c, h, idx, delta):
    n = len(h)
    offset = n - 1
    # First term contribution (h[idx] as first factor)
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # Second term contribution (h[idx] as second factor)
    for j in range(0, idx + 1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(i, j, c, h):
    delta_i = 0 - h[i]
    delta_j = 1 - h[j]
    
    # Apply first delta
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    # Apply second delta
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def targeted_swap(c, h):
    n = len(h)
    offset = n - 1
    max_idx = np.argmax(c)
    max_shift = max_idx - offset
    if max_shift == 0:  # Skip shift 0 as it gives 0
        return None, None
    
    # Find candidate indices for swap
    candidates_i = []
    candidates_j = []
    
    for i in range(n):
        if h[i] == 1:
            j = i + max_shift
            if 0 <= j < n and h[j] == 0:
                candidates_i.append(i)
        elif h[i] == 0:
            j = i + max_shift
            if 0 <= j < n and h[j] == 1:
                candidates_j.append(i)
    
    if not candidates_i or not candidates_j:
        return None, None
        
    i = random.choice(candidates_i)
    j = random.choice(candidates_j)
    return i, j

def hill_climb(h, n_iter, time_left, dx):
    n = len(h)
    c = compute_correlation(h)
    best_val = max_overlap(h, dx)
    best_h = h.copy()
    
    # Precompute sets for faster swaps
    ones = [i for i in range(n) if h[i] == 1]
    zeros = [i for i in range(n) if h[i] == 0]
    
    start_time = time.time()
    for iteration in range(n_iter):
        if time.time() - start_time > time_left:
            break
            
        # Try targeted swap first
        i, j = targeted_swap(c, h)
        
        if i is not None and j is not None:
            swap(i, j, c, h)
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            # Fallback to random swap
            if len(ones) > 0 and len(zeros) > 0:
                i = random.choice(ones)
                j = random.choice(zeros)
                swap(i, j, c, h)
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
        
        current_val = max_overlap(h, dx)
        if current_val < best_val:
            best_val = current_val
            best_h = h.copy()
        
        if current_val < 0.379:
            break
    
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Use 120 sample points as it's a good balance between resolution and speed
    n_points = 120
    dx = 2.0 / n_points
    
    # Get initial solution if available
    h_initial = None
    if initial_h_values is not None:
        h_initial = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(initial_h_values)), 
                              initial_h_values)
        # Clip to [0,1]
        h_initial = np.clip(h_initial, 0, 1)
        # Normalize sum to n/2
        current_sum = np.sum(h_initial)
        h_initial = h_initial * (n_points / 2) / current_sum
    else:
        # Generate binary starting solution
        h_initial = np.zeros(n_points)
        num_ones = n_points // 2
        ones_idx = np.random.choice(n_points, num_ones, replace=False)
        h_initial[ones_idx] = 1
    
    # Ensure sum exactly to n/2
    current_sum = np.sum(h_initial)
    diff = (n_points / 2) - current_sum
    if abs(diff) > 0.001:
        # Distribute adjustment to nearest points
        for i in range(n_points):
            if diff > 0 and h_initial[i] < 1:
                adjustment = min(diff, 1 - h_initial[i])
                h_initial[i] += adjustment
                diff -= adjustment
                if diff <= 0:
                    break
            elif diff < 0 and h_initial[i] > 0:
                adjustment = min(-diff, h_initial[i])
                h_initial[i] -= adjustment
                diff += adjustment
                if diff >= 0:
                    break
    
    # Initialize and train best solution
    best_h = h_initial.copy()
    best_c5 = max_overlap(h_initial, dx)
    
    # Perform multiple rounds of hill climbing with time budget
    remaining_budget = budget_s - (time.time() - start_time)
    if remaining_budget <= 10:
        return list(h_initial), best_c5, n_points
    
    # Run hill climbing with iterations based on remaining time
    n_iter = min(100000, int(remaining_budget * 1000))
    
    # Try to improve
    time_left = budget_s - (time.time() - start_time)
    if time_left > 2:
        try:
            improved_h, improved_c5 = hill_climb(best_h, n_iter, time_left, dx)
            if improved_c5 < best_c5:
                best_h = improved_h
                best_c5 = improved_c5
        except:
            pass
    
    return list(best_h), best_c5, n_points
```
