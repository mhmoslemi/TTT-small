```python
import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is our C5 bound
    c5_bound = np.max(corr) * dx
    return c5_bound

def initial_h_values():
    """Initial construction from the problem description"""
    n = 45
    h = np.zeros(n)
    # Simple construction
    h[:n//2] = 1.0
    h[n//2:] = 0.0
    return h

def objective_function(h_flat):
    """Objective function to minimize"""
    n_points = len(h_flat)
    h = np.clip(h_flat, 0, 1)
    # Normalize to make integral = 1
    h = h * (n_points / (2 * np.sum(h)))
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # start with a moderate size
    dx = 2.0 / n_points
    h = initial_h_values()
    
    # Initial state
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimize using differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # The optimization loop - we try to improve the solution
    while time.time() - start_time < budget_s - 1:
        try:
            # Use differential evolution with restarts
            result = differential_evolution(
                objective_function,
                bounds,
                maxiter=50,
                popsize=15,
                atol=1e-6,
                seed=seed,
                disp=False
            )
            
            # Apply constraints: h should be in [0,1], sum should be n_points/2
            h_new = np.clip(result.x, 0, 1)
            h_new = h_new * (n_points / (2 * np.sum(h_new)))
            
            c5_new = evaluate_erdos_solution(h_new)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
            # Perturb slightly for next iteration
            if len(h_new) >= 5:
                h_new = h_new + np.random.normal(0, 0.02, len(h_new))
            
            # Restart the optimization with new starting guess
            if time.time() - start_time > budget_s - 5:
                break
                
        except Exception:
            break
            
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
```
