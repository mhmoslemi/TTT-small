import numpy as np

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h_values"""
    h = np.array(h_values, dtype=float)
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation: correlation of h with (1-h)
    # np.correlate with mode='full' gives length 2*n-1
    correlation = np.correlate(h, 1 - h, mode='full')
    # Take maximum value (excluding the middle which is for shift 0)
    max_corr = np.max(correlation)
    # The actual bound C5 = max_corr * dx
    return max_corr * dx

def project_to_simplex(v, target_sum):
    """Project vector v onto the simplex with sum = target_sum and bounds [0,1]"""
    n = len(v)
    # Sort in descending order
    sorted_indices = np.argsort(v)[::-1]
    sorted_v = v[sorted_indices]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_v)
    
    # Find the threshold
    idx = -1
    for i in range(n):
        if sorted_v[i] - (target_sum - cumsum[i]) / (i + 1) >= 0:
            idx = i
        else:
            break
    
    if idx == -1:
        # All elements are too large
        lambda_val = target_sum / n
        projected = np.maximum(np.minimum(v, 1), 0)
        # Adjust to maintain the sum
        adjustment = (target_sum - np.sum(projected)) / n
        projected += adjustment
    else:
        # Compute threshold lambda
        lambda_val = (target_sum - cumsum[idx]) / (idx + 1)
        projected = np.maximum(np.minimum(v - lambda_val, 1), 0)
    
    # Return in original order
    result = np.zeros_like(v)
    result[sorted_indices] = projected
    
    # Adjust final sum to target
    adjustment = (target_sum - np.sum(result)) / n
    result += adjustment
    
    # Final check for bounds and sum
    result = np.maximum(np.minimum(result, 1), 0)
    adjustment = (target_sum - np.sum(result)) / n
    result += adjustment
    result = np.clip(result, 0, 1)
    
    # Ensure sum matches by small adjustment
    current_sum = np.sum(result)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        # Distribute difference to all elements equally
        result += diff / n
    
    return np.clip(result, 0, 1)

def compute_gradient(h, max_indices, dx):

    n = len(h)
    g = np.zeros(n)
    
    # For each worst shift, accumulate contributions
    for k in max_indices:
        if k >= 0:
            # For lag k: term h[i] * (1 - h[i+k]) 
            # gradient w.r.t. h[i] = (1 - h[i+k]) if i+k < n 
            # and -h[i-k] if i-k >= 0
            
            # Compute slice contributions using array operations
            # For i in [0, n-1-k], add (1 - h[i+k]) to g[i]
            if n - 1 - k >= 0:
                g[:n-k] += 1 - h[k:]
            
            # For i in [k, n-1], subtract h[i-k] from g[i]
            # This is equivalent to adding -h[i-k] for i in [k, n-1]
            if k < n:
                g[k:] -= h[:n-k]
        else:
            # Negative shift k : handle as positive lag shift
            # If k is negative, shift is by -k = -k (so it's shift of abs(k) in positive direction)
            # So h[i] * (1 - h[i-k]) where we've shifted by |k| to the right
            # gradient: (1 - h[i-k]) to g[i] if i-k < n
            # and -h[i+k] to g[i] if i+k >=0
            
            abs_k = -k
            # g[i] += (1 - h[i-abs_k]) for i in [0, n-1+abs_k] but within valid range
            # effectively for i in [abs_k, n-1] we add -h[i-abs_k]
            # for i in [0, n-1-abs_k] we add 1 - h[i+abs_k]
            
            # i in [0, n-1-abs_k] range: 
            if n - 1 - abs_k >= 0:
                # g[0 : n-1-abs_k] += 1 - h[abs_k : n-1]
                g[:n-abs_k] += 1 - h[abs_k:]
                
            # i in [abs_k, n-1] range:
            if abs_k < n:
                # g[abs_k : n] -= h[: n-abs_k]
                g[abs_k:] -= h[:n-abs_k]
    
    return g

def subgradient_descent_step(h, dx, t_max, max_indices, step_size):
    """Take one gradient descent step"""
    # Compute gradient for the constraint that corresponds to maximum
    g = compute_gradient(h, max_indices, dx)
    
    # Take step
    h_new = h - step_size * g
    
    # Project to feasible set
    h_new = project_to_simplex(h_new, len(h) / 2)
    
    return h_new
