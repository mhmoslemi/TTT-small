import numpy as np
import time
import random

# Global initial_h_values is pre-imported
# We'll use a heuristic approach with various n values

def compute_correlation(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap(h):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h)
    return np.max(c) * dx

def apply_delta(c, h, idx, delta):
    n = len(h)
    offset = n - 1
    # first term: for j from idx to n-1:
    for j in range(idx, n):
        shift = j - idx
        c[shift + offset] += delta * (1 - h[j])
    # second term: for j from 0 to idx:
    for j in range(0, idx+1):
        shift = idx - j
        c[shift + offset] -= delta * h[j]

def swap(i, j, c, h):
    delta_i = 0 - h[i]
    delta_j = 1 - h[j]
    
    apply_delta(c, h, i, delta_i)
    h[i] = 0
    apply_delta(c, h, j, delta_j)
    h[j] = 1

def targeted_swap(c, h):
    n = len(h)
    offset = n - 1
    max_idx = np.argmax(c)
    max_shift = max_idx - offset
    
    # Find candidates for swapping to reduce most influential shift
    A = []
    B = []
    
    for i in range(n):
        j = i + max_shift
        if 0 <= j < n:
            if h[i] == 1 and h[j] == 0:
                A.append(i)
            if h[i] == 0 and h[j] == 1:
                B.append(i)
    
    if A and B:
        i = random.choice(A)
        j = random.choice(B)
        return (i, j)
    else:
        return None

def hill_climb(h, n_iter, time_left, initial_time):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h)
    current_max = np.max(c) * dx
    best_h = h.copy()
    best_max = current_max
    
    # Set of indices with 1s and 0s for faster sampling
    ones = [i for i in range(n) if h[i] == 1]
    zeros = [i for i in range(n) if h[i] == 0]
    
    iter_count = 0
    while iter_count < n_iter and (time.time() - initial_time) < time_left - 2:
        iter_count += 1
        
        # Try targeted swap first
        swap_pair = targeted_swap(c, h)
        if swap_pair is not None:
            i, j = swap_pair
        else:
            # Fallback to random swap
            if len(ones) > 0 and len(zeros) > 0:
                i = random.choice(ones)
                j = random.choice(zeros)
            else:
                continue
        
        # Perform the swap and update
        old_c = c.copy()
        old_h = h.copy()
        old_val = current_max
        
        # Apply swap
        delta_i = 0 - h[i]
        delta_j = 1 - h[j]
        apply_delta(c, h, i, delta_i)
        h[i] = 0
        apply_delta(c, h, j, delta_j)
        h[j] = 1
        
        # Check new maximum
        current_max = np.max(c) * dx
        
        if current_max < best_max:
            best_h = h.copy()
            best_max = current_max
        else:
            # Revert if no improvement
            c[:] = old_c
            h[:] = old_h
            current_max = old_val
            
        # Update sets
        ones = [k for k in range(n) if h[k] == 1]
        zeros = [k for k in range(n) if h[k] == 0]
        
    return best_h, best_max
