import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import math
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Extract the maximum value among all lags
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    # Create a piecewise constant function with 70 points
    n = 70
    dx = 2.0 / n
    h = np.zeros(n)
    # Use a simple construction inspired by optimization
    # We'll try a two-segment step function
    # First segment: from 0 to 0.5 with value 1.5
    # Second segment: from 0.5 to 2 with value 0.25
    # This will make sum(h) * dx = 1
    # Let's compute exact values to satisfy the constraint
    
    # First approach: a symmetric two-step function
    mid = 1.0
    h[:int(n/2)] = 1.2
    h[int(n/2):] = 0.4
    # Normalize to match integral
    integral = np.sum(h) * dx
    h = h / integral * 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Objective function to minimize
    def objective(h):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Compute C5
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds for the optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint function: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Define constraint for scipy.optimize
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Use differential evolution to get a good initial guess
    print("Using differential evolution to find a good starting point...")
    de_result = differential_evolution(objective, bounds, constraints=cons, 
                                      maxiter=50, seed=seed, popsize=15)
    if de_result.success:
        h0 = de_result.x
    
    # Refine with L-BFGS-B
    print("Refining with L-BFGS-B...", end=' ')
    result = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                      options={'maxiter': 1000, 'ftol': 1e-9})
    if result.success:
        h_final = result.x
    else:
        h_final = h0  
    
    # Ensure bounds
    h_final = np.clip(h_final, 0, 1)
    
    # Evaluate the final result
    c5_bound = evaluate_erdos_solution(h_final)
    
    # Return final solution
    elapsed = time.time() - start_time
    print(f"\nCompleted in {elapsed:.2f}s. C5 = {c5_bound:.6f}")
    
    return (h_final, c5_bound, n_points)
