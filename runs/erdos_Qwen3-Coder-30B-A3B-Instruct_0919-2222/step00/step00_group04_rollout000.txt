def evaluate_h(h, dx=2.0 / len(h)):
    """Evaluate the overlap integral for the given step function."""
    # Use np.correlate with mode='full' to compute correlation
    # h is the step function values, 1-h is the complement
    # The result has length 2*len(h) - 1
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap corresponds to the maximum of this correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(v, target_sum):
    """Project vector v onto the capped simplex: sum(v) = target_sum, 0 <= v[i] <= 1."""
    n = len(v)
    # Sort v in descending order
    sorted_v = sorted(v, reverse=True)
    # Calculate cumulative sums
    cumsum = np.cumsum(sorted_v)
    # Find the threshold
    rho = np.argmax(sorted_v[:-1] + 1.0 / (np.arange(n-1) + 1.0) * (target_sum - cumsum[:-1])) + 1
    # Compute the threshold
    threshold = (target_sum - cumsum[rho-1]) / rho
    # Apply projection
    result = np.maximum(np.minimum(v - threshold, 1.0), 0.0)
    # Adjust to meet the exact sum
    adjustment = (target_sum - np.sum(result)) / n
    result += adjustment
    return np.clip(result, 0, 1)
