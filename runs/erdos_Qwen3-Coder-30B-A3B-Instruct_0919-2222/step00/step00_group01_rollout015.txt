from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the overlap integral for given h values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all correlations
    correlations = np.correlate(h, 1 - h, mode="full")
    # Maximum correlation scaled by dx
    max_corr = np.max(correlations)
    return max_corr * dx
