import numpy as np
import cvxpy as cp
import time
from math import ceil

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    best_n = None

    n_list = [80, 120, 160]

        if time.time() - t_start > budget_s - 5:
            break

        # Initial shift set: small shifts are usually critical
        K0 = min(ceil(n/4), 50)
        K = list(range(1, K0 + 1))
        
        # Lazy constraint loop
        max_iter = 10
        iter_count = 0
        while iter_count < max_iter:
            if time.time() - t_start > budget_s - 2:
                break
                
            # Build variables
            x = cp.Variable(n, boolean=True)
            t = cp.Variable()
            
            # Set up auxiliary variables y[i,k] for each k in K
            y_vars = {}
            constraints = []
            
            # Add mass constraint
            constraints.append(cp.sum(x) == n//2)
            
            # Symmetry breaking: first element is 1
            constraints.append(x[0] == 1)
            
            # Prefix non-increasing
            for i in range(n//2 - 1):
                constraints.append(x[i] >= x[i+1])
            
            # Add constraints for shifts in K
            for k in K:
                # y[i,k] = x[i] * x[i+k]
                # Define y_vars for this k
                y_vars[k] = cp.Variable(n-k, boolean=True)
                # McCormick constraints
                for i in range(n-k):
                    constraints.extend([
                        y_vars[k][i] <= x[i],
                        y_vars[k][i] <= x[i+k],
                        y_vars[k][i] >= x[i] + x[i+k] - 1
                    ])
                
                # Constraint for shift k: sum(x) - sum(y) <= t
                constraints.append(cp.sum(x[:n-k]) - cp.sum(y_vars[k]) <= t)
            
            # Objective
            objective = cp.Minimize(t)
            
            # Solve the problem
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=cp.CBC, verbose=False, max_seconds=60)
                if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:
                    x_sol = np.round(x.value).astype(int)

                            best_c5 = c5_bound
                            best_h = x_sol.astype(float).tolist()
                            best_n = n
                    else:
                        # Violation occurred, add that shift to K
                        max_idx = np.argmax(corr_full)
                        # Correction for full correlation indexing
                        k_violated = max_idx - n + 1
                        if k_violated not in K:
                            K.append(k_violated)
                        iter_count += 1
                        continue
                else:
                    # Not solved to optimality, try next n or skip if budget near
                    break
            except Exception:
                break  # Solver failed, break and try next n
            
            iter_count += 1

        if best_h is not None and time.time() - t_start < budget_s - 5:

            A = ones[np.where((ones + k_worst < best_n) & (h_vals[ones + k_worst] == 0))[0]]
            B = zeros[np.where((zeros + k_worst < best_n) & (h_vals[zeros + k_worst] == 1))[0]]

                j = B[0]
