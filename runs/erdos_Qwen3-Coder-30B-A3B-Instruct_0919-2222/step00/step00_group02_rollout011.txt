def evaluate_h(h):
    """Evaluate the maximum overlap integral for a given h."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode='full')
    max_O = O_counts.max()
    return max_O * dx

def max_overlap_counts(h):
    """Return the integer maximum overlap counts."""
    n = len(h)
    O_counts = np.correlate(h, 1 - h, mode='full')
    return O_counts.max()

def random_balanced_binary(n, rng):
    """Generate a random balanced binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    rng.shuffle(h)
    return h

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto box [0,1] and sum constraint using binary search."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def binary_anneal(n, time_limit, rng):
    """Perform simulated annealing on binary vector."""
    h = random_balanced_binary(n, rng)
    cur_val = max_overlap_counts(h)
    best_val = cur_val
    best_h = h.copy()
    start_time = time.time()
    
    # Precompute indices
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    max_iters = 100000
    for iter in range(max_iters):
        if time.time() - start_time > time_limit:
            break
            
        # Temperature schedule
        elapsed = time.time() - start_time
        T = 0.5 * (1 - elapsed / time_limit) + 0.001
        
        # Propose swap
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        # Swap
        h[p], h[q] = 0, 1
        new_val = max_overlap_counts(h)
        delta = new_val - cur_val
        
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
                
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # Revert
            h[p], h[q] = 1, 0
            
        # Update indices
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    return best_h, best_val

    max_iters = 5000

        if time.time() - start_time > time_limit:
            break
            
        O_counts = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(O_counts)
        shift = max_idx - (n - 1)
        
        # Compute gradient for this shift
        grad = np.zeros(n)
        if shift >= 0:
            left = np.zeros_like(h)
            right = np.zeros_like(h)
            left[shift:] = h[:-shift]
            right[:n-shift] = h[shift:]
            grad = 1 - left - right
        else:
            s = -shift
            left = np.zeros_like(h)
            right = np.zeros_like(h)
            left[:n-s] = h[s:]
            right[s:] = h[:-s]
            grad = 1 - left - right
            
        step = 0.1

        h_new = project_onto_box_and_sum(h_new, target_sum=n/2)
        
        obj = evaluate_h(h_new)
        if obj < best_obj:
            best_obj = obj
            best_h = h_new.copy()
            
        h = h_new
        
    return best_h, best_obj
