def evaluate_erdos_solution(h_values):
    """Evaluates the maximum overlap for given h_values"""
    n = len(h_values)
    dx = 2.0 / n
    # Full convolution of h with (1-h)
    conv = correlate(h_values, 1 - h_values, mode='full')
    # Take maximum over all shifts
    max_overlap = np.max(conv)
    return max_overlap * dx
