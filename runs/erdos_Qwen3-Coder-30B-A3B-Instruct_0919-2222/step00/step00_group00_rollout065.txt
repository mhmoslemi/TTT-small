import time
import numpy as np
from scipy import signal

def compute_correlation(h):
    # Return correlation c_k = sum_i h[i] * (1 - h[i+k]), for k = -(n-1)...(n-1)
    # Using correlate with mode='full' and weights 1-h, pivot to get shifts correctly
    n = len(h)
    h1 = 1 - h
    c = signal.correlate(h, h1, mode='full')
    # c has length 2*n-1; shift indices k = -(n-1) ... (n-1) 
    # c[i] corresponds to shift i-(n-1) = k
    return c

def log_sum_exp_stable(x, axis=None):
    # Compute log(sum(exp(x))) in a numerically stable way
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))

def smoothed_objective_and_weights(c, alpha):
    # c: array of length 2*n-1
    # Compute Φ_α = (1/α)*log Σ exp(α * c_k)
    # And weights w_k = exp(α * c_k) / Σ exp(α * c_k)
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    Z = np.sum(exp_c)
    weights = exp_c / Z
    value = (1/alpha) * (np.log(Z) + c_max)
    return value, weights

def compute_gradient_via_convolution(h, weights):
    # Gradient ∇Φ_α(h) = Σ_k w_k * ∇c_k(h)
    # Take: (w_conv 1-h) - (w_conv h) using np.convolve with appropriate offset.
    # w is length 2*n-1
    n = len(h)
    w = weights
    
    # Forward part of gradient: sum over k w_k * (1 - h[i+k]) for valid k
    # This is equivalent to conv(1-h, w_rev) but with offset
    # Compute by conv(w, 1-h) and extract relevant shift components
    forward_part = signal.convolve(np.flip(w), 1-h, mode='full') # [0:2n-1]
    forward_part = forward_part[n-1:] # corresponds to shift k=0 to n-1

    # Backward part: sum over k w_k * h[i-k] for valid k
    backward_part = signal.convolve(np.flip(w), h, mode='full')
    backward_part = backward_part[n-1:] # corresponds to shift k=0 to n-1
    
    # In each position i, gradient := forward_part[i] - backward_part[i] 
    # However we must correctly trim it to align i in range [0,n-1] 
    # i element starts at index 0 in forward_part array
    # The gradient is elements [0, n-1]
    return forward_part[:n] - backward_part[:n]

def project_onto_feasible(y, target_sum):
    # y: array of length n; project onto [0,1]^n and sum == target_sum
    # Use water-filling algorithm: sorted descending parts

    n = len(y)
    if n == 0:
        return y.copy()

    # Clip to [0, 1]
    y_clipped = np.clip(y, 0, 1)
    s = np.sum(y_clipped)
    
    # If exactly sum, no move needed
    if abs(s - target_sum) < 1e-12:
        return y_clipped.copy()

    # Use water filling algorithm (sort reversed)
    idx_sorted = np.argsort(y_clipped)[::-1]
    y_sorted = y_clipped[idx_sorted]
    target = target_sum
    y_result = y_clipped.copy()
    
    # Use the iterative adjustment (similar to np.clip):
    # Start from top, adjust until satisfy sum or all at bound
    free_vars = list(range(n))
    free_idx = np.arange(n)
    
    # make deep copy?
    while len(free_vars) > 0:
        s_free = np.sum(y_result[free_vars])
        excess = s_free - target
        
        if abs(excess) < 1e-12:
            break
            
        if excess > 0:
            # Need to decrease some free components
            # Remove the ones that would go below 0 (they're not free)
            new_free_vars = []
            delta = excess / len(free_vars)
            for var in free_vars:
                new_val = y_result[var] - delta
                if new_val >= 0:
                    y_result[var] = new_val
                    new_free_vars.append(var)
                else:
                    y_result[var] = 0.0  # Set to 0 (no longer free)
            free_vars = new_free_vars
        else:
            # Need to increase
            new_free_vars = []
            delta = -excess / len(free_vars)
            for var in free_vars:
                new_val = y_result[var] + delta
                if new_val <= 1:
                    y_result[var] = new_val
                    new_free_vars.append(var)
                else:
                    y_result[var] = 1.0  # No longer free
            free_vars = new_free_vars

    # Final correction to ensure correct sum
    current_sum = np.sum(y_result)
    diff = target_sum - current_sum
    if abs(diff) > 1e-12:
        # Distribute diff to one of the interior vars with small allocation
        # most 'willing' to accept change
        for i in range(n):
            if 0 < y_result[i] < 1:
                y_result[i] += diff
                break
        else:
            # No interior var; this shouldn't happen in valid cases
            y_result[0] += diff

    return y_result

def swap_update(c, h, i, j, off):
    # Incrementally update correlation c after swapping h[i] and h[j]
    # Note: shifts are indexed from -n+1 to n-1 (total 2*n-1 entries)
    # For shift index k = i - (j + off)  in numpy: k = (i + off) - (j + off) = i - j
    # Change is: added +1 to positions where h[i + k] was 0, subtracted 1 from k where h[j + k] was 1
    # When we modify a single entry 0→1 or 1→0 for index p, affect c[shift] where:
    #   shift = p - q (forward):
    #     for k = q = p to n-1: c[p-q + off] += (1-h[q]) = c[-(q-p) + off]
    #   shift = p - q (backward): 
    #     for k = q = 0 to p: c[p-q + off] -= h[q]
    # Then also swap h[i] and h[j]
    
    # Need helper indices
    n = len(h)
    # shift = p - q
    for q in range(i, n):
        shift = i - q
        c[shift + off] += (1 - h[q]) # contribution from i 
    for q in range(0, i):
        shift = i - q
        c[shift + off] -= h[q]

    for q in range(j, n):
        shift = j - q
        c[shift + off] -= (1 - h[q]) # contribution from j (opposite sign)
    for q in range(0, j):
        shift = j - q
        c[shift + off] += h[q]
    
    # Now update h values
    h[i], h[j] = h[j], h[i]

def targeted_swap(h, c, off):
    # Find worst shift index (c_k = max c[k])  
    # Then decide which indices in h can flip to reduce that shift.
    # A = {i : h[i] == 1, i+k in [0,n) and h[i+k] == 0} contributes 1 to c_k  
    # B = {j : h[j] == 0, j+k in [0,n) and h[j+k] == 1} contributes -1 to c_k  
    max_shift = np.argmax(c)
    max_shift_val = max_shift - (off)
    
    n = len(h)
    A = []
    B = []
    for i in range(n):
        if h[i] == 1:
            k = max_shift_val
            if 0 <= i + k < n and h[i+k] == 0:
                A.append(i)
        
    for j in range(n):
        if h[j] == 0:
            k = max_shift_val
            if 0 <= j + k < n and h[j+k] == 1:
                B.append(j)
    
    if not A or not B:
        # fallback: pick any pair that can swap
        # Here: try to find a single (i,j) such that swapping decreases max(c)
        # NB: just a desktop-FW idea, not well supported.

    if A and B:
        # Pick random from A and B, one swap is guaranteed to reduce c[max_shift] by 1.
        i = A[np.random.randint(len(A))]
        j = B[np.random.randint(len(B))]
        swap_update(c, h, i, j, off)
    else:

def local_binary_search(h, c, best_val, time_left):
    # Given binary solution h, compute c, then do targeted reverse swaps
    # No need for entire swap matrix: just do error-search + swaps
    n = len(h)
    off = n - 1 # offset
    
    min_iter = 1000
    max_iter = 20000 # About an hour with no changes — lots of backbone of redundancy - upper bound safeguard
    it = 0
    t_start = time.time()
    
    while it < max_iter and time.time() - t_start < time_left:
        # Use random shuffle of indices to avoid pattern search bogging
        it += 1
        if it % 1000 == 0:
            # Do a full sweep
            current_max = np.max(c)
            if current_max <= best_val * 1.001:
                break
        
        # Skim smallest of shift from c, swap index
        s_max = np.argmax(c)
        max_c_val = c[s_max]
        # Check skip if value too small
        if max_c_val < 0.1:
            break
        
        # Targeted swap with slight randomization
        if np.random.random() > 0.5:
            targeted_swap(h, c, off)
        else:  # Manual selection
            # Attempt some randomized swap that can move a subset
            # To save compute time, just sample many pairs and try each
            tries = 0
            success = False
            while tries < 10 and not success:
                i = np.random.randint(n)
                j = np.random.randint(n)
                if h[i] == 1 and h[j] == 0:
                    # Check effect
                    # Compute a shallow involved max pass 
                    c_old = c.copy()
                    swap_update(c, h, i, j, off)  
                    max_new = np.max(c)
                    if max_new <= max_c_val - 0.1:
                        # Accept
                        success = True
                    else:
                        # Revert
                        swap_update(c, h, i, j, off)  # revert
                        c[:] = c_old[:]  # Update full history from terminal
                    
                    tries += 1
                    if not success and tries > 5:
                        # Minor heuristic to avoid spending so much on local updates to increasingly flat regions
                        # Only continue the intense target aimed shift early majority intervention
                        break

    return h, c

def upscale(h, n_new):
    # Upscale by 2x by repeating each element
    n_old = len(h)
    h_up = np.repeat(h, 2)  # Each element appears twice
    # If odd, we'll still need uint
    if ((n_old << 1) != n_new):
        # Pad with zeros or a halves? No, just truncate (ff)
        h_up = np.resize(h_up, n_new)  
    else:
        h_up = h_up[:n_new]
    return h_up

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    t_start = time.time()
    
    # Configuration
    ns = [80, 120, 180, 240]  # Discretizations to try
    max_iter_per_n = 10000
    safety_margin = 2.0
    best_h = None
    best_val = float('inf')
    best_n = None
    
    for n in ns:
        if time.time() - t_start > budget_s - safety_margin:
            break        
          
        # Generate initial eligible vector h  
        h0 = np.random.rand(n)
        h0 = project_onto_feasible(h0, n / 2.0)  # Ensure integral constraint
        
        # Continuous optimization
        alpha = 5.0
        alpha_growth = 1.2
        max_alpha = 200.0
        eta = 0.5 * (2.0 / n)
        beta = 0.9
        h_prev = h0.copy()
        h_best = h0.copy()
        c_best = compute_correlation(h_best)  
        val_best = np.max(c_best) * (2.0 / n)
        
        iter_count = 0
        while iter_count < max_iter_per_n and time.time() - t_start < budget_s - safety_margin:
            iter_count += 1
            
            # Compute correlation at current point
            c = compute_correlation(h_best)
            val = np.max(c) * (2.0 / n)
            if val < val_best:
                val_best = val
                h_best[:] = h_best[:]  # Need deep copy

            # Smooth surrogate
            phi_alpha, weights = smoothed_objective_and_weights(c, alpha)
            
            # Gradient via convolution
            grad = compute_gradient_via_convolution(h_best, weights)

            # Apply momentum
            h_candidate = h_best - eta * (grad + beta * (h_best - h_prev))
            
            # Project
            h_new = project_onto_feasible(h_candidate, n/2.0)
            
            # Accept if improve surrogate
            c_new = compute_correlation(h_new)
            val_new = np.max(c_new) * (2.0 / n)
            phi_new, _ = smoothed_objective_and_weights(c_new, alpha)
            
            # Backtrack line search
            if phi_new <= phi_alpha:
                h_best[:] = h_new[:]
                h_prev[:] = h_new[:]
            else:
                # Reduce step size
                eta *= 0.5
                if eta < 1e-12:
                    # To not get trapped)
                    break
            
            # Occasionally increase alpha
            if iter_count % 100 == 0:
                alpha *= alpha_growth
                if alpha > max_alpha:
                    alpha = max_alpha

        # Binary rounding
        indices_sorted_by_val = np.argsort(h_best)[::-1]
        h_bin = np.zeros(n)
        h_bin[indices_sorted_by_val[:n//2]] = 1.0
        
        # Final polishing (binary search)
        c_bin = compute_correlation(h_bin)  
        val_bin = np.max(c_bin) * (2.0 / n)
        
        remaining_time = budget_s - (time.time() - t_start) - safety_margin
        
        if remaining_time > 0.5:
            h_bin, c_bin = local_binary_search(h_bin, c_bin, val_bin, 0.5 * remaining_time)
            val_bin = np.max(c_bin) * (2.0 / n)
        
        if val_bin < best_val:
            best_val = val_bin
            best_h = h_bin.copy()
            best_n = n
            
        # Upscale for next resolution
        if len(ns) > 1 and n != ns[-1]:
            h_up = upscale(h_bin, 2 * n)
            h0 = h_up
        else:
            h0 = np.zeros(n)
            h0[n//2:] = 1.0  # Mean-pad to begin values
    
    # Return best so far
    if best_h is None:
        # Fallback: use coarsest result if any
        best_h = np.zeros(n)
        best_h[n//2:] = 1.0

    return best_h.tolist(), best_val, best_n
