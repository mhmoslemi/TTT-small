from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given step function h"""
    # Compute correlation of h with (1-h)
    correlation = np.correlate(h, 1-h, mode="full")
    # Extract the relevant part (maximum overlap)
    max_corr = np.max(correlation)
    return max_corr * dx
