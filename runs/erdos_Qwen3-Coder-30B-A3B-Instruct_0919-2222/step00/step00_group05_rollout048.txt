import numpy as np
import time
import random
from scipy.optimize import minimize

def preprocess_shift_maps(n):
    """Precompute shift maps for correlation calculations."""
    shift_maps = {}
    for s in range(-n+1, n):
        pairs = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                pairs.append((i, j))
        shift_maps[s] = pairs
    return shift_maps

def evaluate(h):
    """Compute the true maximum overlap using numpy correlation."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def project_to_capped_simplex(v, M):
    """Project vector v onto the capped simplex {h : 0 <= h_i <= 1, sum h_i = M}."""
    n = len(v)
    # Sort indices by v in descending order
    sorted_indices = np.argsort(-v)
    
    # Initialize
    u = np.zeros_like(v)
    u[sorted_indices[0]] = max(0, min(1, v[sorted_indices[0]]))
    sum_u = u[sorted_indices[0]]
    
    # Sweep to find threshold
    for i in range(1, n):
        idx = sorted_indices[i]
        val = v[idx]
        # Try setting element to 1
        new_sum = sum_u + min(1, val)
        if new_sum <= M:
            u[idx] = min(1, val)
            sum_u = new_sum
        else:
            # Compute threshold theta
            theta = (sum_u - M) / (i - sum_u + M)
            # Adjust
            for j in range(i):
                u[sorted_indices[j]] = max(0, min(1, v[sorted_indices[j]] - theta))
            u[idx] = 0
            sum_u = M
            break
    
    if sum_u < M:
        # Distribute remainder
        remain = M - sum_u
        for idx in sorted_indices:
            if u[idx] < 1:
                delta = min(1 - u[idx], remain)
                u[idx] += delta
                remain -= delta
                if remain <= 1e-10:
                    break
    return u

def correlations(h, shift_maps):
    """Compute all correlation values for the given h."""
    n = len(h)
    C = np.zeros(2*n - 1)
    for s, pairs in shift_maps.items():
        if s >= 0:
            corr = 0
            for i, j in pairs:
                corr += h[i] * (1 - h[j])
            C[s] = corr
        else:
            corr = 0
            for i, j in pairs:
                corr += h[i] * (1 - h[j])
            C[s + n - 1] = corr
    return C

def smooth_obj_and_grad(h, alpha, shift_maps):
    """Compute smooth approximation and its gradient."""
    C = correlations(h, shift_maps)
    max_C = np.max(C)
    exp_C = np.exp(alpha * (C - max_C))
    sum_exp = np.sum(exp_C)
    if sum_exp == 0:
        # This should not happen if shifts are not degenerate, but for safety
        w = np.zeros_like(C)
    else:
        w = exp_C / sum_exp
    
    # Gradient computation
    grad = np.zeros_like(h)
    for s, pairs in shift_maps.items():
        if s >= 0:
            idx = s
        else:
            idx = s + len(C) - 1
        w_s = w[idx]
        for i, j in pairs:
            grad[i] += w_s * (1 - h[j] - h[i] * (i==j))  # Exclude self term
        for i, j in pairs:
            grad[i] -= w_s * h[j]  # f, or like it was one Mike

    return np.log(sum_exp) / alpha + max_C, grad

def run_descent(h0, alpha_schedule, eta0, max_iter, shift_maps, time_limit):
    """Run gradient descent for fixed alpha schedule."""
    np.random.seed(int(time.time()) % 1000000)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    start = time.time()
    
    for alpha in alpha_schedule:
        if time.time() - start > time_limit:
            break
        eta = eta0
        for k in range(max_iter):
            if time.time() - start > time_limit:
                break
            val, grad = smooth_obj_and_grad(h, alpha, shift_maps)
            v = h - eta * grad
            h = project_to_capped_simplex(v, len(h) / 2)
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            eta *= 0.95  #decay
            
    return best_h, best_c5

def random_binary(n, M):
    """Generate a random binary vector with exactly M ones."""
    indices = random.sample(range(n), M)
    h = np.zeros(n)
    h[indices] = 1
    return h

def random_fractional(n, M):
    """Generate a random fractional vector and project to feasible simplex."""
    v = np.random.dirichlet([1] * n)
    return project_to_capped_simplex(v, M)

def swap_local_search(h, max_iter, time_limit):
    """Perform targeted swaps to reduce max correlation."""
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate(h)
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    current_max = np.max(corr)
    max_s = np.argmax(corr) - (n-1)
    
    for iteration in range(max_iter):
        if time.time() - start > time_limit:
            break
        # Populate the worst shift related indices efficiently
        if max_s > 0:
            i_list = [i for i in range(n) if (i + max_s) < n and h[i] == 1 and h[i+max_s] == 0]
        else:
            i_list = [i for i in range(n) if (i + max_s) >= 0 and h[i] == 1 and h[i+max_s] == 0]
        
        if not i_list:
            break
        i = random.choice(i_list)
        # Check target index
        if max_s > 0:
            j = [j for j in range(n) if h[j] == 0 and h[j + max_s] == 1]
        else:
            j = [j for j in range(n) if h[j] == 0 and h[j + max_s] == 1]
        if not j:
            continue
        j = random.choice(j)
        
        # Try swap
        h_test = h.copy()
        h_test[i], h_test[j] = h_test[j], h_test[i]
        test_c5 = evaluate(h_test)
        if test_c5 < best_c5:
            h = h_test
            best_c5 = test_c5
            corr = np.correlate(h, 1-h, mode="full")
            current_max = np.max(corr)
            max_s = np.argmax(corr) - (n-1)
        else:
            # Revert if invalid
            pass
    
    return best_h, best_c5
