import numpy as np
from scipy.optimize import differential_evolution
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap integral
    max_overlap = np.max(correlation) * dx
    return max_overlap

def initial_h_values():
    """Return initial construction from the problem"""
    n = 70
    h = np.zeros(n)
    # Simple initial construction
    # Fill first half with 1, second half with 0
    h[:n//2] = 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    start_time = time.time()
    
    n_points = 70
    # Start with initial construction
    h = initial_h_values()
    
    # We'll optimize using a succession of local optimizations
    # to avoid local minima
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Create bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimizer parameters
    maxiter = 100
    disp = False
    
    # Perform optimization
    for i in range(1000):  # Limit number of iterations
        if time.time() - start_time > budget_s - 1:  # Leave 1s for final checks
            break
            
        # Random perturbation to escape local optima
        h_perturbed = h.copy()
        # Add small random perturbations
        perturbation = np.random.normal(0, 0.01, n_points)
        h_perturbed += perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Rescale to ensure integral equals 1
        integral = np.sum(h_perturbed) * (2.0 / n_points)
        h_perturbed = h_perturbed * (1.0 / integral)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_perturbed)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_perturbed.copy()
            
        # Early termination if we are close to target
        if best_c5 < 0.379:
            break
            
        # Try to refine further
        try:
            # Use scipy.optimize.minimize to improve solution slightly
            def objective(h_vals):
                # Reconstruct h array
                h_array = np.array(h_vals)
                # Ensure sum = 1 (as required)
                integral = np.sum(h_array) * (2.0 / n_points)
                h_scaled = h_array * (1.0 / integral)
                h_scaled = np.clip(h_scaled, 0, 1)
                return evaluate_erdos_solution(h_scaled)
            
            # Use differential evolution for global optimization
            res = differential_evolution(objective, bounds, maxiter=100, 
                                        seed=seed+i, disp=False, 
                                        strategy='best1bin', popsize=10)
            
            if res.success:
                h_trial = res.x
                integral = np.sum(h_trial) * (2.0 / n_points)
                h_trial = h_trial * (1.0 / integral)
                h_trial = np.clip(h_trial, 0, 1)
                c5_trial = evaluate_erdos_solution(h_trial)
                
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    
                # Break early if good enough
                if best_c5 < 0.379:
                    break
                    
        except:
            pass
        
        # Update h for next iteration
        h = best_h.copy()
    
    # Final check and return best solution found
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), float(final_c5), n_points)
