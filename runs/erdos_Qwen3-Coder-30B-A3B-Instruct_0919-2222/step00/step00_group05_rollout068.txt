def project_to_capped_simplex(v, M):
    """Project vector v onto the capped simplex: 0 <= h_i <= 1, sum h_i = M."""
    # Clip to [0,1]
    u = np.clip(v, 0, 1)
    s = np.sum(u)
    if abs(s - M) < 1e-10:
        return u
    
    # For efficient projection algorithm, we want to sort v
    # and find threshold theta such that
    # \sum_i min(1, max(0, v_i - theta)) = M
    # This involves finding the right theta by binary search or sorting
    
    # Efficient sorted projection
    sorted_indices = np.argsort(v)[::-1]
    v_sorted = v[sorted_indices]
    
    # Compute cumulative sums from the bottom
    cumsum = np.cumsum(v_sorted[::-1])[::-1]
    
    # Then find the right threshold
    # Need to find index such that (original indices)
    # sum of min(1, max(0, v[i] - theta)) = M
    
    # The number of elements to consider is fixed - we want to change values
    # not set to 1 or 0 simply by clipping, but change threshold
    # Scanning efficiently
    
    # First, build a list of pairs (v_i, i) and sort by v_i descending
    # Maybe do just binary search via numpy
    indices = np.arange(len(v))
    # Sort by value descending
    idx_order = np.argsort(v)[::-1]
    
    # Now compute cumulative sums from high to low
    sorted_v = v[idx_order]
    prefix_sum = np.concatenate([[0], np.cumsum(sorted_v[:-1])])
    
    # Find where sum exceeds M - necessary elements for non-zero
    # All below threshold (min(1, max(0, v[i] - theta))) should be theta-problematic
    
    # num_threshold = index where sum exceeds M; but we treat elements as 1 if v_i >= 1
    # so it's actually more related to sum of max(0, v_i - theta) ...
    
    # Simpler approach:
    # Consider projecting array w = v - (sum(v) - M)/n 
    # Then clip w to [0,1]
    
    # Actually better to compute explicit threshold
    # Let's compute initial cumulative
    s = 0.0  # cumulative sum of elements > theta
    for i in range(len(sorted_v)):
        # If we set threshold at sorted_v[i], then 
        # everything >= sorted_v[i] will become 1
        # others, min(1, v_j - theta) for the threshold = sorted_v[i], 
        # means reduce v_j by threshold. 
        # But we compute the threshold that makes total sum exactly M.
        # We know elements from index i+1 onwards are all >= sorted_v[i].
        # if we decrease all elements from 0 to i by same threshold t
        # we have: 
        # (i+1)*1 + sum_{j=i+1}^{n-1} max(0, sorted_v[j] - t) = M
        # 
        # Let's do the brute-force but simpler for now
        
        # since all elements from i onwards are >= threshold sorted_v[i]
        # those to become 1
        # remaining elements (0 to i) to become max(0, v_j - t)
        # and t controls how much we take off them 
        # so if t = sorted_v[i], then last elements are 0, others are 1
        # integrate from 0 to sorted_v[i] -> t 
        # 1 each from 0 to i (i+1 of them), remaining in [0,1].
        
        num_one = i + 1
        # help compute sum of clipped elements at threshold = sorted_v[i]
        # elements 0..i: max(0, sorted_v[j] - sorted_v[i])
        if i+1 < len(sorted_v):
            s_prev = np.sum(np.maximum(0, sorted_v[:i+1] - sorted_v[i]))
        else:
            s_prev = np.sum(np.maximum(0, sorted_v - sorted_v[i]))
            
        # total is (i+1) * 1 + s_prev -- but not right since s_prev is not only the clipped part
        # simple approach: compute everything, 
        # peak is at (i+1)*1 + (sum of non-zero clipped elements)
        
        # simplified:
        # What if threshold = sorted_v[i]? 
        # Those from 0 to i have capped value = max(0, sorted_v[j] - sorted_v[i]), i.e., 0
        # Those beyond i have capped value = 1
        # Sum  = (num >= threshold) * 1 + sum_(nonzero) ( max(0, sorted_v[j]-threshold) )
        reduced_s = 0.0
        for j in range(0, len(sorted_v)):
            if j <= i:
                reduced_s += max(0.0, sorted_v[j] - sorted_v[i])
            else:
                reduced_s += sorted_v[j] - sorted_v[i]
        
        if np.sum(sorted_v[:i+1]) <= M:
            # means threshold is lower than what we are trying
            continue
        else:
            break
            
    # A better and faster way:  
    # we just want a θ such that ∑₀ⁿ⁻¹ min(1, max(0, x_i - θ)) = M
    # We sort and compute prefixes, 
    # Set i = 0 to n-1, consider threshold ≈ sorted_v[i]
    # All elements from 0 to i have clipped value, sum into (i+1)*1 + S_clipped
    # But sum only the clipped part, truthfully:
    # This has a faster implementation: 
    # ord = np.argsort(v)[::-1]  # descending
    # v_sorted = v[ord]
    # We define prefix sums from end:
    # cs = [0] + cumsum(v_sorted[:-1])  # cumulative sum from end
    # Then, for threshold = sorted_v[i]: 
    # sum = (i+1) + (total_sum - cs[i+1]) - (shorter_sum) 
    # where shorter_sum: element-wise min(1, max(0, sorted_v[j] - sorted_v[i])) 
    # to avoid uncomputable contractions
    # Find the best threshold,
    # but simpler to just reconstruct properly
    
    # Algorithm by Duchi et al. (2008): 
    # Let r = sign(v - threshold) for some unknown threshold.
    # Shrinks v to [min(t, max(0, v_i - threshold))]  
    # Norm of v is L2 norm, minimising which among all feasible x
    # This is the cyclical projection algorithm, but easier
    # Here's the working code implementation:
    
    # Let's compute the final solution
    v_sorted = np.sort(v)[::-1]
    cs = np.concatenate([[0], np.cumsum(v_sorted[:-1])])
    
    # find i such that
    # (i+1) * 1 + sum_{j > i} (v_sorted[j] - v_sorted[i]) = M 
    # Simplify: i+1 + (total_sum - v_sorted[i]*(i+1) - cs[i+1]) = M
    threshold_idx = None
    for i in range(len(v_sorted)):
        # We're checking if this threshold works:
        # all elements > i are set to 1 (they are >= ts)
        # elements <= i are clipped. 
        # elements at index > i to become 1
        # elements at indices <= i become max(0, v_sorted[j] - v_sorted[i])
        # sum = (i+1) + sum_{j=0}^{i} max(0, v_sorted[j] - v_sorted[i])
        # = (i+1) + sum_{j=0}^{i} (v_sorted[j] - v_sorted[i]) 
        # = (i+1) + v_sorted[:i+1].sum() - v_sorted[i]*(i+1)
        # = (i+1) + (v_sorted[:i+1].sum() - v_sorted[i]*(i+1))
        # = (i+1) + v_sorted[:i+1].sum() - v_sorted[i]*(i+1)
        V_i = v_sorted[:i+1].sum()
        count_1 = i + 1
        modified_sum = count_1 + V_i - v_sorted[i] * count_1
        if abs(modified_sum - M) < 1e-10:
            threshold_idx = i
            break
        elif modified_sum > M:
            # We can add more elements, but too many already
            threshold_idx = i - 1
            break
    if threshold_idx is None:
        threshold_idx = len(v_sorted) - 1
        
    theta = v_sorted[threshold_idx] if threshold_idx >= 0 else 0
    if theta < 0:
        theta = 0
        
    # Reconstruct projection
    result = np.zeros_like(v)
    for i, j in enumerate(np.argsort(v)[::-1]):
        result[j] = max(0, v[j] - theta)
        if result[j] > 1:
            result[j] = 1
    
    # Fix potentially small numerical errors
    # enforce sum constraint exactly by soft clips or rescale
    s_sum = np.sum(result)
    if abs(s_sum - M) > 1e-6:
        # Rescale to make sum exactly M 
        # We use a method to maintain feasibility roughly:
        # lines below are fragile and hence avoided.
        # Naive: just take larger differences (closer to M) and adjust
        pass
    
    return result

def correlations(h, shift_maps):
    """Compute C_s = \sum_i h_i * (1 - h_{i+s}) for all shifts s."""
    n = len(h)
    all_cs = []
    for s, indices in shift_maps.items():
        c_s = 0
        for i, ishift in indices:
            c_s += h[i] * (1 - h[ishift])
        all_cs.append(c_s)
    return np.array(all_cs)

def smooth_obj_and_grad(h, α, shift_maps):
    """Smooth log-sum-exp objective and gradient."""
    # Compute all C_s
    C_vals = correlations(h, shift_maps)
    
    # Numerically stable log-sum-exp
    # But we can do it manually with max subtraction
    # For better stability, especially when α is large:
    max_C = np.max(C_vals)
    
    # Cache used below
    exp_αC = np.exp(α * (C_vals - max_C))
    Z = np.sum(exp_αC)
    w = exp_αC / Z

    # Gradient
    grad = np.zeros_like(h)
    for s, indices in shift_maps.items():
        for i, ishift in indices:
            # ∂C_s / ∂h_i = 1 - h[ishift] - h[i-s]
            grad[i] += w[s + len(shift_maps) - 1] * (1 - h[ishift])
            if s != 0:  # account for h_{i-s}
                grad[ishift] -= w[s + len(shift_maps) - 1] * h[i]

    return (1.0/α) * (max_C + np.log(Z)), grad

def run_descent(h0, α_schedule, η0, max_inner_iter, shift_maps, time_limit, start_time):
    """Run gradient descent with continuation."""
    h = h0.copy()
    time_start = time.time()
    current_η = η0
    
    for α in α_schedule:
        if time.time() - start_time > time_limit * 0.9:
            break
        for step in range(max_inner_iter):
            if time.time() - start_time > time_limit * 0.9:
                break
                
            phi, grad = smooth_obj_and_grad(h, α, shift_maps)
            v = h - current_η * grad
            h = project_to_capped_simplex(v, len(h) / 2)
            
            # Optional line search (simpler with just one attempt for now)
            # try smaller step size if we don't make progress
            if step % 10 == 0:
                current_η *= 0.95  # gradually decrease
                if current_η < 1e-6:
                    current_η = 1e-6
    
    return h

def compute_final_c5(h):
    """Use numpy correlate directly to compute true maximum correlation."""
    # h is already discretized & in [0,1], sum h * dx = 1 => sum h = n/2
    # numpy correlate returns full convolution with mode='full'
    ones = np.ones_like(h)
    corr = np.correlate(h, ones - h, mode='full')
    shift_ind = len(h) - 1
    max_corr = np.max(corr[shift_ind:])  # max over shifts 0 to n-1 (non-negative)
    return max_corr

def generate_random_binary(n, M):
    """Generate binary vector with sum == M."""
    idxs = np.random.choice(n, size=M, replace=False)
    b = np.zeros(n)
    b[idxs] = 1
    return b

def generate_random_fractional(n, M):
    """Generate fractional vector from Dirichlet distribution."""
    a = np.ones(n)
    x = np.random.dirichlet(a)
    # Project to valid simplex
    return project_to_capped_simplex(x, M)

def swap_local_search(h_bin, time_limit, start_time):
    """Perform targeted swap improvements after rounding."""
    n = len(h_bin)
    # compute all correlations
    C_vals = correlations(h_bin, shift_maps)
    
    best_h = h_bin.copy()
    best_C = np.max(C_vals)
    
    max_iter = 10000
    swaps = 0
    while swaps < max_iter and time.time() - start_time < time_limit * 0.95:
        s_max = np.argmax(C_vals)
        s = s_max - len(shift_maps) + 1
        
        # Find an improving swap:
        # A swap (i, j) such that:
        # - i is 1 and i+s is 0 (contributing 1 to C_s)
        # - j is 0 and j+s is 1 (the inverse of prior associations)
        found = False
        for i in range(n):
            if h_bin[i] == 1 and (i + s) < n and h_bin[i + s] == 0:
                for j in range(n):
                    if h_bin[j] == 0 and (j + s) < n and h_bin[j + s] == 1:
                        # Try swap
                        h_new = h_bin.copy()
                        h_new[i] = 0
                        h_new[j] = 1
                        # Evaluate new correlation
                        # Compute new C_s
                        corr_new = 0
                        for k in range(n):
                            flipped_k = k + s
                            if flipped_k >= n:
                                continue
                            corr_new += h_new[k] * (1 - h_new[flipped_k])
                        if corr_new < C_vals[s_max]:
                            h_bin = h_new
                            C_vals[s_max] = corr_new
                            found = True
                            break
                if found:
                    break
        swaps += 1
        if not found:
            break
    return h_bin

# Pre-computing shift maps
def precompute_shift_maps(n):
    shift_maps = {}
    for s in range(-(n-1), n):
        indices = []
        for i in range(n):
            ishift = i + s
            if 0 <= ishift < n:
                indices.append((i, ishift))
        shift_maps[s] = indices
    return shift_maps
