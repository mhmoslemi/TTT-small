def solve_milp(n, time_limit):
    import pulp, numpy as np
    prob = pulp.LpProblem('Erdos', pulp.LpMinimize)
    h = pulp.LpVariable.dicts('h', list(range(n)), cat='Binary')
    T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
    prob += T  # objective

    # sum h = n/2
    prob += pulp.lpSum([h[i] for i in range(n)]) == n/2

    # for each shift
    for d in range(0, n):
        # compute list of i such that i+d < n
        max_i = n - d
        if max_i <= 0:
            continue
        y_vars = []
        for i in range(max_i):
            y = pulp.LpVariable(f'y_{i}_{d}', lowBound=0, upBound=1, cat='Continuous')
            y_vars.append(y)
            # Constraints:
            prob += y <= h[i]
            prob += y <= h[i+d]
            prob += y >= h[i] + h[i+d] - 1
        # shift constraint:
        prob += pulp.lpSum(y_vars) + T >= n/2
    # Solve
    solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit, threads=1)
    result_status = prob.solve(solver)
    status = pulp.LpStatus[prob.status]
    # Extract h values
    h_vals = np.array([pulp.value(h[i]) for i in range(n)], dtype=float)
    # Since binary, must be integer 0/1; convert to 0/1 by rounding
    h_binary = np.rint(h_vals).astype(int)
    # Compute C5:
    dx = 2.0 / n
    conv = np.correlate(h_binary, 1 - h_binary, mode='full')
    C5 = conv.max() * dx
    return h_binary.astype(float), C5, n
