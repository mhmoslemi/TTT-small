def evaluate_overlap(h):
    """Evaluate the overlap integral C_5 for a given step function h."""
    n = len(h)

    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value of the convolution gives the C_5 bound
    max_val = np.max(corr)
    return max_val * dx
