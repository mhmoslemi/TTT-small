import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value divided by dx
    return np.max(correlation) * dx

def initial_h_values(n_points):
    """Return initial construction - a piecewise constant function."""
    # Start with a refined version that's a good candidate
    h = np.zeros(n_points)
    # Split into segments with different values
    # Start with baseline from initial construction
    step = 2.0 / n_points
    # Make 8 segments
    segments = 8
    for i in range(segments):
        start = i * n_points // segments
        end = (i+1) * n_points // segments
        if i % 2 == 0:
            h[start:end] = 0.5
        else:
            h[start:end] = 0.0
    # Normalize to have integral = 1.0
    h = h * (n_points / (2.0 * np.sum(h)))
    return h

def objective(h):
    """Objective function to minimize."""
    return evaluate_erdos_solution(h)

def constraint_eq(h):
    """Equality constraint for integral = 1."""
    n_points = len(h)
    dx = 2.0 / n_points
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    # Seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Reasonable number of points to allow optimization
    dx = 2.0 / n_points
    
    # Initialize with better starting point
    h0 = initial_h_values(n_points)
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set up constraint dictionary
    constr = {'type': 'eq', 'fun': constraint_eq}
    
    def wrapped_objective(h_params):
        return objective(h_params)
    
    # Track best solution
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    # Simple optimization: start with initial guess, then try randomized search
    # Then refine using differential evolution
    
    # Start with differential evolution
    try:
        result = differential_evolution(
            wrapped_objective,
            bounds,
            constraints=constr,
            seed=seed,
            maxiter=int(1000),
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            atol=1e-5,
            callback=lambda x, convergence: None,
            disp=False
        )
        if not result.success:
            raise Exception("Differential evolution did not converge")
        current_h = result.x
        current_c5 = evaluate_erdos_solution(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            return best_h, best_c5, n_points
    except Exception as e:
        pass  # continue with optimization below
    
    # Try a few random restarts if we're running low on time
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    
    if time_left > 50:  # allow 50 sec for any further improvements
        n_restarts = int(time_left / 10)
        for _ in range(n_restarts):
            # Random perturbation around best solution
            h_try = best_h.copy()
            # Small random perturbations
            for i in range(n_points):
                if np.random.rand() < 0.2:
                    delta = np.random.uniform(-0.1, 0.1)
                    h_try[i] = h_try[i] + delta
                    h_try[i] = np.clip(h_try[i], 0, 1)
            # Ensure constraint is satisfied
            integral = np.sum(h_try) * dx
            h_try = h_try * (1.0 / integral)
            c5_new = evaluate_erdos_solution(h_try)
            if c5_new < best_c5:
                best_h = h_try
                best_c5 = c5_new
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    # Final check if we can get a better result with closer grid resolution
    if n_points < 500:  # increase resolution only if reasonable
        n_points_high = 200
        dx_high = 2.0 / n_points_high
        h_high = np.zeros(n_points_high)
        for i in range(n_points_high):
            idx = int(i * n_points / n_points_high)
            h_high[i] = best_h[idx]
        # Renormalize
        integral = np.sum(h_high) * dx_high
        h_high = h_high * (1.0 / integral)
        c5_high = evaluate_erdos_solution(h_high)
        if c5_high < best_c5:
            best_h = h_high
            best_c5 = c5_high
            n_points = n_points_high
    
    return best_h, best_c5, n_points
