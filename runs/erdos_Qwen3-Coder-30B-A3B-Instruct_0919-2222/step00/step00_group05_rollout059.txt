import numpy as np
import time
import random
from scipy.special import logsumexp

# Helper functions for the optimization

def project_to_capped_simplex(v, M):
    """Project v onto the capped simplex: {x | 0 <= x_i <= 1, sum x_i = M}"""
    v = np.array(v)
    n = len(v)
    u = np.clip(v, 0, 1)
    s = np.sum(u)
    if np.abs(s - M) < 1e-10:
        return u.tolist()
    # Sort v in descending order
    idx = np.argsort(v)[::-1]
    u_sorted = u[idx]
    # Find threshold theta such that sum of min(1, max(0, v_i - theta)) = M
    # This is done by trying different thresholds
    # Simple approach: binary search on a coarse grid of possible thresholds
    # Better approach: use the standard projection algorithm variant
    # Let's use a direct implementation:
    # First project to box, then adjust the mass
    if s > M:
        # Need to reduce mass
        threshold = (s - M) / n  # this is a good approximation for initial starting point
        # But to be exact, solve the problem directly:
        # The O(n log n) method is to compute cumulative sums
        u_sorted = np.sort(u)[::-1]
        cumsum = np.cumsum(u_sorted)
        # Find the smallest k such that u_sorted[k] < M - (n - k) * u_sorted[k]
        # That would be
        # sum_{i=0}^{k-1} u_sorted[i] + (n-k) * u_sorted[k] = M
        # => sum_{i=0}^{k-1} u_sorted[i] = M - (n-k)*u_sorted[k]
        # Homebrew using sorted lists for efficiency:
        # We compute partial sums in reverse order and check condition
        k_star = None
        prev_sum = 0
        for k in range(n):
            cumulative = cumsum[k] if k < n-1 else cumsum[k]
            remaining = n - k
            if cumulative + (remaining * u_sorted[k]) <= M:
                # Threshold can be up to u_sorted[k]
                # Also test further back
                if k_star is None or k <= k_star:
                    k_star = k
            else:
                break
        # Now backward check:
        # Candidates for threshold in [u_sorted[k], u_sorted[k+1]] when k could be k_star
        # More reliable way:
        # For finite number of such thresholds, just find minimum based on k and u_sorted[k]
        
        # Use the standard algorithm:
        diff = np.array([u_sorted[i] * (n - i) - M for i in range(n)])
        # The solution is where diff changes sign or is 0
        # We seek largest k such that diff[k] <= 0
        valid_indices = np.where(diff <= 0)[0]
        if len(valid_indices) == 0:
            # All positive, reduce to max value average case:
            theta = u_sorted[0] - (M / n)
            if theta <= 0:
                theta = 0
        else:
            k_max = max(valid_indices)
            theta = u_sorted[k_max]
        # Correct the case
        result = np.maximum(0, v - theta)
        result = np.minimum(1, result)
        # Adjust mass:
        current_sum = np.sum(result)
        if np.abs(current_sum - M) > 1e-8:
            # Direct adjustment:
            delta = (M - current_sum) / n
            result = np.clip(result + delta, 0, 1)
        return result.tolist()
    else:
        # Need to increase mass;
        # Use the unclipped values, if sum < M
        # Which means we can add equally to all unclipped values
        # But we're already clipped to [0,1], so it's been clipped already...
        # Just independent adjustment:
        adjustment = (M - s) / n
        clipped = np.clip(u + adjustment, 0, 1)
        return clipped.tolist()

def preprocess_shifts(n):
    """Precompute valid shift indices for each shift"""
    shift_map = {}
    # Shift from -n+1 to n-1
    for s in range(1-n, n):
        pair_list = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                pair_list.append((i, j))
        shift_map[s] = pair_list
    return shift_map

def correlations(h, shift_map):
    """Compute the correlation vector Cs"""
    C = []
    for s in sorted(shift_map.keys()):
        corr = 0.0
        for i, j in shift_map[s]:
            corr += h[i] * (1 - h[j])
        C.append(corr)
    return np.array(C)

def smooth_obj_and_grad(h, alpha, shift_map):
    """Compute the smooth surrogate Phi_alpha and its gradient"""
    C = correlations(h, shift_map)
    max_C = np.max(C)
    # Numerically stable logsumexp
    logZ = logsumexp(alpha * (C - max_C)) + max_C * alpha
    logZ = logZ / alpha
    z = np.exp(alpha * (C - max_C))
    weights = z / np.sum(z)
    
    # Gradient
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for s in sorted(shift_map.keys()):
            for k, j in shift_map[s]:
                if k == i:
                    grad[i] += weights[s] * (1 - h[j] - h[i - s])
    return logZ, grad

def descent_one_alpha(h0, alpha, eta0, max_iter, shift_map, time_limit):
    """Single descent run for one alpha, with time limit"""
    start_time = time.time()
    h = np.array(h0)
    eta = eta0
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        obj, grad = smooth_obj_and_grad(h, alpha, shift_map)
        v = h - eta * grad
        h = np.array(project_to_capped_simplex(v, len(h) / 2))
        # Adaptive step size (for simplicity just decrease)
        eta *= 0.9
    return h.tolist()

def run_continuation(h_start, alpha_schedule, eta0, shift_map, time_limit):
    """Run the full α-continuation for one seed"""
    start_time = time.time()
    current = h_start
    for alpha in alpha_schedule:
        remaining_time = time_limit - (time.time() - start_time)
        if remaining_time <= 5:
            break
        # Max iters can depend on alpha
        max_iter = min(100, max(10, int(500 / (alpha + 1))))
        current = descent_one_alpha(current, alpha, eta0, max_iter, shift_map, remaining_time * 0.9)
    return current

# Required seeds

def random_binary(n, M):
    """Generate random binary vector with exactly M ones"""
    positions = np.random.choice(n, size=int(M), replace=False)
    h = [0.0] * n
    for i in positions:
        h[i] = 1.0
    return h

def random_fractional(n, M):
    """Generate random fractional vector, then project to feasible set"""
    h = np.random.dirichlet([1.0] * n)
    h = project_to_capped_simplex(h, M)
    return h

def evaluate(h):
    """Evaluate the actual C5 based on exact overlap"""
    n = len(h)
    # Use numpy correlate for exact computation
    c = np.correlate(h, 1 - np.array(h), mode="full")
    c = np.array(c)
    return np.max(c)

def swap_local_search(h_bin, time_limit):
    """Perform targeted improvements via swaps"""
    start_time = time.time()
    h = np.array(h_bin)
    n = len(h)
    best_c5 = evaluate(h)

    # Get current worst shift
    def get_worst_shift():
        c_full = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(c_full)
        max_val = c_full[max_idx]
        return max_idx - len(h) + 1, max_val

    worst_shift, current_max = get_worst_shift()

    swaps = 0
    while time.time() - start_time < time_limit and swaps < 5000:
        # Find source (1 that contributes to worst) and target (0 that could become 1 to help)
        # Check all pairs i, j such that:
        # i = witness that 0 has 1 at shift; so h[i]=0, h[j]=1 where j = i + worst_shift
        # All pairs are valid.
        found = False
        # At the beginning, sample a range:
        search_range = min(100, n)
        for _ in range(1000):
            # Try to find a propagating swap that reduces the worst shift by 2
            i = random.randint(0, n - 1)
            j = i + worst_shift
            if 0 <= j < n:
                if h[i] == 0 and h[j] == 1:
                    # Try to swap
                    new_h = h.copy()
                    new_h[i] = 1
                    new_h[j] = 0
                    new_max = evaluate(new_h)
                    if new_max < current_max:
                        h = new_h
                        current_max = new_max
                        found = True
                        break
        if not found:
            # Try a more systematic search
            best_i = None
            best_j = None
            best_delta = 0
            for i in range(n):
                j = i + worst_shift
                if 0 <= j < n:
                    if h[i] == 0 and h[j] == 1:
                        new_h = h.copy()
                        new_h[i] = 1
                        new_h[j] = 0
                        new_c5 = evaluate(new_h)
                        delta = new_c5 - current_max
                        if delta < best_delta:
                            best_delta = delta
                            best_i = i
                            best_j = j
            if best_i is not None:
                h[best_i] = 1
                h[best_j] = 0
                current_max = evaluate(h)
            else:
                break
        # After a swap, recompute the worst shift
        worst_shift, _ = get_worst_shift()
        swaps += 1
    return h.tolist() if current_max < best_c5 else h_bin
