import numpy as np

def evaluate_h(h):
    """Evaluate the overlap integral for a given h vector"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = corr.max()
    return max_overlap * dx

def compute_mismatch_sets(h, shift):
    """Identify mismatched positions for a given shift"""
    n = len(h)
    if shift >= 0:
        mismatches_1 = np.where((h[:-shift] == 1) & (h[shift:] == 0))[0]
        mismatches_0 = np.where((h[:-shift] == 0) & (h[shift:] == 1))[0]
    else:
        shift = -shift
        mismatches_1 = np.where((h[shift:] == 1) & (h[:-shift] == 0))[0]
        mismatches_0 = np.where((h[shift:] == 0) & (h[:-shift] == 1))[0]
    return mismatches_1, mismatches_0

def improve_worst_shift(h, max_shift):
    """Improve current h by swapping mismatched pairs"""
    n = len(h)
    mismatches_1, mismatches_0 = compute_mismatch_sets(h, max_shift)
    if len(mismatches_1) > 0 and len(mismatches_0) > 0:
        i = np.random.choice(mismatches_1)
        j = np.random.choice(mismatches_0)
        h[i], h[j] = h[j], h[i]
        return True
    else:
        # Fallback to random balanced swap if no mismatches
        one_indices = np.where(h == 1)[0]
        zero_indices = np.where(h == 0)[0]
        if len(one_indices) > 0 and len(zero_indices) > 0:
            i = np.random.choice(one_indices)
            j = np.random.choice(zero_indices)
            h[i], h[j] = h[j], h[i]
            return True
    return False

def binary_local_search(h, max_iter=2000):
    """Local search to improve binary h values"""
    # Ensure it's binary and balanced
    h = np.round(h).astype(int)
    n = len(h)
    if np.sum(h) != n // 2:
        # Fix the imbalance by swapping ones and zeros
        one_ids = np.where(h == 1)[0]
        zero_ids = np.where(h == 0)[0]
        adjust = len(one_ids) - n // 2
        if adjust > 0:
            ids_to_change = one_ids[:adjust]
            h[ids_to_change] = 0
        else:
            ids_to_change = zero_ids[:abs(adjust)]
            h[ids_to_change] = 1

    # Start with seeds
    h_start = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = corr.max()
    max_shift = corr.argmax() - (n - 1)
    
    for it in range(max_iter):
        if it % (n // 2) == 0:
            corr = np.correlate(h, 1 - h, mode="full")
            max_overlap = corr.max()
            max_shift = corr.argmax() - (n - 1)
        if max_overlap == 0:
            break
        improved = improve_worst_shift(h, max_shift)
        if not improved:
            return h.copy(), max_overlap
    return h.copy(), max_overlap

def simulated_annealing(h, max_iter=5000):
    """Run simulated annealing from binary initial vector"""
    h = h.copy()
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    best_val = corr.max()
    best_h = h.copy()

    # Start with temperature T0 = 0.5 and decay to 1e-4 over time
    for t in range(max_iter):
        fraction = float(t) / max_iter
        T = 0.5 * (1.0 - fraction) + 1e-4 * fraction
        
        # Propose random balanced swap
        one_indices = np.where(h == 1)[0]
        zero_indices = np.where(h == 0)[0]
        if len(one_indices) > 0 and len(zero_indices) > 0:
            i = np.random.choice(one_indices)
            j = np.random.choice(zero_indices)
            
            # Perform swap
            h[i], h[j] = h[j], h[i]
            new_corr = np.correlate(h, 1 - h, mode="full")
            new_max = new_corr.max()
            delta = new_max - best_val
            
            if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                if new_max < best_val:
                    best_val = new_max
                    best_h = h.copy()
            else:
                # Revert swap
                h[i], h[j] = h[j], h[i]
        else:
            break
    return best_h, best_val

def initialize_binary(n):
    """Initialize a balanced binary vector of length n"""
    h = np.zeros(n, dtype=int)
    ones = np.random.choice(n, size=n//2, replace=False)
    h[ones] = 1
    return h

def get_grad_for_shift(h, shift):
    """Compute gradient for a fixed shift on h"""
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        grad[:-shift] += 1
        grad[shift:] -= h[:-shift]
    else:
        shift = -shift
        grad[shift:] += 1
        grad[:-shift] -= h[shift:]
    return grad

def project_h(h, target_sum):
    """Project h onto the box [0,1] and sum constraint"""
    # Water-filling projection (Euclidean)
    v = h.copy()
    # Clamp to [0,1]
    v = np.clip(v, 0, 1)
    # Water-filling for constraint sum == target_sum
    l_min = np.min(v) - 1.0
    l_max = np.max(v) + 1.0
    def phi(lam):
        return np.sum(np.clip(v - lam, 0, 1)) - target_sum
    # Binary search for lambda
    for _ in range(40):
        lam = (l_min + l_max) / 2
        p = phi(lam)
        if p > 0:
            l_min = lam
        else:
            l_max = lam
    lam = (l_min + l_max) / 2
    new_h = np.clip(v - lam, 0, 1)
    return new_h

def gradient_descent_soft_max(h, alpha=40, steps=200, eta0=0.2):
    """Gradient descent using soft-max surrogate"""
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    
    for t in range(steps):
        # Compute all overlap counts
        o = np.zeros(2*n - 1)
        for k in range(2*n - 1):
            # Compute for shift k - (n - 1)
            shift = k - (n - 1)
            if shift >= 0:
                o[k] = np.sum(h[:-shift] * (1 - h[shift:]))
            else:
                o[k] = np.sum(h[-shift:] * (1 - h[:-shift]))
        
        # Compute soft-max gradient
        om = np.max(o)
        exp_o = np.exp(alpha * (o - om))
        p = exp_o / np.sum(exp_o)
        
        grad = np.zeros(n)
        for k in range(2*n - 1):
            shift = k - (n - 1)
            if shift >= 0:
                grad[:-shift] += p[k]
                grad[shift:] -= p[k] * h[:-shift]
            else:
                shift = -shift
                grad[shift:] += p[k]
                grad[:-shift] -= p[k] * h[shift:]
        
        # Update
        eta = eta0 / np.sqrt(t + 1)
        h += -eta * grad
        
        # Project
        h = project_h(h, target_sum)
        
        # Evaluate current max (for monitoring)
        max_val = np.max(o)
    return h, max_val
