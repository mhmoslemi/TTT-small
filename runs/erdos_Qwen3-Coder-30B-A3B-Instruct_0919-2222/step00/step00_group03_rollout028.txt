import numpy as np
from scipy.optimize import differential_evolution
import time
from math import sqrt

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant C5 for a given step function h"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the full convolution with 1-h
    h = np.array(h_values)
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the convolution times dx
    max_conv = np.max(conv)
    c5 = max_conv * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 70
    h = np.zeros(n)
    # Simple step function: 1 on [0,1) and 0 on [1,2)
    # scaled so that integral is 1
    for i in range(n):
        x = i * 2.0 / n
        if x < 1.0:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h

def objective_function(h_values):
    """
    Objective function to minize: negative of C5 because scipy.optimize.minimize minimizes
    """
    c5 = evaluate_erdos_solution(h_values)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define bounds: [0,1] for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Since we need the integral constraint, we'll solve this using a modified approach
    # Total sum constraint: sum(h) * dx = 1, i.e., sum(h) = n_points / 2
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Use differential evolution to handle the constraints better
    def wrapper_obj(h):
        # Normalize to satisfy the integral constraint
        h = np.array(h)
        scaled_h = h * (n_points / 2.0) / np.sum(h)
        # Ensure values are within bounds [0,1]
        scaled_h = np.clip(scaled_h, 0, 1)
        return objective_function(scaled_h)
    
    try:
        result = differential_evolution(
            wrapper_obj,
            bounds,
            seed=seed,
            maxiter=int(1000000 / n_points),  # Roughly adjust iterations based on n_points
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            atol=1e-6,
            tol=1e-6,
            callback=lambda x, convergence: None,
            disp=False
        )
        best_h = result.x
        # Normalize to satisfy the integral constraint again
        best_h = best_h * (n_points / 2.0) / np.sum(best_h)
        best_h = np.clip(best_h, 0, 1)
        c5_value = evaluate_erdos_solution(best_h)
        return (list(best_h), c5_value, n_points)
    except Exception as e:
        print("Error:", e)
        # Return initial solution if optimization fails
        c5_initial = evaluate_erdos_solution(h0)
        return (list(h0), c5_initial, n_points)
