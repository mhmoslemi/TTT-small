import numpy as np
import time
import random
from scipy import signal
from math import log, exp

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Discretisation sizes to try
    ns = [80, 120, 180, 240]
    best_h = None
    best_c5 = float('inf')
    best_n = None
    t0 = time.time()
    safety_margin = 2.0
    
    def project_onto_feasible(y, target_sum):
        # Project y onto [0,1]^n and sum = target_sum
        y = np.clip(y, 0., 1.)
        current_sum = np.sum(y)
        diff = target_sum - current_sum
        if abs(diff) < 1e-12:
            return y
        # Use water-filling algorithm
        if diff > 0:
            temp = y.copy()
            temp[temp == 1.] = float('inf')
            indices = np.argsort(temp)[::-1]
            free = [i for i in indices if y[i] < 1.]
            while diff > 0 and len(free) > 0:
                delta = 1. - y[free[0]]
                if delta > diff:
                    y[free] += diff / len(free)
                    diff = 0
                else:
                    d = delta * len(free)
                    y[free] = 1.
                    diff -= d
                    free = [i for i in free if y[i] < 1.]
        else:
            temp = y.copy()
            temp[temp == 0.] = float('-inf')
            indices = np.argsort(temp)
            free = [i for i in indices if y[i] > 0.]
            while diff < 0 and len(free) > 0:
                delta = y[free[0]]
                if -delta > -diff:
                    y[free] += diff / len(free)
                    diff = 0
                else:
                    d = -delta * len(free)
                    y[free] = 0.
                    diff -= d
                    free = [i for i in free if y[i] > 0.]
        return y

    def compute_correlation(h):
        # h = array of length n, full correlation
        h = np.asarray(h)
        c = np.correlate(h, 1 - h, mode="full")
        return c
    
    def smoothed_objective_and_weights(c, alpha):
        max_c = np.max(c)
        shifted = c - max_c
        exps = np.exp(alpha * shifted)
        s = np.sum(exps)
        weights = exps / s
        smooth_val = (1.0 / alpha) * (log(s) + max_c)
        return smooth_val, weights
    
    def grad_h(h, weights):
        h = np.asarray(h)
        n = len(h)
        # Precompute
        forward = np.cumsum(1 - h[::-1])[::-1]
        backward = np.cumsum(h)
        grad = np.zeros(n)
        # Simpler expression using convolution
        # For gradient of c_k with respect to h_i:
        #  = (1 - h_{i+k}) if i+k < n
        #  - h_{i-k} if i-k >= 0
        # w^T * [Δc_0, ..., Δc_{2n-2}]
        # Use convolution of w with (1-h) and h
        w = np.asarray(weights)
        ones = np.ones(n)
        conv1 = signal.convolve(w, (1 - h), mode='full')
        conv2 = signal.convolve(w, h, mode='full')
        # The gradient for each i is weighted sum of localized (1-h) and h terms
        grad = np.zeros(n)
        for i in range(n):
            # Weighted sum of shifts
            # sum_k w_k * [(1 - h_{i+k}) - h_{i-k}]
            grad[i] = conv1[i + n - 1] - conv2[i + n - 1]
        return grad
    
    def init_point(n):
        # Start from uniform distribution and project
        h = np.random.rand(n)
        h = project_onto_feasible(h, n/2)
        return h
    
    def local_binary_search(h_int, c_start, time_left):
        # Binary polishing using swap operations
        h = h_int.copy()
        c = c_start.copy()
        n = len(h)
        ONES = set(np.where(h == 1)[0])
        ZEROS = set(np.where(h == 0)[0])
        improved = True
        init_time = time.time()
        attempt_count = 0
        max_attempts = 1000000  # Max attempts before exit
        
        while improved and attempt_count < max_attempts and (time.time() - init_time < time_left - 1.0):
            improved = False
            max_shift = np.argmax(c)
            offset = n - 1
            
            A = [i for i in ONES if i + max_shift < n and h[i + max_shift] == 0]
            B = [j for j in ZEROS if j + max_shift < n and h[j + max_shift] == 1] 
            
            if A and B:
                i = random.choice(A)
                j = random.choice(B)
                assert h[i] == 1 and h[j] == 0
                # Swap indices i and j
                for q in range(i, n):
                    shift = q - i
                    c[shift + offset] -= 1 - h[q]
                for q in range(0, i+1):
                    shift = i - q
                    c[shift + offset] += h[q]
                for q in range(j, n):
                    shift = q - j
                    c[shift + offset] += h[q]
                for q in range(0, j+1):
                    shift = j - q
                    c[shift + offset] -= 1 - h[q]
                
                # Update h and index sets
                h[i] = 0
                h[j] = 1
                ONES.remove(i)
                ZEROS.remove(j)
                ONES.add(j)
                ZEROS.add(i)
                if np.max(c) < max_shift:
                    improved = True
                # Check if new c is max, so update this at start each time
                max_shift = np.argmax(c)
                attempt_count += 1
            else:
                # Random swaps to explore
                if ONES and ZEROS:
                    i = random.choice(list(ONES))
                    j = random.choice(list(ZEROS))
                    assert h[i] == 1 and h[j] == 0
                    # Swap indices i and j, update c values
                    altered = set()
                    update_c = lambda idx, delta: altered.add(idx)
                    for q in range(i, n):
                        shift = q - i
                        c[shift + offset] -= delta_h[q]
                        update_c(q)
                    for q in range(0, i+1):
                        shift = i - q
                        c[shift + offset] += delta_h[q]
                        update_c(q)
                    for q in range(j, n):
                        shift = q - j
                        c[shift + offset] += delta_h[q]
                        update_c(q)
                    for q in range(0, j+1):
                        shift = j - q
                        c[shift + offset] -= delta_h[q]
                        update_c(q)
                attempt_count += 1
        return h, compute_correlation(h)
    
    def upsample(h_coarse, n_new):
        # Linear interpolation
        n_old = len(h_coarse)
        old_x = np.linspace(0, 2, n_old, endpoint=False)
        new_x = np.linspace(0, 2, n_new, endpoint=False)
        h_new = np.interp(new_x, old_x, h_coarse)
        # Project it exactly to sum = n_new/2
        h_new = project_onto_feasible(h_new, n_new/2)
        return h_new
    
    # Main optimization loop
    for i, n in enumerate(ns):
        if time.time() - t0 > budget_s - safety_margin:
            break
        h_best_current = init_point(n)
        h_cont = h_best_current.copy()
        alpha = 5.0
        max_alpha = 200
        eta = 0.5 * 2.0 / n  # Step size
        momentum = 0.9
        h_prev = h_cont.copy()
        best_val = float('inf')
        c_best = None
        convergence_iters = 0
        max_convergence_years = 5000
        # Inner loop
        iteration = 0
        while iteration < 20000 and (time.time() - t0) < budget_s - safety_margin:
            if iteration % 100 == 0 and iteration > 0:
                if (time.time() - t0) > budget_s - safety_margin:
                    break
            
            c = compute_correlation(h_cont)
            smooth_val, weights = smoothed_objective_and_weights(c, alpha)
            g = grad_h(h_cont, weights)
            
            # Check if we can decrease by taking a step
            h_candidate = h_cont - eta * (g + momentum * (h_cont - h_prev))
            h_candidate = project_onto_feasible(h_candidate, n/2)
            
            c_candidate = compute_correlation(h_candidate)
            smooth_val_cand, _ = smoothed_objective_and_weights(c_candidate, alpha)
            
            if smooth_val_cand <= smooth_val:
                h_cont = h_candidate
                h_prev = h_cont.copy()
                eta *= 1.02  # Adaptive step
                convergence_iters = 0
                # Check true value
                true_val = np.max(c_candidate) * (2.0 / n)
                if true_val < best_val:
                    best_val = true_val
                    c_best = c_candidate.copy()
                    h_best_current = h_cont.copy()
            else:
                eta *= 0.5  # Line search backtracking
                convergence_iters += 1
                
            if iteration % 20 == 0 and alpha < max_alpha:
                alpha *= 1.2
            iteration += 1
            
            if convergence_iters > max_convergence_years:
                break
        
        # Convert to binary
        h_bin = h_best_current.copy()
        perm_indices = np.argsort(h_bin)[::-1]
        n_half = int(n / 2)
        h_bin[perm_indices[:n_half]] = 1
        h_bin[perm_indices[n_half:]] = 0
        
        # Round to match sum
        actual_sum = np.sum(h_bin)
        diff = actual_sum - n_half
        if abs(diff) > 1e-10:
            if diff > 0:
                # Remove one "1"
                idx_ones = np.where(h_bin == 1)[0]
                to_zero = np.random.choice(idx_ones)
                h_bin[to_zero] = 0
            else:
                # Add one "1"
                idx_zeros = np.where(h_bin == 0)[0]
                to_one = np.random.choice(idx_zeros)
                h_bin[to_one] = 1
                
        h_after_pruning = h_bin.copy()
        # Binary polishing using O(n) swap schemes
        time_left = budget_s - (time.time() - t0) - 1.0
        if time_left > 0.1:
            h_final, c_final = local_binary_search(h_after_pruning, compute_correlation(h_after_pruning), time_left)
            t_val = np.max(c_final) * (2.0 / n)
            # If final is better than current best either globally or for this resolution, update
            if t_val < best_c5:
                best_h = h_final
                best_c5 = t_val
                best_n = n
        else:
            t_val = np.max(compute_correlation(h_after_pruning)) * (2.0 / n)
            if t_val < best_c5:
                best_c5 = t_val
                best_h = h_after_pruning
                best_n = n
        if i + 1 < len(ns):
            # Use this best result to initialize next problem
            h_prev_size = ns[i+1]
            if h_prev_size > n:
                h_upscaled = upsample(h_final if not (best_h is None) else h_after_pruning, h_prev_size)
                h_best_current = h_upscaled.copy()
                  
    return (best_h.tolist() if best_h is not None else None, best_c5, best_n)
