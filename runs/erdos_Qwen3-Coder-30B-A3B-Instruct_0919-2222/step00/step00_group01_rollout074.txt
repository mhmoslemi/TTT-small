def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given step function h"""
    # Compute the overlap integral for all shifts
    # Using np.correlate with mode='full' gives us all shifts
    correlation = np.correlate(h, 1-h, mode='full')
    # Extract the relevant part (shifts from -1 to 1)
    # The full correlation has length 2*n - 1
    n = len(h)
    # We want the maximum over all valid shifts (0 to 2 in original domain)
    # Shifts correspond to indices n-1 to 2*n-2 inclusive
    max_correlation = np.max(correlation[n-1:])
    return max_correlation * dx
