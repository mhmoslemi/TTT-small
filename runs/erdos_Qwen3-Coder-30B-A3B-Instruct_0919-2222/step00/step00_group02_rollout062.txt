def evaluate_overlap(h):
    """Compute C_5 = max_k int h(x)(1-h(x+k)) dx for step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_correlation = corr.max()
    return max_correlation * dx

def get_worst_shift_indices(h):
    """Return indices of mismatched pairs for the worst shift."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    k_best = corr.argmax() - (len(h) - 1)
    max_correlation = corr.max()
    return k_best, max_correlation

def swap_mismatched_pairs(h, k_best):
    """Try to swap a 1 and 0 that reduce mismatch for shift k_best."""
    n = len(h)
    # Compute mismatched positions
    if k_best >= 0:
        m1 = [i for i in range(n) if h[i] == 1 and (i + k_best < n and h[i + k_best] == 0)]
        m0 = [j for j in range(n) if h[j] == 0 and (j + k_best < n and h[j + k_best] == 1)]
    else:
        m1 = [i for i in range(n) if h[i] == 1 and (i - k_best >= 0 and h[i - k_best] == 0)]
        m0 = [j for j in range(n) if h[j] == 0 and (j - k_best >= 0 and h[j - k_best] == 1)]
    if m1 and m0:
        i = random.choice(m1)
        j = random.choice(m0)
        return i, j
    else:
        # fallback to random balanced swap
        ones = [i for i in range(n) if h[i] == 1]
        zeros = [j for j in range(n) if h[j] == 0]
        if ones and zeros:
            i = random.choice(ones)
            j = random.choice(zeros)
            return i, j
    return None, None

def binary_local_search(h_init, max_time_s):
    """Directed binary local search based on worst-shift swaps."""
    start_time = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_count = np.correlate(h, 1 - h, mode='full').max()
    n = len(h)
    iters = 0
    consecutive_no_improvement = 0
    max_iters_per_cycle = 2 * n
    cycle_count = 0
    while time.time() - start_time < max_time_s:
        k_best, cur_count = get_worst_shift_indices(h)
        if cur_count <= best_count:
            best_count = cur_count
            best_h = h.copy()
        if cur_count == 0:
            break
        # Attempt swaps
        i, j = swap_mismatched_pairs(h, k_best)
        if i is not None:
            h[i], h[j] = h[j], h[i]
            cur_count = np.correlate(h, 1 - h, mode='full').max()
            if cur_count < best_count:
                best_count = cur_count
                best_h = h.copy()
            consecutive_no_improvement = 0
        else:
            consecutive_no_improvement += 1
        iters += 1
        if consecutive_no_improvement > n:
            break 
        # Random swap if stuck
        if iters % 100 == 0 and time.time() - start_time < max_time_s:
            ones = [i for i in range(n) if h[i] == 1]
            zeros = [j for j in range(n) if h[j] == 0]
            if ones and zeros:
                i = random.choice(ones)
                j = random.choice(zeros)
                h[i], h[j] = h[j], h[i]
            consecutive_no_improvement = 0 
    return best_h, best_count

def simulated_annealing(h_init, max_time_s):
    """Basic simulated annealing with exponential cooling."""
    start_time = time.time()
    h = h_init.copy()
    T = 0.5
    T_min = 1e-4
    alpha = 0.95
    max_iters = 5 * len(h) * np.log(len(h))
    iters = 0
    best_h = h.copy()
    best_count = np.correlate(h, 1 - h, mode='full').max()
    while time.time() - start_time < max_time_s and T > T_min and iters < max_iters:
        # Proposal: random balanced swap
        ones = [i for i in range(len(h)) if h[i] == 1]
        zeros = [j for j in range(len(h)) if h[j] == 0]
        if ones and zeros:
            i = random.choice(ones)
            j = random.choice(zeros)
            # Compute delta in overlap
            old_corr = np.correlate(h, 1 - h, mode='full')
            h[i], h[j] = h[j], h[i]
            new_corr = np.correlate(h, 1 - h, mode='full')
            
            # Since we're minimising max overlap, accept if better or with prob
            delta = new_corr.max() - old_corr.max()
            if delta <= 0 or random.random() < np.exp(-delta / T):
                # accept
                if new_corr.max() < best_count:
                    best_count = new_corr.max()
                    best_h = h.copy()
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]
            
        T *= alpha
        iters += 1
    return best_h, best_count

def generate_balanced_binary(n):
    """Generate a balanced binary vector."""
    h = np.zeros(n, dtype=int)
    ones = random.sample(range(n), n // 2)
    for idx in ones:
        h[idx] = 1
    return h

def binary_search(n, max_time_s):
    """Perform multiple binary runs for n, return best."""
    h_best = None
    count_best = float('inf')
    start_time = time.time()
    restarts = 5
    while time.time() - start_time < max_time_s - 1 and restarts > 0:
        h = generate_balanced_binary(n)
        # deterministic descent
        h1, c1 = binary_local_search(h, max_time_s / 4)
        # annealing
        h2, c2 = simulated_annealing(h1, max_time_s / 4)
        if c2 < count_best:
            count_best = c2
            h_best = h2.copy()
        restarts -= 1
    if h_best is None:
        h = generate_balanced_binary(n)
        h_best, count_best = binary_local_search(h, max_time_s)
    return h_best, count_best

def shade_box_projection(v, target_sum):
    """Projected gradient for balancing h subject to 0/1 constraint."""
    # Water-filling to project onto sum=0.5*n and unit box
    v = np.array(v)
    m = len(v)
    sum_v = np.sum(v)
    if abs(sum_v - target_sum) < 1e-10:
        return v.copy()
    # Simple version: scale to target sum
    scale = target_sum / sum_v
    x = v * scale
    # clamp to [0,1]
    x = np.clip(x, 0, 1)
    # re-scale to sum = target_sum
    scale = target_sum / np.sum(x)
    x = x * scale
    return x

def pw_multiply(v1, v2, k):
    """Compute pairwise product with shift k."""
    n = len(v1)
    s = np.zeros(n)
    if k >= 0:
        for i in range(min(n, n - k)):
            s[i] = v1[i] * (1 - v2[i + k])
    else:
        for i in range(min(n, n + k)):
            s[i] = v1[i] * (1 - v2[i - k])
    return s

    best_h = h.copy()

            break

            o = corr
            o_max = o.max()
            if o_max < best_c:
                best_c = o_max
                best_h = h.copy()
            # Softmax-scaled gradients
            exp_vals = np.exp(alpha * (o - o_max))
            p_weights = exp_vals / np.sum(exp_vals)
            
            # Gradient
            grad = np.zeros(n)
            for k in range(-n + 1, n):
                if k >= 0:
                    g_k = np.zeros(n)
                    g_k[:min(n, n - k)] = (1 - h[k:])
                else:
                    g_k = np.zeros(n)
                    g_k[-k:] = (1 - h[:-k])
                grad += p_weights[k + n - 1] * g_k
                
            # update via projected gradient descent  
            h_new = h - eta_0 / np.sqrt(t + 1) * grad
            h_new = np.clip(h_new, 0, 1)
            # enforced sum
            h_new = shade_box_projection(h_new, target_sum)
            h = h_new
            iters += 1
    
    return best_h, best_c
