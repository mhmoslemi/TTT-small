import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract relevant part (k from 0 to 2)
    c5_values = corr[n-1:] * dx
    return np.max(c5_values)

def initial_h_values():
    # This is a standard construction from the problem
    n = 45
    h = np.zeros(n)
    # Simple step function with alternating pattern
    for i in range(n):
        if i < n // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    # Normalize to match constraint
    h = h * 2 / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function to minimize
    def obj(h):
        return evaluate_c5(h)
    
    # Constraint function for integral constraint
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Optimization
    constraint = {'type': 'eq', 'fun': constraint_func}
    
    # Use multiple restarts to avoid local optima
    best_h = h0.copy()
    best_c5 = obj(best_h)
    
    iter_num = 0
    while time.time() - start_time < budget_s - 1.0:
        iter_num += 1
        # Try different random initializations
        h_init = np.random.rand(n_points)
        h_init = h_init / np.sum(h_init) / dx  # Normalize
        h_init = np.clip(h_init, 0, 1)
        
        # Project onto feasible set (some projection for any errors)
        h_init = h_init * 2 / np.sum(h_init)
        h_init = np.clip(h_init, 0, 1)
        
        # Simple gradient descent-like optimization
        # This is a hacky approach - for a proper solution,
        # one would need to use a real optimizer with gradients
        # For now, let's do a variation of descent with simulated annealing approach
        
        # Local search
        for i in range(100):
            if time.time() - start_time > budget_s - 1.0:
                break
            # Perturb one element
            idx = np.random.randint(n_points)
            old_val = h_init[idx]
            # Try small change
            delta = (np.random.rand() - 0.5) * 0.1
            new_val = old_val + delta
            new_val = np.clip(new_val, 0, 1)
            h_init[idx] = new_val
            
            # Adjust other values to maintain constraint
            sum_diff = new_val - old_val
            if abs(sum_diff) > 1e-10:
                # Compensate in other variables
                other_indices = [j for j in range(n_points) if j != idx]
                other_idx = np.random.choice(other_indices)
                h_init[other_idx] = np.clip(h_init[other_idx] - sum_diff * 0.5, 0, 1)
            
            # Check if it's better
            c5_new = obj(h_init)
            if c5_new < best_c5:
                best_h = h_init.copy()
                best_c5 = c5_new
        
        # Slightly randomized: try to spread points more
        # Try to find a better local optimum periodically
        if iter_num % 3 == 0:
            # Replace some values randomly to explore
            for i in range(n_points // 5):
                idx = np.random.randint(n_points)
                h_init[idx] = np.random.rand()
        
        # Small random modification to escape
        for i in range(5):
            idx = np.random.randint(n_points)
            h_init[idx] = np.clip(h_init[idx] + (np.random.rand() - 0.5) * 0.05, 0, 1)
        
        # Keep best
        c5_current = obj(h_init)
        if c5_current < best_c5:
            best_h = h_init.copy()
            best_c5 = c5_current
            
    return (best_h, best_c5, n_points)
