def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for a given step function h."""

    corr = signal.correlate(h, 1 - h, mode="full")
    # Scale by dx
    corr *= dx
    # Return maximum correlation
    return np.max(corr)
