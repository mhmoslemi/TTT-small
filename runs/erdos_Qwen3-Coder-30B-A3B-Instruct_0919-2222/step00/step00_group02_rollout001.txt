import numpy as np
import time
import random

def evaluate_h(h):
    """Return the maximum overlap integral C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    O_counts = np.correlate(h, 1 - h, mode="full")
    max_O = O_counts.max()
    return max_O * dx

def max_overlap_counts(h):
    """Return the maximum overlap count (integer) for a given step function h."""
    O_counts = np.correlate(h, 1 - h, mode="full")
    return O_counts.max()

def random_balanced_binary(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    ones = rng.sample(range(n), n // 2)
    h[ones] = 1
    return h

def project_onto_box_and_sum(v, target):
    """Project vector v onto box [0,1] and sum constraint."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0, 1)

def binary_anneal(n, time_limit, rng, target_max):

    cur_val = max_overlap_counts(h)
    best_val = cur_val

            break

        q = rng.choice(zeros)

                if best_val <= target_max:
                    # Early exit
                    return best_h, best_val
        else:
            h[p], h[q] = 1, 0
