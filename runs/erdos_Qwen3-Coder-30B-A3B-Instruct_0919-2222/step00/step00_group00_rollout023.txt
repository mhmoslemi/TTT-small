import numpy as np
from scipy.optimize import differential_evolution
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlations = np.correlate(h, 1 - np.array(h), mode="full")
    # The overlap integral is the maximum of these correlations
    max_overlap = np.max(correlations) * dx
    return max_overlap

def initial_h_values():
    """Return the initial construction"""
    # Generate the initial construction (n=45 samples)
    n = 45
    h = np.zeros(n)
    dx = 2.0 / n
    # Simple initial construction
    for i in range(n):
        x = i * dx
        if x < 1.0:
            h[i] = 1.0
        else:
            h[i] = 0.0
    # Adjust to satisfy constraint: integral = 1
    h = h * (1.0 / np.sum(h) * dx)
    return h

def obj_func(h):
    """Objective function to minimize"""
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction
    h_init = initial_h_values()
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint that integral should equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Optimization using differential evolution
    result = differential_evolution(
        obj_func,
        bounds,
        seed=seed,
        maxiter=int(1000 * budget_s / 600),  # Scale based on budget
        popsize=10,
        mutation=(0.5, 1.0),
        recombination=0.7,
        atol=1e-6,
        tol=1e-6
    )
    
    # Get final solution
    h_final = result.x
    
    # Ensure the constraint is satisfied
    h_final = h_final * (1.0 / (np.sum(h_final) * dx))
    
    # Evaluate the final solution
    c5_bound = evaluate_erdos_solution(h_final)
    
    # Early termination if budget expires
    elapsed = time.time() - start_time
    if elapsed >= budget_s * 0.95:
        return (h_final, c5_bound, n_points)

    # Try a local optimization approach as well
    # Create a more refined search with gradient-based method
    # But since we're limited in time and resources, we'll stick with DE
    
    return (h_final, c5_bound, n_points)
