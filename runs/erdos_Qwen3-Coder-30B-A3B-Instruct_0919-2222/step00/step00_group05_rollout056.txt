import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Read the initial solution
    h0 = np.array(initial_h_values)
    n = len(h0)
    M = n / 2.0
    dx = 2.0 / n
    
    def evaluate_solution(h):
        # Use the same computation as the evaluator
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # 1. Solve LP relaxation
    if time.time() - start_time > budget_s * 0.95:
        return list(h0), evaluate_solution(h0), n

    # Build model
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Convex hull constraints for y_{i,s}
    # y_{i,s} <= h_i
    # y_{i,s} <= h_{i+s}
    # y_{i,s} >= h_i + h_{i+s} - 1
    # y_{i,s} >= 0
    
    # Need to consider all valid s values from -n+1 to n-1
    # But avoid spurious constraints where indices are out of bounds
    # Instead, keep track of actual correlations:
    # For each s in [-n+1, n-1], define Q_s
    # We'll compute these explicitly in the solve function
    # Add the auxiliary variables directly for the pairs
    
    # List of constraints
    constraints = []
    constraints.append(cp.sum(h) == M)
    constraints.append(t >= 0)
    
    # Precompute all valid (i,s) pairs
    # For each s from -n+1 to n-1, valid i indices are those where i+s is in [0,n-1]
    s_range = list(range(-n + 1, n))
    y_vars = {}
    y_vars_dims = {}
    for s in s_range:
        i_vals = []
        for i in range(n):
            ip = i + s
            if 0 <= ip < n:
                i_vals.append(i)
        if len(i_vals) > 0:
            y_vars[s] = cp.Variable(len(i_vals))
            y_vars_dims[s] = i_vals
    
    # Add McCormick constraints
    for s in s_range:
        if s not in y_vars:
            continue
        i_vals = y_vars_dims[s]
        y = y_vars[s]
        for k, i in enumerate(i_vals):
            ip = i + s
            # y_k <= h_i, y_k <= h_ip, y_k >= h_i + h_ip - 1, y_k >= 0
            constraints.append(y[k] <= h[i])
            constraints.append(y[k] <= h[ip])
            constraints.append(y[k] >= h[i] + h[ip] - 1)
            constraints.append(y[k] >= 0)
    
    # Add shift bounds
    for s in s_range:
        if s not in y_vars:
            continue
        i_vals = y_vars_dims[s]
        y = y_vars[s]
        total_y = cp.sum(y)
        # Total_y >= M - t
        constraints.append(total_y >= M - t)
            
    # Objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try solving with ECOS (may fail and we'll provide fallback later)
    try:
        # Warm start with initial solution, but probably not the right way for CVXPY
        # Rather just wait for it to converge
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000)
    except Exception as e:
        pass
    
    h_star = None
    if prob.status == cp.OPTIMAL:
        h_star = h.value
    else:
        # Fallback to initial solution
        h_star = h0
        
    # Convert to feasible discrete function
    # From the relaxed solution h_star, generate a set of candidates
    
    # 1. Check if it's binary already
    h_star_rounded = np.round(h_star)
    if np.allclose(h_star_rounded, h_star, atol=1e-8) and np.all((h_star_rounded == 0) | (h_star_rounded == 1)):
        h_best = h_star_rounded
        best_obj = evaluate_solution(h_best)
    else:
        # Make multiple random samples
        R = 200  # Number of random trials
        candidates = []
        for i in range(R):
            # Sample with probability h_star
            if time.time() - start_time > budget_s * 0.9:
                break
            h_sample = (np.random.random(n) < h_star).astype(float)
            # Adjust to exactly M ones
            ones_indices = np.where(h_sample == 1)[0]
            zeros_indices = np.where(h_sample == 0)[0]
            while len(ones_indices) > M:
                idx = np.random.choice(ones_indices)
                h_sample[idx] = 0
                ones_indices = np.where(h_sample == 1)[0]
            while len(ones_indices) < M:
                idx = np.random.choice(zeros_indices)
                h_sample[idx] = 1
                zeros_indices = np.where(h_sample == 0)[0]
            assert abs(np.sum(h_sample) - M) < 1e-10
            candidates.append((h_sample, evaluate_solution(h_sample)))
        
        # Determine best among random samples
        h_best = min(candidates, key=lambda x: x[1])[0]
        best_obj = min(candidates, key=lambda x: x[1])[1]
        
    # Local polish via targeted swapping 
    def get_correlations(h):
        corr = correlate(h, 1 - h, mode='full') * dx
        return corr
    
    def local_search(h_candidate, max_iter=500):
        h_polish = h_candidate.copy()
        corr = get_correlations(h_polish)
        s_max = np.argmax(corr)
        current_c5 = np.max(corr)
        iteration = 0
        while iteration < max_iter and time.time() - start_time < budget_s * 0.9:
            improved = False
            # Find one of the worst shifts
            s_max = np.argmax(corr)
            # Iterate through possible swaps
            # For now, this is a simplified approach
            for i in range(n):
                j = (i + s_max) % n
                # Check if we can improve
                if h_polish[i] == 1 and h_polish[j] == 0:
                    # Try flip of i and i+s in current shift
                    # Now check all shifts for change
                    delta = 0
                    # Check how much the worst correlation can change
                    # For s_max 
                    a = h_polish[i] * (1 - h_polish[j])
                    b = h_polish[i] * (1 - h_polish[i+s_max])
                    c = h_polish[j] * (1 - h_polish[i+s_max])
                    # We only care if s_max already has contribution of 1
                    # (i has 1, j has 0 => i contributes to s_max correlation)
                    if h_polish[i] == 1 and h_polish[j] == 0:
                        # Now imagine swapping them
                        # Update the total
                        new_h = h_polish.copy()
                        new_h[i] = 0
                        new_h[j] = 1
                        new_corr = get_correlations(new_h)
                        new_max = np.max(new_corr)
                        if new_max < current_c5 and new_max < 0.3807:  # And below target
                            h_polish = new_h.copy()
                            corr = new_corr
                            current_c5 = new_max
                            improved = True
                            break
            if not improved:
                break
            iteration += 1
        return h_polish, current_c5
        
    # Run polishing
    h_polish, c5_polish = local_search(h_best)
    best_obj = min(best_obj, c5_polish)
    
    # Possibly try MIP if we haven't hit the bar yet and got fractional
    if best_obj > 0.3807 and not np.allclose(h_star_rounded, h_star, atol=1e-8):
        # Use a mixed integer solver via cvxpy again to get better bounds
        h_binary = cp.Variable(n, integer=True)
        t_mip = cp.Variable()
        constraints_mip = [
            cp.sum(h_binary) == M,
            t_mip >= 0,
        ]
        
        # Add constraints
        for s in s_range:
            if s not in y_vars:
                continue
            i_vals = y_vars_dims[s]
            y = cp.Variable(len(i_vals))
            y_vars[s] = y
            # And add all the relevant McCormick constraints
            ti = 0
            for i in i_vals:
                ip = i + s
                # y_k <= h_i, y_k <= h_ip, y_k >= h_i + h_ip - 1, y_k >= 0
                constraints_mip.append(y[ti] <= h_binary[i])
                constraints_mip.append(y[ti] <= h_binary[ip])
                constraints_mip.append(y[ti] >= h_binary[i] + h_binary[ip] - 1)
                constraints_mip.append(y[ti] >= 0)
                ti += 1
            constraints_mip.append(cp.sum(y) >= M - t_mip)
        
        prob_mip = cp.Problem(cp.Minimize(t_mip), constraints_mip)
        try:
            prob_mip.solve(solver=cp.CBC, verbose=False, max_seconds=budget_s * 0.5)
        except:
            pass
        # If MIP ran to completion and gave us a feasible solution
        if prob_mip.status == cp.OPTIMAL:
            h_binary_val = h_binary.value
            if h_binary_val is not None:
                # Must be integer anyway
                h_final = h_binary_val
                _, new_c5 = local_search(h_final, max_iter=200)
                if new_c5 < best_obj:
                    best_obj = new_c5
                    h_best = h_final
                    
    # Always return the best solution
    if hasattr(h_best, 'tolist'):
        h_final = h_best.tolist()
    else:
        h_final = list(h_best)
    
    if hasattr(best_obj, 'item'):
        c5_final = float(best_obj.item())
    else:
        c5_final = float(best_obj)
        
    return h_final, c5_final, n
