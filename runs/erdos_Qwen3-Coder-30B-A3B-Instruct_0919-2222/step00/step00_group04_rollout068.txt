import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    N = 120
    dx = 2.0 / N
    target_time = budget_s - 30  # Reserve 30 seconds for final processing
    
    # Compute shift matrices A_k
    A = [np.eye(N, k=k) + np.eye(N, k=-k) for k in range(N)]  # for k = 0 to N-1
    # For computational efficiency, only consider a sparse set of shifts
    shift_indices = list(range(0, N, 2))  # Every 2nd shift, use fewer samples

    # SDP variables
    h = Variable(N)
    X = Variable((N, N), symmetric=True)

    # Initial guess
    h0 = np.ones(N) * 0.5
    h0[0:20] = 0
    h0[20:40] = 1
    h0[40:60] = 0
    h0[60:80] = 1
    h0[80:100] = 0
    h0[100:120] = 1
    h0 = h0 / np.sum(h0) * N / 2  # Ensure mass constraint

    # Objective: maximize γ
    gamma = Variable()
    
    # Constraints
    constraints = [
        0 <= h,
        h <= 1,
        sum(h) == N / 2,
        X >> 0,  # X is PSD
    ]
    
    for i in shift_indices:
        Ak = np.eye(N, k=i) + np.eye(N, k=-i)  # Non-zero only on the k-th diagonal
        # Append \Tr(A_k * X) >= gamma
        constraints.append(trace(Ak @ X) >= gamma)
        # Also impose min
        track_eye = np.eye(N)
        M = np.zeros((N, N))
        for i in range(N):
            if i + i < N:
                M[i, i + i] = 1.0
        constraints.append(trace(M @ X) >= gamma)

    # Fix main diagonal entries of X to match h
    for i in range(N):
        constraints.append(X[i, i] == h[i])

    # Minimize 1 - gamma to maximize the minimum auto-correlation
    prob = Problem(Maximize(gamma), constraints)

    # Bisection loop on error bound
    tau_low = 0.35
    tau_high = 0.50
    min_C5 = 0.5
    best_h = h0
    
    while time.time() - start_time < target_time:
        # Initialize tau
        tau = (tau_low + tau_high) /2
        gamma_val = 1 - tau  # Since we maximize gamma to minimize (1-gamma)
        
        # Tighten the constraints slightly to take into account any numerical round-offs
        new_constraints = constraints + [gamma >= gamma_val]
        prob_new = Problem(Maximize(gamma), new_constraints)
        
        try:
            prob_new.solve(solver=ECOS, verbose=False)
        except:
            try:
                prob_new.solve(solver=SCS, verbose=False)
            except:
                # Fall back to a quick approximation
                pass

        if prob_new.status in ["optimal", "optimal_inaccurate"]:
            # Retrieved h subject to the constraints
            h_value = h.value
            if h_value is not None:
                # Project h to ensure it satisfies \int h dx = 1 and bounds
                h_projected = np.clip(h_value, 0, 1)
                total = np.sum(h_projected) * dx
                h_projected = h_projected * (1.0 / total) * 1.0
                h_projected = np.clip(h_projected, 0, 1)

                # Compute actual C5
                corr_val = np.correlate(h_projected, 1 - h_projected, mode='full') * dx
                max_overlap = np.max(corr_val)
                
                if max_overlap < min_C5:
                    min_C5 = max_overlap
                    best_h = h_projected.copy()
                
                # Narrow bracket
                tau_high = tau
            else:
                tau_low = tau
        else:
            # Infeasible
            tau_low = tau
            
        # Early exit if we're getting close to the target
        if min_C5 < 0.3807:
            break

    # Refined optimization with projected subgradient descent on the final h
    best_h_refined = best_h.copy()

    def compute_correlation(h_in):
        corr = np.correlate(h_in, 1 - h_in, mode='full') * dx
        return np.max(corr)

    def project_simplex(x, target_sum):
        # Waterfilling
        # Ensure x lies in [0, 1], and sum(x) == target_sum
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if np.abs(total - target_sum) < 1e-10:
            return x
        # Adjust all values using thresholding
        sorted_indices = np.argsort(x[::-1])  # Descending order
        cumsum = 0
        threshold = 0
        for i in range(len(x)):
            idx = sorted_indices[i]
            if cumsum + x[idx] <= target_sum:
                cumsum += x[idx]
            else:
                threshold = (target_sum - cumsum) / (len(x) - i)
                break
        result = np.maximum(x - threshold, 0)
        return result

    # Sub-gradient descent refinement
    max_iter = 2000
    alpha0 = 0.1
    for t in range(max_iter):
        if time.time() - start_time > target_time - 5:
            break
        corr_vals = np.correlate(best_h_refined, 1 - best_h_refined, mode='full') * dx
        k_star = np.argmax(corr_vals)

        g = np.zeros_like(best_h_refined)
        for i in range(len(best_h_refined)):
            if i <= len(best_h_refined) - 1 - k_star:
                g[i] += (1.0 - best_h_refined[i + k_star])
            if i >= k_star:
                g[i] -= best_h_refined[i - k_star]

        alpha = alpha0 / np.sqrt(t + 1)
        new_h = best_h_refined - alpha * g
        new_h = project_simplex(new_h, 1.0)
        best_h_refined = new_h

    # Final C5 calculation
    final_corr = np.correlate(best_h_refined, 1 - best_h_refined, mode='full') * dx
    final_C5 = np.max(final_corr)
    
    return (best_h_refined.tolist(), final_C5, N)
