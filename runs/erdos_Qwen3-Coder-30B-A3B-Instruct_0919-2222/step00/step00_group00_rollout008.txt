import time
import numpy as np
import cvxpy as cp
from cvxpy import ECOS_BB, CBC, GLPK
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None

    # Try different discretization sizes
    n_values = [80, 120, 160]
    
    for n in n_values:
        if time.time() - t_start > budget_s - 5:
            break
            
        dx = 2.0 / n
        K0 = max(1, n // 4)  # Start with small set of shifts
        
        # Initialize variables
        x = cp.Variable(n, boolean=True)
        t = cp.Variable()
        
        # Hard constraints
        constraints = [cp.sum(x) == n // 2]  # Exactly half entries are 1
        constraints.append(x[0] == 1)  # Symmetry breaking
        # Non-increasing constraint for leading block
        for i in range(n // 2 - 1):
            constraints.append(x[i] >= x[i+1])
            
        # Track y variables and constraints by shift
        y_vars = {}
        shift_constraints = []
        active_shifts = list(range(1, min(K0 + 1, n)))
        
        # Add first batch of constraints
        for k in active_shifts:
            y_vars[k] = cp.Variable(n - k, boolean=True)
            # McCormick envelopes
            for i in range(n - k):
                constraints.append(y_vars[k][i] <= x[i])
                constraints.append(y_vars[k][i] <= x[i+k])
                constraints.append(y_vars[k][i] >= x[i] + x[i+k] - 1)
                constraints.append(y_vars[k][i] >= 0)
            # Correlation constraint
            shift_constraints.append(cp.sum(x) - cp.sum(y_vars[k]) <= t)
        
        constraints.extend(shift_constraints)
        
        # Solve initial problem
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        # Update model with default solver and solve
        solver_options = {'solver': CBC}
        try:
            prob.solve(**solver_options, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                x_val = x.value.astype(int)
                t_val = t.value if t.value is not None else 0
                c5_bound = t_val * dx
                
                # Validate against full set of shifts
                full_correlations = correlate(x_val, 1 - x_val, mode='full')
                max_correlation = np.max(full_correlations)
                # Check if full validation needed
                if max_correlation <= t_val + 1e-6:
                    if c5_bound < best_c5:
                        best_c5 = c5_bound
                        best_h = x_val.astype(float)
                        best_n = n
                else:
                    # Refine with full shifts and keep going
                    pass
            else:
                # Fallback if infeasible
                continue
        except Exception:
            continue

        # Refinement through lazy constraint generation
        if time.time() - t_start > budget_s - 5:
            break
            
        additional_shifts_added = 0
        max_additional_shifts = 10
        while additional_shifts_added < max_additional_shifts:
            if time.time() - t_start > budget_s - 5:
                break
                
            # Compute full correlations and find worst
            full_correlations = correlate(x_val, 1 - x_val, mode='full')
            max_k = np.argmax(full_correlations)
            max_corr = full_correlations[max_k]
            
            # If the current bound is tight enough, we're done
            if max_corr <= t_val + 1e-6:
                break
                
            # Add violating shift
            k = max_k - (n - 1)  # Undo offset in correlation indexing
            if k > 0 and k not in active_shifts:
                active_shifts.append(k)
                additional_shifts_added += 1
                
                # Add new variable and constraints
                y_vars[k] = cp.Variable(n - k, boolean=True)
                for i in range(n - k):
                    constraints.append(y_vars[k][i] <= x[i])
                    constraints.append(y_vars[k][i] <= x[i+k])
                    constraints.append(y_vars[k][i] >= x[i] + x[i+k] - 1)
                    constraints.append(y_vars[k][i] >= 0)
                
                shift_constraints.append(cp.sum(x) - cp.sum(y_vars[k]) <= t)
                
                # Rebuild and re-solve
                constraints.extend(shift_constraints[-1:])
                prob = cp.Problem(cp.Minimize(t), constraints)
                
                try:
                    prob.solve(**solver_options, verbose=False)
                    if prob.status in ["optimal", "optimal_inaccurate"]:
                        x_val = x.value.astype(int)
                        t_val = t.value if t.value is not None else 0
                        c5_bound = t_val * dx
                        
                        if c5_bound < best_c5:
                            best_c5 = c5_bound
                            best_h = x_val.astype(float)
                            best_n = n
                    else:
                        pass
                except Exception:
                    break
                    
        # Post-processing with local swap
        if time.time() - t_start > budget_s - 5:
            break
        
        if best_h is not None:
            h_val = best_h.copy()
            # Run swaps until no more improvements
            while True:
                full_corr = correlate(h_val, 1 - h_val, mode='full')
                max_k_idx = np.argmax(full_corr)
                max_val = full_corr[max_k_idx]
                
                # If we're within tolerance, stop
                if max_val <= t_val + 1e-6:
                    break
                    
                # Identify set A and B for swapping
                k = max_k_idx - (n - 1)
                A = [i for i in range(n) if h_val[i] > 0.5 and i + k < n and h_val[i + k] < 0.5]
                B = [j for j in range(n) if h_val[j] < 0.5 and j + k < n and h_val[j + k] > 0.5]
                
                if not A or not B:
                    break
                    
                # Pick one pair to swap
                i = A[0]
                j = B[0]
                h_val[i] = 0
                h_val[j] = 1
                
                # Validate updated maximum
                full_corr_new = correlate(h_val, 1 - h_val, mode='full')
                max_val_new = np.max(full_corr_new)
                
                # If it worsened, swap it back
                if max_val_new >= max_val:
                    h_val[i] = 1
                    h_val[j] = 0
                else:
                    t_val = max_val_new
                    if t_val * dx < best_c5:
                        best_c5 = t_val * dx
                        best_h = h_val.copy()
                        best_n = n
                        
        # If best is not computed (encountered failure initially), skip to next
        if best_h is None:
            continue
            
    if best_h is None:
        # Fallback to a symmetric construction
        best_h = [1] * (n // 2) + [0] * (n // 2)
        best_h = [float(x) for x in best_h]
        best_c5 = 0.5 * dx  # naive upper bound for uniform distribution
        best_n = n

    return (best_h, best_c5, best_n)
