import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap constant C5 for given step function h"""
    # Ensure h is a numpy array
    h = np.array(h)
    # Compute the overlap integral using full correlation
    c = correlate(h, 1 - h, mode='full')
    # The integral is scaled by dx = 2 / len(h)
    dx = 2.0 / len(h)
    return np.max(c) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    end_time = start_time + budget_s * 0.99  # Leave some margin
    
    # Parameters
    n_points = 401  # Odd number for symmetry
    dx = 2.0 / n_points
    
    # Initialize the step function
    h_values = np.ones(n_points) * 0.5  # Start with uniform
    h_values = h_values.astype(float)
    
    # Evaluate initial solution
    c5_bound = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    best_c5 = c5_bound
    
    # Generate shapes to try
    def triangle_shape(x):
        return np.maximum(0, 1 - np.abs(x - 1))
    
    def cosine_shape(x, p):
        return np.cos(np.pi/2 * (x - 1)) ** p
    
    def polynomial_shape(x, a):
        return np.maximum(0, 1 - np.abs(x - 1)**a)

    # Try different shapes
    candidates = []
    
    # Triangle
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    h_tri = triangle_shape(x)
    h_tri = h_tri * (n_points/2) / np.sum(h_tri)
    h_tri = np.clip(h_tri, 0, 1)
    c5_tri = evaluate_erdos_solution(h_tri)
    candidates.append((h_tri.copy(), c5_tri, "triangle"))
    
    # Cosine shapes
    for p in [0.5, 1.0, 2.0, 4.0, 8.0]:
        h_cos = cosine_shape(x, p)
        h_cos = h_cos * (n_points/2) / np.sum(h_cos)
        h_cos = np.clip(h_cos, 0, 1)
        c5_cos = evaluate_erdos_solution(h_cos)
        candidates.append((h_cos.copy(), c5_cos, f"cos^{p}"))
        
    # Polynomial shapes
    for a in [1.0, 2.0, 3.0]:
        h_poly = polynomial_shape(x, a)
        h_poly = h_poly * (n_points/2) / np.sum(h_poly)
        h_poly = np.clip(h_poly, 0, 1)
        c5_poly = evaluate_erdos_solution(h_poly)
        candidates.append((h_poly.copy(), c5_poly, f"poly^{a}"))
    
    # Find the best among candidates
    for h,test_c5,name in candidates:
        if test_c5 < best_c5:
            best_h = h.copy()
            best_c5 = test_c5
    
    # Save initial best
    h_values = best_h.copy()
    c5_bound = best_c5
    
    # Local optimization using hill climbing
    max_iters = 200000  # Limit iterations for time budget
    epsilon = 0.05
    decay_rate = 0.99995  # Gradual decay of epsilon
    
    # Compute initial correlation for faster updates
    prev_c = correlate(h_values, 1 - h_values, mode='full')
    prev_max_c = np.max(prev_c)
    
    iterations = 0
    while time.time() < end_time and iterations < max_iters:
        # Choose random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
            
        # Determine delta amount to move
        delta_max = min(h_values[i], 1 - h_values[j], epsilon)
        if delta_max <= 0:
            continue
        delta = delta_max * np.random.random()
        
        # Propose new values
        h_new = h_values.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Adjust to maintain sum constraint
        sum_diff = np.sum(h_new) - n_points/2
        if abs(sum_diff) > 1e-10:
            # Adjust the first element
            h_new[0] -= sum_diff
            h_new = np.clip(h_new, 0, 1)
            
        # Evaluate new C5
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept if improvement
        if c5_new < c5_bound:
            h_values = h_new.copy()
            c5_bound = c5_new
            
            # Update best if needed
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
        # Gradually reduce epsilon
        epsilon *= decay_rate
        if epsilon < 1e-5:
            epsilon = 1e-5
            
        iterations += 1
        
        # Check time periodically
        if iterations % 1000 == 0 and time.time() > end_time - 5:
            break
            
    return (best_h.tolist(), best_c5, n_points)
