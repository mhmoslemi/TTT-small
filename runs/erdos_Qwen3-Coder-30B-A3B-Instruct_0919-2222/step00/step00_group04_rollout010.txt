import numpy as np
import time
import random

def project_to_simplex(v, m):
    """Project vector v onto the simplex with sum = m and bounds [0,1]"""
    n = len(v)
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_v)
    # Find threshold index
    idx = np.arange(1, n + 1)
    condition = sorted_v - (m - cumsum) / idx > 0
    if not np.any(condition):
        # All components should be zero
        return np.zeros_like(v)
    k = np.argmax(condition)
    # Compute the threshold
    tau = (cumsum[k] - m) / (k + 1)
    # Apply thresholding and clipping
    result = np.maximum(0, np.minimum(1, v - tau))
    return result

def compute_bound(h, dx):
    """Compute C5 bound from step function h"""
    # correlation via np.correlate with full mode
    # h and 1-h (both float64)
    h = np.asarray(h, dtype=np.float64)
    correlation = np.correlate(h, 1 - h, mode='full')
    # correlation array is of length 2*n - 1
    # max over all shifts corresponds to max in correlation array
    # index = n-1 for k=0
    max_correlation = np.max(correlation)
    return max_correlation * dx

def compute_gradient(h, dx):
    """Compute subgradient for the current max shift"""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    # Compute correlation array
    correlation = np.correlate(h, 1 - h, mode='full')
    # Find all indices where the maximum is achieved
    max_val = np.max(correlation)
    max_indices = np.where(np.abs(correlation - max_val) < 1e-10)[0]
    # for each such index k, we compute partial derivative wrt h_i
    # In full correlation, index i corresponds to shift k = i - (n-1)
    g = np.zeros(n)
    # compute gradient w.r.t. to each h_i
    for k_idx in max_indices:
        shift = k_idx - (n - 1)
        if shift >= 0:
            # We take elements from 0 to n-1-shift such that i+k is valid
            # i ranges from 0 to n-1-shift
            # So g_i += (1 - h_{i+shift}) for i = 0 to n-1-shift
            # for i in [0, n-1-shift), add to g[i]
            g[:n-shift] += 1 - h[shift:]
        else:
            # shift < 0, we look at indices where h_{i-shift} appears
            # i ranges from 0-shift to n-1
            # since shift is negative, i from -shift to n-1
            # so g for i from -shift to n-1, we subtract h_{i-shift}
            # i from 0 to n-1-shift: where h_{i-shift} valid
            g[-shift:] -= h[:-shift]
    return g

def gradient_step(h, step_size, dx):
    # gradient computation
    g = compute_gradient(h, dx)
    h_new = h - step_size * g
    # project onto the constrained set
    h_new = project_to_simplex(h_new, len(h) / 2)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Global params
    n_points = 200
    dx = 2.0 / n_points
    # Set random seed
    np.random.seed(seed)
    random.seed(seed)
    
    # Initialize
    # initial_h_values is global variable
    try:
        from initial_h_values import initial_h_values
        # Upsample this to get n_points 
        if len(initial_h_values) <= n_points:
            # tile it to full size
            reps = (n_points + len(initial_h_values) - 1) // len(initial_h_values)
            h = np.tile(initial_h_values, reps)[:n_points]
            # Normalize by scaling such that sum = n_points / 2
            h *= (n_points / 2) / np.sum(h)
        else:
            # downsample with interpolation?
            h = np.interp(np.linspace(0, len(initial_h_values), n_points), 
                          np.arange(len(initial_h_values)), initial_h_values)
            # rescale to ensure sum = n_points/2
            h *= (n_points / 2) / np.sum(h)
    except ImportError:
        # Fallback: random construction
        h = np.random.rand(n_points)
        h *= (n_points / 2) / np.sum(h)
    h = np.asarray(h)
    
    # Local optimization
    best_bound = compute_bound(h, dx)
    best_h = h.copy()
    
    iteration = 0
    start_time = time.time()
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        # adaptive step size
        step_size = 0.1 / np.sqrt(iteration + 1)
        
        h = gradient_step(h, step_size, dx)
        
        current_bound = compute_bound(h, dx)
        
        # update best solution
        if current_bound < best_bound:
            best_bound = current_bound
            best_h = h.copy()
        
        # Occasionally perturb slightly to avoid local optima
        if iteration % 500 == 0:
            # Add slight Gaussian noise
            h += np.random.normal(0, 0.001, n_points)
            # project to feasible set
            h = project_to_simplex(h, n_points / 2)
            # recompute best if better
            current_bound = compute_bound(h, dx)
            if current_bound < best_bound:
                best_bound = current_bound
                best_h = h.copy()
    
    return best_h.tolist(), best_bound, n_points
