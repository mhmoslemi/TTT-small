import numpy as np
import time

from math import log

def evaluate(h):
    """Evaluates C5 = max of correlation integrals."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_capped_simplex(v, M):
    """Projects vector v onto [0,1]^n with sum == M using O(n log n) algorithm."""
    n = len(v)
    # Sort values descending and track indices
    idx_sorted = np.argsort(-v)
    v_sorted = v[idx_sorted]
    # Compute cumulative sums
    cumsum_sorted = np.cumsum(v_sorted)
    # Find threshold index
    thresh_idx = -1
    for i in range(n):
        if v_sorted[i] <= (M - cumsum_sorted[i]) / (n - i):
            thresh_idx = i
        else:
            break
    if thresh_idx == -1:
        theta = (cumsum_sorted[-1] - M) / n
        thres = 0.0
    else:
        theta = (cumsum_sorted[thresh_idx - 1] - M) / (n - thresh_idx + 1)
        thres = v_sorted[thresh_idx - 1]
    # Create final projection
    h = np.zeros_like(v)
    for i in range(n):
        val = v[i]
        if val > thres:
            h[i] = min(1.0, max(0.0, val - theta))
        else:
            h[i] = min(1.0, max(0.0, val))
    return h

def precompute_shifts(n):
    """Returns a list of non-empty shift index mappings."""
    # shift s: list of (i, i+s) pairs that are both valid indices [0,n-1]
    shift_maps = {}
    for s in range(-n+1, n):
        pairs = [(i, i + s) for i in range(n) if 0 <= i + s < n]
        if pairs:
            shift_maps[s] = pairs
    return shift_maps

def correlations(h, shift_maps):
    """Compute all correlation values C_s(h) for valid shifts."""
    n = len(h)
    corr_vals = []
    for s, pairs in shift_maps.items():
        c = 0.0
        for i, j in pairs:
            c += h[i] * (1.0 - h[j])
        corr_vals.append(c)
    return np.array(corr_vals)

def smooth_obj_and_grad(h, alpha, shift_maps):
    """Compute smooth log-sum-exp approximation and gradient."""
    C = correlations(h, shift_maps)
    # numerical stabilization
    C_max = np.max(C)
    if C_max > 1e3:
        C_stable = C - C_max
    else:
        C_stable = C
    exp_C = np.exp(alpha * C_stable)
    Z = np.sum(exp_C)
    w = exp_C / Z
    Phi_alpha = (1.0 / alpha) * (np.log(Z) + C_max)
    # Gradient computation
    grad = np.zeros_like(h)
    for s, pairs in shift_maps.items():
        sign = 1
        w_s = w[s + n - 1]  # index into w array
        for i, j in pairs:
            grad[i] += w_s * (1.0 - h[j])
            grad[j] -= w_s * h[i]
    return Phi_alpha, grad

def run_descent(h0, alpha_schedule, eta0, max_iter, shift_maps, budget_s):
    """Projected gradient descent with continuation."""
    h = h0.copy()
    n = len(h)
    M = n // 2
    eta = eta0
    best_C = float('inf')
    best_h = h.copy()
    start = time.time()
    for alpha in alpha_schedule:
        # reset step size
        eta = eta0
        for k in range(max_iter):
            if time.time() - start > budget_s * 0.95:
                break
            Phi, g = smooth_obj_and_grad(h, alpha, shift_maps)
            v = h - eta * g
            h = project_to_capped_simplex(v, M)
            C = evaluate(h)
            if C < best_C:
                best_C = C
                best_h = h.copy()
            # adaptive step size
            if k < max_iter - 1:
                C_prev = evaluate(h)
                if C_prev > C:  # backward line search
                    eta *= 0.9
    return best_h, best_C

def random_binary(n, M):
    """Generate a random binary vector with exactly M ones."""
    indices = np.random.permutation(n)[:M]
    h = np.zeros(n)
    h[indices] = 1.0
    return h

def random_fractional(n, M):
    """Generate a random fractional vector with sum = M by Dirichlet."""
    # sample from Dirichlet(1, 1, ..., 1)
    sample = np.random.dirichlet([1.0] * n)
    # scale to sum = M
    h = sample * M
    return h
