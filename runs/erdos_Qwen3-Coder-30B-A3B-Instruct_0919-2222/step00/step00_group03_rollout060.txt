import time
import numpy as np
import pulp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def solve_erdos_milp(n, time_limit):
    """Solve the Erdos minimum overlap problem using MILP with binary h variables."""
    start_time = time.time()
    
    # Create the problem
    prob = pulp.LpProblem("ErdosMinOverlap", pulp.LpMinimize)
    
    # Variables: h_i binary, T continuous
    h = pulp.LpVariable.dicts('h', list(range(n)), lowBound=0, upBound=1, cat='Binary')
    T = pulp.LpVariable('T', lowBound=0, cat='Continuous')
    
    # Constraint: sum h_i = n/2
    prob += pulp.lpSum([h[i] for i in range(n)]) == n/2
    
    # Linearization constraints for products h_i * h_{i+d}
    # For each shift d from 1 to n-1 (including d=0 implicitly by constraints)
    # We compute y_i_d = h_i * h_{i+d} and add McCormick envelope constraints
    y_vars = {}
    
    for d in range(1, n):  # Skip d=0 for now; handle later
        for i in range(n - d):
            y = pulp.LpVariable(f"y_{i}_{d}", lowBound=0, upBound=1, cat='Continuous')
            y_vars[(i, d)] = y
            # McCormick envelope constraints:
            prob += y <= h[i]
            prob += y <= h[i+d]
            prob += y >= h[i] + h[i+d] - 1
    
    # Shift constraints: sum over y >= n/2 - T
    for d in range(1, n):
        y_sum = pulp.lpSum([y_vars[(i, d)] for i in range(n - d)])
        prob += y_sum >= n/2 - T
    
    # Objective: minimize T
    prob += T
    
    # Solve
    solver = pulp.PULP_CBC_CMD(msg=0, maxSeconds=time_limit)
    prob.solve(solver)
    
    status = pulp.LpStatus[prob.status]
    if status != 'Optimal':
        return None, None
    
    # Retrieve solution
    h_vals = np.array([pulp.value(h[i]) for i in range(n)], dtype=float)
    t_val = pulp.value(T)
    
    return h_vals, t_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # We'll try various n values from large to small to prioritize finding a better solution sooner
    n_values = [200, 180, 160, 140, 120, 100, 80, 60, 50]  # Larger first to get better bounds quickly
    
    # Start with initial_h_values from the library as a strong baseline
    try:
        init_h = np.array(initial_h_values)  # Likely float array
        init_c5 = evaluate_erdos_solution(init_h)
        if init_c5 < best_c5:
            best_c5 = init_c5
            best_h = init_h.copy()
            best_n = len(init_h)
        print(f"Initial solution n={len(init_h)}, C5={init_c5:.6f}")
    except:
        init_h = None
    
    i = 0
    while i < len(n_values):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        n = n_values[i]
        i += 1
        
        # Compute available time for this solve
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 5:
            break
            
        # Solve MILP for specific n
        h, t = solve_erdos_milp(n, min(remaining_time, 300))  # Cap individual solve at 300s
        
        if h is not None and t is not None:
            # Compute actual C5 bound
            c5_bound = evaluate_erdos_solution(h)  # This corresponds to t * dx
            
            print(f"Solved n={n}, C5 from LP bound={t:.6f}, Actual C5={c5_bound:.6f}")
            
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h.copy()
                best_n = n
                
                if best_c5 < 0.379:
                    print(f"Found target bound C5={best_c5:.6f}")
                    break
        
    # If we found nothing better than initial, return initial
    if best_h is None:
        best_h = init_h
        best_c5 = evaluate_erdos_solution(best_h)
        best_n = len(best_h)
        
    return (best_h.tolist(), best_c5, best_n)
