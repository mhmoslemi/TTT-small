import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except:
        # Fallback to a simple construction
        n_points = 200
        h = np.ones(n_points) * 0.5  # Uniform distribution
    
    dx = 2.0 / n_points
    
    def evaluate_c5(h_vals):
        h_arr = np.array(h_vals)
        conv = np.correlate(h_arr, 1 - h_arr, mode="full") * dx
        return np.max(conv)
    
    # Define objective
    def objective(h_vals):
        return evaluate_c5(h_vals)
    
    # Constraints
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    def constraint_bounds(h_vals):
        return np.min(h_vals) - 0
    
    def constraint_bounds_high(h_vals):
        return 1 - np.max(h_vals)
    
    # Apply bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': constraint_bounds},
        {'type': 'ineq', 'fun': constraint_bounds_high}
    ]
    
    # Optimize
    result = differential_evolution(
        objective,
        bounds,
        constraints=constraints,
        seed=seed,
        maxiter=100,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        tol=1e-6
    )
    
    best_h = result.x
    c5_bound = evaluate_c5(best_h)
    
    # Store results
    best_h = np.array(best_h)
    c5_bound = evaluate_c5(best_h)
    n_points = len(best_h)
    
    # Refine basic solution if needed within remaining time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10 and c5_bound > 0.38:
        # Try perturbing the solution multiple times
        best_c5 = c5_bound
        best_solution = best_h.copy()
        
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            perturbed = best_solution + np.random.normal(0, 0.001, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            # Project onto constraint
            sum_h = np.sum(perturbed)
            perturbed = perturbed * (n_points / 2) / sum_h
            
            new_c5 = evaluate_c5(perturbed)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_solution = perturbed
        
        if best_c5 < c5_bound:
            c5_bound = best_c5
            best_h = best_solution
    
    return (best_h.tolist(), c5_bound, n_points)
