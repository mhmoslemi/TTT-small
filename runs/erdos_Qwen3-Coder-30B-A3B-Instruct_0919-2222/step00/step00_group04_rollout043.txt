def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    # Compute the overlap integral using convolution
    # This computes max_k int h(x)(1 - h(x+k)) dx
    # Using full convolution mode to get all possible shifts
    conv = np.correlate(h, 1-h, mode="full")
    # The maximum value is our C5 constant
    c5 = np.max(conv)
    # Normalize by dx
    dx = 2.0 / n
    c5 *= dx
    return c5
