def evaluate_erdos_solution(h):
    """Evaluate C5 bound for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum value is the one we want
    c5 = np.max(corr) * dx
    return c5
