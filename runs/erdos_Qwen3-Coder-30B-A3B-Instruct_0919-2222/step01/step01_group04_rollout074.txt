import numpy as np
from scipy.signal import convolve
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def project_onto_box_and_sum(v, target):
    """Project vector v onto box [0,1] and sum constraint"""
    # Find lambda such that sum_i clip(v_i + lambda, 0, 1) = target.
    # Binary search is robust
    lo = -np.max(v) - 2.0  # Ensure sum < target
    hi = 1 - np.min(v) + 2.0  # Ensure sum > target
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def grad_for_lag(h, lag):
    """Compute gradient of h^T * (1 - h_{shift}) w.r.t h for a specific shift."""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # Overlap region i in [0, N - lag)
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]  # derivative w.r.t h_i
        grad[i + lag] -= h[i]      # derivative w.r.t h_{i+lag}
    else:
        lag = -lag
        # Overlap region i in [lag, N)
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad
