def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def project_sum_constraint(h, n_points):
    """Project h to sum(h)=n/2 by scaling."""
    total = np.sum(h)
    if abs(total) < 1e-10:
        # Avoid division by zero
        return np.ones(n_points) * (0.5 * n_points / n_points)
    h_new = h * (n_points / 2) / total
    return np.clip(h_new, 0, 1)
