def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_to_feasible(h, target_sum, dx):
    # Clip to [0,1] and then renormalize
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h
    adjustment = (target_sum - current_sum) / len(h)
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def waterfilling_projection(h, target_sum, dx):

        for i in range(n):

def smooth_max(h, beta=100.0):
    """
    Softmax approximation to max over all overlaps.
    Since overlaps are correlations, we consider the full correlation array.
    """
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    corr *= dx
    return np.logaddexp.reduce(beta * corr) / beta

def accelerated_gradient_descent(h_init, target_sum, max_iter=10000, dx=0.02857142857142857, beta=100.0, alpha=0.005, momentum=0.9):
    h = h_init.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for t in range(max_iter):
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Approximate gradient
        # Use simpler estimation for gradient using mini-movement exploration
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum, dx)
            c_plus = evaluate_erdos_solution(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, target_sum, dx)
            c_minus = evaluate_erdos_solution(h_minus)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Accelerated gradient step
        v = momentum * v + alpha * grad
        h -= v
        h = project_to_feasible(h, target_sum, dx)
        
        # Adjust beta to tighten smooth approx
        beta = min(1000.0, beta * 1.001)
        
        # Early stopping
        if t % 500 == 0:
            if abs(c5 - best_c5) < 1e-6:
                break
                
    return best_h, best_c5
