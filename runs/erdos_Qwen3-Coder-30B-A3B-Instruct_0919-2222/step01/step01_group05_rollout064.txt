import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt
import cvxpy as cp
from cvxpy import Minimize, Problem, Bool, sum as cp_sum

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem parameters
    n_points = 42  # Given by initial_h_values
    dx = 2.0 / n_points
    w = n_points / 2  # Target sum

    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Exact integer programming for binary case
    solver_time_limit = 250  # seconds for MILP
    time_remaining = budget_s - (time.time() - start_time)
    
    if time_remaining > 10:
        # Build and solve MILP
        # Variables
        b = cp.Variable(n_points, boolean=True)
        z = {}
        s = {}
        t = cp.Variable()
        
        # Precompute overlapping pairs for linearization of bilinear term.
        # For shift d, pair (i, i+d) is valid if i+d < n
        # Use the same logic as the correlation: avoid single-sided padding
        # For all d such that 1 <= d <= n-1
        constraints = []
        z_vars = []
        
        for d in range(1, n_points):
            for i in range(n_points - d):
                # Variables for McCormick envelope
                z[(i, d)] = cp.Variable(boolean=True)
                z_vars.append(z[(i, d)])
                
                # McCormick constraints
                constraints += [
                    z[(i, d)] <= b[i],
                    z[(i, d)] <= b[i+d],
                    z[(i, d)] >= b[i] + b[i+d] - 1
                ]
                
                # Compute s_d
                if d not in s:
                    s[d] = cp.Variable(integer=True)
                    s[-d] = s[d]  # Symmetric
                s[d] += z[(i, d)]
                
                # If shift d is positive, we can also track d and -d independently
                if d == 1:
                    s[-1] = s[1]  # For -1 shift
        
        # Set up full repetition of shifts to include all valid pairs up to n-1
        # This doesn't need to be exhaustive but limits full computation to d=1 to n-1
        for d in range(1, n_points):
            s[d] = cp.Variable(integer=True)
            for i in range(n_points - d):
                z[(i, d)] = cp.Variable(boolean=True)
                constraints += [
                    z[(i, d)] <= b[i],
                    z[(i, d)] <= b[i+d],
                    z[(i, d)] >= b[i] + b[i+d] - 1
                ]
                s[d] += z[(i, d)]
        
        # Sum constraint
        constraints += [cp_sum(b) == w]
        
        # Constraint and bound: t ≥ w - s_d for all d (d=1 to n-1)
        for d in range(1, n_points):
            constraints += [t >= w - s[d]]
        
        # Objective
        objective = Minimize(t)
        
        # Create problem and solve
        prob = Problem(objective, constraints)
        
        # Try fitting solver
        try:
            prob.solve(solver=cp.CBC, verbose=False, maxTime=solver_time_limit)
            
            # Extract solution
            if prob.status in ["optimal", "optimal_inaccurate"]:
                sol = b.value
                if sol is not None:
                    h_binary = np.array([1.0 if x else 0.0 for x in sol])
                    # Renormalize to target integral
                    h_init = h_binary * (n_points / (2 * np.sum(h_binary)))
                    c5_binary = evaluate_erdos_solution(h_init)
                    if c5_binary < best_c5:
                        best_h = h_init.copy()
                        best_c5 = c5_binary
                        
        except Exception as e:
            pass  # Fall back gracefully
            
    # Phase 2: Fractional refinement using simulated annealing
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 5 and best_h is None:
        # Fallback to `initial_h_values`
        h_init = initial_h_values.copy()
        best_h = h_init * (n_points / (2 * np.sum(h_init)))

    if best_h is not None:
        C5 = evaluate_erdos_solution(best_h)
        if C5 < best_c5:
            best_c5 = C5

        # Simulated Annealing
        T0 = 0.02 * best_c5
        alpha = 0.995
        T = T0
        Δ = 0.2

        last_improvement = 0
        max_no_improvement = 5000
        iteration = 0
        
        h_current = best_h.copy()
        c5_current = C5
        
        while time.time() - start_time < budget_s * 0.98:
            if last_improvement > max_no_improvement:
                break
                
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            if i == j:
                continue
                
            # Ensure i is from 1 to 0 and j from 0 to 1
            if h_current[i] <= 0 or h_current[j] >= 1:
                continue
                
            δ = np.random.rand() * min(h_current[i], 1 - h_current[j], Δ)
            
            h_new = h_current.copy()
            h_new[i] -= δ
            h_new[j] += δ
                
            # Always normalize to target average
            h_new = h_new * (n_points / (2 * np.sum(h_new)))
            h_new = np.clip(h_new, 0, 1)
            
            c5_new = evaluate_erdos_solution(h_new)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                last_improvement = 0
            else:
                if np.random.rand() < np.exp(-(c5_new - c5_current) / T):
                    h_current = h_new
                    c5_current = c5_new
                last_improvement += 1
                
            T *= alpha
            Δ = max(1e-6, Δ * 0.9999)
            iteration += 1
    
    # Final adjustment
    if best_h is not None:
        # Renormalize one more time to satisfy sum constraint exactly
        best_h = best_h * (n_points / (2 * np.sum(best_h)))
    else:
        # Fallback
        h_init = initial_h_values.copy()
        best_h = h_init * (n_points / (2 * np.sum(h_init)))
        
    # Final evaluation
    c5_final = evaluate_erdos_solution(best_h)

    return (best_h, c5_final, n_points)
