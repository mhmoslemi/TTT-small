def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

        # Add symmetry breaking to reduce search space
        constraints.append(b[0] == 1)
        
        # Build all terms s_d = sum_i z_{i,d} with z_{i,d} = b_i * b_{i+d} 
        # Use McCormick envelopes: z <= b_i, z <= b_{i+d}, z >= b_i + b_{i+d} - 1

        # We will build constraints one by one for clarity (two-loop approach).
        s_vars = []

        for d in range(1, n):

                constraints.extend([
                    z[i, d] <= b[i],
                    z[i, d] <= b[i+d],
                    z[i, d] >= b[i] + b[i+d] - 1
                ])
            
            # Sum of z variables across i gives s_d
            constraints.append(s_d == cp.sum([z[i, d] for i in range(n - d)]))

            constraints.append(t >= w - s_d)
        
        # Add t >= 0
        constraints.append(t >= 0)

        objective = cp.Minimize(t)
