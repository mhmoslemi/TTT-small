    return np.max(corr)

def project_to_feasible(h, target_sum):

        return h

    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    n = len(h)
    # Find the index where we need to truncate
    for i in range(n):
        if cumsum[i] > target_sum:
            break
    else:
        i = n
    # Above threshold
    threshold = (target_sum - cumsum[i-1]) / (n - i) if i < n else 0
    # Clamp and reposition
    h_projected = np.array([max(0, min(1, x)) for x in h])
    # Verify sum and reproject if needed
    current_sum = np.sum(h_projected)
    adjustment = (target_sum - current_sum) / n
    h_projected += adjustment
    h_projected = np.clip(h_projected, 0, 1)
    return h_projected

def smooth_max(h, beta=100):
    """Use softmax to approximate max over indices."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return -np.log(np.mean(np.exp(-beta * corr))) / beta

def smooth_max_grad_simple(h, beta=100):
    # Approximate gradient via finite differences
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    exp_weights = np.exp(-beta * corr)
    exp_weights_sum = np.sum(exp_weights)
    weights = exp_weights / (exp_weights_sum + 1e-12)
    
    n = len(h)
    grad_loss = np.zeros_like(h)
    
    # Finite diff to get gradient (approx)
    eps = 1e-6
    base_val = smooth_max(h, beta)
    for i in range(n):
        h_pert = h.copy()
        h_pert[i] += eps
        h_pert = project_to_feasible(h_pert, n / 2.0)
        pert_val = smooth_max(h_pert, beta)
        grad_loss[i] = (pert_val - base_val) / eps
    
    return grad_loss
