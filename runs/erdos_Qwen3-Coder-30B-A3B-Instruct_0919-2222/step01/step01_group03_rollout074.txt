import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different optimization strategies
    
    # Strategy 1: Local optimization with better initialization
    try:
        # Use a more sophisticated initialization
        h_init = np.zeros(n_points)
        # Try a piecewise constant construction
        h_init[:n_points//3] = 0.3
        h_init[n_points//3:2*n_points//3] = 0.7
        h_init[2*n_points//3:] = 0.2
        
        # Normalize to integral = 1
        h_init = h_init / (np.sum(h_init) * dx) * 1.0
        h_init = np.clip(h_init, 0, 1)
        
        local_result = minimize(
            objective,
            h_init,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Strategy 2: Random restarts with different constructions
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate different initialization strategies
        h_rand = np.random.rand(n_points) * 0.6  # Between 0 and 0.6
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Strategy 3: Try a completely different approach: make a two-peak function
    try:
        h_peeky = np.zeros(n_points)
        peak1_idx = n_points // 4
        peak2_idx = 3 * n_points // 4
        h_peeky[peak1_idx] = 0.9
        h_peeky[peak2_idx] = 0.9
        h_peeky = h_peeky / (np.sum(h_peeky) * dx) * 1.0
        # Smooth a bit
        for i in range(n_points):
            h_peeky[i] = max(0, min(1, h_peeky[i] - 0.05))
        h_peeky = np.clip(h_peeky, 0, 1)
        
        local_result = minimize(
            objective,
            h_peeky,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1500, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Strategy 4: Use differential evolution on a grid of solutions
    try:
        # Create a coarser grid solution
        h_grid = np.zeros(n_points)
        h_grid[:n_points//2] = 0.2
        h_grid[n_points//2:] = 0.8
        h_grid = h_grid / (np.sum(h_grid) * dx) * 1.0
        h_grid = np.clip(h_grid, 0, 1)
        
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=10,
            seed=seed + restart,
            callback=lambda x, convergence: None,
            atol=1e-7,
            rtol=1e-7
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
