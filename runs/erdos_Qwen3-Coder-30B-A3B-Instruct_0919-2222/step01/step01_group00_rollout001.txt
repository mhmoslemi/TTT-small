import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points, n_steps):
    """Build h array from step heights"""
    seg_len = n_points // n_steps
    h = np.repeat(heights, seg_len)
    return h

def evaluate_heights(heights, n_points, n_steps, penalty_factor=100.0):
    """Evaluate step function defined by heights"""
    h = build_h(heights, n_points, n_steps)
    c5 = evaluate_h(h)
    # Add penalty for sum constraint
    sum_heights = np.sum(heights)
    penalty = penalty_factor * (sum_heights - n_steps/2)**2
    return c5 + penalty

def project_to_bounds_and_sum(heights, target_sum):
    """Project heights to [0,1] and adjust to achieve target sum"""
    # Clip to bounds [0,1]
    heights = np.clip(heights, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(heights)
    diff = target_sum - current_sum
    if abs(diff) < 1e-8:
        return heights
    # Apply uniform adjustment
    heights += diff / len(heights)
    # Clip again
    heights = np.clip(heights, 0, 1)
    return heights
