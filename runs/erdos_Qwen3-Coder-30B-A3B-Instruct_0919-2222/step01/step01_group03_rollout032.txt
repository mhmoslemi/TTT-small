import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try evolutionary optimization with custom strategy
    try:
        # Use a custom evolutionary strategy
        # First, try a more aggressive global search
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=20,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-8,
            rtol=1e-8,
            mutation=(0.5, 1.0),
            recombination=0.7
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except Exception as e:
        print(f"DE failed: {e}")
        pass

    # Grid search in promising regions
    # Create a coarse grid to sample promising regions
    try:
        grid_points = 50
        grid_h = np.linspace(0, 1, grid_points)
        best_grid_c5 = float('inf')
        best_grid_h = None
        
        for i in range(10):  # Try 10 different grids
            if time.time() - start_time > budget_s * 0.8:
                break
            h_grid = np.random.choice(grid_h, n_points)
            h_grid = h_grid / (np.sum(h_grid) * dx) * 1.0
            h_grid = np.clip(h_grid, 0, 1)
            c5_grid = objective(h_grid)
            if c5_grid < best_grid_c5:
                best_grid_c5 = c5_grid
                best_grid_h = h_grid.copy()
                
        if best_grid_h is not None and best_grid_c5 < best_c5:
            best_c5 = best_grid_c5
            best_h = best_grid_h.copy()
    except Exception as e:
        print(f"Grid search failed: {e}")
        pass

    # Try local optimizations on multiple restarts
    restarts = 15
    for restart in range(restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate a different initialization strategy
        h_restart = np.random.rand(n_points)  # Random uniform
        h_restart = h_restart / (np.sum(h_restart) * dx) * 1.0
        h_restart = np.clip(h_restart, 0, 1)
        
        # Apply some smoothing to the restart
        h_smooth = np.convolve(h_restart, np.ones(5)/5, mode='same')
        h_smooth = h_smooth / (np.sum(h_smooth) * dx) * 1.0
        h_restart = np.clip(h_smooth, 0, 1)
        
        # Local optimization
        try:
            local_result = minimize(
                objective,
                h_restart,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except Exception as e:
            print(f"Local opt failed: {e}")
            pass

    # Try pattern-based refinement (try specific structure)
    # Construct piecewise constant with fewer segments
    try:
        n_segments = 5
        segments = np.random.rand(n_segments)
        segments = segments / np.sum(segments) * 2  # Normalize to have total length = 2
        segment_lengths = np.round(np.array(segments) * n_points).astype(int)
        segment_lengths = np.clip(segment_lengths, 1, n_points - 1)
        
        # Adjust the last segment to ensure total length = n_points
        segment_lengths[-1] = n_points - np.sum(segment_lengths[:-1])
        if segment_lengths[-1] < 1:
            # If we must adjust, take the second-to-last and recompute
            diff = 1 - segment_lengths[-1]
            segment_lengths[-2] -= diff
            segment_lengths[-1] = 1
            
        h_piecewise = np.zeros(n_points)
        start_idx = 0
        
        for i in range(n_segments):
            segment_length = segment_lengths[i]
            h_piecewise[start_idx:start_idx + segment_length] = np.random.rand() * 0.8 + 0.1  # between 0.1 and 0.9
            start_idx += segment_length
            
        h_piecewise = h_piecewise / (np.sum(h_piecewise) * dx) * 1.0
        h_piecewise = np.clip(h_piecewise, 0, 1)
        
        c5_piecewise = objective(h_piecewise)
        if c5_piecewise < best_c5:
            best_c5 = c5_piecewise
            best_h = h_piecewise.copy()
    except Exception as e:
        print(f"Piecewise construction failed: {e}")
        pass

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
