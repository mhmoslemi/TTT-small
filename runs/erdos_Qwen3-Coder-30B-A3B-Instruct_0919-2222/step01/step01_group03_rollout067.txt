import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist, squareform
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Initialize optimization with population-based search
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=20,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-7,
            rtol=1e-7
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Local search to refine
    try:
        local_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 3000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Try a combination of different heuristic strategies
    # Strategy 1: Random perturbations
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate a random perturbed version of best solution
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        # Rescale to maintain integral = 1
        h_perturb = h_perturb / (np.sum(h_perturb) * dx) * 1.0
        h_perturb = np.clip(h_perturb, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_perturb,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Strategy 2: Greedy iterative refinement
    # Try steep descent with adaptive step sizes
    h_current = best_h.copy()
    last_c5 = float('inf')
    step_size = 0.01
    tolerance = 1e-9
    max_iter = 1000
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        current_c5 = objective(h_current)
        if abs(last_c5 - current_c5) < tolerance:
            break
        last_c5 = current_c5
        
        # Compute gradient numerically (approximate)
        epsilon = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / (np.sum(h_plus) * dx) * 1.0
            h_plus = np.clip(h_plus, 0, 1)
            
            h_minus = h_current.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus / (np.sum(h_minus) * dx) * 1.0
            h_minus = np.clip(h_minus, 0, 1)
            
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * epsilon)
        
        # Update with gradient step
        h_new = h_current - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        # Rescale to maintain integral constraint
        h_new = h_new / (np.sum(h_new) * dx) * 1.0
        h_new = np.clip(h_new, 0, 1)
        
        # Accept if better or fallback
        new_c5 = objective(h_new)
        if new_c5 < current_c5:
            h_current = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_current.copy()
        else:
            step_size *= 0.95

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
