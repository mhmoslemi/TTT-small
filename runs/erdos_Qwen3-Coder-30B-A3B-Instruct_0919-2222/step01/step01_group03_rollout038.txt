```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist, squareform
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try several different optimization approaches
    
    # Approach 1: Differential evolution with adaptive parameters
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=20,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-7,
            rtol=1e-7,
            strategy='best1bin'
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Approach 2: Local optimization from DE result if available or from h0
    if time.time() - start_time < budget_s * 0.9:
        try:
            local_result = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Approach 3: Multiple random restarts with adaptive sampling
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate random initialization with various distributions
        if restart % 3 == 0:
            # Gaussian-like centered distribution
            h_rand = np.random.normal(0.5, 0.15, n_points)
            h_rand = np.clip(h_rand, 0, 1)
        elif restart % 3 == 1:
            # Focused on upper values
            h_rand = np.random.beta(2, 1, n_points)
        else:
            # Uniform distribution with specific scaling to target integral
            h_rand = np.random.rand(n_points) * 0.7 + 0.15
            h_rand = np.clip(h_rand, 0, 1)
        
        # Ensure correct integral
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1500, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Approach 4: Gradient-free optimization if time allows
    if time.time() - start_time < budget_s * 0.9:
        try:
            # Try another variant of DE with different strategy
            de_result = differential_evolution(
                objective,
                bounds,
                maxiter=150,
                popsize=25,
                seed=seed,
                callback=lambda x, convergence: None,
                atol=1e-7,
                rtol=1e-7,
                strategy='rand1bin'
            )
            if de_result.success:
                h_opt = np.clip(de_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
```
