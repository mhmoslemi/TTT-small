import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector v onto [0,1] bounds and sum constraint"""
    # First clip to [0,1]
    v = np.clip(v, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(v)
    if np.abs(current_sum - target_sum) < 1e-10:
        return v
    # Use water-filling approach to adjust
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase some values
        idx = np.argsort(v)
        for i in range(len(v)-1, -1, -1):
            if diff <= 0:
                break
            idx_i = idx[i]
            max_increase = 1.0 - v[idx_i]
            increase = min(max_increase, diff)
            v[idx_i] += increase
            diff -= increase
    else:
        # Need to decrease some values
        idx = np.argsort(v, kind='mergesort')
        for i in range(len(v)):
            if diff >= 0:
                break
            idx_i = idx[i]
            max_decrease = v[idx_i]
            decrease = max(-max_decrease, diff)
            v[idx_i] += decrease
            diff -= decrease
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create shift index lists to avoid rebuild
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points - s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))

    # Define objective function using soft max approximation
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        # Compute raw correlations without dx yet
        correlations = np.correlate(h, 1 - h, mode="full")
        # Soft max approximation to mimic max
        beta = 50.0  # Can be adjusted or adaptive
        max_c = np.max(correlations)
        exp_terms = np.exp(beta * (correlations - max_c))
        weights = exp_terms / np.sum(exp_terms)
        # Compute weighted objective
        weighted_sum = np.sum(weights * correlations)
        return -weighted_sum  # Minimize negative for maximization

    # Optimization with gradient descent using projected steps
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Initialize with triangular shape
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    h = np.clip(1 - np.abs(x-1)/1, 0, 1)  # triangular
    total = h.sum()
    h = h * (target_sum / total)
    h = project_onto_bounds_and_sum(h, target_sum)
    
    # Gradient descent params
    max_iter = 50000
    eta0 = 0.1
    beta_start = 1.0
    beta_end = 100.0
    beta_inc = (beta_end - beta_start) / max_iter
    
    best_c5 = float('inf')
    last_c5 = float('inf')
    patience = 0
    patience_limit = 1000  # Stop if no improvement for this many iterations
    
    beta = beta_start
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Update beta
        beta = min(beta_end, beta_start + beta_inc * iter_num)
        
        # Compute raw correlations
        correlations = np.correlate(h, 1 - h, mode="full")
        max_c = np.max(correlations)
        
        # Compute weights for gradient
        exp_terms = np.exp(beta * (correlations - max_c))
        weights = exp_terms / np.sum(exp_terms)
        
        # Compute gradient
        grad = np.zeros_like(h)
        # Derivative of sum_i h_i is 1 for each i
        # Derivative of sum_i h_i h_{i+s} is h_{i+s} + h_{i-s} for each i
        total_weight = np.sum(weights)
        grad += total_weight  # Add 1 * sum(weights) to each

        # Add contribution from each shift
        for shift_idx, (i, j) in enumerate(shift_indices):
            if len(i) == 0: continue
            w = weights[shift_idx]
            if w == 0: continue
            # For first term: weight contributes +1 to grad[i]
            grad[i] += w * 1
            # For product term: weight contributes  -h[j] to grad[i] and -h[i] to grad[j]
            grad[i] += -w * h[j]
            grad[j] += -w * h[i]

        # Step
        eta = eta0 / (1 + iter_num ** 0.5)
        h_new = h - eta * grad
        
        # Project
        h_new = project_onto_bounds_and_sum(h_new, target_sum)
        h = h_new
        
        # Every 100 iterations, recompute actual c5
        if iter_num % 100 == 0 or iter_num == max_iter - 1:
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                patience = 0
            else:
                patience += 1
                if patience > patience_limit:
                    break

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
