import numpy as np
import time
import cvxpy as cp
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, d):
    """Apply shift d to h, return shifted array (zero-padded)"""
    n = len(h)
    if d >= 0:
        shifted = np.concatenate([h[d:], np.zeros(d)])
    else:
        shifted = np.concatenate([np.zeros(-d), h[:d]])
    return shifted

def project_onto_box_and_sum(v, target_sum):
    """Project v onto the hyperplane sum = target_sum and box [0,1]^n"""
    v = np.array(v)
    lb = 0
    ub = 1
    low = 0
    high = 1000.0
    for _ in range(40):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, lb, ub)
        s = np.sum(x)
        if s < target_sum:
            high = mid
        else:
            low = mid
    x = np.clip(v - low, lb, ub)
    # Adjust for numerical errors
    s = np.sum(x)
    diff = target_sum - s
    if abs(diff) > 1e-10:
        # Compensate uniformly
        idx = np.argwhere((x > lb) & (x < ub))
        if len(idx) > 0:
            adjust = diff / len(idx)
            for i in idx.flatten():
                x[i] = np.clip(x[i] + adjust, lb, ub)
    return x

def build_lp_model(n, binary=False):
    """Build an LP model for the Erdős problem with linear relaxation of products"""
    h = cp.Variable(n, name='h')
    y = {}
    t = cp.Variable(1, name='t')
    
    constraints = [
        cp.sum(h) == n / 2,
        h >= 0,
        h <= 1
    ]
    
    # Generate all possible shifts
    shifts = list(range(1-n, n))  # -n+1 to n-1, but zero is implicitly considered
    if binary:
        h.type = 'binary'
    
    for d in shifts:
        if d >= 0:
            for i in range(n - d):
                if i + d < n:
                    if (i, d) not in y:
                        y[(i, d)] = cp.Variable(name=f'y_{i}_{d}')
                    # McCormick envelope
                    constraints.extend([
                        y[(i, d)] >= h[i] + h[i+d] - 1,
                        y[(i, d)] <= h[i],
                        y[(i, d)] <= h[i+d],
                        y[(i, d)] >= 0
                    ])
        else:
            for i in range(-d, n):
                if i + d >= 0:
                    if (i, d) not in y:
                        y[(i, d)] = cp.Variable(name=f'y_{i}_{d}')
                    # McCormick envelope
                    constraints.extend([
                        y[(i, d)] >= h[i] + h[i+d] - 1,
                        y[(i, d)] <= h[i],
                        y[(i, d)] <= h[i+d],
                        y[(i, d)] >= 0
                    ])
    
    # Add constraints for correlation bounds
    for d in shifts:
        sum_y = 0
        if d >= 0:
            for i in range(n - d):
                sum_y += y[(i, d)]
        else:
            for i in range(-d, n):
                sum_y += y[(i, d)]
        constraints.append(t >= 0.5 - sum_y / n)
    
    # Problem definition
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    return prob, h, y, t

def solve_with_time_limit(prob, solver_name, time_limit_s=30):
    """Solve a problem with a time limit"""
    start = time.time()
    prob.solve(solver=cp.GLPK if solver_name == 'GLPK' else cp.CLARA, max_time=time_limit_s)
    return prob.status, prob.value

def subgradient_refine(h_start, end_time):
    """Refine an initial h vector with projected sub-gradient descent"""
    h = np.array(h_start)
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    iteration = 0
    stagnant = 0
    last_improvement = 0

    while time.time() < end_time:
        iteration += 1
        if iteration % 100 == 0:
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = iteration
            if iteration - last_improvement > 5000:
                # Perturb
                noise = np.random.normal(0, 0.05, n)
                h = h + noise
                h = project_onto_box_and_sum(h, n / 2)
                last_improvement = iteration
                continue

        # Identify the worst shift d* that maximizes C(d)
        correlations = []
        shifts = list(range(1 - n, n))  # generate all shifts
        for d in shifts:
            shifted = shift_array(h, d)
            correlation = np.sum(h * (1 - shifted)) * dx
            correlations.append(correlation)
        d_star = shifts[np.argmax(correlations)]

        # Estimate direction (gradient of C(d_star))
        g = np.zeros(n)
        shifted = shift_array(h, d_star)
        g = 1.0 - shifted - shift_array(h, -d_star)

        # Gradient descent step
        alpha = 0.5 / np.sqrt(iteration)
        h_new = h - alpha * g

        # Project onto feasible region
        h_new = project_onto_box_and_sum(h_new, n / 2)
        # Update h
        h = h_new
    return best_h, best_c5
