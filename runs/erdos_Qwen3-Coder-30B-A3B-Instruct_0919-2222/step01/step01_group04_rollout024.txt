import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def project_onto_box_and_sum(v, target):
    """Project vector v onto [0,1] box and sum=target constraint using binary search"""
    # Find λ such that sum_i clip(v_i + λ, 0, 1) = target.
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def compute_gradient_for_lag(h, lag):
    """Compute gradient of sum_i h_i (1 - h_{i+lag}) w.r.t h_i. """
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # Overlap region i in [0, N - lag)
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]
        grad[i + lag] -= h[i]
    else:
        lag = -lag
        # Overlap region i in [lag, N)
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad

def smooth_binary_vector(b):
    """Smooth binary vector with a triangular kernel"""
    # Kernel [1/4, 1/2, 1/4] corresponds to [0.25, 0.5, 0.25]
    kernel = np.array([0.25, 0.5, 0.25])
    h = np.convolve(b, kernel, mode='same')
    return h
