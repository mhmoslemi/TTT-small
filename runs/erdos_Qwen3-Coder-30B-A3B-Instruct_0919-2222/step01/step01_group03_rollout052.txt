import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist, squareform
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try a new approach: genetic algorithm-inspired optimization with adaptive sampling
    try:
        # Start with a more sophisticated way to initialize
        h_init = np.zeros(n_points)
        # Piecewise constant with a spike in the middle
        h_init[10:32] = 0.7
        # Normalize
        h_init = h_init / (np.sum(h_init) * dx) * 1.0
        h_init = np.clip(h_init, 0, 1)
        
        # Then do local optimization with a good starting point
        local_result = minimize(
            objective,
            h_init,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        print(f"First optimization result: {c5_opt}")
    except:
        pass

    # Try another initialization: sin-like shape
    try:
        x = np.linspace(0, 2, n_points)
        h_sin = 0.5 + 0.3 * np.sin(np.pi * x)
        h_sin = np.clip(h_sin, 0, 1)
        # Normalize
        h_sin = h_sin / (np.sum(h_sin) * dx) * 1.0
        local_result = minimize(
            objective,
            h_sin,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Random restarts with specific patterns
    for restart in range(30):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Pattern-based initialization
        h_rand = np.random.rand(n_points)
        # Create some localized structures
        if restart % 3 == 0:
            # Mix of high/low
            h_rand[:n_points//3] = 0.8
            h_rand[n_points//3:2*n_points//3] = 0.2
            h_rand[2*n_points//3:] = 0.8
        elif restart % 3 == 1:
            # Some sinusoidal momentum
            x = np.linspace(0, 2, n_points)
            h_rand = 0.5 + 0.3 * np.sin(2 * np.pi * x + restart)
        else:
            # Random as fallback
            h_rand = np.random.rand(n_points)
        
        h_rand = np.clip(h_rand, 0, 1)
        # Normalize
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 800, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Final refinement if time allows
    if time.time() - start_time < budget_s * 0.9:
        try:
            h_final = best_h.copy()
            local_result = minimize(
                objective,
                h_final,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
