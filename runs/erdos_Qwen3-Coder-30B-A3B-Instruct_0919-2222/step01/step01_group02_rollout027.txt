def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

    # Then we want to maximize the minimal autocorrelation of a
    # which is equivalent to minimizing the maximal overlap integral.

    # SDP Approach 
    # Define variables
    a = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    
    # Define constraints in SDP
    # 1. X = a * a.T (rank1 constraint)
    # We do not enforce it directly, but exploit the fact that it is a moment matrix
    # 2. Add constraints on X and a
    # For each i, X[i,i] = a[i]^2. We add constraints 0 <= X[i,i] <= 1 and -1 <= a[i] <= 1,
    # thus a[i]^2 <= 1.
    
    # Linear constraints
    constraints = [
        cp.sum(a) == 0,  # sum of a = 0
    ]
    
    # Add bounds on a 
    for i in range(n):
        constraints.append(-1 <= a[i])
        constraints.append(a[i] <= 1)

    # Add constraints on rows of X
    for i in range(n):
        constraints.append(X[i,i] == a[i]**2)

    for i in range(n):
        for j in range(n):
            constraints.append(X[i,j] <= 1)
            constraints.append(X[i,j] >= -1)

    # C(d) = dx * ( (n - |d|)/4 - 1/4 * sum a_i * a_{i+d} )
    # We want to minimize max_d of this expression

    # Define the Toeplitz matrices P_d as shift operators that extract

    # Let a be the vector we're trying to optimize (here a_i = 2 h_i - 1, so a in [-1,1])
    # Objective: minimize max_{d} [dx * ( (n - |d|)/4 - (1/4) sum_i a_i * a_{i+d} )]
    
    # We can bound minimum autocorrelation with SDP easily:
    # So we avoid solving large SDPs by using an efficient constructive method:

    # This is a very different approach from earlier strategies.

    def compute_c5_from_h(h):
        return evaluate_erdos_solution(h)

    # We define a step function shape with few segments
    def representative_h_function(params):
        if len(params) != 6:  # [a1, a2, a3, s1, s2, t]
            return 1e10  # Invalid input
        a1, a2, a3, s1, s2, t = params

        a3 = max(0, min(1, a3))

            h[int(s1/dx):int(s2/dx)] = a2 
        if t > s2:
            h[int(s2/dx):] = a3
            
        # Normalize
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n/2) / sum_h
            
        return compute_c5_from_h(h)

        return np.log(np.sum(np.exp(x / eps))) * eps

        # Instead of max, compute smoothed maximum
        overlap_values = np.array([
            np.sum(h * np.roll(1 - h, shift))
            for shift in range(len(h))
        ])
        return smooth_max(overlap_values, eps) * dx
    
    def compute_gradient_h(h, eps=1e-3):
        n = len(h)

        # compute approximate gradient numerically
        grad = np.zeros_like(h)
        for i in range(n):

            h_plus[i] += 1e-6
            h_plus = h_plus * (n / 2) / np.sum(h_plus)
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            h_minus = h_minus * (n / 2) / np.sum(h_minus)
            grad[i] = (objective_smooth(h_plus, eps) - objective_smooth(h_minus, eps)) / (2e-6)
        return grad

    current_h = best_h.copy()

        if time.time() - t_start > budget_s - 1:

        # Compute gradient
        g = compute_gradient_h(current_h)

        updated_h = current_h - lr * g

        updated_h = np.clip(updated_h, 0, 1)

        # Update current h 
        current_h = updated_h
        
    # Final evaluation
    final_c5 = evaluate_erdos_solution(current_h)
    return (current_h.tolist(), final_c5, len(current_h))
