import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try a combination of optimization techniques with adaptive parameters
    # 1. Differential evolution with different population sizes and iterations
    pop_sizes = [15, 20, 25]
    max_iters = [100, 150, 200]
    for popsize in pop_sizes:
        for maxiter in max_iters:
            if time.time() - start_time > budget_s * 0.8:
                break
            try:
                de_result = differential_evolution(
                    objective,
                    bounds,
                    maxiter=maxiter,
                    popsize=popsize,
                    seed=seed,
                    atol=1e-6,
                    rtol=1e-6,
                    mutation=(0.5, 1.0),
                    recombination=0.7
                )
                if de_result.success:
                    h_opt = np.clip(de_result.x, 0, 1)
                    c5_opt = objective(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
            except:
                pass
    
    # 2. Local optimization with different methods
    methods = ['SLSQP', 'L-BFGS-B', 'TNC']
    for method in methods:
        if time.time() - start_time > budget_s * 0.9:
            break
        try:
            local_result = minimize(
                objective,
                best_h,
                method=method,
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # 3. Random restarts with more diverse initializations
    diversities = [0.1, 0.3, 0.5, 0.7, 0.9]
    for diversity in diversities:
        if time.time() - start_time > budget_s * 0.95:
            break
        for restart in range(10):
            if time.time() - start_time > budget_s * 0.95:
                break
            h_rand = np.random.rand(n_points) * diversity
            # Scale to ensure sum constraint is met
            h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
            h_rand = np.clip(h_rand, 0, 1)
            
            try:
                local_result = minimize(
                    objective,
                    h_rand,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                )
                if local_result.success:
                    h_opt = np.clip(local_result.x, 0, 1)
                    c5_opt = objective(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
            except:
                pass
    
    # 4. try a hybrid approach
    # Create a new generation of solutions using current best as base and random noise
    num_hybrid_starts = 5
    for i in range(num_hybrid_starts):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Start with a slight random perturbation of best solution
        h_perturbed = best_h + (np.random.rand(n_points) - 0.5) * 0.1
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Renormalize to satisfy constraint
        h_perturbed = h_perturbed / (np.sum(h_perturbed) * dx) * 1.0
        
        try:
            local_result = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
