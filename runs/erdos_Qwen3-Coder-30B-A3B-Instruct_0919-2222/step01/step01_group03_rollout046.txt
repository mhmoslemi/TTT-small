import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different algorithms and restart strategies
    methods_to_try = [
        # Global optimization
        ('differential_evolution', {'maxiter': 50, 'popsize': 15}),
        # Local optimization with different methods
        ('Nelder-Mead', {}),
        ('Powell', {}),
        ('CG', {}),
        ('BFGS', {}),
    ]
    
    # First, a global search with differential evolution
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=100,
            popsize=15,
            seed=seed,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Local optimization using multiple methods with warm start
    for method, opts in methods_to_try[1:]:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        try:
            # Use best solution as initialization
            local_result = minimize(
                objective,
                best_h,
                method=method,
                bounds=bounds,
                constraints=constraints,
                options={**opts, 'maxiter': 1000},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # More diverse restarts with various initializations
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Different initialization strategies
        choice = random.randint(0, 2)
        
        if choice == 0:
            # Random perturbation of current best
            h_rand = best_h * (0.9 + 0.2 * np.random.rand(n_points))
        elif choice == 1:
            # Two step function
            h_rand = np.zeros(n_points)
            h_rand[:n_points//2] = 1.0
            h_rand[n_points//2:] = 0.0
            h_rand = h_rand * (1.0 / np.sum(h_rand) / dx)
        else:
            # Gaussian-like shape
            x = np.linspace(0, 2, n_points)
            mu = 1.0
            sigma = 0.3
            h_rand = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            h_rand = h_rand / (np.sum(h_rand) * dx)
        
        h_rand = np.clip(h_rand, 0, 1)
        
        # Local optimization
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final refinement with dedicated local search
    try:
        local_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
