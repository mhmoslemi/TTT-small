import numpy as np
import time
from scipy.fft import rfft, irfft

def proj_capped_simplex(v, target_sum):
    """
    Project vector v onto capped simplex: sum = target_sum, 0 <= v[i] <= 1.
    Uses water-filling algorithm.
    """
    v = np.asarray(v)
    if np.all(v >= 0) and np.all(v <= 1) and abs(v.sum() - target_sum) < 1e-12:
        return v.copy()
    # Binary search for the Lagrange multiplier
    l_min, l_max = -1e6, 1e6
    tol = 1e-10
    for _ in range(100):
        l = (l_min + l_max) / 2
        w = np.clip(v - l, 0, 1)
        sum_w = w.sum()
        if abs(sum_w - target_sum) < tol:
            return w
        elif sum_w > target_sum:
            l_min = l
        else:
            l_max = l
    return np.clip(v - l, 0, 1)

def full_corr(h):
    """Compute correlation of h with (1 - h) using FFT."""
    n = len(h)
    H = rfft(h, n=2*n-1)
    G = rfft(1 - h, n=2*n-1)
    corr = irfft(H * np.conj(G), n=2*n-1)
    return corr

def evaluate_h(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    conv = full_corr(h)
    max_val = np.max(conv)
    return max_val * dx
