import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points, seg_len):
    # Build full h array from step heights
    return np.repeat(heights, seg_len)

def evaluate_heights(heights, n_points, seg_len, penalty_factor=10.0):
    # Build h from heights
    h = build_h(heights, n_points, seg_len)
    # Evaluate C5
    c5 = evaluate_h(h)
    # Add penalty for sum constraint
    sum_heights = np.sum(heights)
    penalty = penalty_factor * (sum_heights - n_points / (2 * seg_len)) ** 2
    return c5 + penalty

    # Ensure all values are in [0,1]
    heights = np.clip(heights, 0, 1)
    # Adjust to get exact sum
    current_sum = np.sum(heights)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return heights

    if current_sum > target_sum:
        # Need to reduce
