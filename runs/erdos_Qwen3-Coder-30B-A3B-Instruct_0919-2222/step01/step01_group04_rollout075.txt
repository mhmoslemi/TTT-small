import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def project_onto_box_and_sum(v, target):
    """Project vector v onto box [0,1] and sum constraint"""
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def grad_for_lag(h, lag):
    """Compute gradient of correlation at lag with respect to h"""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]
        grad[i + lag] -= h[i]
    else:
        lag = -lag
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad

    current_sum = np.sum(h)
    h = h * (n_points / 2) / current_sum
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Binary initialization stage
    if budget_s > 30:
        binary_best = None
        best_min_overlap = -1
        
        # Generate many random binary patterns to find good one
        num_samples = 5000
        for _ in range(num_samples):
            if time.time() - start_time > budget_s - 15:
                break
            b = np.zeros(n_points, dtype=int)
            ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
            b[ones_indices] = 1
            
            # Compute correlation
            corr = np.correlate(b, b, mode="full")
            # Get positive lags only (excluding zero)
            min_overlap = np.min(corr[n_points:])
            if min_overlap > best_min_overlap:
                best_min_overlap = min_overlap
                binary_best = b.copy()
        
        if binary_best is not None:
            # Smooth it
            kernel = np.array([0.25, 0.5, 0.25])
            h_smooth = np.convolve(binary_best, kernel, mode='same')
            h_smooth = np.clip(h_smooth, 0, 1)
            # Project to satisfy sum constraint
            h = project_onto_box_and_sum(h_smooth, n_points / 2)
            
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    # Continuous refinement with Adam
    max_iter = 20000
    beta1, beta2, epsilon = 0.9, 0.999, 1e-8
    lr = 0.05
    decay = 1e-6
    
    # Initialize Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    iteration = 0
    while iteration < max_iter and time.time() - start_time < budget_s - 5:
        # Compute correlations for all lags
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        c5_vals = corr * dx
        
        # Get the maximum value and corresponding lags
        max_val = np.max(c5_vals)
        best_lags = np.where(np.abs(c5_vals - max_val) < 1e-10)[0]
        # Convert to actual shift values
        shift_indices = best_lags - (n_points - 1)
        
        # Compute gradient from all worst lags (can average)
        grad = np.zeros(n_points)
        for lag in shift_indices:
            grad += grad_for_lag(best_h, lag)
        grad /= len(shift_indices)
        
        # ADAM update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        
        # Update parameters
        lr_current = lr / (1 + decay * iteration)
        best_h = best_h - lr_current * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project onto feasible set
        best_h = project_onto_box_and_sum(best_h, n_points / 2)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = best_h.copy()
        
        # Occasionally add noise to escape local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            best_h += noise
            best_h = np.clip(best_h, 0, 1)
            best_h = project_onto_box_and_sum(best_h, n_points / 2)
            new_c5 = evaluate_c5(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
        
        iteration += 1
    
    return (best_h, best_c5, n_points)
