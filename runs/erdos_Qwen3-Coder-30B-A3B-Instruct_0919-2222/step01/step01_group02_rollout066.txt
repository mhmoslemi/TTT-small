import numpy as np
import cvxpy as cp
import time
import random
from scipy.optimize import differential_evolution

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def init_h_from_sdp(n_points):
    """Initialize h using SDP relaxation and random rounding"""
    # Variables
    a = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [
        X == a @ a.T,
        cp.diag(X) <= 1,  # X_ii <= 1
        cp.sum(a) == 0,   # sum(a) = 0
        # For each lag, apply the SDP constraint on autocorrelation
    ]
    
    # Add autocorrelation constraints (can be relaxed)
    # TODO: Implement full SDP constraints here, this is approximation
    # For now, we will approximate the constraints via sampling from the solutions
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
    except Exception:
        # Fall back to initializing with a simple function
        return np.ones(n_points) * 0.5  # Simple uniform
    
    # Extract a from solution
    a_val = a.value
    if a_val is None:
        return np.ones(n_points) * 0.5  # Fallback

    # Apply rounding step (simple Gaussian rounding approach)
    # Can be improved with better rounding techniques
    # Here, we use simple projection to [-1,1] and normalization

    # Project to [-1, 1]
    a_clipped = np.clip(a_val, -1, 1)
    
    # Adjust to have zero sum and appropriate average
    mean_a = np.mean(a_clipped)
    a_zero_mean = a_clipped - mean_a

    # Normalize such that sum = 0 and integral = 1
    # Total integral should be 1: (sum(a) * dx) = 1 => sum(a) = n/2 * dx = 1/2 * n / n * dx = dx = 2/n
    # Actually, ∫ h = 1 => sum(h)*dx = 1 => sum(h) = n/2
    # Since h = (a+1)/2, we have sum(h) = (sum(a) + n)/2 = 1 => sum(a) = 0
    # So already aligned with constraint.
    dx = 2.0 / n_points
    scaled_a = a_zero_mean
    h = (scaled_a + 1) / 2

    return np.clip(h, 0, 1)

def parametric_optimize_h():
    """Optimize a piecewise constant or linear shape using differential evolution."""
    
    # Function to evaluate target over a low-dimensional parameters space
    # Parametric shape: h = a on [0,s], b on [s,t], c on [t,2]
    def evaluate_from_params(params, n=70):
        a, b, c, s, t = params
        # Ensure domain constraints
        s = max(0, min(s, t))
        t = max(s, min(t, 2))
        # Clamp values to feasible domain
        a = max(0, min(1, a))
        b = max(0, min(1, b))
        c = max(0, min(1, c))
        # Length of intervals
        len1 = s
        len2 = t - s
        len3 = 2 - t
        # Constrain average to 1
        avg_condition = a * len1 + b * len2 + c * len3
        target_avg = 1.0
        # Scale if needed to match constraint
        scale = target_avg / avg_condition if avg_condition > 0 else 1
        h = np.concatenate([
            np.full(int(round(n * len1 / 2)), a * scale),
            np.full(int(round(n * len2 / 2)), b * scale),
            np.full(int(round(n * len3 / 2)), c * scale)
        ])
        # Adjust length to exactly n
        h = h[:n]  # In case too many
        if len(h) < n:
            h = np.pad(h, (0, n - len(h)), mode='constant', constant_values=h[-1] if len(h)>0 else 0)
        else:
            h = h[:n]
        return evaluate_erdos_solution(h)
    
    # Start values and bounds
    bounds = [(0, 1), (0, 1), (0, 1), (0, 2), (0, 2)]
    
    # Run optimization
    result = differential_evolution(
        lambda p: evaluate_from_params(p, 70),
        bounds,
        maxiter=5000,
        popsize=30,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False
    )
    
    # Recreate final h
    final_params = result.x
    h = np.concatenate([
        np.full(int(round(70 * final_params[3] / 2)), final_params[0]),
        np.full(int(round(70 * (final_params[4] - final_params[3]) / 2)), final_params[1]),
        np.full(int(round(70 * (2 - final_params[4]) / 2)), final_params[2])
    ])
    
    h = h[:70]
    if len(h) < 70:
        h = np.pad(h, (0, 70 - len(h)), mode='constant', constant_values=h[-1] if len(h)>0 else 0)
    else:
        h = h[:70]
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 70
    
    # Phase 1: Try parametric optimization (fast)
    best_h = parametric_optimize_h()
    best_c5 = evaluate_erdos_solution(best_h)
    
    elapsed = time.time() - start_time
    
    # Phase 2: If time allows and not yet sat, use iterative improvement
    while elapsed < budget_s * 0.75:
        # Create a modified version: slight Gaussian jitter
        h_mod = best_h + np.random.normal(0, 0.01, n_points)
        h_mod = np.clip(h_mod, 0, 1)
        # Normalize
        total = np.sum(h_mod)
        h_mod *= (n_points / 2) / total
        # Evaluate
        c5_test = evaluate_erdos_solution(h_mod)
        if c5_test < best_c5:
            best_h = h_mod
            best_c5 = c5_test
        
        elapsed = time.time() - start_time

    return (best_h.tolist(), best_c5, n_points)
