def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def compute_product_sums(h):
    n = len(h)
    # Compute Σ_i h_i * h_{i+s} for all shifts s from -(n-1) to (n-1)
    # Use np.correlate with mode='full' for computation, but we'll do it manually
    # to be precise about overlap handling.
    p_sums = np.zeros(2*n - 1)
    for s in range(-(n-1), n):
        # i from 0 to n-1 such that i+s also in [0,n-1]
        start_i = max(0, -s)
        end_i = min(n, n - s)
        if end_i > start_i:
            p_sums[s + (n-1)] = np.sum(h[start_i:end_i] * h[start_i+s:end_i+s])
    return p_sums

def compute_gradient_for_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # For forwards shift
        g[:n-s] += h[s:]
        g[s:] += h[:n-s]
    else:
        # For backwards shift
        s_abs = -s
        g[s_abs:] += h[:n-s_abs]
        g[:n-s_abs] += h[s_abs:]
    return g
