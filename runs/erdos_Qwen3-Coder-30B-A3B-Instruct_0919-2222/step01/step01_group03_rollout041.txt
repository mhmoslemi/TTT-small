import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto box constraints [lower,upper] and sum constraint"""
    # First clip to [lower, upper]
    v = np.clip(v, lower, upper)
    # Then adjust to meet sum constraint using binary search for lambda
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Binary search for the right shift
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, lower, upper)
        s = np.sum(w)
        if s > target_sum:
            high = mid
        else:
            low = mid
    
    return np.clip(v - (low + high) / 2.0, lower, upper)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the constraint that integral equals 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Pre-compute shift indices for gradient computation
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points - s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))
    
    # Gradient computation for softmax approximation
    def compute_gradient(h, f, weights):
        grad = np.zeros_like(h)
        
        # Add constant contribution from sum_i h_i (i.e., sum_s w_s)
        grad += np.sum(weights)
        
        # Add contribution from sum_i h_i h_{i+s} terms
        for s, (i, j) in enumerate(shift_indices):
            weight = weights[s]
            if weight == 0:
                continue
            # grad[i] += weight * (1 - h[j])
            # grad[j] += weight * (-h[i])
            grad[i] += weight * (1 - h[j])
            grad[j] -= weight * h[i]
        return grad
    
    # Define objective function (we'll directly minimize C5)
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Use the improved gradient-based optimization
    best_c5 = float('inf')
    best_h = h0.copy()
    
    max_iter = 10000
    eta0 = 0.2
    beta_max = 100.0
    beta_increment = beta_max / (max_iter / 1000)
    
    # Initialize with the starting solution
    h = h0.copy()
    c5_current = evaluate_erdos_solution(h)
    
    if c5_current < best_c5:
        best_c5 = c5_current
        best_h = h.copy()
    
    # Gradient-based optimization with momentum
    beta = 1.0
    prev_grad = np.zeros_like(h)
    momentum_gamma = 0.9
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Update beta
        if beta < beta_max:
            beta = min(beta + beta_increment, beta_max)
        
        # Compute raw correlations
        correlations = np.correlate(h, 1 - h, mode="full")
        f = correlations
        
        # Compute softmax weights
        max_f = np.max(f)
        exp_terms = np.exp(beta * (f - max_f))
        weights = exp_terms / np.sum(exp_terms)
        
        # Compute gradient
        grad = compute_gradient(h, f, weights)
        
        # Apply momentum
        grad = momentum_gamma * prev_grad + (1 - momentum_gamma) * grad
        
        # Step size
        eta = eta0 / np.sqrt(it + 1)
        
        # Update h
        h_new = h - eta * grad
        
        # Project onto constraints
        h_new = project_onto_bounds_and_sum(h_new, n_points / 2)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Update for next iteration
        h = h_new
        prev_grad = grad
    
    # Final check with optimized solution
    return (best_h, best_c5, n_points)
