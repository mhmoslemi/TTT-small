import numpy as np
import time
import random
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, d):
    """Compute h shifted by d positions (with zero padding)"""
    n = len(h)
    if d > 0:
        # Shift right (left-pad with zeros)
        return np.concatenate([np.zeros(d), h[:-d]])
    elif d < 0:
        # Shift left (right-pad with zeros)
        return np.concatenate([h[-d:], np.zeros(-d)])
    else:
        return h.copy()

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto the set {x : 0 ≤ x[i] ≤ 1, sum x = target_sum}"""
    # Ensure v is within [0, 1]
    v = np.clip(v, 0, 1)
    # Check if sum is already close
    s = np.sum(v)
    if abs(s - target_sum) < 1e-10:
        return v.copy()
    # Perform bisection on lambda to find correct projection
    def sum_with_lambda(l):
        return np.sum(np.clip(v - l, 0, 1))
    
    # Find bounds for lambda
    # If we increase lambda more and more, sum decreases
    # Initial guess for lambda that gives target sum
    # We seek lambda such that sum(v - lambda) = target_sum
    lower, upper = -1.0, 2.0
    while sum_with_lambda(lower) > target_sum:
        lower -= 1
    while sum_with_lambda(upper) < target_sum:
        upper += 1

    # Bisection on lambda
    for _ in range(40):
        lmid = (lower + upper) / 2.0
        summid = sum_with_lambda(lmid)
        if summid > target_sum:
            lower = lmid
        else:
            upper = lmid
    lambda_opt = (lower + upper) / 2.0
    
    # Apply projection
    x = np.clip(v - lambda_opt, 0, 1)
    # Final adjustment to ensure sum is exactly correct
    current = np.sum(x)
    diff = target_sum - current
    # Add difference uniformly to elements not at bound (to maintain feasibility)
    if abs(diff) > 1e-10:
        # Find non-bound elements
        not_bound_upper = np.where(x < 1)[0]
        not_bound_lower = np.where(x > 0)[0]
        not_bound = np.intersect1d(not_bound_upper, not_bound_lower)
        if len(not_bound) > 0:
            increment = diff / len(not_bound)
            x[not_bound] += increment
            # Clamp again to enforce bounds
            x = np.clip(x, 0, 1)
    
    return x

def build_lp_model(n, binary=False):
    """Build the LP model to relax the problem.
    Returns:
        h_vars: list of variables h[i]
        y_vars: nested dictionary y[i,d]
        t_var: scalar variable used for the upper bound on max correlation
        constraints: list of all constraints
    """
    h_vars = cp.Variable(n)
    y_vars = {}
    constraints = [
        cp.sum(h_vars) == n / 2,
        h_vars >= 0,
        h_vars <= 1
    ]
    
    for d in range(-(n-1), n):
        y_vars[d] = {}
        for i in range(n):
            if 0 <= i + d < n:
                y_vars[d][i] = cp.Variable(1, name=f'y_{i}_{d}')
                # McCormick constraints:
                constraints.append(y_vars[d][i] >= 0)
                constraints.append(y_vars[d][i] <= h_vars[i])
                constraints.append(y_vars[d][i] <= h_vars[i+d])
                constraints.append(y_vars[d][i] >= h_vars[i] + h_vars[i+d] - 1)
                if binary:
                    # For binary case, more specifically:
                    # Constraint valid: y[i,d] >= min(h[i], h[i+d]) if h are binary.
                    # But in formulation we can't enforce y == h[i]*h[i+d] exactly if h[i] are not binary.
                    # Keep existing McCormick as it's tighter for continuous h.
                    pass
    
    return h_vars, y_vars, constraints

def solve_lp(n, solver='GLPK', timeout_s=30):
    """Solve LP relaxation for given n, return the candidate h."""
    try:
        h_vars, y_vars, constraints = build_lp_model(n, binary=False)
        
        t_var = cp.Variable(1, name='t')
        # Add the constraints for each shift d:
        for d in range(-(n-1), n):
            expr = 0
            for i in range(n):
                if 0 <= i + d < n:
                    expr += y_vars[d][i]
            # dx = 2 / n
            dx = 2.0 / n
            constraints.append(t_var >= (n/2) - expr / dx)
        
        # Objective: minimize t
        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        
        # Solve with timeout
        prob.solve(solver=solver, verbose=False, time_limit=timeout_s)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_vals = h_vars.value
            return h_vals
        else:
            # Fallback: just return uniform distribution
            return np.ones(n) * 0.5 * n / 2 / n
    
    except Exception as e:
        print(f"LP for n={n} failed: {e}")
        return np.ones(n) * 0.5 * n / 2 / n

def solve_milp(n, timeout_s=30):
    """Solve MILP for binary h."""
    try:
        h_vars = cp.Variable(n, integer=True)
        y_vars = {}
        constraints = [
            cp.sum(h_vars) == n / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        for d in range(-(n-1), n):
            y_vars[d] = {}
            for i in range(n):
                if 0 <= i + d < n:
                    y_vars[d][i] = cp.Variable(1, name=f'y_{i}_{d}')
                    # McCormick constraints:
                    constraints.append(y_vars[d][i] >= 0)
                    constraints.append(y_vars[d][i] <= h_vars[i])
                    constraints.append(y_vars[d][i] <= h_vars[i+d])
                    constraints.append(y_vars[d][i] >= h_vars[i] + h_vars[i+d] - 1)
        
        # Objective: minimize upper bound t.
        t_var = cp.Variable(1, name='t')
        # Add the constraints for each shift d:
        for d in range(-(n-1), n):
            expr = 0
            for i in range(n):
                if 0 <= i + d < n:
                    expr += y_vars[d][i]
            # If N/2 - sum(y[i,d]) <= t * dx then t >= (N/2 - sum(y[i,d])) / dx
            dx = 2.0 / n
            constraints.append(t_var >= (n/2) - expr / dx)
        
        # Objective to minimize t
        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        
        prob.solve(solver='CBC', verbose=False, time_limit=timeout_s)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            # Extract and retain only h values.
            h = h_vars.value
            if isinstance(h, list):
                h = np.array(h)
            return h
        else:
            return np.ones(n) * 0.5  # unsatisfactory fallback
    
    except Exception as e:
        print(f"MILP for n={n} failed: {e}")
        return np.ones(n) * 0.5

def subgradient_refine(h_start, max_iter=100000, time_limit_s=600):
    """Refine h_start using a projected subgradient method for max over shifts."""
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Keep track of time
    start_t = time.time()
    
    iter_count = 0
    last_improvement = 0
    
    # Sample smaller set of shifts for efficiency when n is large
    candidate_shifts = list(range(-(n//2), n//2 + 1))
    if len(candidate_shifts) > 50:
        # Only sample every few shifts to keep it manageable
        candidate_shifts = [s for s in candidate_shifts if (s % 3) == 0]
        # Also make sure d=0 is included if not already
        if 0 not in candidate_shifts:
            candidate_shifts.append(0)
    
    while iter_count < max_iter and (time.time() - start_t) < time_limit_s:
        iter_count += 1
        
        # Evaluate current h
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            last_improvement = iter_count
        
        # Decide which shift d gives maximal value of Correlation
        max_c = -float('inf')
        best_d = 0
        # Loop over a subset of possible shifts
        for d in candidate_shifts:
            # Correlation value is dp = dx * (sum(h) - sum(h[i]*h[i+d]))
            corr = 0.0
            for i in range(n):
                if 0 <= i + d < n:
                    corr += h[i] * h[i+d]
            val = dx * (n/2 - corr)
            if val > max_c:
                max_c = val
                best_d = d
                
        # Compute subgradient direction
        # g[i] = 1 - h[i+d] - h[i-d]  (accounting for zero-padding)
        g = np.full(n, 1.0)
        if best_d > 0:
            g[:-best_d] -= h[best_d:]
        elif best_d < 0:
            g[-best_d:] -= h[:best_d]
        else:
            g -= 2 * h
        
        # Normalise step size
        step = 0.5 / np.sqrt(iter_count)
        # Tentative update
        v = h - step * g
        
        # Projected update
        h_new = project_onto_box_and_sum(v, n/2)
        
        h = h_new
        
        # Perturb periodically if no improvement
        if iter_count - last_improvement > 5000:
            # Add noise
            noise = np.random.normal(0, 0.05, n)
            h += noise
            h = project_onto_box_and_sum(h, n/2)
            last_improvement = iter_count
            # Adjust step size to be larger for re-start
            step = 0.5
        
        # Occasionally output to be prepared
        if iter_count % 5000 == 0:
            eval_c5 = evaluate_erdos_solution(h)
            if eval_c5 < best_c5:
                best_h = h.copy()
                best_c5 = eval_c5
    
    return best_h, best_c5
