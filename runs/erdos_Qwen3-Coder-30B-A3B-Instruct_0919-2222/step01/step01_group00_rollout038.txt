import numpy as np
import time
from scipy.optimize import minimize

def proj_capped_simplex(v, target_sum):
    # Project onto the capped simplex: sum = target_sum, 0 <= v[i] <= 1
    v = np.array(v)
    n = len(v)
    # Binary search for lambda such that sum of clip(v + lambda, 0, 1) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(v + mid, 0, 1)
        sum_proj = np.sum(proj)
        if sum_proj < target_sum:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + lo, 0, 1)
    return proj

def full_corr(h):
    # Use FFT for full correlation; return real part.
    h = np.array(h)
    n = len(h)
    H = np.fft.rfft(h, n=2*n-1)
    G = np.fft.rfft(1-h, n=2*n-1)
    corr = np.fft.irfft(H * np.conj(G), n=2*n-1)
    return corr.real

def evaluate_h(h_values):
    # Compute C5
    h = np.array(h_values)
    dx = 2.0 / len(h)
    conv = np.correlate(h, 1-h, mode='full')
    return np.max(conv) * dx

def evaluate_h_fft(h_values):
    # Slower but more reliable for debugging
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = full_corr(h)
    return np.max(corr) * dx

def sa_step(h_s, block_len, T, δ_max=0.2):
    # Given block heights h_s, make a random move (delta mass swap)
    i, j = np.random.choice(len(h_s), size=2, replace=False)
    δ = np.random.uniform(-δ_max, δ_max)
    h_s_new = h_s.copy()
    h_s_new[i] += δ
    h_s_new[j] -= δ
    h_s_proj = proj_capped_simplex(h_s_new, target_sum=len(h_s)/2)
    return h_s_proj

def simulate_annealing(initial_h_s, n_points, block_len, T0, α, max_iter, budget_s, start_time):
    h_best = initial_h_s.copy()
    obj_best = np.inf
    T = T0
    iteration = 0
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = sa_step(h_best, block_len, T)
        obj_new = evaluate_h_fft(np.repeat(h_new, block_len))
        Δ = obj_new - obj_best
        if Δ < 0 or np.random.rand() < np.exp(-Δ / T):
            h_best = h_new
            obj_best = obj_new
        T *= α
        iteration += 1
    return h_best, obj_best

def pgd_step(h_s, block_len, α, η_start=0.05, epochs=200):
    h = np.repeat(h_s, block_len)
    η = η_start
    for epoch in range(epochs):
        corr = full_corr(h)
        c_max = np.max(corr)
        w = np.exp(α * (corr - c_max))
        w /= np.sum(w)
        g = np.zeros_like(h)
        # Compute gradient w.r.t h
        for i in range(len(h)):
            # dr = 1 - (h[i+1] if i+1<n else 0) - (h[i-1] if i-1>=0 else 0)
            # for corr at index i -> h[i] contributes 1 and h[i+1],h[i-1]
            # So we compute ∇_{h_i} corr_k based on flip of h_i and contributions from i+1, i-1
            # But since corr_k = Σ_t h[t] - Σ_t h[t]*h[t+k], grad of corr_k w.r.t h_i = 1 if i in k's support.
            # Actually, we want to approximate smooth max's gradient w.r.t h_s
            grad_wrt_h = np.zeros_like(h)
            for k in range(1, len(corr)-1):  # offset for correlation full mode
                if i+k < len(h):
                    grad_wrt_h[i] += w[k] * (-h[i+k])  # assuming h[i+k] due to k shift
                if i-k >= 0:
                    grad_wrt_h[i] += w[k] * (-h[i-k])
                grad_wrt_h[i] += w[k] * 1.0
            g[i] = grad_wrt_h[i]
        g_s = np.array([np.mean(g[i:i+block_len]) for i in range(0, len(g), block_len)])
        h_s -= η * g_s
        h_s = proj_capped_simplex(h_s, len(h_s)/2)
        h = np.repeat(h_s, block_len)
        η *= 0.95
    return h_s

def binary_climb(n_points):
    # binary hill climb
    h = np.zeros(n_points)
    ones = np.random.permutation(n_points)
    h[ones[:n_points//2]] = 1
    best_h = h.copy()
    best_c5 = np.inf
    for _ in range(5 * n_points * 1000):
        if _ % 1000 == 0:
            c5 = evaluate_h_fft(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        i = np.random.choice(np.where(h == 1)[0])
        j = np.random.choice(np.where(h == 0)[0])
        h[i], h[j] = h[j], h[i]
        c5 = evaluate_h_fft(h)
        if c5 >= best_c5:
            h[i], h[j] = h[j], h[i]  # revert
        elif c5 < best_c5:
            best_c5 = c5
    return best_h, best_c5

    np.random.seed(seed)
    start_time = time.time()
    n_points = 140
    block_len = 10  # n_points // 14 (S = 14)
    S = n_points // block_len

    # Initial

    h_initial = proj_capped_simplex(h_initial, target_sum=S/2)
    
    # Evaluate baseline
    baseline = evaluate_h_fft(h_initial)
    
    # Simulated Annealing
    T0 = 0.1 * baseline
    α = 0.997
    max_iter_SA = 10000
    sa_budget = 0.3 * budget_s
    best_h_sa = h_initial.copy()
    best_c5_sa = baseline
    for ch in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        δ_start = 0.2 / (2**ch)
        h_start = np.random.permutation(h_initial)
        h_opt, c5_opt = simulate_annealing(h_start, n_points, block_len, T0, α, max_iter_SA, sa_budget, start_time)
        if c5_opt < best_c5_sa:
            best_h_sa = h_opt
            best_c5_sa = c5_opt
    
    # PGD refinement
    pgd_budget = 0.3 * budget_s
    h_pgd = best_h_sa.copy()
    for α_val in [5, 20, 50, 200]:
        if time.time() - start_time > budget_s - 10:
            break
        h_pgd = pgd_step(h_pgd, block_len, α_val, epochs=200)
        c5_new = evaluate_h_fft(np.repeat(h_pgd, block_len))
        if c5_new < best_c5_sa:
            best_c5_sa = c5_new
            best_h_sa = h_pgd.copy()

    # Binary search
    bin_budget = 0.15 * budget_s
    start_bin = time.time()
    best_h_binary, best_c5_binary = binary_climb(n_points)
    if best_c5_binary < best_c5_sa:
        best_h_sa = best_h_binary
        best_c5_sa = best_c5_binary

    # Final refinement to full vector
    full_h = np.repeat(best_h_sa, block_len)
    full_h = proj_capped_simplex(full_h, target_sum=n_points/2)
    final_c5 = evaluate_h(full_h)
    
    return (full_h, final_c5, n_points)
