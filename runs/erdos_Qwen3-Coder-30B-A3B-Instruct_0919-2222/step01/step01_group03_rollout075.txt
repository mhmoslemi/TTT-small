import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different approaches with different sampling methods and algorithms
    iterations = 0
    max_iterations = 20
    
    for iter_no in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
        iterations += 1
        
        # Random perturbations
        h_perturbed = best_h.copy() + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed / (np.sum(h_perturbed) * dx) * 1.0
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        try:
            # Local optimization with this perturbed start
            local_result = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
        # Slightly different strategy - make some structural modifications
        try:
            h_modified = best_h.copy()
            # Randomly perturb some segments more heavily
            indices = np.random.choice(n_points, size=max(1, n_points // 5), replace=False)
            for idx in indices:
                h_modified[idx] = np.clip(h_modified[idx] + np.random.normal(0, 0.05), 0, 1)
            
            # Normalize
            h_modified = h_modified / (np.sum(h_modified) * dx) * 1.0
            h_modified = np.clip(h_modified, 0, 1)
            
            local_result = minimize(
                objective,
                h_modified,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final improvement with a hybrid approach
    h_final = best_h.copy()
    
    # Try a new hybrid approach: make it binary-like (0 or 1), but still close to integral constraint
    h_binary = np.zeros_like(h_final)
    # Distribute mass in more discrete way
    total_mass = np.sum(h_final) * dx
    # Spread it in a symmetric way
    central_region = n_points // 4
    mid_point = n_points // 2
    h_binary[mid_point - central_region // 2:mid_point + central_region // 2] = 1.0
    # Normalize to match integral constraint
    h_binary = h_binary / (np.sum(h_binary) * dx) * total_mass

    try:
        local_result = minimize(
            objective,
            h_binary,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
