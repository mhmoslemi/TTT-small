import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def project_onto_box_and_sum(v, target):
    """Project vector v onto [0,1] box and sum=target constraint."""
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def grad_for_lag(h, lag):
    """Compute gradient of correlation for a specific lag."""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # Overlap region with positive lag
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]  # derivative w.r.t h_i
        grad[i + lag] -= h[i]      # derivative w.r.t h_{i+lag}
    else:
        lag = -lag
        # Overlap region with negative lag
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad

def get_worst_lags(h, tol=1e-6):
    """Get lags that achieve maximum correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    # Find all lags that achieve maximum
    best_lags = []
    for i in range(len(corr)):
        if abs(corr[i] - max_corr) < tol:
            # Convert index to actual shift
            shift = i - (n - 1)
            best_lags.append(shift)
    return best_lags, corr * dx

        num_samples = min(2000, int(1000 * remaining_time / budget_s))

            # Compute correlations
            corr = np.correlate(b, b, mode='full')
            # Only consider positive lags (and their symmetric counterparts)
            min_overlap = np.min(corr[n_points:])  # avoid zero lag which is sum(b**2)
            if min_overlap > best_overlap:
                best_overlap = min_overlap
                # Smooth binary before getting h
                kernel = np.array([0.25, 0.5, 0.25])
                smoothed = np.convolve(b, kernel, mode='same')
                smoothed = np.clip(smoothed, 0, 1)
                # Project to sum constraint
                tmp_h = project_onto_box_and_sum(smoothed, target_sum)
                c5_sample = evaluate_c5(tmp_h)
                if c5_sample < best_c5:
                    best_c5 = c5_sample
                    best_h = tmp_h.copy()

    # Initialize Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)

    decay = 0.0001

            break
            
        # Get current worst lags
        best_lags, _ = get_worst_lags(best_h)
        
        # Compute gradient for the worst lags (average their gradient)
        avg_grad = np.zeros(n_points)
        for l in best_lags:
            grad = grad_for_lag(best_h, l)
            avg_grad += grad
        avg_grad /= len(best_lags)
        
        # Step with Adam
        m = beta1 * m + (1 - beta1) * avg_grad
        v = beta2 * v + (1 - beta2) * (avg_grad ** 2)

        v_hat = v / (1 - beta2 ** (iter_count + 1))
        best_h = best_h - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        iter_count += 1
        
        # Occasionally add noise for escape
        if iter_count % 500 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            best_h = best_h + noise
            best_h = project_onto_box_and_sum(best_h, target_sum)
    
    return (best_h, best_c5, n_points)
