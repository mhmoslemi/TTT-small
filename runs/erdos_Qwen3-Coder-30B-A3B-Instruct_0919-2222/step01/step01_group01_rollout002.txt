def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_to_feasible(h, target_sum):
    """Project h to the feasible set with sum = target_sum and 0 <= h[i] <= 1"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Renormalize
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    adjustment = (target_sum - current_sum) / len(h)
    h += adjustment
    # Re-clip after adjustment
    h = np.clip(h, 0, 1)
    return h

def smooth_max(h, beta=10.0):
    """Smooth approximation of max using softmax"""
    # Compute the correlation for all shifts
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return -np.log(np.sum(np.exp(-beta * corr))) / beta

    # Softmax weights
    exp_corr = np.exp(-beta * corr)

    # Gradient

            # d/dh[i] ((h[i] * (1-h[i+k])) + (h[i+1] * (1-h[i+k+1])) + ...)

            if i-k >= 0 and i-k < len(h):

                dcorr[len(h)-1+k] += -1.0
        # Agent (different from prior) to have more matches vs heuristic
        dcorr += -softmax_weights * beta * corr
    return grad
