import numpy as np
import time
import cvxpy as cp
from cvxpy import GLPK, GLPK_MI

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def project_sum_constraint(h, target_sum):
    """Project h to sum to target_sum using bisection"""
    low, high = 0.0, 1.0
    while high - low > 1e-10:
        mid = (low + high) / 2
        h_proj = np.clip(h + mid, 0, 1)
        h_proj *= target_sum / np.sum(h_proj)
        if np.sum(h_proj) > target_sum:
            high = mid
        else:
            low = mid
    h_proj = np.clip(h + low, 0, 1)
    h_proj *= target_sum / np.sum(h_proj)
    return h_proj
