def project_simplex(v, s=1.0):
    """Project v onto the probability simplex with sum = s using water-filling."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(u) + 1)
    candid = u - cssv / ind
    theta = np.max(np.where(candid >= 0, candid, -np.inf))
    return np.maximum(v - theta, 0)
