import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def project_onto_box_and_sum(v, target):
    """Project vector v onto box [0,1] and sum constraint"""
    # Find λ s.t. sum_i clip(v_i + λ, 0, 1) = target.
    # Use binary search as described.
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def compute_gradient_for_lag(h, lag):
    """Compute gradient of correlation with shift lag"""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # Overlap region i in [0, N - lag)
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]
        grad[i + lag] -= h[i]
    else:
        lag = -lag
        # Overlap region i in [lag, N)
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad

def random_binary(N, weight):
    """Generate a random binary vector with specified weight"""
    b = np.zeros(N, dtype=int)
    indices = np.random.permutation(N)[:int(weight)]
    b[indices] = 1
    return b

def compute_min_overlap(b):
    """Compute the minimum overlap for a binary vector"""
    corr = np.correlate(b, b, mode="full")
    # Exclude the zero lag
    return np.min(corr[len(corr)//2 + 1:])

def binary_hill_climbing(N, time_limit=20.0):
    """Find a good binary pattern"""
    start_time = time.time()
    best_b = random_binary(N, N//2)
    best_min_overlap = compute_min_overlap(best_b)
    
    while time.time() - start_time < time_limit:
        # Try a random swap
        one_idx = np.where(best_b == 1)[0]
        zero_idx = np.where(best_b == 0)[0]
        if len(one_idx) > 0 and len(zero_idx) > 0:
            i = np.random.choice(one_idx)
            j = np.random.choice(zero_idx)
            # Try the swap
            new_b = best_b.copy()
            new_b[i] = 0
            new_b[j] = 1
            new_min_overlap = compute_min_overlap(new_b)
            if new_min_overlap > best_min_overlap:
                best_b = new_b
                best_min_overlap = new_min_overlap
    
    return best_b

            shift = lag - (n_points - 1)

        m = beta1 * m + (1 - beta1) * grad
