import numpy as np
import time
import random
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, k):
    """Shift array by k positions, with zero padding"""
    n = len(arr)
    if k > 0:
        return np.concatenate([np.zeros(k), arr[:-k]])
    elif k < 0:
        return np.concatenate([arr[-k:], np.zeros(-k)])
    else:
        return arr.copy()

def project_onto_box_and_sum(x, target_sum):
    """Project x onto the constraints: 0≤xi≤1, sum(xi)=target_sum"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    
    # If already close to target, return
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Use bisection on lambda to adjust toward target sum
    # Define scalar function: sum(clip(x - lambda, 0, 1)) - target_sum
    # f(lambda) = sum(clip(x - lambda, 0, 1)) - target_sum
    # This is a monotone decreasing function of lambda
    # We want argmin lambda s.t. f(lambda)=0
    
    # Bisection search
    lambda_min = x.min() - 1.0
    lambda_max = x.max() + 1.0
    
    for _ in range(40):
        lambda_mid = (lambda_min + lambda_max) / 2.0
        x_projection = np.clip(x - lambda_mid, 0, 1)
        actual_sum = np.sum(x_projection)
        
        if actual_sum > target_sum:
            lambda_min = lambda_mid
        else:
            lambda_max = lambda_mid
    
    lambda_opt = (lambda_min + lambda_max) / 2.0
    
    result = np.clip(x - lambda_opt, 0, 1)
    
    # Final adjustment to ensure exact sum
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        # Add residual to entries that are not at bound
        nonbound = []
        for i in range(len(result)):
            if result[i] > 0 and result[i] < 1:
                nonbound.append(i)
        # Add to all non-bound entries
        if nonbound:
            add_per = residual / len(nonbound)
            for i in nonbound:
                result[i] += add_per
            result = np.clip(result, 0, 1)
    
    return result

def build_lp_model(n, binary=False):
    """Build the linear program with (possibly binary) variables"""
    # Variables
    h = cp.Variable(n, name="h")
    t = cp.Variable(name="t")  # Upper bound on C5
    if binary:
        h = cp.Variable(n, boolean=True, name="h")
    else:
        h = cp.Variable(n, name="h")
    
    # Y variables for products h[i] * h[i+d]
    # Let Y[d][i] = h[i] * h[i+d]
    y = {}
    for d in range(-(n-1), n):
        if d == 0:
            continue
        y[d] = cp.Variable(n - abs(d), name=f"y_{d}")

    # Constraints
    constraints = []
    
    # Box constraints
    constraints += [0 <= h, h <= 1]
    
    # Sum constraint
    constraints += [cp.sum(h) == n / 2]
    
    # McCormick constraints: 
    # For each shift d, over indices i with i+d < n:
    for d in range(-(n-1), n):
        if d == 0:
            continue
        # indices to process: i in [0, n - |d| - 1], such that i + d < n (i < n - d) and i >= 0
        start = max(0, -d)
        end = min(n - d, n)
        indices = list(range(start, end))
        
        for i in indices:
            dd = i + d
            y_var = y[d][i - start]  # Element in y[d] indexing from i
            # y >= max(0, h[i] + h[dd] - 1)
            # y <= min(h[i], h[dd])
            # Note: y[i,d] = h[i]*h[dd]
            constraints += [y_var >= 0]
            constraints += [y_var >= h[i] + h[dd] - 1]
            constraints += [y_var <= h[i]]
            constraints += [y_var <= h[dd]]
    
    # Replace C(d) = dx*(n/2 - sum_i h[i]*h[i+d])
    # Using linearisations: sum_i h[i]*h[i+d] = sum_i y[i,d]  
    # The constraint for max shift C(d): 
    # dx*(n/2 - sum_i y[i,d]) <= t
    # -> sum_i y[i,d] >= n/2 - t/dx
    for d in range(-(n-1), n):
        if d == 0:
            continue
        start = max(0, -d)
        end = min(n - d, n)
        indices = list(range(start, end))
        y_vars = [y[d][i - start] for i in indices] 
        # Total product sum
        sum_y = cp.sum(y_vars) if len(y_vars) > 0 else 0
        # dx is 2.0 / n
        dx = 2.0 / n
        constraints += [sum_y >= n / 2 - t / dx]
    
    # Objective
    obj = cp.Minimize(t)
    
    prob = cp.Problem(obj, constraints)
    return prob

def solve_model(prob, solver="GLPK_MI", timeout_s=20):
    """Solve the problem with timeout"""
    # Set solver with time limit
    try:
        prob.solve(solver=solver, time_limit=timeout_s)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:
            return True
        else:
            return False
    except Exception as e:
        return False

def subgradient_refine(h_start, end_time):
    """Refine starting vector using projected subgradient descent"""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    no_improvement_count = 0
    
    # Start learning rate
    alpha0 = 0.1
    iter_count = 0

    while time.time() < end_time:
        iter_count += 1
        current_c5 = evaluate_erdos_solution(h)
        # Find shift with worst correlation
        corr_full = np.correlate(h, 1 - h, mode='full') * dx
        d_star = np.argmax(corr_full)
        d_star = d_star - (n - 1)  # Convert from 0..2n-2 to -n+1..n-1

        # Compute direction g = 1 - shift(h, d*) - shift(h, -d*)
        g = np.ones(n)
        if d_star != 0:
            g = g - shift_array(h, d_star)
            g = g - shift_array(h, -d_star)

        # Update by subgradient step
        alpha = alpha0 / (1.0 + 0.01 * iter_count)  # decreasing factor
        v = h - alpha * g  # No clipping yet
        h = project_onto_box_and_sum(v, target_sum)

        # Check for improvement
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5 - 1e-10:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Perturbation strategy
        if no_improvement_count > 3000:
            noise = np.random.normal(0, 0.05, n)
            h = project_onto_box_and_sum(h + noise, target_sum)
            no_improvement_count = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Set random seeds
    random.seed(seed)
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    end_time = start_time + budget_s
    
    best_h = None
    best_c5 = float('inf')
    
    # Test a few discretization sizes
    n_values = [64, 70, 72, 80]
    prev_best_c5 = float('inf')
    best_n = 70
    
    for n in n_values:
        if time.time() >= end_time - 5:
            break
        print(f"Trying n={n}")
        try:
            prob = build_lp_model(n, binary=False)
            success = solve_model(prob, "GLPK", timeout_s=15)
            if success:
                h_lp = prob.variables()[0].value
                c5_eval = evaluate_erdos_solution(h_lp)
                print(f"LP solution at n={n}: C5 = {c5_eval}")
                if c5_eval < best_c5:
                    best_c5 = c5_eval
                    best_h = h_lp.copy()
                    best_n = n
                if c5_eval < 0.3807:
                    # Found desired bound; return immediately
                    return (best_h.tolist(), best_c5, best_n)
        except:
            continue

    # Try to do a MILP, possibly on smaller size
    attempts = []
    for n in n_values:
        if time.time() >= end_time - 10:
            break
        try:
            prob = build_lp_model(n, binary=True)
            success = solve_model(prob, "CBC", timeout_s=30)
            if success:
                h_milp = prob.variables()[0].value
                c5_eval = evaluate_erdos_solution(h_milp)
                print(f"MILP solution at n={n}: C5 = {c5_eval}")
                attempts.append((h_milp, c5_eval, n))
                if c5_eval < best_c5:
                    best_c5 = c5_eval
                    best_h = h_milp.copy()
                    best_n = n
                if c5_eval < 0.3807:
                    # Found desired bound; return immediately
                    return (best_h.tolist(), best_c5, best_n)
        except:
            continue

    # If best_h isn't already at a good solution, use subgradient refinement
    if best_c5 >= 0.3807:
        print("Running local refinement with projected subgradient descent...")
        try:
            refined_h, refined_c5 = subgradient_refine(best_h, end_time)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
                print(f"Final refined C5 = {best_c5}")
        except Exception as e:
            print("Refinement error:", e)

    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, best_n)
