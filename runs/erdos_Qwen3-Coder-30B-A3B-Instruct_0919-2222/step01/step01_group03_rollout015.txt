import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # First optimization with differential evolution for global search
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=10,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # More specialized local optimization with different strategies
    if time.time() - start_time < budget_s * 0.8:
        # Try a different initialization scheme
        h_trial = np.zeros(n_points)
        # Create a more concentrated function
        center = n_points // 2
        width = n_points // 4
        for i in range(max(0, center - width), min(n_points, center + width)):
            h_trial[i] = 1.0
        h_trial = h_trial / (np.sum(h_trial) * dx)  # Normalize to integral = 1
        h_trial = np.clip(h_trial, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

        # Try a smoother version of the solution
        h_smooth = np.convolve(best_h, np.ones(5)/5, mode='same')
        h_smooth = h_smooth / (np.sum(h_smooth) * dx)  # Normalize
        h_smooth = np.clip(h_smooth, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_smooth,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Local refinement with various local methods
    for method_name in ['L-BFGS-B', 'TNC', 'Powell']:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            local_result = minimize(
                objective,
                best_h,
                method=method_name,
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Try random restarts with different initializations
    for restart in range(3):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate random initialization with different proportions
        h_rand = np.random.rand(n_points) * 0.8 + 0.1  # Between 0.1 and 0.9
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
    # Additional gradient-free optimization
    if time.time() - start_time < budget_s * 0.95:
        try:
            de_result2 = differential_evolution(
                objective,
                bounds,
                maxiter=100,
                popsize=20,
                seed=seed+1,
                callback=lambda x, convergence: None,
                atol=1e-7,
                rtol=1e-7
            )
            if de_result2.success:
                h_opt = np.clip(de_result2.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
