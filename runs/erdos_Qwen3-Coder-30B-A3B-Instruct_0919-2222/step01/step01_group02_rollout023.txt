import numpy as np
import time

import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, d):
    """Shift h by d indices, with zero padding."""
    n = len(h)
    if d > 0:
        return np.concatenate(([0.] * d, h[:-d]))
    elif d < 0:
        return np.concatenate((h[-d:], [0.] * (-d)))
    else:
        return h.copy()

def project_onto_box_and_sum(v, target_sum, lb=0.0, ub=1.0):
    """Project v onto [lb,ub]^n and the hyperplane sum = target_sum using bisection."""
    n = len(v)
    
    def project(lambda_):
        return np.clip(v - lambda_, lb, ub)
    
    def sum_projected(lambda_):
        return np.sum(project(lambda_))
    
    # Bisection to find lambda such that sum_projected(lambda_) == target_sum
    low, high = -1.0, 1.0
    for _ in range(40):
        mid = (low + high) / 2.0
        s = sum_projected(mid)
        if s > target_sum:
            low = mid
        else:
            high = mid
    
    lambda_opt = (low + high) / 2.0
    return project(lambda_opt)

def build_lp_model(n, binary=False):
    """Builds LP/MILP model with McCormick constraints."""
    h = cp.Variable(n, boolean=binary)
    t = cp.Variable()
    
    # Constraint: sum h = n/2
    constraints = [cp.sum(h) == n / 2]
    
    # Add McCormick constraints y[i,d] = h[i]*h[i+d]
    y_vars = {}
    for i in range(n):
        for d in range(-n+1, n):
            if 0 <= i+d < n:
                j = i+d
                if binary:
                    # For binary variables, exact constraints
                    y = cp.Variable(boolean=True)
                    y_vars[(i,d)] = y
                    constraints.extend([
                        y <= h[i],
                        y <= h[j],
                        y >= h[i] + h[j] - 1,
                    ])
                else:
                    # For continuous variables, use McCormick relaxation
                    y = cp.Variable()
                    y_vars[(i,d)] = y
                    constraints.extend([
                        y >= 0,
                        y <= h[i],
                        y <= h[j],
                        y >= h[i] + h[j] - 1,
                    ])
    
    # Add constraint for each shift: sum h - sum y <= t/dx
    dx = 2.0 / n
    for d in range(-n+1, n):
        if d == 0:
            continue
        expr = 0
        for i in range(n):
            if 0 <= i + d < n:
                expr += y_vars[(i,d)]
        constraints.append((n/2 - expr) * dx <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    return prob, h, t, y_vars

def solve_with_timeout(prob, solver="GLPK", time_limit=20):
    """Solve the problem with a hard time limit."""
    try:
        prob.solve(solver=solver, verbose=False, max_time=time_limit)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return True, prob.value
        else:
            return False, None
    except Exception:
        return False, None

        iteration += 1

            best_c5 = c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        if last_improvement > patience:
            # Perturb
            h = h + np.random.normal(0, 0.05, size=n)
            h = project_onto_box_and_sum(h, n / 2)
            last_improvement = 0
            continue

        # Find worst shift d*:
        corr = np.correlate(h, 1 - h, mode='full')
        d_star = np.argmax(corr) - (n - 1)

        # Compute the gradient
        shifted_h = shift_array(h, -d_star)
        g = 1.0 - shifted_h - shift_array(h, d_star)
        g = np.roll(g, d_star, axis=0)  # fixed indexing

        # Update stepsize
        alpha = alpha0 / np.sqrt(iteration)
        
        # Tentative update
        v = h - alpha * g
        
        # Project
        h = project_onto_box_and_sum(v, n / 2)
        
    return best_h, best_c5
