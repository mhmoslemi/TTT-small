import numpy as np
from scipy.fft import rfft, irfft
import time

def proj_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= v_i <= 1, sum v_i = target_sum"""
    v = np.array(v, dtype=float)
    n = len(v)
    v_max = 1.0
    v_min = 0.0
    v = np.clip(v, v_min, v_max)
    total = np.sum(v)
    if np.abs(total - target_sum) < 1e-12:
        return v
    # Binary search for the lagrange multiplier
    lmin, lmax = -1000.0, 1000.0
    for _ in range(100):
        lam = (lmin + lmax) / 2
        v_pr = np.clip(v + lam, v_min, v_max)
        total_pr = np.sum(v_pr)
        if total_pr > target_sum:
            lmin = lam
        else:
            lmax = lam
        if np.abs(total_pr - target_sum) < 1e-10:
            break
    return np.clip(v + (lmin + lmax)/2, v_min, v_max)

def full_corr(h):
    """Use FFT to compute full correlation, return as same shape as np.correlate(full)"""
    n = len(h)
    # Pad to avoid circular correlation
    h_pad = np.pad(h, (0, n - 1), mode='constant')
    g_pad = np.pad(1 - h, (0, n - 1), mode='constant')
    H = rfft(h_pad)
    G = rfft(g_pad)
    corr = irfft(H * np.conj(G), n=2*n - 1)
    return corr

def evaluate_h(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    conv = full_corr(h)
    max_val = np.max(conv)
    return max_val * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    budget_remaining = lambda: budget_s - (time.time() - start_time)

    # Parameters and initialization
    n_points = 140
    dx = 2.0 / n_points
    S = 14               # number of blocks

    if len(h_initial) == n_points:
        init_h = h_initial.copy()
    else:
        # create a symmetric step function

        # promote to full (can be optimized more)
        h_initial = init_h.copy()
    init_h_s = np.array([np.mean(h_initial[i*block_len:(i+1)*block_len]) for i in range(S)])
    init_h_s = proj_capped_simplex(init_h_s, S / 2.0)

    best_h_s = init_h_s.copy()
    best_c5 = evaluate_h(np.repeat(best_h_s, block_len))

    # ==============================
    # Phase 1: Simulated Annealing
    # ==============================
    def sa_step(h_s, T):
        i, j = np.random.choice(S, 2, replace=False)
        delta = np.random.uniform(-0.2, 0.2)
        h_s_new = h_s.copy()
        h_s_new[i] += delta
        h_s_new[j] -= delta
        h_s_new = proj_capped_simplex(h_s_new, S / 2.0)
        return h_s_new

    def sa_run(h_s0, T0, alpha, max_iter):
        h_current = h_s0.copy()
        c_current = evaluate_h(np.repeat(h_current, block_len))
        best_h = h_current.copy()
        best_c = c_current
        T = T0
        for i in range(max_iter):
            if budget_remaining() < 5:
                break
            h_candidate = sa_step(h_current, T)
            c_candidate = evaluate_h(np.repeat(h_candidate, block_len))
            delta = c_candidate - c_current
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h_current = h_candidate
                c_current = c_candidate
                if c_current < best_c:
                    best_h = h_current.copy()
                    best_c = c_current
            T *= alpha
        return best_h, best_c

    # Run 3 independent SA runs
    sa_best_h_s = None
    sa_best_c5 = np.inf
    temps = [0.1*best_c5, 0.05*best_c5, 0.025*best_c5]
    alphas = [0.997, 0.998, 0.9985]
    for j in range(3):
        if budget_remaining() < 10:
            break
        T0 = temps[j]
        alpha = alphas[j]
        h_s_init = np.random.dirichlet(np.ones(S)) * (S/2.0)
        h_s_init = proj_capped_simplex(h_s_init, S/2.0)
        h_s_final, c5_final = sa_run(h_s_init, T0, alpha, int(1e5))
        if c5_final < sa_best_c5:
            sa_best_c5 = c5_final
            sa_best_h_s = h_s_final.copy()

    if sa_best_h_s is not None:
        best_h_s = sa_best_h_s.copy()
        best_c5 = sa_best_c5

    # ==============================
    # Phase 2: PGD with log-sum-exp smoothing
    # ==============================
    def smooth_max_grad(h_s, alpha=5):
        h_full = np.repeat(h_s, block_len)
        corr = full_corr(h_full)
        # Normalize for numerical stability
        c_max = np.max(corr)
        exp_c = np.exp(alpha * (corr - c_max))
        w = exp_c / np.sum(exp_c)
        # Gradient construction
        dw = np.zeros_like(h_full)
        for k in range(len(corr)):
            shift_k = k - (len(h_full) - 1) // 2
            if shift_k == 0:
                dw += w[k] * np.ones_like(h_full)
            else:
                hx = np.roll(h_full, shift_k)
                hx[shift_k > 0] = 0  # out-of-range for right shifts
                if shift_k < 0:
                    hx[:abs(shift_k)] = 0  # out-of-range for left shifts
                dw += w[k] * (-hx - np.roll(hx, -shift_k))
        # Collapse to block gradient
        g_block = np.array([np.mean(dw[i*block_len:(i+1)*block_len]) for i in range(S)])
        return g_block

    def pgd_run(h_s0, max_epochs=200):
        h_s = h_s0.copy()
        for epoch in range(max_epochs):
            if budget_remaining() < 5:
                break
            grad = smooth_max_grad(h_s)
            lr = 0.05 * (0.95 ** epoch)
            h_s -= lr * grad
            h_s = proj_capped_simplex(h_s, S / 2.0)
        return h_s

    # Run one or two PGD restarts
    iters = [0, 1]
    for i in iters:
        if budget_remaining() < 10:
            break
        pgd_h_s = pgd_run(best_h_s.copy())  # start from best SA state
        pgd_c5 = evaluate_h(np.repeat(pgd_h_s, block_len))
        if pgd_c5 < best_c5:
            best_c5 = pgd_c5
            best_h_s = pgd_h_s.copy()

    # ==============================
    # Phase 3: Binary hill climber
    # ==============================
    def binary_hill_climb(n_bits=140):
        # Start with uniformly random binary vector with exactly n_bits/2 ones
        h_bin = np.zeros(n_bits)
        ones = np.random.choice(n_bits, n_bits//2, replace=False)
        h_bin[ones] = 1.0
        best_c5 = evaluate_h(h_bin)
        best_h = h_bin.copy()
        for _ in range(20000 * n_bits):
            if budget_remaining() < 5:
                break
            i = np.random.choice(n_bits)
            if h_bin[i] == 0:
                continue  # invalid move
            j = np.random.choice(n_bits)
            if h_bin[j] == 1:
                continue  # invalid move
            # Swap i and j
            h_bin[i] = 0
            h_bin[j] = 1
            c5_new = evaluate_h(h_bin)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_bin.copy()
            else:
                # revert
                h_bin[i] = 1
                h_bin[j] = 0
        return best_h, best_c5

    binary_h, binary_c5 = binary_hill_climb(n_points)
    if binary_c5 < best_c5:
        best_c5 = binary_c5
        best_h_s = np.array([np.mean(binary_h[i*block_len:(i+1)*block_len]) for i in range(S)])

    # ==============================
    # Final processing and return
    # ==============================
    final_h = np.repeat(best_h_s, block_len)
    final_h = proj_capped_simplex(final_h, 1.0)

    final_c5 = evaluate_h(final_h)

    return (final_h, final_c5, n_points)
