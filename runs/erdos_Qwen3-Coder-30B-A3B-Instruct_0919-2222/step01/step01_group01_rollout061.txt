def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def compute_product_sums(h):
    """Compute product sum for each shift. Assume h has n elements."""
    n = len(h)
    # Product sum for shift s = sum_i h[i] * h[i+s] over valid i (i+s in range)
    # Full mode correlation via correlate with zero padding gives correct answer
    # but let's compute it manually for clarity: compute for each shift using overlap
    # The shift is from -(n-1) to (n-1)
    # For shift s in range, overlapping indices i, j = i+s
    # We compute for each shift s, sum of h[i]*h[j] where both are valid
    # This is equivalent to correlations cp.correlate(h, h, mode='full') where h is extended with zeros
    
    # Actually, we want h[i]*(1-h[i+s]), which in turn equals h[i] - h[i]*h[i+s].
    # So if h_s is shifted version of h: C[s] = sum(h) - sum(h[i]*h[i+s])  
    # Which is sum_h - sum(h.*shifted_h) 
    
    # Compute h.*h (element-wise product) over all shifts
    # Using the full correlation convention that correlates h with itself, then subtracted from sum
    # The middle dimension is 2n-1, indices from -(n-1) to (n-1)
    # Given: np.correlate(h, h, mode='full') gives convolution with zero padding
    
    # Use: np.correlate(h, h, mode='full') - but we want h times (1-h) or (h-h^2) so it's:
    # C[s] = sum_h - sum(h[i]*h[i+s]) = (sum h) - prod_sum[s] 
    # We compute backwards: prod_sum = correlate(h,h) = sum h[i]*h[i+s], then take difference
    
    # To compute correlation of h with (1-h) directly
    corr_h_1h = np.correlate(h, 1-h, mode="full") 
    return corr_h_1h

def project_simplex(h, target_sum):
    """Project h onto the simplex with sum = target and bounds [0,1] using binary search."""
    # Find lambda such that sum_clip(h - lambda) = target
    # tr = target_sum - sum(h) 
    # Clip (h - lambda) to [0,1]: f(lambda) = sum_i clip(h_i - lambda, 0, 1) = target
    high = max(h)   # if lambda >= max(h), then h_i - lambda <= 0 for all, clip = 0, sum = 0
    low = min(h) - 1 # if lambda <= min(h)-1, then h_i - lambda >= 1, clip = 1, sum = n
    
    def f(lambda_val):
        return np.sum(np.clip(h - lambda_val, 0, 1))

    for _ in range(50):
        mid = (low + high) / 2.0
        val = f(mid)
        if val > target_sum:
            low = mid
        else:
            high = mid
    
    projected = np.clip(h - low, 0, 1)
    # Adjust for numerical errors - rescale to target sum
    rescale_factor = target_sum / np.sum(projected)
    return projected * rescale_factor

def subgradient_ascent_with_projection(h, max_iter=2000, step_size=0.01):
    """Perform gradient ascent using subgradients on min product sum."""
    n = len(h)
    target_sum = n / 2.0
    
    for i in range(max_iter):
        # Compute product sums: C(s) = sum_h - prod_sums[s]  = sum_h - Σ h_i h_{i+s}
        # The min is C_max = max C(s) = max (sum_h - prod_sums[s]) = sum_h - min(prod_sums[s])
        # For better optimization: we want to maximize the minimal product sum: right to minimize max C(s) = minimize (sum_h - min(prod_sums)) = minimize min(prod_sums)
        # So: try to maximize min(prod_sums[s])
        prod_sums = np.correlate(h, h, mode="full")
        # Compute C(s) = sum(h) - prod_sums[s]
        C_s = target_sum - prod_sums  # sum_h = n/2
        
        # Find worst shift (min immediate product term)
        idx_min = np.argmin(prod_sums)
        s_min = idx_min - (n - 1)  # Real shift value
        
        # Compute subgradient of the product sum:
        # Gradient of sum_i h_i h_{i+s} w.r.t h_i = h_{i+s} + h_{i-s} (in shifted index set)
        # This is vectorized elsewhere - just compute gradient for worst shift
        g = np.zeros_like(h)
        # For shift s = s_min
        if s_min >= 0:
            g[:n - s_min] += h[s_min:]
            g[s_min:] += h[:n - s_min]
        else:
            # Negative shift: g[i] += h[i-g] if i-g >= 0
            s_abs = -s_min
            g[s_abs:] += h[:n - s_abs]
            g[:n - s_abs] += h[s_abs:]
        
        # Step
        h_new = h + step_size * g
        # Project to constraint
        h = project_simplex(h_new, target_sum)
        
    return h
