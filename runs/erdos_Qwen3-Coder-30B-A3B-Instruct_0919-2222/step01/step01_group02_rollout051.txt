import numpy as np
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, shift):
    """Shift array by shift amount (positive = right shift)"""
    n = len(arr)
    if shift >= 0:
        # For right shift: arr[i] -> arr[i-shift] for i >= shift
        # Zero pad at beginning (after shifting, first shift elements are lost)
        out = np.zeros(n)
        if shift < n:
            out[shift:] = arr[:n-shift]
    else:
        # For left shift: arr[i] -> arr[i+shift] for i < n+shift
        # Zero pad at end
        out = np.zeros(n)
        s = -shift
        if s < n:
            out[:n-s] = arr[s:]
    return out

def project_onto_box_simplex(v, lower=0.0, upper=1.0, target_sum=None):
    """Project v onto the set {x : lower <= x <= upper, sum(x) = target_sum}"""
    if target_sum is None:
        target_sum = len(v) / 2.0
    
    # Find good lambda by bisection
    left, right = -10.0, 10.0
    for _ in range(60):
        mid = (left + right) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            left = mid
        else:
            right = mid
            
    return np.clip(v - (left + right)/2.0, lower, upper)

def subgradient_optimize(n, end_time, start_h=None):
    """Optimize h using projected subgradient method"""
    # Initialize: either start_h or random
    if start_h is not None and len(start_h) == n:
        h = np.array(start_h)
    else:
        h = np.random.rand(n)
        
    # Normalize to meet sum constraint
    h = project_onto_box_simplex(h)
    
    best_h = h.copy()
    best_C = evaluate_erdos_solution(h)
    
    iteration = 0
    last_improve_iter = 0
    stagnation = 0
    
    while time.time() < end_time:
        iteration += 1
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Find index of max correlation
        idx = np.argmax(corr)
        d = idx - (n - 1)  # The shift amount
        
        # Compute direction using shift arrays
        shift_plus = shift_array(h, d)
        shift_minus = shift_array(h, -d)
        direction = shift_plus + shift_minus - 1.0
        
        # Step size (decreasing over time)
        alpha = 0.2 / np.sqrt(iteration)
        
        # Update h
        v = h + alpha * direction
        h = project_onto_box_simplex(v)
        
        # Evaluate
        cur_C = evaluate_erdos_solution(h)
        if cur_C < best_C - 1e-10:
            best_C = cur_C
            best_h = h.copy()
            last_improve_iter = iteration
            stagnation = 0
        else:
            stagnation += 1
            
        # If stagnated for a while, add some randomness
        if stagnation > 5000:
            h = best_h + np.random.normal(scale=0.02, size=n)
            h = project_onto_box_simplex(h)
            stagnation = 0
    
    return best_h, best_C

def run(seed=42, budget_s=600, **kwargs):
    # Set random seeds
    random.seed(seed)
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    best_h = None
    best_C = float('inf')
    best_n = None
    
    # Try various n values
    n_values = [70, 80, 90, 100, 110]
    
    for n in n_values:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
            
        # Allocate time for this iteration
        time_for_this = min(remaining * 0.3, 150)
        end_time = time.time() + time_for_this
        
        # Initialize using provided initial values if available
        start_h = None
        try:
            if 'initial_h_values' in globals():
                start_h = initial_h_values[:n] if len(initial_h_values) >= n else None
        except:
            pass
            
        h, C = subgradient_optimize(n, end_time, start_h)
        
        if C < best_C:
            best_C = C
            best_h = h
            best_n = n
            
        # Early stopping if we get below our target
        if best_C < 0.3807:
            break
    
    return (best_h.tolist(), float(best_C), int(best_n))
