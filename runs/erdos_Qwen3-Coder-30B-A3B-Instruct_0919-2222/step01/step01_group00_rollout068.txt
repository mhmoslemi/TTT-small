import numpy as np
from scipy.optimize import minimize
import time

def proj_capped_simplex(v, target_sum):
    # Project vector v onto the capped simplex: 0 <= v <= 1, sum(v) = target_sum
    # Uses binary search on lambda to find correct projection
    v = np.array(v, dtype=float)
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - target_sum) < 1e-12:
        return v
    elif s < target_sum:
        # Need to increase values
        # Solution is v + (target_sum - s) / n * ones
        return np.clip(v + (target_sum - s) / len(v), 0, 1)
    else:
        # Need to decrease values, use binary search on lambda
        lb, ub = 0.0, 1.0
        for _ in range(100):
            lam = (lb + ub) / 2.0
            newv = np.clip(v - lam, 0, 1)
            s_new = np.sum(newv)
            if abs(s_new - target_sum) < 1e-10:
                break
            elif s_new > target_sum:
                lb = lam
            else:
                ub = lam
        return np.clip(v - lam, 0, 1)

def full_corr(h):
    # Compute full correlation via FFT: h conv (1 - h)
    n = len(h)
    h_extended = np.concatenate([h, np.zeros(n-1)])
    h_conj = np.fft.rfft(h_extended)
    
    # Compute 1 - h as a vector, also extended
    inv_h = np.concatenate([1 - h, np.zeros(n-1)])
    inv_h_conj = np.fft.rfft(inv_h)
    
    # Convolution in frequency domain
    corr = np.fft.irfft(h_conj * np.conj(inv_h_conj), n=2*n-1)
    # The result already corresponds to full mode in np.correlate
    return corr

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = full_corr(h)
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx
