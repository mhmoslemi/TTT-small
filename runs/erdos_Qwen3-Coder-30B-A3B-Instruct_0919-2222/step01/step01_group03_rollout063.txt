import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Initialize with uniform distribution
    h_uniform = np.ones(n_points) * 0.5
    h_uniform = h_uniform / (np.sum(h_uniform) * dx) * 1.0
    
    try:
        result = minimize(
            objective,
            h_uniform,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Try enhanced evolution algorithm
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=20,
            seed=seed,
            atol=1e-8,
            rtol=1e-8,
            callback=lambda x, convergence: None
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Grid-based construction
    # Try a piecewise uniform construction
    n_piece = 6
    h_piece = np.zeros(n_points)
    for i in range(n_piece):
        start_idx = i * n_points // n_piece
        end_idx = (i + 1) * n_points // n_piece
        h_piece[start_idx:end_idx] = 1.0 / n_piece
    h_piece = h_piece / (np.sum(h_piece) * dx) * 1.0
    
    try:
        result = minimize(
            objective,
            h_piece,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Local search around best solution
    max_local_iters = 10000
    for i in range(max_local_iters):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Perturb current solution
        h_pert = best_h.copy()
        perturbation = np.random.normal(0, 0.01, n_points)
        h_pert = h_pert + perturbation
        
        # Project back to feasible domain
        h_pert = np.clip(h_pert, 0, 1)
        
        # Scale to satisfy constraint
        h_pert = h_pert / (np.sum(h_pert) * dx) * 1.0
        
        # Check objective
        c5_pert = objective(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
