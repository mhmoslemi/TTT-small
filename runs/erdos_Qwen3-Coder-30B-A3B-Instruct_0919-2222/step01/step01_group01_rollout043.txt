def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_to_simplex(v, s=1.0):
    """Project vector v onto the simplex {x : x >= 0, sum(x) = s} using water filling."""
    v = np.asarray(v)
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n + 1) > (css - s)) + 1
    theta = (css[rho - 1] - s) / rho
    return np.maximum(v - theta, 0)

def smooth_max(x, beta=100):
    """Smooth approximation of max using softmax."""
    return np.log(np.sum(np.exp(beta * x))) / beta

def smooth_max_grad(x, beta=100):
    """Gradient of smooth max."""
    exp_x = np.exp(beta * x)
    return exp_x / np.sum(exp_x)
