```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def solve_lp_relaxation(n_points):
    """Solve the linear programming relaxation with McCormick envelopes"""
    # Variables
    h = cp.Variable(n_points)
    u = cp.Variable((n_points, 2*n_points - 1))  # u[i, d] = h_i * h_{i+d}
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,
        h >= 0,
        h <= 1
    ]

    # McCormick envelopes for product terms
    for d in range(-n_points + 1, n_points):
        for i in range(n_points):
            j = i + d
            if 0 <= j < n_points:
                constraints += [
                    u[i, d + n_points - 1] >= 0,
                    u[i, d + n_points - 1] <= h[i],
                    u[i, d + n_points - 1] <= h[j],
                    u[i, d + n_points - 1] >= h[i] + h[j] - 1
                ]

    # Compute s_d and enforce t >= w - s_d
    w = n_points / 2
    for d in range(-n_points + 1, n_points):
        s_d = cp.sum([u[i, d + n_points - 1] for i in range(n_points) if 0 <= i + d < n_points])
        constraints += [t >= w - s_d]

    # Objective
    objective = cp.Minimize(t)

    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
    except:
        pass
    return None

def project_to_simplex(h, target_sum):
    """Project h to the simplex {h : Σ h = target_sum, h_i ∈ [0,1]}"""
    # Shift h to mean-zero
    mean_h = np.mean(h)
    h_shifted = h - mean_h
    # Adjust to match sum
    adjustment = (target_sum - np.sum(h_shifted)) / len(h_shifted)
    h_projected = h_shifted + adjustment
    # Clamp to [0, 1]
    h_projected = np.clip(h_projected, 0, 1)
    # Renormalize
    scale = target_sum / np.sum(h_projected)
    return h_projected * scale

def subgradient_step(h, d_star, dx):
    """Compute subgradient w.r.t. max shift d_star"""
    n = len(h)
    g = np.zeros_like(h)
    for i in range(n):
        if i + d_star >= 0 and i + d_star < n:
            g[i] -= h[i + d_star]
        if i - d_star >= 0 and i - d_star < n:
            g[i] -= h[i - d_star]
    return g

def project_gradient_descent(h_init, max_iter=500, eta=0.01, tol=1e-7):
    """Projected subgradient descent on the non-smooth C5"""
    h = h_init.copy()
    n = len(h)
    target_sum = n / 2
    dx = 2.0 / n
    
    for i in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        d_star = np.argmax(corr)
        g = subgradient_step(h, d_star, dx)
        h_new = h - eta * g
        # Project onto simplex
        h_new = project_to_simplex(h_new, target_sum)
        if np.linalg.norm(h - h_new) < tol:
            break
        h = h_new
        eta *= 0.99  # Decay step size
    return h

def sim Annealing_step(h, corr, T, n_points, dx):
    """Perform one simulated annealing step"""
    # Choose two indices at random
    i, j = np.random.choice(n_points, 2, replace=False)
    # Compute change in_corr due to exchange
    old_corr = corr
    delta = 0
    # Consider the contributions to the shifts where i/j appear
    for d in range(-n_points + 1, n_points):
        if i + d < n_points and i + d >= 0 and j + d < n_points and j + d >= 0:
            delta += (h[i] - h[j]) * (h[j+d] - h[i+d])
    delta = -delta * dx
    # Compute proposal
    if delta < 0 or np.random.rand() < np.exp(-delta / T):
        h[i], h[j] = h[j], h[i]
        corr += delta
    return h, corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Precompute possible n values to test
    n_candidates = [42, 48, 54, 60]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Save time for final evaluation
    remaining_budget = budget_s * 0.95

    for n in n_candidates:
        if time.time() - start_time > remaining_budget:
            break
            
        dx = 2.0 / n
        h0 = solve_lp_relaxation(n)
        if h0 is None:
            # Fallback to random initialization if LP fails
            h0 = np.random.rand(n)
            h0 = h0 * (n / (2 * np.sum(h0)))

        # Evaluate LP-based initialization
        c5_initial = evaluate_erdos_solution(h0)
        if c5_initial < best_c5:
            best_c5 = c5_initial
            best_h = h0.copy()
            best_n = n

        # Binary rounding from LP solution: take top half
        h_binary = np.zeros(n)
        indices = np.argsort(h0)[::-1][:n//2]
        h_binary[indices] = 1.0
        h_binary = project_to_simplex(h_binary, n / 2)
        c5_binary = evaluate_erdos_solution(h_binary)
        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_binary
            best_n = n

        # Simulated Annealing of fractional h
        if time.time() - start_time > remaining_budget * 0.9:
            break
        h_current = h0.copy()
        corr_current = np.correlate(h_current, 1 - h_current, mode="full")
        max_corr = np.max(corr_current)
        c5_current = (max_corr) * dx
        T0 = 0.02
        T = T0
        i_sa = 0
        sa_budget = remaining_budget * 0.3
        while i_sa < 5000 and time.time() - start_time < sa_budget:
            h_new, corr_new = sim_annealing_step(h_current, corr_current, T, n, dx)
            corr_current = corr_new
            c5_new = max(corr_new) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
                h_current = h_new.copy()
            else:
                h_current = h_new
                corr_current = corr_new
            i_sa += 1
            T = T * 0.995
                
        # Gradient Descent refinement
        if time.time() - start_time > remaining_budget * 0.95:
            break
        refined = project_gradient_descent(h_current)
        c5_refined = evaluate_erdos_solution(refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = refined
            best_n = n           

    # Final check and normalization
    if best_h is not None:
        # Ensure it's normalized
        best_h = project_to_simplex(best_h, best_n / 2)
        # Final evaluation
        final_c5 = evaluate_erdos_solution(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h, best_c5, best_n)
```
