def direct_optimization(h0, bounds, cons, objective, start_time, budget_s):
    # Run several optimizations with small random perturbations
    best_h = h0.copy()
    best_c5 = objective(best_h)
    
    # Track time
    remaining_time = budget_s - (time.time() - start_time) - 1
    if remaining_time < 1:
        return best_h, best_c5
        
    # Do multiple restarts
    for i in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Slight random perturbation to seed
        pert = np.random.normal(0, 0.03, len(h0))
        x0 = np.clip(h0 + pert, 0, 1)
        
        # Optimize with SLSQP
        try:
            res = minimize(objective, x0, bounds=bounds, constraints=cons,
