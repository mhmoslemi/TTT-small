import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points, n_steps):
    # Build full h array from step heights
    seg_len = n_points // n_steps
    # Repeat each height seg_len times
    h = np.repeat(heights, seg_len)
    return h

def evaluate_heights(heights, n_points, n_steps):
    # Build full h array and compute C5 with penalty for sum constraint
    h = build_h(heights, n_points, n_steps)
    c5 = evaluate_h(h)
    # Add penalty for sum constraint
    sum_heights = np.sum(heights)
    penalty = 10.0 * (sum_heights - n_steps/2)**2
    return c5 + penalty

def project_to_bounds_and_sum(heights, target_sum):
    # Project heights to [0,1] and sum to target_sum
    # Simple algorithm: adjust all heights so sum equals target
    # while keeping each height in [0,1]
    h = np.array(heights)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to match sum
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if np.abs(diff) < 1e-6:
        return h
    # The adjustment method: uniformly distribute adjustment
    adjusted = h + diff / len(h)
    # Ensure all values are in [0,1], adjust others accordingly
    min_val = np.min(adjusted)
    max_val = np.max(adjusted)
    if min_val < 0 or max_val > 1:
        # Use water-filling method for projection
        # We solve for amount to adjust each to keep them in [0,1] and sum to target_sum
        if min_val < 0:
            adjusted = np.clip(adjusted, 0, 1)
            deficit = target_sum - np.sum(adjusted)
            if deficit > 0:
                # Increase values that are 0 to make sum correct
                idx = np.where(adjusted == 0)[0]
                if len(idx) > 0:
                    adjustment = deficit / len(idx)
                    adjusted[idx] = adjustment
                    # Make sure no value exceeds 1
                    adjusted = np.clip(adjusted, 0, 1)
            else:
                # Value below 0, so values used to get deficit
                extra = np.sum(adjusted) - target_sum
                if extra > 0:
                    # Decrease some values that are 1
                    idx = np.where(adjusted == 1)[0]
                    if len(idx) > 0:
                        reduction = extra / len(idx)
                        adjusted[idx] = np.maximum(0, adjusted[idx] - reduction)
                    adjusted = np.clip(adjusted, 0, 1)
        elif max_val > 1:
            adjusted = np.clip(adjusted, 0, 1)
            excess = np.sum(adjusted) - target_sum
            if excess > 0:
                # Decrease some values that are 1 to make sum correct
                idx = np.where(adjusted == 1)[0]
                if len(idx) > 0:
                    reduction = excess / len(idx)
                    adjusted[idx] = np.maximum(0, adjusted[idx] - reduction)
                    adjusted = np.clip(adjusted, 0, 1)
    else:
        adjusted = np.clip(adjusted, 0, 1)
    return adjusted

        initial_sum = np.sum(initial_heights)
        initial_heights = initial_heights * (n_steps/2) / initial_sum
    
    bounds = [(0, 1) for _ in range(n_steps)]
    
    best_c5 = float('inf')
    best_heights = None
    
    # Run multiple random restarts
    max_restarts = min(10, int(300 / (n_points * 0.001)))
