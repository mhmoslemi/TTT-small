import numpy as np
import time
import random
import cvxpy as cp

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, shift):
    n = len(h)
    if shift == 0:
        return h.copy()
    elif shift > 0:
        return np.concatenate([np.zeros(shift), h[:-shift]])
    else:
        return np.concatenate([h[-shift:], np.zeros(-shift)])

def project_onto_box_and_sum(x, target_sum, box_min=0, box_max=1):
    # Clip x to box [0,1]
    x = np.clip(x, box_min, box_max)
    current_sum = np.sum(x)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return x
    # Adjust by a small amount uniformly
    if diff > 0:
        # Need to increase some values
        # Choose those farthest from upper bound
        sort_idx = np.argsort(x)
        for i in range(len(x)):
            if diff < 1e-10:
                break
            max_increase = box_max - x[sort_idx[i]]
            actual_increase = min(max_increase, diff)
            x[sort_idx[i]] += actual_increase
            diff -= actual_increase
    else:
        # Need to decrease values
        sort_idx = np.argsort(-x)
        for i in range(len(x)):
            if diff > -1e-10:
                break
            max_decrease = x[sort_idx[i]] - box_min
            actual_decrease = min(max_decrease, -diff)
            x[sort_idx[i]] -= actual_decrease
            diff += actual_decrease
    return x

def build_lp_model(n, binary=False, sum_target=None):
    if sum_target is None:
        sum_target = n / 2
    h = cp.Variable(n, name="h")
    t = cp.Variable(1, name="t")
    # Binary flag for integer constraints
    if binary:
        h = cp.Variable(n, boolean=True, name="h")
    else:
        h = cp.Variable(n, name="h")
    constraints = [cp.sum(h) == sum_target]
    constraints += [h[i] >= 0 for i in range(n)]
    constraints += [h[i] <= 1 for i in range(n)]

    n_pairs = 0
    # Create auxiliary variables for each shifted pair
    # y[i,d] = h[i] * h[i+d] for all valid d
    # Total number of terms = sum_over_d (n - |d|) = n^2
    y_vars = {}
    for d in range(1 - n, n):
        d_abs = abs(d)
        for i in range(n - d_abs):
            if d > 0:
                y_vars[(i, d)] = cp.Variable(1, name=f"y_{i}_{d}") if not binary else cp.Variable(1, boolean=True, name=f"y_{i}_{d}")
            else:
                y_vars[(i, d)] = cp.Variable(1, name=f"y_{i}_{d}") if not binary else cp.Variable(1, boolean=True, name=f"y_{i}_{d}")

    # McCormick constraints
    for d in range(1 - n, n):
        if d == 0:
            continue  # Already handled by h[i] variables
        d_abs = abs(d)
        for i in range(n - d_abs):
            k = i + d
            if d > 0:
                y = y_vars[(i, d)]
                constraints += [y >= 0,
                                y <= h[i],
                                y <= h[k],
                                y >= h[i] + h[k] - 1]
            else:
                y = y_vars[(i, d)]
                constraints += [y >= 0,
                                y <= h[k],
                                y <= h[i],
                                y >= h[k] + h[i] - 1]

    # Constraint for each shift:
    # sum_i h[i] - sum_i h[i] h[i+d] <= t * dx = t / n
    dx = 2.0 / n
    for d in range(1 - n, n):
        if d == 0:
            continue
        d_abs = abs(d)
        # Compute indices for h[i] and h[i+d]
        h_term = cp.sum(h[:n - d_abs])
        prod_term = 0
        for i in range(n - d_abs):
            k = i + d
            if d > 0:
                prod_term += y_vars[(i, d)]
            else:
                prod_term += y_vars[(k, d)]
        constraints += [h_term - prod_term <= t / dx]
    return cp.Problem(cp.Minimize(t), constraints), h

def solve_with_timeout(problem, solver='GLPK_CMD', timeout=30):
    """Wrap cvxpy's solving to respect timeout."""
    import time
    start = time.time()
    try:
        problem.solve(solver=solver, maxTime=timeout, verbose=False)
        elapsed = time.time() - start
        if elapsed > timeout:
            return False
        return True
    except Exception:
        return False

def subgradient_refine(h_start, end_time, max_iter=100000):
    # Measure current C5
    n = len(h_start)
    dx = 2.0 / n
    h = np.array(h_start, dtype=float)
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    sum_target = n / 2

    iteration = 0
    no_improve_count = 0
    max_no_improve = 5000
    alpha = 0.5

    # Precompute full correlation for initial setup (but better evaluate each time we do shift check)
    # Start iterations
    while time.time() < end_time and iteration < max_iter:
        iteration += 1
        # Get all correlations
        corr_vals = np.correlate(h, 1 - h, mode='full')
        best_shift = np.argmax(corr_vals)
        # useful to handle 0 vs others, or there could be hundreds of shifts wide.
        # convert from full index to relative shift
        shift_len = len(h)
        actual_shift = best_shift - (shift_len - 1)  # we index p=0,...,2n-2 to represent shifts -n+1,...,n-1
        # take shifted h
        shifted_h = shift_array(h, actual_shift)
        # x - shifted_h is effect on correlation as derivative for worst shift
        # compute gradient descent up to some step alpha (that decreases over time).
        # g(i) = h[i+shift] + h[i-shift] - 1 = 1 - reduced from 1 - (h[i] * (1-h[i+shift])) + if boundaries paid attention to...
        # Invent a less formal gradient check: <h[i], h[i+shift]> becomes equivalent to <h[i], h[i+shift] * 1> so do some derivative ...
        # Invert as direction for gradient to decrease worst
        # g[i] := 1 - h[i+shift] - h[i-shift] (formal dash for gradient descent)
        g = np.ones(shift_len)
        if actual_shift > 0:
            # shift_h[0] = 0, 
            g[:actual_shift] = 0
            g[actual_shift:] -= h[:-actual_shift]
        elif actual_shift < 0:
            g[actual_shift:] = 0
            g[:actual_shift] -= h[-actual_shift:]
        else:
            g = 1 - 2 * h

        # back to gradient formula idea:
        # Find shift d* that makes max of C = dx * (sum h_i - sum h_i * h_{i+d*})
        # so we are differentiating the shift part part: C = sum h_i * h_{i+d} 
        # partial / partial h[i] = h_{i+d} (they're at most shifted by d*) for a fixed d*
        # So, determining direction directly as gradient for temporary worst shift
        g = 1 - shifted_h - shift_array(h, -actual_shift)
        # Compute update step
        step_size = alpha / np.sqrt(1 + iteration)  # diminishing
        # Add small noise when stuck?
        # Take out some detailed tolerance here, assume that all values clipped due to elbow shape do make an angle into corner.
        # Local update
        u = h - step_size * g  # note, we are going in opposite sign to maximize.
        # Project onto feasible region
        u = project_onto_box_and_sum(u, sum_target)
        # Check performance
        candidate_c5 = evaluate_erdos_solution(u)
        # Update if improvement
        if candidate_c5 < current_c5:
            h = u.copy()
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = u.copy()
                best_c5 = candidate_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
        # Perturb if stuck
        if no_improve_count > max_no_improve:
            # add noise slightly around
            noise = np.random.normal(0, 0.05, n)
            h += noise
            h = project_onto_box_and_sum(h, sum_target)
            no_improve_count = 0
            current_c5 = evaluate_erdos_solution(h)

    # Return result from best solution among all
    return best_h, best_c5

    for n in target_sizes:
