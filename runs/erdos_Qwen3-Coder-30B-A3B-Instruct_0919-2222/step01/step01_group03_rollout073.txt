import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different optimization approaches
    
    # 1. Differential evolution with different parameters
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=300,
            popsize=25,
            seed=seed,
            atol=1e-8,
            rtol=1e-8
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # 2. Local optimization starting from DE result
    try:
        local_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 3000, 'ftol': 1e-10, 'gtol': 1e-10},
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # 3. Try random restarts with different strategies
    for restart in range(20):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random restart with trend-based initialization
        h_rand = np.random.rand(n_points)
        
        # Try to create some structure - perturb the initial guess a bit
        h_copy = h0.copy()
        h_copy += (np.random.rand(n_points) - 0.5) * 0.1
        h_copy = np.clip(h_copy, 0, 1)
        
        # Try projecting to normalization constraint
        h_rand = h_copy / (np.sum(h_copy) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1500, 'ftol': 1e-10, 'gtol': 1e-10},
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # 4. Try a variant of gradient-free optimization with more iterations
    try:
        # Use COBYLA as an alternative method
        cobyla_result = minimize(
            objective,
            best_h,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 5000, 'tol': 1e-10},
        )
        if cobyla_result.success:
            h_opt = np.clip(cobyla_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
