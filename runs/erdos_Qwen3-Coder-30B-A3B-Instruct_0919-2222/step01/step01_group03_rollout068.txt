import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Iterative refinement approach using multiple local optimization starts
    
    # 1. Global search with DE
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=15,
            seed=seed,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # 2. Local optimization from the DE result
    try:
        local_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # 3. Random restarts with constructed starting points
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Try a different construction
        h_rand = np.random.rand(n_points) 
        # Scale to have correct integral
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        
        # Make a small perturbation around the best solution so far
        if restart % 3 == 0 and restart > 0:
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_rand = h_perturbed
        
        h_rand = np.clip(h_rand, 0, 1)
        
        # Local optimization
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final optimization with a more sophisticated approach
    # Try some piecewise constant constructions
    def create_piecewise_h():
        # Try creating piecewise constant function with fewer intervals
        h = np.zeros(n_points)
        intervals = [4, 6, 8, 10]
        for n_intervals in intervals:
            try:
                # For n_intervals pieces
                intervals = np.linspace(0, n_points, n_intervals+1, dtype=int)
                scaled_vals = np.random.rand(n_intervals)
                scaled_vals = scaled_vals / np.sum(scaled_vals) * (n_points / n_intervals) * 1.0
                for i in range(n_intervals):
                    start, end = intervals[i], intervals[i+1]
                    h[start:end] = scaled_vals[i]
                
                # Renormalize
                h = h / (np.sum(h) * dx) * 1.0
                h = np.clip(h, 0, 1)
                return h
            except:
                continue
        return h0
    
    # Try a few piecewise uniform constructions
    for _ in range(5):
        h_piecewise = create_piecewise_h()
        
        try:
            local_result = minimize(
                objective,
                h_piecewise,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
