import numpy as np
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, shift):
    """Shift array by 'shift' positions with zero padding."""
    n = len(arr)
    if shift >= 0:
        # Shift left by shift (index i -> i+shift)
        out = np.concatenate([arr[shift:], np.zeros(shift)])
    else:
        # Shift right by |shift| (index i -> i+shift)
        s = -shift
        out = np.concatenate([np.zeros(s), arr[:n-s]])
    return out

def project_onto_box_simplex(v, lower=0., upper=1., target_sum=None):
    """Project vector v onto box constraints [lower,upper] and sum constraint."""
    if target_sum is None:
        target_sum = len(v) / 2.0
    
    # Bisection for λ to satisfy sum(clamp(v_i - λ, lower, upper)) = target_sum
    lo = -10.0
    hi = 10.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, lower, upper)

def subgradient_optimize(n, end_time, start_h=None):
    """Optimize h using subgradient method."""
    # Initialize h
    if start_h is not None and len(start_h) == n:
        h = np.array(start_h)
    else:
        h = np.random.rand(n)
    
    # Ensure sum is exactly n/2
    h = project_onto_box_simplex(h, target_sum=n/2)
    
    best_h = h.copy()
    best_C = evaluate_erdos_solution(h)
    iteration = 0
    last_improve_iter = 0
    stagnation = 0
    
    while time.time() < end_time:
        iteration += 1
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # find max index
        idx = np.argmax(corr)
        d = idx - (n - 1)  # shift amount
        
        # compute direction: h_shifted_plus + h_shifted_minus - 1
        shift_plus = shift_array(h, d)
        shift_minus = shift_array(h, -d)
        direction = shift_plus + shift_minus - 1.0
        
        # step size
        α = 0.5 / np.sqrt(iteration)
        
        # update
        v = h + α * direction
        h = project_onto_box_simplex(v)
        
        # evaluate
        cur_C = evaluate_erdos_solution(h)
        if cur_C < best_C - 1e-10:
            best_C = cur_C
            best_h = h.copy()
            last_improve_iter = iteration
            stagnation = 0
        else:
            stagnation += 1
            
        # Periodic random restart
        if stagnation > 5000:
            h = np.random.rand(n)
            h = project_onto_box_simplex(h, target_sum=n/2)
            stagnation = 0
            
    return best_h, best_C

def run(seed=42, budget_s=600, **kwargs):
    # Set random seeds
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    end_time = start_time + budget_s
    
    best_h = None
    best_C = float('inf')
    best_n = None
    
    # Try several values of n
    # Try to use initial_h_values if available, else start from scratch
    initial_h = None
    
    try:
        # Check if global initial_h_values is available
        initial_h_values
        if 'initial_h_values' in globals() and len(initial_h_values) > 10:
            initial_h = initial_h_values.copy()
    except:
        pass
    
    # Try different sizes
    n_values = [70, 80, 90, 100, 110]
    
    for n in n_values:
        if time.time() >= end_time - 10:
            break
            
        # Allocate time for this n
        remaining = end_time - time.time()
        if remaining < 20:
            break
            
        t_alloc = min(remaining * 0.4, 200)
        sub_end_time = time.time() + t_alloc
        
        try:
            h, C = subgradient_optimize(n, sub_end_time, start_h=initial_h)
            if C < best_C:
                best_C = C
                best_h = h
                best_n = n
        except:
            continue
            
        if best_C < 0.3807:
            break
    
    if best_h is None:
        # Fallback to standard procedure
        n = 70
        h = np.ones(n) * 0.5
        h *= (n / 2) / np.sum(h)
        best_h = h
        best_C = evaluate_erdos_solution(h)
        best_n = n
        
    return (best_h.tolist(), float(best_C), int(best_n))
