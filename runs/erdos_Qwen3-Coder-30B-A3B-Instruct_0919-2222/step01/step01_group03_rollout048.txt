import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different optimization strategies
    # Strategy 1: Differential Evolution with restarts
    for restart in range(3):
        if time.time() - start_time > budget_s * 0.8:
            break
        try:
            de_result = differential_evolution(
                objective,
                bounds,
                maxiter=150,
                popsize=20,
                seed=seed + restart,
                callback=lambda x, convergence: None,
                atol=1e-7,
                rtol=1e-7
            )
            if de_result.success:
                h_opt = np.clip(de_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Strategy 2: Local optimization from best solution so far
    try:
        local_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 3000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Strategy 3: Random restarts with different initialization methods
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate different initializations
        # Method 1: Small perturbations of current best
        h_rand = best_h + np.random.normal(0, 0.05, n_points)
        h_rand = np.clip(h_rand, 0, 1)
        
        # Scale to match constraint
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Strategy 4: Try different objective functions to diversify search
    # Try to minimize a modified objective that penalizes large gradients
    def smooth_objective(h):
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        # Add penalty for large gradients to encourage smoother solutions
        grad_penalty = 0.0
        if len(h) > 1:
            grad = np.gradient(h)
            grad_penalty = np.sum(np.abs(grad)) * 0.01
        return c5 + grad_penalty
    
    try:
        # Use a variant of differential evolution with smooth objective
        de_result = differential_evolution(
            smooth_objective,
            bounds,
            maxiter=100,
            popsize=15,
            seed=seed + 100,
            callback=lambda x, convergence: None,
            atol=1e-7,
            rtol=1e-7
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
