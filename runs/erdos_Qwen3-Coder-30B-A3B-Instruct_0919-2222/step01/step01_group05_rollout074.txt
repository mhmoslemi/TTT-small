```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Grid sizes to try
    grid_sizes = [42, 48, 60]
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Use initial values if proper size
        if n == 42:
            h_init = initial_h_values.copy()
        else:
            h_init = np.random.rand(n)
        
        h_init = h_init * (n / (2 * np.sum(h_init)))
        
        # Build the MILP problem
        # Variables: b_i, z_{i,d}, s_d, t
        b = Variable(n, boolean=True)
        s = [Variable(1, integer=True) for _ in range(n)]
        z = {}
        z_vars = {}
        
        # Domain consistency for z variables
        # Each z_i,d variable represents b_i * b_{i+d}
        for d in range(1, n):  # shifts from 1 to n-1
            for i in range(n - d):
                key = (i, d)
                z[key] = Variable(1, boolean=True)
                z_vars[key] = z[key]
        
        # Constraints
        constraints = []
        
        # Fix b_0 = 1 to break symmetry
        constraints.append(b[0] == 1)
        
        # Sum constraint
        constraints.append(sum(b) == n // 2)
        
        # McCormick envelopes for z_{i,d} = b_i * b_{i+d}
        for d in range(1, n):
            for i in range(n - d):
                key = (i, d)
                constraints.extend([
                    z[key] <= b[i],
                    z[key] <= b[i + d],
                    z[key] >= b[i] + b[i + d] - 1
                ])
        
        # Compute s_d = sum of z_{i,d}
        for d in range(n):
            if d == 0:
                # For d = 0, the overlap is with self, like previous
                idxs = list(range(n))
                tail = [z[(i, d)] for i in idxs if (i, d) in z]
                constraints.append(s[d] == sum(tail))
            else:
                # For d > 0
                idxs = list(range(n - d))
                tail = [z[(i, d)] for i in idxs if (i, d) in z]
                constraints.append(s[d] == sum(tail))
        
        # Max shift constraint for all d
        t = Variable(1)
        for d in range(n):
            constraints.append(t >= n//2 - s[d])
        
        # Objective: minimize t
        obj = cp.Minimize(t)
        
        prob = cp.Problem(obj, constraints)
        
        # Try solvers in order
        solver_list = [cp.CBC, cp.GLPK_MI, cp.SCIP]
        success = False
        sol = None
        for solver in solver_list:
            try:
                prob.solve(solver=solver, verbose=False, maxTime=250)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    sol = b.value
                    success = True
                    break
            except:
                continue
        if not success:
            continue  # Skip this n if no solver succeeded
            
        # Convert to numpy array
        h_binary = np.array([int(round(b_val)) for b_val in sol])
        
        # Compute the best overlap bound
        dx = 2.0 / n
        h_float = h_binary.astype(float)
        corr = np.correlate(h_float, 1 - h_float, mode="full")
        c5_milp = max(corr) * dx
        
        if c5_milp < best_c5:
            best_h = h_float.copy()
            best_c5 = c5_milp
            best_n = n
            
    # If no MILP solution found, fallback to using initial_h_values
    if best_h is None:
        h_fallback = initial_h_values.copy()
        h_fallback = h_fallback * (42 / (2 * np.sum(h_fallback)))
        best_h = h_fallback.copy()
        best_c5 = evaluate_erdos_solution(best_h)
        best_n = 42
    
    # Now perform fractional refinement with Simulated Annealing
    h_start = best_h.copy()
    c5_start = best_c5
    # Use more samples to refine if needed
    n_samples = 60
    
    # Refinement procedure
    # Sync the normalization constant
    h_current = h_start * (n_samples / (2 * np.sum(h_start)))
    
    loop_time_limit = budget_s - (time.time() - start_time)
    start_refinement = time.time()
    
    # Parameters
    T0 = 0.02 * c5_start
    alpha = 0.995
    delta_max = 0.2
    max_iter = 1000000
    early_stop = 5000
    last_improvement = 0
    current_c5 = None
    
    best_h_refined = h_current.copy()
    best_c5_refined = 1e10
    
    # Initial energy calculation
    current_c5 = evaluate_erdos_solution(h_current)
    best_c5_refined = current_c5
    
    # Fine-tune with simulated annealing
    for iter_i in range(max_iter):
        if iter_i - last_improvement > early_stop:
            break
        if time.time() - start_refinement > loop_time_limit * 0.95:
            break
            
        # Pick two random indices
        i = np.random.randint(0, n_samples)
        j = np.random.randint(0, n_samples)
        
        # Avoid same index
        while i == j:
            j = np.random.randint(0, n_samples)
            
        # Find min(u_i, 1 - u_j, delta_max)
        u_i = h_current[i]
        u_j = h_current[j]
        delta = min(u_i, 1 - u_j, delta_max)
        
        if delta <= 1e-8:
            continue
            
        # Sample amount
        delta_u = np.random.uniform(0, delta)
        
        # Make the update
        h_new = h_current.copy()
        h_new[i] -= delta_u
        h_new[j] += delta_u
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to sum to n_samples / 2
        h_new = h_new * (n_samples / (2 * np.sum(h_new)))
        
        # Evaluate new energy
        c5_new = evaluate_erdos_solution(h_new)
        
        # Metropolis acceptance criterion
        if c5_new <= current_c5:
            h_current = h_new
            current_c5 = c5_new
            if c5_new < best_c5_refined:
                best_c5_refined = c5_new
                best_h_refined = h_new.copy()
                last_improvement = iter_i
        else:
            # Accept with probability exp( -(C5_new - C5_old) / T )
            T = T0 * (alpha ** iter_i)
            if T > 1e-10:
                prob_accept = np.exp(-(c5_new - current_c5) / T)
                if np.random.rand() < prob_accept:
                    h_current = h_new
                    current_c5 = c5_new
                    if c5_new < best_c5_refined:
                        best_c5_refined = c5_new
                        best_h_refined = h_new.copy()
                        last_improvement = iter_i

    # Final check and return
    final_h = best_h_refined.copy()
    final_c5 = best_c5_refined
    final_n = n_samples
    
    # Ensure normalisation
    final_h = final_h * (final_n / (2 * np.sum(final_h)))
    final_h = np.clip(final_h, 0, 1)
    
    # Final evaluation of the actual value
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h, final_c5, final_n)
```
