import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try multiple different optimization approaches
    
    # 1. Differential evolution with custom parameters
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=100,
            popsize=20,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-7,
            rtol=1e-7
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # 2. Local optimization starting from the best found so far
    try:
        local_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # 3. Try a hybrid evolution+local approach
    # Generate a more sophisticated initial solution
    h_sophisticated = np.zeros(n_points)
    # Create a two-part function: high values at the beginning, low at the end
    split_point = n_points // 3
    h_sophisticated[:split_point] = 0.8
    h_sophisticated[split_point:] = 0.2
    h_sophisticated = h_sophisticated / (np.sum(h_sophisticated) * dx) * 1.0
    
    try:
        local_result = minimize(
            objective,
            h_sophisticated,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # 4. Gradient-based local search for better convergence
    # Use a different initial point to avoid local minimums
    try:
        h_random = np.random.rand(n_points)
        h_random = h_random / np.sum(h_random) / dx * 1.0
        h_random = np.clip(h_random, 0, 1)
        
        local_result = minimize(
            objective,
            h_random,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-11, 'gtol': 1e-11},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # 5. Random restarts with adaptive sampling
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Create a more random, less-structured function
        h_rand = np.random.rand(n_points) * 0.2 + 0.4  # values around 0.4-0.6
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            # First with L-BFGS-B
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-11, 'gtol': 1e-11},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
