import numpy as np
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, shift):
    """Shift array by given amount with zero-padding"""
    n = len(arr)
    if shift >= 0:
        return np.concatenate([arr[shift:], np.zeros(shift)])
    else:
        s = -shift
        return np.concatenate([np.zeros(s), arr[:n-s]])

def project_onto_box_simplex(v, lower=0., upper=1., target_sum=None):
    """Project vector onto box constraints and sum constraint using bisection"""
    if target_sum is None:
        target_sum = len(v) / 2
    # Bisection for λ to satisfy sum(clamp(v_i - λ, lower, upper)) = target_sum
    lo = -10.0
    hi = 10.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, lower, upper)
