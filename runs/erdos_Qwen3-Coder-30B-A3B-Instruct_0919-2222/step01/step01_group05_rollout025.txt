import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    # Ensure reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    remaining_budget = budget_s
    
    # Define grid sizes to try
    grid_sizes = [42, 48, 60]
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # For each grid size
    for n in grid_sizes:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Define solver time limit (in seconds)
        solver_time_limit = min(250, remaining_budget / len(grid_sizes))
        
        # Binary model
        b = cp.Variable(n, integer=True)
        z = {}
        s = {}
        
        # For each shift d
        total_shift_d = 0
        # For d = 1, ..., n-1
        for d in range(1, n):
            for i in range(n - d):
                z[i, d] = cp.Variable(integer=True)
                total_shift_d += 1
        
        # Use symmetry for d < 0, but model only d >= 0
        # s_d are the number of overlaps for shift d
        for d in range(n):
            s[d] = cp.Variable(integer=True)  # Integers, linear anyway
        
        # Constraints for z - bilinear terms
        constraints = []
        for d in range(1, n):
            for i in range(n - d):
                # z[i][d] <= b[i]
                constraints.append(z[i, d] <= b[i])
                # z[i][d] <= b[i+d]
                constraints.append(z[i, d] <= b[i+d])
                # z[i][d] >= b[i] + b[i+d] - 1
                constraints.append(z[i, d] >= b[i] + b[i+d] - 1)
        
        # Define s_d
        for d in range(n):
            if d == 0:
                # s[0] = sum of b_i itself (i.e., always 1 since s[0] counts overlaps at shift 0)
                # Actually just sum of z_i,0 where 0 is self-overlap - but that is always a contribution
                # we’ll fix this
                one_term = b[:n].sum()
                # s[0] = sum of b_i*b_{i} = sum of b_i (since binary)
                s[0] = one_term  # Same as sum(b)
            else:
                # s[d] = sum_{i=0}^{n-d-1} z[i][d]
                expr = cp.sum([z[i, d] for i in range(n - d)])
                constraints.append(s[d] == expr)
        
        # Sum constraint
        constraints.append(cp.sum(b) == n / 2)
        
        # Prevent symmetry
        constraints.append(b[0] == 1)
        
        # To make it binary at all, we’ll constrain b to [0,1]
        constraints.extend([b[i] >= 0 for i in range(n)])
        constraints.extend([b[i] <= 1 for i in range(n)])
        
        # Constraint that for each z[i,d], it is binary
        for i in range(n - 1):
            for d in range(1, n - i):
                constraints.append(z[i, d] >= 0)
                constraints.append(z[i, d] <= 1)
        
        # Objective: minimize t
        t = cp.Variable()
        # For all shifts d:
        for d in range(n):
            constraints.append(t >= n/2 - s[d])
        objectives = [t]
        
        # Create problem
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        # Solve with solver options
        try:
            prob.solve(solver=cp.CBC, verbose=False, maximumSeconds=solver_time_limit)
        except Exception:
            continue  # If solver fails, move to next n
            
        # Extract values
        if prob.status == 'optimal':
            b_val = b.value
        else:
            # Fallback to random solution if MILP failed
            b_val = np.random.rand(n)
            b_val = np.clip(b_val, 0, 1)
            b_val = b_val * (n / (2 * np.sum(b_val)))  # normalize
    
        # Convert to float for further processing
        h_milp = np.array(b_val, dtype=float)
        h_milp = np.clip(h_milp, 0, 1)
        h_milp = h_milp * (n / (2 * np.sum(h_milp)))
        c5_milp = evaluate_erdos_solution(h_milp)
        
        # If this is better than current best, try iterative refinement
        if c5_milp < best_c5:
            best_h = h_milp.copy()
            best_c5 = c5_milp
            best_n = n
        
    # Apply iterative refinement here if necessary
    if best_h is not None and remaining_budget > 300:
        h_refined = best_h.copy()
        t0 = 0.02 * best_c5
        temperature = t0
        alpha = 0.995
        delta_max = 0.2
        step_limit = 100000
        step = 0
        
        # Track return
        best_h_ref = h_refined.copy()
        best_c5_ref = best_c5
        
        while step < step_limit and time.time() - start_time < 0.98 * budget_s:
            i = np.random.randint(0, len(h_refined))
            j = np.random.randint(0, len(h_refined))
            if i == j:
                continue
                
            # Find valid delta
            delta_max_i = min(h_refined[i], 1 - h_refined[j])
            delta = min(delta_max, delta_max_i)
            if delta <= 0:
                continue
            delta = np.random.uniform(0, delta)
            
            # Propose move
            new_h = h_refined.copy()
            new_h[i] -= delta
            new_h[j] += delta
            
            c5_old = evaluate_erdos_solution(h_refined)
            c5_new = evaluate_erdos_solution(new_h)
            
            # Accept or reject
            if c5_new <= c5_old:
                h_refined = new_h.copy()
            else:
                prob_accept = np.exp(-(c5_new - c5_old) / temperature)
                if np.random.rand() < prob_accept:
                    h_refined = new_h.copy()
            
            # Update temperature
            temperature *= alpha
            step += 1
            
            # Normalize after move
            sum_h = np.sum(h_refined)
            if abs(sum_h - n/2) > 1e-12:
                h_refined = h_refined * (n / 2) / sum_h
                h_refined = np.clip(h_refined, 0, 1)
                
            # Keep best so far
            c5_new_ref = evaluate_erdos_solution(h_refined)
            if c5_new_ref < best_c5_ref:
                best_h_ref = h_refined.copy()
                best_c5_ref = c5_new_ref
                
        best_h = best_h_ref.copy()
        best_c5 = best_c5_ref
        
    # Final re-normalization
    if best_h is not None:
        best_h = best_h * (best_n / (2 * np.sum(best_h)))
        best_h = np.clip(best_h, 0, 1)
