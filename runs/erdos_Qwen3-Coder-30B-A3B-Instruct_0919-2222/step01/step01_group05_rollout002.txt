import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt
import cvxpy as cp
from cvxpy import ECOS

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def project_onto_simplex(h, target_sum):
    """Project vector h onto the simplex with sum = target_sum"""
    # Shift to center around zero
    mean_h = np.mean(h)
    h_shifted = h - mean_h
    # Re-center to satisfy target sum
    shift = (target_sum - np.sum(h_shifted)) / len(h_shifted)
    h_proj = h_shifted + shift
    # Clamp to [0, 1]
    h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def lp_initialize(n):
    """Initialize h using convex relaxation via an LP with McCormick envelopes."""
    dx = 2.0 / n
    # LP variables
    h = cp.Variable(n, name="h")
    u = cp.Variable((n, 2*n-1), name="u")
    t = cp.Variable(1, name="t")
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [
        cp.sum(h) == n / 2,
        h >= 0,
        h <= 1
    ]
    
    # McCormick envelopes for u[i,d] = h[i] * h[i+d]
    for i in range(n):
        for d in range(-n+1, n):
            if 0 <= i+d < n:
                idx = d + n - 1  # index in u array
                constraints += [
                    u[i, idx] >= 0,
                    u[i, idx] <= h[i],
                    u[i, idx] <= h[i+d],
                    u[i, idx] >= h[i] + h[i+d] - 1
                ]
    
    # Define s_d = sum_{i: 0 <= i+d < n} u[i, d+n-1], add t constraint
    for d in range(-n+1, n):
        if d >= 0:
            s_d = sum(u[i, d+n-1] for i in range(n-d))
        else:
            s_d = sum(u[i, d+n-1] for i in range(-d, n))
        constraints.append(t >= n/2 - s_d)
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return np.array(h.value)
    except:
        pass
    # Fallback to even distribution if failed
    return np.ones(n) * (n / 2) / n

def binary_round(h):
    """Round the fractional h to a binary vector with correct sum"""
    n = len(h)
    # Sort indices by decreasing h[i]
    idxs = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    count = 0
    for i in idxs:
        if count < n // 2:
            h_bin[i] = 1
            count += 1
        else:
            break
    return h_bin

def compute_worst_shift(h):
    """Compute the shift d that maximizes integral h*(1-h_d)"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.argmax(corr)

def perform_swap(h, d_star):
    """Try to swap elements to reduce C5(d_star)"""
    n = len(h)
    # Find best i where h[i] = 1 and h[i+d_star] = 0
    # and best j where h[j] = 0 and h[j+d_star] = 1
    for i in range(n):
        if h[i] == 1:
            i_plus_d = i + d_star
            if 0 <= i_plus_d < n and h[i_plus_d] == 0:
                for j in range(n):
                    if h[j] == 0:
                        j_plus_d = j + d_star
                        if 0 <= j_plus_d < n and h[j_plus_d] == 1:
                            # Perform swap
                            h[i], h[j] = h[j], h[i]
                            return True
    return False

def hill_climb_binary(h, max_iter=1000):
    """Hill climb using targeted swap moves"""
    h = h.copy()
    for iteration in range(max_iter):
        d_star = compute_worst_shift(h)
        if not perform_swap(h, d_star):
            break
    return h

def simulated_annealing(h0, n_points, max_iters=2000, T0=0.02):
    """Simulated Annealing with fractional moves"""
    T = T0
    h = h0.copy()
    T = max(1e-6, T0 * 0.995)
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for iteration in range(max_iters):
        # Pick a pair to adjust
        i, j = np.random.randint(0, n_points, 2)
        if i == j:
            continue
        
        # Limit delta based on available space
        delta_max = min(h[i], 1 - h[j], 0.1)  # added limit
        delta = np.random.uniform(0, delta_max)
        
        # Apply move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_points / (2 * np.sum(h_new)))
        
        # Accept the new solution
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        elif np.random.rand() < np.exp(-(c5_new - current_c5) / T):
            h = h_new
            current_c5 = c5_new

        # Cool temperature
        T = max(1e-6, T * 0.995)
    
    return best_h, best_c5

def projected_gradient_descent(h0, n_points, max_iter=200):
    """Gradient descent with projection"""
    h = h0.copy()
    step = 0.01 * 2.0 / n_points  # ad-hoc scaled stepsize
    for _ in range(max_iter):
        d_star = compute_worst_shift(h)
        # Compute subgradient
        g = np.zeros(n_points)
        # For shift d_star, the derivative w.r.t h[i] is -h[i-d_star] - h[i+d_star]
        for i in range(n_points):
            j = i + d_star
            k = i - d_star
            # Clamp to valid indices
            if 0 <= j < n_points:
                g[i] -= h[j]
            if 0 <= k < n_points:
                g[i] -= h[k]
        
        # Update
        h = h - step * g
        # Make sum correct
        h = project_onto_simplex(h, n_points / 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Candidate n values, ordered by increasing cost with a focus on reasonable large n
    n_candidates = [42, 48, 54, 60]  # less than 1k, for manageable run
    best_c5 = float('inf')
    best_h = None
    best_n = None
    remaining_budget = budget_s
    
    for n in n_candidates:
        if time.time() - start_time > remaining_budget * 0.9:
            break
            
        dx = 2.0 / n
        
        # LP Initialization
        h_lp = lp_initialize(n)
        if h_lp is None:
            h_lp = np.ones(n) * 0.5  # fallback
        
        # Normalize to sum = n/2
        h_lp = h_lp * (n / (2 * np.sum(h_lp)))
        c5_lp = evaluate_erdos_solution(h_lp)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            best_n = n
            
        # Binary clustering + Swap improvement
        h_bin = binary_round(h_lp)
        h_improved = hill_climb_binary(h_bin)
        c5_improved = evaluate_erdos_solution(h_improved)
        if c5_improved < best_c5:
            best_c5 = c5_improved
            best_h = h_improved.copy()
            best_n = n
            
        # Simulated Annealing
        sa_max = int(500 * n / 42)
        h_sa, c5_sa = simulated_annealing(h_improved, n, sa_max)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n
            
        # Gradient Descent
        h_grad = projected_gradient_descent(h_sa, n, max_iter=100)
        c5_grad = evaluate_erdos_solution(h_grad)
        if c5_grad < best_c5:
            best_c5 = c5_grad
            best_h = h_grad.copy()
            best_n = n
            
        # Avoid redundant checks
        remaining_budget = budget_s - (time.time() - start_time)
    
    # Final normalization (ensure exact integral)
    best_h = best_h * (best_n / (2 * np.sum(best_h)))
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, best_n)
