    def solve_milp_binary(h0, n):
        # Create variables
        b = cp.Variable(n, boolean=True)

        s = {}  # s[d] = overlap for shift d
        t = cp.Variable()

        # Constraints
        constraints = [cp.sum(b) == n // 2, b >= 0, b <= 1, t >= 0]
        
        # Add symmetry breaking
        constraints.append(b[0] == 1)
        
        for d in range(1, n):
            for i in range(n - d):
                key = (i, d)
                z[key] = cp.Variable(boolean=True)
                constraints.extend([
                    z[key] <= b[i],
                    z[key] <= b[i+d],
                    z[key] >= b[i] + b[i+d] - 1
                ])
            s[d] = cp.sum([z[(i, d)] for i in range(n - d)])
            constraints.append(t >= n//2 - s[d])

        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        solver = CBC  # Fallback to GLPK_MI if CBC unavailable
        try:
            prob.solve(solver=solver, verbose=False, maxTime=200)
        except:
            try:
                prob.solve(solver=GLPK_MI, verbose=False, maxTime=200)
            except:
                return h0, 1e6  # Failed fallback

            h_sol = b_sol.astype(float)
            return h_sol, 1e6  # Return binary solution
        else:
            return h0, 1e6  # Fallback to init

        return best_h, best_c5

        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = max(corr) * dx
