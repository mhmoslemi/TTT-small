import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set initial parameters
    grid_sizes = [42, 48, 60]
    best_h = None
    best_c5 = float('inf')
    best_n = None

    start_time = time.time()
    remaining_budget = budget_s

    # Phase 1: solve MILP for binary case using CBC
    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Allow 300s for this phase, or remaining time if less
        milp_time = min(300, remaining_budget / len(grid_sizes))
        
        try:
            # Variables
            b = cp.Variable(n, integer=True)
            t = cp.Variable()
            
            # Create z variables for products
            # z_{i,d} to represent b_i * b_{i+d}
            num_z = 0
            z_vars = {}
            z_cons = []
            s = {}
            # For each shift d from 1 to n-1
            for d in range(1, n):
                for i in range(n - d):
                    z = cp.Variable(integer=True)
                    z_vars[(i, d)] = z
                    num_z += 1
                    # McCormick inequalities  
                    z_cons.append(z <= b[i])
                    z_cons.append(z <= b[i + d])
                    z_cons.append(z >= b[i] + b[i + d] - 1)
            
            # For s_d = sum z_{i,d}
            for d in range(n):
                s[d] = cp.sum([z_vars[(i, d)] for i in range(n - d) if (i, d) in z_vars]) 
            
            # Objective: minimize t
            objective = cp.Minimize(t)
            constraints = [
                cp.sum(b) == n // 2,
                b >= 0,
                b <= 1,
                t >= n // 2 - s[d] for d in range(n)
            ]
            constraints += z_cons

            # Add symmetry-breaking via b_0 = 1
            constraints.append(b[0] == 1)
            
            # Solve
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False, maxSeconds=milp_time)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_milp = b.value
                # Convert to float vector
                h_milp = np.array([float(x) for x in h_milp])
                # Normalize sum to exactly n/2
                h_milp = h_milp * (n / 2) / np.sum(h_milp)
                # Evaluate C5
                c5 = evaluate_erdos_solution(h_milp)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_milp.copy()
                    best_n = n
        except:
            continue

    # If we found a good binary solution (or we had one already as initial_h_values),
    # proceed to fractional refinement using Simulated Annealing
    # unless already sufficiently small
    
    if best_h is None:
        # Start with a reasonable initialization if we did not get a better binary solution
        h = initial_h_values.copy()
        h = h * (len(h) / (2 * np.sum(h)))
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = len(h)

    # Phase 2: Fractional refinement with Simulated Annealing
    if best_h is not None:
        h_desc = best_h.copy()
        c5_desc = best_c5
        
        dx = 2.0 / best_n
        # Define temperature schedule
        T0 = 0.02 * c5_desc
        T = T0
        alpha = 0.995
        delta_max = 0.2
        step_count = 0
        keep_going = True

        # Use remaining time for SA
        sa_start = time.time()
        sa_timeout = 0.95 * (budget_s - (time.time() - start_time))

        # Track last improvement
        last_improvement = 0
        patience = 5000
        
        # Timing critical to avoid overflow
        def is_timeout():
            return time.time() - sa_start >= sa_timeout
            
        while not is_timeout():
            # Generate a move: pick two indices i,j where h[i] > 0 and h[j] < 1
            idx_i = np.random.choice(np.where(h_desc > 1e-10)[0])
            idx_j = np.random.choice(np.where(h_desc < 1 - 1e-10)[0])
            if idx_i == idx_j:
                continue
                
            # Compute transfer amount
            max_delta = min(h_desc[idx_i], 1 - h_desc[idx_j], delta_max)
            if max_delta <= 1e-8:
                continue
                
            delta = np.random.uniform(0, max_delta)
            
            # Perform move
            old_h = h_desc.copy()
            old_c5 = c5_desc
            h_desc[idx_i] -= delta
            h_desc[idx_j] += delta
            
            # Ensure it's still in [0,1]
            h_desc = np.clip(h_desc, 0, 1)
            
            # Renormalize to maintain integral
            h_desc = h_desc * (best_n / 2) / np.sum(h_desc)
            
            # Compute new C5
            c5_new = evaluate_erdos_solution(h_desc)

            # Accept or reject
            if c5_new <= old_c5:
                c5_desc = c5_new
                step_count += 1
                last_improvement = step_count
            else:
                p = np.exp(-(c5_new - old_c5) / T)
                if np.random.uniform() < p:
                    c5_desc = c5_new
                    step_count += 1
                    last_improvement = step_count
                else:
                    # restore state
                    h_desc = old_h
            T = T0 * (alpha ** step_count)  # cooling schedule
            delta_max = 0.5 * delta_max  # shrinking step size
            
            if step_count - last_improvement > patience:
                break
        
        # Final check of normalization and evaluation
        h_final = np.clip(h_desc, 0, 1)
        h_final = h_final * (best_n / 2) / np.sum(h_final)
        final_c5 = evaluate_erdos_solution(h_final)
        
        # Return best
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final
            best_n = len(h_final)

    # Soft final clips
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (best_n / 2) / np.sum(best_h)

    return (best_h, best_c5, best_n)
