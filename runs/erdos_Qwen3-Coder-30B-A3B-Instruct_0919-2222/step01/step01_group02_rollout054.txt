import numpy as np
import cvxpy as cp
import time
import math
from cvxpy import GLPK, GLPK_MI, ECOS, SCS
from scipy.optimize import minimize

def binary_search_feasible(h, n_points, dx, C_target, max_iters=100):
    """Binary search for feasibility of a given C_target using LP"""
    low = 0.0
    high = C_target
    best_feasible = high
    epsilon = 1e-4  # tolerance for stopping

    for _ in range(max_iters):
        if high - low < epsilon:
            break
        mid = (low + high) / 2.0
        try:
            # Formulate feasibility problem
            h_vars = cp.Variable(n_points)
            t = cp.Variable()
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2, t <= mid]
            
            # For each shift, create linear constraint from McCormick-like approx
            # Instead of full quadratic, we just check if feasible given fixed t
            # Since we already have h from previous steps, we can verify directly using correlation
            
            corr_values = np.correlate(h, 1-h, mode='full') * dx
            if np.max(corr_values) <= mid:
                low = mid
                best_feasible = mid
            else:
                high = mid
        except:
            high = mid
    return best_feasible

def project_sum_to_n2(h, n_points):
    """Project h such that sum(h) = n/2"""
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-8:
        h = h * (n_points / 2) / total
    return h

def truncate_to_unit_interval(h):
    """Ensure h values are in [0,1]"""
    return np.clip(h, 0, 1)

def evaluate_c5(h):
    """Evaluate C5 for a given h vector"""
    n = len(h)
    dx = 2.0 / n
    # Full correlation; max gives C5
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def sequential_linear_programming_routine(h, n_points, dx, max_iter=5000, tolerance=1e-5):
    """Refine h using sequential linearizations"""
    h = h.copy()
    old_c5 = float('inf')
    
    for iteration in range(max_iter):
        corr_values = np.correlate(h, 1-h, mode='full') * dx
        worst_shift_idx = np.argmax(corr_values)
        worst_corr = corr_values[worst_shift_idx]
        
        # Formulate a linear approximation around current h for the worst shift
        # Linearize: sum h_i * h_{i + shift} ≈ sum (constant terms + linear approx)
        # For each d, we build:
        # sum of h[i] * h[i+d] ≈ 
        # h[i]^old * h[i+d]^old + h[i]^old * (h[i+d] - h[i+d]^old) + h[i+d]^old * (h[i] - h[i]^old)
        # So it simplifies to sum(h[i]^old * h[i+d] + h[i+d]^old * h[i] - h[i]^old * h[i+d]^old)
        # Which we can rearrange to
        # (sum of h[i]^old * h[i+d]) + (sum of h[i+d]^old * h[i]) - sum(h[i]^old * h[i+d]^old)
        # We build a constraint version of 
        # t >= dx*( n/2 - (sum of linear_terms for shift num) )

        # We linearize only for the worst shift
        d = -(n_points - 1) + worst_shift_idx
        
        # approximate Q(d) = sum_{i} h[i] * h[i+d] with linearization around h_old
        y_linearized = 0.0
        linear_term = 0.0
        cross_term = 0.0

        # iterate over indices for this shift d
        # A more symmetric approach: build sum h[i]*h[i+d]
        # Linear approximation around the old value:
        # h[i]*h[i+d] ≈ h[i]^old * h[i+d]^old + h[i]^old*(h[i+d] - h[i+d]^old) + h[i+d]^old*(h[i] - h[i]^old)
        # Which sums to:
        # h[i]^old * h[i+d]^old + h[i]^old * h[i+d] - h[i]^old * h[i+d]^old + h[i+d]^old * h[i] - h[i]^old * h[i+d]^old
        # == h[i+d]^old * h[i] + h[i]^old * h[i+d] - h[i]^old * h[i+d]^old

        # We rewrite a full linearized term for shift d
        # y(d) = sum_{i} y_{i,d} with y_{i,d} = h[i]*h[i+d]
        # At current h, y_{i,d} ≈ h[i]^old * h[i+d]^old + h[i]^old*(h[i+d]-h[i+d]^old) + h[i+d]^old*(h[i]-h[i]^old)

        linearized_sum = 0.0
        
        # however for numerical efficiency, we build in a lower-bound way:
        # let's create variables for the full shift d correlation, use linear programming
        
        # build linear problem
        # variables: [h[i]] and [t]
        # we want to bound the worst shift's correlation via linear inequalities
        # Based on our analysis, we can write: 
        # for shift d: sum_{i} h[i] * h[i+d]
        # At iteration k: linearize as:
        # approx sum = h_i^old * h_{i+d}^old + h_i^old*(h_{i+d} - h_{i+d}^old) + h_{i+d}^old*(h_i - h_i^old) = H(i,d)
        # Then: sum H(i,d) = constant + sum(h_i^old * h_{i+d} - h_{i+d}^old * h_i^old) = C + (sum h_i^old * h_{i+d}) - (sum h_{i+d}^old * h_i^old)
        
        # The simple approach is to create the linear constraints using soft-linearization
        # But for the full matrix of constraints: see our binary formulation helpers
        
        # Instead, force solve a clean second order linearization version by computing 
        # the function value to a vector row-wise and estimating using current h
        
        # More factual: we do partial linearization, decompose in terms of gradient:
        # Let's build an LP with approximations:
        h_updated = h.copy()
        h_prev = h.copy()
        
        # hard linearized constraint building
        # Shift d comes from worst shift index
        
        shift_d = d
        h_linearized = []
        
        for i in range(n_points):
            j = i + shift_d
            if 0 <= j < n_points:
                term1 = h_prev[i] * h_prev[j]
                term2 = h_prev[i] * (h_updated[j] - h_prev[j])
                term3 = h_prev[j] * (h_updated[i] - h_prev[i])
                linearized_term = term1 + term2 + term3
                h_linearized.append(linearized_term)
        
        linearized_value = sum(h_linearized)
        t = cp.Variable()
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Goal is to make sure 
        # t >= dx*(n/2 - value)
        # implies t >= dx*(n/2 - linearized_value) 
        constraints.append(t >= dx*(n_points/2 - linearized_value))
        
        # objective: minimize t
        # but we want a single relinearization where each d we have its own linear form
        # This is complex in that we cannot avoid outer loop (due to needing to identify worst d each iteration)
        
        # Instead we simplify to re-solve a problem with 1 d choice each solve
        # In each iteration:
        # 1. compute current correlations for all d
        # 2. pick worst d
        # 3. build that linear constraint
        # 4. minimize t with that constraint added (along with penalty for previous variables)
        # But we’re already on a form where previous h is known and delta is just positive or negative
        
        # So approach is:
        # linearized sum(i) h[i]*h[i+d] = sum(i) [h[i]^old * h[i+d]^old + h[i]^old*(h[i+d]-h[i+d]^old) + h[i+d]^old*(h[i]-h[i]^old)]
        # = constant + sum(h[i]^old*(h[i+d]-h[i+d]^old) + h[i+d]^old*(h[i]-h[i]^old))
        # 
        # So we use current estimate to define the linear inequalities:
        total = 0.0
        for i in range(n_points):
            j = i + shift_d
            if 0 <= j < n_points:
                total += h_prev[i] * h_prev[j] + \
                         h_prev[i] * (h_updated[j] - h_prev[j]) + \
                         h_prev[j] * (h_updated[i] - h_prev[i])
        # Approximate value
        approx_total = total

        # one-dimensional linear constraint in t as objective with linearized term:
        obj = cp.Minimize(t)
        constrs = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2,
            t >= dx * (n_points/2 - approx_total)
        ]
        prob = cp.Problem(obj, constrs)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_vars.value
                h_updated = project_sum_to_n2(h_new, n_points)
                h_updated = truncate_to_unit_interval(h_updated)  # don't go out of bounds
                h = h_updated.copy()
            else:
                # if we fail to get a solution, keep previous
                pass
        except:
            pass
        
        c5_new = evaluate_c5(h)
        if abs(c5_new - old_c5) < tolerance:
            break
        old_c5 = c5_new
            
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    
    # Phase 0: attempt initial feasibility using a fractional configuration
    h = np.ones(n_points) * 0.5
    h = project_sum_to_n2(h, n_points)
    
    # Goal: try to improve with a structured initial guess
    # Inspired by known constructions:
    # random + smooth or piecewise const
    # Try a smooth wavelet-like adjusted to sum constraint
    try:
        # Try to do something better than uniform
        # Try a piecewise continuous function like Max(0, 1 - |x - 1|) for bit of smoothness
        # but fit to our grid over [0,2]
        x = np.linspace(0, 2, n_points)
        # example: sinc(1.5x - 1) that peaks near x=1, but funky interpolate.
        # instead: try weighted average of shifted ones:
        h_smooth = np.zeros(n_points)
        for a in [0.1, 0.4, 0.7, 1.0, 1.3, 1.6, 1.9]:
            h_smooth += np.exp(-10*(x - a)**2)

        h_smooth = h_smooth / np.max(h_smooth)
        h = project_sum_to_n2(h_smooth, n_points).copy()
        h = truncate_to_unit_interval(h)
    except:
        pass
    
    # Phase 1: Basic convex-relaxed optimization  
    # Use some kind of constant offset to initialize better
    h = h.copy()

    # Phase 2: SLP loop ("convex-concave linearization") - iterative relinearization
    h_final = sequential_linear_programming_routine(h, n_points, dx, max_iter=2000)
    c5_result = evaluate_c5(h_final)

    # Final polishing using stochastic local search
    h_best = h_final.copy()
    c5_best = c5_result
    
    for _ in range(int(50000)):
        if time.time() - start_time > budget_s - 5:
            break
        # Small random perturbation
        i, j = np.random.randint(0, n_points, 2)
        if i == j:
            continue
    
        delta = np.random.random() * 0.02
        # ensure valid move
        if h_best[i] >= delta and h_best[j] + delta <= 1:
            h_test = h_best.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_sum_to_n2(h_test, n_points)
            h_test = truncate_to_unit_interval(h_test)
            
            c5_test = evaluate_c5(h_test)
            if c5_test < c5_best:
                h_best = h_test.copy()
                c5_best = c5_test
    
    return h_best.tolist(), c5_best, n_points
