import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try using different algorithms at different phases:
    # Phase 1: Global optimization with differential evolution
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=10,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
        
    # Phase 2: Simulated Annealing with local search
    try:
        # Try different local search strategies
        for i in range(3):
            # Use a slightly different neighborhood
            h_temp = best_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h_temp += noise
            h_temp = np.clip(h_temp, 0, 1)
            
            # Normalize
            h_temp = h_temp / (np.sum(h_temp) * dx) * 1.0
            h_temp = np.clip(h_temp, 0, 1)
            
            # Local optimization
            local_result = minimize(
                objective,
                h_temp,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    
    except:
        pass
    
    # Phase 3: Random restarts with different initializations
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Try adaptive initialization
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)  
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Phase 4: Try a few specific constructions that may work better
    # Try 2-plateau construction
    try:
        h_plateau = np.zeros(n_points)
        n1 = n_points // 3
        n2 = n_points - n1*2
        h_plateau[:n1] = 0.3
        h_plateau[n1:n1+n2] = 0.8
        h_plateau[n1+n2:] = 0.3
        h_plateau = h_plateau / (np.sum(h_plateau) * dx) * 1.0
        h_plateau = np.clip(h_plateau, 0, 1)
        
        local_result = minimize(
            objective,
            h_plateau,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Try another structure
    try:  
        h_peak = np.zeros(n_points)
        center = n_points // 2
        width = n_points // 4
        for i in range(max(0, center - width//2), min(n_points, center + width//2)):
            h_peak[i] = 0.8
        h_peak = h_peak / (np.sum(h_peak) * dx) * 1.0
        h_peak = np.clip(h_peak, 0, 1)
        
        local_result = minimize(
            objective,
            h_peak,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
        
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
