import numpy as np
import time
from scipy.fft import rfft, irfft

def proj_capped_simplex(v, target_sum):
    """Project v onto the capped simplex: sum = target, 0 <= v_i <= 1."""
    v = np.array(v)
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - target_sum) < 1e-12:
        return v
    if s < target_sum:
        # Add evenly
        diff = target_sum - s
        v += diff / len(v)
        return np.clip(v, 0, 1)
    # Need to subtract
    # Binary search for lambda
    l, u = 0.0, 1.0
    for _ in range(100):
        mid = (l + u) / 2
        proj = np.clip(v - mid, 0, 1)
        s_proj = np.sum(proj)
        if s_proj > target_sum:
            l = mid
        else:
            u = mid
    return np.clip(v - l, 0, 1)

def full_corr(h):
    """Compute full correlation of h with (1 - h) using FFT."""
    n = len(h)
    H = rfft(h, n=2*n-1)
    G = rfft(1-h, n=2*n-1)
    corr = irfft(H * np.conj(G), n=2*n-1)
    return corr.real

def evaluate_h(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = full_corr(h)
    max_val = np.max(corr)
    return max_val * dx

    S = 14
    block_len = n_points // S

    # Precompute shifts for GC computation
    def compute_grad(h_s, c):
        # c is full correlation vector
        # approximate smooth max with α=10
        alpha = 10.0
        c_shifted = c - np.max(c)
        exp_c = np.exp(alpha * c_shifted)
        w = exp_c / np.sum(exp_c)
        # Vectorized gradient: for each i, K_i = \sum_k w_k (1 - h_{i+k} - h_{i-k})  
        g = np.zeros_like(h_s)
        for k in range(len(c)):
            # extract if valid (hardcoded shifts for valid i)
            shift0 = np.roll(h_s, k) if k > 0 else np.zeros_like(h_s)
            shift1 = np.roll(h_s, -k) if k > 0 else np.zeros_like(h_s)
            g += w[k] * (1 - shift0 - shift1)
        return g

    def sa_chain(h_s0, delta_max=0.2, n_steps=5000):
        h_s = h_s0.copy()
        h = np.repeat(h_s, block_len)
        obj_prev = evaluate_h(h)
        best_h_s = h_s.copy()
        best_obj = obj_prev
        T = 0.1 * obj_prev  # initial temp
        alpha = 0.997  # cooling factor
        for i in range(n_steps):
            # Time check
            if time.time() - start_time > 0.3 * budget_s - 1:
                break
            # Break early if T too low
            if T < 1e-6:
                break

            # propose move
            i1 = np.random.randint(0, S)
            i2 = np.random.randint(0, S)
            if i1 == i2:
                continue
            delta = np.random.uniform(-delta_max, delta_max)
            h_s_new = h_s.copy()
            h_s_new[i1] += delta
            h_s_new[i2] -= delta
            h_s_new = proj_capped_simplex(h_s_new, target_sum=S/2.)
            h_new = np.repeat(h_s_new, block_len)
            obj_new = evaluate_h(h_new)

            # accept
            delta_obj = obj_new - obj_prev
            if delta_obj < 0 or np.random.rand() < np.exp(-delta_obj / T):
                h_s = h_s_new
                obj_prev = obj_new
                if obj_prev < best_obj:
                    best_obj = obj_prev
                    best_h_s = h_s.copy()

        return best_h_s, best_obj

            g = np.zeros_like(h_s)
            for k in range(len(c)):

                shift1 = np.roll(h_s, -k) if k > 0 else np.zeros_like(h_s)
                g += w[k] * (1 - shift0 - shift1)

            # Collapse to block level
            g_s = np.mean(g.reshape((S, block_len)), axis=1)

            # Update
            eta = 0.05 * (0.95 ** t)
            h_s = h_s - eta * g_s
            h_s = proj_capped_simplex(h_s, target_sum=S/2.)
