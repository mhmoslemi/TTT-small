import numpy as np
import time
import cvxpy as cp
from cvxpy import GLPK, GLPK_MI, ECOS, SCS

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):

    time_left = lambda: budget_s - (time.time() - start_time)

        c5_small = evaluate_erdos_solution(h_small)
        return (h_small.tolist(), c5_small, n_points)

    def solve_binary_milp(n, max_time=30):
        if time_left() < max_time + 20:
            return None, None

        t = cp.Variable()

        Y = {}
        A = []
        for d in range(-n+1, n):  # all possible shifts
            if d == 0:
                continue
            d_abs = abs(d)
            if d < 0:
                for i in range(d_abs, n):
                    Y[(i, d)] = cp.Variable(boolean=True)
                    A.append(Y[(i, d)] <= h_binary[i])
                    A.append(Y[(i, d)] <= h_binary[i + d_abs])
                    A.append(Y[(i, d)] >= h_binary[i] + h_binary[i + d_abs] - 1)
                    A.append(Y[(i, d)] >= 0)
            else:
                for i in range(n - d_abs):
                    Y[(i, d)] = cp.Variable(boolean=True)
                    A.append(Y[(i, d)] <= h_binary[i])
                    A.append(Y[(i, d)] <= h_binary[i + d_abs])
                    A.append(Y[(i, d)] >= h_binary[i] + h_binary[i + d_abs] - 1)
                    A.append(Y[(i, d)] >= 0)

        obj = cp.Minimize(t)
        
        # Ensure t >= dx * (n/2 - sum Y[i,d]) for all shifts
        for d in range(-n+1, n):
            if d != 0:
                if d < 0:
                    expr = sum(Y[(i, d)] for i in range(-d, n))  # d is negative
                else:
                    expr = sum(Y[(i, d)] for i in range(n - d))
                A.append(t >= dx * (n / 2 - expr))
        
        # Constraints
        A.append(cp.sum(h_binary) == n / 2)
        
        prob = cp.Problem(obj, A)
        try:
            # solve in given time
            prob.solve(solver=GLPK_MI, verbose=False, time_limit=max_time)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:
                return h_binary.value, t.value
        except:
            pass
        return None, None

            else:
                for i in range(n - d_abs):
                    Y[(i, d)] = cp.Variable()
                    A.append(Y[(i, d)] <= h[i])
                    A.append(Y[(i, d)] <= h[i + d_abs])
                    A.append(Y[(i, d)] >= h[i] + h[i + d_abs] - 1)
                    A.append(Y[(i, d)] >= 0)

                else:
                    expr = sum(Y[(i, d)] for i in range(n - d))
                A.append(t >= dx * (n / 2 - expr))
        
        A.append(cp.sum(h) == n / 2)
        A.append(h >= 0)
        A.append(h <= 1)
        
        prob = cp.Problem(obj, A)
        try:

            prob.solve(solver=GLPK, verbose=False, time_limit=30)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:

        except:

        return None, None

    best_h_lp = None
    best_c5_lp = float('inf')
    if time_left() > 40:
        h_lp, t_val = solve_continuous_lp(80)  # Try a higher resolution for better bisection
        if h_lp is not None and t_val < best_c5_lp:
            best_h_lp = h_lp
            best_c5_lp = t_val

    # Use LP solution as warm start if no binary was found
    if best_h_milp is None:
        if best_h_lp is not None:
            h_current = best_h_lp.copy()

            h_current *= (n_points / 2) / np.sum(h_current)

    # Phase 3. SCC refinement (convex relaxation of bilinears)
    def scc_refinement(h_init, n_max_iter=2000, min_improvement=1e-5):
        h = h_init.copy()
        t_prev = float('inf')
        
        for _ in range(n_max_iter):
            if time_left() < 20:

            for d in range(-n_points + 1, n_points):

                    expr_terms = []
                    for i in range(n_points - d_abs):

            # recompute Q using linearization of all shifts

            # do a first-order approximation manually here directly
            h_old = h.copy()

            # Demanding formulation, iterative update step size could be good
            
            # Approximate general Q(d) with H(d,h)
            # Let Q(d) = sum over i of h[i]*h[i+d], and h_old is fixed through recent loop
            # Linearize as h_old[i]*h[i+d] + h_old[i+d]*h[i] - h_old[i]*h_old[i+d]
