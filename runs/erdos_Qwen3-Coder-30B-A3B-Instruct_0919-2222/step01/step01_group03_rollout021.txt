import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector v onto [0,1] hypercube and sum constraint"""
    # First clip to [0,1]
    v = np.clip(v, 0, 1)
    
    # Adjust to satisfy sum constraint using water-filling
    current_sum = np.sum(v)
    if np.abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Simple projection using bisection
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(w)
        if s > target_sum:
            low = mid
        else:
            high = mid
    
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Initial optimization with differential evolution for global search
    best_c5 = float('inf')
    best_h = h0.copy()
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=10,
            seed=seed,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Then local optimization
    try:
        from scipy.optimize import minimize
        constraints = {'type': 'eq', 'fun': constraint_func}
        local_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Try random restarts with different initializations
    for restart in range(3):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate random initialization
        h_rand = np.random.rand(n_points) * 0.5  # Between 0 and 0.5
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            from scipy.optimize import minimize
            constraints = {'type': 'eq', 'fun': constraint_func}
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Try to improve with gradient-based approach using better initialization
    # Start from better random fields with centered peaks
    np.random.seed(seed + 100)
    best_h_fine = best_h.copy()
    best_c5_fine = best_c5
    
    # Tweak approach: create better initial pattern
    h_init = np.random.rand(n_points)  # Random
    # Normalize to make it sum to 1  
    h_init = h_init / np.sum(h_init) * (n_points / 2)
    h_init = np.clip(h_init, 0, 1)
    
    try:
        constraints = {'type': 'eq', 'fun': constraint_func}
        local_result = minimize(
            objective,
            h_init,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5_fine:
                best_c5_fine = c5_opt
                best_h_fine = h_opt.copy()
    except:
        pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h_fine)
    
    return (best_h_fine, final_c5, n_points)
