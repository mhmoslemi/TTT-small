def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def solve_lp_relaxation(n_points):
    """Solve a convex LP using McCormick envelopes to get a fractional h_star."""
    dx = 2.0 / n_points
    h = cp.Variable(n_points)
    w = n_points / 2.0
    u = cp.Variable((n_points, 2 * n_points - 1))
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == w]
    # Define u[i, d] ≈ h[i] * h[i + d] for valid d
    for i in range(n_points):
        for d in range(-n_points + 1, n_points):
            if 0 <= i + d < n_points:
                d_idx = d + n_points - 1  # Offset for indexing u
                constraints.extend([
                    u[i, d_idx] >= 0,
                    u[i, d_idx] <= h[i],
                    u[i, d_idx] <= h[i + d],
                    u[i, d_idx] >= h[i] + h[i + d] - 1
                ])
    # Compute s_d (sum for each shift d)
    s = [cp.sum([u[i, d + n_points - 1] for i in range(n_points) if 0 <= i + d < n_points]) for d in range(-n_points + 1, n_points)]
    # C5(d) = w - s_d, we want to minimize max C5(d)
    for k in range(len(s)):
        constraints.append(t >= w - s[k])
    problem = Problem(Minimize(t), constraints)
    try:
        problem.solve(solver='ECOS', verbose=False)
        if problem.status in ['optimal', 'optimal_inaccurate']:
            return np.array(h.value)
        else:
            print(f"LP solver status: {problem.status}")
    except Exception as e:
        print(f"Error solving LP: {e}")
    # Fallback
    return np.ones(n_points) * 0.5

def project_onto_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum."""
    # Shift to zero mean
    mean_b = np.mean(h)
    h_proj = h - mean_b
    # Re-center to desired sum
    h_proj += target_sum / len(h)
    # Clamp to [0, 1]
    h_proj = np.clip(h_proj, 0, 1)
    # Fix sum via rescaling
    actual_sum = np.sum(h_proj)
    if actual_sum > 0:
        h_proj = h_proj * target_sum / actual_sum
    return h_proj

def update_h_from_gradient(h, worst_shifts, dx):
    """Update by computing a subgradient."""
    # Squash to a long vector we can perform detailed handy computation on
    n_points = len(h)
    grad = np.zeros_like(h)
    for d in worst_shifts:
        # Add contributions from h_i * h_{i+d} to change in direction
        # For shift +d: d>0, so h_i contributes -h_{i+d}
        # For shift -d: d<0, so h_i contributes -h_{i+d}
        # This requires attention to indexing
        if d > 0:
            # Contribution for h_i -> - h_{i+d} (for i in range(n_points-d))
            for i in range(n_points - d):
                grad[i] -= h[i + d]
        elif d < 0:
            # Contribution for h_{i+d} -> - h_i (for i in range(-d, n_points))
            for i in range(-d, n_points):
                grad[i] -= h[i + d]
        else:  # d == 0: no change in gradient
            pass
    # Change direction is negative gradient
    step = 0.05 * dx
    h_updated = h - step * grad
    return project_onto_simplex(h_updated, n_points / 2)

def round_h_binary(h, n_points):
    """Round h to binary vector with correct sum."""
    # Clamp h to [0,1]
    h = np.clip(h, 0, 1)
    # Create a binary vector 
    sorted_idx = np.argsort(h)[::-1]
    target_sum = n_points / 2
    h_binary = np.full(n_points, False)
    for i in range(int(target_sum)):
        h_binary[sorted_idx[i]] = True
    return h_binary.astype(float)

def get_worst_shifts(h, n_points):
    """Compute all shift indices that correspond to maxima of correlation."""
    # Use full
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = max(corr)
    max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
    # Map full mode indices to actual shifts
    shifts = max_indices - (n_points - 1)
    return list(shifts)

def improve_binary_by_swaps(h, n_points):
    """Improve a binary vector h by swapping 1s with 0s to improve worst-case."""
    # Start with current h
    h = h.copy()
    # Determine position of 0s and 1s
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    if len(ones_idx) == 0 or len(zeros_idx) == 0:
        return h

    previous_c5 = evaluate_erdos_solution(h)
    max_iter = 10000
    for _ in range(max_iter):
        # Compute all shifts and worst shifts
        shifts = get_worst_shifts(h, n_points)
        if len(shifts) == 0:
            break
        d_star = shifts[0]
        # Find a pair to swap
        pairs = []  # (i, j) such that h[i]=1 and h[j]=0, and difference is right for shift d_star
        for i in ones_idx:
            j = i + d_star
            if 0 <= j < n_points and h[j] == 0:
                pairs.append((i, j))
        # Likewise for reverse: h[j]=1 and h[i]=0
        for j in zeros_idx:
            i = j - d_star
            if 0 <= i < n_points and h[i] == 1:
                pairs.append((i, j))
        if not pairs:
            break
        # Try to improve by moving a pair to reduce max correlation
        best_improvement = 0
        best_pair = None
        for i, j in pairs:
            # Temporarily move
            temp_h = h.copy()
            temp_h[i] = 0
            temp_h[j] = 1
            new_c5 = evaluate_erdos_solution(temp_h)
            improvement = previous_c5 - new_c5
            if improvement > best_improvement:
                best_improvement = improvement
                best_pair = (i, j)
        if best_pair is None:
            break
        # Apply the improvement
        i, j = best_pair
        h[i], h[j] = h[j], h[i]
        previous_c5 -= best_improvement
    return h

def simulate_annealing(h_start, n_points, max_time):
    """Apply simulated annealing to improve h if it's fractional."""
    import random
    from time import time as now
    h = h_start.copy()
    current_c5 = evaluate_erdos_solution(h)
    T = 0.1 * current_c5 if current_c5 != 0 else 0.01
    cooling_rate = 0.999
    iter_ct = 0
    max_iter = 100000
    start_time = now()

    while iter_ct < max_iter and now() - start_time < max_time:
        # Small random moves
        i, j = random.randint(0, n_points - 1), random.randint(0, n_points - 1)
        if i == j:
            continue
        delta = random.uniform(0.01, 0.05)
        delta = min(delta, h[i], 1 - h[j])
        if delta == 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_onto_simplex(h_new, n_points / 2)
        
        new_c5 = evaluate_erdos_solution(h_new)
        delta_e = new_c5 - current_c5
        
        if delta_e < 0 or random.random() < np.exp(-delta_e / T):
            h = h_new
            current_c5 = new_c5
        T *= cooling_rate
        iter_ct += 1

    return h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None

    # Grid points to consider
    candidates = [42, 48, 60]
    # Early time safety margin
    time_margin = 10
    for n in sorted(candidates):
        if time.time() - start_time > budget_s - time_margin:
            break
        dx = 2.0 / n
        # Step 1: LP relaxation to get fractional h
        h = solve_lp_relaxation(n)
        computed_c5 = evaluate_erdos_solution(h)
        if computed_c5 < best_c5:
            best_c5 = computed_c5
            best_h = h.copy()
            best_n = n

        # Step 2: Binary rounding + swarm hill climbing
        # Round to 0 or 1
        h_binary = round_h_binary(h, n)
        h_binary = improve_binary_by_swaps(h_binary, n)
        computed_c5 = evaluate_erdos_solution(h_binary)
        if computed_c5 < best_c5:
            best_c5 = computed_c5
            best_h = h_binary.copy()
            best_n = n

        # Step 3: Apply simulated annealing try to further improve
        remaining = budget_s - (time.time() - start_time)
        if remaining > 5:
            h_sa, c5_sa = simulate_annealing(h, n, remaining * 0.3)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n

        # Step 4: Projected gradient descent refinement
        if n > 10:  # Scaling check
            remaining = budget_s - (time.time() - start_time)
            if remaining > 10:
                # Compute current worst shifts
                shifts = get_worst_shifts(best_h, n)
                # Iteratively apply subgradient updates
                for _ in range(50):
                    if time.time() - start_time > budget_s - 3:
                        break
                    best_h = update_h_from_gradient(best_h, shifts, dx)
                    computed_c5 = evaluate_erdos_solution(best_h)
                    if computed_c5 < best_c5:
                        best_c5 = computed_c5
                    else:
                        # No improvement, break
                        break

    # Final normalization to ensure sum = n/2
    if best_h is not None and np.sum(best_h) != n/2:
        best_h = best_h * (n / (2 * np.sum(best_h)))

    # Re evaluate final
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, best_n)
