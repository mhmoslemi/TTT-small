import numpy as np
import time
import random
import cvxpy as cp


def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)


def shift_array(h, d):
    """Shift array h by d positions, with zero-padding"""
    n = len(h)
    if d > 0:
        return np.concatenate([h[d:], np.zeros(d)])
    elif d < 0:
        return np.concatenate([np.zeros(-d), h[:d]])
    else:
        return h.copy()


def project_onto_box_and_sum(v, target_sum, box_min=0.0, box_max=1.0):
    """Project vector v onto convex polytope defined by box constraints and sum constraint"""
    n = len(v)
    # Binary search for the correct lambda
    lb, ub = -100.0, 100.0
    for _ in range(40):
        lam = (lb + ub) / 2.0
        x = np.clip(v - lam, box_min, box_max)
        s = np.sum(x)
        if s > target_sum:
            lb = lam
        else:
            ub = lam
    lam = (lb + ub) / 2.0
    x = np.clip(v - lam, box_min, box_max)
    # Correct small numerical error in sum
    residual = target_sum - np.sum(x)
    # Distribute correction to non-bound entries
    idxs = np.where((x > box_min) & (x < box_max))[0]
    if len(idxs) > 0:
        correction = residual / len(idxs)
        for i in idxs:
            x[i] = np.clip(x[i] + correction, box_min, box_max)
    return x


def build_lp_model(n, binary=False):
    """Build the LP or MILP model for the Erdos problem"""
    h = cp.Variable(n, name="h")
    t = cp.Variable(1, name="t")
    y = cp.Variable((n, 2*n - 1), name="y")  # i, d
    constraints = []
    # Sum constraint
    constraints.append(cp.sum(h) == n / 2)
    # Box constraints
    if binary:
        constraints.append(h >= 0)
        constraints.append(h <= 1)
    else:
        constraints.append(cp.min(h) >= 0)
        constraints.append(cp.max(h) <= 1)
    # McCormick constraints for each y[i,d]
    for i in range(n):
        for d in range(-n + 1, n):
            if 0 <= i + d < n:
                # y[i,d] >= h[i] + h[i+d] - 1
                constraints.append(y[i, d + (n - 1)] >= h[i] + h[i + d] - 1)
                # y[i,d] <= h[i]
                constraints.append(y[i, d + (n - 1)] <= h[i])
                # y[i,d] <= h[i+d]
                constraints.append(y[i, d + (n - 1)] <= h[i + d])
                # y[i,d] >= 0
                if not binary:
                    constraints.append(y[i, d + (n - 1)] >= 0)
                if binary:
                    # For binary h, these are exact constraints
                    # But if h is real, we can still apply the inequalities
                    pass
            else:
                # Set to zero or make variables constant if within bounds
                pass
    # Objective constraints
    for d in range(-n + 1, n):
        start_idx = max(0, -d)
        end_idx = min(n, n - d)
        if start_idx < end_idx:
            s = cp.sum(y[start_idx:end_idx, d + (n - 1)])
            constraints.append((n / 2 - s) * 2.0 / n <= t)
    if binary:
        # enforce binary for h
        constraints.append(h == cp.round(h))
    objective = cp.Minimize(t)
    return h, y, t, constraints, objective


def solve_model(h, y, t, constraints, objective, solver="GLPK_MI", time_limit=30):
    """Solve the convex or mixed-integer program"""
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver, maxTime=time_limit, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return h.value, t.value
        else:
            # fallback to solving with CLP for LP
            if 'MI' in solver:
                try:
                    prob.solve(solver='CLP', maxTime=time_limit, verbose=False)
                except:
                    return None, None
            return h.value, t.value
    except:
        return None, None
    return None, None


def subgradient_refine(h_start, end_time):
    """Refine h_start using projected sub-gradient descent"""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    mu = 0.5  # initial step size
    k = 1
    counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    d_best = 0
    while time.time() < end_time:
        # Find worst shift
        corr = np.correlate(h, 1 - h, mode='full') * dx
        d_best = np.argmax(corr) - (n - 1)
        c5 = corr[d_best + (n - 1)]
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Compute sub-gradient of the worst shift
        g = 1 - shift_array(h, d_best) - shift_array(h, -d_best)
        # Check for stagnation
        if counter > 5000:
            h += np.random.normal(0, 0.01, n)
            h = project_onto_box_and_sum(h, n / 2.0)
            counter = 0
        # Update step size
        alpha = mu / np.sqrt(k)
        # Gradient step
        v = h - alpha * g
        # Project back
        h = project_onto_box_and_sum(v, n / 2.0)
        k += 1
        counter += 1
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(corr)
