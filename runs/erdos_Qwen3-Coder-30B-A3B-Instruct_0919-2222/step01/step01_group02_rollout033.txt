import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, k):
    """Shift array by k positions (positive means left shift)"""
    n = len(arr)
    if k == 0:
        return arr.copy()
    elif k > 0:
        # Left shift: move to the left, fill with zeros
        return np.concatenate([arr[k:], np.zeros(k)])
    else:
        # Right shift: move to the right, fill with zeros
        return np.concatenate([np.zeros(-k), arr[:k]])

def project_onto_box_and_sum(v, target_sum, box_min=0, box_max=1):
    """
    Projects vector v onto the constraints:
    * box: box_min <= v[i] <= box_max
    * sum: sum(v) = target_sum
    """
    n = len(v)
    # We want sum(x) = target_sum, with min <= x[i] <= max

    # Estimate the Lagrange multiplier lambda
    # The idea is to adjust v by -lambda to get the right sum
    def get_sum_with_lambda(lam):
        x = np.clip(v - lam, box_min, box_max)
        return np.sum(x)

    # Binary search for lambda such that sum(x(lambda)) = target_sum
    tol = 1e-12

    # Bracket lambda
    low_lam, high_lam = -100, 100
    while get_sum_with_lambda(low_lam) < target_sum:
        low_lam *= 2
    while get_sum_with_lambda(high_lam) > target_sum:
        high_lam *= 2

    # Bisection loop
    for i in range(40):
        mid_lam = (low_lam + high_lam) / 2
        s = get_sum_with_lambda(mid_lam)
        if s < target_sum:
            high_lam = mid_lam
        else:
            low_lam = mid_lam
    lambda_opt = (low_lam + high_lam) / 2

    # compute final x
    x = np.clip(v - lambda_opt, box_min, box_max)
    return x

def build_lp_model(n, binary=False):
    """
    Build a linear program to minimize max_k integral h(x)(1-h(x+k)) dx
    Takes binary=True for a mixed integer program, otherwise solves continuous relaxation.
    """
    # We'll use h[i] for i in [0, n-1], and fix constraints
    h = cp.Variable(n)
    t = cp.Variable(1)
    # auxiliary variables y[i,d] = h[i] * h[i+d]
    # We will generate all such pairs indexed by (i,d)
    y_dict = {}

    # Build all possible indices and shifts
    total_y = 0
    
    # Determine all non-zero products
    # The indices where h[i]*h[i+d] is supported are where both i, i+d are in [0,n-1]
    for d in range(-(n-1), n):
        for i in range(n):
            if 0 <= i + d < n:
                # We will define y[i,d] for this valid pair
                yij = cp.Variable(1)
                y_dict[(i, d)] = yij
                total_y += 1

    # McCormick constraints for each y[i,d]
    # Create constraints in a tight way
    constraints = []
    for d in range(-(n-1), n):
        for i in range(n):
            if 0 <= i + d < n:
                y = y_dict[(i, d)]
                h_i = h[i]
                h_ipd = h[i+d]
                # y <= h[i]
                constraints.append(y <= h_i)
                # y <= h[i+d]
                constraints.append(y <= h_ipd)
                # y >= h[i] + h[i+d] - 1
                constraints.append(y >= h_i + h_ipd - 1.)
                # y >= 0
                constraints.append(y >= 0)

    # Constraint on sum of h
    constraints.append(cp.sum(h) == n / 2)
    # Box constraints on h
    for i in range(n):
        constraints.append(h[i] >= 0)
        constraints.append(h[i] <= 1)

    # Max over all shifts: For each shift, compute integral term dx * (sum h - sum y)
    for d in range(-(n-1), n):
        # Relevant indices
        y_terms = []
        for i in range(n):
            if 0 <= i + d < n:
                y_terms.append(y_dict[(i, d)])
        sum_y = cp.sum(y_terms)
        # integral term = dx * (n/2 - sum_y)
        integral_term = (n / 2 - sum_y)
        # Add constraint that integral_term <= t
        constraints.append(integral_term <= t)

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    return prob, h, t, y_dict

def subgradient_refine(h_start, n_points, end_time):
    """
    Refine the solution with projected subgradient to improve C5.
    """
    h = h_start.copy()
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    # Initial gradient loop
    iter_count = 0
    last_improvement = 0
    learning_rate = 0.3
    min_rate = 1e-8
    while iter_count < 10000000 and time.time() < end_time:
        c5 = evaluate_erdos_solution(h)
        corr_vals = np.correlate(h, 1-h, mode='full')
        # Most negative
        max_k_idx = np.argmax(corr_vals)
        max_k = max_k_idx - (n_points - 1)
        d_star = max_k  # can be negative

        # Gradient for this shift
        g = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + d_star < n_points:
                g[i] = 1.0 - h[i + d_star]
            if 0 <= i - d_star < n_points and d_star != 0:
                g[i] -= h[i - d_star]

        # Update step
        rate = max(learning_rate / (1 + iter_count ** 0.5), min_rate)
        v = h - rate * g

        # Project onto feasible set
        h_new = project_onto_box_and_sum(v, target_sum=n_points/2)

        h = h_new
        iter_count += 1

        # Evaluate new C5
        c5_new = evaluate_erdos_solution(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            last_improvement = iter_count

        # Early stopping if no progress
        if iter_count - last_improvement > 1000:
            # Gaussian perturbation to escape local optima
            h = h + np.random.normal(0, 0.02, size=n_points)
            h = project_onto_box_and_sum(h, target_sum=n_points/2)
            last_improvement = iter_count

    return best_h, best_c5
