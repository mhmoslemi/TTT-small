import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def project_onto_simplex(x, target_sum):
    """Project x onto the simplex with sum = target_sum"""
    # Shift to center around zero sum
    mean_diff = (np.sum(x) - target_sum) / len(x)
    x = x - mean_diff
    # Clip to [0, 1]
    x = np.clip(x, 0, 1)
    # Adjust if necessary due to clipping
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        # Distribute adjustment on unclipped elements
        idxs = np.where((x > 0) & (x < 1))[0]
        if len(idxs) > 0:
            adjust = diff / len(idxs)
            for i in idxs:
                x[i] = x[i] + adjust
                if x[i] < 0:
                    x[i] = 0
                elif x[i] > 1:
                    x[i] = 1
    return x

def subgradient_worst_shift(h, worst_shifts):
    """Compute subgradient for the worst shifts"""
    n = len(h)
    grad = np.zeros(n)
    for shift in worst_shifts:
        # Positive shift
        if shift > 0:
            for i in range(n - shift):
                grad[i] -= h[i + shift]
        elif shift < 0:
            for i in range(-shift, n):
                grad[i] -= h[i + shift]
        else:  # shift == 0
            # No contribution from shift 0 in subgradient
            pass
    return grad

def projected_gradient_descent(h, target_sum, max_iter=500, lr=0.1):
    """Run projected gradient descent to improve solution h"""
    n = len(h)
    w = target_sum
    h = h.copy()
    lr = lr
    for i in range(max_iter):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Identify worst shifts
        max_val = max(corr)
        worst_shifts = [k - (len(corr) - 1) // 2 for k in range(len(corr)) if corr[k] == max_val]
        
        # Compute subgradient
        grad = subgradient_worst_shift(h, worst_shifts)
        # Update
        h = h - lr * grad
        # Project
        h = project_onto_simplex(h, w)
        lr *= 0.999  # Decay learning rate
        
        # Early stop if movement too small
        if i > 10 and np.allclose(h, np.roll(h, -1), atol=1e-6):
            break
    return h

def sa_move(h, i, j, delta):
    """Perform a mass transfer from i to j (delta must be small)"""
    # Ensure no invalid moves
    if h[i] < delta or h[j] > (1 - delta):
        return h
    h = h.copy()
    h[i] -= delta
    h[j] += delta
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    return h

def simulated_annealing(h, target_sum, max_iter=2000, initial_temp=0.02, cooling=0.995):
    """Simulated Annealing with mass transfer moves"""
    n = len(h)
    h = h.copy()
    w = target_sum
    c5 = evaluate_erdos_solution(h)
    current_temp = initial_temp
    best_h = h.copy()
    best_c5 = c5
    
    for i in range(max_iter):
        # Select a pair of indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # Determine delta (limited to avoid overshoot)
        delta_max = min(h[i], 1 - h[j], 0.1)
        if delta_max <= 0:
            continue
        delta = np.random.uniform(0.001, delta_max)
        new_h = sa_move(h, i, j, delta)
        new_c5 = evaluate_erdos_solution(new_h)
        
        # Accept or reject
        if new_c5 <= c5 or np.random.rand() < np.exp(-(new_c5 - c5) / current_temp):
            h = new_h
            c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        # Cool down
        current_temp *= cooling
    return best_h, best_c5

def binary_rounding(h, n):
    """Round fractional h to binary with sum = n/2"""
    # Sort indices by h value (descending)
    idxs = np.argsort(h)[::-1]
    round_h = np.zeros(n)
    count = 0
    for i in idxs:
        if count < n // 2:
            round_h[i] = 1
            count += 1
        else:
            round_h[i] = 0
    return round_h

def improve_binary(h, target_sum, max_iter=1000):
    """Improves binary h through worst-shift swaps"""
    n = len(h)
    w = target_sum
    h = h.copy()
    c5 = evaluate_erdos_solution(h)
    for _ in range(max_iter):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = max(corr)
        worst_shifts = [k - (len(corr) - 1) // 2 for k in range(len(corr)) if corr[k] == max_val]
        worst_shift = worst_shifts[0]  # take first one

        improved = False
        # Try swaps that reduce the worst shift
        for i in range(n):
            i_shift = (i + worst_shift) % n
            if h[i] == 1 and h[i_shift] == 0:
                # Try to find a complement
                for j in range(n):
                    j_shift = (j + worst_shift) % n
                    if h[j] == 0 and h[j_shift] == 1:
                        # Perform swap
                        h[i] = 0
                        h[j] = 1
                        new_c5 = evaluate_erdos_solution(h)
                        if new_c5 < c5:
                            improved = True
                            c5 = new_c5
                        else:
                            # Revert
                            h[i] = 1
                            h[j] = 0
                        break
            if improved:
                break
        if not improved:
            break
    return h, c5

def lp_initialize(n):
    """Initialize h using LP relaxation with McCormick envelopes"""
    h = cp.Variable(n)
    w = n / 2
    u = cp.Variable((n, 2*n - 1))  # Including all possible shifts
    t = cp.Variable()
    
    # Constraint: sum = n / 2
    constraints = [cp.sum(h) == w]
    # Box constraint
    constraints += [0 <= h[i] <= 1 for i in range(n)]
    # McCormick relaxations for u[i, d]
    for i in range(n):
        for d in range(1 - n, n):
            if 0 <= i + d < n:
                j = d + (n - 1)
                # u[i, d] >= max(0, h[i] + h[i+d] - 1)
                constraints += [u[i, j] >= h[i] + h[i+d] - 1]
                # u[i, d] >= 0
                constraints += [u[i, j] >= 0]
                # u[i, d] <= h[i]
                constraints += [u[i, j] <= h[i]]
                # u[i, d] <= h[i+d]
                constraints += [u[i, j] <= h[i+d]]

    # Objective: minimize t s.t. t >= w - sum(u_i,d) for all d
    s = []
    for d in range(1 - n, n):
        j = d + (n - 1)
        s_d = cp.sum(u[:, j])  # Overlapping sum of u_ij where both i and i+d valid
        constraints += [t >= w - s_d]
    objective = cp.Minimize(t)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return np.array(h.value)
    else:
        return np.full(n, 0.5)  # fallback to uniform

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 42
    
    # Test multiple n values
    n_list = [42, 48, 60]
    
    for n in n_list:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        # Initialize using LP
        h_raw = lp_initialize(n)
        # Renormalize
        h = h_raw * (n / (2 * np.sum(h_raw)))
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
            
        # Phase 1: Binary rounding
        h_bin = binary_rounding(h, n)
        h_bin, c5_bin = improve_binary(h_bin, n/2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n
            
        # Phase 2: Simulated Annealing
        if time.time() - start_time > budget_s * 0.95:
            break
        h_sa, c5_sa = simulated_annealing(h, n/2, max_iter=1000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n
    
        # Phase 3: Projected gradient descent
        if time.time() - start_time > budget_s * 0.95:
            break
        h_grad = projected_gradient_descent(h_sa, n/2, max_iter=200)
        c5_grad = evaluate_erdos_solution(h_grad)
        if c5_grad < best_c5:
            best_c5 = c5_grad
            best_h = h_grad.copy()
            best_n = n
    
    # Final normalization
    if best_h is not None:
        best_h = best_h * (best_n / (2 * np.sum(best_h)))
    
    return (best_h, best_c5, best_n)
