import numpy as np
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, shift):
    """Shift array by given amount with zero padding"""
    n = len(arr)
    if shift >= 0:
        out = np.concatenate([arr[shift:], np.zeros(shift)])
    else:
        s = -shift
        out = np.concatenate([np.zeros(s), arr[:n-s]])
    return out

def project_onto_box_simplex(v, lower=0., upper=1., target_sum=None):
    """Project vector v onto box constraints with target sum"""
    if target_sum is None:
        target_sum = len(v) / 2
    
    # Bisection for lambda to satisfy sum(clamp(v_i - λ, lower, upper)) = target_sum
    lo = -10.0
    hi = 10.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, lower, upper)

        # step size

        # update

            best_C = cur_C
            best_h = h.copy()
            last_improve_iter = iteration
            stagnation = 0
        else:
            stagnation += 1
            
        # random restart if stuck
        if stagnation > 5000:
            h = best_h + np.random.normal(scale=0.05, size=n)
            h = project_onto_box_simplex(h)
            stagnation = 0

    return best_h, best_C
