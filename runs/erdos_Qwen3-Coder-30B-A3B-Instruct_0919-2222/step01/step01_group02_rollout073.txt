import numpy as np
import time
import random
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, shift):
    """Shift array h by shift positions (negative = left shift)"""
    n = len(h)
    if shift == 0:
        return h.copy()
    elif shift > 0:
        return np.concatenate([np.zeros(shift), h[:-shift]])
    else:
        return np.concatenate([h[-shift:], np.zeros(-shift)])

def project_onto_box_and_sum(v, target_sum, box_min=0, box_max=1):
    """Project vector v onto hyperplane sum = target_sum with box constraints [box_min, box_max]"""
    n = len(v)
    v = np.asarray(v)
    lower = np.full(n, box_min)
    upper = np.full(n, box_max)
    
    # Compute a good initial guess for lambda
    def sum_clip(lam):
        return np.sum(np.clip(v - lam, lower, upper))
    
    # Binary search for lambda
    low, high = -100.0, 100.0
    epsilon = 1e-10
    for _ in range(40):
        mid = (low + high) / 2.0
        sum_val = sum_clip(mid)
        if sum_val > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    
    x = np.clip(v - lam, lower, upper)
    # Adjust for potential small discrepancies
    diff = target_sum - np.sum(x)
    # Distribute the remaining difference
    x += diff / n
    return np.clip(x, lower, upper)

def build_lp_model(n, binary=False):
    """Build LP model for C5 minimization"""
    # Create variables
    h = cp.Variable(n)
    y = cp.Variable((n, 2*n - 1))  # y[i,d] for d in range(-n+1, n)
    t = cp.Variable(1)
    
    # Create constraints
    constraints = [h >= 0, h <= 1]
    constraints.append(cp.sum(h) == n / 2)
    constraints.append(t >= 0)
    
    # McCormick constraints for y[i,d] = h[i] * h[i+d]
    for i in range(n):
        for d in range(1 - n, n):  # d = -(n-1) to (n-1)
            if 0 <= i + d < n:
                j = i + d
                constraints.append(y[i, d + n - 1] >= h[i] + h[j] - 1)
                constraints.append(y[i, d + n - 1] <= h[i])
                constraints.append(y[i, d + n - 1] <= h[j])
                if binary:
                    constraints.append(y[i, d + n - 1] >= 0)
                else:
                    constraints.append(y[i, d + n - 1] >= 0)
    
    # Shifted sums constraints
    for d in range(1 - n, n):
        if d >= 0:
            # g_i = h[i+d] for all valid i
            shift_expr = cp.sum(cp.multiply(h, shift_array(h, d)))
        else:
            # g_i = h[i+d] for valid i (d < 0, so i+d>=0)
            shift_expr = cp.sum(cp.multiply(h, shift_array(h, d)))
        if d >= 0:
            shift_expr = cp.sum(cp.multiply(h[:n-d], h[d:]))
        else:
            shift_expr = cp.sum(cp.multiply(h[-d:], h[:-d]))
        constraints.append(t >= n/2 - shift_expr)
    
    if binary:
        # All h are binary
        constraints += [h == cp.Variable(n, boolean=True)]
        # Adapt y to be binary also (but due to linearization the exact product is constraint)
        # Potentially redundant as McCormick handles this ensurin
        for i in range(n):
            for d in range(1 - n, n):
                if 0 <= i + d < n:
                    # Already enforced by McCormick - no additional constraints needed for binary case
                    pass
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    return prob, h, y, t

def solve_model(prob, solver='CLP', timeout=60):
    """Solve the model with time limit"""
    prob.solve(solver=solver, verbose=False, max_iters=10000, time_limit=timeout)
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return True, prob.value, prob.variables()[0].value
    return False, None, None

def subgradient_refine(h_start, end_time, max_iter=50000):
    """Refine with projected subgradient descent"""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)

    alpha0 = 0.5
    k = 0
    
    # Fixed decay rate for simplicity
    decay_rate = 0.999
    
    no_improvement_counter = 0
    stall_threshold = 5000
    
    while time.time() < end_time and k < max_iter:
        k += 1
        C5 = evaluate_erdos_solution(h)
        # Find the shift that gives the maximum overlap
        corr = np.correlate(h, 1 - h, mode='full') * dx
        d_star = np.argmax(corr) - (len(h) - 1)
        
        # Build gradient for shift d_star
        g = np.zeros(n)
        if d_star >= 0:
            g[:n-d_star] = 1.0 - h[d_star:]
        else:
            g[-d_star:] = 1.0 - h[:d_star]
        g = 1.0 - g - shift_array(h, -d_star)
        
        # Update with projected gradient step
        alpha = alpha0 / np.sqrt(k)  # Step size
        v = h - alpha * g
        h_new = project_onto_box_and_sum(v, target_sum)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(h_new)
        
        # Accept or reject
        if new_C5 < C5:
            h = h_new
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1
            
        # Perturbation on stall
        if no_improvement_counter > stall_threshold:
            no_improvement_counter = 0
            perturb = np.random.normal(0, 0.05, n)
            h += perturb
            h = project_onto_box_and_sum(h, target_sum)
            # Re-evaluate
            new_C5 = evaluate_erdos_solution(h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
    
    return best_h, best_C5
