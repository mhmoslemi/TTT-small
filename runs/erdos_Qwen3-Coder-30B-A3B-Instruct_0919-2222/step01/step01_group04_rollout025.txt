import numpy as np
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def project_onto_box_and_sum(v, target):
    """Project vector v onto box [0,1] and sum constraint"""
    # Find λ s.t. sum_i clip(v_i + λ, 0, 1) = target.
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def compute_gradient_for_lags(h, lags):
    """Compute gradient for given lags (average if multiple)"""
    N = len(h)
    grad = np.zeros(N)
    for lag in lags:
        if lag >= 0:
            # Overlap region i in [0, N - lag)
            i = np.arange(N - lag)
            grad[i] += 1 - h[i + lag]
            grad[i + lag] -= h[i]
        else:
            lag = -lag
            # Overlap region i in [lag, N)
            i = np.arange(lag, N)
            grad[i] += 1 - h[i - lag]
            grad[i - lag] -= h[i]
    return grad / len(lags) if len(lags) > 0 else grad

def generate_random_binary(N, weight):
    """Generate a random binary vector with given weight"""
    b = np.zeros(N, dtype=int)
    ones = np.random.choice(N, weight, replace=False)
    b[ones] = 1
    return b

def compute_min_overlap(b):
    """Compute the minimum overlap for a binary vector"""
    # correlation of b with itself, full mode
    corr = np.correlate(b, b, mode="full")
    # exclude zero lag
    min_overlap = np.min(corr[len(b):])  # only positive lags
    return min_overlap

def binary_maximin(N, weight=35, time_limit=20):
    """Find a good binary pattern via random sampling"""
    best_b = None
    best_min_overlap = -1
    
    start_time = time.time()
    iterations = 0
    
    while time.time() - start_time < time_limit:
        b = generate_random_binary(N, weight)
        min_overlap = compute_min_overlap(b)
        iterations += 1
        if min_overlap > best_min_overlap:
            best_min_overlap = min_overlap
            best_b = b.copy()
    
    return best_b, best_min_overlap

    m = np.zeros(n_points)

    lr_base = 0.05
    
    # Initial evaluation
    c5_current = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5_current
    
    # Gradient descent loop
    iteration = 0
    while time.time() - start_time < budget_s - 5:  # 5 sec margin
        # Compute current C5 and worst lags
        corr = np.correlate(h, 1 - h, mode="full")
        c5_vals = corr * dx
        max_c5 = np.max(c5_vals)
        
        # Find indices where C5 is near the maximum
        eps = 1e-6
        worst_lags = np.where(np.abs(c5_vals - max_c5) < eps)[0]
        if len(worst_lags) == 0:
            worst_lags = [np.argmax(c5_vals)]
        
        # Convert indices to shift values
        shifts = worst_lags - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient_for_lags(h, shifts)
        
        # ADAM update
        iteration += 1
        lr = lr_base / (1 + 0.00001 * iteration)  # Learn rate decay
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        
        h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project onto feasible set
        h = project_onto_box_and_sum(h, target_sum)
        
        # Evaluate
        c5_current = evaluate_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Occasionally add noise
        if iteration % 1000 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = np.clip(h + noise, 0, 1)
            h = project_onto_box_and_sum(h, target_sum)
    
    return (best_h, best_c5, n_points)
