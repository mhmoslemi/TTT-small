import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def compute_binary_min_overlap(b):
    """Compute minimum overlap for binary vector b"""
    n = len(b)
    corr = np.correlate(b, b, mode="full")
    # Extract positive lags (excluding zero lag)
    min_overlap = np.min(corr[n:])
    return min_overlap

def random_binary(n, weight):
    """Generate random binary vector with given weight"""
    b = np.zeros(n, dtype=int)
    indices = np.random.choice(n, int(weight), replace=False)
    b[indices] = 1
    return b

def binary_maximin(N, weight, time_limit):
    """Find good binary pattern using random sampling"""
    best_b = None
    best_min_overlap = -1
    start_time = time.time()
    
    # Try several random binary vectors
    max_samples = 10000
    for _ in range(max_samples):
        if time.time() - start_time > time_limit:
            break
        b = random_binary(N, weight)
        min_overlap = compute_binary_min_overlap(b)
        if min_overlap > best_min_overlap:
            best_min_overlap = min_overlap
            best_b = b.copy()
    
    return best_b

def smooth_binary(b, kernel):
    """Smooth binary vector with kernel"""
    h = np.convolve(b, kernel, mode='same')
    return h

def project_onto_box_and_sum(v, target):
    """Project vector onto box [0,1] and sum constraint"""
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def gradient_for_lag(h, lag):
    """Compute gradient for correlation at lag"""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # i from 0 to N - lag (both indices must exist)
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]
        grad[i + lag] -= h[i]
    else:
        lag = -lag
        # i from lag to N
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad
