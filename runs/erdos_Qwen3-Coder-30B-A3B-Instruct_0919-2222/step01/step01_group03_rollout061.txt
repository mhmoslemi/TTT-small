import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto bounds [lower, upper] and sum constraint"""
    # First clip to [lower, upper]
    v = np.clip(v, lower, upper)
    # Adjust to match target sum
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    # Simple linear adjustment
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return v
    # Distribute adjustment evenly
    v += diff / len(v)
    # Clip again after adjustment
    v = np.clip(v, lower, upper)
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Sample points for gradient descent approach
    n_sample = 100
    dx_sample = 2.0 / n_sample
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Initialize with the provided function
    initial_h = initial_h_values()
    c5_initial = evaluate_erdos_solution(initial_h)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = initial_h.copy()
    
    # Multi-start approach to find better solution
    num_starts = 10
    n_eval = 0
    
    # Gradient-based optimization
    max_iter = 10000
    eta0 = 0.1
    
    # Pre-compute shift indices for faster gradient calculation
    shift_ind = []
    for s in range(-(n_sample-1), n_sample):
        if s >= 0:
            i = np.arange(0, n_sample-s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_sample, dtype=int)
            j = i + s
        shift_ind.append((i, j))
    
    # BFGS-style gradient descent with projection
    h_best = best_h.copy()
    h_current = h_best.copy()
    c5_current = best_c5
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random restart occasionally
        if it > 0 and it % 500 == 0:
            h_current = np.random.rand(n_sample) * 0.5
            h_current = h_current / (np.sum(h_current) * dx_sample) * 1.0
            h_current = np.clip(h_current, 0, 1)
            
        # Compute correlation
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        # Softmax smooth approximation
        beta = 5.0 + it / 2000.0
        f = corr / dx_sample  # Raw correlation values
        max_f = np.max(f)
        exp_terms = np.exp(beta * (f - max_f))
        w = exp_terms / np.sum(exp_terms)
        
        # Compute gradient
        grad = np.zeros_like(h_current)
        for s, (i, j) in enumerate(shift_ind):
            if w[s] == 0:
                continue
            weight = w[s]
            grad[i] += weight * (1 - h_current[j])
            grad[j] += weight * (-h_current[i])
        
        # Update step size
        eta = eta0 / (1 + it / 500)
        h_new = h_current - eta * grad
        
        # Project to bounds and sum
        h_new = project_onto_bounds_and_sum(h_new, n_sample / 2)
        h_current = h_new
        
        # Evaluate and update best
        c5_test = evaluate_erdos_solution(h_current)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_current.copy()
    
    # As a final touch, try a few local refinements
    local_max_iter = 2000
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Run SLSQP from current best
        test_h = best_h.copy()
        try:
            from scipy.optimize import minimize
            res = minimize(
                objective,
                test_h,
                method='SLSQP',
                bounds=[(0, 1)] * len(test_h),
                constraints={'type': 'eq', 'fun': constraint_func},
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9}
            )
            if res.success:
                test_h = np.clip(res.x, 0, 1)
                c5_test = evaluate_erdos_solution(test_h)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = test_h.copy()
        except:
            pass

    return (best_h, best_c5, len(best_h))
