import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector v onto the feasible region with sum constraint and bounds [0,1]"""
    # First clip to [0,1]
    v = np.clip(v, 0, 1)
    # Adjust to satisfy sum constraint
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Use water-filling approach to adjust values
    # sort the values and make adjustments
    sorted_indices = np.argsort(v)[::-1]  # descending order
    remaining_sum = target_sum - current_sum
    
    v_copy = v.copy()
    for i in sorted_indices:
        if remaining_sum > 0:
            # Can increase v[i] to 1
            delta = min(1 - v_copy[i], remaining_sum)
            v_copy[i] += delta
            remaining_sum -= delta
        elif remaining_sum < 0:
            # Need to decrease v[i]
            delta = min(v_copy[i], -remaining_sum)
            v_copy[i] -= delta
            remaining_sum += delta
        if abs(remaining_sum) < 1e-10:
            break
    
    return v_copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Create index pairs for each shift to avoid recomputation
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points - s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))
    
    # Set up optimization
    max_iter = int(1e5)
    eta0 = 0.1
    beta = 1.0
    beta_max = 100.0
    beta_increment = (beta_max - beta) / (max_iter // 1000)
    
    # Initialize with best known
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Get starting triangular shape
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    h = np.clip(1 - np.abs(x-1), 0, 1)  # make triangular
    h = h * (n_points / 2.0) / np.sum(h)  # normalize to sum to n_points/2
    h = project_onto_bounds_and_sum(h, n_points / 2.0)
    
    h_current = h.copy()
    c5_current = evaluate_erdos_solution(h_current)
    if c5_current < best_c5:
        best_c5 = c5_current
        best_h = h_current.copy()

    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Compute the correlation values f_i for each shift
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        f = corr  # We'll use them directly as they're already scaled by dx

        # Compute weights (soft maximum)
        max_f = np.max(f)
        if max_f > 0:
            exp_terms = np.exp(beta * (f - max_f))
            w = exp_terms / np.sum(exp_terms)
        else:
            # If all f are equal or very small, use uniform weights
            w = np.ones_like(f) / len(f)

        # Compute gradient (scaled by weights)
        grad = np.zeros_like(h_current)
        
        # Compute constant term
        # The derivative of sum_i h_i term is 1
        grad += np.sum(w)

        # Compute product term contributions
        for shift_id, (i, j) in enumerate(shift_indices):
            if len(i) == 0:
                continue
            weight = w[shift_id]
            if weight == 0:
                continue
            # contrib_i = 1 - h[j]
            # contrib_j = - h[i]
            grad[i] += weight * (1 - h_current[j])
            grad[j] += weight * (-h_current[i])
            
        # Step with projected gradient descent
        eta = eta0 / (1 + it / 1000)  # Adaptive step size
        h_new = h_current - eta * grad
        
        # Project
        h_new = project_onto_bounds_and_sum(h_new, n_points/2.0)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept new solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_current = h_new.copy()
        else:
            h_current = h_new.copy()
            
        # Increase beta gradually for better approximation
        beta = min(beta_max, beta + beta_increment)
        
        # Periodically re-evaluate
        if it % 1000 == 0:
            current_c5 = evaluate_erdos_solution(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
