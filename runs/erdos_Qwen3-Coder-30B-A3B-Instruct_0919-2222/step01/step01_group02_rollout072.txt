import numpy as np
import time
import random
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, d):
    """Shift h by d positions (positive d = right shift)"""
    n = len(h)
    if d == 0:
        return h.copy()
    elif d > 0:
        return np.concatenate([np.zeros(d), h[:-d]])
    else:
        return np.concatenate([h[-d:], np.zeros(-d)])

def project_onto_box_and_sum(v, target_sum, box_min=0, box_max=1):
    """Project v onto the set {x | box_min ≤ x ≤ box_max, sum(x) = target_sum}."""
    n = len(v)
    # Find lambda such that sum(clip(v - lambda, box_min, box_max)) == target_sum
    # By bisection
    lower, upper = -1000.0, 1000.0  # Large bounds to ensure convergence
    for _ in range(40):
        mid = (lower + upper) / 2
        x = np.clip(v - mid, box_min, box_max)
        s = np.sum(x)
        if s < target_sum:
            upper = mid
        else:
            lower = mid
    lambda_opt = (lower + upper) / 2
    result = np.clip(v - lambda_opt, box_min, box_max)
    # Correct rounding errors in sum
    total = np.sum(result)
    diff = target_sum - total
    if abs(diff) > 1e-10:
        # Distribute error to smallest values to stay near the boundary
        idx = np.argsort(result)
        i = 0
        while abs(diff) > 1e-12 and i < n:
            if result[idx[i]] < box_max:
                inc = min(abs(diff), box_max - result[idx[i]])
                result[idx[i]] += inc
                diff -= inc
            i += 1
        # Or just add residual to smallest
        if abs(diff) > 1e-12:
            i = idx[0]
            result[i] = min(box_max, result[i] + diff)
    return result

def build_lp_model(n, binary=False):
    """
    Create a cvxpy model for the LP relaxation.
    If binary=True, make h binary variables.
    """
    # Create variables:
    h = cp.Variable(n, name='h')
    y = cp.Variable((n, 2*n - 1), name='y')  # for every shift d ∈ [-n+1, n-1]
    t = cp.Variable(1, name='t')
    
    # Sum constraint: sum(h) = n/2
    sum_con = cp.Equality(cp.sum(h), n / 2)
    
    # Box constraints
    bounds = [h >= 0, h <= 1]
    
    # McCormick constraints:
    # For all d from -n+1 to n-1, we have pairs: i and i+d, so i+d < n, i >= 0
    
    # For each i, d pair:
    # y[i, d] in [0,1], y[i,d] <= h[i], y[i,d] <= h[i+d], y[i,d] >= h[i] + h[i+d] - 1
    
    # Map d to index in y: d ∈ [-n+1, n-1], so index = d + n - 1
    constraints = [sum_con] + bounds
    
    # For each d: iterate through all valid i where i+d < n
    for d in range(-n + 1, n):
        # Map d to index in y matrix
        d_idx = d + n - 1
        # i ranges such that i+d < n (i < n-d) and 0 <= i  
        start_i = max(0, -d)
        end_i = min(n, n - d)
        for i in range(start_i, end_i):
            y_idx = (i, d_idx)
            # y[i,d] >= h[i] + h[i+d] - 1
            constraints.append(y[y_idx] >= h[i] + h[i + d] - 1)
            # y[i,d] <= h[i]
            constraints.append(y[y_idx] <= h[i])
            # y[i,d] <= h[i+d]
            constraints.append(y[y_idx] <= h[i + d])
            # y[i,d] >= 0
            constraints.append(y[y_idx] >= 0)
            # if binary, also fix y[i,d] = h[i]*h[i+d] = h[i]*h[i+d]
            # for the relaxation it's enough to have the above inequalities
    
    # Add the shift constraints
    for d in range(-n + 1, n):
        d_idx = d + n - 1
        # Sum over i of h[i] - y[i,d] <= t/dx
        # dx = 2/n
        # => (sum(h) - sum(y[:,d])) / (2/n) <= t
        #  => (n/2 - sum(y[:,d])) * n/2 <= t
        # => sum(y[:,d]) >= n/2 - 2*t/n
        # But this is equivalent to
        # sum(h) - sum(y[:,d]) <= t / dx
        #  n/2 - sum(y[:,d]) <= t*n/2  # ='s 2/n
        #  -sum(y[:,d]) <= t*n/2 - n/2
        #  sum(y[:,d]) >= n/2 - t*n/2 = n/2*(1-t)
        
        # The direct constraint:
        # sum(h) - sum(y[:, d]) <= t / (2/n) = t * n / 2
        expr = cp.sum(h) - cp.sum(y[:, d_idx])
        constraints.append(expr <= t * n / 2)
    
    # Objective
    obj = cp.Minimize(t)
    
    prob = cp.Problem(obj, constraints)
    return prob, h, y, t

def solve_with_time_limit(prob, solver='GLPK_MI', timelimit=30):
    # attempts to solve with time constraint built-in.
    # Most solvers allow it via settings.
    try:
        # GLPK supports time limit via "maxTime" if package provides
        # Check version availability
        prob.solve(solver=solver, maxTime=timelimit, verbose=False)
        status = prob.status
        if status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return True
        else:
            return False
    except Exception as e:
        print(f"Unexpected solver error: {e}")
        return False

            h = h_new.copy()
        
        if iter_count % check_every == 0:
            # Regular check that time isn't exhausted
            if time.time() >= end_time - 3:   # Heuristic
                break

    return best_h, best_c5
