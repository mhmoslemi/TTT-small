import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Start with the initial solution
    best_c5 = evaluate_erdos_solution(h0)
    best_h = h0.copy()
    
    # Try a hybrid approach: first use differential evolution on a coarse grid,
    # then fine tune with local optimization
    try:
        # Use a more focused approach with different strategies
        for attempt in range(3):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Optimizer 1: Differential Evolution with mutated bounds
            de_bounds = [(0, 1) for _ in range(n_points)]
            de_result = differential_evolution(
                objective,
                de_bounds,
                maxiter=50,
                popsize=10,
                seed=seed + attempt,
                atol=1e-6,
                rtol=1e-6,
                mutation=(0.5, 1.0),
                recombination=0.7
            )
            
            if de_result.success:
                h_opt = np.clip(de_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            
            # Optimizer 2: Local optimization on mutated starting points
            h_test = best_h.copy()
            # Add some noise
            noise_scale = 0.1
            h_test += np.random.normal(0, noise_scale, n_points)
            # Rescale to integral constraint
            h_test = h_test / (np.sum(h_test) * dx) * 1.0
            h_test = np.clip(h_test, 0, 1)
            
            local_result = minimize(
                objective,
                h_test,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        
    except Exception as e:
        # fallback to simple approach if something fails
        pass

    # Try different strategies if we're still within time
    if time.time() - start_time < budget_s * 0.8:
        # Heuristic: try some manual tuning based on visual inspection
        # Try putting mass on edges and center
        h_manual = np.zeros(n_points)
        # Add peak in center
        center_idx = n_points // 2
        h_manual[center_idx-2:center_idx+3] = 0.2
        # Add mass on edges
        h_manual[0] = 0.1
        h_manual[-1] = 0.1
        # Normalize
        h_manual = h_manual / (np.sum(h_manual) * dx) * 1.0
        h_manual = np.clip(h_manual, 0, 1)
        c5_manual = evaluate_erdos_solution(h_manual)
        if c5_manual < best_c5:
            best_c5 = c5_manual
            best_h = h_manual.copy()

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
