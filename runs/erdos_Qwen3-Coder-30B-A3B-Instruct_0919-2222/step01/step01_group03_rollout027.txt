import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Random restarts with different initializations
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate random initialization
        h_rand = np.random.rand(n_points) * 0.5  # Between 0 and 0.5
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            # Local optimization with different methods
            for method in ['SLSQP', 'L-BFGS-B', 'TNC']:
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                local_result = minimize(
                    objective,
                    h_rand,
                    method=method,
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10},
                    callback=lambda x: None
                )
                if local_result.success:
                    h_opt = np.clip(local_result.x, 0, 1)
                    c5_opt = objective(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
        except:
            pass
    
    # If we have a better starting point, try another optimization approach
    if best_c5 > 0.38:
        # Implement a more targeted approach: try to make a binary-like solution
        h_binary = np.zeros(n_points)
        # Distribute 1s and 0s in a way that might improve over the uniform case
        # Place two rectangular peaks in different parts of the domain
        half = n_points // 2
        h_binary[:half] = 0.5
        h_binary[half:] = 0.5
        # Adjust the spacing and heights of these regions
        step = max(1, n_points // 7)
        # Add some variation
        for i in range(0, n_points, step):
            if i < n_points:
                h_binary[i] = min(1.0, 0.5 + np.random.rand() * 0.3)
        
        h_binary = h_binary / (np.sum(h_binary) * dx) * 1.0
        h_binary = np.clip(h_binary, 0, 1)
        
        try:
            # Optimization
            local_result = minimize(
                objective,
                h_binary,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
