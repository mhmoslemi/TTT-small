import numpy as np

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def compute_product_sums(h):
    """Compute sum_i h_i * h_{i+s} for each shift s using convolution."""
    n = len(h)
    # Using np.correlate to get sum_i h[i] * h[i+s] for all shifts
    # Note: this includes zero padding (but doesn't matter since h is 0 outside [0,1])
    # Actually, let's compute accurate overlapped product for each shift
    prod_sums = np.correlate(h, h, mode='full')
    # For -n+1 to n-1 shifts, we want only the overlapping region relative to h
    # The full correlation from np.correlate is length 2n-1 = 139 for n=70
    # The middle index corresponds to shift 0, so we center index = n-1
    # We extract only regions of valid overlaps
    return prod_sums

def compute_correlation(h):
    """Compute correlation vector C(s) = sum h_i * (1-h_{i+s}) for each shift."""
    n = len(h)
    dx = 2.0 / n
    total_sum = n / 2.0
    prod_sums = np.correlate(h, h, mode='full')
    # For each shift s, the valid product indices are determined by overlap
    # Instead, we use the definition:
    # C(s) = sum h_i * (1 - h_{i+s}) over valid i where i+s is in [0,n-1]
    # This is same as total_sum - sum h_i h_{i+s}  (over valid indices)
    # Correlation convention: we use np.correlate exactly as before
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return corr

def projected_gradient_descent(h_init, max_iter=2000, step_size=0.001, beta=20):
    """Projected gradient descent to minimize max overlap integral.
    Use smooth approximation to make gradient computation manageable."""
    
    n = len(h_init)
    dx = 2.0 / n
    h = np.array(h_init)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # convergence tracking
    prev_c5 = float('inf')
    patience = 0
    patience_limit = 100
    
    for t in range(max_iter):
        # Compute current correlation values
        corr = compute_correlation(h)
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Stop if improving very slowly
        if abs(prev_c5 - current_c5) < 1e-8:
            patience += 1
            if patience > patience_limit:
                break
        else:
            patience = 0
        prev_c5 = current_c5
        
        # Compute weights for smooth approximation of max (log-sum-exp style)
        exp_c = np.exp(beta * corr)
        tot_exp = np.sum(exp_c)
        weights = exp_c / tot_exp
        
        # Compute gradient of log-sum-exp approximation of max
        # Gradient is weighted average of individual gradients
        # dC(s)/dh_i = 1 - h_{i+s} when both in range, 1 otherwise
        grad = np.zeros(n)
        for s_idx in range(2 * n - 1):
            s = s_idx - (n - 1)
            weight = weights[s_idx]
            
            # Determine valid indices for this shift
            if s >= 0:
                # shift forward
                start = s
                end = n
                # gradient update for valid indices
                if start < n:
                    # Add weight to each h_i for such i's when shifted i+s is valid
                    grad[:end-start] += weight * (1 - h[start:end])
                    # For the shifted part, subtract weight * h_{i+s}
                    grad += weight * (h[:end-start] - h[start:end]) 
                else:
                    # No overlapping at all
                    grad += weight
                    
            else:
                # shift backward
                start = 0
                end = n + s  # the real end boundary (since negative shift)
                if end > 0:
                    grad[start:end] += weight * (1 - h[:end])
                    # Recompute for shifted contributions
                    grad -= weight * h[start:end] 
                else:
                    # No valid overlaps
                    grad += weight
                    
        # One cleaner version for gradient updating:
        # Direct computation using convolution logic for each shift
        grad = np.zeros(n)
        for s in range(-(n-1), n):
            weight = weights[s + n - 1]  # shift index in full array
            
            # Add 1 for out-of-bound indices
            if s > 0:
                # shift forward to right
                # Where i in [0, n-s), and i+s in valid range [s, n-1]
                # All indices i get 1 contribution minus overlapping   (1-h[i+s])
                start, end = 0, n-s
                if end > start:
                    grad[start:end] += weight * (1 - h[end:])
                    grad[start:end] -= weight * h[start:end]  # correct double count
            elif s < 0:
                # shift backward to left
                # Where i in [1-s, n) only, i+s in [0, n-1]
                start, end = -s, n
                if end > start:
                    grad[start:end] -= weight * h[:end-start]
                    grad[start:end] += weight * (1 - h[:end-start])
            else:
                # s == 0: forward and backward identical
                grad += weight * (1 - h)  # total               
        
        # Multiplied with step size
        h_new = h - step_size * grad
        
        # Project to feasibility: clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint: ensure sum = n/2
        current_sum = np.sum(h_new)
        adjustment = (n / 2.0 - current_sum) / n
        h_new += adjustment
        h_new = np.clip(h_new, 0, 1)
        
        # Update
        h = h_new
    
    return best_h.tolist(), best_c5
