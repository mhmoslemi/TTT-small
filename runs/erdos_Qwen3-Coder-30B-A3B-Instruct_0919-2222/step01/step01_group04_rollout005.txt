import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def project_onto_box_and_sum(v, target):
    """Project vector v onto box [0,1] and sum constraint"""
    # Find λ s.t. sum_i clip(v_i + λ, 0, 1) = target.
    # Use binary search as described.
    lo = -np.max(v) - 2.0  # ensure sum lower bound zero
    hi = 1 - np.min(v) + 2.0 # ensure sum upper bound N
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def compute_gradient_for_lag(h, lag):
    """Compute gradient of correlation for a given lag"""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # Overlap region i in [0, N - lag)
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]  # derivative w.r.t h_i
        grad[i + lag] -= h[i]      # derivative w.r.t h_{i+lag}
    else:
        lag = -lag
        # Overlap region i in [lag, N)
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad

    best_binary = None
    start_time = time.time()
    elapsed = 0
    
    # Try up to 10000 random binary patterns
    for i in range(10000):
        if elapsed > 1.0:
            break
        b = np.zeros(n_points, dtype=int)
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        b[ones_indices] = 1
        
        # Compute correlations of binary with itself
        corr = np.correlate(b, b, mode="full")
        # Get all positive lags except zero (i = 0 is center)
        min_overlap = np.min(corr[n_points:])
        
        if min_overlap > best_min_overlap:
            best_min_overlap = min_overlap
            best_binary = b.copy()
        
        elapsed = time.time() - start_time
    
    # Convert binary pattern to initial continuous function by smoothing
    # with a triangular kernel
    kernel = np.array([0.25, 0.5, 0.25])
    smoothed = np.convolve(best_binary, kernel, mode='same')
    h0 = smoothed / np.sum(smoothed) * target_sum  # Normalize to correct sum
    
    # Clip and project to [0,1] and correct sum
    h0 = np.clip(h0, 0, 1)
    h0 = project_onto_box_and_sum(h0, target_sum)
    
    # Gradient descent stage
    # Initialize Adam parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Start with the smoothed version
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    lr = 0.05
    decay = 1e-6
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        # Evaluate current state
        corr = np.correlate(h, 1 - h, mode="full")
        c5_values = corr * dx
        
        # Find worst lag(s)
        max_c5 = np.max(c5_values)
        indices = np.where(np.abs(c5_values - max_c5) < 1e-8)[0]
        worst_lags = indices - (n_points - 1)  # Convert back to actual lag indices
        
        # Compute average gradient
        grad = np.zeros(n_points)
        for lag in worst_lags:
            grad += compute_gradient_for_lag(h, lag)
        grad /= len(worst_lags)
        
        # Update Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        
        # Apply update
        h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        lr = lr / (1 + decay * iteration)
        
        # Project back to constraints
        h = project_onto_box_and_sum(h, target_sum)

        if iteration % 500 == 0:

            h = h + noise
            h = project_onto_box_and_sum(h, target_sum)
        
        # Check if new best
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        iteration += 1
    
    return (best_h, best_c5, n_points)
