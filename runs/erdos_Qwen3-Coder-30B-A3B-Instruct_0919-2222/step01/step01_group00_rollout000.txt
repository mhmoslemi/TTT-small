import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points, n_segments):
    # Build full h array from segment heights
    seg_len = n_points // n_segments
    if seg_len * n_segments != n_points:
        # Adjust for remainder
        heights = np.pad(heights, (0, n_points - seg_len * n_segments), 'edge')
    h = np.repeat(heights, seg_len)
    return h

def evaluate_heights(heights, n_points, n_segments, penalty_factor=1e3):
    h = build_h(heights, n_points, n_segments)
    c5 = evaluate_h(h)
    # Add penalty for sum constraint
    sum_heights = np.sum(heights)
    penalty = penalty_factor * (sum_heights - n_segments/2)**2
    return c5 + penalty

def project_to_bounds_and_sum(heights, target_sum):
    # Simple iterative projection to ensure bounds [0,1] and sum=target
    heights = np.clip(heights, 0, 1)
    current_sum = np.sum(heights)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return heights
    
    # Simple algorithm: adjust elements to get target sum while staying in bounds
    # Equal adjustment
    if diff > 0:
        # Need to increase some elements (up to 1)
        # For all elements < 1, give diff/n
        new_heights = heights + diff/len(heights)
        # Ensure all not exceed 1
        new_heights = np.clip(new_heights, 0, 1)
    elif diff < 0:
        # Need to decrease some elements
        # Compute relative adjustments required
        diff_abs = -diff
        # Adjust proportionally
        weights = np.ones_like(heights)
        # Reduce proportional to current values (avoid extreme reduction)
        if np.sum(heights) > 0:
            reduce_factor = diff_abs / np.sum(heights)
            new_heights = heights * (1 - reduce_factor)
        else:
            new_heights = np.ones_like(heights) * 0.5
        
        # Final clipping to [0,1]
        new_heights = np.clip(new_heights, 0, 1)
    else:
        new_heights = heights
    
    # Ensure exact sum now
    current_sum = np.sum(new_heights)
    adjust = target_sum - current_sum
    if abs(adjust) < 1e-10:
        return new_heights
    
    # Now we have new_heights, adjust slightly to meet exact sum
    # Add uniform adjustment (clip to bounds)
    # Brute force (simpler approach): approximate rounding directly to exact sum
    # Just scale to target
    factor = target_sum / np.sum(new_heights)
    new_heights = new_heights * factor
    new_heights = np.clip(new_heights, 0, 1)
    
    # Final exact sum
    current_sum = np.sum(new_heights)
    adjust = target_sum - current_sum
    new_heights += adjust / len(new_heights)
    new_heights = np.clip(new_heights, 0, 1)
    
    return new_heights

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 140
    n_segments = 14  # 14 segments, each of 10 points
    
    penalty_factor = 10.0
    
    # Initial random feasible heights
    initial_heights = np.random.uniform(0, 1, n_segments)
    # Adjust to near target sum (n_segments / 2)
    initial_heights = project_to_bounds_and_sum(initial_heights, n_segments / 2)
    
    bounds = [(0, 1) for _ in range(n_segments)]
    
    best_c5 = float('inf')
    best_heights = None
    
    # Differential evolution with multiple restarts
    max_iter = 200
    popsize = 15  # Sensible for n_segments = 14
    
    for restart in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation of initial heights
        perturbed = np.random.rand(n_segments) * 0.2 + 0.4
        starting_heights = project_to_bounds_and_sum(perturbed, n_segments / 2)
        
        try:
            result = differential_evolution(
                lambda heights: evaluate_heights(heights, n_points, n_segments, penalty_factor),
                bounds,
                maxiter=max_iter,
                popsize=popsize,
                seed=seed + restart,
                disp=False,
                mutation=(0.5, 1.0),
                recombination=0.7,
                atol=1e-8
            )
            
            if result.success:
                final_heights = result.x
                # Final projection to ensure sum constraint and bounds
                final_heights = project_to_bounds_and_sum(final_heights, n_segments / 2)
                
                h_final = build_h(final_heights, n_points, n_segments)
                c5_final = evaluate_h(h_final)
                
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_heights = final_heights.copy()
        except Exception as e:
            pass

    if best_heights is None:
        best_heights = initial_heights

    # Final projection to ensure valid and sum constraint
    best_heights = project_to_bounds_and_sum(best_heights, n_segments / 2)
    final_h = build_h(best_heights, n_points, n_segments)
    final_c5 = evaluate_h(final_h)
    
    return (final_h, final_c5, n_points)
