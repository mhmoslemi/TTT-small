import numpy as np
import time
import cvxpy as cp
from cvxpy import GLPK_MI, GLPK, ECOS, SCS

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def bisection_on_feasibility(h, low, high, target_c5, n_points, dx):
    """Use bisection to tighten the bound on C5 by testing feasibility"""
    eps = 1e-4
    while high - low > eps:
        mid = (low + high) / 2
        # Feasibility check: solve LP with fixed t = mid
        try:
            # Build constraints
            h_vars = cp.Variable(n_points)
            t = mid  # Fixed upper bound
            # Objective: just feasibility
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            # For each shift d, add the linearized constraint C(d) <= t
            for d in range(-n_points + 1, n_points): # All shifts including negatives
                if d >= 0:
                    # h[i] h[i+d] linearized
                    expr = sum(h_vars[i] * h_vars[i+d] for i in range(n_points - d))
                else:
                    # d < 0, indices are reversed
                    expr = sum(h_vars[i-d] * h_vars[i] for i in range(-d, n_points))
                # Approximate bilinears using the previous solution h
                linear = 0
                if d >= 0:
                    lhs = [h_vars[i] * h[i+d] for i in range(n_points - d)]
                else:
                    lhs = [h_vars[i-d] * h[i] for i in range(-d, n_points)]
                # Build the linear approximation using previous h
                linear_expr = 0
                for i in range(n_points - abs(d)):
                    if d >= 0:
                        ui = h[i]
                        vi = h[i+d]
                    else:
                        ui = h[i-d]
                        vi = h[i]
                    linear_expr += ui * h_vars[i+d] + vi * h_vars[i] - ui * vi
                constraints.append(dx*(n_points/2 - linear_expr) <= t)
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=GLPK, verbose=False)
            if prob.status == 'optimal':
                # Feasible
                high = mid
            else:
                # Infeasible
                low = mid
        except:
            low = mid
    return high

def project_to_feasible_box_sum(x, target_sum):
    """
    Project vector x onto box constraints [0,1] and sum constraint
    """
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if np.isclose(current_sum, target_sum):
        return x
    # Simple bisection to preserve sum
    step = (target_sum - current_sum) / len(x)
    x += step
    x = np.clip(x, 0, 1)
    return x

def optimize_step_1_milp(h_initial, n_points):
    """Phase 1: Attempt MILP with binary variables"""
    dx = 2.0 / n_points
    h_vars = cp.Variable(n_points, integer=True)
    t = cp.Variable()
    # Objective: minimize t
    # Constraints:
    # h_i in [0, 1] - binary implies 0 or 1
    # sum h = n/2
    # t >= dx * (n/2 - sum h_i * h_{i+d}) for all d
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points/2]
    # Loop over all shifts
    for d in range(-n_points + 1, n_points):
        if d >= 0:
            # shift forward
            expr = cp.sum(h_vars[i] * h_vars[i+d] for i in range(n_points - d))
        else:
            # shift backward
            expr = cp.sum(h_vars[i-d] * h_vars[i] for i in range(-d, n_points))
        # Constraint: t >= dx*(n/2 - expr)
        constraints.append(t >= dx * (n_points/2.0 - expr))
    prob = cp.Problem(cp.Minimize(t), constraints)
    # Solve with a timeout
    timeout = 30  # seconds
    try:
        result = prob.solve(solver=GLPK_MI, verbose=False, time_limit=timeout)
        if prob.status == 'optimal':
            h_sol = h_vars.value.round()
            return h_sol.tolist(), result
        else:
            return h_initial.tolist(), float('inf')
    except:
        return h_initial.tolist(), float('inf')

def optimize_step_2_lp_relaxation(h_initial, n_points):
    """Phase 2: Solve relaxed continuous LP"""
    dx = 2.0 / n_points
    h_vars = cp.Variable(n_points, continuous=True)
    t = cp.Variable()
    # Objective: minimize t
    # Variable constraints
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points/2]
    # Shift constraints using linearization (same idea as before)
    # Loop over d
    for d in range(-n_points + 1, n_points):
        # Quadratic term: h[i] * h[i+d]
        if d >= 0:
            expr = cp.sum(h_vars[i] * h_vars[i+d] for i in range(n_points - d))
        else:
            expr = cp.sum(h_vars[i-d] * h_vars[i] for i in range(-d, n_points))
        # Thus: t >= dx * (n/2 - expr)
        constraints.append(t >= dx * (n_points/2.0 - expr))
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=GLPK, verbose=False, time_limit=10)
        h_sol = h_vars.value
        if prob.status == 'optimal':
            return h_sol.tolist(), t.value
        else:
            return h_initial.tolist(), float('inf')
    except:
        return h_initial.tolist(), float('inf')

def optimize_step_3_scc_refinement(h_initial, n_points, max_iter=1000):
    """Phase 3: Sequential Linear Programming (SCC) with trust region"""
    dx = 2.0 / n_points
    h_current = np.array(h_initial)
    h_old = h_current.copy()
    c5_prev = evaluate_erdos_solution(h_current)
    tol = 1e-5
    for iteration in range(max_iter):
        h_new = np.zeros_like(h_current)
        try:
            h_vars = cp.Variable(n_points)
            t = cp.Variable()
            # Current h and values used in approximations
            # Constraint for each shift d
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points/2]
            for d in range(-n_points + 1, n_points):
                if d >= 0:
                    expr = cp.sum(h_vars[i] * h_vars[i+d] for i in range(n_points - d))
                else:
                    expr = cp.sum(h_vars[i-d] * h_vars[i] for i in range(-d, n_points))
                # Approximate h[i]*h[i+d] around old values
                linear_expr = 0
                for i in range(n_points - abs(d)):
                    if d >= 0:
                        ui = h_old[i]
                        vi = h_old[i+d]
                        linear_expr += ui * h_vars[i+d] + vi * h_vars[i] - ui * vi
                    else:
                        ui = h_old[i-d]
                        vi = h_old[i]
                        linear_expr += ui * h_vars[i] + vi * h_vars[i-d] - ui * vi
                constraints.append(t >= dx * (n_points/2.0 - linear_expr))

            prob = cp.Problem(cp.Minimize(t), constraints)
            prob.solve(solver=GLPK, verbose=False, time_limit=20)
            if prob.status != 'optimal':
                # Break from optimality issues
                break
            h_new = h_vars.value

            # Project back to feasible box + sum
            h_new = project_to_feasible_box_sum(h_new, n_points/2)

            # Check improvement in objective
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new <= c5_prev - tol:
                h_current = h_new.copy()
                h_old = h_new.copy()
                c5_prev = c5_new
            else:
                # Possibly take smaller step
                h_current = 0.95 * h_current + 0.05 * h_new
                h_current = project_to_feasible_box_sum(h_current, n_points/2)
                c5_prev = evaluate_erdos_solution(h_current)
                if abs(c5_prev - c5_new) < tol:
                    break
        except:
            break

    return h_current.tolist(), c5_prev
