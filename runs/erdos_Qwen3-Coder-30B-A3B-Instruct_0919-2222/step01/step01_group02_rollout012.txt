import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, d):
    """Shift array by d positions"""
    if d == 0:
        return arr.copy()
    elif d > 0:
        return np.concatenate([np.zeros(d), arr[:-d]])
    else:
        return np.concatenate([arr[-d:], np.zeros(-d)])

def project_onto_box_and_sum(v, target_sum, lb=0, ub=1):
    """Project vector v onto box [lb,ub]^n and hyperplane sum=x"""
    n = len(v)
    # initial projection onto box
    x = np.clip(v, lb, ub)
    # Check current sum
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x.copy()

    # Bisection for Lagrange multiplier
    def sum_proj(lam):
        return np.sum(np.clip(v - lam, lb, ub))

    low, high = -10, 10
    for _ in range(40):
        mid = (low + high) / 2
        s = sum_proj(mid)
        if s > target_sum:
            low = mid
        else:
            high = mid

    lam = (low + high) / 2
    x = np.clip(v - lam, lb, ub)
    # Ensure exact sum with small correction
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        # Find first entry not at bound and add diff
        for i in range(n):
            if x[i] < ub:
                x[i] += diff
                break
        # In case that doesn't work, distribute diff evenly
        if abs(np.sum(x) - target_sum) > 1e-10:
            x += diff / n
    return np.clip(x, lb, ub)

def build_lp_model(n, binary=False):
    """Build the LP or MILP model for the given n"""
    # Variables
    h = cp.Variable(n, name='h')
    t = cp.Variable(name='t')
    if binary:
        h = cp.Variable(n, boolean=True, name='h')

    # Auxiliary variables y[i,d] = h[i] * h[i+d]
    y = {}
    y_vars = []
    for d in range(-n+1, n):
        # Iterate over valid indices
        for i in range(n):
            if 0 <= i+d < n:
                name = f'y_{i}_{d}'
                y[(i, d)] = cp.Variable(name=name)
                y_vars.append(y[(i, d)])

    # Constraints
    constraints = []
    # Box constraints
    constraints.extend([h[i] >= 0 for i in range(n)])
    constraints.extend([h[i] <= 1 for i in range(n)])

    # Sum constraint
    constraints.append(cp.sum(h) == n / 2)

    # McCormick constraints
    for d in range(-n+1, n):
        for i in range(n):
            if 0 <= i+d < n:
                y_var = y[(i, d)]
                constraints.extend([
                    y_var >= 0,
                    y_var <= h[i],
                    y_var <= h[i+d],
                    y_var >= h[i] + h[i+d] - 1
                ])

    # Shift constraints
    for d in range(-n+1, n):
        if d == 0:
            obj = cp.sum(h) - cp.sum(y[(i, d) for i in range(n)])
        else:
            obj = cp.sum(h) - cp.sum([y[(i, d)] for i in range(n) if 0 <= i+d < n])
        constraints.append(obj * (2.0 / n) <= t)

    # Objective
    objective = cp.Minimize(t)

    # Formulate the problem
    if binary:
        prob = cp.Problem(objective, constraints)
    else:
        prob = cp.Problem(objective, constraints)

    return h, t, y, prob

def solve_model(prob, end_time, solver='GLPK_MI'):
    """Solve the model with timeout"""
    try:
        prob.solve(solver=solver, max_time=10)
        return True, prob.value
    except Exception:
        return False, float('inf')

    best_c5 = evaluate_erdos_solution(h)
    t_start = time.time()
    it = 0
    no_improvement = 0
    max_no_improvement = 5000

    while time.time() < end_time and no_improvement < max_no_improvement:
        it += 1
        # Find worst shift
        corr = np.correlate(h, 1-h, mode='full')
        max_idx = np.argmax(corr)
        # Shift between -n+1 and n-1
        d = max_idx - (2*n - 2) // 2

        # Compute gradient
        g = np.zeros(n)
        for i in range(n):
            g[i] = 1.0 - (h[min(i+d, n-1)] if i+d < n else 0) - (h[max(i-d, 0)] if i-d >= 0 else 0)

        # Step size
        alpha = 0.5 / np.sqrt(it)

        # Update with projected subgradient descent
        v = h - alpha * g
        h_new = project_onto_box_and_sum(v, n/2)

        # Evaluate
        c5 = evaluate_erdos_solution(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            no_improvement = 0
        else:
            no_improvement += 1
        h = h_new.copy()

    return best_h, best_c5
