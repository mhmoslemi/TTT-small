import numpy as np
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def project_onto_box_and_sum(v, target):
    """Project vector v onto box [0,1] and sum constraint"""
    # Find λ s.t. sum_i clip(v_i + λ, 0, 1) = target.
    lo = -np.max(v) - 2.0  # ensure sum lower bound zero
    hi = 1 - np.min(v) + 2.0 # ensure sum upper bound N
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def compute_gradient_for_lag(h, lag):
    """Compute gradient of correlation at shift lag"""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # Overlap region i in [0, N - lag)
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]  # derivative w.r.t h_i
        grad[i + lag] -= h[i]      # derivative w.r.t h_{i+lag}
    else:
        lag = -lag
        # Overlap region i in [lag, N)
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad

def random_binary(N, weight):
    """Generate random binary vector with given weight"""
    b = np.zeros(N, dtype=int)
    ones_positions = np.random.choice(N, size=int(weight), replace=False)
    b[ones_positions] = 1
    return b

def compute_min_overlap(b):
    """Compute minimum overlap over all shifts for binary sequence"""
    corr = np.correlate(b, b, mode="full")
    # Get positive lags that are not zero
    min_overlap = np.min(corr[len(corr)//2 + 1:])  # Avoiding the middle (zero lag)
    return min_overlap

def binary_maximin(N, weight, time_limit=5.0):
    """Generate good binary pattern by maximizing minimum overlap"""
    start_time = time.time()
    best_binary = None
    max_min_overlap = -1
    
    # Try many random binary patterns
    for _ in range(5000):
        if time.time() - start_time > time_limit:
            break
        b = random_binary(N, weight)
        min_overlap = compute_min_overlap(b)
        if min_overlap > max_min_overlap:
            max_min_overlap = min_overlap
            best_binary = b.copy()
    
    # Simple hill climbing
    if best_binary is not None:
        current_binary = best_binary.copy()
        current_min_overlap = max_min_overlap
        while time.time() - start_time < time_limit * 0.8:
            # Try a hill climb step
            ones_indices = np.where(current_binary)[0]
            zeros_indices = np.where(~current_binary)[0]
            
            if len(ones_indices) == 0 or len(zeros_indices) == 0:
                break
                
            # Choose one 1 and one 0 to swap
            i = np.random.choice(ones_indices)
            j = np.random.choice(zeros_indices)
            
            # Perform swap
            temp_binary = current_binary.copy()
            temp_binary[i], temp_binary[j] = temp_binary[j], temp_binary[i]
            
            new_min_overlap = compute_min_overlap(temp_binary)
            if new_min_overlap > current_min_overlap:
                current_binary = temp_binary
                current_min_overlap = new_min_overlap
        
        best_binary = current_binary
    
    return best_binary if best_binary is not None else random_binary(N, weight)

def smooth_binary(b, kernel=[0.25, 0.5, 0.25]):
    """Smooth a binary vector with given kernel"""
    # Convolve with kernel if kernel isn't scalar
    if isinstance(kernel, (list, np.ndarray)):
        h0 = np.convolve(b, kernel, mode='same')
        return h0
    else:
        # Use a 3-point kernel
        h0 = np.convolve(b, [0.25, 0.5, 0.25], mode='same')
        return h0
