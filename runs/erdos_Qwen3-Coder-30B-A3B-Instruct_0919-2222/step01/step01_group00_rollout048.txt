import numpy as np
import time

def proj_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: sum(v) = target_sum, 0 <= v <= 1."""
    v = np.array(v, dtype=float)
    target_sum = float(target_sum)
    
    if np.sum(v) == target_sum:
        return v
    if np.sum(v) > target_sum:
        # Need to reduce
        lmbd = np.sort(v)[::-1]
        cumsum = np.cumsum(lmbd)
        # Find lambda such that sum(min(v_i, lambda)) = target_sum
        # This is equivalent to finding the largest k such that
        # (sum of first k elements is > target_sum) and sum of first k+1 <= target_sum
        # (k+1) * lmbd[k] - (cumsum[k] - v[k]) <= target_sum
        # This is a bit tricky, so we use the binary search approach
        # Inner loop for exact computation when needed
        # if sum(v) is so small it's some tiny fraction, just scale
        
        # Simpler and more robust:
        # Let b = v - lmbd[0] for first element check 
        # This will be inaccurate but works
        # We compute as if all elements equal to lambda until some condition
        
        # https://stackoverflow.com/questions/11532586/orthogonal-projection-onto-the-unit-simplex
        # Waterfilling approach:
        v = np.sort(v)[::-1]
        s = 0
        for i in range(len(v)):
            s += v[i]
            if (s - (i + 1) * v[i]) >= target_sum:
                break
        else:
            # So we're setting everything to target_sum / len(v)
            lambda_star = target_sum / len(v)
            result = np.clip(v + lambda_star, 0, 1)
            return result * (target_sum / np.sum(result))
        lambda_star = (s - target_sum) / (i + 1)
        result = np.clip(v - lambda_star, 0, 1)
        return result

def full_corr(h):
    """Fast full convolution using FFT (assumes h of even length to make corr match full)."""
    n = len(h)
    h_padded = np.concatenate([h, np.zeros(n-1)])
    h_conj = np.conj(np.fft.rfft(h_padded))
    g_conj = np.conj(np.fft.rfft(np.concatenate([1-h, np.zeros(n-1)])))
    corr = np.fft.irfft(h_conj * g_conj, n=2*n-1)
    return corr[:2*n-1]

def evaluate_h(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    conv = full_corr(h)
    max_val = np.max(conv)
    return max_val * dx
