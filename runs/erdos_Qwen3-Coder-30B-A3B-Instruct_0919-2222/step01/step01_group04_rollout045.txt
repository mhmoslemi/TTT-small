import numpy as np
import time
from scipy import correlate

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def compute_correlation(h):
    """Compute all correlation values for all shifts"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return corr

def project_onto_box_and_sum(v, target_sum):
    """Project vector onto box [0,1] and sum constraint"""
    # Binary search for appropriate shift
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def smooth_binary(b):
    """Smooth binary vector with triangular kernel"""
    kernel = np.array([0.25, 0.5, 0.25])
    smoothed = np.convolve(b, kernel, mode='same')
    return smoothed

def random_binary(N, weight):
    """Generate a random binary vector with given weight"""
    b = np.zeros(N, dtype=int)
    indices = np.random.choice(N, size=int(weight), replace=False)
    b[indices] = 1
    return b

def compute_min_overlap(b):
    """Compute minimum overlap of binary vector with shifted version"""
    n = len(b)
    corr = np.correlate(b, b, mode="full")
    # Exclude zero lag
    min_overlap = np.min(corr[n:])
    return min_overlap

def binary_hill_climb(N, weight, time_limit):
    """Improve binary pattern using hill climbing"""
    start_time = time.time()
    
    # Generate a random binary vector as starting point
    current_b = random_binary(N, weight)
    current_min_overlap = compute_min_overlap(current_b)
    
    # Generate many candidates and pick the best
    num_candidates = 2000
    best_b = current_b.copy()
    best_min_overlap = current_min_overlap
    
    for _ in range(num_candidates):
        if time.time() - start_time > time_limit:
            break
        candidate_b = random_binary(N, weight)
        candidate_min_overlap = compute_min_overlap(candidate_b)
        if candidate_min_overlap > best_min_overlap:
            best_b = candidate_b.copy()
            best_min_overlap = candidate_min_overlap
    
    # Now do local search (hill climbing) with swaps
    while time.time() - start_time < time_limit:
        # Try a random swap: flip a 1 to 0 and a 0 to 1
        ones = np.where(best_b == 1)[0]
        zeros = np.where(best_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Try swap
        trial_b = best_b.copy()
        trial_b[i] = 0
        trial_b[j] = 1
        
        trial_min_overlap = compute_min_overlap(trial_b)
        if trial_min_overlap > best_min_overlap:
            best_b = trial_b
            best_min_overlap = trial_min_overlap
    
    return best_b

def compute_gradient_for_lag(h, lag):
    """Compute gradient of C5 for specific lag using the correlation indices"""
    N = len(h)
    grad = np.zeros(N)
    
    if lag >= 0:
        # Overlap region i in [0, N - lag)
        i_indices = np.arange(0, N - lag)
        grad[i_indices] += 1 - h[i_indices + lag]  # derivative w.r.t h_i
        grad[i_indices + lag] -= h[i_indices]      # derivative w.r.t h_{i+lag}
    else:
        # Lag is negative: lag = -k for positive k
        k = -lag
        # Overlap region i in [k, N)
        i_indices = np.arange(k, N)
        grad[i_indices] += 1 - h[i_indices - k]
        grad[i_indices - k] -= h[i_indices]
    
    return grad

def compute_subgradient(h, corr_values, dx):
    """Compute subgradient for the maximum correlation"""
    # Get all lag indices that achieve maximum correlation
    max_corr_val = np.max(corr_values)
    # Find all indices (lags) with values close to max
    tol = 1e-8
    best_lags = np.where(np.abs(corr_values - max_corr_val) < tol)[0]
    
    if len(best_lags) == 0:
        # Fallback if nothing is close to max
        best_lags = [np.argmax(corr_values)]
    
    # Convert 0-based index to actual lag
    # For np.correlate with mode='full', index = lag + n - 1
    actual_lags = best_lags - (len(corr_values) - 1)  # Convert indices back to lags
    # Compute average gradient from these lags
    subgradient = np.zeros(len(h))
    for lag in actual_lags:
        grad = compute_gradient_for_lag(h, lag)
        subgradient += grad
    
    # Normalize by number of lags contributing gradients
    if len(actual_lags) > 0:
        subgradient /= len(actual_lags)
    
    return subgradient
