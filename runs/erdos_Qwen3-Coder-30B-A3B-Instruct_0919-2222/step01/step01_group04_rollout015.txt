import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def compute_min_overlap(h):
    """Compute minimum overlap for binary h"""
    n = len(h)
    corr = np.correlate(h, h, mode='full')
    # Skip zero lag
    min_overlap = np.min(corr[n:])
    return min_overlap

def project_onto_box_and_sum(v, target):
    """Project vector onto box [0,1] and sum constraint"""
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def binary_hill_climbing(n_points, target_sum, max_time=20):
    """Generate good binary pattern using hill climbing"""
    start_time = time.time()
    best_h = None
    best_min_overlap = -1
    
    # Generate initial pattern with random weight=35
    h = np.zeros(n_points, dtype=int)
    ones_indices = np.random.choice(n_points, int(target_sum), replace=False)
    h[ones_indices] = 1
    
    current_min_overlap = compute_min_overlap(h)
    best_h = h.copy()
    best_min_overlap = current_min_overlap
    
    # Hill climbing for a while
    iterations = 0
    while time.time() - start_time < max_time and iterations < 5000:
        # Try random swaps
        i = np.random.choice(n_points)
        if h[i] == 1:
            # Find a zero to swap with
            zeros = np.where(h == 0)[0]
            if len(zeros) == 0:
                continue
            j = np.random.choice(zeros)
            # Perform swap
            h[i] = 0
            h[j] = 1
            new_min_overlap = compute_min_overlap(h)
            if new_min_overlap > current_min_overlap:
                current_min_overlap = new_min_overlap
                best_h = h.copy()
                best_min_overlap = current_min_overlap
            else:
                # Revert
                h[j] = 0
                h[i] = 1
        iterations += 1
    
    return best_h

def smooth_binary_pattern(b, kernel=[0.25, 0.5, 0.25]):
    """Smooth binary pattern with kernel"""
    n = len(b)
    # Use convolution with specified kernel
    smoothed = np.convolve(b, kernel, mode='same')
    # Clip and reproject to sum = target
    target_sum = n / 2
    smoothed = np.clip(smoothed, 0, 1)
    smoothed = project_onto_box_and_sum(smoothed, target_sum)
    return smoothed
