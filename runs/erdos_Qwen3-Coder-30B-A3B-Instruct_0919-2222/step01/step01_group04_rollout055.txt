import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def compute_binary_min_overlap(b):
    """Compute minimum overlap of binary vector with its shifts"""
    n = len(b)
    corr = np.correlate(b, b, mode="full")
    # exclude zero lag
    min_overlap = np.min(corr[n:])
    return min_overlap

def random_binary(n, weight):
    """Generate a random binary vector with given weight"""
    b = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, size=int(weight), replace=False)
    b[ones_indices] = 1
    return b

def binary_hill_climbing(n, weight, time_limit):
    """Use hill climbing to find good binary pattern"""
    start_time = time.time()
    
    # Start with a random binary vector
    best_b = random_binary(n, weight)
    best_min_overlap = compute_binary_min_overlap(best_b)
    
    # Try swapping 1 and 0 entries to improve minimum overlap
    attempts = 0
    max_attempts = 10000
    
    while time.time() - start_time < time_limit and attempts < max_attempts:
        # Pick a random 1 and a random 0 to swap
        ones = np.where(best_b == 1)[0]
        zeros = np.where(best_b == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Create new candidate
        new_b = best_b.copy()
        new_b[i] = 0
        new_b[j] = 1
        
        # Compute new minimum overlap
        new_min_overlap = compute_binary_min_overlap(new_b)
        
        if new_min_overlap > best_min_overlap:
            best_b = new_b
            best_min_overlap = new_min_overlap
            
        attempts += 1
        
    return best_b, best_min_overlap

def smooth_binary(b, kernel=[0.25, 0.5, 0.25]):
    """Smooth binary vector with given kernel"""
    n = len(b)
    # Use convolution with specified kernel
    h = np.convolve(b, kernel, mode='same')
    # Handle edge effects by extending the kernel to be symmetric
    return h

def project_onto_box_and_sum(v, target):
    """Project vector onto [0,1] box constrained to sum to target"""
    # Binary search for lambda such that 
    # sum_i clip(v_i + lambda, 0, 1) = target
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def grad_for_lag(h, lag):
    """Compute gradient for correlation at lag"""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # Overlap region i in [0, N - lag)
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]
        grad[i + lag] -= h[i]
    else:
        lag = -lag
        # Overlap region i in [lag, N)
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad

    dx = 2.0 / n_points

    # Try random binary patterns to find a good one
    best_b = random_binary(n_points, target_sum)
    best_min_overlap = compute_binary_min_overlap(best_b)
    
    num_samples = 10000  # Quite a few samples for search
    for _ in range(num_samples):
        if time.time() - start_time > binary_time:

        attempt_b = random_binary(n_points, target_sum)
        attempt_min_overlap = compute_binary_min_overlap(attempt_b)
        if attempt_min_overlap > best_min_overlap:
            best_b = attempt_b
            best_min_overlap = attempt_min_overlap
            
    # Now do some hill-climbing
    if binary_time > 0 and time.time() - start_time < binary_time:
        hill_b, hill_min = binary_hill_climbing(
            n_points, target_sum, binary_time - (time.time() - start_time)
        )
        if hill_min > best_min_overlap:
            best_b = hill_b
            best_min_overlap = hill_min

    # Smoothing to get continuous h
    h = smooth_binary(best_b, [0.25, 0.5, 0.25])
    h = np.clip(h, 0, 1)
    h = project_onto_box_and_sum(h, target_sum)
   
    # Use Adam optimizer for refinement
    # Parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.05
    decay_rate = 0.00001
    
    # Initialize Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    current_h = h.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_c5(current_h)
    last_improvement = 0
    iteration = 0

    # Main optimization loop
    while time.time() - start_time < budget_s - 5:  # ~5s margin for cleanup
        # Compute all correlation values:
        corr = np.correlate(current_h, 1 - current_h, mode="full")
        c5_vals = corr * dx
        max_val = np.max(c5_vals)
        best_lags = np.where(np.abs(c5_vals - max_val) < 1e-8)[0]

        # Compute gradient for the max lag(s)
        grad = np.zeros(n_points)
        for lag_idx in best_lags:
            lag = lag_idx - (n_points - 1)
            grad += grad_for_lag(current_h, lag)
        
        if len(best_lags) > 1:
            grad /= len(best_lags)
            
        # Apply Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        
        lr_scaled = lr / (np.sqrt(v_hat) + epsilon)
        current_h -= lr_scaled * m_hat
        
        # Schedule learning rate
        lr *= 1.0 / (1.0 + decay_rate * iteration)
        
        # Project onto feasible region
        current_h = project_onto_box_and_sum(current_h, target_sum)
        
        # Evaluate new C5
        c5_new = evaluate_c5(current_h)
        
        # Check for improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = current_h.copy()
            last_improvement = iteration
            
        # Occasionally add noise to escape local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            current_h += noise
            current_h = project_onto_box_and_sum(current_h, target_sum)
        
        # Early exit if not progressing
        if iteration - last_improvement > 10000:
            break

    return (best_h, best_c5, n_points)
