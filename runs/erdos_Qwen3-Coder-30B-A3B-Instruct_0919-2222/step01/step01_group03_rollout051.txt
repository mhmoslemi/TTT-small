import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector v onto the simplex with sum = target_sum and bounds [0,1]"""
    # Clip v to [0,1]
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if current_sum == target_sum:
        return v
    
    # Use water-filling approach to adjust to target sum
    v_sorted = np.sort(v)[::-1]  # Descending order
    cumsum = np.cumsum(v_sorted)
    n = len(v_sorted)
    
    # Find the threshold
    for i in range(n):
        if cumsum[i] <= target_sum:
            continue
        threshold = (target_sum - cumsum[i-1]) / (i + 1) if i > 0 else target_sum / (i + 1)
        break
    else:
        threshold = (target_sum - cumsum[n-1]) / n if n > 0 else 0
    
    # Apply threshold
    result = np.maximum(v - threshold, 0)
    # Adjust for exact sum (small numerical errors)
    adjustment = (target_sum - np.sum(result)) / len(result)
    result += adjustment
    result = np.clip(result, 0, 1)
    return result

def compute_gradient(h, f, w):
    """Compute gradient of the smooth objective with weights w applied to each shift"""
    n = len(h)
    grad = np.zeros_like(h)
    
    # Compute shifts and their index pairs
    shift_indices = []
    for s in range(-(n-1), n):
        if s >= 0:
            i = np.arange(0, n-s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n, dtype=int)
            j = i + s
        shift_indices.append((i, j))
    
    # Add gradient components for each shift
    for shift_idx, (i, j) in enumerate(shift_indices):
        if shift_idx >= len(w):
            continue
        weight = w[shift_idx]
        if weight == 0:
            continue
            
        # For i: +weight * (1 - h[j]) from sum h_i term
        # For j: +weight * (1 - h[i]) from sum h_i term  
        grad[i] += weight * (1 - h[j])
        grad[j] += weight * (1 - h[i])
        
        # For i: -weight * h[j] from product term
        # For j: -weight * h[i] from product term
        grad[i] -= weight * h[j]
        grad[j] -= weight * h[i]
        
    # Adjust for initial constant term (1)
    # The shift s where s = 0 (full overlap): contributions from h_i sum_i h_i
    # Each i gets +1 from this shift
    # This is implicit already in above terms, but gradient should be -w_sum * 1 = -1
    # But better way: derivative of sum is 1 per component, so we should have added sum(w) 
    # to each component (but we subtracted negative weights fully included)   
    # Actually, recompute: total derivative of sum_i h_i term is len(shifts) * sum(w) = sum(w) not fixed?
    # Since sum_s w_s = 1 by softmax, we should adjust: grad += sum(w) (sum_w is 1)  
    # But, that's already conceptual. Derivative of all terms = 1
    # More carefully:
    # grad = sum_s w_s * (grad f_s)
    # f_s = sum_i h_i - sum_i h_i h_{i+s}
    # derivative of sum_i h_i is 1
    # But we do not need to keep derivative of constant part
    # since it combines with same influence when computing weights
    
    # Final adjustment: Take care of simplified approach via matrix inversion vs full equation
    # We can verify gradient derivative or not add constant since per convention we do not distinguish
    # If   L = sum_s w_s * ( sum_i h_i - sum_i h_i h_{i+s} ) = (sum_s w_s) sum_i h_i - sum_s w_s sum_i h_i h_{i+s}
    # Then dL/dh_k = sum_s w_s * 1 - sum_s w_s * ( h_{k+s} + h_{k-s} )  
    # So: 
    grad -= np.sum(w)  # We can just add 1 to each component based on weight
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Precompute shift indices to reduce overhead in optimization loop
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points-s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))
    
    # Optimization parameters
    max_iter = 10000  # The number of iterations to try
    eta0 = 0.1
    beta = 1.0
    beta_max = 100.0
    beta_increment = (beta_max - beta) / (max_iter // 1000) if max_iter // 1000 else 1.0
    
    # Initial values
    h = h0.copy()
    h = project_onto_bounds_and_sum(h, target_sum)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Gradient-descent-like optimization
    try:
        for it in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Compute correlation for finding shift indices
            corr = np.correlate(h, 1 - h, mode="full")
            f = corr  # These are the non-scaled values
            
            # Compute softmax weights (shift-based)
            max_f = np.max(f)
            exp_terms = np.exp(beta * (f - max_f))
            w = exp_terms / np.sum(exp_terms) 
            
            # Compute gradient using precomputed indices
            grad = np.zeros_like(h)
            for shift_idx, (i, j) in enumerate(shift_indices):
                if shift_idx >= len(w):
                    continue
                weight = w[shift_idx] 
                if weight == 0:
                    continue
                    
                # For every shift s, we get h_i + h_j terms contributing to grad
                # f_s = sum_{i over overlap} h_i * (1 - h_{i+s})  
                #     = sum_{i over overlap} h_i - sum_{i over overlap} h_i h_{i+s}
                # Derivative w.r.t. h_k:
                # - from first part: +1 (if k in overlap)  
                # - from product part: -h_{k+s} (if k+s in overlap) 
                #                    -h_{k-s} (if k-s in overlap) 
                
                # however deriving gradient carefully directly through shifts:
                # At each pair (!) we have contribution (-w[k]*h[j]) for i, -w[k]*h[i] for j
                # and possibly +w[k]* 1 contribution for both i and j
                # So, underivative of f_s is: sum_{i} w_s * (1 - h_{i+s}) - w_s * h_{i+s}
                
                # Better approach: 
                # Direct derivation: f_s = sum_{i in overlap} h_i - sum_{i in overlap} h_i h_{i+s}
                # Gradient w.r.t. h_k:
                # 1) From sum h_i (if k in overlap): += w_s  
                # 2) From product -h_i h_{i+s}: (if k=i, then -w_s * h_{k+s}) (was earlier)
                #                           (if k=i+s, then -w_s * h_{k-s}) (later)
                # that results in gradient *= (1 - h[j]) contributions added to i term, 
                # and (-h[i]) on j term. We also added -h[i], -h[j] from product.

                # The net gradient to compute subproblem becomes:
                # grad[i] += w_s * (1 - h[j])
                # grad[j] += w_s * (+ (-h[i]))
                grad[i] += weight * (1 - h[j])
                grad[j] -= weight * h[i]
                
            # perform a step
            # Apply step size with adaptive increment
            step = eta0 / (1 + 0.01 * it)
            
            h_new = h - step * grad
            
            # project back onto feasible region
            h_new = project_onto_bounds_and_sum(h_new, target_sum)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            # Update h for next iteration
            h = h_new
            
            # Update temperature parameter (smoothing)
            beta = min(beta + beta_increment, beta_max)
            
    except Exception as e:
        pass
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
