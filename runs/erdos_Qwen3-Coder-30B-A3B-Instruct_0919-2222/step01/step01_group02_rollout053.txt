import numpy as np
import time
import random
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, d):
    """Returns h shifted by d positions, with zero padding"""
    n = len(h)
    if d == 0:
        return h.copy()
    elif d > 0:
        return np.concatenate([h[d:], np.zeros(d)])
    else:
        return np.concatenate([np.zeros(-d), h[:d]])

def project_onto_box_and_sum(v, target_sum, box_low=0, box_high=1):
    """
    Project vector v onto intersection of box [box_low, box_high]^n and hyperplane sum = target_sum.
    Returns projected vector.
    """
    n = len(v)
    # Binary search on Lagrange multiplier lambda
    low, high = -np.max(v - box_high), np.max(box_low - v)
    # Use tolerance for convergence
    tol = 1e-10
    for _ in range(40):  # max iterations
        mid = (low + high) / 2.0
        x = np.clip(v - mid, box_low, box_high)
        s = np.sum(x)
        if s < target_sum:
            high = mid
        else:
            low = mid
    x = np.clip(v - low, box_low, box_high)
    # Small correction to enforce sum exactly
    resid = target_sum - np.sum(x)
    # Spread residual over non-bound entries
    to_adjust = np.where((x > box_low) & (x < box_high))[0]
    if len(to_adjust) > 0:
        adjust_per = resid / len(to_adjust)
        x[to_adjust] += adjust_per
        x = np.clip(x, box_low, box_high)
    return x

def build_lp_model(n, binary=False):
    # Create variables
    h = cp.Variable(n)
    y = cp.Variable((n, 2*n - 1))  # shifted indices in range [-n+1, n-1]
    t = cp.Variable()

    # Constraints for McCormick envelope
    constraints = []
    for d in range(-(n - 1), n):
        for i in range(n):
            if 0 <= i + d < n:
                # Need to map shift d to index in y, adjust index offset
                y_idx = i * (2 * n - 1) + (d + n - 1)  # maps d in [-n+1, n-1] to [0, 2*n-2]
                idx_d = d + n - 1
            
                # Use a more systematic approach:
                # Create matrix of variables that need bounds, and use proper slicing in constraints
                if 0 <= i and i+d < n:
                    # The reason for writing constraints this way instead of simple blocks:
                    # For i from 0 to n-1 and for all valid shift d, we construct one way to express this.
                    constraints += [
                        y[i, idx_d] >= h[i] + h[i+d] - 1,
                        y[i, idx_d] >= 0,
                        y[i, idx_d] <= h[i],
                        y[i, idx_d] <= h[i+d],
                    ]

    # Sum constraint:
    constraints += [cp.sum(h) == n / 2]
    
    # Main constraints for C5 bound:
    for d in range(-(n - 1), n):
        start_idx = max(0, -d)
        end_idx = min(n, n - d)
        if end_idx > start_idx:
            idx_arr = np.arange(start_idx, end_idx)
            # sum of y for this shift
            y_d = cp.sum(y[idx_arr, d + n - 1])
            # note that SIP = n/2
            # C(d) = dx*(n/2 - sum(y))
            # Want : C(d) <= t*dx
            # => dx*(n/2 - sum(y)) <= t*dx
            # => n/2 - sum(y[i, idx_d]) <= t
            # => sum(y) >= (n/2) - t
            # => sum(y[i, idx_d]) <= (n/2) - t
            if binary:
                # binary, allow y to be set up as exact product
                pass  # constraints written already above
            else:
                # This section ensures correct indexing for the shifted sums
                # Correct formulation:
                constraints += [(
                    n/2 - cp.sum(y[cp.arange(start_idx, end_idx), d + n - 1])
                ) <= t]

    # Add non-negativity
    constraints += [h >= 0, h <= 1]

    if binary:
        constraints += [h == cp.Variable(n, boolean=True)]
        # In a proper MILP, we have to define y as binary or constrain with products explicitly
        # We are using simplified direct linearization for now, assuming constraints y >= ...
        # This ensures we don't require CP to impose exact product, but our constraints will still relax y even for binary h

    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    return prob, h, t, y


def solve_with_timeout(prob, solver='GLPK_MI', time_limit=20):
    """Solve a cvxpy problem with a soft timeout."""
    try:
        # If solver has an available built-in time limit
        prob.solve(solver=solver, maxTime=time_limit, verbose=False)
        return prob.status
    except Exception as e:
        # If some error occurred or timeout exceeded
        print(f"Solver timed out or raised: {e}")
        return 'timed_out'


def subgradient_refine(h_start, end_time, n_points=None):

    if n_points is None:
        n_points = len(h_start)
    h = h_start.copy()
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    iteration = 0
    stall_count = 0
    max_stall = 5000
    α0 = 0.5
    β = 0.999

    while time.time() < end_time:
        iteration += 1
        # Compute C5 and find worst d
        corr_values = np.correlate(h, 1 - h, mode='full')
        d_best = np.argmax(corr_values) - (n_points - 1)
        C_best = corr_values[d_best + (n_points - 1)]

        # Compute direction g = 1 - (shifted h) - (shifted h) for d_best
        h_shifted_forward = shift_array(h, d_best)
        h_shifted_backward = shift_array(h, -d_best)
        g = 1 - h_shifted_forward - h_shifted_backward

        # Update h with step size α
        α = α0 / (1 + β * iteration)
        h_new = h - α * g
        h_new = project_onto_box_and_sum(h_new, n_points / 2)

        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new)
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            # Perturb
            h = best_h.copy()
            h += np.random.normal(0, 0.05, n_points)
            h = project_onto_box_and_sum(h, n_points / 2)
            stall_count = 0

    return best_h, best_c5
