import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different approaches with variable number of points and restarts
    for n_points_try in [30, 42, 50, 60]:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate different initializations
        h0_try = np.zeros(n_points_try)
        dx_try = 2.0 / n_points_try
        
        # Try some heuristic constructions
        if n_points_try == 30:
            # Try piecewise constant with 3 segments
            h0_try[:10] = 0.25
            h0_try[10:20] = 0.5
            h0_try[20:] = 0.75
        elif n_points_try == 50:
            # Try piecewise constant with 4 segments
            h0_try[:12] = 0.1
            h0_try[12:25] = 0.4
            h0_try[25:37] = 0.7
            h0_try[37:] = 0.9
        elif n_points_try == 60:
            # Try some monotonic construction
            h0_try = np.linspace(0.1, 0.9, n_points_try)
        else:
            # Default case
            h0_try.fill(0.5)
            
        # Normalize
        h0_try = h0_try / (np.sum(h0_try) * dx_try) * 1.0
        h0_try = np.clip(h0_try, 0, 1)
        
        # Initial optimization with differential evolution for global search
        try:
            de_result = differential_evolution(
                objective,
                [(0, 1) for _ in range(n_points_try)],
                maxiter=50,
                popsize=10,
                seed=seed,
                callback=lambda x, convergence: None,
                atol=1e-6,
                rtol=1e-6
            )
            if de_result.success:
                h_opt = np.clip(de_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    n_points = n_points_try
        except:
            pass

    # Try a specialized approach: similar shaped functions
    for i in range(3):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Start with a good solution from the exploration
        h = best_h.copy()
        if i == 0:
            # Try to make a symmetric function
            h = np.repeat(np.linspace(0.1, 0.9, len(h)), 1)
        elif i == 1:
            # Try making a peak in the middle
            h = np.exp(-((np.arange(len(h)) - len(h)/2)**2) / (2 * (len(h)/4)**2))
            h = h / (np.sum(h) * dx) * 1.0
        else:
            # Use a sawtooth-like pattern
            h = np.abs(np.sin(np.linspace(0, 4*np.pi, len(h))))
            h = h / (np.sum(h) * dx) * 1.0
            
        h = np.clip(h, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(len(h))],
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
        # Try another restart with a random but more concentrated distribution
        h = np.random.rand(len(h)) * 0.5  # Between 0 and 0.5
        h = h / (np.sum(h) * dx) * 1.0  # Scale to sum = 1
        h = np.clip(h, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(len(h))],
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
