def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, shift):
    """Shift array by 'shift' positions, padding with zeros"""
    n = len(arr)
    if shift >= 0:
        # Shift left (positive shift): move elements to higher indices, pad with 0s
        return np.concatenate([arr[shift:], np.zeros(shift, dtype=arr.dtype)])
    else:
        # Shift right (negative shift): move elements to lower indices, pad with 0s
        s = -shift
        return np.concatenate([np.zeros(s, dtype=arr.dtype), arr[:n-s]])

def project_onto_box_simplex(v, lower=0., upper=1., target_sum=None):

        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, lower, upper)
