import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time
import random

def proj_capped_simplex(v, target_sum):
    # Project vector v onto the capped simplex {v : 0 <= v_i <= 1, sum(v) = target_sum}
    # Uses water-filling approach
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - target_sum) < 1e-12:
        return v
    elif s < target_sum:
        # Need to add more mass
        delta = (target_sum - s) / len(v)
        v = np.clip(v + delta, 0, 1)
        return v
    else:
        # Need to reduce mass
        # Sort in descending order
        v_sorted = np.sort(v)[::-1]
        # Binary search for lambda
        def gap_sum(lmbda):
            return np.sum(np.maximum(v_sorted - lmbda, 0)) - target_sum
        # Root finding
        lb = v_sorted[-1] - 1
        ub = v_sorted[0]
        eps = 1e-10
        while ub - lb > eps:
            mid = (lb + ub) / 2
            val = gap_sum(mid)
            if val >= 0:
                lb = mid
            else:
                ub = mid
        lmbda = (lb + ub) / 2
        projected = np.maximum(v - lmbda, 0)
        # Scale to exactly match target_sum
        scale = target_sum / np.sum(projected)
        projected *= scale
        return projected

def full_corr(h):
    # Compute correlation via FFT
    n = len(h)
    h_extended = np.concatenate([h, np.zeros(n - 1)])
    g_extended = np.concatenate([1 - h, np.zeros(n - 1)])
    corr_full = fftconvolve(h_extended, g_extended[::-1], mode='full')
    return corr_full[:2*n-1]

def evaluate_h(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    conv = full_corr(h)
    return np.max(conv) * dx

    block_len = n_points // S

    final_c5 = evaluate_h(best_h)
