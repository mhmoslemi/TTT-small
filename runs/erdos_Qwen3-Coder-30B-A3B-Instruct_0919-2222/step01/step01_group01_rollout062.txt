def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_simplex(v, s=1.0):
    """Project vector v onto the simplex with sum = s using the water-filling algorithm."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - s
    ind = np.arange(1, len(u) + 1)
    rho = np.where(u > css / ind, 1, 0)
    theta = np.sum(rho * u - rho * css / ind) / np.sum(rho)
    return np.maximum(v - theta, 0)

def smooth_max(x, beta=10.0):
    """Smooth approximation of max using softmax."""
    return np.log(np.sum(np.exp(beta * x))) / beta

def smooth_max_grad(x, beta=10.0):
    """Gradient of smooth max approximation."""
    exp_x = np.exp(beta * x)
    return exp_x / np.sum(exp_x) / beta

def compute_objective(h, dx, beta=10.0):
    """Compute smooth approximation of max correlation."""
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return smooth_max(corr, beta)

def compute_gradient(h, dx, beta=10.0):
    """Compute gradient of smooth objective with respect to h."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    grad_corr = np.zeros_like(h)
    
    # Gradient w.r.t. each point
    for i in range(n):
        # Approximate gradient by finite differences or analytic
        delta = 1e-6
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += delta
        h_minus[i] -= delta
        
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
        corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full") * dx
        
        grad_corr[i] = (smooth_max(corr_plus, beta) - smooth_max(corr_minus, beta)) / (2 * delta)
    
    return grad_corr

def projected_gradient_descent(h0, dx, max_iter=5000, eta=0.01, beta=1.0, beta_schedule=1.05):
    """
    Projected gradient descent with adaptive beta.
    """
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_objective(h, dx, beta)
    
    for t in range(max_iter):
        # Compute gradient
        grad = compute_gradient(h, dx, beta)
        
        # Update
        h_new = h - eta * grad
        
        # Project to domain [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to simplex (sum constraint)
        h_new = project_simplex(h_new, 1.0)
        
        # Update beta
        beta = beta * beta_schedule
        
        # Evaluate
        val = compute_objective(h_new, dx, beta)
        
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
        
        # Early stopping if no significant improvement
        if t % 100 == 0:
            if t > 0 and abs(compute_objective(h, dx, beta) - best_val) < 1e-6:
                break
    
    return best_h, best_val

def local_search(h, dx, max_iter=3000):
    """Refine solution using local exchanges."""
    h = h.copy()
    best_h = h.copy()
    best_val = evaluate_erdos_solution(h)
    
    for i in range(max_iter):
        # Random exchange
        i1, i2 = random.sample(range(len(h)), 2)
        delta = random.uniform(0.001, 0.02)
        
        # Small exchange
        if h[i1] >= delta and h[i2] + delta <= 1:
            h_new = h.copy()
            h_new[i1] -= delta
            h_new[i2] += delta
            
            # Project
            h_new = np.clip(h_new, 0, 1)
            # Re-normalize
            mean_adjust = (1.0 - np.sum(h_new)) / len(h_new)

            h_new = np.clip(h_new, 0, 1)
            
            new_val = evaluate_erdos_solution(h_new)
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
                h = h_new.copy()
    
    return best_h, best_val
