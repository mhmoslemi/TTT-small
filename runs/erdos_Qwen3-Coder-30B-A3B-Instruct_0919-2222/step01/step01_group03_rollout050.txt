import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum, lower=0.0, upper=1.0):
    """Project vector onto box constraints and sum constraint using bisection"""
    # Ensure that target_sum is within feasible range
    if target_sum < 0 or target_sum > len(v):
        raise ValueError("Target sum out of feasible bounds")
    
    v = np.asarray(v)
    # First clip to [lower, upper]
    v_clipped = np.clip(v, lower, upper)
    
    # Initial estimate: the sum of clipped values
    current_sum = np.sum(v_clipped)
    
    # If already correct, return
    if abs(current_sum - target_sum) < 1e-10:
        return v_clipped
    
    # Use bisection to find lambda such that sum of clipped values equals target
    low = -1.0
    high = 1.0
    tolerance = 1e-8
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        v_adjusted = np.clip(v - mid, lower, upper)
        new_sum = np.sum(v_adjusted)
        
        if new_sum > target_sum:
            high = mid
        else:
            low = mid
            
        if abs(new_sum - target_sum) < tolerance:
            break
    
    return np.clip(v - (low + high) / 2.0, lower, upper)

def compute_gradient(h, weights):
    """Compute gradient of the softened max objective with current weights."""
    n = len(h)
    grad = np.zeros(n)
    total_weight = np.sum(weights)
    
    # Add constant component from sum_i h_i term
    grad += total_weight
    
    # Accumulate contributions from each shift
    for shift_idx, (i_arr, j_arr) in enumerate(shift_indices):
        if weights[shift_idx] == 0:
            continue
            
        weight = weights[shift_idx]
        
        # Process contribution from i_arr to j_arr
        grad[i_arr] += weight * (1 - h[j_arr])
        # Process contribution from j_arr back to i_arr  
        grad[j_arr] -= weight * h[i_arr]
        
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Precompute shift indices for efficient gradient evaluation
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points - s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))
    
    # Determine best solution so far
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Gradient descent with soft maximum
    max_iter = 50000
    eta0 = 0.5
    beta = 1.0
    beta_max = 100.0
    beta_increment = (beta_max - beta) / (max_iter // 1000)
    
    h_current = h0.copy()
    target_sum = n_points / 2.0
    
    # Initialize with best known solution
    current_c5 = evaluate_erdos_solution(h_current)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_current.copy()
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Softmax approximation to the maximum
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        f = corr.copy()
        max_f = np.max(f)
        exp_terms = np.exp(beta * (f - max_f))
        weights = exp_terms / np.sum(exp_terms)
        
        # Compute gradient
        grad = compute_gradient(h_current, weights)
        
        # Update step size based on iteration count
        step_size = eta0 / np.sqrt(iter_count + 1)
        
        # Update solution
        h_new = h_current - step_size * grad
        
        # Project onto constraints: box + sum constraint
        h_new = project_onto_bounds_and_sum(h_new, target_sum)
        
        h_current = h_new
        
        # Periodically evaluate actual C5
        if iter_count % 100 == 0:
            current_c5 = evaluate_erdos_solution(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
        if iter_count % 1000 == 0:
            beta = min(beta + beta_increment, beta_max)
    
    # Final check of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
