import numpy as np
import time
import cvxpy as cp
from cvxpy import GLPK_MI, GLPK, ECOS, SCS


def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)


def init_h_uniform(n_points):
    return np.full(n_points, 0.5)


def project_sum(h, target_sum):
    """Project h to satisfy sum(h) = target_sum using bisection"""
    scale = target_sum / np.sum(h)
    return h * scale


def solve_milp_h_binary(n_points, max_time_s):
    """Solve the binary integer programming version for a farely small n."""
    t0 = time.time()
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    h = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Track the quadratic terms
    Q_terms = []
    constraints = []

    # Add box constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(cp.sum(h) == target_sum)
    constraints.append(t >= 0)

    # Add linearization of product terms
    y_vars = {}
    for d in range(-n_points + 1, n_points):
        for i in range(n_points):
            if 0 <= i + d < n_points:
                y = cp.Variable(boolean=True)
                y_vars[(i, d)] = y
                constraints += [
                    y <= h[i],
                    y <= h[i+d],
                    y >= h[i] + h[i+d] - 1,
                    y >= 0
                ]
                Q_terms.append(y)
    
    constraints.append(t >= dx * (target_sum - cp.sum(Q_terms)))

    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=GLPK_MI, time_limit=max_time_s)
    except Exception:
        pass  # fallback if no solution in time

    if prob.status == 'optimal':
        h_val = h.value.astype(int)
        return h_val.tolist(), evaluate_erdos_solution(h_val), n_points
    else:
        return None, float('inf'), n_points


def solve_relaxation_lp(n_points, max_time_s):
    """Solve the continuous LP relaxation to get a fractional solution."""
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        t >= 0
    ]

    Q_terms = []
    for d in range(-n_points + 1, n_points):
        for i in range(n_points):
            if 0 <= i + d < n_points:
                y = cp.Variable()
                y_vars[(i, d)] = y
                constraints += [
                    y <= h[i],
                    y <= h[i+d],
                    y >= h[i] + h[i+d] - 1,
                    y >= 0
                ]
                Q_terms.append(y)
                
    constraints.append(t >= dx * (target_sum - cp.sum(Q_terms)))

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=GLPK, time_limit=max_time_s)
    except Exception:
        pass

    if prob.status == 'optimal':
        h_val = h.value
        return h_val.tolist(), evaluate_erdos_solution(h_val), n_points
    else:
        return None, float('inf'), n_points


def scc_refinement(h_start, max_iter, max_time_s):
    """Sequential convex-concave refinement."""
    t0 = time.time()
    n_points = len(h_start)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.array(h_start, dtype=float)
    
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    for iter in range(max_iter):
        if time.time() - t0 > max_time_s:
            break
            
        # Compute full correlation and find worst-shift
        corr = np.correlate(h, 1 - h, mode='full')
        wshift = np.argmax(corr)
        wshift_val = corr[wshift]
        
        # construct linearizations for all shifts
        t = cp.Variable()
        h_vars = cp.Variable(n_points)
        constrs = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == target_sum]
        t_constrs = []
        
        for d in range(-n_points + 1, n_points):
            Q_term = 0
            for i in range(n_points):
                if 0 <= i + d < n_points:
                    Q_term += h[i] * h[i+d] + h[i] * (h_vars[i+d] - h[i+d]) + h[i+d] * (h_vars[i] - h[i])
                    # detour above is linearization in h of h[i]*h[i+d] around h.
            t_constrs.append(t >= dx * (target_sum - Q_term))
        
        constrs += t_constrs
        obj = cp.Minimize(t)
        
        prob = cp.Problem(obj, constrs)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                h_new = h_vars.value
                if h_new is not None:
                    h = h_new
                    c5_new = evaluate_erdos_solution(h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
            else:
                # fallback to keeping previous h
                pass
        except:
            pass
            
    return best_h.tolist(), best_c5, n_points


def bisection_feasibility(h_init, c5_upper, n_points, max_steps=500):
    """Use feasibility LP bisection to tighten bound."""
    dx = 2.0 / n_points
    lower = 0.
    upper = c5_upper
    for _ in range(max_steps):
        mid = (lower + upper) / 2.0
        h = cp.Variable(n_points)
        t = mid
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points/2]
        
        Q_terms = []
        for d in range(-n_points+1, n_points):
            for i in range(n_points):
                if 0 <= i + d < n_points:
                    y = cp.Variable()
                    constraints += [y <= h[i], y <= h[i+d], y >= h[i] + h[i+d] - 1, y >= 0]
                    Q_terms.append(y)
        constraints.append(t >= dx * (n_points/2 - cp.sum(Q_terms)))
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=GLPK, verbose=False)
            if prob.status == 'optimal':
                upper = mid
            else:
                lower = mid
        except:
            lower = mid
    return upper


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    n_points = 70
    time_left = lambda: budget_s - (time.time() - t0)
    
    # Step 1: try small MILP if enough time
    if time_left() >= 120:
        h_milp_result, c5_milp, _ = solve_milp_h_binary(n_points, 30)
        if h_milp_result is not None and c5_milp < 0.3807:
            return (h_milp_result, c5_milp, n_points)
    
    # Step 2: fallback to relaxation if needed
    if time_left() >= 100:
        h_relax, c5_relax, _ = solve_relaxation_lp(n_points, 50)
        if h_relax is not None:
            h_start = h_relax
        else:
            h_start = init_h_uniform(n_points)
    else:
        h_start = init_h_uniform(n_points)
    
    # Step 3: SCC refinement
    if time_left() >= 200:
        h_scc, c5_scc, _ = scc_refinement(h_start, 5000, time_left())
        h_final = h_scc
        c5_final = c5_scc
    else:
        h_final = h_start
        c5_final = evaluate_erdos_solution(h_start)

    # Step 4: bisection adoption if time
    if time_left() >= 60:
        target_c5 = bisection_feasibility(h_final, c5_final, n_points, 100)
        if target_c5 < 0.3807:
            c5_final = target_c5

    # Step 5: polishing
    if time_left() >= 30:
        last_c5 = c5_final
        h_best = np.array(h_final)
        for _ in range(500):
            if time.time() - t0 > budget_s - 1:
                break
            i, j = np.random.choice(n_points, 2, replace=False)
            h_try = h_best.copy()
            delta = 0.02 * np.random.choice([-1, 1])
            
            if 0 <= h_try[i] - delta <= 1 and 0 <= h_try[j] + delta <= 1:
                h_try[i] -= delta
                h_try[j] += delta
                h_try = project_sum(h_try, n_points / 2)
                c5_try = evaluate_erdos_solution(h_try)
                if c5_try < last_c5:
                    h_best = h_try
                    last_c5 = c5_try
                    
        h_final = h_best.tolist()
        c5_final = last_c5

    # Final evaluation
    res_c5 = evaluate_erdos_solution(h_final)
    return (h_final, res_c5, n_points)
