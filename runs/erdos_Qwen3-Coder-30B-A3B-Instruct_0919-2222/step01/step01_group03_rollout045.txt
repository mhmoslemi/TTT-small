import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try a hybrid approach
    # Start with a more targeted initialization
    h_test = np.zeros(n_points)
    
    # Try different initialization strategies
    # Strategy 1: Single peak
    peak_pos = n_points // 2
    peak_height = 1.0
    for i in range(n_points):
        # Gaussian-like peak
        h_test[i] = peak_height * np.exp(-0.5 * ((i - peak_pos) / (n_points/8))**2)
    
    # Normalize
    h_test = h_test / (np.sum(h_test) * dx) * 1.0
    h_test = np.clip(h_test, 0, 1)
    
    try:
        local_result = minimize(
            objective,
            h_test,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Try different fixed patterns
    patterns = []
    
    # Pattern 1: Two peaks
    h_two_peak = np.zeros(n_points)
    h_two_peak[:n_points//3] = 0.6
    h_two_peak[n_points//3:2*n_points//3] = 0.2
    h_two_peak[2*n_points//3:] = 0.6
    h_two_peak = h_two_peak / (np.sum(h_two_peak) * dx) * 1.0
    h_two_peak = np.clip(h_two_peak, 0, 1)
    patterns.append(h_two_peak)
    
    # Pattern 2: Optimized sawtooth
    h_saw = np.linspace(0.2, 0.8, n_points)
    h_saw = h_saw / (np.sum(h_saw) * dx) * 1.0
    h_saw = np.clip(h_saw, 0, 1)
    patterns.append(h_saw)
    
    # Pattern 3: Chaotic-like pattern
    h_chaos = np.random.rand(n_points) * 0.6 + 0.2
    h_chaos = h_chaos / (np.sum(h_chaos) * dx) * 1.0
    h_chaos = np.clip(h_chaos, 0, 1)
    patterns.append(h_chaos)
    
    for h_test in patterns:
        try:
            local_result = minimize(
                objective,
                h_test,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1500, 'ftol': 1e-9, 'gtol': 1e-9}
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Differential evolution for global search, but try to set parameters
    try:
        # Try more diverse search strategies
        de_options = {
            'maxiter': 150,
            'popsize': 20,
            'seed': seed,
            'atol': 1e-7,
            'rtol': 1e-7
        }
        
        de_result = differential_evolution(
            objective,
            bounds,
            **de_options
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
        
    # Try with a new optimal parameter and smarter constraint handling
    # We use a two-stage optimization with rescaling
    def rescaled_objective(h):
        # Rescale the function domains to avoid multiple operations
        h_scaled = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h_scaled)
    
    # Try several random restarts but more intelligently
    n_restarts = 10
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate template by random distribution but keep some structure
        h_start = np.random.rand(n_points) * 0.6 + 0.2
        # Manually ensure integral constraint is met
        h_start = h_start / (np.sum(h_start) * dx) * 1.0
        h_start = np.clip(h_start, 0, 1)
        
        try:
            local_result = minimize(
                rescaled_objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = rescaled_objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            continue
    
    # Final local refinement
    try:
        final_result = minimize(
            rescaled_objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1500, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if final_result.success:
            h_opt = np.clip(final_result.x, 0, 1)
            c5_opt = rescaled_objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
