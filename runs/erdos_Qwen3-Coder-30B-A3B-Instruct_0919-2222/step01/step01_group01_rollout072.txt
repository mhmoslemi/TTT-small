def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_to_feasible(h, target_sum):
    """Project h to feasible set: 0 <= h[i] <= 1 and sum(h) = target_sum"""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Water-filling approach
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        # Sort indices in ascending order of h values
        idx = np.argsort(h)
        remaining = diff
        for i in range(len(h)):
            if remaining <= 0:
                break
            j = idx[i]
            max_increase = 1.0 - h[j]
            increase = min(max_increase, remaining)
            h[j] += increase
            remaining -= increase
    else:
        # Need to decrease values
        # Sort indices in descending order of h values
        idx = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(len(h)):
            if remaining <= 0:
                break
            j = idx[i]
            max_decrease = h[j]
            decrease = min(max_decrease, remaining)
            h[j] -= decrease
            remaining -= decrease
    return h

def smooth_max_gradient(h, beta=100.0):
    """Compute gradient of soft max approximation to max of correlations using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Joints are indices [0, 1, ..., 2*n-2]; 
    # corr[i] corresponds to shift i - (n-1)
    # We want to find gradient of max_k corr[k]
    # Using smooth approximation that is:
    # log(sum_k exp(beta * corr[k])) / beta
    # gradient of this w.r.t. h = ...
    # Through chain rule, we need gradient of each corr[k]
    # But rather than computing the interaction accurately, 
    # we compute gradient E[corr_i] ~ Gaussian random directions, to approximate the subgradient
    # Simplified approach below that simulates a "first-order" subgradient

    # So for large beta, we approximate:
    # max_k corr[k] ~ (1/beta) * log(sum_k exp(beta * corr[k]))
    # grad w.r.t. h = (exp(beta * corr) * gradient_c_corr) / sum(exp(beta * corr))
    
    # Num is somewhat complex to compute; instead approximate:
    # solve: minimize \|g - e_i\| for e_i entry 1 at i, g in convex hull of gradients
    
    # Simpler solution: use finite difference or exact gradient w.r.t. one representative
    # We will select the index that corresponds to the maximum, and compute the gradient there
    max_idx = np.argmax(corr)
    
    # Gradient assuming h_i contributes d h_i / dx contributions to each corr_k that depends on h_i
    # Take a pseudo-gradient: track the exact derivative involved (proportional to h or 1-h)
    # But that's tricky without the full indices. For now, we compute an approximate gradient:
    
    # Get the discrete support for correlation calculations
    # Shift by max_idx - (n-1) 
    shift = max_idx - (n - 1)
    
    # Replace with a prettier way to compute gradient
    # Classic gradient w.r.t. correlations 
    # We compute:
    # let corr = h * (1-h)[shift] 
    # d corr/d h_i = (1-h_i) * delta_{i,shift}
    # So we need exact shift
    
    # Might be better to use finite differences directly:
    # Now actually compute basic gradient via finite differencing.
    # For better solver, avoid lambda...
    grad = np.zeros_like(h)

    # manually compute the indices for max correlation contribution
    # corr[i] = integral h(x) * (1-h(x+shift)) dx = sum over discretized terms
    
    # With dx default as fixed, and h step function over n bins
    # For a fixed shift k (shift index in 0..2n-2), 
    # correlations involve terms of h[x]* (1 - h[x+k]) 
    # So if x is index i in [0,n-1], and x+k is index j in [0,n-1], then 
    # we consider contribution to corr = h[i] * (1-h[j])  
    # For all combinations for fixed k
    
    # So let's recompute and compute approx gradients on this:
    grad = np.zeros_like(h)
    for i in range(n):
        # for configuration h with i-th element perturbed, compute resulting change in corr(k=some,some_idx) that's max
        max_corr_idx = np.argmax(corr)
        shift_for_max = max_corr_idx - (n-1)  # reconstruct actual shift

        if 0 <= i + shift_for_max < n:
            # i-th element affects corr when x causes first term with h[i], 
            # x + shift_for_max causes second term in h
            # Note: indicating index for case of positive contribution to corr due to shift
            grad[i] += -1.0  # since h[i]* (1 - h[i+shift_for_max]) contribution        
    return grad

def compute_gradient_approx(h, beta=1e5):
    # Compute the closed form of gradient w.r.t. one max term.
    # Use the chunked sums from the indices involved.
    # For convenience, compute a version approximating gradient around maximum
    n = len(h)
    dx = 2.0 / n

    corr = np.correlate(h, 1 - h, mode="full")
    max_i = np.argmax(corr)
    max_shift = max_i - (n - 1)

    # Create influence matrix: how much increasing h[k] affects corr[shift]
    # Actually this is more precisely about i + adj == j  where corr[k] = sum over i h[i] * (1 - h[i + shift])
    grad = np.zeros(n)
    # For each i: how much does h[i] contribute to corr_{shift} = h[i]* (1-h[i+shift])
    for i in range(n):
        pos = i + max_shift
        if 0 <= pos < n:
            # Influence of h[i] on that term= exists
            grad[i] += 1.0 - h[pos]   # For h[i] * (1-h[pos]), contribution to d/d(h[i]) = (1-h[pos])
    return grad

def projected_gradient_descent(h0, target_sum, max_iter=20000, lr=1e-3, beta=1e2, print_every=1000):
    """Projected gradient descent with smooth max"""
    h = h0.copy()
    h = project_to_feasible(h, target_sum)
    dx = 2.0 / len(h)
    loss_history = []
    
    for i in range(max_iter):
        # Compute gradient by finite difference or approximation
        grad = compute_gradient_approx(h, beta)
        # Gradient step
        h_new = h - lr * grad
        h_new = project_to_feasible(h_new, target_sum)
        h = h_new
        
        if i % print_every == 0:
            val = evaluate_erdos_solution(h)
            loss_history.append(val)
            # print(f"iter {i}: {val}")
            
        # Dynamic learning rate decay
        if i > 5000:
            # Slow down and increase temperature of fake softmax
            lr *= 0.9999
        beta = min(1e7, beta * 1.0001)  # make it grow slowly
        
    val = evaluate_erdos_solution(h)
    return h, val
