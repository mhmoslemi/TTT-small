import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Hybrid approach: Local optimization with different strategies
    
    # Strategy 1: Direct local optimization with different starting points 
    for strategy in range(3):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        h_start = h0.copy()
        
        # Different starting constructions
        if strategy == 0:  # Original
            pass
        elif strategy == 1:  # Random restarts with different initialization
            h_start = np.random.rand(n_points) * 0.5
            h_start = h_start / (np.sum(h_start) * dx) * 1.0  # Scale to sum = 1
        else:  # Piecewise constant with better distribution
            h_start = np.zeros(n_points)
            # Try wave-like pattern
            x = np.linspace(0, 2, n_points)
            h_start = 0.5 + 0.3 * np.sin(2 * np.pi * x) 
            h_start = np.clip(h_start, 0, 1)
            
        try:
            local_result = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Strategy 2: Differential evolution with better mutation and crossover
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=150,
            popsize=20,
            seed=seed,
            mutation=(0.5, 1.0),
            recombination=0.7,
            callback=lambda x, convergence: None,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Strategy 3: A hybrid with topk selection and intensification
    # Start with de_result if available, or h0
    h_current = best_h.copy()
    for iter_no in range(100):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perturb solution
        h_perturbed = h_current.copy()
        # Add small noise to some points (not all)
        noise_indices = np.random.choice(n_points, size=n_points//3, replace=False)
        for i in noise_indices:
            h_perturbed[i] += np.random.normal(0, 0.01)
        
        # Clip and adjust
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Re-normalize to integral = 1
        h_perturbed = h_perturbed / (np.sum(h_perturbed) * dx) * 1.0
        
        try:
            local_result = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_current = h_opt.copy()
        except:
            pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
