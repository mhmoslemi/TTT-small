import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def lp_initialize(n):
    """Use convex relaxation to get a fractional initial point"""
    h = cp.Variable(n, name="h")
    u = cp.Variable((n, 2*n - 1), name="u")  # u[i,d] for shift d from -n+1 to n-1
    t = cp.Variable(name="t")

    constraints = [
        cp.sum(h) == n / 2,
        h >= 0,
        h <= 1,
        t >= 0
    ]

    # Add McCormick envelope constraints for u[i,d].
    for i in range(n):
        for d in range(-(n - 1), n):
            if 0 <= i + d < n:
                idx = d + (n - 1)
                constraints += [
                    u[i, idx] >= 0,
                    u[i, idx] <= h[i],
                    u[i, idx] <= h[i + d],
                    u[i, idx] >= h[i] + h[i + d] - 1
                ]

    # Define shifts and s_d values
    for d in range(-(n - 1), n):
        if d < 0:
            s_d = cp.sum([u[i, d + (n - 1)] for i in range(n + d, n)])
        else:
            s_d = cp.sum([u[i, d + (n - 1)] for i in range(n - d)])
        constraints.append(t >= n/2 - s_d)

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        # Fall back to a random solution if optimization fails
        h_val = np.random.rand(n)
        h_val = h_val * (n / (2 * np.sum(h_val)))
        return h_val

    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE:
        return np.array(h.value)
    else:
        # Fallback if optimization fails
        h_val = np.random.rand(n)
        h_val = h_val * (n / (2 * np.sum(h_val)))
        return h_val

def project_onto_simplex(v, target_sum):
    """Project a vector to the simplex with given sum, with box constraints [0,1]"""
    m = len(v)
    v = np.clip(v, 0, 1)
    # Shift by mean
    mean_diff = (np.sum(v) - target_sum) / m
    v -= mean_diff
    # Clamp to [0,1]
    v = np.clip(v, 0, 1)
    # Renormalize to sum to target_sum
    v *= target_sum / np.sum(v)
    return v

def subgradient(h, worst_shifts):
    """Compute subgradient of C5 at h (for given worst shifts)."""
    n = len(h)
    grad = np.zeros(n)
    # For each d in worst_shifts, add contribution of h_{i+d} and h_{i-d}
    for d in worst_shifts:
        if d >= 0:
            # Add h_{i+d} for indices i where i+d < n
            for i in range(n - d):
                grad[i] -= h[i + d]
            # Add h_{i-d} for indices i where i-d >= 0
            for i in range(d, n):
                grad[i] -= h[i - d]
        else:
            # Add h_{i+d} for indices i where i+d >= 0
            for i in range(-d, n):
                grad[i] -= h[i + d]
            # Add h_{i-d} for indices i where i-d < n
            for i in range(n):
                if i - d < n:
                    grad[i] -= h[i - d]
    return -grad  # Since we compute for max, subgradient of -min is the negative

def gradient_descent(h0, max_iter=300):
    """Apply projected gradient descent to refine h"""
    h = h0.copy()
    n = len(h)
    target_sum = n / 2
    for k in range(max_iter):
        # Find current worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = max(corr)
        worst_shifts = [i for i in range(len(corr)) if abs(corr[i] - max_val) < 1e-6]
        # Compute subgradient
        g = subgradient(h, worst_shifts)
        # Projected gradient step
        alpha = 0.02 / (1 + k / 100)
        h = h + alpha * g
        # Projection
        h = project_onto_simplex(h, target_sum)
    return h

def binary_rounding_and_swap_improve(h_frac, n):
    """Round fractional solution to binary and improve via swaps"""
    # Binary rounding (take top n/2 to 1)
    sorted_indices = np.argsort(h_frac)[::-1]
    h_binary = np.zeros(n)
    for i in range(n//2):
        h_binary[sorted_indices[i]] = 1
    # Improve using swap moves (target worst shift)
    while True:
        corr = np.correlate(h_binary, 1 - h_binary, mode="full")
        d_star = np.argmax(corr)
        # For shift d*, get the actual d
        d_actual = d_star - (n - 1)
        # Find indices for swap move
        found = False
        for i in range(n):
            j = i + d_actual
            if 0 <= j < n:
                if h_binary[i] == 1 and h_binary[j] == 0:
                    # Try to find a j' where h_j = 0, h_{j+d*} = 1 for a suitable j
                    # This is O(n^2) but with n=42, accept for simplicity
                    for k in range(n):
                        l = k + d_actual
                        if 0 <= l < n:
                            if h_binary[k] == 0 and h_binary[l] == 1:
                                h_binary[i], h_binary[k] = h_binary[k], h_binary[i]
                                h_binary[j], h_binary[l] = h_binary[l], h_binary[j]
                                found = True
                                break
                    if found:
                        break
        if not found:
            break
    return h_binary

def simulated_annealing(h_init, max_iter=10000, T0=0.02):
    """Perform simulated annealing on fractional values"""
    h_current = h_init.copy()
    n = len(h_current)
    target_sum = n / 2
    h_current = project_onto_simplex(h_current, target_sum)
    C5_current = evaluate_erdos_solution(h_current)
    best_h = h_current.copy()
    best_c5 = C5_current
    T = T0
    alpha = 0.999
    for i in range(max_iter):
        if i % 200 == 0:
            T *= alpha
        # Make a random step (small mass transfer)
        i1 = np.random.randint(0, n)
        i2 = np.random.randint(0, n)
        if i1 == i2:
            continue
        # Sample delta from a small uniform interval
        delta_max = min(h_current[i1], 1 - h_current[i2], T)
        if delta_max <= 0:
            continue
        delta = np.random.uniform(0, delta_max)
        # Apply the move
        h_new = h_current.copy()
        h_new[i1] -= delta
        h_new[i2] += delta
        # Project back onto simplex and box
        h_new = project_onto_simplex(h_new, target_sum)
        C5_new = evaluate_erdos_solution(h_new)
        if C5_new < C5_current or np.random.rand() < np.exp(-(C5_new - C5_current) / T):
            h_current = h_new
            C5_current = C5_new
        if C5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = C5_current
    return best_h, best_c5
