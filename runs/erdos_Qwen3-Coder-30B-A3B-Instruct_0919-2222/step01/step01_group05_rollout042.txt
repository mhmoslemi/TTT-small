import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import math

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = max(corr) * dx
    return c5

def project_onto_simplex(h, target_sum):
    # Correct the sum by scaling
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    new_h = h * (target_sum / current_sum)
    return new_h

def lp_initialize(n):
    # Solve the LP relaxation for initial guess
    h_vars = Variable(n)
    u_vars = {}
    for d in range(-n+1, n):
        u_vars[d] = [Variable() for i in range(max(0, i) if d >= 0 else max(0, i+d), min(n, i-d) if d < 0 else min(n, i+d))]

    # Define constraints
    constraints = [
        sum(h_vars) == n / 2,
        h_vars >= 0,
        h_vars <= 1
    ]
    
    # Add McCormick envelope constraints
    for d in range(-n+1, n):
        for i in range(n):
            if 0 <= i+d < n:
                u = Variable()
                u_vars[d][i] = u
                constraints.extend([
                    u >= 0,
                    u <= h_vars[i],
                    u <= h_vars[i+d],
                    u >= h_vars[i] + h_vars[i+d] - 1
                ])

    # Objective: minimize max d of sum of u_{i, d} for each shift d
    t = Variable()
    obj = Minimize(t)

    for d in range(-n+1, n):
        s_d = 0
        for i in range(n):
            if 0 <= i+d < n:
                s_d += u_vars[d][i]
        constraints.append(t >= n/2 - s_d)

    prob = Problem(obj, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return np.array([float(h_vars[i].value) for i in range(n)])

def worst_shift(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    d_star = np.argmax(corr)
    return d_star - (n-1)

def binary_rounding(h, w):
    rounded = np.zeros_like(h)
    sorted_indices = np.argsort(h)[::-1]
    num_ones = int(w)
    rounded[sorted_indices[:num_ones]] = 1
    return rounded

def swap_improve(h):
    h = h.copy()
    n = len(h)
    while True:
        d_star = worst_shift(h)
        found_swap = False
        for i in range(n):
            j = (i + d_star) % n
            if h[i] == 1 and h[j] == 0:
                h[i] = 0
                h[j] = 1
                found_swap = True
                break
            j = (i - d_star) % n if d_star > 0 else (i + d_star) % n
            if h[i] == 0 and h[j] == 1:
                h[i] = 1
                h[j] = 0
                found_swap = True
                break
        if not found_swap:
            break
    return h

def sa_step(h, T, d_star):
    n = len(h)
    i = np.random.randint(0, n)
    j = (i + np.random.choice([-1, 1]) * np.random.randint(1, n//4)) % n
    if i == j:
        return h
    delta_max = min(h[i], 1 - h[j], 0.2)
    if delta_max <= 0:
        return h
    delta = np.random.uniform(0, delta_max)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    h_new = np.clip(h_new, 0, 1)
    return h_new

def subgradient_step(h, t):
    n = len(h)
    # Compute subgradient wrt worst shift d_star
    d_star = worst_shift(h)
    g = np.zeros(n)
    for i in range(n):
        j1 = (i - d_star) % n
        j2 = (i + d_star) % n
        if 0 <= i + d_star < n:
            g[i] += h[(i + d_star) % n]
        if 0 <= i - d_star < n:
            g[i] += h[(i - d_star) % n]
    # Step size, decay over time
    eta = 0.1 / (1 + t)
    h_new = h - eta * g
    h_new = project_onto_simplex(h_new, n/2)
    h_new = np.clip(h_new, 0, 1)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Test various grid sizes carefully chosen to fit within budget expectation
    n_values = [42, 48, 60]
    
    for n in n_values:
        if time.time() - start_time > budget_s * 0.8:
            break
            
        dx = 2.0 / n
        h0 = lp_initialize(n)
        # Normalize
        h0 = project_onto_simplex(h0, n / 2)
        
        # Evaluate initial LP result
        c5_0 = evaluate_erdos_solution(h0)
        if c5_0 < best_c5:
            best_h = h0.copy()
            best_c5 = c5_0
            best_n = n
                
        # Binary rounding
        h_binary = binary_rounding(h0, n/2)
        h_binary_clean = project_onto_simplex(h_binary, n/2)
        c5_bin = evaluate_erdos_solution(h_binary_clean)
        if c5_bin < best_c5:
            best_h = h_binary_clean.copy()
            best_c5 = c5_bin
            best_n = n
            
        # Hill climbing on binary
        h_climb = swap_improve(h_binary_clean)
        c5_climb = evaluate_erdos_solution(h_climb)
        if c5_climb < best_c5:
            best_h = h_climb.copy()
            best_c5 = c5_climb
            best_n = n
            
        # Simulated annealing
        if time.time() - start_time > budget_s * 0.95:
            break
        h_sa = h_climb.copy()
        T = 0.05 * c5_climb
        alpha = 0.995
        max_iter = 10000
        for t in range(max_iter):
            if time.time() - start_time > budget_s * 0.98:
                break
            d_star = worst_shift(h_sa)
            h_new = sa_step(h_sa, T, d_star)
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < c5_climb or np.random.rand() < np.exp(-(c5_new - c5_climb)/T):
                h_sa = h_new
                c5_climb = c5_new
                if c5_climb < best_c5:
                    best_h = h_sa.copy()
                    best_c5 = c5_climb
                    best_n = n
            T *= alpha
                
        # Gradient step
        for t in range(500):
            if time.time() - start_time > budget_s * 0.99:
                break
            h_gd = subgradient_step(h_sa, t)
            c5_gd = evaluate_erdos_solution(h_gd)
            if c5_gd < best_c5:
                best_h = h_gd.copy()
                best_c5 = c5_gd
                best_n = n
            h_sa = h_gd

    if best_h is not None:
        # Final projection
        h_final = project_onto_simplex(best_h, best_n/2)
        h_final = np.clip(h_final, 0, 1)
        final_c5 = evaluate_erdos_solution(h_final)
        return (h_final, final_c5, best_n)
    else:
        return (np.ones(42) * 0.5, 0.5, 42)
