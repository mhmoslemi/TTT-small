import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points, seg_len):
    """Build full h array from segment heights"""
    return np.repeat(heights, seg_len)

def evaluate_heights(heights, n_points, seg_len, penalty_factor=10.0):
    """Evaluate the objective function for given heights"""
    h = build_h(heights, n_points, seg_len)
    c5 = evaluate_h(h)
    # Add penalty for sum constraint
    sum_heights = np.sum(heights)
    penalty = penalty_factor * (sum_heights - len(heights)/2)**2
    return c5 + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use 140 points for a good trade-off
    n_points = 140
    seg_len = 10  # 140/14 = 10 points per segment
    S = n_points // seg_len
    
    # Initial configuration: try a step function with S=14 segments
    initial_heights = np.array([0.8] * (S//2) + [0.2] * (S//2))
    initial_heights = initial_heights / np.sum(initial_heights) * S/2

    # Bounds for each height segment
    bounds = [(0, 1) for _ in range(S)]
    
    # Optimization parameters
    maxiter = 100
    popsize = 10
    
    result = None
    best_c5 = float('inf')
    best_heights = None
    
    # Run multiple restarts with different random seeds
    num_restarts = 3
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + restart)
        
        # Random perturb initial heights
        random_perturb = np.random.normal(0, 0.1, S)
        heights_start = initial_heights + random_perturb
        heights_start = np.clip(heights_start, 0, 1)
        heights_start = heights_start / np.sum(heights_start) * S/2
        
        # Optimization
        try:
            de_result = differential_evolution(
                lambda x: evaluate_heights(x, n_points, seg_len),
                bounds,
                maxiter=maxiter,
                popsize=popsize,
                seed=seed + restart,
                disp=False,
                polish=True
            )
            
            if de_result.success:
                heights_opt = de_result.x
                # Ensure sum constraint is exactly met
                heights_opt = heights_opt / np.sum(heights_opt) * S/2
                h_opt = build_h(heights_opt, n_points, seg_len)
                c5_opt = evaluate_h(h_opt)
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_heights = heights_opt.copy()
                    
        except Exception as e:
            continue
        
        # Early stopping
        if time.time() - start_time > budget_s - 1:
            break
    
    # If no optimization succeeded, use initial
    if best_heights is None:
        best_heights = initial_heights
    
    # Refine with local optimization
    try:
        refined_result = differential_evolution(
            lambda x: evaluate_heights(x, n_points, seg_len),
            bounds,
            maxiter=50,
            popsize=8,
            seed=seed,
            disp=False,
            polish=True
        )
        if refined_result.success:
            heights_refined = refined_result.x
            heights_refined = heights_refined / np.sum(heights_refined) * S/2
            h_refined = build_h(heights_refined, n_points, seg_len)
            c5_refined = evaluate_h(h_refined)
            
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_heights = heights_refined
    except Exception as e:
        pass
    
    # Final evaluation
    h_final = build_h(best_heights, n_points, seg_len)
    final_c5 = evaluate_h(h_final)
    
    return (h_final, final_c5, n_points)
