import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt
import cvxpy as cp
from cvxpy import ECOS

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def lp_initialize(n_points):
    """Solve LP relaxation using McCormick envelopes"""
    dx = 2.0 / n_points
    h = cp.Variable(n_points)
    u = cp.Variable((n_points, 2 * n_points - 1))  # for each shift d ∈ [−(n−1), n−1]
    t = cp.Variable()
    constraints = [cp.sum(h) == n_points / 2]
    constraints += [0 <= h[i] <= 1 for i in range(n_points)]
    # Add McCormick inequalities for u[i,d]
    for i in range(n_points):
        for d in range(-(n_points - 1), n_points):
            if 0 <= i + d < n_points:
                d_idx = d + n_points - 1  # Map d to index in u
                constraints += [
                    u[i, d_idx] >= 0,
                    u[i, d_idx] <= h[i],
                    u[i, d_idx] <= h[i + d],
                    u[i, d_idx] >= h[i] + h[i + d] - 1
                ]
    # Compute s_d for all shifts
    s = [cp.sum([u[i, d + n_points - 1] for i in range(n_points) if 0 <= i + d < n_points]) for d in range(-(n_points - 1), n_points)]
    # Constraint: t >= w - s_d for all d
    w = n_points / 2
    constraints += [t >= w - s[d + n_points - 1] for d in range(-(n_points - 1), n_points)]
    problem = cp.Minimize(t)
    prob = cp.Problem(problem, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return np.array(h.value)
    except:
        pass
    return None

def project_to_simplex(h, target_sum):
    """Project h onto the simplex: sum(h) = target_sum and h ∈ [0,1]^n."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to add to h
        idx = np.argsort(h)
        for i in range(len(h) - 1, -1, -1):
            if diff <= 0:
                break
            if h[idx[i]] < 1:
                add = min(1 - h[idx[i]], diff)
                h[idx[i]] += add
                diff -= add
    else:
        # Need to subtract from h
        idx = np.argsort(h, kind='mergesort')[::-1]
        for i in range(len(h)):
            if diff >= 0:
                break
            if h[idx[i]] > 0:
                subtract = min(h[idx[i]], -diff)
                h[idx[i]] -= subtract
                diff += subtract
    return h

def swap_improve(h, dx):
    """Perform greedy worst-shift swap improvement on binary h"""
    n = len(h)
    h_copy = h.copy()
    while True:
        corr = np.correlate(h_copy, 1 - h_copy, mode='full')
        best_d = np.argmax(corr)
        max_val = corr[best_d]
        if max_val <= n / 2:  # No point in improving if already optimal
            break
        d = best_d - (n - 1)
        # Find indices with h[i]=1, h[i+d]=0 and reverse
        # Find pairs
        swapped = False
        for i in range(n):
            j = i + d
            if 0 <= j < n and h_copy[i] == 1 and h_copy[j] == 0:
                # Look for a another pair i', j' where h[i'] = 0, h[j'] = 1
                for i2 in range(n):
                    j2 = i2 + d
                    if 0 <= j2 < n and h_copy[i2] == 0 and h_copy[j2] == 1:
                        h_copy[i], h_copy[i2] = h_copy[i2], h_copy[i]
                        h_copy[j], h_copy[j2] = h_copy[j2], h_copy[j]
                        swapped = True
                        break
                if swapped:
                    break
        if not swapped:
            break
    return h_copy

def sa_move(h, i, j, delta):
    """Transfer mass delta from index i to j"""
    h = h.copy()
    if delta == 0:
        return h
    h[i] = max(0, h[i] - delta)
    h[j] = min(1, h[j] + delta)
    return h

def simulated_annealing(h_initial, dx, budget_s):
    """Run fractional simulated annealing on h"""
    n = len(h_initial)
    h_curr = h_initial.copy()
    h_best = h_curr.copy()
    c5_best = evaluate_erdos_solution(h_curr)
    temp_start = 0.05 * c5_best
    temp = temp_start
    max_iter = 5000
    iter_count = 0
    # Track improvements
    last_improvement = 0
    while iter_count < max_iter and temp > 1e-8:
        iter_count += 1
        if iter_count - last_improvement > 1000:
            break
        # Choose random indices to transfer
        i, j = np.random.choice(n, 2, replace=False)
        delta_max = min(h_curr[i], 1 - h_curr[j], 0.2)
        if delta_max <= 0:
            continue
        delta = np.random.uniform(0, delta_max)
        h_new = sa_move(h_curr, i, j, delta)
        h_new = project_to_simplex(h_new, n / 2)
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            last_improvement = iter_count
        if iter_count % 100 == 0:
            temp *= 0.99  # Cooling schedule
        # Else accept with probability
        if c5_new >= c5_curr:
            p = np.exp(-(c5_new - c5_curr) / temp)
            if np.random.rand() < p:
                h_curr = h_new.copy()
                c5_curr = c5_new
            else:
                continue
        else:
            h_curr = h_new.copy()
            c5_curr = c5_new
    return h_best, c5_best

def subgradient_descent(h_initial, dx, budget_s):
    """Refine fractional value with projected gradient descent on max in overlap"""
    n = len(h_initial)
    h = h_initial.copy()
    h = project_to_simplex(h, n / 2)
    max_iter = 200
    step_size = 0.1 * dx
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        d_star = np.argmax(corr)
        d = d_star - (n - 1)
        # Compute subgradient of C5 at h for shift d_star
        g = np.zeros(n)
        for i in range(n):
            j1 = i + d
            j2 = i - d
            if 0 <= j1 < n:
                g[i] -= h[j1]
            if 0 <= j2 < n:
                g[i] -= h[j2]
        # Gradient step
        h = h - step_size * g
        h = project_to_simplex(h, n / 2)
    return h, evaluate_erdos_solution(h)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Consider several grid sizes for better chance of leveraging discrete structure
    grid_sizes = [42, 48, 54, 60]
    
    for n in grid_sizes[:2]:  # Let's avoid very high n due to time
        if time.time() - start_time > budget_s * 0.9:
            break
        dx = 2.0 / n
        # LP initialization
        h_init = lp_initialize(n)
        if h_init is None:
            #_FALLBACK
            h_init = np.random.rand(n) * 0.5
            h_init = h_init * (n / (2 * np.sum(h_init)))
        else:
            h_init = project_to_simplex(h_init, n / 2)

        # Evaluate LP solution
        c5_eval = evaluate_erdos_solution(h_init)
        if c5_eval < best_c5:
            best_h = h_init.copy()
            best_c5 = c5_eval
            best_n = n

        # Binary rounding + swap improvement
        binary_rounded = np.zeros(n)
        idxs = np.argsort(h_init)[::-1][:n//2]
        binary_rounded[idxs] = 1
        h_binary = project_to_simplex(binary_rounded, n / 2)
        h_binary = swap_improve(h_binary, dx)
        c5_binary = evaluate_erdos_solution(h_binary)
        if c5_binary < best_c5:
            best_h = h_binary.copy()
            best_c5 = c5_binary
            best_n = n

        # Simulated Annealing
        if time.time() - start_time > budget_s * 0.8:
            break
        h_sa, c5_sa = simulated_annealing(h_init, dx, budget_s * 0.4)
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa
            best_n = n

        # Subgradient descent on refined solution
        if time.time() - start_time > budget_s * 0.95:
            break
        h_final, c5_final = subgradient_descent(best_h, dx, budget_s * 0.1)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
            best_n = n

    # Final correction to exact integral
    best_h = project_to_simplex(best_h, best_n / 2)
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, best_n)
