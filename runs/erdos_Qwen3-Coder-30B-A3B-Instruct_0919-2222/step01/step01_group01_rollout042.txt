def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def projected_gradient_descent(h_init, max_iter=5000, eta0=0.01, beta0=1.0, 
                              beta_growth=1.005, eps=1e-8, dx=2.0/70):
    """
    Projected Gradient Descent to minimize max correlation.
    Uses soft maximum approximation of max.
    """
    h = h_init.copy()
    n = len(h)
    target_sum = n / 2.0
    
    def softmax_max(x, beta):
        # Compute smooth approximation to max
        exp_x = np.exp(beta * x)
        return np.sum(exp_x * x) / np.sum(exp_x)
    
    def softmax_max_grad(x, beta):
        # Gradient of smooth maximum
        exp_x = np.exp(beta * x)
        num = np.sum(exp_x * (beta * x + 1)) * np.sum(exp_x) - np.sum(exp_x * x) * np.sum(exp_x * beta)
        den = np.sum(exp_x)**2
        return num / den
    
    def project_simplex(v, s=1.0):
        # Water-filling algorithm to project onto simplex
        v = np.array(v)
        u = np.sort(v)[::-1]
        rho = 0
        for i in range(len(u)):
            if np.sum(u[:i+1]) - (i+1) * s > 0:
                rho = i
        theta = (np.sum(u[:rho+1]) - s) / (rho + 1)
        return np.maximum(v - theta, 0)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    beta = beta0
    eta = eta0
    
    for t in range(max_iter):
        # Compute current correlation matrix
        corr = np.correlate(h, 1 - h, mode="full") 
        k_max = np.argmax(corr) - len(h) + 1
        max_val = corr[k_max + len(h) - 1]
        
        # gradients wrt each element
        grad = np.zeros_like(h)
        for i in range(len(h)):
            for j in range(len(h)):
                k = j - i
                if 0 <= k + len(h) - 1 < len(corr):
                    if abs(corr[k + len(h) - 1] - max_val) < 1e-8:
                        grad[i] += (1 - h[j]) * dx
                        grad[j] -= h[i] * dx
        
        # Update with learned step size
        h_new = h - eta * grad 

        # Project to simplexes
        h_new = project_simplex(h_new, target_sum)
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new objective
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject 
        if c5_new < best_c5:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
        elif random.random() < 0.01:  # Accept some worse steps for escape
            h = h_new

        # Adaptive step size and learning rate
        eta = max(1e-6, eta * 0.99995)
        beta = beta * beta_growth
        
        if t % 500 == 0:
            if math.isnan(c5_new) or abs(c5_new - best_c5) < eps:
                break
    
    return best_h, best_c5

def local_exchange(h, max_iter=1000):
    """Local search: exchange small mass to improve C5."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    current_c5 = evaluate_erdos_solution(h)
    
    for _ in range(max_iter):
        # Pick two indices to shift mass between
        i = random.randint(0, n-1)
        j = random.randint(0, n-1)
        if i == j:
            continue
            
        # Propose move
        delta = random.uniform(0.005, 0.05)
        delta = min(delta, h[i], 1.0 - h[j])
        
        if delta < 1e-6:
            continue
            
        # New h after move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        
        # Project back to sum constraint
        adjustment = (n / 2.0 - current_sum) / n
        h_new += adjustment
        h_new = np.clip(h_new, 0, 1)
        
        new_c5 = evaluate_erdos_solution(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
    
    return h, current_c5
