import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points, seg_len):
    # Build full h array from heights using repeated segments
    h = np.repeat(heights, seg_len)
    return h

def evaluate_heights(heights, n_points, seg_len, penalty_factor=10.0):
    # Build h from heights
    h = build_h(heights, n_points, seg_len)
    # Compute C5 (max correlation)
    c5 = evaluate_h(h)
    # Add penalty for sum constraint
    target_sum = len(heights) / 2.0
    sum_heights = np.sum(heights)
    penalty = penalty_factor * (sum_heights - target_sum)**2
    return c5 + penalty

def project_to_bounds_and_sum(heights, target_sum):
    # Project heights to [0,1] and adjust to match target sum
    # Use iterative approach: clip and adjust
    h = np.clip(heights, 0, 1)
    current_sum = np.sum(h)
    if np.abs(current_sum - target_sum) < 1e-6:
        return h
    # Simple approach: subtract offset from each
    offset = (current_sum - target_sum) / len(h)
    h = h - offset
    # Clip and rebalance
    h = np.clip(h, 0, 1)
    # Small adjustments to fix fine errors
    diff = np.sum(h) - target_sum
    if abs(diff) > 1e-6:
        # Find indices to adjust
        rearrange = np.argsort(h)
        # Decrease from highest to lowest to decrease sum

            if diff > 1e-6:
                decrease = min(diff, h[i])

                diff -= decrease

                break

            if diff < -1e-6:
                increase = min(-diff, 1 - h[i])
                h[i] += increase
                diff += increase
            else:
                break
    return h
