def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def compute_product_sums(h):
    """Compute sum_i h_i * h_{i+s} for all valid shifts s"""
    n = len(h)
    p_sums = np.correlate(h, h, mode="full")
    # Only use valid overlapping sums
    # This produces 2n-1 values, but we want to have overlapping regions of valid length
    # For shift s in range -(n-1) to (n-1), length of overlap region is n - |s|
    # We want to substitute p_sums by a computation that includes only valid region products
    # Restore arrays: the first n-1 elements are for negative shifts, last n-1 for positive, middle for zero

    # Since we want to model h_0 * (1-h_{0+s}) + h_1*(1-h_{1+s}) + ...
    # and we use np.correlate with full mode: z[i] = sum_j h[j] * (1-h[i+j]) 
    # but because h is extended by zero, if i+j outside domain, then 1-h[i+j] = 1.
    # Let's compute correct overlaps:
    h_extended = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    product_sums = np.zeros(2*n-1)

    for s_idx in range(2*n-1):
        s = s_idx - (n-1)
        # compute sum of h[i] * (1-h[i+s]) over overlapping region
        # where h is on [0,n-1] and h[i+s] is non-zero for i: i+s in [0,n-1]
        # i.e., valid range: max(0, -s) <= i <= min(n-1, n-1-s) 
        start = max(0, -s)
        end = min(n, n - s)
        if start < end:
            # computes h[i] * (1 - h[i+s]) for i from start to end-1
            # which is sum_i h[i] - sum_i h[i]*h[i+s]
            p = np.sum(h[start:end] * (1 - h_extended[start+s: end+s]))
            # but shape of h_extended[start+s: end+s] : it is h[i+s] with indices
            # from 0 to (end+s) - (start+s) = end-start, which matches h[start:end]
            # So sum is h[i] (1-h[i+s]) for i = start to end-1
        else:
            # no overlap, so it is just zero product.
            p = 0

        # But note that np.correlate already includes both h[i]*h[j] where j=i+s 
        # We're actually implemented replacing h with 1-h to proxy h*(1-h) vs h*(1-h)
        # So direct correlate results hold:
        # Just take h and 1-h.
        # Easier to just work with correct definition of what each term should be:

        # Compute sum h_i * (1 - h_{i+s}) for i such that both in [0,n-1]
        # Which means for shift s, overlapping intervals for i such that 0 <= i <= n-1 and 0 <= i+s <= n-1,
        # i.e., i in [0,n-1] and i in [-s,n-1-s]
        # So i >= max(0,-s) and i <= min(n-1, n-1-s) = min(n-1, n-1-s) (actually min(n-1, n-1-s) = n-1-s if s <= 0, else n-1)
        # For s in [-n+1, n-1], 
        # i in [max(0,-s), min(n-1,n-1-s)] = [max(0,-s), min(n-1,n-1-s)]
        # This out of range when s < -(n-1) or s > n-1. 
        # Coeffs are already computed there, but we should avoid this.

        # Instead, compute sums as sum of (h[i] - h[i]*h[i+s]) over i in valid range.
        # Or naive definition:  For termed shifted index i → i+s, 
        # (i+s) ∈ [0,n-1] but effectively it overlaps with equation:
        # overlap ranges = shifted index in [0,n-1] => advanced range is [i: i+s ∈ [0,n-1]] => i ∈ [-s, n-1-s]
        # => i ∈ [0,n-1] AND [ -s, n-1-s] 
        # i ≥ max(0,-s) and i ≤ min(n-1, n-1-s) = min(n-1, n-1-s)
        # Notice that this logic yields correct for both case s positive and negative.
        # Over coefficients are h_j at i+s.

        # Original approach: let's just drop for now the uncertainty and focus on implementation correctly.
        # It's critical to get the correct interaction.
        # So for i to be valid, i ∈ [0, n-1], and max(0, -s) ≤ i ≤ min(n-1, n-1-s)
        k_start = max(0, -s)
        k_end = min(n-1, n-1-s) + 1
        if k_start < k_end:
            if s > 0:
                product_sums[s_idx] = np.sum(h[k_start:k_end] * (1 - h[k_start+s:k_end+s]))
            else:
                s_abs = -s
                # We have i in [k_start, k_end) with i<=n-1 and 0 <= i+s_abs<=n-1, i.e., i <= n-1-s_abs
                product_sums[s_idx] = np.sum(h[k_start:k_end] * (1 - h[k_start+s_abs:k_end+s_abs]))
        else:
            # No overlapped indices
            product_sums[s_idx] = 0
    
    return product_sums

def projected_gradient_descent(h0, max_iter=5000, step_size=0.1):
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    proj_step = np.zeros_like(h)
    
    for iteration in range(max_iter):
        # Compute all product sums: p_s = sum_i h_i * h_{i+s} over overlapping positions
        # Plus use np.correlate to compute all shift products including out-of-bound terms with 1 value
        # g = np.zeros(n)
        # For all shifts s from -(n-1) to (n-1):
        
        # Naive loop:
        p_sums = np.zeros(2 * n - 1)
        for s in range(1 - n, n):
            start = max(0, -s)
            end = min(n, n - s)
            # compute sum h_i * h_{i+s}: consider valid indices
            if start < end:  # there's overlap
                if s > 0:
                    # sum over k in [start, end), compare h[k] * h[k+s]
                    p_sums[s + (n - 1)] = np.sum(h[start:end] * h[start+s:end+s])
                else:
                    # s <= 0, so sum over k in [start, end) => h[start] to h[end-1], with h[k+s] where k+s in [0,n-1]
                    # shifter = -s
                    # i in [start, end) => k, shifted -> k + s = k - shifter = k - (-s) = k + s
                    # But let's think:
                    # partially correct: h[start:end] and step=1
                    # second term h[k+s] with k in [start, end): i = k+s, range=[start+s, end+s] 
                    # should be valid: range of indices for h[k+s] in [0,n-1].
                    # [start+s, start+s+end-start) = [start+s, end+s)
                    # require 0 <= start+s and end+s <= n-1.
                    # That is: start+s >= 0 => s >= -start and end+s <= n-1 => s <= n-1-end  
                    # 0 <= start and end <= n-1, thus -start <= 0 and n-1-end >= 0
                    # Condition is s >= -start and s <= n-1-end. So if we compute over k in [start, end) 
                    #    we must ensure that for s being the shift,  

                    # Ensure that range [start, end) is such that the h[j] terms are valid as j = k+s.
                    # Ensure 0 <= k+s <= n-1 for all k in [start, end)
                    # So: s >= -start and s <= n-1-end 
                    # That is within bounds [-start, n-1-end]
                    # Our computation works as h[start:end] corresponds to index k such that k ∈ [start, end).
                    # Then we have h[k+s] over k+s ∈ [start+s, end+s).
                    # For valid h values, we need 0 <= start+s and end+s <= n-1
                    # For safeguards:
                    # inner_start = max(0, start + s)
                    # inner_end = min(n, end + s)
                    # The ones where at least one of start + k, end + k intersection of [start, end) with [0, n)
                    # Rather, commit to consistent path:
                    
                    # We are computing sum of h[k] * h[k+s] over k ∈ [start, end) where start to end sweeps valid overlap
                    # new bounds for h_index from the output h[k] * h[k+s]
                    # both must be in [0, n), where index k is valid and index k+s is valid.
                    inner_start = start + s
                    inner_end = end + s
                    # Check they are within [0,n)
                    if inner_start < 0:
                        inner_start = 0
                    if inner_end > n:
                        inner_end = n
                    # Adjust start and end correctly for valid region of k such that both k and k+s are valid
                    valid_start = max(start, -s)  # This should be 0 if s >= 0, or for valid k+s
                    valid_end = min(end, n - s)  # This should be n if s >= 0, or for valid k+s
                    if valid_start < valid_end:
                        p_sums[s + (n - 1)] = np.sum(h[start:end] * h[start+s : start+s+(end-start)])
            else:
                p_sums[s + (n - 1)] = 0
        # Let's compute again using a robust transformation:
        # For any shift, define its valid indices as i in [0,n) with i+s also in [0,n)
        # i in [0,n) and i+s in [0,n) =>
        # 0 <= i <= n-1 and 0 <= i+s <= n-1 =>
        # 0 <= i <= n-1 and -s <= i <= n-1-s
        # i >= max(0, -s) and i <= min(n-1, n-1-s)
        # p_s = sum_i h_i * h_{i+s} where both indices in [0, n), in [0, n)  
        product_sums = np.zeros(2 * n - 1)
        for s_idx in range(2 * n - 1):
            s = s_idx - (n - 1)
            # i valid hinges: 0 <= i <= n-1 and 0 <= i+s <= n-1
            # => i >= 0, i >= -s, and i <= n-1, i <= n-1-s
            # => i in [0, n-1] ∩ [-s,n-1-s]
            # This is [max(0, -s), min(n-1, n-1-s)]
            i_start = max(0, -s)
            i_end = min(n - 1, n - 1 - s) + 1
            if i_start < i_end:
                p = np.sum(h[i_start:i_end] * h[i_start + s:i_end + s])
                product_sums[s_idx] = p
            else:
                product_sums[s_idx] = 0
                
        # Now C(s) = integral h(x) * (1 - h(x + s)) = sum_i h_i * (1 - h_{i+s})
        # We're implementing it via the exact formula:
        # = sum_i h_i - sum_i h_i * h_{i+s} = total - product_sums[s]
        C_values = target_sum - product_sums

        # Step size schedule
        if iteration > 0:
            step_size = max(0.01, step_size * 0.9995)
        
        # Pick worst shift and compute gradient
        worst_shift_idx = np.argmin(C_values)
        s_worst = worst_shift_idx - (n - 1)
        
        # Compute gradient for the worst shift s_worst:
        # gradient wrt h_i of C(s) = sum_i h_i * (1 - h_{i+s_worst}) = total - sum(h_i * h_{i+s})
        # = 1 - h_{i+s_worst} if i and i+s_worst both valid, else 1 due to missing term.
        # So dC(s)/dh_i = 1 - h[i+s] if valid, else 1. This is the replacement of derivative.
        # Let's find all valid contributions.
        # The expression for dC(s)/dh_i = (1 - h_{i+s_worst}) if i+s_worst valid else 1.  
        # So we compute:
        grad = np.full_like(h, 1.0)
        # For range of i where 0 <= i <= n - 1 and 0 <= i + s_worst <= n - 1 
        if s_worst > 0:
            i_start = 0
            i_end = n - s_worst
            grad[i_start:i_end] -= h[i_start + s_worst:i_end + s_worst]
        elif s_worst < 0:
            i_start = -s_worst
            i_end = n 
            grad[i_start:i_end] -= h[i_start + s_worst:i_end + s_worst]
        else:
            # s_worst sign is zero: dC(s)/dh_i = 1 - h_i (for i ∈ [0,n-1]) 
            grad[:] -= h
        
        # Apply step
        h_new = h - step_size * grad
        # Project to feasible region
        # First clip to [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)
        # Adjust sum
        current_sum = np.sum(h_new)
        adjustment = (target_sum - current_sum) / n
        h_new += adjustment
        h_new = np.clip(h_new, 0.0, 1.0) 
        # Ensure sum < target_sum, which is not sharper
        h_new = h_new + (target_sum - np.sum(h_new)) / n
        
        h = h_new.copy()
        # Evaluate
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return best_h, best_c5
