import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.special import softmax

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_to_feasible(h, target_sum):
    """Project h to the feasible set: 0 <= h <= 1, sum(h) = target_sum"""
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to match target sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    delta = (target_sum - current_sum) / len(h)
    h += delta
    h = np.clip(h, 0, 1)
    return h

def smooth_max(h, beta=10.0):
    """Smooth approximation of max using softmax"""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.log(np.sum(np.exp(beta * corr))) / beta

def smooth_max_gradient(h, beta=10.0):
    """Gradient of smooth max with respect to h"""
    corr = np.correlate(h, 1 - h, mode="full")
    # softmax of corr weights
    weights = softmax(beta * corr)
    # Gradient of mean of corr w.r.t h[0], it is the first half of the convolution of (1-h) with the flips of weights.
    # Let's calculate densely
    n = len(h)
    # Compute gradient of corr[i] = sum(h[k] * (1-h[i+k])) = sum(h[k] * (1-h[l])) for l = i+k
    # contribution to gradient w.r.t h[k] of corr[i] is (1 - h[i+k]) when k <= i, and
    # (h[i + k - n] - 1) when k > i
    # But better: use np.correlate's full mode to obtain the large correlation matrix
    # To compute ∇h_i (sum w_j * corr_j) = sum w_j * ∂corr_j/∂h_i
    # The derivative ∂corr_j/∂h_i = 2*delta_{j,i} + 2*delta_{j,i+n} + 2*delta_{j,i-n} etc.
    # But more directly use the fact that ∂corr[j] = (1-h) * delta_{j} applied from (1-h)
    # and convolve is transpose of flip->conv_then_flip
    # Still use known formula for smooth max
    # W = softmax(beta * corr)
    gradient_of_corr = np.zeros_like(h)
    for i in range(len(h)):
        # gradient of corr[j] w.r.t h[i]
        # correlated with 1-h (the output, so contributions from (1-h) matching to h[i])
        # corr[0] = h[0] * (1-h[0]) + h[1] * (1-h[1]) ... etc.
        # It's a matrix computation in the full walk.

        # Simpler approach: numerical gradient by finite differences
        EPS = 1e-6
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += EPS
        h_minus[i] -= EPS
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
        corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full")
        gradient_of_corr[i] = (np.sum(weights * corr_plus) - np.sum(weights * corr_minus)) / (2 * EPS)
    return gradient_of_corr

def proximal_gradient_descent(h_start, target_sum, max_iter=10000, step_size=1e-3, beta=10.0, tol=1e-6):
    h = h_start.copy()
    current_value = smooth_max(h, beta)
    for i in range(max_iter):
        # gradient of smooth max
        grad = smooth_max_gradient(h, beta)
        # update with step size
        h_new = h - step_size * grad
        # project
        h_new = project_to_feasible(h_new, target_sum)
        # evaluate smooth max
        new_value = smooth_max(h_new, beta)
        # adapt step size
        if abs(new_value - current_value) < tol:
            break
        h = h_new
        current_value = new_value
        if i % 1000 == 0:
            beta *= 1.02  # Increase β gradually for sharpening approx to max
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Define and solve the optimization problem using cvxpy to get bounds
    h_vars = cp.Variable(n_points)
    # Variables are h(x) for x in [0, 2]

    # Define constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == target_sum,
    ]

    # The objective is the max product correlation, so we compute one that sample per
    # For this we will compute a key smoothed approximation.
    # Start with the existing construction from initial_h_values
    init_h = np.array(initial_h_values)
    best_h = init_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    start_time = time.time()
    elapsed = 0
    restarts = 0
    max_restarts = 100  # Stop after a few restarts

    while elapsed < budget_s - 5 and restarts < max_restarts:
        # Generate multiple starts
        inits = generate_initial_h(n_points)
        for i, h0 in enumerate(inits):
            if elapsed >= budget_s - 5:
                break  

            h_fin = proximal_gradient_descent(h0, target_sum)
            h_fin = project_to_feasible(h_fin, target_sum)
            c5 = evaluate_erdos_solution(h_fin)
            if c5 < best_c5:
                best_h = h_fin.copy()
                best_c5 = c5

        elapsed = time.time() - start_time
        restarts += 1

    # Final local search refinement using exchanges
    final_h = best_h.copy()
    current_c5 = best_c5
    max_moves = 10000
    moves = 0
    # Simple localized exchange with small deltas
    def random_exchange():
        nonlocal final_h, current_c5
        if moves >= max_moves:
            return
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        delta = np.random.uniform(0.001, 0.01)
        if final_h[i] - delta < 0 or final_h[j] + delta > 1:
            return
        new_h = final_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_h = project_to_feasible(new_h, target_sum)
        new_c5 = evaluate_erdos_solution(new_h)
        if new_c5 < current_c5:
            final_h = new_h
            current_c5 = new_c5
        moves += 1

    while elapsed < budget_s - 5:
        random_exchange()
        elapsed = time.time() - start_time

    return (final_h.tolist(), current_c5, len(final_h))
