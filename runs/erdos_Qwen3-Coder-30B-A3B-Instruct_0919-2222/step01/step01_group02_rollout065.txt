def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def project_sum_constraint(h, target_sum):
    """Project h onto the constraint ∑ h = target_sum"""
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    factor = target_sum / current_sum
    h_projected = h * factor
    return h_projected
