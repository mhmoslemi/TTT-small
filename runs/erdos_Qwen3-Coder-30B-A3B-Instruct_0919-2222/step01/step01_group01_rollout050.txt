def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def compute_product_sums(h):
    # Compute sum_i h[i] * h[i+s] for all shifts s
    n = len(h)
    prod_sums = np.correlate(h, h, mode='full')
    return prod_sums

def compute_C_sums(h):
    # Compute C(s) = sum(h) - product_sum(s) 
    n = len(h)
    sum_h = n / 2.0
    prod_sums = compute_product_sums(h)
    # shift offset is from -(n-1) to (n-1) -> indices 0 to 2*(n-1)
    C_sums = sum_h - prod_sums
    return C_sums, prod_sums

def project_to_simplex(h, target_sum):
    # Project h to the simplex with sum = target_sum and clipping to [0,1]
    # Uses binary search on lambda to find projection parameter
    n = len(h)
    low = np.min(h) - 1.0
    high = np.max(h)
    
    def proj_sum(lam):
        proj = np.clip(h - lam, 0, 1)
        return np.sum(proj)
    
    for _ in range(50):
        mid = (low + high) / 2.0
        s = proj_sum(mid)
        if s > target_sum:
            low = mid
        else:
            high = mid
    proj = np.clip(h - low, 0, 1)
    return proj

        # Compute gradient for max_C_index shift (we want to decrease this)
        grad = np.zeros(n)

        grad += 1.0  # For terms not in product part of C(s)
        
        # Subtract weighted product terms specifically
        shift = max_C_index - (n - 1)  # Convert index to actual shift
        
        # Contribution from products: subtract weight * h[i+shift] for overlap
        if shift >= 0:
            # overlapping indices i such that i+shift in [0,n)

            end_idx = n - shift
            # over range [start_idx, end_idx)
            indices = slice(start_idx, end_idx)
            # contribute - weight * h[i+shift] to grad[i] 
            grad[indices] -= weights[max_C_index] * h[start_idx + shift:end_idx + shift]
        else:
            # shift < 0: indices i such that i + shift in [0,n)
            # i in [0, n + shift), and we want to do i+shift = 0..n-1 
            # => i in [-shift, n-shift] => 0 <= i <= n-shift
            # , i in [0, -shift) does not affer
            # i in [min(n-shift, n), n) = [n-shift, n) :
            # j = i + shift (i in [0, n+shift)) should be in [0, n)
            # i in [0, n-shift) => i+shift in [shift, n-shift)
            # h indices that contribute:
            # from range [0, n-shift) to [0 + shift = shift, (n-shift) + shift = n )]
            # But need that: i in such as 0 <= i < n and 0 <= i+shift <= n-1
            start_idx = 0
            end_idx = min(n, n + shift)  # cannot go below 0, up to n-shift
            indices = slice(start_idx, end_idx)
            # there is a shift to contribution
            grad[indices] -= weights[max_C_index] * h[start_idx + shift:end_idx + shift] 
        
        # perform step and project
        h_new = h - alpha * grad
        h_new = project_to_simplex(h_new, target_sum)
        h = h_new
        
    return best_h, best_c5
