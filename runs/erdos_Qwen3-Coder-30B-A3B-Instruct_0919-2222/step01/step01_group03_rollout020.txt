import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector v onto bounds [0,1] and sum constraint"""
    # First clip to [0,1]
    v = np.clip(v, 0, 1)
    
    # Adjust to meet sum constraint
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Use water-filling approach
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return v
    
    # For small differences, adjust proportionally
    if abs(diff) < 1e-8:
        return v
    
    # Adjust by subtracting mean to maintain sum
    mean_adjustment = diff / len(v)
    v = v - mean_adjustment
    
    # Clamp to [0,1] and adjust again if necessary
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    diff = target_sum - current_sum
    
    # If still have small difference, redistribute
    if abs(diff) > 1e-10:
        # Find positions that can be adjusted
        if diff > 0:  # Need to increase
            indices = np.argsort(v)
            for i in reversed(indices):  # Start from smallest
                if diff <= 0:
                    break
                available = 1.0 - v[i]
                to_add = min(available, diff)
                v[i] += to_add
                diff -= to_add
        else:  # Need to decrease
            indices = np.argsort(v)
            for i in indices:  # Start from smallest
                if diff >= 0:
                    break
                available = v[i]
                to_subtract = min(available, -diff)
                v[i] -= to_subtract
                diff += to_subtract
    
    return np.clip(v, 0, 1)

    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)

    best_h = project_onto_bounds_and_sum(h0.copy(), target_sum)
    best_c5 = objective(best_h)

    from scipy.optimize import minimize
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}

    try:

            h_opt = np.clip(result.x, 0, 1)
