import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, S, lower=0.0, upper=1.0):
    """Project vector v onto the feasible region with sum=S and bounds [lower, upper]"""
    # First clip to bounds
    v = np.clip(v, lower, upper)
    # Adjust to meet sum constraint
    current_sum = np.sum(v)
    if abs(current_sum - S) < 1e-10:
        return v
    diff = S - current_sum
    if abs(diff) < 1e-10:
        return v
    # Simple adjustment: distribute difference evenly
    v += diff / len(v)
    # Clip again to maintain bounds
    v = np.clip(v, lower, upper)
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    from scipy.optimize import minimize
    
    # First do a global optimization with differential evolution
    try:
        de_result = differential_evolution(
            lambda h: evaluate_erdos_solution(h),
            bounds,
            maxiter=50,
            popsize=10,
            seed=seed,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = evaluate_erdos_solution(h_opt)
            h0 = h_opt
    except:
        pass
    
    # Try improved optimization with fewer iterations
    max_iter = 1000
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Gradient descent with projection for fine tuning
    # We'll implement a simple projected gradient descent
        
    # Precompute shift index pairs for efficiency
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points - s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))
    
    # Optimized gradient descent
    h_current = h0.copy()
    eta = 0.1
    
    # Function to compute gradient
    def compute_gradient(h):
        # Compute correlation without dx (raw values)
        corr = np.correlate(h, 1 - h, mode="full")
        # Weight matrix for soft max of correlation (to approximate max)
        max_corr = np.max(corr)
        exp_terms = np.exp(20 * (corr - max_corr))
        weights = exp_terms / np.sum(exp_terms)
        
        # Compute gradient using weight-scaled contributions
        grad = np.zeros_like(h)
        
        for s, (i, j) in enumerate(shift_indices):
            weight = weights[s]
            if weight == 0:
                continue
            # For shift s, derivative contributions:
            # - On index i: +weight *(1 - h[j])
            # - On index j: +weight *(-h[i])
            grad[i] += weight * (1 - h[j])
            grad[j] += weight * (-h[i])
        
        return grad

    # Run optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        gradient = compute_gradient(h_current)
        
        # Take gradient step
        h_new = h_current - eta * gradient
        
        # Project onto feasible set
        h_new = project_onto_bounds_and_sum(h_new, n_points / 2)
        
        # Evaluate C5
        c5_new = evaluate_erdos_solution(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_current = h_new.copy()
        eta *= 0.999  # Decay learning rate
        
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
