def full_corr(h):
    """Compute full correlation of h with (1-h) using FFT."""
    n = len(h)
    H = np.fft.rfft(h, n=2*n-1)
    G = np.fft.rfft(1-h, n=2*n-1)
    corr = np.fft.irfft(H * np.conj(G), n=2*n-1)
    return corr.real

def proj_capped_simplex(v, target_sum):
    """Project vector v onto capped simplex: sum = target_sum, 0 <= v[i] <= 1."""
    v = np.array(v, dtype=float)
    # Binary search for lambda
    l, r = -1.0, 1.0
    tol = 1e-12
    max_iter = 100

    def f(lam):
        proj = np.clip(v + lam, 0, 1)
        return np.sum(proj) - target_sum

    for _ in range(max_iter):
        mid = (l + r) / 2
        val = f(mid)
        if abs(val) < tol:
            lambda_opt = mid
            break
        if val > 0:
            r = mid
        else:
            l = mid
    else:
        lambda_opt = (l + r) / 2
    
    proj = np.clip(v + lambda_opt, 0, 1)
    return proj

def evaluate_h(h_values):
    """Original evaluate_h with dx scaling."""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    return max_val * dx
