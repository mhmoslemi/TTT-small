import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Constraint dictionary
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try to improve by using a different approach:
    # 1. Direct optimization using L-BFGS-B with random restarts
    # 2. Hybrid with geometric or pattern-based initialization
    
    def geometric_initialization():
        """Create an initial function with a geometric decay"""
        h = np.zeros(n_points)
        # Create a triangular shape centered in the interval
        mid = n_points // 2
        width = n_points // 3
        for i in range(n_points):
            if abs(i - mid) < width:
                h[i] = 1.0 - abs(i - mid) / width
        # Scale to integrate to 1
        h = h / (np.sum(h) * dx)
        return np.clip(h, 0, 1)
    
    def peak_initialization():
        """Create initial function with peaks"""
        h = np.zeros(n_points)
        # Create a few peaks
        peaks = [(n_points//4, 0.8), (n_points//2, 1.2), (3*n_points//4, 0.8)]
        for i in range(n_points):
            for (pos, height) in peaks:
                dist = abs(i - pos)
                # Gaussian-like decay
                h[i] += height * np.exp(-0.5 * (dist / (n_points/10))**2)
        # Scale to integrate to 1
        h = h / (np.sum(h) * dx)
        return np.clip(h, 0, 1)
    
    # Try geometric initialization
    try:
        h_geo = geometric_initialization()
        c5_geo = objective(h_geo)
        if c5_geo < best_c5:
            best_c5 = c5_geo
            best_h = h_geo.copy()
    except:
        pass
    
    # Try peak initialization
    try:
        h_peak = peak_initialization()  
        c5_peak = objective(h_peak)
        if c5_peak < best_c5:
            best_c5 = c5_peak
            best_h = h_peak.copy()
    except:
        pass
    
    # Try random search from multiple initializations
    n_random_restarts = 15
    for i in range(n_random_restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random initialization with some constraints 
        h_rand = np.random.rand(n_points) * 0.8 + 0.1  # Between 0.1 and 0.9
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        # Local optimization starting from this random point
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Try if using population-based algorithm gives better results
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=30,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-9,
            rtol=1e-9
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Final refinement with local optimization from the best solution
    try:
        local_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
