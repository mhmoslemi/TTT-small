def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_to_simplex(v, s=1.0):
    """Project vector v onto the simplex with sum = s using water-filling"""
    # Sort in descending order
    n = len(v)
    u = np.sort(v)[::-1]
    rho = 0
    for i in range(n):
        if u[i] - (np.sum(u[:i+1]) - s) / (i+1) > 0:
            rho = i
        else:
            break
    theta = (np.sum(u[:rho+1]) - s) / (rho+1)
    return np.maximum(v - theta, 0)

def smooth_max_gradient(h, beta=10.0):
    """Compute gradient of smoothed max using softplus"""
    dx = 2.0 / len(h)

    corr = np.correlate(h, 1 - h, mode="full")
    # Convert to probabilities with softmax
    exp_corr = np.exp(beta * corr)
    sum_exp = np.sum(exp_corr)
    prob_corr = exp_corr / sum_exp
    
    # Gradient of smoothed max w.r.t h
    grad = np.zeros_like(h)
    for i in range(len(h)):
        temp = np.zeros_like(h)
        for k in range(len(h)):
            if i - k >= 0:
                temp[i-k] = (1 - h[k]) * prob_corr[k + len(h) - 1]
            else:
                temp[i-k + len(h)] = (1 - h[k]) * prob_corr[k + len(h) - 1]
        grad[i] = np.sum(temp) * dx
    
    return grad

    h_new = h - step_size * grad
    h_new = project_to_simplex(h_new, 1.0)
    h_new = np.clip(h_new, 0, 1)
