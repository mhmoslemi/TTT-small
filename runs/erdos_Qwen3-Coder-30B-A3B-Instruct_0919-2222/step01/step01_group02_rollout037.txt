import numpy as np
import time
import cvxpy as cp
from scipy.optimize import differential_evolution
from cvxpy import SCS

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def solve_sdp_relaxation(n_points):
    """Solve SDP relaxation to find optimal a vector"""
    # Variables
    a = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    
    # Constraints
    constraints = [
        cp.diag(X) == cp.square(a),  # X[i,i] = a[i]^2
        cp.sum(a) == 0,  # Sum constraint
        X >> 0,  # X is positive semidefinite
        cp.abs(a) <= 1  # Bounds on a
    ]
    
    # Objective: minimize max correlation
    # We will form the correlation matrix terms later
    # For now, we just solve the SDP.
    # We'll gradually add constraints during refinement
    
    # Note: The desired objective in SDP may be complex, 
    # so we compute it empirically after rounding
    problem = cp.Problem(cp.Minimize(0), constraints)
    
    # Solution attempt
    try:
        problem.solve(solver=SCS, verbose=False, max_iters=5000, eps=1e-5)
        return a.value, X.value
    except:
        return None, None

def round_sdp_solution(a_sol, X_sol):
    """Round SDP solution to discrete form"""
    if a_sol is None:
        return None
    
    # Simple rounding: Use Gaussian projection
    # Project X = a * a^T to get a feasible vector
    try:
        # Compute approximate a from X
        # Get Cholesky factor L of X, then generate g ~ N(0, I)
        # Set a = L * g, project to [-1,1] and scale to sum=0
        
        # Use a Cholesky-like decomposition for direct recovery
        # This is a tweak - we'll still compute SVD or use approximate eigen
        eigenvals, eigenvecs = np.linalg.eigh(X_sol)
        # For numerical stability
        eigenvals = np.maximum(eigenvals, 0)
        
        L = eigenvecs @ np.diag(np.sqrt(eigenvals))
        
        # Sample a vector, project
        g = np.random.randn(len(a_sol))
        a_rounded = L @ g
        a_rounded = np.clip(a_rounded, -1, 1)
        
        # Rescale to exact zero sum
        mean_a = np.mean(a_rounded)
        a_rounded = a_rounded - mean_a
        
        # Scale to match target sum
        a_rounded = a_rounded * np.sqrt(n_points) / np.linalg.norm(a_rounded)
        
        # Refine to correct sum
        current_sum = np.sum(a_rounded)
        a_rounded = a_rounded - current_sum/n_points
        
        return a_rounded
    except Exception as e:
        # Fallback to a simple smoothing
        return np.clip(a_sol, -1, 1)  

def project_to_box(a, a_min=-1, a_max=1):
    """Project a vector onto box [a_min, a_max]^n"""
    return np.clip(a, a_min, a_max)

def project_to_sum_zero(a):
    """Project a vector onto sum-zero constraint"""
    avg = np.mean(a)
    return a - avg

def smoothed_max_correlation(h, epsilon=1e-5):
    """Smoothed version of maximum correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all lags
    correlations = np.correlate(h, 1-h, mode='full') * dx
    # Remove zero-lag
    correlations = correlations[n-1:]  # Only lags [0, n-1] 
    # Softmax approximation to max
    return -np.log(np.mean(np.exp(-correlations/epsilon))) * epsilon

def parametric_h_function(x, *params):
    """Parametric form of h(x) with 5 parameters for 3 segments"""
    # x in [0, 2]
    a, b, c, s, t = params
    # make sure segments are valid
    s = np.clip(s, 0.1, t-0.1)
    t = np.clip(t, s+0.1, 1.9)
    
    if x < s:
        h_val = a
    elif x < t:
        h_val = b
    else:
        h_val = c
    return np.clip(h_val, 0, 1)

def compute_h_from_parametric(*params):
    """Compute step function h from piecewise constant parameters"""
    n_points = 70
    dx = 2.0 / n_points
    x_vals = np.arange(0, 2, dx)
    h_vals = np.array([parametric_h_function(x, *params) for x in x_vals])
    # Normalize sum to exactly 1 
    h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    return h_vals

def objective_parametric(params):
    """Objective function to minimize for parametric approach"""
    try:
        h = compute_h_from_parametric(*params)
    except Exception:
        return 1e10  # Penalty for invalid params
    
    c5_value = evaluate_erdos_solution(h)
    return c5_value

def refine_with_local_search(h0, iterations=100):
    """Use gradient descent to slightly improve h0"""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    
    # Try small perturbations around center
    for iter in range(iterations):
        # Compute gradients wrt to each value of h
        currents = evaluate_erdos_solution(h)
        h_plus = h.copy()
        h_min = h.copy()
        eps = 1e-4
        
        best_h = h
        best_val = currents
        
        # Sample small changes
        for i in range(n):
            # Positive step
            h_plus[i] = min(1.0, h[i] + eps)
            val_plus = evaluate_erdos_solution(h_plus)
            if val_plus < best_val:
                best_val = val_plus
                best_h = h_plus.copy()
            
            # Negative step
            h_min[i] = max(0.0, h[i] - eps)
            val_min = evaluate_erdos_solution(h_min)
            if val_min < best_val:
                best_val = val_min
                best_h = h_min.copy()
            
            h_plus[i] = h[i]
            h_min[i] = h[i]
                
        h = best_h
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Set np seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()
    result_time = start_time + budget_s
    
    n_points = 70
    
    # Try parametric optimization with differential evolution initially
    bounds = [(0, 1), (0, 1), (0, 1), (0.1, 1.9), (0.1, 1.9)]
    
    # Differential evolution time budget
    de_time_left = (result_time - time.time())
    
    try:
        # Use DE to find reasonable parameters for 3-segment h
        res = differential_evolution(
            objective_parametric, 
            bounds, 
            maxiter=500, 
            popsize=15, 
            seed=seed,
            timeout=min(300, de_time_left)
        )
        if res.success:
            h_start = compute_h_from_parametric(*res.x)
        else:
            # Fallback
            h_start = np.ones(n_points) * 0.5
            h_start *= (n_points / 2) / np.sum(h_start)
    except Exception:
        h_start = np.ones(n_points) * 0.5
        h_start *= (n_points / 2) / np.sum(h_start)
        
    # Refine using local search
    h_final = refine_with_local_search(h_start, 50)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_final)
    best_h = h_final.tolist()
    best_c5 = final_c5
    n_points = len(h_final)
    
    return (best_h, best_c5, n_points)
