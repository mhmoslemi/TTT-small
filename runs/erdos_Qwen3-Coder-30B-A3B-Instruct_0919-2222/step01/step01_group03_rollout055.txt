import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()

    # Try a larger range of optimization approaches
    
    # 1. Differential Evolution with more iterations
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=20,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-8,
            rtol=1e-8
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # 2. Local optimization with different methods
    methods = ['SLSQP', 'L-BFGS-B', 'TNC']
    for method in methods:
        if time.time() - start_time > budget_s * 0.9:
            break
        try:
            local_result = minimize(
                objective,
                best_h,
                method=method,
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 3000, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # 3. Random restarts with more diverse initializations
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Strategy: Create a more structured initial guess that tends to have lower overlap
        h_rand = np.random.rand(n_points)
        # Increase probability of small values to create stepped profile
        h_rand = np.where(h_rand > 0.7, 0, h_rand)
        h_rand = np.where(h_rand > 0.3, 1.0, h_rand)
        # Scale to meet integral constraint
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # 4. One more heuristic approach: use previous best and perturb carefully
    if time.time() - start_time < budget_s * 0.9:
        # Try another kind of optimization with reduced iterations for more restarts
        try:
            # Adaptive restart with perturbations
            h_perturb = best_h.copy()
            # Apply small random perturbations to already good solution
            noise = np.random.normal(0, 0.05, n_points)
            h_perturb += noise
            h_perturb = np.clip(h_perturb, 0, 1)
            # Renormalize
            h_perturb = h_perturb / (np.sum(h_perturb) * dx) * 1.0
            
            local_result = minimize(
                objective,
                h_perturb,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
