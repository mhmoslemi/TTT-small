```python
import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points, seg_len):
    return np.repeat(heights, seg_len)

def evaluate_heights(heights, n_points, seg_len, penalty_factor=1e3):
    h = build_h(heights, n_points, seg_len)
    c5 = evaluate_h(h)
    sum_heights = np.sum(heights)
    penalty = penalty_factor * (sum_heights - len(heights)/2)**2
    return c5 + penalty

def project_to_bounds_and_sum(heights, target_sum):
    # Simple projection onto [0,1] and sum constraint
    heights = np.array(heights)
    heights = np.clip(heights, 0, 1)
    current_sum = np.sum(heights)
    if abs(current_sum - target_sum) < 1e-8:
        return heights
    diff = target_sum - current_sum
    # Distribute difference proportionally to current values
    adjustment = diff / len(heights)
    heights += adjustment
    # Clip to [0,1] and renormalize
    heights = np.clip(heights, 0, 1)
    current_sum = np.sum(heights)
    if abs(current_sum - target_sum) > 1e-8:
        scale = target_sum / current_sum
        heights *= scale
    return heights

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 140  # We'll use 140 points
    S = 14  # Number of steps (should be even for symmetry)
    seg_len = n_points // S  # Each step has same number of points
    target_sum = S / 2  # For ensuring sum of heights = S/2
    
    # Initial random solution
    initial_heights = np.random.rand(S)
    initial_heights = initial_heights * 2  # Scale to [0,2] initially
    initial_heights = np.clip(initial_heights, 0, 1)
    initial_heights = project_to_bounds_and_sum(initial_heights, target_sum)
    
    bounds = [(0, 1) for _ in range(S)]
    
    def objective(heights):
        return evaluate_heights(heights, n_points, seg_len)
    
    best_heights = initial_heights.copy()
    best_c5 = evaluate_heights(best_heights, n_points, seg_len)
    best_h = build_h(best_heights, n_points, seg_len)
    best_h_projected = project_to_bounds_and_sum(best_heights, target_sum)
    best_c5_projected = evaluate_h(best_h_projected)
    
    de = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=200,
        popsize=10,
        disp=False
    )
    
    if not de.success:
        de = differential_evolution(
            objective,
            bounds,
            seed=seed+1,
            maxiter=200,
            popsize=10,
            disp=False
        )
    
    final_heights = de.x
    final_h = build_h(final_heights, n_points, seg_len)
    
    # Final projection to sum constraint
    final_heights_projected = project_to_bounds_and_sum(final_heights, target_sum)
    
    # Evaluate final solutions
    c5_nonprojected = evaluate_heights(final_heights, n_points, seg_len)
    c5_projected = evaluate_h(build_h(final_heights_projected, n_points, seg_len))
    
    # Choose the best solution so far
    if c5_projected < best_c5_projected:
        final_heights = final_heights_projected
        best_c5 = c5_projected
    else:
        best_c5 = best_c5_projected
        
    # Refine with SLSQP if time allows
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        
        h_refined = build_h(final_heights, n_points, seg_len)
        bounds_refined = [(0, 1) for _ in range(n_points)]
        
        def objective_refined(h):
            return evaluate_h(h)
            
        def constraint_refined(h):
            return np.sum(h) * (2.0 / n_points) - 1.0 
        
        cons = {'type': 'eq', 'fun': constraint_refined}
        
        try:
            res = minimize(objective_refined, h_refined, bounds=bounds_refined, constraints=cons, 
                           method='SLSQP', options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
                          )
            if res.success:
                h_slsqp = res.x
                c5_slsqp = evaluate_h(h_slsqp)
                if c5_slsqp < best_c5:
                    best_h = h_slsqp
                    best_c5 = c5_slsqp
        except Exception:
            pass

    final_h = build_h(final_heights, n_points, seg_len)
    final_h = project_to_bounds_and_sum(final_heights, target_sum)
    final_c5 = evaluate_h(final_h)
    
    # Final refinement with SLSQP on projected solution
    if time.time() - start_time < budget_s - 5:
        bounds_refined = [(0, 1) for _ in range(n_points)]
        
        def objective_refined(h):
            return evaluate_h(h)
            
        def constraint_refined(h):
            return np.sum(h) * (2.0 / n_points) - 1.0 
        
        cons = {'type': 'eq', 'fun': constraint_refined}
        
        try:
            res = minimize(objective_refined, final_h, bounds=bounds_refined, constraints=cons, 
                           method='SLSQP', options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6}
                          )
            if res.success:
                final_h = res.x
                final_c5 = evaluate_h(final_h)
        except Exception:
            pass

    return (final_h, final_c5, n_points)
```
