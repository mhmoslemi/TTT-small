import numpy as np
from scipy.optimize import minimize
from math import sqrt
import cvxpy as cp
import time
from copy import deepcopy

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def project_simplex(h, target_sum):
    """Project h to the simplex with given target_sum."""
    # Shift to make sum = 0
    mean_val = np.mean(h)
    h_shifted = h - mean_val
    h_proj = h_shifted + target_sum / len(h_shifted)
    # Clamp to [0, 1]
    h_proj = np.clip(h_proj, 0, 1)
    # Adjust to exact sum by averaging
    current_sum = np.sum(h_proj)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        indices = np.argsort(h_proj)
        # Adjust from smallest to largest to maintain order
        to_add = diff / len(indices)
        for i in indices:
            if abs(diff) < 1e-10:
                break
            max_add = 1 - h_proj[i]
            actual_add = min(to_add, max_add, diff)
            h_proj[i] += actual_add
            diff -= actual_add
    return h_proj

def lp_init(n):
    """Initialize via linear programming relaxation using McCormick envelopes."""
    dx = 2.0 / n
    h = cp.Variable(n)
    u = cp.Variable((n, 2*n - 1))  # positive index: 0 to 2n-2
    t = cp.Variable(1)

    # Build constraints for all valid i, d pairs
    constraints = [
        cp.sum(h) == n/2,  # integral constraint
        0 <= h, h <= 1,    # bounds
        t >= 0              # t non-negative
    ]

    # Add McCormick constraints for u[i,d] = h[i] * h[i+d]
    for i in range(n):
        for d in range(-(n-1), n):
            if 0 <= i+d < n:
                d_idx = d + (n - 1)  # offset to make index valid 0 to 2n-2
                # u[i,d] >= 0
                constraints.append(u[i, d_idx] >= 0)
                # u[i,d] <= h[i]
                constraints.append(u[i, d_idx] <= h[i])
                # u[i,d] <= h[i+d]
                constraints.append(u[i, d_idx] <= h[i+d])
                # u[i,d] >= h[i] + h[i+d] - 1
                constraints.append(u[i, d_idx] >= h[i] + h[i+d] - 1)

    # Compute s_d = sum of u[i,d] over all valid i for each d
    for d in range(-(n-1), n):
        if d == 0:
            continue
        # index offset
        d_idx = d + (n - 1)
        # Sum over valid indices
        s_d = cp.sum(u[range(n), d_idx])
        # t >= n/2 - s_d
        constraints.append(t >= n/2 - s_d)

    # Minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return np.array(h.value).flatten()
    except Exception:
        pass
    # fallback to uniform
    return np.full(n, 0.5)

def binary_round(hf, n):
    """Round continuous h to a binary vector with correct sum"""
    # Select n//2 largest components to make 1, rest 0
    idx = np.argsort(-hf)
    h_binary = np.zeros(n)
    h_binary[idx[:n//2]] = 1
    return h_binary

def worst_shift_d(h):
    """Find the d that causes maximum overlap"""
    corr_h = np.correlate(h, 1-h, mode='full')
    # TODO: Should really return the multiple shifts if there's a tie
    return np.argmax(corr_h)

def local_swap_improve(h, n):
    """Try swapping 0-1 pairs to improve worst-shift"""
    d = worst_shift_d(h)
    round_d = -(n-1) + d
    # Our definition: h(x') = move direction (d>0 means shift right, d<0 left)
    corr_h = np.correlate(h, 1-h, mode='full')
    d_idx = np.argmax(corr_h)

    # Scan for a beneficial swap
    for i in range(n):
        # see if h has 1 at i and 0 at offset (i + round_d)
        if h[i] == 1 and i + round_d >= 0 and i + round_d < n:
            if h[i + round_d] == 0:
                # Look from the other side for h[j] == 0 and h[j + round_d] == 1
                j = i + round_d
                if j >= 0 and j < n:
                    if h[j] == 0:
                        # find any position in the future where h[j+d]==1
                        j2 = None
                        for k in range(n):
                            if h[k] == 1 and k + round_d == j:
                                j2 = k
                                break
                        if j2 is not None:
                            # Swap to improve worst shift
                            h[i], h[j2] = h[j2], h[i]
                            break  # early stop, we improved
    return h

def sa_step(h, t, delta_max, n):
    """Single step of simulated annealing by transferring mass"""
    # pick i, j
    i, j = np.random.choice(n, size=2, replace=False)
    # limit the move size
    delta = np.random.uniform(0, delta_max)
    delta = min(delta, h[i], 1 - h[j])
    if delta <= 0:
        return h
    # do transfer
    h[i] -= delta
    h[j] += delta
    # clip
    h[i] = max(0, min(1, h[i]))
    h[j] = max(0, min(1, h[j]))
    return h

def gradient_descent_step(h, lr=0.05):
    """Perform a projected gradient descent step with subgradient."""
    corr_h = np.correlate(h, 1-h, mode='full')
    d_idx = np.argmax(corr_h)
    d = d_idx - (len(h) - 1)
    grad = np.zeros_like(h)
    # Fill gradient based on worst shift d
    for i in range(len(h)):
        # If i+d falls in range
        if 0 <= i+d < len(h):
            grad[i] -= h[i+d]
        # If i-d falls in range
        if 0 <= i-d < len(h):
            grad[i] -= h[i-d]
    # Apply gradient step
    h = h - lr * grad
    # Project back to correct simplex
    h = project_simplex(h, len(h) / 2.)
    return h

        c5_current = evaluate_erdos_solution(hlp)

        # Round
        h_binary = binary_round(hlp, n)
        for _ in range(1000):  # rate limited swap iterations
            old_h = h_binary.copy()
            h_binary = local_swap_improve(h_binary, n)
            new_c5 = evaluate_erdos_solution(h_binary)
            if new_c5 < c5_current:
                c5_current = new_c5
                if new_c5 < best_c5:
                    best_h = h_binary.copy()
                    best_c5 = new_c5
                    best_n = n
            if time.time() - start_time > budget_s * 0.95:
                break
            if np.array_equal(old_h, h_binary):
                break

        # Simulated Annealing
        h_sa = h_binary.copy()
        t = 0.05 * c5_current
        max_steps = 2000
        delta_max = 0.2
        sa_steps = 0
        while sa_steps < max_steps and time.time() - start_time < budget_s * 0.95:
            h_old = h_sa.copy()
            h_sa = sa_step(h_sa, t, delta_max, n)
            new_c5 = evaluate_erdos_solution(h_sa)
            if new_c5 < c5_current:
                c5_current = new_c5
                if new_c5 < best_c5:
                    best_h = h_sa.copy()
                    best_c5 = new_c5
                    best_n = n
            # Metropolis acceptance
            if new_c5 > c5_current:
                delta = new_c5 - c5_current
                if np.random.rand() > np.exp(-delta / t):
                    # revert
                    h_sa = h_old
            t *= 0.995
            delta_max *= 0.999
            sa_steps += 1
            if t < 1e-8:

                    best_c5 = new_c5
                    best_n = n
