def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_simplex(v, s=1.0):
    """Project vector v onto the simplex with sum = s using water-filling."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(u) + 1)
    rho = np.argmax(u * ind > cssv) if np.any(u * ind > cssv) else len(u) - 1
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0)

def smooth_max_gradient(h, beta=100.0):
    """Compute gradient of smooth max approximation with temperature beta."""
    dx = 2.0 / len(h)
    h_vec = np.array(h)
    corr_full = np.correlate(h_vec, 1 - h_vec, mode='full')
    k_max = len(corr_full) // 2
    # For each k value (from -k_max to k_max)
    # Gradient of smooth max over correlations
    # Use softmax over the full correlation vector
    corr_full = corr_full * dx
    max_corr = np.max(corr_full)
    exp_corr = np.exp(beta * (corr_full - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    # The gradient w.r.t. h of the smooth max
    G = np.zeros_like(h_vec)
    for i in range(len(h_vec)):
        for k in range(len(corr_full)):
            if k < len(h_vec):  # h[i] appears in corr[k]
                delta = int(k - len(h_vec) + 1)  # corresponds to k in range [-k_max, k_max]
                G[i] += weights[k] * (1 - h_vec[i + delta] if i + delta < len(h_vec) else 0) * dx

    return G

def gradient_descent_step(h, lr=0.01, beta=100.0):
    """One step of gradient descent with smoothing."""
    grad = smooth_max_gradient(h, beta)
    h_new = h - lr * grad
    # Project back to [0,1] and to sum=1
    h_new = np.clip(h_new, 0, 1)
    h_new = project_simplex(h_new, 1.0)  # normalize to sum=1
    return h_new

def local_exchange_hill_climb(h, max_iter=1000):
    """Perform a local hill climbing exchange of values."""
    h = np.array(h)
    dx = 2.0 / len(h)
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5

    for _ in range(max_iter):
        i = random.randint(0, len(h)-1)
        j = random.randint(0, len(h)-1)
        delta = random.uniform(0.001, 0.02)
        if h[i] < delta:
            continue

        h_test = h.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = np.clip(h_test, 0, 1)

        # Renormalize to sum=1
        h_test = project_simplex(h_test, 1.0)
        
        test_c5 = evaluate_erdos_solution(h_test)
        if test_c5 < current_c5:
            h = h_test
            current_c5 = test_c5
            if test_c5 < best_c5:
                best_h = h_test.copy()
                best_c5 = test_c5

    return best_h.tolist(), best_c5
