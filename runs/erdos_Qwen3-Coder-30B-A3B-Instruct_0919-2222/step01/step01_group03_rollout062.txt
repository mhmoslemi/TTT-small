import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Use simulated annealing approach
    def simulated_annealing():
        current_h = h0.copy()
        current_c5 = objective(current_h)
        best_h_sa = current_h.copy()
        best_c5_sa = current_c5
        
        # Parameters
        T = 1.0
        T_min = 1e-8
        alpha = 0.999
        iterations = 10000
        
        for i in range(iterations):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Perturb the current solution
            new_h = current_h.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.05)
            new_h[idx] += delta
            new_h = np.clip(new_h, 0, 1)
            
            # Scale to maintain constraint sum = 1
            new_h = new_h / (np.sum(new_h) * dx) * 1.0
            new_h = np.clip(new_h, 0, 1)
            
            new_c5 = objective(new_h)
            
            # Accept or reject the new solution
            if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
                current_h = new_h.copy()
                current_c5 = new_c5
                
                if current_c5 < best_c5_sa:
                    best_h_sa = current_h.copy()
                    best_c5_sa = current_c5
            
            # Cooling schedule
            T = max(T_min, T * alpha)
            
            # Early stopping if we've been improving for a while
            if best_c5_sa < 0.381:
                break
                
        return best_h_sa, best_c5_sa
    
    # Try simulated annealing to escape local minima
    try:
        h_sa, c5_sa = simulated_annealing()
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    except:
        pass
    
    # Try a hybrid approach with local optimization
    try:
        # Use differential evolution for a global search
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=10,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # After global search, use local optimization
    try:
        local_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Random restarts with variations of the approach
    max_restarts = 5
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation of solution
        h_rand = best_h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.3:  # 30% chance to change
                h_rand[i] += np.random.normal(0, 0.02)
        
        h_rand = np.clip(h_rand, 0, 1)
        # Maintain the constraint
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
