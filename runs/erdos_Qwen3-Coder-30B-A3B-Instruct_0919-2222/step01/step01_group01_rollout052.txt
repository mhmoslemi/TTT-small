def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_sum_constraint(h, target_sum):
    """Project h onto simplex: sum(h) = target_sum, h[i] in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    delta = (target_sum - current_sum) / len(h)
    h += delta
    h = np.clip(h, 0, 1)
    return h

def smooth_max(h, beta=100.0):
    """Smooth approximation to max using softmax"""
    # Compute all correlations
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    # Softmax approximation
    return -np.log(np.sum(np.exp(-beta * (corr - np.max(corr))))) / beta + np.max(corr)
