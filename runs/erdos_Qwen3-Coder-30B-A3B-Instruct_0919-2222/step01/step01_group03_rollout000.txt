import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector v onto [0,1] and sum=target_sum using water-filling"""
    # First clip to [0,1]
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Use water-filling to adjust
    # If sum > target, we need to reduce some values
    # If sum < target, we need to increase some values
    # But we must maintain bounds [0,1]
    if current_sum > target_sum:
        # Need to decrease values
        # We want to decrease elements that are highest
        # We can subtract a constant from all
        diff = current_sum - target_sum
        # Try to subtract diff/n from all elements
        v_new = v - diff / len(v)
        if np.all(v_new >= 0):
            return v_new
        else:
            # Use water-filling to adjust properly
            return water_fill(v, target_sum)
    else:
        # Need to increase values
        diff = target_sum - current_sum
        v_new = v + diff / len(v)
        if np.all(v_new <= 1):
            return v_new
        else:
            # Use water-filling to adjust properly
            return water_fill(v, target_sum)

def water_fill(v, target_sum):
    """Water-filling algorithm for projecting onto sum constraint"""
    # First clip to [0,1]
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Sort values in descending order to know which to raise first
    sorted_indices = np.argsort(v)[::-1]
    v_sorted = v[sorted_indices]
    
    # Check if we can raise all values
    total_needed = target_sum - current_sum
    remaining = target_sum
    i = 0
    while i < len(v_sorted) and remaining > 0:
        # Can we raise all at position 0..i to at least v_sorted[i]?
        # Amount needed to reach value v_sorted[i]
        if i == 0:
            new_value = v_sorted[i]
        else:
            new_value = v_sorted[i-1]  # Previous value in sorted order
        
        if new_value > v_sorted[i]:
            # We will raise to this level
            needed = (new_value - v_sorted[i]) * (i + 1)
            if needed <= remaining:
                # We can raise all to this level
                # Actually we want to raise elements 0 to i to new_value
                for j in range(i+1):
                    v_sorted[j] = new_value
                remaining -= needed
            else:
                # Raise equally to fill the remaining
                # If remaining > 0, we raise by remaining/(i+1)
                diff = remaining / (i + 1)
                for j in range(i+1):
                    v_sorted[j] += diff
                remaining = 0
                break
        i += 1
    
    # If we still have remaining, we can raise to 1
    if remaining > 0:
        if i < len(v_sorted):
            # distribute remaining to indices i onwards
            # temporary placeholder; this should not occur in theory
            pass
        else:
            remaining_per = remaining / len(v_sorted)
            for j in range(len(v_sorted)):
                v_sorted[j] = min(v_sorted[j] + remaining_per, 1.0)
    
    # Rearrange back to original order
    v_result = v.copy()
    for i, idx in enumerate(sorted_indices):
        v_result[idx] = v_sorted[i]
    
    return v_result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define optimization parameters
    max_iter = 50000
    eta0 = 0.2
    beta_max = 100.0
    beta_increment = beta_max / (max_iter // 1000)
    
    # Precompute shift indices to avoid recomputation
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points - s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Gradient descent optimization
    h = h0.copy()
    beta = 1.0  # Start with softmax beta
    
    # Initialize history
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute raw function values
        corr = np.correlate(h, 1 - h, mode="full")
        f = corr  # Already scaled with dx in evaluate_erdos_solution
        
        # Compute softmax weights
        max_f = np.max(f)
        # Numerical stability: subtract max
        exp_terms = np.exp(beta * (f - max_f))
        
        # Avoid overflow in case max_f is very negative
        if np.all(exp_terms == 0):
            w = np.ones_like(exp_terms) / len(exp_terms)
        else:
            w = exp_terms / np.sum(exp_terms)
        
        # Compute gradient
        grad = np.zeros_like(h)
        
        # Add constant term contribution from sum_i h_i
        total_w = np.sum(w)
        grad += total_w  # uniformly add 1 (since sum h_i is constrained)

        # Add gradient contributions from product terms
        for shift_idx, (i_arr, j_arr) in enumerate(shift_indices):
            if len(i_arr) == 0:
                continue
            weight = w[shift_idx]
            if weight > 1e-12:
                # Add gradient contribution from shift
                # d/dh_i of (-h_i h_{i+s}) term: -h_{i+s} (i in I_s)
                # Also d/dh_{i+s} of (-h_i h_{i+s}) term: -h_i
                grad[i_arr] += weight * (-h[j_arr])
                grad[j_arr] += weight * (-h[i_arr])
        
        # Step size
        eta = eta0 / np.sqrt(it + 1)
        
        # Update step
        h_new = h - eta * grad
        
        # Project back to valid domain and sum constraint
        h_new = project_onto_bounds_and_sum(h_new, n_points / 2)
        h = h_new
        
        # Update beta
        beta = min(beta + beta_increment, beta_max)
        
        # Evaluate current solution
        if it % 1000 == 0:
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return best solution found
    return (best_h, final_c5, n_points)
