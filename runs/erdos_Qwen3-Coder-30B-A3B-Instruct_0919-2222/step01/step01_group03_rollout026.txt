import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Strategy 1: Try different starting points
    starting_points = [h0]
    
    # Add a few more diverse starting points
    # Start with a ramp-like shape
    ramp = np.linspace(0.1, 0.9, n_points)
    ramp = ramp / (np.sum(ramp) * dx)
    starting_points.append(np.clip(ramp, 0, 1))
    
    # Start with a block-like shape
    block = np.zeros(n_points)
    mid = n_points // 2
    block[mid-5:mid+5] = 1.0
    block = block / (np.sum(block) * dx)
    starting_points.append(np.clip(block, 0, 1))
    
    # Start with a smooth bell-shaped function
    x = np.linspace(0, 2, n_points)
    bell = np.exp(-((x - 1)**2) / 0.1)
    bell = bell / (np.sum(bell) * dx)
    starting_points.append(np.clip(bell, 0, 1))
    
    # Try multiple random restarts
    for i in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
        h_rand = np.random.rand(n_points) * 0.5
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)
        starting_points.append(h_rand)
        
    # Test different optimization strategies
    for h_start in starting_points:
        h_start = np.clip(h_start, 0, 1)
        
        # Pool-based optimization: multiple methods
        try:
            # Try differential evolution
            de_result = differential_evolution(
                objective,
                bounds,
                maxiter=50,
                popsize=10,
                seed=seed + 1,
                callback=lambda x, convergence: None,
                atol=1e-6,
                rtol=1e-6
            )
            if de_result.success:
                h_opt = np.clip(de_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
        # Try local optimization with SLSQP
        try:
            local_result = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
    # Additional gradient-based refinement using a more direct implementation
    if time.time() - start_time < budget_s * 0.9:
        # Try with a broader sampling approach
        for method in ['L-BFGS-B', 'TNC']:
            try:
                local_result = minimize(
                    objective,
                    best_h,
                    method=method,
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
                    callback=lambda x: None
                )
                if local_result.success:
                    h_opt = np.clip(local_result.x, 0, 1)
                    c5_opt = objective(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
            except:
                pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
