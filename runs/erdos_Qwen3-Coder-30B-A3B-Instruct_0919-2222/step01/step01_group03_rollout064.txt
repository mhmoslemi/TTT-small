import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try using a hybrid method
    # 1. Start with a greedy approach
    h_greedy = h0.copy()
    h_greedy = h_greedy / (np.sum(h_greedy) * dx) * 1.0
    
    # Adjust by iterative improvements
    current_h = h_greedy.copy()
    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Try a better global optimization approach
    try:
        # Use a more extensive differential evolution
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=20,
            seed=seed,
            strategy='best1exp',
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.7
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except Exception as e:
        pass

    # Local refinement with more intensive optimization
    try:
        # Try L-BFGS-B with multiple restarts
        for i in range(3):
            if time.time() - start_time > budget_s * 0.9:
                break
            h_rand = np.random.rand(n_points) * 0.7 + 0.15  # Keep it away from extremes
            h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
            
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
    except Exception as e:
        pass

    # Final refinement with SLSQP
    try:
        local_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except Exception as e:
        pass

    # Additional search with specialized strategy for achieving good overlaps
    def improved_search(n_restarts=5):
        nonlocal best_c5, best_h
        for restart in range(n_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Better heuristic: create a function that is skewed
            h_candidate = np.zeros(n_points)
            # Create a two-peak shape
            peak1_center = n_points // 4
            peak2_center = 3 * n_points // 4
            sigma = n_points // 8
            
            for i in range(n_points):
                h_candidate[i] = 0.5 * (np.exp(-0.5 * ((i - peak1_center) / sigma)**2) + 
                                       np.exp(-0.5 * ((i - peak2_center) / sigma)**2)) 
            # Normalize it to required integral
            h_candidate = h_candidate / (np.sum(h_candidate) * dx) * 1.0
            h_candidate = np.clip(h_candidate, 0, 1)
            
            try:
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
            except Exception as e:
                continue

    improved_search(5)
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
