import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Initialize with the known good solution (from existing record)
    # This should be an informed initialization to guide the optimization
    try:
        # Start with a piecewise constant solution
        h_piecewise = np.zeros(n_points)
        # Distribute the mass more efficiently - place more mass in
        # regions where overlapping integrals can be minimized
        # Put ~0.36 at the end, ~0.26 in the middle, and ~0.38 at the start
        # This is a custom heuristic that tries to create a better overlap pattern
        n1 = n_points // 3
        n2 = n_points // 3
        n3 = n_points - n1 - n2
        h_piecewise[:n1] = 0.38
        h_piecewise[n1:n1+n2] = 0.26  
        h_piecewise[n1+n2:] = 0.36
        
        # Normalize to sum to 1
        h_piecewise = h_piecewise / (np.sum(h_piecewise) * dx) * 1.0
        h_piecewise = np.clip(h_piecewise, 0, 1)
        
        # Local optimization around this solution
        local_result = minimize(
            objective,
            h_piecewise,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Try multiple restarts with different strategies  
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random restart on a more sparse grid
        h_rand = np.random.rand(n_points) * 0.5  # Between 0 and 0.5
        # Scale to sum to 1
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
        # Try another pattern: decay on the left side and step up on right side
        h_decay_step = np.zeros(n_points)
        # Linear decay from right to left with a good sum (should be about 1)
        for i in range(n_points):
            h_decay_step[i] = max(0, 1.0 - 0.8 * i / n_points)
        h_decay_step = h_decay_step / (np.sum(h_decay_step) * dx) * 1.0
        h_decay_step = np.clip(h_decay_step, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_decay_step,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
