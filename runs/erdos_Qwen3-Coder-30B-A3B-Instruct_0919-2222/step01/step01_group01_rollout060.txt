def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def compute_product_sums(h):
    """Compute sum_i h_i * h_{i+s} for all shifts s"""
    n = len(h)
    # Convolution with zero padding gives sum_i h_i * h_{i+s}
    # We use np.correlate with mode="full" to compute correlations
    prod_sums = np.correlate(h, h, mode="full")
    return prod_sums

def compute_objective(h):
    """Compute objective value max(C(s)) for all shifts s"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    prod_sums = compute_product_sums(h)
    # Correlation sum_h - prod_sums 
    corr_s = sum_h - prod_sums
    return np.max(corr_s)

def project_to_simplex(h, target_sum):
    """Project h to simplex sum h_i = target_sum, 0 <= h_i <= 1."""
    # Use iterative algorithm
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for lambda
    low = np.min(h) - 1.0
    high = np.max(h)
    
    for _ in range(50):
        mid = (low + high) / 2.0
        projected = np.clip(h - mid, 0, 1)
        projected_sum = np.sum(projected)
        if projected_sum > target_sum:
            low = mid
        else:
            high = mid
            
    return np.clip(h - low, 0, 1)

def gradient_ascent_step(h, alpha, max_iter=1000):
    """Run projected gradient ascent to minimize max correlation."""
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_objective(h)
    
    # Schedule for learning rate
    for i in range(max_iter):
        # Compute product sums for all shifts
        prod_sums = compute_product_sums(h)
        # Compute C(s) = sum_h - prod_sums
        sum_h = np.sum(h)
        corr_s = sum_h - prod_sums  # This is our C(s)
        
        # Find worst shift (min product sum) 
        s_worst = np.argmin(prod_sums)
        worst_shift = s_worst - (n - 1)  # Convert index to actual shift

        # Compute gradient g_i = (1 - h[i+s]) for all valid pairs
        g = np.zeros(n)
        if worst_shift >= 0:
            # Forward shift
            g[:n - worst_shift] += h[worst_shift:]
            # Backward shift 
            g[worst_shift:] += h[:n - worst_shift]
        else:
            # Negative shift (backward)
            backward_shift = -worst_shift
            g[backward_shift:] += h[:n - backward_shift]
            g[:n - backward_shift] += h[backward_shift:]
        
        # Actually, another way to compute: for all pairs (i, i+s) in valid range
        g = np.zeros(n)
        for s in range(-(n - 1), n):
            if s >= 0:
                # i from 0 to n-s, i+s from s to n-1 
                start_i = 0
                end_i = n - s
                if start_i < end_i:
                    g[start_i:end_i] += h[s:end_i + s]
            else:
                # s < 0, i from -s to n-1, i+s from 0 to n+s-1  
                backward_shift = -s
                start_i = backward_shift
                end_i = n
                if start_i < end_i:
                    g[start_i:end_i] += h[:end_i - backward_shift]

        # Gradient ascent step
        h_new = h + alpha * g

        # Project to feasible region
        h_new = project_to_simplex(h_new, target_sum)
        
        # Update h
        h = h_new

        # Update best solution
        c5 = compute_objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
