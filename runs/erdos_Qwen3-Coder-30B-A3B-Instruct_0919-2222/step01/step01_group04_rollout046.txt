import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

    best_c5 = evaluate_c5(h)

    def objective(x):

        x = np.clip(x, 0, 1)
