from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points, n_steps):
    """Build h array from step heights."""
    # Each step is repeated equally
    seg_len = n_points // n_steps
    if seg_len * n_steps != n_points:
        seg_len = n_points // n_steps
        # Adjust so we fit exactly
        n_steps = n_points // seg_len
    # Repeat each height
    h = np.repeat(heights, seg_len)
    # If we have leftover points, distribute one extra to first few
    extra = n_points - seg_len * n_steps
    if extra > 0:
        h = np.pad(h, (0, extra), 'constant', constant_values=heights[-1])
    return h
