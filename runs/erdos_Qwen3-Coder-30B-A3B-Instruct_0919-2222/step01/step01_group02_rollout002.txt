import numpy as np
import time
import random
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, shift):
    """Return h shifted by shift positions (with zero padding)"""
    n = len(h)
    if shift > 0:
        return np.concatenate([np.zeros(shift), h[:-shift]])
    elif shift < 0:
        return np.concatenate([h[-shift:], np.zeros(-shift)])
    else:
        return h.copy()

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto the box [0,1]^n and the hyperplane sum = target_sum"""
    # Wrap v to [0,1]
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return v
    # Use a fast projection approach
    # Sort indices by difference from target
    if diff > 0:
        # Need to increase some values
        # Sort indices by descending value (prefer increasing higher values)
        idxs = np.argsort(v)[::-1]
        v_copy = v.copy()
        for i in idxs:
            if diff <= 0:
                break
            can_increase = 1.0 - v_copy[i]
            increase = min(can_increase, diff)
            v_copy[i] += increase
            diff -= increase
        return v_copy
    else:
        # Need to decrease some values
        idxs = np.argsort(v)
        v_copy = v.copy()
        for i in idxs:
            if diff >= 0:
                break
            can_decrease = v_copy[i]
            decrease = min(can_decrease, -diff)
            v_copy[i] -= decrease
            diff += decrease
        return v_copy

def build_lp_model(n, binary=False):
    """Build a linear program for C5 using McCormick relaxation"""
    h = cp.Variable(n, name="h")
    t = cp.Variable(name="t")
    objective = cp.Minimize(t)
    
    # Constraints on h
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2]
    
    # Use auxiliary variables y[i,d] for h[i] * h[i+d]
    # packed into a matrix to be indexed later
    y = cp.Variable((n, 2*n-1), name="y")  # for all shifts d in [-n+1, n-1]
    
    # For each valid pair (i, d)
    for d in range(-n+1, n):
        idx_d = d + n - 1  # Map shift d to index 0..2n-2
        for i in range(max(0, -d), min(n, n - d)):
            # McCormick constraints
            constraints.append(y[i, idx_d] >= 0)
            constraints.append(y[i, idx_d] <= h[i])
            constraints.append(y[i, idx_d] <= h[i + d])
            constraints.append(y[i, idx_d] >= h[i] + h[i + d] - 1)
            
    # For each shift d, relate t to the correlation
    for d in range(-n+1, n):
        idx_d = d + n - 1
        # Sum over valid i's, a row of the y matrix (i, d)
        if d >= 0:
            valid_indices = range(n - d)
        else:
            valid_indices = range(-d, n)
            
        # Sum of yi,d over valid indices
        corr_term = cp.sum(y[valid_indices, idx_d])
        constraints.append(n/2 - corr_term <= t)
    
    if binary:
        h = cp.Variable(n, boolean=True, name="h")
    else:
        h = cp.Variable(n, name="h")

    # Return a formal model which we can solve, including last set of constraints
    # Rebuild the matrix m and also constraints
    model = cp.Problem(objective, constraints)
    return model, h, t, y

def solve_model(model, time_limit=30):
    """Solve a CVXPY model with a per-solve timeout"""
    try:
        # Select a solver that is likely available (e.g., GLPK)
        model.solve(solver=cp.GLPK, verbose=False, max_time=time_limit)
        return model.status == cp.OPTIMAL or model.status == cp.OPTIMIZED_INFEASIBLE
    except:
        return False

def subgradient_refine(h_start, end_time):
    """Refine h using projected sub-gradient descent on the max shift"""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Incremental decay on learning rate
    lr_max = 0.5
    decay = 0.999
    min_lr = 1e-5
    
    counter = 0
    max_counter = 30000
    no_improvement = 0
    max_no_improvement = 5000
    
    while time.time() < end_time and counter < max_counter:
        counter += 1
        c5_current = evaluate_erdos_solution(h)
        
        # Find the worst shift
        shifts = np.arange(-n+1, n)
        correlations = np.zeros(len(shifts))
        for i, s in enumerate(shifts):
            shifted_h = shift_array(h, s)
            correlations[i] = np.sum(h * (1 - shifted_h)) * dx
        
        d_best = shifts[np.argmax(correlations)]
        
        # Compute negative gradient of the worst shift
        g = np.zeros(n)
        if d_best >= 0:
            g[:n-d_best] = 1 - h[d_best:] - h[:n-d_best]
        else:
            g[-d_best:] = 1 - h[d_best:] - h[-d_best:]
            
        # Direction is negative gradient
        dir_g = -g
        
        # Step size
        lr = max(min_lr, lr_max * (decay ** counter))
        
        # Predict update
        v = h + lr * dir_g
        
        # Project the update
        h_new = project_onto_box_and_sum(v, n/2)
        
        # New value
        c5_new = evaluate_erdos_solution(h_new)
        
        if c5_new < c5_current:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > max_no_improvement:
            # Add noise to escape local minima
            noise = np.random.normal(0, 0.05, n)
            v = h + noise
            h = project_onto_box_and_sum(v, n/2)
            no_improvement = 0
        
    return best_h, best_c5
