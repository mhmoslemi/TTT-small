import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Some additional manipulations for better results
    # Try different number of points
    for n_try in [32, 42, 50, 60]:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        h_try = np.zeros(n_try)
        h_try.fill(1.0 / 2.0)  # Initial guess
        dx_try = 2.0 / n_try
        
        try:
            local_result = minimize(
                objective,
                h_try,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_try)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) * dx_try - 1.0},
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    n_points = n_try
        except:
            pass
            
    # Use simulated annealing (or similar) for better global search
    # Try several population sizes and restarts
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate diversified initializations
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            de_result = differential_evolution(
                objective,
                bounds,
                maxiter=100,
                popsize=20,
                seed=seed + restart,
                callback=lambda x, convergence: None,
                atol=1e-7,
                rtol=1e-7,
                recombination=0.9,
                mutation=(0.5, 1.0)
            )
            if de_result.success:
                h_opt = np.clip(de_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
        # Local refinement with different methods
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
    # Try increasingly complex initializations based on the best so far
    # Smooth version of best current solution
    h_smooth = np.copy(best_h)
    # Apply a small amount of smoothing while preserving constraint
    for _ in range(5):
        h_new = np.zeros_like(h_smooth)
        for i in range(len(h_smooth)):
            total = 0.0
            count = 0
            for j in range(max(0, i - 2), min(len(h_smooth), i + 3)):
                total += h_smooth[j]
                count += 1
            h_new[i] = total / count
        h_smooth = h_new
        # Renormalize
        h_smooth = h_smooth / (np.sum(h_smooth) * dx) * 1.0

    try:
        local_result = minimize(
            objective,
            h_smooth,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
