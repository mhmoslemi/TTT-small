def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, shift):
    """Shift array by shift amount with zero padding"""
    n = len(arr)
    if shift >= 0:
        # Shift left by shift amount (i -> i+shift)
        return np.concatenate([arr[shift:], np.zeros(shift)])
    else:
        # Shift right by |shift| amount (i -> i+shift where shift<0)
        s = -shift
        return np.concatenate([np.zeros(s), arr[:n-s]])

def project_onto_box_simplex(v, lower=0., upper=1., target_sum=None):
    """Project vector v onto box [lower,upper] and hyperplane sum = target_sum"""
    if target_sum is None:
        target_sum = len(v) / 2
    
    # Bisection for λ to satisfy sum(clamp(v_i - λ, lower, upper)) = target_sum
    lo = -10.0
    hi = 10.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, lower, upper)

    iteration = 0
    last_improve_iter = 0

        iteration += 1

        corr = np.correlate(h, 1 - h, mode='full')

        # Compute direction vector using shifts

        shift_minus = shift_array(h, -d)
        direction = shift_plus + shift_minus - 1.0
        
        # Step size

        # Project onto constraints
        h = project_onto_box_simplex(v, target_sum=n/2)
        
        # Evaluate
        cur_C = evaluate_erdos_solution(h)
        if cur_C < best_C - 1e-10:
            best_C = cur_C
            best_h = h.copy()
            last_improve_iter = iteration
            stagnation = 0
        else:
            stagnation += 1
            
        # Random perturbation if stagnated
        if stagnation > 5000:
            perturb = best_h + np.random.normal(scale=0.05, size=n)
            h = project_onto_box_simplex(perturb, target_sum=n/2)
            stagnation = 0
            
    return best_h, best_C
