import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto bounds and sum constraint"""
    # First clip to [lower, upper]
    v = np.clip(v, lower, upper)
    
    # Adjust to meet target sum using water-filling
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Sort in descending order to find indices to reduce
    sorted_indices = np.argsort(-v)
    
    # Water-filling approach
    remaining_sum = target_sum - current_sum
    if remaining_sum > 0:
        # Need to increase values
        for idx in sorted_indices:
            if remaining_sum <= 0:
                break
            increase = min(remaining_sum, upper - v[idx])
            v[idx] += increase
            remaining_sum -= increase
    elif remaining_sum < 0:
        # Need to decrease values
        for idx in sorted_indices:
            if remaining_sum >= 0:
                break
            decrease = min(-remaining_sum, v[idx] - lower)
            v[idx] -= decrease
            remaining_sum += decrease
    
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Optimize using a hybrid approach with gradient descent and random search
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Precompute shift indices for vectorized gradient computation
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points - s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))
    
    # Gradient-based optimization with projected gradient descent
    def compute_gradient(h, weights):
        """Compute gradient for current h and weights"""
        grad = np.zeros_like(h)
        # Total weight for constant term (sum_s w_s * 1)
        total_w = np.sum(weights)
        grad += total_w
        # Gradient from product terms
        for shift_idx, (i, j) in enumerate(shift_indices):
            weight = weights[shift_idx]
            if weight != 0 and len(i) > 0:
                grad[i] += weight * (-h[j])
                grad[j] += weight * (-h[i])
        return grad
    
    # Main gradient descent optimization loop
    max_iter = 30000
    eta0 = 0.05
    beta = 1.0
    beta_max = 100
    beta_inc = (beta_max - beta) / (max_iter // 1000)
    
    h_current = h0.copy()
    h_current = project_onto_bounds_and_sum(h_current, n_points / 2)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation for current h
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        f = corr
        max_f = np.max(f)
        
        # Softmax weights with temperature
        exp_terms = np.exp(beta * (f - max_f))
        w = exp_terms / np.sum(exp_terms)
        
        # Compute gradient
        gradient = compute_gradient(h_current, w)
        
        # Adaptive step size
        eta = eta0 / np.sqrt(iter_num + 1)
        
        # Update h
        h_new = h_current - eta * gradient
        
        # Project back onto constraints
        h_new = project_onto_bounds_and_sum(h_new, n_points / 2)
        
        # Accept if better
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_current = h_new.copy()
        
        # Update beta
        beta = min(beta + beta_inc, beta_max)
    
    # Multi-start random search to escape local minima
    for restart in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate random initial h
        h_rand = np.random.rand(n_points) * 0.5
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = project_onto_bounds_and_sum(h_rand, n_points / 2)
        
        # Simple hill climbing
        for _ in range(5000):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Try swapping two random values
            i, j = np.random.randint(0, n_points, 2)
            if i == j:
                continue
                
            # Swap and check if improving
            h_trial = h_rand.copy()
            h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
            h_trial = project_onto_bounds_and_sum(h_trial, n_points / 2)
            c5_trial = evaluate_erdos_solution(h_trial)
            
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
            else:
                # Accept with low probability sometimes
                acceptance_prob = min(1.0, np.exp(-(c5_trial - best_c5) / 0.001))
                if np.random.rand() < acceptance_prob:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                
            h_rand = h_trial.copy()
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
