def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_simplex(v, s=1.0):
    """Projects v onto the capped simplex {x : sum(x) = s, 0 <= x_i <= 1} using water-filling algorithm."""
    v = np.asarray(v)
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Find the index where cumulative sum exceeds s
    cu = np.cumsum(u)
    rho = np.argmax(cu > s) if cu[-1] > s else n
    # Compute the threshold
    theta = (cu[rho - 1] - s) / rho if rho > 0 else (u[0] - s) / n
    # Project
    result = np.maximum(v - theta, 0)
    # Scale by sum to make it exactly s (should be exact, but safe)
    result *= s / np.sum(result)
    return result

def smooth_max(x, beta=10.0):
    """Smooth approximation of max using log-sum-exp."""
    return np.log(np.sum(np.exp(beta * x))) / beta

def smooth_max_gradient(x, beta=10.0):
    """Analytic gradient of smooth max."""
    exp_x = np.exp(beta * x)
    Z = np.sum(exp_x)
    return (beta * exp_x) / (Z * beta)

def projected_gradient_descent(h0, max_iter=5000, lr=0.01, beta_init=1.0, beta_growth=1.05, tol=1e-6):
    h = h0.copy()
    n = len(h)
    s = n / 2.0  # Required sum
    beta = beta_init
    prev = np.inf
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        sm = smooth_max(corr, beta)
        if abs(prev - sm) < tol:
            break
        prev = sm
        
        grad = np.correlate(smooth_max_gradient(corr, beta), 1-h, mode="full")
        grad += np.correlate(h, smooth_max_gradient(1-corr, beta), mode="full")  # Term from second exponential
        h -= lr * grad
        
        # Project onto feasible set
        h = project_simplex(h, s)
        h = np.clip(h, 0, 1)
        
        # Adaptive beta and learning rate
        beta *= beta_growth
        if it > 1000:
            lr *= 0.999
        
        if it % 500 == 0:
            print(f"Iter {it}, C₅ ≈ {sm:.6f}, β={beta:.2f}")
    
    return h

def local_exchange_search(h, max_iter=1000, step_size=0.01, rand_prob=0.1):

    s = n / 2.0

        if random.random() < rand_prob:
            h = project_simplex(h, s)
            h = np.clip(h, 0, 1)
        else:
            delta = random.uniform(0.001, step_size)
            if h[i] >= delta and h[j] <= 1 - delta:
                h[i] -= delta
                h[j] += delta
                # Renormalize
                current_sum = np.sum(h)
                adjust = (s - current_sum) / n
                h += adjust
                h = np.clip(h, 0, 1)
                h = project_simplex(h, s)
            
    return h
