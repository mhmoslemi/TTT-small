import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.fft import fft, ifft
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different optimization strategies
    # 1. DE with adaptive parameters
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=20,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-8,
            rtol=1e-8
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # 2. Local optimization from DE result and random starts
    # Start with the DE result
    try:
        local_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # 3. Try multiple random restarts with different initialization patterns
    random_starts = [
        np.random.rand(n_points) * 0.5,
        np.ones(n_points) * 0.5,
        np.linspace(0.1, 0.9, n_points),
        np.concatenate([np.ones(n_points//2) * 0.3, np.ones(n_points - n_points//2) * 0.7]),
        np.concatenate([np.ones(n_points//3) * 0.3, np.ones(n_points - n_points//3) * 0.7])
    ]
    
    for i, h_rand in enumerate(random_starts):
        if i > 0 and time.time() - start_time > budget_s * 0.8:
            break
        h_rand = np.clip(h_rand, 0, 1)
        # Normalize so integral = 1
        h_rand = h_rand * (1.0 / (np.sum(h_rand) * dx))
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-12},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # 4. Try some manual heuristics - start with a two-piece construction
    # Potentially good regions: one high-value section and one low-value section
    def create_two_segment_h(alpha=0.3, beta=0.7):
        h = np.ones(n_points)
        midpoint = n_points // 2
        h[:midpoint] = alpha
        h[midpoint:] = beta
        h = h * (1.0 / (np.sum(h) * dx))
        return np.clip(h, 0, 1)
        
    for alpha, beta in [(0.2, 0.8), (0.3, 0.7), (0.25, 0.75), (0.4, 0.6)]:
        try:
            h_manual = create_two_segment_h(alpha, beta)
            c5_manual = objective(h_manual)
            if c5_manual < best_c5:
                best_c5 = c5_manual
                best_h = h_manual.copy()
        except:
            continue

    # Final polishing with gradient-based optimization
    try:
        final_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-15},
            callback=lambda x: None
        )
        if final_result.success:
            h_opt = np.clip(final_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
