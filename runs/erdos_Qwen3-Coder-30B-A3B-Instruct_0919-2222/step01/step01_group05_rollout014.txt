import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Return C5 value for given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = max(corr) * dx
    return c5

def prepare_milp_model(n):
    """Return a MILP model for binary h with n points"""
    # Variables
    b = cp.Variable(n, boolean=True)
    z = {}
    s = {}
    t = cp.Variable()
    
    # Define all z_{i,d} variables with index pairs
    for d in range(1, n):
        for i in range(n - d):
            z[(i, d)] = cp.Variable(boolean=True)
            # Create a unique key for reverse pairing (d, i) where i = n-d-1, ... 0
            # But we only need to process each pair once
    
    # Set up constraints for z
    constraints = []
    for d in range(1, n):
        for i in range(n - d):
            # z[i,d] <= b[i], z[i,d] <= b[i+d]
            constraints.append(z[(i, d)] <= b[i])
            constraints.append(z[(i, d)] <= b[i+d])
            # z[i,d] >= b[i] + b[i+d] - 1
            constraints.append(z[(i, d)] >= b[i] + b[i+d] - 1)
    
    # Because we are using symmetry, and s_d >= s_{-d}, we will only model shift d >= 0
    for d in range(n):
        s[d] = cp.sum([z[(i, d)] for i in range(n - d)])
        # s_d = Sum over valid i of z[i,d]
        # Now, our tight ROC in the original problem implies:
        # t >= w - s_d
        # Once z are modeled, this is an easier linear constraint:
        constraints.append(t >= n/2 - s[d]) 

    # Set half-integral constraint
    constraints.append(cp.sum(b) == n/2)
    constraints.append(t >= 0)

    # Set symmetry-breaking constraint
    constraints.append(b[0] == 1)
    # Optional: break reflection symmetry
    # constraints.append(b[0] >= b[1])
    
    # Objective
    obj = cp.Minimize(t)

    # Create and return the problem
    prob = cp.Problem(obj, constraints)
    return prob, b, z, s, t

def optimize_milp(prob, b, solver="CBC", timeout_s=250):
    """Run MILP solver and return binary solution, or best infeasible feasible"""
    try:
        if solver == "CBC":
            prob.solve(solver=cp.CBC, verbose=False, maxSeconds=timeout_s)
        elif solver == "GLPK_MI":
            prob.solve(solver=cp.GLPK_MI, verbose=False, maxSeconds=timeout_s)
        elif solver == "SCIP":
            prob.solve(solver=cp.SCIP, verbose=False, maxSeconds=timeout_s)
        else:  # fallback
            prob.solve(solver=cp.CBC, verbose=False, maxSeconds=timeout_s)
    except Exception as e:
        pass  # Could be time limit
    
    # Extract the best feasible solution using .value
    if b.value is not None and hasattr(b.value, "__len__"):
        b_sol = np.array([int(round(x)) for x in b.value])
        if np.sum(b_sol) == 42 / 2:
            return b_sol
        else:
            # Renormalize or round to match sum constraint
            return np.clip(b_sol, 0, 1)
    else:
        # Return binary solution rounded off
        b_val = np.array([int(round(x)) for x in b.value])
        return np.clip(b_val, 0, 1)

def run_simulated_annealing(h0, max_time_s):
    """Perform simulated annealing on h with exact integral constraint"""
    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    start_time = time.time()
    T = 0.02 * best_c5  # Initial temp
    alpha = 0.995  # Cooling rate
    delta_max = 0.2
    no_improvement_steps = 0
    max_no_improvement = 5000
    
    while time.time() - start_time < max_time_s:
        # Choose two positions at random
        i = np.random.randint(0, n-1)
        j = np.random.randint(0, n-1)
        if i == j:
            continue
        h = best_h.copy()  # new trial
        delta = np.random.uniform(0, min(h[i], 1-h[j], delta_max))
        h[i] -= delta
        h[j] += delta
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Fix sum to n/2 exactly
        h = h * (n/2) / np.sum(h)
        
        # Compute new C5
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h
            best_c5 = new_c5
            no_improvement_steps = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                best_h = h
                best_c5 = new_c5
                no_improvement_steps = 0
            else:
                no_improvement_steps += 1
                if no_improvement_steps > max_no_improvement:
                    break
        
        # Cool down
        T *= alpha
        if delta_max > 1e-5:
            delta_max *= 0.999

    return best_h, best_c5

def initial_h_values():
    # Use the standard construction
    h = np.zeros(42)
    h[:21] = 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.infty
    best_h = None
    best_n = None

    # Sizes to try
    grid_sizes = [42, 48, 60]

    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            # MILP stage
            prob, b, z, s, t = prepare_milp_model(n)
            timeout = min(300, budget_s / 2)
            h_bin = optimize_milp(prob, b, "CBC", timeout)
            h_bin = np.array(h_bin, dtype=float)

            # Evaluate
            c5 = evaluate_erdos_solution(h_bin)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_bin
                best_n = n

            # Continue to refinement with SA
            remain_time = budget_s - (time.time() - start_time)
            if remain_time > 30:
                refine_h, refine_c5 = run_simulated_annealing(h_bin, remain_time)
                if refine_c5 < best_c5:
                    best_c5 = refine_c5
                    best_h = refine_h
                    best_n = n

        except Exception as e:
            continue  # Skip failed iterations

    if best_h is None:
        # Fallback to initial h if nothing else succeeds
        h_fallback = initial_h_values().copy()
        c5_fallback = evaluate_erdos_solution(h_fallback)
        return (h_fallback, c5_fallback, 42)

    # Ensure integral is preserved exactly
    best_h = best_h * (best_n / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_erdos_solution(best_h)

    return (best_h, final_c5, best_n)
