import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, shift):
    """Shift array by 'shift' positions, zero-padded"""
    n = len(h)
    if shift == 0:
        return h.copy()
    elif shift > 0:
        return np.concatenate([np.zeros(shift), h[:-shift]])
    else:
        return np.concatenate([h[-shift:], np.zeros(-shift)])

def project_onto_box_and_sum(v, target_sum, lb=0.0, ub=1.0):
    """Projects vector v onto box [lb,ub] and then onto sum constraint."""
    n = len(v)
    
    def project_sum(v, lb, ub, target_sum):
        # Newton-like bisection to find lambda
        low, high = -100.0, 100.0
        for _ in range(40):  # Converges quickly
            mid = (low + high) / 2.0
            projected = np.clip(v - mid, lb, ub)
            sum_v = np.sum(projected)
            if sum_v > target_sum:
                high = mid
            else:
                low = mid
        return np.clip(v - low, lb, ub)
    
    return project_sum(v, lb, ub, target_sum)

def solve_with_time_limit(problem, solver='GLPK', timeout=20):
    # Solve with a time limit if supported by solver
    start = time.time()
    try:
        # Try specific solver name as string
        if "GLPK" in solver:
            problem.solve(solver=cp.GLPK, max_time=timeout)
        elif "CBC" in solver:
            problem.solve(solver=cp.CBC, max_time=timeout)
        else:
            problem.solve(solver=cp.GLPK, max_time=timeout)
    except Exception as e:
        # Solver failure or time limit reached
        print(f"[solver failed] {e}")
        return False
    return True

def subgradient_refine(h_start, target_c5, end_time):
    """Refine initial h using projected subgradient descent."""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    max_iter = 1000000
    iter_count = 0
    fbest = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_f = fbest
    counter = 0
    alpha = 0.5  # Initial step size
    beta = 0.999
    stall_thresh = 5000
    np.random.seed(42)  # deterministic  
    
    while iter_count < max_iter:
        if time.time() > end_time:
            break
            
        fcurrent = evaluate_erdos_solution(h)
        if fcurrent < best_f:
            best_h = h.copy()
            best_f = fcurrent

        # Identify worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        d = max_k - n + 1  # Compute from full correlation index pack
                    
        # Compute gradient direction
        # g[i] = 1 - h[i + d] - h[i - d] if exists
        g = np.zeros_like(h)
        if d > 0:
            g[d:] = 1 - h[:-d]  # for i+d<n, so g[i]=-h[i+d]
            if d < n: g[:-d] += h[d:]
        else:
            g[:d] = 1 - h[-d:]  # i-d >=0
            if d < 0: g[-d:] += h[:d]

        if d == 0:
            g = 1 - 2 * h
            
        # Update: h <- h - alpha * g
        v = h - alpha * g
        h = project_onto_box_and_sum(v, n / 2, 0, 1)
        # permute small noise for escaping stations:
        counter += 1
        if counter > stall_thresh and np.random.rand() < 0.01:
            h = h.copy()
            noise = np.random.normal(0, 0.05, n)
            h += noise
            h = project_onto_box_and_sum(h, n / 2, 0, 1)
            counter = 0
            
        iter_count += 1
        alpha = alpha * beta
        
        if counter % 1000 == 0:
            fr = evaluate_erdos_solution(h)
            if fr < fbest:
                best_h = h.copy()
                best_f = fr
            if fr < target_c5 and time.time() + 5 < end_time:
                print(f"Early convergence at iter={iter_count}, C5={fr:.6f}")
                return best_h, best_f
        
    return best_h, best_f
