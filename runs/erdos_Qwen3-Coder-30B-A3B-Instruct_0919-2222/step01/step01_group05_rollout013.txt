import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt
import cvxpy as cp
from cvxpy import ECOS

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def solve_lp_relaxation(n):
    """Solve the LP relaxation for the discrete Erdős constant."""
    dx = 2.0 / n
    w = n / 2.0

    # Variables
    h = cp.Variable(n)
    u = cp.Variable((n, 2*n - 1)) # u[i, d] for d in -n+1 to n-1

    # Objective
    t = cp.Variable()
    obj = cp.Minimize(t)

    # Constraints
    constraints = [
        cp.sum(h) == w,
        h >= 0,
        h <= 1,
        t >= w - cp.sum(u, axis=0)
    ]

    # Add McCormick envelope constraints: u[i, d] >= h[i] + h[i+d] - 1
    for d in range(-n+1, n):
        for i in range(max(0, -d), min(n, n-d)):
            if i + d >= 0 and i + d < n:
                constraints += [
                    u[i, d + n - 1] >= 0,
                    u[i, d + n - 1] <= h[i],
                    u[i, d + n - 1] <= h[i + d],
                    u[i, d + n - 1] >= h[i] + h[i + d] - 1
                ]

    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status == 'optimal':
            return h.value
    except Exception:
        pass
    return None

def project_onto_simplex(x, target_sum):
    """Project vector x onto the simplex with sum = target_sum."""
    x = np.asarray(x)
    n = len(x)
    mu = (np.sum(x) - target_sum) / n
    x_projected = x - mu
    x_projected = np.clip(x_projected, 0, 1)
    scale = target_sum / np.sum(x_projected)
    x_projected = x_projected * scale
    return x_projected

def subgradient(h, d_star):
    """Compute subgradient for worst shift d_star."""
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        if i + d_star >= 0 and i + d_star < n:
            g[i] -= h[i + d_star]
        if i - d_star >= 0 and i - d_star < n:
            g[i] -= h[i - d_star]
    return g

def projected_gradient_descent(h, max_iter=300, lr=0.1):
    """Run projected gradient descent on the Erdős function."""
    n = len(h)
    w = n / 2.0
    dx = 2.0 / n

    for i in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        d_star = np.argmax(corr)
        grad = subgradient(h, d_star)
        h_new = h - lr * grad
        h_new = project_onto_simplex(h_new, w)
        h = h_new
        if i % 20 == 0:
            c5 = evaluate_erdos_solution(h)
            if c5 < 0.3800:
                break

    return h

def swap_improve_step(h, d_star):
    """Perform an improvement swap for a given worst shift d_star."""
    n = len(h)
    for i in range(n):
        if h[i] == 1:
            try:
                j = i + d_star
                if 0 <= j < n and h[j] == 0:
                    h_new = h.copy()
                    h_new[i] = 0
                    h_new[j] = 1
                    return h_new
            except:
                continue
    return h

def swap_hill_climber(h_binary):
    """Perform deterministic hill-climbing based on swaps."""
    n = len(h_binary)
    while True:
        corr = np.correlate(h_binary, 1 - h_binary, mode="full")
        d_star = np.argmax(corr)

        # Try to find a beneficial swap
        found = False
        for i in range(n):
            if h_binary[i] == 1:
                try:
                    j = i + d_star
                    if 0 <= j < n and h_binary[j] == 0:
                        h_binary[i] = 0
                        h_binary[j] = 1
                        found = True
                        break
                except:
                    continue

        if not found:
            break

    # Final evaluation after swaps
    return h_binary

def simulated_annealing(h_init, n, max_iter=800, temp_init=0.02, cooling=0.995):
    """Simulated annealing with transfer moves."""
    current = h_init.copy()
    current_c5 = evaluate_erdos_solution(current)
    best = current.copy()
    best_c5 = current_c5
    
    temp = temp_init
    for i in range(max_iter):
        if temp < 1e-8:
            break

        # Choose two random indices
        i_idx = np.random.randint(0, n)
        j_idx = np.random.randint(0, n)

        if i_idx == j_idx:
            continue

        max_delta = min(current[i_idx], 1 - current[j_idx], 0.1)
        if max_delta <= 0:
            continue

        delta = np.random.uniform(0, max_delta)
        current[i_idx] -= delta
        current[j_idx] += delta

        current = np.clip(current, 0, 1)

        # Renormalize
        total = np.sum(current)
        current = current * (n / 2.0) / total

        new_c5 = evaluate_erdos_solution(current)
        if new_c5 < best_c5:
            best = current.copy()
            best_c5 = new_c5

        # Accept with probability
        if new_c5 > current_c5:
            prob = np.exp(-(new_c5 - current_c5) / temp)
            if np.random.rand() > prob:
                # Revert
                current = current.copy()  # This is okay because we haven't modified it yet

        temp *= cooling

    return best, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Sample n values to try
    n_candidates = [42, 48, 54, 60]
    
    best_h = None
    best_c5 = float('inf')
    best_n = None

    for n in n_candidates:
        if time.time() - start_time > budget_s * 0.95:
            break

        # Solve LP relaxation
        h_lp = solve_lp_relaxation(n)
        if h_lp is None:
            continue

        dx = 2.0 / n
        h_num = h_lp.copy()
        h_num = h_num * (n / (2 * np.sum(h_num)))

        # Evaluate it first
        c5 = evaluate_erdos_solution(h_num)
        if c5 < best_c5:
            best_h = h_num.copy()
            best_c5 = c5
            best_n = n

        # Rounding + binary swap refinement
        h_binary = np.zeros(n)
        idxs = np.argsort(h_num)[::-1][:n // 2]
        h_binary[idxs] = 1
        h_binary = swap_hill_climber(h_binary)
        c5_binary = evaluate_erdos_solution(h_binary)
        if c5_binary < best_c5:
            best_h = h_binary.copy()
            best_c5 = c5_binary
            best_n = n

        # Then SA
        h_sa, c5_sa = simulated_annealing(h_num, n, max_iter=600)
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa
            best_n = n

        # Then gradient descent
        h_grad = projected_gradient_descent(h_sa, max_iter=200, lr=0.1)
        c5_grad = evaluate_erdos_solution(h_grad)
        if c5_grad < best_c5:
            best_h = h_grad.copy()
            best_c5 = c5_grad
            best_n = n

    # Final boost with full optimization
    if time.time() - start_time < budget_s * 0.9:
        # Refine once more with a direct call to minimize with bounds if possible.
        try:
            bounds = [(0, 1) for _ in range(best_n)]
            def obj(h):
                return evaluate_erdos_solution(h)
            
            res = minimize(obj, best_h, bounds=bounds, method='L-BFGS-B', 
                           options={'maxiter': 50})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass

    # Normalize to exact integral constraint
    if best_h is not None:
        sum_h = np.sum(best_h)
        if sum_h > 0:
            scale = (best_n / 2.0) / sum_h
            best_h = best_h * scale

    return (best_h, best_c5, best_n)
