import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different approaches to improve the solution
    # Approach 1: Evolutionary algorithm with finer control
    try:
        # Do more iterations with DE
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=300,
            popsize=20,
            seed=seed,
            callback=lambda x, convergence: None,
            tol=1e-8,
            recombination=0.9
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Approach 2: Local search starting from DE solution
    if time.time() - start_time < budget_s * 0.8:
        try:
            h_start = best_h.copy()
            # Try multiple optimization methods
            methods = ['SLSQP', 'L-BFGS-B', 'TNC']
            for method in methods:
                if time.time() - start_time > budget_s * 0.9:
                    break
                try:
                    local_result = minimize(
                        objective,
                        h_start,
                        method=method,
                        bounds=bounds,
                        constraints=constraints,
                        options={'maxiter': 3000, 'ftol': 1e-10, 'gtol': 1e-10},
                        callback=lambda x: None
                    )
                    if local_result.success:
                        h_opt = np.clip(local_result.x, 0, 1)
                        c5_opt = objective(h_opt)
                        if c5_opt < best_c5:
                            best_c5 = c5_opt
                            best_h = h_opt.copy()
                except:
                    continue
        except:
            pass

    # Approach 3: Random restarts with smarter initializations
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Try a few different initialization strategies
        h_rand = np.random.rand(n_points)
        
        # Scale to desired integral
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        
        # Clip to [0,1] and refine
        h_rand = np.clip(h_rand, 0, 1)
        
        # Add some structure to prevent local minima
        if restart % 3 == 0:
            # Try a piecewise constant approach with 3 segments
            mid1, mid2 = n_points // 3, 2 * n_points // 3
            h_rand[:mid1] = 0.3
            h_rand[mid1:mid2] = 0.6
            h_rand[mid2:] = 0.8
        elif restart % 3 == 1:
            # Try a bell-shaped distribution
            x_vals = np.linspace(0, 2, n_points)
            mean, std = 1, 0.3
            h_rand = np.exp(-0.5 * ((x_vals - mean) / std) ** 2)
            h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        # For restart % 3 == 2, use uniform random as above
            
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Approach 4: Gradient-free optimization with more diversity
    if time.time() - start_time < budget_s * 0.8:
        # Try a hybrid using basin hopping with advisor
        try:
            from scipy.optimize import basinhopping
            
            def objective_wrapper(h):
                return objective(h)
            
            def take_step(h):
                # Small random perturbations
                step_size = 0.1
                rand_pert = np.random.normal(0, step_size, len(h))
                h_new = h + rand_pert
                # Ensure h stays in feasible region [0, 1]
                h_new = np.clip(h_new, 0, 1)
                return h_new
            
            h_start = best_h.copy()
            # Use basin hopping to escape local minima
            bh_result = basinhopping(
                objective_wrapper,
                h_start,
                take_step=take_step,
                niter=100,
                stepsize=0.05,
                minimizer_kwargs={
                    'method': 'SLSQP',
                    'bounds': bounds,
                    'constraints': constraints,
                    'options': {'ftol': 1e-9, 'gtol': 1e-9}
                },
                callback=lambda x, f, accept: None
            )
            
            if bh_result.success:
                h_opt = np.clip(bh_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
