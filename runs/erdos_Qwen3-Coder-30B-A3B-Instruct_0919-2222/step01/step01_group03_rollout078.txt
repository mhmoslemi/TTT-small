import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different initialization strategies:
    # 1. Starting with the initial solution
    try:
        local_result = minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # 2. Try different global optimization methods
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=150,
            popsize=20,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-8,
            rtol=1e-8
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # 3. Local optimization from a random solution
    for restart in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate random initialization with a bit more structure
        h_rand = np.random.rand(n_points)
        # Try to make it closer to the target integral constraint
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1.0
        
        # Add some smoothness to reduce oscillation penalties
        h_rand = np.clip(h_rand, 0, 1) 
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1500, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
    # 4. Try a more aggressive hill climbing approach with mutated solutions
    for restart in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        
        h_test = best_h.copy()
        # Perturb the best solution a bit
        idx = np.random.randint(0, n_points)
        h_test[idx] += (np.random.rand() - 0.5) * 0.2
        h_test = np.clip(h_test, 0, 1)
        # Re-scale to meet constraint
        h_test = h_test / (np.sum(h_test) * dx) * 1.0
        
        try:
            local_result = minimize(
                objective,
                h_test,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
