import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different optimization approaches
    
    # 1. Differential Evolution first
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=20,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-8,
            rtol=1e-8
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # 2. Local optimization with different starting points
    # Try multiple restarts with different heuristics
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random initialization with scaling
        h_rand = np.random.rand(n_points) * 0.5
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
        # Try a more targeted approach: perturb the best solution so far
        if restart < 5 and restart > 0:
            h_pert = best_h.copy()
            # Add small random perturbations
            for i in range(n_points):
                if np.random.rand() < 0.3:
                    perturbation = np.random.normal(0, 0.05)
                    h_pert[i] = np.clip(h_pert[i] + perturbation, 0, 1)
            
            try:
                h_pert = h_pert / (np.sum(h_pert) * dx) * 1.0  # Scale to sum = 1
                local_result = minimize(
                    objective,
                    h_pert,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10},
                    callback=lambda x: None
                )
                if local_result.success:
                    h_opt = np.clip(local_result.x, 0, 1)
                    c5_opt = objective(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
            except:
                pass
    
    # 3. Try grid refinement for the best solution
    try:
        local_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
