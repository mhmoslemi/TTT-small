import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

# Precompute distance matrix for accelerated computation
def compute_distance_matrix(n_points):
    """Precompute distance matrix for correlation computations"""
    x = np.linspace(0, 2, n_points)
    return cdist(x.reshape(-1, 1), x.reshape(-1, 1), metric='euclidean')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try multiple strategies with different algorithms
    
    # Strategy 1: Differential evolution with custom restarts
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=100,
            popsize=15,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Strategy 2: Local optimization from different random starting points
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Generate random initialization - several different patterns
        pattern = restart % 3
        if pattern == 0:
            h_rand = np.random.rand(n_points) * 0.5
        elif pattern == 1:
            # Try a slightly more structured approach
            h_rand = np.ones(n_points) * 0.5 + (np.random.rand(n_points) - 0.5) * 0.2
        else:
            # Try a linear decreasing pattern
            h_rand = np.linspace(0.1, 0.9, n_points)
            h_rand = np.abs(h_rand - np.mean(h_rand)) + np.random.rand(n_points) * 0.1
            
        # Scale to satisfy constraint
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Strategy 3: Basin hopping variant
    try:
        from scipy.optimize import basinhopping
        
        def objective_no_clip(h):
            return evaluate_erdos_solution(h)
        
        # Use a smaller variance for perturbations
        minimizer_kwargs = {"method": "SLSQP", "bounds": bounds, 
                           "constraints": constraints, "options": {"ftol": 1e-9, "gtol": 1e-9}}
        
        # Basin hopping with different starting points
        for i in range(5):
            if time.time() - start_time > budget_s * 0.9:
                break
            # Generate a random sample with constraints
            h_init = np.random.rand(n_points) * 0.5
            h_init = h_init / (np.sum(h_init) * dx) * 1.0
            h_init = np.clip(h_init, 0, 1)
            
            bh_result = basinhopping(
                objective_no_clip,
                h_init,
                niter=50,
                T=1.0,
                stepsize=0.1,
                minimizer_kwargs=minimizer_kwargs,
                callback=lambda x, f, accept: None
            )
            
            if bh_result.success:
                h_opt = np.clip(bh_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
    except:
        pass
    
    # Strategy 4: Gradient-free with alternating optimization
    # Grid search to find better local optima
    if time.time() - start_time < budget_s * 0.8:
        # Try a simple subdivision approach
        # For small values, modify the initial solution in small steps
        for i in range(5):
            if time.time() - start_time > budget_s * 0.9:
                break
                
            # Create a perturbed version of best solution
            h_pert = best_h.copy()
            num_changes = min(5, i * 2 + 1)  # Increase number of modifications
            
            for _ in range(num_changes):
                pos = np.random.randint(0, n_points)
                change = np.random.normal(0, 0.05)
                h_pert[pos] += change
                
            # Ensure bounds
            h_pert = np.clip(h_pert, 0, 1)
            
            # Rescale to fix constraint
            h_pert = h_pert / (np.sum(h_pert) * dx) * 1.0
            
            try:
                c5_pert = objective(h_pert)
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h_pert.copy()
            except:
                pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
