import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def project_onto_simplex(h, target_sum):
    """Project h onto the simplex {x : sum(x) = target_sum, x >= 0}"""
    h = np.array(h)
    h = h - np.mean(h) + target_sum / len(h)
    h = np.clip(h, 0, None)
    h = h * (target_sum / np.sum(h))
    return h

def solve_lp_initialization(n_points):
    """Solve a convex LP relaxation to get a fractional starting point"""
    dx = 2.0 / n_points
    w = n_points / 2
    
    # Variables
    h = cp.Variable(n_points)
    u = cp.Variable((n_points, 2 * n_points - 1))  # for each h_i and shift d
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [
        cp.sum(h) == w,
        h >= 0,
        h <= 1
    ]
    
    # Add McCormick envelopes for the product terms
    # u[i,d] >= h[i] + h[i+d] - 1
    # u[i,d] <= h[i]
    # u[i,d] <= h[i+d]
    # u[i,d] >= 0
    for d in range(-n_points + 1, n_points):
        for i in range(n_points):
            if 0 <= i + d < n_points:
                if d >= 0:
                    idx_u = i + d
                else:
                    idx_u = i + n_points - 1 + d  # Index in u array
                constraints.append(u[i, d + n_points - 1] >= h[i] + h[i + d] - 1)
                constraints.append(u[i, d + n_points - 1] <= h[i])
                constraints.append(u[i, d + n_points - 1] <= h[i + d])
                constraints.append(u[i, d + n_points - 1] >= 0)
    
    # Add constraints relating u to the shift C5 calculation
    for d in range(-n_points + 1, n_points):
        s_d = 0
        for i in range(n_points):
            if 0 <= i + d < n_points:
                if d >= 0:
                    idx_u = i + d
                else:
                    idx_u = i + n_points - 1 + d
                s_d += u[i, d + n_points - 1]
        constraints.append(t >= w - s_d)
    
    # Solve with ECOS solver
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return h.value
    except Exception:
        pass
    return None  # fallback

def swap_worst_shift(h, verbose=False):
    """Improve a binary h by swapping 1s and 0s to reduce worst shift"""
    n = len(h)
    last_best = evaluate_erdos_solution(h)
    done = False
    
    # Compute the full cross-correlation
    corr = np.correlate(h, 1 - h, mode="full")
    
    while not done:
        d_max = np.argmax(corr)
        # The actual shift is between -n+1 and n-1
        d_best = d_max - (n - 1)
        
        # Find first 1 and 0 at d_best distance
        best_move = None
        for i in range(n):
            j = (i + d_best) % n
            if h[i] == 1 and h[j] == 0:
                # Try to find matching 0 and 1
                for k in range(n):
                    l = (k - d_best) % n
                    if h[k] == 0 and h[l] == 1:
                        best_move = (i, j, k, l)
                        break
            if best_move:
                break
        
        if best_move:
            # Perform the swap
            i, j, k, l = best_move
            h[i], h[j], h[k], h[l] = h[k], h[l], h[i], h[j]
            corr = np.correlate(h, 1 - h, mode="full")
            new_val = evaluate_erdos_solution(h)
            if new_val >= last_best:
                h[i], h[j], h[k], h[l] = h[k], h[l], h[i], h[j]
                break
            else:
                last_best = new_val
        else:
            done = True
        
        if verbose:
            print(f"Improvement via swaps. Current C5: {last_best:.6f}")

    return h

def simulated_annealing_step(h, T):
    """Simple simulated annealing mass-transfer move"""
    n = len(h)
    h_new = h.copy()
    
    # Select two random points
    i = np.random.randint(n)
    j = np.random.randint(n)
    
    if i == j:
        return h_new
    
    # Define transfer size as min(h[i], 1 - h[j], 0.1)
    delta = min(h[i], 1 - h[j], 0.1)
    
    # Perform the move
    h_new[i] -= delta
    h_new[j] += delta
    
    # Ensure bounds
    h_new = np.clip(h_new, 0, 1)
    
    # Renormalize
    h_new = project_onto_simplex(h_new, n / 2)
    
    return h_new

def get_subgradient(h, d_star):
    """Get subgradient for the current worst shift d_star"""
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        if i + d_star >= 0 and i + d_star < n:
            g[i] -= h[i + d_star]
        if i - d_star >= 0 and i - d_star < n:
            g[i] -= h[i - d_star]
    return g

def projected_gradient_descent(h, num_iter=500, learning_rate=0.1):
    """Project gradient descent to reduce the max C5"""
    n = len(h)
    w = n / 2
    for k in range(num_iter):
        # Get the current maximum shift
        corr = np.correlate(h, 1 - h, mode="full")
        d_max = np.argmax(corr)
        d_star = d_max - (n - 1)
        
        # Compute the gradient
        g = get_subgradient(h, d_star)
        
        # Update step
        h_new = h - learning_rate * g
        h_new = project_onto_simplex(h_new, w)
        
        # Decrease learning rate slowly
        learning_rate *= 0.995
        
        # Check for convergence
        # For now we just accept the new value
        h = h_new
        
    return h
