import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Generate different initializations based on different strategies
    init_strategies = []
    
    # Strategy 1: Constant function
    init_strategies.append(np.ones(n_points) * 0.5)
    
    # Strategy 2: Piecewise constant
    h_piecewise = np.zeros(n_points)
    h_piecewise[:n_points//2] = 1.0
    h_piecewise[n_points//2:] = 0.0
    init_strategies.append(h_piecewise)
    
    # Strategy 3: Sine-based approach
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 + 0.2 * np.sin(np.pi * x)
    h_sine = np.clip(h_sine, 0, 1)
    init_strategies.append(h_sine)
    
    # Strategy 4: Gaussian-based approach
    x = np.linspace(0, 2, n_points)
    h_gaussian = np.exp(-16 * (x - 1)**2) * 2
    h_gaussian = np.clip(h_gaussian, 0, 1)
    init_strategies.append(h_gaussian)
    
    # Strategy 5: Random with scaling to meet constraint
    h_rand_scaled = np.random.rand(n_points) * 0.8 + 0.1
    h_rand_scaled = h_rand_scaled / (np.sum(h_rand_scaled) * dx) * 1.0
    init_strategies.append(np.clip(h_rand_scaled, 0, 1))
    
    # First, run local optimization with all initial values
    for h_init in init_strategies:
        try:
            local_result = minimize(
                objective,
                h_init,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Then use differential evolution for global exploration
    # With reduced budget for DE since we're already at good local solutions
    de_budget = budget_s * 0.5
    de_start = time.time()
    
    try:
        # Try multiple DE starts with different settings
        de_options = [
            {'maxiter': 50, 'popsize': 10, 'seed': seed},
            {'maxiter': 30, 'popsize': 15, 'seed': seed+1},
        ]
        
        for de_opt in de_options:
            if time.time() - de_start > de_budget * 0.8:
                break
                
            de_result = differential_evolution(
                objective,
                bounds,
                maxiter=de_opt['maxiter'],
                popsize=de_opt['popsize'],
                seed=de_opt['seed'],
                callback=lambda x, convergence: None,
                atol=1e-6,
                rtol=1e-6
            )
            if de_result.success:
                h_opt = np.clip(de_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
    except:
        pass
    
    # Try alternating local optimizations with global search
    # Fine-tune with local optimizer after each DE pass or other global method
    steps = 0
    while time.time() - start_time < budget_s * 0.95:
        steps += 1
        
        # Use a random restart or local refinement
        if steps % 2 == 0:
            # Try a random different initialization
            h_new = np.random.rand(n_points) * 0.8 + 0.1
            h_new = h_new / (np.sum(h_new) * dx) * 1.0
            h_new = np.clip(h_new, 0, 1)
        else:
            # Randomly perturb best solution
            h_new = best_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h_new = h_new + noise
            h_new = np.clip(h_new, 0, 1)
            # Re-scale to satisfy constraint
            h_new = h_new / (np.sum(h_new) * dx) * 1.0
            h_new = np.clip(h_new, 0, 1)
            
        try:
            local_result = minimize(
                objective,
                h_new,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final optimization with increased precision
    try:
        local_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
