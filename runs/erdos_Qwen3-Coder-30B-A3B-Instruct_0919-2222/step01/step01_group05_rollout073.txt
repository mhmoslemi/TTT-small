class Pack:
    def __init__(self, h=None, n=None):
        self.h = h
        self.n = n

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def lp_initialization(n_points):
    """Initialize h using a linear programming relaxation with McCormick envelopes"""
    # Variables
    h = cp.Variable(n_points, bounds=(0, 1))
    w = n_points / 2
    
    # Initialize all u_{i,d}'s (these are auxiliary variables;
    # don't bother creating all n^2 of them here, just tally the number required)
    u = {}
    
    # Set up problem variables for correlations
    total_vars = 0
    # Max d: n-1 so we need full deck from -n+1 to n-1
    # Number of required pairs for each d
    offset = n_points - 1
    s_vars = {}  # s_{d} = sum_{i} u_{i,d}  
    t = cp.Variable()
    
    # Initialize constraints into groups
    # Formulate all constraints piecewise
    constraints = []
    constraints += [cp.sum(h) == w]
    
    # Try to start with cheap propagation
    for d in range(-offset, offset + 1):
        # only look for overlapping ones
        first_idx = max(0, -d)
        last_idx = min(n_points, n_points - d)
        if last_idx <= first_idx:
            continue  # no intersections
        sd_vars = []
        i_pairs = []
        for i in range(first_idx, last_idx):
            u_idx = (i, d)
            u_node = cp.Variable(uplo='lower', shape=(2, 2))
            u[u_idx] = cp.Variable(bounds=(0, 1))
            i_pairs.append((i, u[u_idx]))
            
            constraints.append(u[u_idx] <= h[i])
            constraints.append(u[u_idx] <= h[i+d])
            constraints.append(u[u_idx] >= h[i] + h[i+d] - 1)
            constraints.append(u[u_idx] >= 0)
            
            sd_vars.append(u[u_idx])
            
        s_vars[d] = cp.sum(sd_vars)
        constraints.append(t >= w - s_vars[d])
        
    # Objective
    objective = cp.Minimize(t)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
    except:
        pass
    # Fallback
    return np.ones(n_points) * 0.5

def project_to_simplex(h, total):
    """Project h to simplicial constraint ∑h = total"""
    mean_deviation = (np.sum(h) - total) / len(h)
    h_proj = h - mean_deviation
    h_proj = np.clip(h_proj, 0, 1)
    factor = total / np.sum(h_proj)
    h_proj *= factor
    return h_proj

def swap_improve(h, d_star):
    """Improve by a targeted worst shift swap (binary only)"""
    n = len(h)
    h_copy = h.copy()
    
    # Find all indices where we can potentially swap
    # For a specific fixed d_star, i must be such that:
    # h[i] == 1, h[i+d_star] == 0
    # j must be: h[j] == 0, h[j+d_star] == 1
    
    # Precompute shifted index list
    valid_shifts = []
    for i in range(n):
        idx0 = i + d_star
        if 0 <= idx0 < n:
            valid_shifts.append((i, idx0))
            
    swaps = []
    for (i, j) in valid_shifts:
        if h_copy[i] == 1 and h_copy[j] == 0:
            swaps.append((i, j))
    
    if not swaps:
        return h_copy
    
    # Try all swaps
    best = h_copy.copy()
    current_c5 = evaluate_erdos_solution(best)
    
    for i, j in swaps:
        trial = h_copy.copy()
        trial[i], trial[j] = trial[j], trial[i]  # swap
        trial = project_to_simplex(trial, n/2)
        trial_c5 = evaluate_erdos_solution(trial)
        if trial_c5 < current_c5:
            best = trial
            current_c5 = trial_c5
            
    return best

def best_binary_from_float(h_float, n_points):
    """Convert float h to a good approximation binary h"""
    h = np.array(h_float)
    weights = np.argsort(h)[::-1]
    ones = weights[:int(n_points / 2)]
    h_bin = np.zeros(n_points)
    h_bin[ones] = 1.0
    return project_to_simplex(h_bin, n_points / 2)

def sa_move(h, delta_max):
    """Move by transferring mass between indices"""
    n_points = len(h)
    i, j = np.random.choice(n_points, size=2, replace=False)
    delta = min(delta_max, h[i], 1.0 - h[j])
    if delta <= 0:
        return h.copy()
    h[i] -= delta
    h[j] += delta
    h = np.clip(h, 0, 1)
    return project_to_simplex(h, n_points / 2)

def sa_anneal(h_start, n_points, budget_s):
    """Run simulated annealing on h"""
    start_t = time.time()
    max_time = budget_s * 0.40  # 40% for this phase
    h_current = h_start.copy()
    c5_current = evaluate_erdos_solution(h_current)
    T = 0.05 * c5_current  # Initial temperature based on current value
    c = 0.995  # Cooling coefficient
    
    delta_max = 0.2  # Start small and decay
    i = 0
    
    while time.time() - start_t < max_time and i < 3000:
        new_h = sa_move(h_current, delta_max)
        new_c5 = evaluate_erdos_solution(new_h)
        delta = new_c5 - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = new_h
            c5_current = new_c5
        T *= c
        delta_max *= 0.999
        i += 1
    return h_current, c5_current

def subgradient_descent(h_start, n_points, budget_s):
    """Projected sub-gradient descent to reduce C5"""
    start_t = time.time()
    max_steps = 1000
    eta = 1e-2  # learning rate
    h = h_start.copy()
    h = project_to_simplex(h, n_points / 2)  # normalize
    c5 = evaluate_erdos_solution(h)
    
    for _ in range(max_steps):
        if time.time() - start_t > budget_s * 0.1:
            break
        # Compute the full correlation and find worst d
        corr = np.correlate(h, 1 - h, mode='full')
        d_star = np.argmax(corr)
        # Compute subgradient term
        g = np.zeros(n_points)
        # For shift +d_star (positive)
        for i in range(n_points):
            j = i + d_star
            if 0 <= j < n_points:
                g[i] += -h[j]
            # For shift -d_star (negative)
            j = i - d_star
            if 0 <= j < n_points:
                g[i] += -h[j]
        # Update h
        h_new = h - eta * g
        h_new = project_to_simplex(h_new, n_points / 2)
        new_c5 = evaluate_erdos_solution(h_new)
        if new_c5 < c5:
            h = h_new
            c5 = new_c5
        else:
            eta *= 0.99  # adaptive decrease
    return h, c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    time_free = budget_s - 0.5
    
    # Try n_values with increasing size
    n_values = [42, 48, 54, 60]  # sorted by size

    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    start_time = time.time()
    
    for n in n_values:
        if time.time() - start_time > time_free * 0.90:
            break

        # Reset dx as needed
        dx = 2.0 / n

        # Step 1. LP Initialization
        h_init = lp_initialization(n)
        if h_init is None:
            h_init = np.ones(n) * 0.5
        # Normalize to sum to n/2
        h_init *= n / (2 * np.sum(h_init))
        
        # Step 2. Binary rounding
        h_binary = best_binary_from_float(h_init, n)
        # Compute its C5
        c5_binary = evaluate_erdos_solution(h_binary)
        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_binary.copy()
            best_n = n
        
        # Step 3. Targeted swap improvement
        corr = np.correlate(h_binary, 1 - h_binary, mode='full')
        d_best = np.argmax(corr)
        swapped = swap_improve(h_binary, d_best)
        swapped_c5 = evaluate_erdos_solution(swapped)
        if swapped_c5 < best_c5:
            best_c5 = swapped_c5
            best_h = swapped.copy()
            best_n = n
            
        # Step 4. Simulated Annealing (only if still good enough)
        if swapped_c5 < 0.381:
            sa_h, sa_c5 = sa_anneal(swapped, n, time_free)
            if sa_c5 < best_c5:
                best_c5 = sa_c5
                best_h = sa_h
                best_n = n
        
        # Step 5. Gradient descent refinement
        grad_h, grad_c5 = subgradient_descent(best_h, n, time_free)
        if grad_c5 < best_c5:
            best_c5 = grad_c5
            best_h = grad_h
            best_n = n
            
    # Final correction (exact normalization)
    if best_h is not None:
        # Ensure sum = n/2
        factor = (best_n / 2.0) / np.sum(best_h)
        best_h = best_h * factor
        
    return (best_h, best_c5, best_n)
