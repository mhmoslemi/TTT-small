import numpy as np
import time
import cvxpy as cp
from cvxpy import GLPK, GLPK_MI, ECOS, SCS

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    # Set random seeds
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Choose base resolution
    n_points = 70
    
    # Keep track of best solution
    best_h = None
    best_c5 = float('inf')
    
    # 1. Binomial MILP
    solver = GLPK_MI
    try:
        # Limit time for MILP
        time_milp = min(120, budget_s * 0.2)
        time_left = budget_s - (time.time() - start_time)
        if time_left < 20:
            raise TimeoutError()
        cur_time = time.time()
        # Try smaller first
        for nTry in [60, 70]:
            if time.time() - cur_time > time_milp:
                break
            try:
                h_bin = np.zeros(nTry)
                vars_h = cp.Variable(nTry, boolean=True)
                t = cp.Variable()
                # Shift indices
                shift_range = list(range(-nTry+1, nTry))  # From -69 to 69
                var_y = {}  # y[i, d] = h[i]*h[i+d]
                for d in shift_range:
                    var_y[d] = cp.Variable(nTry - abs(d), boolean=True)
                # Constraints by shift
                obj_rows = []
                for d in shift_range:
                    i_valid = range(max(0, -d), min(nTry, nTry - d))
                    if len(i_valid) == 0:
                        continue
                    expr = cp.sum(var_y[d])
                    const = cp.Variable()
                    obj_rows.append(t >= 1.0 / nTry * (nTry / 2 - expr))
                # McCormick for y[i,d]
                for d in shift_range:
                    i_valid = range(max(0, -d), min(nTry, nTry - d))
                    if len(i_valid) == 0:
                        continue
                    for i in i_valid:
                        k = i + d
                        var_y[d][i] <= vars_h[i]
                        var_y[d][i] <= vars_h[k]
                        var_y[d][i] >= vars_h[i] + vars_h[k] - 1
                        var_y[d][i] >= 0
                # Sum constraint
                cons_sum = cp.sum(vars_h) == nTry / 2
                # Maximize h 1s
                prob = cp.Problem(cp.Minimize(t), obj_rows + [cons_sum])
                prob.solve(solver=solver, time_limit=time_milp, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_opt = vars_h.value.astype(float)
                    if h_opt is not None:
                        h_opt = h_opt.astype(float)
                        c5 = evaluate_erdos_solution(h_opt)
                        if c5 < best_c5:
                            best_h = h_opt
                            best_c5 = c5
                if prob.status == cp.OPTIMAL or prob.status == cp.SOLVER_ERROR:
                    break
            except Exception as e:
                pass
                
        if best_c5 < 0.4:
            # Early return if MILP succeeded
            return best_h.tolist(), best_c5, len(best_h)
    except:
        pass

    # 2. Continuous LP relaxation        
    if time.time() - start_time > budget_s - 5:
        return best_h.tolist(), best_c5, len(best_h) if best_h is not None else 70
    try:
        # Attempt higher resolution LP
        n = 80
        h_continuous = cp.Variable(n, name='h')
        t = cp.Variable(name='t')
        
        # Create shifted correlations as linear functions
        shift_indices = list(range(-n + 1, n))
        obj_rows = []
        
        for d in shift_indices:
            start_idx = max(0, -d)
            end_idx = min(n, n - d)
            if end_idx <= start_idx:
                continue
            # Sum of h[i]*h[i+d] for valid indices
            expr = cp.sum(cpmultiply(h_continuous[start_idx:end_idx], 
                                     h_continuous[start_idx+d:end_idx+d]))
            obj_rows.append(t >= 1.0/n*(n/2 - expr))
            
        # Other constraints
        cons_box = [h_continuous >= 0, h_continuous <= 1]
        cons_sum = cp.sum(h_continuous) == n/2
        
        prob = cp.Problem(cp.Minimize(t), obj_rows + cons_box + cons_sum)
        prob.solve(solver=GLPK, time_limit=budget_s * 0.2, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_cand = h_continuous.value
            c5 = evaluate_erdos_solution(h_cand)
            if c5 < best_c5:
                best_h = h_cand
                best_c5 = c5
    except:
        pass
    
    if best_h is None:
        # Fallback
        best_h = np.ones(70) * 0.5 if n_points == 70 else np.ones(80) * 0.5
        best_h *= (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h)
    
    if time.time() - start_time > budget_s - 5:
        return best_h.tolist(), best_c5, len(best_h)

    # 3. SCC refinement (Piecewise linear relaxation)  
    h = best_h.copy()
    dx = 2.0 / len(h)
    
    # Iteration loop
    max_iter = 10000
    t_prev = float('inf')
    for iter_k in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        current_c5 = evaluate_erdos_solution(h)
        if abs(current_c5 - t_prev) < 1e-6:
            break
        t_prev = current_c5
        
        # Compute all shift correlations
        h = h.copy()
        c_original = np.correlate(h, 1 - h, mode='full')
        # Get range
        correlations = c_original * dx
        max_corr_idx = np.argmax(correlations)
        d_worst = max_corr_idx - len(h) + 1  # Convert index to shift
        
        # Approximate h[i]*h[i+k] around current h
        # Linearisons using Taylor series
        # Q[d] = Sum(h[i] * h[i+d])
        # To linearize h[i]*h[i+d] = 
        #   h[i]*h[i+d]_old + h[i]_old *(h[i+d] - h[i+d]_old) + 
        #   h[i+d]_old *(h[i]  - h[i]_old)  
        
        # Compute linear approximation: h[i]*h[i+d] ≈ coeff_const + coeff1 * h[i] + coeff2 * h[i+d]
        h_new = cp.Variable(n_points, name="h_new")
        w_t = cp.Variable(1, name="t")
        
        # Compute the exact worst shift (for now)
        s_static = np.roll(h, -d_worst)
        # Constraint for that shift only
        corr_worst = []
        for i in range(len(h)):
            if i + d_worst >= 0 and i + d_worst < len(h):
                corr_worst.append(h[i] * h[i + d_worst])
        corr_sum = sum(corr_worst)
        # So h[i]*h[i+d_worst] ≈ h_old[i]*h_old[i+d_worst] + ...
        # MP:   Δv = v_{old} * (v_new - v_old) + v_new * (v_old - v_new) 
        # Simplified:
        expr_approx = 0.
        for i in range(len(h)):
            if i + d_worst >= 0 and i + d_worst < len(h):
                expr_approx += h[i] * h[i + d_worst]  # Just not linearize, do it exactly for simplicity
        # Approximate Q:
        # Ugly but will use a convex formulation directly
        # Problem is: Max all shifts, but that's not convex.
        # We do sequential linearization on one of the worst.
        
        # Better: we build a simple primal LP with one of the shifts as single active constraint
        common_term = (n_points / 2) - corr_sum # Known constant in this context  
        rows = [w_t >= dx * common_term]  # for the worst
        rows.append(h_new >= 0)
        rows.append(h_new <= 1)
        rows.append(cp.sum(h_new) == n_points / 2)
        
        # Rebuild normal linearized problem using our h
        constr_eq = [cp.sum(h_new) == n_points / 2, h_new >= 0, h_new <= 1]
        # Only the one that gives highest value now 
        max_shif = -d_worst  # i+k = i+d , so k=-d
        shift_range_scc = range(-len(h)+1, len(h))
        loss_terms = []
        expr_old = cp.Constant(0)
        for d in shift_range_scc:
            # Add constraint approx - but not a forbidden shift?
            if d == max_shif:
                # For computing C(d) := dx*(n/2 - sum h_i * h_{i+d})
                i_range = list(range(max(0, -d), min(len(h), len(h)-d)))
                if len(i_range) > 0:
                    y = cp.Variable(len(i_range))
                    for i in i_range:
                        k = i+d
                        if k >= 0 and k < len(h):
                            # Mixed binarization trick for taylor expansion
                            coeffs = [h[i], h[k]]
                            # Actual floor/ceil skip - loss of lines
                            # Use a more elaborate approx will just use Taylor around current value
                            y[i - max(0, -d)] = h[i]*h[k]
                            expr_old += y[i - max(0, -d)]
                    # Approximate:
                    approx_q = expr_old
                    loss_terms.append(dx * ((n_points / 2) - approx_q))
        loss_terms = [w_t >= t for t in loss_terms]
        prob_primal = cp.Problem(cp.Minimize(w_t), [h_new == h] + constr_eq)
        prob_primal.solve(solver=ECOS, time_limit=5, verbose=False)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
    # 4. Final polishing
    if time.time() - start_time < budget_s - 10:
        h = best_h.copy()
        t_end = time.time() + 5
        while time.time() < t_end:
            i, j = np.random.choice(len(h), 2, replace=False)
            delta = np.random.uniform(-0.05, 0.05)
            if h[i] + delta >= 0 and h[j] - delta >= 0 and h[i] + delta <= 1 and h[j] - delta <= 1:
                h[i] += delta
                h[j] -= delta
                # Maintain sum
                h[:i] = h[:i] / np.sum(h[:i]) * (n_points / 2 - h[i] - h[j] + h[i] + h[j])
                h[i] = h[i] + delta
                h[j] = h[j] - delta
                # Re-norm to sum
                total = np.sum(h)
                h *= (n_points / 2) / total
                new_c5 = evaluate_erdos_solution(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
    
    return best_h.tolist(), best_c5, len(best_h)
