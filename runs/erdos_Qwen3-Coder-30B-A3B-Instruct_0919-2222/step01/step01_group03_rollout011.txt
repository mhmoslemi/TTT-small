import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector v onto [0,1] hypercube with sum constraint"""
    # First clip to [0,1]
    v = np.clip(v, 0, 1)
    # Adjust to match target sum
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    # Use water-filling approach
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        # Sort indices by current value and increase below 1
        idx = np.argsort(v)
        for i in idx:
            if diff <= 0:
                break
            max_increase = 1.0 - v[i]
            increase = min(max_increase, diff)
            v[i] += increase
            diff -= increase
    else:
        # Need to decrease values
        # Sort indices by current value and decrease above 0
        idx = np.argsort(v)[::-1]
        for i in idx:
            if diff >= 0:
                break
            max_decrease = v[i]
            decrease = max(-max_decrease, diff)
            v[i] += decrease
            diff -= decrease
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate shift indices for gradient computation
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points - s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))

    # Define the objective function for optimization
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Simple gradient descent with projection
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Attempt continuous optimization with gradient descent
    max_iter = int(2e5)
    eta0 = 0.1
    beta_max = 100
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute current correlations and objective values
        correlations = np.correlate(best_h, 1 - best_h, mode="full")
        f_values = correlations / dx  # Raw integral values without dx
        max_f = np.max(f_values)
        
        # Softmax weights with temperature beta
        beta = 1.0 + (beta_max - 1.0) * (iter / max_iter)
        exp_terms = np.exp(beta * (f_values - max_f))
        weights = exp_terms / np.sum(exp_terms)
        
        # Compute gradient
        gradient = np.zeros_like(best_h)
        for shift_idx, (i_arr, j_arr) in enumerate(shift_indices):
            weight = weights[shift_idx]
            if weight > 1e-12:
                gradient[i_arr] += weight * (1.0 - best_h[j_arr])
                gradient[j_arr] -= weight * best_h[i_arr]
        
        # Determine step size with decreasing schedule
        eta = eta0 / (1.0 + iter**0.5)
        
        # Update
        h_new = best_h - eta * gradient
        h_new = project_onto_bounds_and_sum(h_new, target_sum)
        
        # Evaluate new step
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Try random restarts with different initializations to escape local minima
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate random initialization
        h_rand = np.random.rand(n_points) * 0.5  # Between 0 and 0.5
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        # Apply gradient descent to this initialization
        h_curr = h_rand.copy()
        for iter in range(1000):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            correlations = np.correlate(h_curr, 1 - h_curr, mode="full")
            f_values = correlations / dx
            max_f = np.max(f_values)
            
            beta = 1.0 + (beta_max - 1.0) * (iter / 1000.0)
            exp_terms = np.exp(beta * (f_values - max_f))
            weights = exp_terms / np.sum(exp_terms)
            
            gradient = np.zeros_like(h_curr)
            for shift_idx, (i_arr, j_arr) in enumerate(shift_indices):
                weight = weights[shift_idx]
                if weight > 1e-12:
                    gradient[i_arr] += weight * (1.0 - h_curr[j_arr])
                    gradient[j_arr] -= weight * h_curr[i_arr]
            
            eta = 0.05 / (1.0 + iter**0.5)
            h_new = h_curr - eta * gradient
            h_new = project_onto_bounds_and_sum(h_new, target_sum)
            h_curr = h_new.copy()
            
            if iter % 100 == 0:
                c5_curr = evaluate_erdos_solution(h_curr)
                if c5_curr < final_c5:
                    final_c5 = c5_curr
                    best_h = h_curr.copy()
    
    return (best_h, final_c5, n_points)
