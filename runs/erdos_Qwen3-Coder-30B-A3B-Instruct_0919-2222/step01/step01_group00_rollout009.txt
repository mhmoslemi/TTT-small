import numpy as np
import time
from scipy.fft import rfft, irfft

def proj_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= v_i <= 1 and sum v_i = target_sum"""
    v = np.array(v, dtype=float)
    if np.isclose(np.sum(v), target_sum):
        return v
    # water-filling approach
    v = np.sort(v)[::-1]
    cumsum = np.cumsum(v)
    n = len(v)
    lam = (cumsum - target_sum) / np.arange(1, n+1)
    # find largest j such that v[j] > lam[j]
    mask = v > lam
    if not np.any(mask):
        # all elements are zero
        return np.zeros_like(v)
    j = np.argmax(mask)
    if j == 0:
        # all elements may be adjusted
        lambda_val = lam[0]
    else:
        lambda_val = lam[j-1]
    new_v = np.clip(v - lambda_val, 0, 1)
    if not np.isclose(np.sum(new_v), target_sum, atol=1e-12):
        # fallback to rescaling + clipping
        s = np.sum(new_v)
        if s > 1e-10:
            new_v = new_v * (target_sum / s)
        new_v = np.clip(new_v, 0, 1)
    return new_v

def full_corr(h):
    """Fast correlation using FFT: correlate h with (1-h)"""
    n = len(h)
    H = rfft(h, n=2*n-1)
    G = rfft(1-h, n=2*n-1)
    corr = irfft(H * np.conj(G), n=2*n-1)
    return corr

def compute_C5_fast(h):
    """Evaluate C5 using FFT correlation."""
    corr = full_corr(h)
    c5 = np.max(corr) * (2.0 / len(h))
    return c5

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def sa_search(h_s, n_points, target_sum, n_iter=50000, T0=0.1, alpha=0.997, delta_max=0.2, seed=42):
    """Simulated Annealing implementation with block-level changes."""
    np.random.seed(seed)
    best_h_s = h_s.copy()
    best_C5 = compute_C5_fast(np.repeat(best_h_s, n_points // len(best_h_s))) if len(best_h_s) < n_points else compute_C5_fast(best_h_s)
    h_s = best_h_s.copy()
    T = T0
    for i in range(n_iter):
        if i % 10000 == 0:
            if time.time() > time.time() + 10: # check if we will run out of time
                break

        # Proposal
        i1, i2 = np.random.choice(len(h_s), 2, replace=False)
        delta = np.random.uniform(-delta_max, delta_max)
        new_h_s = h_s.copy()
        new_h_s[i1] += delta
        new_h_s[i2] -= delta

        # projection
        new_h_s = proj_capped_simplex(new_h_s, target_sum)

        # evaluate
        new_h = np.repeat(new_h_s, n_points // len(new_h_s))
        new_C5 = compute_C5_fast(new_h)

        # accept
        delta_C5 = new_C5 - best_C5
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h_s = new_h_s.copy()
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h_s = h_s.copy()

        T *= alpha
        if T < 1e-4:
            break

    return best_h_s, best_C5

def pgd_refine(h_s, n_points, target_sum, alpha0=5, alpha_factor=1.5, max_epochs=200, eta0=0.05, eta_decay=0.95, seed=42):
    """Projected gradient descent with smooth max."""
    np.random.seed(seed)
    h_s = proj_capped_simplex(h_s, target_sum)
    h = np.repeat(h_s, n_points // len(h_s))
    
    alpha = alpha0
    eta = eta0
    h_s = h_s.copy()
    
    for epoch in range(max_epochs):
        if time.time() > time.time() + 10: # check if we will run out of time
            break

        # Full correlation
        corr = full_corr(h)
        # Smooth max
        c_max = np.max(corr)
        corr_shift = corr - c_max
        w = np.exp(alpha * corr_shift)
        w = w / np.sum(w)

        # Gradient
        n = len(h)
        g = np.zeros_like(h)
        for idx in range(len(corr)):
            k = idx - (n - 1)
            if k > 0:
                if idx + k < len(corr):
                    k_neg = idx - k
                    g[k_neg] += w[idx]
                if idx >= k:
                    g[idx - k] += w[idx]
            else:
                if idx + abs(k) < len(corr):
                    g[idx + abs(k)] += w[idx]
                if idx >= abs(k):
                    k_neg = idx - abs(k)
                    g[k_neg] += w[idx]
        g = 1.0 - g  # Gradient of smooth max with respect to h

        # Collapse gradient to block level
        block_len = n_points // len(h_s)
        g_s = np.array([np.mean(g[i*block_len:(i+1)*block_len]) for i in range(len(h_s))])
        
        # update
        h_s_new = h_s - eta * g_s
        h_s = proj_capped_simplex(h_s_new, target_sum)
        h = np.repeat(h_s, block_len)
        
        # adapt learning rate
        eta *= eta_decay
        # increase smoothing parameter
        if epoch % 10 == 0:
            alpha *= alpha_factor
            if alpha > 200:
                alpha = 200

    return h_s, compute_C5_fast(np.repeat(h_s, block_len))

def binary_hill_climb(n_points, max_iter=500000, seed=42):
    """Binary hill-climbing."""
    np.random.seed(seed)
    # Generate random binary with half 1s
    h = np.zeros(n_points)
    ones = np.random.choice(n_points, n_points//2, replace=False)
    h[ones] = 1
    C5 = compute_C5_fast(h)
    best_h = h.copy()
    best_C5 = C5
    for i in range(max_iter):
        if i % 10000 == 0:
            if time.time() > time.time() + 10:
                break
        # pick two indices: one 1, one 0
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            continue
        i1 = np.random.choice(ones_indices)
        i2 = np.random.choice(zeros_indices)
        # swap
        h[i1] = 0
        h[i2] = 1
        C5_new = compute_C5_fast(h)
        if C5_new < best_C5:
            best_h = h.copy()
            best_C5 = C5_new
        else:
            # restore
            h[i1] = 1
            h[i2] = 0
    return best_h, best_C5
