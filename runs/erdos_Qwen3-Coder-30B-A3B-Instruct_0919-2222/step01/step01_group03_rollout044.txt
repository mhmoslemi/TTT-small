import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist, squareform
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # First, try with a large number of iterations with differential evolution
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=20,
            seed=seed,
            atol=1e-8,
            rtol=1e-8
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except Exception as e:
        pass

    # Then local optimization using L-BFGS-B for better precision
    try:
        local_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except Exception as e:
        pass

    # Generate random initializations and refine them
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perturb the best solution
        h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Refine with constraint optimization
        try:
            # Normalize to satisfy integral constraint
            h_perturbed = h_perturbed / (np.sum(h_perturbed) * dx) * 1.0
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
            local_result = minimize(
                objective,
                h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1500, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except Exception as e:
            pass

    # Try a least-squares approach with local search variation
    try:
        # Strengthen the local search by starting from a highly optimized point
        h_candidate = best_h.copy()
        
        # Perform a few iterations of gradient descent with projection
        for i in range(1000):
            # Compute gradient approximately using finite difference
            h_eval = np.copy(h_candidate)
            if i % 50 == 0 and time.time() - start_time > budget_s * 0.95:
                break
            c5_current = objective(h_eval)
            epsilon = 1e-5
            grad = np.zeros_like(h_eval)
            for j in range(len(h_eval)):
                h_plus = np.copy(h_eval)
                h_plus[j] += epsilon
                h_plus = np.clip(h_plus, 0, 1)
                c5_plus = objective(h_plus)
                grad[j] = (c5_plus - c5_current) / epsilon

            # Gradient ascent (as we want to minimize)
            h_new = h_eval - 0.001 * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Constraint ensuring integral remains 1.0
            h_new = h_new / (np.sum(h_new) * dx) * 1.0
            
            # Check if we've converged
            if np.linalg.norm(h_new - h_eval) < 1e-8:
                break
                
            h_candidate = h_new
            
        c5_final = objective(h_candidate)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_candidate.copy()
    except Exception as e:
        pass

    return (best_h, best_c5, n_points)
