import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    correlations = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    # Use a more sophisticated initial construction
    n = 80
    h = np.zeros(n)
    # Create a piecewise constant function with strategic values
    # Divide into 5 segments
    segment_length = n // 5
    h[:segment_length] = 0.7
    h[segment_length:2*segment_length] = 0.3
    h[2*segment_length:3*segment_length] = 0.7
    h[3*segment_length:4*segment_length] = 0.3
    h[4*segment_length:] = 0.7
    # Scale to ensure integral equals 1
    h = h * (n / (2 * np.sum(h)))
    return h

def project_onto_bounds_and_sum(v, target_sum):
    # Project vector v onto [0,1] and maintain sum constraint
    # First clip to [0,1]
    v = np.clip(v, 0, 1)
    # Adjust to hit target sum using water-filling approach
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    elif current_sum > target_sum:
        # Need to decrease values
        # Sort in decreasing order to identify which to reduce
        indices = np.argsort(v)[::-1]  # descending order
        remaining_sum = current_sum - target_sum
        for i in indices:
            if remaining_sum <= 0:
                break
            decrease = min(v[i], remaining_sum)
            v[i] -= decrease
            remaining_sum -= decrease
    else:
        # Need to increase values
        indices = np.argsort(v)  # ascending order
        remaining_sum = target_sum - current_sum
        for i in indices:
            if remaining_sum <= 0:
                break
            increase = min(1 - v[i], remaining_sum)
            v[i] += increase
            remaining_sum -= increase
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Precompute shift index pairs for efficiency
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points-s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))
    
    # Objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # We'll use a combination of gradient descent and local search
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Gradient descent approach with projected updates
    max_iter = 30000
    eta0 = 0.3
    beta = 1.0
    beta_max = 100.0
    beta_increment = (beta_max - beta) / (max_iter // 200)
    
    # Initialize
    h = h0.copy()
    h = project_onto_bounds_and_sum(h, n_points / 2)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Bookkeeping
    hist_c5 = []
    hist_iter = []
    
    # Optimization using projected gradient descent with softmax smoothing
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation values
        corr = np.correlate(h, 1-h, mode="full")
        # Compute f values (unweighted sum inside correlation)
        f = corr
        
        # Softmax weights
        max_f = np.max(f)
        exp_terms = np.exp(beta * (f - max_f))
        w = exp_terms / np.sum(exp_terms)
        
        # Compute gradient
        grad = np.zeros(n_points)
        # Constant term from sum_i h_i
        grad += np.sum(w)  # This equals 1 due to normalization
        
        # Product terms from sum_i h_i * h_{i+s}
        for s, (i, j) in enumerate(shift_indices):
            weight = w[s]
            if weight == 0:
                continue
            # Gradient contribution from h_i * (1 - h_{i+s})
            # This is (1 - h_{i+s}) for h_i and (-h_{i+s}) for h_{i+s}
            grad[i] += weight * (1 - h[j])
            grad[j] += weight * (-h[i])
        
        # Step with adaptive learning rate
        eta = eta0 / np.sqrt(it + 1)
        h_new = h - eta * grad
        
        # Project to feasible region
        h_new = project_onto_bounds_and_sum(h_new, n_points / 2)
        h = h_new
        
        # Update beta
        if it > 0 and it % 200 == 0:
            beta = min(beta + beta_increment, beta_max)
        
        # Record progress
        if it % 1000 == 0:
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                hist_c5.append(best_c5)
                hist_iter.append(it)
        
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
    
    # Do a final boost using local tweak
    try:
        # Random local search for boost
        for _ in range(2000):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Random perturbation within bounds
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            delta = np.random.uniform(-0.05, 0.05)
            h_test = best_h.copy()
            old_val1 = h_test[idx1]
            old_val2 = h_test[idx2]
            h_test[idx1] = np.clip(h_test[idx1] + delta, 0, 1)
            h_test[idx2] = np.clip(h_test[idx2] - delta, 0, 1)
            
            # Maintain integral constraint via adjustment
            diff = (h_test[idx1] - old_val1) + (h_test[idx2] - old_val2)
            total_adjust = diff
            if total_adjust != 0:
                # Distribute the adjustment to other points with small weight to keep sum near target
                adjust_indices = list(range(n_points))
                adjust_indices.remove(idx1)
                adjust_indices.remove(idx2)
                if adjust_indices:
                    # Pick one additional index to adjust
                    adjust_idx = adjust_indices[np.random.randint(len(adjust_indices))]
                    h_test[adjust_idx] = np.clip(h_test[adjust_idx] - diff, 0, 1)
            
            # Re-project to meet sum constraint
            h_test = project_onto_bounds_and_sum(h_test, n_points / 2)
            test_c5 = evaluate_erdos_solution(h_test)
            if test_c5 < best_c5:
                best_c5 = test_c5
                best_h = h_test.copy()
    except:
        pass
    
    return (best_h, best_c5, n_points)
