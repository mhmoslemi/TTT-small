import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_and_lag(h):
    """Compute C5 and the lag(s) that achieve the maximum overlap"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    c5_value = np.max(corr) * dx
    # Find all lags that achieve the maximum
    max_corr = np.max(corr)
    best_lags = np.where(np.abs(corr - max_corr) < 1e-10)[0]
    return c5_value, best_lags

def project_onto_box_and_sum(v, target):
    """Project vector onto box [0,1] and sum constraint"""
    # Find λ s.t. sum_i clip(v_i + λ, 0, 1) = target.
    # Use binary search
    lo = -np.max(v) - 2.0  # ensure sum lower bound zero
    hi = 1 - np.min(v) + 2.0 # ensure sum upper bound n
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def grad_for_lag(h, lag):
    """Compute gradient of correlation at a specific lag"""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # Overlap region i in [0, N - lag)
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]  # derivative w.r.t h_i
        grad[i + lag] -= h[i]      # derivative w.r.t h_{i+lag}
    else:
        lag = -lag
        # Overlap region i in [lag, N)
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad

def random_binary(N, weight):
    """Generate a random binary vector with specified weight"""
    b = np.zeros(N, dtype=int)
    ones = np.random.choice(N, size=int(weight), replace=False)
    b[ones] = 1
    return b

def compute_binary_min_overlap(b):
    """Compute minimum overlap (number of 1's) of binary vector with its shifts"""
    corr = np.correlate(b, b, mode='full')
    # Get positive lags
    min_overlap = np.min(corr[len(b):])
    return min_overlap

def binary_maximin(N, weight, time_limit=20):
    """Find a binary vector with maximized minimum overlap"""
    start_time = time.time()
    best_min_overlap = -1
    best_binary = None
    
    # Try a good batch of random binary vectors
    num_trials = min(5000, 1000000 // N)  # Adjust based on N size
    
    for _ in range(num_trials):
        if time.time() - start_time > time_limit:
            break
        b = random_binary(N, weight)
        min_overlap = compute_binary_min_overlap(b)
        if min_overlap > best_min_overlap:
            best_min_overlap = min_overlap
            best_binary = b.copy()
            
    return best_binary
