import numpy as np
import time
from scipy.optimize import minimize
from scipy.linalg import blas

def proj_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= v <= 1 and sum(v) = target_sum."""
    v = np.array(v).copy()
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    # Water-filling to find the threshold t
    cumsum = np.cumsum(v_sorted)
    thresholds = (cumsum - target_sum) / np.arange(1, n + 1)
    # Find largest k such that thresholds[k] <= v_sorted[k]
    k = 0
    for i in range(n):
        if thresholds[i] <= v_sorted[i]:
            k = i + 1
    if k == 0:
        # All values would need to be 0
        return np.zeros_like(v)
    t = thresholds[k-1]  # This is the threshold that makes sum exactly target_sum
    result = np.clip(v - t, 0, 1)
    return result

def full_corr(h):
    """Compute full correlation h * (1-h) using FFT"""
    n = len(h)
    H = np.fft.rfft(h, n=2*n-1)
    G = np.fft.rfft(1-h, n=2*n-1)
    corr = np.fft.irfft(H*np.conj(G), n=2*n-1)
    return corr.real

def evaluate_h(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    corr = full_corr(h)
    max_val = np.max(corr)
    return max_val * dx
