def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def compute_product_sums(h):
    n = len(h)
    # Compute sum_i h_i h_{i+s} for all shifts s = -(n-1) to (n-1)
    # This is equivalent to correlate h with itself (with zero padding)
    # But for our use case, we don't actually need to consider the circular case correctly.
    # Our sum ignores out-of-bound terms (h=0 there), so we compute sum over valid pairs.
    # Using np.correlate with mode="full" computes sum_i h[i] * h[i+s] over all i, j = i+s.
    # but due to zero padding, terms where h[i+s] = 0 are ignored. So we need to reimplement
    # using manual loop over shift range to avoid artifacts.
    
    # Instead, we'll use correlation appropriately: use full mode with zero padding
    # and then extract overlaps: the contribution of i to shift s is h[i] * h[i+s] 
    # if both i and i+s are in [0, n-1]. But forcing boundary to zero just means 
    # product contribution is zero if i+s out of range.
    
    # Therefore: for shift s in range -(n-1), (n-1):
    #   - For i in [0,n-1] such that 0 <= i+s <= n-1, add h[i] * h[i+s] to sum
    #   - Which is equivalent to support of evaluation being in plane [0,n-1]
    #   - The number of such i is n - |s| (for s in [-n+1, n-1]) 
    #   So direct computation via np.correlate gives us what we need for ends to 0
    
    # Actually, a very thorough correct computation is below:
    n = len(h)
    product_sums = np.zeros(2*n - 1) # for shifts -n+1 to n-1 inclusive
    
    # Loop over all valid shifts
    for s in range(-n+1, n):
        total = 0.0
        # Find valid i such that both i and i+s are in [0,n-1]
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                total += h[i] * h[j]
        product_sums[s + n - 1] = total
    
    return product_sums

def project_to_simplex(h, target_sum):
    # Uses iterative projection to project onto simplex with 
    # sum = target and 0 <= x_i <= 1
    # Algorithm: Binary search for lambda such that sum clip(h - lambda) = target
    n = len(h)  
    low = np.min(h) - 1.0
    high = np.max(h)
    
    def project_with_lambda(lam):
        proj = np.clip(h - lam, 0, 1)
        return np.sum(proj)
        
    for _ in range(50):
        mid = (low + high) / 2
        s = project_with_lambda(mid)
        if s > target_sum:
            low = mid
        else:
            high = mid
            
    final_proj = np.clip(h - low, 0, 1)  
    # Adjust slightly to ensure the exact sum target
    adjustment = (target_sum - np.sum(final_proj)) / len(final_proj)
    final_proj += adjustment
    
    # Projection with slight bounds clean-up
    final_proj = np.clip(final_proj, 0, 1)
    return final_proj
