import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def project_onto_simplex(h, target_sum):
    """Project h onto the simplex with given target sum"""
    h = h.copy()
    # Shift to make sum zero
    mean = np.sum(h) - target_sum
    h -= mean / len(h)
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # If still not exact, adjust carefully
    delta = target_sum - np.sum(h)
    if abs(delta) > 1e-10:
        # Distribute the residual
        idxs = np.argsort(h)
        if delta > 0:
            # Need to add
            for i in reversed(idxs):
                if delta <= 0:
                    break
                add = min(delta, 1 - h[i])
                h[i] += add
                delta -= add
        else:
            # Need to subtract
            for i in idxs:
                if delta >= 0:
                    break
                subtract = min(-delta, h[i])
                h[i] -= subtract
                delta += subtract
    return h

def solve_lp_relaxation(n_points):
    """Solve the LP relaxation with McCormick envelopes"""
    # Variables: h_i in [0,1], u_{i,d} (product approximations)
    h = cp.Variable(n_points)
    u = cp.Variable((n_points, 2 * n_points - 1))
    t = cp.Variable()
    
    # Constraint: sum(h) = n/2
    constraints = [cp.sum(h) == n_points / 2]
    constraints += [0 <= h[i] for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]
    
    # Add McCormick envelope constraints and compute sums
    # For each shift d=-(n-1) .. n-1
    # Index index d corresponds to column d+n-1 in u
    for d in range(-n_points + 1, n_points):
        col = d + n_points - 1
        for i in range(n_points):
            if 0 <= i+d < n_points:
                # The product h_i * h_{i+d} must be approximated by u_{i,d}
                # Importantly, use valid indices
                u_idx = i
                v_idx = i + d
                # McCormick envelopes:
                constraints.append(u[u_idx, col] >= 0)
                constraints.append(u[u_idx, col] <= h[u_idx])
                constraints.append(u[u_idx, col] <= h[v_idx])
                constraints.append(u[u_idx, col] >= h[u_idx] + h[v_idx] - 1)

    # For each shift d compute s_d = sum_{i} u_{i,d}
    for d in range(-n_points + 1, n_points):
        col = d + n_points - 1
        # Create sum over matching indices
        indices = [i for i in range(n_points) if 0 <= i+d < n_points]
        s_d = cp.sum([u[i, col] for i in indices])
        # C5(d) = w - s_d = n/2 - s_d
        constraints.append(t >= n_points / 2 - s_d)
    
    # Minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_sol = h.value
            return h_sol
        else:
            return None
    except Exception as e:
        return None

def worst_shift(h):
    """Returns the shift that maximizes integral h(x) * (1 - h(x+k))"""
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (len(h) -1)  # Convert to actual shift
    return shift, corr[max_idx]

def binary_rounding(h_float, n_points):
    """Round continuous h to binary with exact half sums"""
    # Get indices of largest values to set to 1
    idxs = np.argsort(h_float)[::-1][:n_points // 2]
    h_bin = np.zeros(n_points)
    h_bin[idxs] = 1.0
    return h_bin

def local_swap_improvement(h):
    """Hill climb with swap moves to reduce worst shift"""
    h_copy = h.copy()
    while True:
        worst_d, _ = worst_shift(h_copy)
        # Try to find a beneficial swap
        improved = False
        indices_one = np.where(h_copy == 1)[0]
        indices_zero = np.where(h_copy == 0)[0]
        
        # Build bounds: make sure index stays within [0, n_points)
        for i in indices_one:
            # check if h[i+worst_d] == 0
            j = i + worst_d
            if 0 <= j < len(h_copy) and h_copy[j] == 0:
                # try to swap
                h_copy[i] = 0;
                h_copy[j] = 1;
                improved = True
                break
            # Also backward check for h[i+worst_d] == 1
            j = i - worst_d
            if 0 <= j < len(h_copy) and h_copy[j] == 1:
                h_copy[i] = 0;
                h_copy[j] = 1;
                improved = True
                break
        if not improved:
            break
    return h_copy

def simulated_annealing(h_init, budget_s, start_time):
    """Simulated Annealing with incremental wave of moves"""
    h = h_init.copy()
    c5 = evaluate_erdos_solution(h)
    start_temp = 0.02 * c5 if c5 > 0 else 0.02
    temp = start_temp
    T0 = start_temp
    alpha = 0.995
    t0 = time.time()
    step_count = 0
    
    while True:
        step_count += 1
        # Time check
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        delta_max = min(0.2, 0.5)  # Ensure any small move is possible
        # Select indices
        i, j = np.random.choice(len(h), size=2, replace=False)
        
        # Compute change in objective
        old_val = h[i] * h[j]
        if i + j < len(h) and i+j >= 0:
            p1 = h[i] * h[i+j]
            p2 = h[j] * h[j+(j-i)]
        else:
            p1 = p2 = 0
            
        # Accept change if better or with some probability
        delta = -1 * old_val + new_val
        
        if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
            h[i] = max(0, min(1, h[i] - delta));
            h[j] = max(0, min(1, h[j] + delta));
            # Enforce sum constraint
            h = project_onto_simplex(h, len(h) / 2)
            c5 = evaluate_erdos_solution(h)
        else:
            pass  # do not accept
            
        # Update temperature
        temp = T0 * (alpha ** step_count)
    return h, c5

def projected_gradient_refinement(h, max_iter=300):
    """Subgradient descent with projections"""
    h_new = h.copy()
    
    for i in range(max_iter):
        # Compute worst shift
        _, _ = worst_shift(h_new)
        d = np.random.randint(-len(h_new)+1, len(h_new))
        # Approximate steps using parametrized derivative:
        # assume all shifts are relevant (simplifying assumption)
        g = np.zeros(len(h_new))
        for j in range(len(h_new)):
            k = j + d
            if 0 <= k < len(h_new):
                g[j] -= h_new[k]
            k = j - d
            if 0 <= k < len(h_new):
                g[j] -= h_new[k]
        # Gradient step
        h_new -= 0.1 * g
        h_new = project_onto_simplex(h_new, len(h_new) / 2)
        if i % 100 == 0:
            c5 = evaluate_erdos_solution(h_new)
            if abs(c5 - min(1.0, c5)) < 1e-6:
                break
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    start_time = time.time()
    
    best_c5 = float("inf")
    best_h = None
    best_n = None

    # Try a few resolutions with increasing complexity:
    candidate_n = [42, 48, 54, 60]  # n upper bound to avoid excessive runtimes
    for n_points in candidate_n:
        if time.time() - start_time > 0.98 * budget_s:
            break
        dx = 2.0 / n_points
        # Solve LP relaxation
        h_relaxed = solve_lp_relaxation(n_points)
        if h_relaxed is None or not np.isfinite(np.sum(h_relaxed)):
            continue  # skip
        # Penalize same value by small adjustment
        h_scaled = h_relaxed * (n_points / (2 * np.sum(h_relaxed)))
        score = evaluate_erdos_solution(h_scaled)
        if score < best_c5:
            best_c5 = score
            best_h = h_scaled.copy()
            best_n = n_points

        # Binary rounding & hill climbing
        h_binary = binary_rounding(h_scaled, n_points)
        h_improved = local_swap_improvement(h_binary)
        score_bin = evaluate_erdos_solution(h_improved)
        if score_bin < best_c5:
            best_c5 = score_bin
            best_h = h_improved.copy()
            best_n = n_points

        # Simulated Annealing on fractional input
        if time.time() - start_time > 0.95 * budget_s:
            break
        h_sa, score_sa = simulated_annealing(h_scaled, budget_s, start_time)
        if score_sa < best_c5:
            best_c5 = score_sa
            best_h = h_sa.copy()
            best_n = n_points

        # Gradient refinement
        if time.time() - start_time > 0.95 * budget_s:
            break
        h_refined = projected_gradient_refinement(h_sa, max_iter=200)
        score_new = evaluate_erdos_solution(h_refined)
        if score_new < best_c5:
            best_c5 = score_new
            best_h = h_refined.copy()
            best_n = n_points

    # Final normalization to exact sum
    if best_h is not None:
        best_h = best_h * (best_n / (2 * np.sum(best_h)))

    return (best_h, best_c5, best_n)
