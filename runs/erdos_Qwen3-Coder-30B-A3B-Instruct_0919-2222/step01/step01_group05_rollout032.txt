import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def lp_initialize(h0, n_points):
    """Solve the McCormick LP relaxation to get an initial fractional solution."""
    dx = 2.0 / n_points
    w = n_points / 2
    h_vars = cp.Variable(n_points)
    u_vars = {}

    # Constraints
    constraints = [
        cp.sum(h_vars) == w,
        0 <= h_vars,
        h_vars <= 1,
    ]
    
    # Add McCormick inequalities (only for valid overlaps)
    total_u_vars = 0
    for d in range(-n_points+1, n_points):
        # Only consider overlaps where both indices exist
        valid_indices = [i for i in range(n_points) if 0 <= i+d < n_points]
        for i in valid_indices:
            if (i, d) not in u_vars:
                u_vars[(i, d)] = cp.Variable()
                total_u_vars += 1
        
        # McCormick constraints for this shift d
        for i in valid_indices:
            u = u_vars[(i, d)]
            h_i = h_vars[i]
            h_d = h_vars[i+d]
            constraints += [
                u >= 0,
                u <= h_i,
                u <= h_d,
                u >= h_i + h_d - 1,
            ]
    
    # Compute s_d = sum u_{i,d}
    s_d = []
    for d in range(-n_points+1, n_points):
        valid_indices = [i for i in range(n_points) if 0 <= i+d < n_points]
        expr = sum(u_vars[(i,d)] for i in valid_indices)
        s_d.append(expr)
    
    # Objective t = max of (w - s_d) = min of s_d
    t = cp.Variable()
    for d in range(-n_points+1, n_points):
        s = s_d[d + n_points - 1]  # Correct index in the list
        constraints.append(t >= w - s)
    objective = cp.Minimize(t)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=ECOS, verbose=False)

    if prob.status in ["optimal", "optimal_inaccurate"]:
        return h_vars.value
    else:
        # Fallback
        return h0

def project_onto_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum"""
    h = np.array(h)
    h = h - np.mean(h) + target_sum / len(h)
    h = np.clip(h, 0, 1)
    scale = target_sum / np.sum(h)
    return h * scale

def worst_shift(h):
    """Compute index of worst shift d* that maximizes C5(d)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.argmax(corr)

def swap_improve(h, worst_d):
    """Use worst-shift swapping to improve solution"""
    n = len(h)
    h_swapped = h.copy()
    # Find indices for swapping: 1 -> 0 with d* offset
    candidates_i = np.where(h_swapped == 1)[0]
    candidates_j = np.where(h_swapped == 0)[0]

    found = False
    for i in candidates_i:
        if (i + worst_d) >= 0 and (i + worst_d) < n:
            if h_swapped[i + worst_d] == 0:
                # Looking for j such that j + worst_d == 1
                for j in candidates_j:
                    if (j + worst_d) >= 0 and (j + worst_d) < n:
                        if h_swapped[j + worst_d] == 1:
                            h_swapped[i], h_swapped[j] = h_swapped[j], h_swapped[i]
                            found = True
                            break
                if found:
                    break
    return h_swapped

def projected_gradient_descent(h, max_iter=1000, initial_step=0.01):
    """Perform projected subgradient descent to refine fractional solution"""
    n = len(h)
    dx = 2.0 / n
    w = n / 2
    h = np.array(h)
    step = initial_step * dx
    
    for k in range(max_iter):
        # Find the worst shift d*
        corr = np.correlate(h, 1 - h, mode="full")
        d_star = np.argmax(corr)
        d = d_star - (n - 1)

        # Compute subgradient at d_star
        subgrad = np.zeros(n)
        for i in range(n):
            if 0 <= i + d < n:
                subgrad[i] -= h[i + d]
            if 0 <= i - d < n:
                subgrad[i] -= h[i - d]
        
        # Update
        h_new = h - step * subgrad

        # Project onto simplex
        h = project_onto_simplex(h_new, w)
        
        # Reduce step size
        step *= 0.999
    return h

def simulated_annealing(h, max_iter=2000, T0=0.05):
    """Simulated annealing to escape local optima with fractional moves"""
    n = len(h)
    dx = 2.0 / n
    w = n / 2
    T = T0
    h_current = h.copy()
    c5_current = evaluate_erdos_solution(h_current)
    
    for _ in range(max_iter):
        i, j = np.random.randint(0, n, 2)
        # Ensure i ≠ j
        if i == j:
            continue
            
        # Compute delta
        delta = h_current[i] + h_current[j] - 1
        if delta < 0:
            # Reverse move
            new_h = h_current.copy()
            d = min(h_current[i], 1 - h_current[j], 0.1)
            new_h[i], new_h[j] = new_h[i] - d, new_h[j] + d

            new_h = np.clip(new_h, 0, 1)
            new_h = project_onto_simplex(new_h, w)
            c5_new = evaluate_erdos_solution(new_h)
            
            if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / (T + 1e-10)):
                h_current, c5_current = new_h, c5_new
        
        # Decrement temperature
        T *= 0.995
    return h_current
