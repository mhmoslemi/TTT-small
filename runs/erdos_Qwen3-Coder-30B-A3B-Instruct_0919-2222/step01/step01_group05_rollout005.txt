from scipy.optimize import minimize
import time

from cvxpy import Minimize, Problem

                z[(i, d)].lower_bound = 0
                # z_{i,d} <= b_i
                # z_{i,d} <= b_{i+d}
                # z_{i,d} >= b_i + b_{i+d} - 1
                constraints = [
                    z[(i, d)] <= b[i],
                    z[(i, d)] <= b[i+d],
                    z[(i, d)] >= b[i] + b[i+d] - 1
                ]
                
        # Overlap counts
        for d in range(n_points):
            if d == 0:
                s[d] = cp.sum(b)
            else:
                s[d] = cp.sum([z[(i, d)] for i in range(n_points - d)])

    def evaluate_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return max(corr) * dx
    
    def update_h(h_vec, i, j, delta):
        h_new = h_vec.copy()
        h_new[i] -= delta
        h_new[j] += delta
        return h_new

    h = np.clip(h, 0, 1)
    h = h * (n_points / (2 * np.sum(h)))
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    T0 = 0.02 * best_c5
    alpha = 0.995
    T = T0
    max_steps = 50000
    delta_max = 0.2
    
    sa_steps = 0
    
    while time.time() - start_time < budget_s * 0.95:
        sa_steps += 1
        if sa_steps % 1000 == 0:
            T = max(T * alpha, 1e-5)
        
        # perturb
        i = np.random.choice(n_points)
        j = np.random.choice(n_points)
        if i == j:
            continue
        # Only allow moves that keep within [0,1]
        delta_max_curr = min(h[i], 1 - h[j], delta_max)
        if delta_max_curr <= 0:
            continue
        delta = np.random.uniform(0, delta_max_curr)
        
        h_new = update_h(h, i, j, delta)
        h_new = np.clip(h_new, 0, 1)

    # final check
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
