import cvxpy as cp
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

            X >> 0,  # Positive semidefinite
            cp.trace(X) == n,  # Trace constraint
            cp.sum(a) == 0,  # Zero mean
        ]

        for i in range(n):
            constraints.append(X[i, i] <= 1)

        # Note: this uses a conservative approach by relaxing max over shifts
        # and tightening with iterative refinement
        max_overlap = cp.Variable()
        for d in range(1, n - 1):  # for all shifts d=1,...,n-2
            shifted_sum = 0
            for k in range(n - d):
                shifted_sum += a[k] * a[k + d]

            constraints.append(corr_d <= max_overlap)

        prob = cp.Problem(obj, constraints)
