import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def compute_binary_min_overlap(b):
    """Compute minimum overlap for binary vector b"""
    n = len(b)
    corr = np.correlate(b, b, mode="full")
    # Exclude zero lag
    min_overlap = np.min(corr[n:])  # positive lags only
    return min_overlap

def random_binary(n, weight):
    """Generate random binary vector with given weight"""
    b = np.zeros(n, dtype=int)
    ones = np.random.choice(n, weight, replace=False)
    b[ones] = 1
    return b

def binary_maximin_search(n_points, weight, time_limit):
    """Generate binary vector with maximum minimal overlap"""
    best_b = None
    best_min_overlap = -1
    start_time = time.time()
    
    # Try many random examples
    for _ in range(20000):  # Each trial is cheap
        b = random_binary(n_points, weight)
        min_overlap = compute_binary_min_overlap(b)
        if min_overlap > best_min_overlap:
            best_min_overlap = min_overlap
            best_b = b.copy()
        
        if time.time() - start_time > time_limit:
            break
    
    return best_b, best_min_overlap

def smooth_binary(b, kernel=[0.25, 0.5, 0.25]):
    """Smooth binary vector with kernel"""
    return np.convolve(b, kernel, mode='same')

def project_onto_box_and_sum(v, target):
    """Project v onto [0,1] box and sum constraint"""
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def gradient_for_lag(h, lag):
    """Compute gradient of C5 with respect to shift lag"""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # Overlap region i in [0, N - lag)
        i = np.arange(N - lag)
        grad[i] += 1
        grad[i] += -h[i + lag]  # derivative from first term
        grad[i + lag] += -h[i]   # derivative from second term
    else:
        lag = -lag
        # Overlap region i in [lag, N)
        i = np.arange(lag, N)
        grad[i] += 1
        grad[i] += -h[i - lag]
        grad[i - lag] += -h[i]
    return grad

                lag = idx - (n_points - 1)
                avg_grad += gradient_for_lag(best_h, lag)
            
            avg_grad /= len(max_indices) if len(max_indices) > 0 else 1
            
            # Adam step
            m = beta1 * m + (1 - beta1) * avg_grad
            v = beta2 * v + (1 - beta2) * avg_grad**2

            elapsed = time.time() - start_time
            
            # Early stopping if no improvement for a while

                print(f"Early stopping at iteration {iterations}")
                break
                
        print(f"Final C5: {best_c5:.6f} at iteration {iterations}")
    
    return (best_h, best_c5, n_points)
