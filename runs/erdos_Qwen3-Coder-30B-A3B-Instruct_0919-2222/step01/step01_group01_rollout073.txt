def project_to_feasible(h, target_sum):
    # Clip to [0,1] range
    h = np.clip(h, 0, 1)

    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Use water-filling algorithm to adjust
    diff = target_sum - current_sum

        # Need to increase values, start from smallest
        indices = np.argsort(h)
        for i in indices:
            if diff <= 0:
                break
            increase = min(diff, 1.0 - h[i])
            h[i] += increase
            diff -= increase
    else:

        for i in indices:
            if diff >= 0:
                break
            decrease = min(-diff, h[i])
            h[i] -= decrease
            diff += decrease
    
    return h
