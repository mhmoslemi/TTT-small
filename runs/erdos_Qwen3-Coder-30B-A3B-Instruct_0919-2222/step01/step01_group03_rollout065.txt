import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different optimization approaches
    
    # 1. Direct optimization with random restarts
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Try different initializations
        if restart == 0:
            h_init = h0.copy()
        elif restart == 1:
            # Try a piecewise constant function
            h_init = np.zeros(n_points)
            h_init[:n_points//2] = 0.5
            h_init[n_points//2:] = 1.5
            h_init = h_init / (np.sum(h_init) * dx) * 1.0
        elif restart == 2:
            # Try a simple peak at midpoint
            h_init = np.full(n_points, 0.5)
            h_init[n_points//2] = 1.0
        else:
            # Random initialization
            h_init = np.random.rand(n_points) * 0.5
            h_init = h_init / (np.sum(h_init) * dx) * 1.0
            
        h_init = np.clip(h_init, 0, 1)
        
        try:
            # Use L-BFGS-B for local optimization
            result = minimize(
                objective,
                h_init,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            
            if result.success:
                h_opt = np.clip(result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
        # Also try more iterations with a second optimization
        try:
            # Use SLSQP if L-BFGS-B failed
            result = minimize(
                objective,
                h_init,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            ) 
            
            if result.success:
                h_opt = np.clip(result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # 2. Try a hybrid approach with smaller discretization for general shape
    if time.time() - start_time < budget_s * 0.7:
        # Try a smaller grid for rough shape optimization
        n_small = 20
        dx_small = 2.0 / n_small
        
        def small_objective(h_small):
            # Interpolate h_small to full grid
            h_full = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, n_small), h_small)
            # Ensure h is within bounds [0,1]
            h_full = np.clip(h_full, 0, 1)
            return evaluate_erdos_solution(h_full)
            
        small_bounds = [(0, 1) for _ in range(n_small)]
        try:
            result = differential_evolution(
                small_objective,
                small_bounds,
                maxiter=50,
                popsize=10,
                seed=seed,
                atol=1e-6,
                rtol=1e-6
            )
            if result.success:
                h_interpolated = np.interp(np.linspace(0, 2, n_points), 
                                         np.linspace(0, 2, n_small), result.x)
                h_interpolated = np.clip(h_interpolated, 0, 1)
                c5_approach = objective(h_interpolated)
                if c5_approach < best_c5:
                    best_c5 = c5_approach
                    best_h = h_interpolated.copy()
        except:
            pass

    # 3. Final refinement with dirty tricks - try greedy improvement
    if time.time() - start_time < budget_s * 0.9:
        # Try to improve best result further with fine adjustments
        h_refined = best_h.copy()
        # Create small perturbations
        for i in range(100):
            if time.time() - start_time > budget_s * 0.99:
                break
                
            # Make small random change to grid value
            idx = np.random.randint(n_points)
            perturbation = np.random.normal(0, 0.02)
            h_test = h_refined.copy()
            h_test[idx] += perturbation
            h_test = np.clip(h_test, 0, 1)
            
            # Adjust other values to maintain integral constraint
            total_diff = np.sum(h_test) * dx - 1.0
            if abs(total_diff) > 1e-6:
                adjustment = -total_diff / n_points
                h_test += adjustment
                h_test = np.clip(h_test, 0, 1)
            
            c5_test = objective(h_test)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
                h_refined = h_test.copy()
                
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
