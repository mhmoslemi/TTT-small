def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)
