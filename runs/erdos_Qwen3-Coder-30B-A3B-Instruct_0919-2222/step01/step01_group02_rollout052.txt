import numpy as np
import time
import random
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, d):
    """Shift h by d positions (d can be negative)"""
    n = len(h)
    if d == 0:
        return h
    elif d > 0:
        return np.concatenate([np.zeros(d), h[:-d]])
    else:  # d < 0
        return np.concatenate([h[-d:], np.zeros(-d)])

def project_onto_box_and_sum(v, target_sum):
    """Project v onto [0,1]^n and the hyperplane sum(x) = target_sum"""
    n = len(v)
    low, high = 0.0, 1.0
    v = np.clip(v, low, high)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-12:
        return v
    # Bisection to find the right Lagrange multiplier
    def check_sum(lam):
        new_v = np.clip(v - lam, low, high)
        return np.sum(new_v) - target_sum
    
    a, b = -10.0, 10.0  # reasonable range
    for _ in range(40):
        lam = (a + b) / 2.0
        val = check_sum(lam)
        if val > 0:
            a = lam
        else:
            b = lam
    lam = (a + b) / 2.0
    result = np.clip(v - lam, low, high)
    return result

def build_lp_model(n, binary=False):
    """
    Build a linear program for C5 optimization with n points.
    If binary=True, then h variables are integers in {0,1}.
    """
    # Variables
    h = cp.Variable(n, boolean=binary)
    t = cp.Variable()
    y = cp.Variable((n, n), nonneg=True)
    
    # Constraints
    constraints = [
        cp.sum(h) == n/2,
        h >= 0,
        h <= 1,
    ]

    # McCormick constraints for linearization of y[i,j] = h[i]*h[j]
    # Need to iterate over valid i, j pairs for shift d = j-i
    for i in range(n):
        for j in range(n):
            d = j - i  # shift amount
            if 0 <= i + d < n:
                # y[i, j] = h[i]*h[i+d] 
                # y[i,j] >= 0
                # y[i,j] >= h[i] + h[i+d] -1 
                # y[i,j] <= h[i]
                # y[i,j] <= h[i+d]
                if binary:
                    constraints += [y[i,j] >= h[i] + h[i+d] - 1]
                    constraints += [y[i,j] <= h[i]]
                    constraints += [y[i,j] <= h[i+d]]
                else:
                    # For continuous, this is still valid
                    constraints += [y[i,j] >= 0]
                    constraints += [y[i,j] >= h[i] + h[i+d] - 1]
                    constraints += [y[i,j] <= h[i]]
                    constraints += [y[i,j] <= h[i+d]]

    # Constraint on max shifts
    # C(d) = dx*(n/2 - sum over h[i] * h[i+d])
    dx = 2.0 / n
    
    for i in range(n):
        # compute for actual shifts (some might be missing because of zero-padding)
        shifts = []
        for d in range(-n+1, n):
            if 0 <= i+d < n:
                shifts.append(y[i, i+d])
        # sum will be over offsets n-d maximum
        sum_shifts = sum(shifts)
        constraints.append(t >= dx * (n/2 - sum_shifts))
        # alternative setup (this approximates the data we're looping over)
        # For ALL shifts, we will add this:
        # for d in range(-n+1, n):
        #     if 0 <= i+d < n:
        #         d_ind = i + d
        #         constraints.append(t >= dx * (n/2 - y[i,d_ind]))
    # More direct way for checking every shift and making sure C(d) <= t
    # But this is more efficiently done during programming round.

    # Or extremely simple approach:
    # Goal: minimize t such that
    # for all shifts d from -n+1 to n-1:
    #     sum_i h[i] - sum_i y[i,i+d] <= t*dx
    
    # So make: sum_i h[i] - sum_i y[i,i+d] <= t*dx
    # as stated in strategy: sum h is fixed
    # ``` t >= dx*((n/2) - sum_i y[i,i+d] ) ```

    # But let's just enumerate it properly!
    for d in range(-n+1, n):
        col = []
        for i in range(n):
            if 0 <= i+d < n:
                col.append(y[i, i+d])
        if col:
            sum_y = cp.sum(col)
            # Minimise t subject to
            # t >= dx*(n/2 - sum_y)
            constraints += [t >= dx*(n/2 - sum_y)]
    
    # Objective
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    return prob, h, t, y

def solve_with_time_limit(prob, max_time_s, solver='GLPK_MI'):
    """
    Solve the cvxpy problem with a maximum time limit.
    """
    try:
        start = time.time()
        # Solve with a time limit
        solve_opts = {
            'maxTime': max_time_s,
            'msg': 0,  # Suppress solver output
        }
        prob.solve(solver=solver, **solve_opts)
        end = time.time()
        return True, end - start
    except Exception as e:
        return False, 0.0

def subgradient_refine(h_start, end_time):
    """
    Perform projected subgradient descent.
    """
    # Use some reasonable value for max iter
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    time_limit = end_time
    not_improved = 0
    max_not_improved = 5000
    alpha0 = 0.5
    decay = 0.999
    
    t_start = time.time()
    iter_count = 0
    while time.time() < time_limit:
        iter_count += 1
        corr = np.correlate(h, 1-h, mode='full')*dx
        d_star = np.argmax(corr)
        # Adjust! The correlation is indexed from 0 to 2n-2, so shift is d_star - n+1
        actual_d = d_star - (n - 1)
        
        # Compute negative gradient for i-th element is:
        grad = np.zeros(n)
        for i in range(n):
            # Handle shift
            if i+actual_d < n and i+actual_d >= 0:
                grad[i] = 1 - h[i+actual_d]
            else:
                grad[i] = 1.0
            # Handle left shift too
            if i-actual_d >= 0 and i-actual_d < n:
                grad[i] -= h[i-actual_d]
        
        # Step size
        alpha = alpha0 / np.sqrt(iter_count)
        # New candidate
        v = h - alpha * grad
        # Project
        projected = project_onto_box_and_sum(v, n/2)
        # Update
        h = projected
        # Evaluate
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            not_improved = 0
        else:
            not_improved += 1
        # Perturb if needed
        if not_improved > max_not_improved:
            # Add small gaussian noise
            noise = np.random.normal(0, 0.05, n)
            h = project_onto_box_and_sum(h + noise, n/2)
            not_improved = 0
        if time.time() - t_start > 0.1:
            t_start = time.time()
    return best_h, best_c5
