import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt
import cvxpy as cp
from cvxpy import ECOS

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def project_onto_simplex(x, target_sum):
    """Project x onto the simplex {y : sum(y) = target_sum, y in [0,1]^n}"""
    n = len(x)
    x = np.asarray(x)
    # Shift to zero mean
    mu = (np.sum(x) - target_sum) / n
    x -= mu
    # Clamp to [0,1]
    x = np.clip(x, 0, 1)
    # Adjust so sum equals target_sum
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        # Distribute the adjustment
        idxs = np.argsort(x)
        # Distribute diff in ascending order of x
        for i in range(n):
            if diff > 0:
                d = min(diff, 1 - x[idxs[i]])
                x[idxs[i]] += d
                diff -= d
            else:
                break
    return x

def subgradient(h, d_star):
    """Compute subgradient for shift d_star"""
    n = len(h)
    g = np.zeros(n)
    # Include the contribution from d_star and -d_star
    # h_i contributes to both C5(d_star) and C5(-d_star)
    for i in range(n):
        if i + d_star >= 0 and i + d_star < n:
            g[i] += -h[i + d_star]
        if i - d_star >= 0 and i - d_star < n:
            g[i] += -h[i - d_star]
    return g

def worst_shift(h):
    """Find the shift that maximizes the overlap integral"""
    corr = np.correlate(h, 1 - h, mode="full")
    d_star = np.argmax(corr)
    return d_star

def swap_improve(h, d_star):
    """Perform α-optimal swaps with respect to d_star"""
    # Find indices with h_i = 1 and h_{i+d_star} = 0
    # and vice versa
    n = len(h)
    swapped = False
    # Simple O(n^2) loop to find beneficial swaps
    for i in range(n):
        if h[i] == 1:
            j = (i + d_star) % n
            if h[j] == 0:
                # Find a benefical swap
                for k in range(n):
                    if h[k] == 0:
                        l = (k + d_star) % n
                        if h[l] == 1:
                            # Perform swap
                            h[i], h[k] = 0, 1
                            h[j], h[l] = 1, 0
                            swapped = True
                            break
            if swapped:
                break

    return swapped

def lp_initialize(n):
    """Solve the linear programming relaxation of the problem"""
    dx = 2.0 / n
    h = cp.Variable(n)
    w = n / 2
    t = cp.Variable()
    
    # Not all d pairs (i,d) overlap, we just consider valid indices
    # For d ∈ [-n+1, ..., n-1], constraints for all overlaps
    # We define auxiliary variables u_{i,d} for each i,d valid
    u = {}
    s = {}
    
    for d in range(-n + 1, n):
        u[d] = cp.Variable(n, nonneg=True)
        s[d] = cp.sum(u[d])  # sum_i u_{i,d}
        
    # Part 1: Constraints
    constraints = [cp.sum(h) == w]
    for i in range(n):
        constraints.append(0 <= h[i])
        constraints.append(h[i] <= 1)
        
    # Part 2: McCormick envelopes
    for d in range(-n+1, n):
        for i in range(n):
            # skip if i+d is out of bounds
            if i + d < 0 or i + d >= n:
                continue
            u[d][i].value = 0
            constraints.append(u[d][i] >= 0)
            constraints.append(u[d][i] <= h[i])
            constraints.append(u[d][i] <= h[i + d])
            constraints.append(u[d][i] >= h[i] + h[i + d] - 1)
            
    # Part 3: Objective bound
    for d in range(-n + 1, n):
        # C5(d) = w - sum_i h_i h_{i+d} ≈ w - sum_i u_{i,d}
        constraints.append(t >= w - s[d])
    
    # Part 4: Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS)
    except:
        # On failure, return just unit vectors as often they work
        return np.ones(n) * 0.5
    
    if prob.status == cp.OPTIMAL:
        h_sol = h.value.copy()
        # Clip to [0,1]
        h_sol = np.clip(h_sol, 0, 1)
        # Renormalize to sum = n/2
        h_sol = h_sol * (n / (2 * np.sum(h_sol)))
        return h_sol
    else:
        return np.ones(n) * 0.5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # List of candidate discretization sizes to try
    n_candidates = [42, 48, 60]
    best_h = None
    best_c5 = float('inf')
    best_n = None

    for n in n_candidates:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n

        # Step 1: Convex relaxation using LP with McCormick
        h_lp = lp_initialize(n)
        
        # Normalize to sum = n/2
        h_lp = h_lp * (n / (2 * np.sum(h_lp)))
        
        # Check initial LP solution
        c5_lp = evaluate_erdos_solution(h_lp)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            best_n = n

        # Step 2: Binary rounding and deterministic swap improvement
        # Get binary candidate (items with top n/2 values)
        h_rnd = h_lp.copy()
        # Use top n/2 items as 1s
        idxs = np.argsort(h_rnd)[::-1]
        h_binary = np.zeros(n)
        for i in range(n // 2):
            h_binary[idxs[i]] = 1
        
        # Refine binary solution using hops
        swapped = True
        while swapped:
            d_star = worst_shift(h_binary)
            swapped = swap_improve(h_binary, d_star)
            c5_new = evaluate_erdos_solution(h_binary)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_binary.copy()
                best_n = n

        # Step 3: Simulated Annealing refinement
        h_sa = h_binary.copy()
        c5_sa = evaluate_erdos_solution(h_sa)

        # Cooling parameters
        temp_start = max(0.01, 0.05 * c5_sa)
        temp = temp_start
        alpha = 0.995
        max_iter = 1000
        no_improvements = 0
        max_no_improvements = 500
        for _ in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break

            # Make a move by transferring a fraction of mass
            i, j = np.random.randint(0, n, 2)
            delta_max = min(h_sa[i], 1 - h_sa[j], 0.2)
            if delta_max <= 0:
                continue
            delta = np.random.uniform(0, delta_max)
            
            h_new = h_sa.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize
            h_new = h_new * (n / (2 * np.sum(h_new)))
            
            c5_new = evaluate_erdos_solution(h_new)
            
            # Accept or reject
            if c5_new < c5_sa:
                h_sa = h_new
                c5_sa = c5_new
                no_improvements = 0
            else:
                if np.random.rand() < np.exp(-(c5_new - c5_sa) / temp):
                    h_sa = h_new
                    c5_sa = c5_new
                no_improvements += 1
            
            if no_improvements > max_no_improvements:
                break
            temp *= alpha

        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n

        # Step 4: Gradient descent refinement
        h_refine = h_sa.copy()
        c5_refine = evaluate_erdos_solution(h_refine)
        eta = 0.01  # Initial step size
        for step in range(1000):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Compute subgradient
            d_star = worst_shift(h_refine)
            g = subgradient(h_refine, d_star)
            # Projected gradient step
            h_new = h_refine - eta * g
            h_new = project_onto_simplex(h_new, n / 2)
            h_new = np.clip(h_new, 0, 1)
            c5_new = evaluate_erdos_solution(h_new)
            
            # Check improvement
            if c5_new < c5_refine:
                h_refine = h_new
                c5_refine = c5_new
            else:
                # Early stopping on no improvement
                break
            eta = max(1e-6, eta * 0.99)  # Decreasing learning rate

        if c5_refine < best_c5:
            best_c5 = c5_refine
            best_h = h_refine.copy()
            best_n = n

    # Final check to make sure the sum is exactly 1.0 when dx = 2/n
    best_h_final = best_h * (best_n / (2 * np.sum(best_h)))
    final_c5 = evaluate_erdos_solution(best_h_final)

    return (best_h_final, final_c5, best_n)
