import numpy as np
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, shift):
    """Shift array by given amount using zero padding"""
    n = len(arr)
    if shift >= 0:
        # Shift left by shift: elements [shift:] go to [0:n-shift], rest 0
        return np.concatenate([arr[shift:], np.zeros(shift)])
    else:
        # Shift right by |shift|: elements [:n+shift] go to [abs(shift):], rest 0
        s = -shift
        return np.concatenate([np.zeros(s), arr[:n-s]])

def project_onto_box_simplex(v, lower=0., upper=1., target_sum=None):
    """Project onto box constraints [lower,upper] and sum constraint
    Uses bisection on Lagrange multiplier λ"""
    if target_sum is None:
        target_sum = len(v) / 2
    # Binary search for λ to make sum equal to target_sum
    lo, hi = -100.0, 100.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, lower, upper)

def subgradient_optimize(n, end_time, start_h=None):
    """Optimize h using projected subgradient method"""

    if start_h is not None and len(start_h) == n:
        h = np.array(start_h)
    else:
        h = np.random.rand(n)

    best_h = h.copy()

    iteration = 0
    last_improve_iter = 0
    stagnation = 0
    
    α0 = 0.5  # initial step size
    
    while time.time() < end_time:
        iteration += 1
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Find shift with maximum correlation
        idx = np.argmax(corr)

        # Compute direction vector
        shift_plus = shift_array(h, d)
        shift_minus = shift_array(h, -d)
        direction = shift_plus + shift_minus - 1.0
        
        # Step size (decreasing)
        α = α0 / np.sqrt(iteration)
        
        # Update h
        v = h + α * direction
        h = project_onto_box_simplex(v)
        
        # Evaluate new solution
        cur_C = evaluate_erdos_solution(h)
        if cur_C < best_C - 1e-10:
            best_C = cur_C
            best_h = h.copy()
            last_improve_iter = iteration
            stagnation = 0
        else:
            stagnation += 1
            
        # Periodically perturb if no improvement
        if stagnation > 5000 and time.time() < end_time:
            # Apply random perturbation
            perturb = best_h + np.random.normal(scale=0.05, size=n)
            h = project_onto_box_simplex(perturb)
            stagnation = 0
    
    return best_h, best_C
