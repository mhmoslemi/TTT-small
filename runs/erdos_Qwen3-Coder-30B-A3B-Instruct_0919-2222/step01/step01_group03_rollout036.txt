import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try several different optimization strategies with different initializations
    strategies = []
    
    # Strategy 1: Differential Evolution with multiple restarts
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=100,
            popsize=15,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Strategy 2: Local optimization starting from DE result
    try:
        local_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 5000, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Strategy 3: Multiple random restarts with carefully chosen initializations
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate different initializations:
        if restart % 3 == 0:
            # Peaks and valleys in different patterns
            h_rand = np.random.rand(n_points) * 0.3 + 0.35  # Center around 0.35
        elif restart % 3 == 1:
            # Some parts very high, some very low
            h_rand = np.random.rand(n_points)
            # Make some parts high and some low
            h_rand = np.where(h_rand > 0.7, 0.8, np.where(h_rand < 0.3, 0.2, h_rand))
        else:
            # Smooth approximation
            h_rand = np.random.rand(n_points) * 0.5
        # Scale to make integral equal to 1
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Strategy 4: Coarse to fine grid optimization
    # Try a coarse grid first to get a global view
    if len(best_h) >= 20:
        coarse_h = best_h[::2]  # Take every other value
        coarse_dx = 2.0 / len(coarse_h)
        
        def coarse_objective(h):
            # Reconstruct full size h from coarse by repeating elements
            h_full = np.repeat(h, 2)
            if len(h_full) > len(best_h):
                h_full = h_full[:len(best_h)]
            elif len(h_full) < len(best_h):
                h_full = np.pad(h_full, (0, len(best_h) - len(h_full)), 'constant')
            return evaluate_erdos_solution(h_full)
        
        coarse_bounds = [(0, 1) for _ in range(len(coarse_h))]
        
        try:
            coarse_result = minimize(
                coarse_objective,
                coarse_h,
                method='L-BFGS-B',
                bounds=coarse_bounds,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8},
                callback=lambda x: None
            )
            if coarse_result.success:
                h_coarse = coarse_result.x
                h_full = np.repeat(h_coarse, 2)
                if len(h_full) > len(best_h):
                    h_full = h_full[:len(best_h)]
                elif len(h_full) < len(best_h):
                    h_full = np.pad(h_full, (0, len(best_h) - len(h_full)), 'constant')
                
                h_opt = np.clip(h_full, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
