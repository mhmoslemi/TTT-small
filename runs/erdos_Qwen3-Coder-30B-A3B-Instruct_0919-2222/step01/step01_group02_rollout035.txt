import numpy as np
import cvxpy as cp
import time
import math
from cvxpy import GLPK, GLPK_MI, ECOS, SCS

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def project_sum_constraint(h, target_sum):
    """Project h onto sum = target_sum while preserving bounds [0,1]"""
    # Use iterative bisection projection
    h = np.array(h)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Clip to [0,1], then reweight
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.full_like(h, target_sum/len(h))
    else:
        h *= target_sum / current_sum
    return np.clip(h, 0, 1)

def solve_binary_milp(n, time_limit_s):
    dx = 2.0 / n
    # Variable setup
    h = cp.Variable(n, binary=True)
    t = cp.Variable()
    
    # Objective
    obj = cp.Minimize(t)
    
    # Constraints
    constrs = [t >= 0]
    # Add constraints for each shift (entire range)
    for d in range(-n+1, n):
        # compute effective indices for shift
        if d >= 0:
            valid_indices = np.arange(0, n - d)
            shifted_indices = np.arange(d, n)
        else:
            valid_indices = np.arange(-d, n)
            shifted_indices = np.arange(0, n + d)
        if len(valid_indices) == 0:
            continue
            
        # Quad approximation used in relaxations
        Q_expr = cp.sum(cp.multiply(h[valid_indices], h[shifted_indices]))
        constrs.append(t >= dx * (n/2 - Q_expr))
        
    # sum constraint
    constrs.append(cp.sum(h) == n/2)
    
    # Build and solve
    prob = cp.Problem(obj, constrs)
    start_time = time.time()
    try:
        prob.solve(solver=GLPK_MI, verbose=False, time_limit=time_limit_s)
        elapsed = time.time() - start_time
        if prob.status == 'optimal':
            h_val = h.value
            t_val = t.value
            return h_val, t_val, elapsed
        elif prob.status == 'time_limit':
            h_val = h.value
            t_val = t.value
            return h_val, t_val, elapsed
        else:
            return None, None, elapsed
    except:
        return None, None, time.time() - start_time
    
def solve_lp_relaxation(n, time_limit_s):
    dx = 2.0 / n
    # Variable setup
    h = cp.Variable(n, continuous=True)
    t = cp.Variable()
    
    # Objective
    obj = cp.Minimize(t)
    
    # Constraints
    constrs = [cp.sum(h) == n/2, h >= 0, h <= 1]
    
    # Add constraints for each shift (entire range)
    for d in range(-n+1, n):
        # compute effective indices for shift
        if d >= 0:
            valid_indices = np.arange(0, n - d)
            shifted_indices = np.arange(d, n)
        else:
            valid_indices = np.arange(-d, n)
            shifted_indices = np.arange(0, n + d)
        if len(valid_indices) == 0:
            continue
            
        # Here Q_expr is just sum of products
        Q_expr = cp.sum(cp.multiply(h[valid_indices], h[shifted_indices]))
        # Use McCormick inequalities for improved LP relaxation:
        #  y_{i,d} = h[i] * h[i+d]
        #  y_{i,d} <= h[i]
        #  y_{i,d} <= h[i+d]
        #  y_{i,d} >= h[i] + h[i+d] - 1
        #  y_{i,d} >= 0
        # We'll do it symbolically by introducing Y variables
        # But for efficiency here, just add the constraint and let cvxpy
        # approximate or handle as part of the LP hull
        
        # Use standard convex hull relaxation trick: just add
        # t >= dx*( n/2 - Q_expr ) where Q_expr = sum products
        
        # But make it in a linear form.
        # For now, we'll ignore objects for now
        # and work around with variables if necessary, as they might be hard to do directly
        constrs.append(t >= dx * (n/2 - Q_expr))
        
    # Build and solve
    prob = cp.Problem(obj, constrs)
    start_time = time.time()
    try:
        prob.solve(solver=GLPK, verbose=False, time_limit=time_limit_s)
        elapsed = time.time() - start_time
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_val = h.value
            t_val = t.value
            return h_val, t_val, elapsed
        else:
            return None, None, elapsed
    except:
        return None, None, time.time() - start_time

def scc_refinement(h_initial, n, max_iter=2000, time_limit=200, tol=1e-5):
    dx = 2.0 / n
    h = np.array(h_initial)
    
    start_time = time.time()
    prev_value = np.inf
    
    for i in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Compute all Q values for each shift
        Q_values = []
        for d in range(-(n-1), n):
            if d >= 0:
                valid_indices = np.arange(0, n - d)
                shifted_indices = np.arange(d, n)
            else:
                valid_indices = np.arange(-d, n)
                shifted_indices = np.arange(0, n + d)
            if len(valid_indices) == 0:
                Q_values.append(0.0)
                continue
            q_val = np.sum(h[valid_indices] * h[shifted_indices])
            Q_values.append(q_val)
        
        # Update t by computing worst case
        c5_values = [dx*(n/2 - Q_val) for Q_val in Q_values]
        t_new = max(c5_values)
        
        if abs(prev_value - t_new) < tol:
            break
        prev_value = t_new
        
        # Rationalize for the worst shift
        # Get the shift index for the worst case
        worst_idx = np.argmax(c5_values)
        d_worst = worst_idx - (n-1)
        
        # Expected: shift d = d_worst
        # Approximate Q(d_worst) as sum of linearized terms
        # For computing this part, compute the linearization point
        # Linearization point: h_old, current h
        
        # Create the LP constraint for this shift
        # Introduce temporary variable h and t
        if d_worst >= 0:
            valid_indices = np.arange(0, n - d_worst)
            shifted_indices = np.arange(d_worst, n)
        else:
            valid_indices = np.arange(-d_worst, n)
            shifted_indices = np.arange(0, n + d_worst)
        if len(valid_indices) == 0:
            continue
            
        # At each position k in valid_indices sum
        # h_k * h_{k+d_worst}   becomes
        # h_k^old * h_{k+d_worst}^old + h_k^old * (h_{k+d_worst} - h_{k+d_worst}^old) + h_{k+d_worst}^old * (h_k - h_k^old)
        
        import cvxpy as cp
        
        h_var = cp.Variable(n, continuous=True)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == n/2,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # Add objective for this constraint
        Q_linearized = cp.sum(
            h[valid_indices] * h[shifted_indices] # old point
            + cp.multiply(h[valid_indices], h_var[shifted_indices] - h[shifted_indices])
            + cp.multiply(h[shifted_indices], h_var[valid_indices] - h[valid_indices])
        )
        
        # Add this constraint (if needed for max over all shifts using thus this one)
        # But simplify this to be safe, just focus on transformation
        factors = [h[valid_indices], h[shifted_indices]]
        
        # More robust: build full system
        # Approximate element-wise
        # We create a full quadratic term linearization around current point
        Q_lin_expr = 0.0
        for j in range(len(valid_indices)):
            i1 = valid_indices[j]
            i2 = shifted_indices[j]
            val = h[i1] * h[i2]
            linear_part = h[i1] * (h_var[i2] - h[i2]) + h[i2] * (h_var[i1] - h[i1])
            Q_lin_expr += val + linear_part
            
        constraints.append(t_var >= dx * (n/2 - Q_lin_expr))
        
        # Minimize t_var
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h = h_var.value
                h = project_sum_constraint(h, n/2)
                continue
            else:
                break
        except:
            break
    
    return h, evaluate_erdos_solution(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start at n=70 or larger if needed
    n_values = [70, 80, 60]
    
    best_h = None
    best_c5 = float('inf')
    
    # Phase 0: iterate small n
    for n in n_values:
        if time.time() - start_time > budget_s - 20:
            break
        if n < 64:
            continue
            
        # Try binary milp
        mx_seconds = min(30, budget_s - (time.time() - start_time) - 5)
        h_binary, t_binary, elapsed = solve_binary_milp(n, mx_seconds)
        if h_binary is not None:
            # Project to [0,1] if needed
            h_binary = np.clip(h_binary.astype(float), 0, 1)
            h_binary = project_sum_constraint(h_binary, n/2)
            c5_current = evaluate_erdos_solution(h_binary)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_binary
            if best_c5 < 0.3808:
                return (h_binary.tolist(), c5_current, n)
                
    # Phase 1: fallback to LP relaxation
    if best_h is None:
        target_n = 80
        if time.time() - start_time > budget_s - 100:
            target_n = 60
        time_left = budget_s - (time.time() - start_time)
        if time_left > 10:
            h_lp, t_lp, _ = solve_lp_relaxation(target_n, min(100, time_left - 20))
            if h_lp is not None:
                h_refined, c5_refined = scc_refinement(h_lp, target_n, time_limit=min(100, time_left - 20))
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
                    if best_c5 < 0.3808:
                        return (h_refined.tolist(), c5_refined, target_n)

    # Phase 2: do SCC refinement for any current best
    if best_h is not None:
        h_scc, c5_scc = scc_refinement(best_h, len(best_h), time_limit=budget_s - (time.time() - start_time) - 20)
        if c5_scc < best_c5:
            best_c5 = c5_scc
            best_h = h_scc
            
    # Bisection to tighten value if needed
    if best_h is not None:
        time_left = budget_s - (time.time() - start_time) - 5
        if time_left > 20:
            #-binary search on t for feasibility
            low = 0.37
            high = best_c5
            prev_high = high
            tolerance = 1e-4
            max_iter = 100 if time_left > 50 else 50
            
            for _ in range(max_iter):
                if time.time() - start_time > budget_s - 5:
                    break
                mid = (low + high) / 2
                # try to solve feasibility test at this t
                n = len(best_h)
                dx = 2.0 / n
                
                # Single feasibility LP this time
                h_var = cp.Variable(n, continuous=True)
                # Constraints
                constrs = [cp.sum(h_var) == n/2, h_var >= 0, h_var <= 1]
                
                # Constraints for shift inequalities
                for d in range(-(n-1), n):
                    if d >= 0:
                        valid_indices = np.arange(0, n - d)
                        shifted_indices = np.arange(d, n)
                    else:
                        valid_indices = np.arange(-d, n)
                        shifted_indices = np.arange(0, n + d)
                    if len(valid_indices) == 0:
                        continue
                    Q_expr = cp.sum(cp.multiply(h_var[valid_indices], h_var[shifted_indices]))
                    constrs.append(dx * (n/2 - Q_expr) <= mid)  # This makes it a feasibility test.
                
                prob = cp.Problem(cp.Minimize(0), constrs)
                try:
                    prob.solve(solver=GLPK, verbose=False, time_limit=0.1)
                    if prob.status in ['optimal', 'optimal_inaccurate']:
                        high = mid
                    else:
                        low = mid
                except:
                    low = mid
                if abs(high - low) < tolerance:
                    break
                prev_high = high
            
            if high < best_c5:
                # Re-evaluate final h with lower matrix
                # But the main approach only lower-bounds it here; we can skip modifying h
                # for simplicity. The current one probably is at a lower bound.
                pass
            
    # Final polishing with stochastic hill climb
    if best_h is not None:
        time_left = budget_s - (time.time() - start_time) - 5
        if time_left > 10:
            current_h = np.array(best_h)
            current_c5 = evaluate_erdos_solution(current_h)
            for _ in range(10000):
                if time.time() - start_time > budget_s - 5:
                    break
                i, j = np.random.choice(len(best_h), 2, replace=False)
                delta = np.random.uniform(-0.02, 0.02)
                new_h = current_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
                new_h = project_sum_constraint(new_h, len(new_h)/2)
                new_c5 = evaluate_erdos_solution(new_h)
                if new_c5 < current_c5:
                    current_h = new_h.copy()
                    current_c5 = new_c5
            best_h = current_h
            best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, len(best_h))
