def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while keeping bounds [0,1]"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Adjust all elements equally to meet the target sum
    adjustment = (target_sum - current_sum) / len(h)
    h_new = h + adjustment
    h_new = np.clip(h_new, 0, 1)
    
    # If this doesn't exactly hit the target, adjust a few elements more
    new_sum = np.sum(h_new)
    diff = target_sum - new_sum
    
    if abs(diff) > 1e-10:
        # Find the indices where we can adjust
        indices = np.argsort(h_new)
        i = 0
        while abs(diff) > 1e-10 and i < len(h_new):
            if diff > 0:
                # Need to increase
                max_increase = 1.0 - h_new[indices[i]]
                increase = min(max_increase, diff)
                h_new[indices[i]] += increase
                diff -= increase
            else:
                # Need to decrease
                max_decrease = h_new[indices[i]]
                decrease = min(max_decrease, -diff)
                h_new[indices[i]] -= decrease
                diff += decrease
            i += 1
