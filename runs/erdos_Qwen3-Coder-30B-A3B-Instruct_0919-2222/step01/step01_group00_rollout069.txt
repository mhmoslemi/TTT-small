import numpy as np
from scipy.fft import rfft, irfft
import time

def proj_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: sum = target, 0 <= v <= 1"""
    v = np.asarray(v)
    n = len(v)
    if np.abs(np.sum(v) - target_sum) < 1e-12:
        return v.copy()
    # Binary search for lambda
    # step size for increasing lambda
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    tmp = (cumsum - target_sum) / np.arange(1, n + 1)
    idx = np.where(v_sorted > tmp)[0]
    if len(idx) == 0:
        # All elements are > target_sum / n => set to target_sum/n and clip 1
        return np.clip(np.full_like(v, target_sum/n), 0, 1)
    # Choose index where lambda would be < v_sorted[idx[-1]]
    k = idx[-1] + 1
    lam = (cumsum[k-1] - target_sum) / k
    result = np.clip(v - lam, 0, 1)
    return result

def full_corr(h):
    # Use real FFT for optimal efficiency
    n = len(h)
    # Convolution via FFT: full mode (2*n - 1 length result)
    H = rfft(h, n=2*n-1)
    G = rfft(1-h, n=2*n-1)
    c = irfft(H * np.conj(G), n=2*n-1)
    return c

def evaluate_h(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    conv = full_corr(h)
    max_val = np.max(conv)
    return max_val * dx
