import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def solve_lp_relaxation(n_points):
    """Solve the LP relaxation for initial fractional h using McCormick envelopes"""
    dx = 2.0 / n_points
    # Decision variables
    h = cp.Variable(n_points, bounds=(0, 1))
    u = cp.Variable((n_points, 2 * n_points - 1))  # All shifts in [-n+1, n-1]
    t = cp.Variable()

    # Construct u_{i,d} for valid indices
    # d_index maps shift d to index in u[:, :]: shift d map to index d + (n-1)
    # u[i, d_idx] represents h_i * h_{i+d}
    constraints = []
    for d in range(-n_points + 1, n_points):
        d_idx = d + (n_points - 1)
        for i in range(n_points):
            if 0 <= i + d < n_points:
                u[i, d_idx].value = h[i].value * h[i + d].value
                constraints.extend([
                    u[i, d_idx] >= 0,
                    u[i, d_idx] <= h[i],
                    u[i, d_idx] <= h[i + d],
                    u[i, d_idx] >= h[i] + h[i + d] - 1
                ])

    # sum(h) = n/2 since dx = 2/n
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Define sums over d (s_d)
    s = []
    for d in range(-n_points + 1, n_points):
        d_idx = d + (n_points - 1)
        sum_d = cp.sum(u[:, d_idx])  # Sum of u[i, d_idx] for valid i
        s.append(sum_d)
    
    # Each C5(d) = w - s_d
    w = n_points / 2  # Integral constraint
    C5_expr = [w - s_d for s_d in s]
    constraints.append(t >= cp.max(C5_expr))

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        # Get value if successful
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return np.array([h.value[i] for i in range(n_points)])
    except:
        pass
    # Fallback to normalized random guess
    return np.random.rand(n_points) * 0.5

def round_to_binary(h, n_points):
    """Round fractional h to binary vector with exactly n/2 ones"""
    # Sort by descending order (for top halves)
    idx_sorted = np.argsort(h)[::-1]
    h_bin = np.zeros(n_points)
    h_bin[idx_sorted[:n_points // 2]] = 1.0
    return h_bin

def swap_improve(h, n_points):
    """Improve h by targeted swaps that reduce the worst correlation."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    max_d = max_idx - (n_points - 1)  # Convert index to shift d
    if max_d == 0:
        return h, corr[max_idx] * dx

    # Find points to swap (i such that h[i]=1, h[i+max_d]=0 and vice versa)
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]
    
    # Try to find a compatible pair
    improved = True
    while improved:
        improved = False
        for i in indices_1:
            j = i + max_d
            if 0 <= j < n_points and h[j] == 0:
                # Swap i and j
                h[i], h[j] = h[j], h[i]
                corr = np.correlate(h, 1 - h, mode="full")
                improved = True
                break
    return h, max(corr) * dx

def simulated_annealing(h0, n_points, time_limit):
    """Simulated annealing with fraction mass transfer"""
    start_time = time.time()
    dx = 2.0 / n_points
    
    current_h = h0.copy()
    best_h = current_h.copy()
    current_c5 = evaluate_erdos_solution(current_h)
    best_c5 = current_c5

    T0 = 0.05 * current_c5
    T = T0
    alpha = 0.995
    max_iter = 100000
    iter_no_improvement = 0
    max_no_improvement = 5000
    
    step = 0
    while step < max_iter and time.time() - start_time < time_limit - 1:
        # Randomly select two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(0, min(current_h[i], 1 - current_h[j], 0.2))
        # Direction: i to j
        if np.random.rand() < 0.5:
            delta = -delta
        new_h = current_h.copy()
        new_h[i] = current_h[i] - delta
        new_h[j] = current_h[j] + delta
        # Ensure valid range
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / (2 * np.sum(new_h)))  # Rescale to integral constraint

        new_c5 = evaluate_erdos_solution(new_h)

        # Accept if better or with probability
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            current_h = new_h
            current_c5 = new_c5
            iter_no_improvement = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            iter_no_improvement += 1
            if iter_no_improvement > max_no_improvement:
                T *= alpha
                iter_no_improvement = 0
        step += 1

    return best_h, best_c5

def project_onto_simplex(v, target_sum):
    """Project vector v onto the simplex {sum = target_sum, v[i] in [0,1]}"""
    # Subtract its mean to center
    mean_val = np.mean(v)
    v_shifted = v - mean_val
    # Scale to achieve sum = target_sum or max value = 1
    v_scaled = v_shifted * (target_sum / (np.sum(v_shifted) + 1e-10))
    # Clamp to [0, 1]
    v_clamped = np.clip(v_scaled, 0, 1)
    # Rescale
    v_scaled_corrected = v_clamped * (target_sum / (np.sum(v_clamped) + 1e-10))
    return v_scaled_corrected

def projected_gradient_descent(h0, n_points, max_iter=500):
    """Apply projected sub-gradient descent for final tuning"""
    dx = 2.0 / n_points
    h = h0.copy()
    current_c5 = evaluate_erdos_solution(h)
    alpha = 0.1 * dx  # Learning rate, decreasing gradually

    for it in range(max_iter):
        if it > 0 and it % 50 == 0:
            alpha *= 0.9  # schedule

        corr = np.correlate(h, 1-h, mode='full')
        worst_shift_idx = np.argmax(corr)
        worst_d = worst_shift_idx - (n_points - 1)
        
        # Compute sub-gradient for the worst shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # h_i goes to i + worst_d, i - worst_d
            if 0 <= i + worst_d < n_points:
                gradient[i] -= h[i + worst_d]
            if 0 <= i - worst_d < n_points and worst_d != 0:
                gradient[i] -= h[i - worst_d]
                
        h_new = h - alpha * gradient
        h_new = project_onto_simplex(h_new, n_points / 2)
        h_new = np.clip(h_new, 0, 1)

        new_c5 = evaluate_erdos_solution(h_new)
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
        else:
            # Early stop if no improvement
            break

    return h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Try different resolution
    n_list = [42, 48, 54, 60]  # ordered by increasing complexity
    remaining_time = budget_s
    
    for n in n_list:
        if time.time() - start_time > remaining_time - 5:
            break
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
            
        dx = 2.0 / n
        
        # Step 1: LP relaxation
        h_lp = solve_lp_relaxation(n)
        h_lp = np.clip(h_lp, 0, 1)
        h_lp = h_lp * (n / (2 * np.sum(h_lp)))  # renormalize
        
        c5 = evaluate_erdos_solution(h_lp)
        if c5 < best_c5:
            best_h = h_lp.copy()
            best_c5 = c5
            best_n = n
            
        # Step 2: Binary rounding + hill climbing
        h_bin = round_to_binary(h_lp, n)
        h_bin, c5_bin = swap_improve(h_bin, n)
        if c5_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c5_bin
            best_n = n
            
        # Step 3: Simulated Annealing
        remaining_time_half = (remaining_time * 0.35)
        sa_h, sa_c5 = simulated_annealing(h_bin, n, remaining_time_half)
        if sa_c5 < best_c5:
            best_h = sa_h.copy()
            best_c5 = sa_c5
            best_n = n
            
        # Step 4: Final gradient refinement
        remaining_time_final = (remaining_time * 0.2)
        ref_h, ref_c5 = projected_gradient_descent(sa_h, n, 100)
        if ref_c5 < best_c5:
            best_h = ref_h.copy()
            best_c5 = ref_c5
            best_n = n
    
    # Final correction to exactly satisfy the integral constraint
    best_h = best_h * (best_n / (2 * np.sum(best_h)))
    best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, best_n)
