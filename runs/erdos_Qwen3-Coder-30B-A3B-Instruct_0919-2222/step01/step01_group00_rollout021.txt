import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points, seg_len):
    # Repeat each height to create the full h array
    return np.repeat(heights, seg_len)

def evaluate_heights(heights, n_points, seg_len, penalty_factor=10.0):
    h = build_h(heights, n_points, seg_len)
    c5 = evaluate_h(h)
    # Add penalty for sum constraint
    sum_heights = np.sum(heights)
    penalty = penalty_factor * (sum_heights - n_points / (2 * seg_len))**2
    return c5 + penalty

def project_to_bounds_and_sum(heights, target_sum):
    # Project heights to [0,1] and adjust to make sum equal target_sum
    # Using iterative approach
    heights = np.array(heights)
    heights = np.clip(heights, 0, 1)
    current_sum = np.sum(heights)
    if abs(current_sum - target_sum) < 1e-8:
        return heights
    # Rescale to target
    if current_sum > 0:
        scales = target_sum / current_sum
        heights = heights * scales
    heights = np.clip(heights, 0, 1)
    return heights
