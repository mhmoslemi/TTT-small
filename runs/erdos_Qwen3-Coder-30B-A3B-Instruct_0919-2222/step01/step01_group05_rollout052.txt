import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from math import sqrt

def lp_initialize(n_points):
    """Initialize fractional h using LP relaxation with McCormick envelopes"""
    # Variables
    h = cp.Variable(n_points)
    u = cp.Variable((n_points, 2 * n_points - 1))  # All shifts d in [-n+1,n-1]
    t = cp.Variable()

    # Mappings to shift indices
    shift_indices = list(range(-(n_points-1), n_points))

    # Constraints
    constraints = []
    constraints += [h[i] >= 0 for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]
    constraints += [cp.sum(h) == n_points / 2]

    for di, d in enumerate(shift_indices):
        for i in range(n_points):
            if 0 <= i + d < n_points:
                # McCormick envelopes: u[i,d] \approx h[i]*h[i+d]
                constraints += [u[i, di] >= 0]
                constraints += [u[i, di] <= h[i]]
                constraints += [u[i, di] <= h[i + d]]
                constraints += [u[i, di] >= h[i] + h[i + d] - 1]

    # Compute s_d for each shift
    for di, d in enumerate(shift_indices):
        s_d = cp.sum(u[:, di])
        C5_d = n_points / 2 - s_d
        constraints += [t >= C5_d]

    # Objective
    objective = cp.Minimize(t)

    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, reltol=1e-6)
        h_sol = h.value

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5


def project_onto_simplex(h, target_sum):
    """Project vector h onto simplex with equality constraint sum(h) = target_sum"""
    h = h.copy()
    # Shift to sum to zero
    mean_excess = np.mean(h) - target_sum / len(h)
    h -= mean_excess
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum
    actual_sum = np.sum(h)
    diff = target_sum - actual_sum
    if abs(diff) > 1e-10:
        h += diff / len(h)
    h = np.clip(h, 0, 1)
    return h


def subgradient(h, worst_shifts):
    """Compute subgradient of max_d (sum h_i * h_{i+d}) """
    n = len(h)
    g = np.zeros_like(h)

    for d in worst_shifts:
        for i in range(n):
            if 0 <= i + d < n:
                g[i] -= h[i + d]
            if 0 <= i - d < n and d != 0:
                g[i] -= h[i - d]
    return g


def projected_gradient_descent(h0, max_iter=300, eta=0.1):
    """Projected (sub)gradient descent to reduce the max C5 value"""
    h = h0.copy()
    target_sum = len(h) / 2

    for k in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        d_max = np.argmax(corr)
        worst_shifts = [d_max - len(h) + 1 + i for i in range(len(h)) if corr[i] == corr[d_max]]
        g = subgradient(h, worst_shifts)
        h = h - eta * g
        h = project_onto_simplex(h, target_sum)
        eta *= 0.999
        if k % 50 == 0:
            c5_val = evaluate_erdos_solution(h)
            if c5_val < 1e-5:
                break

    return h


def get_worst_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.argmax(corr) - len(h) + 1


def binary_rounding(h_frac, n_points):
    """Convert fractional solution to binary with same weight"""
    h = h_frac.copy()
    ones_indices = np.argsort(h)[::-1][:n_points // 2]
    bin_h = np.zeros(n_points)
    bin_h[ones_indices] = 1
    return bin_h


def swap_improve(h, max_iter=1000):
    """Worst-shift improvement with deterministic swaps"""
    h = h.copy()
    n = len(h)
    for _ in range(max_iter):
        worst_d = get_worst_shift(h)
        found = False

        # try to find swap
        if worst_d > 0:
            # check if any i has h[i]=1, h[i+worst_d]=0
            for i in range(n-worst_d):
                if h[i] == 1 and h[i+worst_d] == 0:
                    # Find match at j where h[j]=0 and h[j+worst_d]=1
                    for j in range(n-worst_d):
                        if h[j] == 0 and h[j+worst_d] == 1:
                            # Swap
                            h[i], h[j] = h[j], h[i]
                            h[i+worst_d], h[j+worst_d] = h[j+worst_d], h[i+worst_d]
                            found = True
                            break
                    if found:
                        break
        elif worst_d < 0:
            # reverse direction
            for i in range(-worst_d, n):
                if h[i] == 1 and h[i+worst_d] == 0:
                    for j in range(n):
                        if h[j] == 0 and h[j+worst_d] == 1:
                            h[i], h[j] = h[j], h[i]
                            h[i+worst_d], h[j+worst_d] = h[j+worst_d], h[i+worst_d]
                            found = True
                            break
                    if found:
                        break
        else:
            # worst_d == 0 — just break, since all h_i h_{i+0} = h_i^2
            break

        if not found:
            break

    return h


def simulated_annealing(h0, max_iter=2000, T0=0.02, alpha=0.995):
    """Simulated Annealing to explore fractional solutions"""
    h = h0.copy()
    current_c5 = evaluate_erdos_solution(h)
    n = len(h)
    target_sum = n / 2
    T = T0

    for k in range(max_iter):
        # Make a small random fractional move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        delta = np.random.uniform(0, 0.2)
        delta = min(delta, h[i], 1-h[j])

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_onto_simplex(h_new, target_sum)

        new_c5 = evaluate_erdos_solution(h_new)
        delta_c5 = new_c5 - current_c5

        if delta_c5 <= 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            current_c5 = new_c5

        T = T * alpha

    return h


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Try several resolutions
    n_candidates = list(range(42, 61, 6))
    best_c5 = float('inf')
    best_h = None
    best_n = None

    for n_points in n_candidates:
        if time.time() - start_time > budget_s * 0.95:
            break

        # Initialize
        h_initial = lp_initialize(n_points)
        c5_initial = evaluate_erdos_solution(h_initial)

        # Binary rounding
        h_binary = binary_rounding(h_initial, n_points)
        c5_binary = evaluate_erdos_solution(h_binary)

        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_binary
            best_n = n_points

        # Try deterministic swaps
        h_swapped = swap_improve(h_binary)
        c5_swapped = evaluate_erdos_solution(h_swapped)
        if c5_swapped < best_c5:
            best_c5 = c5_swapped
            best_h = h_swapped
            best_n = n_points

        # Run simulated annealing from best binary
        h_sa = simulated_annealing(h_binary, max_iter=int(max_iter*0.5))
        c5_sa = evaluate_erdos_solution(h_sa)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            best_n = n_points

        # Final refine with projected gradient descent
        h_final = projected_gradient_descent(h_sa)
        c5_final = evaluate_erdos_solution(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            best_n = n_points

    # Final normalization & evaluation
    if best_h is not None:
        best_h = best_h * (best_n / (2 * np.sum(best_h)))
        final_c5 = evaluate_erdos_solution(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5

    return (best_h, best_c5, best_n)
