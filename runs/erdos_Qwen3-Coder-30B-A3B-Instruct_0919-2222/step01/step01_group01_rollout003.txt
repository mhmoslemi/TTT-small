def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_simplex(v, s=1.0):
    """Projects vector v onto the simplex {x : x >= 0, sum(x) = s} using water-filling."""
    v = np.asarray(v)
    n = len(v)
    u = np.sort(v)[::-1]
    cssu = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssu - s))[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        theta = (cssu[rho[-1]] - s) / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)

def smooth_max(h, beta=100.0):
    """Smooth approximation to max using softmax."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return -np.log(np.sum(np.exp(-beta * corr))) / beta

def smooth_max_grad(h, beta=100.0):
    """Gradient of smooth_max with respect to h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    exp_term = np.exp(-beta * corr)
    Z = np.sum(exp_term)
    exp_term = exp_term / Z
    # Compute gradient of corr w.r.t. h via convolution
    grad_corr = np.zeros_like(h)
    for i in range(len(h)):
        grad_corr[i] = np.sum(exp_term[i:] * (1 - h[i:]) * dx) - np.sum(exp_term[:i] * h[:i] * dx)
    
    return -beta * grad_corr / Z
