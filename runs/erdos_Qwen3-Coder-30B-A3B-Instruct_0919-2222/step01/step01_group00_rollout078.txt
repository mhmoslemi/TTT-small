import numpy as np
import time
from scipy.fft import rfft, irfft

def proj_capped_simplex(v, target_sum):
    """
    Project vector v onto capped simplex: sum = target_sum, 0 <= v[i] <= 1
    """
    v = np.array(v)
    n = len(v)
    # Binary search for lambda
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find the largest k such that v_sorted[k] > (target_sum - cumsum[k-1]) / (k+1)
    # This is the 'k' such that sum of clipped values is target_sum.
    # Special case at beginning is when k=0
    lambda_vals = (cumsum - target_sum) / np.arange(1, n+1)
    k = np.searchsorted(lambda_vals, v_sorted, side='right') - 1
    if k < 0:
        # All elements are larger than what's allowed
        lam = target_sum / n
        return np.clip(v - lam, 0, 1)
    else:
        # tv = target_sum
        # sum_{j=0}^{k} max(v_sorted[j] - lambda, 0)
        # = sum_{j=0}^{k} (v_sorted[j] - lambda)
        # = v_sorted[0..k] sum - (k+1)*lambda
        # so lambda = (v_sorted[0..k] sum - target_sum) / (k+1)
        lambda_opt = (cumsum[k] - target_sum) / (k+1)
        v_projected = np.clip(v_sorted - lambda_opt, 0, 1)
        return v_projected

def full_corr(h):
    """
    Fast full correlation using FFT: correlation of h with (1 - h)
    """
    h = np.asarray(h)
    h = h - np.mean(h)  # Shift to avoid DC
    n = len(h)
    # conv = correlate(h, (1 - h), mode='full') but faster via FFT
    # Use rfft due to symmetry: if h is real, rfft has fewer elements
    H = rfft(h, n=2*n-1)  # length 2*n-1
    G = rfft(1-h, n=2*n-1)  # also same
    corr = irfft(H * np.conj(G), n=2*n-1)
    # corr has length 2*n - 1
    return corr[:n]  # Just overlap part

def evaluate_h(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    conv = full_corr(h)
    return np.max(conv) * dx

def sa_search(initial_h, n_points, S, max_time_s):
    # Use block-based representation. Split into S blocks, each of length block_len = n_points // S
    block_len = n_points // S
    h_s = np.array(initial_h).reshape(S, block_len).mean(axis=1)

    def obj(h_s):
        h = np.repeat(h_s, block_len)
        return evaluate_h(h)

    # Plan: use Metropolis with different step sizes
    T0 = 0.1 * obj(h_s)
    T = T0
    alpha = 0.997
    best_h_s = h_s.copy()
    best_obj = obj(h_s)
    cbest = best_obj
    time_limit = time.time() + max_time_s

    delta_max_list = [0.3, 0.15, 0.075]
    for seed_idx, delta_max in enumerate(delta_max_list):
        np.random.seed(seed_idx)
        h_s_copy = h_s.copy()
        T = T0
        while T > 1e-4 and time.time() < time_limit:
            i, j = np.random.randint(0, S, size=2)
            if i == j:
                continue
            delta = np.random.uniform(-delta_max, delta_max)
            h_s_new = h_s_copy.copy()
            h_s_new[i] += delta
            h_s_new[j] -= delta
            h_s_new = proj_capped_simplex(h_s_new, S / 2)
            obj_new = obj(h_s_new)
            obj_old = obj(h_s_copy)
            delta_obj = obj_new - obj_old
            prob = np.exp(-delta_obj / T) if T > 1e-10 else 0
            if delta_obj < 0 or np.random.rand() < prob:
                h_s_copy = h_s_new
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_h_s = h_s_copy.copy()
            T *= alpha
            if best_obj < cbest:
                cbest = best_obj
    return best_h_s, best_obj

def pgd_refine(h_s, n_points, S, block_len, max_time_s):
    # Reuse same setup as SA. h_s has S block heights.
    # Run gradient descent with log-sum-exp (smoothed max)
    # Target is single shift k that maximizes correlation
    def obj_smooth(h_s, alpha=10.0):
        h_full = np.repeat(h_s, block_len)
        c = full_corr(h_full)
        # Smoothed max
        c_c = c - np.max(c)
        exp_c = np.exp(alpha * c_c)
        sm = np.log(np.sum(exp_c)) / alpha
        return sm

    def grad_smooth(h_s, alpha=10.0):
        h = np.repeat(h_s, block_len)
        c = full_corr(h)
        c_c = c - np.max(c)
        exp_c = np.exp(alpha * c_c)
        w = exp_c / np.sum(exp_c)
        # Now compute gradient with respect to each h[i] using indices where convolution occurs
        g = np.zeros_like(h)
        for k in range(len(w)):
            shift = k  # k-th entry in corr = h * (1-h) shifted by k.
            # This is conv[i] = sum h[i+t] * (1 - h[i+t+k]) = sum_{x: x+k] h[x] * (1-h[x+k])
            # So ∂c_k / ∂ h[i] = 1 - h[i+k] - h[i-k]
            h_k = np.roll(h, k, axis=0)
            h_kl = np.roll(h, -k, axis=0)
            g += w[k] * (1 - h_k - h_kl)
        g = g[:len(h_s)*block_len]
        # Aggregate gradient back to h_s
        grad = np.zeros(S)
        for i in range(S):
            grad[i] = np.mean(g[i*block_len:(i+1)*block_len])
        return grad

    # Initialize
    h_s = h_s.copy()
    alpha = 5.0
    eta = 0.05
    best_obj = obj_smooth(h_s, alpha)
    best_h_s = h_s.copy()
    max_iter = 200
    iter_count = 0
    start = time.time()

    while iter_count < max_iter:
        if time.time() - start > max_time_s:
            break
        g = grad_smooth(h_s, alpha)
        h_s = h_s - eta * g
        h_s = proj_capped_simplex(h_s, S / 2)
        iter_count += 1
        new_val = obj_smooth(h_s, alpha)
        if new_val < best_obj:
            best_obj = new_val
            best_h_s = h_s.copy()
        # Adaptive updates
        if iter_count % 50 == 0:
            eta *= 0.95
            alpha = min(alpha * 1.5, 200.0)
            if alpha > 100:
                eta *= 0.98
    return best_h_s, best_obj

def binary_hill_climb(n_points, max_time_s):
    # Generate binary vector with exactly n_points/2 ones
    h_bin = np.zeros(n_points)
    ones_idx = np.random.permutation(n_points)[:n_points//2]
    h_bin[ones_idx] = 1
    c_best = evaluate_h(h_bin)
    cbest = c_best

    t_start = time.time()
    iter_max = 5000000  # 5m iterations -- should be plenty for fast evaluation
    for i in range(iter_max):
        if time.time() - t_start > max_time_s:
            break
        i1 = np.random.choice(np.where(h_bin == 1)[0])
        i0 = np.random.choice(np.where(h_bin == 0)[0])
        h_new = h_bin.copy()
        h_new[i1] = 0
        h_new[i0] = 1
        c_new = evaluate_h(h_new)
        if c_new < c_best:
            c_best = c_new
            h_bin = h_new
        if c_best < cbest:
            cbest = c_best
    return h_bin, c_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Prepare data structures and parameters
    n_points = 140
    S = 14  # number of block heights
    block_len = n_points // S

    # Step 1: Baseline
    baseline_h = np.array(initial_h_values)
    baseline_c5 = evaluate_h(baseline_h)
    best_c5 = baseline_c5
    best_h = baseline_h.copy()

    # Step 2: Simulated Annealing
    remaining = budget_s * 0.30
    h_s_best_sa, c5_sa = sa_search(baseline_h, n_points, S, remaining)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = np.repeat(h_s_best_sa, block_len)

    # Step 3: Projected Gradient Descent
    remaining -= (time.time() - start_time)
    if remaining > 1.0:
        # Run twice: once from SA best, second from updated version
        h_s_pg, c5_pg = pgd_refine(h_s_best_sa, n_points, S, block_len, remaining * 0.7)
        best_h_new = np.repeat(h_s_pg, block_len)
        c_current = evaluate_h(best_h_new)
        if c_current < best_c5:
            best_c5 = c_current
            best_h = best_h_new

        # Second refined run
        remaining -= (time.time() - start_time)
        if remaining > 1.0:
            h_s_pg2, c5_pg2 = pgd_refine(h_s_pg, n_points, S, block_len, remaining * 0.5)
            best_h_new2 = np.repeat(h_s_pg2, block_len)
            c_current2 = evaluate_h(best_h_new2)
            if c_current2 < best_c5:
                best_c5 = c_current2
                best_h = best_h_new2

    # Step 4: Binary hill climb
    binary_remaining = budget_s * 0.15
    binary_h, c5_bin = binary_hill_climb(n_points, binary_remaining)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = binary_h

    # Final projection for accuracy
    # Convert back to full length and recheck
    if len(best_h) != n_points:
        best_h = np.repeat(np.array(best_h).reshape(-1), block_len)
    else:
        pass  # binary already correct length
    
    # Now enforce constraints properly using simplex projection
    dx = 2.0 / n_points
    # Project back to exact sum constraint
    xl = np.clip(best_h, 0, 1)
    prop_sum = np.sum(xl) * dx
    ref_sum = 1.0
    scale_factor = ref_sum / prop_sum
    if abs(prop_sum - ref_sum) > 1e-10:
        xl = xl * scale_factor
        # And clip
        xl = np.clip(xl, 0, 1)
        # Re-calibrate
        prop_sum = np.sum(xl) * dx
        if abs(prop_sum - ref_sum) > 1e-10:
            # Recalculate sum and small adjustment
            xl = xl * (ref_sum / prop_sum)

    final_c5 = evaluate_h(xl)
    return (xl, final_c5, n_points)
