```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different initialization strategies
    init_strategies = [
        # Original initialization
        lambda: h0.copy(),
        # Random with constraints
        lambda: np.random.rand(n_points) * 0.5,
        # Piecewise constant
        lambda: np.concatenate([np.ones(n_points//3), np.zeros(n_points - n_points//3)]) * (3.0 / n_points),
        # Step function
        lambda: np.concatenate([np.ones(n_points//2), np.zeros(n_points - n_points//2)]) * (2.0 / n_points),
        # Sigmoid-like
        lambda: 1.0 / (1.0 + np.exp(-10 * (np.linspace(0, 2, n_points) - 1)))
    ]
    
    for i, init_func in enumerate(init_strategies):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        try:
            h_init = init_func()
            h_init = h_init / (np.sum(h_init) * dx) * 1.0  # Scale to sum = 1
            h_init = np.clip(h_init, 0, 1)
            
            # Local optimization
            local_result = minimize(
                objective,
                h_init,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Try multiple restarts with random initializations
    for i in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Use different random seeds for better exploration
        np.random.seed(seed + i)
        h_rand = np.random.rand(n_points) * 0.8  # Keep in [0, 0.8]
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Try genetic algorithm as alternative to standard local opt
    try:
        # Use differential evolution with careful setup to reduce running time
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=10,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-5,
            rtol=1e-5
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Final fine-tuning with more runs
    for restart in range(3):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        np.random.seed(seed + 10 + restart)
        h_rand = np.random.rand(n_points) * 0.6 + 0.2  # Between 0.2 and 0.8
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1500, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
```
