def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    # Set random seeds
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 0: Parameter selection
    n_list = [70, 64, 80]  # prioritize middle size
    n = 70  # default
    if initial_h_values is not None and isinstance(initial_h_values, list):
        n = len(initial_h_values)
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.ones(70) * 0.5

    # Final optimized h will be stored here
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Phase 1: Try MILP
    time_left = budget_s - (time.time() - start_time)
    if time_left > 120:
        try:
            for n_try in n_list:
                if time.time() - start_time > budget_s - 100:
                    break
                try:
                    h_milp = solve_milp_exact(n_try, time_left - 100)
                    if h_milp is not None:
                        c5_val = evaluate_erdos_solution(h_milp)
                        if c5_val < best_c5:
                            best_h = h_milp.copy()
                            best_c5 = c5_val
                            time_left = budget_s - (time.time() - start_time)
                except:
                    continue
        except: pass

    # Phase 2: Continuous Relaxation
    time_left = budget_s - (time.time() - start_time)
    if time_left > 100:
        try:
            h_relaxed = solve_lp_relaxation(80, time_left - 50)
            if h_relaxed is not None:
                c5_val = evaluate_erdos_solution(h_relaxed)
                if c5_val < best_c5:
                    best_h = h_relaxed.copy()
                    best_c5 = c5_val
        except: pass

    # Phase 3: Sequential Linear Programming (SCC)
    time_left = budget_s - (time.time() - start_time)
    if time_left > 200:
        try:
            h_scc = sequential_linearization(best_h, time_left - 150)
            if h_scc is not None:
                c5_val = evaluate_erdos_solution(h_scc)
                if c5_val < best_c5:
                    best_h = h_scc.copy()
                    best_c5 = c5_val
        except: pass

    # Phase 4: Bisection over t
    time_left = budget_s - (time.time() - start_time)
    if time_left > 60:
        try:
            new_h = bisection_feasibility(best_h, best_c5, time_left - 30)
            if new_h is not None:
                c5_val = evaluate_erdos_solution(new_h)
                if c5_val < best_c5:
                    best_h = new_h.copy()
                    best_c5 = c5_val
        except: pass

    # Phase 5: Stochastic polishing
    time_left = budget_s - (time.time() - start_time)
    if time_left > 50:
        try:
            polished_h = stochastic_polish(best_h, time_left - 10)
            if polished_h is not None:
                c5_val = evaluate_erdos_solution(polished_h)
                if c5_val < best_c5:
                    best_h = polished_h.copy()
                    best_c5 = c5_val
        except: pass

    return (best_h.tolist(), best_c5, len(best_h))

def solve_milp_exact(n_points, time_limit_s):
    # Solve binary MILP with squeeze
    dx = 2.0 / n_points
    h = cp.Variable(n_points, boolean=True)
    t = cp.Variable()

    # Constraints on h (sum)
    constraints = [cp.sum(h) == n_points / 2]

    # Build all shift terms
    all_shifts = []
    for d in range(-n_points+1, n_points):
        # Type check
        if not isinstance(d, int):
            continue
        shift = []
        for i in range(n_points):
            if 0 <= i+d < n_points:
                shift.append((i, i+d))  # valid indices
        all_shifts.append((d, shift))

    # Add auxiliary variables and McCormick constraints for all pairs
    y = {}
    obj_terms = []
    for d, pairs in all_shifts:
        # explicit zero padding
        if not pairs:
            continue
        y_key = f"d_{d}"
        y[y_key] = cp.Variable(len(pairs), boolean=True)
        # add constraints
        for idx, (i, j) in enumerate(pairs):
            # constraint: y[...][idx] <= h[i], h[j]
            # y[...][idx] >= h[i] + h[j] - 1
            # y[...][idx] >= 0
            constraints.append(y[y_key][idx] <= h[i])
            constraints.append(y[y_key][idx] <= h[j])
            constraints.append(y[y_key][idx] >= h[i] + h[j] - 1)
            constraints.append(y[y_key][idx] >= 0)
        obj_terms.append(cp.sum(y[y_key]))

    # Add inequality t >= dx*(n/2 - sum_of_products) for each d
    for d, _ in all_shifts:
        if not any((d, shift) for shift in all_shifts if shift[0] == d):
            continue
        y_key = f"d_{d}"
        if y_key in y:
            constraints.append(t >= dx * (n_points/2 - cp.sum(y[y_key])))

    # Solve
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=GLPK_MI, time_limit=time_limit_s)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            final_h = h.value.astype(float)
            return final_h
    except:
        return None
    return None

def solve_lp_relaxation(n_points, time_limit_s):
    dx = 2.0 / n_points
    h = cp.Variable(n_points, pos=True)
    t = cp.Variable()

    # Constraint: sum = n/2
    constraints = [cp.sum(h) == n_points / 2]

    # Build all shift terms and linearize
    all_shifts = []
    for d in range(-n_points+1, n_points):
        pairs = []
        for i in range(n_points):
            if 0 <= i+d < n_points:
                pairs.append((i, i+d))
        if pairs:
            all_shifts.append((d, pairs))

    y = {}
    objective = 0
    for d, pairs in all_shifts:
        if not pairs:
            continue
        y_key = f"d_{d}"
        y[y_key] = cp.Variable(len(pairs), pos=True, name=y_key)
        # Add linear McCormick constraints
        for idx, (i, j) in enumerate(pairs):
            constraints.append(y[y_key][idx] <= h[i])
            constraints.append(y[y_key][idx] <= h[j])
            constraints.append(y[y_key][idx] >= h[i] + h[j] - 1)
            constraints.append(y[y_key][idx] >= 0)
        objective += cp.sum(y[y_key])

    # Inequality constraint t >= dx*(n/2 - sum(y_{i,d}))
    constraints.append(t >= dx * (n_points/2 - objective))

    # Solve problem
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=GLPK, time_limit=time_limit_s)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return h.value
    except:
        return None
    return None

def sequential_linearization(initial_h, max_time_s):
    start = time.time()
    dx = 2.0 / len(initial_h)
    h = np.array(initial_h)
    h_best = h.copy()

    def compute_shifted_products(h, d):
        y = []
        for i in range(len(h)):
            j = i + d
            if 0 <= j < len(h):
                y.append(h[i] * h[j])
        return np.sum(y)

    def get_linearized_h(h_old):
        h_new = np.copy(h_old)
        best_d = 0
        max_qval = 0
        # find max shift's actual Q
        for d in range(-len(h_old)+1, len(h_old)):
            if abs(d) < len(h_old):
                q_val = compute_shifted_products(h_old, d)
                if q_val > max_qval:
                    max_qval = q_val
                    best_d = d
        # Linearationalize around this worst d
        contribs = []
        for i in range(len(h_old)):
            j = i + best_d
            if 0 <= j < len(h_old):
                contribs.append(h_old[i] * h_old[j])  # This is exact term

        linear_sum = 0
        for i in range(len(h_old)):
            j = i + best_d
            if 0 <= j < len(h_old):
                linear_sum += (
                    h_old[i] * h_old[j] +
                    h_old[i] * (h_old[j] - h_old[j]) +  # H_i * (h_j - h_old_j)
                    h_old[j] * (h_old[i] - h_old[i])   # h_j * (h_i - h_old_i)
                )
                # This linearization is exact for fixed h_old, because it is a 1st order Taylor.
                # But we can use the math required
        m = len(h_old)
        # Formulate delayed:
        # minimize t
        # subject to t >= dx*(m/2 - sum over i: term_i)
        # where term_i = val +
        #         h_old[i] * (h[i+d] - h_old[i+d]) +
        #         h_old[i+d] * (h[i] - h_old[i])

        def new_linear_term(i):
            j = i + best_d
            if 0 <= j < m:
                return h_old[i]*h_old[j] + h_old[i]*(h[j] - h_old[j]) + h_old[j]*(h[i] - h_old[i])
            return 0.0

        linear_terms = np.array([new_linear_term(i) for i in range(m)])
        return linear_terms


    iter_counter = 0
    while time.time() - start < max_time_s - 20 and iter_counter < 1000:
        # get a feasible step:
        old_h = h.copy()
        h_new = h.copy()
        # Reinvent the LP-based SCC approach:
        # minimize t
        # s.t.
        #   t >= dx * ( n/2 - q1(h) ), for some worst d
        #   0 <= h[i] <= 1
        #   sum(h) = n/2

        # But now we use the Pearson moment form:

        # Compute worst shift
        max_val = -1e9
        worst_d = 0
        for d in range(-len(h)+1, len(h)):
            s = 0
            for i in range(len(h)):
                j = i + d
                if 0 <= j < len(h):
                    s += h[i] * h[j]
            if s > max_val:
                max_val = s
                worst_d = d

        # Now form approximate problem for this shift
        A = np.zeros((len(h), len(h)))
        b = np.zeros(len(h))
        # iterate through entries:
        for i in range(len(h)):
            j = i + worst_d
            if 0 <= j < len(h):
                # We add linearization constraint for term h[i]*h[j]
                # h[i]*h[j] - h[i]*h_old[j] - h_old[i]*h[j] + h_old[i]*h_old[j]
                A[i, i] += 1
                A[i, j] += 1  # linear term: h_old[j] * h[i] = h_old[j]*h[i] bcoz fixed
                A[j, i] += 1  # . . . h_old[i] * h[j]
                A[j, j] += 1  # same constant scaled
                b[i] += h[i] * h[j]  # contstant term

        # Quadratic parts from linearization to form:
        # h_new[i] * h_new[j]  ≈  h_old[i] * h_old[j] + h_old[i](h_new[j] - h_old[j]) + h_old[j](h_new[i] - h_old[i])
        # So constraint is:
        # t >= dx * (n/2 - [const + h_old[j] (h_new[i] - h_old[i]) + h_old[i] (h_new[j] - h_old[j])])

        # The final solution:
        t = cp.Variable()
        h_var = cp.Variable(len(h))
        constraints = [
            cp.sum(h_var) == len(h)/2,
            h_var >= 0,
            h_var <= 1,
        ]
        y = []
        for d in range(-len(h)+1, len(h)):
            t_base = 0
            for i in range(len(h)):
                j = i + d
                if 0 <= j < len(h):
                    y.append(h_var[i] * h_var[j])
                    t_base += h_var[i] * h_var[j]
            constraints.append(t >= dx * (len(h)/2 - t_base))

        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False, time_limit=max_time_s/2 - 5)
            if prob.status == 'optimal':
                h = h_var.value
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    h_best = h.copy()
                    best_c5 = current_c5
        except:
            pass
        iter_counter += 1
    return h_best

def bisection_feasibility(h_start, current_bound, max_time_s):
    dx = 2.0 / len(h_start)
    n = len(h_start)
    time_left = time.time() + max_time_s
    low = 0.0
    high = current_bound
    tolerance = 1e-4
    h_best = h_start.copy()
    best_bound = current_bound

    def _check_feasible(target_c5):
        # Reconstruct problem wit h variable and fixed t
        h_vars = cp.Variable(n)
        # stub for construction — just solve with fixed t
        obj = cp.Minimize(0)
        constraints = [
            cp.sum(h_vars) == n/2,
            h_vars >= 0,
            h_vars <= 1
        ]
        all_d = []
        for d in range(-n+1, n):
            if abs(d) < n:
                total = 0
                # compute sum h(i) * h(i+d) for this d:
                for i in range(n):
                    j = i + d
                    if 0 <= j < n:
                        total += h_vars[i] * h_vars[j]
                all_d.append(total)
                # We use no special inequality here, just inline use goal
                # Directly compute contribution for current t:
                # t >= dx * (n/2 - sum)
                constraints.append(cp.sum(h_vars) * dx >= target_c5 + dx * n/2 - cp.sum(all_d))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(verbose=False, solver=GLPK, time_limit=max_time_s/5)
            return prob.status in ['optimal', 'optimal_inaccurate']
        except:
            return False

    while high - low > tolerance and time.time() < time_left:
        mid = (low + high) / 2.0
        if _check_feasible(mid):
            high = mid
        else:
            low = mid
        if time.time() > time_left:
            break
    return h_best

def stochastic_polish(h_start, max_time_s):
    h = h_start.copy()
    dx = 2.0 / len(h)
    current_c5 = evaluate_erdos_solution(h)

    def project_onto_sum_box(h_vec, target_sum):
        s = np.sum(h_vec)
        factor = target_sum / s
        h_vec = h_vec * factor
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec

    start_time = time.time()
    while time.time() - start_time < max_time_s - 5:
        old_h = h.copy()
        old_c5 = current_c5
        # Swap move: draw two idx with difference noise
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        if i == j:
            continue
        delta = np.random.uniform(-0.05, 0.05)
        if not (0 <= h[i] - delta <= 1 and 0 <= h[j] + delta <= 1):
            continue
        h[i] -= delta
        h[j] += delta
        # Ensure sum constraint (project to correct support)
        h = project_onto_sum_box(h, len(h)/2)
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < current_c5:
            current_c5 = new_c5
        else:
            h = old_h.copy()
            current_c5 = old_c5
    return h
