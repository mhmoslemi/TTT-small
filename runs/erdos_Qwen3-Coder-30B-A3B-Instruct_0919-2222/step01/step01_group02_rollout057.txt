import numpy as np
import time
import cvxpy as cp
from scipy.linalg import eigh
from scipy.optimize import differential_evolution

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    # Set random seeds
    np.random.seed(seed)
    start_time = time.time()
    
    # Define constants
    n_points = 70
    dx = 2.0 / n_points
    
    # Define SDP variables and problem setup
    # We use the transformation a = 2h - 1 to map h in [0,1] to a in [-1,1]
    # We want to minimize max_d C(d) = dx * (n/2 - sum(a_i a_{i+d})/4)
    # This is equivalent to maximizing min_d sum(a_i a_{i+d})
    
    # Use SDP relaxation approach
    def solve_sdp(n):
        # Using cvxpy to solve the SDP relaxation of the problem
        # Variables
        a = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            X == a @ a.T,
            cp.diag(X) <= 1,
            cp.sum(a) == 0,
            t >= 0
        ]
        
        # Objective
        objective = cp.Minimize(t)
        
        # Problem instance
        prob = cp.Problem(objective, constraints)
        
        # Solve with SCS and return a
        prob.solve(solver=cp.SCS, eps=1e-6, max_iters=5000)
        
        if prob.status == cp.OPTIMAL:
            return a.value
        else:
            return None
    
    # Attempt to initialize from SDP solution
    a_init = solve_sdp(n_points)
    if a_init is not None:
        # Project back to h
        h = (a_init + 1) / 2
        # Renormalize
        h *= (n_points / 2) / np.sum(h)
    else:
        # Fallback
        h = np.ones(n_points) * 0.5
        h *= (n_points / 2) / np.sum(h)
        
    # Evaluate initial solution
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Refinement using local search
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec)

    def project_to_feasible(h_vec):
        # Ensure constraints
        h_vec = np.clip(h_vec, 0, 1)
        # Renormalize to sum to n_points / 2
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec

    def safe_gradient_descent(h0, max_iter=100, tol=1e-6):
        h = h0.copy()
        for i in range(max_iter):
            old_val = objective(h)
            # Simple gradient ascent step with projection
            h = project_to_feasible(h + 0.001 * (1 - 2*h))
            new_val = objective(h)
            if abs(old_val - new_val) < tol:
                break
        return h
    
    # Refine using local gradient ascent
    refined_h = safe_gradient_descent(best_h)
    refined_c5 = evaluate_erdos_solution(refined_h)
    if refined_c5 < best_c5:
        best_h = refined_h
        best_c5 = refined_c5
    
    # Final check of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Incorporate a small amount of secondary refinement using differential evolution
    def hp_obj(hv):
        return evaluate_erdos_solution(np.clip(hv, 0, 1) * (n_points / 2) / np.sum(np.clip(hv, 0, 1)))

    # Differential evolution for further improvement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run for a short time
    time_limit = start_time + budget_s - 5 
    de_start = time.time()
    
    # Limit samples if necessary
    samples = min(1000, int(50000 * (time.time() - de_start) / (time_limit - de_start)) or 1000)
    result = differential_evolution(
        hp_obj, 
        bounds,
        seed=seed+123,
        maxiter=100,
        popsize=10,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-10,
        callback=lambda x, convergence: None,
        workers=1,
        disp=False
    )
    
    # Process DE result
    if result.success:
        h_DE = np.clip(result.x, 0, 1) * (n_points / 2) / np.sum(np.clip(result.x, 0, 1))
        c5_DE = evaluate_erdos_solution(h_DE)
        
        if c5_DE < best_c5:
            best_h = h_DE
            best_c5 = c5_DE
    
    return (best_h.tolist(), best_c5, len(best_h))
