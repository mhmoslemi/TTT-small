import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C_5 bound
    c5 = max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Try several grid sizes
    grid_sizes = [42, 48, 60]
    best_h = None
    best_c5 = float('inf')
    best_n = None

    for n in grid_sizes:
        if time.time() - start_time > 0.8 * budget_s:
            break

        # Construct MILP model for binary values
        b = cp.Variable(n, integer=True)
        z = {}
        s = [cp.Variable(1, integer=True) for d in range(n)]
        t = cp.Variable(1)

        # Linearization constraints for z_{i,d}
        # d = 1 to n-1
        for d in range(1, n):
            for i in range(n - d):
                z_var = cp.Variable(1, integer=True)
                z[(i, d)] = z_var
                # McCormick envelopes
                constraints = [
                    z_var <= b[i],
                    z_var <= b[i + d],
                    z_var >= b[i] + b[i + d] - 1
                ]
                # Add constraints to list
                for c in constraints:
                    s[d].constraints.append(c)

        # Use association with sum of z variables
        # Sum over specified overlapping indices for each shift d
        # Make s_d = sum_{i} z_{i,d} where valid
        for d in range(n):
            s[d] = cp.Variable(1, integer=True)
        
        # Construct objective: minimize t
        constraints = [cp.sum(b) == n // 2, 
                       t >= 0]

        # Add constraints for each s_d
        for d in range(n):
            start_idx = 0
            end_idx = n - d
            idxs = range(start_idx, end_idx)
            s_expr = cp.sum([z[(i, d)] for i in idxs])
            constraints.extend([s_expr == s[d], t >= n/2 - s[d]])
        
        # Add b in [0,1]
        for i in range(n):
            constraints.append(b[i] >= 0)
            constraints.append(b[i] <= 1)

        # Symmetry breaking
        constraints.append(b[0] == 1)

        # Build objective and solve
        prob = cp.Problem(cp.Minimize(t), constraints)

        try:
            # Allow 200s for solving MILP (configurable)
            prob.solve(solver=cp.CBC, verbose=False, maxSeconds=200)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                b_val = b.value
                c5_milp = t.value
                if c5_milp < best_c5:
                    best_h = b_val
                    best_c5 = c5_milp
                    best_n = n
        except:
            pass  # Skip if MILP failed to solve

    # If no binary MILP solution found, use initial_h_values
    if best_h is None:
        best_h = initial_h_values.copy()
        best_n = len(best_h)
        best_c5 = evaluate_erdos_solution(best_h)

    # Stage 2: Fractional refinement using Simulated Annealing
    dx = 2.0 / best_n
    h = np.array(best_h, dtype=float)
    h = h * (best_n / (2 * np.sum(h)))  # Ensure sum = n/2

    best_h_refined = h.copy()
    best_c5_refined = evaluate_erdos_solution(h)
    
    # Simulated Annealing parameters
    temp = 0.02 * best_c5_refined
    alpha = 0.995
    max_iter = 50000
    no_improvement_threshold = 5000
    no_improvement_count = 0

    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break

        # Pick two random indices
        i1 = np.random.randint(0, best_n)
        i2 = np.random.randint(0, best_n)
        if i1 == i2:
            continue

        # Pick transfer amount
        delta_max = min(h[i1], 1 - h[i2], 0.05)
        if delta_max <= 0:
            continue
        delta = np.random.uniform(0, delta_max)

        # Make a copy and modify
        h_new = h.copy()
        h_new[i1] -= delta
        h_new[i2] += delta

        # Ensure 0 <= h_new <= 1
        h_new = np.clip(h_new, 0, 1)

        # Renormalize to ensure sum = n/2
        h_new = h_new * (best_n / (2 * np.sum(h_new)))

        # Compute new bound
        c5_new = evaluate_erdos_solution(h_new)

        # Acceptance rule
        if c5_new < best_c5_refined:
            h = h_new
            best_c5_refined = c5_new
            no_improvement_count = 0
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(c5_new - best_c5_refined) / temp):
                h = h_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1

        # Adjust temperature
        temp *= alpha

        if no_improvement_count > no_improvement_threshold:
            temp = 0.00001  # Final low temperature

    return (best_h_refined, best_c5_refined, best_n)
