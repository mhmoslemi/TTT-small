import numpy as np
from scipy.fft import rfft, irfft
import time
from math import exp

def proj_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= v[i] <= 1, sum(v) = target_sum"""
    v = np.array(v)
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if v_sum == target_sum:
        return v
    elif v_sum > target_sum:
        # Use water-filling approach
        v = np.sort(v)[::-1]
        cumsum = np.cumsum(v)
        n = len(v)
        i = 0
        while i < n and v[i] + (target_sum - cumsum[i]) / (i + 1) > 1:
            i += 1
        if i == 0:
            w = np.ones(n) * target_sum/n
        else:
            alpha = (target_sum - cumsum[i - 1]) / i
            w = np.maximum(v - alpha, 0)
            if i < n:
                w[i:] = 0
        return w
    else:
        # This shouldn't happen if we maintain always sum <= target
        # But to be safe, scale up
        scale = target_sum / v_sum
        v = v * scale
        return np.clip(v, 0, 1)

def full_corr(h):
    """Compute correlation of h[t] with (1-h[t+k]) over all shifts k"""
    n = len(h)
    h_ext = np.concatenate([h, np.zeros(n - 1)])
    g_ext = np.concatenate([1 - h, np.zeros(n - 1)])
    H = rfft(h_ext)
    G = rfft(g_ext)
    corr = irfft(H * np.conj(G), n=2*n - 1)
    return corr

def evaluate_h(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    conv = full_corr(h)
    max_val = np.max(conv)
    return max_val * dx

    def sa_run(h_s_start, T0, alpha, max_steps, delta_max):
        h_s = h_s_start.copy()
        current_C5 = evaluate_h(np.repeat(h_s, block_len))
        best_C5 = current_C5
        best_h_s = h_s.copy()
        T = T0

                break
            # Make a move
            i, j = np.random.choice(S, 2, replace=False)
            delta = np.random.uniform(-delta_max, delta_max)
            h_s_new = h_s.copy()
            h_s_new[i] += delta
            h_s_new[j] -= delta
            h_s_new = proj_capped_simplex(h_s_new, target_sum/S)
            new_C5 = evaluate_h(np.repeat(h_s_new, block_len))
            delta_C5 = new_C5 - current_C5
            if delta_C5 <= 0 or np.random.rand() < np.exp(-delta_C5 / T):
                h_s = h_s_new
                current_C5 = new_C5

            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h_s = h_s.copy()
            T *= alpha
            if T < 1e-4:

            steps += 1
        return best_h_s, best_C5

            weights = exp_acts / exp_acts_sum

            n = len(h_full)

            # Iterate over all shifts k
            for k in range(len(corr)):

                    shifted = np.concatenate([np.zeros(k_idx), h_full[:n-k_idx]])

                    grad += weights[k] * (1 - shifted - np.roll(shifted, k_idx))
                else:

                    grad += weights[k] * (1 - shifted - np.roll(shifted, -k_idx))

                h_s = h_s_new
                C5_prev = new_C5
                lr *= 0.95  # Adaptive learning rate
            else:
                lr *= 0.95
                if np.random.rand() < 0.01:
                    h_s = h_s_new  # small chance of accepting worse step
        return h_s, C5_prev

            # Pick random 1 and 0
