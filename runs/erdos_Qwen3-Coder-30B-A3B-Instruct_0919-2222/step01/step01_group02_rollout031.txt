import numpy as np
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, shift):
    """Shift array by given amount using zero padding"""
    n = len(arr)
    if shift >= 0:
        # shift left (positive shift): h[i] -> h[i+shift]
        return np.concatenate([arr[shift:], np.zeros(shift)])
    else:
        # shift right (negative shift): h[i] -> h[i+shift] 
        s = -shift
        return np.concatenate([np.zeros(s), arr[:n-s]])

def project_onto_box_simplex(v, lower=0., upper=1., target_sum=None):
    """Project vector v onto box [lower,upper] and simplex with given target_sum"""
    # Bisection to find λ such that sum(clamp(v - λ, lower, upper)) = target_sum
    n = len(v)
    if target_sum is None:
        target_sum = n / 2
        
    # Initial bounds
    lo, hi = -10.0, 10.0
    
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    # Apply final projection
    return np.clip(v - (lo + hi) / 2, lower, upper)

def subgradient_optimize(n, end_time, start_h=None):
    """Optimize h using projected subgradient method"""
    # Initialize h
    if start_h is not None:
        h = np.array(start_h).copy()
        h = project_onto_box_simplex(h)

        h = np.random.rand(n)

    best_C = evaluate_erdos_solution(h)
    iteration = 0
    last_improve_iter = 0
    stagnation = 0
    
    # Time stepping
    while time.time() < end_time:
        iteration += 1

        corr = np.correlate(h, 1 - h, mode='full')
        # Find shift that maximizes correlation

        # Compute direction: h[i+d] + h[i-d] -1 for valid indices
        shift_plus = shift_array(h, d)
        shift_minus = shift_array(h, -d)
        direction = shift_plus + shift_minus - 1.0
        
        # Step size
        α = 0.5 / np.sqrt(iteration)  # decreasing step size
        
        # Update step
        v = h + α * direction
        # Project onto feasible set
        h = project_onto_box_simplex(v)
        
        # Evaluate
        cur_C = evaluate_erdos_solution(h)
        if cur_C < best_C - 1e-10:
            best_C = cur_C
            best_h = h.copy()
            last_improve_iter = iteration
            stagnation = 0
        else:
            stagnation += 1
            
        # If no improvement for long, random restart
        if stagnation > 5000:
            # Restart with best found so far plus noise
            perturb = best_h + np.random.normal(scale=0.05, size=n)
            h = project_onto_box_simplex(perturb)
            stagnation = 0
            
    return best_h, best_C
