import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, S, lower=0.0, upper=1.0):
    """Project vector v onto bounds [lower,upper] and sum constraint S"""
    # First clip to [lower, upper]
    v = np.clip(v, lower, upper)
    # Adjust to meet sum constraint S
    current_sum = np.sum(v)
    if abs(current_sum - S) < 1e-10:
        return v
    diff = S - current_sum
    if abs(diff) < 1e-10:
        return v
    # Use simple adjustment: reduce sum by diff
    # Average adjustment per element needed to get diff
    # But we want to avoid pushing elements outside bounds
    # So distribute diff uniformly or use more sophisticated projection
    # Simple projection: subtract mean adjustment to get back to target.
    # If all elements are already at bound, we should not reduce further
    # Instead, perform water-filling approach
    
    # Let's use a more robust approach:
    # Use scipy.optimize to minimize distance to v subject to constraints
    # But simpler: just adjust total sum by shifting
    # Sort v descending
    idx_sorted = np.argsort(v)[::-1]
    v_sorted = v[idx_sorted]
    # Start from largest, adjust to sum to S
    adjusted_sum = np.sum(v_sorted)
    if abs(adjusted_sum - S) < 1e-10:
        return v
    # We reduce from largest first
    if adjusted_sum > S:
        # Need to reduce sum by (adjusted_sum - S)
        delta = adjusted_sum - S
        i = 0
        while i < len(v_sorted) and delta > 0:
            to_reduce = min(v_sorted[i] - lower, delta)
            v_sorted[i] -= to_reduce
            delta -= to_reduce
            i += 1
    elif S > adjusted_sum:
        # Need to increase sum by (S - adjusted_sum)
        delta = S - adjusted_sum
        i = 0
        while i < len(v_sorted) and delta > 0:
            to_increase = min(upper - v_sorted[i], delta)
            v_sorted[i] += to_increase
            delta -= to_increase
            i += 1
    # Reorder back
    v_out = v.copy()
    for i, orig_idx in enumerate(idx_sorted):
        v_out[orig_idx] = v_sorted[i]
    # Final check
    while abs(np.sum(v_out) - S) > 1e-10:
        # small fix
        diff = S - np.sum(v_out)
        # We can distribute difference 
        # Find element that can be adjusted most
        # Try to increase first non-full element
        # or decrease a full element
        j = np.argmax(v_out)  # index of max element which we can decrease
        if v_out[j] > lower:
            delta = min(diff, v_out[j] - lower)
            v_out[j] -= delta
            diff -= delta
        else:
            break
        
    return v_out

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Precompute shift indices for vectorized gradient
    shift_indices = []
    for s in range(-(n_points-1), n_points):
        if s >= 0:
            i = np.arange(0, n_points-s, dtype=int)
            j = i + s
        else:
            i = np.arange(-s, n_points, dtype=int)
            j = i + s
        shift_indices.append((i, j))
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Objective function for global optimization
    def objective(h):
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Use differential evolution for global search
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=15,
            seed=seed,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Gradient descent with approximate max and temperature-based approach
    # Try local search with projected gradient
    def compute_gradient(h, weights):
        """Compute gradient for weighted sum of correlation terms"""
        n = len(h)
        grad = np.zeros_like(h)
        # First term: derivative of sum_i h_i is constant 1
        # which contributes: sum_s w_s * 1. But that's already incorporated
        # Second term derivative: for each shift, sum_i h_i h_{i+s}
        # We compute for all shifts 
        total_weight = np.sum(weights)
        # Add constant component: each element + total_weight
        grad += total_weight
        
        # Now gradient from each shift i term
        for k, weight in enumerate(weights):
            if weight == 0:
                continue
            i, j = shift_indices[k]
            if len(i) == 0:
                continue
            grad[i] += weight * (-h[j])   # from term -h_i h_{i+s}
            grad[j] += weight * (-h[i])   # from term -h_{i+s} h_i
            
        return grad
    
    # Optimization settings
    max_iter = 30000
    eta0 = 0.5
    beta = 1.0
    beta_max = 100.0
    beta_increment = (beta_max - beta) / (max_iter // 100)
    
    # Start from best found solution
    h = best_h.copy()
    # Project to ensure it meets sum = n/2 exactly
    target_sum = n_points / 2.0
    h = project_onto_bounds_and_sum(h, target_sum)
    
    # Gradient descent loop with adaptive step
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Compute correlation values
        correlations = np.correlate(h, 1 - h, mode="full")
        f = correlations   # Leave out dx for weight computation
        
        # Softmax weights
        max_f = np.max(f)
        exp_terms = np.exp(beta * (f - max_f))
        w = exp_terms / np.sum(exp_terms)
        
        # Compute gradient
        grad = compute_gradient(h, w)
        
        # Adaptive step size
        eta = eta0 / np.sqrt(it + 10)
        
        # Update step
        h_new = h - eta * grad
        
        # Project
        h_new = project_onto_bounds_and_sum(h_new, target_sum)
        h = h_new
        
        # Update beta for soft max
        if it > 0 and it % 500 == 0:
            beta = min(beta + beta_increment, beta_max)
            
        # Evaluate
        if it % 500 == 0:
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final optimization with improved restarts
    num_restarts = 3
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        np.random.seed(seed + restart)
        h_restart = np.maximum(0.0, np.minimum(1.0, h + np.random.normal(0, 0.05, n_points)))
        h_restart = project_onto_bounds_and_sum(h_restart, target_sum)
        
        # Simple local search: random swaps
        for _ in range(3000):
            # Randomly select two positions
            i, j = np.random.choice(n_points, 2, replace=False)
            # Swap values
            h_swap = h_restart.copy()
            h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
            # Project to maintain sum
            h_swap = project_onto_bounds_and_sum(h_swap, target_sum)
            
            # Evaluate
            new_c5 = evaluate_erdos_solution(h_swap)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_swap.copy()
        
        h_restart = best_h.copy()  # Restart the GD from best so far
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
