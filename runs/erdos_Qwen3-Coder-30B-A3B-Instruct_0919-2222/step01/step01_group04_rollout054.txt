import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values is our C5
    return np.max(corr) * dx

def compute_binary_min_overlap(b):
    """Compute minimum overlap for binary vector b"""
    n = len(b)
    # Compute correlation of b with itself
    corr = np.correlate(b, b, mode="full")
    # Find minimum for non-zero lags
    min_overlap = np.min(corr[n:])  # from index n onwards, which are positive lags
    return min_overlap

def random_binary(N, weight):
    """Generate random binary vector with given weight"""
    b = np.zeros(N, dtype=int)
    indices = np.random.choice(N, size=int(weight), replace=False)
    b[indices] = 1
    return b

def binary_maximin(N, weight, time_limit=20):
    """Find good binary pattern using random sampling"""
    best_b = None
    best_min_overlap = -1
    start_time = time.time()
    
    # Try many random binary vectors
    num_trials = 10000
    for _ in range(num_trials):
        if time.time() - start_time > time_limit:
            break
            
        b = random_binary(N, weight)
        min_overlap = compute_binary_min_overlap(b)
        
        if min_overlap > best_min_overlap:
            best_min_overlap = min_overlap
            best_b = b.copy()
    
    return best_b, best_min_overlap

def smooth_binary(b, kernel=[0.25, 0.5, 0.25]):
    """Apply smoothing to binary vector"""
    h = np.convolve(b, kernel, mode='same')
    # Handle edge effects by padding zeros if necessary
    # Ensure smooth result stays in [0, 1]
    h = np.clip(h, 0, 1)
    return h

def project_onto_box_and_sum(v, target):
    """Project v onto box [0,1] and sum constraint"""
    # Use binary search to find lambda such that sum of clipped values equals target
    lo = -np.max(v) - 2.0
    hi = 1 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def grad_for_lag(h, lag):
    """Compute gradient of correlation with given lag"""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # Overlap region i in [0, N - lag)
        i = np.arange(N - lag)
        grad[i] += 1 - h[i + lag]  # derivative w.r.t h_i
        grad[i + lag] -= h[i]      # derivative w.r.t h_{i+lag}
    else:
        lag = -lag
        # Overlap region i in [lag, N)
        i = np.arange(lag, N)
        grad[i] += 1 - h[i - lag]
        grad[i - lag] -= h[i]
    return grad
