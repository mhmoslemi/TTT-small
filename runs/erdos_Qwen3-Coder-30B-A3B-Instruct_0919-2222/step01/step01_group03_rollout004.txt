import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Multi-start optimization with different strategies
    # Strategy 1: Differential Evolution
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=150,
            popsize=20,
            seed=seed,
            atol=1e-7,
            rtol=1e-7,
            mutation=(0.5, 1.0),
            recombination=0.7
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Strategy 2: Multiple Local Optimizations with Different Starting Points
    for restart in range(8):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate diverse random initializations
        # Method 1: Mix of constant and spike patterns
        if restart % 3 == 0:
            h_rand = np.random.rand(n_points) * 0.5
            h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        elif restart % 3 == 1:
            # Create a more concentrated pattern
            h_rand = np.ones(n_points) * 0.1
            # Add some spikes
            indices = np.random.choice(n_points, size=n_points//4, replace=False)
            h_rand[indices] = 0.8
            h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        else:  # restart % 3 == 2
            # Create buckets
            h_rand = np.zeros(n_points)
            bucket_size = n_points // 6
            for i in range(6):
                start = i * bucket_size
                end = (i+1) * bucket_size if i < 5 else n_points
                h_rand[start:end] = np.random.rand(1)[0] * 0.5 + 0.25
            h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
            
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1500, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Strategy 3: Gradient-free methods on good candidates
    # If we have a good solution already, try a local refinement
    try:
        # Try to refine with a more aggressive local search
        local_result = minimize(
            objective,
            best_h,
            method='Nelder-Mead',
            options={'maxiter': 1000, 'xatol': 1e-10, 'fatol': 1e-10},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
