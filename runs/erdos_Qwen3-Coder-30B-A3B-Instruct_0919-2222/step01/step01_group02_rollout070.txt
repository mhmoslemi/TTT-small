import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, shift):
    """Helper function to shift array with zero padding"""
    n = len(arr)
    if shift >= 0:
        # Shift left by shift positions with zero padding at end
        out = np.concatenate([arr[shift:], np.zeros(shift)])
    else:
        # Shift right by |shift| positions with zero padding at start
        s = -shift
        out = np.concatenate([np.zeros(s), arr[:n-s]])
    return out

def project_onto_box_simplex(v, lower=0., upper=1., target_sum=None):
    """Project vector onto box constraints [lower, upper] and sum constraint"""
    if target_sum is None:
        target_sum = len(v) / 2
    # Bisection for λ to satisfy sum(clamp(v_i - λ, lower, upper)) = target_sum
    lo = -1e3
    hi = 1e3
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, lower, upper)
