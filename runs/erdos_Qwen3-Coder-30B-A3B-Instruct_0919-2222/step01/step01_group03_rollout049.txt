import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Use advanced optimization with multiple strategies
    # Strategy 1: Global search with differential evolution
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=20,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-8,
            rtol=1e-8
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Strategy 2: Local search with L-BFGS-B
    try:
        local_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Strategy 3: Random restarts with adaptive perturbations
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break

        # Generate a perturbed initial guess that satisfies constraints
        h_rand = np.random.rand(n_points)
        # Make the vector more structured to avoid pure randomness
        # Try a piecewise constant construction
        if n_points >= 8:
            segments = np.random.randint(2, 5)
            segment_sizes = np.random.choice([1, 2, 3], segments, p=[0.4, 0.4, 0.2])
            segment_sizes = np.array(segment_sizes)
            total_segments = np.sum(segment_sizes)
            if total_segments < n_points:
                segment_sizes[-1] += n_points - total_segments
            h_rand = np.zeros(n_points)
            idx = 0
            for segment_size in segment_sizes:
                value = np.random.rand() * 0.8 + 0.1  # Between 0.1 and 0.9
                h_rand[idx:idx + segment_size] = value
                idx += segment_size
        else:
            h_rand = np.random.rand(n_points) * 0.5  # Between 0 and 0.5
        
        # Scale to meet constraint
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            # Local optimization
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Strategy 4: Random sampling and local search (for more diversity)
    for restart in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break

        # Generate multiple random samples that satisfy the integral constraint
        h_rand = np.random.rand(n_points)
        # Apply a sigmoid to make it slightly smoother
        h_rand = 1.0 / (1.0 + np.exp(-5 * (h_rand - 0.5)))
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
        h_rand = np.clip(h_rand, 0, 1)

        try:
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
