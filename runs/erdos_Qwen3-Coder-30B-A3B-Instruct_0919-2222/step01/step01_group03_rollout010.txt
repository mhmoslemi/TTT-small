import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector v onto [0,1] and sum constraint"""
    # First clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Adjust to meet sum constraint
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Simple adjustment using water-filling approach
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return v
    
    # Distribute difference
    if diff > 0:
        # Need to increase values
        # Increase all by same amount (minimum possible)
        increment = diff / len(v)
        v = np.minimum(v + increment, 1.0)
        
        # Adjust for overflows
        over_sum = np.sum(v) - target_sum
        if over_sum > 0:
            # Reduce all entries equally from highest
            reduction = over_sum / len(v)
            v = np.maximum(v - reduction, 0.0)
    elif diff < 0:
        # Need to decrease values
        decrement = -diff / len(v)
        v = np.maximum(v - decrement, 0.0)
        
        # Adjust for underflows
        under_sum = target_sum - np.sum(v)
        if under_sum > 0:
            # Increase all entries equally from lowest
            increment = under_sum / len(v)
            v = np.minimum(v + increment, 1.0)
    
    # Final correction to match sum exactly
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust most extreme value
        diff = target_sum - current_sum
        idx = np.argmax(v) if diff > 0 else np.argmin(v)
        v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
    
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Better initial shape: try triangle with careful scaling
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    h_triangle = np.clip(1 - np.abs(x-1)/1, 0, 1)
    h_triangle = h_triangle * (target_sum / np.sum(h_triangle))
    h0 = h_triangle.copy()
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Use binary search to find good middle point
    # Try few different local searches
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Large scale simulated annealing
    def attempt_sa(n_runs=10, iterations_per_run=50000):
        nonlocal best_h, best_c5
        for _ in range(n_runs):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Start from triangular
            h = h0.copy()
            current_c5 = best_c5
            
            # Temperature schedule
            temp = 0.01
            temp_decay = 0.9995
            
            for iter_num in range(iterations_per_run):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                # Perturb two points
                i = np.random.randint(0, n_points)
                j = np.random.randint(0, n_points)
                
                # Save old values
                old_i = h[i]
                old_j = h[j]
                
                # Try small random change
                delta = np.random.uniform(-0.05, 0.05)
                h[i] += delta
                h[j] -= delta
                
                # Maintain bounds and sum
                h = project_onto_bounds_and_sum(h, target_sum)
                
                # Calculate new C5
                new_c5 = evaluate_erdos_solution(h)
                
                # Metropolis accept
                if new_c5 < current_c5:
                    current_c5 = new_c5
                elif np.random.random() < np.exp(-(new_c5 - current_c5) / temp):
                    # Accept the worse solution
                    current_c5 = new_c5
                else:
                    # Revert
                    h[i] = old_i
                    h[j] = old_j
                
                # Cool down
                temp *= temp_decay
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    # Parallel local improvements using gradient descent
    def refine_with_gradient_descent(max_iter=10000):
        nonlocal best_h, best_c5
        h = best_h.copy()
        current_c5 = best_c5
        
        # Precompute shift indices
        shift_indices = []
        for s in range(-(n_points-1), n_points):
            if s >= 0:
                i = np.arange(0, n_points-s, dtype=int)
                j = i + s
            else:
                i = np.arange(-s, n_points, dtype=int)
                j = i + s
            shift_indices.append((i, j))
        
        # Annealing parameters
        eta = 0.05
        beta = 1.0
        beta_max = 50.0
        beta_inc = (beta_max - beta) / max_iter
        
        for it in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Compute raw values for all shifts
            corr_raw = np.correlate(h, 1-h, mode='full')
            f = corr_raw  # Already in correct form without dx
            
            # Softmax weights with increasing beta
            max_f = np.max(f)
            exp_terms = np.exp(beta * (f - max_f))
            w = exp_terms / np.sum(exp_terms)
            
            # Compute weighted gradient
            grad = np.zeros_like(h)
            
            # Add gradient contributions from all shifts
            for shift_idx, (i_arr, j_arr) in enumerate(shift_indices):
                weight = w[shift_idx]
                if weight <= 1e-10:
                    continue
                    
                # Add contribution from first part: + weight
                grad[i_arr] += weight * 1.0
                # Add contribution from product: - weight * h[j]
                grad[i_arr] -= weight * h[j_arr]
                # Add contribution from h[i] term in product: - weight * h[i]
                grad[j_arr] -= weight * h[i_arr]
            
            # Update step
            h_new = h - eta * grad
            
            # Project onto constraint set
            h_new = project_onto_bounds_and_sum(h_new, target_sum)
            
            # Evaluate
            new_c5 = evaluate_erdos_solution(h_new)
            
            # Accept change if better
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            
            # Lower learning rate and increase beta
            eta = max(0.001, eta * 0.9995)
            beta = min(beta_max, beta + beta_inc)
    
    # Run SA first
    attempt_sa(n_runs=3, iterations_per_run=20000)
    
    # Then gradient refinement
    refine_with_gradient_descent(max_iter=8000)
    
    return (best_h, best_c5, n_points)
