import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 140  # Good balance between resolution and computational cost
    n_steps = 14    # Number of steps for piecewise constant function
    dx = 2.0 / n_points
    
    # Compute segment length
    seg_len = n_points // n_steps
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    
    # Helper function to build h from step heights
    def build_h(heights):
        return np.repeat(heights, seg_len)
    
    # Objective function for differential evolution
    def objective(heights):
        # Build full h array
        h = build_h(heights)
        # Compute C5
        c5 = evaluate_h(h)
        # Add penalty for sum constraint
        sum_heights = np.sum(heights)
        penalty = 1000.0 * (sum_heights - n_steps/2)**2
        return c5 + penalty
    
    # Perform multiple random restarts
    max_restarts = 50
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random initialization for step heights
        heights = np.random.uniform(0, 1, n_steps)
        
        # Optimization using differential evolution
        bounds = [(0, 1) for _ in range(n_steps)]
        try:
            result = differential_evolution(
                objective,
                bounds,
                maxiter=100,
                popsize=15,
                seed=seed + restart,
                strategy='best1bin',
                disp=False
            )
            if result.success:
                heights_opt = result.x
                h_opt = build_h(heights_opt)
                c5_new = evaluate_h(h_opt)
                if c5_new < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_new
        except Exception:
            pass
    
    # Final refinement with SLSQP if possible
    if best_h is not None:
        try:
            # Project onto simplex to ensure sum constraint exactly
            def project_onto_simplex(heights, target_sum):
                # Adjust to desired sum while keeping in [0,1]
                heights = np.array(heights)
                original_sum = np.sum(heights)
                diff = target_sum - original_sum
                if abs(diff) < 1e-10:
                    return heights
                    
                # Simple approach: adjust all heights
                heights = heights + diff / len(heights)
                # Clip to [0,1]
                heights = np.clip(heights, 0, 1)
                # Re-scale if necessary (not ideal but simple)
                new_sum = np.sum(heights)
                if abs(new_sum) > 1e-10:
                    heights = heights * target_sum / new_sum
                # Clip again to ensure in [0,1]
                heights = np.clip(heights, 0, 1)
                return heights
            
            # Project to ensure correct sum
            adjusted_heights = project_onto_simplex(
                np.array(best_h[:n_steps*seg_len]).reshape(n_steps, seg_len).mean(axis=1), 
                n_steps/2
            )
            
            h_projected = build_h(adjusted_heights)
            final_c5 = evaluate_h(h_projected)
            
            if final_c5 < best_c5:
                best_h = h_projected
                best_c5 = final_c5
                
        except Exception:
            pass
    
    # Return final solution
    return (best_h, best_c5, n_points)
