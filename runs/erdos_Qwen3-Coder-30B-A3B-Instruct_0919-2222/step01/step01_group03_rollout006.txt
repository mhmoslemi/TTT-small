import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    # Since we're looking for a minimum, we do not need to negate
    # the objective function
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different optimization strategies
    # Strategy 1: Differential evolution
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=15,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Strategy 2: Local optimization with multiple restarts
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Try different initialization strategies 
        if restart < 5:
            # Random initialization that respects the integral constraint
            h_init = np.random.rand(n_points)
            h_init = h_init / (np.sum(h_init) * dx) * 1.0
        else:
            # Initialize with a more structured function
            # Try a piecewise constant function with some variation
            h_init = np.ones(n_points) * 0.5
            # Add some small variations to break symmetry
            variations = np.random.normal(0, 0.05, n_points)
            h_init += variations
            h_init = np.clip(h_init, 0, 1)
            
        try:
            local_result = minimize(
                objective,
                h_init,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Strategy 3: Evolutionary optimization with custom approach
    if time.time() - start_time < budget_s * 0.9:
        # Try a simple evolutionary approach with some heuristic rules
        population_size = 20
        generations = 30
        
        # Initialize population
        population = []
        for i in range(population_size):
            h_rand = np.random.rand(n_points) * 0.5
            h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0
            h_rand = np.clip(h_rand, 0, 1)
            population.append(h_rand)
        
        for gen in range(generations):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Evaluate fitness
            fitness_values = []
            for h in population:
                fit = objective(h)
                fitness_values.append(fit)
            
            # Select best individuals
            indices = np.argsort(fitness_values)[:population_size//2]
            selected = [population[i] for i in indices]
            
            # Create new population (mutation, crossover, selection)
            new_population = selected.copy()
            for _ in range(population_size - len(selected)):
                # Random mutation
                parent = selected[np.random.randint(0, len(selected))]
                child = parent + np.random.normal(0, 0.05, n_points)
                child = np.clip(child, 0, 1)
                # Enforce constraint
                child = child / (np.sum(child) * dx) * 1.0
                child = np.clip(child, 0, 1)
                new_population.append(child)
            
            population = new_population
        
        # Check if any of these improved the best solution
        for h in population:
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
