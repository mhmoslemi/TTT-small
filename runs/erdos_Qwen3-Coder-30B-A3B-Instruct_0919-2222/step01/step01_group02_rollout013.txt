import numpy as np
import time
import random
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(h, d):
    """Shift array h by d positions, padding with zeros"""
    n = len(h)
    if d == 0:
        return h.copy()
    elif d > 0:
        return np.concatenate([np.zeros(d), h[:-d]])
    else:
        return np.concatenate([h[-d:], np.zeros(-d)])

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto box [0, 1]^n and hyperplane sum = target_sum"""
    n = len(v)
    
    # Sort components for binary search
    v_sorted = np.sort(v)
    v_unique = np.unique(v_sorted)
    
    # Binary search for lambda
    low, high = v_sorted[0] - 1, v_sorted[-1] + 1
    
    # We want sum of clip(v - lambda, 0, 1) == target_sum
    def compute_sum(lam):
        clipped = np.clip(v - lam, 0, 1)
        return np.sum(clipped)
    
    # Ensure we can bracket the solution
    low_sum = compute_sum(high)
    high_sum = compute_sum(low)
    
    # Handle edge cases where low or high is already correct
    if np.isclose(compute_sum(low), target_sum):
        return np.clip(v - low, 0, 1)
    if np.isclose(compute_sum(high), target_sum):
        return np.clip(v - high, 0, 1)
    
    # Bisection search
    for _ in range(40):
        mid = (low + high) / 2
        mid_sum = compute_sum(mid)
        if mid_sum > target_sum:
            low = mid
        else:
            high = mid
    
    final_lam = (low + high) / 2
    result = np.clip(v - final_lam, 0, 1)
    
    # Linear correction to exactly match the sum if necessary
    actual_sum = np.sum(result)
    diff = target_sum - actual_sum
    if abs(diff) > 1e-10:
        # Add diff to elements that are not at upper bound
        candidates = np.where(result < 1)[0]
        if len(candidates) > 0:
            to_add = diff / len(candidates)
            for i in candidates:
                result[i] = min(1, result[i] + to_add)
    return result

def build_lp_model(n, binary=False):
    """Build LP or MILP with McCormick constraints."""
    # Variables
    h = cp.Variable(n, name="h")
    t = cp.Variable(name="t")
    
    all_y = {}
    all_constraints = []
    
    # Domain bounds
    if binary:
        h_bounds = [h >= 0, h <= 1]
        h_binary = [cp.Integral(h)]
    else:
        h_bounds = [h >= 0, h <= 1]
    
    # Sum constraint to match integral constraint ∫_0^2 h(x)dx = 1
    # dx = 2 / n, so sum * dx = 1 means sum = n / 2
    sum_con = cp.sum(h) == n / 2
    
    all_constraints.extend(h_bounds)
    all_constraints.append(sum_con)
    
    # Add McCormick constraints for y[i,d] = h[i] * h[i+d]
    for d in range(-n+1, n):
        for i in range(n):
            if 0 <= i + d < n:
                if binary:
                    # In integer case, we can model with equality
                    y = cp.Variable(name=f"y_{i}_{d}")
                    all_y[(i, d)] = y
                    # Add McCormick constraints:
                    all_constraints.extend([
                        y >= 0,
                        y <= h[i],
                        y <= h[i + d],
                        y >= h[i] + h[i + d] - 1
                    ])
                else:
                    # In continuous case, same constraints but let y be continuous
                    y = cp.Variable(name=f"y_{i}_{d}")
                    all_y[(i, d)] = y
                    all_constraints.extend([
                        y >= 0,
                        y <= h[i],
                        y <= h[i + d],
                        y >= h[i] + h[i + d] - 1
                    ])
    
    # For each d, add constraints C(d) <= t
    for d in range(-n+1, n):
        expr = 0
        if d == 0:
            # Only diagonal case
            expr = cp.sum(h) - cp.sum([all_y[(i, 0)] for i in range(n)])
        else:
            # For shift d
            l = []
            for i in range(n):
                if 0 <= i + d < n:
                    l.append(all_y[(i, d)])
            expr = cp.sum(h) - cp.sum(l)
        all_constraints.append(expr * (2.0 / n) <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, all_constraints)
    return prob, h, t, all_y

def solve_with_time_limit(prob, time_limit_s):
    """Solve problem with given time limit."""
    try:
        prob.solve(solver=cp.GLPK, verbose=False, max_time=time_limit_s)
    except:
        prob.solve(solver=cp.CLARABEL, verbose=False, max_time=time_limit_s)
    return prob.status

def subgradient_refine(h_start, end_time, n):
    """Refine h_start using projected subgradient method."""
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h_start)
    stall_counter = 0
    last_improvement = 0
    step_size = 0.5
    
    # Compute initial shift that gives max score
    def compute_max_corr():
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        return corr[k], k
    
    c5, k_star = compute_max_corr()
    iter_count = 0
    
    while time.time() < end_time and iter_count < 1000000:
        iter_count += 1
        # Compute gradient w.r.t. indices of shift k* (open boundary)
        g = np.zeros(n)
        if k_star == 0:
            g = 1 - 2 * h
        else:
            s = shift_array(h, k_star)
            s2 = shift_array(h, -k_star)
            g = 1 - s - s2
            
        # Update
        v = h - step_size * g
            
        # Project
        h_new = project_onto_box_and_sum(v, n / 2)
            
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new)
        
        # Accept if better
        if new_c5 < c5:
            h = h_new
            c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
            last_improvement = iter_count
            stall_counter = 0
        else:
            stall_counter += 1
            if stall_counter > 5000:
                # Re-randomize and restart with noise
                h = best_h.copy()
                noise = np.random.normal(0, 0.05, n)
                h += noise
                h = project_onto_box_and_sum(h, n / 2)
                stall_counter = 0
            
        # Step size decay
        if iter_count % 100 == 0:
            step_size *= 0.999
        if iter_count % 10000 == 0:
            # Every 10k steps, recompute k_star
            c5, k_star = compute_max_corr()

    return best_h, best_c5
