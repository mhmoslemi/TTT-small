```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Initial optimization using differential evolution for global search
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=100,
            popsize=15,
            seed=seed,
            atol=1e-6,
            rtol=1e-6
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass

    # Enhanced local search using L-BFGS-B
    try:
        local_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x: None
        )
        if local_result.success:
            h_opt = np.clip(local_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Try random restarts with multistart optimization
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate different initializations with different strategies
        #策略1: 均匀分布，但置换中间值
        h_rand = np.random.rand(n_points) * 0.5  # Between 0 and 0.5
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        #策略2: 高斯扰动
        h_gauss = best_h + np.random.normal(0, 0.05, n_points)
        h_gauss = np.clip(h_gauss, 0, 1)
        h_gauss = h_gauss / (np.sum(h_gauss) * dx) * 1.0
        
        #策略3: 双峰构造
        h_bimodal = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h_bimodal[i] = 0.5 + 0.2 * np.sin(2 * np.pi * i / mid)  
            else:
                h_bimodal[i] = 0.5 + 0.2 * np.sin(2 * np.pi * (i - mid) / (n_points - mid))
        h_bimodal = h_bimodal / (np.sum(h_bimodal) * dx) * 1.0
        h_bimodal = np.clip(h_bimodal, 0, 1)
        
        # 用3种初始化方法尝试
        for h_init in [h_rand, h_gauss, h_bimodal]:
            try:
                local_result = minimize(
                    objective,
                    h_init,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10},
                    callback=lambda x: None
                )
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except:
                continue
    
    # Fine-tune with more targeted search if time permits
    if time.time() - start_time < budget_s * 0.9:
        try:
            # More aggressive optimization from best solution
            refined_result = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-14},
            )
            if refined_result.success:
                h_opt = np.clip(refined_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
```
