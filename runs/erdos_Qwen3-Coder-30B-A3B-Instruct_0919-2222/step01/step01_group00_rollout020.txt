import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points, seg_len):
    # Build full h array from segment heights
    h = np.repeat(heights, seg_len)
    return h[:n_points]

def evaluate_heights(heights, n_points, seg_len, penalty_factor=1e4):
    # Construct full h array
    h = build_h(heights, n_points, seg_len)
    # Compute C5 with penalty on sum constraint
    c5 = evaluate_h(h)
    # Add penalty for sum constraint
    target_sum = len(heights) / 2.0
    sum_error = np.sum(heights) - target_sum
    penalty = penalty_factor * (sum_error ** 2)
    return c5 + penalty

def project_to_bounds_and_sum(heights, target_sum):
    # Project heights to [0,1] range and adjust to meet sum constraint
    # Simple method: clip, then adjust via scaling
    heights = np.clip(heights, 0, 1)
    current_sum = np.sum(heights)
    if abs(current_sum - target_sum) < 1e-8:
        return heights
    # Linear adjustment to meet target
    adjustment = (target_sum - current_sum) / len(heights)
    heights += adjustment
    # Clip to [0,1]
    heights = np.clip(heights, 0, 1)
    return heights
