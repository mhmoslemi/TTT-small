import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Method 1: Direct optimization with augmented Lagrangian for better constraint handling
    try:
        # Try a more robust constraint-handling method by using trust-constr
        result = minimize(
            objective,
            h0,
            method='trust-constr',
            bounds=bounds,
            constraints=constraints,
            options={'gtol': 1e-9, 'ftol': 1e-9, 'maxiter': 1000}
        )
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Method 2: Particle swarm optimization variation
    # Let's try a different approach with direct search
    def optimize_with_patterns():
        # Let's create a smarter starting pattern
        # Try a pattern that reduces overlap more efficiently
        
        # Use elitist evolutionary approach with fixed structure to start
        h_pattern = np.zeros(n_points)
        
        # Start with a few distinct regions
        # Create a trapezoidal-like shape to reduce overlap
        mid = n_points // 2
        width = n_points // 3
        
        # Fill central region with higher values
        start = max(0, mid - width//2)
        end = min(n_points, mid + width//2)
        h_pattern[start:end] = 1.0
        
        # Normalize to integral constraint
        h_pattern = h_pattern / (np.sum(h_pattern) * dx) * 1.0
        
        # Apply local optimization starting from this pattern
        h_opt = h_pattern.copy()
        try:
            local_result = minimize(
                objective,
                h_opt,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                return h_opt, c5_opt
        except:
            return h_opt, objective(h_opt)
        
        return h_opt, objective(h_opt)
    
    # Run pattern-based optimization
    try:
        pattern_h, pattern_c5 = optimize_with_patterns()
        if pattern_c5 < best_c5:
            best_c5 = pattern_c5
            best_h = pattern_h.copy()
    except:
        pass
    
    # Method 3: Gradient-free evolutionary optimization with restarts
    # Generate multiple random patterns and climb from them
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Try patterns with specific structure
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to integral = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        # Simple optimization from random point
        try:
            result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9}
            )
            if result.success:
                h_opt = np.clip(result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final tuning with smaller step sizes and trust-constr
    try:
        final_result = minimize(
            objective,
            best_h,
            method='trust-constr',
            bounds=bounds,
            constraints=constraints,
            options={'gtol': 1e-12, 'ftol': 1e-12, 'maxiter': 1000}
        )
        if final_result.success:
            h_final = np.clip(final_result.x, 0, 1)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    except:
        pass
    
    return (best_h, best_c5, n_points)
