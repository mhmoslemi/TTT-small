import numpy as np
import time
import random
import cvxpy as cp
from cvxpy import GLPK, CBC

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def shift_array(h, shift):
    """Shift array h by 'shift' positions, zero-padded"""
    n = len(h)
    result = np.zeros(n)
    if shift > 0:
        result[shift:] = h[:-shift]
    elif shift < 0:
        result[:shift] = h[-shift:]
    else:
        result[:] = h[:]
    return result

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto box [0,1] and hyperplane sum = target_sum"""
    lower, upper = 0.0, 1.0
    v = np.clip(v, lower, upper)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    # Bisection on Lagrange multiplier
    def sum_with_lambda(lam):
        return np.sum(np.clip(v - lam, lower, upper))
    low, high = -10.0, 10.0
    for _ in range(40):
        mid = (low + high) / 2
        s = sum_with_lambda(mid)
        if s > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    projected = np.clip(v - lam, lower, upper)
    # Correct any numerical drift
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-10:
        # Distribute the difference among entries not at bounds
        indices = np.where((projected > lower) & (projected < upper))[0]
        if len(indices) > 0:
            delta = diff / len(indices)
            for i in indices:
                projected[i] = np.clip(projected[i] + delta, lower, upper)
    return projected

def build_lp_model(n, binary=False):
    """Build linear or mixed-integer program model"""
    h = cp.Variable(n, integer=binary)
    y = {}
    t = cp.Variable()
    constraints = [
        cp.sum(h) == n / 2,
        h >= 0,
        h <= 1
    ]
    for d in range(-n+1, n):
        if d >= 0:
            indices_i = list(range(n-d))
            indices_j = list(range(d, n))
        else:
            indices_i = list(range(-d, n))
            indices_j = list(range(n+d))
        if indices_i:
            y[d] = cp.Variable(len(indices_i), integer=binary)
            # McCormick constraints
            for i, j in zip(indices_i, indices_j):
                y[d][i-d if d < 0 else i] = cp.Variable(integer=binary) if binary else cp.Variable()
                constraints.extend([
                    y[d][i-d if d < 0 else i] >= h[i] + h[j] - 1,
                    y[d][i-d if d < 0 else i] <= h[i],
                    y[d][i-d if d < 0 else i] <= h[j],
                    y[d][i-d if d < 0 else i] >= 0
                ])
            # Shifting sum constraint
            constraints.append(
                cp.sum(y[d]) >= n / 2 - t
            )
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    return prob, h, y, t

def solve_with_time_limit(prob, solver, time_limit):
    """Solve the problem with a time limit"""
    start = time.time()
    try:
        prob.solve(solver=solver, verbose=False, max_time=time_limit)
        return prob.status
    except Exception as e:
        return 'infeasible'

def subgradient_refine(h_start, end_time, n):
    """Refine solution using projected sub-gradient descent"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    iteration = 0
    stagnation = 0
    while time.time() < end_time:
        iteration += 1
        # Find maximizing shift d*
        corr = np.correlate(h, 1 - h, mode='full')
        d_star = np.argmax(corr) - (len(corr) // 2)
        dx = 2.0 / n
        
        # Construct gradient g
        h_shifted = shift_array(h, d_star)
        g = 1.0 - h_shifted  # For shift d*, g[i] = 1 - h[i+d_star]
        if d_star != 0:
            h_shifted_rev = shift_array(h, -d_star)
            g += 1.0 - h_shifted_rev
        
        # Step size
        alpha = 0.5 / np.sqrt(iteration)  # or use exp decay if desired
        v = h - alpha * g
        
        # Project v
        h = project_onto_box_and_sum(v, n / 2)
        
        # Check improvement
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 5000:
            # Perturb
            h = h + np.random.normal(0, 0.05, n)
            h = project_onto_box_and_sum(h, n / 2)
            stagnation = 0
    return best_h, best_c5
