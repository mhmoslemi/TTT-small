import numpy as np
import time

def proj_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= v_i <= 1, sum(v) = target_sum."""
    # Method: water-filling using binary search for lambda
    v = np.array(v, dtype=np.float64)
    v = np.clip(v, 0, 1)
    total = np.sum(v)
    if abs(total - target_sum) < 1e-12:
        return v
    if total > target_sum:
        # Need to lower values
        def F(lam):
            proj = np.clip(v - lam, 0, 1)
            return np.sum(proj) - target_sum
        low, high = 0, np.max(v)
        for _ in range(100):
            mid = (low + high) / 2
            f_val = F(mid)
            if f_val > 0:
                low = mid
            else:
                high = mid
        lam = (low + high) / 2
        return np.clip(v - lam, 0, 1)
    else:
        # Need to raise values to sum to target sum
        def F(lam):
            # We know this is increasing in lam since we return (sum - target)
            proj = np.clip(v + lam, 0, 1)
            return np.sum(proj) - target_sum
        # Initial lower bound is 0.0
        low, high = 0.0, 1.0
        for _ in range(100):
            mid = (low + high) / 2
            f_val = F(mid)
            if f_val < 0:
                low = mid
            else:
                high = mid
        lam = (low + high) / 2
        return np.clip(v + lam, 0, 1)


def full_corr(h):
    """Compute full convolution correlation between h and (1-h) using FFT."""
    n = len(h)
    H = np.fft.rfft(h, n=2*n-1)
    G = np.fft.rfft(1-h, n=2*n-1)
    corr = np.fft.irfft(H * np.conj(G), n=2*n-1)
    return corr


def evaluate_h(h_values):
    """Evaluate the C5 value using the sliding correlation."""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = full_corr(h)
    max_val = np.max(corr)
    return max_val * dx


def SaMove(h_s, delta_max):
    """Perform one SA move (swap mass between two blocks, then project)."""
    i, j = np.random.choice(len(h_s), size=2, replace=False)
    delta = np.random.uniform(-delta_max, delta_max)
    h_s_new = h_s.copy()
    h_s_new[i] += delta
    h_s_new[j] -= delta
    return proj_capped_simplex(h_s_new, target_sum=len(h_s)/2)
