import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    # Compute C5 using the convolution approach
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Maximum value in the correlation result
    max_val = np.max(conv)
    return max_val * dx

def build_h(heights, n_points):
    # Build h from heights for piecewise constant representation
    n_steps = len(heights)
    seg_len = n_points // n_steps
    if seg_len * n_steps != n_points:
        raise ValueError("n_points must be divisible by n_steps")
    return np.repeat(heights, seg_len)

def evaluate_heights(heights, n_points, target_sum):
    # Evaluate objective function for given heights
    h = build_h(heights, n_points)
    penalty = 1e3 * (np.sum(heights) - target_sum)**2
    return evaluate_h(h) + penalty
