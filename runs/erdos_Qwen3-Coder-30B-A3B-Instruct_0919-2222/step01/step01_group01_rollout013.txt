def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_to_feasible(h, target_sum):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Water-filling to enforce sum = target_sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    if current_sum > target_sum:
        # Need to decrease values
        # Sort descending
        sorted_h = np.sort(h)[::-1]
        cumsum = np.cumsum(sorted_h)
        n = len(sorted_h)
        # Find index where cumulative sum exceeds target_sum
        i = np.argmax(cumsum >= target_sum)
        if i == 0 and cumsum[i] > target_sum:
            # All values must be reduced
            h = h * (target_sum / current_sum)
        else:
            # Find the threshold
            threshold = (cumsum[i] - target_sum) / (n - i)
            threshold = max(0, threshold)
            h = np.clip(h - threshold, 0, 1)
            # Renormalize to exact target_sum
            h = h * (target_sum / np.sum(h))
    else:
        # Need to increase values
        # If not more than one can increase (sum is small)
        # Just fill up remaining space
        delta = target_sum - current_sum
        h = h + delta / len(h)
        h = np.clip(h, 0, 1)
    return h

def smooth_max_gradient(h, beta=10.0):
    """Smoothes the maximum with softmax approximation"""
    # For the purpose of this minimize, we'll consider C5 = max_k ∫ h(x)(1-h(x+k))dx
    # The integral is handled in structured form by np.correlate
    
    # When using smooth max, we approximate max_i f_i(x) by log(sum_i exp(beta * f_i(x))) / beta
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    # Use gradual approximation with β increasing from some small value
    # For smooth gradient, we take derivative numerically
    # But for efficiency, we'll just compute and use gradient direction
    
    # Return gradient approximation relative to smooth max strength beta
    return corr

def compute_smoothed_objective(h, beta=10.0):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    # Return smooth maximum using softmax approximation
    # max(corr) =~ 1/beta * log(sum_i exp(beta * corr[i]))
    return np.log(np.sum(np.exp(beta * corr))) / beta

def projected_gradient_descent(h_start, target_sum, max_iter=1000, step_size=1e-3,
                              beta_start=1.0, beta_growth=1.02, tol=1e-6):
    h = h_start.copy()
    n_points = len(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for t in range(max_iter):
        # Compute gradient
        dx = 2.0 / n_points
        # Approximate gradient through finite differences
        # Let us compute approximate gradient by finite diff of the smooth objective
        h_copy = h.copy()
        eps = 1e-5
        grad = np.zeros_like(h)
        obj_base = compute_smoothed_objective(h, beta=beta_start * (beta_growth ** t))
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            obj_plus = compute_smoothed_objective(h_plus, beta=beta_start * (beta_growth ** t))
            grad[i] = (obj_plus - obj_base) / eps
        
        # Update step
        h = h - step_size * grad
        
        # Project
        h = project_to_feasible(h, target_sum)
        
        # Check early termination based on change in objective or gradient magnitude
        new_c5 = evaluate_erdos_solution(h)
        
        # For numerical stability, we softmax with possibly larger beta
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
    return best_h, best_c5

        sum_h = np.sum(h)

        h = np.clip(h, 0, 1)
    return h
