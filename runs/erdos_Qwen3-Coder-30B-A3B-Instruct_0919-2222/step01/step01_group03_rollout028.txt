import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum over all k values
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Initial construction for h"""
    n = 42
    h = np.zeros(n)
    # Simple construction: constant function that integrates to 1
    h.fill(1.0 / 2.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define the objective function for optimization
    def objective(h):
        # Ensure h is within bounds [0,1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: integral must equal 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP method
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Try different optimization approaches
    
    # 1. Differential evolution with higher precision
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=150,
            popsize=20,
            seed=seed,
            atol=1e-8,
            rtol=1e-8,
            disp=False
        )
        if de_result.success:
            h_opt = np.clip(de_result.x, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # 2. Local optimization with different methods
    methods = ['L-BFGS-B', 'SLSQP', 'TNC', 'Powell']
    for method in methods:
        if time.time() - start_time > budget_s * 0.8:
            break
        try:
            local_result = minimize(
                objective,
                best_h,
                method=method,
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
    # 3. Random restarts with more diverse strategies
    # Random initialization with truncated normal
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
        # Generate random initialization with some structure
        h_rand = np.random.exponential(0.5, n_points)  # Exponential for spikier shape
        h_rand = h_rand / (np.sum(h_rand) * dx) * 1.0  # Scale to sum = 1
        h_rand = np.clip(h_rand, 0, 1)
        
        try:
            local_result = minimize(
                objective,
                h_rand,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 800, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
    # 4. Gradient-free approach with simulated annealing-like behavior
    # Use a different grid-based approach for fine-tuning
    try:
        # Create a mesh of diverse initial configurations  
        # Try structured (piecewise constant) approaches
        h_structured = np.zeros(n_points)
        
        # Try a piecewise approach with more variation
        # Fill first half with higher values and second half with lower
        mid = n_points // 2
        h_structured[:mid] = 0.6
        h_structured[mid:] = 0.4
        h_structured = h_structured / (np.sum(h_structured) * dx) * 1.0
        
        if time.time() - start_time > budget_s * 0.95:
            pass
        else:
            # Local optimization on structured grid
            local_result = minimize(
                objective,
                h_structured,
                method='L-BFGS-B', 
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-11, 'gtol': 1e-11},
                callback=lambda x: None
            )
            if local_result.success:
                h_opt = np.clip(local_result.x, 0, 1)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
    except:
        pass
    
    # Final evaluation if we haven't timed out
    if time.time() - start_time < budget_s * 0.95:
        final_c5 = evaluate_erdos_solution(best_h)
        best_c5 = final_c5
    
    return (best_h, best_c5, n_points)
