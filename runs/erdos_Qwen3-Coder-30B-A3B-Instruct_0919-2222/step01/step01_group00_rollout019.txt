import numpy as np
from scipy.fft import rfft, irfft
import time
from math import exp, log

def proj_capped_simplex(v, target_sum):
    """Project vector `v` onto capped simplex {v : 0 ≤ v_i ≤ 1, sum v_i = target_sum}"""
    v = np.array(v, dtype=float)
    n = len(v)
    # Binary search on lambda
    lmbd_low, lmbd_high = -1.0, 1.0
    max_iter = 100
    for _ in range(max_iter):
        lmbd = (lmbd_low + lmbd_high) / 2.0
        proj = np.clip(v + lmbd, 0, 1)
        s = np.sum(proj)
        if s < target_sum:
            lmbd_high = lmbd
        else:
            lmbd_low = lmbd
    lmbd = (lmbd_low + lmbd_high) / 2.0
    return np.clip(v + lmbd, 0, 1)

def full_corr(h):
    """Compute full correlation h * (1 - h) using FFT."""
    h = np.array(h)
    n = len(h)
    # Total length for full convolution
    size = 2 * n - 1
    H = rfft(h, n=size)
    G = rfft(1 - h, n=size)
    corr = irfft(H * np.conj(G), n=size)
    return corr.real

def evaluate_h(h_values):
    """Standard evaluation (for verification)."""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Convolution with 1-h
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx
