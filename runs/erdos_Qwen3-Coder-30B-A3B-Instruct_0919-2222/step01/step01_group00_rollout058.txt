import numpy as np
from scipy.fft import rfft, irfft
import time
import random

def proj_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: sum = target_sum, 0 <= v_i <= 1."""
    v = np.array(v)
    n = len(v)
    if np.sum(v) == target_sum:
        return v
    # Sort v in descending order
    v_sorted = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(v_sorted)
    # Find the threshold index
    idx = -1
    for i in range(n):
        if v_sorted[i] - (target_sum - cumsum[i]) / (i + 1) >= 0:
            idx = i
    if idx == -1:
        # All elements should be set to zero or target/len (but with 0<=...)
        # Infeasible under these constraints, let's just regularize it
        result = np.clip(v, 0, target_sum/n)
        return result * (target_sum / np.sum(result))
    
    t = (target_sum - cumsum[idx]) / (idx + 1)
    # Compute the projection
    result = np.clip(v - t, 0, 1)
    return result

def full_corr(h):
    """
    Compute full correlation h * (1-h) using FFT.
    Results in same size as convolve(full).
    """
    n = len(h)
    # zero-padded fft
    H = rfft(h, n=2*n-1)
    G = rfft(1-h, n=2*n-1)
    corr = irfft(H * np.conj(G), n=2*n-1)
    return corr.real

def compute_c5(h_vals):
    corr = full_corr(h_vals)
    return np.max(corr) * (2.0 / len(h_vals))

def get_block_averages(h_s, block_len):
    """Expand h_s into full array by repeating each element block_len times."""
    n_full = len(h_s) * block_len
    h_full = np.repeat(h_s, block_len)
    return h_full

    def sa_move(h_s, δ_max):

            if time.time() - start_time > budget_s - 10:

            c5_new = compute_c5(get_block_averages(h_new, block_len))

            j = np.random.choice(np.where(h_bin == 0)[0])
            h_bin[i] = 0
            h_bin[j] = 1
            c5_new = compute_c5(h_bin)
            if c5_new < c5_best:
                c5_best = c5_new
                best_h_bin = h_bin.copy()
            else:
                h_bin[i] = 1
                h_bin[j] = 0
        return (best_h_bin, c5_best)
    
    if time.time() - start_time < budget_s - 10:
        bin_h, bin_c5 = binary_hill_climb()
        if bin_c5 < best_c5:
            best_c5 = bin_c5
            best_h_s = bin_h
    
    # Final expansion and projection
    final_h = get_block_averages(best_h_s, block_len)
    final_h = proj_capped_simplex(final_h, n_points/2)
    final_c5 = compute_c5(final_h)
    
    return (final_h, final_c5, n_points)
