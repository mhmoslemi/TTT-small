import numpy as np
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def shift_array(arr, shift):
    """Shift array by given amount, padding with zeros"""
    n = len(arr)
    if shift >= 0:
        return np.concatenate([arr[shift:], np.zeros(shift)])
    else:
        s = -shift
        return np.concatenate([np.zeros(s), arr[:n-s]])

def project_onto_box_simplex(v, lower=0., upper=1., target_sum=None):
    """Project vector onto box and simplex constraints"""
    if target_sum is None:
        target_sum = len(v) / 2
    
    # Bisection for lambda to find correct projection
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, lower, upper)

    """Optimize h using projected subgradient method"""
    # Initialize h
    if start_h is not None and len(start_h) == n:
        h = np.array(start_h)
    else:
        h = np.random.rand(n)
    
    # Normalize to satisfy sum constraint
    h = project_onto_box_simplex(h * (n/2 / h.sum()))
    
    best_h = h.copy()
    best_C = evaluate_erdos_solution(h)

    last_improve_iter = 0

        iteration += 1
        
        # Compute correlation to find worst shift
        corr = np.correlate(h, 1 - h, mode='full')

        # Compute direction vector
        shift_plus = shift_array(h, d)
        shift_minus = shift_array(h, -d)
        direction = shift_plus + shift_minus - 1.0
        
        # Step size
        α = 0.5 / np.sqrt(iteration + 1)
        
        # Update
        v = h + α * direction
        h = project_onto_box_simplex(v)
        
        # Evaluation
        cur_C = evaluate_erdos_solution(h)
        if cur_C < best_C - 1e-10:
            best_C = cur_C
            best_h = h.copy()
            last_improve_iter = iteration
            stagnation = 0
        else:
            stagnation += 1
            
        # Random restart if stagnation
        if stagnation > 5000:
            h = project_onto_box_simplex(best_h + np.random.normal(scale=0.05, size=n))
            stagnation = 0
            
        # Early termination if target achieved
        if best_C < 0.3807:
            break

    return best_h, best_C
