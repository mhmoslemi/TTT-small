import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog
from cvxpy import Minimize, Problem, ECOS, SCS, GLPK, GLPK_MI

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode (zero-padded)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum correlation value
    return np.max(corr)

def project_sum_constraint(h, target_sum):
    """Project h onto the sum constraint sum(h) = target_sum"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    delta = (target_sum - current_sum) / len(h)
    h_proj = h + delta
    return np.clip(h_proj, 0, 1)

def get_shift_matrix(n, d):
    """Return a sparse shift matrix for shift d"""
    # Create block diagonal structure for shift d
    if d >= 0:
        rows = np.arange(n - d)
        cols = np.arange(d, n)
        data = np.ones(n - d)
    else:
        # Negative shift
        rows = np.arange(-d, n)
        cols = np.arange(n + d)
        data = np.ones(n + d)
    return np.array((data, (rows, cols)), shape=(n, n))

            # But focus on small valid shifts to limit matrix size
            shift_matrix_list = {}
            for d in range(-n+1, n):
                if d == 0:
                    shift_matrix_list[d] = np.eye(n)
                else:
                    shift_matrix_list[d] = get_shift_matrix(n, d)

            # We directly build shifted product terms

                if not np.any(valid):

                row_indices = i_row[valid]

                y_ij = cp.Variable(len(row_indices), boolean=True)
                
                # Approximate bilinear product h[i] * h[i+d]
                # McCormick master constraints:
                for k, (i, j) in enumerate(zip(row_indices, col_indices)):
                    # h_i_org, h_j_shifted
                
                    # y_ij = h_i * h_j
                    # Linear relaxations:
                    # y_ij <= h_i   y_ij <= h_j   y_ij >= h_i + h_j - 1   y_ij >= 0
                    constraints.append(y_ij[k] <= h_vars[i])
                    constraints.append(y_ij[k] <= h_vars[j])
                    constraints.append(y_ij[k] >= h_vars[i] + h_vars[j] - 1)
                    constraints.append(y_ij[k] >= 0)
                
                # Add linear objective term
                sum_term = cp.sum(y_ij)
                constraints.append(t_var >= dx * (n/2 - sum_term))
                
            # Projective constraints
            constraints.append(cp.sum(h_vars) == target_sum)
            
            obj = Minimize(t_var)
            prob = Problem(obj, constraints)
            
            prob.solve(solver=GLPK_MI, time_limit=min(30, time_left()))
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                sol_h = h_vars.value
                if sol_h is not None:
                    # Convert to continuous via rounding
                    h_continuous = np.round(sol_h)
                    h_continuous = project_sum_constraint(h_continuous, target_sum)
                    c5 = evaluate_erdos_solution(h_continuous)
                    if c5 < best_c5:

    if time_left() > 100 and best_h is None:

            # Box constraints
            constraints += [h_vars >= 0, h_vars <= 1]

                    constraints.append(y_ij_new[k] <= h_vars[i])

                    constraints.append(y_ij_new[k] >= h_vars[i] + h_vars[j] - 1)
                    constraints.append(y_ij_new[k] >= 0)
                    # For objective: linearize and sum this to add to t
                sum_term = cp.sum(y_ij_new)
                constraints.append(t_var >= dx * (n/2 - sum_term))
                
            # Objective
            obj = Minimize(t_var)
            prob = Problem(obj, constraints)
            
            prob.solve(solver=GLPK, time_limit=min(50, time_left()))
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                sol_h = h_vars.value
                if sol_h is not None:
                    h_continuous = np.clip(sol_h, 0, 1)
                    h_continuous = project_sum_constraint(h_continuous, target_sum)
                    c5 = evaluate_erdos_solution(h_continuous)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_continuous
        except:
            pass
    
    # Phase 3: SCC refinement loop
    if time_left() > 100 and best_h is not None:
        h_old = best_h.copy()
        h_new = h_old.copy()
        t_old = best_c5
        t_new = t_old
        max_iter = 10000  # Limit iterations
        
        count = 0
        active_timing = time.time()

        while count < max_iter and (time.time() - start_time) < budget_s - 50:

            # Find maximum d such that induced Q(d) dominates
            # Compute correlation for all d and get max one
            reduced_h = np.pad(h_old, (n-1, n-1), mode="constant", constant_values=0)
            corr = np.correlate(reduced_h, 1-reduced_h, mode="full")
            d_offsets = np.arange(-n+1, n)
            max_corr = np.max(corr)
            
            max_idx = np.argmax(corr)
            best_d = d_offsets[max_idx]
            
            # Select (approximate) shift coefficient
            if abs(best_d) >= n - 1:
                continue  # ignore for now (no real diagonal term anyway)
            
            # Build linearized Q(d)
            # Let i go from 0 to n-1
            i_row = np.arange(n)  
            j_col = np.arange(n) + best_d

            col_idx = j_col[valid]

            # Now build constraint

            # new linearized constraint term
            # Rearrangement of product approximation
            # h[i]*h[i+d] ≈ h_old[i]*h_old[i+d] + h_old[i]*(h[i+d]-h_old[i+d]) + h_old[i+d]*(h[i]-h_old[i])

            constraints.append(cp.sum(h_vars) == target_sum)

            status = prob.solve(solver=GLPK, time_limit=1)

                h_new = np.clip(h_new, 0, 1)
                h_new = project_sum_constraint(h_new, target_sum)

                    best_c5 = c5
                    best_h = h_new.copy()

    if time_left() > 20 and best_h is not None:
        h_final = best_h.copy()

            # Compute C5 and accept improvement
