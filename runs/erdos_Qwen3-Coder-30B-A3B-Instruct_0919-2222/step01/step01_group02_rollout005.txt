# Helper function to compute the maximum overlap from a step function
def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

# Helper to project h onto feasible set: sum = n/2, 0 <= h <= 1
def project_h(h, n):
    dx = 2.0 / n
    target_sum = n / 2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    h = h * scale
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points

    # Phase 0: Try binary MILP for small n
    # Heuristic: start with smaller n if needed.
    chosen_n = 70
    if chosen_n <= 50:
        h = np.zeros(chosen_n)
        h[10:30] = 1.0  # Some initial pattern that satisfies sum condition
        h = project_h(h, chosen_n)
    else:
        h = np.ones(chosen_n) * 0.5
    
    # Renormalize just to ensure that sum is exactly n/2
    h = h * (chosen_n / 2) / np.sum(h)
    
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Phase 1 & 2: Try binary MILP or relax fractional solution if it's small
    # We approximate this section with an initialized solution and work from it.
    
    # Phase 3: SCC-style linearizations
    max_iter = 5000
    current_h = h.copy()
    last_c5 = best_c5
    
    t_elapsed = time.time() - start_time
    remain_time = budget_s - t_elapsed

    for i in range(max_iter):
        t_elapsed = time.time() - start_time
        if t_elapsed >= budget_s - 1:
            break
            
        # Compute Q(d) approximations around current h
        # Approximate each h[i]*h[i+d] as linearized around current value
        C5_vals = []
        for d in range(-n_points+1, n_points):
            # compute Q(d) = sum_i h[i]*h[i+d]
            q_val = 0.0
            for i in range(max(0, -d), min(n_points, n_points - d)):
                q_val += current_h[i] * current_h[i+d]
            # from Q = dx*(n/2 - Q(d))
            c5_val = dx * (n_points/2 - q_val)
            C5_vals.append(c5_val)
        
        current_c5 = max(C5_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        if abs(current_c5 - last_c5) < 5e-6:
            break
        last_c5 = current_c5
        
        # Simple gradient descent in the upper bound approximation
        # For simplicity, approximate gradient using finite differencing
        # In practice, one should productionize this by solving dual
        new_h = current_h.copy()
        update_step = 0.01
        direction = np.zeros_like(h)
        
        for d in range(-n_points+1, n_points):
            # compute contribution to gradient
            grad = np.zeros_like(h)
            for i in range(max(0, -d), min(n_points, n_points - d)):
                if i >= 0 and i+d < n_points:
                    grad[i] += 1
                    grad[i+d] += 1
            
            # normalize with small smoothing factor
            if np.sum(grad) > 1e-10:
                grad /= np.sum(grad)
                direction += grad

        # Apply update to make new_h
        new_h = current_h.copy() - update_step * direction
        
        # Project solution into valid set again
        new_h = project_h(new_h, chosen_n)
        current_h = new_h
        
    # Phase 4: Final polishing with hill climbing
    polish_time = time.time() - start_time
    final_remaining_time = budget_s - polish_time - 1

        t_elapsed = time.time() - start_time
        while time.time() - start_time < budget_s - 0.1:
            # stochastic swap
            i, j = np.random.choice(n_points, size=2, replace=False)
            delta = np.random.uniform(-0.02, 0.02)
            if i != j and 0 <= i < n_points and 0 <= j < n_points:
                new_h = current_h.copy()

                    continue
                new_h[i] += delta
                new_h[j] -= delta
                new_h = project_h(new_h, chosen_n)
                new_c5 = evaluate_erdos_solution(new_h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h
                
        current_h = best_h.copy()

    return (best_h.tolist(), best_c5, len(best_h))
