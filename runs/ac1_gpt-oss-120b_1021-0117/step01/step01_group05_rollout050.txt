        else:
            parent = None
            n = 3809
    except Exception:
        parent = None
        n = 3809

    # Initialize best solution
    if parent is not None:

    else:
        best = _initial_binomial(n)

    best_score, best_conv, best_max = _fast_score(best)

    # Ensure best is finite
    if not math.isfinite(best_score):
        # fallback to binomial
        best = _initial_binomial(n)
        best_score, best_conv, best_max = _fast_score(best)

    # Temperature schedule parameters
    # start temperature roughly proportional to current score
    T0 = max(0.1, best_score * 0.05)   # 5% of score
    # function to compute temperature
    def temperature():
        remaining = deadline - time.monotonic()
        total = deadline - start_time
        frac = remaining / total if total > 0 else 0.0
        return max(1e-9, T0 * frac)

    # Main loop
    while time.monotonic() < deadline:
        # copy best for mutation
        cand = best.copy()
        # choose operation
        op = rng.integers(0, 5)
        if op == 0:
            # aggressive mass shift from top product pairs
            # compute current max convolution index
            kmax = int(np.argmax(best_conv))

                prod = cand[i_range] * cand[kmax - i_range]

                max_idx = i_range[prod.argmax()]

import numpy as np
import time
import math

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1, positive decreasing."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, max value) using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())        # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function that reduces the AC1 bound.
    Returns a plain Python list of coefficients.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin before the hard timeout
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Problem size and optional parent (height_sequence_1) handling
    # ------------------------------------------------------------------
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Initialise best solution (parent if present, else binomial seed)
    # ------------------------------------------------------------------
    if parent is not None:
        best = parent.copy()
    else:
        best = _initial_binomial(n)

    best_score, best_conv, _ = _fast_score(best)

    # If the parent turned out invalid, fall back to binomial
    if not math.isfinite(best_score):
        best = _initial_binomial(n)
        best_score, best_conv, _ = _fast_score(best)

    # ------------------------------------------------------------------
    # Temperature schedule (simulated annealing)
    # ------------------------------------------------------------------
    T0 = max(0.1, best_score * 0.05)          # initial temperature
    total_time = deadline - start_time

    def temperature():
        """Linear cooling, never below a tiny epsilon."""
        remaining = deadline - time.monotonic()
        frac = remaining / total_time if total_time > 0 else 0.0
        return max(1e-9, T0 * frac)

    # ------------------------------------------------------------------
    # Main stochastic optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best.copy()
        op = rng.integers(0, 5)

        if op == 0:
            # ----------------------------------------------------------
            # Aggressive mass shift from the most contributing pairs
            # ----------------------------------------------------------
            kmax = int(np.argmax(best_conv))
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            # number of pairs to touch this iteration (1‑3)
            pairs = rng.integers(1, 4)
            for _ in range(pairs):
                i_range = np.arange(lo, hi + 1)
                prod = cand[i_range] * cand[kmax - i_range]
                idx = i_range[prod.argmax()]
                jdx = kmax - idx
                if idx == jdx:
                    # diagonal term – skip (removing it would not change the sum)
                    continue
                alpha = rng.uniform(0.06, 0.25)          # fraction to move
                rem_i = cand[idx] * alpha
                rem_j = cand[jdx] * alpha
                cand[idx] -= rem_i
                cand[jdx] -= rem_j
                # distribute the removed mass to a random low‑value location
                tgt = rng.integers(0, n)
                while tgt == idx or tgt == jdx:
                    tgt = rng.integers(0, n)
                cand[tgt] += rem_i + rem_j

            # ----------------------------------------------------------
            width = rng.integers(1, max(2, n // 40))

            # Smooth a random neighbour pair
            # ----------------------------------------------------------
            i = rng.integers(0, n - 1)

            cand[i] = avg
            cand[i + 1] = avg

        elif op == 3:

            # ----------------------------------------------------------

            # ----------------------------------------------------------
            idx = rng.integers(0, n)

    # ------------------------------------------------------------------
    return [float(v) for v in best]
