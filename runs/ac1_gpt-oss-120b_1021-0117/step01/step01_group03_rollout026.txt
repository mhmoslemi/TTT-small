import time
import numpy as np

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (size = next power of two)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (no clipping, sum>0)."""
    n = a.size
    b = _conv_self_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the target bound."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------------
    # Determine problem size and an optional parent sequence.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    else:
        parent = None
        n = 5582                     # fallback length used in the reference program

    # ----------------------------------------------------------------------
    # Helper to evaluate a candidate and keep the best seen so far.
    # ----------------------------------------------------------------------
    best_arr = None
    best_score = np.inf

    def consider(arr: np.ndarray):
        nonlocal best_arr, best_score
        sc = _objective(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_arr = arr.copy()

    # ----------------------------------------------------------------------
    # Start from the supplied parent (if any) and from a family of power‑law
    # sequences a_i = (i+1)^(-p).  Scan p in a sensible interval.
    # ----------------------------------------------------------------------
    if parent is not None:
        consider(parent)

    idx = np.arange(1, n + 1, dtype=np.float64)   # i+1 values

    # coarse grid
    for p in np.linspace(0.5, 0.7, 21):
        cand = idx ** (-p)
        consider(cand)

    # refine around the best p found so far (store it)
    # we retrieve the p that produced best_arr by recomputing once more
    # (this cheap step avoids keeping an extra variable during the loops)
    if best_arr is not None:
        # find the p that gave the current best (nearest match by objective)
        # start with a coarse guess and then narrow; to avoid extra FFTs we
        # simply keep the best from the previous coarse grid and refine near
        # its middle of the interval.
        # For simplicity we take a centred interval of width 0.05.
        # If the coarse grid already hit the centre (0.6) we still refine.
        coarse_mid = 0.6
        lo = max(0.5, coarse_mid - 0.05)
        hi = min(0.9, coarse_mid + 0.05)
        for p in np.linspace(lo, hi, 61):
            cand = idx ** (-p)
            consider(cand)

    # ----------------------------------------------------------------------
    # Light local search: move a tiny amount of mass from a donor that
    # contributes most to the current maximal convolution entry to a
    # recipient that contributes least.  Accept only improving moves.
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        if best_arr is None:
            break
        # current convolution and its maximal index
        conv = _conv_self_fft(best_arr)
        k_max = int(conv.argmax())

        # contributions to the peak: a[i] * a[k_max-i] where defined
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = best_arr[i[mask]] * best_arr[j[mask]]

        donor = int(i[mask][np.argmax(contrib[mask])])
        # recipient chosen as the index with the smallest contribution
        # (if it coincides with donor we pick a random alternative)
        cand_rec = int(i[mask][np.argmin(contrib[mask])])
        if cand_rec == donor:
            cand_rec = int(rng.integers(0, n))

        donor_mass = best_arr[donor]
        if donor_mass <= 0.0:
            break
        transfer_max = min(donor_mass, 1000.0 - best_arr[cand_rec])
        if transfer_max <= 0.0:
            break
        delta = transfer_max * rng.random() * 0.04  # up to 4 % of transferable mass
        trial = best_arr.copy()
        trial[donor] -= delta
        trial[cand_rec] += delta
        np.clip(trial, 0.0, 1000.0, out=trial)

        sc = _objective(trial)
        if sc < best_score:
            best_score = sc
            best_arr = trial

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats).
    # ----------------------------------------------------------------------
    if best_arr is None:
        # fallback – a uniform tiny sequence (guaranteed valid)
        best_arr = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_arr]
