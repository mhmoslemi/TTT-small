import time
import numpy as np

# ----- helper utilities -------------------------------------------------
def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4**i via recurrence, i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of linear convolution a*a (single FFT)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # keep only the linear part
    conv = conv[: 2 * a.size - 1]
    return float(conv.max())


def _score_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact AC1 bound for an already‑normalized (sum≈1) sequence."""
    s = a.sum()
    if s == 0.0:
        return np.inf
    maxb = _conv_max_fft(a, fft_len)
    return float(2 * n * maxb / (s * s))


# ----- main search -----------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative height list with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- 1. initialise ------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        cur = np.array(height_sequence_1, dtype=np.float64)
        cur = np.clip(cur, 0.0, 1000.0)
    except Exception:
        n = 256
        cur = None

    if cur is None or cur.size != n:
        cur = _generate_binomial_seq(n)

    # normalise to unit L1 (the bound is scale‑invariant)
    total = cur.sum()
    if total <= 0.0:
        cur = np.ones(n, dtype=np.float64) / n
    else:
        cur = cur / total

    # ---- 2. FFT layout ------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ---- 3. evaluate starter ------------------------------------------
    cur_score = _score_fft(cur, n, fft_len)
    best = cur.copy()
    best_score = cur_score

    # target – stop early if we already beat it
    TARGET = 1.50274349

    # ---- 4. main search loop -----------------------------------------
    while time.monotonic() < deadline:
        # compute full convolution (needed for the gradient direction)
        fa = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        k_star = int(conv.argmax())

        # indices that actually contribute to this convolution entry
        i_start = max(0, k_star - (n - 1))
        i_end = min(k_star, n - 1)          # inclusive
        block = cur[i_start : i_end + 1]
        if block.size == 0:
            break

        # source = largest coefficient in the block, target = smallest
        i_max = i_start + int(block.argmax())
        i_min = i_start + int(block.argmin())

        # ---- propose a directed move ---------------------------------
        if i_max != i_min and cur[i_max] > 1e-14:
            # move a moderate fraction from the large entry to the small one
            frac = rng.uniform(0.02, 0.25)   # tuning range
            delta = cur[i_max] * frac
            new = cur.copy()
            new[i_max] -= delta
            new[i_min] += delta
        else:

            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:

            new[i] -= delta

    try:
        _ = evaluate_sequence(final)
        return final
    except Exception:

        return [float(x) for x in cur]
