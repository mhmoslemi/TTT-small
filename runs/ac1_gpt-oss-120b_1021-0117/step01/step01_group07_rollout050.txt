def propose_candidate(...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5

    # initialize best: use height_sequence_1 if exists else use catalan or uniform.
    ...
    best_sum = best.sum()
    best = best / best_sum # sum=1
    best_score = evaluate_sequence(best.tolist())
    best_seq = best.copy()

    # Pre-allocate FFT size
    n = best.size
    fft_size = 1 << ((2*n-1).bit_length())
    
    step_frac = 0.1
    min_step = 1e-6

    while time.monotonic() < deadline:
        # compute convolution via FFT
        fa = np.fft.rfft(best, fft_size)
        conv = np.fft.irfft(fa * fa, fft_size)[:2*n-1]
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # compute gradient
        grad = best.copy()
        start = max(0, max_idx - (n-1))
        end = min(n-1, max_idx)
        if start <= end:
            i = np.arange(start, end+1)
            j = max_idx - i
            # j will automatically be within [0,n-1]
            grad[i] += best[j]

        # Determine step size
        max_grad = grad.max()
        if max_grad <= 0:
            break
        step = step_frac * best.max() / max_grad
        if step <= 0:
            step = min_step

        # candidate
        cand = best - step * grad
        np.clip(cand, 0, None, out=cand)
        cand_sum = cand.sum()
        if cand_sum <= 0:
            # perhaps step too large, reduce
            step_frac *= 0.5
            continue
        cand /= cand_sum

        # evaluate
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            step_frac = min(step_frac * 1.2, 1.0)
            # optional early exit
            if best_score <= 1.50274349:
                break
        else:
            step_frac = max(step_frac * 0.5, min_step)

        # occasional random move
        if rng.random() < 0.05:
            i,j = rng.choice(n, size=2, replace=False)
            if best[i] > 0:
                # transfer random fraction up to, say, up to 30% of best[i]
                transfer = best[i] * rng.random() * 0.3
                cand2 = best.copy()
                cand2[i] -= transfer
                cand2[j] += transfer
                # normalize (should remain sum=1)
                cand2_sum = cand2.sum()
                if cand2_sum > 0:
                    cand2 /= cand2_sum
                    cand2_score = float(evaluate_sequence(cand2.tolist()))
                    if np.isfinite(cand2_score) and cand2_score < best_score:
                        best = cand2
                        best_score = cand2_score
                        if best_score <= 1.50274349:
                            break

    return [float(v) for v in best]

import time
import numpy as np

def _catalan_weights(n: int) -> np.ndarray:
    """First n Catalan‑style weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _powerlaw_weights(n: int, p: float) -> np.ndarray:
    """Weights proportional to (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = 1.0 / (idx ** p)
    return w


def _exp_weights(n: int, a: float) -> np.ndarray:
    """Weights proportional to exp(-a*i)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-a * i)

            no_improve += 1

        # --------------------------------------------------------------
        # d)  Occasional random mass transfer (exploration)
        # --------------------------------------------------------------
        if rng.random() < 0.04:
            i, j = rng.choice(n, size=2, replace=False)
            if best[i] > 0.0:
                eps = best[i] * rng.random() * 0.35
                cand2 = best.copy()
                cand2[i] -= eps
                cand2[j] += eps
                # renormalise (sum stays 1 up to round‑off)
                cand2_sum = cand2.sum()
                if cand2_sum > 0:
                    cand2 /= cand2_sum
                    cand2_score = float(evaluate_sequence(cand2.tolist()))
                    if np.isfinite(cand2_score) and cand2_score < best_score:
                        best = cand2
                        best_score = cand2_score
                        no_improve = 0
                        if best_score <= 1.50274349:
                            break

        # --------------------------------------------------------------
        # e)  Small global jitter – helps escape plateaux
        # --------------------------------------------------------------
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.012, size=n))
            cand3 = best * jitter
            c3_sum = cand3.sum()
            if c3_sum > 0:
                cand3 /= c3_sum
                cand3_score = float(evaluate_sequence(cand3.tolist()))
                if np.isfinite(cand3_score) and cand3_score < best_score:
                    best = cand3
                    best_score = cand3_score
                    no_improve = 0
                    if best_score <= 1.50274349:
                        break

        # --------------------------------------------------------------
        # f)  Reset if stagnation persists
        # --------------------------------------------------------------
        if no_improve >= 300:
            # apply a modest random perturbation to leave the basin
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            best = best * jitter
            best /= best.sum()
            no_improve = 0
            step_frac = max(step_frac * 0.5, min_step_frac)

    # ------------------------------------------------------------------
    # 6.  Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
