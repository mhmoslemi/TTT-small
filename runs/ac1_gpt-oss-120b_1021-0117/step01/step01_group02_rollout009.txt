import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimisation of a non‑negative step function to minimise
    the AC1 upper bound 2·n·max(conv)/sum².
    Uses a greedy mass‑transfer guided by the most influential convolution
    entries and a per‑move line‑search (ternary) for the optimal transferred
    amount.  The routine works with the parent construction ``height_sequence_1``
    when it is present; otherwise it falls back to a binomial‑series start.
    """
    rng = np.random.default_rng(seed)

    # ---- timing ------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ---- initialise --------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:                       # no parent supplied
        n = 5000
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))

    # clip and normalise (sum is irrelevant for the score but convenient)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() == 0.0:
        a[:] = 1.0 / n
    else:
        a = a / a.sum()

    # ---- utilities ---------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        """Direct computation of the evaluator’s score (vector already clipped)."""
        conv = np.convolve(vec, vec)
        return 2.0 * len(vec) * conv.max() / (vec.sum() ** 2)

    def shifted(arr: np.ndarray, shift: int) -> np.ndarray:
        """
        Return a length (2*len(arr)-1) array that equals `arr` shifted by `shift`
        positions (zero‑filled elsewhere).  ``shift`` is non‑negative.
        """
        L = arr.shape[0]
        out = np.zeros(2 * L - 1, dtype=np.float64)
        out[shift: shift + L] = arr
        return out

    # ---- baseline candidates ------------------------------------------------
    best = a.copy()
    best_score = _score(best)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # ---- optimisation loop -------------------------------------------------
    # parameters
    K = 5                      # number of top convolution entries to use
    min_delta = 1e-12

    while time.monotonic() < deadline:
        conv = np.convolve(best, best)
        cur_max = conv.max()

        # ----- active set: K largest convolution entries --------------------
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            part = np.argpartition(-conv, K - 1)
            top_idx = part[:K]

        # ----- impact of each coefficient on the selected entries -----------
        # impact[i] = sum_{t in top_idx} a[t-i]  (zero where out of range)
        impact = np.zeros_like(best)
        for t in top_idx:
            i_start = max(0, t - (n - 1))
            i_end = min(t, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            impact[idx] += best[t - idx]

        src = int(np.argmax(impact))
        dest = int(np.argmin(impact))

        if src == dest:
            # random local perturbation when no clear move exists
            width = int(rng.integers(1, max(2, n // 40)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            cand = best.copy()
            cand[start:stop] *= np.exp(rng.normal(0.0, 0.20))
            cand = np.clip(cand, 0.0, 1000.0)
            cand /= cand.sum()
            cand_score = _score(cand)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
            continue

        # ----- compute linear effect of moving mass from src to dest ----------
        shift_src = shifted(best, src)
        shift_dest = shifted(best, dest)
        diff = shift_dest - shift_src               # change in conv per unit mass

        max_delta = best[src]                       # cannot move more than we have
        if max_delta < min_delta:
            continue

        # ----- ternary search for the optimal delta (convex 1‑D problem) -----
        lo, hi = 0.0, max_delta
        for _ in range(25):                        # ~25 iterations give high precision
            d1 = lo + (hi - lo) * 0.33
            d2 = lo + (hi - lo) * 0.66
            m1 = (conv + d1 * diff).max()
            m2 = (conv + d2 * diff).max()
            if m1 < m2:
                hi = d2
            else:
                lo = d1
        delta_opt = (lo + hi) * 0.5
        if delta_opt < min_delta:
            continue

        # ----- apply the move ------------------------------------------------
        candidate = best.copy()
        candidate[src] -= delta_opt
        candidate[dest] += delta_opt
        candidate = np.clip(candidate, 0.0, 1000.0)

        # renormalise to avoid tiny drift (sum stays 1 analytically)
        s = candidate.sum()
        if s == 0.0:
            continue
        candidate /= s

        cand_score = _score(candidate)

        if cand_score < best_score:
            best = candidate
            best_score = cand_score
        else:
            # occasional random shake to escape plateaux
            if rng.random() < 0.002:
                width = int(rng.integers(1, max(2, n // 45)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = start + width
                shake = best.copy()
                shake[start:stop] *= np.exp(rng.normal(0.0, 0.25))
                shake = np.clip(shake, 0.0, 1000.0)
                shake /= shake.sum()
                shake_score = _score(shake)
                if shake_score < best_score:
                    best = shake
                    best_score = shake_score

    # ---- final sanity check -------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
