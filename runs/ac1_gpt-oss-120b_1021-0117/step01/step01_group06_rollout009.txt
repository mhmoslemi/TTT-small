import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    # find the rho index
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1‑bound evaluation (identical to the judge's function)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ------------------------------------------------------------
# Gradient of the active (max) convolution entry w.r.t. a
# ------------------------------------------------------------
def _grad_of_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Gradient of the currently maximal convolution coefficient."""
    n = a.shape[0]
    k = int(np.argmax(conv))                # index of the active entry
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    grad = np.zeros_like(a)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad, k
