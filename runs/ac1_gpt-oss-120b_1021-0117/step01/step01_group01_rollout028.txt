import time
import numpy as np

def _score_fft(arr: np.ndarray, n: int, fft_len: int) -> float:
    """Fast evaluation of the evaluator's objective using FFT."""
    fa = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s == 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (the classical decreasing sequence)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _mix_seq(rng: np.random.Generator, n: int) -> np.ndarray:
    """Mixture of two power‑law tails: w·i^{-p1} + (1‑w)·i^{-p2}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    p1 = rng.uniform(0.1, 1.0)
    p2 = rng.uniform(0.1, 1.0)
    w = rng.random()
    a = w * i ** (-p1) + (1.0 - w) * i ** (-p2)
    return a


def _piecewise_seq(rng: np.random.Generator, n: int, k: int = 4) -> np.ndarray:
    """Monotone decreasing piecewise‑linear profile with k segments."""
    # random break points in (0,1)
    pts = np.sort(rng.random(k - 1))
    pts = np.concatenate(([0.0], pts, [1.0]))
    # random heights, then enforce decreasing order
    h = rng.random(k)
    h = np.sort(h)[::-1]
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.interp(x, pts, h)
    return a

    except Exception:                     # fallback when no parent is present
        n = 256

    # FFT layout (fixed for the whole run)
