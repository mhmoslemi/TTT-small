def fast_score(arr, L):
    # arr: np.ndarray of floats
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*len(arr)-1]
    maxb = conv.max()
    s = arr.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * len(arr) * maxb / (s * s)

import numpy as np
import time, math

def _gen_binomial(n: int) -> np.ndarray:
    """Classic binomial coefficients a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimizes
    the AC1 upper bound.  The routine combines a small family of parametric
    shapes with a fast stochastic hill‑climbing that focuses on the indices
    contributing most to the current maximal convolution coefficient.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # ------------------------------------------------------------------ #
    # problem size ------------------------------------------------------ #
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256
        parent = None

    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 2).bit_length())

    # ------------------------------------------------------------------ #
    # fast AC1 score (FFT based, identical to evaluator up to rounding) #
    def _fast_score(arr: np.ndarray) -> float:
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        maxb = float(conv.max())
        s = float(arr.sum())
        if s < 1e-12:
            return float('inf')
        return 2.0 * n * maxb / (s * s)

    # ------------------------------------------------------------------ #
    # generate a handful of parametric starting shapes ------------------ #
    cand_list = []

    # 1) parent if available
    if parent is not None:
        a = parent.copy()
        np.clip(a, 0.0, 1000.0, out=a)

        if a_sum < 1e-12:
            a[0] = 1.0
            a_sum = a.sum()
        a /= a_sum
        cand_list.append(a)

    # 2) classic binomial
    b = _gen_binomial(n)

    b /= b.sum()

        if beta.sum() == 0:

        beta /= beta.sum()

    cur_score = best_score

    # ------------------------------------------------------------------ #

    temperature = 1.0

    p_block    = 0.3   # scale a random contiguous block
