import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by `evaluate_sequence`.  The search explores
    simple polynomially decreasing shapes a_i ∝ (n‑i)^p for a few integer
    exponents p and keeps the best one found before the time budget expires.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)  # keep a safety margin

    # Determine the problem length n.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        # Very unlikely, but fall back to a modest size if the parent is missing.
        n = 256

    # Pre‑allocate the descending integer vector (n, n‑1, …, 1) as float64.
    base = np.arange(n, 0, -1, dtype=np.float64)

    best_seq = None
    best_score = float("inf")

    # Explore a handful of polynomial exponents.
    # p = 0 gives the uniform vector, p = 1 is linear, higher p produce steeper
    # decay. Empirically the optimum lies around p = 5 for the present n.
    for p in range(0, 9):  # 0 … 8
        # Compute a_i ∝ (n‑i)^p, normalise to sum = 1.
        arr = base ** p
        total = arr.sum()
        if total == 0.0:        # should never happen, but be defensive
            continue
        arr = arr / total

        # Evaluate the candidate via the authoritative scorer.
        try:
            score = evaluate_sequence(arr.tolist())
        except Exception:
            score = float("inf")

        # Keep the best one seen so far.
        if score < best_score:
            best_score = score
            best_seq = arr.copy()

        # Time check – stop early if we are out of budget.
        if time.monotonic() >= deadline:
            break

    # If for any reason no candidate succeeded, fall back to a uniform vector.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Final safety: ensure the sequence passes the evaluator (it will be clipped
    # internally, but we keep it inside the allowed range).
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        # As a last resort, return a simple uniform distribution.
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Convert to plain Python floats as required by the harness.
    return [float(x) for x in best_seq]
