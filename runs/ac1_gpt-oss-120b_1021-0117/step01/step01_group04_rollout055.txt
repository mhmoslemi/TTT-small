import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a[i] = C(2i,i) / 4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_and_conv(arr: np.ndarray, n: int, L: int):
    """
    Returns (score, max_conv_value, index_of_max) using FFT.
    Score = 2*n*max_conv/(sum(arr)**2)
    """
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[: 2 * n - 1]
    idx = int(conv.argmax())
    mx = float(conv[idx])
    s = float(arr.sum())
    if s < 1e-12:
        score = float('inf')
    else:
        score = 2.0 * n * mx / (s * s)
    return score, mx, idx


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----- problem size ----------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.size)
    except Exception:
        n = 256               # very unlikely fallback
        parent = None

    L = _next_pow2(2 * n - 1)

    # ----- initial candidate ------------------------------------------------
    base = _gen_binomial(n)
    best = base.copy()
    best_score, _, _ = _score_and_conv(best, n, L)

    if parent is not None:
        parent_score, _, _ = _score_and_conv(parent, n, L)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    cur = best.copy()
    cur_score, cur_max, cur_kmax = _score_and_conv(cur, n, L)

    # ----- annealing parameters --------------------------------------------
    temperature = 1.0
    decay = 0.9999
    min_temp = 0.01
    target = 1.50274349

    # ----- main search loop -------------------------------------------------
    while time.monotonic() < deadline:
        r = rng.random()

        # 1) targeted mass move (focus on the lag with maximal convolution)
        if r < 0.40:
            # indices that contribute to the maximal lag
            lo = max(0, cur_kmax - (n - 1))
            hi = min(n - 1, cur_kmax)
            i = int(rng.integers(lo, hi + 1))
            j = cur_kmax - i
            src = i if cur[i] >= cur[j] else j
            if cur[src] <= 0.0:
                continue
            # move a fraction of the larger entry to a random destination
            frac = rng.random() * 0.45
            delta = cur[src] * frac
            dest = int(rng.integers(0, n))
            while dest == src:
                dest = int(rng.integers(0, n))
            cand = cur.copy()
            cand[src] -= delta
            cand[dest] += delta

        # 2) block scaling (moderate adjustments)
        elif r < 0.78:
