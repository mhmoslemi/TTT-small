import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """FFT‑based full self‑convolution (real input)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective used by the evaluator (scale‑invariant)."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def _power_series(n: int, p: float, beta: float) -> np.ndarray:
    """a_i = 1/(i+beta+1)^p  (non‑negative, decreasing)."""
    idx = np.arange(n, dtype=np.float64)
    return 1.0 / np.power(idx + beta + 1.0, p)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step‑function height list with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    total_budget = max(0.0, float(budget_s) - 5.0)   # safety margin
    deadline = time.monotonic() + total_budget

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    else:
        n = 5582
        parent = None

    # baseline candidates
    sqrt_seq = _sqrt_series(n)
    best = sqrt_seq.copy()
    best_score = _score(best)
    if parent is not None:
        parent_score = _score(parent)
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # -------------------- quick parametric search --------------------
    # power‑law family a_i = (i+beta+1)^(-p)
    param_end = deadline - max(30.0, total_budget * 0.2)   # reserve time for final hill‑climb
    p_grid = np.linspace(0.3, 2.0, 181)   # step ≈0.009
    beta_grid = np.linspace(0.0, 20.0, 21)  # step 1
    for p in p_grid:
        for beta in beta_grid:
            cand = _power_series(n, p, beta)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
            if time.monotonic() > param_end:
                break
        if time.monotonic() > param_end:
            break

    # pre‑compute convolution of the current best for targeted moves
    best_conv = _conv_self(best)

    # -------------------- hill‑climbing with targeted mass transfer --------------------
    early_len = n // 10                # region where values are typically large
    late_len  = n // 10                # region where values are typically small

    while time.monotonic() < deadline:
        cand = best.copy()
        op = rng.random()

        if op < 0.60:
            # ---------- targeted transfer ----------
            # use the peak location of the current best convolution
            kmax = int(np.argmax(best_conv))
            i0 = max(0, kmax - (n - 1))
            i1 = min(n - 1, kmax)
            i_range = np.arange(i0, i1 + 1, dtype=np.int64)
            j_range = kmax - i_range
            # product of paired entries that contribute to the peak
            prod = best[i_range] * best[j_range]
            if prod.size == 0:
                continue
            idx = int(np.argmax(prod))
            i_max = i_range[idx]
            j_max = j_range[idx]
            src = i_max if best[i_max] >= best[j_max] else j_max

            # choose a destination far from src (prefer the opposite tail)
            if src < n // 2:
                dst = rng.integers(n - late_len, n)   # far right
            else:
                dst = rng.integers(0, early_len)      # far left
            if dst == src:
                continue
            max_transfer = min(cand[src], 1000.0 - cand[dst])
            if max_transfer <= 0.0:
                continue
            delta = max_transfer * rng.random() * 0.07   # up to 7 % transfer
            cand[src] -= delta
            cand[dst] += delta

        elif op < 0.85:
            # ---------- block scaling ----------
            width = int(rng.integers(max(2, n // 300), max(4, n // 30)))
            start = rng.integers(0, n - width + 1)
            factor = float(np.exp(rng.normal(0.0, 0.13)))   # log‑normal perturbation
            cand[start:start + width] *= factor

        else:
            # ---------- small additive noise ----------
            noise = rng.normal(0.0, 0.015, size=n)
            cand += noise

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            best_conv = _conv_self(best)

    # return as a plain list of python floats
    return [float(v) for v in best]
