def ac1_score(arr, L, n):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = conv.max()
    s = arr.sum()
    if s < 1e-12: return float('inf')
    return 2.0 * n * max_b / (s * s)

def lp_optimize(a, M, selected_k):

import numpy as np
import time, math
import cvxpy as cp

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound using FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _lp_optimize(a: np.ndarray, M: float, ks: np.ndarray) -> np.ndarray | None:
    """
    Solve max sum(b)  s.t. b ≥ 0,  conv(a,b)[k] ≤ M for k in ks.
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)
    constraints = []
    for k in ks:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        coeff = a[idx]                     # constant coefficients
        b_idx = k - idx                    # matching indices of b
        constraints.append(cp.sum(cp.multiply(coeff, b[b_idx])) <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=2000)
        except Exception:
            return None
    if b.value is None:
        return None
    return np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
