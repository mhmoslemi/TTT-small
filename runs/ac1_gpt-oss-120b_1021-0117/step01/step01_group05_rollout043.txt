    # ------------------------------------------------------------------
    temp = max(0.1, best_score * 0.2)   # initial temperature
    it = 0
    while time.monotonic() < deadline:
        cand = cur.copy()
        r = rng.random()
        if r < 0.45:
            cand = _move_mass_maxpair(cand, cur_conv, rng)
        elif r < 0.65:
            cand = _block_scale(cand, rng)
        elif r < 0.80:
            cand = _single_scale(cand, rng)
        elif r < 0.90:
            cand = _smooth_pair(cand, rng)
        else:
            cand = _redistribute(cand, rng)

        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score, cand_conv, _ = _fast_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            cur_conv = cand_conv
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # temperature schedule
        temp *= 0.9995
        it += 1

        # occasional re‑seed from the best solution to avoid stagnation
        if it % 2000 == 0 and rng.random() < 0.2:
            cur = best.copy()
            cur_score, cur_conv, _ = _fast_score(cur)

    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
