import time
import numpy as np

# ------------------- helper functions ------------------- #

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic Sidon‑like decreasing sequence a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _geometric_sequence(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i = r**i (r in (0,1))."""
    return np.power(r, np.arange(n, dtype=np.float64))


def _power_sequence(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = 1/(i+1)**p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return 1.0 / np.power(idx, p)


def _alternating_order(arr: np.ndarray) -> np.ndarray:
    """Place values alternating from the left and right ends (spreads large terms)."""
    n = arr.size
    new = np.empty_like(arr)
    left, right = 0, n - 1
    pos = 0
    take_left = True
    while left <= right:
        if take_left:
            new[left] = arr[pos]
            left += 1
        else:
            new[right] = arr[pos]
            right -= 1
        pos += 1
        take_left = not take_left
    return new


def _normalize(arr: np.ndarray) -> np.ndarray:
    """Scale to have unit sum, keep non‑negative."""
    s = arr.sum()
    if s == 0:
        return arr
    return arr / s


def _score(arr: np.ndarray) -> float:
    """Exact score matching evaluate_sequence (clipping, sum‑normalisation)."""
    a = np.clip(arr, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    maxb = conv.max()
    n = a.size
    return 2.0 * n * maxb / (total * total)
