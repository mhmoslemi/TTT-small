import time, numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # safety margin
    
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size == 0:
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    
    n = parent.size
    
    # Normalize to sum 1 for simplicity
    parent_norm = parent / max(parent.sum(), 1e-12)
    
    # start with parent_norm
    best = parent_norm.copy()
    # also try uniform
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    best = best if evaluate_sequence(best.tolist()) <= evaluate_sequence(uniform.tolist()) else uniform.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Also try Catalan weights for extra baseline
    def catalan_weights(n):
        w = np.empty(n, dtype=np.float64)
        w[0] = 1.0
        for i in range(1, n):
            w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
        return w
    cat = catalan_weights(n)
    cat_norm = cat / cat.sum()
    cat_score = float(evaluate_sequence(cat_norm.tolist()))
    if cat_score < best_score:
        best = cat_norm.copy()
        best_score = cat_score
    
    # Precompute index array for convenience
    indices = np.arange(n)
    
    # helper compute score using FFT
    def score_of(a):
        # ensure nonnegative
        a = np.maximum(a, 0.0)
        n = a.shape[0]
        # length for FFT
        size = 1
        while size < 2*n - 1:
            size <<= 1
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
        maxb = conv.max()
        sum_a = a.sum()
        if sum_a < 0.01:
            return np.inf
        return 2.0 * n * maxb / (sum_a*sum_a)
    
    # main loop
    # adjustable parameters
    move_prob = [0.4, 0.3, 0.2, 0.1] # param_sampling, random_transfer, gradient_move, jitter
    # define each move function
    # 1: parameter sampling (beta shape)
    def param_beta():
        # sample alphas in [0,5]
        alpha = rng.uniform(0.0, 5.0)
        beta = rng.uniform(0.0, 5.0)
        x = indices / (n-1) if n>1 else indices
        # to avoid 0**0, set x[0]=0, but (0**alpha) is 0 for alpha>0, fine.
        a = np.power(x, alpha) * np.power(1 - x, beta)
        # add small epsilon to avoid all zeros
        a += 1e-12
        a /= a.sum()
        return a
    
    # 2: random transfer
    def random_transfer(current):
        i, j = rng.choice(n, size=2, replace=False)
        if current[i] == 0.0:
            return current
        eps = rng.random() * current[i] * 0.5  # up to 50% of mass at i
        new = current.copy()
        new[i] -= eps
        new[j] += eps
        # renorm just to be safe (should sum to 1)
        new_sum = new.sum()
        if new_sum > 0.0:
            new /= new_sum
        return new
    
    # 3: gradient move based on max lag
    def gradient_move(current):
        n_ = current.shape[0]
        # compute conv via FFT
        size = 1
        while size < 2*n_ - 1:
            size <<= 1
        fa = np.fft.rfft(current, n=size)
        conv = np.fft.irfft(fa*fa, n=size)[:2*n_-1]
        kmax = int(conv.argmax())
        # contributions from pairs (i, j) s.t. i+j = kmax
        # for each i, j = kmax - i
        partner = kmax - indices
        mask = (partner >= 0) & (partner < n_)
        # compute contribution values
        contrib = np.zeros(n_, dtype=np.float64)
        contrib[mask] = current[mask] * current[partner[mask]]
        total = contrib.sum()
        if total == 0.0:
            return current
        # choose i weighted by contribution
        probs = contrib / total
        i = rng.choice(n_, p=probs)
        # choose target j random (could be same)
        j = rng.integers(n_)
        if i == j:
            return current
        # epsilon proportion of current[i]
        eps = rng.random() * current[i] * 0.3
        new = current.copy()
        new[i] -= eps
        new[j] += eps
        new_sum = new.sum()
        if new_sum > 0.0:
            new /= new_sum
        return new
    
    # 4: jitter (multiplicative noise)
    def jitter_move(current):
        # multiplicative log-normal noise
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        new = current * noise
        new_sum = new.sum()
        if new_sum > 0.0:
            new /= new_sum
        return new
    
    move_funcs = [param_beta, random_transfer, gradient_move, jitter_move]
    
    # initial best known score
    # loop with early exit if target reached
    target = 1.50274349
    
    # run loop
    while time.monotonic() < deadline:
        # select move type based on probabilities
        r = rng.random()
        cum = 0.0
        for prob, func in zip(move_prob, move_funcs):
            cum += prob
            if r < cum:
                # for param_beta we need generate fresh, others need current best as input
                if func is param_beta:
                    cand = func()
                else:
                    cand = func(best)
                break
        else:
            cand = best.copy()
        # evaluate score
        s = score_of(cand)
        if np.isfinite(s) and s < best_score:
            best = cand
            best_score = s
            # adjust probabilities to focus more on good moves?
            # maybe increase prob for the successful move type
            # but keep simple
            if best_score <= target:
                # early break
                break
        # Occasionally, every few iterations, try smoothing of best by averaging with param sample
        if rng.random() < 0.02:
            # blend with a new param sample
            sample = param_beta()
            blended = 0.5*best + 0.5*sample
            blended /= blended.sum()
            s2 = score_of(blended)
            if s2 < best_score:
                best = blended
                best_score = s2
    # final fallback
    # If best_score still infinite (shouldn't), fallback to uniform
    if not np.isfinite(best_score):
        best = uniform.copy()
    return [float(v) for v in best]

# ---------- helper functions (top‑level) ----------
def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def _fft_conv_max(a: np.ndarray, size: int) -> float:
    """
    Return max value of the full linear convolution a*a using an FFT of length `size`.
    """
    # real‑to‑real FFT is sufficient – a is real, non‑negative
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _score_fft(a: np.ndarray, size: int) -> float:
    """
    Fast proxy for evaluate_sequence – returns the same quantity
    (2*n*max_conv/(sum_a**2)) using FFT instead of np.convolve.
    """
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    n = a.shape[0]
    maxb = _fft_conv_max(a, size)
    return 2.0 * n * maxb / (sum_a * sum_a)


def _param_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑style shape a_i ∝ x^α (1−x)^β with random α,β∈[0,5]."""
    α = rng.uniform(0.0, 5.0)

    a = np.power(x, α) * np.power(1.0 - x, β)
    a += 1e-12          # avoid strict zeros
    a /= a.sum()
    return a

        return cur
    eps = rng.random() * cur[i] * 0.5          # ≤50 % of a[i]

    out[i] -= eps
    out[j] += eps
    out_sum = out.sum()
    if out_sum > 0.0:

    return out


def _gradient_move(cur: np.ndarray, size: int, rng: np.random.Generator) -> np.ndarray:
    """

    a_i·a_{k−i}, and shift a little mass away from the most‑responsible index.

    if tot == 0.0:

    out[i] -= eps
    out[j] += eps
    s = out.sum()
    if s > 0.0:

    return out

    s = out.sum()
