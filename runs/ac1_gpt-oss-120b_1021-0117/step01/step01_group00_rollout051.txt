def compute_score(seq):
    # seq is np.ndarray of float64 length n
    seq = np.clip(seq, 0.0, 1000.0)   # ensure.
    sum_val = np.sum(seq)
    if sum_val < 0.01:
        return float('inf')
    # compute convolution: full length 2n-1
    # Use FFT with length m = 2**int(np.ceil(np.log2(2*n-1))) for speed.
    m = 1 << (2*n-1).bit_length()
    # Actually for size m >= 2n-1
    # Use np.fft.rfft for real to smaller memory.
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]  # because convolution as auto-correlation same orientation.
    max_b = conv.max()
    return 2 * len(seq) * max_b / (sum_val**2)
```

import numpy as np, time, math

def _sidon_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
    return a

def ac1_score(seq):
    # compute max conv using FFT
    n = seq.shape[0]
    m = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np
import time

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic Sidon‐like coefficients a_i = C(2i,i)/4**i (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _ac1_score(seq: np.ndarray, fft_len: int) -> float:
    """Fast evaluation of the AC1 bound via FFT (same objective as evaluate_sequence)."""
    n = seq.shape[0]
    # FFT of the sequence (real → rfft)
    f = np.fft.rfft(seq, n=fft_len)
    # convolution = irfft(product)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Hill‑climbing search that tries to lower the AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # obtain problem size (use the provided parent if it exists)
    # ------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        n = parent.shape[0]
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # ------------------------------------------------------------------
    # initial construction – Sidon‑like multiset of values
    # ------------------------------------------------------------------
    base_vals = _sidon_like_sequence(n)
    base_vals = np.clip(base_vals, 0.0, 1000.0)

    # initial candidate: the ordered Sidon sequence
    cur_seq = base_vals.copy()
    # (a random permutation of the same multiset can be tried later)
    fft_len = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT
    cur_score = _ac1_score(cur_seq, fft_len)

    # keep the best sequence found so far
    best_seq = cur_seq.copy()
    best_score = cur_score

    # parent might already be better – try it once
    parent_score = _ac1_score(parent, fft_len)
    if parent_score < best_score:
        best_seq = parent.copy()
        best_score = parent_score
        cur_seq = parent.copy()
        cur_score = parent_score

    # ------------------------------------------------------------------
    # main iterative improvement loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # full convolution (FFT)
        fcoeff = np.fft.rfft(cur_seq, n=fft_len)
        conv_full = np.fft.irfft(fcoeff * fcoeff, n=fft_len)[: 2 * n - 1]
        max_b = conv_full.max()
        kmax = int(conv_full.argmax())

        # contributions of each index to the maximal shift
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(cur_seq)
        contrib[mask] = cur_seq[mask] * cur_seq[partner[mask]]

        # indices with largest / smallest contribution
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))

        # ------------------------------------------------------------------
        # propose a mass‑transfer move (remove a little from i_max, add to i_min)
        # ------------------------------------------------------------------
        if i_max != i_min:
            eps = min(0.05 * cur_seq[i_max], 0.05 * (1000.0 - cur_seq[i_min]))
            if eps > 0:
                cand = cur_seq.copy()
                cand[i_max] -= eps
                cand[i_min] += eps
                cand_score = _ac1_score(cand, fft_len)

                if cand_score < best_score:
                    best_seq = cand.copy()
                    best_score = cand_score
                if cand_score < cur_score:
                    cur_seq = cand
                    cur_score = cand_score
                    continue   # move successful – start next iteration

        # ------------------------------------------------------------------
        # if the targeted move does not help, try a random swap of two values
        # ------------------------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i != j:
            cand = cur_seq.copy()
            cand[i], cand[j] = cand[j], cand[i]
            cand_score = _ac1_score(cand, fft_len)
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
            if cand_score < cur_score:
                cur_seq = cand
                cur_score = cand_score
                continue

        # occasional small random perturbation (simulated annealing style)
        # ------------------------------------------------------------------
        if rng.random() < 0.002:
            noise = rng.normal(scale=0.02, size=n)
            cand = np.clip(cur_seq + noise, 0.0, 1000.0)
            cand_score = _ac1_score(cand, fft_len)
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
            if cand_score < cur_score:
                cur_seq = cand
                cur_score = cand_score

            cur_seq = base_vals[perm]
            cur_score = _ac1_score(cur_seq, fft_len)
            if cur_score < best_score:
                best_seq = cur_seq.copy()
                best_score = cur_score

    # ------------------------------------------------------------------

    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
