import time
import numpy as np

def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of a where a*a = 1 + … + x^{2n-2} (truncated)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _conv_full(vec: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution vec*vec using FFT, truncated to length 2*len-1."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * vec.size - 1]


def _gradient_at(vec: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max‑convolution w.r.t. vec at shift k."""
    n = vec.size
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(vec)
    grad[mask] = 2.0 * vec[j[mask]]
    return grad


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise a probability vector (sum == 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = _sqrt_series(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= s                     # now a sums to 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # initial evaluation
    # ------------------------------------------------------------------
    conv = _conv_full(a, L)
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    cur_score = 2.0 * n * max_val
    best_score = cur_score
    best_vec = a.copy()

    # ------------------------------------------------------------------
    # optimisation loop – sub‑gradient descent + occasional random move
    # ------------------------------------------------------------------
    step0 = 0.12                     # initial step magnitude
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # gradient of the active convolution entry
        grad = _gradient_at(a, max_idx)

        # diminishing step size
        step = step0 / np.sqrt(it)

        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        tot = cand.sum()
        if tot == 0.0:
            cand = np.full_like(a, 1.0 / n)
            tot = 1.0
        cand /= tot                     # back to simplex

        # evaluate candidate
        conv_c = _conv_full(cand, L)
        max_idx_c = int(conv_c.argmax())
        max_val_c = float(conv_c[max_idx_c])
        score_c = 2.0 * n * max_val_c

        # accept if improvement
        if score_c < cur_score:
            a = cand
            conv = conv_c
            max_idx = max_idx_c
