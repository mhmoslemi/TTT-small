import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: fast linear convolution via real FFT (O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective used by the evaluator (same formula, fast implementation)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Classic sqrt‑series (Catalan‑type) – used when no parent is supplied.
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a
