def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1 via recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 upper bound (uses FFT for speed)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound
    is reduced below the target 1.50274349.  The algorithm starts from the
    classic binomial construction (or a provided parent) and performs a
    fast simulated‑annealing walk that preferentially moves mass from
    indices contributing most to the current maximal convolution towards
    indices that do not influence that maximum.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------
    # Length of the problem (use parent if supplied)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        # fallback size (should never be used in the official test)
        n = 256
        parent = None

    L = _next_pow2(2 * n - 1)          # FFT length

    # ------------------------------------------------------------
    # Initial candidate
    base = _gen_binomial(n)
    if parent is not None:
        parent_score = _ac1_score(parent, n, L)
        base_score = _ac1_score(base, n, L)
        if parent_score < base_score:
            cur = parent.astype(np.float64).copy()
            cur_score = parent_score
        else:
            cur = base.copy()
            cur_score = base_score
    else:
        cur = base.copy()
        cur_score = _ac1_score(cur, n, L)

    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------
    # Simulated‑annealing parameters
    temperature = 1.0
    decay = 0.9995
    target_bound = 1.50274349

    # ------------------------------------------------------------
    # Main optimisation loop
    while time.monotonic() < deadline:
        # current convolution and its maximum index
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # weight vector w_i = a_{max_idx-i} when the index is valid,
        # otherwise 0.  This is the sub‑gradient of the max‑convolution.
        w = np.zeros_like(cur)
        low = max(0, max_idx - (n - 1))
        high = min(max_idx, n - 1)
        if low <= high:
            i = np.arange(low, high + 1, dtype=int)
            w[i] = cur[max_idx - i]

        # choose a source index among those that contribute most
        contrib = cur * w
        src = int(np.argmax(contrib))
        if contrib[src] == 0.0:
            # fallback: random move if gradient information is unavailable
            src = int(rng.integers(0, n))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue
            frac = rng.random() * 0.3
            delta = cur[src] * frac
        else:
            # target indices where w == 0 (no influence on current max)
            zero_mask = (w == 0)
            zero_idx = np.nonzero(zero_mask)[0]
            if zero_idx.size == 0:
                # extremely unlikely, fall back to random target
                dst = int(rng.integers(0, n))
            else:
                dst = int(rng.choice(zero_idx))

            # transfer a modest fraction of mass
            frac = rng.random() * 0.3
            delta = cur[src] * frac
            if delta <= 0.0:
                continue

        # build candidate by moving delta from src to dst
        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _ac1_score(cand, n, L)

        # keep the best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target_bound:
                break

        # simulated‑annealing acceptance for the current walk
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # occasional restart from the best solution to avoid drift
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ------------------------------------------------------------
    # Return best sequence as plain Python floats
    return [float(x) for x in best]
