import time
import numpy as np

def _ac1_bound(seq: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound using FFT (exact for non‑negative real sequences)."""
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return np.inf
    fft_seq = np.fft.rfft(seq, L)
    conv = np.fft.irfft(fft_seq * fft_seq, L)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (sum_a * sum_a)


def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    return idx ** (-p)


def _exponential_sequence(n: int, c: float) -> np.ndarray:
    """a_i = exp(-c*i/n)."""
    r = np.exp(-c / n)
    return r ** np.arange(n, dtype=np.float64)


def _mixed_sequence(n: int, p: float, c: float) -> np.ndarray:
    """a_i = (i+1)^{-p} * exp(-c*i/n)."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    r = np.exp(-c / n)
    return (idx ** (-p)) * (r ** np.arange(n, dtype=np.float64))

    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score):
        # fallback to the very first valid parent if something went wrong
        final_seq = np.full_like(best_seq, 1.0)
    else:
        final_seq = best_seq

    return [float(v) for v in final_seq]
