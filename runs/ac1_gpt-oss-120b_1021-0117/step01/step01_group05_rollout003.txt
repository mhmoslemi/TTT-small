```
def conv_and_score(a):
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[:2*n-1]
    M = conv.max()
    S = a.sum()
    score = 2.0 * n * M / (S*S) if S>0 else float('inf')
    return score, conv, M
```

```
                    best_score = cand_score
                    best_conv = cand_conv

        # else (if no improvement) or src==tgt, do random block scaling
        # Random block scaling operation similar to previous op 0/3
        op_choice = rng.integers(0, 3)
        candidate = best.copy()
        if op_choice == 0:

            i = rng.integers(0, n-1)

            candidate[i] = avg

        else:

            factor = np.exp(rng.normal(0.0, 0.30))

        np.clip(candidate, 0.0, 1000.0, out=candidate)
        # renormalize sum to 1 to avoid drift
        cand_sum = candidate.sum()
        if cand_sum > 0:

        cand_score, cand_conv, _ = conv_and_score(candidate)
        if cand_score < best_score:
            best = candidate
            best_score = cand_score
            best_conv = cand_conv

    # return list
    return [float(x) for x in best]
```

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution + AC1 score (exact)
# ----------------------------------------------------------------------
def _conv_score(seq: np.ndarray):
    """Return (score, full convolution array, max conv value) for a non‑negative sequence."""
    n = seq.shape[0]
    # FFT length: next power of two >= 2n‑1
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = float(conv.max())
    s = float(seq.sum())
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return score, conv, max_val

    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a
