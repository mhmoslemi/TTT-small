import numpy as np, time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1 via recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound using FFT (linear convolution)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------ #
    # Determine problem size and obtain a parent if offered.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256                     # fallback length
    L = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------ #
    # Build an initial pool of simple parametric candidates.
    best = None
    best_score = float('inf')

    def consider(arr):
        nonlocal best, best_score
        s = _ac1_score(arr, n, L)
        if s < best_score:
            best = arr.copy()
            best_score = s

    # 1) Parent (if any)
    if parent is not None:
        consider(parent)

    # 2) Classic binomial
    binom = _gen_binomial(n)
    consider(binom)

    # 3) Exponential decays
    for lam in np.linspace(5e-5, 5e-4, 30):
        arr = np.exp(-lam * np.arange(n, dtype=np.float64))
        consider(arr)

    # 4) Geometric sequences
    for r in np.linspace(0.95, 0.9995, 30):
        arr = r ** np.arange(n, dtype=np.float64)
        consider(arr)

    # 5) Sine‑power shapes (decreasing from the left)
    sin_vals = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    for alpha in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        arr = sin_vals ** alpha
        consider(arr)

    # ------------------------------------------------------------------ #
    # Hill‑climbing / simulated‑annealing loop.
    cur = best.copy()
    cur_score = best_score
    temperature = 1.0
    decay = 0.9995
    target = 1.50274349

    # Helper to keep a copy while staying cheap.
    def copy_array(a):
        return a.copy()

    while time.monotonic() < deadline:
        # pick move type
        move_type = rng.random()
        cand = copy_array(cur)

        if move_type < 0.4:
            # ---- block flattening (random location) ----
            w_low = max(1, n // 200)
            w_high = max(2, n // 12)
            width = int(rng.integers(w_low, w_high + 1))
            start = int(rng.integers(0, n - width + 1))
            block_avg = cand[start:start + width].mean()
            global_avg = cand.mean()

            best_score = cand_score

        accept = False
        if cand_score < cur_score:

        if accept:

            cur_score = cand_score

        # temperature schedule
        temperature = max(temperature * decay, 1e-6)

        # rare restart to best
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ------------------------------------------------------------------ #
    # Return the best sequence discovered.
    return [float(x) for x in best]
