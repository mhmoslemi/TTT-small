    # Find indexes i in range where partner exists:

def _targeted_move(a, conv, rng):

    n = a.size

    k_max = int(np.argmax(conv))

    # index range for i

    i_min = max(0, k_max - (n-1))

    i_max = min(k_max, n-1)

    i_range = np.arange(i_min, i_max+1)

    j_range = k_max - i_range

    prod = a[i_range] * a[j_range]

    idx = np.argmax(prod)

    i_star = i_range[idx]

    j_star = j_range[idx]

    min_val = min(a[i_star], a[j_star])

    if min_val < 1e-12:

        # cannot move, but we can fall back to random move

        return None

    # choose delta fraction randomly up to 10% of min_val

    delta = min_val * rng.random() * 0.1  # up to 10%

    cand = a.copy()

    if i_star == j_star:

        cand[i_star] -= delta

        # choose a random distinct index r

        r = rng.integers(0, n)

        if r == i_star:

            r = (r + 1) % n

        cand[r] += delta

    else:

        cand[i_star] -= delta

        cand[j_star] -= delta

        # choose two random distinct indices not among i_star/j_star

        r1 = rng.integers(0, n)

        while r1 == i_star or r1 == j_star:

            r1 = rng.integers(0, n)

        r2 = rng.integers(0, n)

        while r2 == i_star or r2 == j_star or r2 == r1:

            r2 = rng.integers(0, n)

        cand[r1] += delta

        cand[r2] += delta

    np.clip(cand, 0.0, 1000.0, out=cand)

    return cand

import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (next power‑of‑2 length)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def _targeted_move(a: np.ndarray, conv: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small amount from the pair that dominates the maximal convolution."""
    n = a.size
    k_max = int(np.argmax(conv))

    i_min = max(0, k_max - (n - 1))
    i_max = min(k_max, n - 1)
    i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
    j_range = k_max - i_range

    prod = a[i_range] * a[j_range]
    idx = int(np.argmax(prod))
    i_star = int(i_range[idx])
    j_star = int(j_range[idx])

    # ensure there is transferable mass
    min_val = min(a[i_star], a[j_star]) if i_star != j_star else a[i_star]
    if min_val <= 1e-12:
        return None

    # up to 30 % of the smaller entry, randomised
    delta = min_val * rng.random() * 0.30
    cand = a.copy()

    if i_star == j_star:
        # self‑pair: take delta from the entry and give it elsewhere
        cand[i_star] -= delta
        r = rng.integers(0, n)
        while r == i_star:
            r = rng.integers(0, n)
        cand[r] += delta
    else:
        # remove delta from both members of the dominant pair
        cand[i_star] -= delta
        cand[j_star] -= delta
        # redistribute to two random locations (different from the donors)
        r1 = rng.integers(0, n)
        while r1 == i_star or r1 == j_star:
            r1 = rng.integers(0, n)
        r2 = rng.integers(0, n)
        while r2 == i_star or r2 == j_star or r2 == r1:
            r2 = rng.integers(0, n)
        cand[r1] += delta
        cand[r2] += delta

    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _random_swap(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two random positions."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a.copy()
    cand = a.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score non‑negative step function under the given budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise – use the supplied parent if it exists, otherwise the sqrt‑series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = base.size
    else:
        n = 5582
        base = _sqrt_series(n)

    best = base.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    no_improve = 0
    while time.monotonic() < deadline:
        op = rng.random()

        # ---------- targeted reduction move ----------
        if op < 0.55:
            # need the current convolution for this move
            conv = _conv_self(best)
            cand = _targeted_move(best, conv, rng)
            if cand is None:
                continue

        # ---------- simple pairwise swap ----------
        elif op < 0.80:
            cand = _random_swap(best, rng)

        # ---------- block scaling ----------
        elif op < 0.95:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = rng.integers(0, n - width + 1)
            factor = float(np.exp(rng.normal(0.0, 0.12)))  # ≈[0.7,1.4]
            cand = best.copy()
            cand[start:start + width] *= factor

        # ---------- tiny additive noise ----------
        else:
            cand = best.copy()
            noise = rng.normal(0.0, 0.02, size=n)
            cand += noise

        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1

        # ---------- occasional random re‑shuffle to escape plateaus ----------
        if no_improve >= 2500:
            perm = rng.permutation(n)
            cand = best[perm]
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
                no_improve = 0
            else:
                # accept the shuffled version as a new starting point
                best = cand
                best_score = sc
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best discovered candidate as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
