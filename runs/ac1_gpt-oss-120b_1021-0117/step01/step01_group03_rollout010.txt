        contrib = best[idx] * best[rev_idx]
        # choose source as max contribution
        src_idx = idx[np.argmax(contrib)]
        src_val = best[src_idx]
        if src_val <= min_transfer_abs:
            # no mass to move; break or try random
            # fallback random move
            # use random transfer as earlier.

def _score(a):
    n = a.size
    m = 1 << ((2*n -1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[:2*n-1]
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i, length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def _conv_fft(a: np.ndarray, m: int) -> np.ndarray:
    """Linear convolution a*a via real FFT (length m, power of two)."""
    fa = np.fft.rfft(a, n=m)
    return np.fft.irfft(fa * fa, n=m)[: 2 * a.size - 1]


def _score(a: np.ndarray, m: int = None) -> float:
    """Exact objective used by the evaluator (fast via FFT)."""
    n = a.size
    if m is None:
        m = 1 << ((2 * n - 1).bit_length())
    conv = _conv_fft(a, m)
    max_b = float(conv.max())
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

    sqrt_seq = _sqrt_series(n)

    # pick the better of parent / sqrt_series as start
    if parent is not None:
        cand0 = parent.copy()
        score0 = _score(cand0)
        score_s = _score(sqrt_seq)
        if np.isfinite(score_s) and score_s < score0:
            best = sqrt_seq.copy()
            best_score = score_s
        else:
            best = cand0

    else:
        best = sqrt_seq.copy()
        best_score = _score(best)

    # pre‑compute FFT length (power of two)
    m = 1 << ((2 * n - 1).bit_length())

    # target (stop early if reached)
    TARGET = 1.50274349

    # control parameters
    max_frac = 0.07          # max fraction of source mass to move
    min_delta = 1e-9
    no_imp = 0               # counter for stagnation

    # ----- main optimisation loop -----
    while time.monotonic() < deadline:
        # compute convolution of current best to locate the active peak
        conv_best = _conv_fft(best, m)
        k_max = int(conv_best.argmax())
        # overlapping index range for this peak
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)

        # ----- gradient‑guided mass transfer -----
        if i_max - i_min + 1 < n:  # there exists an index outside the overlap
            idx = np.arange(i_min, i_max + 1)
            rev_idx = k_max - idx
            contrib = best[idx] * best[rev_idx]                     # contribution to the peak
            src = idx[np.argmax(contrib)]                           # worst contributor
            src_val = best[src]
            # pick a destination outside the overlapping block
            dst = None
            for _ in range(5):
                cand = rng.integers(0, n)
                if cand < i_min or cand > i_max:
                    dst = cand
                    break
            if dst is not None and src_val > min_delta:
                capacity = 1000.0 - best[dst]
                if capacity > 0.0:
                    delta = src_val * max_frac * rng.random()
                    delta = min(delta, capacity, src_val)
                    if delta > min_delta:
                        cand_seq = best.copy()
                        cand_seq[src] -= delta
                        cand_seq[dst] += delta
                        np.clip(cand_seq, 0.0, 1000.0, out=cand_seq)
                        sc = _score(cand_seq, m)
                        if sc < best_score:
                            best = cand_seq
                            best_score = sc
                            no_imp = 0

            cand += rng.normal(0.0, 0.02, size=n)

            t = rng.random() * 0.2          # up to 20 % of the sqrt‑series
            cand = (1.0 - t) * cand + t * sqrt_seq

        sc = _score(cand, m)
        if sc < best_score:

            best_score = sc
            no_imp = 0
            if best_score <= TARGET:

        else:
            no_imp += 1

        # occasional aggressive re‑normalisation when stuck
        if no_imp > 5000:
            # a gentle global smoothing (moving average) that preserves sum
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(best, kernel, mode='same')
            # renormalise to original sum
            smooth *= best.sum() / smooth.sum()
            np.clip(smooth, 0.0, 1000.0, out=smooth)
            sc = _score(smooth, m)
            if sc < best_score:
                best = smooth
                best_score = sc
            no_imp = 0

    # ----- return best found -----
    return [float(v) for v in best]
