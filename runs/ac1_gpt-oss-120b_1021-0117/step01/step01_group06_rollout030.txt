def _subgradient(a, k):
    n = a.shape[0]
    grad = np.zeros_like(a)
    # offset range
    start = max(0, k - (n-1))
    end = min(n-1, k)

    seg = a[start:end+1][::-1]  # length = end-start+1

def _conv_fft(a):
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    return conv
```

import time
import numpy as np

# ------------------------------------------------------------
# projection onto the simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# fast convolution via FFT (real-to-real)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# score = 2 * n * max(conv) / (sum(a)**2)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ------------------------------------------------------------
# subgradient of max(conv) at index k
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, k: int) -> np.ndarray:
    n = a.size
    grad = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(n - 1, k)
    if start <= end:
        idx = np.arange(start, end + 1)
        rev = a[idx][::-1]               # a[start:end+1] reversed
        grad[k - idx] = 2.0 * rev
    return grad


# ------------------------------------------------------------
# analytic "sqrt series" (a * a = 1 + … + x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
