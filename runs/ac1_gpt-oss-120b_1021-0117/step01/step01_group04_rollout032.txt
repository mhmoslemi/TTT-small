def proj_simplex(v):
    # v: np array, not necessarily positive, sum must become 1.
    # From Duchi et al. (2008).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _proj_simplex(v):
    # Project onto simplex (nonnegative, sum=1)
    # Using algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all values <= 0
        return np.full_like(v, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Works for any vector.

            best_score = cand_score

        if cand_score < cur_score:

        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            accept = rng.random() < prob
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay_temp

        # occasional random perturbation to escape stagnation
        if rng.random() < 0.005:
            # random block scaling

def _proj_simplex(v):
    # project v onto {w >=0, sum w = 1}
    # Inspired by Duchi et al.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return uniform
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1] - 1  # zero-indexed
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _ac1_score(arr: np.ndarray, L: int) -> float:
    """Fast AC1 upper bound using FFT (2·n·max(conv)/sum²)."""
    n = arr.shape[0]
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # keep a small safety margin for the sandbox
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---- initial sequence -------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        n = int(parent_arr.size)
        init = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        n = 256
        init = _gen_binomial(n)

    # Normalise to a probability vector (scale does not matter for the score)
    init = init / init.sum()

    # baseline candidates
    L = _next_pow2(2 * n - 1)
    best = init.copy()
    best_score = _ac1_score(best, L)

    # uniform
    uniform = np.full(n, 1.0 / n)
    us = _ac1_score(uniform, L)
    if us < best_score:
        best, best_score = uniform.copy(), us

    # binomial (useful when parent is absent)
    binom = _gen_binomial(n)
    binom = binom / binom.sum()
    bs = _ac1_score(binom, L)
    if bs < best_score:
        best, best_score = binom.copy(), bs

    cur = best.copy()
    cur_score = best_score

    # parameters for the search
    temperature = 0.5
    decay_temp = 0.9995
    step0 = 0.2
    it = 0

    # ---- main optimisation loop -------------------------------------------
    while time.monotonic() < deadline:
        # full convolution of the current vector
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)
        conv = conv[: 2 * n - 1]
        max_idx = int(conv.argmax())

        r = rng.random()
        if r < 0.45:
            # ---- gradient step (sub‑gradient of the max‑convolution) ----
            i = np.arange(n)
            j = max_idx - i
            mask = (j >= 0) & (j < n)
            g = np.zeros_like(cur)
            g[mask] = 2.0 * cur[j[mask]]
            it += 1
            step = step0 / math.sqrt(it)
            cand = cur - step * g
            cand = _proj_simplex(cand)
        elif r < 0.75:
            # ---- targeted mass transfer based on the contributing products ----
            # product of each entry with its mirror for the current worst index
            j = max_idx - np.arange(n)
            mask = (j >= 0) & (j < n)
            prod = np.zeros_like(cur)
            prod[mask] = cur[mask] * cur[j[mask]]
            i_high = int(np.argmax(prod))
            i_low = int(np.argmin(prod))
            if i_high != i_low:
                frac = rng.random() * 0.3            # up to 30 % transfer
                delta = cur[i_high] * frac
                cand = cur.copy()
                cand[i_high] -= delta
                cand[i_low] += delta
                cand = _proj_simplex(cand)
            else:
                continue
        else:
            # ---- random block scaling (helps escape local plateaux) ----
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.08))
            cand = cur.copy()
            cand[start:start + width] *= factor
            cand = _proj_simplex(cand)

        cand_score = _ac1_score(cand, L)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay_temp

        # occasional restart from the best solution seen so far
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 0.5
            it = 0

    # Return the best sequence as plain Python floats
    return [float(x) for x in best]
