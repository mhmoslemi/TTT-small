    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)

    # get n from height_sequence_1 length if exists else fallback:
    try: n = len(height_sequence_1)
        init_array = np.array(height_sequence_1, copy=False)
    except Exception:
        n = 256
        # initialize some shape, maybe binomial
        # define _generate_binomial_seq (as before)
        init_array = _generate_binomial_seq(n)

    # Use init = init_array, but we need to ensure sum>0
    # Normalize sum to 1 (or any constant)
    a = np.maximum(init_array.astype(np.float64), 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones(n, dtype=np.float64)/n
    else:
        a = a / total

    # Optionally apply random smoothing.
    # Compute initial conv using FFT:
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]

    cur_max = float(conv.max())
    best_a = a.copy()
    best_max = cur_max

    # Greedy random moves
    # Determine delta fraction range: maybe adapt based on iteration.
    # We'll limit to delta fraction up to 0.1 of source value.
    while time.monotonic() < deadline:
        # Randomly pick i,j
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j += 1
        ai = a[i]
        if ai < 1e-12:
            continue
        frac = rng.uniform(0.01, 0.15)  # moderate fraction
        delta = ai * frac
        # Update conv copy
        conv_new = conv.copy()
        # apply source
        conv_new[i:i+n] -= 2.0 * delta * a
        # apply target
        conv_new[j:j+n] += 2.0 * delta * a
        # quadratic self
        conv_new[2*i] += delta*delta
        conv_new[2*j] += delta*delta
        # cross term
        sum_idx = i + j
        conv_new[sum_idx] -= 2.0 * delta*delta

        new_max = float(conv_new.max())
        if new_max < cur_max - 1e-12:   # improvement
            # accept
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
    # after loop return best_a (as list) after maybe some final scaling safety
    final = best_a.tolist()
    # Ensure numeric safety:
    final = [float(min(max(x,0.0),1000.0)) for x in final]
    # Evaluate to be sure.
    try:
        _ = evaluate_sequence(final)
    except Exception:
        # fallback to initial normalized uniform
        final = (a / a.sum()).tolist()
    return final

    else:

    # ensure sum approx 1.

    fft_len = 1 << ((conv_len - 1).bit_length())

        fa = np.fft.rfft(arr, n=fft_len)
        return np.fft.irfft(fa*fa, n=fft_len)[:conv_len]

    conv = compute_conv(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    # Greedy loop
    while time.monotonic() < deadline:

        if rng.random() < 0.7:
            # random transfer move
            i = rng.integers(0, n)
            # ensure a[i] > small threshold
            if a[i] <= 1e-12:
                continue
            # choose j != i

            delta = a[i] * frac
            # apply incremental conv update
            conv_new = conv.copy()
            conv_new[i:i+n] -= 2.0 * delta * a
            conv_new[j:j+n] += 2.0 * delta * a
            conv_new[2*i] += delta*delta

            conv_new[i+j] -= 2.0 * delta*delta  # maybe i+j < conv_len always

                # accept
                a[i] -= delta
                a[j] += delta
                conv = conv_new
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
        else:

            k = int(conv.argmax())
            # compute gradient slice indices
            i_low = max(0, k - (n - 1))

                conv = conv_new
                cur_max = new_max

                    best_max = new_max
                    best_a = a.copy()
        # small break if time near deadline
        if time.monotonic() > deadline - 0.5:
            break

    final = best_a.tolist()
    final = [float(min(max(x, 0.0), 1000.0)) for x in final]
    try:
        _ = evaluate_sequence(final)
    except Exception:

    return final

def _exponential_seq(n: int, c: float) -> np.ndarray:
    """
    Return a length‑n non‑negative step sequence a_i = r^i,
    where r = 1 - c/n (0 < r < 1).  The sequence is normalized
    to have sum 1.
    """
    r = 1.0 - c / n
    if r <= 0.0:                     # safety – fall back to uniform
        return np.ones(n, dtype=np.float64) / n
    # use exp(log(r) * i) for numerical stability
    i = np.arange(n, dtype=np.float64)
    a = np.exp(i * np.log(r))
    s = a.sum()
    if s == 0.0:
        return np.ones(n, dtype=np.float64) / n
    a /= s
    return a

    candidate = [float(x) for x in a]

    # Quick sanity check with the authoritative evaluator.
    # If anything unexpected happens we fall back to a uniform vector.
    try:
        _ = evaluate_sequence(candidate)   # will raise if invalid
        return candidate
    except Exception:
        # fallback – uniform distribution (sum = 1, definitely feasible)
        fallback = (np.ones(n, dtype=np.float64) / n).tolist()
        return [float(x) for x in fallback]
