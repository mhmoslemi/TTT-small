def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,float(budget_s)-0.5)
    try:
        n = len(height_sequence_1)
        a = height_sequence_1.astype(np.float64).copy()
    except Exception:
        n = 5000
        a = np.full(n, 1.0/n, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.full_like(a, 1.0 / a.size)
    a = a / a.sum()   # normalize to 1 for simplification

    def conv_full(v):
        N = v.shape[0]
        size = 2*N - 1
        m = 1 << (size-1).bit_length()
        fv = np.fft.fft(v, m)
        c = np.fft.ifft(fv*fv).real[:size]
        return c

    def score(v):
        c = conv_full(v)
        return 2.0 * n * c.max() / (v.sum()**2)

    # fallback uniform
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    best = a
    best_score = score(best)
    uniform_score = score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # parameters
    step = 0.15   # initial step
    min_step = 1e-5
    decay = 0.99
    grow = 1.02
    max_step = 0.5

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # compute max conv index
        c = conv_full(best)
        k = int(c.argmax())
        # compute partner vector for index k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(best)
        partner[idx] = best[k - idx]
        # gradient (subgradient of max conv)
        g = 2.0 * partner
        # propose step
        cand = best - step * g
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            cand = cand / s
        else:
            cand = uniform.copy()
        cand_score = score(cand)

        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # increase step a little to accelerate
            step = min(max_step, step * grow)
        else:
            # reduce step
            step = max(min_step, step * decay)

        # occasional random perturbation to escape flat regions
        if rng.random() < 0.004:
            # multiply a random block by random factor >0
            width = int(rng.integers(1, max(2, n // 40)))
            start = int(rng.integers(0, max(1, n - width + 1)))

            pert = np.clip(pert, 0.0, None)
            pert_sum = pert.sum()
            if pert_sum > 0:
                pert = pert / pert_sum
                pert_score = score(pert)
                if pert_score < best_score:
                    best = pert
                    best_score = pert_score
                    step = min(max_step, step * grow)

        # occasional full random restart if stuck for many iterations

            if rng.random() < 0.05:

    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    return [float(v) for v in final]
