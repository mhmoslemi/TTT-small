def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,budget_s - 2.0) # a margin
    if 'height_sequence_1' in globals():
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = base_arr.size
    else:
        n=256
        base_arr = np.ones(n)
    # compute sqrt_series shape
    def sqrt_series(n):
        a=np.empty(n)
        a[0]=1.0
        for k in range(1,n):
            if k==1:
                s=0.0
            else:
                s=np.dot(a[1:k], a[k-1:0:-1])
            a[k]=(1.0 - s)/2.0
        return a
    sqrt_arr = sqrt_series(n)
    # normalise both to sum 1
    base_arr = np.maximum(base_arr,0.0)
    if base_arr.sum() ==0:
        base_arr = np.ones(n)
    base_arr = base_arr / base_arr.sum()
    sqrt_arr = sqrt_arr / sqrt_arr.sum()
    # compute scores
    def conv_max(a):
        L=1 << ((2*n-1).bit_length())
        fv=np.fft.rfft(a, n=L)
        conv=np.fft.irfft(fv*fv, n=L)[:2*n-1]
        return conv
    def score_of(a):
        conv = conv_max(a)
        return 2.0*n*conv.max()/(a.sum()**2)
    # Evaluate both
    best = base_arr.copy()
    best_sc = score_of(best)
    # maybe replace with sqrt if better
    sqrt_sc = score_of(sqrt_arr)
    if sqrt_sc < best_sc:
        best = sqrt_arr.copy()
        best_sc = sqrt_sc
    # main loop
    # hyperparameters
    max_iters = 200000
    # we want step size decreasing
    step_frac = 0.2
    # tolerance for considering max positions
    tol_rel = 1e-12
    iter_count=0
    while time.monotonic() < deadline and iter_count < max_iters:
        iter_count+=1
        # evaluate conv and max indices
        conv = conv_max(best)
        max_val = conv.max()
        # indices where conv close to max
        # using absolute tolerance
        idxs = np.where(conv >= max_val - max_val*1e-12)[0]
        # compute gradient g
        g = np.zeros(n)
        for k in idxs:
            # shift positive
            if k>0:
                g[:n-k] += best[k:]
                g[k:] += best[:n-k]
            else:
                # k ==0
                g += 2*best
        # choose source i and target j
        # source among top 10% gradient where a>threshold
        # pick i = argmax(g * (best>eps))
        eps=1e-12
        mask = best>eps
        if not np.any(mask):
            break
        # compute weighted gradient
        weighted = g*mask
        i = int(np.argmax(weighted))
        # target j = argmin(g)
        j = int(np.argmin(g))
        # ensure i!=j
        if i==j:
            # random other
            j = int(rng.integers(0,n))
            if j==i:
                continue
        # move delta
        delta = best[i] * step_frac
        if delta < 1e-12:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # cand sum unchanged
        # enforce nonnegativity
        if cand[i] < -1e-12:
            continue
        cand = np.maximum(cand,0.0)
        cand = cand / cand.sum()  # renorm to sum=1 (redundant but ensure)
        sc = score_of(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand
            # maybe reduce step size for finer search
            step_frac = max(0.01, step_frac*0.99)
        # occasional random perturbation
        if rng.random() < 0.01:
            # pick random indices
            i2 = int(rng.integers(0,n))
            j2 = int(rng.integers(0,n-1))
            if j2 >= i2: j2+=1
            if best[i2] > 0:
                delta2 = best[i2] * rng.random()*0.3
                cand2 = best.copy()
                cand2[i2] -= delta2
                cand2[j2] += delta2
                cand2 = np.maximum(cand2,0.0)
                cand2 = cand2 / cand2.sum()
                sc2 = score_of(cand2)
                if sc2 < best_sc:
                    best_sc = sc2
                    best = cand2
        # occasionally recompute step_frac to wander
        if iter_count % 5000 == 0:
            step_frac = 0.2 * (1.0 - iter_count / max_iters)
    # final early exit if time
    # Scale back to any magnitude (no need)
    # Return list of floats
    return [float(x) for x in best]

def _conv_fft(a):

    n = a.shape[0]

        if src == tgt:

            src = int(rng.integers(0, n))
            tgt = int(rng.integers(0, n-1))

                tgt += 1

        if delta <= 0:
            # reduce step and continue
            step_frac = max(0.01, step_frac * 0.9)
            continue

        cand = best.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # Ensure nonnegative and renormalize (just in case)
        cand = np.maximum(cand, 0.0)

        if cand_sum == 0:

        cand = cand / cand_sum

        sc = _score(cand)
        if sc < best_score:

            best = cand

import time

# ------------------------------------------------------------
# elementary utilities
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of the series a with a*a = 1+…+x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # s = sum_{i=1}^{k-1} a_i * a_{k-i}
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _conv_fft(vec: np.ndarray) -> np.ndarray:
    """Linear convolution of vec with itself via FFT (zero‑padded)."""
    n = vec.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(vec: np.ndarray) -> float:
    """Exact score used by the evaluator (sum may be any positive number)."""
    n = vec.shape[0]
    conv = _conv_fft(vec)
    return float(2.0 * n * conv.max() / (vec.sum() ** 2))


def _max_gradient(a: np.ndarray, conv: np.ndarray = None):
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Returns (gradient vector, indices achieving the max, max value).
    """

        d = np.abs(p - js)                 # distances in [0, n-1]
        g[js] += 2.0 * a[d]                # factor 2 because each term appears twice
    return g, idxs, max_val
