import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: generate the classic binomial (arcsine) seed
# ----------------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ----------------------------------------------------------------------
# Fast convolution + score using a single FFT length (reused everywhere)
# ----------------------------------------------------------------------
def _make_fft_len(n: int) -> int:
    # we need length >= 3n-2 for the convolution with a reversed copy
    need = 3 * n - 2
    return 1 << (need - 1).bit_length()


def _fast_score(seq: np.ndarray, fft_len: int, n: int):
    """Return (score, full convolution, max index, max value)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv_full = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    max_val = np.max(conv_full)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv_full, -1, max_val
    score = 2.0 * n * max_val / (s * s)
    max_idx = int(np.argmax(conv_full))
    return float(score), conv_full, max_idx, max_val


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Determine problem size and optional parent construction
    # ------------------------------------------------------------------
    n = 3809  # default length if nothing else is provided
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Initialise best sequence (global best) and working sequence (cur)
    # ------------------------------------------------------------------
    best = _initial_binomial(n)
    best = best / np.sum(best)  # L1‑normalised for numeric stability
    if parent is not None:
        cand = parent / np.sum(parent)
        # evaluate the parent early – keep whichever is better
        fft_len = _make_fft_len(n)
        cand_score, _, _, _ = _fast_score(cand, fft_len, n)
        best_score, best_conv, best_kmax, _ = _fast_score(best, fft_len, n)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            best_conv = None  # will be recomputed below
    else:
        best_score, best_conv, best_kmax, _ = _fast_score(best, _make_fft_len(n), n)

    # Ensure we have the convolution for the current best
    fft_len = _make_fft_len(n)
    best_score, best_conv, best_kmax, _ = _fast_score(best, fft_len, n)

    # Working copy used for simulated‑annealing moves
    cur = best.copy()
    cur_score = best_score
    cur_conv = best_conv
    cur_kmax = best_kmax

    # Temperature schedule for occasional uphill moves
    T_start = 0.02
    T_end = 0.0005

    # ------------------------------------------------------------------
    # Main optimisation loop (simulated annealing + targeted moves)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # linear temperature decay
        t_now = time.monotonic()
        elapsed = t_now - (deadline - budget_s + 2.0)

                transfer = candidate[i] * alpha
                candidate[i] -= transfer
                candidate[j] += transfer

        # --------------------------------------------------------------
        # 6 – global multiplicative jitter (small random noise)
        # --------------------------------------------------------------
        else:
            sigma = 0.03
            jitter = np.exp(rng.normal(0.0, sigma, size=n))
            candidate *= jitter

        # ------------------------------------------------------------------
        # Normalise, clip and evaluate
        # ------------------------------------------------------------------
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        s = candidate.sum()
        if s > 0.0:
            candidate /= s

        cand_score, cand_conv, cand_kmax, _ = _fast_score(candidate, fft_len, n)

        # ------------------------------------------------------------------
        # Acceptance (simulated annealing)
        # ------------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probabilistic uphill move
            prob = math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = candidate
            cur_score = cand_score
            cur_conv = cand_conv
            cur_kmax = cand_kmax

                best = candidate.copy()
                best_score = cand_score
                best_conv = cand_conv
                best_kmax = cand_kmax

    # ----------------------------------------------------------------------
    # Return best candidate as a plain Python list (the evaluator will
    # recompute the exact score itself).
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
