    rng=np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-0.5)
    
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim==1 and parent_arr.size>0 and np.isfinite(parent_arr).all():
            n = parent_arr.size
            best = np.clip(parent_arr, 0.0, 1000.0).copy()
        else:
            raise Exception()
    except Exception:
        n = 3809
        best = _initial_binomial(n)
    
    # compute initial score/convolution
    best_conv = _conv_fft(best)
    best_score = 2.0 * n * best_conv.max() / (best.sum()**2)
    
    # If parent (above) not better than binomial, compute binomial score above; we already have best = whichever.
    # We'll keep best and conv.
    
    # Main loop
    while time.monotonic() < deadline:
        # Choose operation based on probabilities
        op = rng.integers(0, 4)  # 0: focused move, 1: block scaling, 2: smooth neighbor, 3: random single index
        candidate = best.copy()
        if op == 0:
            # targeted mass redistribution
            conv = best_conv
            kmax = int(conv.argmax())
            lo = max(0, kmax - (n-1))
            hi = min(kmax, n-1)
            idxs = np.arange(lo, hi+1)
            products = best[idxs] * best[kmax - idxs]
            # choose max product where both >0
            max_idx = idxs[products.argmax()]
            i = int(max_idx)
            j = int(kmax - i)
            # if i == j or either zero, fallback to random move
            if i == j or best[i] <= 0 or best[j] <= 0:
                # fallback to simple random single index scaling
                idx = rng.integers(0, n)

                candidate[idx] *= factor

                # candidate currently = best copy
                low_idx = int(np.argmin(candidate))
                if low_idx in (i, j):

                    # get next lowest via argsort
                    low_candidates = np.argsort(candidate)

# ----- utilities -----------------------------------------------------------

def _initial_binomial(n: int) -> np.ndarray:
    """Generate decreasing positive a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_fft(x: np.ndarray) -> np.ndarray:
    """Full convolution x*x using power‑of‑two FFT."""
    n = x.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    return conv


def _score_from(conv: np.ndarray, total: float, n: int) -> float:
    """Exact AC1 score from pre‑computed convolution and sum."""
    if total <= 0.0:
        return float("inf")
    return 2.0 * n * np.max(conv) / (total * total)
