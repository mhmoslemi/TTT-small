            g[j] = 2.0 * a[max_idx - j]  # ∂ (a*a)_{max_idx} / ∂ a_j
        cand = a - step * g
        cand = _project_simplex(cand)

        # evaluate candidate
        max_val_c, max_idx_c = _conv_max_and_index(cand, L)
        cand_score = 2.0 * n * max_val_c

        # accept if we have a genuine improvement
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            max_idx = max_idx_c
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # occasional simulated‑annealing acceptance (temperature ≈ step)
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-8)):
                a = cand
                cur_score = cand_score
                max_idx = max_idx_c

        # ------- occasional random mass transfer for extra exploration -------
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)

                max_val2, max_idx2 = _conv_max_and_index(cand2, L)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    max_idx = max_idx2
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()

        # ------- step‑size decay (ensures convergence) -------
        step = max(min_step, step * decay)

        # cheap early exit if we are about to run out of time
        if time.monotonic() + 0.5 > deadline:
            break

    # return plain Python list – the evaluator expects a list of floats
    return [float(x) for x in best]
