def conv_max(a):
    N = 1 << ((2*len(a)-1).bit_length())
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[:2*len(a)-1]
    return float(conv.max())

    cur_score = best_score
    # Precompute N for FFT
    N_fft = 1 << ((2*n-1).bit_length())
    # Simulated annealing parameters
    temp0 = 1.0
    temp = temp0
    # schedule: exponential decay
    decay = 0.9995
    # Define move functions
    while time.monotonic() < deadline:
        # choose move type
        move_type = rng.integers(0,3)
        cand = current_seq.copy()
        if move_type == 0:
            # transfer mass between two positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # choose delta limited by cand[i]
            max_delta = cand[i] * 0.5  # up to half of i's mass
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            cand[i] -= delta
            cand[j] += delta
        elif move_type == 1:
            # uniform addition epsilon to all entries, and subtract from a random early index
            eps = 1e-3 * rng.random()  # small epsilon
            # ensure not too big
            cand += eps
            # subtract from early index to keep max low
            i = rng.integers(0, n//10)  # early
            subtract = min(eps * (n//10), cand[i]*0.5)
            cand[i] -= subtract
        else:
            # small perturbation of a single entry
            i = rng.integers(0, n)
            delta = (rng.random() - 0.5) * cand[i] * 0.1
            new_val = cand[i] + delta
            if new_val < 0:
                new_val = 0.0
            cand[i] = new_val

        # clip to bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        # evaluate only if sum not too small
        sum_c = cand.sum()
        if sum_c < 0.01:
            continue
        # compute convolution max using FFT
        # compute conv via FFT
        fa = np.fft.rfft(cand, N_fft)
        conv = np.fft.irfft(fa * fa, N_fft)
        max_b = conv.max()
        score = 2 * n * max_b / (sum_c * sum_c)
        # accept based on score
        if score < cur_score or rng.random() < math.exp((cur_score - score) / max(temp, 1e-12)):
            current_seq = cand
            cur_score = score
            if score < best_score:
                best_seq = cand.copy()
                best_score = score
        # decrease temp
        temp *= decay
        # occasional restart
        if rng.random() < 0.001:
            # restart from best
            current_seq = best_seq.copy()
            cur_score = best_score
    return [float(x) for x in best_seq]

import numpy as np

def _sidon_like_sequence(n):
    """Classic a_i = C(2i,i)/4**i sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _next_pow_two(x: int) -> int:
    """Small utility: smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

                S_j = jig.sum()
                if S_j >= 0.01:
                    score_j = 2.0 * n * M_j / (S_j * S_j)
                    if score_j < cur_score:
                        cur = jig
                        cur_score = score_j
                        if score_j < best_score:
                            best_seq = jig.copy()
                            best_score = score_j
            # very occasional restart from the current best
            if rng.random() < 0.0004:
                cur = best_seq.copy()
                cur_score = best_score
                lr = 0.2

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list).
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
```
