def _initial_binomial(n: int) -> np.ndarray:
    """Generate decreasing positive a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _fast_score(seq: np.ndarray):

            idx = rng.integers(0, n)
            factor = math.exp(rng.normal(0.0, sigma_single))
            cand[idx] *= factor

        elif op == 2:  # targeted mass relocation from current max‑convolution pair
            k_max = int(np.argmax(cur_conv))
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            # perform a few small relocations in one mutation
            for _ in range(rng.integers(1, 4)):
                i = rng.integers(lo, hi + 1)
                j = k_max - i
                if i == j:
                    continue
                alpha = rng.uniform(alpha_low, alpha_high)
                # remove a fraction from both ends
                rem_i = cand[i] * alpha
                rem_j = cand[j] * alpha
                cand[i] -= rem_i
                cand[j] -= rem_j
                total = rem_i + rem_j
                # split the mass to two low‑value positions
                t1 = rng.integers(0, n)
                while t1 == i or t1 == j:
                    t1 = rng.integers(0, n)
                t2 = rng.integers(0, n)
                while t2 == i or t2 == j or t2 == t1:
                    t2 = rng.integers(0, n)
                cand[t1] += total * 0.5
                cand[t2] += total * 0.5

        elif op == 3:  # local neighbour averaging
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        elif op == 4:  # global gentle smoothing
            smooth = np.convolve(cand, kernel, mode='same')
            cand = (1.0 - gamma_smooth) * cand + gamma_smooth * smooth

        elif op == 5:  # additive Gaussian jitter (small)
            sigma_add = 0.02 * np.mean(cand) if np.mean(cand) > 0 else 0.01
            cand += rng.normal(0.0, sigma_add, size=n)

        else:  # op == 6 : enforce symmetry (mirroring)
            cand = (cand + cand[::-1]) * 0.5

        # ---- enforce bounds ----------------------------------------------------
        np.clip(cand, 0.0, 1000.0, out=cand)

        # ---- evaluate candidate -------------------------------------------------
        cand_score, cand_conv, _ = _fast_score(cand)

        # ---- temperature (linear cooling) --------------------------------------
        elapsed = time.monotonic() - (deadline - total_time)
        frac = elapsed / total_time if total_time > 0 else 1.0
        temperature = max(T0 * (1.0 - frac), Tmin)

        # ---- acceptance test ----------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            cur_conv = cand_conv
            # update global best
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                best_conv = cand_conv

    # -------------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # -------------------------------------------------------------------------
    return [float(x) for x in best]
