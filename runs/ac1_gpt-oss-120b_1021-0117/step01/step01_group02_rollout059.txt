    if total < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total  # normalize sum=1

    uniform = np.full(n, 1.0 / n)
    # compute initial score for a and uniform
    def conv_fft(seq):
        # m size
        return np.fft.irfft(np.fft.rfft(seq, m) * np.fft.rfft(seq, m), m)[:2*n-1]
    # Determine sizes
    m = 1 << int(math.ceil(math.log2(2*n - 1)))
    m2 = 1 << int(math.ceil(math.log2(3*n)))

    # compute score function
    def score_of(seq):
        conv = np.fft.irfft(np.fft.rfft(seq, m) * np.fft.rfft(seq, m), m)[:2*n-1]
        maxv = conv.max()
        return 2.0 * n * maxv

    # initial scores
    cur_score = score_of(a)
    best_score = cur_score
    best_seq = a.copy()
    uniform_score = score_of(uniform)
    if uniform_score < best_score:
        best_score = uniform_score
        best_seq = uniform.copy()
        a = uniform.copy()  # maybe start from uniform
        cur_score = uniform_score

    # parameters
    alpha = 0.1  # initial step size
    gamma = 1.0   # initial smoothing parameter
    max_gamma = 30.0
    stagnation = 0
    iteration = 0

    # main loop
    while time.monotonic() < deadline:

        cur_score = 2.0 * n * max_val

            best_score = cur_score
            best_seq = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # compute weights
        # gamma may be increased gradually to focus
        if stagnation > 200 and gamma < max_gamma:
            gamma = min(max_gamma, gamma * 1.05)
            stagnation = 0  # reset after increase

        w = np.exp(gamma * (conv - max_val))

            break

    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total

    # fallback uniform
    uniform = np.full(n, 1.0 / n)

    # compute FFT lengths
    m = 1 << int(math.ceil(math.log2(2*n - 1)))
    m2 = 1 << int(math.ceil(math.log2(3*n)))

    def convolve_self(seq):
        fft = np.fft.rfft(seq, m)
        return np.fft.irfft(fft * fft, m)[:2*n-1]

    def compute_score(seq):
        conv = convolve_self(seq)
        return 2.0 * n * conv.max()

    cur_score = compute_score(a)
    uniform_score = compute_score(uniform)
    if uniform_score < cur_score:
        a = uniform.copy()
        cur_score = uniform_score

    best_seq = a.copy()
    best_score = cur_score

    alpha = 0.1
    gamma = 1.0
    max_gamma = 30.0
    stagnation = 0
    it = 0

    while time.monotonic() < deadline:
        conv = convolve_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 200 and gamma < max_gamma:
            gamma = min(max_gamma, gamma * 1.05)
            stagnation = 0

        # weights emphasizing maximal entries
        w = np.exp(gamma * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0:
            w_norm = w
        else:
            w_norm = w / w_sum

        # gradient via convolution of a_rev and w_norm
        a_rev = a[::-1]
        fft_a_rev = np.fft.rfft(a_rev, m2)
        fft_w = np.fft.rfft(w_norm, m2)

        step_alpha = alpha
        success = False
        for _ in range(5):
            cand = a - step_alpha * grad
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()

                continue
            cand = cand / s
            cand_score = compute_score(cand)
            if cand_score < cur_score:

                cur_score = cand_score
                success = True
                alpha = step_alpha * 1.1
                break

        a = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        # fallback: exponential decay (always positive)
        a = np.exp(-np.arange(n) / (n / 10.0))

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s                     # normalise to sum == 1

    # ----------------------------------------------------------------------
    # uniform fallback (its score is exactly 2)
    uniform = np.full(n, 1.0 / n)

    # convolution length utilities (powers of two for FFT speed)
    m = 1 << int(math.ceil(math.log2(2 * n - 1)))   # >= 2n-1
    m2 = 1 << int(math.ceil(math.log2(3 * n)))    # >= 3n-2

    def convolve_self(v):
        """Linear convolution v*v (full length) using FFT."""
        f = np.fft.rfft(v, m)
        return np.fft.irfft(f * f, m)[: 2 * n - 1]

    def score_of(v):
        """Objective 2*n*max(conv(v,v))  (sum(v) == 1)."""
        c = convolve_self(v)
        return 2.0 * n * c.max()

    # initialise best solution
    cur_score = score_of(a)
    if score_of(uniform) < cur_score:
        a = uniform.copy()
        cur_score = score_of(a)

    best_seq = a.copy()
    best_score = cur_score

    # ----------------------------------------------------------------------
    # algorithm parameters
    alpha = 0.5          # step size for gradient descent
    step_factor = 0.4    # fraction of source mass moved in transfer step
    min_step = 1e-7
    gamma = 0.5          # smoothing for max‑focus weights
    max_gamma = 30.0
    stagnation = 0
    it = 0

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # current convolution and the index attaining its maximum
        conv = convolve_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # update global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # ------------------------------------------------------------------
        # adapt smoothing when progress stalls
        if stagnation > 250 and gamma < max_gamma:
            gamma = min(max_gamma, gamma * 1.07)
            stagnation = 0

        # ------------------------------------------------------------------
        # ---- 1. SIMPLE MASS‑TRANSFER STEP (like the previous approach) ----
        # partner[i] = a[max_idx - i] when the index is valid, else 0
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(a)
        partner[idx] = a[k - idx]

        src = int(partner.argmax())
        dest_candidates = np.where(partner == partner.min())[0]
        dest = int(rng.choice(dest_candidates)) if dest_candidates.size else rng.integers(0, n)

        if src != dest:
            delta = a[src] * step_factor
            if delta < min_step:
                delta = min_step
            if delta > a[src]:
                delta = a[src]
            cand = a.copy()
            cand[src] -= delta
            cand[dest] += delta
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s > 0:
                cand = cand / s
                cand_score = score_of(cand)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    # slightly increase step size on success
                    step_factor = min(0.9, step_factor * 1.03)
                    # continue to next iteration (skip gradient step for now)
                    it += 1
                    if best_score < 1.48:
                        break
                    continue

        # ------------------------------------------------------------------
        # ---- 2. GRADIENT‑DESCENT STEP (smooth max focus) -----------------
        # build smoothing weights centred at the maximum entries
        w = np.exp(gamma * (conv - max_val))          # 1 at maximal locations
        w_sum = w.sum()
        w_norm = w / w_sum if w_sum != 0 else w

        # gradient = 2 * (a_rev * w_norm) evaluated at suitable slice
        a_rev = a[::-1]
        fft_a_rev = np.fft.rfft(a_rev, m2)
        fft_w = np.fft.rfft(w_norm, m2)
        grad_full = np.fft.irfft(fft_a_rev * fft_w, m2)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]       # length n

        # linesearch on the gradient direction
        step = alpha
        improved = False
        for _ in range(5):
            cand = a - step * grad
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s <= 0:
                step *= 0.5
                continue
            cand = cand / s
            cand_score = score_of(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                alpha = step * 1.1                      # a little larger next time
                break
            step *= 0.5
        if not improved:
            alpha = max(1e-4, alpha * 0.8)               # shrink if stuck

        # ------------------------------------------------------------------
        # ---- 3. OCCASIONAL RANDOM SHAKES TO ESCAPE LOCAL MINIMA ---------
        if rng.random() < 0.02:
            width = int(rng.integers(max(1, n // 300), max(2, n // 30)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            cut = a.copy()
            factor = np.exp(rng.normal(0.0, 0.25, size=width))
            cut[start:start + width] *= factor
            cut = np.clip(cut, 0.0, None)
            s = cut.sum()
            if s > 0:
                cut = cut / s
                cut_score = score_of(cut)
                if cut_score < cur_score:
                    a = cut
                    cur_score = cut_score

        it += 1
        if best_score < 1.48:            # target comfortably passed
            break

    # ----------------------------------------------------------------------
    # final sanity check – the evaluator will also clip, but we keep it tidy
    final = np.clip(best_seq, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
