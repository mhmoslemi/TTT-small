            frac = rng.uniform(0.05, 0.15)

            cand[i_max] -= removed
            # spread the removed mass over a few random positions
            k = rng.integers(2, 6)
            targets = rng.choice(n, size=k, replace=False)
            cand[targets] += removed / k

        elif op == 3:                             # smooth a neighbour pair
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        else:                                      # tiny random jitter
            i = rng.integers(0, n)
            delta = rng.normal(0.0, 0.05 * cand[i] + 0.001)
            cand[i] = max(0.0, cand[i] + delta)

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score, _, _ = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            best_conv = _

    # ------------------------------------------------------------------
    # Return the best found sequence as a pure Python list of floats.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
