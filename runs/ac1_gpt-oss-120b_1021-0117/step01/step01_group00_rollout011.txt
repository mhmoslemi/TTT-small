import numpy as np
import time

# ---------- Helper functions (top‑level) ----------
def _max_conv_fft(a: np.ndarray) -> float:
    """Maximum coefficient of the full convolution a*a using a power‑of‑2 FFT."""
    n = a.shape[0]
    length = 2 * n - 1
    N = 1 << (length.bit_length())          # smallest power of two ≥ length
    A = np.fft.rfft(a, n=N)
    B = A * A                               # element‑wise product (no conjugate)
    conv = np.fft.irfft(B, n=N)
    return float(conv[:length].max())


def _fast_ac1_score(a: np.ndarray) -> float:
    """Fast AC1 upper bound (exact, but uses FFT for the max‑convolution)."""
    s = a.sum()
    if s <= 0.0:
        return np.inf
    n = a.shape[0]
    maxb = _max_conv_fft(a)
    return 2.0 * n * maxb / (s * s)


def _sidon_like(n: int) -> np.ndarray:
    """Classic Sidon‑like decreasing sequence a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a
