def compute_bound(seq):
    # seq is np.ndarray of shape (n,) float
    n = seq.shape[0]
    # zero pad to next power of 2 maybe 2*n
    size = 1 << ((2*n - 1).bit_length())
    # Compute convolution via FFT

def compute_score(seq):
    n = seq.shape[0]
    size = fft_size (precomputed)
    fft_seq = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_seq * fft_seq, size)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    return 2.0 * n * max_b / (sum_a * sum_a)

    AC1 upper bound as defined by the pre‑imported `evaluate_sequence`.

    best_seq = parent.copy()

    # ----------------------------------------------------------------------
    fft_size = 1 << ((2 * n - 1).bit_length())  # next power of two

    def fast_score(arr: np.ndarray) -> float:
        """Return the AC1 bound for `arr` using FFT (no safety checks)."""
        f = np.fft.rfft(arr, fft_size)
        conv = np.fft.irfft(f * f, fft_size)[: 2 * n - 1]
        max_b = np.max(conv)
        s = np.sum(arr)
        # guard against division by zero (should never happen)
        return 2.0 * n * max_b / (s * s) if s > 0 else np.inf
