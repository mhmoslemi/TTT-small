import time
import math
import numpy as np

# Global placeholder for FFT length (set inside propose_candidate)
_FFT_M = None


def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, decreasing positive."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _initial_exponential(n: int, scale: float = 0.01) -> np.ndarray:
    """a_i = exp(-scale * i)"""
    return np.exp(-scale * np.arange(n, dtype=np.float64))


def _initial_powerlaw(n: int, p: float = 0.5) -> np.ndarray:
    """a_i = (i+1)^-p, normalized to similar L1 as binomial."""
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a


def _initial_triangular(n: int) -> np.ndarray:
    """Linear decreasing from 1 to 0."""
    return np.linspace(1.0, 0.0, n, dtype=np.float64)


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution array, max conv) using global FFT length."""
    global _FFT_M
    n = seq.shape[0]
    f = np.fft.rfft(seq, n=_FFT_M)
    conv = np.fft.irfft(f * f, n=_FFT_M)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val
