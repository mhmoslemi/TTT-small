    a = np.empty(n, dtype=np.float64)

def _ac1_score(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[:2*n-1]
    max_b = conv.max()

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i for i=0..n-1 (recurrence)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_and_info(arr: np.ndarray, n: int, L: int):
    """
    Return (score, idx_of_max, max_value) using FFT convolution.
    Score = 2 * n * max(conv) / (sum(arr)**2)
    """
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    idx = int(conv.argmax())
    max_val = float(conv[idx])
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf'), idx, max_val
    return float(2.0 * n * max_val / (s * s)), idx, max_val

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----- problem size ----------------------------------------------------

        n = 256
