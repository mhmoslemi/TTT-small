def _fast_score(seq):
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]

import numpy as np
import time

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a decreasing sequence a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution array, max(conv)) using FFT."""
    n = seq.shape[0]
    # pad to next power of two covering full linear convolution length
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Hill‑climbing / simulated‑annealing search for a low AC1 bound."""
    rng = np.random.default_rng(seed)
