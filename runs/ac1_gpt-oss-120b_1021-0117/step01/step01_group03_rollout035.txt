def interleave_extremes(arr):
    # arr sorted descending (e.g., sqrt_series). produce pattern [arr[0], arr[-1], arr[1], arr[-2], ...]
    result = np.empty_like(arr)
    left = 0
    right = arr.size - 1
    idx = 0
    while left <= right:
        result[idx] = arr[left]
        idx += 1
        left += 1
        if left <= right:
            result[idx] = arr[right]
            idx += 1
            right -= 1
    return result

def _conv_self(a):
   # compute linear convolution using real FFT, using size m = next power of two >= 2*n-1
   n = a.shape[0]
   m = 1 << ((2*n - 1).bit_length())
   fa = np.fft.rfft(a, n=m)
   prod = fa * fa
   res = np.fft.irfft(prod, n=m)[:2*n-1]
   return res

def _score_fast(a, conv):
   n = a.shape[0]
   s = a.sum()
   if s < 0.01:
      return np.inf
   return 2.0 * n * conv.max() / (s * s)

def _sqrt_series(n):
   a = np.empty(n, dtype=np.float64)
   a[0] = 1.0
   for i in range(1,n):
       a[i] = a[i-1] * (2.0*i - 1.0) / (2.0*i)  # formula: (2i-1)/(2i)
   return a

def interleave_extremes(a):
    n = a.shape[0]
    out = np.empty_like(a)
    left = 0
    right = n - 1
    idx = 0
    while left <= right:
        out[idx] = a[left]
        idx += 1
        left += 1
        if left <= right:
            out[idx] = a[right]
            idx += 1
            right -= 1
    return out

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (next power of two)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence, length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def _interleave_extremes(arr: np.ndarray) -> np.ndarray:
    """Place the largest, smallest, second‑largest, second‑smallest, …"""
    n = arr.size
    out = np.empty_like(arr)
    left, right, pos = 0, n - 1, 0
    while left <= right:
        out[pos] = arr[left]
        left += 1
        pos += 1
        if left <= right:
            out[pos] = arr[right]
            right -= 1
            pos += 1
    return out


def _score_fast(a: np.ndarray, conv: np.ndarray) -> float:
    """Objective 2·n·max(conv) / (sum(a))²."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * a.size * conv.max() / (s * s)
