def project_simplex(v):
    # ensure sum(v)=1 and v>=0
    # Using algorithm from https: // arxiv.org/pdf/1309.1541.pdf
    # Input: any vector (not necessarily sum=1)
    # Output: projection onto simplex {x >=0, sum x =1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # expects numpy 1d
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero? return uniform
        return np.full(n, 1.0/n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    def conv_of(v):

        c = np.fft.irfft(F * F, N)[:2*n-1]

        g = np.zeros_like(a)
        i_start = max(0, k - (n-1))

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, N: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    Fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(Fa * np.conj(Fa), n=N)[: 2 * a.size - 1]
    return c


# ----------------------------------------------------------------------
def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Score = 2 * n * max(conv) (sum of a is 1)."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- initialise ----------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        # fallback: simple binomial‑like series
        n = 5000
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
        s = 1.0
    a = a / s  # normalise to sum == 1

    # ---- FFT helper ----------------------------------------------------
    N = _next_pow2(2 * n - 1)

    def conv(v):
        return _conv_fft(v, N)

    def score(v, cv=None):
        if cv is None:
            cv = conv(v)
        return _score_from_conv(cv, n)

    cur_conv = conv(a)
    cur_score = score(a, cur_conv)

    best = a.copy()
    best_score = cur_score

    # ---- hyper‑parameters ------------------------------------------------
    step_factor = 0.35          # fraction of selected source mass to move
    sigma = 0.25                # log‑normal perturbation scale
    min_step_factor = 1e-4

    iteration = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ----- gradient of the active maximal shifts --------------------
        max_val = cur_conv.max()
        # indices whose convolution is very close to the max
        tol = max_val * 1e-8 + 1e-14
        active_idxs = np.where(cur_conv >= max_val - tol)[0]

        grad = np.zeros_like(a)
        for k in active_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

            rand_vec = _project_simplex(rand_vec)

                cur_conv = rand_conv
                cur_score = rand_score
                if rand_score < best_score:
                    best = rand_vec.copy()
                    best_score = rand_score

        # safety guard – avoid endless tiny steps
        if step_factor < 1e-6 and sigma < 1e-6:
            break

    # ---- final sanity check ---------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
