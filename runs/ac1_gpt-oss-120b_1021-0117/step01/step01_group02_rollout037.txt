        src = int(partner.argmax())

        else:
            dest = int(rng.choice(dest_candidates))

        if src == dest:
            # random tweak
            width = rng.integers(1, max(2, n//50))
            start = rng.integers(0, max(1, n - width + 1))
            cand = best.copy()

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) whose AC1 upper bound
    is as small as possible.  The routine first tries a few simple parametric
    families, keeps the best one, then refines it with a fast greedy/random
    local‑search that respects the evaluator’s constraints.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # timing
    # ----------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # length of the problem
    # ----------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 5000                     # fallback if no parent is supplied

    # ----------------------------------------------------------------------
    # scoring helper (identical to evaluate_sequence but without clipping)
    # ----------------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        c = np.convolve(vec, vec)
        return float(2.0 * n * c.max() / (vec.sum() ** 2))

    # ----------------------------------------------------------------------
    # start from the trivial uniform distribution (score = 2)
    # ----------------------------------------------------------------------
    best = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = _score(best)

    # ----------------------------------------------------------------------
    # 1) explore a few compact parametric families
    # ----------------------------------------------------------------------
    # (a) power‑law decay a_i ∝ (i+1)^(-p)
    for p in np.linspace(0.0, 2.0, 41):
        if time.monotonic() >= deadline:
            break
        arr = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        arr /= arr.sum()
        sc = _score(arr)
        if sc < best_score:
            best, best_score = arr, sc

    # (b) exponential decay a_i ∝ exp(-λ i)
    for lam in np.logspace(-6, -1, 20):
        if time.monotonic() >= deadline:
            break
        arr = np.exp(-lam * np.arange(n, dtype=np.float64))
        arr /= arr.sum()
        sc = _score(arr)
        if sc < best_score:
            best, best_score = arr, sc

    # (c) polynomial taper a_i ∝ (1 - i/(n-1))^q   (decreasing, q≥0)
    for q in np.linspace(0.0, 5.0, 31):
        if time.monotonic() >= deadline:
            break
        if q == 0.0:
            arr = np.ones(n, dtype=np.float64)
        else:
            arr = ((n - 1 - np.arange(n, dtype=np.float64)) / (n - 1)) ** q
        arr /= arr.sum()
        sc = _score(arr)
        if sc < best_score:
            best, best_score = arr, sc

    # ----------------------------------------------------------------------
    # 2) greedy / stochastic refinement (mass‑transfer + occasional shake)
    # ----------------------------------------------------------------------
    step_factor = 0.7          # how much of a source bin to move
    min_step = 1e-9

    # current solution (may diverge from `best` during search)
    cur = best.copy()
    cur_score = best_score

    while time.monotonic() < deadline:
        # ---- compute the convolution and its maximal index ----
        conv = np.convolve(cur, cur)
        k = int(conv.argmax())

        # ---- partner[i] = cur[k-i] where defined, else 0 ----
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(cur)
        partner[idx] = cur[k - idx]

        # ---- pick a source (largest contribution) and a destination (smallest) ----
        src = int(partner.argmax())
        min_val = float(partner.min())
        dest_candidates = np.where(partner == min_val)[0]
        if dest_candidates.size == 0:
            dest = rng.integers(0, n)
        else:
            dest = int(rng.choice(dest_candidates))

        # ---- propose a deterministic mass transfer ----
        delta = cur[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > cur[src]:
            delta = cur[src]

        cand = cur.copy()
        cand[src] -= delta
        cand[dest] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        # renormalise (keeps the L1‑norm fixed, which leaves the objective unchanged)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        cand_score = _score(cand)

        # ---- accept if it improves the current solution ----
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            # keep the global best as well
            if cand_score < best_score:
                best, best_score = cand, cand_score
            # (slightly increase step size after a success)
            step_factor = min(0.95, step_factor * 1.005)
        else:
            # decrease step size after a failure
            step_factor = max(0.02, step_factor * 0.997)

        # ---- occasional random “shake” to escape plateaus ----
        if rng.random() < 0.004:
            width = int(rng.integers(max(1, n // 250), max(2, n // 30)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            shake = cur.copy()
            shake[start:start + width] *= float(np.exp(rng.normal(0.0, 0.18)))
            shake = np.clip(shake, 0.0, 1000.0)
            shake /= shake.sum()
            shake_score = _score(shake)
            if shake_score < cur_score:
                cur, cur_score = shake, shake_score
                if shake_score < best_score:
                    best, best_score = shake, shake_score

    # ----------------------------------------------------------------------
    # final sanity check – evaluator also clips, but we enforce basics
    # ----------------------------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
