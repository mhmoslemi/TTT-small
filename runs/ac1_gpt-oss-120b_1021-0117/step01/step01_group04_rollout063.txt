def _ac1_score(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[:2*n-1]
    max_b = float(conv.max())
    s = arr.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def _gen_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1,n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except Exception:
        n = 256
        parent = None

    L = 1 << ((2*n - 2).bit_length()) # next pow2

    # compute initial candidates
    base = _gen_binomial(n)
    best = base.copy()
    best_score = _ac1_score(best, n, L)
    if parent is not None:
        pscore = _ac1_score(parent, n, L)
        if pscore < best_score:
            best = parent.copy()
            best_score = pscore

    cur = best.copy()
    cur_score = best_score

    temperature = 1.0
    decay = 0.9998

    early_end = n // 4  # early region
    late_start = (3 * n) // 4

    # define small utility functions
    while time.monotonic() < deadline:
        move_type = rng.random()
        if move_type < 0.5:
            # early->late mass transfer
            i = int(rng.integers(0, early_end))
            j = int(rng.integers(late_start, n))
            if cur[i] <= 0:
                continue
            frac = rng.random() * 0.5  # up to 50% of source moved
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        elif move_type < 0.8:
            # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.09))
            cand = cur.copy()
            cand[start:start+width] *= factor
        else:
            # general random transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * 0.3
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # clip bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _ac1_score(cand, n, L)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # break early if beat target
            if best_score < 1.50274349:
                break

        # acceptance
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score

        temperature *= decay

        # occasional restart
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    return [float(x) for x in best]

import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic search that preferentially moves mass from the early,
    high‑impact part of the sequence to the late, low‑impact part.
    This reduces the peak of the autocorrelation while keeping the
    ℓ₁‑norm (and thus the AC1 bound) essentially unchanged.
    """
    rng = np.random.default_rng(seed)
    # a few seconds reserved for cleanup / return
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----- problem size ----------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256
        parent = None

    L = _next_pow2(2 * n - 1)

    # ----- initial candidates -----------------------------------------------
    base = _gen_binomial(n)                # classic binomial construction
    best = base.copy()
    best_score = _ac1_score(best, n, L)

    if parent is not None:
        parent_score = _ac1_score(parent, n, L)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    cur = best.copy()
    cur_score = best_score

    # ----- search parameters -------------------------------------------------
    temperature = 1.0
    decay = 0.9997

    early_end = n // 4          # indices that heavily affect interior convolution
    late_start = (3 * n) // 4   # indices that mostly affect the tail

    # ----- main stochastic loop ---------------------------------------------
    while time.monotonic() < deadline:
        move = rng.random()

        if move < 0.55:            # focused mass transfer from early to late
            i = int(np.argmax(cur[:early_end]))
            j = int(np.argmin(cur[late_start:]) + late_start)
            if i == j or cur[i] <= 0.0:
                continue
            # move a modest fraction of the source mass
            frac = rng.random() * 0.25    # up to 25 %
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        elif move < 0.80:          # block scaling (soft perturbation)
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.09))
            cand = cur.copy()
            cand[start:start + width] *= factor

        else:                      # unrestricted random transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * 0.30
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # enforce evaluator bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _ac1_score(cand, n, L)

        # record global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:          # target achieved
                break

        # simulated‑annealing style acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # occasional reset to the best found so far
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    return [float(x) for x in best]
