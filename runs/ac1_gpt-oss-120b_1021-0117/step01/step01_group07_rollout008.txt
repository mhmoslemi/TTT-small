    start_time = time.monotonic()

    # Catalan
    n = None

        w = np.empty(n)
        w[0] = 1.0
        for i in range(1, n):
            w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
        return w
    # Try parent if defined
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:

        n = int(parent.size)
        cand = np.clip(parent, 0.0, 1000.0)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum
        else:
            cand = np.ones_like(cand) / len(cand)

def _catalan_weights(n: int) -> np.ndarray:
