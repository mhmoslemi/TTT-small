def autocorr(a):
    # returns convolution of a with a, length 2n-1
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    # product
    B = A * A
    conv = np.fft.irfft(B, n=size)[:2*n-1]
    return conv

def crossconv(w, a):
    # conv of w (len m) with a (len n) returning len m+n-1
    m = len(w)
    size = 1 << ((m+n-1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(W * A, n=size)[:m+n-1]
    return conv

import time
import numpy as np

def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a power‑of‑two FFT (real inputs)."""
    m = len(x) + len(y) - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    Z = X * Y
    return np.fft.irfft(Z, n=size)[:m]


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Gradient‑based flattening of the self‑convolution (step‑function heights)."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------ length & init
    try:
        n = int(len(height_sequence_1))
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:                                 # fallback if no parent
        n = 5000
        a = np.exp(-np.arange(n) / (n / 10.0))

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
        s = a.sum()
    a = a / s                                          # enforce Σa = 1

    # ------------------------------------------------------------------- scoring helpers
    b = conv_fft(a, a)                                 # current self‑convolution
    best = a.copy()
    best_score = 2.0 * n * b.max()                     # Σa = 1 ⇒ denominator = 1

    # uniform fallback (score = 2)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    bu = conv_fft(uniform, uniform)
    uniform_score = 2.0 * n * bu.max()
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = best.copy()
        b = bu

    # ------------------------------------------------------------------- optimisation parameters
    max_iter = 20000
    lam = 5.0                                          # soft‑max temperature
    lam_target = 80.0
    lam_step = (lam_target - lam) / max_iter
    lr = 0.2                                           # learning‑rate
    lr_decay = 0.9998
    min_lr = 1e-5
    no_improve = 0
    target_score = 1.50274349

        # ---- smooth max weights ------------------------------------------------
        ex = lam * b

        w = np.exp(ex - maxexp)

        if w_sum == 0.0:
