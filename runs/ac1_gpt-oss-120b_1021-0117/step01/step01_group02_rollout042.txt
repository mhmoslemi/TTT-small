import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the AC1 upper bound for a candidate sequence.
# ----------------------------------------------------------------------
def _score(seq: np.ndarray, n: int) -> float:
    """Return 2*n*max(conv(seq,seq))/(sum(seq)^2)."""
    conv = np.convolve(seq, seq)
    maxb = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return np.inf
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Helper: generate a decreasing power‑law sequence.
# ----------------------------------------------------------------------
def _power_law(n: int, p: float, lam: float) -> np.ndarray:
    """Return a non‑negative sequence a_i = 1/(i+lam)^p, normalised to sum 1."""
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / np.power(idx + lam, p)
    # normalise (the sum is always >0 for lam>0)
    a = a / a.sum()
    # clamp to the evaluator's admissible range
    return np.clip(a, 0.0, 1000.0)
