import time
import numpy as np

def _score_fft(arr: np.ndarray, n: int, fft_len: int) -> float:
    """Fast exact score: 2*n*max(conv)/sum(arr)^2."""
    fa = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    s = arr.sum()
    return float(2 * n * max_b / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a second margin for safety
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # --------------------------------------------------------------
    # 1. Determine length and initialise a feasible sequence
    try:
        n = int(len(height_sequence_1))
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s <= 0.0:
            a = np.ones(n, dtype=np.float64) / n
        else:
            a = a / s                     # normalise sum to 1
    except Exception:
        # fallback: binomial‐type start (used in the reference solution)
        n = 256
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(n - 1):
            a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
        a = a / a.sum()

    # --------------------------------------------------------------
    # 2. Pre‑compute FFT size and helper objects
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # current best solution
    cur = a.copy()
    cur_score = _score_fft(cur, n, fft_len)
    best = cur.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # 3. Main optimisation loop (gradient‑guided hill‑climbing + occasional random jiggle)
    stagnation = 0
    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1

        # compute convolution of the current iterate
        fa = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_idx = int(conv.argmax())

        # ---- gradient of the active peak (linear convolution)
        g = np.zeros_like(cur)
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1)
            g[i_range] = 2.0 * cur[max_idx - i_range]

        # ---- pick a source (large contribution) and a target (small weight)
        contrib = cur * g
        src = int(np.argmax(contrib))
        if cur[src] <= 1e-16:                     # safety fallback
            src = rng.integers(0, n)
            while cur[src] <= 1e-16:
                src = rng.integers(0, n)

        # target = index with minimal gradient weight
        tgt = int(np.argmin(g))
        if tgt == src:
            # second‑best target
            mask = np.arange(n) != src
            tgt = int(np.argmin(np.where(mask, g, np.inf)))

        # step size shrinks as time runs out
        time_left = deadline - time.monotonic()
        frac = 0.005 + 0.15 * (time_left / (budget_s - 1.0))

        if delta <= 0.0:

        cand[src] -= delta
