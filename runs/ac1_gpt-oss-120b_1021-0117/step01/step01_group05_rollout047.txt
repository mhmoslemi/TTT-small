def _score_and_conv(seq, m):
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2*len(seq)-1]
    mx = conv.max()
    s = seq.sum()
    if s <= 0: return float('inf'), conv, mx
    score = 2.0*len(seq)*mx/(s*s)
    return float(score), conv, mx

import numpy as np, time

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1) / (2*i)
    return a

def _fast_score(seq, m):
    # seq assumed 1D float, sum positive
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2*len(seq)-1]
    mx = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float('inf'), conv, mx
    score = 2.0 * len(seq) * mx / (s*s)
    return float(score), conv, mx

def _project_simplex(v):
    """Project vector v onto the probability simplex (nonnegative, sum=1)."""
    # Duchi et al. algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Ensure sum exactly 1 (might be off due to floating error)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        # fallback to uniform
        w = np.full_like(v, 1.0/len(v))
    return w
