import time
import numpy as np

def _conv_max_and_partner(v: np.ndarray, n: int, L: int):
    """Full convolution via FFT, return max value, partner vector for the max shift."""
    fv = np.fft.rfft(v, L)
    conv = np.fft.irfft(fv * fv, L)[: 2 * n - 1]
    k = int(conv.argmax())
    maxv = conv[k]

    # partner[i] = v[k-i] when indices are valid, else 0
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end + 1, dtype=np.int64)
    partner = np.zeros_like(v)
    partner[idx] = v[k - idx]
    return maxv, partner, k


def _score(v: np.ndarray, n: int, L: int) -> float:
    """Score defined by the problem (2*n*max(conv)/sum(v)^2)."""
    maxv, _, _ = _conv_max_and_partner(v, n, L)
    s = v.sum()
    if s <= 0.0:
        return float("inf")
    return float(2.0 * n * maxv / (s * s))
