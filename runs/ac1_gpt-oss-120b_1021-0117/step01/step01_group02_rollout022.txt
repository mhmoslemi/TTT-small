    sums = {0}
    current = 1
    while True:
        # candidate must be < n

import time
import numpy as np

# ----------------------------------------------------------------------
def _mian_chowla_positions(n: int, include_zero: bool = False) -> list[int]:
    """
    Generate a Sidon set (all pairwise sums distinct) inside [0, n).
    Greedy construction (Mian‑Chowla sequence).  Returns list of positions.
    """
    positions: list[int] = []
    used_sums: set[int] = set()
    if include_zero:
        positions.append(0)
        used_sums.add(0)          # sum 0+0
    cand = 1 if not include_zero else 1
    while cand < n:
        # check self‑sum
        if (cand + cand) in used_sums:
            cand += 1
            continue
        # check sums with existing positions
        conflict = False
        for p in positions:
            if (cand + p) in used_sums:
                conflict = True
                break
        if conflict:
            cand += 1
            continue
        # accept candidate
        for p in positions:
            used_sums.add(cand + p)
        used_sums.add(cand + cand)
        positions.append(cand)
        cand += 1
    return positions

# ----------------------------------------------------------------------
def _build_sequence_from_positions(n: int, pos: list[int]) -> np.ndarray:
    """
    Create a non‑negative step‑function of length n.
    Entries at the given positions are set to the maximal allowed height 1000.
    """
    seq = np.zeros(n, dtype=np.float64)
    if pos:
        seq[pos] = 1000.0
    return seq

    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # length of the problem instance
    try:

    except Exception:
        n = 5000

    # --------------------------------------------------------------
    # baseline: uniform distribution (score = 2)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform
    best_score = float('inf')
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    # --------------------------------------------------------------
    # attempt Sidon sets (two variants)
    for include_zero in (False, True):
        if time.monotonic() > deadline:
            break
        pos = _mian_chowla_positions(n, include_zero=include_zero)
        seq = _build_sequence_from_positions(n, pos)

        # safety clipping – should already satisfy limits
        seq = np.clip(seq, 0.0, 1000.0)

        # evaluate – cheap because convolution size ~2n
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:
            continue

        if score < best_score:
            best_seq = seq
            best_score = score

    # --------------------------------------------------------------

    if time.monotonic() < deadline:

        # rebuild the used‑sum set from them
        used = set()
        for i in cur_pos:
