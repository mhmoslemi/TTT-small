def simplex_proj(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def golden_section_search(f, a, b, tol):
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = f(c)
    fd = f(d)
    while b - a > tol:
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = f(d)
    # return best point
    if fc < fd:
        return c, fc
    else:
        return d, fd
```

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 5000
    # Uniform fallback
    best = np.full(n, 1.0/n, dtype=np.float64)
    best_score = 2.0  # uniform score
    # Precompute indices
    idx = np.arange(n, dtype=np.float64)
    m = 1 << ((2 * n - 1).bit_length())
    phi = (np.sqrt(5) - 1) / 2
    # define scoring function
    def score_for_lam(lam):
        # lam may be zero -> uniform
        if lam <= 0:
            a = np.full(n, 1.0/n, dtype=np.float64)
        else:
            a_raw = np.exp(-lam * idx)
            a = a_raw / a_raw.sum()
        FA = np.fft.rfft(a, m)
        conv = np.fft.irfft(FA * FA, m)
        max_b = conv[:2*n-1].max()
        return 2.0 * n * max_b, a
    # golden-section search limits
    lam_lo = 0.0
    lam_hi = 25.0 / n  # corresponds to t up to 25
    # initial evaluations
    c = lam_hi - phi * (lam_hi - lam_lo)
    d = lam_lo + phi * (lam_hi - lam_lo)
    fc, a_c = score_for_lam(c)
    fd, a_d = score_for_lam(d)
    # store best so far
    if fc < best_score:
        best, best_score = a_c, fc
    if fd < best_score:
        best, best_score = a_d, fd
    # iterate limited by deadline
    it = 0
    max_iters = 60  # enough for convergence
    while time.monotonic() < deadline and it < max_iters:
        if fc < fd:
            lam_hi = d
            d = c
            fd = fc
            a_d = a_c
            c = lam_hi - phi * (lam_hi - lam_lo)
            fc, a_c = score_for_lam(c)
        else:
            lam_lo = c
            c = d
            fc = fd
            a_c = a_d
            d = lam_lo + phi * (lam_hi - lam_lo)
            fd, a_d = score_for_lam(d)
        # update best
        if fc < best_score:
            best, best_score = a_c, fc
        if fd < best_score:
            best, best_score = a_d, fd
        it += 1
    # final safety: evaluate best using evaluate_sequence
    # already best has sum=1, but just in case.
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    return [float(v) for v in final]

import time
import numpy as np
from math import sqrt, log

def _score_lambda(lam: float, n: int, idx: np.ndarray, m: int):
    """
    For a given decay parameter lam compute the AC1 upper bound score
    and also return the normalized sequence.
    """
    if lam <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        raw = np.exp(-lam * idx)
        a = raw / raw.sum()
    # linear convolution via FFT (zero‑padded to length m ≥ 2n‑1)
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)
    max_b = conv[: 2 * n - 1].max()
    return 2.0 * n * max_b, a

    m = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
