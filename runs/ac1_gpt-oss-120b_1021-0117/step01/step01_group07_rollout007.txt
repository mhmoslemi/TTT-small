def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # load parent or fallback
    try: ...
    except: parent = np.ones(256)
    n = parent.size
    # compute Catalan
    w = np.empty(n, dtype=np.float64)
    w[0]=1.0
    for i in range(1,n):
        w[i] = w[i-1] * ((2*i)*(2*i-1))/(i*i*4.0)
    a_catalan = w / w.sum()
    # Normalize parent
    parent_norm = parent / parent.sum()
    # pick best initial
    candidates = [a_catalan, parent_norm]
    best = candidates[0]; best_score = float(evaluate_sequence(best.tolist()))
    for cand in candidates[1:]:
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best = cand; best_score = sc
    # gradient descent loop
    step_frac = 0.04
    min_step_frac = 1e-4
    # old jitter probability
    jitter_prob = 0.008
    while time.monotonic() < deadline:
        # compute convolution via FFT
        N = 1 << ((2*n - 1).bit_length())
        # compute FFT of best once per loop
        X = np.fft.rfft(best, n=N)
        conv_full = np.fft.irfft(X*X, n=N)[:2*n-1]
        # find max index
        p = int(conv_full.argmax())
        # compute gradient for max shift
        grad = np.zeros_like(best)
        lo = max(0, p - (n - 1))
        hi = min(p, n - 1)
        # indices range
        idx = np.arange(lo, hi+1)
        grad[idx] = 2 * best[p - idx]
        # Step
        if grad.max() > 0:
            max_step = best.max() / grad.max()
            step = step_frac * max_step
            cand = best - step * grad
            # enforce nonnegativity
            np.clip(cand, 0, None, out=cand)
            s = cand.sum()
            if s > 0:
                cand /= s
                sc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc
                    step_frac = min(step_frac * 1.15, 1.0)
                    if best_score <= 1.50274349:
                        break
                else:
                    step_frac = max(step_frac * 0.7, min_step_frac)
        # Random pairwise transfer occasionally
        if rng.random() < 0.02:
            i, j = rng.choice(n, size=2, replace=False)
            if best[i] > 0:
                eps = rng.random() * best[i] * 0.5
                cand = best.copy()
                cand[i] -= eps
                cand[j] += eps
                s = cand.sum()
                if s>0:
                    cand /= s
                    sc = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(sc) and sc < best_score:
                        best = cand; best_score = sc
        # jitter
        if rng.random() < jitter_prob:
            jitter = np.exp(rng.normal(0.0, 0.017, size=n))
            cand = best * jitter
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum
                sc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc
        # occasional reset: if no improvement for many loops, maybe reinitialize from parent or Catalan
        # Could track iteration count since last improvement
        # Could be added but maybe not needed

    return [float(v) for v in best]

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    Return a non‑negative step‑function (list of heights) whose AC1 bound
    is as low as possible within the allotted time.
    """

            raise ValueError

    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)

    # --------------------------------------------------------------
    # 2.  Helper: fast AC1 score (sum is assumed to be 1)
    # --------------------------------------------------------------
    N_fft = 1 << ((2 * n - 1).bit_length())

    def fast_score(vec: np.ndarray) -> float:
        """2·n·max(conv(vec,vec)) for a vector whose sum≈1."""
        X = np.fft.rfft(vec, n=N_fft)
        conv = np.fft.irfft(X * X, n=N_fft)[: 2 * n - 1]
        return 2.0 * n * float(conv.max())

    # --------------------------------------------------------------
    # 3.  Basic starts: Catalan‑type, parent, and a few parametric families
    # --------------------------------------------------------------
    # Catalan‑style (central binomial) weights
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    a_catalan = w / w.sum()

    parent_norm = parent / parent.sum()

    best = a_catalan.copy()
    best_score = fast_score(best)
    # compare with the provided parent
    parent_score = fast_score(parent_norm)
    if parent_score < best_score:

    # power‑law family a_i ∝ (i+1)^–α  (α around 0.5)
