                best = cand
                best_score = cand_score
                step_factor = max(0.02, step_factor * 0.996)   # slow decay on success
        else:
            # occasional uphill acceptance to escape plateaus
            if rng.random() < 0.001:
                cur = cand
                cur_score = cand_score

        # random “shake” – multiplicative noise on a short segment
        if rng.random() < shake_prob:
            width = max(2, int(rng.uniform(0.001, 0.02) * n))
            start = rng.integers(0, n - width + 1)
            tmp = cur.copy()
            tmp[start:start + width] *= np.exp(rng.normal(0.0, 0.22, size=width))
            tmp = np.clip(tmp, 0.0, 1000.0)
            tmp = tmp / tmp.sum()
            tmp_score = score_of(tmp, full_convolution(tmp, L), n)
            if tmp_score < best_score:
                best = tmp
                best_score = tmp_score
                cur = tmp
                cur_score = tmp_score

        # gentle cooling of the step size irrespective of success
        step_factor = max(0.005, step_factor * 0.9995)

    # ----- final safety checks ---------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
