def _ac1_score_and_conv(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = conv.max()
    s = arr.sum()
    if s == 0:
        return float('inf'), conv
    return 2.0 * n * max_b / (s * s), conv
```

import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1 (Catalan-like)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """AC1 upper bound for a candidate (FFT based)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _ac1_score_and_conv(arr: np.ndarray, n: int, L: int):
    """Return (score, full linear convolution)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf'), conv
    return 2.0 * n * max_b / (s * s), conv

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # ---- problem size ----------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256
        parent = None

    L = _next_pow2(2 * n - 1)

    # ---- initial solutions -----------------------------------------------
    base = _gen_binomial(n)
    best = base.copy()
    best_score = _ac1_score(best, n, L)

    if parent is not None:
        parent_score = _ac1_score(parent, n, L)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    cur = best.copy()
    cur_score, conv_cur = _ac1_score_and_conv(cur, n, L)

    # ---- annealing parameters --------------------------------------------
    temperature = 1.0
    decay = 0.9995

    # ---- main stochastic search -------------------------------------------
    while time.monotonic() < deadline:
        r = rng.random()
        cand = cur.copy()

        if r < 0.40:          # targeted mass transfer
            # locate lag with maximal convolution
            k_max = int(np.argmax(conv_cur))
            lag = k_max - (n - 1)
            offset = abs(lag)
            if offset == 0:
                src = int(np.argmax(cur))
            else:
                prod = cur[: n - offset] * cur[offset:]
                i_max = int(np.argmax(prod))
                idx1 = i_max
                idx2 = i_max + offset
                src = idx1 if cur[idx1] > cur[idx2] else idx2
            tgt = int(rng.integers(0, n))
            while tgt == src:
                tgt = int(rng.integers(0, n))
            frac = rng.random() * 0.5
            delta = cur[src] * frac
            cand[src] -= delta
            cand[tgt] += delta

        elif r < 0.65:        # block scaling
            w = int(rng.integers(max(1, n // 300), max(2, n // 30)))
            start = int(rng.integers(0, max(1, n - w + 1)))
            factor = math.exp(rng.normal(0.0, 0.09))
            cand[start:start + w] *= factor

        elif r < 0.90:        # random mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.5
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta

        else:                 # swap two entries
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                cand[i], cand[j] = cand[j], cand[i]

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _ac1_score(cand, n, L)

        # ---- update best ---------------------------------------------------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:            # target reached
                break

        # ---- simulated annealing acceptance --------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            cur_score, conv_cur = _ac1_score_and_conv(cur, n, L)

        # ---- temperature decay ---------------------------------------------
        temperature *= decay

        # ---- occasional restart from best ----------------------------------
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            cur_score, conv_cur = _ac1_score_and_conv(cur, n, L)
            temperature = 1.0

    return [float(x) for x in best]
