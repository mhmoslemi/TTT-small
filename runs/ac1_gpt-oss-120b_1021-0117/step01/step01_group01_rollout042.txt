def _conv_max_fft(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*a.size - 1]
    return float(conv.max())

def _score(a, n, fft_len):

    if s == 0:
        return float('inf')
    max_b = _conv_max_fft(a, fft_len)
    return 2.0 * n * max_b / (s * s)

def _project_simplex(v):
    # returns projection onto simplex {x>=0, sum(x)=1}
    # from https: adaptation
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper functions (FFT convolution, scoring, projection)
# ------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a single FFT (full length)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_full_fft(a, fft_len).max())


def _score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluator proxy (no scaling needed, sum may differ from 1)."""
    s = a.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max_fft(a, fft_len)
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_at_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of the active convolution term (2 * shifted copy of a)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if k_max < n:

    else:
        start = k_max - (n - 1)
        # overlap length = n - start
        g[start:] = 2.0 * a[n - 1 : start - 1 : -1]
    return g
