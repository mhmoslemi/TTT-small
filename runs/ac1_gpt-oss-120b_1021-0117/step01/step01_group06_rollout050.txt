def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of a where a*a = 1+…+x^{2n-2} (all ones on the diagonal)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _max_conv(v: np.ndarray) -> float:
    """Maximum entry of the full (linear) convolution v*v using FFT."""
    n = v.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(conv.max())


def _score(v: np.ndarray) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = v.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (v.sum() ** 2))
