def propose_candidate(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.1  # leaves little margin

    try:
        n = len(height_sequence_1)
        init = height_sequence_1.copy()
    except NameError:
        n = 256
        init = np.ones(n, dtype=float)

    # Ensure nonnegative within bounds, clip
    init = np.clip(init, 0, 1000)
    s = init.sum()
    if s == 0:
        init = np.ones(n)/n
        s = 1.0
    # Normalize sum to e.g., 1 (any scaling won't affect objective)
    init = init / s

    best = init.copy()
    best_score = _score_fft(best, n, fft_len)

    cur = best.copy()
    cur_score = best_score

    # Pre-compute FFT length
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len-1).bit_length())

    # Temperature schedule
    T0 = 0.1
    T_end = 0.0005
    total_iters = 0
    # We can compute theoretical total iterations allowed based on time.
    # We'll adjust temperature linearly based on elapsed ratio.

    while time.monotonic() < deadline:
        # simple acceptance probability
        # Choose i,j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1

        if cur[i] <= 0:
            continue
        # choose fraction move of cur[i]
        frac = rng.uniform(0.01, 0.3)  # moderate
        delta = cur[i] * frac
        new_val_i = cur[i] - delta
        new_val_j = cur[j] + delta
        if new_val_i < 0:
            continue
        new = cur.copy()
        new[i] = new_val_i
        new[j] = new_val_j

        # optional: keep other entries unchanged
        # Compute score
        new_score = _score_fft(new, n, fft_len)
        # determine acceptance
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        else:
            # compute temperature based on elapsed fraction
            elapsed = (time.monotonic() - start) / (deadline - start)
            T = T0 * (1 - elapsed) + T_end * elapsed
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur = new
                cur_score = new_score

    final = [float(x) for x in best]
    # ensure compliance
    try:
        _ = evaluate_sequence(final)
        return final
    except Exception:
        # fallback
        return [float(x) for x in init]

import time
import numpy as np

def _score_fft(arr: np.ndarray, n: int, fft_len: int) -> float:
    """Fast proxy for evaluate_sequence (identical for non‑negative inputs)."""
    conv_len = 2 * n - 1
    fa = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    maxb = conv.max()
    s = arr.sum()
    return float(2.0 * n * maxb / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 upper bound.
    The function respects the supplied time budget and uses a simulated‑annealing
    hill‑climber seeded from the provided `height_sequence_1` when available.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)   # keep a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Obtain a starting sequence
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        cur = np.array(height_sequence_1, dtype=np.float64)
    except Exception:                     # fallback if parent not supplied
        n = 256
        cur = np.ones(n, dtype=np.float64)

    # Clip to evaluator's hard limits and normalise the L¹‑norm (scaling does not affect the score)
    np.clip(cur, 0.0, 1000.0, out=cur)
    tot = cur.sum()
    if tot <= 0.0:
        cur.fill(1.0 / n)
        tot = cur.sum()
    cur /= tot                               # now sum(cur) == 1.0

    # ------------------------------------------------------------------
    # 2. Pre‑compute FFT layout
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 3. Scoring helpers
    # ------------------------------------------------------------------
    cur_score = _score_fft(cur, n, fft_len)
    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing loop (mass‑transfer moves)
    # ------------------------------------------------------------------
    # Temperature schedule (scaled to the current score)
    T0 = cur_score * 0.05
    T_end = cur_score * 0.001
    it = 0

    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        # --- choose a source index (must have some mass) and a distinct target ---
        i = rng.integers(n)
        if cur[i] <= 1e-14:

        j = rng.integers(n - 1)
        if j >= i:

        progress = (now - start) / (deadline - start)
        max_frac = 0.2 * (1.0 - progress) + 0.02   # decays from ~0.22 to 0.02
