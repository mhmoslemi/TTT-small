import time

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4^i sequence, decreasing, positive."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a

def _score_fft(arr: np.ndarray, n: int, fft_len: int) -> float:
    """Fast proxy for evaluate_sequence (exact for non‑negative data)."""
    fa = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s == 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)
