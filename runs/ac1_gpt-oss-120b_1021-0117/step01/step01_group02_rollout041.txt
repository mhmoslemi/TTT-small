    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

            return np.inf
    sequence = [float(x) for x in sequence]

    # linear convolution via FFT, returns (max_value, argmax_index)

import time
import numpy as np

def _conv_max(arr: np.ndarray, fft_size: int):
    """Linear convolution of arr with itself via FFT.
    Returns (maximum value, index of that maximum)."""
    A = np.fft.rfft(arr, n=fft_size)
    B = A * A
    conv = np.fft.irfft(B, n=fft_size)
    n = arr.shape[0]
    valid_len = 2 * n - 1
    c = conv[:valid_len]
    return float(c.max()), int(c.argmax())

def _score_from_max(max_val: float, n: int) -> float:
    """Score = 2 * n * max(conv) (sum is normalized to 1)."""
    return 2.0 * n * max_val

    """Search for a non‑negative step‑function that lowers the AC1 bound."""
