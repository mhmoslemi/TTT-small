import time, numpy as np

def _conv_max_fft(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2 * a.size - 1]
    return float(conv.max())

def _score_fft(a, n, fft_len):
    max_b = _conv_max_fft(a, fft_len)
    s = float(a.sum())
    return 2.0 * n * max_b / (s * s)

def _generate_binomial_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a

def _make_powerlaw_seq(n, p, beta, reverse=False):
    i = np.arange(n, dtype=np.float64)
    if reverse:
        idx = n - 1 - i
    else:
        idx = i
    raw = 1.0 / np.power(idx + beta, p)
    total = raw.sum()
    if total <= 0:
        # fallback uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / total

import time
import numpy as np

def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of linear convolution a*a using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def _score_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast proxy for the evaluator (identical for non‑negative inputs)."""
    max_b = _conv_max_fft(a, fft_len)
    s = float(a.sum())
    return 2.0 * n * max_b / (s * s)


def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4**i via recurrence; all entries positive and decreasing."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a

    """a_i ∝ 1/(i+β)^p (or reversed), then normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    idx = n - 1 - i if reverse else i
    raw = 1.0 / np.power(idx + beta, p)
    tot = raw.sum()
    if tot <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / tot


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a short margin for the harness
    deadline = start + max(0.0, float(budget_s) - 2.0)

    # ---- length and initial sequence ---------------------------------
    try:
        n = int(len(height_sequence_1))
        cur = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 256
        cur = _generate_binomial_seq(n)

    cur = np.clip(cur, 0.0, 1000.0)
    s = cur.sum()
    cur = cur / s if s > 0.0 else np.full(n, 1.0 / n, dtype=np.float64)

    # ---- FFT layout --------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ---- baseline score -----------------------------------------------
    cur_score = _score_fft(cur, n, fft_len)
    best = cur.copy()
    best_score = cur_score

    # ---- 1) grid search over simple power‑law families -----------------
    p_vals = np.linspace(0.3, 1.5, 13)          # 13 values
    beta_vals = np.linspace(0.0, 5.0, 11)       # 11 values
    for p in p_vals:
        for beta in beta_vals:
            if time.monotonic() >= deadline:
                break
            for rev in (False, True):
                seq = _make_powerlaw_seq(n, p, beta + 1e-12, reverse=rev)
                seq = np.clip(seq, 0.0, 1000.0)
                seq = seq / seq.sum()
                sc = _score_fft(seq, n, fft_len)
                if sc < best_score:
                    best = seq.copy()
                    best_score = sc
                    cur = seq.copy()
                    cur_score = sc

    # ---- 2) simulated‑annealing hill climbing -------------------------
    init_temp = 0.02                     # initial temperature
    total_time = max(0.0, deadline - start)
    last_restart = time.monotonic()
    restart_interval = 30.0             # seconds between random restarts

    while time.monotonic() < deadline:
        now = time.monotonic()

        # periodic random restart to escape possible traps
        if now - last_restart > restart_interval:
            raw = rng.exponential(scale=1.0, size=n)
            cur = raw / raw.sum()
            cur_score = _score_fft(cur, n, fft_len)
            last_restart = now

        # propose moving a fraction of mass from i to j (i≠j)
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cur[i] < 1e-12:
            continue
        frac = rng.uniform(0.001, 0.5)
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_score = _score_fft(new, n, fft_len)
        delta_score = new_score - cur_score

        # linearly decaying temperature
        remaining = deadline - now
        temp = init_temp * (remaining / total_time) if total_time > 0 else 0.0

        # Metropolis acceptance
        if delta_score < 0.0 or rng.random() < np.exp(-delta_score / (temp + 1e-15)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

    # ---- final conversion ---------------------------------------------
    final = [float(x) for x in best]
    # sanity check (the evaluator may raise on pathological inputs)
    try:
        _ = evaluate_sequence(final)
    except Exception:
        fallback = _generate_binomial_seq(n)
        fallback = fallback / fallback.sum()
        return [float(x) for x in fallback]

    return final
