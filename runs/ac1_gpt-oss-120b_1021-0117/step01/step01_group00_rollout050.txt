def gradient_step(a, lr):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return a
    # compute convolution via FFT
    Nfft = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, Nfft)
    conv = np.fft.irfft(A*A, Nfft)[:2*n-1]
    M = conv.max()
    k_star = conv.argmax()
    # compute gradM vector
    gradM = np.zeros_like(a)
    i_min = max(0, k_star - (n-1))
    i_max = min(n-1, k_star)
    idx = np.arange(i_min, i_max+1)
    gradM[idx] = 2 * a[k_star - idx]
    # full gradient
    scalar = 2 * n / (S**2)
    grad = scalar * (gradM - 2 * M / S)
    # update
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

def fast_score(a):
    S = a.sum()
    if S < 1e-12:
        return np.inf
    n = a.shape[0]
    Nfft = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, Nfft)
    conv = np.fft.irfft(A * A, Nfft)[:2*n - 1]
    M = conv.max()
    return 2.0 * n * M / (S * S)
```

import numpy as np

# ---------- helper functions ----------
def _fast_score(a: np.ndarray, Nfft: int) -> float:
    """Return the AC1 bound using an FFT‑based convolution (fast)."""
    n = a.shape[0]
    A = np.fft.rfft(a, Nfft)
    conv = np.fft.irfft(A * A, Nfft)[: 2 * n - 1]
    M = conv.max()
    S = a.sum()
    if S <= 0.0:
        return float("inf")
    return 2.0 * n * M / (S * S)


def _gradient_step(a: np.ndarray, lr: float, Nfft: int):
    """One projected gradient step for the AC1 bound."""
    n = a.shape[0]
    A = np.fft.rfft(a, Nfft)
