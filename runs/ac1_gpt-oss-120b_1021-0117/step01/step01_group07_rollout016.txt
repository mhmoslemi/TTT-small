        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        parent = np.clip(parent, 0, 1000)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    # normalized candidates
    a_parent = parent.copy()
    a_parent = np.clip(a_parent, 0, None)
    a_parent /= a_parent.sum()
    # Catalan
    def catalan_weights(n):
        w = np.empty(n, dtype=np.float64)
        w[0] = 1.0
        for i in range(1, n):
            w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
        return w
    a_catalan = catalan_weights(n)
    a_catalan = a_catalan / a_catalan.sum()
    # choose best start
    def score_of(arr):
        s = arr.sum()
        if s <= 0:
            return np.inf
        b = np.convolve(arr, arr)
        maxb = b.max()
        return 2.0 * n * maxb / (s*s)
    best = a_parent.copy()
    best_score = score_of(best)
    cand2_score = score_of(a_catalan)
    if cand2_score < best_score:

        best_score = cand2_score

    best_fallback = best.copy()
    best_fallback_score = best_score

    min_step_factor = 1e-5
    tolerance = 1e-12

    # main loop
    iteration = 0
    while time.monotonic() < deadline:

        a = best

        b = np.convolve(a, a)

    # returns projection onto simplex {x>=0, sum=1}

import time
import numpy as np

def _project_simplex(v):
    # project onto probability simplex (sum=1, v>=0)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all nonpositive, project to uniform?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _score(a):
    # compute score
    s = a.sum()
    if s <= 0:
        return np.inf
    b = np.convolve(a, a)
    maxb = b.max()
    return 2.0 * a.size * maxb / (s*s)

    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # Generate candidate initial shapes
    init_candidates = []

    # parent normalized
    a_parent = parent.copy()
    if a_parent.sum() == 0:
        a_parent = np.ones_like(a_parent)
    a_parent = a_parent / a_parent.sum()
    init_candidates.append(a_parent)

    # Catalan weights
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
    a_catalan = w / w.sum()
    init_candidates.append(a_catalan)

    # Uniform
    init_candidates.append(np.ones(n, dtype=np.float64) / n)

    # Linear ramp up
    ramp_up = np.arange(1, n+1, dtype=np.float64)
    ramp_up = ramp_up / ramp_up.sum()
    init_candidates.append(ramp_up)

    ramp_down = ramp_up[::-1]
    init_candidates.append(ramp_down)

    # Symmetric triangle

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find rho = max{i | u_i - cssv_i/(i+1) > 0}
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho_candidates = np.where(rho_arr > 0)[0]
    if rho_candidates.size == 0:
        # all entries non‑positive → return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho_candidates[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Exact AC1 upper bound for a (no clipping – the evaluator will clip)."""
    s = a.sum()
    if s <= 0.0:
        return np.inf
    b = np.convolve(a, a)          # linear convolution, length 2n‑1
    return 2.0 * a.size * b.max() / (s * s)
