# Analytic √‑series (used when no parent is supplied)

def _sqrt_series(n: int) -> np.ndarray:

    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0

        while tgt == src:
            tgt = rng.integers(0, n - 1)

        # decide transfer size
        if rng.random() < BIG_MOVE_PROB:
            frac = rng.random() * 0.8 + 0.2   # move 20‑80 % of src mass
        else:
            frac = rng.random() * 0.25        # move up to 25 % of src mass
        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # numerical safety (tiny negatives can appear due to floating point)
        cand[src] = max(cand[src], 0.0)

        # renormalise – avoids drift and satisfies the evaluator's sum‑check
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        # ----------------------------------------------------------
        # Evaluate candidate
        # ----------------------------------------------------------
        conv_cand, cand_sc = _conv_score(cand, L)

        # ----------------------------------------------------------
        # Simulated‑annealing acceptance test
        # ----------------------------------------------------------
        elapsed = time.monotonic() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        accept = False
        if cand_sc < cur_sc:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(cand_sc - cur_sc) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_sc = cand_sc
            conv = conv_cand
            if cand_sc < best_sc:
                best_sc = cand_sc
                best = cand.copy()

        # ----------------------------------------------------------
        # Occasional global jitter to escape plateaus
        # ----------------------------------------------------------
        if it % 2000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            cand2 = cur + noise
            cand2 = np.maximum(cand2, 0.0)
            s2 = cand2.sum()
            if s2 > 0.0:
                cand2 /= s2
                conv2, cand2_sc = _conv_score(cand2, L)
                if cand2_sc < best_sc:
                    best_sc = cand2_sc
                    best = cand2.copy()
                # optional acceptance as current state
                if cand2_sc < cur_sc or (T > 0.0 and rng.random() < np.exp(-(cand2_sc - cur_sc) / T)):
                    cur = cand2
                    cur_sc = cand2_sc
                    conv = conv2

    # --------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------
    return [float(x) for x in best]
