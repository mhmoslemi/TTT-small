import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Hill‑climbing search for a non‑negative step‑function that minimises the
    AC1 upper bound defined by ``evaluate_sequence``.
    The algorithm starts from the better of the provided parent (if any) and a
    Catalan‑style sequence, then repeatedly transfers mass from the largest
    coefficient towards the smallest while keeping the total sum equal to one.
    All updates to the convolution are performed analytically in O(n) time.
    """
    rng = np.random.default_rng(seed)
    # safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    target_score = 1.50274349

    # ----------------------------------------------------------------------
    # 1.   Base sequence (parent or Catalan)
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)

    # normalise to sum = 1  (objective is scale‑invariant)
    a = parent.astype(np.float64)
    s = a.sum()
    if s < 0.01:
        a = np.ones_like(a) / n
    else:
        a /= s

    # Catalan‑style construction
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    cat = w / w.sum()

    # pick the better of the two starts
    score_a = float(evaluate_sequence(a.tolist()))
    score_cat = float(evaluate_sequence(cat.tolist()))
    if score_cat < score_a:
        a = cat
        best_score = score_cat
    else:
        best_score = score_a

    best = a.copy()
    conv = np.convolve(best, best)            # length 2n‑1
    cur_max = conv.max()                     # max of a*a

    # ----------------------------------------------------------------------
    # 2.   Hill‑climbing loop (mass transfer)
    # ----------------------------------------------------------------------
    move_frac = 0.9          # start with aggressive transfers
    min_move_frac = 1e-5
    stagnant = 0             # iterations without improvement

    while time.monotonic() < deadline:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best = best.copy()
            best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1

        # early exit if target reached
        if cur_score <= target_score:
            break

        # ------------------------------------------------------------------
        # guided move: pull mass from the largest coefficient towards the
        # smallest (most beneficial for reducing the convolution peak)
        # ------------------------------------------------------------------
        if rng.random() < 0.8:
            src = int(np.argmax(best))
            tgt = int(np.argmin(best))
            if src == tgt:
                src, tgt = rng.choice(n, size=2, replace=False)
        else:
            src, tgt = rng.choice(n, size=2, replace=False)

        if best[src] <= 0.0:
            continue

        # propose a transfer – fraction of src's mass
        delta = best[src] * move_frac * rng.random()
        if delta <= 0.0:
            continue

        # exact update of the convolution (linear in delta)
        delta_vec = 2.0 * delta * best
        cand_conv = conv.copy()
        cand_conv[src:src + n] -= delta_vec
        cand_conv[tgt:tgt + n] += delta_vec
        new_max = cand_conv.max()

        if new_max + 1e-14 < cur_max:          # improvement
            best[src] -= delta
            best[tgt] += delta
            conv = cand_conv
            cur_max = new_max
            move_frac = min(move_frac * 1.05, 0.9)
            stagnant = 0
        else:

                cand_score = 2.0 * n * cand_conv.max()

                    move_frac = min(move_frac * 1.2, 0.9)
