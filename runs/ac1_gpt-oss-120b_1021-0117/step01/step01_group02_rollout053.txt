def convolve_self(a):
    """Linear convolution a*a using FFT, returns length 2*len(a)-1 array."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def compute_score(a):
    """Return (score, conv) for sequence a (sum may be arbitrary)."""

    # Guard against zero sum (should not happen)
    if s == 0:
        return float('inf'), conv
    score = 2.0 * a.shape[0] * max_b / (s * s)
    return score, conv

            best_a = a.copy()
            best_score = cur_score
            stagnation = 0
        else:
            stagnation += 1

        # occasional global jitter
        if rng.random() < 0.001:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            cand = a.copy()
            cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.15)))
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand / cand.sum()
            cand_score, cand_conv = compute_score(cand)
            if cand_score < best_score:
                a, conv, best_a, best_score = cand, cand_conv, cand, cand_score

        if stagnation > 4000:
            # give up after a long dead‑end
            break

    final = np.clip(best_a, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    return [float(v) for v in final]
