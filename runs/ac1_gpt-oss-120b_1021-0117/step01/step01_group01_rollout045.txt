import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1, decreasing positive sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _prepare_initial(n, parent_seq):
    """Return a normalized non‑negative start vector of length n."""
    if parent_seq is not None:
        seq = np.array(parent_seq, dtype=np.float64)
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total > 0:
            return seq / total
    # fallback: binomial shape
    a = _generate_binomial_seq(n)
    return a / a.sum()


def _conv_self_max(a, fft_len, conv_len):
    """Maximum entry of a*a via FFT, also returns full convolution."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(conv.max()), conv


def _conv_cross_max(a, b, fft_len, conv_len):
    """Maximum entry of a*b via FFT, also returns full convolution."""
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)[:conv_len]
    return float(conv.max()), conv


def _solve_feasible_g(f, M, init_active, n, fft_len, conv_len, max_expand=8):
    """Linear program: maximise sum(g) s.t. g>=0 and f*g <= M on active set.
    Returns a feasible g (sum(g)=1) or None if LP fails or cannot be made feasible."""
    active = list(init_active)
    # keep a set for quick membership test
    active_set = set(active)

    for _ in range(max_expand):
        # --- build sparse matrix for active constraints -----------------
        rows, cols, data = [], [], []
        for r, k in enumerate(active):
            start_j = max(0, k - n + 1)
            end_j = min(k, n - 1)
            if start_j > end_j:
                continue
            j_range = np.arange(start_j, end_j + 1, dtype=np.int64)
            coeffs = f[k - j_range]                     # f_{k-j}
            rows.extend([r] * len(j_range))
            cols.extend(j_range.tolist())
            data.extend(coeffs.tolist())
        if not rows:               # no constraints -> any non‑negative g works
            g = np.ones(n, dtype=np.float64) / n
            return g
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b_ub = np.full(len(active), M, dtype=np.float64)

        # --- solve LP (maximise sum(g) == minimise -sum(g)) ----------
        c = -np.ones(n, dtype=np.float64)
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs', options={'presolve': True})
        if not res.success:
            return None
        g = res.x
        s = g.sum()
        if s <= 0:
            return None
        g /= s                                   # normalise to sum 1

        # --- verify all constraints, add violators if needed ------------
        _, conv_fg = _conv_cross_max(f, g, fft_len, conv_len)
        viol = np.where(conv_fg > M * (1 + 1e-8))[0]
        if viol.size == 0:
            return g
        # add a few new violators
        added = 0
        for vi in viol:
            if vi not in active_set:
                active.append(int(vi))
                active_set.add(int(vi))
                added += 1
                if added >= 4:
                    break
        # continue loop, rebuilding matrix with enlarged active set
    return None   # gave up after expansions
