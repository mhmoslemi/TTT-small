import numpy as np, time, math

def _conv_self(a):
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n - 1]
    return conv

def _score(a):
    # compute same as evaluator (but no clipping maybe)
    n = a.size
    conv = _conv_self(a)
    maxb = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * maxb / (sum_a * sum_a)
