def proj_simplex(v):
    # ensure sum of positive entries =1 and v_i>=0.
    # Using standard algorithm: sort descending, compute rho, theta.
    # Return np.maximum(v - theta, 0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def proj_simplex(v):

    # if sum(v) == 0 then return uniform.

import time, numpy as np

def _proj_simplex(v):
    # from [Duchi et al. 2008]
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0 gives uniform distribution
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1] - 1  # index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: gradient of the maximal convolution coefficient
# ----------------------------------------------------------------------
def _maxconv_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i
    # each pair (i,j) contributes a[j] to g[i] and a[i] to g[j]
    g[i] += a[j]
    g[j] += a[i]
    return g


# ----------------------------------------------------------------------
# Helper: build the (2n‑1)×n convolution matrix for a fixed a
# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Sparse matrix C so that C @ x = a * x (full convolution)."""
    n = a.shape[0]
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    data = np.repeat(a, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) with a low
    AC1 upper bound.  The routine starts from the supplied parent
    (height_sequence_1) when available, otherwise from Catalan‑type
    weights, and then performs a projected sub‑gradient descent combined
    with occasional stochastic moves.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise from the parent if possible
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0:
            raise ValueError
        init = np.clip(init, 0.0, 1000.0)
    except Exception:
        # Catalan weights as a generic fallback (length 256)
        n_fallback = 256
        w = np.empty(n_fallback, dtype=np.float64)
        w[0] = 1.0
        for i in range(1, n_fallback):
            w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
        init = w

    # normalise to a probability vector
    if init.sum() < 1e-12:
        init = np.ones_like(init) / init.size
    a = init / init.sum()
    n = a.shape[0]

    # best solution seen so far
    best = a.copy()
    best_max = np.max(np.convolve(best, best))
    best_obj = 2.0 * n * best_max

    # target (derived from the required bound)
    target_obj = 1.50274349
    target_max = target_obj / (2.0 * n)

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop
    # ------------------------------------------------------------------
    step = 0.5                      # initial step scale for sub‑gradient
    min_step = 1e-5
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        max_val = conv.max()
        idx = int(np.argmax(conv))

        # early exit if we have already passed the target
        if max_val <= target_max:
            best = a.copy()
            best_max = max_val
            break

        # sub‑gradient step
        grad = _maxconv_gradient(a, idx)
        step_sz = step / np.sqrt(it + 1.0)
        cand = _proj_simplex(a - step_sz * grad)

        cand_max = np.max(np.convolve(cand, cand))
        if cand_max + 1e-15 < best_max:
            best = cand.copy()
            best_max = cand_max
            a = cand
            step = min(step * 1.05, 2.0)          # a little acceleration
        else:
            step = max(step * 0.7, min_step)

        # --------------------------------------------------------------
        # Occasional stochastic jitter to escape plateaux
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.018, size=n))
            cand_j = _proj_simplex(a * jitter)
            cand_j_max = np.max(np.convolve(cand_j, cand_j))
            if cand_j_max < best_max:
                a = cand_j
                best = cand_j
                best_max = cand_j_max

        # --------------------------------------------------------------
        # Small random mass transfer (mass‑conserving)
        # --------------------------------------------------------------
        if rng.random() < 0.03:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                eps = rng.random() * a[i] * 0.25
                cand_t = a.copy()
                cand_t[i] -= eps
                cand_t[j] += eps
                cand_t = _proj_simplex(cand_t)
                cand_t_max = np.max(np.convolve(cand_t, cand_t))
                if cand_t_max < best_max:
                    a = cand_t
                    best = cand_t
                    best_max = cand_t_max

        # --------------------------------------------------------------
        # Periodic linear‑programming refinement (very cheap, few calls)
        # --------------------------------------------------------------
        if it % 50 == 0 and time.monotonic() < deadline - 0.5:
            # Build the convolution matrix for the current a
            C = _conv_matrix(a)
            M = max_val
            g = cp.Variable(n, nonneg=True)
            t = cp.Variable(nonneg=True)
            constraints = [C @ g <= M, cp.sum(g) == 1]   # keep sum(g)=1 for scaling
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
                if prob.status in [cp.OPTIMAL, cp.FEASIBLE] and g.value is not None:
                    g_val = np.clip(g.value, 0.0, None)
                    # mix a little bit toward the LP solution
                    mix = 0.12
                    cand_lp = _proj_simplex((1 - mix) * a + mix * g_val)
                    cand_lp_max = np.max(np.convolve(cand_lp, cand_lp))
                    if cand_lp_max < best_max:
                        a = cand_lp
                        best = cand_lp
                        best_max = cand_lp_max
            except Exception:
                pass  # ignore LP failures – continue with sub‑gradients

        it += 1
        # safety break when time is almost exhausted
        if time.monotonic() > deadline - 0.2:
            break

    # ------------------------------------------------------------------
    # 3.  Return the best solution found
    # ------------------------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    s = final.sum()
    if s < 1e-8:
        final = np.full_like(final, 1.0 / final.size)
    else:
        final = final / s                     # keep the sum = 1 (scale‑invariant)
    return [float(v) for v in final]
