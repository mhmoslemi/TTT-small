    shift = np.zeros_like(seq)

def _fast_score(seq, m):
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*len(seq)-1]
    max_val = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a == 0:
        return float('inf'), conv, max_val
    score = 2.0 * len(seq) * max_val / (sum_a * sum_a)
    return float(score), conv, max_val

import numpy as np

# ------------------------------------------------------------
# Helper functions (top level, no closures)
# ------------------------------------------------------------

def _initial_binomial(n: int) -> np.ndarray:
    """Monotonically decreasing binomial seed a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray, m: int = None):
    """Return (score, full convolution, max value) using FFT."""
    n = seq.shape[0]
    if m is None:
        m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val

            accept = True
        else:
            # probability of accepting a worse move
            if temperature > 0.0:
                prob = np.exp((best_score - cand_score) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            best = cand
            best_score = cand_score
            best_conv = cand_conv

        # ----------------------------------------------------------
        # Temperature schedule
        # ----------------------------------------------------------
        temperature *= 0.99995
        if best_score <= min_target:
            break   # we have already beaten the target

    # ----------------------------------------------------------------
    # Optional deterministic refinement: a few pure gradient steps
    # ----------------------------------------------------------------
    for _ in range(120):
        k_max = int(np.argmax(best_conv))
        max_c = best_conv[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idxs = np.arange(lo, hi + 1)
        shift = np.zeros_like(best)
        shift[idxs] = best[k_max - idxs]
        eta = 0.03 * (0.97 ** _)
        cand = best - eta * (shift - max_c * best)
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_sum = np.sum(cand)
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        cand_score, cand_conv, _ = _fast_score(cand, fft_len)
        if cand_score < best_score:
            best, best_score, best_conv = cand, cand_score, cand_conv

    # ----------------------------------------------------------------
    # Return a plain Python list – the evaluator will re‑evaluate it.
    # ----------------------------------------------------------------
    return [float(x) for x in best]
