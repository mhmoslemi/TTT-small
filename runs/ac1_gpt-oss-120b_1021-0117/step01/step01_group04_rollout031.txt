def _next_pow2(x: int) -> int:

    return 1 << (x - 1).bit_length()


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """
    Fast evaluation of the AC1 bound using an FFT.
    Returns 2*n*max(conv)/sum(arr)**2.
    """
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[: 2 * n - 1]               # relevant part
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)
