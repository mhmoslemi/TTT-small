```
def _fast_score(seq):
    n = seq.shape[0]
    m = fft_len # global variable compute externally.
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_val = conv.max()
    sum_a = seq.sum()
    if sum_a <= 0:
        return float('inf'), conv, max_val
    score = 2.0 * n * max_val / (sum_a * sum_a)
    return float(score), conv, max_val
```

            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        elif op == 4:                                 # high‑to‑low redistribution
            total = cand.sum()
            if total > 0:
                probs = cand / (total + 1e-30)
                hi = rng.choice(n, p=probs)
                low_mask = cand < np.percentile(cand, 10)
                low_idxs = np.where(low_mask)[0]
                if low_idxs.size:
                    lo = rng.choice(low_idxs)
                    frac = rng.uniform(0.05, 0.2)
                    move = cand[hi] * frac
                    cand[hi] -= move
                    cand[lo] += move

        else:                                          # far‑shift mass
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if abs(i - j) > n // 4:
                frac = rng.uniform(0.05, 0.15)
                move = cand[i] * frac
                cand[i] -= move
                cand[j] += move

        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score, cand_conv, _ = _fast_score(cand, n, m)
        delta = cand_score - current_score
        accept = False
        if delta <= 0:
            accept = True
        elif temp > 0 and rng.random() < np.exp(-delta / temp):
            accept = True

        if accept:
            current = cand
            current_score = cand_score
            current_conv = cand_conv
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                best_conv = cand_conv

        temp = max(end_temp, temp * decay)
        it += 1
        if it % 20000 == 0:
            current = best.copy()
            current_score = best_score
            current_conv = best_conv

    # ---------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ---------------------------------------------------------
    return [float(v) for v in best]
