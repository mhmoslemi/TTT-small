def _next_pow2(x: int) -> int:
    """Small utility to get the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (length 2|a|-1)."""
    n = a.shape[0]
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
