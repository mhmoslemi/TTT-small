def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine n
    try:
        n = int(len(height_sequence_1))
        init = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 5000
        # initialize as binomial distribution
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (1.0 - 1.0 / (2.0*i))
        init = a

    # normalize to sum 1
    if init.sum() == 0:
        init = np.full(n, 1.0/n, dtype=np.float64)
    a = init / init.sum()
    # ensure nonnegative, clip
    a = np.clip(a, 0.0, 1000.0)

    best = a.copy()
    best_score = 2.0 * n * np.convolve(best, best).max()  # sum=1

    # fallback uniform if needed
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    uniform_score = 2.0 * n * np.convolve(uniform, uniform).max()
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # Precompute FFT size
    L = 2*n - 1
    M = 3*n - 2
    N = 1 << int(math.ceil(math.log2(M)))  # power of two >= M
    # hyperparameters
    alpha = 30.0  # smoothmax hardness
    lr = 0.5      # initial learning rate
    min_lr = 1e-4
    no_improve = 0
    max_no_improve = 200  # after which we random perturb

    # main loop
    while time.monotonic() < deadline:
        # compute self-convolution via FFT
        a_fft = np.fft.rfft(a, n=N)
        conv_fft = a_fft * a_fft
        conv_full = np.fft.irfft(conv_fft, n=N)[:L]
        maxc = conv_full.max()
        # evaluate current score
        cur_score = 2.0 * n * maxc
        # compute smooth weights
        # shift for stability
        exp_arg = alpha * (conv_full - maxc)  # max zero -> exp <=1
        # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to uniform weights
            weight = np.full_like(w, 1.0 / w.size)
        else:
            weight = w / w_sum

        # gradient via correlation
        a_rev = a[::-1]  # reversed
        weight_fft = np.fft.rfft(weight, n=N)
        a_rev_fft = np.fft.rfft(a_rev, n=N)

        cand = a - lr * grad
        cand = np.maximum(cand, 0.0)

            cand = np.full_like(a, 1.0/n)
            s = 1.0
        cand = cand / s  # re‑normalize to sum 1

        cand_fft = np.fft.rfft(cand, n=N)
        conv_cand_fft = cand_fft * cand_fft
        conv_cand = np.fft.irfft(conv_cand_fft, n=N)[:L]
        cand_score = 2.0 * n * conv_cand.max()

        if cand_score < best_score:
            best = cand
            best_score = cand_score
            a = cand  # continue from the improvement
            # optionally increase learning rate modestly
            lr = min(1.0, lr * 1.05)
            no_improve = 0
        else:

                best_score = cand2_score
                a = cand2

    if final.sum() < 0.01:

    return [float(v) for v in final]
