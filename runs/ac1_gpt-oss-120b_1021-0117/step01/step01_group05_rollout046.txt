def improve_with_lp(f, max_iter=10, t=0.1, budget_deadline):
    best_f = f
    best_score = evaluate_sequence(f.tolist())
    for iteration in range(max_iter):
        if time.monotonic() > deadline: break
        # compute convolution
        conv_f = np.convolve(best_f, best_f)
        M = conv_f.max()
        # solve LP
        g = solve_lp(f=best_f, M=M)
        if g is None:
            break
        sum_f = best_f.sum()
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # scale g to same sum
        g_scaled = g * (sum_f / sum_g)
        # convex combination
        cand = (1 - t) * best_f + t * g_scaled
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_f = cand
            best_score = cand_score
        else:
            # maybe reduce t or break
            t *= 0.5
            if t < 1e-4:
                break
    return best_f, best_score
```

def solve_lp(f, M):
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    # define conv using cp.Constant
    f_const = cp.Constant(f)
    conv = cp.conv(f_const, g)
    constraints = [conv <= M]
    # sum(g) objective maximize to allow addition
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return np.array(g.value, dtype=np.float64)
    else:
        return None
```

def _move_from_max_pair(seq, conv, rng):
    n = seq.shape[0]
    k_max = int(np.argmax(conv))
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    i = rng.integers(lo, hi + 1)
    j = k_max - i
    if i == j:
        return seq  # can't move from diagonal
    # decide fraction to move
    alpha = rng.uniform(0.05, 0.12)
    # ensure not exceed values
    if seq[i] <= 0 or seq[j] <= 0:
        return seq
    removed_i = seq[i] * alpha
    removed_j = seq[j] * alpha
    seq[i] -= removed_i
    seq[j] -= removed_j
    total = removed_i + removed_j
    tgt = rng.integers(0, n)
    # avoid indices i,j
    while tgt == i or tgt == j:
        tgt = rng.integers(0, n)
    seq[tgt] += total
    return seq

import time
import numpy as np
import cvxpy as cp

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a decreasing non‑negative sequence a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _fast_score(seq: np.ndarray):
    """
    Compute AC1 bound using FFT (returns score, full convolution, max value).
    """
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())   # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val

def _solve_lp(f: np.ndarray, M: float, time_left: float):
    """
    Solve   max sum(g)   s.t.   g >= 0 ,   conv(f,g) <= M .
    Returns a NumPy array g or None on failure/timeout.
    """
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    f_const = cp.Constant(f)
    conv = cp.conv(f_const, g)          # length 2n-1
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv <= M])
    # Use a fast LP solver; fall back if unavailable.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500,
                    abstol=1e-7, reltol=1e-7, feastol=1e-7, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, warm_start=True, msg=False)
        except Exception:
            return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    g_val = np.asarray(g.value, dtype=np.float64)
    np.clip(g_val, 0.0, 1000.0, out=g_val)
    return g_val
