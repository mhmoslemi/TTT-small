import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt-series (a*a = 1+...+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Fast AC1 score (same formula as evaluate_sequence)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # leave a small head‑room for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----- initialise ------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = _project_simplex(a)                     # enforce sum=1, non‑neg
        n = a.size
    else:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform fallback

    # If the parent is not the sqrt‑series, try the analytic one as an
    # alternative initial point and keep the better of the two.
    sqrt_init = _sqrt_series(n)
    sqrt_init = sqrt_init / sqrt_init.sum()
    if _score(sqrt_init) < _score(a):
        a = sqrt_init.copy()

    best_vec = a.copy()
    best_sc = _score(best_vec)

    # pre‑computed helper
    L = 1 << ((2 * n - 1).bit_length())
    idx = np.arange(n, dtype=np.int64)

    # ----- optimisation loop ----------------------------------------
    step0 = 0.5          # initial step magnitude
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # current convolution & the active index
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        m = int(conv.argmax())                     # max‑index
        # sub‑gradient of max(conv) w.r.t. a
        mask = (m - idx >= 0) & (m - idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[m - idx[mask]]           # g_j = 2*a_{m-j}

        # --------- projected sub‑gradient step ----------------------
        eta = step0 / np.sqrt(it)                 # diminishing step
        cand = a - eta * g
        cand = _project_simplex(cand)
        sc = _score(cand)

        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
            a = cand                                 # move forward
            continue

        # backtracking if the full step was too aggressive
        improved = False
        eta_bt = eta
        for _ in range(4):
            eta_bt *= 0.5
            cand_bt = a - eta_bt * g
            cand_bt = _project_simplex(cand_bt)
            sc_bt = _score(cand_bt)

                best_vec = cand_bt.copy()
                a = cand_bt
                improved = True
                break
        if improved:
            continue

        # --------- occasional random perturbation --------------------

                    a = cand2

    # ----- final fallback (always valid) -----------------------------
    return [float(x) for x in best_vec]
