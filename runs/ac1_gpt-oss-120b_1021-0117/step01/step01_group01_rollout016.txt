import time
import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 using recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of a*a (full linear convolution)."""
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())


def _score_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """Score 2*n*max(conv)/(sum(a)^2) – the evaluator’s formula."""
    max_b = _conv_max_fft(a, fft_len)
    s = a.sum()
    if s == 0.0:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for final checks
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- 1. Load or generate a start sequence --------------------
    try:
        n = int(len(height_sequence_1))
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        n = 256
        a = _generate_binomial_seq(n)

    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total                     # normalise to sum = 1

    # ---- 2. Prepare FFT layout -----------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # power of two >= conv_len

    # ---- 3. Initial scoring --------------------------------------
    cur = a.copy()
    cur_score = _score_fft(cur, n, fft_len)
    best = cur.copy()
    best_score = cur_score

    # ---- 4. Optimisation parameters -------------------------------
    frac_low, frac_high = 0.05, 0.30          # amount of mass moved each step
    smooth_prob = 0.02                       # chance to apply a smooth update
    smooth_kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    smooth_kernel /= smooth_kernel.sum()
    idx_all = np.arange(n, dtype=np.int64)

    # ---- 5. Main loop (targeted hill‑climbing) -------------------
    while time.monotonic() < deadline:
        # compute full convolution and locate the dominant shift
        conv_full = _conv_full_fft(cur, fft_len)
        k0 = int(np.argmax(conv_full))

        # contributions of each index to that dominating shift
        partner = k0 - idx_all
        valid = (partner >= 0) & (partner < n)
        if valid.any():
            contrib = np.zeros(n, dtype=np.float64)
            contrib[valid] = cur[idx_all[valid]] * cur[partner[valid]]
            i_src = int(np.argmax(contrib))
            # pick a destination with minimal contribution (prefer zero)
