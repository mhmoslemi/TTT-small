def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine n
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    else:
        n = 5582
        parent = None

    # Helper functions
    def conv_self(a):
        m = 1 << ((2 * a.size - 1).bit_length())
        fa = np.fft.rfft(a, n=m)
        return np.fft.irfft(fa*fa, n=m)[:2*a.size-1]

    def score(a):
        # quick internal scoring, same as evaluator after clipping (but we don't clip because a already inside bounds)
        b = conv_self(a)
        maxb = b.max()
        s = a.sum()
        if s < 0.01: 
            return np.inf
        return 2.0 * a.size * maxb / (s * s)

    # Basic shapes
    def sqrt_series(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (2.0 * (i-1) + 1.0) / (2.0 * i)
        return a

    def hann_window(n):
        i = np.arange(n, dtype=np.float64)
        return 0.5 * (1 - np.cos(2 * np.pi * i / (n-1)))

    def beta_shape(alpha, beta):
        i = np.arange(1, n+1, dtype=np.float64)
        x = i / (n+1)
        return (x ** alpha) * ((1 - x) ** beta)

    # Setup initial candidates
    candidates = []
    sqrt_seq = sqrt_series(n)
    candidates.append(sqrt_seq)
    if parent is not None:
        candidates.append(parent)
    # add few param shapes
    candidates.append(hann_window(n))
    # random beta shapes
    for _ in range(3):
        a = rng.uniform(0.2,2.0)
        b = rng.uniform(0.2,2.0)
        candidates.append(beta_shape(a,b))
    # Evaluate and pick best as current
    best_score = np.inf
    best_seq = None
    for c in candidates:
        sc = score(c)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = c.copy()
    current_seq = best_seq.copy()
    current_score = best_score

    # Main loop: mixture of gradient steps and random transfers
    iter = 0
    while time.monotonic() < deadline:
        iter += 1
        # Choose operation
        r = rng.random()
        if r < 0.6:
            # gradient descent step on max index
            conv = conv_self(current_seq)
            k = int(np.argmax(conv))
            # gradient
            g = current_seq.copy()
            # compute shifted addition
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            if i_end >= i_start:
                idx = np.arange(i_start, i_end+1)
                g[idx] += current_seq[k - idx]
            # project to zero-sum direction
            g_centered = g - g.mean()
            # step size schedule
            step = 0.2 / np.sqrt(iter)
            cand = current_seq - step * g_centered
        elif r < 0.8:
            # random mass transfer similar to baseline
            cand = current_seq.copy()
            early_cut = n // 4
            late_cut = 3 * n // 4
            src = rng.integers(0, early_cut)
            dst = rng.integers(late_cut, n)
            max_transfer = min(cand[src], 1000.0 - cand[dst])
            if max_transfer > 0:
                delta = max_transfer * rng.random() * 0.04
                cand[src] -= delta
                cand[dst] += delta
        elif r < 0.95:
            # block scaling
            cand = current_seq.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = rng.integers(0, max(1, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.1)))
            cand[start:start+width] *= factor
        else:
            # noise perturb
            cand = current_seq + rng.normal(0.0, 0.01, size=n)

        # enforce bounds and renorm sum
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s < 0.01:
            continue
        cand *= (current_seq.sum() / s)  # keep total mass constant

        cand_score = score(cand)
        if np.isfinite(cand_score) and cand_score < current_score:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
        else:

            if temp > 0 and rng.random() < np.exp((current_score - cand_score) / temp):
                current_seq = cand
                current_score = cand_score

def _conv_self(a: np.ndarray, m: int) -> np.ndarray:
    """Linear convolution a*a via real FFT (length m is power of two)."""
    fa = np.fft.rfft(a, n=m)
    return np.fft.irfft(fa * fa, n=m)[: 2 * a.size - 1]


def _score(a: np.ndarray, m: int) -> float:
    """Exact objective used by the authoritative evaluator."""
    if a.sum() < 0.01:
        return np.inf
    b = _conv_self(a, m)
    return 2.0 * a.size * b.max() / (a.sum() ** 2)

    """Classic a_i = C(2i,i)/4^i (length n)."""

    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)

                cand[src] -= delta
                cand[dst] += delta
        # --------------------------------------------------------------
        # 3) Block‑wise multiplicative scaling
        # --------------------------------------------------------------
        elif r < 0.90:
            cand = cur_seq.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = rng.integers(0, max(1, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.09)))
            end = min(n, start + width)
            cand[start:end] *= factor
        # --------------------------------------------------------------
        # 4) Tiny additive Gaussian noise
        # --------------------------------------------------------------
        else:
            cand = cur_seq + rng.normal(0.0, 0.012, size=n)

        # --------------------------------------------------------------
        # Enforce bounds, renormalise to unit L1, and evaluate
        # --------------------------------------------------------------
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s < 0.01:
            continue
        cand = cand / s
        sc = _score(cand, m)

        # --------------------------------------------------------------
        # Accept improvements (and occasional uphill moves)
        # --------------------------------------------------------------
        improved = False
        if np.isfinite(sc) and sc < cur_score:
            improved = True
        else:
            # simple simulated‑annealing acceptance
            temp = 0.12 * max(0.0, 1.0 - it / 25000.0)
            if temp > 0 and rng.random() < np.exp((cur_score - sc) / temp):
                improved = True

        if improved:
            cur_seq = cand
            cur_score = sc
            if sc < best_score:
                best_seq = cand.copy()
                best_score = sc

        # --------------------------------------------------------------
        # Occasional exploration of a fresh parametric shape
        # --------------------------------------------------------------
        if it % 4200 == 0:
            typ = rng.integers(0, 3)
            if typ == 0:   # power‑law
                p = rng.uniform(0.1, 0.9)
                new = np.arange(1, n + 1, dtype=np.float64) ** (-p)
            elif typ == 1:  # beta distribution
                a = rng.uniform(0.2, 4.0)
                b = rng.uniform(0.2, 4.0)
                new = _beta_shape(n, a, b)
            else:           # sine‑power
                pw = rng.uniform(0.5, 3.5)
                new = _sin_power(n, pw)
            np.clip(new, 0.0, 1000.0, out=new)
            new = new / new.sum()
            sc_new = _score(new, m)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_seq = new.copy()
                best_score = sc_new
                cur_seq = new.copy()
                cur_score = sc_new

    # ------------------------------------------------------------------
    # Return the best discovered sequence as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
