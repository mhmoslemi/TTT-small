def _conv_fft(vec):
    n = vec.size

    fv = np.fft.rfft(vec, n=L)

    n = v.shape[0]
    u = np.sort(v)[::-1]

    rho_arr = u - css / np.arange(1, n+1)

    else:

import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# First n coefficients of the series a where a*a = 1+...+x^{2n-2}
# (the truncated expansion of (1-x)^(-1/2))
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# convolution via FFT together with the AC1 score
# ------------------------------------------------------------
def _conv_and_score(vec: np.ndarray):
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    maxb = conv.max()
    s = vec.sum()
    score = float(2.0 * n * maxb / (s * s))
    return maxb, conv, score

# ------------------------------------------------------------

    # --------------------------------------------------------
    # initialise – use the supplied parent if available
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
    else:

        base = _sqrt_series(n)
    n = int(base.shape[0])

    # put on the probability simplex (sum = 1, non‑negative)
    base = np.maximum(base, 0.0)
    if base.sum() == 0.0:
        base = np.full(n, 1.0 / n, dtype=np.float64)
    cur = _project_simplex(base / base.sum())

    _, _, best_score = _conv_and_score(cur)
    best = cur.copy()

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    target = 1.50274349          # stop early if we reach the target bound
    prob_grad = 0.35             # chance to perform a gradient step
    step_max = 0.25              # maximal step size for gradient moves
    no_improve = 0               # counter for stagnation

    # --------------------------------------------------------
    # main search loop (time‑bounded)

    while time.monotonic() < deadline:
