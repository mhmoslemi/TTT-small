import time
import numpy as np

def _score(arr: np.ndarray) -> float:
    """Exact score used by the evaluator (faster than calling it)."""
    n = arr.shape[0]
    conv = np.convolve(arr, arr)
    max_b = np.max(conv)
    s = arr.sum()
    if s <= 0.0:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search a non‑negative step‑function that lowers the AC1 upper bound.

    The routine mixes greedy mass‑transfer moves targeted at the current
    maximal autocorrelation entry, simple projected gradient steps and occasional
    random jitter.  It respects the time budget and always returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

            raise ValueError
    except Exception:

    # Clip to the bounds required by the evaluator.
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 0.01:
        parent = np.ones_like(parent)

    # Normalise to unit sum (the score is scale‑invariant).
    a = parent / parent.sum()
    n = a.shape[0]

    # ------------------------------------------------------------------
    # 2. Prepare a fallback (uniform) and pick the better starter.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best = a if _score(a) <= _score(uniform) else uniform
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 3. Main optimisation loop.
    # ------------------------------------------------------------------
    # Adaptive parameters.
    transfer_frac = 0.05          # fraction of mass moved in a transfer
    min_transfer = 0.005
    eta = 0.5                     # step size for the gradient‑like move
    min_eta = 1e-4
    it = 0

    while time.monotonic() < deadline:
        # --------------------------------------------------------------

        # --------------------------------------------------------------
        conv = np.convolve(best, best)
