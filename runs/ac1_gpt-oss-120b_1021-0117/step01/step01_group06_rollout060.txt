def _conv_and_max(vec):
    n = len(vec)
    L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv*fv, n=L)[:2*n-1]
    max_b = conv.max()
    return conv, max_b

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        # Ensure non-negative and within bounds; they already are.
    else:
        n = 256
        a = _sqrt_series(n).astype(np.float64)
    # Normalize to sum=1
    if a.sum() == 0:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.size
    # Precompute FFT length
    L = 1 << ((2 * n - 1).bit_length())
    # Helpers:
    def conv_max(vec):
        fv = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        max_b = conv.max()
        max_idx = int(conv.argmax())
        return conv, max_b, max_idx
    # Evaluate initial
    _, max_b, max_idx = conv_max(a)
    best_vec = a.copy()
    best_sc = 2.0 * n * max_b / (a.sum() ** 2)  # sum is 1
    cur_vec = a.copy()
    cur_sc = best_sc
    # main loop
    it = 0
    max_iter = 2000000  # big bound
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        # compute gradient of the current max shift
        # we have conv and max_idx from previous step? need compute again if cur_vec changed via random step not yet evaluated.
        # For simplicity compute fresh each iteration.
        conv, max_b, idx = conv_max(cur_vec)
        cur_sc = 2.0 * n * max_b / (cur_vec.sum() ** 2)
        # gradient for shift idx
        g = np.zeros_like(cur_vec)
        k = idx
        if k < n:
            if k > 0:
                g[:n - k] += cur_vec[k:]
                g[k:] += cur_vec[:n - k]
            else:
                # k == 0, gradient = 2 * cur_vec
                g = 2.0 * cur_vec
        else:
            # k >= n
            off = k - n + 1
            length = n - off
            if length > 0:
                idx_range = np.arange(off, n)
                g[idx_range] = cur_vec[k - idx_range]
        # step size - simple diminishing
        eta = 1.0 / (1.0 + 0.001 * it)  # start near 1, quickly shrink
        # propose descent step
        cand = cur_vec - eta * g
        cand = _project_simplex(cand)
        # evaluate
        _, cand_max, _ = conv_max(cand)
        cand_sc = 2.0 * n * cand_max / (cand.sum() ** 2)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
        # acceptance rule (simulated annealing)
        # temperature schedule
        temp = 0.05 * (1.0 - it / max_iter) + 0.001
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            # metropolis
            prob = np.exp((cur_sc - cand_sc) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = cand
            cur_sc = cand_sc
        # occasional random transfer to escape
        if it % 500 == 0:
            # random move
            i = rng.integers(0, n)
            if cur_vec[i] == 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur_vec[i] * rng.random() * 0.2
            tmp = cur_vec.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            _, tmp_max, _ = conv_max(tmp)
            tmp_sc = 2.0 * n * tmp_max / (tmp.sum() ** 2)
            if tmp_sc < best_sc:
                best_sc = tmp_sc
                best_vec = tmp.copy()
            # acceptance similar to above
            if tmp_sc < cur_sc or rng.random() < np.exp((cur_sc - tmp_sc) / (temp + 1e-12)):
                cur_vec = tmp
                cur_sc = tmp_sc
    # Return best vector as list
    return [float(x) for x in best_vec]

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = u - css / np.arange(1, n + 1)
    rho_idx = np.where(rho > 0)[0]
    if rho_idx.size == 0:
        theta = 0.0
    else:
        rho = rho_idx[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)

    for k in range(1, n):

            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0

def _conv_max(vec: np.ndarray, L: int):

    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    idx = int(conv.argmax())
    return conv, float(conv[idx]), idx

    g = np.zeros_like(a)
    np.add.at(g, i, a[j])
