import numpy as np
import time

# ----- convolution helpers ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two >= x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution via FFT of length ``fft_len`` (power of two)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    return np.fft.irfft(fx * fy, n=fft_len)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2*len(a)-1)."""
    full = _conv_full_fft(a, a, fft_len)
    n = a.size
    return full[: 2 * n - 1]


# ----- classic sqrt‑series (fallback) ----------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a

    # final check (using the authoritative evaluator for safety)
    final_score = evaluate_sequence(a.tolist())
    if not np.isfinite(final_score) or final_score > best_score:
        # fallback to the best found earlier
        a = best_a

    # return plain Python list
    return [float(v) for v in a]
