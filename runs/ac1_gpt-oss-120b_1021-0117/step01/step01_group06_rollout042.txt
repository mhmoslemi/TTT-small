import time
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = css[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Coefficients of the series a where a*a = 1 + x + … + x^{2n-2}
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
# Fast evaluation of the AC1 bound (identical to evaluate_sequence)
# ----------------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (vec.sum() ** 2))


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # initialise the sequence
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() == 0.0:
            n = a.size
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
    else:
        n = 256
        a = _sqrt_series(n)
        a = a / a.sum()

    best = a.copy()
    best_sc = _score(best)

    # --------------------------------------------------------------
    # pre‑compute FFT length (must accommodate both convolution & correlation)
    # --------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())          # >= 3n‑2, works for all transforms

    # --------------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------------
    step0 = 0.5                # initial step magnitude
    max_iter = 200000          # hard cap; the time check will stop earlier
    alpha_start = 5.0
    alpha_end = 50.0

    i = np.arange(n)           # reused index vector

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1

        # ----- convolution a*a (needed for the smooth max) -----
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]

        # ----- smooth approximation of the max (log‑sum‑exp) -----
        # linearly increase the sharpness parameter alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)
        cm = conv.max()
        exp_w = np.exp(alpha * (conv - cm))
        w = exp_w / exp_w.sum()                     # weights, length 2n‑1

        # ----- gradient of the smoothed max w.r.t. a -----
        w_rev = w[::-1]                             # reversed weights
        fw = np.fft.rfft(w_rev, n=L)                # FFT of reversed weights
        # correlation w ⋆ a = (w_rev * a) in frequency domain
        corr_full = np.fft.irfft(fa * fw, n=L)
        # gradient_i = 2 * Σ_j w_{i+j} a_j  → slice (2n‑2‑i)
        idxs = (2 * n - 2) - i
        grad = 2.0 * corr_full[idxs]

        # ----- normalized step (projected sub‑gradient descent) -----
        norm = np.linalg.norm(grad) + 1e-12
        direction = grad / norm
        step = step0 / np.sqrt(it)                  # diminishing step size
        cand = a - step * direction
        cand = _project_simplex(cand)                # stay on the simplex

        sc = _score(cand)
        if sc < best_sc:

            best = cand.copy()

        # move the current point forward – always accept the proposal
        a = cand

                a = tmp.copy()
