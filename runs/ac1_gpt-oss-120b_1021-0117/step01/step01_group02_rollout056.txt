def simplex_projection(v):
    """Project v onto the probability simplex {x >=0, sum(x)=1}."""
    # from https://gist.github.com/...
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    deadline = time.monotonic() + max(0.0, budget_s - 0.5) # leaves 0.5 sec buffer

    try:

        a = init.copy()
    except Exception:

        a[0] = 1.0
        for i in range(1,n):
            a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
        a = a / a.sum()

    # compute initial best_score via evaluate_sequence (or our own)
    # use evaluate_sequence for safety; but that uses np.convolve which is O(n^2), but called only few times
    # We can use our fast compute for speed but result identical (within rounding). Use evaluate_sequence for guarantee.
    try:

        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_score = float('inf')
    best = a.copy()

    # parameters
    step_factor = 0.2  # initial step

    step_decay = 0.995   # per iteration if no improvement
    # Random generator
    rng = np.random.default_rng(seed)

    def conv_self(v):
        L = 1 << ((2*len(v)-1).bit_length())
        fa = np.fft.rfft(v, L)
        conv = np.fft.irfft(fa * fa, L)[:2*len(v)-1]
        return conv

    # projection onto simplex
    def simplex_proj(v):
        # ensure shape 1d
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        # rho is last index where u_i > cssv_i / i
        cond = u - cssv / ind > 0
        rho = np.where(cond)[0]
        if rho.size == 0:
            # then all v <= 0, return uniform distribution
            return np.full_like(v, 1.0 / len(v))
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    # main loop
    it = 0
    while time.monotonic() < deadline:
        # compute convolution via FFT
        conv = conv_self(a)
        max_val = conv.max()
        k = int(conv.argmax())
        # compute partner vector
        partner = np.zeros_like(a)

        i_end = min(k, n - 1)
        # Here, i_start <= i_end
        if i_start <= i_end:

            partner[idx] = a[k - idx]

        # For simplicity, use one step:
        step = step_factor

        # project onto simplex
        cand = simplex_proj(cand)

        # compute candidate score quickly
        conv_cand = conv_self(cand)
        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max / (cand.sum()**2)   # sum=1

        # accept if lower
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            a = cand   # move to cand for next iteration
        else:
            # optionally reject and reduce step size
            step_factor = max(min_step_factor, step_factor * step_decay)

                if cand2_score < best_score:

                    best_score = cand2_score

        it += 1

    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]

    best = a.copy()

import time
import numpy as np

def simplex_projection(v):
    """Project a vector onto the simplex {x >=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive
        w = np.full_like(v, 1.0 / n)
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to eliminate numerical drift
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def convolve_self(v, fft_len):
    """Return linear convolution of v with itself (real) using FFT of length fft_len."""
    fv = np.fft.rfft(v, fft_len)
    cv = np.fft.irfft(fv * fv, fft_len)
    # keep only the first 2*len(v)-1 entries
    return cv[:2*len(v)-1]

        a = np.empty(n, dtype=np.float64)

    cur_conv = convolve_self(a, fft_len)

import time
import numpy as np

def simplex_projection(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def convolve_self(v, fft_len):
    """Linear self‑convolution of v with itself using an FFT of length fft_len."""
    fv = np.fft.rfft(v, fft_len)
    cv = np.fft.irfft(fv * fv, fft_len)
    return cv[: 2 * len(v) - 1]
