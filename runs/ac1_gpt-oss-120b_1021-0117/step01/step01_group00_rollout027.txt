def _ac1_score(arr, n, Nfft):
    s = arr.sum()
    if s < 0.01:

    B = A * A

import numpy as np

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)
