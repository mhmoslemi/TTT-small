# Helper factories for candidate sequences

def _power_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """a_i = (i+delta)^(-alpha), normalised to sum 1."""
    x = np.arange(n, dtype=np.float64) + delta
    a = np.power(x, -alpha)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def _symmetric_power(alpha: float, delta: float, n: int) -> np.ndarray:
    """a_i = (i+delta)^(-alpha) + (n-1-i+delta)^(-alpha), normalised."""
    x = np.arange(n, dtype=np.float64) + delta
    a = np.power(x, -alpha)
    a = a + a[::-1]
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a
