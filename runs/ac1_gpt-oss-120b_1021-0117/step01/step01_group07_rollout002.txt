def conv_max(a):
    n = a.shape[0]
    m = 1 << (2*n-1).bit_length()
    A = np.fft.rfft(a, m)
    b = np.fft.irfft(A * A, m)[:2*n-1]
    return b.max(), b.argmax()
```

def simplex_proj(v):
    # find projection of vector v onto simplex {x >=0, sum x = 1}
    # as per Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# --------------------------------------------------------------
def _simplex_proj(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1  # zero‑based index of the last positive entry
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Helper: max of the aperiodic autocorrelation via FFT (matrix‑free)
# --------------------------------------------------------------
def _conv_max(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length fft_len."""
    A = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(A * A, fft_len)[:conv_len]
    return float(c.max())

    minimises the AC1 upper bound as evaluated by ``evaluate_sequence``.
    The routine combines a quick parametric sweep, a simple projected
    gradient descent and occasional random jitter, all within the
    allotted time budget.
    """

    # leave a small safety margin for the sandbox
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------
    # 1.  Obtain a fallback parent (if supplied) and basic size
    # ----------------------------------------------------------
    try:

        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)

    # ----------------------------------------------------------
    # 2.  Baseline candidates: uniform and the provided parent
    # ----------------------------------------------------------
    best = np.ones(n, dtype=np.float64) / n          # uniform (sum = 1)
    best_score = float(evaluate_sequence(best.tolist()))

    # try the supplied parent (renormalised to sum 1)

        cand_score = float(evaluate_sequence(cand.tolist()))

    return [float(v) for v in best]
