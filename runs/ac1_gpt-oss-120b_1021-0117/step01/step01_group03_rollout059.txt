import numpy as np, time, math, scipy.optimize

def _conv_self(a):
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[:2*n-1]
    return conv

def _score(a):
    n = a.size
    conv = _conv_self(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

import numpy as np

# fast convolution via real FFT (linear convolution, not circular)
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exactly the objective used by the evaluator (fast version)."""
    n = a.size
    conv = _conv_self(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a

                    max_transfer = min(cand[src], 1000.0 - cand[dst])
                    if max_transfer > 0.0:
                        delta = max_transfer * rng.random() * 0.07
                        cand[src] -= delta
                        cand[dst] += delta

        elif op < 0.65:
            # generic mass transfer between two random positions
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src != dst:
                max_transfer = min(cand[src], 1000.0 - cand[dst])
                if max_transfer > 0.0:
                    delta = max_transfer * rng.random() * 0.10
                    cand[src] -= delta
                    cand[dst] += delta

        elif op < 0.85:
            # block scaling (multiplicative perturbation)
            width = max(1, int(rng.uniform(n // 200, n // 12)))
            start = rng.integers(0, n - width + 1)
            factor = float(np.exp(rng.normal(0.0, 0.14)))
            cand[start:start + width] *= factor

        else:
            # additive Gaussian noise
            cand += rng.normal(0.0, 0.025, size=n)

        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)

        # accept improvements outright, otherwise possibly via annealing
        if sc < cur_score or rng.random() < math.exp(-(sc - cur_score) / max(temperature, 1e-9)):
            cur_seq = cand
            cur_score = sc
            if sc < best_score:
                best_seq = cand
                best_score = sc

        # cool down
        temperature = max(temperature * decay, 1e-5)

    # ------------------------------------------------------------------
    # Return the best candidate discovered (as plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
