            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            partner = np.zeros_like(a)
            partner[idx] = a[k - idx]

            src = int(partner.argmax())
            # destination: any index where partner is minimal (often zero)
            min_val = float(partner.min())
            dest_cand = np.where(partner == min_val)[0]
            dest = int(rng.choice(dest_cand)) if dest_cand.size else int(rng.integers(0, n))

            if src == dest:
                # occasional random local scaling when no useful move is found
                width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = min(n, start + width)
                cand = a.copy()
                cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
                cand = np.clip(cand, 0.0, 1000.0)
                cand = _normalise(cand, total)
                new_max = np.convolve(cand, cand).max()
                if new_max < best_local_max:
                    a, conv = cand, np.convolve(cand, cand)
                    best_local, best_local_max = a.copy(), new_max
                continue

            delta = a[src] * step_factor
            if delta < min_step:
                delta = min_step
            if delta > a[src]:
                delta = a[src]

            # ---- fast update of convolution ---------------------------------------
            conv_new = conv.copy()
            conv_new[src:src + n] -= 2.0 * delta * a
            conv_new[dest:dest + n] += 2.0 * delta * a
            conv_new[2 * src] += delta * delta
            conv_new[2 * dest] += delta * delta
            conv_new[src + dest] -= 2.0 * delta * delta

            new_max = conv_new.max()
            if new_max < best_local_max:
                # accept improvement
                a[src] -= delta
                a[dest] += delta
                conv = conv_new
                best_local, best_local_max = a.copy(), new_max
                step_factor = max(0.02, step_factor * 0.998)
                stagnation = 0
            else:
                # occasional uphill acceptance to escape plateaus
                if rng.random() < 0.001:
                    a[src] -= delta
                    a[dest] += delta
                    conv = conv_new
                stagnation += 1

            # occasional random shake of a segment
            if rng.random() < 0.005:
                width = int(rng.integers(max(1, n // 300), max(2, n // 50)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = min(n, start + width)
                shake = a.copy()
                shake[start:stop] *= float(np.exp(rng.normal(0.0, 0.15)))
                shake = np.clip(shake, 0.0, 1000.0)
                shake = _normalise(shake, total)
                shake_max = np.convolve(shake, shake).max()
                if shake_max < best_local_max:
                    a, conv = shake, np.convolve(shake, shake)
                    best_local, best_local_max = a.copy(), shake_max

            # periodic full recompute to curb accumulated rounding error
            if stagnation % 5000 == 0:
                conv = np.convolve(a, a)

        return best_local, best_local_max

    # -------------------------------------------------------------------------------
    for init in init_candidates:
        if time.monotonic() >= deadline:
            break
        seq, seq_max = _optimize(init)
        score = 2.0 * n * seq_max / (seq.sum() ** 2)
        if score < best_score:
            best_score, best_seq = score, seq.copy()

    # ---- final safety checks --------------------------------------------------------
    final = np.clip(best_seq, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
