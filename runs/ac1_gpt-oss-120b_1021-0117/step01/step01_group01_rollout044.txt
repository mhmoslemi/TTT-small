def proj_simplex(v):
    # v: np.ndarray shape (n,)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _generate_binomial_seq(n: int) -> np.ndarray:
    """a[i] = C(2i,i)/4**i  for i = 0..n-1 (decreasing, positive)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _generate_powerlaw_seq(n: int, p: float = 0.5) -> np.ndarray:
    """a[i] = (i+1)^{-p}, then normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    # Find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are <=0 -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1].astype(int) - 1  # zero‑based
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise
    s = w.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / n)
    return w / s


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple:
    """Return (max value, argmax index) of the linear convolution a*a."""
    fa = np.fft.rfft(a, n=fft_len)

    conv = conv[: 2 * a.size - 1]
    max_idx = int(np.argmax(conv))
    return float(conv[max_idx]), max_idx
