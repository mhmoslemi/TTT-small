import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Catalan weights (central binomial / 4**i)
# ----------------------------------------------------------------------
def _catalan_weights(n: int) -> np.ndarray:
    """Return the first n Catalan‑style weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        # recurrence: w[i] = w[i-1] * (2i)*(2i-1) / (i*i*4)
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w
