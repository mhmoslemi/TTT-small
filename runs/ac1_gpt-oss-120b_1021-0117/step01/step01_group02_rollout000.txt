def score(v):
    n = v.shape[0]
    # compute linear convolution length 2n-1 using FFT of size next power of 2 > 2n
    m = int(2**np.ceil(np.log2(2*n-1)))
    fv = np.fft.rfft(v, n=m)

    maxc = conv.max()
    s = v.sum()
    return 2.0 * n * maxc / (s**2)

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Simulated‑annealing style search for a low‑peak self‑convolution.
    Returns a non‑negative step‑function (list of floats)."""
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialise -------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
    except Exception:                     # no parent supplied
        n = 5000
        init = np.empty(n, dtype=np.float64)
        init[0] = 1.0
        for i in range(1, n):
            init[i] = init[i - 1] * (1.0 - 1.0 / (2.0 * i))

    init = np.clip(init, 0.0, 1000.0)
    tot = init.sum()
    if tot <= 0.0:
        init = np.full(n, 1.0 / n, dtype=np.float64)
        tot = 1.0
    init = init / tot                     # normalise to sum == 1

    # ---- score helper -----------------------------------------------------
    def _score(v):
        c = np.convolve(v, v)              # linear (non‑circular) convolution
        return 2.0 * n * c.max() / (v.sum() ** 2)

    # ---- start from the parent or a uniform fallback -----------------------
    best = init.copy()
    best_score = _score(best)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score = _score(uniform)
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score

    cur = best.copy()
    cur_score = best_score

    # ---- annealing parameters ---------------------------------------------
    T0 = best_score * 0.05                 # start temperature ~5 % of score
    T_min = 1e-9
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)   # gentle smoother

    # ---- main loop ---------------------------------------------------------
    while time.monotonic() < deadline:
        # progress‑dependent temperature
        elapsed = time.monotonic() - start
        total = deadline - start
        prog = elapsed / total if total > 0 else 1.0
        temp = max(T_min, T0 * (1.0 - prog))

        op = rng.random()
        cand = None

        if op < 0.40:
            # ---- targeted reduction of the current worst autocorr shift -----
            conv = np.convolve(cur, cur)
            k = int(conv.argmax())                       # shift with maximal value
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            partner = cur[k - idx]                       # values that pair with idx
            prod = cur[idx] * partner
            # pair contributing most to the peak
            src1 = idx[np.argmax(prod)]
            src2 = k - src1
            # move a modest fraction away from the pair
            frac = 0.20

            # pick low‑mass destinations

            dst1, dst2 = int(dst[0]), int(dst[1])

            cand = cur.copy()

            cand[dst1] += d1
            if src2 != src1:
                cand[src2] = max(0.0, cand[src2] - d2)
                cand[dst2] += d2
            cand = np.clip(cand, 0.0, 1000.0)

        elif op < 0.80:
            # ---- random sub‑array scaling ---------------------------------
            max_len = max(2, n // 10)
            width = int(rng.integers(1, max_len + 1))
