def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    try: n = int(len(height_sequence_1))
    except Exception: n = 5000

    # scoring function
    def _score(v):
        c = np.convolve(v, v)
        maxb = c.max()
        total = v.sum()
        if total == 0: return np.inf
        return 2.0 * n * maxb / (total * total)

    best = np.full(n, 1.0 / n, dtype=np.float64)  # uniform as fallback
    best_score = _score(best)

    # Add inverse sqrt shape
    i = np.arange(n, dtype=np.float64)
    a_inv_sqrt = 1.0 / np.sqrt(i + 1.0)  # alpha=1 prevents division by zero
    a_inv_sqrt = a_inv_sqrt / a_inv_sqrt.sum()
    s = _score(a_inv_sqrt)
    if s < best_score:
        best = a_inv_sqrt
        best_score = s

    # grid search of power law shapes
    p_vals = np.linspace(0.35, 0.7, 15)  # from 0.35 to 0.70
    alpha_vals = np.geomspace(0.5, 20.0, 15)  # geometric spacing
    for p in p_vals:
        for alpha in alpha_vals:
            if time.monotonic() >= deadline: break
            a = 1.0 / ((i + alpha) ** p)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a = a / a_sum
            s = _score(a)
            if s < best_score:
                best = a
                best_score = s

    # optional: try reflected version of power law at the end (i.e., decreasing from both sides)
    # not needed now.

    # local improvement using gradient-like moves
    step_factor = 0.4
    min_step = 1e-7
    while time.monotonic() < deadline:
        conv = np.convolve(best, best)
        maxb = conv.max()
        k = int(conv.argmax())
        # compute partner
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # vectorized
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(best)
        partner[idx] = best[k - idx]

        # product contributions
        prod = best * partner
        src = int(np.argmax(prod))

        min_partner = partner.min()
        dest_candidates = np.where(partner == min_partner)[0]
        if dest_candidates.size == 0:
            dest = rng.integers(0, n)
        else:
            dest = int(rng.choice(dest_candidates))
        if src == dest:
            # random perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = best.copy()
            cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand / cand.sum() * best.sum()
            cand_score = _score(cand)
            if cand_score < best_score:
                best, best_score = cand, cand_score
            continue

        delta = best[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > best[src]:
            delta = best[src]
        cand = best.copy()
        cand[src] -= delta
        cand[dest] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        # Renormalize to preserve sum (which is (original) 1 after normalization)
        s = cand.sum()
        if s == 0:
            continue
        cand = cand / s * best.sum()
        cand_score = _score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
            step_factor = max(0.02, step_factor * 0.998)
        else:
            # occasionally random shake
            if rng.random() < 0.005:
                width = int(rng.integers(max(1, n // 300), max(2, n // 50)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = min(n, start + width)
                shake = best.copy()
                shake[start:stop] *= float(np.exp(rng.normal(0.0, 0.15)))
                shake = np.clip(shake, 0.0, 1000.0)
                shake = shake / shake.sum() * best.sum()
                shake_score = _score(shake)
                if shake_score < best_score:
                    best, best_score = shake, shake_score

    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    return [float(v) for v in final]

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step‑function with low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # length of the problem
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 5000          # fallback length

    # ------------------------------------------------------------
    # helper: evaluate the objective used by the official scorer
    # ------------------------------------------------------------
    def _score(v: np.ndarray) -> float:
        c = np.convolve(v, v)
        maxb = c.max()
        s = v.sum()
        if s == 0.0:
            return float("inf")
        return 2.0 * n * maxb / (s * s)

    # ------------------------------------------------------------
    # initial candidates
    # ------------------------------------------------------------
    # uniform
    best = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = _score(best)

    # parent if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.0:
            parent = parent / parent.sum()
            ps = _score(parent)
            if ps < best_score:
                best, best_score = parent, ps
    except Exception:
        pass

    # simple inverse‑sqrt shape (α=1 to avoid division by zero)
    i = np.arange(n, dtype=np.float64)
    inv_sqrt = 1.0 / np.sqrt(i + 1.0)
    inv_sqrt = inv_sqrt / inv_sqrt.sum()
    sc = _score(inv_sqrt)
    if sc < best_score:
        best, best_score = inv_sqrt, sc

    # ------------------------------------------------------------
    # grid search over power‑law families
    # ------------------------------------------------------------
    p_vals = np.linspace(0.35, 0.70, 15)                     # exponent
    alpha_vals = np.geomspace(0.5, 20.0, 15)                # shift
    for p in p_vals:
        for alpha in alpha_vals:
            if time.monotonic() >= deadline:
                break
            a = 1.0 / ((i + alpha) ** p)                  # monotone power law
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a = a / a_sum
            sc = _score(a)
            if sc < best_score:
                best, best_score = a, sc

            # symmetric version  a_i ∝ ((i+α)(n‑1‑i+α))^{-p}
            b = 1.0 / (( (i + alpha) * (n - 1 - i + alpha) ) ** p)
            b_sum = b.sum()
            if b_sum == 0.0:
                continue
            b = b / b_sum
            sc = _score(b)
            if sc < best_score:
                best, best_score = b, sc

    # ------------------------------------------------------------
    # local refinement – gradient‑guided mass transfer
    # ------------------------------------------------------------
    step_factor = 0.4
    min_step = 1e-7

    while time.monotonic() < deadline:
        conv = np.convolve(best, best)
        maxb = conv.max()
        k = int(conv.argmax())

        # partner[i] = best[k-i] where defined
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(best)
        partner[idx] = best[k - idx]

        # contribution of each index to the current maximum
        contrib = best * partner
        src = int(np.argmax(contrib))

        # destination: an index that contributes the least to this max
        min_part = partner.min()
        dest_candidates = np.where(partner == min_part)[0]
        if dest_candidates.size == 0:
            dest = rng.integers(0, n)
        else:
            dest = int(rng.choice(dest_candidates))

        if src == dest:
            # a tiny random perturbation when no sensible move exists
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = best.copy()
            cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand / cand.sum() * best.sum()
            cand_score = _score(cand)
            if cand_score < best_score:
                best, best_score = cand, cand_score
            continue

        # move a fraction of mass from src to dest
        delta = best[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > best[src]:
            delta = best[src]

        cand = best.copy()
        cand[src] -= delta
        cand[dest] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        # renormalise to keep the total mass unchanged (up to numerical error)
        total = cand.sum()
        if total == 0.0:
            continue
        cand = cand / total * best.sum()

        cand_score = _score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
            step_factor = max(0.02, step_factor * 0.998)
        else:
            # occasional stochastic shake to escape plateaus
            if rng.random() < 0.005:
                width = int(rng.integers(max(1, n // 300), max(2, n // 50)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = min(n, start + width)
                shake = best.copy()
                shake[start:stop] *= float(np.exp(rng.normal(0.0, 0.15)))
                shake = np.clip(shake, 0.0, 1000.0)
                shake = shake / shake.sum() * best.sum()
                shake_score = _score(shake)
                if shake_score < best_score:
                    best, best_score = shake, shake_score

    # ------------------------------------------------------------
    # final safety clamp
    # ------------------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
