def project_to_simplex(v):
    # ensure nonnegative and sum to 1
    # standard algorithm:
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import time
import math
import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 via recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step sequence with a low AC1 bound.

    The routine uses a simple simulated‑annealing hill‑climber that works
    directly with the FFT‑based convolution proxy used by the authoritative
    evaluator.  It starts from the provided parent `height_sequence_1` (if
    present) or from the classical binomial sequence, normalises the sum to
    one and then iteratively moves a fraction of mass between two positions.
    The best candidate found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Load / construct an initial sequence of length n
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        a0 = np.array(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, 1000.0)
        if a0.sum() < 1e-12:                     # fall back if degenerate
            raise ValueError()
        a0 = a0 / a0.sum()
    except Exception:
        n = 256
        a0 = _generate_binomial_seq(n)
        a0 = a0 / a0.sum()

    # also keep a pure binomial start as a second candidate
    bin_seq = _generate_binomial_seq(n)
    bin_seq = bin_seq / bin_seq.sum()

    # ------------------------------------------------------------------
    # 2. Prepare the FFT layout (used for every score evaluation)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def _score(arr: np.ndarray) -> float:
        """Exact AC1 bound for a normalised sequence (sum == 1)."""
        fa = np.fft.rfft(arr, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # 3. Choose the better of the two initial candidates
    # ------------------------------------------------------------------
    score_a0 = _score(a0)
    score_bin = _score(bin_seq)
    cur = a0 if score_a0 <= score_bin else bin_seq
    cur_score = min(score_a0, score_bin)
    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing search
    # ------------------------------------------------------------------
    # initial temperature proportional to the current score
    temperature = cur_score * 0.05
    min_temp = 1e-9
    cooling = 0.9999

    # helper to pick a non‑zero source index efficiently
    def _pick_source(vec):
        # try a few times; fall back to uniform if all are tiny
        for _ in range(5):
            i = rng.integers(0, n)
            if vec[i] > 1e-12:
                return i
        # worst case: pick the max entry
        return int(np.argmax(vec))

    while time.monotonic() < deadline:
        # ---- propose a move ------------------------------------------------
        src = _pick_source(cur)
        dst = rng.integers(0, n - 1)
        if dst >= src:
            dst += 1

        # move a modest fraction of the source entry
        frac = rng.uniform(0.01, 0.20)
        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta

        # ---- evaluate ------------------------------------------------------
        cand_score = _score(cand)
        dscore = cand_score - cur_score

        # ---- acceptance (metropolis) ---------------------------------------
        if dscore < 0.0 or rng.random() < math.exp(-dscore / max(temperature, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # ---- temperature schedule -----------------------------------------
        temperature = max(temperature * cooling, min_temp)

    # ------------------------------------------------------------------
    # 5. Return best candidate (ensuring evaluator compatibility)
    # ------------------------------------------------------------------
    result = [float(x) for x in best]
    try:
        _ = evaluate_sequence(result)   # hidden validation
    except Exception:
        # As a safety net, fall back to the initial normalised sequence
        result = [float(x) for x in cur]
    return result
