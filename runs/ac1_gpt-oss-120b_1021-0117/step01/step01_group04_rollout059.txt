import numpy as np, time, math

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1 (a decreasing ~1/sqrt(i))."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """FFT based evaluation of the AC1 bound."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a (zero‑padded)."""
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * len(a) - 1]

def _greedy_fill(f: np.ndarray, M: float, rng: np.random.Generator) -> np.ndarray:
    """
    Simple packing heuristic:
    maximise sum(g) s.t. f*g <= M componentwise.
    Returns the raw (unscaled) g.
    """
    n = f.size
    conv = np.zeros(2 * n - 1, dtype=np.float64)   # current f*g
    g = np.zeros(n, dtype=np.float64)
    order = rng.permutation(n)                     # random order for robustness
    # pre‑compute mask for non‑zero f to avoid division by zero
    nz_mask = f > 0.0
    f_nz = f[nz_mask]
    for j in order:
        start = j
        end = j + n
        # capacity remaining on this slice
        cap = M - conv[start:end]
        # elementwise ratio; ignore zero entries of f
        ratios = np.empty_like(f, dtype=np.float64)
        ratios.fill(np.inf)
        ratios[nz_mask] = cap / f_nz
        # smallest admissible increment for index j
        delta = ratios.min()
        if delta <= 0.0:
            continue
        # leave a tiny safety margin to avoid numerical overflow
        delta *= 0.999
        g[j] = delta
        conv[start:end] += delta * f    # update all affected convolutions
    return g

            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.45
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta
        else:                                      # gradient‑guided move
            # identify the lag where the convolution attains its maximum
            conv = _conv_self(cand, L)
            idx = int(conv.argmax())
            # gradient of that entry w.r.t. each coefficient
            grad = np.zeros_like(cand)
            if idx == 0:
                grad = 2.0 * cand
            else:
                # positions where f contributes to lag idx
                grad[:n - idx] += cand[idx:]
                grad[idx:] += cand[:n - idx]
            # pick source (large grad) and sink (small grad)
            src = int(rng.choice(n, p=np.abs(grad) / np.abs(grad).sum()))
            sink = int(rng.choice(n, p=(1.0 / (np.abs(grad) + 1e-12)) /
                                      (1.0 / (np.abs(grad) + 1e-12)).sum()))
            if src != sink and cand[src] > 0.0:
                frac = rng.random() * 0.35
                delta = cand[src] * frac
                cand[src] -= delta
                cand[sink] += delta

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _ac1_score(cand, n, L)

        # record best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay
        if rng.random() < prob_restart:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    return best.tolist()
