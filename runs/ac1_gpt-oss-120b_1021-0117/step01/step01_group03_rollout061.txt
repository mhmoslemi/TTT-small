def _gradient_step(a, rng):
    conv = _conv_self(a)
    k = np.argmax(conv)
    # compute w
    n = a.size
    w = np.zeros_like(a)
    lo = max(0, k - (n-1))
    hi = min(k, n-1)
    idx = np.arange(lo, hi+1)
    w[idx] = a[k - idx]
    product = a * w
    src = np.argmax(product)
    dst = np.argmin(product) # could be zero product indices
    if product[src] <= 0:
        return a  # nothing to move
    # compute max transferable
    max_transfer = min(a[src], 1000.0 - a[dst])
    if max_transfer <= 1e-12:
        return a
    delta = max_transfer * rng.random() * 0.02  # up to 2% of transferable mass
    a[src] -= delta
    a[dst] += delta
    return a

import numpy as np

# --------------------------------------------------------------
# Core utilities (FFT convolution, scoring, simple families)
# --------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution using a real FFT (fast, O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _fast_score(a: np.ndarray):
    """Return (score, max_conv, sum) without extra copies."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf, max_b, s
    return float(2.0 * n * max_b / (s * s)), max_b, s


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def _power_series(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def _lin_decay_series(n: int, alpha: float) -> np.ndarray:
    """a_i = (1 - i/(n-1))^{alpha} (non‑negative, ends at 0)."""
    if n == 1:
        return np.ones(1, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    return (1.0 - t) ** alpha
