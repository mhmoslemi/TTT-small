import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    """Positive decreasing sequence a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_self(seq: np.ndarray) -> np.ndarray:
    """Full convolution of seq with itself using FFT (real inputs)."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    return conv


def _fast_score(seq: np.ndarray):
    """Return (score, conv, max_conv) for a sequence."""
    conv = _conv_self(seq)
    max_v = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, max_v
    score = 2.0 * len(seq) * max_v / (s * s)
    return float(score), conv, max_v


def _greedy_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve (approximately) the linear program:
        max   sum g_j
        s.t.  (f * g)_k <= M   for all k,   g_j >= 0,   g_j <= 1000.
    The implementation uses a simple greedy packing that respects the
    interval constraints of the convolution matrix.
    """
    n = f.shape[0]
    # remaining capacity for each convolution index
    r = np.full(2 * n - 1, M, dtype=np.float64)
    g = np.zeros_like(f)
    # allocate mass to indices with smallest f_i first
    order = np.argsort(f)          # ascending (small coefficient -> large allowance)
    for i in order:
        fi = f[i]
        if fi <= 0.0:
            # zero coefficient imposes no restriction; fill up to the box limit
            delta = 1000.0
        else:
            lo = i
            hi = i + n               # exclusive
            min_rem = r[lo:hi].min()
            allowed = min_rem / fi
            if allowed <= 0.0:
                delta = 0.0
            else:
                delta = allowed
                if delta > 1000.0:
                    delta = 1000.0
        if delta > 0.0:
            g[i] = delta
            if fi > 0.0:
                r[i:i + n] -= fi * delta
    return g

            cand[i], cand[j] = cand[j], cand[i]

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score, _, _ = _fast_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()

    # ------------------------------------------------------------
    # Safety post‑process: ensure a tiny positive mass if necessary.
    # ------------------------------------------------------------
    if best_seq.sum() < 0.01:
        best_seq[0] = 0.01

    return [float(x) for x in best_seq]
