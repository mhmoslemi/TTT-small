    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Initial sequence (parent if available, otherwise a classic series)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = a.size
    else:
        n = 5582                      # fallback length
        a = _sqrt_series(n)
    # normalise sum to the conventional √(2n) – keeps numbers moderate
    target_sum = math.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    np.clip(a, 0.0, 1000.0, out=a)
    a = a * (target_sum / a.sum())

    # ------------------------------------------------------------------
    # Helper to evaluate the exact objective used by the official scorer
    # ------------------------------------------------------------------
    def _score_vec(x):
        conv = _conv_self(x)
        return 2.0 * n * conv.max() / (x.sum() * x.sum())

    # best solution seen so far
    best = a.copy()
    best_score = _score_vec(best)

    cur = a.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # Simulated‑annealing + sub‑gradient descent on the max‑lag term
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    max_iters = 300000
    it = 0
    lr = 0.3                     # learning rate for gradient moves
    while time.monotonic() < deadline and it < max_iters:
        # decide which kind of perturbation to apply
        if rng.random() < 0.38:                     # sub‑gradient step
            conv = _conv_self(cur)
            k = int(conv.argmax())
            M = conv.max()
            partner = k - np.arange(n)              # index paired with each i
            mask = (partner >= 0) & (partner < n)
            g = np.zeros_like(cur)
            g[mask] = cur[partner[mask]]            # a_{k-i}
            # sub‑gradient of 2*n*max(conv)/S^2 w.r.t a_i (scale omitted,
            # absorbed into lr)
            subgrad = g - (2.0 * M / cur.sum())
            cand = cur - lr * subgrad
        else:                                        # random exploratory move
            cand = cur.copy()
            op = rng.random()
            if op < 0.55:                            # mass transfer early→late
                src = rng.integers(0, n // 4)
                if cand[src] > 1e-12:
                    dst = rng.integers(3 * n // 4, n)
                    max_xfer = min(cand[src], 1000.0 - cand[dst])
                    if max_xfer > 0.0:
                        delta = max_xfer * rng.random() * 0.05
                        cand[src] -= delta
                        cand[dst] += delta
            elif op < 0.80:                          # block scaling
                width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
                start = rng.integers(0, max(1, n - width + 1))
                factor = float(math.exp(rng.normal(0.0, 0.12)))
                cand[start:start + width] *= factor
            else:                                    # additive Gaussian noise
                cand += rng.normal(0.0, 0.02, size=n)

        # clip + re‑normalise to keep the sum fixed (scale‑invariant objective)
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand = cand * (target_sum / cand.sum())
        new_score = _score_vec(cand)

        # acceptance rule (always accept improvement)
        if new_score < cur_score:
            cur, cur_score = cand, new_score
            if new_score < best_score:
                best, best_score = cand.copy(), new_score
        else:
            # occasional uphill move – linear temperature schedule
            T = 0.02 * (1.0 - it / max_iters) + 1e-7
            if rng.random() < math.exp(-(new_score - cur_score) / T):
                cur, cur_score = cand, new_score

        # modest learning‑rate decay and occasional momentum reset
        if it % 12000 == 0 and it > 0:
            lr *= 0.7
        it += 1

    return [float(v) for v in best]
