    # Use compute_score (FFT) to evaluate.

        nonlocal best, best_score
        score = _ac1_score(arr, n, L)
        if score < best_score:
            best = arr.copy()
            best_score = score
        return score

    # base candidates
    cand_seq = []
    if parent is not None:
        cand_seq.append(parent.copy())
    # binomial
    cand_seq.append(_gen_binomial(n))
    # sine shape

    for r in [0.9, 0.95, 0.98]:
        cand_seq.append(r**i)
    # maybe power law
    for p in [0.5, 0.75, 1.0]:
        seq = (i+1)**(-p)
        cand_seq.append(seq)

    for seq in cand_seq:
        seq = np.clip(seq, 0., 1000.)
        add_candidate(seq)

    # Set current = best
    cur = best.copy()
    cur_score = best_score

    # main loop
    temperature = 1.0
    decay = 0.9998
    # random move probabilities among types: block scaling, mass transfer, smoothing, random perturb.
    while time.monotonic() < deadline:
        # choose move type
        move_type = rng.integers(0,5)
        cand = cur.copy()
        if move_type == 0:
            # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0., 0.1))
            cand[start:start+width] *= factor
        elif move_type == 1:
            # mass transfer between two random positions
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                frac = rng.random() * 0.3
                delta = cur[i] * frac
                cand[i] -= delta
                cand[j] += delta
        elif move_type == 2:
            # smoothing: replace a point with average of neighbors
            idx = int(rng.integers(1, n-1))
            avg = (cur[idx-1] + cur[idx+1]) * 0.5
            cand[idx] = avg
        elif move_type == 3:
            # transfer from top contributors to random lower
            # compute conv max index
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]

            end_i = min(n-1, kmax)
            for i_idx in range(start_i, end_i+1):

                contrib[i_idx] = cur[i_idx] * cur[j_idx]
            total = contrib.sum()
            if total > 0:

                src = int(rng.choice(n, p=contrib/total))
                # choose dest as index with minimal product with its counterpart
                # compute potential contribution if moved
                # Use counterpart indices community
                # Let's pick dest randomly weighted inversely to contrib
                inv = (total - contrib) + 1e-12

                # move fraction of src to dst
                frac = rng.random() * 0.2
                delta = cur[src] * frac
                cand[src] -= delta
                cand[dst] += delta
        else: # move_type == 4
            # random perturbation
            idx = int(rng.integers(n))
            delta = rng.normal(0., cur[idx] * 0.05 + 1e-6)
            cand[idx] = max(0., cand[idx] + delta)

        # clipping bounds
        np.clip(cand, 0., 1000., out=cand)

        cand_score = _ac1_score(cand, n, L)
        # update best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # acceptance: similar as before
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
        temperature *= decay

        # occasional restart
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # Return best as list of floats
    return [float(x) for x in best]

import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound using FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # -------------------- problem size --------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256
        parent = None

    L = _next_pow2(2 * n - 1)

    # -------------------- initial candidates --------------------
    best = None
    best_score = float("inf")

    def consider(arr):
        nonlocal best, best_score
        arr = np.clip(arr, 0.0, 1000.0)
        score = _ac1_score(arr, n, L)
        if score < best_score:
            best = arr.copy()
            best_score = score
        return score

    # parent
    if parent is not None:
        consider(parent)

    # classic binomial
    consider(_gen_binomial(n))

    # sine half‑wave
    i = np.arange(n, dtype=np.float64)
    consider(np.sin(np.pi * (i + 0.5) / n))

    # power‑law families
    for p in (0.4, 0.5, 0.7, 1.0):
        consider((i + 1) ** (-p))

    # exponential families
    for r in (0.90, 0.95, 0.98):
        consider(r ** i)

    # Gaussian‑like bump (centered)
    mid = (n - 1) / 2.0
    for sigma in (n / 6.0, n / 4.0):
        consider(np.exp(-0.5 * ((i - mid) / sigma) ** 2))

    # -------------------- start local search --------------------
    cur = best.copy()
    cur_score = best_score
    temperature = 1.0
    decay = 0.9995

    # deterministic active‑set smoothing (quickly reduces the peak)
    while time.monotonic() < deadline:
        # compute current convolution & max index
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        kmax = int(np.argmax(conv))
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        contrib = cur[idx_range] * cur[kmax - idx_range]
        if contrib.size == 0 or contrib.max() == 0:
            break
        src = int(idx_range[np.argmax(contrib)])
        # choose a destination outside the active window if possible
        all_idx = np.arange(n, dtype=np.int64)
        mask = (all_idx < i_start) | (all_idx > i_end)
        if mask.any():
            dst = int(rng.choice(all_idx[mask]))
        else:
            dst = int(rng.integers(n))
        # move a modest fraction
        frac = rng.random() * 0.25
        delta = cur[src] * frac
        cur[src] -= delta
        cur[dst] += delta
        np.clip(cur, 0.0, 1000.0, out=cur)
        cur_score = _ac1_score(cur, n, L)
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            if best_score < 1.50274349:
                return [float(x) for x in best]
        # stop deterministic phase after a few thousand steps
        if cur_score <= best_score * 0.9995:
            break

    # -------------------- stochastic walk --------------------
    while time.monotonic() < deadline:
        move_type = rng.integers(0, 5)
        cand = cur.copy()

        if move_type == 0:                     # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand[start:start + width] *= factor

            i = int(rng.integers(n))
            j = int(rng.integers(n))

                cand[j] += delta

            cand[idx] = 0.5 * (cand[idx - 1] + cand[idx + 1])

                mask = (all_idx < i_start) | (all_idx > i_end)
                if mask.any():
                    dst = int(rng.choice(all_idx[mask]))

                    dst = int(rng.integers(n))
                frac = rng.random() * 0.30
                delta = cur[src] * frac
                cand[src] -= delta
                cand[dst] += delta

            idx = int(rng.integers(n))
            delta = rng.normal(0.0, max(1.0, cur[idx] * 0.05))
            cand[idx] = max(0.0, cand[idx] + delta)

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _ac1_score(cand, n, L)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # occasional restart from best
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    return [float(x) for x in best]
