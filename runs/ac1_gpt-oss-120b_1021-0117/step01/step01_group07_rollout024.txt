import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) with a low AC1 bound.
    The routine builds a few deterministic candidates from the available parent
    sequence, keeps the best one and then performs cheap hill‑climbing swaps
    until the time budget expires.
    """
    rng = np.random.default_rng(seed)
    # leave a safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----------------------------------------------------------------------
    # 1. Load the parent construction (if present) and make a safe fallback.
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:
        # a simple fallback – 256 equal entries (sum > 0)
        parent = np.full(256, 1.0, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)          # respect the evaluator’s bounds
    n = int(parent.size)

    # ----------------------------------------------------------------------
    # 2. Build a small portfolio of deterministic candidates.
    # ----------------------------------------------------------------------
    candidates = []

    # 2a. The raw (clipped) parent – as‑is.
    candidates.append(parent.copy())

    # 2b. Ascending sort – small values first, large values last.
    candidates.append(np.sort(parent))

    # 2c. Descending sort – the opposite order (often worse, kept for safety).
    candidates.append(np.sort(parent)[::-1])

    # 2d. Interleaved low/high pattern.
    s = np.sort(parent)
    low = s[: (n + 1) // 2]            # smaller half (ascending)
    high = s[(n + 1) // 2:][::-1]      # larger half (descending)
    inter = np.empty(n, dtype=np.float64)
    inter[0::2] = low[: len(inter[0::2])]
    inter[1::2] = high[: len(inter[1::2])]
    candidates.append(inter)

    # ----------------------------------------------------------------------
    # 3. Pick the best among the deterministic candidates.
    # ----------------------------------------------------------------------
    best = None
    best_score = np.inf
    for cand in candidates:
        # evaluate – conversion to list is required by the provided evaluator
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = cand.copy()
    # If for any reason none were valid, fall back to a uniform vector.
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # 4. Simple hill‑climbing: random swaps that keep the sum unchanged.
    #    Accept only improvements (deterministic descent).
    # ----------------------------------------------------------------------
    current = best.copy()
    current_score = best_score
    target = 1.50274349         # stop early if we reach the published target

    # A very cheap early‑exit check.
    if best_score <= target:
        return [float(v) for v in best]

    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:

        # perform the swap in‑place
        current[i], current[j] = current[j], current[i]

        sc = float(evaluate_sequence(current.tolist()))
        if np.isfinite(sc) and sc < current_score:

            current_score = sc
            if sc < best_score:
                best_score = sc
                best = current.copy()
                if best_score <= target:

    # 5. Return the best sequence found (as plain Python list).
    # ----------------------------------------------------------------------
