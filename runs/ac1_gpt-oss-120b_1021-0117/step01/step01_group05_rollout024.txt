import numpy as np, time, math

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1,n):
        a[i] = a[i-1] * (2*i-1) / (2*i)
    return a

def _fast_score(seq):
    n = seq.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f*f, n=m)[:2*n-1]
    M = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float('inf'), conv, M
    return 2.0 * n * M / (s*s), conv, M

import numpy as np
import time
import math

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a decreasing non‑negative sequence a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, max value) using a power‑of‑two FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    mx = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, mx
    return 2.0 * n * mx / (s * s), conv, mx
