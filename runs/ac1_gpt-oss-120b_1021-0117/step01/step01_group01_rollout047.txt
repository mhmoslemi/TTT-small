def proj_simplex(v):
    # Project onto simplex sum(v)=1, v>=0
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time

import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 via recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a

def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using one FFT (full length)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv

        new_score, new_conv = _score_and_conv(new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur, conv, cur_score = new, new_conv, new_score
            if new_score < best_score:
                best, best_score = new.copy(), new_score

        # ----- 4.3 Temperature schedule & occasional perturbation -----
        T = max(T * cooling, 1e-12)

        if it % 1024 == 0:
            # occasional global shake: blend with a small Dirichlet draw
            beta = 0.03
            rand = rng.dirichlet(np.ones(n) * 0.5)
            shaken = (1.0 - beta) * cur + beta * rand
            shaken = shaken / shaken.sum()
            shaken_score, shaken_conv = _score_and_conv(shaken)
            if shaken_score < cur_score:
                cur, conv, cur_score = shaken, shaken_conv, shaken_score
                if shaken_score < best_score:
                    best, best_score = shaken.copy(), shaken_score

    # ------------------------------------------------------------------
    # 5. Return the best candidate (ensuring evaluator safety)
    # ------------------------------------------------------------------
    final = [float(x) for x in best]
    try:
        evaluate_sequence(final)   # trigger any hidden checks
    except Exception:
        # fall back to the initial binomial shape if something went wrong
        fallback = _generate_binomial_seq(n)
        fallback = fallback / fallback.sum()
        final = [float(x) for x in fallback]
    return final
