            src, dst = i, j
        else:
            src, dst = j, i

        if current[src] < delta:          # avoid negativity
            continue
        cand = current.copy()
        cand[src] -= delta
        cand[dst] += delta

        sc = _score(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc
            current = cand
            avg = current.mean()          # recompute average for next step

    # ------------------------------------------------------------------ #
    # 3. Final validation with the official evaluator.
    # ------------------------------------------------------------------ #
    # Convert to Python list of floats (the evaluator expects list)
    result = [float(v) for v in best_seq]
    # One final sanity check – if something went wrong, fall back to uniform
    try:
        final_score = evaluate_sequence(result)
        if not np.isfinite(final_score):
            raise Exception
    except Exception:
        result = [1.0] * n
    return result
