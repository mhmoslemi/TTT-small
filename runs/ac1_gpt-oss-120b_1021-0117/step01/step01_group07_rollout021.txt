def lp_step(a, M):
    import cvxpy as cp
    from scipy import sparse
    n = a.shape[0]
    # Build sparse A via dia_matrix
    diags = np.tile(a, (n, 1)).T  # shape (n, n) but we need as list of arrays
    # We need each diagonal offset d with values a[d] repeated
    data = np.empty((n, n), dtype=np.float64)
    # but easier: data[i, :] = a[i] (for each i) as each diagonal's values
    # Build using scipy.sparse.diags? but that expects diag arrays of appropriate length.
    # Use list comps:
    diags_data = []
    offsets = []
    for d in range(n):
        diag_vals = np.full(n, a[d])
        diags_data.append(diag_vals)
        offsets.append(d)
    A = sparse.dia_matrix((diags_data, offsets), shape=(2*n-1, n))
    # Convert to CSC for cvxpy
    A_csc = A.tocsc()
    g = cp.Variable(n)
    constraints = [g >= 0, A_csc @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    # Solve with small max_iters and warm start? CP has parameters:
    prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-6, reltol=1e-6, feastol=1e-6, verbose=False)
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
        g_val = g.value
        return g_val
    else:
        return None

       return w

       best = cat.copy()
       best_score = cat_score

   # Function to evaluate and update best

        nonlocal best, best_score

            return
        s = cand.sum()
        if s < 1e-12:
            return

        return False

   # Random transfer function

      prob = rng.random()

      cand[i] -= delta

      return cand

def _max_gradient(seq):
    conv = np.convolve(seq, seq)
    M = conv.max()
    max_idx = np.flatnonzero(conv == M)
    indicator = np.zeros_like(conv)
    indicator[max_idx] = 1.0
    grad = np.convolve(indicator, seq[::-1])[len(seq)-1:len(seq)-1+len(seq)]
    return grad

def _random_transfer(seq, rng):
    n = seq.size
    i, j = rng.choice(n, size=2, replace=False)
    if seq[i] == 0.0:
        return seq
    # choose fraction random in (0, 0.5)
    frac = rng.random() * 0.5
    delta = seq[i] * frac
    cand = seq.copy()
    cand[i] -= delta
    cand[j] += delta
    return _normalize(cand)

def _lp_step(seq, M, cp, sparse):
    # if cp or sparse not available, return None
    if cp is None or sparse is None:
        return None
    n = seq.size
    # Build dia matrix
    # We need to provide data as list of ndarrays sized n each.
    data = [np.full(n, seq[d]) for d in range(n)]
    offsets = list(range(n))
    A = sparse.dia_matrix((data, offsets), shape=(2*n-1, n)).tocsc()
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-7, reltol=1e-7, feastol=1e-7, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=2500, verbose=False)
        except Exception:
            return None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
        return None
    g_val = g.value
    if g_val is None:
        return None
    # ensure nonnegative
    g_val = np.maximum(g_val, 0.0)
    # normalize
    s = g_val.sum()
    if s <= 0:
        return None
    return g_val / s

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------
def _catalan_weights(n: int) -> np.ndarray:
    """Catalan–type weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _normalize(v: np.ndarray) -> np.ndarray:
    s = v.sum()
    return v if s <= 0 else v / s


def _evaluate(v: np.ndarray) -> float:
    """Thin wrapper around the supplied evaluator."""
    return float(evaluate_sequence(v.tolist()))


def _max_convolution(v: np.ndarray):
    conv = np.convolve(v, v)
    return conv.max(), conv


def _max_gradient(v: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """Gradient of the max‑convolution w.r.t. each entry of v."""
    idx = np.flatnonzero(conv == max_val)
    ind = np.zeros_like(conv)
    ind[idx] = 1.0
    # gradient = convolution of the indicator with reversed v
    grad = np.convolve(ind, v[::-1])[len(v) - 1 : len(v) - 1 + len(v)]
    return grad


def _mirror_step(v: np.ndarray, eps: float) -> np.ndarray:
    """Exponentiated‑gradient step using the exact max‑gradient."""
    max_val, conv = _max_convolution(v)
    grad = _max_gradient(v, conv, max_val)
    gmax = grad.max()
    if gmax <= 0:
        return v
    scale = eps * (grad / gmax)
    new_v = v * np.exp(-scale)
    return _normalize(new_v)


def _random_transfer(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = v.size
    i, j = rng.choice(n, size=2, replace=False)
    if v[i] <= 0.0:
        return v
    frac = rng.random() * 0.5           # up to 50 % of donor mass
    delta = v[i] * frac
    cand = v.copy()
    cand[i] -= delta
    cand[j] += delta
    return _normalize(cand)


def _jitter(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    sigma = 0.02
    factor = np.exp(rng.normal(0.0, sigma, size=v.shape))
    return _normalize(v * factor)


def _smooth(v: np.ndarray) -> np.ndarray:
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    sm = np.convolve(v, kernel, mode="same")
    return _normalize(sm)


def _lp_step(v: np.ndarray, bound: float):
    """Linear‑programming step: maximise 1ᵀg s.t. f·g ≤ bound, g≥0."""
    try:
        import cvxpy as cp
        from scipy import sparse
    except Exception:
        return None

    n = v.size
    # Build a sparse Toeplitz‑like matrix with constant diagonals = v[d]
    data = [np.full(n, v[d]) for d in range(n)]
    offsets = list(range(n))
    A = sparse.dia_matrix((data, offsets), shape=(2 * n - 1, n)).tocsc()

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(
            solver=cp.ECOS,
            warm_start=True,
            max_iters=2000,
            abstol=1e-7,
            reltol=1e-7,
            feastol=1e-7,
            verbose=False,
        )
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=2500, verbose=False)
        except Exception:
            return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE):
        return None
    g_val = g.value
    if g_val is None:
        return None
    g_val = np.maximum(g_val, 0.0)
    s = g_val.sum()
    return None if s <= 0 else _normalize(g_val)
