def _targeted_transfer(a, conv, rng):
    n = a.size
    # find lag of max conv
    lag = int(conv.argmax())
    # ensure lag <= n-1; if not, map symmetrical
    if lag > n - 1:
        lag = 2 * (n - 1) - lag  # symmetry
    # contributions
    left = a[:lag+1]
    right = a[lag::-1]
    contrib = left * right
    idx = int(contrib.argmax())
    jdx = lag - idx
    # choose source as larger coefficient among idx and jdx
    if a[idx] >= a[jdx]:
        src = idx
    else:
        src = jdx
    # destination early region
    early_end = max(1, n // 20)  # at least 1
    dst = int(rng.integers(0, early_end))
    # ensure src != dst ; if same, choose different
    if src == dst:
        # try other early region
        alt_dst = early_end + int(rng.integers(0, n - early_end))
        dst = alt_dst
    # compute delta
    max_transfer = min(a[src], 1000.0 - a[dst])
    if max_transfer <= 0:
        return None  # no move possible
    # random factor up to 20% of source mass
    delta = max_transfer * rng.random() * 0.2
    if delta <= 1e-12:
        return None
    cand = a.copy()
    cand[src] -= delta
    cand[dst] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _block_scale(a, rng):
    n = a.size
    # width between n/200 and n/10
    min_w = max(1, n // 200)
    max_w = max(2, n // 10)
    width = int(rng.integers(min_w, max_w+1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    factor = float(math.exp(rng.normal(0.0, 0.12)))  # multiplicative factor
    cand = a.copy()
    # multiply block directly
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)  # 1 sec margin

    if 'height_sequence_1' in globals():
        a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = a0.size
    else:
        n = 5582
        a0 = None

    # fallback to sqrt series
    sqrt_series = _sqrt_series(n)
    # decide start
    cur = sqrt_series
    cur_score = _score_cur(cur) # compute via conv
    best = cur.copy()
    best_score = cur_score

    if a0 is not None:
        parent_score = _score_cur(a0)
        if np.isfinite(parent_score) and parent_score < best_score:
            cur = a0.copy()
            cur_score = parent_score
            best = cur.copy()
            best_score = cur_score

    # Pre-compute conv_cur for use in targeted move
    conv_cur = _conv_self(cur)

    # Temperature schedule
    temp_start = max(0.01, 0.05 * best_score)
    # degrade factor per iteration
    cooling = 0.9997  # close to 1

    it = 0
    while time.monotonic() < deadline:
        # choose op
        p = rng.random()
        cand = None
        if p < 0.45:
            cand = _targeted_transfer(cur, conv_cur, rng)
        elif p < 0.70:
            cand = _block_scale(cur, rng)
        elif p < 0.90:
            cand = _random_transfer(cur, rng)
        else:

                best_score = cand_score

        it += 1

        if best_score < 1.50274349:

import numpy as np, time, math

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (next power of two)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv

def _score_and_conv(a: np.ndarray):
    """Return (score, full convolution) for a."""
    n = a.size
    conv = _conv_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf, conv
    score = 2.0 * n * conv.max() / (s * s)
    return float(score), conv

def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a

def _targeted_transfer(cur: np.ndarray, conv_cur: np.ndarray, rng) -> np.ndarray | None:
    n = cur.size
    lag = int(conv_cur.argmax())
    if lag > n - 1:
        lag = 2 * (n - 1) - lag        # symmetry of linear convolution
    # contributions for this lag
    left = cur[: lag + 1]
    right = cur[lag :: -1]
    contrib = left * right
    idx = int(contrib.argmax())
    jdx = lag - idx
    src = idx if cur[idx] >= cur[jdx] else jdx

    # destination: a very early index where the reversed partner is tiny
    early_end = max(1, n // 20)
    dst = int(rng.integers(0, early_end))
    if dst == src:
        dst = early_end + int(rng.integers(0, n - early_end))
    max_transfer = min(cur[src], 1000.0 - cur[dst])
    if max_transfer <= 0.0:
        return None
    delta = max_transfer * rng.random() * 0.2   # up to 20% of transferable mass
    if delta <= 1e-12:
        return None
    cand = cur.copy()
    cand[src] -= delta
    cand[dst] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _block_scale(cur: np.ndarray, rng) -> np.ndarray:
    n = cur.size
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(math.exp(rng.normal(0.0, 0.12)))
    cand = cur.copy()
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _random_transfer(cur: np.ndarray, rng) -> np.ndarray | None:
    n = cur.size
    early = n // 4
    late = 3 * n // 4
    src = int(rng.integers(0, early))
    dst = int(rng.integers(late, n))
    if cur[src] <= 1e-12:
        return None
    max_transfer = min(cur[src], 1000.0 - cur[dst])
    if max_transfer <= 0.0:

    cand = cur.copy()
    cand[src] -= delta
    cand[dst] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _add_noise(cur: np.ndarray, rng) -> np.ndarray:
    cand = cur + rng.normal(0.0, 0.02, size=cur.size)
    np.clip(cand, 0.0, 1000.0, out=cand)
