def gradient_at_k(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    # j range from start to end inclusive
    # compute indices j = start...end
    # compute reversed indices: k - j
    # vectorized:
    j = np.arange(start, end + 1)
    g[j] = 2.0 * a[k - j]
    return g
```

def project_simplex(v):
    # assume v is 1D np.ndarray
    # returns projection onto simplex sum=1, v>=0.
    # Implementation from Duchi et al.
    n = v.shape[0]

    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

# ----- helper utilities ---------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry c_k = (a * a)[k] with respect to a.
    For non‑negative a the sub‑gradient is
        g_j = 2 * a_{k-j}   if 0 <= k-j < n, else 0
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        j = np.arange(start, end + 1, dtype=int)
        g[j] = 2.0 * a[k - j]
    return g


def _global_jitter(a: np.ndarray, rng: np.random.Generator, sigma: float = 0.018) -> np.ndarray:
    """Small multiplicative noise followed by simplex projection."""
    jitter = np.exp(rng.normal(0.0, sigma, size=a.shape))
    cand = a * jitter
    return _project_simplex(cand)


def _score_of(a: np.ndarray) -> float:
    """Exact objective value using the provided evaluator."""
    # the evaluator is scaling‑invariant; we can compute the score directly:
    conv = np.convolve(a, a)
    return float(2.0 * a.shape[0] * conv.max() / (a.sum() ** 2))
