import numpy as np, time

def conv_self_fft(a):
    n = a.size
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    return np.fft.irfft(fa*fa, n=m)[:2*n-1]

def score(a):
    b = conv_self_fft(a)
    maxb = b.max()
    s = a.sum()
    return 2.0 * a.size * maxb / (s*s) if s>0 else np.inf

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)  # keep margin
    # start vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 5582
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1,n):
            a[i] = a[i-1] * (2.0*i-1) / (2.0*i)   # compute sqrt series
    # clip
    np.clip(a, 0.0, 1000.0, out=a)
    # normalize sum to 1.0
    total = a.sum()
    if total == 0:
        a[0] = 1.0
        total = 1.0
    a = a / total

    best = a.copy()
    best_score = score(best)

    # also compute baseline of sqrt if parent else not

    # loops
    while time.monotonic() < deadline:
        # Determine which move to try
        move_type = rng.random()
        cand = best.copy()
        if move_type < 0.6:
            # gradient-based transfer
            b = conv_self_fft(cand)
            peak = int(np.argmax(b))
            n = cand.size
            # build g
            g = np.zeros_like(cand)
            i_start = max(0, peak - (n-1))
            i_end = min(n-1, peak)
            # indices
            idx = np.arange(i_start, i_end+1)
            j = peak - idx
            g[i_start:i_end+1] = cand[j]
            # compute product contributions
            contrib = cand * g
            src_idx = int(np.argmax(contrib))
            dst_idx = int(np.argmin(contrib))
            # ensure source has mass to give
            if cand[src_idx] > 1e-12 and cand[dst_idx] < 1000.0:
                max_transfer = min(cand[src_idx], 1000.0 - cand[dst_idx])
                if max_transfer > 0:
                    delta = max_transfer * rng.random() * 0.2  # up to 20% of transferable
                    cand[src_idx] -= delta
                    cand[dst_idx] += delta
        elif move_type < 0.85:
            # block scaling preserving sum by renormalization
            n = cand.size
            # choose random block length
            min_len = max(2, n // 200)
            max_len = max(min_len+1, n // 12)
            width = int(rng.integers(min_len, max_len+1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.15)))
            cand[start:start+width] *= factor
            # Clip to bounds before renorm
            np.clip(cand, 0.0, 1000.0, out=cand)
            # Renorm sum to 1
            cand = cand / cand.sum()
        else:
            # small additive noise
            noise = rng.normal(0.0, 0.01, size=cand.shape)
            cand += noise
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand = cand / cand.sum()
        # evaluate
        sc = score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
        # occasional random restart to avoid stagnation? maybe not needed.

    # Ensure final output is list of floats and in correct range
    # Even though we kept sum=1, convert to list.
    out = [float(max(0.0, min(1000.0, v))) for v in best]
    return out

import numpy as np, time, math

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (next power‑of‑two length)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact score used by the official evaluator (no clipping)."""
    n = a.size
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _norm_one(a: np.ndarray) -> np.ndarray:
    """Scale vector so that its sum equals 1 (preserves non‑negativity)."""
    s = a.sum()
    if s == 0:
        a[0] = 1.0
        s = 1.0
    a /= s
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score non‑negative step function within the time budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # Initialise: use the supplied parent if available, otherwise a cheap
    # decreasing exponential shape.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0, out=a)
    else:
        n = 5582
        a = np.exp(-np.arange(n, dtype=np.float64) / (n / 10.0))
        a = np.clip(a, 0.0, 1000.0, out=a)

    a = _norm_one(a)                     # keep sum = 1 (scale‑invariant)
    n = a.size
    best = a.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # Helper to recompute the convolution centre and gradient‑like weights.
    # ------------------------------------------------------------------
    def _peak_weights(vec):
        b = _conv_fft(vec)
        p = int(np.argmax(b))
        maxb = b[p]
        # build weight array g where g[i] = vec[p-i] if index valid else 0
        g = np.zeros_like(vec)
        i0 = max(0, p - (n - 1))
        i1 = min(n - 1, p)
        idx = np.arange(i0, i1 + 1)
        g[idx] = vec[p - idx]
        return p, maxb, g

    # ------------------------------------------------------------------
    # Main stochastic optimisation loop.
    # ------------------------------------------------------------------
    early_cut = n // 4
    late_cut = 3 * n // 4

    while time.monotonic() < deadline:
        cand = best.copy()
        op = rng.random()

        # --------------------------------------------------------------
        # 1) Gradient‑inspired mass transfer targeting the current peak.
        # --------------------------------------------------------------
        if op < 0.45:
            peak, maxb, g = _peak_weights(cand)
            # gradient proxy: larger g[i] makes a[i] more harmful for the peak
            grad = g - 2.0 * maxb * cand                # sum == 1 => denominator omitted
            pos = grad > 0
            neg = grad < 0
            if pos.any() and neg.any():
                src = int(np.argmax(np.where(pos, grad, -np.inf)))
                dst = int(np.argmin(np.where(neg, grad, np.inf)))
                if cand[src] > 0.0 and cand[dst] < 1000.0:
                    mx = min(cand[src], 1000.0 - cand[dst])
                    if mx > 0.0:
                        delta = mx * rng.random() * 0.25   # up to 25 % of transfer‑able mass
                        cand[src] -= delta
                        cand[dst] += delta

        # --------------------------------------------------------------

            if cand[src] > 1e-12 and cand[dst] < 1000.0:

                if mx > 0.0:
                    delta = mx * rng.random() * 0.10   # up to 10 %
                    cand[src] -= delta
                    cand[dst] += delta

        # 3) Block scaling (followed by renormalisation to sum‑1).

        elif op < 0.95:

            max_len = max(min_len + 1, n // 12)
            width = int(rng.integers(min_len, max_len + 1))
            start = int(rng.integers(0, n - width + 1))

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list of floats.
    # ------------------------------------------------------------------
    out = [float(v) for v in best]
    return out
