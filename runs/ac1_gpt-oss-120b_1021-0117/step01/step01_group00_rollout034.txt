    # sort ascending so smallest first.

    # assign smallest values to edges
    res = np.empty(n, dtype=np.float64)
    if n >= 1:
        res[0] = vals_sorted[0]  # smallest
    if n >= 2:
        res[-1] = vals_sorted[1]  # second smallest
    # remaining values
    remaining = vals_sorted[2:]
    m = len(remaining)
    # We'll create pairs for interior positions i in [1, n-2]
    # Pair extremes: largest with smallest alternating across interior.
    # For interior, we can take remaining sorted descending: large_to_small = remaining[::-1]
    large_to_small = remaining[::-1]  # descending
    half = m // 2
    # assign largest half to left interior positions (starting at index 1)
    res[1:1+half] = large_to_small[:half]  # largest values
    # assign remaining (including smallest half) to right interior positions in reverse:
    # Right interior positions: indices n-1-half ... n-2 (excluding last element which is index n-1)
    right_vals = large_to_small[half:][::-1]  # smallest interior values
    # compute start index of right interior slice:
    # number of right interior slots = m - half = ceil(m/2)
    # Those occupy positions from n-1 - (m - half) to n-2 inclusive.
    # Let start = n - 1 - (m - half)
    start = n - 1 - (m - half)
    # This slice may include the index n-1 (last), but we already used it; ensure we stop before n-1.
    res[start:n-1] = right_vals
    # If m is odd, there is a middle element that hasn't been placed.
    # For odd m, half = floor(m/2), remaining slot is interior position at index 1+half (or start-1)
    if m % 2 == 1:
        middle_val = large_to_small[half]  # middle value
        # Identify the unfilled interior index:
        # positions not filled: we filled left positions from 1 to 1+half-1 inclusive.
        # Then we have a gap at index left_mid = 1+half
        # Or we might fill right side leaving gap.
        # We'll fill the index at (n-1)//2 maybe.
        idx = (n // 2)  # integer centre
        # ensure not overwritten edges
        if res[idx] == 0:  # not yet set

import time
import numpy as np

def _power_law_sequence(n: int, alpha: float) -> np.ndarray:
    """Return a decreasing positive sequence a_i = (i+1)^(-alpha)."""
    # i+1 starts at 1, so the first term is 1.0
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-alpha)
    return np.clip(seq, 0.0, 1000.0)
