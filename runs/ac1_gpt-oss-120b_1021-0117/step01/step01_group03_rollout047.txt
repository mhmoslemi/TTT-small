def _conv_self(a):
    n = a.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    # real FFT: rfft length m, product, irfft
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[:2*n-1]
    return conv
```

def gradient_descent(a, deadline):
    best = a.copy()
    best_score = _score(best)
    conv_best = _conv_self(best)
    n = best.size
    while time.monotonic() < deadline:
        k_star = int(np.argmax(conv_best))
        # compute gradient vector
        g = np.zeros_like(best)
        imin = max(0, k_star - (n-1))
        imax = min(k_star, n-1)
        idx = np.arange(imin, imax+1)
        g[idx] = 2.0 * best[k_star - idx]
        # source: max g among idx (or all)
        src = int(idx[np.argmax(g[idx])])
        # destination: random index where g==0
        zero_inds = np.where(g == 0)[0]
        if zero_inds.size == 0:
            dst = int(rng.integers(0,n))
        else:
            dst = int(rng.choice(zero_inds))
        # ensure src != dst
        if src == dst:
            dst = int((src + 1 + rng.integers(0, n-1)) % n)
        max_transfer = min(best[src], 1000.0 - best[dst])
        if max_transfer <= 1e-12:
            # add small noise across vector
            cand = best + rng.normal(0.0, 0.001, size=n)
        else:
            delta = max_transfer * rng.random() * 0.1
            cand = best.copy()
            cand[src] -= delta
            cand[dst] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            conv_best = _conv_self(best)
        # else optionally try small random noise acceptance (not required)
        # optionally after many non-improvements, break and return.
    return best, best_score

import numpy as np

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective (scale‑invariant)."""
    n = a.size
    conv = _conv_self(a)
    maxb = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * maxb / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a
