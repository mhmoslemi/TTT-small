def _conv_self(a, m):
    fa = np.fft.rfft(a, n=m)
    return np.fft.irfft(fa*fa, n=m)[:2*a.size-1]

def _eval_score(a, m):
    n = a.size
    conv = _conv_self(a, m)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

import numpy as np, time, math

def _conv_self(a, m):
    fa = np.fft.rfft(a, n=m)
    return np.fft.irfft(fa*fa, n=m)[:2*a.size-1]

def _eval_score(a, m):
    n = a.size
    conv = _conv_self(a, m)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s*s)

    def try_candidate(cand):
        nonlocal best, best_score
        sc = _eval_score(cand, m)
        if sc < best_score:
            best = cand
            best_score = sc

    # Precompute index arrays
    idx = np.arange(n, dtype=np.float64)

    uniform = np.ones(n, dtype=np.float64)

    for r in r_vals:
        geom = np.power(r, idx)
        for p in p_vals:
            cand = (1-p) * uniform + p * geom
            try_candidate(cand)

    # V shape geometric
    for r in r_vals:
        geomV = np.power(r, idx_min)
        for p in p_vals:
            cand = (1-p) * uniform + p * geomV
            try_candidate(cand)

    # Power law left
    alphas = np.linspace(0.0, 2.0, 25)
    for alpha in alphas:
        powl = np.power(idx+1, -alpha)
        for p in p_vals:

    # Power law V
    for alpha in alphas:

        for p in p_vals:

    # local search loop
    while time.monotonic() < deadline:
        mode = rng.random()
        cand = best.copy()
