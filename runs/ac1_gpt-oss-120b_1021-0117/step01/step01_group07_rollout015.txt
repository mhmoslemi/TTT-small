def project_simplex(v):
    # v is np.ndarray 1D
    # returns projection onto {x >= 0, sum x = 1}
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho: largest i s.t. u_i - cssv_i/(i+1) > 0
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all non-positive, return uniform or zeros? We could return uniform distribution.
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def proj_simplex(v):
    # Project onto simplex {x>=0, sum=1}
    # Implementation as reference.
    # Input: 1D array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all nonpositive
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]

import time
import numpy as np

# ---------- helper utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find rho = max { i | u_i - cssv_i / i > 0 }
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Convolution a*a using real FFT, returns full length 2*n-1 array."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """Generate a small list of plausible start vectors (sum=1)."""
    cand = []

    # 1. parent if supplied
    if parent_arr is not None and parent_arr.size == n:

        s = p.sum()

            cand.append(p / s)

    # 2. Catalan shape
    w = _catalan_weights(n)

    idx = np.arange(n, dtype=np.float64)

    for p in (0.5, 1.0, 2.0):

        cand.append(v / v.sum())

    return cand

            return best_vec.tolist()

    # ---- sub‑gradient descent on f(a)=max_k (a*a)_k (sum fixed to 1) ----
    a = best_vec.copy()
    iter_no_improve = 0
    alpha0 = 0.8                # initial step magnitude
    min_alpha = 1e-4

    while time.monotonic() < deadline:
        # compute full convolution and locate maximal index
        conv = _conv_self_fft(a, n, fft_len)
        max_b = float(conv.max())
        k = int(conv.argmax())

        # objective (sum(a)=1) -> score = 2*n*max_b
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            iter_no_improve += 1

        # gradient of the active term b_k = a * a evaluated at k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            g[idx] = 2.0 * a[k - idx]

        # adaptive step size (diminishing)
        t = max(1, iter_no_improve + 1)
        alpha = max(min_alpha, alpha0 / np.sqrt(t))

        # projected sub‑gradient update
        cand = a - alpha * g
        cand = _project_onto_simplex(cand)

        # occasional random jitter to escape plateaus
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.015, size=n))
            cand = cand * jitter
            cand = _project_onto_simplex(cand)

        a = cand

        # guard against pathological stagnation
        if iter_no_improve > 800:
            # restart from current best with a small random perturbation
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            a = best_vec * perturb
            a = _project_onto_simplex(a)
            iter_no_improve = 0

    # fallback: if nothing improved, return the best we have
    return best_vec.tolist()
