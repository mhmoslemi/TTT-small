def _fast_score(seq):
    n = seq.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f*f, n=m)[:2*n-1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0:
        return float('inf'), conv, max_val
    score = 2.0 * n * max_val / (s*s)
    return float(score), conv, max_val
```

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: a decreasing positive seed (binomial coefficients)
# ----------------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    """Return a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ----------------------------------------------------------------------
# Helper: fast score + convolution via FFT (linear‑time up to log factor)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, max(conv))."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val
