import time
import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4**i, i=0..n-1, decreasing positive sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a

def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of linear convolution a*a using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())

def _score_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact score used by the authoritative evaluator (sum is 1)."""
    max_b = _conv_max_fft(a, fft_len)
    return float(2 * n * max_b)          # sum(a) = 1 → denominator = 1

    deadline = start + max(0.0, float(budget_s) - 1.0)   # leave a second margin

    #  Problem size
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))

        n = 256

    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    #  Starting point – normalized binomial sequence (good baseline)

    a = _generate_binomial_seq(n)
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()

    else:

    cur = a.copy()

    best = cur.copy()
