import time
import numpy as np

def _max_conv_fft(seq: np.ndarray):
    """Return (max_convolution_value, argmax_index) using a real FFT."""
    n = seq.shape[0]
    # Length for linear convolution = 2n-1, next power of two for FFT efficiency
    N = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, N)
    conv = np.fft.irfft(fft * fft, N)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives – ignore them
    conv = np.real(conv)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Hill‑climbing / mass‑transfer search for a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # initial sequence (use the provided parent if available)
    parent = globals().get("height_sequence_1")
    if parent is not None:
        a = np.asarray(parent, dtype=np.float64).copy()
    else:
        a = None

    if a is None or a.size == 0:
        n = 5000
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = int(a.shape[0])

    # enforce feasibility and normalise the sum to 1 (scale does not matter)
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total

    # ------------------------------------------------------------------ #
    # a simple uniform fallback (score = 2)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    def _score(vec: np.ndarray) -> float:
        maxc, _ = _max_conv_fft(vec)
        s = vec.sum()
        return 2.0 * n * maxc / (s * s)

    best = a.copy()
    best_score = _score(best)

    # compare with uniform
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # ------------------------------------------------------------------ #
    # hill‑climbing parameters
    step_factor = 0.5          # fraction of source mass to move
    min_step_factor = 0.02
    max_step_factor = 0.9
    decay_no_improve = 0.995
    grow_improve = 1.01
    no_improve_counter = 0
    max_no_improve_before_shake = 200

    # time management
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    seq = best.copy()
    while time.monotonic() < deadline:
        # -------------------------------------------------------------- #
        # current worst convolution
        cur_max, cur_k = _max_conv_fft(seq)

        # -------------------------------------------------------------- #
        # identify the pair (i, j) that contributes most to cur_k
        i_start = max(0, cur_k - (n - 1))
        i_end = min(cur_k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = cur_k - i_range
        contrib = seq[i_range] * seq[j_range]
        idx_local = int(np.argmax(contrib))
        i_sel = int(i_range[idx_local])
        j_sel = int(j_range[idx_local])

        # pick the larger coordinate as the source of mass
        src = i_sel if seq[i_sel] >= seq[j_sel] else j_sel

        # -------------------------------------------------------------- #
        # destination: a (globally) minimal entry, breaking ties randomly
        min_val = seq.min()
        dest_cands = np.where(seq <= min_val + 1e-12)[0]
        dest = int(rng.choice(dest_cands))
        if dest == src:
            # pick any other index if clash
            dest = int(rng.integers(0, n))
            while dest == src:
                dest = int(rng.integers(0, n))

        # -------------------------------------------------------------- #
        # move a fraction of the source mass

        if cand_sum <= 0.0:

        cand /= cand_sum

        cand_score = _score(cand)

        # -------------------------------------------------------------- #

        if cand_score + 1e-14 < best_score:

    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:

    return [float(v) for v in final]
