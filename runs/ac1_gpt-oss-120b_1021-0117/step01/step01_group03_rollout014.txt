def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # Determine length
    if 'height_sequence_1' in globals():

    if parent is not None:

        if sc_parent < best_score:
            best = parent.copy()
            best_score = sc_parent
    # simple parameter families
    # power law

        if sc < best_score:
            best = seq
            best_score = sc

    for r in np.linspace(0.96, 0.9995, 13):
        if time.monotonic() >= deadline:
            break
        seq = r ** np.arange(n, dtype=np.float64)

        if sc < best_score:

import time
import numpy as np

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (fast, O(n log n))."""
    n = a.size
    # length needed for full convolution
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _ratio(a: np.ndarray) -> float:
    """Exact objective (2·n·max(conv)/sum^2). Assumes a already clipped."""
    n = a.size
    conv = _conv_self(a)
    m = conv.max()
    s = a.sum()
    # sum is guaranteed >0 in all our constructions
    return 2.0 * n * m / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence of length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a
