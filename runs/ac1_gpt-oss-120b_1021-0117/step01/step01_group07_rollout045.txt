def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 3.0)  # safe margin

    # length n:
    try: parent = np.asarray(height_sequence_1, dtype=np.float64); n = parent.size
    except Exception: n = 2000; parent = np.ones(n, dtype=np.float64)

    # Normalize sum to 1 for evaluation
    def norm(arr):
        s = arr.sum()
        if s <= 0:
            return arr
        return arr / s

    best_arr = None
    best_score = np.inf

    # Evaluate existing parent normalized
    if parent.ndim == 1 and parent.size>0:
        parent_norm = norm(parent.clip(0,1000))
        score = evaluate_sequence(parent_norm.tolist())
        if np.isfinite(score) and score < best_score:
            best_arr = parent_norm
            best_score = score

    # also evaluate uniform random start maybe
    # keep fallback as uniform?
    # Actually uniform seldom good; but we can use.

    # Evaluate a few deterministic simple initial families
    # Family 1: power law with beta offset
    i = np.arange(n, dtype=np.float64)
    for alpha in [0.5, 0.6, 0.7, 0.8, 1.0, 1.2, 1.5]:
        for beta in [0.0, 0.5, 1.0, 2.0]:
            a = (i + beta + 1e-12) ** (-alpha)
            a = norm(a)
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_arr = a.copy()
                best_score = score

    # Family 2: beta distribution shape: a_i = (i+1)^p * (n-i)^q
    # Actually use x = (i+0.5)/n; a = (x**p) * ((1-x)**q)
    for p in [0.1, 0.5, 1.0, 2.0, 3.0]:
        for q in [0.1, 0.5, 1.0, 2.0, 3.0]:
            x = (i + 0.5) / n
            a = (x**p) * ((1 - x)**q)
            a = norm(a)
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_arr = a.copy()
                best_score = score

    # Family 3: exponential: a_i = exp(-lam * i)
    for lam in [0.001, 0.002, 0.005, 0.01,0.02,0.05,0.1]:
        a = np.exp(-lam * i)
        a = norm(a)
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_arr = a.copy()
            best_score = score

    # Family 4: symmetric double exponential (like a Gaussian)
    for sigma in [n/10, n/5, n/4, n/3]:
        mu = (n-1) / 2.0
        a = np.exp(- ((i - mu)**2) / (2 * sigma**2))
        a = norm(a)

        if np.isfinite(score) and score < best_score:
            best_arr = a.copy()
            best_score = score

    # Now local random search from best_arr.

        best_arr = norm(best_arr)

    # Local search parameters: start with step fraction maybe gradually decreasing
    # We'll use two types: random mass transfer (like previous) and jitter (multiplicative noise)
    # We'll track iteration count.

    # Param adapt: step size for transfer epsilon proportion.
    eps_scale = 0.1

    while time.monotonic() < deadline:
        # Random mass transfer:

        if best_arr[i_idx] == 0:

        cand = best_arr.copy()
        cand[i_idx] -= eps

        # Occasionally apply multiplicative jitter:

            if best_score <= 1.50274349:

                best_arr = cand2
                best_score = score2
                if best_score <= 1.50274349:
                    break

    # Return best_arr as python list.
    return [float(x) for x in best_arr]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that attempts to
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    The search combines inexpensive parametric families, random Dirichlet
    starts and a projected sub‑gradient descent on the objective.
    """
    rng = np.random.default_rng(seed)
    # Reserve a safety margin for the sandbox timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1.  Determine problem size and a safe fallback candidate.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:
        # If no parent is provided, start from a uniform vector.
        parent = np.ones(2000, dtype=np.float64)

    n = int(parent.size)

    # Helper: project onto the probability simplex (non‑negative, sum = 1).
    def _project_simplex(v: np.ndarray) -> np.ndarray:
        s = v.sum()
        if s <= 0:
            # fallback to uniform if something went wrong
            return np.full_like(v, 1.0 / v.size)
        w = v / s
        # ensure non‑negativity (tiny negatives may appear after updates)
        w = np.clip(w, 0.0, None)
        # renormalise again to avoid loss of mass
        w_sum = w.sum()
        if w_sum == 0:
            return np.full_like(v, 1.0 / v.size)
        return w / w_sum

    # Helper: evaluate a numpy vector (cost = O(n log n) via np.convolve).
    def _score(vec: np.ndarray) -> float:
        return float(evaluate_sequence(vec.tolist()))

    # Initialise best candidate.
    best_arr = _project_simplex(parent.clip(0.0, 1000.0))
    best_score = _score(best_arr)

    # ------------------------------------------------------------------
    # 2.  Exhaust a few cheap parametric families (power‑law, beta‑like,
    #     exponential, Gaussian). Keep the best.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)

    # ---- power‑law a_i = (i + beta)^(-alpha) ------------------------
    for alpha in (0.4, 0.5, 0.6, 0.7, 0.8, 1.0, 1.2, 1.5):
        for beta in (0.0, 0.5, 1.0, 2.0):
            a = (i + beta + 1e-12) ** (-alpha)
            a = _project_simplex(a)
            s = _score(a)
            if s < best_score:
                best_arr, best_score = a, s

    # ---- beta‑shape a_i ∝ x^p (1‑x)^q  (x = (i+½)/n) -----------------
    x = (i + 0.5) / n
    for p in (0.1, 0.3, 0.5, 1.0, 2.0, 3.0):
        for q in (0.1, 0.3, 0.5, 1.0, 2.0, 3.0):
            a = (x ** p) * ((1.0 - x) ** q)
            a = _project_simplex(a)
            s = _score(a)
            if s < best_score:

    # ---- exponential a_i = exp(-lam * i) ----------------------------
    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05):
        a = np.exp(-lam * i)
        a = _project_simplex(a)
        s = _score(a)
        if s < best_score:
            best_arr, best_score = a, s

    # ---- symmetric Gaussian‑like bells --------------------------------
    mu = (n - 1) / 2.0

        a = _project_simplex(a)

    #   g_i = (2 n / S^2) * 2 a_{k0-i} - 4 n * M / S^3,
    # where S = sum a and M = (a*a)_{k0}, with out‑of‑range terms omitted.

        if S <= 0.0:

            S = 1.0

            step = min(max_step, step * 1.2)

            if best_arr[i1] > 0.0:
                eps = rng.random() * step * best_arr[i1]
                cand2 = best_arr.copy()
                cand2[i1] -= eps
                cand2[i2] += eps
                cand2 = _project_simplex(cand2)
                s2 = _score(cand2)
                if s2 < best_score:
                    best_arr, best_score = cand2, s2

        # --------------------------------------------------------------
        if rng.random() < 0.07:

            cand3 = best_arr * jitter
            cand3 = _project_simplex(cand3)

            if s_new < best_score:
                best_arr, best_score = a, s_new
                grad_success = 0
                step = 0.5

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found (as plain Python floats).
    # ------------------------------------------------------------------
    return [float(v) for v in best_arr]
