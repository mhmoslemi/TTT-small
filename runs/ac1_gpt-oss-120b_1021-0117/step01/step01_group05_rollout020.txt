def reduce_pair(seq):
    # compute conv and find k max
    conv = compute conv (fast)
    k_max = np.argmax(conv)
    lo = max(0, k_max - (n-1))
    hi = min(k_max, n-1)
    # compute products
    idx = np.arange(lo, hi+1)
    prod = seq[idx] * seq[k_max - idx]
    i = idx[np.argmax(prod)]
    j = k_max - i
    if i == j:
        # choose neighbor maybe
        i = max(0, i-1)
        j = k_max - i
    # amount to move
    alpha = rng.uniform(0.05, 0.12)
    delta_i = seq[i] * alpha
    delta_j = seq[j] * alpha
    if delta_i <= 0 or delta_j <= 0:
        return False
    seq[i] -= delta_i
    seq[j] -= delta_j
    total = delta_i + delta_j
    # choose target low index
    t = rng.integers(0, n)
    # add mass (could split across a few)
    seq[t] += total
    # clip box
    np.clip(seq, 0, 1000, out=seq)

    # Perform FFT
    f = np.fft.rfft(seq, n=m)  # m precomputed

    score = 2.0 * n * max_val / (s * s)

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

    partners = seq[kmax - idx]
    sum_seq = seq.sum()
    M = conv[kmax]
    diff = partners * sum_seq - M
    pos = diff > 0
    cand = seq.copy()
    if np.any(pos):
        idx_pos = idx[pos]
        i = idx_pos[np.argmax(diff[pos])]
        alpha = rng.uniform(0.04, 0.12)
        delta_i = cand[i] * alpha
        cand[i] -= delta_i
        j = kmax - i
        total_removed = delta_i
        if j != i:
            delta_j = cand[j] * alpha
            cand[j] -= delta_j
            total_removed += delta_j
        # add to the smallest entry (heuristic low‑partner location)
        t = int(np.argmin(cand))
        cand[t] += total_removed
    return cand


def _candidate_block_scaling(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = seq.shape[0]
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.07))
    cand = seq.copy()
    cand[start:start + width] *= factor
    return cand


def _candidate_single_scale(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Perturb a single coefficient by a log‑normal factor."""
    n = seq.shape[0]
    cand = seq.copy()
    idx = rng.integers(0, n)
    factor = np.exp(rng.normal(0.0, 0.25))
    cand[idx] *= factor
    return cand


def _candidate_smooth(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Replace two neighbours by their average (local smoothing)."""
    n = seq.shape[0]
    cand = seq.copy()
    i = rng.integers(0, n - 1)
    avg = (cand[i] + cand[i + 1]) * 0.5
    cand[i] = avg
    cand[i + 1] = avg
    return cand


def _candidate_swap_mass(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a small fraction of mass from a larger entry to a smaller one."""
    n = seq.shape[0]
    cand = seq.copy()
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    if cand[i] > cand[j]:
        frac = rng.uniform(0.02, 0.10)
        delta = cand[i] * frac
        cand[i] -= delta
        cand[j] += delta
    else:
        frac = rng.uniform(0.02, 0.10)
        delta = cand[j] * frac
        cand[j] -= delta
        cand[i] += delta
    return cand


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search that iteratively reshapes a non‑negative step function
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Determine problem size and a valid parent (if supplied by harness)
    # ------------------------------------------------------------------
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Initialise the sequence (parent, binomial seed or simple decay)
    # ------------------------------------------------------------------
    if parent is not None:
        cur = parent.copy()
    else:
        cur = _initial_binomial(n)

    fft_len = 1 << ((2 * n - 1).bit_length())

    # Evaluate the initial candidate
    best_score, best_conv, _ = _score_and_conv(cur, n, fft_len)
    best = cur.copy()

    # Try a simple 1/√i decay as an alternative start
    alt = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    alt_score, alt_conv, _ = _score_and_conv(alt, n, fft_len)
    if alt_score < best_score:
        best = alt.copy()
        best_score = alt_score
        best_conv = alt_conv

    # Use the best discovered as the current working point
    cur = best.copy()
    cur_score = best_score
    cur_conv = best_conv.copy()

    # ------------------------------------------------------------------
    # Main stochastic optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        op = rng.integers(0, 5)
        if op == 0:
            cand = _candidate_targeted(cur, cur_conv, rng)
        elif op == 1:
            cand = _candidate_block_scaling(cur, rng)
        elif op == 2:
            cand = _candidate_single_scale(cur, rng)
        elif op == 3:
            cand = _candidate_smooth(cur, rng)
        else:
            cand = _candidate_swap_mass(cur, rng)

        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score, cand_conv, _ = _score_and_conv(cand, n, fft_len)

        # Record global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            best_conv = cand_conv.copy()

        # Acceptance for the current walk (hill climbing + simple annealing)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            cur_conv = cand_conv
        else:
            elapsed = time.monotonic() - start_time
            total = deadline - start_time
            frac = elapsed / total if total > 0 else 1.0
            T = max(1e-8, 0.02 * (1.0 - frac))
            if rng.random() < math.exp(-(cand_score - cur_score) / T):
                cur = cand
                cur_score = cand_score
                cur_conv = cand_conv

    # Return the best sequence found as a plain Python list
    return [float(x) for x in best]
