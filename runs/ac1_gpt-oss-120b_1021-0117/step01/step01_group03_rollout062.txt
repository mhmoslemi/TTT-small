def simplex_projection(v):

    # if sum(v) == 1 and all>=0, return v
    # else apply algorithm: sort descending
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1

import numpy as np, time

def _conv_self(a):
    n = a.size
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa*fa
    conv = np.fft.irfft(prod, n=m)[:2*n-1]
    return conv

def _score(a):
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s*s))
```

import numpy as np
import time

# ----------------------------------------------------------------------
# Core utilities – they are deliberately lightweight and matrix‑free.
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective from the authoritative evaluator."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def _make_power(alpha: float, shift: float, n: int) -> np.ndarray:
    """Generate a_i = (i + shift)^{-alpha} for i = 0 .. n‑1."""
    idx = np.arange(n, dtype=np.float64)
    return np.power(idx + shift, -alpha)
