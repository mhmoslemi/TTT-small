def _next_pow2(x):
    return 1 << (x-1).bit_length()

def ac1_score(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa*fa, n=L)
    # Conv length L, relevant part 0..2n-2
    max_b = conv[:2*n-1].max()
    s = arr.sum()
    if s < 1e-12:
        return np.inf
    return (2.0 * n * max_b) / (s * s)

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound using FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    max_b = float(conv[: 2 * n - 1].max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _smooth_centered(arr: np.ndarray, radius: int) -> np.ndarray:
    """Replace each entry by the average over a centered window of size 2*radius+1."""
    if radius <= 0:
        return arr.copy()
    n = arr.shape[0]
    pad = radius
    padded = np.concatenate((np.zeros(pad, dtype=np.float64),
                             arr,
                             np.zeros(pad, dtype=np.float64)))
    csum = np.cumsum(padded, dtype=np.float64)
    window = 2 * radius + 1
    sums = csum[window:] - csum[:-window]      # length n
    out = sums / window
    # renormalise to keep the total sum unchanged
    s_old = arr.sum()
    s_new = out.sum()
    if s_new > 0:
        out *= s_old / s_new
    return out
