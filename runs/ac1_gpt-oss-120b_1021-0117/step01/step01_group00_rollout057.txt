def _fast_ac1(seq, n, L):
    # compute AC1 upper bound using FFT
    # seq numpy array (float64)
    fa = np.fft.rfft(seq, n=L)

    sum_a = seq.sum()

import numpy as np
import time
import math

def _fast_ac1(seq, n, L):
    """Return the AC1 bound using FFT (no clipping)."""
    # convolution of seq with itself via FFT
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

def _sidon_like_sequence(n):
    """C(2i,i)/4**i, decreasing, length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # small safety margin before the hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # ------------------------------------------------------------------
    # problem size & parent (if supplied)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
        parent = np.clip(parent, 0.0, 1000.0)
        if not np.isfinite(parent).all() or parent.sum() < 0.01:
            raise Exception
        have_parent = True
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)
        have_parent = False

    # ------------------------------------------------------------------

    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------

    if fast_sid < best_fast - 1e-12:
        score_sid = float(evaluate_sequence(seq.tolist()))
        if score_sid < best_score:

            best_fast = fast_sid

    # current working solution
    current = best_seq.copy()

    # ------------------------------------------------------------------

    start_time = time.monotonic()
    total_time = deadline - start_time

    while time.monotonic() < deadline:

        prog = (now - start_time) / total_time if total_time > 0 else 1.0
        temp = temp_start * (1.0 - prog) + temp_end * prog

        cand = current.copy()
