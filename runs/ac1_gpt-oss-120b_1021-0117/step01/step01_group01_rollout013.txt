import time, numpy as np

def _fft_len_for_n(n):
    conv_len = 2 * n - 1
    return 1 << ((conv_len - 1).bit_length())

def _conv_full(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv

def _score_from_conv(conv, n, sum_a):
    max_b = conv.max()
    return float(2 * n * max_b / (sum_a * sum_a))

def _compute_score_and_conv(a, fft_len):
    conv = _conv_full(a, fft_len)
    sum_a = a.sum()
    score = _score_from_conv(conv, a.size, sum_a)
    return score, conv

def generate_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * (2*i + 1) / (2*i + 2)
    a /= a.sum()
    return a

def generate_exponential(n, r):
    # r in (0,1)
    idx = np.arange(n, dtype=np.float64)
    a = r**idx
    a /= a.sum()
    return a

def generate_powerlaw(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = 1.0 / (idx**p)
    a /= a.sum()
    return a

import time
import numpy as np

# ---------- low‑level convolution utilities ----------
def _fft_len_for_n(n: int) -> int:
    """Smallest power of two >= 2*n‑1 (length needed for linear convolution)."""
    conv_len = 2 * n - 1
    return 1 << ((conv_len - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a single FFT (real‑valued)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact AC1 score from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (sum_a * sum_a))


def _score_and_conv(a: np.ndarray, fft_len: int):
    """Return (score, full convolution) for a non‑negative vector."""
    conv = _conv_full(a, fft_len)
    s = a.sum()
    return _score_from_conv(conv, a.size, s), conv


# ---------- candidate generators ----------
def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i normalized to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    a /= a.sum()
    return a


def _gen_exponential(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i = r**i, normalized to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    a /= a.sum()
    return a


def _gen_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i = 1/(i+1)^p, normalized to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


# ---------- main search routine ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative heights with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # problem size (provided by the harness)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    fft_len = _fft_len_for_n(n)

    # ----- 1. quick sweep over a few analytic families -----
    pool = [_gen_binomial(n)]
    for r in np.linspace(0.90, 0.999, 8):
        pool.append(_gen_exponential(n, r))
    for p in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        pool.append(_gen_powerlaw(n, p))

    best = None
    best_score = np.inf
    for a in pool:
        sc, _ = _score_and_conv(a, fft_len)
        if sc < best_score:
            best_score = sc
            best = a.copy()

    # ----- 2. local stochastic optimisation (simulated annealing) -----
    cur = best.copy()
    cur_score, cur_conv = _score_and_conv(cur, fft_len)
    best = cur.copy()
    best_score = cur_score

    # annealing schedule parameters
    T0 = 0.05               # initial temperature
    total_time = deadline - start
    min_frac = 0.01
    max_frac = 0.20
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        # linearly decreasing temperature
        elapsed = time.monotonic() - start
        T = T0 * max(0.0, 1.0 - elapsed / total_time)

        # locate shift with the current maximal convolution value
        kmax = int(np.argmax(cur_conv))
        i_low = max(0, kmax - (n - 1))
        i_high = min(n - 1, kmax)
        idx = np.arange(i_low, i_high + 1)
        rev = cur[kmax - idx]
        contrib = cur[idx] * rev

        # source index – the pair contributing most to the peak
        src = int(idx[np.argmax(contrib)])

        # try to pick a target index outside the overlapping window
        tgt = None
        for _ in range(5):
            cand = rng.integers(0, n)
            if cand < i_low or cand > i_high:
                tgt = int(cand)
                break
        if tgt is None:
            # fallback: any index different from src
            tgt = int(rng.integers(0, n - 1))
            if tgt >= src:
                tgt += 1

        if cur[src] <= 1e-12:
            # source depleted – do a cheap random move instead
            i = rng.integers(0, n)
            if cur[i] <= 1e-12:
                continue
            src = int(i)
            j = rng.integers(0, n - 1)
            if j >= src:
                j += 1
            tgt = int(j)

        # move a modest fraction of mass
        frac = rng.uniform(min_frac, max_frac)
        delta = cur[src] * frac
        cand = cur.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # evaluate candidate
        cand_score, cand_conv = _score_and_conv(cand, fft_len)

        # acceptance test
        if cand_score < cur_score:
            accept = True
        else:
            accept = (T > 0.0) and (rng.random() < np.exp(-(cand_score - cur_score) / T))

        if accept:
            cur = cand
            cur_score = cand_score
            cur_conv = cand_conv
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                if best_score < 1.50274349:
                    break

        # occasional random perturbation to keep diversity
        if rng.random() < 0.001:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if cur[i] <= 1e-12:
                continue
            d = cur[i] * rng.uniform(min_frac, max_frac)
            alt = cur.copy()
            alt[i] -= d
            alt[j] += d
            alt_score, alt_conv = _score_and_conv(alt, fft_len)
            if alt_score < cur_score or (T > 0.0 and rng.random() < np.exp(-(alt_score - cur_score) / T)):
                cur = alt
                cur_score = alt_score
                cur_conv = alt_conv
                if alt_score < best_score:
                    best_score = alt_score
                    best = alt.copy()
                    if best_score < 1.50274349:
                        break

    # ----- 3. final fallback and return -----
    final = [float(x) for x in best]
    try:
        _ = evaluate_sequence(final)   # sanity check – will raise on hidden issues
    except Exception:
        # if something went wrong, fall back to a uniform distribution
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        final = [float(x) for x in uniform]
    return final
