import scipy.sparse as sp

def build_convolution_ub(f):
    n = len(f)
    rows = 2*n-1
    diags = []
    offsets = []
    for d in range(n):
        offsets.append(d)  # diagonal offset (row - col)
        diags.append(np.full(n, f[d]))
    A = sp.diags(diags, offsets, shape=(rows, n), format='csr')
    return A

def _conv_and_score(seq, n, m):
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_val = conv.max()
    sum_val = seq.sum()
    if sum_val <= 0:
        return float('inf'), conv, max_val
    score = 2.0 * n * max_val / (sum_val * sum_val)
    return score, conv, max_val

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_and_score(seq: np.ndarray, n: int, m: int):
    """Return (score, full convolution, max(conv)) using FFT."""
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = float(conv.max())

    if total <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (total * total)
    return float(score), conv, max_val
