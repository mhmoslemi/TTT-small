Projection onto simplex: algorithm:

```
def proj_simplex(v):
    # Projects onto simplex {x >=0, sum x = 1}
    # Using sorting method.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

    A = np.fft.rfft(a, m)

def proj_simplex(v):
    # projects onto simplex of sum 1
    # reference: Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all v_i <= 0
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all at module level – no closures or lambdas)
# ----------------------------------------------------------------------


def _catalan_weights(n: int) -> np.ndarray:
    """Return first n Catalan‑style weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a power‑of‑two FFT (O(n log n))."""
    n = a.shape[0]
    size = 2 * n - 1
    m = 1 << (size - 1).bit_length()          # next power of two ≥ size
    A = np.fft.rfft(a, m)
    conv = np.fft.irfft(A * A, m)[:size]
    return conv


def _fast_score(a: np.ndarray) -> float:
    """Fast version of the AC‑1 bound (uses FFT, assumes a≥0)."""
    s = a.sum()
    if s == 0.0:
        return float('inf')
    conv = _full_convolution(a)
    return 2.0 * a.shape[0] * conv.max() / (s * s)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex {x≥0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of heights) whose AC‑1 bound
    is as small as possible within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------
    # Load parent if it exists, otherwise fall back to a short uniform.
    # --------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
        n = int(parent_arr.size)
    except Exception:
        n = 256
        parent_arr = None

    # --------------------------------------------------------------
    # Build a modest portfolio of deterministic initial candidates.
    # --------------------------------------------------------------
    candidates = []

    # 1) normalized parent (if any)
    if parent_arr is not None:
        cand = parent_arr / parent_arr.sum()
        candidates.append(cand)

    # 2) Catalan‑style
    cand = _catalan_weights(n)
    cand = cand / cand.sum()
    candidates.append(cand)

    # 3) Uniform
    candidates.append(np.full(n, 1.0 / n))

    # 4) Power‑law decreasing a_i = (i+1)^{-p}
    for p in (0.2, 0.5, 0.8, 1.2, 1.5):
        idx = np.arange(n, dtype=np.float64) + 1.0
        cand = idx ** (-p)
        candidates.append(cand / cand.sum())

    # 5) Exponential decreasing a_i = exp(-λ i)
    for lam in (0.0005, 0.001, 0.002, 0.004):
        idx = np.arange(n, dtype=np.float64)
        cand = np.exp(-lam * idx)
        candidates.append(cand / cand.sum())

    # 6) Heavy‑edge “arcsine‑type” a_i ∝ 1/√((i+½)(n‑i‑½))
    i_mid = np.arange(n, dtype=np.float64) + 0.5
    cand = 1.0 / np.sqrt(i_mid * (n - i_mid))
    candidates.append(cand / cand.sum())

    # 7) Very heavy edge a_i ∝ 1/((i+1)(n‑i))
    idx = np.arange(n, dtype=np.float64) + 1.0
    cand = 1.0 / (idx * (n - idx + 1.0))
    candidates.append(cand / cand.sum())

    # --------------------------------------------------------------
    # Pick the best among the deterministic set (using fast & exact scores).
    # --------------------------------------------------------------
    best = None
    best_fast = float('inf')
    best_true = float('inf')
    for cand in candidates:
        fs = _fast_score(cand)
        if fs < best_fast:
            true = float(evaluate_sequence(cand.tolist()))
            if true < best_true:
                best = cand.copy()
                best_fast = fs
                best_true = true

    # Safety fallback – should never happen
    if best is None:
        best = np.full(n, 1.0 / n)
        best_true = float(evaluate_sequence(best.tolist()))
        best_fast = _fast_score(best)

    # --------------------------------------------------------------
    # Projected‑gradient descent on the simplex.
    # --------------------------------------------------------------
    step = 0.5            # initial step length
    min_step = 1e-5
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        # 1) compute convolution of the current best and locate active index
        conv_best = _full_convolution(best)
        k_max = int(np.argmax(conv_best))

        # 2) gradient of the active autocorrelation term
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        # derivative = a_i + a_{k-i} when the counterpart exists
        grad[mask] = best[mask] + best[j[mask]]

        # ----------------------------------------------------------
        # a) Projected‑gradient candidate
        # ----------------------------------------------------------
        cand = _proj_simplex(best - step * grad)
        cand_fast = _fast_score(cand)

        improved = False
        if cand_fast + 1e-12 < best_fast:
            cand_true = float(evaluate_sequence(cand.tolist()))
            if cand_true + 1e-12 < best_true:
                best = cand
                best_fast = cand_fast
                best_true = cand_true
                improved = True

        # ----------------------------------------------------------
        # b) Simple pairwise transfer (fallback if a) stalls)
        # ----------------------------------------------------------
        if not improved:
            i_high = int(np.argmax(grad))
            i_low = int(np.argmin(best))          # move to a currently small entry
            if i_high != i_low and best[i_high] > 0.0:
                eps = step * best[i_high]
                eps = min(eps, best[i_high] * 0.5)   # be conservative
                cand2 = best.copy()
                cand2[i_high] -= eps
                cand2[i_low] += eps
                cand2 = _proj_simplex(cand2)
                cand2_fast = _fast_score(cand2)
                if cand2_fast + 1e-12 < best_fast:
                    cand2_true = float(evaluate_sequence(cand2.tolist()))
                    if cand2_true + 1e-12 < best_true:
                        best = cand2
                        best_fast = cand2_fast
                        best_true = cand2_true
                        improved = True

        # ----------------------------------------------------------
        # c) Occasional random jitter to escape plateaus
        # ----------------------------------------------------------
        if it % 100 == 0 and rng.random() < 0.3:
            jitter = np.exp(rng.normal(0.0, 0.018, size=n))
            cand3 = best * jitter
            cand3 = cand3 / cand3.sum()
            cand3 = _proj_simplex(cand3)
            cand3_fast = _fast_score(cand3)
            if cand3_fast + 1e-12 < best_fast:
                cand3_true = float(evaluate_sequence(cand3.tolist()))
                if cand3_true + 1e-12 < best_true:
                    best = cand3
                    best_fast = cand3_fast
                    best_true = cand3_true
                    improved = True

        # ----------------------------------------------------------
        # Step‑size adaptation
        # ----------------------------------------------------------
        if improved:
            step = min(step * 1.1, 1.0)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                step = max(step * 0.5, min_step)
                no_improve = 0

        # Early exit on target bound
        if best_true <= 1.50274349:
            break

        it += 1

    # --------------------------------------------------------------
    # Final sanity: clip, renormalise and return as a plain list.
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    s = best.sum()
    if s == 0.0:
        best = np.full_like(best, 1.0 / n)
    else:
        best = best / s

    return [float(v) for v in best]
