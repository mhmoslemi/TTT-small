def _conv_and_max(a):
    n = a.size
    L = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv*fv, n=L)[:2*n-1]
    max_val = conv.max()

import time
import numpy as np

# ------------------------------------------------------------
# basic utilities
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    """Exact AC1 score (identical to evaluate_sequence)."""
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (vec.sum() ** 2))


def _score_and_max_idx(vec: np.ndarray):
    """Return (score, index of maximal autocorrelation)."""
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    max_val = conv.max()
    max_idx = int(conv.argmax())
    score = float(2.0 * n * max_val / (vec.sum() ** 2))
    return score, max_idx


def _target_move(a: np.ndarray, k: int):
    """
    Pick a source index that contributes most to the peak at k,
    and a destination index that currently carries the smallest mass.
    """
    n = a.size
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    contrib = a[i] * a[j]                     # contribution of each ordered pair
    best = int(contrib.argmax())
    i_src, j_src = i[best], j[best]
    src = i_src if a[i_src] >= a[j_src] else j_src   # take the larger entry
    dst = int(np.argmin(a))
    if dst == src:                                 # avoid self‑transfer
        # second smallest if possible
        order = np.argsort(a)
        dst = int(order[1]) if a.size > 1 else src
    return src, dst


def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients a satisfying a*a = 1+…+x^{2n‑2} (truncated)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # give ourselves a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -----------------------------------------------------------------
    # initialise the vector (use the supplied parent if present)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():

    else:
        # fallback: analytic sqrt‑series of moderate length
        n = 256
        a = _sqrt_series(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 1e-12:
        a[:] = 1.0 / a.size
    else:
        a /= total                     # normalise to sum = 1

    cur = a.copy()
    best = cur.copy()
    best_score = _score(best)

    # parameters for the simulated‑annealing style search
    init_temp = 0.02
    final_temp = 0.001
    max_iters = 10_000_000          # huge upper bound; loop stops by deadline
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        # recompute score and the location of the current peak
        cur_score, peak_idx = _score_and_max_idx(cur)

        # decide on a move: mostly targeted, occasionally random
        if rng.random() < 0.75:
            src, dst = _target_move(cur, peak_idx)
        else:
            src = rng.integers(0, cur.size)
            dst = rng.integers(0, cur.size - 1)
            if dst >= src:
                dst += 1

        # step size adapts with iteration count (larger early, smaller later)
        progress = it / 200_000.0                # after 200k steps temperature is low
        temp = max(final_temp, init_temp * (1.0 - progress))
        max_fraction = 0.30 * (temp / init_temp)  # shrink fraction as temp drops
        frac = rng.random() * max_fraction
        delta = frac * cur[src]

        # ensure we do not create a negative entry
        if delta > cur[src]:
            delta = cur[src]

        if delta <= 0.0:
            it += 1
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta

        # tiny safeguard against round‑off negativity
        if cand[src] < 0.0:
            cand[src] = 0.0

        cand_score = _score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur = cand
            cur_score = cand_score
            # keep the global best seen so far
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()

        # occasional global perturbation to escape plateaus
        if it % 5000 == 0 and rng.random() < 0.15:
            i = rng.integers(0, cur.size)
            j = rng.integers(0, cur.size - 1)
            if j >= i:
                j += 1
            eps = rng.random() * 0.02 * cur[i]
            eps = min(eps, cur[i])
            tmp = cur.copy()
            tmp[i] -= eps
            tmp[j] += eps
            sc = _score(tmp)
            if sc < best_score:
                best_score = sc
                best = tmp.copy()
                cur = tmp.copy()
                cur_score = sc

        # renormalise modestly every few thousand steps to fight drift
        if it % 2000 == 0:
            s = cur.sum()
            if abs(s - 1.0) > 1e-12:
                cur /= s

        it += 1

    # return the best vector found (as plain Python floats)
    return [float(x) for x in best]
