import time
import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, decreasing, a_0 = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, conv, max_conv) using power‑of‑two FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    maxv = float(conv.max())
    s = float(seq.sum())
    if s < 0.01:
        return float("inf"), conv, maxv
    return float(2.0 * n * maxv / (s * s)), conv, maxv

            j = k_max - i
            if i != j:
                alpha = rng.uniform(0.05, 0.22)
                rem_i = cand[i] * alpha
                rem_j = cand[j] * alpha
                cand[i] -= rem_i
                cand[j] -= rem_j
                total_removed = rem_i + rem_j
                # add to a safe tail position (does not affect conv[k_max])
                if k_max < n - 1:
                    tgt = rng.integers(k_max + 1, n)
                else:
                    tgt = rng.integers(0, n)
                while tgt == i or tgt == j:
                    tgt = rng.integers(0, n)
                cand[tgt] += total_removed

        # --------------------------------------------------------------
        # 3 – smooth two neighbours (averaging)
        # --------------------------------------------------------------
        elif op == 3:
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        # --------------------------------------------------------------
        # 4 – swap a front fragment with a tail fragment (preserving mass)
        # --------------------------------------------------------------
        elif op == 4:
            block = rng.integers(1, max(2, n // 40))
            i = rng.integers(0, n - block + 1)
            j = rng.integers(0, n - block + 1)
            if i != j:
                tmp = cand[i:i + block].copy()
                cand[i:i + block] = cand[j:j + block]
                cand[j:j + block] = tmp

        # --------------------------------------------------------------
        # 5 – global additive bump to the tail (increase sum)
        # --------------------------------------------------------------
        else:
            tail_start = rng.integers(n // 2, n)
            add = rng.uniform(0.0, 0.03) * cand[tail_start:].mean()
            cand[tail_start:] += add

        # enforce box constraints
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score, cand_conv, _ = _fast_score(cand)

        # record best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            best_conv = cand_conv

        # decide whether to move current state (simulated annealing)
        if cand_score < curr_score:
            accept = True
        else:
            elapsed = time.monotonic() - start
            frac = elapsed / total
            temp = max_temp * (1.0 - frac) + min_temp
            accept = rng.random() < np.exp(-(cand_score - curr_score) / temp)

        if accept:
            curr = cand
            curr_score = cand_score
            curr_conv = cand_conv

    # ------------------------------------------------------------------
    # Return a plain Python list (evaluator will re‑evaluate it exactly).
    # ------------------------------------------------------------------
    return [float(x) for x in best]
