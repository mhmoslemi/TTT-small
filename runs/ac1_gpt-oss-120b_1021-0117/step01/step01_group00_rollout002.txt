import numpy as np
import time

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic Sidon–like decay a_i = prod_{k=1}^i (1-1/(2k))."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _power_sequence(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """a_i = (i+1+offset)^{-p}, normalized to sum 1."""
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    a = idx ** (-p)
    total = a.sum()
    if total == 0.0:
        return a
    return a / total

def _evaluate(seq: np.ndarray) -> float:
    """Wrapper for the provided evaluator (expects a list)."""
    return float(evaluate_sequence(seq.tolist()))

        n = int(parent.size)
        if n == 0:

    except Exception:
        n = 6962
        parent = np.full(n, 1.0, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Initialise best solution with the parent (if valid)
    # ------------------------------------------------------------------
    best_seq = parent.copy()
    best_score = _evaluate(best_seq)
    if not np.isfinite(best_score):

    sid = _sidon_like_sequence(n)
