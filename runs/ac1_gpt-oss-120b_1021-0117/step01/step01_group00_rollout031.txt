def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # get base values
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0:
            raise Exception
    except Exception:
        # generate Sidon-like sequence length fallback. Use default length = 2000?
        nfallback = 2048
        base = np.empty(nfallback, dtype=np.float64)
        base[0]=1.0
        for i in range(1, nfallback):
            base[i] = base[i-1] * (1.0 - 1.0/(2.0*i))
    n = base.shape[0]
    # ensure sum > 0.01
    if base.sum() < 0.01:
        base = np.full(n, 0.01/n)
    # sort descending to have consistent values to permute
    vals = np.sort(base)[::-1]  # descending

    # initial best: parent order (original base)
    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    # also evaluate sorted descending order (maybe better)
    seq_desc = vals.copy()
    score_desc = float(evaluate_sequence(seq_desc.tolist()))
    if score_desc < best_score:
        best_seq = seq_desc
        best_score = score_desc

    # generate candidate permutations from step patterns
    # generate small primes up to maybe 2000
    def gen_primes(limit):
        sieve = np.ones(limit+1, dtype=bool)
        sieve[:2] = False
        for p in range(2, int(limit**0.5)+1):
            if sieve[p]:
                sieve[p*p:limit+1:p] = False
        return np.nonzero(sieve)[0]
    primes = gen_primes(2000)  # produce many primes
    # limit number of steps to try to avoid too many evaluations
    max_step_trials = 200
    step_trials = primes[:max_step_trials]
    # Also include some composite values like random step
    for step in step_trials:
        # try offsets 0..max_offset
        # using up to 3 offsets
        for offset in (0, step//2, step//3):
            perm = (np.arange(n) * step + offset) % n
            cand = np.empty_like(vals)
            cand[perm] = vals
            # Evaluate
            sc = float(evaluate_sequence(cand.tolist()))
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
        # Optional early break if time low
        if time.monotonic() >= deadline - 5.0:
            break

    # random permutations
    max_rand_trials = 400
    for _ in range(max_rand_trials):
        if time.monotonic() >= deadline - 5.0:
            break
        perm = rng.permutation(n)
        cand = vals[perm]
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # simple local hill climbing (swap) from best_seq for remaining time
    current_seq = best_seq.copy()
    current_score = best_score
    # Initialize temperature based on relative scale:
    temp = best_score * 0.1 + 1e-3
    min_temp = 1e-6
    cooling = 0.999
    # loop
    while time.monotonic() < deadline - 1.0 and temp > min_temp:
        # pick two random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap in place
        current_seq[i], current_seq[j] = current_seq[j], current_seq[i]
        sc = float(evaluate_sequence(current_seq.tolist()))
        delta = sc - current_score
        if delta < 0 or math.exp(-delta / temp) > rng.random():
            # accept
            current_score = sc
            if sc < best_score:
                best_score = sc
                best_seq = current_seq.copy()
        else:
            # reject: revert swap
            current_seq[i], current_seq[j] = current_seq[j], current_seq[i]
        temp *= cooling
        # occasional large move: random shuffle small subset to escape
        if rng.random() < 0.001:
            # random sub-permutation of 10 indices
            idxs = rng.choice(n, size=10, replace=False)
            sub_vals = current_seq[idxs].copy()
            rng.shuffle(sub_vals)
            current_seq[idxs] = sub_vals

    return [float(x) for x in best_seq.tolist()]

import numpy as np
import time
import math

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step‑function that lowers the AC1 bound.

    The routine works in three phases:
      1. Build a pool of candidate permutations of a Sidon‑like multiset.
         Deterministic “prime‑step” permutations and a number of random
         shuffles are evaluated; the best is kept.
      2. A targeted local‑search driven by the location of the maximal
         convolution entry tries to move large values away from that spot.
      3. A short simulated‑annealing style hill‑climb finishes the search.

    The function returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin before the hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1.  Input sequence (parent) – use the provided height_sequence_1 if
    #     available, otherwise fall back to the classic Sidon‑like series.
    # ------------------------------------------------------------------
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0:
            raise Exception
    except Exception:
        n = 4096
        base = np.empty(n, dtype=np.float64)
        base[0] = 1.0
        for i in range(1, n):
            base[i] = base[i - 1] * (1.0 - 1.0 / (2.0 * i))
    n = int(base.shape[0])

    # ensure a non‑trivial total mass
    if base.sum() < 0.01:
        base = np.full(n, 0.01 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Helper to evaluate a sequence (expects a NumPy array)
    # ------------------------------------------------------------------
    def _eval(arr: np.ndarray) -> float:
        return float(evaluate_sequence(arr.tolist()))

    # ------------------------------------------------------------------
    # 2.  Prepare the multiset of values – we only ever permute them.
    # ------------------------------------------------------------------
    vals_desc = np.sort(base)[::-1]           # descending order (largest first)

    # start from a few simple baselines
    best_seq = base.copy()
    best_score = _eval(best_seq)

    # descending order
    cand = vals_desc.copy()
    sc = _eval(cand)
    if sc < best_score:
        best_seq, best_score = cand, sc

    # ascending order
    cand = vals_desc[::-1].copy()
    sc = _eval(cand)
    if sc < best_score:
        best_seq, best_score = cand, sc

    # ------------------------------------------------------------------
    # 3.  Systematic “prime‑step” permutations.
    # ------------------------------------------------------------------
    def _primes(limit: int) -> np.ndarray:
        sieve = np.ones(limit + 1, dtype=bool)
        sieve[:2] = False
        for p in range(2, int(limit ** 0.5) + 1):
            if sieve[p]:
                sieve[p * p : limit + 1 : p] = False
        return np.nonzero(sieve)[0]

    primes = _primes(2000)                     # a short list of primes
    max_step_trials = min(len(primes), 150)    # keep the work bounded

    for step in primes[:max_step_trials]:
        # a couple of offsets to diversify the pattern
        offsets = (0, step // 2, step // 3)
        for off in offsets:
            perm = (np.arange(n, dtype=np.int64) * step + off) % n
            cand = np.empty_like(vals_desc)
            cand[perm] = vals_desc
            sc = _eval(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
            if time.monotonic() >= deadline - 0.5:
                break
        if time.monotonic() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # 4.  Random permutations (still cheap compared to the whole budget).
    # ------------------------------------------------------------------
    max_random = 400
    for _ in range(max_random):
        if time.monotonic() >= deadline - 0.5:
            break
        perm = rng.permutation(n)
        cand = vals_desc[perm]
        sc = _eval(cand)
        if sc < best_score:
            best_seq, best_score = cand.copy(), sc

    # ------------------------------------------------------------------
    # 5.  Targeted local search: repeatedly look at the index where the
    #     convolution is maximal and try to move the contributing largest
    #     values to a low‑value location.
    # ------------------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score
    temperature = max(0.1 * current_score, 1e-3)
    cooling = 0.9995
    min_temp = 1e-6

    # pool of indices that currently hold relatively small values
    low_pool = np.argsort(current_seq)[: max(1, n // 10)]

    while time.monotonic() < deadline - 1.0 and temperature > min_temp:
        # convolution and the location of its peak
        conv = np.convolve(current_seq, current_seq)
        max_k = int(np.argmax(conv))

        # all i that participate in the peak (i + j = max_k)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        i_vec = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vec = max_k - i_vec

        # product a_i * a_j for each pair; pick the largest contributor
        prod = current_seq[i_vec] * current_seq[j_vec]
        idx_max = int(i_vec[np.argmax(prod)])
        partner = int(j_vec[np.argmax(prod)])

        # pick a low‑value position, avoiding the two involved indices
        l = int(rng.choice(low_pool))
        while l == idx_max or l == partner:
            l = int(rng.choice(low_pool))

        # single swap proposal
        cand = current_seq.copy()
        cand[idx_max], cand[l] = cand[l], cand[idx_max]
        sc = _eval(cand)
        delta = sc - current_score
        if delta < 0 or math.exp(-delta / temperature) > rng.random():
            current_seq, current_score = cand, sc
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
        else:

            m = int(rng.choice(low_pool))
            while m in (idx_max, partner, l):
                m = int(rng.choice(low_pool))
            cand2 = current_seq.copy()
            cand2[idx_max], cand2[l] = cand2[l], cand2[idx_max]
            cand2[partner], cand2[m] = cand2[m], cand2[partner]
            sc2 = _eval(cand2)
            delta2 = sc2 - current_score
            if delta2 < 0 or math.exp(-delta2 / temperature) > rng.random():
                current_seq, current_score = cand2, sc2
                if sc2 < best_score:
                    best_seq, best_score = cand2.copy(), sc2

        temperature *= cooling

        # occasional small reshuffle to keep the search fluid
        if rng.random() < 0.001:
            perm = rng.permutation(n)
            current_seq = best_seq[perm]
            current_score = _eval(current_seq)

    # ------------------------------------------------------------------
    # 6.  Return the best sequence discovered.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
