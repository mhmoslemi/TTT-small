import numpy as np
import time

def _fast_score(seq: np.ndarray):
    """Return AC1 score, full convolution and its maximum (FFT‑based)."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val


def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -------------------------------------------------
    # Determine problem size and optional parent sequence
    # -------------------------------------------------
    n = 3809  # default length
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # -------------------------------------------------
    # Helper to evaluate a candidate quickly
    # -------------------------------------------------
    def _eval(seq):
        return _fast_score(seq)[0]

    best_seq = None
    best_score = np.inf

    # -------------------------------------------------
    # 1. start from the provided parent (if any)
    # -------------------------------------------------
    if parent is not None:
        sc = _eval(parent)
        best_seq = parent.copy()
        best_score = sc

    # -------------------------------------------------
    # 2. binomial seed (known good baseline)
    # -------------------------------------------------
    bin_seq = _initial_binomial(n)
    sc_bin = _eval(bin_seq)
    if sc_bin < best_score:
        best_seq = bin_seq.copy()
        best_score = sc_bin

    # -------------------------------------------------
    # 3. power‑law families a_i = (i+1)^{-p}
    # -------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)   # 1‑based to avoid zero power
    # explore p in [0.5, 0.9] with a step of 0.01
    for p in np.linspace(0.5, 0.9, 41):
        if time.monotonic() >= deadline:
            break
        cand = idx ** (-p)
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _eval(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc

    # -------------------------------------------------
    # 4. Light stochastic hill‑climbing from the current best
    # -------------------------------------------------
    rng = np.random.default_rng(seed)
    # pre‑compute length‑dependent parameters once
    max_block = max(2, n // 60)   # block width bound for mutations

    while time.monotonic() < deadline:
        cand = best_seq.copy()
        op = rng.integers(0, 4)

        if op == 0:            # block scaling
            width = rng.integers(1, max_block)
            start_i = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            cand[start_i:start_i + width] *= factor

        elif op == 1:          # single index perturbation
            i = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.25))
            cand[i] *= factor

        elif op == 2:          # move a little mass away from current max pair
            # locate max‑convolution index of the present best
            _, best_conv, _ = _fast_score(best_seq)
            k_max = int(np.argmax(best_conv))
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            i = rng.integers(lo, hi + 1)
            j = k_max - i
            if i != j:
                alpha = rng.uniform(0.04, 0.12)
                removed_i = cand[i] * alpha
                removed_j = cand[j] * alpha
                cand[i] -= removed_i
                cand[j] -= removed_j
                total = removed_i + removed_j
                tgt = rng.integers(0, n)
                while tgt == i or tgt == j:
                    tgt = rng.integers(0, n)
                cand[tgt] += total

        else:                  # smooth two neighbours
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _eval(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc

    # -------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # -------------------------------------------------
    return [float(x) for x in best_seq]
