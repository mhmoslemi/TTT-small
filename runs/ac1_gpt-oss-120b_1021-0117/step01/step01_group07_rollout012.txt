def compute_gradient(a, max_idx):
    n = a.shape[0]
    idx = np.arange(n)
    j = max_idx - idx
    mask1 = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask1] = a[j[mask1]]
    k = idx - max_idx
    mask2 = (k >= 0) & (k < n)
    grad[mask2] += a[k[mask2]]
    return grad

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    AC1 upper bound returned by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    # safety margin before the sandbox timeout
    deadline = time.monotonic() + float(budget_s) - 2.0

    # --------------------------------------------------------------
    # Helper utilities
    # --------------------------------------------------------------
    def _catalan_weights(m: int) -> np.ndarray:
        w = np.empty(m, dtype=np.float64)
        w[0] = 1.0
        for i in range(1, m):
            w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
        return w

    def _normalize(v: np.ndarray) -> np.ndarray:
        s = v.sum()
        if s <= 0.0:
            return v
        return v / s

    def _grad_at_max(a: np.ndarray, max_i: int) -> np.ndarray:
        """Gradient of the convolution entry at index max_i (sub‑gradient of the max)."""
        n = a.shape[0]
        idx = np.arange(n)
        j = max_i - idx
        mask1 = (j >= 0) & (j < n)
        k = idx - max_i
        mask2 = (k >= 0) & (k < n)

        grad = np.empty_like(a)
        grad.fill(0.0)
        grad[mask1] = a[j[mask1]]
        grad[mask2] += a[k[mask2]]
        return grad

        n = 256

    cat = _normalize(_catalan_weights(n))

    cat_score = float(evaluate_sequence(cat.tolist()))

    TARGET = 1.50274349            # early‑exit target
    step_frac = 0.05               # fraction of source mass to move
    min_step = 1e-5
    max_step = 0.5
    last_improve = 0

    while time.monotonic() < deadline:

            grad = _grad_at_max(best, max_idx)
