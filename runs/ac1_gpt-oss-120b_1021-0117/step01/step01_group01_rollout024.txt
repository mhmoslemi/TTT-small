import time, numpy as np

def _max_conv_fft(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv.max()

def _score(a, n, fft_len):
    return float(2 * n * _max_conv_fft(a, fft_len) / (a.sum() ** 2))

def _generate_power_law(n, alpha):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)

    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    best_seq = None
    best_score = float('inf')

    # 1. Scan a few simple families (uniform, binomial, power‑law)
    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    cur_score = _score(uniform, n, fft_len)
    if cur_score < best_score:
        best_score, best_seq = cur_score, uniform.copy()

    # binomial (already used by parent)
    # generate using recurrence
    binom = np.empty(n, dtype=np.float64)
    binom[0] = 1.0
    for i in range(n - 1):
        binom[i + 1] = binom[i] * (2 * i + 1) / (2 * i + 2)
    binom /= binom.sum()
    cur_score = _score(binom, n, fft_len)
    if cur_score < best_score:
        best_score, best_seq = cur_score, binom.copy()

    # power‑law sweep
    for alpha in np.linspace(0.55, 0.90, 36):   # step ≈0.01
        if time.monotonic() >= deadline:
            break
        seq = _generate_power_law(n, alpha)
        cur_score = _score(seq, n, fft_len)
        if cur_score < best_score:
            best_score, best_seq = cur_score, seq.copy()

    # 2. Local stochastic refinement around the best candidate
    cur = best_seq.copy()
    cur_score = best_score

    while time.monotonic() < deadline:
        # pick a source index with non‑negligible mass
        i = rng.integers(0, n)
        if cur[i] <= 1e-12:
            continue
        # pick a destination different from i
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # transfer a modest fraction

            cur_score = new_score
            if new_score < best_score:
                best_seq = new.copy()
                best_score = new_score

def _max_conv_fft(a: np.ndarray, fft_len: int) -> float:

    """Exact AC1 upper bound for a (non‑negative, sum may be arbitrary)."""
    return float(2 * n * _max_conv_fft(a, fft_len) / (a.sum() ** 2))


def _power_law(n: int, alpha: float) -> np.ndarray:
    """Return a length‑n probability vector proportional to (i+1)^(-alpha)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # length of the problem (parent may provide a long array)
    try:
        n = int(len(height_sequence_1))
    except Exception:          # safe fallback
        n = 256

    # FFT layout (same for the whole run)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 1. Scan a few compact families (uniform, binomial, power‑law)
    # -----------------------------------------------------------------
    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    cur = uniform
    cur_score = _score(cur, n, fft_len)
    if cur_score < best_score:
        best_score, best_seq = cur_score, cur.copy()

    # binomial (the classic C(2i,i)/4^i sequence)
    binom = np.empty(n, dtype=np.float64)
    binom[0] = 1.0
    for i in range(n - 1):
        binom[i + 1] = binom[i] * (2 * i + 1) / (2 * i + 2)
    binom /= binom.sum()
    cur = binom
    cur_score = _score(cur, n, fft_len)
    if cur_score < best_score:
        best_score, best_seq = cur_score, cur.copy()

    # power‑law sweep (α from 0.55 to 0.90, step ≈0.01)
    for alpha in np.linspace(0.55, 0.90, 36):
        if time.monotonic() >= deadline:
            break
        cur = _power_law(n, alpha)
        cur_score = _score(cur, n, fft_len)
        if cur_score < best_score:
            best_score, best_seq = cur_score, cur.copy()

    # -----------------------------------------------------------------
    # 2. Light stochastic refinement around the best point found
    # -----------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score

    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if cur[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # move a modest fraction of mass from i to j (keeps total unchanged)
        frac = rng.uniform(0.001, 0.05)
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_score = _score(new, n, fft_len)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_seq = new.copy()
                best_score = new_score

    # -----------------------------------------------------------------
    # 3. Return the best candidate (as a plain Python list)
    # -----------------------------------------------------------------
    final = [float(x) for x in best_seq]
    try:
        _ = evaluate_sequence(final)          # sanity check – will raise on invalid data
        return final
    except Exception:
        # fallback to the binomial construction if something went wrong
        return [float(x) for x in binom]
