import time
import numpy as np

def _sqrt_series(n: int) -> np.ndarray:
    """Generate the first n coefficients a_k of the series satisfying
    (a * a)[k] = 1 for 0 <= k < n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    if n > 1:
        a[1] = 0.5
    for k in range(2, n):
        # sum_{i=1}^{k-1} a[i] * a[k-i]
        s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) * 0.5
    # numerical safety
    a = np.maximum(a, 0.0)
    return a

def _score(vec: np.ndarray) -> float:
    """Fast version of evaluate_sequence (same formula)."""
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)
