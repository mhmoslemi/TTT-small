def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    try: 
        n = len(height_sequence_1)
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if not np.isfinite(base).all(): raise
    except Exception:
        n = 256
        base = np.full(n, 1.0, dtype=np.float64)
    # generate Sidon-like if base not good: compute if other metric.
    # We can use base as starting point, or if base's score > 1.6 use sidon.
    seq = base.copy()
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(seq.tolist()))
    # fallback if score is infinite
    if not np.isfinite(best_score):
        seq = _sidon_like_sequence(n)
        best_seq = seq.copy()
        best_score = float(evaluate_sequence(seq.tolist()))
    # if still infinite, set plain uniform.
    # Hill climbing:
    current = best_seq.copy()
    current_score = best_score
    # number of steps, but break when time runs out
    while time.monotonic() < deadline:
        # propose transfer: pick i,j distinct
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # propose delta: fraction of current[i] maybe small.
        max_transfer = current[i]
        if max_transfer <= 0:
            continue
        # choose delta in (0, max_transfer)
        delta = rng.random() * max_transfer * 0.1  # up to 10%
        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure bounds
        if cand[i] < 0 or cand[j] > 1000:
            continue
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < current_score:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
        # maybe also accept occasional worse for annealing? Not needed.

def _sidon_like_sequence(n):
    """Classic a_i = prod_{j=1}^i (1-1/(2j)) sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a
