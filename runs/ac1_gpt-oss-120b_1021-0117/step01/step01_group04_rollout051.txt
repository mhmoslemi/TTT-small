import numpy as np

# ---------- helper utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1 (well‑known decreasing sequence)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution x*x via FFT (zero‑padded to length L)."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]  # keep the linear part only


def _ac1_score(x: np.ndarray, L: int) -> float:
    """Exact AC1 upper bound for a candidate (uses FFT, same formula as evaluator)."""
    n = x.size
    conv = _conv_fft(x, L)
    max_b = float(conv.max())
    s = float(x.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _grad_objective(x: np.ndarray, conv: np.ndarray, max_val: float,
                    max_idx: np.ndarray, s: float, L: int) -> np.ndarray:
    """
    Gradient of the AC1 bound with respect to the coefficient vector.
    Returns ∇f where f = 2n*max(conv)/s^2.
    """
    n = x.size
    # contribution from the max‑convolution term
    grad_max = np.zeros_like(x)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad_max[i] += 2.0 * x[k - i]                     # d/d a_i of conv_k
    # full gradient (chain rule)
    factor = 2.0 * n / (s * s)
    const = 4.0 * n * max_val / (s * s * s)
    return factor * grad_max - const

                break

        # simulated‑annealing acceptance for the current walk
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # occasional reset to the best found so far (helps avoid drifting)
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # return as plain Python floats (the harness will call evaluate_sequence)
    return [float(v) for v in best]
```
