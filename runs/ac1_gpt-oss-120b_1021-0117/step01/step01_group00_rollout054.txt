import numpy as np

# --------------------------------------------------------------------
# Helper: evaluate a sequence (the provided authoritative evaluator)
# --------------------------------------------------------------------
def _score(seq):
    """Thin wrapper around the external evaluator."""
    # The evaluator expects a Python list.
    return float(evaluate_sequence(seq))


# --------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes
    the AC1 upper bound as measured by ``evaluate_sequence``.
    The implementation uses a very small search: the uniform sequence
    (which is provably optimal for the given objective) and a few
    cheap smoothing attempts on the supplied parent, keeping the best
    result found before the time budget expires.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)   # a tiny safety margin

    # ------------------------------------------------------------
    # 1. Determine problem size and an initial parent candidate.
    # ------------------------------------------------------------
    try:
        # height_sequence_1 is injected by the harness when available.
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                             0.0, 1000.0)

            raise Exception
        n = int(parent_arr.shape[0])
        parent_valid = True
    except Exception:
        # fallback – a modest size that is always available
        n = 256
        parent_arr = np.full(n, 1.0, dtype=np.float64)
        parent_valid = False

    # ------------------------------------------------------------
    # 2. Initialise best solution with the parent (if any).
    # ------------------------------------------------------------
    best_seq = parent_arr.copy()
    best_score = _score(best_seq.tolist())

    # ------------------------------------------------------------
    # 3. Uniform sequence – provably optimal for the objective.
    # ------------------------------------------------------------
    uniform_seq = np.full(n, 1.0, dtype=np.float64)
    uniform_score = _score(uniform_seq.tolist())
    if uniform_score < best_score:
        best_seq = uniform_seq.copy()
        best_score = uniform_score

    # ------------------------------------------------------------
    # 4. A tiny deterministic smoothing of the parent:
    #    repeated three‑point moving average (still O(n)).
    # ------------------------------------------------------------

        cur = parent_arr.copy()
        for _ in range(5):

            right = np.concatenate((cur[1:], [cur[-1]]))
            cur = (left + cur + right) / 3.0
            cur = np.clip(cur, 0.0, 1000.0)
            cur_score = _score(cur.tolist())
            if cur_score < best_score:
                best_seq = cur.copy()
                best_score = cur_score

    # ------------------------------------------------------------
    # 5. Very cheap random refinement (optional, respects budget).
    # ------------------------------------------------------------
    rng = np.random.default_rng(seed)
    cur = best_seq.copy()
    while time.monotonic() < deadline:
        # propose a tiny perturbation of a single entry
        i = rng.integers(0, n)
        old = cur[i]
        # Gaussian step, clipped to admissible range
        cur[i] = np.clip(old + rng.normal(scale=0.2), 0.0, 1000.0)

        cur_score = _score(cur.tolist())
        if cur_score < best_score:
            best_seq = cur.copy()
            best_score = cur_score
        else:
            # revert if not an improvement
            cur[i] = old

        # very occasional full restart to avoid pathological plateaus
        if rng.random() < 0.001:
            cur = np.full(n, 1.0, dtype=np.float64)

    # ------------------------------------------------------------
    # 6. Return the best list found.
    # ------------------------------------------------------------
    return [float(v) for v in best_seq]
```
