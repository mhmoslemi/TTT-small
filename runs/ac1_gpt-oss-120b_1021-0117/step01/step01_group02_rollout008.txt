import time, numpy as np

def _project_simplex(v):
    # Projects onto the probability simplex (sum=1, nonnegative)
    # Implements Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # If all false, return uniform
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_self_fft(a):
    n = a.shape[0]
    # FFT length as power of two >= 2*n-1
    m = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A*A, m)[:2*n-1]
    return c

    # v: numpy array
    # Projects onto simplex {x >= 0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

    if not np.any(cond):

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a power‑of‑two FFT (real inputs)."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)[: 2 * n - 1]
    return c

    step_factor = 0.5          # fraction of the source mass to transfer
    min_step = 1e-8
    eps_rel = 1e-12
    it = 0

    while time.monotonic() < deadline:
        it += 1

        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M

        # keep global best
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            best_M = M
            if best_score < 1.5028:      # early success
                break

        # active indices (where convolution reaches its maximum)
        if M > 0:
            active = conv >= M - max(eps_rel, M * 1e-15)
        else:
            active = np.ones_like(conv, dtype=bool)
        active_idx = np.nonzero(active)[0]

        # sub‑gradient of the max‑convolution (size n)
        g = np.zeros(n, dtype=np.float64)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = k - idx
            diff = idx != j
            if np.any(diff):
                g[idx[diff]] += 2.0 * a[j[diff]]
            eq = idx == j
            if np.any(eq):
                g[idx[eq]] += a[idx[eq]]

        # choose source (large gradient & positive mass) and destination
        pos_mask = a > min_step
        src_candidates = np.where(pos_mask)[0]
        if src_candidates.size == 0:
            src = rng.integers(0, n)
        else:
            src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))
        if dest == src:
            dest = int(rng.integers(0, n - 1))
            if dest >= src:
                dest += 1
            dest %= n

        # move a fraction of the source mass
        delta = step_factor * a[src]
        if delta < min_step:
            delta = min_step
        delta = min(delta, a[src])

        cand = a.copy()
        cand[src] -= delta
        cand[dest] += delta
        cand = _project_simplex(cand)

        # evaluate the candidate
        conv_c = _conv_self_fft(cand)
        cand_score = 2.0 * n * conv_c.max()

        if cand_score < cur_score:
            a = cand
            step_factor = max(1e-4, step_factor * 0.9995)
        else:
            # occasional random shake to leave plateaus
            if rng.random() < 0.006:
                width = rng.integers(max(1, n // 200), max(2, n // 30))
                start = rng.integers(0, max(1, n - width + 1))
                stop = start + width
                shake = a.copy()
                shake[start:stop] *= np.exp(rng.normal(0.0, 0.12, size=width))
                shake = _project_simplex(shake)
                s_score = 2.0 * n * _conv_self_fft(shake).max()
                if s_score < cur_score:
                    a = shake
                    step_factor = max(1e-4, step_factor * 0.995)
            else:
                step_factor = max(1e-4, step_factor * 0.9999)

    # ----------------------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final.tolist()]
