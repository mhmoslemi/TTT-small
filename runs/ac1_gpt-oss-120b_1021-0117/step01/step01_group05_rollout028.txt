import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, maximal convolution value)."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val


def _active_indices(conv: np.ndarray, k: int) -> np.ndarray:
    """Return up to k indices of conv having the largest values."""
    length = conv.size
    if k >= length:
        return np.arange(length, dtype=np.int64)
    # argpartition gives unsorted top‑k; we keep them unsorted (order not needed)
    idx = np.argpartition(-conv, k)[:k]
    return idx


def _build_constraint_matrix(f: np.ndarray, ks: np.ndarray) -> sp.csr_matrix:
    """Build sparse matrix A where (A @ g)[row]=∑_i f_i g_{k-i} for k in ks."""
    n = f.size
    rows = []
    cols = []
    data = []
    for ridx, k in enumerate(ks):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # j runs over indices of g that appear in this convolution row
        j = np.arange(lo, hi + 1, dtype=np.int64)
        i = k - j                     # corresponding indices of f
        rows.extend([ridx] * j.size)
        cols.extend(j.tolist())
        data.extend(f[i].tolist())
    return sp.csr_matrix((data, (rows, cols)), shape=(ks.size, n))


def _solve_lp(f: np.ndarray, M: float, ks: np.ndarray, max_iter: int = 2000):
    """Linear program: maximise sum(g) s.t. A@g <= M, g>=0, g<=1000."""
    A = _build_constraint_matrix(f, ks)
    n = f.size
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M, g <= 1000.0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=max_iter, abstol=1e-6,
                    reltol=1e-6, feastol=1e-6)
    except Exception:
        return None
    if g.value is None:
        return None
    return np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)

# ----------------------------------------------------------------------

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
