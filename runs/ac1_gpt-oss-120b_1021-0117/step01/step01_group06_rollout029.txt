def _score(vec):
    n = vec.size
    L = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)

def _build_toeplitz(f):
    n = f.shape[0]
    # Build diagonals
    diags = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
    return A

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    else:
        n = 256
        parent = None

    # create sqrt series
    init = _sqrt_series(n).astype(np.float64)
    init = init / init.sum()

    # determine initial f by choosing best of init and maybe parent normalized
    f = init.copy()
    best_score = _score(f)
    if parent is not None:
        if parent.sum() == 0:
            parent_norm = parent
        else:
            parent_norm = parent / parent.sum()
        parent_score = _score(parent_norm)
        if parent_score < best_score:
            f = parent_norm.copy()
            best_score = parent_score

    best_f = f.copy()

    # LP iterations if cvxpy is available
    if _cvxpy_available:
        max_lp_iter = 4
        t_mix = 0.5
        for lp_iter in range(max_lp_iter):
            if time.monotonic() > deadline:
                break
            # compute M = max convolution
            L = 1 << ((2*n-1).bit_length())
            fv = np.fft.rfft(f, n=L)
            conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
            M = conv.max()
            # Build Toeplitz
            A = _build_toeplitz(f)
            # Solve LP
            g = cp.Variable(n, nonneg=True)
            constraints = [A @ g <= M * np.ones(2*n-1)]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
            try:
                prob.solve(solver=cp.ECOS, warm_start=False, max_iters=5000, abstol=1e-9, reltol=1e-9)
            except Exception:
                break
            if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                break
            g_val = g.value
            if g_val is None:
                break
            sum_g = g_val.sum()
            if sum_g <= 0:
                break
            # normalize g to sum 1
            g_norm = g_val / sum_g
            # mix
            f_new = (1 - t_mix) * f + t_mix * g_norm
            # ensure sum 1 (should be)
            # compute score
            sc_new = _score(f_new)
            if sc_new < best_score:
                best_score = sc_new
                best_f = f_new.copy()
                f = f_new.copy()
            else:
                # No improvement, maybe break or continue with reduced t?
                break

    # hill climbing phase
    # use simple random mass transfer while time remains
    vec = best_f.copy()
    # recompute best_score accordingly
    best_score = _score(vec)
    while time.monotonic() < deadline:
        # random move
        i = rng.integers(0, n)
        if vec[i] == 0.0:
            continue
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # choose delta: up to 30% of source
        frac = rng.random() * 0.30
        delta = vec[i] * frac
        if delta <= 0:
            continue
        cand = vec.copy()
        cand[i] -= delta
        cand[j] += delta
        # project to simplex to keep sum 1 and nonnegative
        cand = _project_simplex(cand)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            vec = cand
            # also update best_f
            best_f = cand.copy()
    # Return best_f
    return [float(x) for x in best_f]

import time
import numpy as np

def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

def _score(vec, L=None):
    n = vec.size
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)

def _max_conv(vec, L=None):
    n = vec.size
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv.max(), conv

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    if "height_sequence_1" in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
    else:
        n = 256
        parent_arr = None

    # initial vectors
    base = _sqrt_series(n)
    base = base / base.sum()
    best_vec = base.copy()
    best_sc = _score(best_vec)
    if parent_arr is not None and parent_arr.size > 0:
        if parent_arr.sum() == 0:
            parent_norm = parent_arr
        else:
            parent_norm = parent_arr / parent_arr.sum()
        sc_parent = _score(parent_norm)
        if sc_parent < best_sc:
            best_vec = parent_norm.copy()
            best_sc = sc_parent

    # pre‑compute FFT length
    L = 1 << ((2 * n - 1).bit_length())

    # main search loop
    cur = best_vec.copy()
    while time.monotonic() < deadline:
        # evaluate convolution
        max_c, conv_vec = _max_conv(cur, L)
        # pick an active lag (first occurrence)
        lag = int(np.argmax(conv_vec))

        # decide move type
        r = rng.random()
        if r < 0.4:
            # Frank‑Wolfe style move towards index with smallest gradient
            i_start = max(0, lag - (n - 1))
            i_end = min(lag, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            grad = np.empty_like(cur)
            grad.fill(0.0)
            grad[idx] = 2.0 * cur[lag - idx]   # gradient of conv at lag
            j_min = int(np.argmin(grad))
            step = rng.random() * 0.4   # up to 40 % of total mass
            cand = (1.0 - step) * cur
            cand[j_min] += step
        elif r < 0.75:
            # simple mass transfer between two indices
            i = rng.integers(0, n)
            if cur[i] == 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            frac = rng.random() * 0.5   # up to 50 % of source amount
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
        else:
            # random Dirichlet mixing
            t = rng.random() * 0.3   # mixing weight
            g = rng.gamma(1.0, size=n)
            g = g / g.sum()
            cand = (1.0 - t) * cur + t * g

        sc = _score(cand, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
            cur = cand.copy()
        # optionally stop early if bound is already very good
        if best_sc < 1.5028:
            break

    return [float(x) for x in best_vec]

import time
import numpy as np

# ------------------------------------------------------------
# projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# analytic sqrt‑series (first n coeffs of a*a = 1+…+x^{2n‑2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# exact AC1 bound (same as evaluate_sequence)
# ------------------------------------------------------------
def _score(vec: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    return 2.0 * vec.size * conv.max() / (vec.sum() ** 2)


# ------------------------------------------------------------
# fast max‑convolution (returns max value and full vector)
# ------------------------------------------------------------
def _max_conv(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    return conv.max(), conv


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # length of the problem
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    else:
        parent = None
        n = 256

    # ---------- initial candidates ----------
    sqrt_vec = _sqrt_series(n).astype(np.float64)
    sqrt_vec /= sqrt_vec.sum()

    best_vec = sqrt_vec.copy()
    L = 1 << ((2 * n - 1).bit_length())      # FFT length (power of two)
    best_sc = _score(best_vec, L)

    if parent is not None and parent.size > 0:
        parent_norm = parent / max(parent.sum(), 1.0)
        parent_sc = _score(parent_norm, L)
        if parent_sc < best_sc:
            best_vec = parent_norm.copy()
            best_sc = parent_sc

    cur = best_vec.copy()                     # current point for local search

    # ---------- main stochastic optimisation ----------
    while time.monotonic() < deadline:
        # current auto‑correlation
        max_c, conv_vec = _max_conv(cur, L)
        lag = int(np.argmax(conv_vec))        # one of the active lags

        r = rng.random()

        # ---- 1. Frank‑Wolfe style move (gradient of the active lag) ----
        if r < 0.4:
            i_start = max(0, lag - (n - 1))
            i_end = min(lag, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            grad = np.zeros_like(cur)
            grad[idx] = 2.0 * cur[lag - idx]          # sub‑gradient of conv[lag]
            min_val = grad.min()
            candidates = np.where(grad == min_val)[0]
            j = int(rng.choice(candidates))
            step = rng.random() * 0.4                 # up to 40 % of total mass
            cand = (1.0 - step) * cur
            cand[j] += step                           # stay on the simplex automatically

        # ---- 2. Simple pairwise mass transfer with simplex projection ----
        elif r < 0.75:
            i = rng.integers(0, n)
            if cur[i] == 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            frac = rng.random() * 0.5                 # up to 50 % of source amount
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)

        # ---- 3. Random Dirichlet mixing (exploratory jump) ----
        else:
            t = rng.random() * 0.3                     # mixing weight
            g = rng.gamma(1.0, size=n)
            g /= g.sum()
            cand = (1.0 - t) * cur + t * g

        # evaluate candidate
        sc = _score(cand, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
            cur = cand.copy()                          # continue from the improvement

        # early exit if we already beat the target bound
        if best_sc < 1.5028:
            break

    # ---------- final return ----------
    return [float(x) for x in best_vec]
