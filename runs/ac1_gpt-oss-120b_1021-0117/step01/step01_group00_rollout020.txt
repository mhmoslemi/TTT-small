def bound(seq):

    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    m = 2*n - 1

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a sequence of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Obtain problem size and a valid fallback (the provided parent, if any)
    # ------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = int(parent.shape[0])
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:                     # no parent supplied – make a simple fallback
        n = 256
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # Helper: build a Sidon (B2) set by greedy insertion.
    # Returns the list of indices that form the set.
    # ------------------------------------------------------------------
    def greedy_sidon(order):
        idx = []
        # sums[i] == True  iff some pair already uses i as i+j
        sums = np.zeros(2 * n - 1, dtype=bool)
        for i in order:
            s_self = i + i
            if sums[s_self]:
                continue
            conflict = False
            for j in idx:
                if sums[i + j]:
                    conflict = True
                    break
            if conflict:
                continue
            # accept i
            idx.append(i)
            sums[s_self] = True
            for j in idx[:-1]:
                sums[i + j] = True
        return idx

    # ------------------------------------------------------------------
    # Try a few different insertion orders (deterministic then random)
    # ------------------------------------------------------------------
    attempts = 0
    while time.monotonic() < deadline and attempts < 15:
        if attempts == 0:
            order = np.arange(n, dtype=int)                # natural order
        else:
            order = rng.permutation(n)                     # random order
        idx = greedy_sidon(order)
        m = len(idx)
        if m < 2:                     # need at least two non‑zeros to beat uniform
            attempts += 1
            continue

        # Build candidate: value 1.0 on the Sidon positions, 0 elsewhere
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = 1.0

        # Evaluate via the official evaluator (fast enough for our n)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score

        attempts += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
