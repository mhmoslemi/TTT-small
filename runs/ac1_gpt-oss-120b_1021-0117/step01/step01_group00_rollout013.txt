def compute_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    # compute convolution via FFT with length L = 1 << (2*n-1).bit_length()
    L = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a*sum_a)

def _conv_max(a: np.ndarray) -> float:
    """Return max of a full convolution a*a using FFT (real inputs)."""
    n = a.size
    # length for FFT (next power of two >= 2n-1)
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return float(np.max(conv))
