import numpy as np, time, math

def _conv_self(a):
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[:2*n-1]
    return conv

def _score(a):
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0 * (i-1) + 1.0) / (2.0 * i)
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Initialize sequence

    # Ensure sum is not too tiny
    if seq.sum() < 0.01:
        seq += 1.0

    best = seq.copy()
    best_score = _score(best)

    # Current sequence for simulated annealing
    cur = seq.copy()
    cur_score = best_score

    # Annealing parameters
    T = 0.1
    decay = 0.99995

    while time.monotonic() < deadline:
        # compute convolution and gradient based on current max positions
        conv = _conv_self(cur)
        max_conv = conv.max()
        max_idxs = np.flatnonzero(conv == max_conv)

        # compute gradient (derivative of max conv w.r.t each element)
        g = np.zeros(n, dtype=np.float64)
        for k in max_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # vectorized portion
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * cur[j]

        # decide move type
        if rng.random() < 0.7:

            src = int(np.argmax(g))
            dst = int(np.argmin(g))
            if src == dst:
                # fallback to random transfer
                src = rng.integers(0, n)
                dst = rng.integers(0, n)
        else:

            src = rng.integers(0, n)
            dst = rng.integers(0, n)

import numpy as np
import time
import math

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (fast and memory‑efficient)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator (fast version)."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score non‑negative step function within the time budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------------------
    # Initialise – use the supplied parent if present, otherwise a sqrt series.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = cur.size
    else:
        n = 5582
        cur = _sqrt_series(n)

    if cur.sum() < 0.01:                     # safeguard against degenerate sum
        cur += 1.0

    best = cur.copy()
    best_score = _score(best)

    cur_score = best_score

    # Simulated‑annealing style parameters
    T = 0.12
    decay = 0.99994

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # Core update – either a gradient‑guided mass transfer,
        # a random transfer, or a block scaling.
        # --------------------------------------------------------------
        op = rng.random()

        if op < 0.55:          # gradient‑guided transfer
            conv = _conv_self(cur)
            k = int(conv.argmax())                # index attaining the current max
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            jdx = k - idx
            grad = np.zeros_like(cur)
            grad[idx] = 2.0 * cur[jdx]             # derivative of max‑conv wrt each a_i

            src = int(np.argmax(grad))
            dst = int(np.argmin(grad))
            if src == dst:                         # very unlikely, fallback to random
