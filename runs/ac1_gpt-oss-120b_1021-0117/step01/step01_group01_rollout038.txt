def _score_fft(a, n, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)
```

import time
import numpy as np

def _fft_len_needed(n: int) -> int:
    """smallest power‑of‑two length for a linear convolution of size n."""
    conv_len = 2 * n - 1
    return 1 << ((conv_len - 1).bit_length())


def _score_fft(vec: np.ndarray, n: int, fft_len: int) -> float:
    """Fast evaluation of the AC‑1 bound for a non‑negative vector."""
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    s = vec.sum()
    # sum is never zero for our constructions; safeguard division anyway
    return 2.0 * n * max_b / (s * s) if s > 0 else np.inf


def _gen_binomial_raw(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (unnormalised)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a

    fft_len = _fft_len_needed(n)

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    best_vec = None
    best_score = np.inf

    def consider(vec: np.ndarray):
        nonlocal best_vec, best_score
        vec = np.clip(vec, 0.0, 1000.0)
        if vec.sum() == 0.0:
            return
        sc = _score_fft(vec, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()

    # uniform (baseline)
    consider(np.ones(n, dtype=np.float64))

    # raw binomial, normalised
    bin_raw = _gen_binomial_raw(n)
    consider(bin_raw / bin_raw.sum())

    # include the parent construction if it exists
    try:

    except Exception:

    # -----------------------------------------------------------------
