def compute_max_conv(seq):
    n = seq.shape[0]
    nfft = 1 << (2*n - 1).bit_length()   # next power of two >= 2n-1

import numpy as np
import time

def compute_max_convolution(seq: np.ndarray, nfft: int) -> float:
    """Return max value of the full self‑convolution of `seq`."""
    # FFT of zero‑padded sequence
    fft_seq = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(fft_seq * fft_seq, n=nfft)
    # Only the first 2*len(seq)-1 entries are the true convolution
    n = seq.shape[0]
    return float(conv[: 2 * n - 1].max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # --------------------------------------------------------------------
    # 1.  Obtain a starting vector (parent if supplied, otherwise Sidon‑like)
    # --------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the harness when available
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        n = int(parent.shape[0])
        have_parent = True
    except Exception:
        # fallback: classic Sidon‑like sequence
        n = 256
        parent = np.empty(n, dtype=np.float64)
        parent[0] = 1.0
        for i in range(1, n):
            parent[i] = parent[i - 1] * (1.0 - 1.0 / (2.0 * i))
        parent = np.clip(parent, 0.0, 1000.0)
        have_parent = False

    # --------------------------------------------------------------------
    # 2.  Build an initial ordering that pairs large with small values.
    #     This is the classic “alternating” pattern derived from the
    #     rearrangement inequality.
    # --------------------------------------------------------------------
    sorted_vals = np.sort(parent)                     # ascending
    seq = np.empty_like(parent)
    seq[::2] = sorted_vals[::-1][: seq[::2].size]    # largest, 2nd largest, …
    seq[1::2] = sorted_vals[: seq[1::2].size]        # smallest, 2nd smallest, …

    # Guard against an (unlikely) zero‑sum situation
    if seq.sum() < 0.01:
        seq = np.maximum(seq, 0.01 / n)

    # --------------------------------------------------------------------
    # 3.  Prepare FFT parameters (power of two ≥ 2n‑1)
    # --------------------------------------------------------------------
    nfft = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------------
    # 4.  Initialise best candidate
    # --------------------------------------------------------------------
    cur = seq.copy()
    cur_max = compute_max_convolution(cur, nfft)
    best_seq = cur.copy()
    best_max = cur_max

    if have_parent:
        parent_max = compute_max_convolution(parent, nfft)
        if parent_max < best_max:
            best_seq = parent.copy()
            best_max = parent_max
            cur = best_seq.copy()
            cur_max = best_max

    # --------------------------------------------------------------------
    # 5.  Main hill‑climbing loop (swap two entries, keep if not worse)
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform the swap in‑place
        cur[i], cur[j] = cur[j], cur[i]
        new_max = compute_max_convolution(cur, nfft)

        if new_max <= cur_max:
            # keep the move
            cur_max = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_seq = cur.copy()
        else:
            # reject: revert the swap
            cur[i], cur[j] = cur[j], cur[i]

        # occasional reshuffle to escape plateaus

            cur[:] = cur[perm]
            cur_max = compute_max_convolution(cur, nfft)
            if cur_max < best_max:
                best_max = cur_max
                best_seq = cur.copy()

    # --------------------------------------------------------------------
    # 6.  Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------------
    return [float(v) for v in best_seq]
