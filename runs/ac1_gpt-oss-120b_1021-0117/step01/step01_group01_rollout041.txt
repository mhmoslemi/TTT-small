def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # Determine n
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # Prepare FFT size
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def conv_max_fft(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(conv.max())

    def score_of(a):
        maxb = conv_max_fft(a)
        s = a.sum()
        return float(2 * n * maxb / (s * s))

    # 1. Generate candidate families
    best = None
    best_score = np.inf

    # binomial base
    binom = np.empty(n, dtype=np.float64)
    binom[0] = 1.0
    for i in range(n - 1):
        binom[i + 1] = binom[i] * (2 * i + 1) / (2 * i + 2)
    # candidate 1: normalized binomial
    cand = binom / binom.sum()
    sc = score_of(cand)
    best, best_score = cand.copy(), sc

    # Exponential family
    for lam in np.geomspace(1e-5, 1e-2, num=15):
        exp_seq = np.exp(-lam * np.arange(n))
        exp_seq = exp_seq / exp_seq.sum()
        sc = score_of(exp_seq)
        if sc < best_score:
            best, best_score = exp_seq.copy(), sc

    # Power law family
    for alpha in np.linspace(0.1, 1.0, num=19):
        idx = np.arange(1, n + 1, dtype=np.float64)
        pl = idx ** (-alpha)
        pl = pl / pl.sum()
        sc = score_of(pl)
        if sc < best_score:
            best, best_score = pl.copy(), sc

    # Gaussian family with std's grid
    for sigma in np.geomspace(0.5, n / 2, num=12):
        x = np.arange(n, dtype=np.float64)
        gauss = np.exp(-0.5 * ((x) / sigma) ** 2)  # centered at 0, decays
        gauss = gauss / gauss.sum()
        sc = score_of(gauss)
        if sc < best_score:
            best, best_score = gauss.copy(), sc

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    hann = hann / hann.sum()
    sc = score_of(hann)
    if sc < best_score:
        best, best_score = hann.copy(), sc

    # Mixed binomial * exp
    for gamma in np.geomspace(1e-5, 5e-3, num=12):
        mix = binom * np.exp(-gamma * np.arange(n))
        mix = mix / mix.sum()
        sc = score_of(mix)
        if sc < best_score:
            best, best_score = mix.copy(), sc

    # Random Dirichlet samples
    for _ in range(8):
        rnd = rng.random(n)
        rnd = rnd / rnd.sum()
        sc = score_of(rnd)
        if sc < best_score:
            best, best_score = rnd.copy(), sc

    # At this point best is some candidate

    cur = best.copy()
    cur_score = best_score

    # Simulated annealing hill climbing
    temp = 1.0
    # schedule for temperature decay
    decay = 0.9995
    min_frac = 0.01
    max_frac = 0.5

    while time.monotonic() < deadline:
        # Choose move type
        if rng.random() < 0.7:
            # single transfer
            i = rng.integers(0, n)
            if cur[i] <= 0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            frac = rng.uniform(min_frac, max_frac)
            delta = cur[i] * frac
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        else:
            # perturb many entries (smooth)
            idx = rng.choice(n, size=min(30, n), replace=False)
            # random additive changes respecting positivity
            changes = cur[idx] * rng.uniform(-0.2, 0.2, size=idx.shape)
            new = cur.copy()
            new[idx] += changes
            # clip negatives
            new = np.where(new < 0, 0.0, new)

        # renormalize to keep sum unchanged
        s = new.sum()
        if s == 0:
            continue
        new = new / s

        new_score = score_of(new)

        # Accept if better or with probability
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(temp, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        # temperature decay
        temp = max(temp * decay, 1e-8)

    # final check and return
    final = [float(x) for x in best]
    # ensure sum >=0.01 to satisfy evaluator
    if sum(final) < 0.01:
        # fallback to uniform
        final = [1.0 / n] * n
    # optional final evaluation (may raise if invalid)
    try:
        _ = evaluate_sequence(final)
    except Exception:
        final = [float(x) for x in cand]  # fallback to binomial normalized
    return final

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)  # keep margin

    # Determine length n
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 256

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def max_conv(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(conv.max())

    def evaluate(a):
        m = max_conv(a)
        s = a.sum()
        return float(2 * n * m / (s * s))

    # generate candidates
    best = None
    best_score = np.inf

    # Binomial base
    binom = np.empty(n, dtype=np.float64)
    binom[0] = 1.0
    for i in range(n-1):
        binom[i+1] = binom[i] * (2*i + 1) / (2*i + 2)
    cand = binom / binom.sum()
    cur_score = evaluate(cand)
    best, best_score = cand.copy(), cur_score

    # exponential families
    for lam in np.geomspace(1e-5, 1e-2, num=12):
        seq = np.exp(-lam * np.arange(n, dtype=np.float64))
        seq = seq / seq.sum()
        sc = evaluate(seq)
        if sc < best_score:
            best, best_score = seq.copy(), sc

    # power law families
    for alpha in np.linspace(0.1, 1.0, 19):
        idx = np.arange(1, n+1, dtype=np.float64)
        seq = idx ** (-alpha)
        seq = seq / seq.sum()
        sc = evaluate(seq)
        if sc < best_score:
            best, best_score = seq.copy(), sc

    # Gaussian families
    for sigma in np.geomspace(0.5, n/2, num=12):
        x = np.arange(n, dtype=np.float64)
        seq = np.exp(-0.5 * (x / sigma) ** 2)
        seq = seq / seq.sum()
        sc = evaluate(seq)
        if sc < best_score:
            best, best_score = seq.copy(), sc

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    hann = hann / hann.sum()
    sc = evaluate(hann)
    if sc < best_score:
        best, best_score = hann.copy(), sc

    # binomial multiplied by growing exponential (shift weight to the tail)
    for gamma in np.linspace(-0.001, 0.001, 13):
        # avoid overflow
        ex = np.exp(gamma * np.arange(n, dtype=np.float64))
        mix = binom * ex
        mix = mix / mix.sum()
        sc = evaluate(mix)
        if sc < best_score:
            best, best_score = mix.copy(), sc

    # random samples
    for _ in range(6):
        rnd = rng.random(n)
        rnd = rnd / rnd.sum()
        sc = evaluate(rnd)
        if sc < best_score:
            best, best_score = rnd.copy(), sc

    # start local search
    cur = best.copy()
    cur_score = best_score
    temp = 1.0
    decay = 0.9998
    min_frac = 0.02
    max_frac = 0.4

    while time.monotonic() < deadline:
        # Random mass transfer move
        i = rng.integers(0, n)
        if cur[i] <= 0:
            continue
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        frac = rng.uniform(min_frac, max_frac)
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # Additional perturbation move with some probability
        if rng.random() < 0.2:
            idx = rng.choice(n, size=min(20, n), replace=False)
            add = new[idx] * rng.uniform(-0.15, 0.15, size=idx.shape)
            new[idx] += add
            new = np.where(new < 0, 0.0, new)

        # Renormalise to keep sum=1 (transfer move already keeps sum)
        s = new.sum()
        if s == 0:
            continue
        new = new / s

        new_score = evaluate(new)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(temp, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temp = max(temp * decay, 1e-8)

    # Final clip and conversion
    final = [float(x) for x in best]
    # Guarantee sum>0.01 (should hold)
    if sum(final) < 0.01:
        final = [1.0 / n] * n
    # Validate if possible
    try:
        _ = evaluate_sequence(final)
    except Exception:
        # fallback to uniform or binomial
        fallback = [float(x) for x in cand]
        final = fallback
    return final

def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution of a with itself via FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of a*a."""
