def gradient_move(a, step_factor):
    n = a.shape[0]
    max_val, conv = fast_max_conv(a)
    idx = int(conv.argmax())
    # compute gradient
    g = np.zeros_like(a)
    lo = max(0, idx - (n-1))
    hi = min(idx, n-1)
    # vectorized
    js = np.arange(lo, hi+1)
    ks = idx - js
    g[js] = 2.0 * a[ks]
    max_g = g.max()
    if max_g <= 0:
        return None, None
    alpha = step_factor * a.max() / max_g
    cand = a - alpha * g
    np.maximum(cand, 0.0, out=cand)
    s = cand.sum()
    if s <= 0:
        return None, None
    cand = cand / s
    return cand, max_val

import time
import numpy as np

# ---------- fast convolution utilities ----------
def fast_max_conv(a: np.ndarray):
    """Return (max value, full convolution) of a with itself using FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return float(conv.max()), conv


# ---------- elementary move generators ----------
def gradient_move(a: np.ndarray, step_factor: float):
    """One gradient‑descent step using the lag with maximal autocorrelation."""
    max_val, conv = fast_max_conv(a)
    idx = int(conv.argmax())
    n = a.shape[0]

    # gradient of the max‑lag term: g_j = 2 * a_{idx-j} when index valid
    g = np.zeros_like(a)
    lo = max(0, idx - (n - 1))
    hi = min(idx, n - 1)
    js = np.arange(lo, hi + 1, dtype=np.int64)
    ks = idx - js
    g[js] = 2.0 * a[ks]

    max_g = g.max()
    if max_g <= 0.0:
        return None

    # step size scaled to keep numbers reasonable
    alpha = step_factor * a.max() / max_g
    cand = a - alpha * g
    np.maximum(cand, 0.0, out=cand)
    s = cand.sum()
    if s <= 0.0:
        return None
    cand /= s
    return cand


def random_transfer_move(a: np.ndarray, rng: np.random.Generator):
    """Move a random fraction of mass from one index to another."""
    n = a.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    if a[i] == 0.0:
        return None
    eps = rng.random() * a[i] * 0.5
    cand = a.copy()
    cand[i] -= eps
    cand[j] += eps
    s = cand.sum()
    if s <= 0.0:
        return None
    cand /= s
    return cand


def jitter_move(a: np.ndarray, rng: np.random.Generator, sigma: float = 0.02):
    """Multiplicative log‑normal jitter."""
    jitter = np.exp(rng.normal(0.0, sigma, size=a.shape[0]))
    cand = a * jitter
    s = cand.sum()
    if s <= 0.0:
        return None
    cand /= s
    return cand


# ---------- parametric families ----------
def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (decreasing power law)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def generate_geometric(n: int, r: float) -> np.ndarray:
    """a_i ∝ r^{i}  (geometric decay)."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    a /= a.sum()
    return a


def catalan_weights(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w
