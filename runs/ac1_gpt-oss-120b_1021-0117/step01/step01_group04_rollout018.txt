import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic optimisation of a non‑negative step function for the AC1 bound.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # reserve a couple of seconds for the final return
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # length and initial vector
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        init = parent.copy()
    except Exception:
        n = 256
        init = _gen_binomial(n)

    L = _next_pow2(2 * n - 1)

    # initial best
    cur = init.copy()
    best = cur.copy()
    best_score = _ac1_score(best, n, L)

    # compare with a fresh binomial if no parent was used
    if parent is None:
        alt = _gen_binomial(n)
        alt_score = _ac1_score(alt, n, L)
        if alt_score < best_score:
            cur = alt.copy()
            best = alt.copy()
            best_score = alt_score

    cur_score = best_score

    # --------------------------------------------------------------
    # hyper‑parameters
    temperature = 1.0
    decay = 0.9997
    min_block = max(2, n // 200)
    max_block = max(2, n // 12)
    transfer_max = 0.4
    grad_eps = 0.02
    target = 1.50274349

    while time.monotonic() < deadline:
        move = rng.random()

        # ----------------------------------------------------------
        # 1) gradient guided mass transfer  (≈45 % of moves)
        if move < 0.45:
            # current convolution & index of the maximal entry
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            # gradient contribution: each a_i appears with partner a_{kmax-i}
            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)

            # double for distinct partners
            self_mask = mask & (rev != idx)
            g[self_mask] *= 2.0

            # direction that preserves total mass (sum(d)=0)
            mean_g = g.mean()
            d = mean_g - g
