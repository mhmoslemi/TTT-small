import numpy as np, time

def _conv_self(a):
    n = a.size
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2*n - 1]
    return conv

def _score(a):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _conv_self(a)
    max_b = b.max()
    return float(2 * n * max_b / (s * s))

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0*i - 1.0) / (2.0*i)
    return a

    # initialise base sequence

        n = parent.size
    else:
        n = 5582
        parent = None

    # sqrt series as fallback
    sqrt_seq = _sqrt_series(n)
    sqrt_seq = sqrt_seq.astype(np.float64)
    # normalize to sum 1
    sqrt_seq /= sqrt_seq.sum()

    best = sqrt_seq.copy()
    best_score = _score(best)

    if parent is not None:
        parent = parent / parent.sum()
        parent_score = _score(parent)
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # Precompute indices array for speed

    # parameters for moves
    early_cut = n // 4
    late_cut = 3 * n // 4

import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _conv_self(a)
    return float(2.0 * n * b.max() / (s * s))


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score non‑negative step function under the given budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Starting point – supplied parent if available, otherwise sqrt series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    else:
        n = 5582
        parent = None

    sqrt_seq = _sqrt_series(n)
    sqrt_seq = sqrt_seq.astype(np.float64)
    sqrt_seq /= sqrt_seq.sum()               # normalise – scale does not affect score

    best = sqrt_seq.copy()
    best_score = _score(best)

    if parent is not None:
        parent = parent / parent.sum()
        parent_score = _score(parent)
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # Parameters for the search.
    # ------------------------------------------------------------------
    idx_arr = np.arange(n, dtype=np.int64)
    early_cut = n // 4
    late_cut = 3 * n // 4

    temperature = best_score * 0.01           # small initial temperature for occasional uphill moves
    decay = 0.99995

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # convolution of the current best – needed to locate the dominant shift
        conv_best = _conv_self(best)
        kmax = int(conv_best.argmax())

        # partner values for that shift (a_{kmax - j} where defined, else 0)
        rev = np.zeros_like(best)
        indices = kmax - idx_arr
        mask = (indices >= 0) & (indices < n)
        rev[mask] = best[indices[mask]]

        # product contribution of each entry to the dominant autocorrelation term
        contr = best * rev

        # ------------------------------------------------------------------
        # Choose a move.
        # ------------------------------------------------------------------
        r = rng.random()
        cand = best.copy()

        if r < 0.40:
            # Targeted mass transfer from high‑contribution indices to a zero‑partner edge.
            thresh = np.percentile(contr, 90)
            src_idx = np.where(contr >= thresh)[0]
            if src_idx.size:
                # remove a random fraction (up to 50 %) from each source
                fraction = rng.random() * 0.5
                delta_vec = cand[src_idx] * fraction
                total_removed = delta_vec.sum()
                cand[src_idx] -= delta_vec

                # move the removed mass to a random edge where rev==0 (no effect on the max term)
                edge_idx = np.where(rev == 0)[0]
                if edge_idx.size:
                    dst = rng.choice(edge_idx)
                else:
                    dst = rng.integers(0, n)
                cand[dst] += total_removed
            else:
                # fall back to a simple early→late transfer
                src = rng.integers(0, early_cut)
                dst = rng.integers(late_cut, n)
                max_tf = min(cand[src], 1000.0 - cand[dst])
                if max_tf > 0:
                    delta = max_tf * rng.random() * 0.3
                    cand[src] -= delta
                    cand[dst] += delta

        elif r < 0.65:
            # More aggressive early→late transfer
            src = rng.integers(0, early_cut)
            dst = rng.integers(late_cut, n)
            max_tf = min(cand[src], 1000.0 - cand[dst])
            if max_tf > 0:
                delta = max_tf * rng.random() * 0.4
                cand[src] -= delta
                cand[dst] += delta

        elif r < 0.85:
            # Block scaling (log‑normal factor around 1)
            width = int(rng.integers(max(2, n // 200), max(3, n // 12)))
            start = rng.integers(0, n - width + 1)
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:start + width] *= factor

        elif r < 0.95:
            # Small additive Gaussian noise
            cand += rng.normal(0.0, 0.02, size=n)

        else:
            # Random permutation – reshuffle without changing multiset
            perm = rng.permutation(n)
            cand = cand[perm]

        # ------------------------------------------------------------------
        # Enforce bounds and renormalise (renorm does not change the objective).
        # ------------------------------------------------------------------
        np.clip(cand, 0.0, 1000.0, out=cand)
        total = cand.sum()
        if total > 0:
            cand /= total

        sc = _score(cand)
        if np.isfinite(sc):
            if sc < best_score:
                best = cand
                best_score = sc
                temperature = best_score * 0.01      # reset temperature after improvement
            else:
                # occasional uphill acceptance
                if temperature > 0 and rng.random() < np.exp((best_score - sc) / temperature):
                    best = cand
                    best_score = sc
                    temperature *= decay

        temperature *= decay

    # ------------------------------------------------------------------
    # Return the best discovered sequence as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
