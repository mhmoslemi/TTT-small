import numpy as np
import time
import math

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Load / construct a starting sequence
    # ------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
    except Exception:
        # fallback: uniform vector of length 256
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    n = int(parent.size)

    # ------------------------------------------------------------------
    # 2. Helper: full convolution via FFT (vectorised, O(n log n))
    # ------------------------------------------------------------------
    def full_convolve(x):
        L = 1 << ((2 * x.size - 1).bit_length())   # next power of two
        fx = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
        return conv

    # ------------------------------------------------------------------
    # 3. Evaluate several simple initialisations and pick the best
    # ------------------------------------------------------------------
    def bound_from(conv, total):
        return float(2.0 * n * conv.max() / (total * total))

    candidates = []

    # a) original parent
    candidates.append(parent.copy())

    # b) a random permutation of the parent (same multiset)
    perm = rng.permutation(n)
    candidates.append(parent[perm].copy())

    # c) uniform distribution with same total mass as parent
    uniform_seq = np.full(n, parent.sum() / n, dtype=np.float64)
    candidates.append(uniform_seq)

    # pick the best start
    best_seq = candidates[0]
    conv = full_convolve(best_seq)
    best_score = bound_from(conv, best_seq.sum())
    best_max = conv.max()
    for cand in candidates[1:]:
        conv_c = full_convolve(cand)
        sc = bound_from(conv_c, cand.sum())
        if sc < best_score:
            best_seq = cand.copy()
            conv = conv_c
            best_score = sc
            best_max = conv.max()

    # ------------------------------------------------------------------
    # 4. Prepare for iterative optimisation (simulated‑annealing style)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    sum_a = a.sum()
    factor = 2.0 * n / (sum_a * sum_a)        # constant throughout the search
    max_conv = best_max
    current_score = factor * max_conv

    # fast in‑place convolution update for a single index change
    def apply_delta(arr, conv_arr, idx, d):
        """Change arr[idx] by d (d may be negative) and update conv_arr."""
        old = arr[idx]                     # value before modification
        # contribution of this index to all sums i+idx
        conv_arr[idx : idx + n] += 2.0 * d * arr
        arr[idx] = old + d
        conv_arr[2 * idx] += d * d         # diagonal term correction

    # annealing parameters
    temperature = max(0.1, current_score)   # start fairly warm
    cooling = 0.9995
    min_temp = 1e-8
    move_counter = 0
    alpha = 0.3                               # max fraction moved from a source

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline and temperature > min_temp:
        move_counter += 1

        if rng.random() < 0.6:   # swap move (more frequent)
            i, j = rng.integers(0, n, size=2, endpoint=False)
            if i == j:
                continue
            d = a[j] - a[i]
            if abs(d) < 1e-12:
                continue

            # perform swap via two delta updates
            apply_delta(a, conv, i, d)
            apply_delta(a, conv, j, -d)

            new_max = conv.max()
            new_score = factor * new_max
        else:                     # transfer move
            i = rng.integers(0, n)
            if a[i] <= 1e-12:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # amount that can be moved while respecting bounds
            max_t = min(alpha * a[i], 1000.0 - a[j])
            if max_t <= 0.0:
                continue
            t = rng.random() * max_t
            if t <= 1e-12:
                continue

            apply_delta(a, conv, i, -t)
            apply_delta(a, conv, j, t)

            new_max = conv.max()
            new_score = factor * new_max

        # ------------------------------------------------------------------
        # 6. Acceptance test (Metropolis)
        # ------------------------------------------------------------------
        if new_score < current_score:
            accept = True
        else:
            # probability of accepting a worsening move
            prob = math.exp(-(new_score - current_score) / temperature) if temperature > 0 else 0.0
            accept = rng.random() < prob

        if accept:
            current_score = new_score
            max_conv = new_max
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                # occasionally re‑scale temperature upward to keep exploration alive
                temperature = max(temperature, best_score * 0.2)
        else:
            # revert the move (reverse order!)
            if move_counter % 5 == 0 or (move_counter and move_counter % 10 == 0):
                # harmless extra check – the branch has been taken already
                pass
            if rng.random() < 0.5:   # match the type we just performed
                # was a swap
                # revert: apply opposite deltas in reverse order
                apply_delta(a, conv, j, d)
                apply_delta(a, conv, i, -d)
            else:
                # transfer revert
                apply_delta(a, conv, j, -t)
                apply_delta(a, conv, i, t)
            # current_score stays unchanged; max_conv unchanged

        # ------------------------------------------------------------------
        # 7. Temperature decay & periodic housekeeping
        # ------------------------------------------------------------------
        temperature *= cooling
        if move_counter % 5000 == 0:
            # small temperature boost to avoid premature freezing
            temperature = max(temperature, best_score * 0.05)

    # ------------------------------------------------------------------
    # 8. Final sanity check (use official evaluator once)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        if not math.isfinite(final_score):
            raise Exception
        # keep the sequence we have
        out_seq = best_seq
    except Exception:
        # fallback to the original parent (guaranteed valid)
        out_seq = parent

    return [float(x) for x in out_seq]
