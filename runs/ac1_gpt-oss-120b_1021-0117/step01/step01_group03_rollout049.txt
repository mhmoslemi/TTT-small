import numpy as np
import time

# ----------------------------------------------------------------------
# FFT based convolution (matrix‑free, O(n log n))
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑two FFT."""
    n = x.size + y.size - 1
    m = 1 << ((n - 1).bit_length())          # next power of two ≥ n
    fx = np.fft.rfft(x, n=m)
    fy = np.fft.rfft(y, n=m)
    prod = fx * fy
    conv = np.fft.irfft(prod, n=m)[:n]
    return conv

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a."""
    return _conv_fft(a, a)

# ----------------------------------------------------------------------
# Objective used by the official evaluator (same formula)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
# Classical decreasing sequence a_i = C(2i,i)/4^i  (length‑n)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return seq

# Main search routine
# ----------------------------------------------------------------------
