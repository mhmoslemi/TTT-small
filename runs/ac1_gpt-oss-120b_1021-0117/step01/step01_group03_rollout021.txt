def _conv_self(a):
    n = a.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]

def _score_fft(a):
    conv = _conv_self(a)
    return 2.0 * a.size * conv.max() / (a.sum()**2)

import numpy as np, time

def _conv_self(a):
    n = a.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    return np.fft.irfft(fa*fa, n=m)[:2*n-1]

def _score_fft(a):
    conv = _conv_self(a)
    return 2.0 * a.shape[0] * conv.max() / (a.sum()**2)

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1,n):
        a[i] = a[i-1] * (2.0*i - 1.0) / (2.0*i)   # derived from earlier formula
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # length
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    else:
        n = 5582
        parent = None

    # initial candidates list
    candidates = []

    # add parent if exists
    if parent is not None:
        candidates.append(parent.copy())

    # sqrt series
    sqrt_seq = _sqrt_series(n)
    sqrt_seq = np.clip(sqrt_seq, 0.0, 1000.0)
    candidates.append(sqrt_seq.copy())

    # power law candidates
    for p in np.linspace(0.3, 2.0, 9):
        i = np.arange(1, n+1, dtype=np.float64)
        a = i ** (-p)
        a /= a.sum()  # normalize sum to 1
        candidates.append(a)

    # exponential decay candidates
    for lam in np.linspace(0.0005, 0.005, 7):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * i)
        a /= a.sum()
        candidates.append(a)

    best_seq = None
    best_score = np.inf

    # function to improve a sequence using gradient transfers
    def improve(seq, end_time):
        a = seq.copy()
        # ensure non-negative, clipped by bounds
        np.clip(a, 0.0, 1000.0, out=a)
        current_score = _score_fft(a)
        # main loop
        while time.monotonic() < end_time:
            # compute convolution and max index
            conv = _conv_self(a)
            m = int(np.argmax(conv))
            # gradient g: g_j = 2*a_{m-j} for valid indices
            g = np.zeros_like(a)
            low = max(0, m - (n - 1))
            high = min(n - 1, m)
            if low <= high:
                idx = np.arange(low, high+1)
                g[idx] = 2.0 * a[m - idx]
            # choose source and destination
            # source: largest g among entries with positive mass
            pos_mask = a > 1e-12
            if not np.any(pos_mask):
                break
            src_candidates = np.where(pos_mask)[0]
            src = src_candidates[np.argmax(g[src_candidates])]
            # destination: smallest g among entries not at upper bound
            up_mask = a < 1000.0 - 1e-12
            if not np.any(up_mask):
                break
            dst_candidates = np.where(up_mask)[0]
            dst = dst_candidates[np.argmin(g[dst_candidates])]
            if src == dst:
                # random perturbation fallback
                src = int(rng.integers(0, n))
                dst = int(rng.integers(0, n))
                if src == dst:
                    continue
            max_transfer = min(a[src], 1000.0 - a[dst])
            if max_transfer <= 0.0:
                continue
            # try decreasing fractions until improvement
            improved = False
            for frac in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
                delta = max_transfer * frac
                cand = a.copy()
                cand[src] -= delta
                cand[dst] += delta
                sc = _score_fft(cand)
                if sc < current_score - 1e-12:
                    a = cand
                    current_score = sc
                    improved = True
                    break
            if not improved:
                # occasional random block scaling
                if rng.random() < 0.03:
                    w = int(rng.integers(max(1, n // 300), max(2, n // 30)))
                    st = int(rng.integers(0, n - w + 1))
                    factor = float(np.exp(rng.normal(0.0, 0.06)))
                    block = a[st:st+w] * factor
                    np.clip(block, 0.0, 1000.0, out=block)
                    cand2 = a.copy()
                    cand2[st:st+w] = block
                    sc2 = _score_fft(cand2)
                    if sc2 < current_score:
                        a = cand2
                        current_score = sc2
            # occasional tiny Gaussian noise
            if rng.random() < 0.02:
                noise = rng.normal(0.0, 0.01, size=n)
                cand3 = a + noise
                np.clip(cand3, 0.0, 1000.0, out=cand3)
                sc3 = _score_fft(cand3)
                if sc3 < current_score:
                    a = cand3
                    current_score = sc3
        return a, current_score

    # allocate time for each initial candidate equally (roughly)

import time
import numpy as np

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (fast)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score_fft(a: np.ndarray) -> float:
    """Score used by the evaluator (scale‑invariant)."""
    conv = _conv_self(a)
    return 2.0 * a.size * conv.max() / (a.sum() ** 2)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic decreasing series a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def _improve(seq: np.ndarray, rng: np.random.Generator,
             end_time: float, n: int):
    """Greedy mass‑transfer search guided by the gradient of the current peak."""
    a = seq.copy()
    np.clip(a, 0.0, 1000.0, out=a)
    cur_score = _score_fft(a)

    while time.monotonic() < end_time:
        conv = _conv_self(a)
        m = int(np.argmax(conv))

        # gradient of the peak conv[m] w.r.t. a
        g = np.zeros_like(a)
        low = max(0, m - (n - 1))
        high = min(n - 1, m)
        if low <= high:
            idx = np.arange(low, high + 1)
            g[idx] = 2.0 * a[m - idx]

        # source = high‑gradient entry that still has mass
        pos_mask = a > 1e-12
        if not np.any(pos_mask):
            break
        src_candidates = np.where(pos_mask)[0]
        src = src_candidates[np.argmax(g[src_candidates])]

        # destination = low‑gradient entry that can receive mass
        up_mask = a < 1000.0 - 1e-12
        if not np.any(up_mask):
            break
        dst_candidates = np.where(up_mask)[0]
        dst = dst_candidates[np.argmin(g[dst_candidates])]

        if src == dst:
            src = int(rng.integers(0, n))
            dst = int(rng.integers(0, n))
            if src == dst:

        max_transfer = min(a[src], 1000.0 - a[dst])
        if max_transfer <= 0.0:

        improved = False
        for frac in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            delta = max_transfer * frac
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
            sc = _score_fft(cand)
            if sc < cur_score - 1e-12:
                a = cand
                cur_score = sc
                improved = True
                break

        if not improved:

                cur_score = sc3
