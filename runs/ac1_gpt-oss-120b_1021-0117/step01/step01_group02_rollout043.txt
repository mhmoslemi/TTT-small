import time
import numpy as np

def _conv_max(v: np.ndarray) -> float:
    """Maximum value of the linear convolution v*v."""
    n = v.size
    # use FFT for speed (real FFT)
    n_fft = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=n_fft)
    conv = np.fft.irfft(fv * fv, n=n_fft)[: 2 * n - 1]
    return float(conv.max())

    """
