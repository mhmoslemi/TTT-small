def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import numpy as np, time, math, random
    rng=np.random.default_rng(seed)
    deadline=time.monotonic()+max(0.0,float(budget_s)-2.0)
    # determine n
    try:
        parent=np.asarray(height_sequence_1,dtype=np.float64)
        if parent.ndim!=1 or parent.size==0:
            raise ValueError
        n=int(parent.size)
    except Exception:
        n=256
        parent=None
    L=_next_pow2(2*n-1)
    # base candidates
    # generate binomial
    base=np.empty(n,dtype=np.float64)
    base[0]=1.0
    for i in range(1,n):
        base[i]=base[i-1]*(2*i-1)/(2*i)
    # compute scores
    best_seq=base.copy()
    best_score=_ac1_score(best_seq,n,L)
    if parent is not None:
        parent_score=_ac1_score(parent,n,L)
        if parent_score<best_score:
            best_seq=parent.copy()
            best_score=parent_score
    # ensure best_seq within bounds (clip)
    np.clip(best_seq,0.0,1000.0,out=best_seq)
    # start local search
    cur=best_seq.copy()
    cur_score=best_score
    temperature=1.0
    decay=0.9995
    no_improve_steps=0
    # loop
    while time.monotonic()<deadline:
        move_type=rng.random()
        cand=cur.copy()
        if move_type<0.45:
            # block scaling
            min_width=max(1, n//300)
            max_width=max(2, n//30)
            width=int(rng.integers(min_width, max_width+1))
            start=int(rng.integers(0, n-width+1))
            factor=math.exp(rng.normal(0.0, 0.08))
            cand[start:start+width]*=factor
        elif move_type<0.8:
            # transfer mass
            i=int(rng.integers(0,n))
            j=int(rng.integers(0,n))
            if i!=j:
                frac=rng.random()*0.6  # up to 60%
                delta=cur[i]*frac
                cand[i]-=delta
                cand[j]+=delta
        elif move_type<0.95:
            # swap two entries
            i=int(rng.integers(0,n))
            j=int(rng.integers(0,n))
            cand[i],cand[j]=cand[j],cand[i]
        else:
            # global small perturbation (multiply random subset)
            mask=rng.random(n)<0.1
            factor=math.exp(rng.normal(0.0,0.05, size=mask.sum()))
            cand[mask]*=factor
        np.clip(cand,0.0,1000.0,out=cand)
        cand_score=_ac1_score(cand,n,L)
        # update best
        if cand_score<best_score:
            best_seq=cand.copy()
            best_score=cand_score
            no_improve_steps=0
            if best_score<1.50274349:
                break
        # acceptance for current
        accept=False
        if cand_score<cur_score:
            accept=True
        else:
            prob=math.exp(-(cand_score-cur_score)/ (temperature+1e-12))
            if rng.random()<prob:
                accept=True
        if accept:
            cur=cand
            cur_score=cand_score
        temperature*=decay
        # restart occasionally after many steps without improvement
        no_improve_steps+=1
        if no_improve_steps>20000:
            cur=best_seq.copy()
            cur_score=best_score
            temperature=1.0
            no_improve_steps=0
        # occasional random reset
        if rng.random()<0.001:
            cur=best_seq.copy()
            cur_score=best_score
            temperature=1.0
    return [float(x) for x in best_seq]

import time, math, numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_and_conv(arr: np.ndarray, n: int, L: int):
    """
    Return (score, full convolution array of length 2*n-1).
    Score = 2*n*max(conv)/(sum(arr)**2).
    """
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf"), conv
    score = 2.0 * n * max_b / (s * s)
    return score, conv


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _gen_half_circle(n: int) -> np.ndarray:
    """Symmetric half‑circle shape (sqrt(1‑t^2) on [-1,1])."""
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0
    a = np.sqrt(np.maximum(0.0, 1.0 - t * t))
    return a

        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score, cand_conv = _ac1_score_and_conv(cand, n, L)

        # update global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            no_improve = 0
            # early exit if we have already beaten the target
            if best_score < 1.50274349:
                break
        else:
            no_improve += 1

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
            cur_conv = cand_conv

        temperature *= decay

        # occasional restart from the best to avoid drift
        if no_improve >= RESTART_AFTER or rng.random() < 0.001:
            cur = best_seq.copy()
            cur_score, cur_conv = _ac1_score_and_conv(cur, n, L)
            temperature = 1.0
            no_improve = 0

    # final safety: make sure the returned sequence respects bounds
    np.clip(best_seq, 0.0, 1000.0, out=best_seq)
    return [float(x) for x in best_seq]
