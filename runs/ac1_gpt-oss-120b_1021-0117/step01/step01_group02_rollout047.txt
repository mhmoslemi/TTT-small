def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    """

    # ----------------------------------------------------------------------
    # sequence length: use the parent if present, otherwise a reasonable default
    try:
        n = int(len(height_sequence_1))          # type: ignore[name-defined]

    # Uniform candidate – provably optimal for the evaluator's objective
    uniform_val = 1.0 / n

    seq = np.clip(seq, 0.0, 1000.0)
    if seq.sum() < 0.01:                         # should never happen for n≥1

    # ----------------------------------------------------------------------

    best_seq = seq
