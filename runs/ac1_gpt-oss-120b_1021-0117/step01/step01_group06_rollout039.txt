import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    # Sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

def _conv_fft(vec: np.ndarray, L: int) -> np.ndarray:
    # compute full convolution via FFT
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    m = vec.size * 2 - 1
    return conv[:m]

def _score_vec(vec: np.ndarray, L: int) -> float:
    conv = _conv_fft(vec, L)
    n = vec.size
    # sum cannot be zero because we keep it normalised
    return 2.0 * n * conv.max() / (vec.sum() ** 2)

def _gradient_for_shift(vec: np.ndarray, shift: int) -> np.ndarray:
    n = vec.size
    g = np.zeros_like(vec)
    s = shift
    # forward contribution
    if s < n:
        g[: n - s] += vec[s:]
        g[s:] += vec[: n - s]
    # if s >= n, contributions are zero
    return g

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Determine length and starting vector
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback length 256
        n = 256
        base = _sqrt_series(n)
    # Normalise to sum = 1 (simplex)
    base /= base.sum()
    n = base.size

    # Pre‑compute FFT length
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = base.copy()
    best_score = _score_vec(best_vec, L)

    # Main optimisation loop
    while time.monotonic() < deadline:
        # Compute convolution and identify the shift that attains the max
        conv = _conv_fft(best_vec, L)
        max_idx = int(conv.argmax())
        # Gradient step aimed at reducing this max entry
        grad = _gradient_for_shift(best_vec, max_idx)
        # try a few step magnitudes
        improved = False
        if np.linalg.norm(grad) > 0:
            # normalise gradient to have comparable magnitude
            grad_norm = np.linalg.norm(grad)
            for factor in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
                step = factor / grad_norm
                cand = _project_simplex(best_vec - step * grad)
                sc = _score_vec(cand, L)
                if sc < best_score:
                    best_score = sc
                    best_vec = cand
                    improved = True
                    break
        # Random small mass‑transfer if gradient did not improve
        if not improved and rng.random() < 0.3:
            i = rng.integers(0, n)
            if best_vec[i] == 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            frac = rng.random() * 0.15  # up to 15 % of source mass
            delta = best_vec[i] * frac
            cand = best_vec.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            sc = _score_vec(cand, L)
            if sc < best_score:
                best_score = sc
                best_vec = cand
                improved = True

        # Occasionally inject Gaussian noise around the current best
        if rng.random() < 0.01:
            noise = rng.normal(scale=0.001, size=n)
            cand = best_vec + noise
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            sc = _score_vec(cand, L)
            if sc < best_score:
                best_score = sc
                best_vec = cand

        # Early exit if we reach the target bound (tiny safety margin)
        if best_score <= 1.5027:
            break

    # Return the best sequence found as plain Python floats
    return [float(x) for x in best_vec]

def _solve_lp(f, M):
    n = f.size
    g = cp.Variable(n, nonneg=True)
    constraints = []
    # shift 0 to 2n-2
    for k in range(2 * n - 1):
        # sum_i f_i * g_{k-i}
        # Equivalent to cp.sum(cp.multiply(f_slice, g_slice))
        # Compute overlap indices:
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)  # inclusive
        # length = i_end - i_start + 1
        # slice f[i_start:i_end+1] and g[k-i_start :: -1? Let's compute j = k-i
        # j = k - i; iterate i: i from i_start to i_end, j runs from k - i_start down to k - i_end.
        # Use f[i_start:i_end+1] and g[k - i_start : k - i_end - 1 : -1]? but easier: compute sum_i f_i * g_{k-i} = sum_i f_i * g_j where j = k - i
        # Equivalent to sum over i using cp.multiply(f[i_start:i_end+1], g[k - i_start : k - i_end -1 : -1]).
        # However we can simply compute cp.sum(cp.multiply(f[i_start: i_end+1], g[(k - i_end):(k - i_start+1)])) reversed.
        # Since the slice order doesn't matter for dot product.
        # So define g_slice = g[(k-i_end):(k-i_start+1)]
        # ensure slices are contiguous forward order.
        g_slice = g[(k - i_end):(k - i_start + 1)]
        f_slice = f[i_start:i_end+1]
        constraints.append(cp.sum(cp.multiply(f_slice, g_slice)) <= M)
    # objective
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000) # might adjust.
    return g.value

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

def _conv_fft(vec: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    m = vec.size * 2 - 1
    return conv[:m]

def _score_vec(vec: np.ndarray, L: int) -> float:
    conv = _conv_fft(vec, L)
    return 2.0 * vec.size * conv.max() / (vec.sum() ** 2)

def _gradient_for_shift(vec: np.ndarray, shift: int) -> np.ndarray:
    n = vec.size
    g = np.zeros_like(vec)
    s = shift
    if s < n:
        g[: n - s] += vec[s:]
        g[s:] += vec[: n - s]
    return g

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of a where a*a = 1+…+x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _conv_fft(vec: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution via FFT (length 2*len(vec)-1)."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    m = vec.size * 2 - 1
    return conv[:m]


def _score_vec(vec: np.ndarray, L: int) -> float:
    """Exact AC1 bound for a non‑negative sequence."""
    conv = _conv_fft(vec, L)
    return 2.0 * vec.size * conv.max() / (vec.sum() ** 2)


def _gradient_for_shift(vec: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry with respect to vec."""
    n = vec.size
    g = np.zeros_like(vec)
    s = shift
    if s < n:
        g[: n - s] += vec[s:]
        g[s:] += vec[: n - s]
    return g
