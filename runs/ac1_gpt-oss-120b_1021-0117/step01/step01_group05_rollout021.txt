    rng=...
    # Determine n and parent or init.
    # init: binomial or parent whichever better.
    best = ...
    best_score, best_conv = fast_score(best)

    # parameters
    temperature = 1.0
    # scaling schedule
    while time.monotonic() < deadline:
        # reduce temperature linearly
        temp = max(0.01, (deadline - time.monotonic())/budget_s)
        # choose operation
        op = rng.integers(0,5)
        candidate = best.copy()
        if op == 0:

            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.05 * temp))
            candidate[start:start+width] *= factor
        elif op == 1:
            # single perturb
            idx = rng.integers(0,n)
            factor = np.exp(rng.normal(0.0, 0.2 * temp))
            candidate[idx] *= factor
        elif op == 2:
            # mass transfer from max pair
            k_max = np.argmax(best_conv)
            lo = max(0, k_max - (n-1))
            hi = min(k_max, n-1)
            i = rng.integers(lo, hi+1)
            j = k_max - i

                # ensure we don't remove more than available
                removed_i = candidate[i] * alpha
                removed_j = candidate[j] * alpha
                candidate[i] -= removed_i
                candidate[j] -= removed_j
                total = removed_i + removed_j
                # add to low-value index
                low_idx = rng.choice(np.where(candidate < np.percentile(candidate, 30))[0])
                candidate[low_idx] += total
            else:
                # handle diagonal: move fraction to low index
                alpha = rng.uniform(0.05, 0.2) * temp
                removed = candidate[i] * alpha
                candidate[i] -= removed
                low_idx = rng.integers(0,n)
                candidate[low_idx] += removed
        elif op == 3:
            # smooth neighbours (averaging)
            i = rng.integers(0,n-1)
            avg = (candidate[i] + candidate[i+1]) * 0.5
            candidate[i] = avg
            candidate[i+1] = avg

def _conv_score(seq):
    n = seq.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    s = np.sum(seq)
    if s <= 0:
        return np.inf, conv
    max_val = np.max(conv)
    score = 2.0 * n * max_val / (s*s)
    return score, conv

    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        T = T0 * (deadline - now) / full_time
        cand = cur.copy()
        op = rng.integers(0, 5)

        if op == 0:  # block scaling
            w = rng.integers(1, max(2, n // 60))
            st = rng.integers(0, n - w + 1)
            factor = math.exp(rng.normal(0.0, 0.06) * (T + 0.01))
            cand[st:st + w] *= factor
        elif op == 1:  # single index perturb
            i = rng.integers(0, n)
            factor = math.exp(rng.normal(0.0, 0.18) * (T + 0.01))
            cand[i] *= factor
        elif op == 2:  # mass from max‑pair
            k = int(np.argmax(cur_conv))
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = rng.integers(lo, hi + 1)
            j = k - i
            alpha = rng.uniform(0.04, 0.15) * (T + 0.01)
            if i != j:
                di = cand[i] * alpha
                dj = cand[j] * alpha
                cand[i] -= di
                cand[j] -= dj
                tot = di + dj
                tgt = rng.integers(0, n)
                cand[tgt] += tot
            else:
                di = cand[i] * alpha
                cand[i] -= di
                tgt = rng.integers(0, n)
                cand[tgt] += di
        elif op == 3:  # neighbour smooth
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg
        else:  # move from max to min
            hi = np.argmax(cand)
            lo = np.argmin(cand)
            alpha = rng.uniform(0.05, 0.25) * (T + 0.01)
            moved = cand[hi] * alpha
            cand[hi] -= moved
            cand[lo] += moved

        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score, cand_conv = _conv_score(cand)

        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / max(1e-12, T)):
            cur = cand
            cur_score = cand_score
            cur_conv = cand_conv
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

    return [float(x) for x in best]
