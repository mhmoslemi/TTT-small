def project_onto_simplex(v):
    # ensure v >=0 after projection, sum = 1
    # typical algorithm: sort descending, find rho, compute theta
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_step(seq, step):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    # compute gradient
    # vect for i<=k:
    idx1 = np.arange(k+1)
    grad = np.zeros_like(seq)
    grad[idx1] = seq[k - idx1]
    # i>=k:
    idx2 = np.arange(k, n)
    grad[idx2] += seq[idx2 - k]
    # subtract step * grad
    cand = seq - step * grad
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s > 0:
        cand = cand / s
    return cand

def subgrad(seq):

def project_simplex(v):
    # ensure sum 1 and nonnegative
    # standard algorithm
    # from https://gist.github.com/...
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero-indexed
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Exact AC1 upper‑bound (fast, no clipping – seq is already feasible)."""
    n = seq.size
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max() / (seq.sum() ** 2))


def _max_index_and_grad(seq: np.ndarray):
    """
    Return the index k where the convolution attains its maximum,
    together with a sub‑gradient of that maximal entry.
    For a given k the gradient is g_i = a_{k-i}  (if the index is in range).
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())

    grad = np.zeros_like(seq)
    # offset = n-1-k  (relation between i and k-i in the reversed array)
    offset = n - 1 - k
    if offset >= 0:
        # i = 0 .. k   -> k-i = k .. 0  -> take a slice of the reversed array
        grad[: k + 1] = seq[offset : offset + k + 1]
    else:
        start = -offset                # first i where k-i becomes non‑negative
        length = n + offset            # how many entries are valid
        grad[start:] = seq[: length]
    return k, grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Classic O(n log n) algorithm – works for any non‑negative vector.
    """
    if v.ndim != 1:
        raise ValueError
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            cand = _project_simplex(cand)
            cand_score = _score(cand)

        # --------------------------------------------------------------
        # d)  Accept if it improves the current point (no uphill moves)
        # --------------------------------------------------------------
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score

        # --------------------------------------------------------------
        # e)  Track the overall best solution seen so far
        # --------------------------------------------------------------
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            # early exit when the target is reached
            if best_score <= 1.50274349:
                break

    # ------------------------------------------------------------------
    # 3. Return the best candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
