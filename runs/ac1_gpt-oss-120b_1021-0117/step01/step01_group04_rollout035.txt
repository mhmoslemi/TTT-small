import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """
    Compute the AC1 upper bound: 2*n*max(conv(arr,arr))/(sum(arr)**2)
    using FFT for speed.
    """
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    # keep only the linear part (length 2*n-1)
    conv = conv[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step function (list of floats) that
    attempts to minimise the AC1 bound.
    """
    rng = np.random.default_rng(seed)
    # leave a safety margin for clean exit
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- Determine length --------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        # fallback length if parent not provided
        n = 256
        parent = None

    # FFT size for convolution (next power of two)
    L = _next_pow2(2 * n - 1)

    # ---- Initial candidates ------------------------------------------------
    # 1. Uniform sequence (theoretical optimum for the evaluated bound)
    uniform = np.full(n, 1.0, dtype=np.float64)
    best = uniform.copy()
    best_score = _ac1_score(best, n, L)

    # 2. If a parent construction is given, keep it if better
    if parent is not None:
        parent_score = _ac1_score(parent, n, L)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # ---- Simple stochastic refinement --------------------------------------
    cur = best.copy()
    cur_score = best_score
    temperature = 0.8
    decay = 0.9995

    # we limit the number of distinct move types to keep overhead low
    while time.monotonic() < deadline:
        # pick a move: scale a random block or transfer mass between two points
        if rng.random() < 0.6:          # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.08))
            cand = cur.copy()
            cand[start:start + width] *= factor
        else:                           # mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * 0.3            # up to 30 % of source moved
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # enforce evaluator’s hard bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _ac1_score(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # early stop if we already beat a reasonable target
            if best_score < 1.5:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # occasional restart from the best solution
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 0.8

    return [float(v) for v in best]
