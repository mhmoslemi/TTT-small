import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‐style sequence w_i = C(2i,i)/4**i for i=0..n-1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _normalize(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and renormalise to total mass 1."""
    v = np.clip(v, 0.0, 1000.0)
    s = v.sum()
    if s <= 0.0:
        return None
    return v / s


def _gradient_move(a: np.ndarray, step_frac: float) -> np.ndarray:
    """A cheap gradient‑inspired descent, preserving the simplex."""
    cum = np.cumsum(a)                 # forward cumulative sum
    grad = 2.0 * cum[::-1]             # reversed tail‑sum (sub‑gradient proxy)
    max_g = grad.max()
    if max_g <= 0.0:
        return a
    step = step_frac * (a.max() / max_g)
    cand = a - step * grad
    np.clip(cand, 0.0, None, out=cand)
    s = cand.sum()
    if s <= 0.0:
        return a
    cand /= s
    return cand


def _random_pairwise(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of mass from one coordinate to another."""
    n = a.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    if a[i] <= 0.0:
        return a
    eps = rng.random() * a[i] * 0.5   # up to half of the donor
    cand = a.copy()
    cand[i] -= eps
    cand[j] += eps
    s = cand.sum()
    if s <= 0.0:
        return a
    cand /= s
    return cand


def _jitter_move(a: np.ndarray, rng: np.random.Generator, sigma: float = 0.018) -> np.ndarray:
    """Multiplicative log‑normal perturbation, followed by re‑normalisation."""
    jitter = np.exp(rng.normal(0.0, sigma, size=a.shape))
    cand = a * jitter
    s = cand.sum()
    if s <= 0.0:
        return a
    cand /= s
    return cand


def _targeted_move(a: np.ndarray, rng: np.random.Generator, eps_frac: float = 0.07) -> np.ndarray:
    """
    Reduce the current maximal convolution value.
    Find the shift k where a*a is maximal,
    locate the pair (i,j) with i+j=k that contributes the most,
    and shift a small amount from the smaller entry to the larger one.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(conv.argmax())

    # overlapping index interval for this k
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    prod = a[i] * a[j]

    idx = int(prod.argmax())
    i_max = int(i[idx])
    j_max = int(j[idx])

    # if the pair collapses to a single index (i==j) pick a different destination
    if i_max == j_max:
        src = i_max
        # choose a destination different from src with the smallest value
        mask = np.arange(n) != src
        if not mask.any():
            return a
        dst = int(np.argmin(np.where(mask, a, np.inf)))
    else:
        # move mass from the smaller of the two to the larger
        if a[i_max] < a[j_max]:
            src, dst = i_max, j_max
        else:
            src, dst = j_max, i_max

    if a[src] <= 0.0:
        return a
    eps = eps_frac * a[src]
    cand = a.copy()
    cand[src] -= eps
    cand[dst] += eps
    cand = np.clip(cand, 0.0, None)
    s = cand.sum()
    if s <= 0.0:
        return a
    cand /= s
    return cand

        # early exit if time is about to run out
        if time.monotonic() > deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # 5.  Return the best candidate found
    # ------------------------------------------------------------------
    return [float(v) for v in best]
