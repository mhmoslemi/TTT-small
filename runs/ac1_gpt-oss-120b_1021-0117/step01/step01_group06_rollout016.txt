# ------------------------------------------------------------
# simple projection onto the probability simplex (non‑negative + sum=1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fall back to uniform if everything vanished
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ------------------------------------------------------------

# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------

    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

    return float(conv[idx]), idx
