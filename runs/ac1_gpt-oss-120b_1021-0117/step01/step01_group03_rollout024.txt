def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound.

    The routine performs a fast simulated‑annealing / targeted‑mass‑transfer
    search using FFT‑based convolution for the objective.

    # ------------------------------------------------------------------

        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = int(parent.size)
    else:
        n = 5582
        parent = None

    # classic Catalan‑type sqrt series
    sqrt_seq = np.empty(n, dtype=np.float64)
    sqrt_seq[0] = 1.0
    for i in range(1, n):
        sqrt_seq[i] = sqrt_seq[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)

        fa = np.fft.rfft(a, n=m_conv)
