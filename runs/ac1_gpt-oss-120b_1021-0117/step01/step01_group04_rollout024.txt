def _ac1_score_and_conv(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 1e-12: return np.inf, conv
    return 2.0 * n * max_b / (sum_a**2), conv

import numpy as np, time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1 (recurrence)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score_and_conv(arr: np.ndarray, n: int, L: int):
    """Return bound and full linear convolution (length 2*n-1)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf'), conv
    return 2.0 * n * max_b / (s * s), conv

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----- initialise -------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256
        parent = None

    L = _next_pow2(2 * n - 1)

    base = _gen_binomial(n)
    cur = base.copy()
    if parent is not None:
        # keep the better of parent / base as start
        pscore, _ = _ac1_score_and_conv(parent, n, L)
        bscore, _ = _ac1_score_and_conv(base, n, L)
        if pscore < bscore:
            cur = parent.copy()
    cur_score, _ = _ac1_score_and_conv(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # fixed reference sum (used for occasional renormalisation)
    ref_sum = float(cur.sum()) if cur.sum() > 0 else 1.0

    temperature = 1.0
    decay = 0.9997
    target = 1.50274349

    # ----- main search ------------------------------------------------------
    while time.monotonic() < deadline:
        move_type = rng.random()
        cand = cur.copy()

        # 1) block scaling ---------------------------------------------------
        if move_type < 0.5:
            width = int(rng.integers(max(1, n // 250), max(2, n // 15)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.13))
            cand[start:start + width] *= factor

        # 2) transfer mass ---------------------------------------------------
        elif move_type < 0.8:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.45
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta

        # 3) active‑shift reduction -----------------------------------------
        else:
            # compute current convolution to locate the peak
            _, conv = _ac1_score_and_conv(cur, n, L)
            s_max = int(conv.argmax())

            # overlapping index interval for this shift
            if s_max < n:
                i_start, i_end = 0, s_max
            else:
                i_start = s_max - (n - 1)
                i_end = n - 1

            i_arr = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_arr = s_max - i_arr
            contrib = cur[i_arr] * cur[j_arr]

            # source: index with largest product for the offending shift
            src_local = int(contrib.argmax())
            src = int(i_arr[src_local])

            # choose a destination outside the overlap if possible
            if i_start > 0 or i_end < n - 1:
                non = np.concatenate(
                    (np.arange(0, i_start, dtype=np.int64),
                     np.arange(i_end + 1, n, dtype=np.int64))
                )
                dst = int(rng.choice(non))
            else:
                # fallback: any index with small contribution
                low_candidates = i_arr[contrib < np.percentile(contrib, 20)]
                if low_candidates.size:
                    dst = int(rng.choice(low_candidates))
                else:
                    dst = int(rng.integers(0, n))
                    while dst == src:
                        dst = int(rng.integers(0, n))

            frac = rng.random() * 0.35
            delta = cand[src] * frac
            cand[src] -= delta
            cand[dst] += delta

        # enforce evaluator bounds ------------------------------------------------
        np.clip(cand, 0.0, 1000.0, out=cand)

        # keep the total mass reasonable (avoid sum < 0.01)
        cur_sum = float(cand.sum())
        if cur_sum < 0.1:

            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # return as plain Python list
    return [float(x) for x in best]
