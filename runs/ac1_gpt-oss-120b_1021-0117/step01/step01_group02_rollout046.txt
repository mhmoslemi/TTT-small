    # determine length n, initialize a

        a0 = np.array(height_sequence_1, dtype=np.float64)
        n = a0.size
        a = np.clip(a0, 0.0, 1000.0)
        if a.sum() < 1e-12:
            a = np.full(n, 1.0/n)
        else:
            a = a / a.sum()
    except Exception:
        n = 5000
        a = np.full(n, 1.0/n)

    # ensure sum > 0.
    # compute nfft
    nfft = 1 << ((2*n - 1).bit_length())

    # precompute uniform vector
    uniform = np.full(n, 1.0/n)

    # define max_conv
    def max_conv(arr):
        A = np.fft.rfft(arr, n=nfft)
        conv = np.fft.irfft(A * A, n=nfft)
        # conv length nfft; return max of first 2*n-1 elements
        return conv[:2*n-1].max()

    # score function
    def compute_score(arr):
        s = arr.sum()
        if s <= 0:
            return np.inf
        mc = max_conv(arr)
        return 2.0 * n * mc / (s*s)

    # initial best
    best = a.copy()
    best_score = compute_score(best)

    # also evaluate uniform as fallback
    uniform_score = compute_score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # parameters
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)
    step_factor = 0.4
    min_step = 1e-8
    iteration = 0
    no_improve_counter = 0
    # thresholds
    max_no_improve_before_decay = 200  # adjust

    delta_fractions = [1.0, 0.5, 0.2]

    # main loop
    while time.monotonic() < deadline:
        iteration += 1

        A = np.fft.rfft(best, n=nfft)
        conv_full = np.fft.irfft(A * A, n=nfft)[:2*n-1]

    n_fft = 1 << ((2 * n - 1).bit_length())   # power of two ≥ 2n‑1
    uniform = np.full(n, 1.0 / n)

    def _max_conv(x):
        """Maximum entry of the linear convolution x*x (full length)."""
        X = np.fft.rfft(x, n=n_fft)
        conv = np.fft.irfft(X * X, n=n_fft)
        return conv[: 2 * n - 1].max()

    def _score(x):
        """Exact score used by the evaluator (sum may drift a little)."""
        s = x.sum()
        if s <= 0.0:
            return np.inf
        return 2.0 * n * _max_conv(x) / (s * s)

    # ----- keep the best seen --------------------------------------------------
    best = a.copy()
    best_score = _score(best)

    # compare with uniform – it is a cheap guaranteed fallback (score = 2)
    uni_score = _score(uniform)
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score

    # ----- optimisation parameters ---------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    step_factor = 0.4           # fraction of source mass moved each try
    min_step = 1e-9
    no_improve = 0
    max_no_improve = 200
    delta_fractions = (1.0, 0.5, 0.2)   # try full, half and small steps
    target_score = 1.50274349

    # ----- main search loop ----------------------------------------------------
    while time.monotonic() < deadline:
        # compute full convolution and its maximum location
        X = np.fft.rfft(best, n=n_fft)
        conv_full = np.fft.irfft(X * X, n=n_fft)[: 2 * n - 1]
        cur_max = conv_full.max()
        k = int(conv_full.argmax())

        # partner[i] = best[k-i] when defined, else 0
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        partner = np.zeros_like(best)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner[idx] = best[k - idx]

        # source = index that contributes most to the current peak
        contributions = best * partner
        src = int(contributions.argmax())
        src_val = best[src]

        # possible destinations: indices where partner == 0 (they do not affect the current peak)
        zero_idxs = np.where(partner == 0)[0]
        if zero_idxs.size:
            dest_candidates = rng.choice(zero_idxs,
                                         size=min(5, zero_idxs.size),
                                         replace=False)
        else:
            # fallback: smallest partner values
            min_part = partner.min()
            cand = np.where(partner == min_part)[0]
            dest_candidates = rng.choice(cand,
                                         size=min(5, cand.size),
                                         replace=False)

        improved = False

        # try moving mass from src to each dest using a few step sizes
        for dest in dest_candidates:
            if dest == src:
                continue
            for frac in delta_fractions:
                delta = src_val * step_factor * frac
                if delta < min_step:
                    continue
                if delta > src_val:
                    delta = src_val
                cand = best.copy()
                cand[src] -= delta
                cand[dest] += delta
                # numeric safety
                if cand[src] < 0.0:
                    continue
                cand_score = _score(cand)
                if cand_score < best_score:
                    best = cand
                    best_score = cand_score
                    improved = True
                    # shrink step size a little after a successful move
                    step_factor = max(0.02, step_factor * 0.998)
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        # ----- occasional random perturbation (shake) ---------------------------
        if rng.random() < 0.012:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            shake = best.copy()
            factor = np.exp(rng.normal(0.0, 0.22, size=width))
            shake[start:stop] *= factor
            total = shake.sum()
            if total > 0.0:
                shake = shake / total * best.sum()
                shake_score = _score(shake)
                if shake_score < best_score:
                    best = shake
                    best_score = shake_score
                    step_factor = max(0.02, step_factor * 0.99)
                    no_improve = 0

        # ----- occasional mixing with the uniform sequence ----------------------
        if rng.random() < 0.018:
            alpha = rng.uniform(0.008, 0.07)
            mixed = (1.0 - alpha) * best + alpha * uniform
            mixed_score = _score(mixed)
            if mixed_score < best_score:
                best = mixed
                best_score = mixed_score
                step_factor = max(0.02, step_factor * 0.99)
                no_improve = 0

        # ----- decay step size if we have been stuck for a while ---------------
        if no_improve >= max_no_improve:
            step_factor = max(0.02, step_factor * 0.9)
            no_improve = 0

        # early exit if target achieved
        if best_score <= target_score:
            break

    # ----- final sanity check ---------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
