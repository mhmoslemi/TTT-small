import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients a with a*a = 1 + … + x^{2n‑2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _max_conv(a: np.ndarray):
    """Return (max value, index) of a*a (linear convolution)."""
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _score(a: np.ndarray) -> float:
    """Exact AC‑1 bound for a (same as evaluate_sequence)."""
    n = a.size
    max_val, _ = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (s * s))

    sqrt_vec /= sqrt_vec.sum()

        g[idx] = 2.0 * vec[kmax - idx]

        # direction opposite to the gradient, projected onto the zero‑sum hyperplane
        d = -g + g.mean()

        # try a step in that direction
        cand = _project_simplex(vec + step * d)
        cand_sc = _score(cand)
        if cand_sc < best_sc - 1e-12:
            best_sc, best_vec, vec = cand_sc, cand, cand
            step = min(step * 1.15, 0.5)  # accelerate when successful
            continue

        # if not better, try a smaller step
        cand2 = _project_simplex(vec + (step * 0.5) * d)
        cand2_sc = _score(cand2)
        if cand2_sc < best_sc:
            best_sc, best_vec, vec = cand2_sc, cand2, cand2
            step *= 0.5
            continue

        # occasional random mass transfer (exploration)
        i = rng.integers(0, n)
        if vec[i] > 0.0:
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = vec[i] * rng.random() * 0.10
            cand3 = vec.copy()
            cand3[i] -= delta
            cand3[j] += delta
            cand3 = _project_simplex(cand3)
            cand3_sc = _score(cand3)
            if cand3_sc < best_sc:
                best_sc, best_vec, vec = cand3_sc, cand3, cand3
                continue

        # adapt step size when nothing improves
        step = max(step * 0.7, min_step)

        # very rare reset to the best seen vector (helps when stuck)
        if rng.random() < 0.001:
            vec = best_vec.copy()

    # ------------------------------------------------------------
    # return the best candidate found
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
