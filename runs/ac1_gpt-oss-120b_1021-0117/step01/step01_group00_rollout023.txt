import time
import math
import numpy as np

def _sidon_like_sequence(n):
    """C(2i,i)/4**i for i=0..n-1, classic Sidon‑like decreasing sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _score(seq):
    """Exact objective used by evaluate_sequence (numpy‑based, fast)."""
    # seq is a 1‑D np.ndarray of non‑negative floats already clipped to [0,1000]
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑empty list of non‑negative step‑function heights that
    (approximately) minimizes the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)  # 1 s head‑room

    # ------------------------------------------------------------------
    # 1.  Base sequence (parent or Sidon‑like)
    # ------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = int(parent.shape[0])
        if parent.ndim != 1 or n == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        n = 256                     # safe fallback length
        parent = np.full(n, 1.0, dtype=np.float64)

    # Sidon‑like sequence of the same length
    sidon = _sidon_like_sequence(n)
    sidon = np.clip(sidon, 0.0, 1000.0)

    # evaluate both candidates quickly
    best_seq = parent.copy()
    best_score = _score(best_seq)
    sidon_score = _score(sidon)
    if sidon_score < best_score:
        best_seq = sidon.copy()
        best_score = sidon_score

    # ------------------------------------------------------------------
    # 2.  Simulated annealing on values (transfer) and order (swap)
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    # temperature schedule
    temperature = max(0.1, cur_score * 0.5)          # start relatively warm
    cooling = 0.9995                                 # exponential decay
    min_temp = 1e-6

    # helper to evaluate and optionally accept a candidate
    def try_candidate(cand, cand_score):
        nonlocal cur_seq, cur_score, best_seq, best_score, temperature
        delta = cand_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / temperature):
            cur_seq = cand
            cur_score = cand_score
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score

    # main SA loop
    while time.monotonic() < deadline:
        # 70 % transfer moves, 30 % pure swaps
        if rng.random() < 0.7:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # max amount we can move from i to j without breaking bounds
            max_move = min(cur_seq[i], 1000.0 - cur_seq[j])
            if max_move <= 0.0:
                continue
            # move a random fraction (up to ~10 % of the allowed amount)
            amount = max_move * rng.random() * 0.1
            cand = cur_seq.copy()
            cand[i] -= amount
            cand[j] += amount
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

            cand[i], cand[j] = cand[j], cand[i]

        cand_score = _score(cand)
        if not np.isfinite(cand_score):

        try_candidate(cand, cand_score)

        # cool down
        temperature = max(min_temp, temperature * cooling)

        # occasional random reshuffle to escape deep local minima

            perm = rng.permutation(n)
            cur_seq = cur_seq[perm]
            cur_score = _score(cur_seq)

    # ------------------------------------------------------------------

    if time.monotonic() < deadline:

        # a) interleave large and small values (ends‑in‑the‑middle pattern)
        inter = np.empty_like(vals)
        left, right = 0, n - 1
        toggle = True
        for v in vals:
            if toggle:
                inter[left] = v
                left += 1
            else:
                inter[right] = v
                right -= 1
            toggle = not toggle
        inter_score = _score(inter)
        if inter_score < best_score:
            best_seq = inter.copy()
            best_score = inter_score

    if time.monotonic() < deadline:
        # b) place values on a stride (circular spread)
        stride = max(1, n // max(2, int(np.sqrt(n))))
        spread = np.empty_like(vals)
        idx = 0

            spread[idx] = v
            idx = (idx + stride) % n

        if spread_score < best_score:

            best_score = spread_score

    # ------------------------------------------------------------------
    # final sanity check – ensure the evaluator will accept it
    final = np.clip(best_seq, 0.0, 1000.0)
    if np.sum(final) < 0.01:
        # fallback: uniform tiny values to guarantee validity
        final = np.full(n, 0.01 / n, dtype=np.float64)

    return [float(x) for x in final.tolist()]
