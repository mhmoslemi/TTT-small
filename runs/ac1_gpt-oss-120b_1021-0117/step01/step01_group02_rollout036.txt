import time
import numpy as np

# ---------- helpers ---------------------------------------------------------

def _conv_max(v: np.ndarray):
    """Return (max(convolution), argmax) of v with itself, using FFT."""
    n = v.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    # numerical noise may produce tiny negative values – ignore
    max_val = float(conv.max())
    arg = int(conv.argmax())
    return max_val, arg


def _score(v: np.ndarray):
    """Exact score used by the evaluator (sum is assumed >0)."""
    n = v.shape[0]
    max_val, _ = _conv_max(v)
    s = float(v.sum())
    if s <= 0.0:
        return np.inf
    return 2.0 * n * max_val / (s * s)


# ---------- main proposal ---------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- initialise --------------------------------------------------------
    try:
        cur = np.array(height_sequence_1, dtype=np.float64)
        n = cur.shape[0]
        cur = np.clip(cur, 0.0, 1000.0)
    except Exception:
        n = 5000
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    if cur.sum() < 0.01:
        cur = np.full_like(cur, 1.0 / cur.size)

    # normalise to unit sum – scale does not affect the bound
    cur = cur / cur.sum()

    # keep a uniform fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best = cur.copy()
    best_score = _score(best)
    u_score = _score(uniform)
    if u_score < best_score:
        best = uniform.copy()
        best_score = u_score

    # ---- parameters --------------------------------------------------------
    step_factor = 0.3          # fraction of source mass moved each step
    min_step = 1e-9
    temp = 0.05                # temperature for occasional uphill moves
    temp_decay = 0.9995
    idx = np.arange(n, dtype=np.int64)

    # ---- optimisation loop -------------------------------------------------
    while time.monotonic() < deadline:
        max_val, k = _conv_max(best)          # active lag
        # full gradient of the active convolution entry
        diff = np.abs(k - idx)
        mask = diff < n

        if src_cands.size == 0:

        src = src_cands[np.argmax(best[src_cands] * grad[src_cands])]

        # destination – smallest gradient (edges of the distribution)
        dest = int(np.argmin(grad))

        if src == dest:
            # tiny random perturbation to break symmetry
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = best.copy()
            cand[start:stop] *= np.exp(rng.normal(0.0, 0.12))
            cand = np.clip(cand, 0.0, 1000.0)
            cand_sum = cand.sum()
            if cand_sum > 0.0:
                cand = cand / cand_sum
                cand_score = _score(cand)
                if cand_score < best_score or rng.random() < np.exp(
                    -(cand_score - best_score) / max(temp, 1e-12)
                ):
                    best, best_score = cand, cand_score
            temp = max(1e-6, temp * temp_decay)
            continue

        # move a fraction of the source to the destination
        delta = best[src] * step_factor
        if delta < min_step:
            delta = min_step
        delta = min(delta, best[src])
        cand = best.copy()
        cand[src] -= delta
        cand[dest] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to keep the sum exactly 1
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s
        cand_score = _score(cand)

        # accept if better, otherwise maybe accept uphill (simulated annealing)
        if cand_score < best_score or rng.random() < np.exp(
            -(cand_score - best_score) / max(temp, 1e-12)
        ):
            best, best_score = cand, cand_score
            step_factor = max(0.01, step_factor * 0.995)
        else:
            # rare chance to increase the step size and escape a plateau
            if rng.random() < 0.001:
                step_factor = min(0.5, step_factor * 1.1)

        temp = max(1e-6, temp * temp_decay)

        # occasional global shake
        if rng.random() < 0.0005:
            shake = best.copy()
            shake *= np.exp(rng.normal(0.0, 0.2, size=n))
            shake = np.clip(shake, 0.0, 1000.0)
            shake = shake / shake.sum()
            shake_score = _score(shake)
            if shake_score < best_score:
                best, best_score = shake, shake_score

    # ---- final sanity check -------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
