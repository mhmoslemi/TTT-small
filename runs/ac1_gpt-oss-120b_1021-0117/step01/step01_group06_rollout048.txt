import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# generate the sqrt‑series coefficients (a * a = 1+…+x^{2n‑2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# full convolution via FFT (linear convolution, real output)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# score = 2 * n * max(conv) / (sum(a)**2)
# ------------------------------------------------------------
def _score_vec(a: np.ndarray) -> float:
    n = a.size
    conv = _conv_full(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise: use the supplied parent if available, otherwise sqrt‑series
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        if a.size == 0:
            n = 256
            a = _sqrt_series(n)
        else:
            n = a.size
    else:
        n = 256
        a = _sqrt_series(n)

    # normalise to the simplex (sum = 1)
    a = a / a.sum()
    a = _project_simplex(a)

    best_vec = a.copy()
    best_sc = _score_vec(best_vec)

    # ------------------------------------------------------------------
    # parameters
    # ------------------------------------------------------------------
    prob_grad = 0.25          # probability to take a gradient step
    max_frac = 0.30           # max fraction transferred in random move
    base_step = 0.02          # initial gradient step size
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # choose the type of move
        # --------------------------------------------------------------
        if rng.random() < prob_grad:
            # ----- gradient step towards reducing the current maximum -----
            conv = _conv_full(a)
            k = int(conv.argmax())                     # index achieving max
            # gradient of the selected quadratic term (2 * reversed a)
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            j = np.arange(low, high + 1, dtype=np.int64)
            g = np.zeros_like(a)
            g[j] = 2.0 * a[k - j]

            # schedule a decreasing step size
            step = base_step / (1.0 + 0.001 * it)
            cand = _project_simplex(a - step * g)
            sc = _score_vec(cand)

            if sc < best_sc:
                best_sc = sc
                best_vec = cand.copy()
                a = cand
        else:
            # ------------------------ random mass transfer -----------------
            i = rng.integers(0, n)
            if a[i] == 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            frac = rng.random() * max_frac
            delta = a[i] * frac
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            sc = _score_vec(cand)

            if sc < best_sc:
                best_sc = sc
                best_vec = cand.copy()
                a = cand
            else:
                # occasional non‑improving jump to escape plateaus
                if rng.random() < 0.001:
                    a = cand

    # return the best vector found
    return [float(x) for x in best_vec]
