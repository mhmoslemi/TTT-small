def swap_mass(seq, rng):

   i = rng.integers(0, n)

   j = rng.integers(0, n)

   if i == j: return

   # amount to move from larger to smaller

   if seq[i] > seq[j]:

        delta = (seq[i] - seq[j]) * rng.uniform(0.02, 0.1)  # fraction of difference

        seq[i] -= delta

        seq[j] += delta

   else:

        delta = (seq[j] - seq[i]) * rng.uniform(0.02, 0.1)

        seq[j] -= delta

        seq[i] += delta

Return.

def _fast_score(seq):
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float('inf'), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val

def op_swap(seq, rng):
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return
    if seq[i] > seq[j]:
        delta = (seq[i] - seq[j]) * rng.uniform(0.02, 0.1)
        seq[i] -= delta
        seq[j] += delta

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, max value) using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val
