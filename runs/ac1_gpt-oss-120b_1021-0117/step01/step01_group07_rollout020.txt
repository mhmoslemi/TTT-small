import time

# --------------------------------------------------------------
# Helper functions (top‑level, no closures)
# --------------------------------------------------------------
def _max_convolution(a: np.ndarray, m: int) -> float:
    """
    Return max of the linear convolution a*a using an FFT of size m.
    a is assumed to sum to 1 (normalisation handled by caller).
    """
    # FFT based auto‑convolution (real signals)
    fft_a = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[: 2 * a.size - 1]
    # tiny negative artefacts are set to zero
    conv = np.maximum(conv, 0.0)
    return float(conv.max())


def _gradient_at_max(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution functional.
    Returns a vector g such that a ← a*exp(-η g) moves a away from the
    active indices where the convolution attains its maximum.
    """
    n = a.size
    # indices where convolution is (almost) maximal
    active = np.where(np.isclose(conv, max_val, rtol=1e-12, atol=0.0))[0]
    g = np.zeros_like(a)
    for k in active:
        # i ranges over all positions where a_i participates in conv_k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - idx                         # the matching partner index
        # derivative contribution: 2*a_j
        np.add.at(g, idx, 2.0 * a[j])
    return g

    Compute the exact AC1 upper bound for a (assumed non‑negative,
    sum(a) == 1) using the fast convolution routine.
    """
    max_conv = _max_convolution(a, m_fft)
    return 2.0 * a.size * max_conv
