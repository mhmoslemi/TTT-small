import time
import numpy as np

# ----- helper utilities -------------------------------------------------
def _fft_len_for(n: int) -> int:
    """Smallest power of two >= 2*n-1 (length for linear convolution)."""
    conv_len = 2 * n - 1
    return 1 << ((conv_len - 1).bit_length())


def _fft_score_and_conv(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """Return AC1 upper bound (with sum(a)=1) and the full linear convolution."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    max_b = conv.max()
    # sum(a) is kept =1, so the bound is 2*n*max_b
    return float(2.0 * a.size * max_b), conv


def _grad_from_max(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Gradient of the peak convolution term w.r.t. a (factor 2 included)."""
    n = a.size
    idx = np.arange(n)
    j = peak_idx - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    return g


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i sequence (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


# ----- main entry point -------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the target bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # ---- initialise ----------------------------------------------------
    try:
        n = int(len(height_sequence_1))          # supplied parent
        a = np.array(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        n = 256
        a = _gen_binomial(n)

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-12:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / s                                 # normalise sum to 1

    fft_len = _fft_len_for(n)

    # ---- evaluate initial candidate ------------------------------------
    best = a.copy()
    best_score, cur_conv = _fft_score_and_conv(best, fft_len)
    cur_score = best_score
    cur = a.copy()
    cur_peak = int(cur_conv.argmax())

    # ---- cheap parametric sweep (power / exp families) -----------------
    def _maybe_update(seq):
        nonlocal best, best_score
        sc, _ = _fft_score_and_conv(seq, fft_len)
        if sc < best_score:
            best = seq.copy()
            best_score = sc

    # power‑law a_i ∝ (i+1)^{-α}
    for alpha in np.linspace(0.1, 2.0, 22):
        if time.monotonic() >= deadline:
            break
        seq = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), alpha)
        seq /= seq.sum()
        _maybe_update(seq)

    # exponential decay a_i ∝ r^i
    for r in np.linspace(0.80, 0.999, 30):
        if time.monotonic() >= deadline:
            break
        seq = np.power(r, np.arange(n, dtype=np.float64))
        seq /= seq.sum()
        _maybe_update(seq)

    # mixed power‑exp a_i ∝ (i+1)^{-α}·r^i
    for α in np.linspace(0.2, 1.5, 12):
        for r in np.linspace(0.85, 0.99, 7):
            if time.monotonic() >= deadline:
                break
            seq = (np.arange(1, n + 1, dtype=np.float64) ** (-α)) * np.power(r, np.arange(n))
            seq /= seq.sum()
            _maybe_update(seq)

    # ---- guided hill‑climbing -----------------------------------------
    # current state already stored in cur / cur_score / cur_conv / cur_peak
    while time.monotonic() < deadline:
        # gradient from the current peak convolution term
        grad = _grad_from_max(cur, cur_peak)

        # source: a positive entry with positive gradient
        src_mask = (cur > 1e-12) & (grad > 0)
        src_candidates = np.where(src_mask)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)

        # destination: smallest gradient (might be zero)
        dst = int(np.argmin(grad))
        if dst == src:

            dst = rng.integers(0, n)
            while dst == src:

        delta = cur[src] * frac
        if delta <= 0.0:

        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta

        cand_score, cand_conv = _fft_score_and_conv(cand, fft_len)

        # greedy acceptance
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            cur_conv = cand_conv
            cur_peak = int(cur_conv.argmax())
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:

            if rng.random() < 0.003:

                if i != j:

                    cur[i] -= d
                    cur[j] += d
                    cur_score, cur_conv = _fft_score_and_conv(cur, fft_len)
                    cur_peak = int(cur_conv.argmax())
                    if cur_score < best_score:
                        best = cur.copy()
                        best_score = cur_score

    # ---- finalise -------------------------------------------------------
    final = [float(x) for x in best]
    # sanity check – ensure the evaluator will accept it
    try:
        _ = evaluate_sequence(final)
    except Exception:
        # fallback to the initial normalised sequence if something went wrong
        final = [float(x) for x in a]

    return final
