    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_and_conv(seq: np.ndarray):
    """
    Compute AC1 upper bound, the full self‑convolution and the index
    of the maximal entry using an FFT of length power‑of‑two.
    Returns (score, conv_array, max_index).
    """
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    max_val = conv[idx]
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, idx
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, idx


def _gradient_for_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at the current maximal shift."""
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    j = k_max - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Determine problem size and optional parent construction
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent_arr.ndim == 1
            and parent_arr.size > 0
            and np.isfinite(parent_arr).all()
        ):
            n = parent_arr.size
            parent = np.clip(parent_arr, 0.0, 1000.0)
        else:
            parent = None
            n = 3809
    except Exception:
        parent = None
        n = 3809

    # ------------------------------------------------------------------
    # Build a few deterministic starters (binomial, exponentials, etc.)
    # ------------------------------------------------------------------
    starters = []

    # binomial (the classical near‑optimal shape)
    binom = _initial_binomial(n)
    binom = binom / binom.sum()
    starters.append(binom)

    # parent if supplied
    if parent is not None:
        parent_norm = parent / parent.sum()
        starters.append(parent_norm)

    # exponential families
    for factor in (1.0, 2.0, 3.0):
        a = np.exp(-factor * np.arange(n) / n)
        a /= a.sum()
        starters.append(a)

    # simple Gaussian‑like bumps
    mu = (n - 1) / 2.0
    for mul in (0.2, 0.3, 0.4):
        sigma = mul * n
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - mu) ** 2) / (2.0 * sigma * sigma))
        a /= a.sum()
        starters.append(a)

    # power‑law decays
    for p in (0.1, 0.3, 0.5, 0.8):
        a = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        a /= a.sum()
        starters.append(a)

    # evaluate all starters, keep the best
    best = starters[0]
    best_score, best_conv, best_kmax = _score_and_conv(best)
    for seq in starters[1:]:
        sc, cv, km = _score_and_conv(seq)
        if sc < best_score:
            best = seq
            best_score, best_conv, best_kmax = sc, cv, km

    # current solution (the one we will perturb)
    cur = best.copy()
    cur_score = best_score
    cur_conv = best_conv
    cur_kmax = best_kmax

    # ------------------------------------------------------------------
    # Parameters for stochastic moves
    # ------------------------------------------------------------------
    smooth_kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)  # sums to 1
    target_score = 1.50274349  # stop early if we reach this

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        candidate = cur.copy()
        op = rng.integers(0, 6)   # six different mutation types

        # --------------------------------------------------------------
        # 0 – block scaling (multiplicative)
        # --------------------------------------------------------------
        if op == 0:
            width = rng.integers(1, max(2, n // 30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.06))
            candidate[start : start + width] *= factor

        # --------------------------------------------------------------
        # 1 – single entry perturbation
        # --------------------------------------------------------------
        elif op == 1:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.20))
            candidate[idx] *= factor

        # --------------------------------------------------------------
        # 2 – directed mass shift away from the current maximal pair
        # --------------------------------------------------------------
        elif op == 2:
            k_max = int(np.argmax(cur_conv))
            grad = _gradient_for_max(candidate, k_max)

            # source: index with largest gradient (most responsible for max)
            src_candidates = np.where(grad > 0)[0]
            if src_candidates.size == 0:
                # fallback: simple point perturb
                idx = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, 0.15))
                candidate[idx] *= factor
            else:
                src = src_candidates[np.argmax(grad[src_candidates])]

                # destination: among zero‑gradient indices with smallest value
                zero_mask = grad == 0
                if zero_mask.any():
                    zeros = np.where(zero_mask)[0]
                    # pick a few random zeros and choose the smallest entry
                    if zeros.size > 10:
                        sample = rng.choice(zeros, size=10, replace=False)
                    else:
                        sample = zeros
                    dst = sample[np.argmin(candidate[sample])]
                else:
                    dst = rng.integers(0, n)

                # transfer a modest fraction of mass
                frac = rng.uniform(0.02, 0.12)
                amount = frac * candidate[src]
                candidate[src] -= amount
                candidate[dst] += amount

        # --------------------------------------------------------------
        # 3 – local averaging of two neighbours (smooth)
        # --------------------------------------------------------------
        elif op == 3:
            i = rng.integers(0, n - 1)
            avg = (candidate[i] + candidate[i + 1]) * 0.5
            candidate[i] = avg
            candidate[i + 1] = avg

        # --------------------------------------------------------------
        # 4 – global smoothing with a small fixed kernel
        # --------------------------------------------------------------
        elif op == 4:
            candidate = np.convolve(candidate, smooth_kernel, mode="same")
            candidate = np.maximum(candidate, 0.0)

        # --------------------------------------------------------------
        # 5 – full sub‑gradient step on the maximal convolution entry
        # --------------------------------------------------------------
        else:  # op == 5
            k_max = int(np.argmax(cur_conv))
            g = _gradient_for_max(candidate, k_max)
            # step size is a tiny fraction of the current magnitude
            eta = rng.uniform(0.001, 0.008) * np.mean(g[g > 0]) if np.any(g > 0) else 0.0
            candidate -= eta * g
            candidate = np.maximum(candidate, 0.0)

        # --------------------------------------------------------------
        # Normalise, clip and evaluate
        # --------------------------------------------------------------
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # Re‑scale to keep total mass = 1 (scale does not affect AC1)
        total = candidate.sum()
        if total <= 0.0:
            continue
        candidate *= (1.0 / total)

        cand_score, cand_conv, cand_kmax = _score_and_conv(candidate)

        # --------------------------------------------------------------
        # Accept improvement (hill‑climbing)
        # --------------------------------------------------------------
        if cand_score < cur_score:
            cur = candidate
            cur_score = cand_score
            cur_conv = cand_conv
            cur_kmax = cand_kmax
            if cand_score < best_score:
                best = candidate
                best_score = cand_score
                best_conv = cand_conv

        # --------------------------------------------------------------
        # Early exit if we have already beaten the target
        # --------------------------------------------------------------
        if best_score < target_score:
            break

    # Return a plain Python list for the evaluator.
    return [float(x) for x in best]
