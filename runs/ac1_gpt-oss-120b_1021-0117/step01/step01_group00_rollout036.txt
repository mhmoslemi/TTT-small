```
def compute_score(seq):
    n = len(seq)
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    n_fft = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(fft_a * fft_a, n=n_fft)
    max_b = conv[:2*n-1].max()
    return float(2 * n * max_b / (sum_a**2))
```

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,float(budget_s) - 2.0)

        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    best_seq = parent.copy()
    best_score = compute_score(best_seq)

    # Create alternative ordering
    sorted_vals = np.sort(parent)[::-1]   # descending
    alt_seq = np.empty_like(parent)
    left = 0
    right = n-1
    toggle = True
    for val in sorted_vals:
        if toggle:
            alt_seq[left] = val
            left += 1
        else:
            alt_seq[right] = val
            right -= 1
        toggle = not toggle
    alt_score = compute_score(alt_seq)
    if alt_score < best_score:
        best_seq = alt_seq.copy()
        best_score = alt_score

    # Also start with sidon-like if length large enough
    def sidon_seq(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
        return a
    if n <= 2000:   # to keep moderate complexity

def compute_score(seq):
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    # compute FFT length
    m = 2*n - 1
    n_fft = 1 << (m - 1).bit_length()
    fft_a = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(fft_a * fft_a, n=n_fft)
    max_b = conv[:m].max()
    return float(2 * n * max_b / (sum_a*sum_a))

import numpy as np
import time

# ---------- helper functions ----------

def _full_conv(seq):
    """Full convolution of seq with itself using FFT (real inputs)."""
    n = seq.shape[0]
    m = 2 * n - 1
    # next power of two for FFT length (fast)
    n_fft = 1 << ((m - 1).bit_length())
    fft_a = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(fft_a * fft_a, n=n_fft)[:m]
    return conv


def _score_from_conv(seq, conv):
    """Score 2*n*max(conv)/(sum(seq)^2) given a pre‑computed convolution."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


def _conv_and_score(seq):
    """Return (conv, score) for the given sequence."""
    conv = _full_conv(seq)
    return conv, _score_from_conv(seq, conv)


def _sidon_like(n):
    """Classic Sidon (central binomial) coefficients a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _arcsine_shape(n):
    """Discrete analogue of the arcsine density 1/sqrt(x(1-x))."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n          # points in (0,1)
    a = 1.0 / np.sqrt(x * (1.0 - x))
    return np.clip(a, 0.0, 1000.0)


def _interleave_desc(vals):
    """Spread the largest values to opposite ends of the array."""
    n = vals.shape[0]
    out = np.empty_like(vals)
    left, right = 0, n - 1
    take_left = True
    for v in vals:
        if take_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        take_left = not take_left
    return out
