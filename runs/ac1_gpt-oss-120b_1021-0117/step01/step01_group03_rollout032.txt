import numpy as np
import time

def _conv_fft(a):
    n = a.size
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[:2*n-1]
    return conv

def _score_fast(a):
    n = a.size
    b = _conv_fft(a)
    return 2.0 * n * b.max() / (a.sum() ** 2)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Get length and initial sequence
    if "height_sequence_1" in globals():
        seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = seq.size
    else:
        n = 5582
        seq = np.ones(n, dtype=np.float64) / n  # uniform fallback

    # Ensure sum not too small
    if seq.sum() < 0.01:
        seq = seq + 0.01 / n

    # Baseline evaluate
    best = seq.copy()
    best_score = _score_fast(best)

    # optionally compute exact for best later
    # Main loop
    iteration = 0
    # Set some parameters
    base_lr = 0.5   # adjust later
    lr_decay = 0.9995
    lr = base_lr

    while time.monotonic() < deadline:
        # compute convolution and max index
        b = _conv_fft(best)
        max_b = b.max()
        k_max = int(np.argmax(b))

        # compute gradient for this max index
        grad = np.zeros_like(best)

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = k_max - idx
        # partner will be between 0 and n-1
        grad[idx] += 2.0 * best[partner]

        # step
        cand = best - lr * grad

        # Clip and rescale sum to original sum
        np.clip(cand, 0.0, 1000.0, out=cand)

        s = cand.sum()
        if s > 0:
            cand *= (best.sum() / s)

        # Evaluate fast score
        sc = _score_fast(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            # maybe increase learning rate a bit after improvement
            lr = min(base_lr, lr / lr_decay)  # actually increase?
        else:
            # no improvement; try random perturbation
            # choose random operation: transfer mass or block scaling
            op = rng.random()
            cand2 = best.copy()
            if op < 0.5:
                # transfer mass
                early = rng.integers(0, n//3)
                late = rng.integers(2*n//3, n)
                amt = min(cand2[early], 1000.0 - cand2[late]) * rng.random() * 0.1

                cand2[late] += amt

                cand2[start:start+width] *= factor
            else:
                noise = rng.normal(0.0, 0.01, size=n)
                cand2 += noise
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            # rescale sum

            if s2 > 0:

import numpy as np
import time


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score_fast(a: np.ndarray) -> float:
    """Fast evaluation of the objective (identical to the official one for
    non‑negative bounded inputs)."""
    n = a.size
    b = _conv_fft(a)
    return 2.0 * n * b.max() / (a.sum() ** 2)
