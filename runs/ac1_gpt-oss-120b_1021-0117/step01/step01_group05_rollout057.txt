def _move_maxpair_to_low(seq, conv, rng):
    n = seq.shape[0]
    k_max = int(np.argmax(conv))
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx_i = np.arange(lo, hi+1)
    idx_j = k_max - idx_i
    contributions = seq[idx_i] * seq[idx_j]
    # choose pair with max contribution
    max_idx = int(np.argmax(contributions))
    i = idx_i[max_idx]
    j = idx_j[max_idx]
    # choose target index with minimal contribution (excluding i and j)
    min_idx = int(np.argmin(contributions))
    t = int(idx_i[min_idx])
    # Ensure t not equal i or j
    if t == i or t == j:
        # find a low contribution index from the whole array (maybe random low)
        # Choose random index with small seq value
        low_candidates = np.where(seq < np.percentile(seq, 10))[0]
        if low_candidates.size > 0:
            t = int(rng.choice(low_candidates))
        else:
            t = int(rng.integers(0, n))
    # fraction to move
    alpha = rng.uniform(0.04, 0.12)
    candidate = seq.copy()
    if i == j:
        removed = candidate[i] * alpha
        candidate[i] -= removed
        total_removed = removed
    else:
        removed_i = candidate[i] * alpha
        removed_j = candidate[j] * alpha
        candidate[i] -= removed_i
        candidate[j] -= removed_j
        total_removed = removed_i + removed_j
    candidate[t] += total_removed
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    # renorm
    s = candidate.sum()
    if s > 0:
        candidate /= s
    return candidate
```

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities – top‑level, no closures
# --------------------------------------------------------------

def _initial_binomial(n: int) -> np.ndarray:
    """Binomial (Catalan) seed a_i = C(2i,i)/4^i, i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_and_max(seq: np.ndarray):
    """Full autocorrelation via FFT, returns (conv_array, max_value)."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    return conv, conv.max()


def _score(seq: np.ndarray):
    """Exact AC1 bound for a sequence (sum‑normalised)."""
    conv, mx = _conv_and_max(seq)
    s = seq.sum()
    if s == 0.0:
        return float("inf"), conv, mx
    return 2.0 * len(seq) * mx / (s * s), conv, mx


def _move_maxpair_to_low(seq: np.ndarray, conv: np.ndarray, rng: np.random.Generator):
    """Shift a small fraction of mass from the most contributing pair
    (for the current maximal convolution index) to a low‑contribution index."""
    n = seq.shape[0]
    k_max = int(np.argmax(conv))
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx_i = np.arange(lo, hi + 1)
    idx_j = k_max - idx_i
    contrib = seq[idx_i] * seq[idx_j]

    # most and least contributing positions
    max_idx = int(np.argmax(contrib))
    i = int(idx_i[max_idx])
    j = int(idx_j[max_idx])

    min_idx = int(np.argmin(contrib))
    t = int(idx_i[min_idx])
    if t == i or t == j:
        # choose a genuinely small entry
        low = np.where(seq < np.percentile(seq, 5))[0]
        t = int(rng.choice(low) if low.size else rng.integers(0, n))

    alpha = rng.uniform(0.04, 0.12)          # fraction to move
    cand = seq.copy()
    if i == j:
        removed = cand[i] * alpha
        cand[i] -= removed
        total = removed
    else:
        rem_i = cand[i] * alpha
        rem_j = cand[j] * alpha
        cand[i] -= rem_i
        cand[j] -= rem_j
        total = rem_i + rem_j
    cand[t] += total
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand


def _block_scaling_move(seq: np.ndarray, rng: np.random.Generator):
    """Multiply a random contiguous block by a log‑normal factor."""
    n = seq.shape[0]
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.07))
    cand = seq.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand


def _single_scaling_move(seq: np.ndarray, rng: np.random.Generator):
    """Scale a single entry by a log‑normal factor."""
    n = seq.shape[0]
    idx = rng.integers(0, n)
    factor = np.exp(rng.normal(0.0, 0.25))
    cand = seq.copy()
    cand[idx] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand


def _smoothing_move(seq: np.ndarray, rng: np.random.Generator):
    """Replace two neighbours by their average (preserves total sum)."""
    n = seq.shape[0]
    i = rng.integers(0, n - 1)
    avg = (seq[i] + seq[i + 1]) * 0.5
    cand = seq.copy()
    cand[i] = avg
    cand[i + 1] = avg
    return cand


def _lp_improve(seq: np.ndarray, M: float, mix: float, rng: np.random.Generator):
    """One LP refinement step: maximise ∑g under (seq * g) ≤ M,
    then convex‑combine with the current sequence."""
    try:
        import cvxpy as cp
    except Exception:
        return seq                     # cvxpy unavailable – skip LP step

    n = seq.shape[0]
    g = cp.Variable(n, nonneg=True)
    f_const = cp.Constant(seq)
    conv = cp.conv(f_const, g)       # length 2n‑1, linear in g
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv <= M])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=4000)
        except Exception:
            return seq
    if g.value is None:
        return seq

    g_vec = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    sum_f = seq.sum()
    sum_g = g_vec.sum()
    if sum_g == 0.0:
        return seq
    g_scaled = g_vec * (sum_f / sum_g)

    cand = (1.0 - mix) * seq + mix * g_scaled
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step‑function that minimises the AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Detect parent construction (height_sequence_1) if supplied.
    # ------------------------------------------------------------------
    n = 3809

    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Initialise sequence – either the provided parent or the binomial seed.
    # ------------------------------------------------------------------
    if parent is not None:
        cur = parent.copy()
    else:
        cur = _initial_binomial(n)

    np.clip(cur, 0.0, 1000.0, out=cur)
    s = cur.sum()
    cur = cur / s if s > 0.0 else np.ones_like(cur) / cur.size

    best_score, best_conv, _ = _score(cur)
    best_seq = cur.copy()

    global_seq = best_seq.copy()
    global_score = best_score

    it = 0
    no_improve = 0
    last_lp = -1

    while time.monotonic() < deadline:
        it += 1
        r = rng.random()
        if r < 0.40:
            cand = _move_maxpair_to_low(best_seq, best_conv, rng)
        elif r < 0.70:
            cand = _block_scaling_move(best_seq, rng)
        elif r < 0.90:
            cand = _single_scaling_move(best_seq, rng)
        else:
            cand = _smoothing_move(best_seq, rng)

        cand_score, cand_conv, _ = _score(cand)

        # ----- global best tracking -----
        if cand_score < global_score:
            global_seq = cand.copy()
            global_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        # ----- acceptance for the next iteration (simple hill‑climb) -----
        if cand_score < best_score:
            best_seq, best_score, best_conv = cand, cand_score, cand_conv
        else:
            # occasional uphill acceptance (simulated annealing)
            T = 0.02 / np.sqrt(it)
            if np.exp(-(cand_score - best_score) / max(T, 1e-12)) > rng.random():
                best_seq, best_score, best_conv = cand, cand_score, cand_conv

        # ----- periodic LP refinement (if cvxpy is available) -----
        if (it - last_lp) >= 200:
            M_cur = best_conv.max()
            lp_seq = _lp_improve(best_seq, M_cur, mix=0.2, rng=rng)
            lp_score, lp_conv, _ = _score(lp_seq)
            if lp_score < global_score:
                global_seq, global_score = lp_seq.copy(), lp_score
                if lp_score < best_score:
                    best_seq, best_score, best_conv = lp_seq, lp_score, lp_conv
            last_lp = it

        # safety exit a few tenths of a second before the hard timeout
        if time.monotonic() + 0.1 >= deadline:
            break

    # Return the best sequence found as a plain Python list.
    return [float(x) for x in global_seq]
