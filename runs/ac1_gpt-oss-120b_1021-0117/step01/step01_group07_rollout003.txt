    import time, numpy as np

    # Determine deadline
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # leave buffer

    # Get parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise Exception
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        parent_arr = None

    # Choose initialization
    n = None
    if parent_arr is not None:
        n = parent_arr.size
        init = parent_arr.copy()
    else:
        # fallback: uniform length 256? Actually we can pick length 256 (hard-coded)
        n = 256
        init = np.ones(n, dtype=np.float64)

    # Normalise to sum 1 (or sqrt(2n) not needed)
    total = init.sum()
    if total < 0.01:
        init = np.ones_like(init) / init.size
    else:
        init = init / total

    best = init
    best_score = float(evaluate_sequence(best.tolist()))
    # maybe parent may have higher sum? Anyway evaluate.

    # Step size parameters
    step_factor = 0.05
    min_step = 1e-5
    rng = np.random.default_rng(seed)

    # thresholds
    target = 1.50274349

    # define function to compute gradient
    def subgrad_max_convolution(a):
        c = np.convolve(a, a)
        M = c.max()
        # find all indices achieving within tolerance relative to M
        # tolerance small absolute needed
        tol = 1e-12
        idx = np.where(c >= M - tol)[0]
        # compute gradient w
        n = a.shape[0]
        w = np.zeros(n, dtype=np.float64)
        # try vectorized loops using broadcasting? Use Python loops over idx
        for k in idx:
            # w[0:k+1] += a[k::-1]
            # w[k:] += a[:n-k]
            w[:k+1] += a[k::-1]
            w[k:] += a[:n-k]
        return w, M

    # main loop
    while time.monotonic() < deadline:
        # decide operation
        op = rng.random()
        cand = None
        if op < 0.55:
            # subgradient descent step
            grad, M = subgrad_max_convolution(best)
            maxg = grad.max()
            if maxg > 0:
                step = step_factor * (best.max() / maxg)
                cand = best - step * grad
                # clip
                np.clip(cand, 0, None, out=cand)
                s = cand.sum()
                if s > 0:
                    cand = cand / s
        elif op < 0.75:
            # random transfer
            i, j = rng.choice(best.shape[0], size=2, replace=False)
            if best[i] > 0:
                frac = rng.random() * 0.5  # up to 0.5 of mass from i
                transfer = frac * best[i]
                cand = best.copy()
                cand[i] -= transfer
                cand[j] += transfer
                # rescale sum (should be same)
                # ensure positivity (transfer from nonzero)
                s = cand.sum()
                if s > 0:
                    cand = cand / s
        else:
            # jitter multiplicative perturbation
            jitter = np.exp(rng.normal(0.0, 0.018, size=best.shape[0]))
            cand = best * jitter
            s = cand.sum()
            if s > 0:
                cand = cand / s

        if cand is None:
            continue

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # adjust step factor depending on operation
            if op < 0.55:
                step_factor = min(step_factor * 1.2, 1.0)
            # early exit
            if best_score <= target:
                break
        else:
            if op < 0.55:
                step_factor = max(step_factor * 0.5, min_step)

    # final result convert to list
    # optionally re-scale to keep sum within [0,1000] but not needed.
    return [float(v) for v in best]

                prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
                try:
                    prob.solve(solver=cp.ECOS, warm_start=True,
                               max_iters=2000, verbose=False)
                except Exception:
                    prob = None
                if prob is not None and prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):

                    if g_val is not None:

                            if cand_score < best_score:

                                best_score = cand_score

import time
import numpy as np

# optional linear programming for large improvements
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def _catalan_weights(n: int) -> np.ndarray:
    """First n Catalan‐style weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _subgrad_max_conv(a: np.ndarray):
    """Sub‑gradient of the max of a*a (non‑negative, sum‑fixed)."""
    conv = np.convolve(a, a)
    M = conv.max()
    # indices where max is attained (tiny tolerance)
    idx = np.where(conv >= M - 1e-12)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in idx:
        # contribution a[k-i] for i≤k
        grad[:k + 1] += a[k::-1]
        # contribution a[i-k] for i≥k
        grad[k:] += a[: n - k]
    return grad, M


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the AC1 upper bound as defined by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    # leave a safety margin for the sandbox
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # ------------------------------------------------------------------
    # 1.  Obtain a starter sequence (parent if available, else Catalan)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
        init = parent.copy()
    except Exception:  # pragma: no cover
        # fallback: Catalan‑style construction of the same length as parent
        # (if parent absent we pick a moderate length)
        n0 = 256
        init = _catalan_weights(n0)
        init = init / init.sum()

    n = int(init.size)
    # normalise to L1 = 1 (the objective is scale‑invariant)
    total = init.sum()
    if total < 0.01:
        init = np.ones_like(init) / n
    else:
        init = init / total

    best = init.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    target_score = 1.50274349

    # --------------------------------------------------------------
    # 2.  Simple stochastic optimiser – main loop
    # --------------------------------------------------------------
    step_frac = 0.05
    min_step = 1e-5

    while time.monotonic() < deadline:
        # ----- occasional LP improvement (only if cvxpy available) -----
        if cp is not None and rng.random() < 0.12:
            time_left = deadline - time.monotonic()
            if time_left > 5.0:          # give some room for a solve
                # current bound M
                _, M_cur = _subgrad_max_conv(best)
                g = cp.Variable(n)
                f_const = cp.Constant(best)
                constraints = [
                    cp.conv(f_const, g) <= M_cur,
                    g >= 0
                ]
                prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
                try:
                    prob.solve(solver=cp.ECOS, max_iters=3000, warm_start=True)
                except Exception:  # pragma: no cover
                    prob = None

                if prob is not None and prob.status in (cp.OPTIMAL,
                                                       cp.OPTIMAL_INACCURATE):
                    g_val = np.asarray(g.value, dtype=np.float64)
                    if g_val is not None:
                        # keep a feasible g, normalise to sum=1
                        g_val = np.maximum(g_val, 0.0)
                        s = g_val.sum()
                        if s > 0.0:
                            g_norm = g_val / s
                            # small convex blend with current best
                            t = 0.2
                            cand = (1 - t) * best + t * g_norm
                            cand = np.clip(cand, 0.0, None)
                            cand /= cand.sum()
                            score = float(evaluate_sequence(cand.tolist()))
                            if np.isfinite(score) and score < best_score:
                                best = cand
                                best_score = score
                                if best_score <= target_score:

        # ----- choose a local move ------------------------------------------------
        move_type = rng.random()
        cand = None

        if move_type < 0.55:                                 # sub‑gradient step
            grad, _ = _subgrad_max_conv(best)
            gmax = grad.max()
            if gmax > 0.0:
                step = step_frac * (best.max() / gmax)
                cand = best - step * grad
                np.clip(cand, 0.0, None, out=cand)
                cand_sum = cand.sum()
                if cand_sum > 0.0:
                    cand /= cand_sum

                eps = rng.random() * best[i] * 0.5
                cand = best.copy()
                cand[i] -= eps
                cand[j] += eps
                cand /= cand.sum()

        elif move_type < 0.95:                               # swap two entries

            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]          # simple exchange

        else:                                                # multiplicative jitter
            jitter = np.exp(rng.normal(0.0, 0.018, size=n))
            cand = best * jitter
            cand /= cand.sum()

        if cand is None:

        # ----- evaluate candidate -------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

                step_frac = max(step_frac * 0.5, min_step)

        # occasional early‑exit safety check
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3.  Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
