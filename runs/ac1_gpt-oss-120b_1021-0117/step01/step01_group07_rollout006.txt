import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------
def _catalan_weights(n: int) -> np.ndarray:
    """Generate Catalan‐style weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _conv_full(a: np.ndarray, L: int, n: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length L."""
    # rfft works on real data and returns the positive frequencies only
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    # numerical noise may produce tiny negatives – clip to 0
    np.maximum(conv, 0.0, out=conv)
    return conv


def _score_from_max(max_conv: float, n: int) -> float:
    """Exact AC1 upper bound when sum(a)=1."""
    return 2.0 * n * max_conv


def _gradient_at_max(a: np.ndarray, idx: int) -> np.ndarray:
    """Gradient of max_{i}(a*a)_i w.r.t. a at the active index idx."""
    n = a.shape[0]
    reflect = idx - np.arange(n)
    mask = (reflect >= 0) & (reflect < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[reflect[mask]]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of heights) whose AC1 upper
    bound is (hopefully) below the target.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1.  Obtain a starting vector (parent or Catalan)
    # ------------------------------------------------------------------
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None

    if parent is not None:
        n = int(parent.size)
        a = parent / parent.sum()
    else:
        n = 256
        a = _catalan_weights(n)
        a = a / a.sum()

    # ensure a is a proper probability vector
    a = np.maximum(a, 0.0)
    a /= a.sum()

    # ------------------------------------------------------------------
    # 2.  Prepare convolution parameters
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    L = 1 << (conv_len - 1).bit_length()  # next power of two ≥ conv_len

    # ------------------------------------------------------------------
    # 3.  Evaluate the starting candidate
    # ------------------------------------------------------------------
    conv = _conv_full(a, L, n)
    best_score = _score_from_max(conv.max(), n)
    best = a.copy()

    # ------------------------------------------------------------------
    # 4.  Quick parametric sweep (power‑law family) to seed a better start
    # ------------------------------------------------------------------
    idxs = np.arange(n, dtype=np.float64) + 0.5  # shift avoids singularity at 0
    for p in np.linspace(0.15, 1.05, 81):  # 81 trial exponents
        cand = idxs ** (-p)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        conv_c = _conv_full(cand, L, n)
        score_c = _score_from_max(conv_c.max(), n)
        if score_c < best_score:
            best = cand.copy()
            best_score = score_c
            a = cand.copy()
        if time.monotonic() >= deadline:
            return [float(v) for v in best]

    # ------------------------------------------------------------------
    # 5.  Main optimisation loop (gradient + stochastic moves)
    # ------------------------------------------------------------------
    TARGET = 1.50274349
    eta = 0.8                # step size for exponential gradient move
    min_eta = 1e-5
    max_eta = 5.0
    no_improve = 0

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # a)  Gradient‑based exponential move
        # --------------------------------------------------------------
        conv = _conv_full(a, L, n)
        i_max = int(conv.argmax())
        grad = _gradient_at_max(a, i_max)

        # try a few scaling factors around the current eta
        improved = False
        for factor in (1.0, 0.5, 2.0):
            step = eta * factor
            # avoid overflow/underflow in the exponent
            exp_factor = np.exp(-np.clip(step * grad, -50.0, 50.0))
            cand = a * exp_factor
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            conv_c = _conv_full(cand, L, n)
            score_c = _score_from_max(conv_c.max(), n)
            if score_c + 1e-12 < best_score:
                best = cand.copy()
                best_score = score_c
                a = cand
                eta = min(max_eta, step * 1.15)
                improved = True
                no_improve = 0
                break
        if improved:
            if best_score <= TARGET:
                break
            continue

        # --------------------------------------------------------------
        # b)  Random pairwise mass transfer (conservative move)
        # --------------------------------------------------------------
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] > 0.0:
            eps = rng.random() * a[i] * 0.5
            cand = a.copy()
            cand[i] -= eps
            cand[j] += eps
            cand /= cand.sum()
            conv_c = _conv_full(cand, L, n)
            score_c = _score_from_max(conv_c.max(), n)
            if score_c + 1e-12 < best_score:
                best = cand.copy()
                best_score = score_c
                a = cand
                no_improve = 0
                if best_score <= TARGET:
                    break
                continue

        # --------------------------------------------------------------
        # c)  Small multiplicative jitter (log‑normal noise)
        # --------------------------------------------------------------
        if rng.random() < 0.07:
            jitter = np.exp(rng.normal(0.0, 0.015, size=n))
            cand = a * jitter
            cand /= cand.sum()
            conv_c = _conv_full(cand, L, n)
            score_c = _score_from_max(conv_c.max(), n)
            if score_c + 1e-12 < best_score:
                best = cand.copy()
                best_score = score_c
                a = cand
                if best_score <= TARGET:
                    break
                continue

        # --------------------------------------------------------------
        # d)  Adaptive step‑size decay if nothing improved
        # --------------------------------------------------------------
        no_improve += 1
        if no_improve > 30:
            eta = max(min_eta, eta * 0.6)
            no_improve = 0

        # occasional fresh restart blending with Catalan shape
        if rng.random() < 0.02:
            cat = _catalan_weights(n)
            cat = cat / cat.sum()
            t = rng.random()
            cand = (1.0 - t) * a + t * cat
            cand /= cand.sum()
            conv_c = _conv_full(cand, L, n)
            score_c = _score_from_max(conv_c.max(), n)
            if score_c + 1e-12 < best_score:
                best = cand.copy()
                best_score = score_c
                a = cand
                eta = 0.8
                if best_score <= TARGET:
                    break

    # ------------------------------------------------------------------
    # 6.  Return the best sequence found
    # ------------------------------------------------------------------
    return [float(v) for v in best]
