def _conv(vec):
    n = vec.size
    size = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(vec, n=size)
    conv = np.fft.irfft(fv * fv, n=size)[:2*n-1]
    return conv
```

import time, numpy as np

def _project_simplex(v):

def _conv(vec):
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    cv = np.fft.irfft(fv * fv, n=L)
    return cv[:2 * n - 1]

def _score(vec):
    n = vec.size
    conv = _conv(vec)
    M = conv.max()
    return 2.0 * n * M / (vec.sum() ** 2)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0) # 5 sec headroom
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        # ensure nonnegative
        base = np.maximum(base, 0.0)
    else:
        n = 256
        base = _sqrt_series(n)
    # normalize to sum=1 via simplex projection
    if base.sum() == 0:
        base = np.full_like(base, 1.0)
    cur = base / base.sum()
    cur = _project_simplex(cur)  # ensures sum=1
    best = cur.copy()
    best_sc = _score(best)

    n = best.size

    # hyperparameters
    max_iter = 200000  # large but we will break on deadline
    # step sizes
    lr = 0.1  # initial gradient step factor (to be tuned)
    # We'll decrease lr gradually
    while time.monotonic() < deadline and max_iter > 0:
        max_iter -= 1

        # occasionally perform random mass transfer
        if rng.random() < 0.3:
            # biased source and target
            # source from first part (0..n//3), target from last part (2n//3..n-1)
            i = rng.integers(0, max(1, n // 3))
            j = rng.integers(2 * n // 3, n)
            # ensure distinct
            if i == j:
                continue
            # transfer fraction of source
            frac = rng.random() * 0.5  # up to 50%
            delta = cur[i] * frac
            if delta <= 0:
                continue
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)  # preserve sum=1
        else:
            # gradient step on current max conv index
            conv = _conv(cur)
            idx = int(np.argmax(conv))
            # compute gradient
            g = np.zeros_like(cur)
            lo = max(0, idx - (n - 1))
            hi = min(idx, n - 1)
            # i range
            # use vectorized indexing
            i_range = np.arange(lo, hi + 1)
            g[i_range] = 2.0 * cur[idx - i_range]

            cand = cur - step * g
            cand = _project_simplex(cand)

        if sc < best_sc:

            best = cand.copy()
            # keep current as best to continue descent
            cur = cand.copy()
            # possibly reduce step size a bit
            lr = max(1e-4, lr * 0.99)

            # occasionally accept worse to escape
            if rng.random() < 0.01:
                cur = cand.copy()
                # maybe increase step a bit
                lr = min(0.5, lr * 1.01)

    return [float(x) for x in best]

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_candidates = np.where(rho_arr > 0)[0]
    if rho_candidates.size == 0:
        theta = 0.0
    else:
        rho = rho_candidates[-1]
        theta = css[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# sqrt-series (a*a = 1+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# convolution via FFT (linear, length 2n‑1)
# ------------------------------------------------------------
def _conv(vec: np.ndarray) -> np.ndarray:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fv = np.fft.rfft(vec, n=L)
    cv = np.fft.irfft(fv * fv, n=L)
    return cv[: 2 * n - 1]

# ------------------------------------------------------------
def _ac1_score(vec: np.ndarray) -> float:
    n = vec.size
    conv = _conv(vec)
    m = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * m / (s * s)

        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        ind = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[ind] = 2.0 * cur[idx - ind]

        # diminishing step size
        eps = step0 / np.sqrt(it)

        cand = cur - eps * g
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_sc = _ac1_score(cand)

        # always move (sub‑gradient method); keep the best ever seen
        cur = cand
        if cand_sc < best_sc:
            best_sc = cand_sc
            best = cand.copy()

        # occasional small random transfers (exploration)
        if it % 50 == 0:
            # move a tiny fraction from an early to a late index
            i = rng.integers(0, max(1, n // 3))
            j = rng.integers(2 * n // 3, n)
            if i != j:
                frac = rng.random() * 0.05  # up to 5 % of source mass
                delta = cur[i] * frac
                cur[i] -= delta
                cur[j] += delta
                cur = _project_simplex(cur)
                cur_sc = _ac1_score(cur)
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best = cur.copy()

    # ------------------------------------------------------------
    # return as plain Python list (evaluator will clamp etc.)
    # ------------------------------------------------------------
    return [float(x) for x in best]
