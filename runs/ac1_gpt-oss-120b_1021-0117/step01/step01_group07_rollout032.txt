def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic()+max(0.0, float(budget_s)-3.0)

    try: parent = np.asarray(height_sequence_1, dtype=np.float64); except: parent = None

    n = len(parent) if parent is not None else 256

    #function to normalize sum to 1
    def norm_to_one(arr):
        s = arr.sum()
        if s == 0: return arr
        return arr / s

    # generate initial candidates
    init_candidates = []

    # parent normalized if exists
    if parent is not None and parent.ndim==1 and parent.size>0 and np.isfinite(parent).all():
        parent_norm = norm_to_one(np.clip(parent,0,1000))
        init_candidates.append(parent_norm)

    # uniform
    init_candidates.append(np.full(n, 1.0/n))

    # Catalan
    # compute catalan weights
    w = np.empty(n, dtype=np.float64)
    w[0]=1.0
    for i in range(1,n):
        w[i] = w[i-1] * ((2*i)*(2*i-1))/(i*i*4.0)
    init_candidates.append(norm_to_one(w))

    # exponentials
    for r in [0.99,0.98,0.96,0.94]:
        a = np.power(r, np.arange(n, dtype=np.float64))
        init_candidates.append(norm_to_one(a))

    # power law
    for p in [0.5,1.0,1.5,2.0]:
        a = 1.0/np.power(np.arange(1,n+1,dtype=np.float64), p)
        init_candidates.append(norm_to_one(a))

    # binomial (using pmf)
    from math import comb
    # compute binom pmf for p in [0.3,0.5,0.7]
    for p in [0.3,0.5,0.7]:
        # compute using recursion: pmf[0] = (1-p)^{n-1}
        k = np.arange(n, dtype=np.float64)
        # log pmf: log comb(n-1, k) + k*log(p) + (n-1-k)*log(1-p)
        # use scipy.special.comb maybe large; but we can compute iteratively:
        pmf = np.empty(n, dtype=np.float64)
        pmf[0] = (1-p)**(n-1)
        for i in range(1,n):
            pmf[i] = pmf[i-1] * ( (n- i) / i ) * (p/(1-p))
        init_candidates.append(norm_to_one(pmf))

    # random Dirichlet vectors
    for _ in range(3):
        a = rng.random(n)
        # raise to power for more variability
        a = a ** rng.uniform(0.5,2.0)
        init_candidates.append(norm_to_one(a))

    # evaluate and pick best
    best = None
    best_score = float('inf')
    for cand in init_candidates:
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # gradient descent loop
    # set parameters
    step_frac = 0.1   # initial fraction of source entry to move
    min_step = 1e-5
    max_step = 0.5
    no_improve = 0

    while time.monotonic() < deadline:
        # compute convolution
        conv = np.convolve(best, best)
        idx_max = int(np.argmax(conv))
        # gradient
        grad = np.zeros(n, dtype=np.float64)
        # for i where idx_max-i in [0,n-1], grad[i]=2*best[idx_max-i]
        # can compute via reversed slice
        # the range of i: i from max(0, idx_max-(n-1)) to min(n-1, idx_max)
        i_start = max(0, idx_max - (n-1))
        i_end = min(n-1, idx_max)
        # compute slice indices
        # For i in range(i_start,i_end+1):
        #   j = idx_max - i
        #   grad[i] = 2*best[j]
        # we can assign vectorized:
        i_range = np.arange(i_start, i_end+1, dtype=np.int64)
        j_range = idx_max - i_range
        grad[i_range] = 2.0 * best[j_range]

        # Identify source (max grad) and sink (min grad)
        # we ignore positions where best[i]==0 since moving from there impossible
        # also ensure we don't move all mass to sink that may already cause high conv but ok.
        # Choose i_max from positions with best>0.
        mask_positive = best > 1e-15
        # if no positive entries? shouldn't happen
        if not mask_positive.any():
            break
        # compute grad for these positions
        grad_pos = grad.copy()
        grad_pos[~mask_positive] = -np.inf
        i_max = int(np.argmax(grad_pos))
        # choose sink among positions with minimal grad (including zeros)
        # optionally prioritize positions where best is small
        # We'll just choose i_min = argmin grad
        i_min = int(np.argmin(grad))
        # Ensure they are distinct; else pick next min
        if i_max == i_min:
            # find second min
            sorted_idx = np.argsort(grad)
            for ii in sorted_idx:
                if ii != i_max:
                    i_min = int(ii)
                    break

        # determine epsilon
        eps = step_frac * best[i_max]
        # ensure eps >0
        if eps <= 0:
            # if step_frac too low, break
            break
        # ensure after move best[i_max] - eps >=0
        eps = min(eps, best[i_max])
        # apply move
        cand = best.copy()
        cand[i_max] -= eps
        cand[i_min] += eps
        # renormalize sum (should stay same), but due to float maybe small diff
        s = cand.sum()
        if s == 0:

        cand /= s

        # evaluate
        score = evaluate_sequence(cand.tolist())
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # improve step size
            step_frac = min(step_frac*1.2, max_step)
            no_improve = 0
            # early exit
            if best_score <= 1.50274349:
                break
        else:
            # not improved: reduce step size
            step_frac = max(step_frac*0.5, min_step)
            no_improve += 1

        # occasional random perturbation
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            cand2 = best * jitter
            cand2 /= cand2.sum()
            score2 = evaluate_sequence(cand2.tolist())
            if np.isfinite(score2) and score2 < best_score:
                best = cand2
                best_score = score2
                step_frac = min(step_frac*1.2, max_step)
                if best_score <= 1.50274349:
                    break

        # if too many iterations without improvement, maybe random restart
        if no_improve > 2000:
            # random reinitialization from saved top candidates or simple distribution
            # choose a random initial candidate from init_candidates
            cand_restart = rng.choice(init_candidates)
            # add small jitter
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            cand_restart = cand_restart * jitter
            cand_restart /= cand_restart.sum()
            candidate_score = evaluate_sequence(cand_restart.tolist())
            if candidate_score < best_score:
                best = cand_restart
                best_score = candidate_score
                step_frac = 0.1
            no_improve = 0

    # return best as list

import time

import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimizes the
    AC1 upper bound defined by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1.  Load parent if available, otherwise fall back to a modest size.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
    except Exception:
        n = 256
        parent = None

    # ------------------------------------------------------------------
    # 2.  Helper: normalise to unit sum.
    # ------------------------------------------------------------------
    def unit_norm(arr):
        s = arr.sum()
        return arr if s == 0 else arr / s

    # ------------------------------------------------------------------
    # 3.  Build a palette of cheap initial candidates.
    # ------------------------------------------------------------------
    init_cands = []

    if parent is not None:
        init_cands.append(unit_norm(parent))

    # uniform
    init_cands.append(np.full(n, 1.0 / n))

    # Catalan‑style (central binomial) weights
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    init_cands.append(unit_norm(w))

    # exponential families
    for r in (0.99, 0.97, 0.95, 0.93):
        a = np.power(r, np.arange(n, dtype=np.float64))
        init_cands.append(unit_norm(a))

    # power‑law families
    for p in (0.5, 1.0, 1.5, 2.0):
        a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        init_cands.append(unit_norm(a))

    # a few random Dirichlet‑style draws
    for _ in range(3):
        a = rng.random(n) ** rng.uniform(0.5, 2.0)
        init_cands.append(unit_norm(a))

    # ------------------------------------------------------------------
    # 4.  Pick the best of the palette.
    # ------------------------------------------------------------------
    best = None
    best_score = float('inf')
    for cand in init_cands:
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # early exit if we already beat the target
    target = 1.50274349
    if best_score <= target:
        return [float(v) for v in best]

    # ------------------------------------------------------------------
    # 5.  Gradient‑descent style local optimisation.
    #    Move a small “epsilon” from the index that most increases the
    #    current maximal autocorrelation to the one that contributes least.
    # ------------------------------------------------------------------
    step_frac = 0.08          # fraction of source mass to relocate
    min_step = 1e-6
    max_step = 0.5
    stagnation = 0           # count of non‑improving iterations

    while time.monotonic() < deadline:
        # convolution of the current best with itself
        conv = np.convolve(best, best)
        idx_max = int(np.argmax(conv))

        # gradient of b_{idx_max} w.r.t. each a_i : 2*a_{idx_max-i} if in range
        grad = np.zeros(n, dtype=np.float64)
        i_start = max(0, idx_max - (n - 1))
        i_end = min(n - 1, idx_max)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = idx_max - i_range
        grad[i_range] = 2.0 * best[j_range]

        # source = index with large gradient AND positive mass
        mask_pos = best > 1e-15
        if not mask_pos.any():
            break
        grad_src = grad.copy()
        grad_src[~mask_pos] = -np.inf
        i_src = int(np.argmax(grad_src))

        # sink = index with smallest gradient (could be zero)
        i_sink = int(np.argmin(grad))
        if i_src == i_sink:
            # pick a distinct sink
            sorted_idx = np.argsort(grad)
            for ii in sorted_idx:
                if ii != i_src:
                    i_sink = int(ii)
                    break

        # move a fraction of the mass from src to sink
        eps = step_frac * best[i_src]
        eps = min(eps, best[i_src])          # keep non‑negative
        if eps <= 0:
            break
        cand = best.copy()
        cand[i_src] -= eps
        cand[i_sink] += eps
        cand /= cand.sum()                   # renormalise (sum stays 1)

        score = evaluate_sequence(cand.tolist())
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            step_frac = min(step_frac * 1.2, max_step)
            stagnation = 0
            if best_score <= target:
                break
        else:
            step_frac = max(step_frac * 0.5, min_step)
            stagnation += 1

        # occasional stochastic jitter to escape plateaux
        if rng.random() < 0.009:
            jitter = np.exp(rng.normal(0.0, 0.018, size=n))
            cand_j = best * jitter
            cand_j /= cand_j.sum()
            sc_j = evaluate_sequence(cand_j.tolist())
            if np.isfinite(sc_j) and sc_j < best_score:
                best = cand_j
                best_score = sc_j
                step_frac = min(step_frac * 1.2, max_step)
                if best_score <= target:

            cand = cand * jitter
            cand /= cand.sum()
            sc = evaluate_sequence(cand.tolist())
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
                step_frac = 0.08
            stagnation = 0

    # ------------------------------------------------------------------
    # 6.  Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
