def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, float(budget_s) - 0.5)   # leave margin
    # determine n
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 256
        # compute sqrt_series
        init = _sqrt_series(n)
    # ensure nonneg and normalize to sum 1 (scale not needed but for simplicity)
    vec = np.maximum(init, 0)
    total = vec.sum()
    if total < 1e-12:
        vec = np.full_like(vec, 1.0)
        total = vec.sum()
    vec = vec / total  # sum = 1

    # one big evaluation function (fast)
    def score(v):
        n_local = v.size
        L = 1 << ((2*n_local - 1).bit_length())
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n_local-1]
        return 2.0 * n_local * conv.max() / (v.sum()**2)

    best_vec = vec.copy()
    best_score = score(best_vec)

    # loop
    # we will combine gradient steps with random mass transfer
    while time.monotonic() < deadline:
        # gradient step
        n_local = best_vec.size
        # compute conv
        L = 1 << ((2*n_local - 1).bit_length())
        fv = np.fft.rfft(best_vec, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n_local-1]
        max_val = conv.max()
        # find all indices near max (relative tolerance)
        tol = 1e-12 * max_val
        max_idx = np.where(conv >= max_val - tol)[0]

        # compute gradient
        grad = np.zeros_like(best_vec)
        for k in max_idx:
            # i ranges [max(0, k-(n-1)), min(n-1, k)]
            i_start = k - (n_local - 1)
            if i_start < 0:
                i_start = 0
            i_end = k if k < n_local - 1 else n_local - 1
            # vectorized: j = k - i
            i_range = np.arange(i_start, i_end + 1)
            grad[i_range] += 2.0 * best_vec[k - i_range]

        # direction = -grad (we want to reduce max)
        # try few step sizes
        improved = False
        step_candidates = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
        for step in step_candidates:
            cand = best_vec - step * grad
            # project onto simplex
            cand = _project_simplex(cand)
            sc = score(cand)
            if sc < best_score - 1e-12:  # improvement
                best_score = sc
                best_vec = cand
                improved = True
                break
        if not improved:
            # random mass transfer fallback: pick i,j and fraction
            i = rng.integers(0, n_local)
            if best_vec[i] == 0:
                continue
            j = rng.integers(0, n_local-1)
            if j >= i:
                j += 1
            frac = rng.random() * 0.3
            delta = best_vec[i] * frac
            cand = best_vec.copy()
            cand[i] -= delta
            cand[j] += delta
            # projection ensure sum=1
            cand = _project_simplex(cand)
            sc = score(cand)
            if sc < best_score:
                best_score = sc
                best_vec = cand

    return [float(x) for x in best_vec]

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = np.dot(a[1:k], a[k-1:0:-1])
        a[k] = (1.0 - s) / 2.0
    return a

import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (identical to evaluate_sequence)
# ------------------------------------------------------------
def _score(v: np.ndarray) -> float:
    """Return 2*n*max(conv(v,v))/(sum(v)**2) using FFT."""
    n = v.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (v.sum() ** 2)


# ------------------------------------------------------------
# Fallback analytic start (sqrt series) if parent not provided
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of a where a*a = 1+...+x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

        # indices where conv is (almost) maximal
