def fast_score(seq):
    n = seq.shape[0]
    m = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_val = conv.max()
    sum_a = seq.sum()
    if sum_a <= 0:
        return float('inf'), None, None
    score = 2.0 * n * max_val / (sum_a * sum_a)
    k_max = int(conv.argmax())
    return score, conv, k_max

import time

def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, argmax index) using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), None, None
    score = 2.0 * n * max_val / (s * s)
    k_max = int(conv.argmax())
    return float(score), conv, k_max

        # Update global best.
        # --------------------------------------------------------------
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            best_conv = cand_conv
            best_k = cand_k
            last_improve = it

        # --------------------------------------------------------------
        # Simulated‑annealing acceptance for the current walk.
        # --------------------------------------------------------------
        if cand_score < cur_score:
            cur_seq, cur_score, cur_conv, cur_k = cand, cand_score, cand_conv, cand_k
        else:
            diff = cand_score - cur_score
            prob = np.exp(-diff / (temperature * cur_score + 1e-12))
            if rng.random() < prob:
                cur_seq, cur_score, cur_conv, cur_k = cand, cand_score, cand_conv, cand_k

        it += 1

        # cooling schedule
        if it % 250 == 0:
            temperature *= 0.94

        # occasional restart if stagnated
        if it - last_improve > 5000:
            cur_seq, cur_score, cur_conv, cur_k = best_seq.copy(), best_score, best_conv, best_k
            last_improve = it

        # keep a small safety margin before the hard timeout
        if time.monotonic() + 0.2 >= deadline:
            break

    # Return a plain Python list – the evaluator will recompute the exact score.
    return [float(v) for v in best_seq]
