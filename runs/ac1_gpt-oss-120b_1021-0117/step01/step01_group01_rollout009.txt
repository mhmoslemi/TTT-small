import time, math, numpy as np

def _fft_len_for_n(n):
    conv_len = 2*n - 1
    return 1 << ((conv_len - 1).bit_length())

def _full_conv(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[:2*len(a)-1]

def _score_fft(a, n, fft_len):
    conv = _full_conv(a, fft_len)
    max_b = conv.max()
    s = a.sum()
    return float(2 * n * max_b / (s * s))

import math
import numpy as np

# ------------------------------------------------------------
# Helper functions (FFT based convolution, scoring, etc.)
# ------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length sufficient for a linear convolution."""
    conv_len = 2 * n - 1
    return 1 << ((conv_len - 1).bit_length())


def _full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a single real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _score_from_conv(conv: np.ndarray, n: int, total: float) -> float:
    """Exact AC1 bound from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total * total))

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)   # keep a safety margin

    if total < 1e-12:

    cur = base / total                                          # sum == 1 (scale‑invariant)
