def eval_score(a):
    # compute convolution via FFT (fast)
    n = a.shape[0]
    conv_len = 2*n -1
    fft_len = 1 << ((conv_len -1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    max_b = conv.max()
    s = a.sum()
    return 2 * n * max_b / (s * s)
```

    fa = np.fft.rfft(a, n=fft_len)
