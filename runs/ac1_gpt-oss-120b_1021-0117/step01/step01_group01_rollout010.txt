def simplex_projection(v):
    # v is numpy array.
    # Returns projection onto probability simplex.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import math
import numpy as np

def _fft_conv_full(x: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution x*x (full) using a single FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[: 2 * x.size - 1]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    # shift to guarantee numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are <=0 → return the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
