def _gradient_step(cur, conv, k_max, rng):
    n = cur.shape[0]
    # Build rev
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    rev = np.empty_like(cur)
    rev.fill(0.0)
    rev[mask] = cur[partner[mask]]
    grad = 2.0 * rev
    avg = grad.mean()
    d = -grad + avg
    max_d = np.abs(d).max()
    if max_d == 0:
        return cur.copy()
    # step fraction factor (scale)
    step_frac = rng.uniform(0.0, 0.05)  # up to 5%
    step = step_frac * cur.max() / max_d
    cand = cur + step * d
    # clip and renorm
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0:
        cand /= s
    return cand

import numpy as np
import time
import math


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _gen_binomial(n: int) -> np.ndarray:
    """Classic decreasing binomial coefficients a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(a: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 upper bound via FFT."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_c = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _gradient_candidate(cur: np.ndarray, k_max: int, rng: np.random.Generator) -> np.ndarray:
    """Move a little in the opposite direction of the gradient of the peak."""
    n = cur.shape[0]
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    rev = np.zeros_like(cur)
    rev[mask] = cur[partner[mask]]
    grad = 2.0 * rev
    avg = grad.mean()
    d = -grad + avg                     # zero‑sum direction
    max_d = np.abs(d).max()
    if max_d == 0:
        return cur.copy()
    # step size: up to a few percent of the current maximum entry
    step_frac = rng.uniform(0.0, 0.05)
    step = step_frac * cur.max() / max_d
    cand = cur + step * d
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0:
        cand /= s
    return cand


def _random_candidate(cur: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Block scaling or mass‑transfer perturbation."""
    n = cur.shape[0]
    cand = cur.copy()
    if rng.random() < 0.6:          # block scaling
        width = int(rng.integers(max(1, n // 200), max(2, n // 12) + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = math.exp(rng.normal(0.0, 0.12))
        cand[start:start + width] *= factor
    else:                           # move a fraction of mass
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            frac = rng.random() * 0.4
            delta = cand[i] * frac
            cand[i] -= delta
            cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0:
        cand /= s
    return cand


def _param_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a fresh candidate from a simple parametric family."""
    typ = rng.integers(0, 3)
    i = np.arange(n, dtype=np.float64)
    if typ == 0:                     # power‑law
        p = rng.uniform(0.1, 2.0)
        off = rng.uniform(0.0, 5.0)
        a = (i + off) ** (-p)
    elif typ == 1:                   # exponential
        lam = rng.uniform(0.0, 0.02)
        a = np.exp(-lam * i)
    else:                            # beta‑like shape on [0,1]
        a = np.linspace(0.0, 1.0, n, dtype=np.float64)
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        a = (a ** (alpha - 1)) * ((1 - a) ** (beta - 1) + 1e-12)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---------- problem size ----------
    parent = None
    if 'height_sequence_1' in globals():
        try:
            tmp = np.asarray(height_sequence_1, dtype=np.float64)
            if tmp.ndim == 1 and tmp.size > 0:
                parent = tmp
        except Exception:
            parent = None

    if parent is not None:
        n = int(parent.size)
    else:

    L = _next_pow2(2 * n - 1)

    # ---------- initial candidates ----------
    base = _gen_binomial(n)
    base = np.clip(base, 0.0, 1000.0, out=base)
    s = base.sum()
    if s > 0:
        base /= s
    best = base.copy()
    best_score = _ac1_score_fft(best, n, L)

    if parent is not None:
        parent = np.clip(parent, 0.0, 1000.0, out=parent)
        ps = parent.sum()
        if ps > 0:
            parent /= ps
            pscore = _ac1_score_fft(parent, n, L)
            if pscore < best_score:
                best = parent.copy()
                best_score = pscore

    cur = best.copy()
    cur_score = best_score

    # ---------- search ----------
    temperature = 1.0
    decay = 0.9995

    while time.monotonic() < deadline:
        # decide move type
        r = rng.random()
        if r < 0.45:                      # gradient step (needs conv)

            cand = _gradient_candidate(cur, k_max, rng)
        elif r < 0.85:                    # stochastic random move
            cand = _random_candidate(cur, rng)
        else:                             # fresh parametric candidate
            cand = _param_candidate(rng, n)

            best_score = cand_score
            if best_score < 1.50274349:
                break

        accept = False
        if cand_score < cur_score:
            accept = True

            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # occasional restart to the best found so far
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

        temperature *= decay

    # ---------- return ----------
    return [float(x) for x in best]
