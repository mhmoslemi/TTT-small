import time
import numpy as np

# ---------- helper utilities (top‑level) ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _self_convolve_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (length fft_len)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # length is len(a)+len(a)-1
    return conv[: 2 * a.size - 1]


def _weighted_convolve_fft(w: np.ndarray, rev_a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convolution of w (length 2n‑1) with rev_a (length n) via FFT."""
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(rev_a, n=fft_len)
    conv = np.fft.irfft(fw * fa, n=fft_len)
    return conv[: w.size + rev_a.size - 1]  # length = 3n‑2


def _softmax_weights(x: np.ndarray, beta: float) -> np.ndarray:
    """Numerically stable softmax with temperature beta."""
    mx = x.max()
    e = np.exp(beta * (x - mx))
    return e / e.sum()

    s = a.sum()
    return float(2 * a.size * maxb / (s * s))
