def gradient_at_max(a):
    conv = np.convolve(a, a)
    kmax = np.argmax(conv)
    # gradient shape: for each i, derivative of conv_kmax w.r.t a_i = 2*a_{kmax - i} if valid else 0
    grad = np.zeros_like(a)
    # compute reversed slice
    # for i where 0 <= kmax - i < n => i in [kmax-(n-1), kmax]
    lo = max(0, kmax - (n - 1))
    hi = min(n-1, kmax)   # inclusive
    # then grad[i] = 2*a[kmax - i]
    # using slicing:
    idx = np.arange(lo, hi+1, dtype=int)
    grad[idx] = 2 * a[kmax - idx]
    return grad, kmax
```

import time
import numpy as np

def _catalan_weights(n: int) -> np.ndarray:
    """First n Catalan‐style weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _normalize(v: np.ndarray) -> np.ndarray:
    s = v.sum()
    if s > 0.0:
        return v / s
    return v


def _score(v: np.ndarray) -> float:
    """Exact same objective as evaluate_sequence (without clipping)."""
    s = v.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(v, v)
    return float(2.0 * v.size * conv.max() / (s * s))

    if a[i_high] <= 0.0:
        return a.copy()

    delta = a[i_high] * rng.random() * 0.3    # up to 30 % of the source mass
    b = a.copy()
    b[i_high] -= delta
    b[i_low] += delta
    # sum is preserved; just guard against tiny negative round‑off
    b = np.clip(b, 0.0, None)
    return b


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the sandbox
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1. Determine length and obtain a fallback parent if present
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
        parent = np.clip(parent, 0.0, 1000.0)
        parent = _normalize(parent)
    except Exception:
        n = 256
        parent = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Start from a good deterministic construction (Catalan)
    # ------------------------------------------------------------------
    best = _normalize(_catalan_weights(n))
    best_score = _score(best)

    # also try the provided parent
    parent_score = _score(parent)
    if parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # ------------------------------------------------------------------
    # 3. Random parametric search (power‑law + optional exponential tail)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - 2.0:
        p = rng.uniform(0.3, 1.2)          # power‑law exponent
        c = rng.uniform(0.1, 5.0)          # offset to avoid zero division
        lam = rng.uniform(0.0, 0.005)      # exponential cutoff
        idx = np.arange(n, dtype=np.float64)
        a = (idx + c) ** (-p)
        if lam > 0.0:
            a *= np.exp(-lam * idx)
        a = _normalize(a)

        sc = _score(a)
        if sc < best_score:
            best = a.copy()
            best_score = sc
            if best_score <= 1.50274349:
                break

        # a few cheap local “flattening” moves from the current best
        for _ in range(2):
            if time.monotonic() >= deadline - 1.0:
                break
            cand = _flatten_step(best, rng)
            sc2 = _score(cand)
            if sc2 < best_score:
                best = cand
                best_score = sc2
                if best_score <= 1.50274349:
                    break

    # ------------------------------------------------------------------
    # 4. Final refinement with cheap stochastic moves
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - 0.5:
        # pairwise mass transfer
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] > 0.0:
            delta = best[i] * rng.random() * 0.2
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                if best_score <= 1.50274349:
                    break

        # occasional multiplicative jitter (helps escape plateaus)
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = _normalize(best * jitter)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                if best_score <= 1.50274349:
                    break

    # ------------------------------------------------------------------
    # 5. Return the best sequence found
    # ------------------------------------------------------------------
    return [float(v) for v in best]
