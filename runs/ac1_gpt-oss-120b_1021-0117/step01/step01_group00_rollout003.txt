                    i_light = rng.integers(0, n)
        else:
            i_light = rng.integers(0, n)
            while i_light == i_heavy:
                i_light = rng.integers(0, n)

        # propose a swap
        cand = current.copy()
        cand[i_heavy], cand[i_light] = cand[i_light], cand[i_heavy]
        cand_score = evaluate_sequence(cand.tolist())

        if np.isfinite(cand_score):
            improve = cand_score < current_score
            accept = improve or rng.random() < math.exp((current_score - cand_score) / T)
            if accept:
                current = cand
                current_score = cand_score
                if cand_score < best_score:
                    best_seq = cand.copy()
                    best_score = cand_score

        # occasional completely random swap to keep diversity
        if rng.random() < 0.005:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                cand2 = current.copy()
                cand2[i], cand2[j] = cand2[j], cand2[i]
                cand2_score = evaluate_sequence(cand2.tolist())
                if np.isfinite(cand2_score):
                    improve2 = cand2_score < current_score
                    accept2 = improve2 or rng.random() < math.exp((current_score - cand2_score) / T)
                    if accept2:
                        current = cand2
                        current_score = cand2_score
                        if cand2_score < best_score:
                            best_seq = cand2.copy()
                            best_score = cand2_score

        # cooling
        T *= decay

        # tiny safety break – leave a half‑second to return the answer
        if time.monotonic() + 0.5 > deadline:
            break

    # ----------------------------------------------------------------------
    # 6. Return the best sequence found (as a plain Python list).
    # ----------------------------------------------------------------------
    return [float(x) for x in best_seq]
