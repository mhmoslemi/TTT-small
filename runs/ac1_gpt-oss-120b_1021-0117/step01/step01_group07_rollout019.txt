def _catalan_weights(n: int) -> np.ndarray:
    """Return the first n Catalan‑style weights C(2i,i)/4**i."""

            if best_score <= TARGET:
                break
        else:
            no_improve_iters += 1
            if no_improve_iters >= MAX_NO_IMPROVE:
                # Large jitter to shake the state.
                jitter = np.exp(rng.normal(0.0, 0.10, size=n))
                a = a * jitter
                a /= a.sum()
                no_improve_iters = 0

    # ------------------------------------------------------------------
    # 3.  Return the best sequence found (plain Python list).
    # ------------------------------------------------------------------
    return [float(v) for v in best]
