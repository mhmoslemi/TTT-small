import numpy as np, time, math

def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    n = a.size
    b = _conv_self(a)
    M = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * M / (s * s)

def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # initialise
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = _sqrt_series(5582)
    a0 = np.clip(a0, 0.0, 1000.0)
    total = a0.sum()
    if total == 0.0:
        a0[0] = 1.0
        total = 1.0
    a = a0 / total                     # work on the probability simplex

    n = a.size
    conv = _conv_self(a)
    M = conv.max()
    cur_score = 2.0 * n * M
    best = a.copy()
    best_score = cur_score

    eta = 0.3                          # initial step size
    iter_cnt = 0
    while time.monotonic() < deadline:
        # --- gradient of the current max convolution ---
        max_mask = conv >= M - 1e-12
        max_idx = np.nonzero(max_mask)[0]
        grad = np.zeros_like(a)
        for k in max_idx:
            i0 = max(0, k - (n - 1))
            i1 = min(k, n - 1)
            idx = np.arange(i0, i1 + 1)
            grad[idx] += a[k - idx]

        # --- additive projected gradient step ---
        cand = a - eta * grad
        np.clip(cand, 0.0, None, out=cand)
        s = cand.sum()
        if s > 0.0:
            cand /= s

        # occasional jitter to keep diversity
        if iter_cnt % 250 == 0:
            jitter = rng.normal(0.0, 0.004, size=n)
            cand = np.clip(cand + jitter, 0.0, None)
            cand_sum = cand.sum()
            if cand_sum > 0.0:
                cand /= cand_sum

        # evaluate candidate
        conv_cand = _conv_self(cand)
        M_cand = conv_cand.max()
        cand_score = 2.0 * n * M_cand

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # acceptance (simulated annealing)
        if cand_score < cur_score:
            a, conv, M, cur_score = cand, conv_cand, M_cand, cand_score
        else:
            # temperature shrinks linearly with remaining time
            time_left = deadline - time.monotonic()
            total_time = float(budget_s) - 5.0
            temp = max(1e-6, 0.05 * (time_left / total_time))
            if rng.random() < math.exp(-(cand_score - cur_score) / temp):
                a, conv, M, cur_score = cand, conv_cand, M_cand, cand_score

        # decay step size
        eta *= 0.9997
        iter_cnt += 1

    # rescale to a safe magnitude (still <1000)
    if best.max() > 1000.0:
        best *= 1000.0 / best.max()
    return [float(v) for v in best]
