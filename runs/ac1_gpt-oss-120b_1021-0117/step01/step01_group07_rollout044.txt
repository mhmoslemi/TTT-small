def _score(seq: np.ndarray) -> float:
    """Exact AC1 bound for a non‑negative sequence (sum may be any positive)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_conv = conv.max()
    s = seq.sum()
    # guard against division by zero (should never happen for valid seq)
    if s <= 0.0:
        return float('inf')
    return 2.0 * n * max_conv / (s * s)
