def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

import time, numpy as np

def _project_simplex(v):
    # Euclidean projection onto simplex {x>=0, sum=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# helper: Euclidean projection onto the probability simplex

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]

    ind = np.arange(1, n + 1)

            frac = rng.random() * 0.15             # up to 15 % of source mass
            delta = a[src] * frac
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = _project_simplex(cand)           # ensure strict feasibility

        sc = _score(cand)

        # ---------- keep the best ----------
        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
            a = cand
            no_imp = 0
            continue

        # ---------- simulated‑annealing style acceptance ----------
        remaining = max(0.0, deadline - time.monotonic())
        temp = 0.002 * (remaining / float(budget_s)) + 1e-9
        if rng.random() < np.exp(-(sc - best_sc) / temp):
            a = cand

        # ---------- restart if stagnating ----------
        no_imp += 1
        if no_imp > max_no_imp:
            jitter = rng.exponential(scale=0.003, size=n)
            a = best + jitter
            a = _project_simplex(a)
            no_imp = 0

    # --------------------------------------------------------
    # return the best candidate found
    # --------------------------------------------------------
    return [float(x) for x in best]
