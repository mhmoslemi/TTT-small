import numpy as np
import time

def _score(seq: np.ndarray, n: int) -> float:
    """AC1 bound for a non‑negative sequence."""
    s = float(seq.sum())
    if s < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    return float(2.0 * n * np.max(conv) / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ---- initialise -------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        cur = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        n = 256
        cur = np.full(n, 1.0, dtype=np.float64)

    if cur.sum() < 0.01:                     # assure a valid denominator
        cur = cur + (0.01 - cur.sum()) / n

    cur_score = _score(cur, n)
    best_seq = cur.copy()
    best_score = cur_score

    # ---- gradient descent parameters --------------------------------------
    lr = 0.5                     # initial step size
    min_lr = 1e-9
    stagnation = 0               # counts iterations without improvement

    while time.monotonic() < deadline:
        # convolution of the current sequence
        conv = np.convolve(cur, cur)
        k_max = int(np.argmax(conv))
        max_b = float(conv[k_max])
        S = float(cur.sum())
        if S < 0.01:                                 # safety guard
            cur = cur + (0.01 - S) / n
            S = float(cur.sum())

        # gradient of 2*n*max_b / S^2  (only the max‑lag contributes)
        fac1 = 2.0 * n / (S * S)
        fac2 = 4.0 * n * max_b / (S * S * S)

        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        paired = np.zeros_like(cur)
        paired[mask] = cur[partner[mask]]

        grad = fac1 * paired - fac2 * cur

        # propose a move
        cand = cur - lr * grad
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _score(cand, n)

        if cand_score < cur_score:                     # accept improvement
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
            lr = min(lr * 1.05, 5.0)                    # a bit more aggressive
            stagnation = 0
        else:                                          # no improvement – shrink step
            lr = max(lr * 0.5, min_lr)
            stagnation += 1

        # occasional random reshuffle to escape plateaus
        if stagnation > 4000:
            perm = rng.permutation(n)
            cur = cur[perm]
            cur_score = _score(cur, n)
            stagnation = 0
            lr = 0.5

    # -----------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    return [float(v) for v in best_seq]
