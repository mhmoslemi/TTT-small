def conv_fft(seq):
    n = seq.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    return conv

def move_peak(seq, conv):
    n = seq.shape[0]
    k_max = int(np.argmax(conv))
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    # compute contributions products list
    # contributions = seq[lo:hi+1] * seq[k_max - np.arange(lo, hi+1)]
    idxs = np.arange(lo, hi+1)
    prod = seq[idxs] * seq[k_max - idxs]
    # Select i index with max product
    i = idxs[np.argmax(prod)]
    j = k_max - i
    # Determine fraction alpha between 0.05 and 0.15
    alpha = np.random.random() * 0.10 + 0.05

            t = np.random.randint(0, n)
        new_seq[t] += total_removed
    # Clip
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)  # reserve slight buffer

    # Determine n, parent
    n=3809
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim ==1 and arr.size>0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
        else:
            parent = None
    except Exception:
        parent = None

    def initial_binomial(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1,n):
            a[i] = a[i-1] * (2*i -1) / (2*i)
        return a

    if parent is not None:
        best_seq = parent.copy()
    else:
        best_seq = initial_binomial(n)

    # compute score and conv
    def fast_score(seq):
        n = seq.shape[0]
        m = 1 << ((2*n -1).bit_length())
        f = np.fft.rfft(seq, n=m)
        conv = np.fft.irfft(f * f, n=m)[:2*n-1]
        max_val = conv.max()
        s = seq.sum()
        if s <= 0.0:
            return float('inf'), conv, max_val
        score = 2.0 * n * max_val / (s * s)
        return float(score), conv, max_val

    best_score, best_conv, _ = fast_score(best_seq)

    # Ensure best_seq is not trivial
    # If parent present and perhaps worse than binomial, compare
    if parent is not None:
        bin_seq = initial_binomial(n)
        bin_score, _, _ = fast_score(bin_seq)
        if bin_score < best_score:
            best_seq = bin_seq
            best_score = bin_score
            best_conv, _ = best_conv, _  # compute conv again later
            best_score, best_conv, _ = fast_score(best_seq)

    # current state for SA
    cur_seq = best_seq.copy()
    cur_score = best_score
    cur_conv = best_conv

    start_time = time.monotonic()
    # SA parameters
    initial_temp = 0.05 * best_score
    final_temp = 0.001 * best_score

    # move functions
    def move_peak(seq, conv):
        n = seq.shape[0]
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idxs = np.arange(lo, hi+1)
        # product contributions
        prod = seq[idxs] * seq[k - idxs]
        i = idxs[np.argmax(prod)]
        j = k - i
        # fraction
        alpha = rng.random() * 0.10 + 0.05   # 0.05-0.15
        new_seq = seq.copy()
        if i == j:
            delta = new_seq[i] * alpha
            new_seq[i] -= delta
            total_removed = delta
        else:
            delta_i = new_seq[i] * alpha
            delta_j = new_seq[j] * alpha
            new_seq[i] -= delta_i
            new_seq[j] -= delta_j
            total_removed = delta_i + delta_j
        # target index
        t = rng.integers(0, n)
        while t == i or t == j:
            t = rng.integers(0, n)
        new_seq[t] += total_removed
        np.clip(new_seq, 0.0, 1000.0, out=new_seq)
        return new_seq

    def move_block_scale(seq):
        n = seq.shape[0]
        width = rng.integers(1, max(2, n // 50))  # smaller blocks
        start = rng.integers(0, n - width + 1)
        factor = math.exp(rng.normal(0.0, 0.06))
        new_seq = seq.copy()
        new_seq[start:start+width] *= factor
        np.clip(new_seq, 0.0, 1000.0, out=new_seq)
        return new_seq

    def move_smooth(seq):
        n = seq.shape[0]
        i = rng.integers(0, n-1)
        avg = (seq[i] + seq[i+1]) * 0.5
        new_seq = seq.copy()
        new_seq[i] = avg
        new_seq[i+1] = avg
        return new_seq

    def move_single_scale(seq):
        n = seq.shape[0]
        idx = rng.integers(0, n)
        factor = math.exp(rng.normal(0.0, 0.25))
        new_seq = seq.copy()
        new_seq[idx] *= factor
        np.clip(new_seq, 0.0, 1000.0, out=new_seq)
        return new_seq

    def move_swap(seq):
        n = seq.shape[0]
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        # swap a fraction of mass
        frac = rng.random() * 0.1 + 0.05
        delta = min(seq[i], seq[j]) * frac
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta  # moving from i to j; keep sum constant
        # also optionally move opposite direction?
        # We'll also move small amount back for symmetry:
        # Actually swap similar: take delta2 from j to i:
        # but we can keep as simple transfer.
        np.clip(new_seq, 0.0, 1000.0, out=new_seq)
        return new_seq

    move_funcs = [move_peak, move_block_scale, move_smooth, move_single_scale, move_swap]

    # main loop
    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        # maybe recompute cur_conv if we changed cur_seq without conv value
        # but we have cur_conv from current state; we need ensure it's up-to-date.
        # We'll recompute conv only if move type uses it (peak). To simplify,
        # compute conv for cur_seq once per iteration before picking move.
        cur_score, cur_conv, _ = fast_score(cur_seq)  # compute again; cheap

        # Choose move type
        op = rng.integers(0, len(move_funcs))
        # generate candidate
        if op == 0:
            cand_seq = move_funcs[op](cur_seq, cur_conv)
        else:
            cand_seq = move_funcs[op](cur_seq)

        # Evaluate
        cand_score, cand_conv, _ = fast_score(cand_seq)

        # temperature schedule
        elapsed = now - start_time
        total_allowed = deadline - start_time
        fraction = elapsed / total_allowed if total_allowed > 0 else 1.0
        temp = initial_temp * (1 - fraction) + final_temp * fraction

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability:
            if temp > 0:
                prob = math.exp(-(cand_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_seq = cand_seq
            cur_score = cand_score
            cur_conv = cand_conv

        # update best overall
        if cand_score < best_score:
            best_seq = cand_seq
            best_score = cand_score
            best_conv = cand_conv

    # Return best as list of float
    return [float(x) for x in best_seq]

def move_peak(seq, conv, rng):
    n = seq.shape[0]
    k = int(np.argmax(conv))
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idxs = np.arange(lo, hi+1)
    prod = seq[idxs] * seq[k - idxs]
    i = idxs[np.argmax(prod)]
    j = k - i
    alpha = rng.random() * 0.15 + 0.05 # 0.05-0.20
    new_seq = seq.copy()
    if i == j:
        delta = new_seq[i] * alpha
        new_seq[i] -= delta
        total_removed = delta
    else:
        delta_i = new_seq[i] * alpha
        delta_j = new_seq[j] * alpha
        new_seq[i] -= delta_i
        new_seq[j] -= delta_j
        total_removed = delta_i + delta_j
    # target index
    t = rng.integers(0, n)
    while t == i or t == j:
        t = rng.integers(0, n)
    new_seq[t] += total_removed
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

    n = seq.shape[0]

        delta = out[i] * alpha
        out[i] -= delta
        total = delta
    else:
        delta_i = out[i] * alpha
        delta_j = out[j] * alpha
        out[i] -= delta_i
        out[j] -= delta_j
        total = delta_i + delta_j
    t = rng.integers(0, n)
    while t == i or t == j:
        t = rng.integers(0, n)
    out[t] += total
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _move_block_scale(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = seq.shape[0]
    width = rng.integers(1, max(2, n // 50))
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.06))
    out = seq.copy()
    out[start:start + width] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _move_smooth(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Replace two neighbours by their average (preserves sum)."""
    n = seq.shape[0]
    i = rng.integers(0, n - 1)
    avg = (seq[i] + seq[i + 1]) * 0.5
    out = seq.copy()
    out[i] = avg
    out[i + 1] = avg
    return out


def _move_single_scale(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a single entry by a log‑normal factor."""
    n = seq.shape[0]
    idx = rng.integers(0, n)
    factor = math.exp(rng.normal(0.0, 0.25))
    out = seq.copy()
    out[idx] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _move_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a small fraction of mass from one index to another."""
    n = seq.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    frac = rng.random() * 0.15 + 0.05
    delta = seq[i] * frac
    out = seq.copy()
    out[i] -= delta
    out[j] += delta
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step‑function that lowers the AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine whether a parent sequence is supplied.
    # ------------------------------------------------------------------
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Initialise best (and current) candidate.
    # ------------------------------------------------------------------
    if parent is not None:
        best_seq = parent.copy()
    else:
        best_seq = _initial_binomial(n)

    best_score, best_conv, _ = _fast_score(best_seq)

    # In case the binomial seed is actually better, switch to it.
    if parent is not None:
        bin_seq = _initial_binomial(n)
        bin_score, _, _ = _fast_score(bin_seq)
        if bin_score < best_score:
            best_seq = bin_seq
            best_score = bin_score
            best_conv, _ = best_conv, _  # placeholder
            best_score, best_conv, _ = _fast_score(best_seq)

    cur_seq = best_seq.copy()
    cur_score = best_score
    cur_conv = best_conv

    start_time = time.monotonic()
    init_temp = 0.05 * best_score
    final_temp = 0.001 * best_score
    stagnant = 0

    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        # --------------------------------------------------------------
        # Choose a move (peak‑focused with higher probability).
        # --------------------------------------------------------------
        r = rng.random()
        if r < 0.45:
            cand_seq = _move_peak(cur_seq, cur_conv, rng)
        elif r < 0.55:
            cand_seq = _move_block_scale(cur_seq, rng)
        elif r < 0.70:
            cand_seq = _move_smooth(cur_seq, rng)
        elif r < 0.85:
            cand_seq = _move_single_scale(cur_seq, rng)
        else:
            cand_seq = _move_swap(cur_seq, rng)

        cand_score, cand_conv, _ = _fast_score(cand_seq)
        if not np.isfinite(cand_score):
            continue

        # --------------------------------------------------------------
        # Simulated‑annealing style acceptance.
        # --------------------------------------------------------------
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        temp = init_temp * (1.0 - frac) + final_temp * frac

        accept = False
        if cand_score < cur_score:
            accept = True
            stagnant = 0
        elif temp > 0.0 and rng.random() < math.exp(-(cand_score - cur_score) / temp):
            accept = True
            stagnant = 0
        else:
            stagnant += 1

        if accept:
            cur_seq = cand_seq
            cur_score = cand_score
            cur_conv = cand_conv

        # --------------------------------------------------------------
        # Update the global best if improved.
        # --------------------------------------------------------------
        if cand_score < best_score:
            best_seq = cand_seq
            best_score = cand_score
            best_conv = cand_conv

        # --------------------------------------------------------------
        # Periodic reset to the best found if we have been stagnant.
        # --------------------------------------------------------------
        if stagnant > 2000:
            cur_seq = best_seq.copy()
            cur_score = best_score
            cur_conv = best_conv
            stagnant = 0

    # Return a plain Python list – the evaluator will recompute the exact score.
    return [float(x) for x in best_seq]
