import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities (all top‑level, no closures)

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n‑1 (positive decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_max_and_kmax(a: np.ndarray, n: int, fft_len: int):
    """Return (max value of a*a, index where it occurs) using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_b = float(conv.max())
    k_max = int(conv.argmax())
    return max_b, k_max


def _pair_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Return an array `p` such that p[i] = a[k-i] when 0<=k-i<n,
    otherwise 0.  This is the vector of partners that appear in the
    convolution term at shift k.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    p = np.zeros_like(a)
    if i_end >= i_start:
        idx = np.arange(i_start, i_end + 1)
        p[i_start : i_end + 1] = a[k - idx]
    return p


# ------------------------------------------------------------
# Main candidate generator
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights (floats) whose AC1 upper bound
    is as small as possible within the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # spare a bit for cleanup

    # ------------------------------------------------------------------
    # 1️⃣  Determine problem size and initialise a starting sequence
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        init = np.array(height_sequence_1, dtype=np.float64)
        if np.any(np.isnan(init)) or np.any(np.isinf(init)):
            init = None
    except Exception:
        n = 256
        init = None

    def _fallback_binomial():
        a = _generate_binomial_seq(n)
        a = np.clip(a, 0.0, 1000.0)
        tot = a.sum()
        if tot == 0.0:
            a = np.ones(n, dtype=np.float64) / n
        else:
            a = a / tot
        return a

    if init is None or init.size != n or init.sum() == 0.0:
        cur = _fallback_binomial()
    else:
        cur = np.clip(init, 0.0, 1000.0)
        tot = cur.sum()
        cur = cur / tot                     # normalise sum to 1 (scale‑invariant)

    # ------------------------------------------------------------
    # 2️⃣  Pre‑compute FFT layout
    # ------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------
    # 3️⃣  Scoring helpers (sum is constant -> denominator fixed)
    # ------------------------------------------------------------
    total = cur.sum()
    denom = total * total
    max_b, k_max = _conv_max_and_kmax(cur, n, fft_len)
    cur_score = 2.0 * n * max_b / denom
    best = cur.copy()
    best_score = cur_score

    no_improve = 0  # counter for stagnation

    # ------------------------------------------------------------
    # 4️⃣  Main directed hill‑climbing loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        # recompute convolution maximum and its location
        max_b, k_max = _conv_max_and_kmax(cur, n, fft_len)

        # compute partners for the binding shift
        partner = _pair_for_shift(cur, k_max)
        contrib = cur * partner                     # contribution of each index

        # ---- source: an index that contributes heavily to the peak ----
        if contrib.max() <= 0.0:
            break
        src_thr = contrib.max() * 0.30
        src_cands = np.where(contrib >= src_thr)[0]
        src = rng.choice(src_cands)

        if cur[src] <= 1e-14:
            continue

        # ---- target: an index that participates little (small partner) ----
        max_partner = partner.max()
        if max_partner == 0.0:
            tgt_cands = np.where(partner == 0.0)[0]
        else:
            tgt_thr = max_partner * 0.10
            tgt_cands = np.where(partner <= tgt_thr)[0]
        # avoid picking the source again
        tgt_cands = tgt_cands[tgt_cands != src]

        if tgt_cands.size == 0:
            tgt = rng.integers(0, n)
            while tgt == src:
                tgt = rng.integers(0, n)
        else:
            tgt = rng.choice(tgt_cands)

        # ---- move a modest fraction of mass ----
        frac = rng.uniform(0.02, 0.12)
        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # evaluate candidate
        cand_max_b, _ = _conv_max_and_kmax(cand, n, fft_len)
        cand_score = 2.0 * n * cand_max_b / denom

        # greedy acceptance
        if cand_score < cur_score - 1e-12:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1

        # ----------------------------------------------------------------
        # Periodic random re‑balancing to escape possible plateaus
        # ----------------------------------------------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and cur[i] > 1e-14:
                frac = rng.uniform(0.01, 0.05)
                d = cur[i] * frac
                cur[i] -= d
                cur[j] += d
                # recompute score after the tweak
                max_b, _ = _conv_max_and_kmax(cur, n, fft_len)
                cur_score = 2.0 * n * max_b / denom
                if cur_score < best_score:
                    best = cur.copy()
                    best_score = cur_score
            no_improve = 0

    # ------------------------------------------------------------
    # 5️⃣  Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------
    final = [float(v) for v in best]
    try:
        _ = evaluate_sequence(final)  # safety check – should be finite
    except Exception:                 # in the unlikely event of trouble
        final = [float(v) for v in cur]
    return final
