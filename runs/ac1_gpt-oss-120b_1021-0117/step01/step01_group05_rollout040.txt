    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    except Exception:
        parent = None

    # define generator for binomial
    def initial_binomial(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (2*i -1) / (2*i)
        return a

    if parent is not None:
        best = parent.copy()
    else:
        best = initial_binomial(n)

    best_score, best_conv, _ = _fast_score(best)

    # Attempt LP improvement once if time permits
    if time.monotonic() < deadline - 10.0:
        # compute M
        M = np.max(best_conv)
        # build sparse constraint matrix
        try:
            import scipy.sparse
            import scipy.optimize
            f = best
            # Build A_ub
            col = np.repeat(np.arange(n, dtype=np.int32), n)
            row = col + np.tile(np.arange(n, dtype=np.int32), n)

                cand_score, cand_conv, _ = _fast_score(cand)

                    best_score = cand_score
                    best_conv = cand_conv

import time
import numpy as np
from scipy import sparse, optimize

def _initial_binomial(n: int) -> np.ndarray:
    """Catalan‑like decreasing seed a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Score = 2 n * max(conv(seq,seq)) / sum(seq)^2, using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    b_max = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, b_max
    return float(2.0 * n * b_max / (s * s)), conv, b_max


def _build_A_ub(f: np.ndarray):
    """Sparse matrix for constraints  (f * g)_k <= M."""
    n = f.shape[0]
    # row = i + j , col = i , value = f[j]
    idx = np.arange(n, dtype=np.int32)
    col = np.repeat(idx, n)                     # length n*n
    row = col + np.tile(idx, n)
    data = np.tile(f, n)
    return sparse.csr_matrix((data, (row, col)), shape=(2 * n - 1, n))

    except Exception:
        parent = None

    # ------------------------------------------------------------------
