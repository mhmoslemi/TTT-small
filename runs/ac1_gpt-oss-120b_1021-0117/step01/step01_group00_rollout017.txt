def interleave_extremes(values):
    n = len(values)
    out = np.empty_like(values)
    left = 0
    right = n-1
    for k, val in enumerate(values):
        if k % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out

            perm = rng.permutation(sidon)

            if np.isfinite(perm_score) and perm_score < best_score:
                best_score = perm_score
                best_seq = perm.copy()
                current = perm.copy()
                current_score = perm_score

    # ------------------------------------------------------------------ #
    # 7. Final sanity check using the official evaluator (tiny overhead)
    # ------------------------------------------------------------------ #
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        if not np.isfinite(final_score):
            # fall back to the last known good candidate (uniform)
            best_seq = uniform.copy()
    except Exception:
        best_seq = uniform.copy()

    # Return a plain Python list of floats as required
    return [float(v) for v in best_seq.tolist()]
