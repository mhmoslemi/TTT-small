    if s <= 0:
        raise error (but s positive)
    n = len(v)

    u = np.sort(v)[::-1]  # decreasing

def subgradient_descent(a, max_iters, start_time, deadline):
    best_local = a.copy()
    best_local_score = float('inf')
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        b = np.convolve(a, a)
        M = b.max()
        score = 2 * n * M
        if score < best_local_score:
            best_local_score = score
            best_local = a.copy()
        # compute gradient from active set
        active_mask = (b >= M - 1e-12).astype(np.float64)
        grad = 2.0 * np.correlate(active_mask, a, mode='valid')
        # step size schedule:
        step = 0.5 / np.sqrt(it+1)
        # scaling
        step_scaled = step * (np.linalg.norm(a) / (np.linalg.norm(grad) + 1e-12))
        new_a = a - step_scaled * grad
        a = proj_simplex(new_a, s=1.0)
        # occasional jitter
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = proj_simplex(a * jitter, s=1.0)
    return best_local, best_local_score
```

    except Exception:

    n = parent.shape[0]

    def proj_simplex(v, s=1.0):
        # reference Duchi et al
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        ind = np.arange(1, len(v)+1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # all zeros, return uniform
            return np.full_like(v, s/len(v))
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        # ensure sum exactly s
        w_sum = w.sum()
        if w_sum > 0:
            w = w * (s / w_sum)
        return w

    # initial candidates: normalized parent and Catalan
    def catalan_weights(n):
        w = np.empty(n, dtype=np.float64)
        w[0] = 1.0
        for i in range(1,n):
            w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
        return w

import time
import math
import numpy as np

def _proj_simplex(v, s=1.0):
    """Project onto the simplex {x >= 0, sum(x) = s}."""
    # Duchi et al. (2008) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        w = np.full_like(v, s / len(v), dtype=np.float64)
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w *= (s / w_sum)
    return w


def _catalan_weights(n):
    """First n Catalan‐style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w

        # diminishing step size, scaled to the relative magnitude of a and grad
        step = 0.6 / math.sqrt(it + 1.0)
        step_scaled = step * (norm_a / norm_g)

        cand = a - step_scaled * grad
        cand = _proj_simplex(cand, s=1.0)
        new_M = np.convolve(cand, cand).max()
        return cand, new_M

    # ------------------------------------------------------------------
    # 5.  Main search loop with occasional restarts
    # ------------------------------------------------------------------
    total_iters = 0
    while time.monotonic() < deadline:
        # restart periodically or when we have made little progress
        if total_iters % 2000 == 0:
            a = rng.gamma(1.0, 1.0, size=n)
            a = a / a.sum()
        else:
            a = global_best.copy()

        # perform a short burst of sub‑gradient steps
        max_local = 600
        for it in range(max_local):
            if time.monotonic() >= deadline:
                break

            # current score
            cur_score = _score(a)
            if cur_score < global_best_score:
                global_best = a.copy()
                global_best_score = cur_score
                if global_best_score <= 1.50274349:
                    break

            # deterministic sub‑gradient move
            a_new, M_new = _subgrad_step(a, it + 1)
            new_score = 2.0 * n * M_new

            # accept if it improves
            if new_score < cur_score:
                a = a_new
                continue

            # otherwise try a cheap random mass transfer
            if rng.random() < 0.3:
                i, j = rng.choice(n, size=2, replace=False)
                if a[i] > 0.0:
                    eps = rng.random() * a[i] * 0.4
                    cand = a.copy()
                    cand[i] -= eps
                    cand[j] += eps
                    cand = _proj_simplex(cand, s=1.0)
                    cand_score = _score(cand)
                    if cand_score < cur_score:
                        a = cand
                        continue

            # occasional log‑normal jitter to escape flat plateaux
            if rng.random() < 0.01:
                jitter = np.exp(rng.normal(0.0, 0.02, size=n))
                a = _proj_simplex(a * jitter, s=1.0)

        total_iters += max_local
        if global_best_score <= 1.50274349:
            break

    # ------------------------------------------------------------------
    # 6.  Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in global_best]
