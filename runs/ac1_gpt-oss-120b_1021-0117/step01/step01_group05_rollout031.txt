    # compute conv to find max index
    conv = current_conv (passed)
    k = np.argmax(conv)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    # compute contributions
    i_arr = np.arange(lo, hi + 1)
    j_arr = k - i_arr
    prod = seq[i_arr] * seq[j_arr]

    idx = np.argmax(prod)
    i = i_arr[idx]
    j = j_arr[idx]
    if i == j:

        alpha = rng.uniform(0.04, 0.12)
        delta = seq[i] * alpha
        seq[i] -= delta

        seq[tgt] += delta
    else:

def fast_score(seq):
    n = seq.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    s = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(s * s, n=m)[:2*n-1]
    max_val = np.max(conv)
    total = np.sum(seq)
    if total <= 0:
        return np.inf, conv
    score = 2.0 * n * max_val / (total*total)
    return float(score), conv
```

def fast_score(seq, fft_len=None):
    n = seq.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=fft_len)
    conv_full = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    max_val = np.max(conv_full)
    total = np.sum(seq)
    if total <= 0.0:
        return float('inf'), conv_full
    score = 2.0 * n * max_val / (total * total)
    return float(score), conv_full
```

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a decreasing positive sequence a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)

def _fast_score(seq: np.ndarray, fft_len: int = None):
    """Return (score, full convolution) using a real FFT."""
    n = seq.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    max_val = np.max(conv)
    total = np.sum(seq)
    if total <= 0.0:
        return float("inf"), conv
    score = 2.0 * n * max_val / (total * total)
    return float(score), conv
