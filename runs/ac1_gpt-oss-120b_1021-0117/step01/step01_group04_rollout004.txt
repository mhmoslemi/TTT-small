def _conv_and_score(x, n, L):
    fa = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    s = float(x.sum())
    if s < 1e-12:
        return max_val, max_idx, float('inf')
    score = 2.0 * n * max_val / (s * s)
    return max_val, max_idx, score

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i for i=0..n-1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound using FFT convolution."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _conv_max(arr: np.ndarray, n: int, L: int):
    """Return (max value, index of max) of the linear convolution a*a."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx

        # -------------------------------------------------------------- #
        # 2) Block scaling (as in the reference implementation)
        elif move_type < 0.8:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.09))
            cand = cur.copy()
            cand[start:start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        # -------------------------------------------------------------- #
        # 3) Local smoothing – replace a block by its average
        else:
            width = int(rng.integers(max(2, n // 200), max(3, n // 20)))
            start = int(rng.integers(0, n - width + 1))
            block_mean = cur[start:start + width].mean()
            cand = cur.copy()
            cand[start:start + width] = block_mean
            np.clip(cand, 0.0, 1000.0, out=cand)

        # -------------------------------------------------------------- #
        cand_score = _ac1_score(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:      # early exit on target
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score

        # temperature schedule
        temperature *= decay

        # occasional restart from the best found so far
        if rng.random() < restart_prob:
            cur = best.copy()
            cur_score = best_score
            temperature = 0.2

    # ------------------------------------------------------------------ #
    # Return result as plain Python floats
    return [float(x) for x in best]
