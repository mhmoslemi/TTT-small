import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (fast, O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv

def _fast_score(a: np.ndarray, conv: np.ndarray = None) -> float:
    """Exact score used by the evaluator (uses pre‑computed convolution if given)."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    if conv is None:
        conv = _conv_self(a)
    return 2.0 * a.size * conv.max() / (s * s)

def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i (decreasing, positive). Length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that
    minimizes the AC1 upper bound defined by `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds (leaving a short
    safety margin) and returns the best sequence found.
