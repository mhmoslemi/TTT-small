    # Main optimisation loop.
    # ------------------------------------------------------------------
    no_improve = 0
    while time.monotonic() < deadline:
        # ----- gradient‑driven improvement --------------------------------
        cand = best.copy()
        _, conv, max_val = _fast_score(cand)

        grad = _grad_contrib(cand, conv, max_val)

        impact = cand * grad
        top_k = max(1, n // 200)               # ~1 % of entries
        if top_k > n // 2:
            top_k = n // 2

        top_idx = np.argpartition(impact, -top_k)[-top_k:]
        low_idx = np.argpartition(cand, top_k)[:top_k]

        # adaptive fraction (larger early, shrinks later)
        elapsed = time.monotonic() - start
        total_time = deadline - start
        alpha = 0.08 * (1.0 - elapsed / total_time) + 0.02   # in [0.02,0.10]

        total_removed = 0.0
        for i in top_idx:
            dec = cand[i] * alpha
            cand[i] -= dec
            total_removed += dec

        if low_idx.size:
            inc = total_removed / low_idx.size
            cand[low_idx] += inc

        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score, cand_conv, _ = _fast_score(cand)

        if cand_score < best_score:
            best = cand
            best_score = cand_score
            best_conv = cand_conv
            no_improve = 0
            continue
        else:
            no_improve += 1

        # ----- occasional random perturbation when stuck ------------------
        if no_improve >= 5:
            no_improve = 0
            cand = best.copy()
            op = rng.integers(0, 4)

            if op == 0:                               # block scaling
                width = rng.integers(1, max(2, n // 30))
                start_i = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.07))
                cand[start_i:start_i + width] *= factor

            elif op == 1:                             # single entry tweak
                idx = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, 0.25))
                cand[idx] *= factor

            elif op == 2:                             # smooth neighbours
                i = rng.integers(0, n - 1)
                avg = (cand[i] + cand[i + 1]) * 0.5
                cand[i] = avg
                cand[i + 1] = avg

            else:                                     # move mass away from a max site
                _, conv2, max_val2 = _fast_score(cand)
                k_max = int(np.argmax(conv2))
                lo = max(0, k_max - (n - 1))
                hi = min(k_max, n - 1)
                i = rng.integers(lo, hi + 1)
                j = k_max - i
                if i != j:
                    frac = rng.uniform(0.05, 0.15)
                    rem_i = cand[i] * frac
                    rem_j = cand[j] * frac
                    cand[i] -= rem_i
                    cand[j] -= rem_j
                    total = rem_i + rem_j
                    tgt = rng.integers(0, n)
                    while tgt == i or tgt == j:
                        tgt = rng.integers(0, n)
                    cand[tgt] += total

            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score, cand_conv, _ = _fast_score(cand)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                best_conv = cand_conv

    # ------------------------------------------------------------------
    # Return a plain Python list for the evaluator.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
