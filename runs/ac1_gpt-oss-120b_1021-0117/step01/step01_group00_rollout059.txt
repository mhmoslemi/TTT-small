    if np.sum(v) == 0:
        # return uniform
        return np.full_like(v, 1.0 / len(v))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def full_convolution(a):
    # compute convolution a*a via FFT to get length 2n-1
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()  # next power of two >= 2n-1
    # compute using real FFT: embed into complex.
    # Use np.fft.rfft with zero padding.
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

    # compute conv via FFT
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)[:2*n-1]
    k = np.argmax(conv)
    # gradient for conv[k]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    if k % 2 == 0:
        i0 = k // 2
        if 0 <= i0 < n:
            g[i0] = a[i0]  # diag correction
    return k, g, conv[k]

import numpy as np

def _project_simplex(v):
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
