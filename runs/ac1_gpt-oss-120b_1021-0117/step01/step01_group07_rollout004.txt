        best_score = pscore

    N = 1 << ((2*n - 1).bit_length())  # next power of two >= 2n-1

    eta = 0.5  # initial step size

        conv = np.fft.irfft(X * X, n=N)[:2*n-1]
        # due to floating errors, ensure nonnegative
        return np.maximum(conv, 0.0)

    def grad_at_max(x, k):

            grad[:k+1] += x[k::-1]

            grad[k:] += x[:n_loc - k]

            grad[L:] += x[L:][::-1]
        return grad

        b = conv_full(best)

import time, numpy as np

def _project_simplex(v):
    """Project v onto the probability simplex {x>=0, sum=1}."""
    # see Duchi et al. 2008
    # works for arbitrary sum (>=0)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum x_i = 1}.  Uses the O(n log n) algorithm from
    Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – fall back to the uniform distribution
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):

    Search for a non‑negative step‑function (list of heights) that
    minimises the AC1 upper bound defined by ``evaluate_sequence``.
    The method starts from the supplied parent (if any) and then runs a
    projected sub‑gradient descent on the max‑convolution, with occasional
    random jitter to escape plateaus.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise from parent or a uniform fallback
    # ------------------------------------------------------------------
    try:

        if parent.sum() < 1e-12:

        parent_norm = parent / parent.sum()
        n = parent_norm.size
    except Exception:
        n = 256
        parent_norm = np.full(n, 1.0 / n, dtype=np.float64)

    # uniform start
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # pick the better of the two as the initial point
    best = uniform.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    parent_score = float(evaluate_sequence(parent_norm.tolist()))
    if parent_score < best_score:
        best = parent_norm.copy()
        best_score = parent_score

    cur = best.copy()          # current iterate for the sub‑gradient method
    cur_score = best_score

    # pre‑compute FFT length (next power of two >= 2n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop (projected sub‑gradient descent)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    step0 = 0.5                # initial step magnitude
    it = 1
    target = 1.50274349

    while time.monotonic() < deadline:
        # ----- convolution and objective -------------------------------------------------
        A = np.fft.rfft(cur, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[:2 * n - 1]
        # tiny negative rounding errors are harmless for the max
        max_b = conv.max()
        score = 2.0 * n * max_b

        # ----- keep the best valid sequence ---------------------------------------------
        if score < best_score:
            best_score = score
            best = cur.copy()
            if best_score <= target:
                break
            # Restart the walk from the new best point
            cur = best.copy()
            cur_score = best_score

        # ----- sub‑gradient of the max‑convolution ---------------------------------------
        k = int(np.argmax(conv))
        grad = np.zeros_like(cur)

        if k < n:
            # first contribution (mirrored prefix)
            grad[:k + 1] += cur[k::-1]
            # second contribution (suffix)
            grad[k:] += cur[: n - k]
        else:
            # only the mirrored part is present when k ≥ n
            L = k - (n - 1)
            grad[L:] += cur[L:][::-1]

        # ----- projected sub‑gradient step -----------------------------------------------
        eta = step0 / np.sqrt(it)
        it += 1
        nxt = cur - eta * grad
        cur = _project_simplex(nxt)

        # ----- occasional random jitter to shake the walk --------------------------------
        if rng.random() < 0.02:
            # log‑normal multiplicative noise, fairly mild
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            cur = _project_simplex(cur * jitter)

    # ------------------------------------------------------------------
    # 3.  Return the best found sequence as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
