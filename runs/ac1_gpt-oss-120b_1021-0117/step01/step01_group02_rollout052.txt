                conv_tmp[j : j + n] += 2.0 * delta * a
                conv_tmp[i : i + n] -= 2.0 * delta * a
                conv_tmp[2 * j] += delta * delta
                conv_tmp[2 * i] += delta * delta
                conv_tmp[i + j] -= 2.0 * delta * delta
                np.maximum(conv_tmp, 0.0, out=conv_tmp)
                tmp_max = conv_tmp.max()
                if tmp_max < best_max:
                    a = a_tmp
                    conv = conv_tmp
                    best_a = a_tmp.copy()
                    best_max = tmp_max
                    step_factor = max(0.001, step_factor * 0.95)

    # -------------------------------------------------------------------------
    final = np.clip(best_a, 0.0, 1000.0)
    total = final.sum()
    if total < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    else:
        final = final / total                     # keep sum ~=1
    return [float(v) for v in final]
