def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine length
    try:
        n = int(len(height_sequence_1))
        base = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 5000
        base = np.arange(n, 0, -1, dtype=np.float64)   # descending
    # Clip and normalise
    base = np.clip(base, 0.0, 1000.0)
    total = base.sum()
    if total <= 0.0:
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        base = base / total

    # Score function
    def _score(v):
        c = np.convolve(v, v)
        return 2.0 * n * c.max() / (v.sum() ** 2)

    best = base.copy()
    best_score = _score(best)

    # Uniform fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score = _score(uniform)
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score

    # Parameters
    step_factor = 0.2
    min_step = 1e-9
    no_improve = 0
    # try random initialization
    # optionally store a few baseline candidates
    target = 1.50274349

    while time.monotonic() < deadline:
        # Compute convolution and index of max
        conv = np.convolve(best, best)
        k = int(conv.argmax())

        # Compute partner vector
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(best)
        partner[idx] = best[k - idx]

        contrib = best * partner
        src = int(contrib.argmax())
        # dest where partner minimal; break ties randomly
        min_val = partner.min()
        dest_candidates = np.where(partner == min_val)[0]
        if dest_candidates.size == 0:
            dest = rng.integers(0, n)
        else:
            dest = int(rng.choice(dest_candidates))

        if src == dest:
            # simple random perturbation
            width = int(rng.integers(2, max(3, n // 100)))
            start = int(rng.integers(0, n - width + 1))
            stop = start + width
            cand = best.copy()
            factor = np.exp(rng.normal(0.0, 0.15, size=width))
            cand[start:stop] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand * (best.sum() / cand.sum())
            cand_score = _score(cand)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                step_factor = max(0.01, step_factor * 0.99)
                no_improve = 0
            continue

        # Randomly decide move type
        move_type = rng.random()
        if move_type < 0.7:
            # mass transfer

            if delta < min_step:
                delta = min_step
            delta = min(delta, best[src])
            cand = best.copy()
            cand[src] -= delta
            cand[dest] += delta
        elif move_type < 0.9:

            cand[src], cand[dest] = cand[dest], cand[src]
        else:

    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    return [float(v) for v in final]

import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Hill‑climbing / simulated‑annealing search for a low‑AC1 step function."""
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the sandbox
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------ #
    # initialise sequence ------------------------------------------------
    try:
        # height_sequence_1 is provided by the harness if it exists
        base = np.array(height_sequence_1, dtype=np.float64)
        n = int(len(base))
    except Exception:               # fallback when no parent is present
        n = 5000
        base = np.arange(n, 0, -1, dtype=np.float64)   # simple decreasing start

    base = np.clip(base, 0.0, 1000.0)
    s = base.sum()
    if s <= 0.0:
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        base = base / s                     # normalise to L1‑norm 1

    # ------------------------------------------------------------------ #
    # scoring helper ----------------------------------------------------
    def _score(v):
        c = np.convolve(v, v)
        return 2.0 * n * c.max() / (v.sum() ** 2)

    best = base.copy()
    best_score = _score(best)

    # uniform sequence is a cheap baseline (score = 2)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score = _score(uniform)
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score

    # ------------------------------------------------------------------ #
    # optimisation parameters -------------------------------------------
    step_factor = 0.20            # fraction of source mass to move
    min_step = 1e-9
    no_improve = 0
    target = 1.50274349           # stop as soon as we beat the target

    # ------------------------------------------------------------------ #
    while time.monotonic() < deadline:
        # ---- current convolution and its worst index --------------------
        conv = np.convolve(best, best)
        k = int(conv.argmax())                     # shift where max occurs

        # ---- partner values for the worst shift -----------------------
        i_start = max(0, k - (n - 1))
        i_end   = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(best)
        partner[idx] = best[k - idx]

        contrib = best * partner                     # contribution of each entry
        src = int(contrib.argmax())                  # biggest offender

        # destination – index where partner is smallest (often zero)
        min_val = partner.min()
        dest_candidates = np.where(partner == min_val)[0]
        if dest_candidates.size == 0:

                step_factor = max(0.01, step_factor * 0.99)
