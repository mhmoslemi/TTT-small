import time
import math
import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _initial_exponential(n: int) -> np.ndarray:
    """Exponential decay a_i = exp(-i/scale), scaled to sqrt(2n) sum."""
    scale = max(1.0, n / 15.0)
    raw = np.exp(-np.arange(n, dtype=np.float64) / scale)
    target = math.sqrt(2.0 * n)
    return raw * (target / raw.sum())

def _fast_score(seq: np.ndarray):
    """Return (score, full convolution array, index of max entry)."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())   # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_idx
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_idx

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    seeds = [_initial_binomial(n), _initial_exponential(n)]
    if parent is not None:
        seeds.append(parent.copy())
    best = None

    best_conv = None
    best_max_idx = None
    for s in seeds:

        if scr < best_score:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    total_time = deadline - time.monotonic()
    T0 = 0.04  # initial temperature (score units)

    # ------------------------------------------------------------------

    while time.monotonic() < deadline:
