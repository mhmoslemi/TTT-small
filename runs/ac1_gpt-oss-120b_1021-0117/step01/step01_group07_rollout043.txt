import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------
def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style central binomial weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _linear_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution a*a via FFT, length 2n‑1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power‑of‑2
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return c


def _gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient of (a*a)[k] w.r.t. a (ordered‑pair version)."""
    n = a.shape[0]
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def _random_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of mass from i to j (mass‑conserving)."""
    n = a.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    if a[i] == 0.0:
        return a.copy()
    eps = rng.random() * a[i] * 0.5
    cand = a.copy()
    cand[i] -= eps
    cand[j] += eps
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand


def _jitter(a: np.ndarray, rng: np.random.Generator, sigma: float = 0.012) -> np.ndarray:
    """Multiplicative log‑normal jitter, then renormalise."""
    pert = np.exp(rng.normal(0.0, sigma, size=a.shape))
    cand = a * pert
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta density: (i+1)^{α-1} (n-i)^{β-1}, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = n - i + 1.0
    logv = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(j)
    # shift to avoid overflow
    logv -= np.max(logv)
    v = np.exp(logv)
    v /= v.sum()
    return v

    self‑convolution, random mass transfers, jitter and occasional Beta‑family
    exploration, all respecting the time budget.
    """

    score_catalan = float(evaluate_sequence(a_catalan.tolist()))

    if score_parent < score_catalan:
