import time, math
import numpy as np

def _sidon_like_sequence(n):
    """Classic Sidon‑like decreasing sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _fast_conv_max(x, L):
    """Maximum value of the full convolution x*x using FFT (linear, zero‑padded)."""
    fx = np.fft.rfft(x, n=L)
    prod = fx * fx
    conv = np.fft.irfft(prod, n=L)[:2 * x.shape[0] - 1]
    return np.max(conv)
