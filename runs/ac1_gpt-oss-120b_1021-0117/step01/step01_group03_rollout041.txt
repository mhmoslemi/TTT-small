def _conv_self(arr):
    n = arr.size
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(arr, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[:2*n-1]
    return conv

import numpy as np
import time
import math

def _conv_self(arr: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real)."""
    n = arr.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(arr: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = arr.size
    conv = _conv_self(arr)
    max_b = conv.max()
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _build_sparse_high_value(n: int, rng: np.random.Generator,
                             limit_per_sum: int = 4,
                             target_m: int = 200) -> np.ndarray:
    """
    Greedy construction of a set of indices with limited
    ordered‐pair contributions per convolution diagonal.
    Each chosen index receives value 1000.
    """
    max_sum = 2 * n - 1
    counts = np.zeros(max_sum, dtype=np.int16)   # ordered‑pair counts per diagonal
    pos = []
    # random order of candidate indices
    order = rng.permutation(n)
    for i in order:
        # self‑pair contribution (i,i) adds 1 to diagonal 2*i
        s_self = 2 * i
        if counts[s_self] + 1 > limit_per_sum:
            continue
        ok = True
        # check contributions with existing positions
        for j in pos:
            s = i + j
            if counts[s] + 2 > limit_per_sum:   # ordered pair (i,j) and (j,i)
                ok = False
                break
        if not ok:
            continue
        # accept i
        pos.append(i)
        counts[s_self] += 1
        for j in pos[:-1]:          # all previous positions
            s = i + j
            counts[s] += 2
        if len(pos) >= target_m:
            break
    a = np.zeros(n, dtype=np.float64)
    if pos:
        a[pos] = 1000.0
    # tiny filler to keep the sum safely >0.01 (negligible for the score)
    a += 1e-9
    return a
