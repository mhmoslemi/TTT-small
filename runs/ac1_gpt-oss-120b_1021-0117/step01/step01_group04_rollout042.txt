import numpy as np, time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a[i] = C(2i,i) / 4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_fft(arr: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution arr*arr using zero‑padded FFT (length L)."""
    fa = np.fft.rfft(arr, n=L)
    conv_full = np.fft.irfft(fa * fa, n=L)
    return conv_full[: 2 * arr.size - 1]


def _score_with_conv(arr: np.ndarray, n: int, L: int):
    """Return (score, max_b, conv) for the given array."""
    conv = _conv_fft(arr, L)
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf'), max_b, conv
    score = 2.0 * n * max_b / (s * s)
    return score, max_b, conv

    Search for a non‑negative step function that (approximately) minimizes the
    AC1 upper bound.  A simple simulated‑annealing / targeted‑move scheme is

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #

            raise ValueError

        parent = None
        n = 256                     # a modest fallback size
    L = _next_pow2(2 * n - 1)

        cur = parent.copy()
    else:
        cur = _gen_binomial(n)

    # enforce problem constraints immediately

    best = cur.copy()

    # ------------------------------------------------------------------ #

    temperature = 1.0

    while time.monotonic() < deadline:

        if rng.random() < 0.70:

            #  targeted move: look at the index producing the current    #
            #  maximal convolution value and shift mass away from the    #
            #  largest contributing pair.                               #
            # ---------------------------------------------------------- #
            max_k = int(np.argmax(cur_conv))
            idx = np.arange(n)
            partner = max_k - idx
            mask = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(cur)
            contrib[mask] = cur[mask] * cur[partner[mask]]
            i_max = int(np.argmax(contrib))

            # pick a low‑value index that is not i_max
            # adding a large constant to i_max guarantees it is not chosen
            i_min = int(np.argmin(cur + (idx == i_max) * 1e6))

            frac = rng.random() * 0.30 + 0.10          # 10‑40 % transfer
