```python
import numpy as np
import time, math

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _gen_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _fast_score(arr, n, L):
    # compute score using FFT, arr is 1D numpy array
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
    M = conv.max()
    s = arr.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * M / (s * s)
```

import numpy as np

import math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1 (classic binomial sequence)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _fast_score(arr: np.ndarray, n: int, L: int) -> float:
    """FFT‑based AC1 upper bound (no scaling performed)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    M = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * M / (s * s)
