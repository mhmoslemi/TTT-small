def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,float(budget_s)-1.5)
    # Determine n
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim==1 and parent_arr.size>0 and np.isfinite(parent_arr).all():
            n = parent_arr.size
            have_parent = True
        else:
            raise Exception()
    except Exception:
        parent_arr = None
        have_parent = False
        n = 3809
    # define helper functions _initial_binomial, _fast_score etc.

    # Candidate generation functions
    def shape_binomial():
        return _initial_binomial(n)
    def shape_power(p):
        i = np.arange(n, dtype=np.float64)+1
        a = i**(-p)
        # Clip to 0-1000 maybe but values small
        return a
    def shape_exponential(lam):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-lam*i)
        return a
    def shape_triangular():
        a = 1.0 - np.arange(n, dtype=np.float64)/(n-1)
        a = np.maximum(a,0)
        return a
    # evaluate each shape
    shapes = []
    shapes.append(shape_binomial())
    # add parent if available
    if have_parent:
        shapes.append(np.clip(parent_arr,0.0,1000.0))
    # power exponents
    for p in (0.3,0.4,0.5,0.6):
        shapes.append(shape_power(p))
    # exponential
    for lam in (0.0005,0.001,0.002):
        shapes.append(shape_exponential(lam))
    # triangular
    shapes.append(shape_triangular())

    best = None; best_score = np.inf; best_conv = None
    for seq in shapes:
        seq = np.clip(seq,0.0,1000.0)
        score,_conv,_max = _fast_score(seq)
        if score < best_score:
            best = seq.copy()
            best_score = score
            best_conv = _conv
    # Main loop
    while time.monotonic() < deadline:
        cand = best.copy()
        op = rng.integers(0,5)
        if op == 0:
            # block scaling
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            cand[start:start+width] *= factor
        elif op == 1:
            # single index
            idx = rng.integers(0,n)
            factor = np.exp(rng.normal(0.0, 0.25))
            cand[idx] *= factor
        elif op == 2:
            # mass move from max pair
            k_max = int(np.argmax(best_conv))
            lo = max(0, k_max - (n-1))
            hi = min(k_max, n-1)
            # if range length >0
            i_range = np.arange(lo, hi+1)
            j_range = k_max - i_range
            # compute contributions
            contrib = cand[i_range] * cand[j_range]
            # pick index with max contribution
            max_idx = np.argmax(contrib)
            i = i_range[max_idx]
            j = j_range[max_idx]
            # decide fraction
            alpha = rng.uniform(0.02, 0.09)
            # compute removed mass
            rem_i = cand[i] * alpha
            rem_j = cand[j] * alpha if j != i else 0.0
            cand[i] -= rem_i
            if j != i:
                cand[j] -= rem_j
            transferred = rem_i + rem_j
            # choose target far from i,j
            tgt = rng.integers(0,n)
            # ensure not i/j
            while tgt == i or tgt == j:
                tgt = rng.integers(0,n)
            cand[tgt] += transferred
        elif op == 3:
            # smoothing two neighbours
            i = rng.integers(0, n-1)
            avg = (cand[i] + cand[i+1]) * 0.5
            cand[i] = avg
            cand[i+1] = avg
        else:
            # redistribution from a high idx to a low idx
            # pick source weighted by value
            probs = cand / (cand.sum() + 1e-12)
            src = rng.choice(n, p=probs)
            # pick target inversely weighted
            inv = 1.0 / (cand + 1e-6)
            inv_probs = inv / inv.sum()
            tgt = rng.choice(n, p=inv_probs)
            if src == tgt:
                continue
            factor = rng.uniform(0.03,0.07)
            delta = cand[src] * factor
            cand[src] -= delta
            cand[tgt] += delta

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score, cand_conv, _ = _fast_score(cand)
        # Simulated annealing acceptance
        if cand_score < best_score or rng.random() < math.exp(-(cand_score - best_score)/(0.01 + (deadline - time.monotonic()))):
            best = cand
            best_score = cand_score
            best_conv = cand_conv

    return [float(x) for x in best]

import numpy as np
import time
import math

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _shape_power(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}  (decreasing power law)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)

def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)

    return np.maximum(1.0 - np.arange(n, dtype=np.float64) / (n - 1), 0.0)
