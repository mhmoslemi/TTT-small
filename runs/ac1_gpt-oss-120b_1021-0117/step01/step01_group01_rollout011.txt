import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, O(n log n) and memory‑linear)
# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def _score_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast proxy for evaluate_sequence (identical for non‑negative inputs)."""
    max_b = _conv_max_fft(a, fft_len)
    s = a.sum()
    # s is never zero in our usage
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------
def _generate_binomial_seq(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i (decreasing, positive)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2.0 * i + 1.0) / (2.0 * i + 2.0)
    return a


def _generate_exponential_seq(alpha: float, idx: np.ndarray) -> np.ndarray:
    """a_i = exp(-alpha * i)."""
    return np.exp(-alpha * idx)


def _generate_powerlaw_seq(p: float, idx: np.ndarray) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    return (idx + 1.0) ** (-p)


# ----------------------------------------------------------------------

    if score_bin < best_score:
