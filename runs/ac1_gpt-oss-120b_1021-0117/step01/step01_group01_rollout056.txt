import time
import numpy as np

def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max_and_idx(a: np.ndarray, fft_len: int, conv_len: int):
    """Return (max value, index of max, full conv array) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv
