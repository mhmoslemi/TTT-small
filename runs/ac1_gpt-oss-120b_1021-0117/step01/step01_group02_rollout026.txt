    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)   # leave 1 sec for return

        init_arr = np.array(height_sequence_1, dtype=np.float64)
    except Exception:

        n = 5000
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (1 - 1/(2*i))
        init_arr = a
    else:
        n = init_arr.size

    total = init_arr.sum()
    if total < 0.01:
        init_arr = np.full(n, 1.0/n)
        total = 1.0
    init_arr = init_arr / total   # sum=1

    a = init_arr.copy()
    best = a.copy()
    best_score = compute_score(best, n)
    # maybe keep uniform baseline as fallback
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    uniform_score = compute_score(uniform, n)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # parameters
    step_factor = 0.2
    min_step = 1e-12
    no_improve_counter = 0
    shake_interval = 500
    iteration = 0

    L = 1 << ((2*n-1).bit_length())
    # pre-allocate FFT array maybe reused? Not needed.

    while time.monotonic() < deadline:
        # compute convolution
        A = np.fft.rfft(a, n=L)
        C = A * A
        conv = np.fft.irfft(C, n=L)[:2*n-1]
        k = int(conv.argmax())
        conv_max = conv[k]

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end+1, dtype=np.int64)
        partner = a[k - idx]

        product = a[idx] * partner

        # compute product_all = a * reversed? Actually we want small product for dest to reduce impact.
        # We can pick dest as index with smallest a; if a small, moving mass there will not increase max conv much.

        if src == dest:

            low_vals = np.where(a <= np.percentile(a, 10))[0]
            if low_vals.size > 0:
                dest = int(rng.choice(low_vals))
            else:
                dest = int(rng.integers(n))

        if src == dest:
            # if still same, skip iteration
            continue

        if delta < min_step:
            delta = min_step

        A2 = np.fft.rfft(a_new, n=L)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----- initialise ---------------------------------------------------------
    try:
        a0 = np.array(height_sequence_1, dtype=np.float64)  # provided parent
        n = a0.size
    except Exception:                     # no parent – fall back to a simple shape
        n = 5000
        a0 = np.empty(n, dtype=np.float64)
        a0[0] = 1.0
        for i in range(1, n):
            a0[i] = a0[i - 1] * (1.0 - 1.0 / (2.0 * i))

    a0 = np.clip(a0, 0.0, 1000.0)
    if a0.sum() < 0.01:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    a0 = a0 / a0.sum()                 # normalise to sum == 1
    a = a0.copy()

    # ----- helper -------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    def _score(vec: np.ndarray) -> float:
        """compute the AC1 upper bound for a vector (sum is assumed > 0)"""
        A = np.fft.rfft(vec, n=L)
        C = A * A
        conv = np.fft.irfft(C, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max() / (vec.sum() ** 2)

    # start from the better of the parent and the uniform vector
    best = a.copy()
    best_score = _score(best)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score = _score(uniform)
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score
        a = best.copy()

    # ----- optimisation parameters --------------------------------------------
    T = 0.01               # initial temperature for simulated‑annealing acceptance
    T_min = 1e-6
    decay = 0.99995        # geometric cooling
    step_factor = 0.25
    min_step = 1e-12
    it = 0

    # current state score (used for Metropolis acceptance)
    cur_score = best_score

    while time.monotonic() < deadline:
        # ---- linear convolution of the current vector ------------------------
        A_fft = np.fft.rfft(a, n=L)
        C_fft = A_fft * A_fft
        conv_full = np.fft.irfft(C_fft, n=L)[: 2 * n - 1]
        k = int(conv_full.argmax())                # index giving the current max
        cur_score = 2.0 * n * conv_full.max() / (a.sum() ** 2)

        # ---- deterministic move: pull mass from the biggest contributor -----
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = a[k - idx]                       # a_{k-i} for each i
        contrib = a[idx] * partner                  # contribution of each i to conv[k]
        src = int(idx[np.argmax(contrib)])          # source index (biggest contribution)

        dest = int(np.argmin(a))                    # destination – smallest coefficient
        if dest == src:
            # choose a different low‑value index (fallback)
            low = np.where(a <= np.percentile(a, 5))[0]
            if low.size:
                dest = int(rng.choice(low))
            else:
                dest = (src + 1) % n

        delta = a[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > a[src]:
            delta = a[src]

        a_det = a.copy()
        a_det[src] -= delta
        a_det[dest] += delta

        # ---- random move (occasionally) ------------------------------------
        if rng.random() < 0.2:
            src_r = int(rng.integers(n))
            while a[src_r] <= min_step:
                src_r = int(rng.integers(n))
            dest_r = int(rng.integers(n))
            while dest_r == src_r:
                dest_r = int(rng.integers(n))
            frac = rng.random() * 0.5
            delta_r = a[src_r] * frac
            a_rand = a.copy()
            a_rand[src_r] -= delta_r
            a_rand[dest_r] += delta_r
            cand = a_rand
        else:
            cand = a_det

        # ---- evaluate candidate ---------------------------------------------
        cand_score = _score(cand)

        # ---- Metropolis acceptance -------------------------------------------
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                # tighten temperature a bit when we find a new best
                T = max(T_min, T * 0.95)

        # ---- cooling ---------------------------------------------------------
        T = max(T_min, T * decay)

        # ---- occasional global shake -----------------------------------------
        if it % 2000 == 0 and it > 0:
            width = rng.integers(max(2, n // 300), max(10, n // 30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            shaken = a.copy()
            shaken[start:start + width] *= factor
            shaken = np.clip(shaken, 0.0, 1000.0)
            shaken /= shaken.sum()               # keep sum == 1
            shake_score = _score(shaken)
            if shake_score < best_score:
                best = shaken.copy()
                best_score = shake_score
                a = shaken.copy()
                cur_score = shake_score
                T = 0.01                          # restart temperature

        it += 1
        if best_score < 1.50274349:               # target met – stop early
            break

    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
