import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level, no closures)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, N: int, conv_len: int) -> np.ndarray:
    """Convolution a*a using a real FFT (size N >= 2·len(a)‑1)."""
    fa = np.fft.rfft(a, N)
    conv_full = np.fft.irfft(fa * fa, N)
    return conv_full[:conv_len]


def _objective(a: np.ndarray, n: int, N: int, conv_len: int) -> float:
    """Return the AC1 upper‑bound for a (the same quantity used by evaluate_sequence)."""
    b = _conv_fft(a, N, conv_len)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) whose AC1 bound
    is as low as possible within the allotted time.
    The routine builds many random sparse vectors (few hundred spikes)
    and keeps the best one.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # 1.  Base length and a safe fallback (uniform or the supplied parent)
    # ------------------------------------------------------------------

        if parent.ndim != 1 or parent.size == 0:
