def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # compute conv max via FFT
    n2 = 2*n - 1
    N = 1 << (n2 - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:n2]
    maxb = conv.max()
    return 2.0 * n * maxb / (s*s)

def compute_conv_fft(a):
    n = a.shape[0]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-0.5)

    # Get parent if available
    try:
        n = int(len(height_sequence_1))
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Determine base shape (Sidon)
    base_seq = _sidon_like_sequence(n)
    base_seq = np.clip(base_seq, 0.0, 1000.0)

    # Evaluate base seq
    base_score = compute_score_fast(base_seq)
    if base_score < best_score:
        best_seq = base_seq.copy()
        best_score = base_score

    # Track sum constant
    sum_base = base_seq.sum()
    # Precompute FFT length
    L = 2*n - 1
    NFFT = 1 << (L - 1).bit_length()

    # Define helper to compute conv max quickly
    def conv_max(a):
        A = np.fft.rfft(a, n=NFFT)
        conv = np.fft.irfft(A * A, n=NFFT)[:L]
        return conv.max(), conv

    # Search loop
    iteration = 0
    # We'll interleave permutation trials and continuous moves.
    while time.monotonic() < deadline:
        # decide which mode
        if rng.random() < 0.6:  # use permutation search most of time
            perm = rng.permutation(n)
            cand = base_seq[perm]
            cand_max, _ = conv_max(cand)
            cand_sum = cand.sum()
            cand_score = 2.0 * n * cand_max / (cand_sum * cand_sum)
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
        else:
            # local continuous move search on current best sequence
            # we use a copy of best_seq as current for moves.
            # To avoid resetting each iteration, maintain a current variable.
            # We'll maintain separate current; init if not defined.
            if iteration == 0:
                current = best_seq.copy()
                current_score = best_score
                current_conv = None
                # compute conv array for current once
                _, current_conv = conv_max(current)
            # propose move
            # choose donor via targeted or random
            if rng.random() < 0.2:
                # targeted
                kmax = int(np.argmax(current_conv))
                i_start = max(0, kmax - (n - 1))
                i_end = min(n - 1, kmax)
                i_range = np.arange(i_start, i_end + 1)
                # compute product contributions
                contributions = current[i_range] * current[kmax - i_range]
                donor = i_range[np.argmax(contributions)]
                partner = kmax - donor
                if current[partner] > current[donor]:
                    donor = partner
            else:
                donor = rng.integers(0, n)
                if current[donor] <= 0:
                    continue
            # pick recipient different
            rec = rng.integers(0, n - 1)
            while rec == donor:
                rec = rng.integers(0, n - 1)

            delta = current[donor] * rng.random()**2
            if delta < 1e-12:
                continue
            # ensure recipient bound

import numpy as np
import time

def _sidon_like_sequence(n):
    """Classic Sidon‑like decreasing sequence a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _conv_max_fft(a, nfft, length):
    """Linear convolution a*a via FFT, return max and the full array."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:length]
    return conv.max(), conv

                         0.0, 1000.0)
