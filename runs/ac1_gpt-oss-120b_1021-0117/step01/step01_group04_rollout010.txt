    fa = np.fft.rfft(arr, n=L)

    M = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * M / (s * s)

import numpy as np, time, math

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    M = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * M / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ---- initial sequence -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        base = parent.copy()
    except Exception:
        n = 256
        base = _gen_binomial(n)

    L = _next_pow2(2 * n - 1)

    # try pure binomial of same length as a secondary start
    try:
        bin_seq = _gen_binomial(n)
        bin_score = _ac1_score(bin_seq, n, L)
    except Exception:
        bin_seq = None
        bin_score = float('inf')

    cur_score = _ac1_score(base, n, L)
    best = base.copy()
    best_score = cur_score
    if bin_seq is not None and bin_score < best_score:
        best = bin_seq.copy()
        best_score = bin_score
        cur_score = bin_score

    cur = best.copy()
    S0 = float(cur.sum())

    # ---- search parameters ------------------------------------------------
    step_factor = 0.08                # target magnitude of each gradient move
    max_no_improve = 2000
    no_improve = 0

    while time.monotonic() < deadline:
        # convolution and peak index
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        p = int(conv.argmax())
        M = float(conv[p])

        # rev[i] = a_{p-i} where defined, else 0
        rev = np.zeros_like(cur)
        low = max(0, p - (n - 1))
        high = min(p, n - 1)
        if low <= high:
            idx = np.arange(low, high + 1)
            rev[idx] = cur[p - idx]

        const = 4.0 * n / (S0 * S0)
        grad = const * (rev - M / S0)
        maxg = np.max(np.abs(grad))
        if maxg == 0:
            break

        improved = False
        for factor in (step_factor, step_factor * 0.5, step_factor * 0.2):
            a = factor / maxg
            cand = cur - a * grad
            np.clip(cand, 0.0, 1000.0, out=cand)
            total = cand.sum()
            if total > 0:
                cand *= (S0 / total)
                np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _ac1_score(cand, n, L)
            if cand_score < cur_score - 1e-12:
                cur = cand
                cur_score = cand_score
                improved = True
                no_improve = 0
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                break

            no_improve += 1
            if rng.random() < 0.12:

                cand = cur.copy()

                if i == j:

                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta

            np.clip(cand, 0.0, 1000.0, out=cand)
            total = cand.sum()
            if total > 0:
                cand *= (S0 / total)
                np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _ac1_score(cand, n, L)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score

                    best = cand.copy()
                    best_score = cand_score

            cur_score = best_score
