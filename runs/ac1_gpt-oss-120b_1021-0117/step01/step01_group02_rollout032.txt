import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Gradient‑descent on a smoothed max‑autocorrelation.
    Returns a list of non‑negative coefficients."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- initial sequence -------------------------------------------------
    try:
        n = len(height_sequence_1)
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() == 0.0:
            raise ValueError
        a = a / a.sum()
    except Exception:
        # binomial‑like start (a_i ≈ 1/√i)
        n = 5000
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
        a = a / a.sum()

    # ---- helper: fast AC1 upper bound ------------------------------------
    size_conv = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    size_corr = 1 << ((3 * n - 2).bit_length())          # next power of two ≥ 3n‑2

    def fast_score(v):
        """2·n·max(conv(v,v)) / (sum(v)^2)  – uses FFT for speed."""
        fa = np.fft.rfft(v, n=size_conv)
        conv = np.fft.irfft(fa * fa, n=size_conv)[:2 * n - 1]
        return 2.0 * n * conv.max() / (v.sum() ** 2)

    best = a.copy()
    best_score = fast_score(best)

    # occasional safety fallback (uniform)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = fast_score(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # ---- optimisation parameters -------------------------------------------
    lr = 0.5               # step size (will adapt)
    alpha = 1.0            # smoothing for the max (will grow)
    min_lr = 1e-5
    max_lr = 10.0

    # ---- main loop ---------------------------------------------------------
    while time.monotonic() < deadline:
        # convolution of current best (FFT, length size_conv)
        fa = np.fft.rfft(best, n=size_conv)
        conv = np.fft.irfft(fa * fa, n=size_conv)[:2 * n - 1]
        max_c = conv.max()

        # smoothed weights for the max (log‑sum‑exp trick)
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max:  grad_j = 2 Σ_i a_i w_{i+j}
        fa_corr = np.fft.rfft(best, n=size_corr)
        fw = np.fft.rfft(w, n=size_corr)
        corr_full = np.fft.irfft(np.conj(fa_corr) * fw, n=size_corr)
        # linear correlation length = n + (2n‑1) – 1 = 3n‑2
        corr = corr_full[:3 * n - 2]
        grad = 2.0 * corr[:n]            # only lags 0 … n‑1 are needed

        # candidate by a gradient step, then projection onto non‑negative simplex
        cand = best - lr * grad
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:                     # avoid degenerate case
            cand = best.copy()
        else:
            cand /= s

        cand_score = fast_score(cand)

        if cand_score < best_score:      # accept improvement
            best = cand
            best_score = cand_score
            lr = min(max_lr, lr * 1.05)   # slowly increase step size
            alpha = min(20.0, alpha * 1.02)   # sharpen the max approximation
            if best_score < 1.5028:      # reached target – stop early
                break
        else:                            # reject – shrink step size
            lr = max(min_lr, lr * 0.70)

        # occasional random shake to escape plateaus
        if rng.random() < 0.001:
            width = max(1, n // 200)
            start = rng.integers(0, max(1, n - width + 1))
            stop = start + width
            shake = best.copy()
            shake[start:stop] *= np.exp(rng.normal(0.0, 0.15, size=width))
            shake = np.clip(shake, 0.0, None)
            shake_sum = shake.sum()
            if shake_sum > 0.0:
                shake /= shake_sum
                sh_score = fast_score(shake)
                if sh_score < best_score:
                    best = shake
                    best_score = sh_score
                    lr = min(max_lr, lr * 1.02)

    # ---- final sanity check -------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final.tolist()]
