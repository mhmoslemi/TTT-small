import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i   (recurrence a0=1, a_i = a_{i-1}*(2i-1)/(2i))."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 upper bound (fast via FFT)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic optimisation that starts from the binomial (or the provided
    parent) and aggressively fills the slack of the autocorrelation while
    also performing guided mass‑transfer moves.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----- problem size ------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256
        parent = None

    L = _next_pow2(2 * n - 1)          # FFT length (next power of two)

    # ----- initial sequence -------------------------------------------
    base = _gen_binomial(n)            # classic binomial (C(2i,i)/4^i)
    best = base.copy()
    best_score = _ac1_score(best, n, L)

    if parent is not None:
        parent_score = _ac1_score(parent, n, L)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    cur = best.copy()
    cur_score = best_score

    # annealing parameters
    temperature = 1.0
    decay = 0.9997

    target_bound = 1.50274349

    while time.monotonic() < deadline:
        # ----- current convolution & slack --------------------------------
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_b = float(conv.max())
        slack = max_b - conv
        slack[slack < 0] = 0.0                      # numerical safety

        cand = cur.copy()
