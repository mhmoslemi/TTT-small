import time
import numpy as np

# ------------------------------------------------------------
# projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# analytic sqrt‑series (first n coeffs of a where a*a = 1+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        s = (
            0.0
            if k == 1
            else float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        )
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# fast AC1 score (identical to the official evaluator)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = vec.sum()
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ------------------------------------------------------------
# gradient of the maximal convolution entry (index k)
# ------------------------------------------------------------
def _grad_at(vec: np.ndarray, k: int) -> np.ndarray:
    n = vec.size
    grad = np.zeros_like(vec)
    j_start = max(0, k - (n - 1))
    j_end = min(k, n - 1)
    idx = np.arange(j_start, j_end + 1)
    grad[idx] = 2.0 * vec[k - idx]
    return grad
