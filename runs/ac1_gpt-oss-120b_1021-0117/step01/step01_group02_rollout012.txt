def _score(arr, n=None):
    if n is None:
        n = arr.shape[0]
    conv = np.convolve(arr, arr)
    max_b = conv.max()
    s = arr.sum()
    if s < 0.01:
        return math.inf
    return float(2.0 * n * max_b / (s * s))

def _partner(arr, k):
    n = arr.shape[0]
    partner = np.zeros_like(arr)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    # i indices from i_start..i_end inclusive
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    partner[idx] = arr[k - idx]
    return partner

import numpy as np
import math

def _score(arr, n=None):
    """Exact score used by the evaluator (fast local version)."""
    if n is None:
        n = arr.shape[0]
    conv = np.convolve(arr, arr)
    max_b = conv.max()
    s = arr.sum()
    if s < 0.01:
        return math.inf
    return float(2.0 * n * max_b / (s * s))

def _partner(arr, k):
    """Vector g where g[i]=arr[k-i] if within bounds, else 0."""
    n = arr.shape[0]
    g = np.zeros_like(arr)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    g[idx] = arr[k - idx]
    return g

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = _score(uniform, n)

        best_score = uniform_score

    no_improve = 0

        conv = np.convolve(best, best)
        k = int(conv.argmax())
        M = float(conv[k])

        conv2 = conv.copy()
        conv2[k] = -np.inf
        M2 = float(conv2.max())

        # ----- partner vector for the peak index -------------------------
        partner = _partner(best, k)

        # source (largest partner) and destination (smallest partner)
        src_candidates = np.flatnonzero(partner == partner.max())
        src = int(rng.choice(src_candidates))
        dest_candidates = np.flatnonzero(partner == partner.min())
        dest = int(rng.choice(dest_candidates))

        if src == dest:

            width = int(rng.integers(max(1, n // 250), max(2, n // 20)))
