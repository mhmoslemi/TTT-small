import numpy as np
import time
import math

def _sidon_like_sequence(n):
    """Classic decreasing Sidon‑like sequence a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _fast_score(arr):
    """Score matching evaluate_sequence but without list conversion."""
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    return float(2.0 * arr.shape[0] * conv.max() / (s * s))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----------------------------------------------------------------------
    # 1. Load parent if present, otherwise fall back to Sidon‑like.
    # ----------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = int(parent.size)
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # Sidon‑like construction (always available)
    sidon = _sidon_like_sequence(n)

    # ----------------------------------------------------------------------
    # 2. Pick the better of the two as the initial reference.
    # ----------------------------------------------------------------------
    best_seq = parent.copy()
    best_score = _fast_score(best_seq)
    sid_score = _fast_score(sidon)
    if sid_score < best_score:
        best_seq = sidon.copy()
        best_score = sid_score

    # Keep a stable reference sequence for random restarts.
    base_seq = best_seq.copy()

    # ----------------------------------------------------------------------
    # 3. Phase A – massive random permutations.
    # ----------------------------------------------------------------------
    # Expect huge improvement by spreading large coefficients.
    while time.monotonic() < deadline:
        cand = base_seq[rng.permutation(n)]
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            # keep the best as new base for fast diffusion
            base_seq = best_seq.copy()
        # break early if an excellent bound is already reached
        if best_score < 1.3:          # comfortably below the target
            break

    # ----------------------------------------------------------------------
    # 4. Phase B – local hill‑climbing with occasional larger moves.
    # ----------------------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score
    max_temp = max(0.1 * best_score, 0.05)      # small temperature to allow escapes
    total_time = deadline - start

    while time.monotonic() < deadline:
        # choose a mutation type
        mode = rng.random()
        cand = current_seq.copy()
        if mode < 0.5:

            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

            if j - i > 1:

                rng.shuffle(segment)
                cand[i:j] = segment

    # 5. Return the best sequence found (as plain Python floats).
    # ----------------------------------------------------------------------
    return [float(x) for x in best_seq]
