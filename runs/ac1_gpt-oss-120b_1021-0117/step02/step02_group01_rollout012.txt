def compute_gradient(a, conv, K=3):
    n = a.size
    # get K largest indices
    if K >= conv.size:
        active = np.arange(conv.size)
    else:
        idx = np.argpartition(-conv, K-1)[:K]
        active = idx
    g = np.zeros_like(a)
    for i in active:
        lo = max(0, i - (n - 1))
        hi = min(n - 1, i)
        # slice of a reversed indices
        rev_slice = a[i - np.arange(lo, hi+1)]
        g[lo:hi+1] += 2.0 * rev_slice
    return g

import time
import numpy as np

# -------------------------------------------------------------
# Projection onto the probability simplex (x >= 0, sum(x) = 1)
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

# -------------------------------------------------------------
# Linear convolution of a with itself using FFT (full length)
# -------------------------------------------------------------
def _full_self_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full linear convolution) using an FFT of length L."""
    f = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * a.size - 1]
    return conv

# -------------------------------------------------------------
# Gradient of the max‑convolution (sub‑gradient using top‑K indices)
# -------------------------------------------------------------
def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int = 4) -> np.ndarray:
    """Compute a sub‑gradient using the top‑k convolution entries."""
    n = a.size
    # indices of the k largest convolution values
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    g = np.zeros_like(a)
    for i in idxs:
        lo = max(0, i - (n - 1))
        hi = min(n - 1, i)
        # vector of a[i-lo : i-hi] reversed
        rev = a[i - np.arange(lo, hi + 1)]
        g[lo : hi + 1] += 2.0 * rev
    return g

# Main entry point
# -------------------------------------------------------------
