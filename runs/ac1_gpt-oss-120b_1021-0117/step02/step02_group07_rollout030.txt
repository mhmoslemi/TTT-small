import numpy as np
import time

# -------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# -------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 – positive decreasing sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fft_conv(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of seq with itself using a power‑of‑two FFT."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    # keep only the linear part (length 2*len(seq)-1)
    return conv[: 2 * seq.size - 1]


def _score(seq: np.ndarray, fft_len: int) -> float:
    """Exact score used by the evaluator (fast FFT version)."""
    conv = _fft_conv(seq, fft_len)
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    n = seq.size
    return 2.0 * n * max_b / (s * s)


def _gradient_direction(seq: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. seq, preserving the sum."""
    n = seq.size
    max_b = conv.max()
    # treat ties – include all entries within a tiny relative tolerance
    eps = max_b * 1e-12
    act = np.where(conv >= max_b - eps)[0]

    grad = np.zeros_like(seq)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_vals = np.arange(lo, hi + 1, dtype=np.int64)
        grad[i_vals] += seq[k - i_vals]
    grad *= 2.0                     # ∂B/∂a_i = 2 a_{k-i}
    # direction that reduces max while keeping sum unchanged
    direction = -grad + grad.mean()
    return direction


def _random_perturb(seq: np.ndarray, rng: np.random.Generator,
                    s_target: float) -> np.ndarray:
    """Small stochastic move (multiplicative, transfer, smoothing)."""
    n = seq.size
    cand = seq.copy()
    op = rng.integers(0, 5)

    if op == 0:                                  # block scaling
        width = rng.integers(1, max(2, n // 50))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.20))
        cand[start:start + width] *= factor

    elif op == 1:                                # single index scaling
        idx = rng.integers(0, n)
        factor = np.exp(rng.normal(0.0, 0.25))
        cand[idx] *= factor

    elif op == 2:                                # move a fraction i → j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j:
            frac = rng.uniform(0.05, 0.15)
            delta = cand[i] * frac
            cand[i] -= delta
            cand[j] += delta

    elif op == 3:                                # smooth two neighbours
        i = rng.integers(0, n - 1)
        avg = (cand[i] + cand[i + 1]) * 0.5
        cand[i] = avg
        cand[i + 1] = avg

    else:                                        # gentle global scaling
        factor = 1.0 + rng.normal(0.0, 0.04)
        cand *= factor

    # enforce hard bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    total = cand.sum()
    if total > 0.0:
        cand *= s_target / total
        np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# -------------------------------------------------------------
# Main search routine (the only required top‑level name)
# -------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the target."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---------------------------------------------------------
    # Determine problem size and optional parent sequence
    # ---------------------------------------------------------
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # ---------------------------------------------------------
    # Normalisation target (any positive constant works; we use √(2n))
    # ---------------------------------------------------------
    s_target = np.sqrt(2.0 * n)

    # ---------------------------------------------------------
    # Build two simple seeds: parent (if any) and binomial
    # ---------------------------------------------------------
    binom = _initial_binomial(n)
    binom = binom * (s_target / binom.sum())

    candidates = [binom]
    if parent is not None:
        # also normalise parent to the same sum – shape is unchanged for score
        parent = parent * (s_target / parent.sum())
        candidates.append(parent)

    # pick the best of the seeds (fast scoring)
    fft_len = 1 << ((2 * n - 1).bit_length())
    best = None
    best_score = float("inf")
    for seq in candidates:
        sc = _score(seq, fft_len)
        if sc < best_score:
            best_score = sc
            best = seq.copy()

    # ---------------------------------------------------------
    # Working copy – initialise current point to the best seed
    # ---------------------------------------------------------
    current = best.copy()
    current_score = best_score

    # ---------------------------------------------------------
    # Main iterative improvement (gradient + occasional random moves)
    # ---------------------------------------------------------
    while time.monotonic() < deadline:
        # decide move type
        if rng.random() < 0.72:                     # gradient‑type move
            conv_curr = _fft_conv(current, fft_len)
            direction = _gradient_direction(current, conv_curr)

            # linear schedule for step size (larger early, tiny later)
            elapsed = time.monotonic() - start
            fraction = elapsed / (deadline - start)
            alpha = 0.07 * (1.0 - fraction) + 0.001   # never zero
            step = alpha * (np.max(current) + 1.0)

            cand = current + step * direction
            np.clip(cand, 0.0, 1000.0, out=cand)
            total = cand.sum()
            if total > 0.0:
                cand *= s_target / total
                np.clip(cand, 0.0, 1000.0, out=cand)
        else:                                         # random perturbation
            cand = _random_perturb(current, rng, s_target)

        cand_score = _score(cand, fft_len)

        # ----- acceptance test (mostly downhill, occasional uphill) -----
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # simple temperature that vanishes linearly
            elapsed = time.monotonic() - start
            fraction = elapsed / (deadline - start)
            temp = 0.04 * (1.0 - fraction) + 1e-9
            if rng.random() < np.exp((current_score - cand_score) / temp):
                accept = True

        if accept:
            current = cand
            current_score = cand_score

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # -------------------------------------------------
        # Very occasional pure hill‑climbing refinement on best
        # -------------------------------------------------
        if rng.random() < 0.005:
            refin = _random_perturb(best, rng, s_target)
            r_score = _score(refin, fft_len)
            if r_score < best_score:
                best = refin.copy()
                best_score = r_score

    # ---------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ---------------------------------------------------------
    return [float(x) for x in best]
