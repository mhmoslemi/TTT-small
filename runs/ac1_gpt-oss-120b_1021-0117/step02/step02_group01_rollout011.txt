def simplex_projection(v):
    # v is non-negative? Not necessarily; can have negative entries
    # Standard Euclidean projection onto simplex {x >=0, sum=1}
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0)
    # In case of rounding error out of sum1, renorm
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.ones_like(w) / n
    return w

def _project_simplex(v):
    # Euclidean projection onto the probability simplex, sum to 1.
    v = np.asarray(v, dtype=np.float64)

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(w, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution via FFT – return max value, its index and the full array
# ----------------------------------------------------------------------
def _conv_max_and_index(x: np.ndarray, L: int):
    """Full self‑convolution of x (real) using FFT.
    Returns (max value, index of the max, full convolution array)."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    max_val = conv.max()
    max_idx = int(conv.argmax())
    return float(max_val), max_idx, conv
