import time
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# Analytic sqrt‑series (coefficients a with a*a = 1+…+x^{2n‑2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ----------------------------------------------------------------------
# Convolution via FFT: return maximal value and its index (full mode)
# ----------------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# ----------------------------------------------------------------------
# Candidate proposal
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --- initialise ----------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)

    a = np.clip(a, 0.0, 1000.0)
    a = _project_simplex(a)               # sum = 1, non‑negative
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val          # sum(a)=1 => denominator =1
    best = a.copy()
    best_score = cur_score

    # --- optimisation parameters ----------------------------------------
    step = 0.5
    min_step = 1e-6
    decay = 0.9994
    prob_swap = 0.015                     # probability of a random mass‑swap

    while time.monotonic() < deadline:
        # ----- gradient step targeting the presently maximal convolution -----
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] = 2.0 * a[max_idx - j]            # ∂(a*a)_max / ∂a_j
        cand = a - step * g
        cand = _project_simplex(cand)

        max_val_c, max_idx_c = _conv_max_and_index(cand, L)
        cand_score = 2.0 * n * max_val_c

        # ----- accept improvement -----------------------------------------
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            max_idx = max_idx_c
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # occasional simulated‑annealing acceptance
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-9)):
                a = cand
                cur_score = cand_score
                max_idx = max_idx_c

        # ----- random mass transfer (exploration) -------------------------
        if rng.random() < prob_swap:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)

                max_val2, max_idx2 = _conv_max_and_index(cand2, L)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    max_idx = max_idx2
                    if cand2_score < best_score:
                        best = cand2.copy()
                        best_score = cand2_score

        # ----- step size schedule -----------------------------------------
        step = max(min_step, step * decay)

        # ----- early termination guard ------------------------------------
        if time.monotonic() + 0.5 > deadline:
            break

    # Return a plain‑Python list as required by the harness
    return [float(x) for x in best]
