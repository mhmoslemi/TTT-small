import time
import numpy as np

# ---------------------------------------------------------------
# Project onto the simplex {x >= 0, sum(x)=1}
# ---------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    theta = css[rho[-1]] / (rho[-1] + 1) if rho.size else 0.0
    return np.maximum(v - theta, 0.0)

# ---------------------------------------------------------------
# Full convolution a*a using FFT (real result)
# ---------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv

# ---------------------------------------------------------------
# Analytic sqrt series (fallback initializer)
# ---------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ---------------------------------------------------------------
