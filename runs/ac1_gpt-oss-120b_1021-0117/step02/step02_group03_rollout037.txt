    rng = np.random.default_rng(seed)
    deadline = time.monotonic()+budget_s-5

    if 'height_sequence_1' in globals():

    # evaluations
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())   # uses evaluator directly
    # precompute FFT length
    L = 1 << ((2*n-1).bit_length())
    
    # gradient descent parameters
    eta0 = 0.8
    beta = 0.9
    vel = np.zeros_like(a)
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # compute conv and max
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        max_val = float(conv.max())
        # get indices of top values near max
        # we can get list of indices where conv >= max_val - eps
        eps = max_val * 1e-6 + 1e-12
        ks = np.where(conv >= max_val - eps)[0]
        # compute gradient sum for all such ks
        grad = np.zeros_like(a)
        for k in ks:
            # compute gradient for lag k
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            idx = np.arange(i_min, i_max + 1)
            partner = k - idx
            grad[idx] += a[partner]
        grad = 2.0 * grad
        # if k even adjust central as before? But that correction for double counting of diagonal?

        # So formula above correct.
        # now normalized gradient
        norm = np.linalg.norm(grad)
        if norm == 0:
            direction = np.zeros_like(a)
        else:

        # Metropolis acceptance

import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑normalize to avoid tiny numerical drift
    s = w.sum()
    if s > 0:
        w /= s
    return w


# ------------------------------------------------------------
# Maximum value of the (full) autocorrelation a*a via FFT
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    """Return max(conv(a, a)) using an FFT of length L."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Gradient of the active convolution lag k (sub‑gradient of max)
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------------
    # initialise – use the provided parent if present, otherwise sqrt series
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: short sqrt series (length 256)
        def _sqrt_series(m):
            out = np.empty(m, dtype=np.float64)
            out[0] = 1.0
            for k in range(1, m):
                if k == 1:
                    s = 0.0
                else:
                    s = float(np.dot(out[1:k], out[k - 1 : 0 : -1]))
                out[k] = (1.0 - s) / 2.0
            return out

        a = _sqrt_series(256).astype(np.float64)

    n = a.shape[0]
    a = a / a.sum()
    a = _project_simplex(a)

    # --------------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------------
    Lfft = 1 << ((2 * n - 1).bit_length())   # next power of two
    cur_max = _max_convolution(a, Lfft)
    best = a.copy()
    best_score = 2.0 * n * cur_max

    # --------------------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------------------
    iteration = 0
    eta0 = 0.9           # base step size
    beta = 0.9           # momentum decay
    vel = np.zeros_like(a)
    no_improve = 0
    TARGET = 1.5028

    while time.monotonic() < deadline:
        iteration += 1

        # current convolution and its argmax
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv.max())
        k = int(conv.argmax())
        cur_score = 2.0 * n * M

        # gradient of the active lag
        grad = _grad_for_k(a, k)
        norm = np.linalg.norm(grad)
        if norm > 0.0:
            direction = grad / norm
        else:
            direction = np.zeros_like(a)

        # simple momentum step
        step = eta0 / np.sqrt(iteration)
        vel = beta * vel - step * direction
        cand = a + vel
        cand = _project_simplex(cand)

        # occasional aggressive random perturbation (helps escape plateaus)
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.18
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

        # evaluate candidate
        cand_M = _max_convolution(cand, Lfft)
        cand_score = 2.0 * n * cand_M

        # keep the best ever seen
        if cand_score < best_score:

            best = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # accept improving (or very occasional worsening) moves
        if cand_score < cur_score or rng.random() < 0.004:
            a = cand

        # restart from a fresh random simplex point if stuck for long
        if no_improve >= 2500:
            a = rng.dirichlet(np.ones(n)).astype(np.float64)
            a = _project_simplex(a)
            no_improve = 0

        # early exit if target met
        if best_score < TARGET:
            break

    # --------------------------------------------------------------------
    # final safety check – ensure the returned list is valid for evaluator
    # --------------------------------------------------------------------
    # (project once more, guard against tiny negatives due to rounding)
    best = _project_simplex(best)
    return [float(x) for x in best]
