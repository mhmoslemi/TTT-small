import time
import math
import numpy as np

# ------------ helper functions -------------------------------------------

def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2*n-1)."""
    A = np.fft.rfft(a, fft_len)
    B = A * A
    full = np.fft.irfft(B, fft_len)
    return full[: 2 * n - 1]


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """Generate a modest collection of plausible start vectors (∑=1)."""
    cand = []

    # parent, if supplied
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # Catalan shape
    w = _catalan_weights(n)
    cand.append(w / w.sum())

    # symmetric power shapes a_i ∝ ((i+1)*(n-i))**p
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for pwr in (0.5, 1.0, 1.5, 2.0, 3.0):
        v = base ** pwr
        cand.append(v / v.sum())

    # sin‑power shapes a_i ∝ sin^p(π*(i+0.5)/n)
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for pwr in (0.5, 1.0, 2.0):
        v = sin_vals ** pwr
        cand.append(v / v.sum())

    return cand


def _score_of(vec: np.ndarray, n: int, fft_len: int) -> float:
    """Return the AC1 upper bound for a vector already normalized to sum = 1."""
    conv = _conv_self_fft(vec, n, fft_len)
    return float(2.0 * n * conv.max())

    bound 1.50274349.
    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    try:

    except Exception:

    fft_len = 1 << ((2 * n - 1).bit_length())

    # ----- build a few starter vectors ------------------------------------
    starters = _initial_candidates(n, parent_arr)
    best_vec = None
    best_score = float("inf")

    for vec in starters:
        sc = _score_of(vec, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
        if best_score <= 1.50274349:
            return best_vec.tolist()

    # ----- parameters for the exponentiated‑gradient descent -------------
    eta0 = 0.8                # initial step magnitude
    jitter_prob = 0.025
    jitter_sigma = 0.02
    max_no_improve = 800
    no_improve = 0
    iteration = 0

    a = best_vec.copy()      # current iterate (always sums to 1)

    # ----- main optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        # compute current convolution and score
        conv = _conv_self_fft(a, n, fft_len)
        max_b = float(conv.max())
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            no_improve += 1

        # active index that attains the maximum
        k = int(conv.argmax())

        # sub‑gradient of the active quadratic term
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]   # derivative = 2·a_{k-i}

        # diminishing step size
        eta = max(1e-4, eta0 / math.sqrt(iteration + 1))

        # exponentiated‑gradient update (keeps positivity)
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex (sum = 1)
        s = a.sum()
        if s <= 0.0:
