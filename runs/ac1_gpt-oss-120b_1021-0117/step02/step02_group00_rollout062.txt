import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT (real-valued input)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return full convolution a*a using FFT (real output)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Gradient of the smooth approximation  (log‑sum‑exp) of max(conv)
# ------------------------------------------------------------
def _grad_smooth(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Compute gradient of  f(a)= (1/gamma) log sum exp(gamma*conv(a))
    with respect to a, assuming sum(a)=1.
    """
    n = a.shape[0]
    # 1) convolution a*a  (full length 2n‑1)
    conv = _conv_fft(a)

    # 2) soft‑max weights
    m = conv.max()
    exp_terms = np.exp(gamma * (conv - m))
    Z = exp_terms.sum()
    w = exp_terms / Z                     # length 2n‑1, non‑negative, sums to 1

    # 3) correlation a_rev * w  (linear convolution)
    L = 2 * n - 1
    N = 1 << ((n + L - 1).bit_length())  # safe length for linear conv
    a_rev = a[::-1]

    A_rev = np.fft.rfft(a_rev, n=N)
    W = np.fft.rfft(w, n=N)
    corr_full = np.fft.irfft(A_rev * W, n=N)   # length N (real)

    start = L - 1
    grad = 2.0 * corr_full[start:start + n]    # slice gives n entries
    return grad
