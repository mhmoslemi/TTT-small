import time
import numpy as np

def _sqrt_series(n: int) -> np.ndarray:
    """
    Compute the coefficients a[0..n-1] of the formal power series
    sqrt(1 + x + … + x^{2n-2}) = Σ a_k x^k.
    The recurrence is:
        a_0 = 1,
        a_k = (1 - Σ_{i=1}^{k-1} a_i * a_{k-i}) / 2   for k ≥ 1.
    All coefficients are non‑negative.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = 0.5 * (1.0 - s)
        # Guard against tiny negative artifacts from round‑off
        if a[k] < 0.0:
            a[k] = 0.0
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients whose AC1 bound
    is expected to be close to the theoretical optimum (~1.0).
    The construction uses the sqrt‑series described above.
    """
    # Preserve a safety margin for the overall sandbox timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Length of the desired sequence: use the provided parent length if
    # available, otherwise fall back to a modest default.
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2048  # reasonable size when no parent is supplied

    # Generate the candidate.
    a = _sqrt_series(n)

    # Clamp to the evaluator's admissible range.
    a = np.clip(a, 0.0, 1000.0)

    # Ensure a positive L1 norm (the series is far above 0.01, but we keep it for safety).
    if a.sum() < 0.01:
        a[0] = 0.01

    # If the time budget is already exhausted, fall back to a trivial uniform vector.
    if time.monotonic() >= deadline:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Return a plain Python list as required.
    return [float(x) for x in a]
