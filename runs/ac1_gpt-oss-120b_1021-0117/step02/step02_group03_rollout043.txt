import time

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------

    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())

    # --------------------------------------------------------------------
    cur_max = _max_convolution(a, Lfft)
    cur_score = 2.0 * n * cur_max
    best = a.copy()
    best_score = cur_score

    # --------------------------------------------------------------------
    # simulated‑annealing style search
    # --------------------------------------------------------------------
    start_time = time.monotonic()
    total_time = deadline - start_time
    max_temp = 0.02                     # initial temperature

    while time.monotonic() < deadline:
        # linear cooling
        elapsed = time.monotonic() - start_time
        T = max_temp * max(0.0, 1.0 - elapsed / total_time)

        # ------------------------------------------------------------
        # propose a small mass transfer
        # ------------------------------------------------------------
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        # pick a different destination index
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # fraction of the source mass to move (adaptively shrink with time)
        frac = rng.random() * 0.1 * max(0.1, 1.0 - elapsed / total_time)
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # ------------------------------------------------------------
        # evaluate candidate
        # ------------------------------------------------------------
        cand_max = _max_convolution(cand, Lfft)
        cand_score = 2.0 * n * cand_max

        # acceptance rule (Metropolis)
        accept = False
        if cand_score < cur_score:
            accept = True
        elif T > 1e-12:
            # probability of uphill move
            prob = np.exp(-(cand_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score

        # keep the global best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            if best_score < 1.5028:
                break

        # occasional aggressive shuffle to escape plateaus
        if rng.random() < 0.008:
            # rebuild a fresh random sparse vector
            m2 = max(2, int(np.sqrt(2 * n) * (1.0 + rng.random() * 1.5)))
            new_a = np.zeros(n, dtype=np.float64)
            idx2 = rng.choice(n, size=m2, replace=False)
            new_a[idx2] = 1.0 / m2
            a = new_a
            cur_score = 2.0 * n * _max_convolution(a, Lfft)

    # final safety projection (keeps sum exactly 1)
    best = _project_simplex(best)

    # return as plain Python list
    return [float(x) for x in best]
