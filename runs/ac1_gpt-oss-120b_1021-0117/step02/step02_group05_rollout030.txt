import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _conv_fft(x: np.ndarray, n: int, L: int) -> np.ndarray:
    """Linear self‑convolution via FFT (full length 2n‑1)."""
    fa = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv

def _ac1_score(arr: np.ndarray, conv: np.ndarray, n: int) -> float:
    """Exact score used by the evaluator (no extra allocations)."""
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    max_b = float(conv.max())
    return 2.0 * n * max_b / (s * s)

            best = cand.copy()
            best_score = cand_score
            best_conv = conv_cand.copy()
            if best_score < target:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # schedule
        temperature *= decay

        # occasional restart from the best solution
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # Return the best candidate found (as plain Python floats)
    return [float(x) for x in best]
