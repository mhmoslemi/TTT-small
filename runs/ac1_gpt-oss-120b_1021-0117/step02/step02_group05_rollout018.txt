import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, block_sigma))
            cand = cur.copy()
            cand[start:start + width] *= factor
        # ---------------------------------------------------------- #
        # 3) random mass transfer between two indices
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * transfer_max
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # enforce evaluator bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score = _ac1_score(cand, n, L)

        # global best update
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # temperature schedule
        temperature *= decay

        # occasional restart from best
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # return the best sequence found
    return [float(x) for x in best]
