def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (positive decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_conv(seq: np.ndarray, fft_len: int, n: int):
    """Return (score, max_conv)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return np.inf, max_b
    return 2.0 * n * max_b / (s * s), max_b
