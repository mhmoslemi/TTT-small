import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _max_conv_fft(a: np.ndarray, L: int) -> float:
    """Maximum value of the autocorrelation a*a using an FFT of length L."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return float(conv.max())

    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # leave a small safety margin before the hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise from the provided parent if possible, otherwise a uniform start
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        cur = parent.copy()
    except Exception:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    # make sure the vector is feasible for the evaluator
    cur = np.clip(cur, 0.0, None)
    s = cur.sum()
    if s == 0.0:
        cur = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur = cur / s                     # normalise to sum = 1

    L = _next_pow2(2 * n - 1)             # FFT length (power of two)

    # current and best scores (fast, exact for sum = 1)
    cur_score = 2.0 * n * _max_conv_fft(cur, L)
    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # hyper‑parameters
    temperature = 1.0
    decay = 0.9997
    grad_eps = 0.12                     # base step size for gradient moves
    block_min = max(2, n // 200)
    block_max = max(2, n // 12)

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        r = rng.random()
        cand_score = None

        # --------------------------------------------------------------
        # 1) gradient‑guided move (≈45 % of the time)
        if r < 0.45:
            # locate the index where autocorrelation attains its maximum
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            # gradient of the active autocorrelation entry
            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            distinct = mask & (rev != idx)
            g[distinct] *= 2.0

            # direction that preserves the total sum
            d = np.mean(g) - g

            # test a few step lengths and keep the best

            best_local_score = cur_score
            for factor in (0.5, 1.0, 2.0):

                score = 2.0 * n * _max_conv_fft(cand, L)
                if score < best_local_score:
                    best_local_score = score
                    best_local_vec = cand
            if best_local_vec is not None:

                cand_score = best_local_score

        # --------------------------------------------------------------
        # 2) block scaling move (≈30 % of the time)
        elif r < 0.75:

            j = int(rng.integers(0, n))

                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta

    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
