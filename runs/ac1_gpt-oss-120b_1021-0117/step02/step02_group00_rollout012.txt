def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialize a0
    if "height_sequence_1" in globals():
        a0 = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        # generate sqrt series like before similar to _sqrt_series
        a0 = np.empty(n, dtype=np.float64)
        a0[0] = 1.0
        for k in range(1, n):
            if k ==1:
                s=0.0
            else:
                s = np.dot(a0[1:k], a0[k-1:0:-1])
            a0[k] = (1.0 - s) / 2.0
    # ensure nonnegative and normalise to sum=1
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s

    n = a0.size

    # define helper funcs:
    def _conv_fft(a):
        L = 1 << ((2 * n - 1).bit_length())
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        return conv

    def _score(a):
        conv = _conv_fft(a)
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    def _project_simplex(v):
        # same as earlier, but handle zero size?
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # best solution tracking
    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # momentum
    m = np.zeros_like(cur)
    beta = 0.9
    alpha0 = 0.5   # initial step amplitude
    it = 0
    no_improve = 0
    K = 5  # number of active indices for gradient averaging
    # thresholds
    improve_target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        conv = _conv_fft(cur)
        max_val = conv.max()
        # get top K indices (worst conv entries)
        if K >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            # argpartition for top K largest
            part = np.argpartition(-conv, K-1)[:K]
            active_idx = part

        # gradient accumulation
        g = np.zeros_like(cur)
        for k in active_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # average gradient over active set
        g = g / len(active_idx)

        # momentum update
        m = beta * m + (1.0 - beta) * g

        # step size diminishing
        alpha = alpha0 / np.sqrt(it)

        cand = cur - alpha * m
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # maybe accept with small probability to escape
            # small probability decreasing with iteration
            prob = np.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # occasional random mass transfer if stuck
        if no_improve > 3000 and rng.random() < 0.2:
            i = rng.integers(0, n)
            if cur[i] > 0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # jitter restart toward best
        if it % 8000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        if best_sc < improve_target:
            break

    # Return best vector as python list
    return [float(x) for x in best_vec]

        # ------------------------------------------------------------------
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # occasional random mass‑transfer move if progress stalls
        # ------------------------------------------------------------------
        if no_improve > 3000 and rng.random() < 0.2:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # ------------------------------------------------------------------
        # jittered restart toward the best known solution
        # ------------------------------------------------------------------
        if it % 8000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # ------------------------------------------------------------------
        # early exit if target reached
        # ------------------------------------------------------------------
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # final fallback: simple local refinement (few random transfers)
    # ----------------------------------------------------------------------
    refin_end = time.monotonic() + min(0.4, deadline - time.monotonic())
    while time.monotonic() < refin_end:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

    # return the best vector found as a plain Python list
    return [float(x) for x in best_vec]
