import time
import numpy as np

# ---------- helper functions ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full convolution a*a using real FFT, returns length 2*n-1 array."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w

            jitter = np.exp(rng.normal(0.0, 0.012, size=n))

            cand = _project_onto_simplex(cand)

        a = cand

        # restart if stuck too long
        if iter_no_improve > 2500:
            pert = np.exp(rng.normal(0.0, 0.035, size=n))
            a = best_vec * pert
            a = _project_onto_simplex(a)
            iter_no_improve = 0

    # Return the best vector found (as a Python list)
    return best_vec.tolist()
