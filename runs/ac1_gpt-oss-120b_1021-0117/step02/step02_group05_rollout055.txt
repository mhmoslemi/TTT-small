import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound evaluation using FFT (same formula as the evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _gradient_on_max(arr: np.ndarray, kmax: int) -> np.ndarray:
    """Sub‑gradient of the max‑convolution entry (index kmax)."""
    n = arr.size
    idx = np.arange(n)
    rev = kmax - idx
    mask = (rev >= 0) & (rev < n)
    g = np.zeros_like(arr)
    g[mask] = arr[rev[mask]]
    dup = mask & (rev != idx)
    g[dup] *= 2.0
    return g

    # evaluate a coarse grid of exponents around 0.5 (the Catalan shape)
    for p in np.linspace(0.30, 0.70, 21):
        cand = _pow_family(p)
        cand_score = _ac1_score(cand, n, L)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

    # also keep the current candidate as starting point
    cur = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # Search parameters.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    temperature = 1.0                # SA temperature
    decay = 0.9996                    # exponential cooling
    min_block = max(2, n // 250)      # block size bounds for scaling moves
    max_block = max(2, n // 30)
    max_frac_transfer = 0.45          # max fraction moved in guided transfer
    sigma_scale = 0.18                # std. dev. for random block scaling
    eps_small = 0.02                  # small perturbation scale (unused here)

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # --------------------------------------------------------------
        # Decide move type.
        move = rng.random()
        # --------------------------------------------------------------
        # 0 – Gradient directed step (≈45 %).
        if move < 0.45:
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            g = _gradient_on_max(cur, kmax)
            mu = g.mean()
            # direction that preserves total mass (sum(d)=0)
            d = mu - g
            # maximum admissible step to keep non‑negativity
            neg = d < 0
            if np.any(neg):
                # avoid division by zero; consider only strictly positive cur entries
                safe = (cur > 1e-12) & neg
                if np.any(safe):
                    alpha_max = np.min(cur[safe] / (-d[safe]))
                else:
                    alpha_max = 0.0
            else:
                alpha_max = np.inf
            # step size is bounded both by alpha_max and a decaying schedule
            step = min(alpha_max * 0.95, temperature * 0.25 + 1e-8)
            if step <= 0:
                # fall back to a tiny random perturbation
                cand = cur + rng.normal(0.0, eps_small, size=n)
            else:
                cand = cur + step * d

        # --------------------------------------------------------------
        # 1 – Block scaling (≈30 %).
        elif move < 0.75:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, sigma_scale))
            cand = cur.copy()
            cand[start:start + width] *= factor

        # --------------------------------------------------------------
        # 2 – Guided mass transfer (≈20 %).
        elif move < 0.95:
            # Use the same gradient as in the first branch (cheap recompute)
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))
            g = _gradient_on_max(cur, kmax)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i != j:
                frac = rng.random() * max_frac_transfer
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                cand = cur.copy()

        # --------------------------------------------------------------
        # 3 – Small random jitter (≈5 %).
        else:
            # add tiny Gaussian noise to a random subset (~1 % of entries)
            mask = rng.random(n) < 0.01
            noise = rng.normal(0.0, eps_small, size=n)
            cand = cur.copy()
            cand[mask] += noise[mask]

        # --------------------------------------------------------------
        # Enforce bounds.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # --------------------------------------------------------------
        # Evaluate candidate quickly.
        cand_score = _ac1_score(cand, n, L)

        # Keep global best.
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # Early exit if we already beat the target.
            if best_score < 1.50274349:
                break

        # --------------------------------------------------------------
        # Simulated‑annealing acceptance.
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability to accept a worse move
            prob = math.exp(-(cand_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # --------------------------------------------------------------
        # Temperature schedule.
        temperature *= decay

        # Occasionally reset to the best seen to avoid drift.
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ------------------------------------------------------------------
    # Final safety: ensure the returned sequence respects evaluator limits.
    np.clip(best, 0.0, 1000.0, out=best)
    if best.sum() < 0.01:                      # guard against pathological sum
        best = np.maximum(best, 1e-4)
    return [float(x) for x in best]
