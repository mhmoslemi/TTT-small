def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..n‑1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fft_len_for_conv(n: int) -> int:
    """Next power of two not smaller than 2·n‑1 (FFT length)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution seq * seq using real FFT, length 2·n‑1."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    return conv[: 2 * len(seq) - 1]


def _score_fft(seq: np.ndarray, fft_len: int):
    """Return (score, conv_array, max_conv)."""
    conv = _conv_fft(seq, fft_len)
    max_b = float(conv.max())
    s = float(seq.sum())
    if s <= 0.0:
        return float("inf"), conv, max_b
    score = 2.0 * len(seq) * max_b / (s * s)
    return score, conv, max_b

            n = int(parent.size)
    except Exception:
        parent = None

    # ------------------------------
    # Initial best candidate
    # ------------------------------
    best = _initial_binomial(n)
    fft_len = _fft_len_for_conv(n)
    best_score, best_conv, _ = _score_fft(best, fft_len)

    if parent is not None:
        p_score, p_conv, _ = _score_fft(parent, fft_len)
        if p_score < best_score:
            best = parent.copy()
            best_score = p_score
            best_conv = p_conv

    # ------------------------------
    # Working copy (current state)
    # ------------------------------
    current = best.copy()
    current_score = best_score

    # ------- parameters for the meta‑heuristic -------
    max_temp = 0.12                         # initial temperature
    # -------------------------------------------------

    # Main optimisation loop
    while time.monotonic() < deadline:
        cand = current.copy()
        op = rng.integers(0, 6)

        # --------------------------------------------------
        # 0 – block scaling (exp‑log normal factor)
        # --------------------------------------------------
        if op == 0:
            width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            cand[start_idx:start_idx + width] *= factor

        # --------------------------------------------------
        # 1 – single‑index multiplicative perturbation
        # --------------------------------------------------
        elif op == 1:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.20))
            cand[idx] *= factor

        # --------------------------------------------------
        # 2 – move a small fraction from the (i,j) pair that
        #     attains the maximal convolution product
        # --------------------------------------------------
        elif op == 2:
            # use current convolution (re‑compute only when needed)
            cur_conv = best_conv if np.array_equal(cand, best) else _conv_fft(current, fft_len)
            kmax = int(cur_conv.argmax())
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            idxs = np.arange(lo, hi + 1, dtype=np.int64)
            js = kmax - idxs
            prods = current[idxs] * current[js]
            pair_pos = int(prods.argmax())
            i = idxs[pair_pos]
            j = js[pair_pos]
            if i != j:
                frac = rng.uniform(0.05, 0.12)
                delta_i = current[i] * frac
                delta_j = current[j] * frac
                total = delta_i + delta_j
                cand[i] -= delta_i
                cand[j] -= delta_j
                # move to the smallest entry (helps increase sum without
                # inflating the peak)
                tgt = int(np.argmin(cand))
                cand[tgt] += total
            else:
                # fallback: treat as single‑index perturbation
                factor = np.exp(rng.normal(0.0, 0.15))
                cand[i] *= factor

        # --------------------------------------------------
        # 3 – smooth two neighbours (averaging)
        # --------------------------------------------------
        elif op == 3:
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        # --------------------------------------------------
        # 4 – early‑to‑late mass transfer (flatten the head)
        # --------------------------------------------------
        elif op == 4:
            early_end = max(1, n // 12)
            late_start = n - early_end
            i = rng.integers(0, early_end)           # early index
            j = rng.integers(late_start, n)          # late index
            frac = rng.uniform(0.07, 0.22)
            delta = cand[i] * frac
            cand[i] -= delta
            cand[j] += delta

        # --------------------------------------------------
        # 5 – gradient step that directly attacks the maximal
        #     convolution entry while keeping the total sum
        # --------------------------------------------------
        else:
            # compute current convolution (only when needed)
            cur_conv = best_conv if np.array_equal(cand, best) else _conv_fft(current, fft_len)
            kmax = int(cur_conv.argmax())
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = kmax - idx

            grad = np.zeros_like(cand)
            grad[idx] = current[jdx]
            diag = idx == jdx
            if np.any(diag):
                grad[idx[diag]] *= 2.0

            # direction that reduces max‑conv while keeping sum constant
            mean_grad = grad.mean()
            direction = -grad + mean_grad
            step = rng.uniform(0.003, 0.015) * (cand.max() + 1.0)
            cand = current + step * direction

        # enforce bounds and non‑negativity
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score, cand_conv, _ = _score_fft(cand, fft_len)

        # --------------------------------------------------
        # Simulated‑annealing acceptance
        # --------------------------------------------------
        elapsed = time.monotonic() - start
        total = deadline - start
        temp = max(0.0, max_temp * (1.0 - elapsed / total))

        accept = False
        if cand_score < current_score:
            accept = True
        elif temp > 0.0:
            # probabilistic uphill move
            prob = np.exp((current_score - cand_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            # keep the best ever seen
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                best_conv = cand_conv

        # early exit if we already beat the target bound
        if best_score < 1.50274349:
            break

    # Return a plain Python list for the evaluator
    return [float(x) for x in best]
