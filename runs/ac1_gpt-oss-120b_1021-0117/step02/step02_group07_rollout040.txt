    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s)-2.)

    # get n and initial sequence

        if arr.ndim == 1 and arr.size>0 and np.isfinite(arr).all():

    # Normalize sum to 1 (or sqrt(2*n) maybe not needed)
    total = a.sum()
    if total == 0:
        a[:] = 1.0 / n
        total = a.sum()
    a = a / total   # sum=1

    # Precompute FFT length
    fft_len = 1 << ((2*n - 1).bit_length())
    # function compute conv and score
    def compute_conv_and_score(seq):
        f = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[: 2*n - 1]
        max_b = conv.max()
        score = 2.0 * n * max_b  # since sum(seq)=1
        return conv, max_b, score

    conv, max_b, score = compute_conv_and_score(a)
    best_seq = a.copy()
    best_score = score
    best_conv = conv

    # parameters for schedule
    iter_count = 0
    # time for each iteration target: we can adapt later

    # main loop
    while time.monotonic() < deadline:
        # compute gradient at max positions
        # tolerance for identifying multiple maxima (absolute)
        eps = max_b * 1e-12 + 1e-15
        ks = np.where(conv >= max_b - eps)[0]   # indices that achieve max (or near)
        # compute gradient
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n-1))
            hi = min(k, n-1)
            i = np.arange(lo, hi+1)
            grad[i] += 2.0 * a[k - i]  # vectorized
        # direction that preserves sum
        mean_grad = grad.mean()
        direction = -grad + mean_grad

        # step size schedule: eta decays with iterations / total allowed based on time
        # compute remaining proportion
        elapsed = time.monotonic() - start
        total_allowed = deadline - start
        progress = elapsed / total_allowed
        # start eta=0.02, end=0.001, linear decay
        eta = 0.02 * (1 - progress) + 0.001 * progress

        # compute step magnitude

        step = eta * np.max(a)  # maybe scale by max
        cand = a + step * direction

        # clip and renormalize

            sum_cand = cand.sum()
        cand = cand / sum_cand  # renormalize to sum 1

            conv = conv_new
            max_b = max_b_new
            score = score_new
            if score_new < best_score:
                best_seq = cand.copy()
                best_score = score_new
                best_conv = conv_new

                best_score = sc2
                best_conv = conv2
                a = cand2
                conv = conv2
                max_b = mb2
                score = sc2

        iter_count += 1

    # final result: best_seq (scaled arbitrary). Since score uses sum, we can return scaled version with any sum.
    # To stay within bounds, raise back to original scale maybe sum=1 works fine (values might be small).
    # Ensure values within [0,1000]; they are because sum normalized and max <=1.
    return [float(x) for x in best_seq]

import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

    try:

            a = np.clip(parent, 0.0, 1000.0)
            n = a.size
        else:
            raise Exception()
    except Exception:
        n = 3809
        a = _initial_binomial(n)

    # Normalise to sum = 1 (scale is irrelevant for the score)
    tot = a.sum()
    if tot == 0.0:

        tot = a.sum()
    a = a / tot

    # ------------------------------------------------------------------

        return conv_full, m, sc

    conv, max_b, cur_score = _conv_and_score(a)
