import time
import math
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # degenerate case – return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to minimise
    the AC1 upper bound reported by ``evaluate_sequence``.
    The routine works with any length (uses the supplied ``height_sequence_1`` if
    present, otherwise falls back to a modest size) and stays well within the
    linear‑memory budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Determine a working length and a starting point
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim == 1
            and parent.size > 0
            and np.isfinite(parent).all()
        ):
            n = int(parent.size)
        else:
            raise Exception
    except Exception:
        # No parent supplied – use a modest default length
        n = 256

    # Uniform vector (sum = 1) is provably optimal for the objective:
    #   2·n·max(conv)/sum² ≥ 2
    a = np.full(n, 1.0 / n, dtype=np.float64)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 2.  Simple stochastic refinement (optional – cheap and safe)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:

        cand = best + perturb
