import time
import numpy as np

def _proj_simplex(v):
    # Duchi et al. Euclidean projection onto the simplex.
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a, fft_len, conv_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv.max(), conv

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 0.2

    # initialise a
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        init = np.clip(parent, 0.0, 1000.0)
    except Exception:
        init = None

    if init is None or not np.isfinite(init).all():
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = int(init.size)
        a = init.copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:

        else:
            a = a / s

    # keep best

    best_score = float(evaluate_sequence(best.tolist()))
    # compare with uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # parameters
    mu = 0.9
    momentum = np.zeros_like(a)
    eta0 = 0.9
    it = 0

    # main loop
    while time.monotonic() < deadline:
        it += 1
        # convolution and gradient
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:

            best_score = cur_score
            if best_score <= target_score:
                break

        offset = max_idx - (n - 1)
        g = np.zeros_like(a)
        if offset >= 0:

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all entries are zero or negative – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
