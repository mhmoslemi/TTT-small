def propose_candidate(seed=42, budget_s=1100):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # Determine length n and parent.
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # Helper generate binomial
    def binomial_init(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (2 * i - 1) / (2 * i)
        return a

    # start candidate
    best = binomial_init(n)
    fft_len = 1 << ((2 * n - 1).bit_length())
    # Evaluate candidate
    def conv_score(seq):
        f = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
        maxb = np.max(conv)
        s = np.sum(seq)
        score = float('inf')
        if s > 0.0:
            score = 2.0 * n * maxb / (s * s)
        return score, maxb, conv

    best_score, best_maxb, best_conv = conv_score(best)

    if parent is not None:
        pscore, _, _ = conv_score(parent)
        if pscore < best_score:
            best = parent.copy()
            best_score = pscore
            best_maxb, best_conv = best_maxb, best_conv

    cur = best.copy()
    cur_score = best_score

    # Precompute max index of convolution
    def max_index(conv):
        return int(np.argmax(conv))

    # Gradient step based on active lag
    # compute direction -gradient for active lag, project to zero-sum
    def active_direction(seq, kmax):
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        raw = np.zeros_like(seq)
        raw[mask] = seq[j[mask]]
        # direction to reduce the active term while keeping sum constant
        # We project raw to zero-mean
        mean_raw = raw.mean()
        direction = -raw + mean_raw
        return direction

    # Main loop
    it = 0
    # Momentum term for gradient style
    momentum = np.zeros_like(cur)
    beta = 0.9

    while time.monotonic() < deadline:
        # compute conv and active lag
        _, maxb, conv = conv_score(cur)
        kmax = max_index(conv)
        # compute direction
        dir_vec = active_direction(cur, kmax)
        # step size schedule: start relatively large and decrease
        # base step proportional to typical magnitude
        scale = (np.max(cur) + 1.0)
        step = (0.1 / (1 + it * 1e-4)) * scale  # decaying
        # momentum update
        momentum = beta * momentum + (1 - beta) * dir_vec
        cand = cur + step * momentum

        # enforce bounds and preserve sum
        np.clip(cand, 0.0, 1000.0, out=cand)
        s_c = np.sum(cand)
        if s_c > 0:
            cand *= (np.sum(cur) / s_c)
            np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score, _, _ = conv_score(cand)

        # acceptance rule: only improve (greedy)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # occasional random perturbation to escape local minima
            if rng.random() < 0.01:
                width = rng.integers(1, max(2, n // 50))
                start_idx = rng.integers(0, n - width + 1)

                np.clip(cur, 0.0, 1000.0, out=cur)

                if s_cur > 0:

                if s_cur > 0:
                    cur *= (np.sum(best) / s_cur)
                    np.clip(cur, 0.0, 1000.0, out=cur)

import numpy as np
import time

def _binomial_init(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_score(seq: np.ndarray, fft_len: int):
    """Return (score, max_b, full convolution array)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), max_b, conv
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), max_b, conv


def _direction_topk(seq: np.ndarray, conv: np.ndarray, top_k: int = 5):
    """
    Compute a descent direction using the top_k largest convolution entries.
    For each selected lag k we add the shifted copy of seq and finally
    project to zero‑mean (so the sum of the direction is zero).
    """
    n = seq.shape[0]
    # indices of the k largest entries (unsorted)
    idxs = np.argpartition(conv, -top_k)[-top_k:]
    agg = np.zeros_like(seq)
    for k in idxs:
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        agg[mask] += seq[j[mask]]
    agg /= float(top_k)
    # project to zero mean
    return -agg + agg.mean()
