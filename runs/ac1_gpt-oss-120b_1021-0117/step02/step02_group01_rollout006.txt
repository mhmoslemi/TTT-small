    alpha_max = 8.0

    prob_random = 0.005

    while time.monotonic() < deadline:
        # compute conv via FFT
        a_fft = np.fft.rfft(a, n=fft_len_conv)

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic sqrt‑series (coefficients a with a*a = 1+…+x^{2n‑2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Return the analytic sqrt‑series of length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative heights that (hopefully) lowers the AC1 bound."""
    rng = np.random.default_rng(seed)
    # give a little margin before the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Initialise a feasible sequence (projected onto the simplex)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: analytic sqrt‑series of modest length
        a = _sqrt_series(256)

    a = _project_simplex(a)                     # ensure sum == 1, non‑negative
    n = a.size
    conv_len = 2 * n - 1
    grad_len = conv_len + n - 1                  # length needed for gradient convolution

    # FFT lengths (next powers of two)
    fft_len_conv = 1 << ((conv_len - 1).bit_length())
    fft_len_grad = 1 << ((grad_len - 1).bit_length())

    # ------------------------------------------------------------------
    # Helper: evaluate the true (non‑smoothed) score of a candidate
    # ------------------------------------------------------------------
    def _true_score(vec: np.ndarray) -> float:
        fv = np.fft.rfft(vec, n=fft_len_conv)
        conv = np.fft.irfft(fv * fv, n=fft_len_conv)[:conv_len]
        return 2.0 * n * float(conv.max())

    cur_score = _true_score(a)
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    step = 0.5               # base step size (will be normalised by ||grad||)
    min_step = 5e-6
    decay = 0.9995

    alpha = 0.5              # initial soft‑max temperature
    alpha_max = 6.0
    alpha_inc = 0.025        # increase every 100 iterations

    prob_random = 0.008      # probability of a random mass‑transfer move
    iter_no_improve = 0
    max_no_improve = 500

    iteration = 0

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- compute convolution of current vector (via FFT) -----
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[:conv_len]

        # ----- smooth approximation of max (log‑sum‑exp) -----
        max_c = conv.max()
        exp_term = np.exp(alpha * (conv - max_c))
        w = exp_term / exp_term.sum()                     # softmax weights, sum to 1

        # ----- gradient of the smoothed objective -----
        # grad_j = 2 * sum_k w_k * a_{k-j}
        w_fft = np.fft.rfft(w, n=fft_len_grad)
        a_rev_fft = np.fft.rfft(a[::-1], n=fft_len_grad)
        conv_rev = np.fft.irfft(w_fft * a_rev_fft, n=fft_len_grad)[:grad_len]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]          # length n

        # ----- gradient step (normalised) -----
        grad_norm = np.linalg.norm(grad) + 1e-15
        cand = a - (step / grad_norm) * grad
        cand = _project_simplex(cand)

        # ----- evaluate candidate with the true objective -----
        cand_score = _true_score(cand)

        # ----- accept/reject -----
        improved = cand_score < cur_score - 1e-12
        if improved:
            a = cand
            cur_score = cand_score
            iter_no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # occasional uphill acceptance (simulated annealing style)
            if rng.random() < math.exp(-(cand_score - cur_score) / max(step, 1e-8)):
                a = cand
                cur_score = cand_score
            iter_no_improve += 1

        # ----- random mass‑transfer perturbation -----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.25
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                cand2_score = _true_score(cand2)
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    iter_no_improve = 0
                if cand2_score < best_score:
                    best_score = cand2_score
                    best = cand2.copy()

        # ----- periodic jitter to escape stagnation -----
        if iter_no_improve >= max_no_improve:
            jitter = rng.normal(scale=0.01, size=n)
            a = _project_simplex(np.maximum(a + jitter, 0.0))
            cur_score = _true_score(a)
            iter_no_improve = 0

        # ----- schedule updates -----
        step = max(min_step, step * decay)
        if iteration % 100 == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        iteration += 1

        # ----- early termination check -----
        if time.monotonic() + 0.5 > deadline:
            break

    # Return the best vector found as a plain Python list.
    return [float(x) for x in best]
