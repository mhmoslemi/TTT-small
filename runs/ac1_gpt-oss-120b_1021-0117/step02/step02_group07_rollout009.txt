def _conv(seq, fft_len):
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    return conv[:2*seq.size-1]
def _score(seq, fft_len):
    conv = _conv(seq, fft_len)
    max_b = conv.max()
    s = seq.sum()
    if s <=0:
        return float('inf'), conv
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv

import time
import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _conv(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution of seq with itself using FFT (real)."""
    f = np.fft.rfft(seq, n=fft_len)
    c = np.fft.irfft(f * f, n=fft_len)
    return c[: 2 * seq.size - 1]

def _score(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution array)."""
    conv = _conv(seq, fft_len)
    max_b = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv
