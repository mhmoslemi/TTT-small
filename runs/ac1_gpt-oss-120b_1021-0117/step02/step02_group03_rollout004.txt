import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _initial_vector():
    """Return a non‑negative start vector of length n."""
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: simple sqrt‑series of length 256
        n = 256
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            if k == 1:
                s = 0.0
            else:
                s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
            a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.ones_like(a)
    return a


def _score(v):
    """Exact AC1 upper bound for a coefficient vector."""
    n = v.shape[0]
    conv = np.convolve(v, v)
    return 2.0 * n * conv.max() / (v.sum() ** 2)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    a = _initial_vector()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # normalise to the canonical sum
    a = a * (target_sum / a.sum())

    best = a.copy()
    best_score = _score(best)

    # algorithmic parameters
    mix = 0.25          # initial mixing weight
    min_mix = 0.02
    max_iters = 30
    it = 0
    no_improve = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # current autocorrelation peak
        conv_self = np.convolve(a, a)
        M = float(conv_self.max())

        # build the sparse convolution matrix F (shape (2n‑1, n))
        # each diagonal offset i contains the constant a[i]
        diags = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
        offsets = list(range(n))
        F = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")

        # linear programme: maximise 1ᵀg  <=> minimise –1ᵀg
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # solve with HiGHS (fast sparse LP). If it fails we abort the LP loop.
        try:
            res = linprog(
                c,
                A_ub=F,
                b_ub=b_ub,
                bounds=(0, None),
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g0 = np.asarray(res.x, dtype=np.float64)
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # rescale g0 so that its sum equals the target_sum
        g = g0 * (target_sum / sum_g0)

        # convex combination with the current iterate
        a_new = (1.0 - mix) * a + mix * g
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * (target_sum / a_new.sum())   # re‑normalise

            best = a_new.copy()
            no_improve = 0
        else:

        a = a_new

        if no_improve >= 5:

        k = int(conv.argmax())

        g = np.zeros_like(best)
