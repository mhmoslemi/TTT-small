import time
import numpy as np

# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all entries are equal after projection
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Starts from the supplied parent (``height_sequence_1``) or a uniform
    fallback, then repeatedly performs a projected gradient step on the
    active convolution shift together with occasional random swaps and
    jitter.  The best sequence encountered is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Load / initialise a base sequence
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        a = parent.copy()
    except Exception:
        # fallback: uniform of moderate length
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = int(a.size)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # 2.  Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    factor = 2.0 * n                  # constant multiplying max(conv)

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # power‑of‑two for FFT

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Parameters for the optimisation loop
    base_eta = 0.8
    jitter_interval = 80
    swap_prob = 0.12
    it = 0

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution & active shift ------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        kmax = int(np.argmax(conv))
        max_conv = conv[kmax]
        cur_score = factor * max_conv

        # ----- keep the best seen ---------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            # early stop if we have reached the target bound
            if best_score <= 1.50274349:
                break

        # ----- random swap move (exploration) ---------------------------
        if rng.random() < swap_prob:
            i, j = rng.choice(n, size=2, replace=False)
            a_swapped = a.copy()
            a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
            fa_sw = np.fft.rfft(a_swapped, n=fft_len)
            conv_sw = np.fft.irfft(fa_sw * fa_sw, n=fft_len)[:conv_len]
            score_sw = factor * np.max(conv_sw)
            if score_sw < cur_score:
                a = a_sw
                # continue to next iteration (swap already yields an improvement)
                it += 1
                continue

        # ----- gradient of the active convolution shift ------------------
        offset = kmax - (n - 1)
        if offset >= 0:
            idx = np.arange(offset, n)
            partner = kmax - idx
        else:
            idx = np.arange(0, n + offset)
            partner = kmax - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[partner]          # sub‑gradient for active shift

        # ----- line‑search over step sizes -------------------------------
        improved = False
        # geometric trial factors
        for factor_eta in (1.0, 0.5, 0.25, 0.125, 0.0625):
            eta = base_eta * factor_eta / np.sqrt(it + 1.0)
            a_trial = _proj_simplex(a - eta * g)
            fa_trial = np.fft.rfft(a_trial, n=fft_len)
            conv_trial = np.fft.irfft(fa_trial * fa_trial, n=fft_len)[:conv_len]
            trial_score = factor * np.max(conv_trial)
            if trial_score < cur_score:
                a = a_trial
                improved = True
                break
        if not improved:
            # fallback: small step then project
            eta = base_eta / np.sqrt(it + 1.0)
            a = _proj_simplex(a - eta * g)

        it += 1

        # ----- occasional multiplicative jitter to escape plateaus -------
        if it % jitter_interval == 0:
            jitter = np.exp(rng.normal(0.0, 0.006, size=n))
            a = _proj_simplex(a * jitter)

    # ------------------------------------------------------------------
    # 4.  Return the best sequence discovered
    # ------------------------------------------------------------------
    return [float(v) for v in best]
