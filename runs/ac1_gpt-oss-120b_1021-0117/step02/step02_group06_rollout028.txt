    for vec in init_cands:
        sc = _score(vec, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
        if best_score <= target:
            return best_vec.tolist()

    # ----- local optimisation: exponentiated‑gradient descent -----
    def local_optimize(start_vec):
        a = start_vec.copy()
        best_local = a.copy()
        best_local_score = _score(a, n, fft_len)

        iter_no_improve = 0
        alpha0 = 0.6
        min_alpha = 1e-5

        while time.monotonic() < deadline:
            conv = _conv_self_fft(a, n, fft_len)
            max_b = float(conv.max())
            k = int(conv.argmax())
            cur_score = 2.0 * n * max_b

            if cur_score < best_local_score - 1e-12:
                best_local_score = cur_score
                best_local = a.copy()
                iter_no_improve = 0
                if cur_score <= target:
                    break
            else:
                iter_no_improve += 1

            # gradient of the active term b_k
            g = np.zeros_like(a)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
                g[idx_range] = 2.0 * a[k - idx_range]

            # step size (diminishing)
            t = max(1, iter_no_improve + 1)
            alpha = max(min_alpha, alpha0 / np.sqrt(t))

            # exponentiated‑gradient update
            a = a * np.exp(-alpha * g)

            # re‑normalise onto the simplex
            s = a.sum()
            if s <= 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

            # occasional stochastic jitter
            if rng.random() < 0.015:
                a = a * np.exp(rng.normal(0.0, 0.03, size=n))
                a = a / a.sum()

            # restart when stuck
            if iter_no_improve > 3000:
                a = best_local * np.exp(rng.normal(0.0, 0.07, size=n))
                a = a / a.sum()
                iter_no_improve = 0

        return best_local, best_local_score

    # refine the current best
    refined_vec, refined_score = local_optimize(best_vec)
    if refined_score < best_score:
        best_score, best_vec = refined_score, refined_vec.copy()
        if best_score <= target:
            return best_vec.tolist()

    # if time permits, try a couple of random restarts
    while time.monotonic() < deadline and best_score > target:
        rnd = rng.random(n)
        rnd = rnd / rnd.sum()
        opt_vec, opt_score = local_optimize(rnd)
        if opt_score < best_score:
            best_score, best_vec = opt_score, opt_vec.copy()
            if best_score <= target:
                break

    # fallback uniform vector (should never be needed)
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return best_vec.tolist()
