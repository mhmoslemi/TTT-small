import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for cleanup
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # -------------------- initial vector -----------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a short uniform vector (any non‑empty works)
        n0 = 256
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    # make it a probability vector (sum = 1) – scale does not affect the score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size

    # -----------------------------------------------------------
    # helpers for the objective
    # -----------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        """Exact score identical to evaluate_sequence for sum(vec)=1."""
        conv = np.convolve(vec, vec)
        return 2.0 * n * conv.max()

    # -----------------------------------------------------------
    # keep the best solution seen so far
    # -----------------------------------------------------------
    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    # -----------------------------------------------------------
    # Adam‐style moments (helps with noisy sub‑gradients)
    # -----------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                     # base step size
    t = 0                        # iteration counter

    # smoothing factor for the soft‑max (starts gentle, grows)
    alpha = 5.0
    alpha_inc = 1.02
    alpha_max = 200.0

    # bookkeeping for stagnation
    no_improve = 0

    # -----------------------------------------------------------
    # main optimisation loop
    # -----------------------------------------------------------
    while time.monotonic() < deadline:
        t += 1

        # full convolution of the current iterate
        conv = np.convolve(cur, cur)                 # length 2n‑1
        maxc = conv.max()

        # ---------- smooth approximation of the max ----------
        # normalize so the largest entry is 1, then apply a soft‑max
        conv_norm = conv / maxc
        shifted = conv_norm - 1.0                     # ≤ 0, zero at the max
        weights = np.exp(alpha * shifted)            # all ≤ 1
        weights /= weights.sum()                     # sum = 1

        # ---------- gradient of the smoothed max ----------
        # g = 2 * (cur ⋆ weights)  (⋆ = correlation)
        corr = np.correlate(cur, weights, mode='full')
        # correlation length = 3n‑2; the slice starting at (2n‑2) gives n values
        start_idx = 2 * n - 2
        grad = 2.0 * corr[start_idx:start_idx + n]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # descent step (negative gradient direction) and projection
        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_score = _score(cand)

        # accept if it improves (or with a small Metropolis probability)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                no_improve = 0
        else:
            # occasional uphill move to escape plateaus
            prob = np.exp(-(cand_score - cur_score) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
            no_improve += 1

        # ---------- occasional random mass‑transfer move ----------
        if no_improve > 4000 and rng.random() < 0.15:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    if tmp_score < best_score:
                        best, best_score = tmp.copy(), tmp_score
                        no_improve = 0

        # ---------- periodic jitter (restart towards best) ----------
        if t % 2500 == 0:
            jitter = rng.normal(scale=8e-4, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_score = _score(tmp)
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
                    no_improve = 0

        # increase the soft‑max sharpness slowly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_inc)

        # early exit if the target is already beaten
        if best_score < 1.50274349:
            break

    # -----------------------------------------------------------
    # return the best vector seen (as a plain Python list)
    # -----------------------------------------------------------
    return [float(x) for x in best]
