import time

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = 1, non‑neg)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where u_i - cssv_i / i > 0
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # degenerate: return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Gradient from a set of convolution shifts
# ----------------------------------------------------------------------
def _grad_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """Return sum of sub‑gradients for the given shift indices."""
    n = a.size

        else:
            idx = np.arange(0, n + offset)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
    return g
