import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series: a*a ≈ 1 + … + x^{2n-2}
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Gradient of the k‑th convolution entry (subgradient of max)
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Max autocorrelation via FFT (full linear convolution)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, Lfft: int) -> float:
    fv = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative floats whose AC1 bound is low.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ----------------------------------------------------------------
    # Initialise from parent if available, else from sqrt series
    # ----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a0 = _sqrt_series(n).astype(np.float64)

    n = a0.shape[0]
    a0 = a0 / a0.sum() if a0.sum() != 0 else np.full(n, 1.0 / n)
    a0 = _project_simplex(a0)

    # Also prepare a pure sqrt‑series candidate for safety
    sqrt_candidate = _sqrt_series(n).astype(np.float64)
    sqrt_candidate = sqrt_candidate / sqrt_candidate.sum()
    sqrt_candidate = _project_simplex(sqrt_candidate)

    # ------------------------------------------------------------
    # Pre‑compute FFT length and initial best
    # ------------------------------------------------------------
    Lfft = 1 << ((2 * n - 1).bit_length())  # next power of two
    best = a0.copy()
    best_max = _max_convolution(best, Lfft)
    best_score = 2.0 * n * best_max

    # compare with sqrt series
    sqrt_max = _max_convolution(sqrt_candidate, Lfft)
    sqrt_score = 2.0 * n * sqrt_max
    if sqrt_score < best_score:
        best = sqrt_candidate.copy()
        best_max = sqrt_max
        best_score = sqrt_score

    # start point for the search
    a = best.copy()

    # ------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------
    eta0 = 0.7                # initial step magnitude
    target_score = 1.5028    # early‑stop threshold
    iteration = 0

    # ------------------------------------------------------------
    # Main optimisation loop (projected sub‑gradient descent)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # Full convolution and active set
        fv = np.fft.rfft(a, n=Lfft)
        conv_full = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv_full.max())
        cur_score = 2.0 * n * M

        # Active indices (within tiny tolerance of the maximum)
        eps = 1e-12
        active = np.where(conv_full >= M - eps)[0]

        # Sub‑gradient: average over active lags
        grad = np.zeros_like(a)

            grad += _grad_for_k(a, int(k))
        norm = np.linalg.norm(grad)
        if norm > 0.0:
            direction = grad / norm
            eta = eta0 / math.sqrt(iteration)
            cand = a - eta * direction
        else:
            cand = a.copy()

        # Keep feasibility
        cand = _project_simplex(cand)

        # Evaluate candidate
        cand_max = _max_convolution(cand, Lfft)
        cand_score = 2.0 * n * cand_max

        # Keep global best

            best = cand.copy()

        # Accept if improvement, otherwise with a tiny random chance
        if cand_score < cur_score or rng.random() < 0.005:

        if rng.random() < 0.02:

                if dst >= src:

                frac = rng.random() * 0.15
                delta = a[src] * frac
                a[src] -= delta
                a[dst] += delta
                a = _project_simplex(a)

        # Very occasional full random restart (Dirichlet draw)
        if rng.random() < 0.001:
            a = rng.dirichlet(alpha=np.ones(n))
            a = _project_simplex(a)

        # Early exit if we reached the target bound
        if best_score < target_score:
            break

    # ------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best]
