import time
import numpy as np

# ----- utilities -------------------------------------------------------------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a using real FFT (length 2*len(a)-1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    n = a.shape[0]
    return full[: 2 * n - 1]


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style coefficients w_i = C(2i,i)/4**i, normalized later."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _initial_candidates(n: int, parent_arr):
    """Return a small list of plausible start vectors (already sum‑to‑1)."""
    cand = []

    # parent if given
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # Catalan distribution
    w = _catalan_weights(n)
    cand.append(w / w.sum())

    # Power‑parabola shapes a_i ∝ ((i+1)*(n-i))**p
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        v = base ** p
        cand.append(v / v.sum())

    # Sin‑power shapes a_i ∝ sin^p(π*(i+0.5)/n)
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sin_vals ** p
        cand.append(v / v.sum())

    # Random Dirichlet draws (concentration 0.5 gives some sparsity)
    rng = np.random.default_rng(12345)
    for _ in range(6):
        v = rng.dirichlet(np.full(n, 0.5))
        cand.append(v)

    return cand
