import time
import numpy as np

# ---------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, x >= 0)
# ---------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.size
    # sort decreasing
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # guard against numerical drift
    w_sum = w.sum()
    if w_sum == 0:
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ---------------------------------------------------------------
# Helper: max of self‑convolution (full mode) using FFT
# ---------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    """Return (max value, argmax) of full convolution vec*vec."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx
