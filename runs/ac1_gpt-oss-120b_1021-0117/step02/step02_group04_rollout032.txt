        if r < 0.5:

            t = max_b / sum_cur
            idx = np.arange(n)

            mirror[mask] = cur[j[mask]]

            cand = cur + delta
            # clip then renorm

                cand = cur.copy()

        if cand_score < best_score:

            best_score = cand_score
            # maybe break early if target reached

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Classic step a_i = C(2i,i)/4**i, i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _max_conv_and_idx(arr: np.ndarray, n: int, L: int):
    """Return (max value, arg‑max index, full convolution)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    k_max = int(conv.argmax())
    return max_b, k_max, conv


def _solve_lp_active(f: np.ndarray, M: float, top_k: int = 30):
    """
