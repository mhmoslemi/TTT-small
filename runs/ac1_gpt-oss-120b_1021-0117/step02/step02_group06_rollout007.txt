import time
import numpy as np

# -------------------- utilities --------------------

def _simplex_normalize(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex by truncating negatives and renormalising."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the full (length 2*len(a)-1) self‑convolution of a using a real FFT.
    """
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * a.size - 1]

    while time.monotonic() < deadline:

        max_b = float(conv.max())
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            iter_no_improve += 1

        # locate an active index and build a sub‑gradient
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx_range] = 2.0 * a[k - idx_range]

        # diminishing step size
        step = step0 / np.sqrt(iter_no_improve + 1)

        # simple projected sub‑gradient step
        a = _simplex_normalize(a - step * g)

        # occasional random jitter to escape plateaus
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.015, size=n))
            a = _simplex_normalize(a * jitter)

        # restart if stagnation lasts too long
        if iter_no_improve > 1500:
            perturb = np.exp(rng.normal(0.0, 0.04, size=n))
            a = _simplex_normalize(best_vec * perturb)
            iter_no_improve = 0

    # final fallback
    return best_vec.tolist()
