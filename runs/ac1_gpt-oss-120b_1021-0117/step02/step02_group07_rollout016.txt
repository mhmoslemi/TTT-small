def _conv(seq, fft_len):
    f = np.fft.rfft(seq, n=fft_len)
    return np.fft.irfft(f * f, n=fft_len)[:2*len(seq)-1]

def _score_from_conv(seq, conv):
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    return 2.0*len(seq)*conv.max()/(s*s)

        else:
            raise Exception
    except Exception:

    if init is None:

        init = np.empty(n, dtype=np.float64)
        init[0] = 1.0

            init[i] = init[i-1] * (2*i-1) / (2*i)
    else:

    best_conv = _conv(best, fft_len)
    best_score = _score_from_conv(best, best_conv)

            best_conv = conv_c

import numpy as np
import time

def _conv(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Convolution of a sequence with itself using a power‑of‑two FFT."""
    f = np.fft.rfft(seq, n=fft_len)
    return np.fft.irfft(f * f, n=fft_len)[: 2 * len(seq) - 1]


def _score(seq: np.ndarray, conv: np.ndarray) -> float:
    """Exact objective value from the evaluator (without the clipping step)."""
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * len(seq) * conv.max() / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # a safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Build an initial candidate (parent, binomial, or a simple family)
    # ------------------------------------------------------------------
    n_default = 3809
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            init = np.clip(arr, 0.0, 1000.0)
        else:
            raise Exception
    except Exception:
        init = None

    if init is None:
        # binomial seed a_i = C(2i,i)/4^i
        n = n_default
        init = np.empty(n, dtype=np.float64)
        init[0] = 1.0
        for i in range(1, n):
            init[i] = init[i - 1] * (2 * i - 1) / (2 * i)
    else:
        n = init.size

    fft_len = 1 << ((2 * n - 1).bit_length())

    best = init.copy()
    best_conv = _conv(best, fft_len)
    best_score = _score(best, best_conv)

    # ------------------------------------------------------------------
    # 2️⃣  Scan a few simple parametric families (power‑law & exponential)
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)

    # power‑law: a_i = (i+1)^{-p}
    for p in np.linspace(0.05, 2.5, 45):
        cand = idx ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        conv = _conv(cand, fft_len)
        sc = _score(cand, conv)
        if sc < best_score:
            best = cand.copy()
            best_conv = conv
            best_score = sc

    # exponential decay: a_i = exp(-λ·i)
    for lam in np.linspace(5e-5, 0.02, 40):
        cand = np.exp(-lam * (idx - 1))
        cand = np.clip(cand, 0.0, 1000.0)
        conv = _conv(cand, fft_len)
        sc = _score(cand, conv)
        if sc < best_score:
            best = cand.copy()
            best_conv = conv
            best_score = sc

    # ------------------------------------------------------------------
    # 3️⃣  Main improvement loop (projected gradient + random moves)
    # ------------------------------------------------------------------
    current = best.copy()
    current_score = best_score
    current_conv = best_conv

    while time.monotonic() < deadline:
        # ----- gradient of the active convolution entry -----
        current_conv = _conv(current, fft_len)
        kmax = int(np.argmax(current_conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idxs = np.arange(lo, hi + 1, dtype=np.int64)
        js = kmax - idxs
        grad = np.zeros_like(current)
        grad[idxs] = 2.0 * current[js]               # ∂M/∂a_i

        # projected direction (keeps the total sum unchanged)
        mean_g = grad.mean()
        direction = -grad + mean_g

        # ----- step size (large at the start, shrinking linearly) -----
        elapsed = time.monotonic() - start
        total = deadline - start
        step_scale = 0.07 * max(0.0, 1.0 - elapsed / total) + 0.001
        base_step = step_scale * (current.max() + 1.0)

        cand_best = None
        cand_best_score = None
        cand_best_conv = None

        # ----- try a few multiples of the projected step -----
        for mul in (0.5, 1.0, 1.5):
            cand = current + base_step * mul * direction
            np.clip(cand, 0.0, 1000.0, out=cand)
            if cand.sum() < 0.01:
                continue
            conv = _conv(cand, fft_len)
            sc = _score(cand, conv)
            if cand_best_score is None or sc < cand_best_score:
                cand_best = cand
                cand_best_score = sc
                cand_best_conv = conv

        # ----- occasional random perturbation (diversification) -----
        if cand_best is None or rng.random() < 0.07:
            op = rng.integers(0, 5)          # five different perturbations
            cand = current.copy()
            if op == 0:                       # single multiplicative factor
                i = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, 0.22))
                cand[i] *= factor
            elif op == 1:                     # block scaling
                w = rng.integers(1, max(2, n // 25))
                s = rng.integers(0, n - w + 1)
                factor = np.exp(rng.normal(0.0, 0.10))
                cand[s:s + w] *= factor
            elif op == 2:                     # pair mass transfer from the active lag
                i = rng.integers(lo, hi + 1)
                j = kmax - i
                if i != j:
                    frac = rng.uniform(0.05, 0.18)
                    di = cand[i] * frac
                    dj = cand[j] * frac
                    cand[i] -= di
                    cand[j] -= dj
                    tgt = rng.integers(0, n)
                    while tgt == i or tgt == j:
                        tgt = rng.integers(0, n)
                    cand[tgt] += di + dj
            elif op == 3:                     # early‑to‑late mass transfer
                early = max(1, n // 12)
                late = n - early
                i = rng.integers(0, early)
                j = rng.integers(late, n)
                frac = rng.uniform(0.07, 0.22)
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta
            else:                             # local smoothing of two neighbours
                i = rng.integers(0, n - 1)
                avg = (cand[i] + cand[i + 1]) * 0.5
                cand[i] = avg
                cand[i + 1] = avg
            np.clip(cand, 0.0, 1000.0, out=cand)
            if cand.sum() >= 0.01:
                conv = _conv(cand, fft_len)
                sc = _score(cand, conv)
                if cand_best_score is None or sc < cand_best_score:
                    cand_best = cand
                    cand_best_score = sc
                    cand_best_conv = conv

        if cand_best is None:
            continue

        # ----- acceptance (simulated annealing flavour) -----
        accept = False
        if cand_best_score < current_score:
            accept = True
        else:
            temp = max(0.0, 0.1 * (1.0 - elapsed / total))
            if temp > 0.0 and rng.random() < np.exp((current_score - cand_best_score) / (temp + 1e-12)):
                accept = True

        if accept:
            current = cand_best
            current_score = cand_best_score
            current_conv = cand_best_conv
            if cand_best_score < best_score:
                best = cand_best.copy()
                best_score = cand_best_score
                best_conv = cand_best_conv

    # Return a plain Python list (the evaluator expects a list, not a NumPy array)
    return [float(v) for v in best]
