    deadline = time.monotonic() + max(0.0,budget_s-1.0)

    Lfft = 1 << ((2*n-1).bit_length())
    best = a.copy()
    best_score = 2.0*n*_max_convolution(a, Lfft)
    v = np.zeros_like(a)  # momentum
    momentum = 0.9
    base_eta = 0.7
    iteration=0
    while time.monotonic() < deadline:
        iteration += 1
        # compute convolution and max
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv*fv, n=Lfft)[:2*n-1]

        norm = np.linalg.norm(grad)

            direction = grad / norm
            v = momentum * v - eta * direction
            cand = a + v
        else:

        # evaluate

            best_score = cand_score
            best = cand.copy()

import time
import numpy as np

# ------------------------------------------------------------
# Simple simplex projection (non‑negative, sum=1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2}) – fallback init.
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Full convolution via FFT (linear‑time in n log n)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Gradient of the active lag k (sub‑gradient of max convolution)
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad
