def _sqrt_series(n: int) -> np.ndarray:
    """Return the non‑negative step‑function whose self‑convolution is constant = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

        if k == 1:

    a = np.maximum(a, 0.0)
