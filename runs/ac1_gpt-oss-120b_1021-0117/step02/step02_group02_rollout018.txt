import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # degenerate – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Starts from the provided parent ``height_sequence_1`` (if present)
    or from a uniform distribution, then performs a momentum‑accelerated
    sub‑gradient descent with adaptive step sizes and occasional jitter.
    Returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. initial sequence (parent or uniform)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:               # parent unavailable or invalid
        parent = None

    if parent is not None:
        a = parent.copy()
        n = a.shape[0]
    else:
        n = 256
        a = np.ones(n, dtype=np.float64)

    # normalise to a probability vector (sum == 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    # ------------------------------------------------------------------
    # 3. bookkeeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))   # score = 2*n*max_conv
    best_max = best_score / (2.0 * n)

    # uniform fallback (sometimes better than the parent)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        best_max = best_score / (2.0 * n)
        a = uniform.copy()

    target_score = 1.50274349

    # FFT parameters (reuse every iteration)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 4. optimisation parameters
    # ------------------------------------------------------------------
    it = 0
    momentum = np.zeros_like(a)
    beta = 0.9                     # momentum factor
    eta0 = 3.0                     # base step size (larger than before)
    gamma = 0.5                    # polyak scaling
    jitter_interval = 100
    jitter_std = 0.015
    no_improve_limit = 4000       # restart after many stagnant iterations
    last_improve = 0
    top_k = 5                      # number of most offending shifts to aggregate

    # ------------------------------------------------------------------
    # 5. main loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution via FFT (real-valued)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # keep track of the best sequence seen so far
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            best_max = max_conv
            last_improve = it
            if best_score <= target_score:
                break

        # ------------------------------------------------------------------
        # gradient from the most offending shifts
        # ------------------------------------------------------------------
        # pick up to `top_k` shifts with largest convolution values
        if top_k >= conv_len:
            active = np.arange(conv_len, dtype=np.int64)
        else:
            active = np.argpartition(conv, -top_k)[-top_k:]

        g = np.zeros_like(a)
        for k in active:
            offset = int(k) - (n - 1)
            if offset >= 0:
                idx = np.arange(offset, n, dtype=np.int64)
                partner = k - idx
                g[idx] += 2.0 * a[partner]
            else:
                idx = np.arange(0, n + offset, dtype=np.int64)
                partner = k - idx
                g[idx] += 2.0 * a[partner]

        # ------------------------------------------------------------------
        # momentum update
        # ------------------------------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * g
        norm_sq = np.dot(momentum, momentum) + 1e-30

        # ------------------------------------------------------------------
        # step size (combine diminishing schedule with Polyak step)
        # ------------------------------------------------------------------
        eta = eta0 / np.sqrt(it)
        gap = max_conv - best_max
        if gap > 0.0:
            poly = gamma * gap / norm_sq
            eta = max(eta, poly)          # take the larger, but not too large
        eta = min(eta, 5.0)                # safety upper bound

        # ------------------------------------------------------------------
        # gradient step + projection onto simplex
        # ------------------------------------------------------------------
        a = _proj_simplex(a - eta * momentum)

        # ------------------------------------------------------------------
        # occasional jitter to escape plateaus
        # ------------------------------------------------------------------
        if (it % jitter_interval) == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)

        # ------------------------------------------------------------------
        # restart if no improvement for a long while
        # ------------------------------------------------------------------
        if it - last_improve > no_improve_limit:
            # re‑inject a small random perturbation around the best found so far
            pert = rng.normal(0.0, 0.02, size=n)
            a = best * (1.0 + pert)
            a = np.clip(a, 0.0, None)
            a = _proj_simplex(a)
            momentum.fill(0.0)
            last_improve = it

        # early exit if time is about to run out
        if it % 500 == 0 and time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 6. return the best sequence (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
