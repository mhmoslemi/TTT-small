import scipy.sparse as sp

def _max_corr_fft(v: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the full autocorrelation v*v using real FFT."""
    A = np.fft.rfft(v, fft_len)
    B = A * A
    conv = np.fft.irfft(B, fft_len)[: 2 * len(v) - 1]
    return float(conv.max())

def _toeplitz_conv_matrix(a: np.ndarray) -> sp.csc_matrix:
    """
    Sparse matrix representation of the convolution operator:
    (a * g)[k] = sum_j a[k-j] * g[j]  for k=0..2n-2, g length n.
    """
    n = a.shape[0]
    # each offset d (0..n-1) has a constant diagonal equal to a[d]
    data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int64)
    return sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

        new_score, new_vec = candidates[0]

        if new_score + 1e-12 < best_score:      # improvement
            best_score, best_vec = new_score, new_vec.copy()
            iter_no_improve = 0
            # gradually enlarge mixing weight if it helped
            mix_factor = min(0.9, mix_factor * 1.1)
        else:
            iter_no_improve += 1
            # shrink mixing weight on stagnation
            mix_factor = max(0.1, mix_factor * 0.7)

        # occasional random restart from the best so far
        if iter_no_improve >= 6:
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            best_vec = best_vec * perturb
            best_vec = best_vec / best_vec.sum()
            best_score = score(best_vec)
            iter_no_improve = 0
            mix_factor = 0.5

        # early exit if we already beat the target
        if best_score <= 1.50274349:
            break

    # ------------------------------------------------------------------
    # final safeguard: ensure non‑empty, finite, non‑negative output
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    if best_vec.sum() < 0.01:                     # should never happen
        best_vec = np.full_like(best_vec, 1.0 / n)

    return best_vec.tolist()
