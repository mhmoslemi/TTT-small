def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # 5 second buffer

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
        a0 = np.clip(parent, 0.0, 1000.0)
    except Exception:
        n = 256
        a0 = np.ones(n, dtype=np.float64)

    # ensure sum>0
    if a0.sum() < 0.01:
        a0 = np.full(n, 1.0/n, dtype=np.float64)
    else:
        a0 = a0 / a0.sum()   # scale to sum=1

    # initial best
    best = a0.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # also consider uniform if better
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # options: track multiple restarts
    # We'll run a loop of 'runs' each using EG for limited iterations.

    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    
    # Parameter schedule
    total_iters = 0
    # Outer restarts loop
    while time.monotonic() < deadline:
        # Choose start a: sometimes random perturbed parent
        if rng.random() < 0.2:
            # random Dirichlet
            a = rng.random(n)
            a = a / a.sum()
        else:
            # start from best found (to refine) with a small jitter
            a = best.copy()
            # small jitter multiplicative
            jitter = np.exp(rng.normal(0.0, 0.001, size=n))
            a = a * jitter
            a = a / a.sum()

        it_local = 0
        # run limited iterations per restart
        max_local_iters = 2000   # adjust based on remaining time
        while it_local < max_local_iters and time.monotonic() < deadline:
            it_local += 1
            total_iters += 1

            # convolution via FFT
            fa = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
            max_conv = conv.max()
            score = 2.0 * n * max_conv
            # update best
            if score < best_score:
                best = a.copy()
                best_score = score
                # early stop if target
                if best_score <= 1.50274349:
                    break

            # get active indices (within tolerance)
            # Use absolute tolerance relative to max
            tol = max_conv * (1 - 1e-12)
            # Because floats, using >= max_conv - eps
            eps = 1e-14
            active = np.where(conv >= max_conv - eps)[0]

            # compute gradient
            g = np.zeros_like(a)
            if active.size > 0:
                for k in active:
                    start = 0 if k < n else k - (n - 1)
                    end = n - 1 if k >= n else k
                    # indices i from start..end
                    i = np.arange(start, end + 1)
                    partner = k - i
                    g[i] += 2.0 * a[partner]
            else:
                # fallback: use index of max
                k = int(np.argmax(conv))
                start = max(0, k - (n - 1))
                end = min(n - 1, k)
                i = np.arange(start, end + 1)
                partner = k - i
                g[i] += 2.0 * a[partner]

            # step size
            eta = 0.15 / math.sqrt(total_iters)   # adjust
            # exponential update (mirror descent)
            a = a * np.exp(-eta * g)
            # clip min to avoid underflow
            # small positive floor
            a = np.maximum(a, 1e-30)
            a = a / a.sum()

            # occasional jitter
            if total_iters % 250 == 0:
                jitter = np.exp(rng.normal(0.0, 0.003, size=n))
                a = a * jitter
                a = a / a.sum()

        if best_score <= 1.50274349:
            break

    return [float(v) for v in best]

import time
import math
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that lowers the
    AC1 upper bound.  Starts from the provided parent (``height_sequence_1``)
    or from a uniform fallback, then performs a momentum‑based projected
    gradient descent on the maximal autocorrelation value.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. time limit
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. base sequence (parent if available)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        # sanity checks
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        a0 = np.clip(parent, 0.0, 1000.0)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # also keep a uniform vector as a cheap alternative
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------

    target_score = 1.50274349
