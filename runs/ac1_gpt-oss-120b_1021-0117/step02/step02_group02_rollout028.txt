import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Uses a momentum‑accelerated sub‑gradient method with a
    top‑k smoothing of the active convolution peaks.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2.  Base sequence (fallback if the parent is unavailable)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # 3.  Initialise best candidate
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(2.0 * n * np.max(np.convolve(best, best)))
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(2.0 * n * np.max(np.convolve(uniform, uniform)))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 4.  Pre‑compute FFT size
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 5.  Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 0.8               # initial step magnitude
    beta = 0.9               # momentum decay
    top_k = 5                # number of top convolution peaks to smooth
    jitter_every = 120       # steps between random perturbations
    jitter_std = 0.01        # std‑dev of log‑normal jitter
    target_score = 1.50274349
    v = np.zeros_like(a)     # momentum buffer
    it = 0
    no_improve = 0
    max_stale = 3000

    # ------------------------------------------------------------------
    # 6.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # -------- convolution (FFT‑based) ---------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # -------- update the best sequence found -------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # -------- gradient from top‑k convolution peaks ------------------
        # (vectorised accumulation)
        idxs = (
            np.argpartition(conv, -top_k)[-top_k:]
            if conv.size > top_k else np.arange(conv.size)
        )
        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                idx = np.arange(offset, n, dtype=np.int64)
                partner = k - idx
                g[idx] += a[partner]
            else:
                idx = np.arange(0, n + offset, dtype=np.int64)
                partner = k - idx
                g[idx] += a[partner]
        g = 2.0 * g / max(1, len(idxs))

        # -------- momentum step & projection -----------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v)

        # -------- occasional random jitter to escape plateaus ----------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v[:] = 0.0
            no_improve = 0

        # -------- long stagnation → restart around best -----------------
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v[:] = 0.0
            no_improve = 0

    # ------------------------------------------------------------------
    # 7.  Return the best sequence (as plain python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
