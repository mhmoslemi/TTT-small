    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    step0 = 0.6          # initial sub‑gradient step size
    jitter_prob = 0.05   # frequency of global jitter
    jitter_std = 0.025   # std of log‑normal jitter
    restart_every = 3000
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # ----- current objective -----
        F = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        cur_max = float(conv_full.max())
        cur_score = 2.0 * n * cur_max

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            if best_score <= 1.50274349:
                break

        # ----- sub‑gradient step (active max index) -----
        k = int(conv_full.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        g = np.zeros_like(a)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            g[idx] = 2.0 * a[k - idx]

        step = max(1e-4, step0 / np.sqrt(iteration))
        a_candidate = a - step * g
        a_candidate = _project_onto_simplex(a_candidate)

        # ----- occasional global jitter -----
        if rng.random() < jitter_prob:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a_candidate = a_candidate * jitter
            a_candidate = a_candidate / a_candidate.sum()

        # ----- periodic diversification -----
        if iteration % restart_every == 0:
            if rng.random() < 0.3:
                # fresh random Dirichlet‑like point
                rnd = rng.exponential(scale=1.0, size=n)
                a_candidate = rnd / rnd.sum()
            else:
                # perturb current best modestly
                pert = np.exp(rng.normal(0.0, 0.04, size=n))
                a_candidate = best_vec * pert
                a_candidate = a_candidate / a_candidate.sum()

        a = a_candidate

    # final sanity‑check and return
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total <= 0:
        best_vec = np.full_like(best_vec, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total
    return best_vec.tolist()
