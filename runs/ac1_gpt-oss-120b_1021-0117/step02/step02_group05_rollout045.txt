def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _fft_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 score using FFT (scale‑invariant)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

    """Return (kmax, gradient vector g) for the maximal convolution entry."""
    fa = np.fft.rfft(arr, n=L)

    self_mask = mask & (rev != idx)

            d = g.mean() - g
            eps = grad_eps0 * temperature
            cand = cur + eps * d
        # ----- 2) block scaling ---------------------------------------------
        elif move < 0.75:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, block_sigma))
            cand = cur.copy()
            cand[start:start + width] *= factor
        # ----- 3) random pair mass transfer ---------------------------------
        elif move < 0.92:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * transfer_max
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        # ----- 4) smoothing move --------------------------------------------
        else:
            # simple three‑point averaging (convolution with [1,1,1]/3)
            kernel = np.array([1.0, 1.0, 1.0])
            cand = np.convolve(cur, kernel, mode='same') / 3.0

        # ----- enforce constraints -------------------------------------------
        cand = _clip_and_renorm(cand, target_sum)

        # ----- cheap evaluation -----------------------------------------------
        cand_score = _fft_score(cand, n, L)

        # ----- update global best ---------------------------------------------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # early exit if excellent
            if best_score < 1.50274349:
                break

        # ----- simulated‑annealing acceptance ---------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # ----- temperature schedule -------------------------------------------
        temperature *= decay

        # ----- occasional restart from best -----------------------------------
        if rng.random() < start_restart_prob:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ---- final local refinement (tiny gradient steps) ------------------------
    while time.monotonic() < deadline:
        # tiny move towards reducing the maximal entry
        _, g = _gradient_info(best, n, L)
        d = g.mean() - g                # direction that lowers peaks
        cand = best + 0.002 * d
        cand = _clip_and_renorm(cand, target_sum)
        cand_score = _fft_score(cand, n, L)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        else:
            break   # no further improvement

    # ---- safety fall‑back ----------------------------------------------------
    # Ensure the returned sequence is valid for the official evaluator.
    try:
        final_score = evaluate_sequence(best.tolist())
        if not math.isfinite(final_score):
            # fallback to parent or a simple uniform vector
            fallback = np.full(n, target_sum / n, dtype=np.float64)
            best = fallback
    except Exception:
        pass

    return [float(x) for x in best]
