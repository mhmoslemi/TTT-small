import time
import numpy as np

# ------------------------------------------------------------
# projection onto the probability simplex (non‑negative, sum=1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# maximum of full convolution a*a using FFT (linear convolution)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# gradient of convolution entry k with respect to a
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# helper: evaluate the bound 2*n*max(conv) (sum is 1 after projection)
# ------------------------------------------------------------
def _eval_score(a: np.ndarray, n: int, L: int) -> float:
    return 2.0 * n * _max_convolution(a, L)


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---- initialise ------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = np.full(n, 1.0, dtype=np.float64)

    n = a.shape[0]
    a = np.maximum(a, 0.0)
    total = a.sum()
    a = a / total if total > 0 else np.full(n, 1.0 / n, dtype=np.float64)
    a = _project_simplex(a)                     # enforce sum = 1

    # ---- optimisation parameters ------------------------------------
    Lfft = 1 << ((2 * n - 1).bit_length())      # next power of two
    best = a.copy()
    best_score = _eval_score(best, n, Lfft)

    cur = a.copy()
    cur_score = best_score
    it = 0
    eta0 = 0.8                                   # initial step size

    # ---- main search loop -------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution, its max and its lag
        fv = np.fft.rfft(cur, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv.max())
        k = int(conv.argmax())

        # sub‑gradient step aimed at the active lag
        grad = _grad_for_k(cur, k)
        nrm = np.linalg.norm(grad)
        if nrm > 0.0:
            eta = eta0 / np.sqrt(it)
            cand = cur - eta * (grad / nrm)
        else:
            cand = cur.copy()
        cand = _project_simplex(cand)

        cand_score = _eval_score(cand, n, Lfft)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()

        # accept improvements or occasional worsening
        if cand_score < cur_score or rng.random() < 0.01:
            cur = cand
            cur_score = cand_score

        # random mass‑transfer to explore plateaus
        if rng.random() < 0.05:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.2
                delta = cur[i] * frac
                cur[i] -= delta
                cur[j] += delta
                cur = _project_simplex(cur)
                cur_score = _eval_score(cur, n, Lfft)
                if cur_score < best_score:
                    best_score = cur_score
                    best = cur.copy()

        # very occasional reversal (different ordering)
        if rng.random() < 0.005:
            cur = cur[::-1]
            cur = _project_simplex(cur)
            cur_score = _eval_score(cur, n, Lfft)
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()

        # early exit if we already beat the target bound
        if best_score < 1.5028:
            break

    # ----- return best found sequence ---------------------------------
    return [float(x) for x in best]
