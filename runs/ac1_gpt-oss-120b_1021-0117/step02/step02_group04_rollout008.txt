def gradient_move(cur, sum_cur, max_b, k_max):
    n = len(cur)
    t = max_b / sum_cur
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    mirror = np.zeros_like(cur)
    mirror[mask] = cur[j[mask]]
    C = 4.0 * n / (sum_cur * sum_cur)
    # use step_lr variable global
    delta = step_lr * C * (t - mirror)
    cand = cur + delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    # renormalize sum to original sum_cur to maintain sum (just scaling)
    s = cand.sum()
    if s > 0:
        cand *= sum_cur / s
    return cand

import math
import time

import numpy as np

# ---------- helper utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (classic binomial step)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound via FFT (same as official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


# ---------- move operators ----------
def _move_gradient(cur: np.ndarray, sum_cur: float, max_b: float,
                  k_max: int, step_lr: float) -> np.ndarray:
    n = cur.shape[0]
    t = max_b / sum_cur
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    mirror = np.zeros_like(cur)
    mirror[mask] = cur[j[mask]]
    C = 4.0 * n / (sum_cur * sum_cur)
    delta = step_lr * C * (t - mirror)
    cand = cur + delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0:
        cand *= sum_cur / s
    return cand


def _move_block_scale(cur: np.ndarray, rng: np.random.Generator,
                      sum_target: float) -> np.ndarray:
    n = cur.shape[0]
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = math.exp(rng.normal(0.0, 0.13))
    cand = cur.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0:
        cand *= sum_target / s
    return cand


def _move_mass_transfer(cur: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = cur.shape[0]
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:

    frac = rng.random() * 0.45  # up to 45 % of source mass
    delta = cur[i] * frac
    cand = cur.copy()

    return cand


def _move_high_low(cur: np.ndarray, sum_cur: float, max_b: float,
                  k_max: int, rng: np.random.Generator) -> np.ndarray:
    n = cur.shape[0]
    t = max_b / sum_cur
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    mirror = np.zeros_like(cur)
    mirror[mask] = cur[j[mask]]
    high = np.flatnonzero(mirror > t)
    low = np.flatnonzero(mirror < t)
    if high.size == 0 or low.size == 0:

    hi = int(rng.choice(high))

            cur_score = best_score
            temperature = 5.0
            step_lr = 0.25
            iter_no_improve = 0

    return [float(x) for x in best]
