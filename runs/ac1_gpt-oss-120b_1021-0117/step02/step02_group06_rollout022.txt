    n = a.size

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_self_conv(a: np.ndarray, n: int, fft_len: int) -> float:
    """Maximum entry of the a*a convolution (real, full length)."""
    A = np.fft.rfft(a, n=fft_len)
    prod = A * A
    conv = np.fft.irfft(prod, n=fft_len)
    return float(conv[: 2 * n - 1].max())


def _conv_matrix(f: np.ndarray, n: int) -> sp.csc_matrix:
    """
    Build sparse matrix T (shape (2n-1, n)) such that T @ b = f * b
    (linear convolution, full length).  Each diagonal i (0≤i<n) is constant f[i].
    """
    offsets = np.arange(n, dtype=int)
    # each diagonal consists of n copies of the scalar f[i]
    diag_data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    return sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format="csc")
