import numpy as np
import time

# -------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# -------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    """Generate a decreasing positive sequence a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_conv(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution array, max value) for a sequence."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    max_b = float(conv.max())
    s = float(seq.sum())
    if s <= 0.0:
        return float("inf"), conv, max_b
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv, max_b


def _gradient_of_max(seq: np.ndarray, kmax: int) -> np.ndarray:
    """
    Sub‑gradient of the (max‑lag) convolution entry with respect to seq.
    For the lag index kmax, the gradient component i is seq[kmax-i] when
    0 ≤ kmax-i < n, otherwise 0.
    """
    n = seq.shape[0]
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad = np.zeros_like(seq)
    grad[idx] = seq[kmax - idx]
    return grad
