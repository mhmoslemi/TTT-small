def lp_improve(a, M, Ksize=20):
    # a shape (n,)
    n = a.shape[0]
    # compute convolution
    fv = np.fft.rfft(a, n=next_pow2(2*n-1))
    conv = np.fft.irfft(fv * fv, n=next_pow2(2*n-1))[:2*n-1]
    # find top K indices
    idx_sorted = np.argsort(conv)[::-1]
    K = idx_sorted[:Ksize]
    # define constraints: for each k in K: sum_i a_i * g_{k-i} <= M
    # We'll create matrix A of shape (Ksize, n)
    A = np.zeros((Ksize, n), dtype=np.float64)
    for row, k in enumerate(K):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # g indices j = k - i
        # for i in [i_min..i_max], set A[row, k - i] = a[i]
        # Equivalent: set A[row, i_range] = a[i_range reversed? Let's compute.
        # For j = k - i, as i varies from i_min..i_max, j runs from k - i_max .. k - i_min
        # We'll just loop.
        for i in range(i_min, i_max+1):
            j = k - i
            A[row, j] = a[i]
    # define variable
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    # maybe add sum constraint? Not needed.
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    result = prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is not None:
        g_opt = np.maximum(g.value, 0.0)
        # scale to sum=1
        sum_g = g_opt.sum()
        if sum_g > 0:
            g_opt = g_opt / sum_g
        else:
            g_opt = a.copy()
        return g_opt
    else:
        return None

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""

    # final normalisation (ensures sum != 0 for the evaluator)
    best = best / max(best.sum(), 1e-12)
    return [float(x) for x in best]
```
