def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv*fv, n=L)[:2*n-1]
    return conv

def _conv_fft_general(x, y):
    L = 1 << ((len(x)+len(y)-1).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx*fy, n=L)[:len(x)+len(y)-1]
    return conv
```

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full(n, 1.0/n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    rho = rho[-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    rho = rho[-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def _conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x * y using FFT."""
    m = len(x) + len(y) - 1
    L = _next_pow2(m)
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:m]
    return conv
