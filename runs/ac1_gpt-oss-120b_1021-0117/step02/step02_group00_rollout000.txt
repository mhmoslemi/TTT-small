    L = 1 << ((len(u)+len(v)-2).bit_length())
    fu = np.fft.rfft(u, n=L)

import time
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (non‑negative and sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# FFT‑based convolution (full linear convolution)
# ------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return full convolution of a and b using real FFTs."""
    m = len(a) + len(b) - 1
    L = 1 << ((m - 1).bit_length())       # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)[:m]
    return conv


# ------------------------------------------------------------
# Objective (same as evaluate_sequence for sum == 1)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    """2 * n * max(conv) for a vector normalised to sum 1."""
    n = vec.size
    conv = _fft_convolve(vec, vec)
    return 2.0 * n * conv.max()

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function whose AC1 bound is low."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # a small safety margin to avoid hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise from the provided parent if present, otherwise
    # generate a simple exponential decay.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 1024
        a = np.exp(-np.arange(n) / 6.0)

    # Normalise to the simplex (sum = 1). Scaling does not affect the score.
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    best_vec = a.copy()
    best_sc = _score(best_vec)

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    step0 = 0.4            # initial step length
    tau_factor0 = 0.3      # initial temperature as fraction of current max
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and score -----
        conv = _fft_convolve(a, a)
        max_conv = conv.max()
        cur_sc = 2.0 * n * max_conv

        # ----- keep best -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
            if best_sc < 1.50274349:
                break
        else:
            no_improve += 1

        # ----- smooth approximation to max (softmax) -----
        # temperature gradually cools
        tau = max(1e-12, max_conv * max(tau_factor0, 0.08 / np.sqrt(it)))
        # shift by the maximum to avoid overflow
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        # grad_j = 2 * sum_k w_k * a_{k-j}
        rev_a = a[::-1]
        g_full = _fft_convolve(w, rev_a)                     # length 3n-2
        grad = 2.0 * g_full[n - 1 : n - 1 + n]                # slice of length n

        # ----- gradient step with diminishing step size -----
        step = step0 / np.sqrt(it)
        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ----- simple acceptance rule (always keep improvement) -----
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
        else:

            prob = np.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # ----- random mass‑transfer move when stagnating -----
        if no_improve > 3000 and rng.random() < 0.15:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.4
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    a, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                        no_improve = 0

        if it % 4500 == 0 and rng.random() < 0.28:

            tmp = np.maximum(tmp, 0.0)

                a, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

    # Return the best solution found as a plain list of Python floats.
    return [float(x) for x in best_vec]
