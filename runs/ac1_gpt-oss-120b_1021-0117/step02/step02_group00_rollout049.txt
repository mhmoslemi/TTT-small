def _project_simplex(v):
    """Project onto probability simplex sum=1, v >=0."""
    # Implementation based on algorithm above.

    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]

    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]

    L_cross = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = _score_vec(best, L_self)

    cur = a.copy()
    momentum = np.zeros_like(a)

    beta = 0.9          # momentum decay
    eta0 = 0.35         # initial step‑size factor
    alpha = 5.0         # softmax sharpness (starts smooth)
    alpha_max = 200.0
    target = 1.50274349
    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # gradient of smoothed max‑convolution
        g = _grad_softmax(cur, alpha, L_self, L_cross)

        # momentum step
        momentum = beta * momentum + (1.0 - beta) * g

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # projected gradient step
        cand = cur - eta * momentum
        cand = _project_simplex(cand)

        cand_score = _score_vec(cand, L_self)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        cur = cand  # move on

        # --------------------------------------------------------------
        # occasional stochastic move to escape plateaus
        # --------------------------------------------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_vec(tmp, L_self)
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score
                    no_improve = 0
                cur = tmp
            else:
                cur = _project_simplex(cur + rng.normal(scale=1e-3, size=n))

        # --------------------------------------------------------------
        # gentle perturbation of the current best point every few thousand steps
        # --------------------------------------------------------------
        if it % 4000 == 0:
            jitter = rng.normal(scale=2e-3, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_score = _score_vec(tmp, L_self)
            if tmp_score < best_score:
                best = tmp.copy()
                best_score = tmp_score
                no_improve = 0
            cur = tmp

        # --------------------------------------------------------------
        # slowly sharpen the softmax (approach true max)
        # --------------------------------------------------------------
        if it % 800 == 0:
            alpha = min(alpha * 1.06, alpha_max)

        # --------------------------------------------------------------
        # early exit if target already reached
        # --------------------------------------------------------------
        if best_score < target:
            break

    # return as plain Python list
    return [float(x) for x in best]
