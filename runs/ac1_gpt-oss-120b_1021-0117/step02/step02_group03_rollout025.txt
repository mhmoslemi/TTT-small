    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0., budget_s - 5.)
    # initialize a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 256
        a = _sqrt_series(n0).astype(np.float64)
    # Normalize to sum = 1
    a = a / a.sum()
    a = _project_simplex(a)   # ensures sum=1
    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())
    best = a.copy()
    best_score = 2 * n * _max_convolution(a, Lfft) # sum=1 -> denominator=1
    iteration=0
    eta0=0.9
    beta=0.5  # momentum factor? maintain prev update vector.
    momentum = np.zeros_like(a)
    # tolerance for active set: top_k=5
    top_k=5
    while time.monotonic() < deadline:
        iteration += 1
        # compute conv
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2*n-1]
        M = float(conv.max())
        # compute score
        cur_score = 2 * n * M
        # update best
        if cur_score < best_score:
            best_score=cur_score
            best = a.copy()
        # if target reached break
        if best_score < 1.5028:
            break
        # determine active lags
        # find indices of top_k largest conv values
        if top_k >= len(conv):
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(-conv, top_k)[:top_k]
        # compute combined gradient
        grad = np.zeros_like(a)
        for k in idxs:
            grad += _grad_for_k(a, int(k))
        # optional weight by (conv[k]-M0) maybe; but we can just sum.
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = eta0 / np.sqrt(iteration)
            # combine momentum
            momentum = beta * momentum + grad / norm
            cand = a - step * momentum
        else:
            cand = a.copy()
        # project
        cand = _project_simplex(cand)
        # evaluate new candidate score
        cand_M = _max_convolution(cand, Lfft)
        cand_score = 2 * n * cand_M
        # accept if improvement else maybe occasional acceptance
        if cand_score < cur_score or rng.random() < 0.003:
            a = cand
        # occasional random mass transfer
        if rng.random() < 0.015:
            # choose i with high mass and j with low mass
            # Weighted random: sample i proportional to a (high mass)
            i = rng.choice(n, p=a / a.sum())

            frac = rng.random() * 0.2
            delta = a[i] * frac
            a[i] -= delta
            a[j] += delta
            a = _project_simplex(a)
        # occasional restart if stuck (score hasn't improved after many iterations)
        # maintain counter of no-improvement steps

import time
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Simple analytic initialization (fallback if no parent)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Maximum of (full) autocorrelation using FFT (real‑to‑real)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Gradient of the autocorrelation at a specific lag k.
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative coefficients with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------------
    # Initialise – use the provided parent if present, otherwise the sqrt series
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = _sqrt_series(256).astype(np.float64)

    # Normalise to sum = 1 and enforce simplex feasibility
    a = a / a.sum()
    a = _project_simplex(a)

    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())          # next power of two for FFT
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(a, Lfft)   # sum = 1 -> denominator = 1

    # --------------------------------------------------------------------
    # Parameters for the search
    # --------------------------------------------------------------------
    iteration = 0
    eta0 = 0.9                     # initial step magnitude
    top_k = 5                      # how many active lags to aggregate
    no_improve = 0
    restart_after = 3000           # iterations with no improvement -> random restart
    momentum = np.zeros_like(a)    # simple momentum term

    while time.monotonic() < deadline:
        iteration += 1

        # ------------------------------------------------------------
        # Current autocorrelation and active lags
        # ------------------------------------------------------------
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M

        # record best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_score < 1.5028:

        # ------------------------------------------------------------
        # Use a relative tolerance to capture ties

        active_idxs = np.nonzero(active_mask)[0]
        if active_idxs.size > top_k:

            active_idxs = np.argpartition(-conv, top_k)[:top_k]

        # ------------------------------------------------------------

            # pick a destination far enough away (avoid immediate neighbours)

            a[src] -= delta
            a[dst] += delta
            a = _project_simplex(a)

        # ------------------------------------------------------------
        # Acceptance rule (mostly greedy, small chance to accept worsening)
        # ------------------------------------------------------------
        cand_M = _max_convolution(cand, Lfft)
        cand_score = 2.0 * n * cand_M
        if cand_score < cur_score or rng.random() < 0.003:
            a = cand

        # ------------------------------------------------------------
        # Random restart if stuck for a while
        # ------------------------------------------------------------
        if no_improve >= restart_after:
            # new random point (Dirichlet with all parameters = 1)
            a = rng.exponential(scale=1.0, size=n)
            a = a / a.sum()
            a = _project_simplex(a)
            momentum.fill(0.0)
            no_improve = 0

    # --------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # --------------------------------------------------------------------
    return [float(x) for x in best]
