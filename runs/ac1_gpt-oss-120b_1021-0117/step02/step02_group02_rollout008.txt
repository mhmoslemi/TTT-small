def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    # get parent if available
    n = ...
    a = initial sequence ...
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    # target early exit
    target_score = 1.50274349
    # parameters
    eta0 = 0.5   # base step
    beta = 0.9   # momentum coefficient
    vel = np.zeros_like(a)
    # precompute conv length
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # compute conv via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break
        # compute gradient from active shifts
        # tolerance: relative 1e-12 * max_val? if max_val is zero? but not.
        tol = max_val * 1e-12 + 1e-15
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        # compute contributions for each active k
        offset_vals = active - (n - 1)  # shape (m,)
        # We'll vectorize: for each k, we can accumulate contributions.
        # simple loop is OK; active size is small (1 or few)
        for k, offset in zip(active, offset_vals):
            if offset >= 0:
                i_start = offset
                i_end = n  # exclusive
                idx = np.arange(i_start, i_end)
                partner = k - idx
                grad[idx] += 2.0 * a[partner]
            else:
                i_end = n + offset
                idx = np.arange(0, i_end)
                partner = k - idx
                grad[idx] += 2.0 * a[partner]
        # momentum
        eta = eta0 / np.sqrt(iteration)
        vel = beta * vel + eta * grad
        # update
        a_new = a - vel
        # project onto simplex
        a = _proj_simplex(a_new)
        # occasional jitter after some iterations
        if iteration % 100 == 0:
            jitter = np.exp(rng.normal(0.0, 0.01, size=n))
            a = _proj_simplex(a * jitter)
    return [float(v) for v in best]

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
