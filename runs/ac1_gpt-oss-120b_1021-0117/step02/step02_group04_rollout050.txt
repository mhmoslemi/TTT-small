import time, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _ac1_score_fft(arr, n):
    L = _next_pow2(2*n-1)
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s*s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)
    # determine length
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256  # fallback length
    # candidate uniform
    uniform = np.full(n, 1.0, dtype=np.float64)
    # evaluate
    best = uniform
    best_score = _ac1_score_fft(best, n)
    # maybe keep parent if better
    if parent is not None:
        parent_score = _ac1_score_fft(parent, n)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    # simple local improvement: try slight smoothing of best using moving average
    cur = best.copy()

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: fast AC1 score (identical to the official evaluator)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int) -> float:
    """Return the AC1 upper bound for a real‑valued array."""
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound defined by ``evaluate_sequence``.
    The implementation works for any parent sequence ``height_sequence_1``
    that the harness may provide; otherwise it falls back to a uniform
    construction of length 256.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Determine problem size and a reasonable starting point
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256

    # Uniform candidate (any constant works – we use 1.0)
    uniform = np.full(n, 1.0, dtype=np.float64)

    # ------------------------------------------------------------------
    # Choose the better of the two obvious baselines
    # ------------------------------------------------------------------
    best = uniform.copy()
    best_score = _ac1_score_fft(best, n)

    if parent is not None:
        parent_score = _ac1_score_fft(parent, n)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # Simple stochastic improvement loop.
    # ------------------------------------------------------------------
    cur = best.copy()
    cur_score = best_score

    # parameters for the annealing style search
    temperature = 1.0
    temp_decay = 0.9997
    frac_range = (0.01, 0.30)       # fraction of mass to move
    max_iter = 0

    while time.monotonic() < deadline:
        max_iter += 1

        # ---- propose a random mass‑transfer move ----
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        frac = float(rng.random()) * (frac_range[1] - frac_range[0]) + frac_range[0]
        delta = cur[i] * frac
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        # ---- evaluate candidate ----
        cand_score = _ac1_score_fft(cand, n)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # ---- simulated‑annealing acceptance ----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # ---- update temperature ----
        temperature *= temp_decay

        # occasional reset to best (helps escaping plateaus)
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ------------------------------------------------------------------
    # Return the best sequence discovered as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
