import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    # shift by its mean then clip, then renormalise the positive part
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """
    Return a non‑negative step‑function (list of floats) that minimises the
    AC1 upper bound as measured by ``evaluate_sequence``.
    The theoretical optimum on the simplex is the uniform vector, so we
    construct that (or fall back to the provided parent if it is already
    at least as good).
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # keep headroom

    # ------------------------------------------------------------------
    # 1.  Try to use the supplied parent (height_sequence_1)
    # ------------------------------------------------------------------
    parent_vec = None
    n = None
    try:
        # height_sequence_1 is supplied by the harness when available
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 0 and np.isfinite(raw).all():
            parent_vec = np.clip(raw, 0.0, 1000.0)
            n = int(parent_vec.size)
    except Exception:
        parent_vec = None

    # ------------------------------------------------------------------
    # 2.  Choose a length if we could not infer one from the parent
    # ------------------------------------------------------------------
    if n is None:
        n = 256                     # modest default that fits easily in budget

    # ------------------------------------------------------------------
    # 3.  Construct the uniform simplex vector (the provable optimum)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 4.  Evaluate both candidates (parent, uniform) and keep the best
    # ------------------------------------------------------------------
    best_vec = uniform
    best_score = float(evaluate_sequence(best_vec.tolist()))

    if parent_vec is not None:
        # Normalise the parent onto the simplex (sum = 1) for a fair comparison
        s = parent_vec.sum()
        if s >= 0.01:
            parent_norm = parent_vec / s
        else:
            parent_norm = uniform.copy()
        # Simple projection onto the simplex guards against tiny negative rounding
        parent_norm = _proj_simplex(parent_norm)
        parent_score = float(evaluate_sequence(parent_norm.tolist()))
        if parent_score < best_score:
            best_vec = parent_norm
            best_score = parent_score

    # ------------------------------------------------------------------
    # 5.  Time‑guard: if we happened to run out of time, just return what we have
    # ------------------------------------------------------------------
    if time.monotonic() >= deadline:
        return [float(v) for v in best_vec]

    # ------------------------------------------------------------------
    # 6.  (Optional) small random jitter – can only increase the score,
    #     but we keep it for completeness in case the parent was already
    #     better and we want to explore a tiny neighbourhood.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    for _ in range(3):          # at most a few cheap trials
        if time.monotonic() >= deadline:
            break
        jitter = np.exp(rng.normal(0.0, 0.001, size=n))
        trial = _proj_simplex(best_vec * jitter)
        trial_score = float(evaluate_sequence(trial.tolist()))
        if trial_score < best_score:
            best_vec = trial
            best_score = trial_score

    # ------------------------------------------------------------------
    # 7.  Return the best sequence found as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best_vec]
