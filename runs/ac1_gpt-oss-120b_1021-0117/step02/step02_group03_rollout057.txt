    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-5.0)

    if "height_sequence_1" in globals():

    n = a.shape[0]
    a = np.maximum(a, 0.0)
    a = a / (a.sum() + 1e-12)
    a = _project_simplex(a)

    Lfft = 1 << ((2*n - 1).bit_length())

    best = a.copy()
    best_score = 2.0 * n * _max_convolution(a, Lfft)  # since sum=1

    # optimization parameters
    eta0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iter = 0
    topk = 5

    while time.monotonic() < deadline:
        # compute convolution and max
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[:2*n-1]
        M = conv.max()

        # Determine top peaks
        # Use argpartition for speed
        # If 2n-1 may be large, we can just get topk.
        if topk < len(conv):
            idxs = np.argpartition(conv, -topk)[-topk:]
        else:
            idxs = np.arange(len(conv))
        # compute gradient sum for these peaks
        grad = np.zeros_like(a)
        for k in idxs:
            grad += _grad_for_k(a, int(k))
        norm = np.linalg.norm(grad)
        if norm > 0:
            # Adam-like update
            g = grad / norm
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1**(iter+1))
            v_hat = v / (1 - beta2**(iter+1))
            step = eta0 / (np.sqrt(v_hat) + eps)
            delta = - step * m_hat
            cand = a + delta
        else:
            cand = a.copy()

        # enforce symmetry (optional)
        cand = (cand + cand[::-1]) * 0.5

        # project onto simplex
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_score = 2.0 * n * _max_convolution(cand, Lfft)

            best = cand.copy()

        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------------
    # initialise – use the provided parent if present, otherwise sqrt series
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = _sqrt_series(256).astype(np.float64)

    n = a.shape[0]
    a = np.maximum(a, 0.0)
    a = a / (a.sum() + 1e-12)          # normalise to sum = 1
    a = _project_simplex(a)            # safety projection

    # --------------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------------
    Lfft = 1 << ((2 * n - 1).bit_length())   # next power of two for convolution
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(a, Lfft)

    # optimisation parameters
    eta0 = 0.8                # base step magnitude
    topk = 5                  # how many of the largest lags to use
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # current convolution and its maximum
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M

        # indices of the top‑k largest autocorrelation values
        if topk < conv.size:
            idxs = np.argpartition(conv, -topk)[-topk:]
        else:
            idxs = np.arange(conv.size)

        # sub‑gradient (sum of gradients for the selected lags)
        grad = np.zeros_like(a)
        for k in idxs:
            grad += _grad_for_k(a, int(k))

        norm = np.linalg.norm(grad)
        if norm > 0.0:
            # simple diminishing‑step sub‑gradient update
            step = eta0 / np.sqrt(it)
            cand = a - step * (grad / norm)
        else:
            cand = a.copy()

        # symmetry often helps flatten the autocorrelation
        cand = (cand + cand[::-1]) * 0.5

        # enforce feasibility (non‑negative, sum = 1)
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_score = 2.0 * n * _max_convolution(cand, Lfft)

        # keep the best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            if best_score < 1.5028:       # early exit if target reached
                break

        # accept the move if it improves or with a tiny exploration probability
        if cand_score < cur_score or rng.random() < 0.005:
            a = cand

        # occasional random mass‑transfer to escape plateaus
        if rng.random() < 0.02:
            src = rng.integers(0, n)
            if a[src] > 0.0:
                dst = rng.integers(0, n)
                if dst != src:
                    frac = rng.random() * 0.15
                    delta = a[src] * frac
                    a[src] -= delta
                    a[dst] += delta
                    a = _project_simplex(a)
                    a = (a + a[::-1]) * 0.5

    # --------------------------------------------------------------------
    # return the best sequence found as a plain Python list
    # --------------------------------------------------------------------
    return [float(x) for x in best]
