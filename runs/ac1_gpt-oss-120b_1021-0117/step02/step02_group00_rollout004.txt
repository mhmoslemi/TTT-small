def _gradient_smooth(a, beta):
    n = a.size
    B = _conv_fft(a)
    # compute w = softmax(beta * B)
    caps = np.max(beta * B)
    expB = np.exp(beta * B - caps)  # stable
    w = expB / np.sum(expB)
    # gradient g_i = 2 * Σ_j w_{i + j} * a_j
    # compute conv of a (size n) and w[::-1] (size 2n-1) full
    conv = np.convolve(a, w[::-1], mode='full')
    # indexing: g_i = 2 * conv[(len(w)-1) - i]
    offset = (len(w) - 1)
    g = 2 * conv[offset - np.arange(n)]
    return g, B.max()

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast full convolution via FFT (real)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Score (matches evaluate_sequence exactly)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.size
    conv = _conv_fft(a)
    maxb = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * maxb / (s * s))


# ------------------------------------------------------------
# Generate the analytic sqrt‑series (used when no parent is given)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Sub‑gradient of max‑convolution (averaged over near‑max peaks)
# ------------------------------------------------------------
def _subgradient(conv: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.size
    maxb = conv.max()
    # take all indices within relative tolerance of the maximum
    tol = maxb * 1e-8
    act = np.where(conv >= maxb - tol)[0]
    g = np.zeros_like(a)
    for k in act:

        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    if act.size > 0:
        g /= act.size
    return g
