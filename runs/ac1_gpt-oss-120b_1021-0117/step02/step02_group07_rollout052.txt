def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep small safety.

    # Determine n and starting sequence
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # define init seq: binomial
    def initial_binomial(n):
        a = np.empty(n, dtype=np.float64)
        a[0]=1.0
        for i in range(1,n):
            a[i] = a[i-1] * (2*i-1)/(2*i)
        return a

    best = initial_binomial(n)
    sum_best = best.sum()
    fft_len = 1 << ((2*n-1).bit_length())
    def conv_max(seq):
        f = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(f*f, n=fft_len)[:2*n-1]
        return conv.max()
    best_max = conv_max(best)
    best_score = 2.0 * n * best_max / (sum_best**2)

    if parent is not None:
        p_max = conv_max(parent)
        p_score = 2.0 * n * p_max / (parent.sum()**2)
        if p_score < best_score:
            best = parent.copy()
            best_max = p_max
            best_score = p_score

    # current = best
    cur = best.copy()
    cur_sum = cur.sum()
    cur_score = best_score

    # Force sum normalization to 1 to simplify; we can just keep original sum as we maintain it via projection step.
    original_sum = cur_sum

    # Main loop
    while time.monotonic() < deadline:
        # compute conv of current
        f = np.fft.rfft(cur, n=fft_len)
        conv_full = np.fft.irfft(f*f, n=fft_len)[:2*n-1]
        kmax = int(np.argmax(conv_full))
        # compute gradient
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = cur.copy()
        grad[mask] += cur[j[mask]]
        # direction = -grad + mean(grad) (sum zero)
        mean_g = np.mean(grad)
        direction = -grad + mean_g

        # step size: start with step factor based on maybe 0.1 * (max(cur)+average|direction|)
        max_val = cur.max()
        step_scale = max_val + 1.0
        # schedule: diminishing step as time passes
        elapsed = time.monotonic() - start
        remaining = deadline - start
        progress = elapsed / (deadline - start)
        # Use step factor maybe 0.1 * (1 - progress) + 0.005
        base_step = (0.12 * (1.0 - progress) + 0.005) * step_scale

        cand = cur + base_step * direction

        # clip and renormalize
        np.clip(cand, 0.0, 1000.0, out=cand)
        sum_cand = cand.sum()
        if sum_cand > 0:
            cand *= (original_sum / sum_cand)
        else:
            cand = cur.copy()

        # evaluate cand
        cand_max = conv_max(cand)
        cand_score = 2.0 * n * cand_max / (original_sum**2)  # use original sum (same as cand sum after scaling)

        # acceptance: if better or with small prob
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            # store best
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                best_max = cand_max
        else:
            # perhaps accept with small probability decreasing
            temp = 0.01 * (1.0 - progress)  # smaller temperature
            if temp > 0 and rng.random() < np.exp((cur_score - cand_score) / (temp + 1e-12)):
                cur = cand
                cur_score = cand_score

        # occasional random perturbation for jitter
        if rng.random() < 0.02:
            # random location scaling
            idx_rand = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.15))
            cur[idx_rand] *= factor
            np.clip(cur, 0.0, 1000.0, out=cur)
            # renorm to original sum
            cur_sum = cur.sum()
            if cur_sum > 0:
                cur *= (original_sum / cur_sum)

            cur_score = 2.0 * n * conv_max(cur) / (original_sum**2)

            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

    # return best list
    return [float(x) for x in best]

import numpy as np
import time

def _binomial_initial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function height list with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---------------------------------------------------------------
    # Determine problem size and possible parent sequence
    # ---------------------------------------------------------------
    n = 3809               # fallback length
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = int(parent.size)
    except Exception:
        parent = None

    # ---------------------------------------------------------------
    # Helper: full autocorrelation via FFT (returns only needed slice)
    # ---------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    def _full_conv(seq: np.ndarray) -> np.ndarray:
        f = np.fft.rfft(seq, n=fft_len)
        return np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]

    # ---------------------------------------------------------------
    # Initialise best candidate (parent vs binomial seed)
    # ---------------------------------------------------------------
    best = _binomial_initial(n)
    best_sum = best.sum()
    conv_best = _full_conv(best)
    best_max = conv_best.max()
    best_score = 2.0 * n * best_max / (best_sum ** 2)

    if parent is not None:
        p_sum = parent.sum()
        conv_parent = _full_conv(parent)
        p_max = conv_parent.max()
        p_score = 2.0 * n * p_max / (p_sum ** 2)
        if p_score < best_score:
            best = parent.copy()
            best_sum = p_sum
            best_max = p_max
            best_score = p_score
            conv_best = conv_parent

    # ---------------------------------------------------------------
    # Current working copy (will be moved around)
    # ---------------------------------------------------------------
    cur = best.copy()
    cur_sum = best_sum               # invariant (scale does not matter)
    cur_score = best_score
    orig_sum = cur_sum               # keep a constant for normalization

    # small counters for stagnation handling
    no_improve_iters = 0

    # ---------------------------------------------------------------
    # Main optimisation loop
    # ---------------------------------------------------------------
    while time.monotonic() < deadline:
        # --- convolution of current sequence (for gradient) ---
        conv_cur = _full_conv(cur)
        kmax = int(np.argmax(conv_cur))

        # --- gradient of the maximal autocorrelation entry ---
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur)
        # derivative of max entry is 2 * a_{kmax-i} where defined
        grad[mask] = 2.0 * cur[j[mask]]

        # --- descent direction with zero‑sum constraint ---
        mean_g = grad.mean()
        direction = -grad + mean_g               # sum(direction) == 0

        # --- step size schedule (large early, shrink later) ---
        elapsed = time.monotonic() - start
        progress = elapsed / (deadline - start)  # 0 → 1
        # base step factor falls from ~0.25 to ~0.01
        base = 0.25 * (1.0 - progress) + 0.01
        step = base * (cur.max() + 0.1)

        # --- candidate via gradient step ---
        cand = cur + step * direction
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s > 0.0:
            cand *= (orig_sum / s)
            np.clip(cand, 0.0, 1000.0, out=cand)
        else:
            cand = cur.copy()

        # --- evaluate candidate ---
        conv_cand = _full_conv(cand)
        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max / (orig_sum ** 2)

        # --- acceptance (simple descent + occasional uphill) ---
        improved = cand_score < cur_score
        if not improved:
            # tiny temperature for rare uphill moves
            temp = 0.02 * (1.0 - progress)
            if temp > 0 and rng.random() < np.exp((cur_score - cand_score) / (temp + 1e-12)):
                improved = True

        if improved:
            cur = cand
            cur_score = cand_score
            no_improve_iters = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                best_max = cand_max
        else:
            no_improve_iters += 1

        # -----------------------------------------------------------
        # occasional random perturbation to escape plateaus
        # -----------------------------------------------------------
        if no_improve_iters > 2500 and rng.random() < 0.6:
            # shake a random block (multiplicative)
            width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            cur[start_idx:start_idx + width] *= factor
            np.clip(cur, 0.0, 1000.0, out=cur)
            cur_sum = cur.sum()
            if cur_sum > 0.0:
                cur *= (orig_sum / cur_sum)
            cur_score = 2.0 * n * _full_conv(cur).max() / (orig_sum ** 2)
            no_improve_iters = 0

    # ---------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ---------------------------------------------------------------
    return [float(v) for v in best]
