import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# gradient of conv_k = sum_i a_i a_{k-i} at the active index k
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# max of the (full) convolution a*a using FFT
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())
