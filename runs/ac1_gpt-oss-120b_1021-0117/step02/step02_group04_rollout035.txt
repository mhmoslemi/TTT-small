import time
import math
import numpy as np

# ---------------------------------------------------------
# small utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _binomial_step(n: int) -> np.ndarray:
    """Classic binomial step a[i] = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 score (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


# ---------------------------------------------------------

        parent = None

    L = _next_pow2(2 * n - 1)

    binom = np.clip(binom, 0.0, 1000.0)
    # normalise to unit sum – the score is scale‑invariant
    binom /= binom.sum()

    if parent is not None:
        cur0 = np.clip(parent.astype(np.float64), 0.0, 1000.0)
        cur0 /= cur0.sum()
        # pick whichever start is already better

        score_parent = _ac1_score_fft(cur0, n, L)
        if score_parent < score_binom:

        else:
            base = binom
