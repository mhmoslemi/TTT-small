    if s < 1e-12:
        base = np.full(n, 1.0 / n)
        s = 1.0
    a = base / s                 # normalise to sum = 1

    # length sufficient for convolutions up to 3n‑2
    L = _next_pow2(3 * n)

    def _score(x: np.ndarray) -> float:
        """Fast evaluation assuming sum(x)=1."""
        A = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()

    best = a.copy()
    best_score = _score(best)

    cur = a.copy()
    cur_score = best_score

    # ----------- hyper‑parameters -----------
    lr = 0.15            # base step size for gradient moves
    lr_decay = 0.9996
    temperature = 1.0
    temp_decay = 0.9995
    alpha = 8.0          # soft‑max sharpness

    # ------------------- main loop -------------------
    while time.monotonic() < deadline:
        # current autocorrelation (full)
        A_cur = np.fft.rfft(cur, n=L)
        conv_cur = np.fft.irfft(A_cur * A_cur, n=L)[: 2 * n - 1]
        max_b = float(conv_cur.max())
        cur_score = 2.0 * n * max_b          # sum(cur) == 1

        # keep global best
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            if best_score < 1.50274349:
                break

        # ----- propose a new candidate -----
        move = rng.random()
        if move < 0.55:                     # soft‑max gradient step
            # soft‑max over convolution values
            w = np.exp(alpha * (conv_cur - max_b))
            w_sum = w.sum()
            w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

            # gradient of the weighted sum of convolutions
            Wr = w[::-1]
            W_fft = np.fft.rfft(Wr, n=L)
            grad_full = np.fft.irfft(A_cur * W_fft, n=L)[: 3 * n - 2]
            grad = 2.0 * grad_full[2 * n - 2 - np.arange(n)]
            gnorm = np.linalg.norm(grad)
            cand = cur - lr * grad / gnorm if gnorm > 0.0 else cur.copy()

        elif move < 0.80:                  # random block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor

        else:                               # mass transfer between two indices
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * 0.4         # up to 40 % of source mass
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # ----- project back to feasible set -----
        cand = np.maximum(cand, 0.0)
        tot = cand.sum()
        if tot < 1e-12:                     # protection against collapse
            cand = best.copy()
            tot = cand.sum()
        cand /= tot                          # enforce sum = 1

        # ----- evaluate candidate -----
        cand_score = _score(cand)

        # ----- simulated‑annealing acceptance -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # ----- schedule updates & occasional restart -----
        lr *= lr_decay
        temperature *= temp_decay
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score

    # return the best sequence found as plain Python floats
    return [float(v) for v in best]
