def project_to_simplex(v):
    """Project vector v (nonnegative) onto simplex (sum=1). Using algorithm from Wang & Carreira-Perpinan."""
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1   # index in Python 0-index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # v non-negative? but might be any real.
    # Return w >=0 with sum=1 and minimal Euclidean distance to v.
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero?
        return np.full(n, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _binomial_initial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1, then normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety – renormalise exactly to 1
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def _full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full (linear) convolution a*a using real FFT of supplied length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Compute the AC1 upper bound from a convolution array."""
    max_b = conv.max()
    if total <= 0.0:
        return np.inf
    return 2.0 * n * max_b / (total * total)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic / projected‑gradient search for a low‑AC1 step‑function.
    Returns a plain Python list of non‑negative coefficients.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a tiny safety margin for the harness timeout
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # --------------------------------------------------------------
    # Problem size and optional parent
    # --------------------------------------------------------------
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # --------------------------------------------------------------
    # Initialise current / best sequence (normalised to sum = 1)
    # --------------------------------------------------------------
    if parent is not None:
        cur = parent.astype(np.float64)
    else:
        cur = _binomial_initial(n)

    cur = _project_simplex(cur)          # ensure exact simplex condition
    fft_len = 1 << ((2 * n - 1).bit_length())

    conv_cur = _full_conv(cur, fft_len)
    best = cur.copy()
    best_score = _score_from_conv(conv_cur, cur.sum(), n)
    best_conv = conv_cur.copy()

    # --------------------------------------------------------------
    # Main search loop
    # --------------------------------------------------------------
    # parameters
    base_step = 0.2                     # basic step magnitude for gradient move
    uniform = np.full(n, 1.0 / n)       # stored for occasional mixing

    while time.monotonic() < deadline:
        # recompute convolution of the current iterate
        conv_cur = _full_conv(cur, fft_len)
        total_cur = cur.sum()          # should be 1 (up to rounding)
        cur_score = _score_from_conv(conv_cur, total_cur, n)

        # locate a maximal entry in the convolution
        kmax = int(np.argmax(conv_cur))

        # random choice of elementary operation
        op = rng.integers(0, 6)

        cand = cur.copy()

        if op == 0:  # block multiplicative scaling
            width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.06))
            cand[start_idx:start_idx + width] *= factor

        elif op == 1:  # single entry scaling
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.18))
            cand[idx] *= factor

        elif op == 2:  # move a small fraction from the most contributing pair
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            idxs = np.arange(lo, hi + 1)
            js = kmax - idxs
            contrib = cand[idxs] * cand[js]
            # pick the pair with largest contribution
            best_pair = np.argmax(contrib)
            i = idxs[best_pair]
            j = js[best_pair]
            if i != j:
                frac = rng.uniform(0.05, 0.12)
                di = cand[i] * frac
                dj = cand[j] * frac
                cand[i] -= di
                cand[j] -= dj
                # relocate the removed mass to a random low‑value position
                tgt = rng.integers(0, n)
                while tgt == i or tgt == j:
                    tgt = rng.integers(0, n)
                cand[tgt] += di + dj

        elif op == 3:  # smooth a neighbour pair (averaging)
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        elif op == 4:  # early‑to‑late mass transfer
            early_end = max(1, n // 12)
            late_start = n - early_end
            i = rng.integers(0, early_end)
            j = rng.integers(late_start, n)
            frac = rng.uniform(0.07, 0.22)
            delta = cand[i] * frac
            cand[i] -= delta
            cand[j] += delta

        else:  # projected sub‑gradient step on the maximal convolution entry
            # gradient of the maximal convolution entry
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            idxs = np.arange(lo, hi + 1, dtype=np.int64)
            grad = np.zeros_like(cand)
            grad[idxs] = 2.0 * cand[kmax - idxs]          # dM/da_i
            # step size shrinks with elapsed time
            elapsed = time.monotonic() - start
            total = deadline - start
            frac_time = max(0.0, 1.0 - elapsed / total)
            step = base_step * frac_time * rng.uniform(0.5, 1.5)
            cand = cand - step * grad

        # ----------------------------------------------------------
        # Projection onto the probability simplex (keeps sum = 1)
        # ----------------------------------------------------------
        cand = _project_simplex(cand)

        # Occasionally blend a tiny amount of uniform mass (helps escape plateaus)
        if rng.random() < 0.03:
            blend = rng.uniform(0.001, 0.005)
            cand = (1.0 - blend) * cand + blend * uniform
            cand = _project_simplex(cand)

        # ----------------------------------------------------------
        # Evaluate candidate
        # ----------------------------------------------------------
        conv_cand = _full_conv(cand, fft_len)
        cand_score = _score_from_conv(conv_cand, cand.sum(), n)

        # ----------------------------------------------------------
        # Keep the best ever seen (greedy)
        # ----------------------------------------------------------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            best_conv = conv_cand.copy()

        # ----------------------------------------------------------
        # Acceptance for the current iterator (simulated annealing style)
        # ----------------------------------------------------------
        # Simple temperature schedule
        temp = 0.08 * (1.0 - (time.monotonic() - start) / (deadline - start))
        accept = False
        if cand_score < cur_score:
            accept = True
        elif temp > 0.0 and rng.random() < np.exp((cur_score - cand_score) / (temp + 1e-12)):
            accept = True

        if accept:
            cur = cand
            # cur_score will be recomputed at loop start

        # safety break if we run out of time inside heavy computation
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found – plain Python list for the evaluator
    # ------------------------------------------------------------------
    return [float(x) for x in best]
