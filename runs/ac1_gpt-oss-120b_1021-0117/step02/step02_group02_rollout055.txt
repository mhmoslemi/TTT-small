import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries summing to 1).  Implements the algorithm of
    Duchi et al. (2008).  Returns a new array of the same shape.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation of the AC1 upper bound.
    A smooth‑max gradient (softmax) with Adam‑style updates and
    simplex projection is used.  The initial vector is either the
    provided parent ``height_sequence_1`` (normalised to sum = 1) or
    a uniform distribution if that fails.
    """
    rng = np.random.default_rng(seed)

    # ------------------- 1. initialise ---------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        a = parent.copy()
        n = a.size
    except Exception:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s  # now sum(a) == 1

    # best solution seen so far
    best = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(best, best))  # sum is 1 → bound = 2 n max(conv)

    # uniform fallback (score = 2)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = 2.0
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------- 2. pre‑compute FFT sizes -----------------------
    conv_len = 2 * n - 1
    conv_fft_len = 1 << ((conv_len - 1).bit_length())          # for a*a
    grad_len = n + conv_len - 1                                 # = 3n-2
    grad_fft_len = 1 << ((grad_len - 1).bit_length())          # for a_rev * w

    # ------------------- 3. optimisation parameters --------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                         # base learning rate for Adam
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment

    target_score = 1.50274349
    it = 0
    last_improve = 0

    # ------------------- 4. main optimisation loop --------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution a * a (full) via FFT ----------------------
        fa = np.fft.rfft(a, n=conv_fft_len)
        conv = np.fft.irfft(fa * fa, n=conv_fft_len)[:conv_len]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            last_improve = it
            if best_score <= target_score:
                break

        # ----- smooth max weights (softmax) -------------------------
        # temperature decays slowly, tied to current max value
        tau_factor = max(0.3 * np.exp(-it / 2000.0), 0.01)
        tau = max_conv * tau_factor + 1e-12
        exp_shift = (conv - max_conv) / tau
        w = np.exp(exp_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / conv_len)
        else:
            w /= w_sum                     # now w sums to 1

        # ----- gradient of smooth max (2 * a_rev * w) ---------------
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, n=grad_fft_len)
        fw = np.fft.rfft(w, n=grad_fft_len)
        grad_full = np.fft.irfft(fa_rev * fw, n=grad_fft_len)[:grad_len]
        g = 2.0 * grad_full[n - 1:n - 1 + n]      # slice of length n

        # ----- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----- projection onto simplex -------------------------------
        a = _proj_simplex(a)

        # ----- occasional jitter to escape stagnation ---------------
        if it % 100 == 0:
            jitter = np.exp(rng.normal(0.0, 0.005, size=n))
            a = _proj_simplex(a * jitter)

        # ----- restart if no improvement for a long while -----------
        if it - last_improve > 5000:
            # re‑inject a bit of uniform mass and continue
            a = (0.9 * a) + (0.1 * uniform)
            a = _proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it

    # ------------------- 5. return best -------------------------------
    return [float(v) for v in best]
