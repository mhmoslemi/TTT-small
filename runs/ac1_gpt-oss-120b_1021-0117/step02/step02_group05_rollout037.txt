def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)  # 1.5 second margin

    # length
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
        cur = parent.copy()
    except Exception:
        n = 4806  # fallback
        cur = _gen_binomial(n) # need to define

    L = 1 << ((2 * n - 1 - 1).bit_length())  # next pow2

    def ac1(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        maxb = conv.max()
        s = arr.sum()
        if s < 1e-12:
            return float('inf')
        return 2.0 * n * maxb / (s * s)

    best = cur.copy()
    best_score = ac1(best)

    cur_score = best_score

    # precompute "top_k" for gradient contributions
    top_k = 3

    # annealing schedule
    temperature = 0.7
    decay = 0.9995

    # step sizes
    grad_eps = 0.1  # start bigger
    block_scale_eps = 0.2
    mass_transfer_frac = 0.2

    # min step adaptation
    min_eps = 1e-4

    while time.monotonic() < deadline:
        move_type = rng.random()
        cand = None
        if move_type < 0.6:  # gradient guided
            # compute gradient from top peaks
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[:2 * n - 1]
            # get top_k indices (could include duplicates if same)
            # use argpartition
            # note: if many ties, there may be < top_k distinct
            idxs = np.argpartition(conv, -top_k)[-top_k:]
            g = np.zeros_like(cur)
            idx = np.arange(n)
            for k in idxs:
                rev = k - idx
                mask = (rev >= 0) & (rev < n)
                # derivative = 2 * cur[rev] where mask true
                g[mask] += 2.0 * cur[rev[mask]]
            # Ensure non-zero direction; if g sum zero, skip
            if g.sum() == 0:
                continue
            # direction preserving sum: subtract mean
            d = g.mean() - g
            # compute epsilon adaptively based on iteration maybe but for simplicity use grad_eps

        elif move_type < 0.85:  # block scaling
            # select random block size and position
            max_width = max(2, n // 8)  # up to 1/8 n
            min_width = max(2, n // 200)
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))

            cand[start:start+width] *= factor

                cand[i] -= delta

            cur_score = best_score

    return [float(x) for x in best]

                best_score = cand_score

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())
