def project_simplex(v):
    # Works for any vector
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2}) – used if no parent.
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Full convolution a*a (same orientation) via FFT.
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray, Lfft: int) -> np.ndarray:
    """Return a*a (full) as a length‑2n‑1 array."""
    fv = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# Maximum of the full convolution.
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, Lfft: int) -> float:
    """Maximum entry of a*a."""
    conv = _full_convolution(a, Lfft)
    return float(conv.max())


# ------------------------------------------------------------
# Gradient of a single active lag k (subgradient of max).
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative coefficients with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ----------------------------------------------------------------
    # Initialise – use the supplied parent if present; otherwise sqrt series.
    # ----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 256
        a = _sqrt_series(n0).astype(np.float64)

    n = a.shape[0]
    # Normalise to the simplex (sum = 1) and enforce non‑negativity.
    a = a / a.sum()
    a = _project_simplex(a)

    # ------------------------------------------------------------
    # Pre‑compute FFT length (next power of two) for efficiency.
    # ------------------------------------------------------------
    Lfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping of the best solution seen so far.
    # ------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(a, Lfft)

    # ------------------------------------------------------------
    # Optimisation parameters.
    # ------------------------------------------------------------
    eta0 = 0.9          # base step size
    beta = 0.85         # momentum coefficient
    v = np.zeros_like(a)   # momentum buffer
    iteration = 0
    # tolerance for treating several lags as “active”
    rel_tol = 1e-12

    while time.monotonic() < deadline:
        iteration += 1

        # ----- current convolution and active lags -----
        conv = _full_convolution(a, Lfft)
        M = float(conv.max())
        cur_score = 2.0 * n * M
        # lags whose value is within a tiny tolerance of the maximum
        tol = max(rel_tol, M * rel_tol)
        active = np.where(conv >= M - tol)[0]

        # ----- gradient (sum of sub‑gradients of active lags) -----
        grad = np.zeros_like(a)
        for k in active:
            grad += _grad_for_k(a, int(k))

        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            # Nothing to move – stop early.
            break

        # ----- momentum‑based projected step -----
        eta = eta0 / np.sqrt(iteration)
        v = beta * v - eta * (grad / grad_norm)
        cand = a + v
        cand = _project_simplex(cand)

    return [float(x) for x in best]
