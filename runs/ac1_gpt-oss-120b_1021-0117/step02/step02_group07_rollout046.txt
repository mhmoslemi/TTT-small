import time
import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a list of non‑negative heights with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ------------------------------------------------------------------
    # Problem size & optional parent sequence
    # ------------------------------------------------------------------
    n = 3809
    parent = None
    try:
        h = globals().get("height_sequence_1")
        if h is not None:
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                parent = np.clip(arr, 0.0, 1000.0)
                n = parent.size
    except Exception:
        parent = None

    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Helper: convolution + score
    # ------------------------------------------------------------------
    def _conv_and_score(seq: np.ndarray):
        f = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
        maxb = np.max(conv)
        s = np.sum(seq)
        if s <= 0.0:
            return float("inf"), conv, maxb
        score = 2.0 * n * maxb / (s * s)
        return float(score), conv, maxb

    # ------------------------------------------------------------------
    # Initialise best candidate
    # ------------------------------------------------------------------
    best = _initial_binomial(n)
    best_score, best_conv, _ = _conv_and_score(best)

    if parent is not None:
        p_score, p_conv, _ = _conv_and_score(parent)
        if p_score < best_score:
            best = parent.copy()
            best_score = p_score
            best_conv = p_conv

    current = best.copy()
    current_score = best_score
    target_sum = np.sum(current)

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # compute current convolution once
        cur_score, cur_conv, _ = _conv_and_score(current)
        current_score = cur_score

        # decide whether to do a gradient step (most of the time)
        if rng.random() < 0.68:
            # --------------------------------------------------------------
            # Gradient step aimed at lowering the maximal autocorrelation
            # --------------------------------------------------------------
            kmax = int(np.argmax(cur_conv))
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)

            grad = np.zeros_like(current)
            if lo <= hi:
                seg = current[lo:hi + 1]
                grad[lo:hi + 1] = 2.0 * seg[::-1]

            # direction that keeps the total sum unchanged
            direction = -grad + grad.mean()

            # decaying step factor (large at start, tiny near the end)
            elapsed = time.monotonic() - start
            total = deadline - start
            max_factor = 0.42
            min_factor = 0.018
            factor = max_factor * (1.0 - elapsed / total) + min_factor * (elapsed / total)

            step = factor * (np.max(current) + 1.0)

            # try a few halving steps until improvement appears
            improved = False
            for _ in range(4):
                cand = current + step * direction
                cand = np.clip(cand, 0.0, 1000.0)

                # rescale to keep original total mass
                s = cand.sum()
                if s > 0.0:
                    cand *= target_sum / s
                cand = np.clip(cand, 0.0, 1000.0)

                # tiny correction for any remaining drift
                residual = target_sum - cand.sum()
                if abs(residual) > 1e-12:
                    cand += residual / n
                    np.clip(cand, 0.0, 1000.0, out=cand)

                cand_score, cand_conv, _ = _conv_and_score(cand)
                if cand_score < current_score:
                    current = cand
                    current_score = cand_score
                    if cand_score < best_score:
                        best = cand.copy()
                        best_score = cand_score
                        best_conv = cand_conv
                    improved = True
                    break
                else:
                    step *= 0.5
            if improved:
                if best_score <= 1.50274349:
                    break
                continue

        # --------------------------------------------------------------
        # Random perturbations (fallback when gradient made no progress)
        # --------------------------------------------------------------
        op = rng.integers(0, 5)
        cand = current.copy()

        if op == 0:  # block multiplicative scaling
            width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.09))
            cand[start_idx:start_idx + width] *= factor

        elif op == 1:  # move a fraction from early to late index
            early_len = max(1, n // 14)
            late_start = n - early_len
            i = rng.integers(0, early_len)
            j = rng.integers(late_start, n)
            frac = rng.uniform(0.07, 0.22)
            delta = cand[i] * frac
            cand[i] -= delta
            cand[j] += delta

        elif op == 2:  # average two neighbours (local smoothing)
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        elif op == 3:  # global smoothing with a small kernel
            # simple three‑point kernel [0.25, 0.5, 0.25]
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            cand = np.convolve(cand, kernel, mode='same')

        else:  # pairwise averaging of two random indices
            i, j = rng.choice(n, size=2, replace=False)
            avg = (cand[i] + cand[j]) * 0.5
            cand[i] = avg
            cand[j] = avg

        # enforce bounds and keep total mass
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0.0:
            cand *= target_sum / s
        cand = np.clip(cand, 0.0, 1000.0)
        residual = target_sum - cand.sum()
        if abs(residual) > 1e-12:
            cand += residual / n
            np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score, cand_conv, _ = _conv_and_score(cand)

        # accept if strictly better, otherwise occasional uphill move
        accept = cand_score < current_score
        if not accept:
            elapsed = time.monotonic() - start
            total = deadline - start
            temp = max(0.0, 0.15 * (1.0 - elapsed / total))
            if temp > 0.0 and rng.random() < np.exp((current_score - cand_score) / (temp + 1e-12)):
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                best_conv = cand_conv
                if best_score <= 1.50274349:
                    break

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
