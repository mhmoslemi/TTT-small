import math
import time

import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Compute the AC1 upper bound using FFT (same as the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for cleanup
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ----- problem size & initial sequence -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        # no parent – use classic binomial shape as a fallback
        parent = None
        n = 256

    L = _next_pow2(2 * n - 1)

    # start from parent if it exists and is not worse than binomial
    if parent is None:
        base = np.ones(n, dtype=np.float64) * (1.0 / n)   # uniform fallback
    else:
        base = parent.astype(np.float64).copy()

        # Binomial shape for comparison
        binom = np.empty(n, dtype=np.float64)
        binom[0] = 1.0
        for i in range(1, n):
            binom[i] = binom[i - 1] * (2 * i - 1) / (2 * i)
        if _ac1_score_fft(binom, n, L) < _ac1_score_fft(base, n, L):
            base = binom

    # ---- initialise best and current states ----
    best = base.copy()
    best_score = _ac1_score_fft(best, n, L)

    cur = best.copy()
    cur_score = best_score

    # ---- hyper‑parameters ----
    temperature = 1.0
    temp_decay = 0.9998
    step_lr = 0.30          # learning rate for gradient‑like move
    lr_decay = 0.9999
    target_score = 1.50274349

    # ----- main optimisation loop -----
    while time.monotonic() < deadline:
        # current convolution information (needed for gradient‑type move)
        fa_cur = np.fft.rfft(cur, n=L)
        conv_cur = np.fft.irfft(fa_cur * fa_cur, n=L)[: 2 * n - 1]
        max_b = float(conv_cur.max())
        k_max = int(conv_cur.argmax())
        sum_cur = float(cur.sum())
        cur_score = 2.0 * n * max_b / (sum_cur * sum_cur)

            right = np.roll(cur, -1)

            # guard against numerical drift
            np.clip(cand, 0.0, 1000.0, out=cand)

        elif move_type < 0.60:
            # 2) Gradient‑inspired block move (mirroring w.r.t. active lag)
            t = max_b / sum_cur
