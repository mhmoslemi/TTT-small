def solve_lp_max_sum(a_vec, M):
    # a_vec: numpy 1-d array length n, normalized sum 1.
    n = len(a_vec)
    a_param = cp.Parameter(n)
    a_param.value = a_vec
    b = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(a_param, b) <= M]   # vector inequality
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=20000)
    # get solution
    b_val = np.array(b.value).flatten()
    # ensure nonnegative and finite
    b_val = np.maximum(b_val, 0)
    return b_val
```

def conv_fft(a, n, fft_len):
    # compute convolution a * a using FFT of length fft_len
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[:2*n-1]
```

import time

import numpy as np

# CVXPY is optional – we use it if available, otherwise fall back to a simple
# sub‑gradient method (the one from the reference implementation).
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVXPY_AVAILABLE = False

# ----- basic linear‑algebra utilities ---------------------------------

def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the full convolution a*a using an FFT of length fft_len.
    Returns the first (2*len(a)-1) entries.
    """
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * a.shape[0] - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray, fft_len: int) -> float:
    """Score used by the evaluator (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    conv = _conv_self_fft(a, fft_len)
    return 2.0 * n * float(conv.max())

# ----- LP solver (matrix‑free) ---------------------------------------

def _make_lp_solver(n: int):
    """
    Construct a reusable CVXPY problem:
        max      sum(b)
        s.t.     conv(a, b) <= M
                 b >= 0
    a and M are parameters that will be updated each iteration.
    """
    a_param = cp.Parameter(n, nonneg=True)
    M_param = cp.Parameter(nonneg=True, shape=())
    b = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(a_param, b) <= M_param]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    return prob, a_param, M_param, b


def _solve_lp_max_sum(a_vec: np.ndarray, M_val: float,
                      prob, a_param, M_param, b_var):
    """
    Solve the LP for a given a and M.  Returns a dense ndarray of the optimal b.
    If the solver fails, returns None.
    """
    a_param.value = a_vec
    M_param.value = M_val
    # ECOS is fast for dense LPs; fall back to SCS if needed.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:
            return None
    if b_var.value is None:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64).ravel()
    # numerical cleanup
    b_val = np.maximum(b_val, 0.0)
    return b_val

# ----- fallback sub‑gradient optimizer --------------------------------

def _sgd_optimize(a0: np.ndarray, fft_len: int, deadline: float):
    """Very simple projected sub‑gradient descent (used if CVXPY absent)."""
    n = a0.shape[0]
    a = a0.copy()
    best = a.copy()
    best_score = _score(a, fft_len)
    rng = np.random.default_rng(1234)
    alpha0 = 0.8
    no_imp = 0

    while time.time() < deadline:
        conv = _conv_self_fft(a, fft_len)
        k = int(conv.argmax())
        g = np.zeros_like(a)
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        idx = np.arange(i0, i1 + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]

        t = max(1, no_imp + 1)
        alpha = max(1e-4, alpha0 / math.sqrt(t))
        cand = a - alpha * g
        cand = _project_onto_simplex(cand)

        # occasional jitter
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.015, size=n))
            cand = cand * jitter
            cand = _project_onto_simplex(cand)

        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best = cand.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = cand
        if time.time() >= deadline - 0.5:
            break
    return best.tolist()
