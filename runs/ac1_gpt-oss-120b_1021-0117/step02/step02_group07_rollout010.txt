def _conv_full(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Return full linear convolution of seq with itself (length 2*len-1)."""
    f = np.fft.rfft(seq, n=fft_len)
    return np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]

    if parent is not None:

    else:

        a[0] = 1.0
        for i in range(1, n):
