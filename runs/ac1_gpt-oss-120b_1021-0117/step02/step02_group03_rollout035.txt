import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Gradient of a single convolution entry k (full convolution)
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # give ourselves a safety margin for the outer timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initialise – use the provided parent if present, else uniform
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 256
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    a = a / a.sum()                     # normalise to sum = 1
    a = _project_simplex(a)              # safety projection

    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    fv = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
    cur_score = 2.0 * n * float(conv.max())
    best = a.copy()
    best_score = cur_score

    # annealing / step‑size schedule
    step_frac = 0.5          # maximum fraction of mass moved in one step
    temperature = cur_score * 0.02
    it = 0
    no_imp = 0

    while time.monotonic() < deadline:
        it += 1

        # recompute convolution of current point (needed for active‑lag gradient)
        fv_cur = np.fft.rfft(a, n=Lfft)
        conv_cur = np.fft.irfft(fv_cur * fv_cur, n=Lfft)[: 2 * n - 1]
        cur_M = float(conv_cur.max())
        cur_score = 2.0 * n * cur_M

        # ----------------------------------------------------
        # propose a new candidate
        # ----------------------------------------------------
        if rng.random() < 0.65:                     # gradient‑guided move
            k = int(conv_cur.argmax())
            grad = _grad_for_k(a, k)
            norm = np.linalg.norm(grad)
            if norm > 0.0:
                # step size decays like 1/sqrt(iter)
                eta = step_frac / math.sqrt(it)
                cand = a - eta * (grad / norm)
            else:
                cand = a.copy()
            cand = _project_simplex(cand)
        else:                                        # random mass transfer
            i = rng.integers(0, n)
            if a[i] <= 0.0:

            j = rng.integers(0, n - 1)
            if j >= i:

            cand[i] -= delta
            cand[j] += delta

        cand_M = float(conv_cand.max())
        cand_score = 2.0 * n * cand_M

        # ----------------------------------------------------

        # ----------------------------------------------------
        improve = cand_score < cur_score
