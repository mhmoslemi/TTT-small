        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        maxb = conv.max()

        if s < 1e-12:

    transfer_max = 0.35

import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a non‑negative step function for the AC1 bound.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # reserve a couple of seconds for the final return
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # length
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
    except Exception:
        n = 2400                     # fallback length if parent missing

    L = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # start from a Catalan‑type (binomial) shape
    base = _gen_binomial(n)

    # test a few simple re‑weightings of the binomial shape
    best = base.copy()
    best_score = _ac1_score(best, n, L)

    ar = np.arange(n, dtype=np.float64) + 1.0
    norm = float(n)

    # explore a modest family a_i = binom_i * ((i+1)/n)^delta
    for delta in np.linspace(0.0, 0.6, 13)[1:]:          # skip delta=0 (already tested)
        factor = (ar / norm) ** delta
        cand = base * factor
        cand_score = _ac1_score(cand, n, L)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

    # --------------------------------------------------------------
    # simulated‑annealing fine‑tuning from the best found so far
    cur = best.copy()
    cur_score = best_score

    # hyper‑parameters
    temperature = 1.0
    decay = 0.9995
    min_block = max(2, n // 200)
    max_block = max(2, n // 12)
    grad_eps = 0.025
    transfer_max = 0.35
    block_std = 0.12
    idx = np.arange(n, dtype=np.int64)

    while time.monotonic() < deadline:
        move = rng.random()

        # ------------------------------------------------------
        # 1) gradient‑guided mass transfer (≈45 % of moves)
        if move < 0.45:
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            dup = mask & (rev != idx)
            g[dup] *= 2.0

            d = g.mean() - g
            cand = cur + grad_eps * d

        # ------------------------------------------------------
        # 2) block scaling (≈35 % of moves)
        elif move < 0.80:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, block_std))
            cand = cur.copy()
            cand[start:start + width] *= factor

        # ------------------------------------------------------
        # 3) random mass transfer (≈20 % of moves)
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * transfer_max
            delta_val = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta_val
            cand[j] += delta_val

        # enforce evaluator bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _ac1_score(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # simulated‑annealing acceptance
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # occasional restart from the best seen
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # --------------------------------------------------------------
    # final result – plain Python floats
    return [float(x) for x in best]
