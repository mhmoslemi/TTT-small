import time
import numpy as np

def _project_onto_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # obtain parent

        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256
    # FFT length for correlation (3n-2)
    L = 1 << ((3*n - 2).bit_length())

    # helper to compute score (assuming sum=1)
    def _score(a):
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]
        return 2.0 * n * np.max(conv)

    # Prepare some base arrays
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)

    # initial candidate list
    candidates = []

    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = np.sum(p)
        if s > 0:
            candidates.append(p / s)

    candidates.append(np.full(n, 1.0 / n))

    # Catalan weights (reuse from previous code)
    cat = np.empty(n, dtype=np.float64)
    cat[0] = 1.0
    for i in range(1, n):
        cat[i] = cat[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
    candidates.append(cat / cat.sum())

    # sin^2 shape
    v = sin_vals**2
    if v.sum() > 0:
        candidates.append(v / v.sum())

    # exponent sweep for base^p
    for p in np.linspace(0.1, 5.0, 30):
        v = np.power(base, p)
        s = v.sum()
        if s > 0:

    for _ in range(20):

        p2 = rng.uniform(0.1, 5.0)
        mix = np.power(base, p1) * np.power(sin_vals, p2)
        s = mix.sum()
        if s > 0:
            candidates.append(mix / s)

    best_vec = None

        if sc < best_score:

            best_vec = vec.copy()

            iter_no_improve = 0
            alpha = 5.0
            total_iter = 0

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
