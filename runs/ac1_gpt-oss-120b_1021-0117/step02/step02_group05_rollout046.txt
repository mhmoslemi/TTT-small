def move_swap(cur, rng):
    i, j = rng.integers(0, n, size=2)
    if i==j: return cur.copy()
    cand = cur.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand

def move_shift(cur, rng):
    # rotate by random shift
    shift = rng.integers(0, n)
    if shift == 0:
        return cur.copy()
    cand = np.roll(cur, shift)
    return cand

def move_block_scaling(cur, rng):
    width = rng.integers(min_block, max_block+1)
    start = rng.integers(0, n-width+1)
    factor = np.exp(rng.normal(0.0, 0.13))
    cand = cur.copy()
    cand[start:start+width] *= factor
    return cand

def move_random_mass(cur, rng):
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return cur.copy()
    frac = rng.random() * transfer_max
    delta = cur[i] * frac
    cand = cur.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand

import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound using FFT (no clipping)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the AC1 upper bound.
    Returns a list of floats.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # obtain initial sequence (parent or simple start)
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        cur = parent_arr.copy()
    except Exception:
        # Catalan‑type start: a_i = C(2i,i)/4**i
        n = 256
        cur = np.empty(n, dtype=np.float64)
        cur[0] = 1.0
        for i in range(1, n):
            cur[i] = cur[i - 1] * (2 * i - 1) / (2 * i)
    n = cur.size
    L = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline candidates
    best = cur.copy()
    best_score = _ac1_score(best, n, L)

    # try a fresh Catalan start if we used a parent and didn't yet test it
    if 'parent_arr' in locals():
        alt = np.empty(n, dtype=np.float64)
        alt[0] = 1.0
        for i in range(1, n):
            alt[i] = alt[i - 1] * (2 * i - 1) / (2 * i)
        alt_score = _ac1_score(alt, n, L)
        if alt_score < best_score:
            best = alt.copy()
            best_score = alt_score

    cur_score = best_score

    # ------------------------------------------------------------------
    # hyper‑parameters
    temperature = 1.0
    decay = 0.9994
    min_block = max(2, n // 250)
    max_block = max(2, n // 10)
    transfer_max = 0.45
    grad_eps = 0.015
    target = 1.50274349
    # probability split (must sum <=1)
    p_grad = 0.45
    p_block = 0.30
    p_swap = 0.10
    p_random = 0.10

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    while time.monotonic() < deadline:
        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) gradient guided mass transfer
        if move_type < p_grad:

            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            g[mask] = cur[rev[mask]]

            g[self_mask] *= 2.0
            d = g.mean() - g          # zero‑sum direction

        # --------------------------------------------------------------
        # 2) block scaling
        elif move_type < p_grad + p_block:

        cand = reorder(best)
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _ac1_score(cand, n, L)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

    # final conversion to pure Python floats
    return [float(x) for x in best]
