def subgrad_descent(a0, max_iter, deadline, momentum_alpha=0.9, base_step=0.6, min_step=1e-5):
    a = a0.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    iter_no_improve = 0
    for t in range(1, max_iter+1):
        if time.monotonic() >= deadline:
            break
        conv = conv_self_fft(a, n, fft_len)
        max_b = conv.max()
        cur_score = 2*n*max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            iter_no_improve = 0
        else:
            iter_no_improve += 1

        # compute subgradient at current active index
        k = int(conv.argmax())
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # For indices i_start..i_end inclusive
        # g[i] = 2*a[k-i]
        # We can fill slice
        idx = np.arange(i_start, i_end+1)
        g[idx] = 2*a[k - idx]

        # momentum update
        v = momentum_alpha * v + (1 - momentum_alpha) * g

        # step size schedule
        step = max(min_step, base_step / np.sqrt(t))

        # update
        a = a - step * v

        # projection onto simplex for nonnegativity and sum=1
        a = _project_onto_simplex(a)

        # occasional jitter
        if np.random.rand() < 0.02:
            jitter = np.exp(np.random.normal(0.0, 0.01, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)

        # if stagnation:
        if iter_no_improve > 2000:
            # random restart near best
            perturb = np.exp(np.random.normal(0.0, 0.03, size=n))
            a = best_a * perturb
            a = _project_onto_simplex(a)
            iter_no_improve = 0

    return best_a, best_score

# Helper: projection onto the probability simplex
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):          # all non‑positive → uniform
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            best_score = sc
            best_vec = a.copy()
            if best_score <= 1.50274349:
                return best_vec.tolist()

    # -------------------------------------------------
    # Phase 2: exponential‑gradient descent with momentum
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    a = best_vec.copy()
    # momentum buffer (same shape)
    momentum = np.zeros_like(a)

    # hyper‑parameters (tuned heuristically)
    base_step = 0.6          # initial step magnitude
    min_step = 1e-5
    momentum_coeff = 0.9
    jitter_prob = 0.015
    jitter_sigma = 0.018
    stagnation_limit = 2500
    restart_sigma = 0.025
    target = 1.50274349

    iter_no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # full self‑convolution and current score
        conv = _conv_self_fft(a, n, fft_len)
        max_b = float(conv.max())
        cur_score = 2.0 * n * max_b

        # record best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_imp = 0
            if best_score <= target:
                break
        else:
            iter_no_imp += 1

        # active index producing the maximal entry
        k = int(conv.argmax())

        # sub‑gradient for the active k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            g[idx] = 2.0 * a[k - idx]

        # momentum update
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * g

        # adaptive step size (diminishing)
        step = max(min_step, base_step / math.sqrt(it))

        # exponential‑gradient (mirror) step
        a = a * np.exp(-step * momentum)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # occasional multiplicative jitter to escape plateaus
        if rng.random() < jitter_prob:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = a / a.sum()

        # restart if no progress for a while
        if iter_no_imp > stagnation_limit:
            pert = np.exp(rng.normal(0.0, restart_sigma, size=n))
            a = best_vec * pert
            a = a / a.sum()
            momentum.fill(0.0)
            iter_no_imp = 0

    # -------------------------------------------------
    # Return the best vector found (as plain Python list)
    return best_vec.tolist()
