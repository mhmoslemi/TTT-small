        cand_max = _max_self_conv(cand, fft_len)
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            a = cand
            alpha = min(alpha * 1.02, 1.0)       # modestly increase step if successful
        else:
            alpha = max(alpha * 0.5, min_alpha)  # shrink step
            if no_improve % 40 == 0:            # occasional jitter on stagnation
                jitter = np.exp(rng.normal(0.0, 0.015, size=n))
                a = _normalize(a * jitter)
                momentum.fill(0.0)

        # long stagnation → restart from best with mild perturbation
        if no_improve > 500:
            pert = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _normalize(best_vec * pert)
            momentum.fill(0.0)
            no_improve = 0

    return best_vec.tolist()
