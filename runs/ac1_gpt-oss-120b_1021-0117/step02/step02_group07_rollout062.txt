import cvxpy as cp

def lp_improve(seq, max_conv):
    n = len(seq)
    a_param = cp.Parameter(n, value=seq, constant=True)
    b = cp.Variable(n, nonneg=True)
    conv_ab = cp.conv(a_param, b)
    constraints = [conv_ab <= max_conv,
                   b <= 1000]  # optional bound
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    if b.value is None:
        return None
    b_opt = np.maximum(0.0, np.asarray(b.value, dtype=np.float64))
    sum_seq = np.sum(seq)
    sum_b = np.sum(b_opt)
    if sum_b == 0:
        return None
    scale = sum_seq / sum_b
    g = b_opt * scale
    return g
```

import numpy as np
import time
import math

# Optional LP solver
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of seq with itself using FFT."""
    f = np.fft.rfft(seq, n=fft_len)
    conv_full = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    return conv_full


def _score(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution, max value)."""
    conv = _conv_fft(seq, fft_len)
    max_b = float(conv.max())
    s = float(seq.sum())
    if s <= 0.0:
        return float("inf"), conv, max_b
    return 2.0 * seq.size * max_b / (s * s), conv, max_b


def _solve_lp(a_seq: np.ndarray, max_conv: float):
    """Linear program: maximise sum(b) s.t. conv(a,b) <= max_conv, b>=0."""
    if cp is None:
        return None
    n = a_seq.shape[0]
    a_param = cp.Parameter(n, value=a_seq, constant=True)
    b = cp.Variable(n, nonneg=True)
    conv_ab = cp.conv(a_param, b)          # length 2n-1
    constraints = [conv_ab <= max_conv, b <= 1000.0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:                      # pragma: no cover
        return None
    if b.value is None:
        return None
    return np.maximum(0.0, np.asarray(b.value, dtype=np.float64))

    fft_len = 1 << ((2 * n - 1).bit_length())
