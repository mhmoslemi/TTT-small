def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0) # margin

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256  # fallback length
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate a simple exponential decay
        a = np.exp(-np.arange(n) / (n/10.0))

    # ensure nonnegative
    a = np.maximum(a, 0.0)
    # project onto simplex sum=1
    a = _project_simplex(a)

    best_a = a.copy()
    conv_a = _conv_full_fft(a, a)
    best_score = 2.0 * n * conv_a.max()

    # hyperparameters
    step0 = 0.1   # initial step size
    beta = 0.5
    beta_max = 10.0
    beta_inc = 0.02  # per iteration
    it = 0
    # for early break if perfect bound < target
    target = 1.50274349

    # Precompute L for FFT
    L = 1 << ((3*n - 2).bit_length())

    while time.monotonic() < deadline:
        it += 1
        # Compute full convolution a*a via FFT
        rfft_a = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(rfft_a * rfft_a, n=L)[:2*n-1]
        max_conv = conv_full.max()
        # Real score
        score = 2.0 * n * max_conv
        # update best
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if best_score < target:
                break

        # Weight for smooth max
        beta = min(beta_max, beta + beta_inc)
        # stable softmax
        shifted = conv_full - max_conv
        exp_beta = np.exp(beta * shifted)
        w = exp_beta / exp_beta.sum()

        # Compute gradient via correlation using FFT
        a_rev = a[::-1]
        rfft_rev = np.fft.rfft(a_rev, n=L)
        rfft_w = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(rfft_rev * rfft_w, n=L)[:3*n-2]
        grad = 4.0 * n * corr_full[n-1 : n-1 + n]

        # step size schedule
        step = step0 / np.sqrt(it)

        # gradient descent step
        cand = a - step * grad
        # project onto simplex (nonneg + sum=1)
        cand = _project_simplex(cand)

        # compute candidate score for accept condition
        # using conv of cand quickly: maybe compute again; but for simplicity compute factor
        rfft_cand = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(rfft_cand * rfft_cand, n=L)[:2*n-1]
        cand_score = 2.0 * n * conv_cand.max()

        if cand_score < score:
            a = cand
            # update best inside next loop iteration
        else:
            # Maybe accept with small probability
            if rng.random() < np.exp(-(cand_score - score) / (step + 1e-12)):
                a = cand

        # occasional random perturbation
        if it % 2000 == 0:
            # jitter best and restart
            jitter = rng.normal(scale=1e-3, size=n)
            a = np.maximum(best_a + jitter, 0.0)
            a = _project_simplex(a)

    # Final fallback: ensure best_a has nonnegative and sum >= 0.01
    best_a = np.maximum(best_a, 0.0)
    # sum already 1 due to projection

    return [float(x) for x in best_a]

def propose_candidate(...):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 1.)

    # initial a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = np.exp(-np.arange(n) / 10.0)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.size

    # precompute L
    L = 1 << ((3*n - 2).bit_length())

    # evaluate function
    def _score_vec(v):
        r = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(r * r, n=L)[:2*n-1]
        return 2.0 * n * conv.max()

    cur_score = _score_vec(a)
    best_a = a.copy()
    best_score = cur_score

    # gradient descent phase
    step0 = 0.05
    beta = 0.5
    beta_max = 12.0
    beta_inc = 0.01

    it = 0
    while time.monotonic() < deadline:
        # gradient descent step
        # compute convolution once & gradient as above
        rfft_a = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(rfft_a * rfft_a, n=L)[:2*n-1]
        max_conv = conv_full.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if best_score < 1.50274349:
                break
        # update beta
        beta = min(beta_max, beta + beta_inc)
        # weights for gradient smoothing
        shifted = conv_full - max_conv
        exp_beta = np.exp(beta * shifted)
        w = exp_beta / exp_beta.sum()
        # gradient
        a_rev = a[::-1]
        rfft_rev = np.fft.rfft(a_rev, n=L)
        rfft_w = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(rfft_rev * rfft_w, n=L)[:3*n-2]
        grad = 4.0 * n * corr_full[n-1 : n-1 + n]
        # step size
        it += 1
        step = step0 / np.sqrt(it)
        cand = a - step * grad
        cand = _project_simplex(cand)
        cand_score = _score_vec(cand)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            # accept with prob
            if rng.random() < np.exp(-(cand_score - cur_score) / (step + 1e-12)):
                a = cand
                cur_score = cand_score
        # occasional random jitter restart
        if it % 3000 == 0:
            jitter = rng.normal(scale=1e-3, size=n)
            a = best_a + jitter
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
        # Check time
        if time.monotonic() >= deadline:
            break

    # Simulated annealing (optional) if time left
    temp = 0.02
    temp_decay = 0.9995
    move_frac = 0.2
    while time.monotonic() < deadline:
        # random pick pair
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * move_frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # note sum unchanged and nonnegatives
        new_score = _score_vec(a_new)
        dscore = new_score - cur_score
        if dscore < 0 or rng.random() < np.exp(-dscore / temp):
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                if best_score < 1.50274349:
                    break
        temp *= temp_decay
        # optional occasional reset to best with jitter
        if rng.random() < 0.001:
            a = best_a + rng.normal(scale=5e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            cur_score = _score_vec(a)
        # time check
        if time.monotonic() >= deadline - 0.2:
            break

    # final best solution
    best_a = np.maximum(best_a, 0.0)
    # ensure simplex
    best_a = _project_simplex(best_a)
    # return as list of floats
    return [float(x) for x in best_a]

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolve_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Return full linear convolution of x and y using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[: len(x) + len(y) - 1]

            continue
        cand = cand / cand.sum()
        conv_c = _full_convolve_fft(cand, cand, L)
        sc = 2.0 * n * np.max(conv_c)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # gradient‑descent phase on a smooth approximation of the max.
    # ------------------------------------------------------------------
    a = best_vec.copy()
    cur_score = best_score
    step0 = 0.02
    beta = 0.5
    beta_max = 12.0
    beta_inc = 0.02
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # full convolution of current iterate
        conv_full = _full_convolve_fft(a, a, L)
        maxc = np.max(conv_full)
        cur_score = 2.0 * n * maxc

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
            if best_score < 1.50274349:
                break
        else:
            no_improve += 1

        # smooth max (log‑sum‑exp) weights
        beta = min(beta_max, beta + beta_inc)
        shifted = conv_full - maxc                # max at zero
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient = 4 * n * correlation(a_rev, w)
        a_rev = a[::-1]
        corr_full = _full_convolve_fft(a_rev, w, L)   # length 3n‑2
        grad = 4.0 * n * corr_full[n - 1: n - 1 + n]

        it += 1
        step = step0 / np.sqrt(it)

        cand = a - step * grad
        cand = _project_simplex(cand)

        # evaluate candidate
        conv_cand = _full_convolve_fft(cand, cand, L)
        cand_score = 2.0 * n * np.max(conv_cand)

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            # occasional Metropolis step to escape plateaus
            if rng.random() < np.exp(-(cand_score - cur_score) / (step + 1e-12)):
                a = cand
                cur_score = cand_score

        # jittered restart every few thousand iterations
        if it % 3000 == 0:
            jitter = rng.normal(scale=5e-4, size=n)
            a = best_vec + jitter
            a = _project_simplex(a)
            cur_score = 2.0 * n * np.max(_full_convolve_fft(a, a, L))

        if time.monotonic() >= deadline - 0.2:
            break

    # ------------------------------------------------------------------
    # Simulated‑annealing style random mass‑transfer moves.
    # ------------------------------------------------------------------
    temp = 0.02
    temp_decay = 0.9997
    move_frac = 0.3

    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * move_frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv_new = _full_convolve_fft(a_new, a_new, L)
        new_score = 2.0 * n * np.max(conv_new)
        dscore = new_score - cur_score

        if dscore < 0.0 or rng.random() < np.exp(-dscore / max(temp, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a_new.copy()
                if best_score < 1.50274349:
                    break
        temp = max(temp * temp_decay, 1e-8)

        # occasional tiny perturbation around the best known point
        if rng.random() < 0.001:
            a = best_vec + rng.normal(scale=3e-4, size=n)
            a = _project_simplex(a)
            cur_score = 2.0 * n * np.max(_full_convolve_fft(a, a, L))

        if time.monotonic() >= deadline - 0.1:
            break

    # final safeguard projection and return
    best_vec = _project_simplex(best_vec)
    return [float(x) for x in best_vec]
