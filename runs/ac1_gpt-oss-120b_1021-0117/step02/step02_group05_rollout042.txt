    init = parent if parent not None else _gen_binomial(n)

    cur = init.astype(np.float64)
    best = cur.copy()
    best_score = _ac1_score(best, n, L)

    # try alternative initializations:
    # uniform
    uni = np.full(n, 0.5) * (np.sum(cur) / (0.5 * n))? Actually we can set uniform with same sum as cur: u = np.full(n, np.sum(cur)/n)
    # but must clamp to [0, 1000]; ok.

    alt = np.full(n, np.sum(cur)/n)
    if _ac1_score(alt,n,L) < best_score: update.

    # main loop
    temperature = 1.0
    decay = 0.9995
    grad_eps = 0.05 # base
    block_min = max(2, n // 500)
    block_max = max(2, n // 10)
    transfer_max = 0.5

    while time.monotonic() < deadline:
        # pick move type
        move = rng.random()
        if move < 0.6:
            # gradient step
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
            kmax = int(np.argmax(conv))
            # compute g
            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            self_mask = mask & (rev != idx)
            g[self_mask] *= 2.0

            # direction preserving sum
            d = g.mean() - g
            # step magnitude
            eps = grad_eps * temperature
            cand = cur + eps * d
        elif move < 0.9:
            # block scaling
            width = int(rng.integers(block_min, block_max+1))
            start = int(rng.integers(0, n - width + 1))
            # scaling factor distribution about 1
            factor = np.exp(rng.normal(0.0, 0.15))
            cand = cur.copy()
            cand[start:start+width] *= factor
        else:
            # random transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * transfer_max
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # clip
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate
        cand_score = _ac1_score(cand, n, L)

        # update best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # if good, maybe lower temperature rediscover? can reset temperature
            # optionally break if < target
            if best_score < 1.50274349:
                break

        # acceptance (simulate annealing)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # occasional restart
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    return [float(x) for x in best]

import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

    rng = np.random.default_rng(seed)

    cur = best_seq.copy()
