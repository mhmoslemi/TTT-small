import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used for fallback random moves
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        # 5) evaluate candidate and decide on acceptance
        # --------------------------------------------------------------
        fa_c = np.fft.rfft(cand, n=Lconv)
        conv_c = np.fft.irfft(fa_c * fa_c, n=Lconv)[: 2 * n - 1]
        cand_sc = 2.0 * n * conv_c.max()

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # occasional Metropolis‑type acceptance to escape plateaus
            prob = np.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc

        # --------------------------------------------------------------
        # 6) occasional random mass‑transfer move (exploration)
        # --------------------------------------------------------------
        if no_improve > 3000 and rng.random() < 0.18:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.45
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)   # keep on simplex
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                        no_improve = 0

        # --------------------------------------------------------------
        # 7) periodic jitter‑restart towards the best solution found
        # --------------------------------------------------------------
        if (it % 5000 == 0) and rng.random() < 0.27:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best vector discovered (clipped to evaluator limits)
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    best_vec = np.minimum(best_vec, 1000.0)
    return [float(x) for x in best_vec]
