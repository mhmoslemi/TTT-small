import cvxpy as cp

def improve_with_lp(f, M, n, max_iter=5, t=0.5):
    for it in range(max_iter):
        # Solve LP
        g = cp.Variable(n)
        constraints = [g >= 0]
        # define M bound
        for k in range(2*n-1):
            terms = []
            # loop through support
            for i in support:
                j = k - i
                if 0 <= j < n:
                    terms.append(f[i] * g[j])
            if terms:
                constraints.append(cp.sum(cp.hstack(terms)) <= M)
            else:
                # no terms, 0 <= M automatically
                pass
        # objective
        obj = cp.Maximize(cp.sum(g))

def mass_transfer_peak(seq, k_max, frac=0.2):
    n=len(seq)
    i = rng.integers(0, n)
    j = k_max - i
    if j <0 or j>=n or i==j:
        return seq.copy()
    # amount to move from both i and j
    delta_i = seq[i] * frac
    delta_j = seq[j] * frac
    total = delta_i + delta_j
    cand = seq.copy()
    cand[i] -= delta_i
    cand[j] -= delta_j
    # choose random destination index(s) not i or j
    dest1 = rng.integers(0, n)
    dest2 = rng.integers(0, n)
    cand[dest1] += total/2
    cand[dest2] += total/2
    np.maximum(cand, 0, out=cand)
    return cand

def move_reduce_peak(arr, k_max, rng, reduction_frac=0.2):
    n = arr.shape[0]
    i_vals = np.arange(n)
    j_vals = k_max - i_vals
    mask = (j_vals >= 0) & (j_vals < n)
    # compute products
    prods = np.empty(n)
    prods[:] = -1.0
    prods[mask] = arr[i_vals[mask]] * arr[j_vals[mask]]
    # find index i_max where product is maximal
    idx = int(np.argmax(prods))
    j = int(k_max - idx)
    if idx == j:
        # product from a_i^2
        # reduce a_i alone
        delta = arr[idx] * reduction_frac
        cand = arr.copy()
        cand[idx] -= delta
        # add delta to random other index
        dest = int(rng.integers(0, n))
        cand[dest] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand
    else:
        # reduce both sides
        delta_i = arr[idx] * reduction_frac
        delta_j = arr[j] * reduction_frac
        total = delta_i + delta_j
        cand = arr.copy()
        cand[idx] -= delta_i
        cand[j] -= delta_j
        # distribute total to two random indices (could be same)
        dest1 = int(rng.integers(0, n))
        dest2 = int(rng.integers(0, n))
        cand[dest1] += total/2
        cand[dest2] += total/2
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 score (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _mirror_gravity_move(cur: np.ndarray, k_max: int, max_b: float, sum_cur: float,
                         step_lr: float, rng) -> np.ndarray:
    """Gradient‑inspired move: push values toward the mirror needed to cut the peak."""
    # target value that would equalise the peak
    t = max_b / sum_cur
    idx = np.arange(cur.size)
    j = k_max - idx
    mask = (j >= 0) & (j < cur.size)
    mirror = np.zeros_like(cur)
    mirror[mask] = cur[j[mask]]
    # scale‑invariant factor
    C = 4.0 * cur.size / (sum_cur * sum_cur)
    delta = step_lr * C * (t - mirror)
    cand = cur + delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _random_block_scale(cur: np.ndarray, rng) -> np.ndarray:
    n = cur.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = math.exp(rng.normal(0.0, 0.12))
    cand = cur.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _mass_transfer_move(cur: np.ndarray, rng) -> np.ndarray:
    n = cur.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return cur
    frac = rng.random() * 0.4
    delta = cur[i] * frac
    cand = cur.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _peak_reduce_move(cur: np.ndarray, k_max: int, rng, frac: float = 0.2) -> np.ndarray:
    """Transfer mass away from the pair that dominates the current peak."""
    n = cur.size
    i_arr = np.arange(n)
    j_arr = k_max - i_arr
    mask = (j_arr >= 0) & (j_arr < n)
    prods = np.full(n, -1.0)
    prods[mask] = cur[i_arr[mask]] * cur[j_arr[mask]]
    idx = int(np.argmax(prods))
    j = int(k_max - idx)
    cand = cur.copy()
    if idx == j:  # diagonal term a_i^2
        delta = cur[idx] * frac
        cand[idx] -= delta
        dest = int(rng.integers(0, n))
        cand[dest] += delta
    else:
        delta_i = cur[idx] * frac
        delta_j = cur[j] * frac
        total = delta_i + delta_j
        cand[idx] -= delta_i
        cand[j] -= delta_j
        dest1 = int(rng.integers(0, n))
        dest2 = int(rng.integers(0, n))
        cand[dest1] += total / 2.0
        cand[dest2] += total / 2.0
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ----- problem size & initial sequence -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256
    L = _next_pow2(2 * n - 1)

    # start from parent if it exists, otherwise a uniform seed
    if parent is not None:
        base = parent.astype(np.float64).copy()
        # ensure validity
        np.clip(base, 0.0, 1000.0, out=base)
    else:
        base = np.full(n, math.sqrt(2 * n) / n, dtype=np.float64)  # uniform sum = sqrt(2n)

    # normalise to the canonical sum sqrt(2n) (the evaluator is scale‑invariant)
    target_sum = math.sqrt(2.0 * n)
    cur_sum = float(base.sum())
    if cur_sum > 0:
        base *= target_sum / cur_sum
    else:
        base[:] = target_sum / n
    np.clip(base, 0.0, 1000.0, out=base)

    best = base.copy()
    best_score = _ac1_score_fft(best, n, L)

    cur = best.copy()
    cur_score = best_score

    # ----- optimisation parameters -----
    temperature = 5.0
    temp_decay = 0.9999
    step_lr = 0.3
    lr_decay = 0.9999

    # ----- main loop -----
    while time.monotonic() < deadline:
        # compute current convolution and relevant statistics
        fa_cur = np.fft.rfft(cur, n=L)
        conv_cur = np.fft.irfft(fa_cur * fa_cur, n=L)[: 2 * n - 1]
        max_b = float(conv_cur.max())
        k_max = int(conv_cur.argmax())
        sum_cur = float(cur.sum())
        cur_score = 2.0 * n * max_b / (sum_cur * sum_cur)

        # ----- select a move -----
        move_type = rng.random()
        if move_type < 0.35:
            cand = _mirror_gravity_move(cur, k_max, max_b, sum_cur, step_lr, rng)
        elif move_type < 0.55:
            cand = _peak_reduce_move(cur, k_max, rng, frac=0.15)
        elif move_type < 0.75:
            cand = _random_block_scale(cur, rng)
        else:
            cand = _mass_transfer_move(cur, rng)

        # ----- evaluate candidate -----
        cand_score = _ac1_score_fft(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # ----- simulated‑annealing acceptance -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # ----- schedule updates -----
        temperature *= temp_decay
        step_lr *= lr_decay

        # occasional restart to the best known point
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 5.0
            step_lr = 0.3

    # return as plain Python floats
    return [float(x) for x in best]
