import time
import numpy as np

# ------------------------------------------------------------
# Projection onto probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        s = 0.0 if k == 1 else float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------
# max of the (full) convolution a*a using FFT
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# gradient of conv_k = sum_i a_i a_{k-i} at given lag k
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # keep a small safety margin for the outer harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------
    # initialise – use provided parent if present, otherwise sqrt series
    # ----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = _sqrt_series(256).astype(np.float64)

    a = a / a.sum()
    a = _project_simplex(a)          # ensure feasibility (sum=1, non‑negative)

    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two for FFT

    # ----------------------------------------------------------------
    # helper for scoring (avoids recomputing denominator)
    # ----------------------------------------------------------------
    def _score(x):
        return 2.0 * n * _max_convolution(x, Lfft)

    best = a.copy()
    best_score = _score(best)

    # ----------------------------------------------------------------
    # momentum variables for sub‑gradient descent
    # ----------------------------------------------------------------
    mom = np.zeros_like(a)
    mu = 0.9                 # momentum decay
    eta0 = 0.7               # initial step magnitude
    it = 0
    atol = 1e-12
    target = 1.5028          # early‑stop threshold

    while time.monotonic() < deadline:
        it += 1

        # current full convolution and its active indices
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv.max())
        active = np.where(conv >= M - atol)[0]
        if active.size == 0:
            break

        # sub‑gradient (average over all active lags)
        grad = np.zeros_like(a)
        for k in active:
            grad += _grad_for_k(a, int(k))
        grad /= float(active.size)

        # normalised direction
        norm = np.linalg.norm(grad)
        if norm > 0.0:
            direction = grad / norm
        else:
            direction = np.zeros_like(a)

        # momentum update
        mom = mu * mom + (1.0 - mu) * direction
        eta = eta0 / np.sqrt(it)
        cand = a - eta * mom
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_score = _score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            if best_score < target:
                break

        # accept if improvement or with a tiny exploration probability
        if cand_score < 2.0 * n * M or rng.random() < 0.005:
            a = cand

        # occasional random mass transfer to escape plateaus
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.12
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

    # ----------------------------------------------------------------
    # Return the best sequence found as plain Python list
    # ----------------------------------------------------------------
    return [float(x) for x in best]
