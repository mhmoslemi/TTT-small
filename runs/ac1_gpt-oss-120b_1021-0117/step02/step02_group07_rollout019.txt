import numpy as np, time, math

def _fft_len(n: int) -> int:
    """Length of FFT (next power of two) for convolution of length n."""
    return 1 << ((2 * n - 1).bit_length())


def _conv(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of seq with itself using FFT."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    return conv[: 2 * seq.size - 1]


def _score(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution, max coefficient)."""
    conv = _conv(seq, fft_len)
    max_b = float(conv.max())
    s = float(seq.sum())
    if s <= 0.0:
        return math.inf, conv, max_b
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv, max_b


def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i, decreasing positive sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _initial_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‐type distribution on n points, sum = 1."""
    # centre of each bin at (i+0.5)
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    w /= w.sum()
    return w
