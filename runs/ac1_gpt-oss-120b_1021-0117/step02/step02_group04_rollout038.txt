import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(a: np.ndarray, n: int, L: int) -> float:
    """Score = 2·n·max(conv) / (sum(a))² using FFT."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- initial sequence -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        cur = parent.copy()
    except Exception:
        n = 256
        cur = _gen_binomial(n)

    np.clip(cur, 0.0, 1000.0, out=cur)
    if cur.sum() < 0.01:
        cur += 0.01 / n

    L = _next_pow2(2 * n - 1)

    best = cur.copy()
    best_score = _ac1_score_fft(best, n, L)

    # ----- optimisation parameters -----------------------------------------
    temperature = 1.0
    temp_decay = 0.9996
    step_lr = 0.20
    lr_decay = 0.9998

    # ----- main loop --------------------------------------------------------
    while time.monotonic() < deadline:
