import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum = 1, v >= 0)."""
    if v.ndim != 1:
        raise ValueError("vector must be 1‑D")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution (FFT) – returns max value and its index
# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray, L: int):
    """Linear convolution via FFT, return (max, index)."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # reserve a couple of seconds for cleanup
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # Initial sequence
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: short uniform vector (length 256)
        n0 = 256
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    # put on the simplex (sum = 1, all non‑negative)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Score helpers
    # --------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        """Exact score according to the supplied evaluator."""
        max_val, _ = _conv_max_fft(vec, L)
        # because we keep sum(vec) == 1, the denominator is 1
        return 2.0 * n * max_val

    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    step = 0.4                # initial gradient step size
    min_step = 1e-5
    decay = 0.9995
    prob_swap = 0.02          # probability of a random swap move
    prob_perturb = 0.01       # small random mass‑transfer
    temperature = 0.1         # for simulated‑annealing acceptance

    # --------------------------------------------------------------
    # Main loop – projected‑gradient + occasional random moves
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- 1. gradient move aimed at reducing the current peak ----
        max_val, max_idx = _conv_max_fft(a, L)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1, dtype=int)
            g[j] = 2.0 * a[max_idx - j]          # ∂(a*a)_max / ∂a_j
        cand = a - step * g
        cand = _project_simplex(cand)

        cand_score = _score(cand)

        # simulated‑annealing style acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(step, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # ---- 2. occasional random swap of two entries ---------------
        if rng.random() < prob_swap:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                cand2 = a.copy()
                cand2[i], cand2[j] = cand2[j], cand2[i]
                # projection is unnecessary – swap preserves simplex
                cand2_score = _score(cand2)
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best = cand2.copy()
                        best_score = cand2_score

        # ---- 3. occasional tiny mass transfer between random slots --
        if rng.random() < prob_perturb:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand3 = a.copy()
                cand3[i] -= delta
                cand3[j] += delta
                cand3 = _project_simplex(cand3)
                cand3_score = _score(cand3)
                if cand3_score < cur_score:
                    a = cand3
                    cur_score = cand3_score
                    if cand3_score < best_score:
                        best = cand3.copy()
                        best_score = cand3_score

        # ---- 4. step‑size and temperature decay --------------------
        step = max(min_step, step * decay)
        temperature *= decay

        # early exit safeguard
        if time.monotonic() + 0.5 > deadline:
            break

    # --------------------------------------------------------------
    # Return the best vector found (plain Python list of floats)
    # --------------------------------------------------------------
    return [float(x) for x in best]
