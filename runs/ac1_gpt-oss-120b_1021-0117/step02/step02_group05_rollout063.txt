import numpy as np, time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound using FFT (no clipping – done outside)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

        which = rng.random()
        if which < 0.33:                                # pure exponential
            log_eps = rng.uniform(-5.0, -0.5)            # epsilon in [1e-5, 0.3]
            eps = 10.0 ** log_eps
            r = 1.0 - eps
            cand = np.power(r, idx)
        elif which < 0.66:                              # pure power law
            p = rng.uniform(0.0, 2.0)
            cand = np.power(idx1, -p)
        else:                                           # combined
            log_eps = rng.uniform(-5.0, -0.5)
            eps = 10.0 ** log_eps
            r = 1.0 - eps
            p = rng.uniform(0.0, 2.0)
            cand = np.power(r, idx) * np.power(idx1, -p)

        np.clip(cand, 0.0, 1000.0, out=cand)
        if cand.sum() < 0.01:
            continue
        cand_score = _ac1_score(cand, n, L)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            cur = cand.copy()
            cur_score = cand_score

    # ----------------------------------------------------------------------
    # Simulated‑annealing refinement.
    temperature = 1.0
    decay = 0.9996
    grad_eps = 0.12          # base step for gradient moves
    transfer_max = 0.45

    while time.monotonic() < deadline:
        move = rng.random()

        # -------------------- 1) gradient move (≈ 50 %) --------------------
        if move < 0.5:
            # convolution & index of maximal entry
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            g *= 2.0                     # derivative of a_i a_{kmax-i}
            d = g.mean() - g             # direction that preserves total mass
            cand = cur + grad_eps * d

        # -------------------- 2) block scaling (≈ 30 %) --------------------
        elif move < 0.8:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.10))
            cand = cur.copy()
            cand[start:start + width] *= factor

        # -------------------- 3) random transfer (≈ 20 %) ---------------
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * transfer_max
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # enforce bounds and keep sum positive
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate
        cand_score = _ac1_score(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score

        temperature = max(temperature * decay, 1e-12)

        # occasional restart from the best seen
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ----------------------------------------------------------------------
    # If nothing succeeded, fall back to a simple uniform vector.
    if best is None:
        best = np.full(n, 1.0, dtype=np.float64)

    # final safe‑clipping (evaluator also clips, but we keep within limits)
    np.clip(best, 0.0, 1000.0, out=best)

    # return as plain Python floats
    return [float(x) for x in best]
