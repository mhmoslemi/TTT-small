import numpy as np

def _sqrt_series(n: int) -> np.ndarray:
    """
    Return the coefficients a_0,…,a_{n-1} of the truncated series
    sqrt(1 + x + … + x^{2n-2}) so that (a*a)_k = 1 for 0 ≤ k ≤ n‑1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:

            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) * 0.5
    return a
