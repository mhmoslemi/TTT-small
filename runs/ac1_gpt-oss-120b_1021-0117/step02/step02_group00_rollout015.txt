def _cross_corr_fft(w, a):   # w length m=2n-1, a length n
    n = a.shape[0]
    m = w.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    W_rev = np.fft.rfft(w[::-1], n=L)
    conv = np.fft.irfft(A * W_rev, n=L)
    # conv length L, but only first n+m-1 entries are linear conv
    # gradient = 2 * conv[m-1 : m-n-1 : -1]
    grad = 2.0 * conv[m-1 : m-n-1 : -1]
    return grad

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution a*a using FFT (real -> real)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2*len(a)-1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


# ------------------------------------------------------------
# Gradient of the smoothed max (log‑sum‑exp) objective.
# ------------------------------------------------------------
def _grad_from_weights(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Given probability weights w (length 2*n-1) and current a (length n),
    compute 2 * cross‑correlation(w, a) via FFT.  L must be a power of two
    >= n + len(w) - 1 (pre‑computed for the problem size).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2*n-1
    A = np.fft.rfft(a, n=L)
    W_rev = np.fft.rfft(w[::-1], n=L)
    conv = np.fft.irfft(A * W_rev, n=L)   # linear convolution a * w[::-1]
    start = m - 1
    end = m - n - 1
    # slice in reverse order to obtain the needed n entries
    return 2.0 * conv[start:end:-1]


# ------------------------------------------------------------
# Objective value (same formula as evaluate_sequence)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Score = 2 * n * max(conv) / (sum(a)**2). Assumes a non‑negative."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initial vector (use provided parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform distribution of modest length
        n_fallback = 256
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # make it a valid simplex point
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s

    n = a0.shape[0]
    m = 2 * n - 1                     # length of autocorrelation
    L = 1 << ((n + m - 1).bit_length())   # FFT size for correlation steps

    # ------------------------------------------------------------
    # State initialization
    # ------------------------------------------------------------
    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # momentum term for accelerated updates
    velocity = np.zeros_like(cur)

    # smoothing parameter (log‑sum‑exp epsilon)
    conv0 = _conv_fft(cur)
    epsilon = max(conv0.max() * 0.2, 1e-6)
    eps_decay = 0.9995
    eps_min = 1e-6

    beta = 0.9               # momentum coefficient
    base_lr = 0.2            # base step size factor
    no_improve = 0

    # ------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---------- convolution of current iterate (self‑correlation) ----------
        A = np.fft.rfft(cur, n=L)
        conv_self = np.fft.irfft(A * A, n=L)[:m]      # length 2*n-1
        M = conv_self.max()

        # ---------- smoothed weights (soft‑max over autocorrelation) ----------
        # Subtract M for numeric stability; values are <= 0
        w = np.exp((conv_self - M) / epsilon)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the smoothed max ----------
        grad = _grad_from_weights(w, cur, L)

        # adaptive learning rate based on gradient magnitude
        max_abs = np.max(np.abs(grad))
        lr = base_lr / (max_abs + 1e-12)

        # ---------- momentum update + projection ----------
        velocity = beta * velocity - lr * grad
        cand = cur + velocity
        cand = _project_simplex(cand)

        # ---------- evaluate candidate ----------
        conv_cand = _conv_fft(cand)
        cand_sc = float(2.0 * n * conv_cand.max() / (cand.sum() ** 2))

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                no_improve = 0
            else:
                no_improve += 1
        else:
            # try a plain gradient step without momentum
            cand2 = cur - lr * grad
            cand2 = _project_simplex(cand2)
            conv_cand2 = _conv_fft(cand2)
            cand2_sc = float(2.0 * n * conv_cand2.max() / (cand2.sum() ** 2))

            if cand2_sc < cur_sc:
                cur, cur_sc = cand2, cand2_sc
                if cand2_sc < best_sc:
                    best_vec, best_sc = cand2.copy(), cand2_sc
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # no improvement – damp momentum
                velocity *= 0.5
                no_improve += 1

        # ---------- schedule for epsilon ----------
        epsilon = max(epsilon * eps_decay, eps_min)

        # ---------- occasional random jitter when stuck ----------
        if no_improve > 2500:
            jitter = rng.normal(scale=1e-3, size=n)
            cur = _project_simplex(best_vec + jitter)
            cur_sc = _score(cur)
            velocity.fill(0.0)
            no_improve = 0

        # ---------- very occasional random restart ----------
        if rng.random() < 0.001:
            rand = rng.random(n)
            rand /= rand.sum()
            cur = _project_simplex(0.7 * cur + 0.3 * rand)
            cur_sc = _score(cur)
            velocity.fill(0.0)

        # ---------- early exit if target reached ----------
        if best_sc < 1.50274349:
            break

    # Return the best vector found as a plain Python list
    return [float(x) for x in best_vec]
