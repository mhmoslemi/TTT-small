import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: project onto the probability simplex (Euclidean proj.)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast full linear convolution via FFT
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Build the constraint matrix for the LP:
#   (f * g)_i = sum_j f_j * g_{i-j} <= M   for all i
# ------------------------------------------------------------
def _build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    n = f.size
    rows = 2 * n - 1
    data = []
    indices = []
    indptr = [0]
    for i in range(rows):
        lo = max(0, i - (n - 1))
        hi = min(i, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        data.extend(f[idx].tolist())
        indices.extend(idx.tolist())
        indptr.append(len(data))
    return sp.csr_matrix((data, indices, indptr), shape=(rows, n))

    else:

        n0 = 256

            else:

            a[k] = (1.0 - s) / 2.0
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    n = a.size
    # store best solution seen
    best = a.copy()
    conv = _conv_fft(best)
    best_sc = 2.0 * n * conv.max()  # sum(a)=1 ⇒ denominator =1
