import time
import numpy as np

# ------------------------------------------------------------
# Simple projection onto the probability simplex (non‑negative,
# sum = 1). Faster than a full simplex projection for our use‑case.
# ------------------------------------------------------------
def _normalize_nonneg(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v / s
    else:
        v = np.full_like(v, 1.0 / v.shape[0])
    return v


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2}), used only if no parent.
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Gradient of the active convolution lag k (hard gradient).
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]   # correct double count for diagonal term
    return grad

    # --------------------------------------------------------------------
    return [float(x) for x in best]
