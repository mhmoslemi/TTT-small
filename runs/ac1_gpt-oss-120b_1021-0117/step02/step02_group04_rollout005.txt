def _gradient_step(a, L, step_lr):
    n = a.size
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    M = conv.max()
    k = int(conv.argmax())
    S = a.sum()
    # compute gradient g_i
    if k == 0:
        g = 2.0 * a
    else:
        g = np.zeros_like(a)
        # contributions from shift k
        # for i where i + k < n: add a[i + k]
        g[:n - k] += a[k:]  # a[i + k] for i = 0..n-k-1
        # for i where i - k >= 0: i >= k, add a[i - k]
        g[k:] += a[:n - k]  # a[i - k] for i = k..n-1
    # compute grad of f (excluding factor 2*n)
    scalar = 2.0 * M / (S ** 3)
    grad = g / (S ** 2) - scalar
    # step: a - step_lr * grad
    cand = a - step_lr * grad
    # clip to >=0
    np.clip(cand, 0.0, None, out=cand)
    # renorm sum to 1
    total = cand.sum()
    if total < 1e-12:
        # fallback to uniform
        cand = np.full_like(a, 1.0 / n)
    else:
        cand *= (1.0 / total)
    return cand

import math
import time
import numpy as np

# --- fast helpers -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 score using FFT (identical to evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _gradient_step(a: np.ndarray, L: int, step_lr: float) -> np.ndarray:
    """One projected gradient step w.r.t. the AC1 objective."""
    n = a.size
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    M = float(conv.max())
    k = int(conv.argmax())
    S = float(a.sum())

    # gradient of the max‑lag term
    if k == 0:
        g = 2.0 * a
    else:
        g = np.zeros_like(a)
        # a[i] * a[i+k]
        g[: n - k] += a[k:]
        # a[i] * a[i-k]
        g[k:] += a[: n - k]

    # full gradient of f = M / S**2 (ignoring constant 2n)
    scalar = 2.0 * M / (S ** 3)
    grad = g / (S ** 2) - scalar

    cand = a - step_lr * grad
    np.clip(cand, 0.0, None, out=cand)
    total = cand.sum()
    if total < 1e-12:
        cand[:] = 1.0 / n
    else:
        cand *= (1.0 / total)
    return cand

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---- initialise ------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:
        parent = None

    n = parent.size if parent is not None else 256
    L = _next_pow2(2 * n - 1)

    if parent is not None:
        base = parent.astype(np.float64).copy()
        base = np.clip(base, 0.0, None)
    else:
        base = np.ones(n, dtype=np.float64)

    # normalise to a probability vector (sum=1)
    s = base.sum()
    if s < 1e-12:
        base[:] = 1.0 / n
    else:
        base *= (1.0 / s)

    best = base.copy()
    best_score = _ac1_score(best, n, L)

    # schedule parameters
    step_lr = 0.07
    step_decay = 0.99996

    # ---- search loop ------------------------------------------------
    while time.monotonic() < deadline:
        r = rng.random()
        # --- candidate generation ------------------------------------
        if r < 0.45:
            # projected gradient from the current best
            cand = _gradient_step(best, L, step_lr)
        elif r < 0.70:
            # beta‑type shape  x^α (1‑x)^β
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            x = np.linspace(0.0, 1.0, n)
            cand = (x ** a) * ((1.0 - x) ** b)
        elif r < 0.85:
            # sine‑power shape  sin(πx)^p
            p = rng.uniform(0.5, 5.0)
            x = np.linspace(0.0, 1.0, n)
            cand = np.sin(math.pi * x) ** p
        else:
            # parabola‑power shape  ((i+1)*(n‑i))^p
            p = rng.uniform(0.1, 3.0)
            idx = np.arange(n, dtype=np.float64)
            base_par = (idx + 1.0) * (n - idx)
            cand = base_par ** p

        # enforce feasibility
        np.clip(cand, 0.0, None, out=cand)
        tot = cand.sum()
        if tot < 1e-12:
            cand[:] = 1.0 / n
        else:
            cand *= (1.0 / tot)

        # short local refinement (2 gradient steps)
        for _ in range(2):
            fa = np.fft.rfft(cand, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            M = float(conv.max())
            k = int(conv.argmax())
            S = float(cand.sum())
            if k == 0:
                g = 2.0 * cand
            else:
                g = np.zeros_like(cand)
                g[: n - k] += cand[k:]
                g[k:] += cand[: n - k]
            scalar = 2.0 * M / (S ** 3)
            grad = g / (S ** 2) - scalar
            cand = cand - step_lr * grad
            np.clip(cand, 0.0, None, out=cand)
            tot = cand.sum()
            if tot < 1e-12:
                cand[:] = 1.0 / n
            else:
                cand *= (1.0 / tot)

        cand_score = _ac1_score(cand, n, L)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # decay step size
        step_lr *= step_decay

        # occasional random restart to avoid stagnation
        if rng.random() < 0.001:
            # reset best to a fresh random shape (keeps current best)
            restart = np.sin(np.pi * rng.random())  # dummy to trigger new shape next loop
            _ = restart  # silence unused variable warning

    # final scaling (any positive scalar leaves the score unchanged)
    final_scale = math.sqrt(2.0 * n)
    best_scaled = best * final_scale
    return [float(x) for x in best_scaled]
