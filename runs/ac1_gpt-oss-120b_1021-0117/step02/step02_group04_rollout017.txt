def _ac1_score(arr, n, L):
    fa = np.fft.rfft(arr, n=L)

def _gradient(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = float(conv.max())
    k_max = int(conv.argmax())
    sum_a = float(arr.sum())
    if sum_a == 0.0:
        return np.zeros_like(arr), max_b, k_max, sum_a
    # build gradient of M
    g = np.zeros_like(arr)
    i = np.arange(arr.shape[0])
    j1 = k_max - i
    mask1 = (j1 >= 0) & (j1 < n)
    g[mask1] += arr[j1[mask1]]
    j2 = i - k_max
    mask2 = (j2 >= 0) & (j2 < n)
    g[mask2] += arr[j2[mask2]]
    # coefficients
    C1 = 2.0 * n / (sum_a * sum_a)
    C2 = 4.0 * n * max_b / (sum_a * sum_a * sum_a)
    grad = C1 * g - C2
    return grad, max_b, k_max, sum_a
```

import math
import time

import numpy as np

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (classic binomial step)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _gradient(arr: np.ndarray, n: int, L: int):
    """Return gradient of the AC1 bound w.r.t. `arr` and auxiliary values."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    k_max = int(conv.argmax())
    sum_a = float(arr.sum())
    # derivative of the max‑convolution term
    g = np.zeros_like(arr)
    idx = np.arange(n)
    j1 = k_max - idx
    m1 = (j1 >= 0) & (j1 < n)
    g[m1] += arr[j1[m1]]
    j2 = idx - k_max
    m2 = (j2 >= 0) & (j2 < n)
    g[m2] += arr[j2[m2]]
    # constants
    C1 = 2.0 * n / (sum_a * sum_a)
    C2 = 4.0 * n * max_b / (sum_a * sum_a * sum_a)
    grad = C1 * g - C2
    return grad, max_b, k_max, sum_a


# ---------- main entry ----------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- problem size & initial sequence -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        base = parent.astype(np.float64)
    except Exception:
        n = 256
        base = _gen_binomial(n)

    L = _next_pow2(2 * n - 1)

    # ensure validity
    np.clip(base, 0.0, 1000.0, out=base)

    # best-so‑far bookkeeping
    best = base.copy()
    best_score = _ac1_score_fft(best, n, L)

    # current state
    cur = best.copy()
    cur_score = best_score

    # optimisation parameters
    step = 0.2
    min_step = 1e-6
    momentum = np.zeros_like(cur)
    beta = 0.9  # momentum factor
    temperature = 1.0
    temp_decay = 0.9995
    target = 1.50274349

    # main loop
    while time.monotonic() < deadline:
        # ---- gradient based move ----
        grad, M, k_max, S = _gradient(cur, n, L)
        momentum = beta * momentum + (1.0 - beta) * grad
        cand = cur - step * momentum
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _ac1_score_fft(cand, n, L)

        # ---- occasional random block scaling ----
        if rng.random() < 0.3:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.08))
            cand2 = cur.copy()
            cand2[start:start + width] *= factor
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            cand2_score = _ac1_score_fft(cand2, n, L)
            if cand2_score < cand_score:
                cand, cand_score = cand2, cand2_score

        # ---- occasional mass‑transfer move ----
        if rng.random() < 0.2:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.4
                delta = cur[i] * frac
                cand3 = cur.copy()
                cand3[i] -= delta
                cand3[j] += delta
                np.clip(cand3, 0.0, 1000.0, out=cand3)
                cand3_score = _ac1_score_fft(cand3, n, L)
                if cand3_score < cand_score:
                    cand, cand_score = cand3, cand3_score

        # ---- update global best (even if not accepted) ----
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        # ---- simulated‑annealing acceptance for the current state ----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            step = min(step * 1.02, 5.0)
        else:
            step = max(step * 0.5, min_step)

        # ---- occasional reset to the best known ----
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            momentum.fill(0.0)
            temperature = 1.0

        temperature *= temp_decay

    return [float(x) for x in best]
