def evaluate_sequence(sequence: list) -> float:
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

    otherwise falls back to a modest default length.
    The sequence a_i ∝ i^p (p=5) is normalised to sum to 1.
    This shape yields a very small maximal autocorrelation and therefore a
    low AC1 upper bound.
    """
    # Length of the problem – use the parent if it exists
    n = len(height_sequence_1) if "height_sequence_1" in globals() else 256

    # Power exponent – p=5 gives an excellent trade‑off between sparsity
    # and maximal self‑convolution for typical n (≈2500).
    p = 5

    # Build the raw values i^p for i = 1 … n
    i = np.arange(1, n + 1, dtype=np.float64)
    raw = i ** p

    # Normalise to sum = 1 (the objective is scale‑invariant)
    a = raw / raw.sum()

    # Return a plain Python list of floats
    return [float(v) for v in a]
