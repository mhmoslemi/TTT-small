import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------
def _scale_to_simplex(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum exactly 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Convolution a*a using real FFT, returns the full (2*n‑1) array."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * a.size - 1]


def _gradient_multi_peak(a: np.ndarray, peaks: np.ndarray) -> np.ndarray:
    """Gradient of Σ_{k∈peaks} (a*a)[k] w.r.t. a (non‑negative a)."""
    n = a.size
    g = np.zeros_like(a)
    for k in peaks:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        j = np.arange(lo, hi + 1, dtype=int)
        g[j] += 2.0 * a[k - j]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # implementation from the previous solution (stable O(n log n))
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Initialise sequence (use provided parent if present)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback – modest length uniform vector
        a = np.ones(256, dtype=np.float64)

    a = _scale_to_simplex(a)               # sum = 1, non‑negative
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())    # power‑of‑two FFT length

    # initial score (sum = 1 ⇒ score = 2·n·max_conv)
    conv = _conv_fft(a, L)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    step = 0.5                     # initial EG step size
    min_step = 5e-6
    decay = 0.9997
    top_k = 4                      # number of convolution peaks used for gradient
    prob_random = 0.006            # frequency of extra random mass transfers

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution and peak set -----
        conv = _conv_fft(a, L)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # pick top‑k indices (unsorted, cheap)
        if top_k == 1:
            peak_idxs = np.array([int(conv.argmax())], dtype=int)
        else:

        # ----- gradient w.r.t. chosen peaks -----
        g = _gradient_multi_peak(a, peak_idxs)
