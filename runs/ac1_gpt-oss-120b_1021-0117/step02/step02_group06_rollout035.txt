def gradient_softmax(a, lambda_val):
    # a shape (n,)
    n = a.shape[0]
    # compute conv = a*a (full)
    conv = conv_self_fft(a, n, fft_len)  # length 2*n-1
    # compute weights softmax (numerically stable)
    maxc = conv.max()
    exps = np.exp(lambda_val * (conv - maxc))
    w = exps / exps.sum()
    # compute reversed w for gradient
    w_rev = w[::-1]   # but w length 2n-1; reversed near center.
    # compute convolution of a with w_rev using FFT
    conv2_len = 1 << ((n + (2*n-1) - 2).bit_length())
    A = np.fft.rfft(a, n=conv2_len)
    B = np.fft.rfft(w_rev, n=conv2_len)
    g_full = np.fft.irfft(A * B, n=conv2_len)
    # extract needed slice: index range (2n-2 - (n-1)) to (2n-2) => (n-1) ... (2n-2)? Let's compute.
    # For j from 0..n-1, g_j = 2 * conv2[2n-2 - j]

import time
import numpy as np

# ---------- helper utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a using real FFT (length 2*len(a)-1)."""
    A = np.fft.rfft(a, n=fft_len)
    full = np.fft.irfft(A * A, n=fft_len)
    n = a.shape[0]
    return full[: 2 * n - 1]


def _gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     lam: float,
                     fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv) w.r.t. a.
    Returns ∇_a ( (1/lam) log Σ exp(lam*conv) ).
    """
    # soft‑max weights
    maxc = conv.max()
    w = np.exp(lam * (conv - maxc))
    w /= w.sum()
    # reversed weights for convolution
    w_rev = w[::-1]                     # length 2*n-1
    # convolution a * w_rev  (full length 3*n-2)
    A = np.fft.rfft(a, n=fft_len_grad)
    B = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(A * B, n=fft_len_grad)
    n = a.shape[0]
    # needed slice: indices n-1 … 2*n-2 (inclusive), length n
    seg = full[n - 1:2 * n - 1]
    grad = 2.0 * seg[::-1]              # 2 * Σ_i w_i a_{i‑j}
    return grad


def _score_of(a: np.ndarray) -> float:
    """Exact AC1 upper bound used by the evaluator."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = a.shape[0]
    conv = _conv_self_fft(a, _next_pow2(2 * n - 1))
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def _next_pow2(x: int) -> int:
    """Next power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """Generate a few plausible normalized start vectors."""
    cand = []

    # use parent if provided
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        if p.sum() > 0.0:
            cand.append(p / p.sum())

    # Catalan‑style weights
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    cand.append(w / w.sum())

    # polynomial bump (i+1)*(n-i) raised to powers
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for pwr in (0.5, 1.0, 1.5, 2.0, 3.0):
        v = base ** pwr
        cand.append(v / v.sum())

    # sinusoidal shapes
    theta = np.pi * (idx + 0.5) / n
    sinv = np.sin(theta)
    for pwr in (0.5, 1.0, 2.0):
        v = sinv ** pwr
        cand.append(v / v.sum())

    # exponential decay shapes
    for alpha in (0.001, 0.005, 0.01, 0.02):
        v = np.exp(-alpha * idx)
        cand.append(v / v.sum())

    return cand
