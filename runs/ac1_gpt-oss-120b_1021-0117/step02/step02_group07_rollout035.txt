def propose_candidate(...):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s)-1.)  # to be safe

    # load parent or initialize binomial
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            n = arr.size
            parent = np.clip(arr, 0.0, 1000.0)
        else:
            raise Exception
    except Exception:
        parent = None
        n = 3809

    def binomial_seq(m):
        a = np.empty(m, dtype=np.float64)
        a[0] = 1.0
        for i in range(1,m):
            a[i] = a[i-1] * (2*i - 1) / (2*i)
        return a

    if parent is not None:
        best_seq = parent.copy()
        best_score,_ = compute_score(best_seq)
    else:
        best_seq = binomial_seq(n)
        best_score,_ = compute_score(best_seq)

    # also evaluate parent if exists and maybe choose better
    if parent is not None and parent is not best_seq:
        parent_score,_ = compute_score(parent)
        if parent_score < best_score:
            best_seq = parent.copy()
            best_score = parent_score

    current = best_seq.copy()
    current_score = best_score

    fft_len = 1 << ((2 * n - 1).bit_length())

    # define conv and score functions using fft
    def conv_fft(seq):
        f = np.fft.rfft(seq, n=fft_len)
        conv_full = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
        return conv_full

    def compute_score(seq):
        conv = conv_fft(seq)
        M = np.max(conv)
        S = np.sum(seq)
        score = 2.0 * n * M / (S * S) if S > 0 else float('inf')
        return float(score), conv

    # parameters
    step0 = 0.01
    min_step = 1e-6
    max_step = 0.1
    iter_cnt = 0
    last_random = 0

    while time.monotonic() < deadline:
        iter_cnt += 1
        # compute gradient based step
        conv = conv_fft(current)
        kmax = int(np.argmax(conv))
        M = conv[kmax]
        S = np.sum(current)
        lo = max(0, kmax - (n-1))
        hi = min(kmax, n-1)
        idx_i = np.arange(lo, hi+1, dtype=np.int64)
        idx_j = kmax - idx_i
        gradM = np.zeros_like(current)
        gradM[idx_i] = 2.0 * current[idx_j]
        # direction = 2*M - S*gradM
        direction = 2.0 * M - S * gradM

        # schedule step size
        elapsed = time.monotonic() - start
        total = deadline - start
        frac = 1.0 - elapsed / total
        step = max(min_step, step0 * frac)

        # try a few multipliers

        best_local_seq = current
        for factor in (0.5, 1.0, 2.0):
            cand = current + step * factor * direction
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score, _ = compute_score(cand)
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local_seq = cand

        # random perturbation every 20 iterations to diversify
        if iter_cnt % 20 == 0:
            op = rng.integers(0,3)
            cand = best_local_seq.copy()
            if op == 0:

                w = rng.integers(1, max(2, n // 40))
                start_idx = rng.integers(0, n - w + 1)
                factor = np.exp(rng.normal(0.0, 0.08))

                best_local_seq = cand

                current = best_local_seq
                current_score = best_local_score
                if current_score < best_score:
                    best_seq = current.copy()
                    best_score = current_score

    return [float(x) for x in best_seq]

                cand[start_idx:start_idx + w] *= factor
            elif op == 1:                                 # single index
                idx = rng.integers(0, n)
                factor = math.exp(rng.normal(0.0, 0.18))
                cand[idx] *= factor
            elif op == 2:                                 # early‑to‑late mass transfer
                early_end = max(1, n // 12)
                late_start = n - early_end
                i = rng.integers(0, early_end)
                j = rng.integers(late_start, n)
                frac = rng.uniform(0.05, 0.2)
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta
            else:                                         # smooth neighbours
                i = rng.integers(0, n - 1)
                avg = (cand[i] + cand[i + 1]) * 0.5
                cand[i] = avg
                cand[i + 1] = avg
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score, _ = _objective_score(cand, n, fft_len)
            if cand_score < best_local_score:
                best_local_seq = cand
                best_local_score = cand_score

        # ---- Acceptance ------------------------------------------------
        if best_local_score < current_score:
            current = best_local_seq
            current_score = best_local_score
            if current_score < best_score:
                best_seq = current.copy()
                best_score = current_score
        else:
            # occasional uphill acceptance (simulated‑annealing style)
            elapsed = time.monotonic() - start
            temp = 0.04 * (1.0 - elapsed / (deadline - start))
            if temp > 0.0 and rng.random() < math.exp((current_score - best_local_score) / (temp + 1e-12)):
                current = best_local_seq
                current_score = best_local_score
                if current_score < best_score:
                    best_seq = current.copy()
                    best_score = current_score

    # Return a plain Python list for the evaluator
    return [float(x) for x in best_seq]
