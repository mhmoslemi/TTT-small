import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution(a: np.ndarray, L: int) -> float:
    """Maximum of the full convolution a*a using FFT (length L)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())

def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the active autocorrelation lag k (full gradient)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # initialise
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # safety: enforce bounds and simplex
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() == 0.0:
        a[:] = 1.0 / a.size
    else:
        a = a / a.sum()
    a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # bookkeeping
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(best, L) / (best.sum() ** 2)

    iteration = 0
    eta0 = 0.7  # initial step size

    while time.monotonic() < deadline:
        iteration += 1

        # current convolution, max and its index
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        M = float(conv.max())
        k = int(conv.argmax())
        cur_score = 2.0 * n * M / (a.sum() ** 2)

        grad = _grad_for_k(a, k)
        norm = np.linalg.norm(grad)
        if norm > 0.0:
            eta = eta0 / np.sqrt(iteration)
            cand = a - eta * (grad / norm)
        else:
            cand = a.copy()

        cand = _project_simplex(cand)

        # evaluate candidate
        cand_sum = cand.sum()
        if cand_sum < 0.01:
            cand_score = np.inf
        else:
            maxb = _max_convolution(cand, L)
            cand_score = 2.0 * n * maxb / (cand_sum ** 2)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()

        # accept if improvement or small random walk
        if cand_score < cur_score or rng.random() < 0.003:
            a = cand

        # occasional random mass transfer to escape plateaus
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.1
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

        if best_score < 1.5028:
            break

    return [float(x) for x in best]
