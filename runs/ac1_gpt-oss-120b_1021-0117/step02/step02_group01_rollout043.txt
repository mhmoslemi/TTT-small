def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def _conv_max(seq, L):
    """Return (max_val, max_idx, full_conv) for autocorrelation of seq."""
    fv = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(seq) - 1]
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx, conv

def _sqrt_series(n):
    """Analytic sqrt series: a*a = 1+…+x^{2n‑2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

                best_score = cand_score
                best = cand.copy()

        temperature = max(1e-6, temperature * temp_decay)

    # ------------------------------------------------------------------
    # return result as plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(seq) - 1]
