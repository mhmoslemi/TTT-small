def _ac1_score_fft(a, L):
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*len(a)-1]
    maxb = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2 * len(a) * maxb / (s * s)
