def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    return conv

import numpy as np, time, math

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2 * n - 1]
    return conv

def _score(vec):
    n = vec.shape[0]
    conv = _conv_fft(vec)
    return 2.0 * n * conv.max() / (vec.sum() ** 2)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast full convolution via FFT
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Objective: 2 * n * max(conv) / (sum(a)**2)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise: use the provided parent if present, otherwise a simple
    # fallback based on the sqrt‑series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 256
        a = np.empty(n0, dtype=np.float64)

        for k in range(1, n0):

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a.fill(1.0 / a.size)
    else:
        a = a / total

    # optional symmetrisation (most good solutions are symmetric)
    a = (a + a[::-1]) * 0.5
    n = a.size

    # current iterate and best‑so‑far
    cur = a.copy()
    cur_sc = _score(cur)

    best = cur.copy()

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        M = conv.max()

        # pick a few indices attaining (or nearly attaining) the maximum
        if top_peaks >= conv.size:

        # compute averaged sub‑gradient of the active peaks
