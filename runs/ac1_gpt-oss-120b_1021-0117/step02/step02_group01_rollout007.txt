    return conv

def propose_candidate(seed=42,budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        s = np.sum(a)
        if s <= 0:
            a = np.ones_like(a)
            s = np.sum(a)
        a = a / s
    else:
        # fallback: sqrt series length maybe 256
        a = _sqrt_series(256)
        a = a / np.sum(a)

    n = a.size
    L = 1 << ((2 * n - 2).bit_length())   # smallest power-of-two >= 2n-1

    # Evaluate initial
    conv = _conv_full(a, L)
    max_c = np.max(conv)
    cur_score = 2.0 * n * max_c
    best = a.copy()
    best_score = cur_score

    # Parameters
    step = 0.5
    min_step = 1e-6
    decay = 0.9995
    rel_tol = 0.02   # active set relative tolerance
    prob_random = 0.02
    stagnation_counter = 0
    stagnation_limit = 2000

    # momentum
    momentum_coeff = 0.9
    momentum = np.zeros_like(a)

    while time.monotonic() < deadline:
        # recompute conv
        conv = _conv_full(a, L)
        max_c = np.max(conv)
        # active indices:
        active = np.where(conv >= max_c * (1 - rel_tol))[0]
        # compute gradient
        g = np.zeros_like(a)
        for i in active:
            lo = max(0, i - (n - 1))
            hi = min(n - 1, i)
            idx = np.arange(lo, hi + 1)
            # a_i_minus_idx = a[i - idx]
            g[idx] += 2.0 * a[i - idx]
        # incorporate momentum
        momentum = momentum_coeff * momentum + (1 - momentum_coeff) * g
        direction = momentum

        # attempt gradient step with backtracking
        improved = False
        cur_step = step
        for _ in range(5):
            cand = a - cur_step * direction
            cand = _project_simplex(cand)
            # compute candidate score
            conv_cand = _conv_full(cand, L)
            max_c_cand = np.max(conv_cand)
            cand_score = 2.0 * n * max_c_cand
            if cand_score < cur_score - 1e-12:
                # accept
                a = cand
                cur_score = cand_score
                max_c = max_c_cand
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best = cand.copy()
                break
            else:
                cur_step *= 0.5
        # update step size
        step = max(min_step, step * decay)

        if not improved:
            stagnation_counter += 1
        else:
            stagnation_counter = 0

        # random mass transfer
        if rng.random() < prob_random:
            i = rng.integers(n)
            if a[i] > 0:
                j = rng.integers(n-1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                conv2 = _conv_full(cand2, L)
                cand2_score = 2.0 * n * np.max(conv2)
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    max_c = np.max(conv2)
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                    stagnation_counter = 0

        # optional restart if stagnated
        if stagnation_counter >= stagnation_limit:
            # restart from best or random
            if rng.random() < 0.5:
                a = best.copy()
            else:
                # random Dirichlet: sample exponential and normalize
                a = rng.random(n)  # uniform, but we can sample exponentials
                a = -np.log(rng.random(n) + 1e-12)  # exponential distribution
                a = a / np.sum(a)
            stagnation_counter = 0
            # step reset
            step = max(0.5, step)
            momentum = np.zeros_like(a)

        # early exit check
        if time.monotonic() + 0.5 > deadline:
            break

    return [float(x) for x in best]

import time
import numpy as np

# --------------------------------------------------------------
# Simplex projection (Euclidean) – keeps sum = 1, entries >= 0
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Analytic sqrt‑series (fallback initializer)
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# --------------------------------------------------------------
# Full convolution via FFT – returns the whole vector
# --------------------------------------------------------------
def _conv_full(vec: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    # keep only the valid length (2*len(vec)-1)
    return conv[: 2 * vec.size - 1]
