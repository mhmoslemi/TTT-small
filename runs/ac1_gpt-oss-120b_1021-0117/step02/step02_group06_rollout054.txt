def _initial_candidates(n, parent_arr):
    candidates = []
    # parent
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0, 1000)
        s = p.sum()
        if s > 0:
            candidates.append(p / s)
    # Catalan weight
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
    candidates.append(w / w.sum())
    # power shape base = (i+1)*(n-i)
    idx = np.arange(n, dtype=np.float64)
    base = (idx+1)*(n-idx)
    for p in (0.5, 1.0, 1.5, 2.0):
        v = base ** p
        candidates.append(v / v.sum())
    # sin power shape
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sin_vals ** p
        candidates.append(v / v.sum())
    return candidates

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """Generate a handful of plausible starting vectors (each sums to 1)."""
    cand = []

    # 1. use parent if present
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # 2. Catalan‑style weights
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    cand.append(w / w.sum())

    # 3. Polynomial “tent” shapes a_i ∝ ((i+1)*(n-i))^p
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        v = base ** p
        cand.append(v / v.sum())

    # 4. Sin‑power shapes a_i ∝ sin^p(π*(i+0.5)/n)
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sin_vals ** p
        cand.append(v / v.sum())

    return cand

    Return a non‑negative step‑function (list of heights) that minimises the
    AC1 upper bound used by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # safety margin

    # ----- obtain parent if available -----

        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            parent_arr = None
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256

    # ----- pick a good initial vector -----
    best_vec = None
    best_score = np.inf
    for vec in _initial_candidates(n, parent_arr):
        v = np.maximum(vec, 0.0)
        s = v.sum()
        if s == 0.0:
            continue
        v = v / s
        L0 = 1 << ((2 * n - 1).bit_length())
        conv = np.fft.irfft(np.fft.rfft(v, L0) * np.fft.rfft(v, L0), n=L0)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = v.copy()
        if best_score <= 1.50274349:
            return best_vec.tolist()

    if best_vec is None:
        # fallback uniform distribution
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * (1.0 / n)   # = 2.0

    # ----- main optimisation loop -----
    # FFT length for all subsequent transforms (needs to accommodate 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    a = best_vec.copy()
    base_alpha = 0.5          # initial step‑size base
    eps = 0.02                # initial smoothing parameter
    min_eps = 1e-6
    eps_decay_iter = 200
    iter_no_improve = 0
    max_no_improve = 1500
    it = 0

    while time.monotonic() < deadline:
        # ----- FFT of current iterate -----
        A = np.fft.rfft(a, L)

        # ----- self‑convolution (linear) -----
        conv_self = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        # ----- evaluate current objective -----
        cur_max = conv_self.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            iter_no_improve += 1

        # ----- smoothed max (soft‑max) -----
        M = cur_max
        cur_eps = max(eps, min_eps)
        # stable exponentials
        exps = np.exp((conv_self - M) / cur_eps)
        w = exps / exps.sum()                     # soft‑max weights, length 2n‑1

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, L)
        conv_rev = np.fft.irfft(A * W_rev, n=L)[: 3 * n - 2]   # linear a * w_rev
        idx = (2 * n - 2) - np.arange(n)                      # positions yielding correlation
        grad = 2.0 * conv_rev[idx]

        # ----- projected gradient step -----
        alpha = base_alpha / np.sqrt(it + 1)
        a_new = a - alpha * grad
        a_new = _project_onto_simplex(a_new)

        # occasional multiplicative jitter to escape plateaus
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.009, size=n))
            a_new = _project_onto_simplex(a_new * jitter)

        a = a_new

        # ----- schedule adjustments -----
        if it % eps_decay_iter == 0 and eps > min_eps:
            eps *= 0.7

        if iter_no_improve > max_no_improve:
            # restart from the current best with a small perturbation
            pert = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _project_onto_simplex(best_vec * pert)
            iter_no_improve = 0
            eps = max(eps, 0.01)   # warm‑up epsilon a bit

        it += 1

    # ----- final answer -----
    return best_vec.tolist()
