import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Analytic sqrt‑series (fallback initializer)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Convolution (full) via FFT, returning max value and its index
# ------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ------------------------------------------------------------
# Gradient step targeting the current maximal autocorrelation entry
# ------------------------------------------------------------
def _gradient_step(a: np.ndarray, max_idx: int, step: float) -> np.ndarray:
    n = a.shape[0]
    lo = max(0, max_idx - (n - 1))
    hi = min(n - 1, max_idx)
    g = np.zeros_like(a)
    if lo <= hi:
        j = np.arange(lo, hi + 1)
        g[j] = 2.0 * a[max_idx - j]          # ∂(a*a)_{max_idx} / ∂ a_j
    cand = a - step * g
    cand = _project_simplex(cand)
    return cand


# ------------------------------------------------------------
# Small random redistribution move (preserves the L1 norm)
# ------------------------------------------------------------
def _random_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = a.shape[0]
    i = rng.integers(n)
    if a[i] <= 0.0:
        return a  # nothing to move
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    # move a fraction (up to 70 %) of a[i] to position j
    frac = rng.random() * 0.7
    delta = a[i] * frac
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


# ------------------------------------------------------------
# Local smoothing move – shift some mass from a point to its neighbours
# ------------------------------------------------------------
def _smooth_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = a.shape[0]
    i = rng.integers(1, n - 1)                # avoid the very ends
    if a[i] <= 0.0:
        return a
    frac = rng.random() * 0.5                # up to half the mass is spread
    delta = a[i] * frac
    b = a.copy()
    b[i] -= delta
    b[i - 1] += delta / 2.0
    b[i + 1] += delta / 2.0
    return b

    step = 0.4                     # gradient step size
    step_decay = 0.9995
    min_step = 1e-5

    temperature = cur_score * 0.02     # initial SA temperature
    temp_decay = 0.9995

    iter_no_improve = 0

                iter_no_improve += 1
        else:
            iter_no_improve += 1

        # decay parameters
        step = max(min_step, step * step_decay)
        temperature = max(1e-12, temperature * temp_decay)

        # occasional restart if stuck for a while
        if iter_no_improve >= max_no_improve_before_restart:
            # re‑initialise with a fresh random point on the simplex
            rand = rng.random(n)
            a = _project_simplex(rand)
            cur_max, cur_idx = _conv_max_and_index(a, L)
            cur_score = 2.0 * n * cur_max
            iter_no_improve = 0

        # early exit if we have already passed the target bound
        if best_score < 1.50274349:
            break

        # spare a little time for returning the answer
        if time.monotonic() + 0.4 > deadline:
            break

    # ---------------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # ---------------------------------------------------------------
    return [float(x) for x in best]
