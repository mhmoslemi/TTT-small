import time

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # ensure exact sum 1 (helps numerical stability)
    diff = 1.0 - w.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny remainder uniformly over non‑zero entries
        nz = w > 0
        if nz.any():
            w[nz] += diff / nz.sum()
        else:
            w[0] = 1.0
    return w

# Analytic sqrt‑series (coefficients a with a*a = 1+…+x^{2n‑2})
# Used only when no parent sequence is supplied.
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

                cand[dst] += delta
                cand = _project_simplex(cand)

                conv_c = _conv_fft(cand, L)
                max_c = float(conv_c.max())
                cand_score = _score(max_c)

                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    max_idx = int(conv_c.argmax())
                    if cand_score < best_score:
                        best_score = cand_score
                        best = cand.copy()

        # --------------------------------------------------------------
        # Step‑size decay (ensures convergence)
        # --------------------------------------------------------------
        step = max(min_step, step * decay)

        # --------------------------------------------------------------
        # Early exit guard to stay within the sandbox deadline
        # --------------------------------------------------------------
        if time.monotonic() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # Return the best candidate (plain Python list of floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
