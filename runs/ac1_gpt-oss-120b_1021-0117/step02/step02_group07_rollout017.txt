def propose_candidate(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.5   # safety

    # Determine n and starting sequence seq
    ...
    # Normalize sum to 1
    seq = seq / np.sum(seq)

    n = len(seq)
    fft_len = 1 << ((2*n-1).bit_length())

    def conv_of(x):
        f = np.fft.rfft(x, n=fft_len)
        return np.fft.irfft(f * f, n=fft_len)[:2*n-1]

    best_seq = seq.copy()
    best_conv = conv_of(best_seq)
    best_max = best_conv.max()
    best_score = 2.0 * n * best_max  # sum^2 = 1

    cur_seq = best_seq.copy()
    cur_max = best_max

    # parameters
    temp0 = 0.1
    total_iters = 0

    while time.monotonic() < deadline:
        # compute convolution and max
        conv = conv_of(cur_seq)
        kmax = int(np.argmax(conv))
        max_v = conv[kmax]

        # compute gradient
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi+1)
        jdx = kmax - idx
        grad = np.zeros_like(cur_seq)
        grad[idx] = cur_seq[jdx]
        diag = idx == jdx
        if np.any(diag):
            grad[idx[diag]] *= 2

        # direction that reduces max while keeping sum zero
        mean_grad = np.mean(grad)
        direction = -grad + mean_grad

        # compute feasible step size to keep nonnegative
        # For entries where direction <0, max step limited by cur_seq / -direction
        neg = direction < 0
        if np.any(neg):
            max_step = np.min(cur_seq[neg] / -direction[neg])
        else:
            max_step = np.inf
        # choose step size as fraction of max_step (e.g., 0.8) and as bounded
        step = min(0.5 * max_step, 0.05)  # cap at 0.05

        # sometimes random step
        if rng.random() < 0.02:  # 2% random large step
            step = rng.uniform(0.05, 0.2) * max_step

        cand = cur_seq + step * direction

        # enforce nonnegative small epsilon
        cand = np.maximum(cand, 0.0)

        # renormalize sum to 1
        s = cand.sum()
        if s == 0:
            continue
        cand = cand / s

        # evaluate candidate
        cand_conv = conv_of(cand)
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max

        # acceptance
        if cand_score < cur_max * 2.0 * n:  # but we compare score
            cur_seq = cand
            cur_max = cand_max
            # update best

                best_seq = cand.copy()
                best_score = cand_score
                best_max = cand_max

                cur_max = cand_max

import numpy as np
import time
import math

def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two length for a linear convolution of
    two length‑n sequences."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of seq with itself using a FFT of size fft_len."""
    f = np.fft.rfft(seq, n=fft_len)
    return np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]


def _score_seq(seq: np.ndarray, n: int, fft_len: int):
    """Return (score, max_convolution) for the given sequence."""
