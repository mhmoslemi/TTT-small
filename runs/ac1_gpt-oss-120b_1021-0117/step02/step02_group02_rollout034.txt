import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation (sub‑gradient with momentum + jitter) for the AC1 upper bound.
    Returns the best non‑negative step‑function heights found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # time budget (leave a few seconds safety margin)
    # ------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------
    # initialise from the provided parent or fall back to uniform
    # ------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
        a = np.clip(parent, 0.0, 1000.0)
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    best = a.copy()

    # cheap uniform alternative
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution via FFT -------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)

        # ----- current objective ----------------------------------------------------
