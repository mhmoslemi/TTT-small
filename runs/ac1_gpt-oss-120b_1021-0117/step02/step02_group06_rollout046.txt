import time
import numpy as np

def _fft_conv(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution of real vectors x and y using FFT of length fft_len."""
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    Z = X * Y
    conv = np.fft.irfft(Z, n=fft_len)
    return conv[: x.size + y.size - 1]

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256
