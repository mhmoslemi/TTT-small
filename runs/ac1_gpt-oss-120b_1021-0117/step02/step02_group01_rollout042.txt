import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution via FFT: return maximum value and its index
# ----------------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int = None):
    n = vec.size
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

    data = np.repeat(a, a.size)  # length n*n
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * a.size - 1, a.size))

            c = -np.ones(n)                # maximise sum(b)  -> minimise -sum(b)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            res = opt.linprog(c, A_ub=C, b_ub=b_ub,
                              bounds=[(0, None)] * n,
                              method='highs')
            if res.success:
                b = res.x
                s = b.sum()
                if s > 0:
                    b = b / s                  # enforce sum = 1
                    b = _project_simplex(b)   # numerical cleanup
                    b_score = _score(b)
                    if b_score < best_score:
                        best = b.copy()
                        best_score = b_score
                    if b_score < cur_score:
                        a = b
                        cur_score = b_score
                        # recompute max_idx for the next iteration
                        _, max_idx = _conv_max_and_index(a, L)

        # ----- step‑size schedule -----
        step = max(min_step, step * decay)
        it += 1

        # safety break close to the deadline
        if time.monotonic() + 0.1 > deadline:
            break

    # return a plain Python list – the evaluator expects a list of floats
    return [float(x) for x in best]
