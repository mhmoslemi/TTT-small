def simplex_projection(v):
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]

import time, numpy as np

def _proj_simplex(v):
    # Euclidean projection onto simplex (sum=1)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero or negative -> uniform
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    return w

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation of the AC1 upper bound.
    Starts from ``height_sequence_1`` (if present) or a uniform fallback,
    then repeatedly reduces the peak of the autocorrelation using a
    projected line‑search gradient step that aggregates the gradients of the
    largest few convolution peaks.
    Returns the best sequence encountered before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Initialise a (probability vector)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                           # enforce sum = 1

    # ------------------------------------------------------------------

    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349

    # FFT size for a*a (linear convolution)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    it = 0
    top_k = 5
    step_factors = [1.0, 0.5, 0.25, 0.125, 0.0625]

    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # current autocorrelation via FFT
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # --------------------------------------------------------------
        # identify a few largest peaks
        # --------------------------------------------------------------
        if top_k < conv_len:
            idxs = np.argpartition(-conv, top_k)[:top_k]

        # --------------------------------------------------------------
        # gradient = sum of gradients for the selected peaks
        # --------------------------------------------------------------
        g = np.zeros_like(a)
        for k in idxs:

            jitter = np.exp(rng.normal(0.0, 0.004, size=n))
            a = _proj_simplex(a * jitter)

        # line‑search: try decreasing step sizes until improvement
        # --------------------------------------------------------------
        improved = False
        for factor in step_factors:
