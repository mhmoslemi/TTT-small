def mass_transfer_from_max_pair(seq):
    conv = _conv_self(seq)
    kmax = np.argmax(conv)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    i = rng.integers(lo, hi+1)
    j = kmax - i
    if i == j:
        # choose another i
        i = lo if i != lo else hi
        j = kmax - i
    # amount to transfer from i and j
    frac = rng.uniform(0.05, 0.15)
    delta_i = seq[i] * frac
    delta_j = seq[j] * frac
    new_seq = seq.copy()
    new_seq[i] -= delta_i
    new_seq[j] -= delta_j
    # distribute total to low-value indices
    total = delta_i + delta_j
    # find L low indices:

    # after changes may be small negative due to rounding; clip & renorm
    np.clip(new_seq, 0.0, None, out=new_seq)

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n‑1 (positive, decreasing)."""
