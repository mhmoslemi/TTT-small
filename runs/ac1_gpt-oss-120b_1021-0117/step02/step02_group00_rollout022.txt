import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full convolution via FFT (fast, linear‑time memory)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Score for a vector with sum == 1  (same as evaluate_sequence)
# ------------------------------------------------------------
def _score_from_max(n: int, max_conv: float) -> float:
    return 2.0 * n * max_conv


# ------------------------------------------------------------
# Gradient built from the K largest convolution entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, K: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    if K >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(conv, -K)[-K:]
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    return g


# ------------------------------------------------------------
# Incremental update of the full convolution after moving delta mass
# from i to j (pre‑move a is supplied).  sign = +1 applies, -1 reverts.
# ------------------------------------------------------------
def _apply_move(b: np.ndarray, a: np.ndarray, i: int, j: int,
                delta: float, sign: int = 1) -> None:
    n = a.shape[0]
    # linear part
    b[i : i + n] -= sign * 2.0 * delta * a
    b[j : j + n] += sign * 2.0 * delta * a
    # quadratic part (indices are always within bounds)
    b[2 * i] += sign * delta * delta
    b[2 * j] += sign * delta * delta
    b[i + j] -= sign * 2.0 * delta * delta
