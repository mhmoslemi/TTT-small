import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- initialise sequence -------------------------------------------------
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if a.ndim != 1 or a.size == 0:
            raise ValueError
    except Exception:
        # fallback: classic binomial shape
        n = 256
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    n = a.size
    a = np.clip(a, 0.0, 1000.0)
    sum0 = a.sum()
    if sum0 < 1e-8:
        a[0] = 1.0
        sum0 = a.sum()

    # FFT length sufficient for all convolutions (linear, not circular)
    L = _next_pow2(3 * n - 2)

    def _conv_self(x):
        """Full convolution x*x using FFT (real‑valued)."""
        fx = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fx * fx, n=L)[: 2 * n - 1]
        return conv

    # ----- keep track of best found -------------------------------------------
    conv = _conv_self(a)
    M = float(conv.max())
    best_score = 2.0 * n * M / (sum0 * sum0)
    best = a.copy()

    # ----- optimisation parameters ---------------------------------------------
    step0 = 0.6              # initial gradient step size
    step_decay = 0.9997
    alpha = 0.5              # smooth‑max temperature
    alpha_inc = 0.0003
    max_alpha = 5.0
    iter_cnt = 0
    no_improve = 0

    # main loop
    while time.monotonic() < deadline:
        # ----- compute current convolution and smooth max ------------------------
        conv = _conv_self(a)
        M = float(conv.max())
        sum_a = float(a.sum())
        # smooth max parameters
        cur_alpha = min(alpha, max_alpha)
        shift = conv - M                     # <= 0
        exp_shift = np.exp(cur_alpha * shift)
        z = exp_shift.sum()
        if z == 0.0:                         # underflow protection
            cur_alpha = 0.1
            exp_shift = np.exp(cur_alpha * shift)
            z = exp_shift.sum()
        p = exp_shift / z
        h = M + math.log(z) / cur_alpha      # smooth max value

        # ----- gradient of smooth objective ------------------------------------
        # g_i = Σ_j a_j * p_{i+j}  (convolution of reversed a with p)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, n=L)
        fp = np.fft.rfft(p, n=L)
        conv_rev = np.fft.irfft(fa_rev * fp, n=L)
        start = n - 1
        g = conv_rev[start : start + n]

        factor = 4.0 * n / (sum0 * sum0)          # (sum_a == sum0 after rescaling)
        grad = factor * (g - (h / sum0) * np.ones_like(a))

        # ----- decide move type -------------------------------------------------
        move_type = rng.random()
        if move_type < 0.85:                     # gradient‑driven move
            step = step0 * (step_decay ** iter_cnt)
            cand = a - step * grad
        else:                                     # occasional random block scaling
            width = int(rng.integers(max(1, n // 250), max(2, n // 15)))
            start_idx = int(rng.integers(0, max(1, n - width + 1)))
            factor_rand = math.exp(rng.normal(0.0, 0.09))
            cand = a.copy()
            cand[start_idx:start_idx + width] *= factor_rand

        # ----- enforce constraints ------------------------------------------------
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s > 0:
            cand *= (sum0 / s)               # renormalise to original total mass
        else:
            cand = a.copy()                  # fallback if everything vanished

        # ----- evaluate candidate ------------------------------------------------
        conv_c = _conv_self(cand)
        M_c = float(conv_c.max())
        score_c = 2.0 * n * M_c / (sum0 * sum0)

        # ----- keep the best -----------------------------------------------------
