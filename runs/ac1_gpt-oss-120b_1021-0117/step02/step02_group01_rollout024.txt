def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # time limit
    deadline = time.monotonic() + budget_s - 5.0
    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 256
        init_seq = np.ones(n, dtype=np.float64) / n
    # Ensure nonnegative:
    init_seq = np.maximum(init_seq, 0.0)

    # We'll define projection onto simplex for sum=1
    def project_simplex(v):
        # standard algorithm from previous code
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho_arr = u - css / np.arange(1, v.size + 1)
        rho = np.where(rho_arr > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        w = v - theta
        w[w < 0] = 0.0
        return w

    # compute initial candidate
    candidate = project_simplex(init_seq)
    # maybe also try sqrt_series if better:
    # but generate sqrt_series for n length once
    def sqrt_series(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            if k == 1:
                s = 0.0
            else:
                s = np.dot(a[1:k], a[k-1:0:-1])
            a[k] = (1.0 - s) / 2.0
        return a

    # compute sqrt series and compare
    sqrt_seq = sqrt_series(n)
    sqrt_seq = project_simplex(sqrt_seq)

    # Evaluate both
    def compute_score(seq):
        # use evaluate_sequence internal for reliable but we can compute quickly as n known
        max_val, _ = conv_max(seq)
        # sum is 1 so score = 2*n*max_val
        return 2.0 * n * max_val

    # compute conv using FFT
    L = 1 << ((2 * n - 1).bit_length())
    def conv_max(seq):
        fv = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        idx = int(conv.argmax())
        return float(conv[idx]), idx

import time
import numpy as np

# -------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# -------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

# -------------------------------------------------------------
# Analytic sqrt-series: a*a = 1+…+x^{2n-2}
# -------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# -------------------------------------------------------------
# Convolution via FFT – return max value and full array
# -------------------------------------------------------------
def _conv_full_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Return the linear convolution x*x (full, length 2*len(x)-1) using FFT."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * len(x) - 1]
    return conv

# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # give ourselves a small safety margin before sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.5)

    # -----------------------------------------------------------------
    # Prepare initial sequence(s)
    # -----------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        base_seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base_seq = np.maximum(base_seq, 0.0)
    else:
        n = 256
        base_seq = np.ones(n, dtype=np.float64) / n

    # alternative analytic start
    sqrt_seq = _sqrt_series(n)
    sqrt_seq = np.maximum(sqrt_seq, 0.0)

    # put both on the simplex (sum = 1) – scaling does not affect the score
    cand1 = _proj_simplex(base_seq)
    cand2 = _proj_simplex(sqrt_seq)

    # pick the better of the two to start
    def _score_from_full_conv(full_conv: np.ndarray, s: float) -> float:
        return float(2.0 * n * full_conv.max() / (s * s))

    L = 1 << ((2 * n - 1).bit_length())
    conv1 = _conv_full_fft(cand1, L)
    conv2 = _conv_full_fft(cand2, L)
    score1 = _score_from_full_conv(conv1, cand1.sum())
    score2 = _score_from_full_conv(conv2, cand2.sum())

    a = cand1 if score1 <= score2 else cand2
    cur_score = min(score1, score2)

    best = a.copy()
    best_score = cur_score

    # -------------------------------------------------------------
    # optimisation parameters
    # -------------------------------------------------------------
    step = 0.25          # initial step size
    min_step = 1e-6
    decay = 0.998
    random_prob = 0.008
    plateau_cnt = 0
    max_plateau = 120   # after this many non‑improving steps we shrink more

    # -------------------------------------------------------------
    # main loop
    # -------------------------------------------------------------
    while time.monotonic() < deadline:

        full = _conv_full_fft(a, L)
        max_val = float(full.max())
        max_idxs = np.flatnonzero(full >= max_val - 1e-14)

        # sub‑gradient of the max‑convolution (sum of gradients for all active indices)
        grad = np.zeros_like(a)
        for k in max_idxs:

    # Return the best sequence found (as plain Python list)
    # -------------------------------------------------------------
    return [float(x) for x in best]
