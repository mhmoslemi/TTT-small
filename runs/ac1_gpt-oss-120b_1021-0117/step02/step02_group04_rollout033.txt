def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # problem size
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.shape[0]
    except Exception:
        parent = None
        n = 256

    L = 1 << ((2*n - 2).bit_length())  # next power of two >= 2n-1

    # fast score function
    def fast_score(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        max_b = float(conv.max())
        s = float(arr.sum())
        if s < 1e-12:
            return float('inf')
        return 2.0 * n * max_b / (s * s)

    # initial candidates
    if parent is not None:
        best = parent.astype(np.float64).copy()
        best_score = fast_score(best)
    else:
        best = np.full(n, 1.0, dtype=np.float64)
        best_score = fast_score(best)

    # also compare binomial
    binom = _gen_binomial(n)
    binom_seq = binom / binom[0] * 1000.0
    binom_score = fast_score(binom_seq)
    if binom_score < best_score:
        best = binom_seq.copy()
        best_score = binom_score

    # main random param search
    idx = np.arange(n, dtype=np.float64)
    # set bounds for lambda and beta
    lam_low, lam_high = 2e-4, 5e-3  # approx
    beta_low, beta_high = 0.0, 2.0

    while time.monotonic() < deadline:
        # sample a candidate shape
        lam = rng.uniform(lam_low, lam_high)
        beta = rng.uniform(beta_low, beta_high)
        w = np.exp(-lam * idx) * np.power(idx + 1.0, beta)
        # ensure all finite
        seq = w / w.max() * 1000.0
        # evaluate
        score = fast_score(seq)
        if score < best_score:
            best = seq.copy()
            best_score = score
            # optional early exit
            if best_score < 1.50274349:
                break
        # occasional gradient refinement from current best
        if rng.random() < 0.02:
            cur = best.copy()
            step_lr = 0.5
            # limited inner loop
            for _ in range(30):
                fa = np.fft.rfft(cur, n=L)
                conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
                max_b = float(conv.max())
                k_max = int(conv.argmax())
                sum_cur = float(cur.sum())
                t = max_b / sum_cur
                # compute mirror

                    best = cand.copy()
                    best_score = cand_score

import time
import math

import numpy as np

# --------------------------------------------------------------
# Small helper utilities (same as the reference solution)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (classic binomial step)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound via FFT (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

# Main candidate proposer
# --------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # give a tiny safety margin for the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Problem size & a first reasonable baseline

    try:

            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256                     # fallback size (should never happen)

    L = _next_pow2(2 * n - 1)

    # fast scoring wrapper
    def fast_score(v):
        return _ac1_score_fft(v, n, L)

    # start from the better of the supplied parent and the classic binomial
    base = _gen_binomial(n) * 1000.0    # binomial scaled to max≈1000
    if parent is not None:
        base = parent.astype(np.float64).copy()
    best = base.copy()
    best_score = fast_score(best)

    # The explicit binomial shape is a good fallback, keep it if better
    binom_seq = _gen_binomial(n) * 1000.0
    binom_score = fast_score(binom_seq)
    if binom_score < best_score:
        best = binom_seq.copy()
        best_score = binom_score

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)          # vector of indices
