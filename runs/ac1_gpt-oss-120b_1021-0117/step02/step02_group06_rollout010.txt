import time
import numpy as np

def _conv_full(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full (2n‑1) convolution of a with itself using real FFT."""
    A = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(A * A, n=fft_len)
    return c[: 2 * a.size - 1]

def _max_conv(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (max value, index) of the self‑convolution of a."""
    conv = _conv_full(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises the
    AC1 upper bound used by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ----- parent sequence (if any) -----
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ----- generate a few sensible seeds -----
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # parent (if present)
    if parent_arr is not None:

        s = p.sum()
        if s > 0.0:

    # simple sine‑shape (often flattening)

    sshape = sin_vals ** 0.5

    # pick the best of the seeds by the *exact* score (no evaluator call)
