import numpy as np
import time

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of seq with itself using FFT (real input)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    return conv


def _score_from_conv(seq_len: int, conv: np.ndarray, s: float) -> float:
    """Score = 2*len*max(conv)/s**2."""
    if s <= 0.0:
        return float("inf")
    return float(2.0 * seq_len * np.max(conv) / (s * s))


def _score_seq(seq: np.ndarray, fft_len: int) -> tuple[float, np.ndarray, float]:
    """Return (score, full convolution, max conv value)."""
    conv = _conv_fft(seq, fft_len)
    s = float(np.sum(seq))
    score = _score_from_conv(seq.size, conv, s)
    return score, conv, float(np.max(conv))


def _gradient_direction(seq: np.ndarray, conv: np.ndarray, k_max: int, M: float) -> np.ndarray:
    """
    Compute the descent direction d = λ·1 - 2·rev,
    where rev[i] = a_{k_max-i} (zero outside) and λ = 2*M/∑a.
    """
    n = seq.size
    s = float(np.sum(seq))
    if s == 0.0:
        return np.zeros_like(seq)
    lam = 2.0 * M / s

    rev = np.zeros_like(seq)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    rev[idx] = seq[k_max - idx]

    # direction = λ·1 - 2·rev
    return lam - 2.0 * rev


def _apply_random_op(seq: np.ndarray, rng: np.random.Generator,
                     conv: np.ndarray, k_max: int) -> np.ndarray:
    """Create a perturbed copy of ``seq`` using one of several cheap moves."""
    n = seq.size
    cand = seq.copy()
    op = rng.integers(0, 5)

    if op == 0:  # block scaling
        width = rng.integers(1, max(2, n // 30))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.06))
        cand[start:start + width] *= factor

    elif op == 1:  # single index scaling
        idx = rng.integers(0, n)
        factor = np.exp(rng.normal(0.0, 0.18))
        cand[idx] *= factor

    elif op == 2:  # early‑to‑late mass transfer
        early_end = max(1, n // 12)
        late_start = n - early_end
        i = rng.integers(0, early_end)
        j = rng.integers(late_start, n)
        frac = rng.uniform(0.07, 0.22)
        delta = cand[i] * frac
        cand[i] -= delta
        cand[j] += delta

    elif op == 3:  # smooth a neighbour pair
        i = rng.integers(0, n - 1)
        avg = (cand[i] + cand[i + 1]) * 0.5
        cand[i] = avg
        cand[i + 1] = avg

    else:  # redistribute a fraction from the pair that attains max conv
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        if lo <= hi:
            i = rng.integers(lo, hi + 1)
            j = k_max - i
            if i != j:
                frac = rng.uniform(0.05, 0.12)
                di = cand[i] * frac
                dj = cand[j] * frac
                cand[i] -= di
                cand[j] -= dj
                tgt = rng.integers(0, n)
                while tgt == i or tgt == j:
                    tgt = rng.integers(0, n)
                cand[tgt] += di + dj

    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ------------------------------------------------------------------
    # Determine problem size and possibly a parent construction
    # ------------------------------------------------------------------
    n = 3809  # default length used by the reference solution
    parent = None
    try:

    if parent is not None:

        if p_score < best_score:
            best = parent.copy()
            best_score = p_score
            best_conv = p_conv
            best_max = p_max

    # Working copy
    cur = best.copy()
    cur_score = best_score
    cur_conv = best_conv
    cur_max = best_max

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # recompute convolution for the current state (cheap)
        cur_conv = _conv_fft(cur, fft_len)
        k_max = int(np.argmax(cur_conv))
        cur_max = float(cur_conv[k_max])
        s_cur = float(np.sum(cur))
        cur_score = _score_from_conv(n, cur_conv, s_cur)

        # --------------------------- Gradient step -----------------------
        direction = _gradient_direction(cur, cur_conv, k_max, cur_max)

        # Determine a safe step size (scale so largest move ≈ 0.3·average)
        avg = s_cur / n
        max_abs = float(np.max(np.abs(direction))) + 1e-12
        base_step = 0.3 * avg / max_abs

        improved = False
        # backtrack line‑search (≤5 attempts)
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step = base_step * factor
            cand = cur + step * direction
            np.clip(cand, 0.0, 1000.0, out=cand)

            sum_cand = float(np.sum(cand))
            if sum_cand <= 0.0:
                continue

            cand *= (s_cur / sum_cand)

            if cand_score < cur_score:
                # accept the improvement

                cur_score = cand_score
                cur_conv = cand_conv
                cur_max = cand_max
                improved = True
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                break  # exit back‑track loop

        # --------------------------- Random fallback --------------------
        if not improved:
            cand = _apply_random_op(cur, rng, cur_conv, k_max)

            # renormalise mass after the random move
            sum_cand = float(np.sum(cand))
            if sum_cand > 0.0:
                cand *= (s_cur / sum_cand)

                cur_conv = cand_conv
                cur_max = cand_max

                    best = cand.copy()
                    best_score = cand_score

            break

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
