import time
import numpy as np

def _project_onto_simplex(v):
    # Euclidean projection onto simplex {x>=0, sum=1}
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _compute_score(a, n, fft_len):
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)

            a = a / total

import time
import numpy as np

# ---------- helpers ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Score = 2 * n * max(conv) / (sum(a)^2) using FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)
