import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# max of the (full) convolution a*a using FFT
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# gradient of conv_k = sum_i a_i a_{k-i} (sub‑gradient of the active lag)
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    return 2.0 * g


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------
    # initialise – use the provided parent if present, otherwise sqrt series
    # ----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 256
        a = _sqrt_series(n0).astype(np.float64)

    n = a.shape[0]
    a = a / a.sum()                     # normalise to sum = 1
    a = _project_simplex(a)             # safety projection

    # ----------------------------------------------------------------
    # bookkeeping
    # ----------------------------------------------------------------
    Lfft = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n‑1
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(a, Lfft)

    # ----------------------------------------------------------------
    # optimisation loop
    # ----------------------------------------------------------------
    iteration = 0
    eta0 = 1.0                     # initial step magnitude
    momentum = np.zeros_like(a)
    beta = 0.85                    # momentum decay
    patience = 0

    while time.monotonic() < deadline:
        iteration += 1

        # current convolution and its maximum
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M

        # gather all (almost) active lags – a tiny tolerance helps robustness
        eps = 1e-12 * M
        active = np.where(conv >= M - eps)[0]

        # aggregate gradient of all active lags
        grad = np.zeros_like(a)
        for k in active:
            grad += _grad_for_k(a, int(k))
        norm = np.linalg.norm(grad)
        if norm > 0.0:
            direction = grad / norm
        else:
            direction = np.zeros_like(a)

        # adaptive step size
        eta = eta0 / np.sqrt(iteration)

        # ---- additive update -------------------------------------------------
        cand_add = a - eta * direction
        cand_add = _project_simplex(cand_add)

        # ---- multiplicative (exponential) update -----------------------------
        # small exponential step; direction is already normalised
        cand_mul = a * np.exp(-eta * direction)
        cand_mul = _project_simplex(cand_mul)

        # ---- momentum‑augmented additive update every 10 iters ---------------
        if iteration % 10 == 0:

        else:

        # evaluate candidates

        for c in cand_list:
            scores.append(2.0 * n * _max_convolution(c, Lfft))

        # choose the best of the three
        best_idx = int(np.argmin(scores))
        cand = cand_list[best_idx]
        cand_score = scores[best_idx]

        # keep the overall best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            patience = 0
        else:
            patience += 1

        # accept the candidate if it improves, or with a tiny exploration chance

                    if j >= i:
                        j += 1
                    frac = rng.random() * 0.15
                    delta = a[i] * frac
                    a[i] -= delta
                    a[j] += delta
                    a = _project_simplex(a)

        # early exit if we already beat the target bound
        if best_score < 1.5028:
            break

        # safeguard: if no progress for a long while, restart from a random simplex point
        if patience > 2500:
            a = _project_simplex(rng.random(n))
            patience = 0

    # ----------------------------------------------------------------
    # return the best sequence found as a plain Python list
    # ----------------------------------------------------------------
    return [float(x) for x in best]
