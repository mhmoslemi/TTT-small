import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # degenerate case – return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimizer for the AC1 upper bound.
    Uses a momentum‑based projected sub‑gradient method with a
    multi‑shift sub‑gradient and occasional jitter.
    Returns the best coefficient list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. Base sequence(s)
    # ------------------------------------------------------------------
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0 or not np.isfinite(parent_raw).all():

    except Exception:
        parent_raw = None

    if parent_raw is not None:
        n = int(parent_raw.size)
        a_parent = np.maximum(parent_raw, 0.0)
        s = a_parent.sum()
        a_parent = a_parent / s if s >= 0.01 else np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 256
        a_parent = None

    # Uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # Triangular (tent) shape
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0          # edges are zero
    tent = np.maximum(tent, 0.0)
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:
        tent = uniform.copy()

    # Random Dirichlet‑like start
    rand_vec = rng.random(n)
    rand_vec = np.maximum(rand_vec, 0.0)
    rand_vec = rand_vec / rand_vec.sum()

    # Gather candidates
    candidates = [uniform, tent, rand_vec]
    if a_parent is not None:
        candidates.append(a_parent)

    # ------------------------------------------------------------------
    # 3. Helper to evaluate the AC1 bound (fast via FFT)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # Choose the best initial candidate
    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 1.0          # initial step magnitude
    mu = 0.9            # momentum decay
    K = 5               # number of top shifts to use for sub‑gradient
    target = 1.50274349
    v = np.zeros_like(a)   # momentum vector
    it = 0

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and current score -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= target:
                break

        # ----- sub‑gradient from top K shifts -----
        if K < conv_len:
            top_idx = np.argpartition(-conv, K)[:K]
        else:
            top_idx = np.arange(conv_len)
        top_vals = conv[top_idx]
        wsum = top_vals.sum()
        if wsum > 0:
            weights = top_vals / wsum
        else:
            weights = np.full_like(top_vals, 1.0 / top_vals.size)

        g = np.zeros_like(a)
        for shift, w in zip(top_idx, weights):
            offset = shift - (n - 1)
            if offset >= 0:
                idx_range = np.arange(offset, n, dtype=np.int64)
                partner = shift - idx_range
            else:
                idx_range = np.arange(0, n + offset, dtype=np.int64)
                partner = shift - idx_range
            g[idx_range] += w * 2.0 * a[partner]

        # ----- momentum update -----
        v = mu * v + (1.0 - mu) * g

        # ----- step size (diminishing) -----
        eta = eta0 / np.sqrt(it)

        # ----- apply step and project -----
        a = _proj_simplex(a - eta * v)

        # ----- occasional jitter to escape plateaus -----
        if it % 200 == 0:
            jitter = np.exp(rng.normal(0.0, 0.01, size=n))
            a = _proj_simplex(a * jitter)

    # ------------------------------------------------------------------
    # 6. Return the best sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
