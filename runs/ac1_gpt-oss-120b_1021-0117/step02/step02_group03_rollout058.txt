    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)  # preserve headroom

    # initialize function to get starting vector
    if 'height_sequence_1' in globals():

    n = a0.shape[0]

    a0 = a0 / a0.sum()
    a0 = _project_simplex(a0)

    best = a0.copy()
    best_M = _max_convolution(best, L_fft)  # compute later after L_fft known

    L_fft = 1 << ((2 * n - 1).bit_length())
    # compute initial best_M
    best_M = _max_convolution(best, L_fft)
    best_score = 2.0 * n * best_M

    # start with a copy for current
    a = best.copy()
    cur_M = best_M
    cur_score = best_score

    iteration = 0
    eta0 = 0.7  # step magnitude
    T0 = 0.01 # temperature scale
    # main loop
    while time.monotonic() < deadline:
        iteration += 1

        # compute conv and max
        # use stored cur_M? But after updates a changed, recompute each loop
        fv = np.fft.rfft(a, n=L_fft)
        conv = np.fft.irfft(fv * fv, n=L_fft)[:2*n-1]
        M = float(conv.max())
        k = int(conv.argmax())
        cur_score = 2.0 * n * M

        # gradient for the active indices (maybe consider near-peak)
        # compute list of indices where conv >= M - tol
        tol = 1e-12
        # we may include just active; for performance maybe just one.
        grad = _grad_for_k(a, k)

        # optionally include other near-peak indices
        # But for speed skip.

        # compute step
        norm = np.linalg.norm(grad)
        if norm > 0:
            eta = eta0 / math.sqrt(iteration)
            direction = -grad / norm
            cand = a + eta * direction
        else:
            cand = a.copy()

        cand = _project_simplex(cand)

        # evaluate candidate conv max
        fv_c = np.fft.rfft(cand, n=L_fft)
        conv_c = np.fft.irfft(fv_c * fv_c, n=L_fft)[:2*n-1]
        M_c = float(conv_c.max())
        cand_score = 2.0 * n * M_c

        # record best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            best_M = M_c

        # acceptance
        if cand_score < cur_score:
            a = cand
            cur_M = M_c
        else:
            # probabilistic acceptance with decreasing temperature
            T = T0 / math.sqrt(iteration)
            if rng.random() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
                a = cand
                cur_M = M_c

        # occasional random mass transfer
        if rng.random() < 0.02:
            # random source
            idx_src = rng.integers(0, n)
            if a[idx_src] > 1e-12:
                idx_dst = rng.integers(0, n-1)
                if idx_dst >= idx_src:
                    idx_dst += 1
                frac = rng.random() * 0.2
                delta = a[idx_src] * frac
                a[idx_src] -= delta
                a[idx_dst] += delta
                a = _project_simplex(a)

        # occasional restart from best (or random)
        if best_score < 1.5028:
            break
        if iteration % 2000 == 0 and rng.random() < 0.1:
            # reinitialize to best or random
            if rng.random() < 0.5:
                a = best.copy()
            else:
                # random Dirichlet
                a = rng.random(n)
                a = a / a.sum()
                a = _project_simplex(a)

    return [float(x) for x in best]

def sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        # compute s = sum_{i=1}^{k-1} a[i] * a[k-i]
        # Equivalent to dot product of a[1:k] and a[k-1:0:-1]
        if k == 1:
            s = 0.0
        else:
            # Use np.dot for speed
            s = np.dot(a[1:k], a[k-1:0:-1])
        a[k] = (1.0 - s) / 2.0
    return a

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# maximum of full convolution a*a using FFT (matrix‑free)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())
