# ------------------------------------------------------------
# Simplex projection (Euclidean) for non‑negative vectors with sum 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    L = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:n]
    return conv

        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                                     # enforce sum = 1

    n = a.size

    # pre‑compute FFT length for the gradient convolution (w * a_rev)
    Lg = 1 << ((3 * n - 2).bit_length())

    # best solution seen so far
    conv = _conv_self_fft(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    # ADAM state
    m = np.zeros_like(a)
    v_adam = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # soft‑max temperature schedule (beta = weight factor)
    beta_start = 1.0 / 5.0            # start fairly smooth
    beta_end = 200.0
    beta_inc = 0.001                  # increase per iteration

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ------------------------------------------------------------
        # evaluate current score
        # ------------------------------------------------------------
        conv = _conv_self_fft(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_score < 1.50274349:
            break

        # ------------------------------------------------------------
        # soft‑max weights over the full convolution
        # ------------------------------------------------------------
        beta = min(beta_end, beta_start + beta_inc * it)
        # stable soft‑max
        c = max_val
        w = np.exp(beta * (conv - c))
        w_sum = w.sum()
        if w_sum == 0.0:                     # fallback to pure max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ------------------------------------------------------------
        # gradient of the smoothed objective: g = 2 * (w * a_rev)[n-1 : n-1+n]
        # ------------------------------------------------------------
        a_rev = a[::-1]
        fx = np.fft.rfft(w, n=Lg)
        fy = np.fft.rfft(a_rev, n=Lg)
        conv_wa = np.fft.irfft(fx * fy, n=Lg)[: len(w) + len(a_rev) - 1]
        start_idx = n - 1
        g = 2.0 * conv_wa[start_idx:start_idx + n]

        # ------------------------------------------------------------
        # ADAM update on the simplex (Euclidean projection afterwards)
        # ------------------------------------------------------------
        lr = 0.05 / (np.sqrt(it) + 1e-12)
        m = beta1 * m + (1.0 - beta1) * g
        v_adam = beta2 * v_adam + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a_new)

        # ------------------------------------------------------------
        # occasional random mass‑transfer move to escape plateaus
        # ------------------------------------------------------------
        if no_improve > 2000 and rng.random() < 0.2:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.3
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = _project_simplex(a_tmp)
                # quick evaluation
                cur = 2.0 * n * _conv_self_fft(a_tmp).max()
                if cur < best_score:
                    best_score = cur
                    best_a = a_tmp.copy()
                    no_improve = 0
                a = a_tmp

        # ------------------------------------------------------------
        # periodic jitter toward the best solution
        # ------------------------------------------------------------
        if it % 3000 == 0:
            jitter = rng.normal(scale=5e-4, size=n)
            a_tmp = best_a + jitter
            a_tmp = np.maximum(a_tmp, 0.0)
            a = _project_simplex(a_tmp)

    # Return the best vector found as a plain Python list
    return [float(x) for x in best_a]
