import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # a little safety margin

    # --------------------------------------------------------------
    # Problem size and optional parent sequence
    # --------------------------------------------------------------
    n = 3809                     # default length
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # --------------------------------------------------------------
    # Helper to evaluate a sequence (FFT based, linear memory)
    # --------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    def _conv(seq):
        f = np.fft.rfft(seq, n=fft_len)
        return np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]

    def _score(seq, conv=None):
        if conv is None:
            conv = _conv(seq)
        max_b = conv.max()
        s = seq.sum()
        if s <= 0.0:
            return np.inf, conv, max_b
        return 2.0 * n * max_b / (s * s), conv, max_b

    # --------------------------------------------------------------
    # Initial candidates: parent (if any) and the classic binomial seed
    # --------------------------------------------------------------
    def _binomial_init(N):
        a = np.empty(N, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, N):
            a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
        return a

    best = _binomial_init(n)
    best_score, best_conv, _ = _score(best)

    if parent is not None:
        p_score, p_conv, _ = _score(parent)
        if p_score < best_score:
            best = parent.copy()
            best_score = p_score
            best_conv = p_conv

    # --------------------------------------------------------------
    # Working copy (current state) – start from the best found so far
    # --------------------------------------------------------------
    current = best.copy()
    cur_score = best_score
    target_sum = current.sum()                     # used only for re‑normalisation

    # --------------------------------------------------------------
    # Main stochastic optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a modification type
        op = rng.integers(0, 4)

        if op == 0:                 # gradient‑descent step on the maximal lag
            # current convolution & lag where it peaks
            cur_conv = _conv(current)
            kmax = int(cur_conv.argmax())

            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = kmax - idx

            grad = np.zeros_like(current)
            grad[idx] = current[jdx]
            # diagonal entries appear twice
            diag = idx == jdx
            if diag.any():
                grad[idx[diag]] *= 2.0

            # direction that reduces the maximal entry while keeping the sum
            mean_grad = grad.mean()
            direction = -grad + mean_grad

            # maximal safe step that keeps all components non‑negative
            neg = direction < 0
            if np.any(neg):
                max_step = np.min(current[neg] / (-direction[neg]))
                max_step *= 0.9                     # a safety factor
            else:
                max_step = np.inf

            if not np.isfinite(max_step) or max_step <= 0.0:
                step = 0.0
            else:
                # step size shrinks linearly with the remaining time
                remain = deadline - time.monotonic()
                frac = remain / (deadline - start)
                step = max_step * rng.uniform(0.15, 0.45) * frac

            cand = current + step * direction
            np.clip(cand, 0.0, 1000.0, out=cand)
            # renormalise to keep the overall scale (invariant for the score)
            s = cand.sum()
            if s > 0.0:
                cand *= (target_sum / s)

        elif op == 1:               # small multiplicative tweak of a random block
            width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.04))   # modest log‑normal factor
            cand = current.copy()
            cand[start_idx:start_idx + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        elif op == 2:               # swap two random positions (exploration)
            cand = current.copy()
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        else:                       # tiny single‑index perturbation
            cand = current.copy()
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.12))
            cand[idx] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        # --------------------------------------------------------------
        # Evaluate the candidate
        # --------------------------------------------------------------
        cand_score, cand_conv, _ = _score(cand)

        # --------------------------------------------------------------
        # Acceptance rule (simulated annealing style)
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start
        total = deadline - start
        temperature = max(0.0, 0.12 * (1.0 - elapsed / total))

        accepted = False
        if cand_score < cur_score:
            accepted = True
        elif temperature > 0.0:
            prob = np.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accepted = True

        if accepted:
            current = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                best_conv = cand_conv

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
