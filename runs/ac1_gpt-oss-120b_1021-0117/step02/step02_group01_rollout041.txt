import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_bool = rho_arr > 0
    if not np.any(rho_bool):
        theta = 0.0
    else:
        rho = np.nonzero(rho_bool)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Maximum of the (real) convolution a*a using FFT (matrix‑free)
# ----------------------------------------------------------------------
def _conv_max_and_index(a: np.ndarray, L: int):
    """Return (max value, index) of the full convolution a*a."""
    # real FFT, pointwise square, inverse FFT
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# ----------------------------------------------------------------------

    rng = np.random.default_rng(seed)

    # leave a small safety margin for the sandbox to kill us
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------

    if 'height_sequence_1' in globals():

            a0[0] = 1.0
            for k in range(1, m):
                if k == 1:
                    s = 0.0
                else:
                    s = float(np.dot(a0[1:k], a0[k - 1 : 0 : -1]))

            return a0
        a = _sqrt_series(256)

    # project onto simplex so that sum(a) = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power of two ≥ 2·n‑1

    # initial score (sum=1 --> score = 2·n·max)
    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    step = 0.5               # initial step size for gradient descent
    min_step = 1e-6
    decay = 0.998

    # ------------------------------------------------------------------

    while time.monotonic() < deadline:

        lo = max(0, max_idx - (n - 1))
