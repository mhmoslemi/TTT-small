import time
import numpy as np

# ---------- utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    for p in (0.5, 1.0, 2.0):
        v = sinv ** p
        cand.append(v / v.sum())

    base = (idx + 1) * (n - idx)

    # ---- obtain parent length (if any) ----
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256

    # ---- initialise candidates ----
    best_vec = None
    best_score = np.inf

    def _score(vec: np.ndarray) -> float:
        """Exact score as used by the evaluator (sum already >= 0)."""
        conv = np.convolve(vec, vec)
        max_b = conv.max()
        total = vec.sum()
        if total < 0.01:
            return np.inf
        return 2.0 * n * max_b / (total * total)

    for cand in _initial_candidates(n, parent_arr):
        sc = _score(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
        if best_score <= 1.50274349:      # early success
            return best_vec.tolist()

    # ---- optimisation loop (soft‑max gradient descent) ----
    a = best_vec.copy()
    it = 0
    temp = 0.2          # temperature for soft‑max (larger = smoother)
    alpha = 0.5         # multiplicative step magnitude

    while time.monotonic() < deadline:
        # evaluate current state
        conv = np.convolve(a, a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b / (a.sum() ** 2)
        if cur_score < best_score:
            best_score, best_vec = cur_score, a.copy()
            if best_score <= 1.50274349:
                break

        # ----- soft‑max weights over convolution entries -----
        shift = conv - conv.max()               # stabilise exp
        w = np.exp(shift / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient via correlation (g = 2 * (a ⋆ w)) -----
        corr = np.correlate(a, w, mode='full')
        start = len(w) - 1                     # lag 0 position
        g = 2.0 * corr[start:start + n]

        # ----- multiplicative update and renormalisation -----
        g = np.clip(g, -50.0, 50.0)             # avoid overflow in exp
        a = a * np.exp(-alpha * g)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # ----- schedule adjustments -----
        it += 1
        if it % 200 == 0:
            alpha = max(alpha * 0.85, 1e-4)
            temp = max(temp * 0.85, 1e-4)

        # ----- occasional random jitter to avoid symmetry traps -----
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.025, size=n))
            a = a * jitter
            a /= a.sum()

        # ----- simple restart if progress stalls -----
        if it % 1500 == 0:
            # keep best so far, but perturb it a little
            pert = np.exp(rng.normal(0.0, 0.03, size=n))
            a = best_vec * pert
            a = a / a.sum()

        # safety: leave a little headroom before hard timeout
        if time.monotonic() + 0.1 > deadline:
            break

    # return the best vector discovered
    return best_vec.tolist()
