    import ...
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.5)
    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():

    else:

        ...

    # Ensure nonnegative and sum 1
    a = np.maximum(a, 0.0)
    if a.sum() <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()

    n = a.size

    best = a.copy()
    best_sc = _score(best)

    # gradient descent parameters
    step0 = 1.0  # initial step size
    beta = 0.9
    m = np.zeros_like(a)

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        conv = _conv_fft(a)
        M = conv.max()
        # compute gradient at active max indices
        active = np.where(conv >= M - 1e-12)[0]

            no_improve = 0

            no_improve += 1

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# FFT‑based full linear convolution
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a (length 2*len(a)-1) via FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x * y via FFT."""
    m = x.shape[0] + y.shape[0] - 1
    L = 1 << (m.bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:m]
    return conv


# ------------------------------------------------------------
# Score (same as evaluate_sequence for sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max()


# ------------------------------------------------------------
# Gradient of the max‑convolution (sub‑gradient)
# ------------------------------------------------------------
def _maxconv_grad(a: np.ndarray) -> tuple:
    """
    Returns (grad, max_conv, full_conv) where grad is the sub‑gradient
    of f(a)=max_k (a*a)_k, max_conv is the maximum value, and full_conv
    is the full convolution array.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    # indices where the maximum is attained (within tiny tolerance)
    active = np.where(conv >= max_val - 1e-14)[0]
    # indicator vector of active positions
    ind = np.zeros_like(conv)
    ind[active] = 1.0
    # gradient = 2 * (ind ⋆ rev(a))[n-1 : n-1+n]
    rev_a = a[::-1]
    tmp = _conv_full(ind, rev_a)
    grad = 2.0 * tmp[n - 1 : n - 1 + n]
    return grad, max_val, conv

    """

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # keep a safety margin

    else:
