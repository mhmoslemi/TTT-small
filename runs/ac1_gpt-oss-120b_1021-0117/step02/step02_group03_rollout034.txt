def _max_convolution_fft(a, L):
    # using real FFT
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*len(a)-1]
    return conv.max()

    while time.monotonic() < deadline:

        k = int(rng.choice(active_indices))

        grad = _grad_for_k(a, k)
        # normalize gradient
        norm = np.linalg.norm(grad)
        if norm==0:
            # random perturbation fallback
            # call random mass transfer below
            pass
        else:
            # try step sizes
            base_eta = eta0 / np.sqrt(iter_cnt)
            # candidate best for this iteration
            cand_best = a
            cand_best_score = best_score  # maybe use current? Actually current score 2*n*M
            cand_best_score = 2.0*n*M
            # try a series of multipliers
            for mult in (1.0, 0.5, 0.25, 0.125, 0.0625):
                eta = base_eta * mult
                cand = a - eta * (grad / norm)  # descent direction
                cand = _project_simplex(cand)
                cand_M = _max_convolution_fft(cand, L)
                cand_score = 2.0 * n * cand_M
                if cand_score < cand_best_score:
                    cand_best_score = cand_score
                    cand_best = cand

            # also keep current unchanged if no improvement
            if cand_best_score < 2.0 * n * M:
                a = cand_best
                M = cand_best_score / (2.0*n)
                # update best global
                if cand_best_score < best_score:
                    best_score = cand_best_score
                    best = cand_best.copy()
                no_improve = 0
            else:

                no_improve += 1

        if rng.random() < 0.02:
            # pick i with positive mass
            i = rng.integers(0, n)

                best = cand.copy()

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Initialize

    # Project onto simplex (sum=1)

    # parameters

    iteration = 0
    no_improve = 0

    while time.monotonic() < deadline:

        cand_best = a
        cand_best_score = best_score  # fallback to previous best (not current)

            # try several scales

                eta = base_eta * mult
                cand = a - eta * (grad / norm)
                cand = _project_simplex(cand)

                cand_score = 2.0 * n * cand_M
                if cand_score < cand_best_score:

                    cand_best = cand
        # accept if better than current state
        cur_score = 2.0 * n * M
        if cand_best_score < cur_score - 1e-12:
            a = cand_best
            M = cand_best_score / (2.0 * n)
            no_improve = 0
        else:

            if rng.random() < 0.001:
                a = cand_best
                M = cand_best_score / (2.0 * n)
                no_improve = 0
            else:
                no_improve += 1

import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Gradient of the full convolution a*a at a given lag k
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of conv_k = sum_i a_i a_{k-i} (full convolution)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    return 2.0 * g


# ------------------------------------------------------------
# Maximum of the (full) convolution a*a using a real FFT
# ------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray, L: int) -> float:
    """Return max_k (a * a)[k] using an FFT of length L (power of two)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.shape[0] - 1]
    return float(conv.max())

# Main entry point
# ------------------------------------------------------------

    # initialise – use the provided parent if present,

    a = a / a.sum()
    a = _project_simplex(a)                 # safety projection onto simplex
