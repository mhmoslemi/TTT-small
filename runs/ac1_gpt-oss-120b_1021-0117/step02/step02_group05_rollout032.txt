def gradient_step(cur, step):
    # compute conv, max index
    conv = compute conv
    kmax = np.argmax(conv)
    # compute grad
    n = len(cur)
    grad = np.zeros_like(cur)
    idx = np.arange(n)
    rev = kmax - idx
    mask = (rev >= 0) & (rev < n)
    grad[mask] = cur[idx[mask]] + cur[rev[mask]]
    # keep grad values where mask false =0
    # gradient direction: d = -grad + grad.mean()
    # (makes sum zero)
    d = -grad + np.mean(grad)
    cand = cur + step * d
    np.clip(cand, 0.0, 1000.0, out=cand)
    # renorm to original sum to keep invariant
    s_cur = cur.sum()
    s_cand = cand.sum()
    if s_cand > 0:
        cand *= s_cur / s_cand
    else:
        # fallback keep cur
        cand = cur.copy()
    return cand

def _score_arr(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_c = conv.max()
    s = arr.sum()
    if s <= 1e-12:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

import numpy as np, time, math

def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _ac1_score(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    maxc = conv.max()
    s = arr.sum()
    if s <= 0:
        return float('inf')
    return 2.0 * n * maxc / (s * s)

def _gradient_move(arr, step, n, L):
    # compute convolution via FFT
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    kmax = int(np.argmax(conv))
    idx = np.arange(n)
    rev = kmax - idx
    mask = (rev >= 0) & (rev < n)
    grad = np.zeros_like(arr)
    # derivative: a_i + a_{kmax-i}
    grad[mask] = arr[mask] + arr[rev[mask]]
    # direction that reduces high values
    d = grad.mean() - grad
    cand = arr + step * d
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _block_scaling(arr, min_block, max_block, rng):
    n = arr.size
    width = int(rng.integers(min_block, max_block+1))
    start = int(rng.integers(0, n - width + 1))
    factor = math.exp(rng.normal(0.0, 0.07))
    cand = arr.copy()
    cand[start:start+width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _smooth_move(arr):
    # simple three‑point smoothing preserving sum
    sm = np.convolve(arr, np.ones(3)/3.0, mode='same')
    # re‑normalize to original sum
    s = arr.sum()
    sm *= s / (sm.sum() + 1e-16)
    np.clip(sm, 0.0, 1000.0, out=sm)
    return sm

def _mass_transfer(arr, max_frac, rng):
    n = arr.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return arr.copy()
    frac = rng.random() * max_frac
    delta = arr[i] * frac
    cand = arr.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- obtain length -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        init_candidates = [parent.copy()]
    except Exception:
        n = 256
        init_candidates = []
    # ---------------------------------------------------------------
    # fallback generator: binomial (Catalan‑type)
    def _gen_binomial(m):
        a = np.empty(m, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, m):
            a[i] = a[i-1] * (2*i-1) / (2*i)
        return a

    init_candidates.append(_gen_binomial(n))

    # ----- explore simple parametric families -------------------------
    idx = np.arange(n, dtype=np.float64) + 1.0

    # power‑law a_i = (i)^{-p}
    for p in np.linspace(0.0, 2.0, 31):
        a = idx ** (-p)
        init_candidates.append(a)

    # exponential a_i = exp(-lambda*i)
    for lam in np.linspace(0.0, 5.0, 31):
        a = np.exp(-lam * (idx-1))
        init_candidates.append(a)

    # pick best among these cheap candidates
    L = _next_pow2(2 * n - 1)
    best = None
    best_score = float('inf')
    for cand in init_candidates:
        # enforce evaluator bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _ac1_score(cand, n, L)
        if sc < best_score:
            best_score = sc
            best = cand.copy()

    cur = best.copy()
    cur_score = best_score

    # ----- main stochastic optimisation ---------------------------------
    temperature = 1.0
    decay = 0.9995
    step = 0.2                # initial gradient step magnitude
    step_decay = 0.9997

    min_block = max(2, n // 200)
    max_block = max(2, n // 12)
    transfer_max = 0.4

    prob_grad = 0.55
    prob_block = 0.20
    prob_smooth = 0.15
    prob_transfer = 0.10

    while time.monotonic() < deadline:
        r = rng.random()
        if r < prob_grad:
            cand = _gradient_move(cur, step, n, L)
        elif r < prob_grad + prob_block:
            cand = _block_scaling(cur, min_block, max_block, rng)
        elif r < prob_grad + prob_block + prob_smooth:
            cand = _smooth_move(cur)
        else:
            cand = _mass_transfer(cur, transfer_max, rng)

        cand_score = _ac1_score(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # early exit if already very good
            if best_score < 1.50274349:
                break

        # simulated‑annealing style acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # schedules
        temperature *= decay
        step *= step_decay
        # occasional restart from the best seen
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0
            step = 0.2

    # make sure we return a plain Python list
    return [float(x) for x in best]

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound (same formula used by the evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_c = float(conv.max())
    s = float(arr.sum())
    if s <= 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _gen_binomial(m: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(m, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, m):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _gradient_move(arr: np.ndarray, step: float, n: int, L: int) -> np.ndarray:
    """Move mass away from indices that contribute most to the current peak."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    kmax = int(np.argmax(conv))

    idx = np.arange(n)
    rev = kmax - idx
    mask = (rev >= 0) & (rev < n)

    grad = np.zeros_like(arr)
    # derivative of conv[kmax] w.r.t. a_i is a_i + a_{kmax-i}
    grad[mask] = arr[mask] + arr[rev[mask]]

    d = grad.mean() - grad                     # zero‑sum direction
    cand = arr + step * d
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _block_scaling(arr: np.ndarray, min_block: int, max_block: int, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block."""
    n = arr.size
    width = int(rng.integers(min_block, max_block + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = math.exp(rng.normal(0.0, 0.07))
    cand = arr.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _smooth_move(arr: np.ndarray) -> np.ndarray:
    """Apply a cheap three‑point moving average while preserving the total sum."""
    sm = np.convolve(arr, np.ones(3) / 3.0, mode="same")
    total = arr.sum()
    sm_sum = sm.sum()
    if sm_sum > 0:
        sm *= total / sm_sum
    np.clip(sm, 0.0, 1000.0, out=sm)
    return sm


def _mass_transfer(arr: np.ndarray, max_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of mass from one index to another."""
    n = arr.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return arr.copy()
    frac = rng.random() * max_frac
    delta = arr[i] * frac
    cand = arr.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
