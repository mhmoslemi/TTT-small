def _conv_and_max(a, L):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*len(a)-1]
    max_val = float(conv.max())
    max_idx = int(conv.argmax())
    return conv, max_val, max_idx

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)
    # initialize
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # generate sqrt series of length 1024 maybe
        n_def = 1024
        a = _sqrt_series(n_def)
    # Project onto simplex (sum=1)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    # initial evaluation
    conv, max_val, max_idx = _conv_and_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score
    # parameters
    step = 0.3
    min_step = 1e-6
    step_decay = 0.999
    momentum = np.zeros_like(a)
    beta = 0.9
    top_k = 5
    random_move_prob = 0.01
    # loop
    while time.monotonic() < deadline:
        # compute gradient based on top_k conv entries
        # find top indices
        if top_k < len(conv):
            top_idxs = np.argpartition(-conv, top_k-1)[:top_k]
        else:
            top_idxs = np.arange(len(conv))
        g = np.zeros_like(a)
        for k in top_idxs:
            lo = k - (n - 1)
            if lo < 0: lo = 0
            hi = k if k < n else n - 1
            if lo <= hi:
                idxs = np.arange(lo, hi+1)
                g[idxs] += 2.0 * a[k - idxs]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        # try step size with backtracking
        accepted = False
        cur_step = step
        for _ in range(5):
            cand = a - cur_step * momentum
            cand = _project_simplex(cand)
            conv_c, max_val_c, _ = _conv_and_max(cand, L)
            cand_score = 2.0 * n * max_val_c
            if cand_score < cur_score:
                # accept
                a = cand
                conv = conv_c
                max_val = max_val_c
                cur_score = cand_score
                accepted = True
                if cand_score < best_score:
                    best_score = cand_score
                    best = cand.copy()
                break
            else:
                cur_step *= 0.5
        # if not accepted, keep a unchanged but maybe still update momentum
        # occasional random perturbation
        if rng.random() < random_move_prob:
            i = rng.integers(0, n)

                cand2 = _project_simplex(cand2)

import time
import numpy as np

# --------------------------------------------------------------
# Simplex projection (L1=1, non‑negative)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# --------------------------------------------------------------
# Analytic sqrt‑series (used only if no parent is supplied)
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

# --------------------------------------------------------------
# Convolution (full) via FFT and extraction of maximum entry
# --------------------------------------------------------------
def _conv_and_max(vec: np.ndarray, L: int):
    """Return full convolution, its maximum value and the index of that max."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    max_val = float(conv.max())
    max_idx = int(conv.argmax())
    return conv, max_val, max_idx
