    rng = np.random.default_rng(seed)
    try get n from height_sequence_1 else n=256
    L = next_pow2(2*n-1)
    # helper compute score
    def compute_score(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
        maxb = conv.max()
        s = arr.sum()
        if s < 1e-12: return np.inf
        return 2.0 * n * maxb / (s*s)
    # init best
    if parent exists:
        base = parent.copy().astype(np.float64)
    else:

    best = base.copy()

    uniform = np.full(n, 1.0/n) # sum=1
    uniform_score = compute_score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

import math
import time

import numpy as np

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Score used by the official evaluator (FFT based)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta(a,b) density on n points, normalized to sum 1."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    y = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    s = y.sum()
    return y / s if s > 0 else np.full(n, 1.0 / n)


def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    """1/(i+1)^p on n points, normalized to sum 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    y = 1.0 / (i ** p)
    s = y.sum()
    return y / s if s > 0 else np.full(n, 1.0 / n)


def _exp_shape(n: int, c: float) -> np.ndarray:
    """exp(-c*i) on n points, normalized to sum 1."""
    i = np.arange(n, dtype=np.float64)
    y = np.exp(-c * i)
    s = y.sum()
    return y / s if s > 0 else np.full(n, 1.0 / n)


def _hann_shape(n: int, q: float) -> np.ndarray:
    """(sin^2) window raised to power q, normalized to sum 1."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    y = (0.5 - 0.5 * np.cos(2.0 * theta)) ** q
    s = y.sum()
    return y / s if s > 0 else np.full(n, 1.0 / n)


# ---------- main optimizer ----------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    # ----- problem size & initial sequence -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        cur = parent.astype(np.float64)
    except Exception:
        # fallback to a simple binomial‑like sequence
        n = 256
        cur = np.empty(n, dtype=np.float64)
        cur[0] = 1.0
        for i in range(1, n):
            cur[i] = cur[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    # Normalise to unit L1‑norm (scale‑invariant for the score)
    cur_sum = cur.sum()
    if cur_sum <= 0:
        cur = np.full(n, 1.0 / n)
    else:
        cur = cur / cur_sum
    best = cur.copy()
    L = _next_pow2(2 * n - 1)
    best_score = _ac1_score_fft(best, n, L)

    # ----- annealing parameters -----
    temperature = 5.0
    temp_decay = 0.9997
    momentum = np.zeros_like(cur)          # optional momentum for smoother moves
    momentum_factor = 0.85
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----- main search loop -----
    while time.monotonic() < deadline:

        cand = cur.copy()

                kmax = int(conv.argmax())

                    break

                j = kmax - idx

                mirror[mask] = refined[j[mask]]
                t = max_b / s

                refined = np.clip(refined, 0.0, 1000.0)

                    best_score = sc

            cur = best.copy()
            temperature = 5.0

        # ----- early exit if target achieved -----
        if best_score < 1.50274349:
            break

    # ----- return best sequence as plain Python floats -----
    return [float(x) for x in best]
