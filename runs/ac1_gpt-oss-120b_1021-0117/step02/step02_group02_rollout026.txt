import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Projected exponentiated‑gradient descent with a smooth‑max surrogate.
    Starts from the supplied parent ``height_sequence_1`` or a uniform fallback,
    repeatedly reduces the maximal autocorrelation peak, and returns the best
    sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2.  Load parent if available, otherwise fall back to uniform
    # ------------------------------------------------------------------
    parent = None
    try:
        parent_raw = height_sequence_1  # may raise NameError
    except Exception:
        parent_raw = None

    if parent_raw is not None:
        try:
            parent = np.asarray(parent_raw, dtype=np.float64)
            if parent.ndim != 1 or parent.size == 0:
                raise ValueError
            if not np.isfinite(parent).all():
                raise ValueError
            parent = np.clip(parent, 0.0, 1000.0)
        except Exception:
            parent = None

    if parent is not None:
        n = int(parent.size)
        a = parent.copy()
    else:
        n = 256
        a = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3.  Normalise to the simplex (sum = 1) and set a valid start
    # ------------------------------------------------------------------
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    # ------------------------------------------------------------------
    # 4.  Keep the best candidate seen so far (including uniform)
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    grad_len = conv_len + n - 1               # length of w * a_rev
    fft_len_g = 1 << ((grad_len - 1).bit_length())

    # hyper‑parameters
    eta0 = 0.5                # initial step magnitude
    eps = 1e-4                # smooth‑max temperature
    eps_min = 5e-5
    eps_decay_interval = 2000
    jitter_interval = 100
    max_no_improve = 2500
    last_improve = 0

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 5.  Full autocorrelation via FFT
        # --------------------------------------------------------------
        Fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]

        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # 6.  Record improvement
        # --------------------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            last_improve = it
            if best_score <= target_score:
                break

        # --------------------------------------------------------------
        # 7.  Smooth‑max weights
        # --------------------------------------------------------------
        if it % eps_decay_interval == 0 and eps > eps_min:
            eps = max(eps_min, eps * 0.8)

        shifted = conv - conv.max()          # for numeric stability
        w = np.exp(shifted / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / conv_len)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 8.  Gradient of the smooth surrogate (matrix‑free)
        # --------------------------------------------------------------
        a_rev = a[::-1]
        Fw = np.fft.rfft(w, n=fft_len_g)
        Fa_rev = np.fft.rfft(a_rev, n=fft_len_g)
        conv_wr = np.fft.irfft(Fw * Fa_rev, n=fft_len_g)
        g = conv_wr[(n - 1):(n - 1 + n)]      # slice of length n

        # --------------------------------------------------------------
        # 9.  Exponentiated‑gradient update, project back to simplex
        # --------------------------------------------------------------
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        total = a.sum()
        if total <= 0 or not np.isfinite(total):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= total

        # --------------------------------------------------------------
        # 10.  Periodic jitter to avoid stagnation
        # --------------------------------------------------------------
        if it % jitter_interval == 0:
            jitter = np.exp(rng.normal(0.0, 0.008, size=n))
            a = a * jitter
            a /= a.sum()

        # --------------------------------------------------------------
        # 11.  Simple stagnation restart
        # --------------------------------------------------------------
        if it - last_improve > max_no_improve:
            # perturb the current best and continue
            perturb = rng.exponential(scale=0.001, size=n)
            a = best + perturb
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = it

    # Ensure the returned list is plain Python floats
    return [float(v) for v in best]
