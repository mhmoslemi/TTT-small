import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_candidates = np.where(rho_arr > 0)[0]
    if rho_candidates.size == 0:
        theta = 0.0
    else:
        rho = rho_candidates[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative floats with a low AC1 bound.
    The routine respects the time budget and keeps the best candidate
    discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initialise – use the provided parent if available
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a modest sqrt‑series of length 256
        a = _sqrt_series(256).astype(np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.ones_like(a)
    a = a / a.sum()
    a = _project_simplex(a)

    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())          # next power of two for FFT

    # --------------------------------------------------------
    # helper: evaluate the AC1 upper bound (sum is 1, so factor 2*n)
    # --------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        fv = np.fft.rfft(vec, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------
    # helper: sub‑gradient of the max‑lag autocorrelation
    # --------------------------------------------------------
    def _subgrad(vec: np.ndarray) -> np.ndarray:
        fv = np.fft.rfft(vec, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = conv.max()
        # numerical tolerance to capture ties
        tol = 1e-12 * max(1.0, M)
        active = np.where(conv >= M - tol)[0]
        g = np.zeros_like(vec)
        for k in active:
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            idx = np.arange(i_min, i_max + 1)
            partner = k - idx
            g[idx] += vec[partner]                # contribution of this lag
        grad = 2.0 * g
        if active.size > 0:
            grad /= active.size
        return grad

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    # optimisation parameters
    vel = np.zeros_like(cur)          # momentum term
    beta = 0.9                        # momentum decay
    eta0 = 0.6                        # base step size for gradient moves
    max_frac = 0.2                    # maximum fraction of source mass to move
    T = 0.01                          # initial temperature for SA moves
    T_min = 5e-6
    cooling = 0.9995
    no_improve = 0
    restart_after = 2000
    iteration = 0
    target = 1.5028

    # --------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ----- decide move type -------------------------------------------------
        if iteration % 5 == 0:                     # gradient‑based move
            grad = _subgrad(cur)
            # momentum update
            vel = beta * vel + (1.0 - beta) * grad
            eta = eta0 / np.sqrt(iteration)
            cand = cur - eta * vel
            cand = _project_simplex(cand)
        else:                                      # random mass‑transfer (SA style)
            # pick source proportionally to its mass
            if cur.sum() == 0.0:
                src = rng.integers(0, n)
            else:
                src = rng.choice(n, p=cur / cur.sum())
            if cur[src] <= 0.0:
                cand = cur.copy()
            else:

                if tgt >= src:
                    tgt += 1
                frac = rng.random() * max_frac
                delta = cur[src] * frac
                cand = cur.copy()
                cand[src] -= delta
                cand[tgt] += delta
                # safeguard against tiny negatives
                if cand[src] < 0.0:
                    cand[src] = 0.0
                cand = _project_simplex(cand)

        cand_score = _score(cand)

        # ----- record best -------------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- accept / reject ----------------------------------------------------
        diff = cand_score - cur_score
        accept = False

        else:
            prob = np.exp(-diff / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score

        # ----- temperature cooling ------------------------------------------------
        T = max(T * cooling, T_min)

        # ----- stagnation restart ------------------------------------------------
        if no_improve >= restart_after:
            # re‑initialise from a fresh Dirichlet draw
            fresh = rng.random(n)
            fresh = fresh / fresh.sum()
            cur = _project_simplex(fresh)
            cur_score = _score(cur)
            vel.fill(0.0)
            no_improve = 0
            T = 0.01

        # ----- early exit --------------------------------------------------------
        if best_score < target:
            break

    # --------------------------------------------------------
    # Return the best sequence discovered
    # --------------------------------------------------------
    return [float(x) for x in best]
