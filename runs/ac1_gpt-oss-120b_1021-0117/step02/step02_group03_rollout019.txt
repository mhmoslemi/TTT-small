# Project onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Simple sqrt‑series (a*a == 1+…+x^{2n-2}) – used if no parent.
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        s = float(np.dot(a[1:k], a[k - 1 : 0 : -1])) if k > 1 else 0.0
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Max of the full convolution a*a using FFT (fast, linear memory)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Main optimisation entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------
    # initialise – use provided parent when present, else a simple start.
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 256
        a = _sqrt_series(n0)

    # simplex normalisation (scale‑invariant)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a_sum
    a = _project_simplex(a)

    n = a.shape[0]

    # FFT size for a*a (next power of two)
    L_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # Adam optimiser parameters (operates in the simplex)
    # ------------------------------------------------------------
    lr = 0.08
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # ------------------------------------------------------------
    # Temperature schedule for smooth max (log‑sum‑exp)
    # ------------------------------------------------------------
    tau_start, tau_end = 0.4, 18.0

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------
    # bookkeeping of the best solution seen
    # ------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(a, L_fft)

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------
        # 1) full autocorrelation (via FFT) – also current score
        # --------------------------------------------------------
        fv = np.fft.rfft(a, n=L_fft)
        conv_full = np.fft.irfft(fv * fv, n=L_fft)[: 2 * n - 1]
        cur_max = float(conv_full.max())
        cur_score = 2.0 * n * cur_max

        # --------------------------------------------------------
        # 2) smooth approximation of the max (soft‑max)
        # --------------------------------------------------------
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / (budget_s - 5.0))
        tau = tau_start + (tau_end - tau_start) * frac

        # stable soft‑max
        shifted = tau * (conv_full - cur_max)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback – pure max
            w = np.zeros_like(conv_full)
            w[np.argmax(conv_full)] = 1.0
        else:
            w /= w_sum

        # --------------------------------------------------------
        # 3) gradient of the smooth objective (matrix‑free via FFT)
        #    g_i = 2 * sum_k w_k * a_{k-i}
        # --------------------------------------------------------
        a_rev = a[::-1]
        Lc = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        w_fft = np.fft.rfft(w, n=Lc)
        a_rev_fft = np.fft.rfft(a_rev, n=Lc)
        conv_wr = np.fft.irfft(w_fft * a_rev_fft, n=Lc)
        start = n - 1
        grad = 2.0 * conv_wr[start:start + n]

        # --------------------------------------------------------
        # 4) Adam update on the unconstrained logits, then simplex proj.
        # --------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = _project_simplex(cand)

        # --------------------------------------------------------
        # 5) evaluate candidate (exact bound) and keep the best.
        # --------------------------------------------------------
        cand_score = 2.0 * n * _max_convolution(cand, L_fft)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            # early exit if target already beaten
            if best_score < 1.5028:
                break

        # accept if it improves the current point, or with a tiny chance
        if cand_score < cur_score or rng.random() < 0.003:
            a = cand

        # --------------------------------------------------------
        # 6) occasional random mass‑transfer for added exploration.
        # --------------------------------------------------------
        if rng.random() < 0.012:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.07
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

    # --------------------------------------------------------------------
    # Return the best sequence discovered as a plain Python list.
    # --------------------------------------------------------------------
    return [float(x) for x in best]
