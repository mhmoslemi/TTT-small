def _fft_convolve(a, b, L):
    # returns full linear convolution truncated to length len(a)+len(b)-1
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[:a.size + b.size - 1]

def _softmax_weights(conv, lam):
    # conv: array length 2n-1
    maxc = conv.max()

        return np.full_like(conv, 1.0 / conv.size)

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT based convolution, softmax, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution via FFT, truncated to the exact length
    len(a)+len(b)-1.  L must be a power of two ≥ that length.
    """
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: a.size + b.size - 1]


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Returns a probability vector proportional to exp(lam * conv),
    numerically stabilized by subtracting the maximum.
    """
    maxc = conv.max()
    exps = np.exp(lam * (conv - maxc))
    s = exps.sum()
    if s == 0 or not np.isfinite(s):
        # fallback to uniform – should never happen under normal use
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s
