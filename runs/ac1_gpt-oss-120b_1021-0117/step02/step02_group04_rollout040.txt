import math
import time

import numpy as np

# ----------------------------------------------------------------------
# utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound using FFT (identical to the evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _make_exponential(n: int, c: float) -> np.ndarray:
    """
    Return a normalized non‑negative step function a_i ∝ exp(-c*i/n).
    The vector already sums to 1.
    """
    # r = exp(-c/n); a_i = r**i * (1-r)/(1-r**n)  (exact normalization)
    exp_factor = np.exp(-c * np.arange(n) / n)
    denom = 1.0 - math.exp(-c)               # 1 - r**n
    scale = (1.0 - math.exp(-c / n)) / denom  # (1-r)/(1-r**n)
    return exp_factor * scale


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound
    is as small as possible.  The routine runs for up to ``budget_s``
    seconds and returns the best sequence seen.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- problem size -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:          # fallback if parent not provided
        parent = None
        n = 256

    L = _next_pow2(2 * n - 1)

    # ----- initial construction -----------------------------------------
    # start from parent if available, otherwise from a simple exponential
    if parent is not None:
        base = parent.astype(np.float64).copy()
        base = np.maximum(base, 0.0)
    else:
        base = _make_exponential(n, c=1.0)

    # normalise to sum‑1 (objective is scale‑invariant)
    s = base.sum()
    if s <= 0.0:
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        base = base / s

    # alternative exponential initialisation; keep the better one
    exp_init = _make_exponential(n, c=1.2)
    if _ac1_score_fft(exp_init, n, L) < _ac1_score_fft(base, n, L):
        base = exp_init

    best = base.copy()
    best_score = _ac1_score_fft(best, n, L)

    cur = best.copy()
    cur_score = best_score

    # ----- optimisation parameters ---------------------------------------
    temperature = 1.0
    temp_decay = 0.9999        # slow annealing
    step_lr = 0.25             # learning rate for gradient‑type move
    lr_decay = 0.9999
    target_score = 1.50274349

    # ----- main loop ----------------------------------------------------
    while time.monotonic() < deadline:
        # convolution of current state
        fa_cur = np.fft.rfft(cur, n=L)
        conv_cur = np.fft.irfft(fa_cur * fa_cur, n=L)[: 2 * n - 1]
        max_b = float(conv_cur.max())
        k_max = int(conv_cur.argmax())
        sum_cur = float(cur.sum())
        if sum_cur < 1e-12:
            break
        cur_score = 2.0 * n * max_b / (sum_cur * sum_cur)

        # ----- propose a candidate ---------------------------------------
        move_type = rng.random()

        elif move_type < 0.7:                   # random block scaling

            cand[start:start + width] *= factor

            cand[i], cand[j] = cand[j], cand[i]

            cur_score = best_score
            temperature = 1.0
            step_lr = 0.25
