import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) for non‑negative vectors with sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast full linear convolution via FFT (real‑to‑real)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Score identical to evaluate_sequence for a normalized vector
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ------------------------------------------------------------------
    # Initialise: use the provided parent if present, otherwise a simple
    # decreasing series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a0 = 1.0 / (np.arange(1, n + 1, dtype=np.float64))
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s
    n = a0.size

    # ------------------------------------------------------------------
    # Keep the best vector seen so far
    # ------------------------------------------------------------------
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    cur = best_vec.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------
    # Parameters for the adaptive sub‑gradient descent
    # ------------------------------------------------------------
    step = 0.5               # current step size
    step_inc = 1.02
    step_dec = 0.5
    min_step = 1e-6
    max_step = 3.0
    momentum = np.zeros_like(cur)
    beta = 0.9               # momentum coefficient
    it = 0
    no_improve = 0

    # ------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution and the active (maximal) index
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))

        # sub‑gradient of the active quadratic term
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # candidate step
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < cur_sc:                     # successful step
            cur, cur_sc = cand, cand_sc
            step = min(max_step, step * step_inc)
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:                                     # not improving
            step = max(min_step, step * step_dec)
            # occasional Metropolis–like acceptance
            prob = np.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # --------------------------------------------------------
        # Random mass‑transfer move (exploration)
        # --------------------------------------------------------
        if rng.random() < 0.03:   # modest frequency
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # --------------------------------------------------------
        # Periodic jitter around the best known solution
        # --------------------------------------------------------
        if it % 2500 == 0:
            if rng.random() < 0.35:
                jitter = rng.normal(scale=5e-4, size=n)
                tmp = best_vec + jitter
                tmp = np.maximum(tmp, 0.0)
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # --------------------------------------------------------
        # Early exit if target reached
        # --------------------------------------------------------
        if best_sc < 1.50274349:
            break

        # --------------------------------------------------------
        # Keep the loop responsive to the deadline
        # --------------------------------------------------------
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break

    # Return a plain Python list
    return [float(x) for x in best_vec]
