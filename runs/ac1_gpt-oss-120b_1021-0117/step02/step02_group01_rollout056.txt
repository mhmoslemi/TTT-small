import time
import numpy as np

# --------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Analytic sqrt‑series (coefficients a with a*a = 1+…+x^{2n‑2})
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# --------------------------------------------------------------
# Full convolution via FFT (returns full vector)
# --------------------------------------------------------------
def _full_conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * len(x) - 1]
    return conv

    # ------------------------------------------------------------------
    # Initial sequence (prefer the provided parent)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(1024)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)               # enforce sum = 1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # initial convolution information
    conv = _full_conv_fft(a, L)
    max_val = float(conv.max())
    max_idx = int(conv.argmax())
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # Parameters for the optimizer
    # --------------------------------------------------------------
    step0 = 0.5                # initial step size (will be scaled)
    beta = 0.9                 # momentum decay
    v = np.zeros_like(a)       # momentum buffer
    iter_cnt = 0
    prob_random = 0.03         # probability of a random mass transfer
    eps_factor = 1e-6          # tolerance for “active” convolution indices

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- compute active set of indices near the maximum -----
        eps = max_val * eps_factor + 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # ----- sub‑gradient (sum of contributions of active indices) -----
        g = np.zeros_like(a)
        for idx in active:
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            j = np.arange(lo, hi + 1, dtype=int)
            g[j] += 2.0 * a[idx - j]

        # ----- momentum update -----
        v = beta * v + (1.0 - beta) * g

        # ----- step size (1/√iteration) and normalisation -----
        step = step0 / np.sqrt(iter_cnt + 1.0)
        # scale so that the largest coordinate moves at most `step`
        max_v = np.max(v) + 1e-12
        step_norm = step / max_v

        # ----- line‑search over a few factors -----
        improved = False
        best_local = None
        best_local_val = max_val
        best_local_idx = max_idx
        for factor in (0.5, 1.0, 2.0):
            cand = a - factor * step_norm * v
            cand = _project_simplex(cand)
            conv_c = _full_conv_fft(cand, L)
            val_c = float(conv_c.max())
            if val_c < best_local_val - 1e-15:
                best_local_val = val_c
                best_local = cand
                best_local_idx = int(conv_c.argmax())
                improved = True

        if improved:
            a = best_local
            conv = _full_conv_fft(a, L)
            max_val = best_local_val
            max_idx = best_local_idx
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

        # ----- occasional random mass transfer (exploration) -----
        if rng.random() < prob_random:
            # pick a source with non‑trivial mass
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size:
                i = rng.choice(nonzero)
                j = rng.integers(0, n - 1)
                while j == i:
                    j = rng.integers(0, n - 1)
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                conv2 = _full_conv_fft(cand2, L)
                val2 = float(conv2.max())
                if val2 < max_val - 1e-15:
                    a = cand2
                    conv = conv2
                    max_val = val2
                    max_idx = int(conv2.argmax())
                    cur_score = 2.0 * n * max_val
                    if cur_score < best_score:
                        best_score = cur_score
                        best = a.copy()

        iter_cnt += 1

        # Early bail‑out if we are very close to the deadline
        if time.monotonic() + 0.3 > deadline:
            break

    # ------------------------------------------------------------------
    # Return a plain Python list – the evaluator expects a list of floats
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
