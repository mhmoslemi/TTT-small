import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (L1‑norm = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Sub‑gradient of (a*a)[k] at a given lag k
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:                # diagonal term counted twice -> subtract once
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# fast evaluation of the AC1 bound (no extra denominator – sum = 1)
# ------------------------------------------------------------
def _score_fft(a: np.ndarray, L: int) -> float:
    """Return 2·n·max(conv) for a normalized (sum=1) vector a."""
    n = a.shape[0]
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative floats whose AC1 bound
    (as evaluated by the provided `evaluate_sequence`) is as low as possible.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------
    # initialise – use the supplied parent if present, else uniform
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = np.full(256, 1.0 / 256.0, dtype=np.float64)   # fallback

    # normalise to the simplex (sum = 1)
    a = a / a.sum()
    a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # length for zero‑padded FFT

    # bookkeeping of the best ever seen
    best = a.copy()
    best_score = _score_fft(best, L)

    # parameters for the sub‑gradient descent
    iteration = 0
    eta0 = 0.6                                   # base step size
    target_score = 1.50274349

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # current convolution and its maximum
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # indices where the maximum is (almost) attained
        eps = 1e-12
        active = np.where(conv >= cur_max - eps)[0]

        # gradient (sum of active sub‑gradients)
        grad = np.zeros_like(a)
        for k in active:
            grad += _grad_for_k(a, int(k))

        norm = np.linalg.norm(grad)

        # generate a few candidate updates with decreasing step magnitudes
        cand_best = a
        cand_best_score = cur_score

        if norm > 0.0:
            base_eta = eta0 / np.sqrt(iteration)
            for mul in (1.0, 0.5, 0.25, 0.125):
                step = base_eta * mul
                cand = a - step * (grad / norm)
                cand = _project_simplex(cand)
                cand_score = _score_fft(cand, L)
                if cand_score < cand_best_score:
                    cand_best_score = cand_score
                    cand_best = cand

        # occasional random mass‑transfer to escape plateaus
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.15
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

        # decide whether to accept the candidate
        if cand_best_score < cur_score or rng.random() < 0.005:
            a = cand_best

        # update the global best
        if cand_best_score < best_score:
            best_score = cand_best_score
            best = cand_best.copy()

        # early exit if we already beat the target
        if best_score <= target_score:
            break

    # --------------------------------------------------------
    # return the best sequence found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best]
