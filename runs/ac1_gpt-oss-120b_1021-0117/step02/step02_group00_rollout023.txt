def _grad_weights(a, weights):
    # a: 1D np.ndarray (n)
    # weights: 1D np.ndarray (2n-1) corresponding to w_k
    # compute g_i = 2 * sum_j a_j * w_{i + j}
    # using FFT correlation
    n = a.shape[0]
    L = 1 << ((len(weights) + n - 1).bit_length())
    # compute FFTs
    FA = np.fft.rfft(a, n=L)
    FW = np.fft.rfft(weights, n=L)
    # correlation: ifft( FW * conj(FA) )
    corr = np.fft.irfft(FW * np.conj(FA), n=L)
    # valid part indices n-1 : n-1+n
    g = 2.0 * corr[n-1 : n-1 + n]
    return g

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast full linear convolution via FFT (a*a)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return full convolution a*a (length 2*len(a)-1) using FFT size L."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of max‑convolution term (active index only)
# ------------------------------------------------------------
def _subgrad_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) with respect to a."""
    n = a.size
    k = int(np.argmax(conv))
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k - idx]
    return g


# ------------------------------------------------------------
# Gradient of smoothed max (soft‑max) using FFT‑based correlation
# ------------------------------------------------------------
def _grad_smooth(a: np.ndarray, weights: np.ndarray, L: int) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k (a*a)_k = 2 * correlation(w, a).
    w has length 2n-1, a length n.
    """
    n = a.size
    FA = np.fft.rfft(a, n=L)
    FW = np.fft.rfft(weights, n=L)
    corr = np.fft.irfft(FW * np.conj(FA), n=L)
    # valid part indices n-1 .. n-1+n-1
    g = 2.0 * corr[n - 1 : n - 1 + n]
    return g
