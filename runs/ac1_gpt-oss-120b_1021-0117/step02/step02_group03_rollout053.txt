def propose_candidate(...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # initial sequence
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = _sqrt_series(256).astype(np.float64)
    # scale to sum 1 via projection onto simplex (norm)
    a = a / max(a.sum(), 1e-12)
    a = _project_simplex(a)

    n = a.shape[0]
    Lfft = 1 << ((2*n - 1).bit_length())

    # compute best
    def conv_max(x):
        fv = np.fft.rfft(x, n=Lfft)
        conv = np.fft.irfft(fv*fv, n=Lfft)[:2*n-1]
        return float(conv.max())

    best = a.copy()
    best_score = 2 * n * conv_max(a)
    # store current score also
    cur_score = best_score
    # parameters
    eta0 = 0.8
    iteration = 0
    # start temperature for simulated annealing
    T0 = 0.01
    # main loop
    while time.monotonic() < deadline:
        iteration += 1
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv*fv, n=Lfft)[:2*n-1]
        M = float(conv.max())
        k = int(conv.argmax())
        cur_score = 2 * n * M

        # gradient
        grad = _grad_for_k(a, k)
        norm = np.linalg.norm(grad)
        if norm > 0:
            eta = eta0 / np.sqrt(iteration)
            # direction step
            cand = a - eta * (grad / norm)
        else:
            cand = a.copy()
        cand = _project_simplex(cand)

        # random perturbation
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.2
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)

        # evaluate candidate
        cand_score = 2 * n * conv_max(cand)
        # acceptance: SA
        if cand_score < cur_score:
            a = cand
        else:
            # temperature decreasing
            T = T0 * np.exp(-iteration / 50000.0) # scaling
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                a = cand

        # record best
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            # optional early break
            if best_score < 1.5028:
                break

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)  # small buffer

    # init a
    if 'height_sequence_1' in globals():

    else:
        # fallback: sqrt_series length 256, padded to maybe 256?
        a = _sqrt_series(256).astype(np.float64)

    # project to simplex
    a = a / max(a.sum(), 1e-12)
    a = _project_simplex(a)

    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())

    def conv_max(x):
        fv = np.fft.rfft(x, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        return float(conv.max())

    # best
    best = a.copy()
    best_score = 2.0 * n * conv_max(a)

    iteration = 0
    # schedule parameters
    total_time = budget_s - 0.2
    explore_frac = 0.6
    explore_until = time.monotonic() + explore_frac * total_time
    eta0_explore = 1.0
    eta0_exploit = 0.5
    temp0 = 0.02  # starting temperature for SA

    while time.monotonic() < deadline:
        iteration += 1
        # compute conv for a
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv.max())
        k = int(conv.argmax())
        cur_score = 2.0 * n * M

        # gradient for active lag
        grad = _grad_for_k(a, k)
        norm = np.linalg.norm(grad)
        # choose step size based on phase
        if time.monotonic() < explore_until:
            eta0 = eta0_explore
        else:
            eta0 = eta0_exploit
        if norm > 0:
            eta = eta0 / np.sqrt(iteration)
            cand = a - eta * (grad / norm)
        else:
            cand = a.copy()
        cand = _project_simplex(cand)

        # occasional random mass transfer
        if rng.random() < 0.05:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.2
                delta = a[i] * frac
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)

        # compute candidate score
        cand_score = 2.0 * n * conv_max(cand)

        # acceptance
        if cand_score < cur_score:
            a = cand
        else:
            # simulated annealing temperature based on iteration

def _grad_for_k(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # initialise
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

        a = np.ones(n_fallback, dtype=np.float64) / n_fallback
    a = a / max(a.sum(), 1e-12)
    a = _project_simplex(a)

    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())
    target_score = 1.5028

        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        return float(conv.max())

    best = a.copy()

    iteration = 0
    eta0 = 1.0               # initial step magnitude
    temp0 = 0.02             # annealing temperature start
    explore_frac = 0.6
    explore_until = time.monotonic() + explore_frac * (deadline - time.monotonic())

    while time.monotonic() < deadline:

        # current convolution & active lag
        fv = np.fft.rfft(a, n=Lfft)

        M = float(conv.max())
        k = int(conv.argmax())
        cur_score = 2.0 * n * M

        # sub‑gradient step

        norm = np.linalg.norm(grad)
        if norm > 0.0:
