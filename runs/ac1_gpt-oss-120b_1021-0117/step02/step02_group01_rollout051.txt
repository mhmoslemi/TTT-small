def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: sqrt_series length maybe 256
        a = _sqrt_series(256)

    # project onto simplex
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2*n - 1).bit_length())  # next power of 2
    # compute initial max conv
    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # parameters
    step = 0.2
    min_step = 1e-6
    decay = 0.9990
    beta = 0.9   # momentum factor
    momentum = np.zeros_like(a)
    prob_random = 0.02
    prob_restart = 0.005

    while time.monotonic() < deadline:
        # gradient for current max index
        lo = max(0, max_idx - (n-1))
        hi = min(n-1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi+1)
            g[j] = 2.0 * a[max_idx - j]
        # update momentum
        momentum = beta * momentum + (1 - beta) * g

        cand = a - step * momentum
        cand = _project_simplex(cand)

        max_val_c, max_idx_c = _conv_max_and_index(cand, L)
        cand_score = 2.0 * n * max_val_c

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            max_idx = max_idx_c
            # reset momentum maybe?
            momentum *= 0
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # probabilistic acceptance
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-6)):
                a = cand
                cur_score = cand_score
                max_idx = max_idx_c

        # random mass transfer
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.5
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max_val2, max_idx2 = _conv_max_and_index(cand2, L)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    max_idx = max_idx2
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
        # occasional restart of a random convex combination of best and a new rand vector
        if rng.random() < prob_restart:
            rand = rng.random(n)
            rand = _project_simplex(rand)
            a_restart = 0.5 * a + 0.5 * rand
            a_restart = _project_simplex(a_restart)
            max_val_r, max_idx_r = _conv_max_and_index(a_restart, L)
            cur_score_r = 2.0 * n * max_val_r
            if cur_score_r < cur_score:
                a = a_restart
                cur_score = cur_score_r
                max_idx = max_idx_r
                momentum = np.zeros_like(a)
                if cur_score_r < best_score:
                    best_score = cur_score_r
                    best = a_restart.copy()

        # decay step
        step = max(min_step, step * decay)

        if time.monotonic() + 0.5 > deadline:
            break

    return [float(x) for x in best]

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_idx = np.where(rho_arr > 0)[0]
    if rho_idx.size:
        rho = rho_idx[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def _conv_max_and_index(a: np.ndarray, L: int):
    """Return (maximum convolution value, its index) for a*a via FFT."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(a) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx
