import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        cur = parent.copy()
    except Exception:
        # fallback: a simple analytically generated start
        n = 4806
        cur = _gen_binomial(n)

    n = int(cur.size)
    L = _next_pow2(2 * n - 1)

    # evaluate the starting point
    cur_score = _ac1_score(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # hyper‑parameters
    temperature = 1.0
    decay = 0.9995
    grad_factor = 0.25          # base step size for gradient‑type moves
    target = 1.50274349

    min_block = max(2, n // 300)
    max_block = max(2, n // 6)

    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        move = rng.random()
        cand = cur.copy()

        # ------------------------------------------------------
        # 1) gradient guided reduction of the current max peak
        if move < 0.55:
            # convolution and index of the peak
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
