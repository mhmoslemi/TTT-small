import time
import numpy as np

def _greedy_set(n: int, L: int, rng: np.random.Generator):
    """
    Build a subset of {0,…,n-1} such that every ordered sum i+j appears
    at most L times.  Returns the chosen indices and the full count array.
    """
    idx = np.arange(n, dtype=np.int32)
    rng.shuffle(idx)
    selected = []
    # number of possible sums = 2*n-1, use a small integer dtype
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

    for cand in idx:
        # self‑pair (i,i) contributes only once
        if sum_counts[cand + cand] + 1 > L:
            continue
        ok = True
        for i in selected:
            s = i + cand
            if sum_counts[s] + 2 > L:           # (i,cand) and (cand,i)
                ok = False
                break
        if not ok:
            continue
        # accept the candidate
        selected.append(int(cand))
        sum_counts[cand + cand] += 1
        # update counts for pairs with earlier elements
        for i in selected[:-1]:
            sum_counts[i + cand] += 2
    return np.array(selected, dtype=np.int32), sum_counts
