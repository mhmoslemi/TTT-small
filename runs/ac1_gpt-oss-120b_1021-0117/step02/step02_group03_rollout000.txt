import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Maximum value of a full self‑convolution using FFT
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Gradient of the convolution at a given lag k
# ------------------------------------------------------------
def _add_grad_for_k(grad: np.ndarray, a: np.ndarray, k: int) -> None:
    n = a.shape[0]
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    grad[idx] += 2.0 * a[partner]
    if k % 2 == 0:
        mid = k // 2
        if 0 <= mid < n:
            grad[mid] -= a[mid]


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------------
    # initialise – use the provided parent if present, otherwise sqrt series
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 256
        a = _sqrt_series(n0)

    # Normalise to the probability simplex (sum = 1, non‑negative)
    a = a / a.sum()
    a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two for FFT

    # --------------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(best, L)
    velocity = np.zeros_like(a)
    iteration = 0
    last_improve = 0
    eta0 = 0.8          # initial step size factor
    beta = 0.8          # momentum coefficient
    target_score = 1.50274349

    # --------------------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # current convolution and its maximum
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M

        # gradient (sum over all active maximal lags)

        norm = np.linalg.norm(grad)
        if norm < 1e-12:

                delta = a[i] * 0.15 * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)
                continue
            else:
                break

        # momentum‑based update
        eta = eta0 / np.sqrt(iteration)
        velocity = beta * velocity - eta * (grad / norm)
        cand = a + velocity
        cand = _project_simplex(cand)
