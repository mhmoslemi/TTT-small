def _project_onto_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

        trial_score = 2.0 * n * float(conv_trial.max())

        if trial_score < cur_score:
            a = a_trial
            step = min(step * 1.05, 1.0)
        else:
            step = max(step * 0.5, 1e-6)
            if rng.random() < 0.04:
                jitter = np.exp(rng.normal(0.0, 0.02, size=n))
                a = _project_onto_simplex(a * jitter)

        iteration += 1

        # occasional restart if stuck
        if iter_no_improve > 600:
            pert = np.exp(rng.normal(0.0, 0.03, size=n))
            a = _project_onto_simplex(best_a * pert)
            step = 0.5
            vel.fill(0.0)
            iter_no_improve = 0

    return best_a.tolist()
