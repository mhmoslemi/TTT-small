import time, numpy as np

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    c = np.fft.irfft(fv*fv, n=L)[:2*n-1]
    return c

def _score_vec(a, L=None):
    n = a.shape[0]
    c = _conv_fft(a, L)
    maxb = c.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * maxb / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise base vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a simple Dirichlet (alpha=1) sample
        n0 = 256
        a0 = rng.random(n0)
        a0 = a0 / a0.sum()
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length, reused

    # store the best solution seen so far
    best_vec = a0.copy()
    best_sc = _score_vec(best_vec, L)

    # optimisation parameters
    alpha0 = 0.9          # initial step size
    beta = 0.9            # momentum factor
    momentum = np.zeros_like(best_vec)
    it = 0
    cur = best_vec.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- sub‑gradient step based on the active convolution entry -----
        conv = _conv_fft(cur, L)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # diminishing step size
        alpha = alpha0 / np.sqrt(it)

        cand = cur - alpha * momentum
        cand = _project_simplex(cand)
        cand_sc = _score_vec(cand, L)

        # accept if improvement, otherwise possibly accept downhill
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):

import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear convolution using a real FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using an FFT of length L."""
    fv = np.fft.rfft(a, n=L)
    c = np.fft.irfft(fv * fv, n=L)[: 2 * a.shape[0] - 1]
    return c


# ------------------------------------------------------------
# Score identical to evaluate_sequence for a summed‑to‑1 vector
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    conv = _conv_fft(a, L)
    maxb = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * maxb / (s * s)
