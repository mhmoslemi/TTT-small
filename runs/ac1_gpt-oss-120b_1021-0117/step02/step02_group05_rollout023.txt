    max_b = float(conv.max())

    return 2.0 * n * max_b / (sum_a * sum_a)

            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            cur_conv = _conv(cur, n, L)

        temperature *= decay

        # occasional restart from the best seen
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            cur_conv = _conv(cur, n, L)
            temperature = 1.0

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    if best is None or best.size == 0:
        best = base_seq
    return [float(x) for x in best]
```
