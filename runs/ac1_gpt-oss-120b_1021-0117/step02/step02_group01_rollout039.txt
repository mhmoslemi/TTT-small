def _initial_conv(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    return conv

    conv = np.fft.irfft(fv * fv, n=L)[:2*vec.size - 1]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    if 'height_sequence_1' in globals():

    max_frac = 0.5
    focus_prob = 0.75

    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # choose donor i, receiver j
        if rng.random() < focus_prob:
            i = rng.integers(0, n)
            j = idx_max - i

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution (FFT) returning full vector, its maximum value and index
# ----------------------------------------------------------------------
def _conv_full_max_idx(vec: np.ndarray, length_fft: int):
    """Return (full convolution a*a, max value, argmax) for real vec."""
    fv = np.fft.rfft(vec, n=length_fft)
    conv = np.fft.irfft(fv * fv, n=length_fft)[: 2 * vec.size - 1]
    idx = int(conv.argmax())
    return conv, float(conv[idx]), idx

    # --------------------------------------------------------------

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

        a = np.arange(n_fallback, 0, -1, dtype=np.float64)
        a = a / a.sum()

    # make sure it is non‑negative and normalise the ℓ₁‑norm to 1
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= total

    n = a.size

    conv, cur_max, idx_max = _conv_full_max_idx(a, L)
    best = a.copy()

    # --------------------------------------------------------------
