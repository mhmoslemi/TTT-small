import time
import numpy as np

# ---------- helper utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (non‑negative, sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        raise ValueError
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → return uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full convolution a*a (real FFT, returns length 2*n‑1 array)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]


# ---------- candidate generation ----------

def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """Return a list of plausible start vectors (each sums to 1)."""
    candidates = []

    # parent, if supplied
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            candidates.append(p / s)

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # triangular shape a_i ∝ (i+1)*(n-i)
    idx = np.arange(n, dtype=np.float64)
    tri = (idx + 1) * (n - idx)
    candidates.append(tri / tri.sum())

    # Catalan‑style weights w_i = C(2i,i)/4**i
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    candidates.append(w / w.sum())

    # sin‑power shapes
    theta = np.pi * (idx + 0.5) / n
    sinv = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sinv ** p
        candidates.append(v / v.sum())

    return candidates
