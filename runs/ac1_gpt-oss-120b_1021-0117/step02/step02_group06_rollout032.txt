def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 0.2  # safety.

    # Determine n
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.shape[0]
    except Exception:
        parent_arr = None
        n = 256

    # Build initial vectors (list of numpy arrays, normalized sum=1)
    cands = []

    # parent
    if parent_arr is not None and parent_arr.size == n and np.isfinite(parent_arr).all():
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cands.append(p / s)

    # Catalan as before
    def catalan_weights(n):
        w = np.empty(n, dtype=np.float64)
        w[0] = 1.0
        for i in range(1, n):
            w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
        return w

    w = catalan_weights(n)
    cands.append(w / w.sum())

    # arcsine shape
    idx = np.arange(n) + 0.5
    ar = 1.0/np.sqrt(idx * (n - idx))
    cands.append(ar / ar.sum())

    # sin powers
    theta = np.pi * (np.arange(n) + 0.5) / n
    sin_vals = np.sin(theta)
    for p in (0.2,0.5,1.0,2.0,3.0):
        v = sin_vals ** p
        cands.append(v / v.sum())

    # power law shape a_i ∝ (i+1)*(n-i) (tent)
    base = (np.arange(n)+1) * (n - np.arange(n))
    for p in (0.5, 1.0, 1.5,2.0):
        v = base ** p
        cands.append(v / v.sum())

    # random few
    for _ in range(5):
        v = rng.random(n)
        v = np.clip(v, 0.0, 1000.0)
        if v.sum() < 1e-12:
            v[rng.integers(n)] = 1.0
        cands.append(v / v.sum())

    # early evaluate each candidate (fast conv using FFT) to pick best
    fft_len = 1 << ((2*n-1).bit_length())
    # We'll compute score = 2*n*max(conv) as sum=1 already.
    best_vec = None
    best_score = np.inf
    for vec in cands:
        conv = np.fft.irfft(np.fft.rfft(vec, fft_len) ** 2, n=fft_len)[:2*n-1]
        cur = 2.0 * n * conv.max()
        if cur < best_score:
            best_score = cur
            best_vec = vec.copy()
    # If best_score < target*? Actually target includes denominator sum^2 but our sum=1 so same.
    target_score = 1.50274349
    if best_score <= target_score:
        return best_vec.tolist()

import time
import numpy as np

def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to sum = 1 (simple simplex projection)."""
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises the
    AC1 upper bound used by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

        n = parent_arr.shape[0]

    except Exception:

        n = 256

    # ----- generate a small set of plausible initial vectors --------------------
    init_vectors = []

        init_vectors.append(_normalize(p))

    # Catalan‑style weights

    init_vectors.append(_normalize(cat))

    # Power‑law “tent’’ shapes
