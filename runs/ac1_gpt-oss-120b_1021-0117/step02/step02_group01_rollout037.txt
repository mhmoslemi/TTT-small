import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# --------------------------------------------------------------
# Simplex projection (non‑negative, sum = 1)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# --------------------------------------------------------------
# Analytic sqrt‑series (fallback when no parent is supplied)
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# --------------------------------------------------------------
# Full convolution via FFT, return max value and its index
# --------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# --------------------------------------------------------------
# Build sparse matrix representing (a * b)[t] ≤ M constraints:
# rows = t = 0..2n-2, cols = j = 0..n-1,
# entry = a[t - j] if 0 ≤ t-j < n else 0
# --------------------------------------------------------------
def _build_convolution_csr(a: np.ndarray) -> sp.csr_matrix:
    n = a.shape[0]
    # each diagonal offset = -i carries constant value a[i]
    diag_data = [np.full(n, a_i, dtype=np.float64) for a_i in a]
    offsets = -np.arange(n, dtype=int)          # 0, -1, -2, …
    A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A
