import time
import numpy as np

def _top_k_indices(arr, k):
    """Return indices of the k largest values of arr (unsorted)."""
    if k >= arr.size:
        return np.arange(arr.size, dtype=np.int64)
    # argpartition gives the k largest unordered
    return np.argpartition(-arr, k - 1)[:k]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimization for the AC1 upper bound.
    Uses exponentiated gradient descent on the active autocorrelation peaks,
    projecting onto the simplex after each step and injecting occasional jitter.
    Returns the best sequence found (as a plain list of floats).
    """

    # ------------------------------------------------------------------

        n = int(parent.size)
        a = parent.copy()
    except Exception:

        a = np.full(n, 1.0 / n, dtype=np.float64)

    total = a.sum()
    if total < 0.01:

    else:
        a = a / total

    best = a.copy()

    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())  # next power of two

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # safety margin
