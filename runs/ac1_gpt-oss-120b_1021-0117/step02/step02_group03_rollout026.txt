import time

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a simple uniform non‑negative step‑function.
    The sequence is normalised to sum = 1, which (by the
    evaluator's formula) yields the theoretical lower bound of 2.
    """
    # Respect the time budget – this routine is instantaneous,
    # but we include the deadline logic for safety.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # Choose a modest length (any length works; bound = 2 for uniform).
    n = 256
    val = 1.0 / n
    # Build the list without allocating huge intermediate structures.
    seq = [val] * n

    # Ensure we return before the deadline.
    if time.monotonic() > deadline:
        # Fallback: return a tiny valid sequence.
