import time
import numpy as np

def _normalize_nonneg(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fall back to uniform
        n = v.shape[0]
        return np.full(n, 1.0 / n, dtype=np.float64)
    return v / s


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Projected (momentum) gradient descent on the AC1 upper bound.
    Uses the supplied parent `height_sequence_1` if available,
    otherwise starts from a uniform vector.
    Returns the best coefficient list found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Initialise a feasible sequence (sum = 1, non‑negative)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        a = _normalize_nonneg(np.clip(parent, 0.0, 1000.0))
        n = a.size
    except Exception:
        # fallback: modest length uniform vector
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Pre‑compute FFT length
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two

    # ------------------------------------------------------------------
    # 3. Baselines (uniform & parent if any)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.8               # initial step magnitude
    beta = 0.9               # momentum coefficient
    v = np.zeros_like(a)     # momentum buffer
    target_score = 1.50274349
    top_k = min(10, conv_len)   # number of shifts used for the sub‑gradient
    jitter_interval = 200
    jitter_sigma = 0.012

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution (full, via FFT) ---------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # ---- current objective (sum a = 1) --------------------------------
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # ---- sub‑gradient from the top‑k shifts -------------------------
        # indices of the k largest convolution values
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        # weights proportional to the convolution magnitude (more weight for larger)
        w = conv[top_idx].astype(np.float64)
        weight_sum = w.sum()
        if weight_sum == 0.0:
            # fallback to uniform weighting
            w = np.ones_like(w)
            weight_sum = w.sum()
        w = w / weight_sum

        g = np.zeros_like(a)
        for idx_shift, weight in zip(top_idx, w):
            # overlapping region of the two copies
            lo = max(0, idx_shift - (n - 1))
            hi = min(n - 1, idx_shift)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            partner = idx_shift - idx
            g[idx] += 2.0 * a[partner] * weight

        # ---- momentum update ---------------------------------------------
        v = beta * v + (1.0 - beta) * g

        # ---- step size (diminishing) ------------------------------------
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * v

        # ---- feasibility projection (non‑negative, sum = 1) ---------------
        a = _normalize_nonneg(a_new)

        # ---- occasional jitter to escape plateaus ------------------------
        if it % jitter_interval == 0:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = _normalize_nonneg(a * jitter)

        # ---- quick time‑out check ----------------------------------------
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
