import time
import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score(seq: np.ndarray, n: int, N: int, Lc: int):
    """Fast evaluation of the AC1 bound."""
    A = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(A * A, n=N)[:Lc]
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return np.inf, conv, max_b
    score = float(2.0 * n * max_b / (s * s))
    return score, conv, max_b

    Lc = 2 * n - 1                     # full auto‑convolution length
    Lcorr = Lc + n - 1                 # length needed for correlation
    N = 1 << (Lcorr.bit_length())      # power‑of‑two >= Lcorr

    # ------------------------------------------------------------------
    # Initial scoring
    # ------------------------------------------------------------------
    best = a.copy()
    best_score, best_conv, _ = _score(best, n, N, Lc)

    # ------------------------------------------------------------------
    # Main optimisation loop (soft‑max gradient descent + occasional jumps)
    # ------------------------------------------------------------------
    current = best.copy()
    current_score = best_score

    # schedule parameters
    beta_start = 5.0
    beta_end = 35.0
    eta_start = 0.6
    eta_end = 0.02

    # counters for stagnation handling
    no_improve = 0
    MAX_STAGNATE = 400

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        prog = elapsed / total if total > 0 else 0.0

        # annealing schedule
        beta = beta_start + (beta_end - beta_start) * prog
        eta = eta_start + (eta_end - eta_start) * prog

        # FFT of current vector
        A_fft = np.fft.rfft(current, n=N)

        # auto‑convolution (full)
        conv = np.fft.irfft(A_fft * A_fft, n=N)[:Lc]

        # soft‑max weights over convolution values
        max_c = np.max(conv)
        w = np.exp(beta * (conv - max_c))
        w_sum = np.sum(w)
        w /= w_sum

        # correlation w * current  (linear, non‑circular)
        W_fft = np.fft.rfft(w, n=N)
        corr_full = np.fft.irfft(W_fft * np.conj(A_fft), n=N)[:n]
        grad = 2.0 * corr_full                     # ∂M/∂a

        # project gradient onto the simplex tangent space
        proj = grad - np.mean(grad)

        # take a descent step, enforce non‑negativity, re‑normalise
        cand = current - eta * proj
        np.clip(cand, 0.0, None, out=cand)
        s_cand = np.sum(cand)
        if s_cand > 0:
            cand /= s_cand
        else:
            cand = current.copy()   # fall back if everything vanished

        # occasional random jump to escape flat regions
        if rng.random() < 0.015:
            width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            tmp = cand.copy()
            tmp[start_idx:start_idx + width] *= factor
            np.clip(tmp, 0.0, None, out=tmp)
            tmp /= np.sum(tmp)
            cand = tmp

        # evaluate candidate
        cand_score, cand_conv, _ = _score(cand, n, N, Lc)

        # accept if improvement (simple greedy)
        if cand_score < current_score:
            current = cand
            current_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                best_conv = cand_conv
        else:
            no_improve += 1
            # occasional uphill acceptance to avoid being stuck
            if rng.random() < 0.001:
                current = cand
                current_score = cand_score

        # if stagnating for a long time, restart from best with a small perturbation
        if no_improve >= MAX_STAGNATE:
            # small random redistribution while staying near best
            perturb = best.copy()
            idx = rng.integers(0, n)
            delta = perturb[idx] * rng.uniform(0.01, 0.07)
            perturb[idx] -= delta
            tgt = rng.integers(0, n)
            perturb[tgt] += delta
            perturb = np.clip(perturb, 0.0, None)
            perturb /= np.sum(perturb)
            current = perturb
            current_score = best_score
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best sequence found (scaled back – any scale works)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
