import time, math, numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    total_time = deadline - start_time

    # --------------------------------------------------------------------
    # initialise – use the provided parent if present, otherwise uniform
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # safety: non‑negative and normalise to sum = 1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / a.shape[0]
    else:
        a /= s

    n = a.shape[0]
    # length for zero‑padded FFT (next power of two >= 2n‑1)
    Lfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # fast score using FFT (2 * n * max convolution) – sum(a)=1
    # ------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        fv = np.fft.rfft(vec, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    best = a.copy()
    best_score = _score(best)
    cur_score = best_score

    target = 1.50274349
    if best_score <= target:
        return best.tolist()

    # ------------------------------------------------------------
    # simulated‑annealing‑style search
    # ------------------------------------------------------------
    # temperature and step size decay linearly with elapsed time
    start_temp, end_temp = 0.04, 1e-6      # acceptance scale
    start_step, end_step = 0.25, 1e-6      # max fraction of selected mass to move

    it = 0
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        it += 1

        # linear schedule
        elapsed = now - start_time
        frac = elapsed / total_time
        temp = start_temp * (1.0 - frac) + end_temp * frac
        step = start_step * (1.0 - frac) + end_step * frac

        # --------------------------------------------------------
        # propose a small mass transfer i -> j
        # --------------------------------------------------------
        i = rng.integers(0, n)
        ai = a[i]
        if ai <= 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = step * ai                # never exceed current mass
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # apply move in‑place
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)

        improved = new_score < cur_score
        if improved:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                if best_score <= target:
                    break
        else:
            # Metropolis acceptance for uphill moves
            if temp > 0.0:
                prob = math.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_score = new_score
                else:
                    # revert move
                    a[i] += delta
                    a[j] -= delta
            else:
                a[i] += delta
                a[j] -= delta

        # --------------------------------------------------------
        # occasional large stochastic remix to escape plateaus
        # --------------------------------------------------------
        if it % 5000 == 0:

            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                if best_score <= target:
                    break

    # --------------------------------------------------------------------
    # return the best sequence found as a plain Python list
    # --------------------------------------------------------------------
    return [float(x) for x in best]
