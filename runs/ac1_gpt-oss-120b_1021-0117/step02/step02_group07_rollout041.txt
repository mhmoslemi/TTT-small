import time
import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution of seq with itself using FFT."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    return conv


def _score_fft(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution array, max conv)."""
    conv = _conv_fft(seq, fft_len)
    max_b = float(np.max(conv))
    s = float(np.sum(seq))
    if s <= 0.0:
        return float("inf"), conv, max_b
    score = 2.0 * seq.size * max_b / (s * s)
    return score, conv, max_b
