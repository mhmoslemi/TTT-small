def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback sqrt series length 1024?
        a = _sqrt_series(1024)
    # project onto simplex (sum=1)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    def conv_max_and_index(vec):
        fv = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*len(vec)-1]
        idx = int(conv.argmax())
        return float(conv[idx]), idx
    max_val, max_idx = conv_max_and_index(a)
    cur_score = 2.0 * n * max_val  # sum=1
    best = a.copy()
    best_score = cur_score
    step = 1.0
    decay = 0.9998
    min_step = 1e-6
    beta = 0.9
    momentum = np.zeros_like(a)
    smooth_factor = 0.02
    prob_random = 0.005
    # optional big_move probability 0.001, move more mass
    while time.monotonic() < deadline:
        # compute gradient at current max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            idx_range = np.arange(lo, hi + 1)
            g[idx_range] = 2.0 * a[max_idx - idx_range]
        # update momentum
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        # project onto simplex
        cand = _project_simplex(cand)
        # smoothing
        if smooth_factor > 0:
            # blur kernel [0.25,0.5,0.25]
            blurred = np.convolve(cand, [0.25, 0.5, 0.25], mode='same')
            cand = (1 - smooth_factor) * cand + smooth_factor * blurred
            cand = _project_simplex(cand)
        # evaluate
        max_val_c, max_idx_c = conv_max_and_index(cand)
        cand_score = 2.0 * n * max_val_c
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            max_idx = max_idx_c
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # simulated annealing acceptance
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-8)):
                a = cand
                cur_score = cand_score
                max_idx = max_idx_c
                if cand_score < best_score:
                    best_score = cand_score
                    best = cand.copy()
        # random pair transfer
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4  # up to 40% of a[i]
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)  # sum preserved but might adjust
                max_val2, max_idx2 = conv_max_and_index(cand2)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    max_idx = max_idx2
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
        # step decay
        step = max(min_step, step * decay)
        # break if approaching deadline
        if time.monotonic() + 0.5 > deadline:
            break
    # final fallback: if best stays None, use original a
    return [float(x) for x in best]

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
