import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_idxs = np.where(rho_arr > 0)[0]
    if rho_idxs.size:
        rho = rho_idxs[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic sqrt‑series (fallback when no parent is supplied)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
# Helper: maximum of a*a using FFT (fast, linear‑time memory)
# ----------------------------------------------------------------------
def _max_convolution(v: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(v) - 1]
    return float(conv.max())
