def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init vector a0
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback uniform
        n = 256
        a0 = np.ones(n, dtype=np.float64) / n

    # enforce nonnegative and normalize to sum 1
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s

    n = a0.size
    # define conv size L = next pow2
    L = 1 << ((2*n-1).bit_length())
    # define functions using L globally for speed
    def conv_full(x):
        fv = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2*n-1]
        return conv

    def score_val(x):
        # assuming sum(x) approx 1
        conv = conv_full(x)
        maxc = conv.max()
        return 2.0 * n * maxc

    # helper: simplex projection (copy from previous)
    def project_simplex(v):
        # similar to earlier version with epsilon
        n_ = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n_+1) > 0)[0]
        if rho.size > 0:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        w = np.maximum(v - theta, 0.0)
        return w

    cur = a0.copy()
    cur_sc = score_val(cur)
    best = cur.copy()
    best_sc = cur_sc

    # hyperparameters

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT, returns length 2*len(a)-1 array."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # --------------------------------------------------------------
        # periodic jitter‑restart toward the best known solution
        # --------------------------------------------------------------
        if it % 4000 == 0 and rng.random() < 0.28:
            jitter = rng.normal(scale=5e-4, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # early exit if we already beat the target
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # return the best sequence found
    # ------------------------------------------------------------------
    return [float(x) for x in best]
