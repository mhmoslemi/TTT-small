def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = np.ones(n, dtype=np.float64)
    # ensure nonnegative and sum to 1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    # best
    best = a.copy()
    best_sc = _score(best)

    # optimizer state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 0.2  # base learning rate
    beta_temp = 0.5  # softmax temperature parameter
    max_beta = 80.0

    it = 0
    last_improve_it = 0

    while time.monotonic() < deadline:
        it += 1

        # compute convolution of current a
        # conv = _conv_fft(a)  # also compute max later
        # compute conv via FFT: a_fft = rfft, conv = irfft(...).
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        maxc = conv.max()
        cur_sc = 2.0 * n * maxc / (a.sum() ** 2)  # a.sum() should be 1

        # update softmax parameter gradually
        if it % 500 == 0 and beta_temp < max_beta:
            beta_temp = min(max_beta, beta_temp * 1.15)

        # softmax weights
        w = np.exp(beta_temp * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # compute gradient (smooth max) via convolution of w with reversed a
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = eta0 / np.sqrt(it)  # decaying learning rate
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        # project onto simplex
        cand = _project_simplex(cand)

        cand_sc = _score(cand)
        # accept if improvement
        if cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
        else:
            # occasional acceptance of worse move with temperature
            prob = np.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                a = cand
                cur_sc = cand_sc

        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            last_improve_it = it

        # Random mass-transfer exploration if stagnating
        if it - last_improve_it > 5000 and rng.random() < 0.15:
            i = rng.integers(0, n)
            if a[i] > 0:
                # choose distinct j
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.4
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best = tmp.copy()
                    best_sc = tmp_sc
                if tmp_sc < cur_sc:
                    a = tmp
                    cur_sc = tmp_sc
                last_improve_it = it

        # Periodic jitter towards best to explore neighbourhood
        if it % 4000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best = tmp.copy()
                best_sc = tmp_sc
                a = tmp
                cur_sc = tmp_sc
                last_improve_it = it

        # Early exit if target reached
        if best_sc < 1.50274349:
            break

    # final local refinement: small pairwise transfers
    # attempt limited steps
    max_local = 2000
    for _ in range(max_local):
        if time.monotonic() >= deadline:
            break
        # choose random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if best[i] == 0:

        delta = best[i] * rng.random() * 0.25
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best = cand
            best_sc = cand_sc

    return [float(x) for x in best]

import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT (real valued)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Objective (same formula as evaluate_sequence for sum==1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))
