import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT (real-valued)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int = None) -> np.ndarray:
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Score used by the official evaluator (scale‑invariant)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_full(a)
    maxc = conv.max()
    s = a.sum()
    if s <= 0:
        return float("inf")
    return 2.0 * n * maxc / (s * s)

# ------------------------------------------------------------

        n = parent.size
    else:
        parent = None
        n = 256

    # ------------------------------------------------------------------
    # Helper generators for diverse initial candidates (all on the simplex)
    # ------------------------------------------------------------------
    def _from_parent():
        if parent is None:
            return None
        a = np.clip(parent, 0.0, None)
        s = a.sum()
        return a / s if s > 0 else np.full_like(a, 1.0 / n)

    def _uniform(_):
        return np.full(n, 1.0 / n)

    def _cosine(_):

        return a / a.sum()

    def _semicircle(_):

        a = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
        return a / a.sum()

            a[k] = (1.0 - s) / 2.0

        return a / a.sum()

    def _random(_):

    generators = [_from_parent, _uniform, _cosine,

    # ------------------------------------------------------------------

        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Final fallback (in case the loop terminated early)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
