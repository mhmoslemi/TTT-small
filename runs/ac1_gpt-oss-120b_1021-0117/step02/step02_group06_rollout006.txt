def simplex_proj(v):
    # from Duchi et al.
    if v.sum() == 1 and (v >= 0).all():
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]

import time
import numpy as np

def _conv_max(a, fft_len, n):
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    conv = np.fft.irfft(B, n=fft_len)
    # conv length fft_len, but only first 2*n-1 are valid (others are zero due to zero padding)
    return float(conv[:2 * n - 1].max())

def _project_simplex(v):
    # Euclidean projection onto simplex (nonnegative sum=1)
    # Works for positive entries after subtraction
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _catalan_weights(n):
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
    return w

def _initial_candidates(n, parent_arr):
    cand = []
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0:
            cand.append(p / s)
    # Catalan shape
    w = _catalan_weights(n)
    cand.append(w / w.sum())
    # Symmetric power shape: ((i+1)*(n-i))**p
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for pwr in (0.5, 1.0, 1.5, 2.0, 3.0):
        v = base ** pwr
        cand.append(v / v.sum())
    # Sin-power shape
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for pwr in (0.5, 1.0, 2.0):
        v = sin_vals ** pwr
        cand.append(v / v.sum())
    # Exponential decay shapes
    for r in (0.99, 0.95, 0.9, 0.85):
        v = r ** idx
        cand.append(v / v.sum())
    # Linear tent
    v = np.concatenate([np.arange(1, n // 2 + 1), np.arange(n // 2, 0, -1)], dtype=np.float64)
    cand.append(v / v.sum())
    return cand

        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256

    fft_len = 1 << ((2 * n - 2).bit_length())  # next power of two >= 2n-1

    # ---- initial candidates ----
    best_vec = None
    best_score = np.inf

    for vec in _initial_candidates(n, parent_arr):
        # Ensure simplex
        vec = _project_simplex(vec)
        max_b = _conv_max(vec, fft_len, n)
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_vec = vec.copy()
        if best_score <= 1.50274349:
            return best_vec.tolist()

    # ---- subgradient descent (as in previous solution) ----
    a = best_vec.copy()
    iter_no_improve = 0
    alpha0 = 0.9
    min_alpha = 1e-4

    while time.monotonic() < deadline:
        conv = _conv_max(a, fft_len, n)  # actually returns max, not full conv
        # But we need index of max to compute gradient; need full conv for index.
        # Compute full conv for gradient.
        A = np.fft.rfft(a, n=fft_len)
        B = A * A
        conv_full = np.fft.irfft(B, n=fft_len)[:2 * n - 1]
        max_b = float(conv_full.max())
        k = int(conv_full.argmax())
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_improve = 0
            if best_score <= 1.50274349:

            cand = cand * jitter

import time
import numpy as np

def _conv_max(a, fft_len, n):
    """Maximum value of a*a via real FFT (full convolution length 2n‑1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    conv = np.fft.irfft(B, n=fft_len)
    return float(conv[: 2 * n - 1].max())

def _project_simplex(v):
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _catalan_weights(n):
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w

def _initial_candidates(n, parent_arr):
    cand = []
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0:
            cand.append(p / s)

    # Catalan shape
    w = _catalan_weights(n)
    cand.append(w / w.sum())

    # Symmetric power shape ((i+1)*(n-i))**p
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for pwr in (0.5, 1.0, 1.5, 2.0, 3.0):
        v = base ** pwr
        cand.append(v / v.sum())

    # Sin‑power shapes
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for pwr in (0.5, 1.0, 2.0):
        v = sin_vals ** pwr
        cand.append(v / v.sum())

    # Exponential decays
    for r in (0.99, 0.95, 0.9, 0.85):
        v = r ** idx
        cand.append(v / v.sum())

    # Linear tent
    half = n // 2
    v = np.concatenate([np.arange(1, half + 1),
                        np.arange(half, 0, -1)], dtype=np.float64)
    if v.size == n:
        cand.append(v / v.sum())

    return cand
