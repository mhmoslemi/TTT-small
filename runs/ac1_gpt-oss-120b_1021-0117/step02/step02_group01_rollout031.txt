import numpy as np

def _sqrt_series(n: int) -> np.ndarray:
    """
    Build the non‑negative sequence a of length n satisfying
    (a * a)[k] = 1 for all k = 0,…,n‑1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # s = sum_{i=1}^{k-1} a[i] * a[k-i]
            s = np.dot(a[1:k], a[k-1:0:-1])
        a[k] = (1.0 - s) / 2.0
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step‑function (list of non‑negative floats) that yields a very
    low AC1 upper bound.  The construction is the analytic sqrt‑series of
    the required length (the length of the provided parent if present).
    """
    # Determine the target length: use the parent's length when available.
    if 'height_sequence_1' in globals():
        n = int(len(height_sequence_1))
    else:
        # fallback length – any positive length works; 256 is modest.
        n = 256

    # Build the analytic sqrt‑series.  It guarantees max(convolution) = 1.
    seq = _sqrt_series(n)

    # Convert to plain Python list of floats as required by the evaluator.
    return [float(x) for x in seq]
