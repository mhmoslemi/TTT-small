import numpy as np, time

def _initial_binomial(n):
   a = np.empty(n, dtype=np.float64)
   a[0] = 1.0
   for i in range(1,n):
      a[i] = a[i-1] * (2*i-1)/(2*i)
   return a

def _score_and_conv(seq, fft_len):
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f*f, n=fft_len)[:2*len(seq)-1]
    maxb = conv.max()
    s = seq.sum()
    if s <= 0:
        return np.inf, conv
    score = 2.0 * len(seq) * maxb / (s * s)
    return float(score), conv

import time

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_and_conv(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution array)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    max_b = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return np.inf, conv
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv
