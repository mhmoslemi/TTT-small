def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.3)

    # initial vector a0

    else:

        a[0] = 1.0
        for k in range(1, n):
            if k == 1:
                s = 0.0
            else:
                s = float(np.dot(a[1:k], a[k-1:0:-1]))
            a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    best = a.copy()
    best_sc = _score(best)

    # Adam params
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05

    # smoothing param
    beta_smooth = 5.0
    beta_max = 40.0
    beta_inc = 1.005 # each iteration multiply?

    it = 0

    while time.monotonic() < deadline:
        it += 1
        # compute full convolution via FFT
        L = 1 << ((2 * n - 1).bit_length())
        # compute FFT of a
        Fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
        # compute loss for early exit maybe
        # compute softmax weights
        # To avoid overflow, shift
        w_exp = np.exp(beta_smooth * conv - np.max(beta_smooth * conv))
        w = w_exp / w_exp.sum()
        # compute correlation gradient using FFT
        # compute FFT of w
        Fw = np.fft.rfft(w, n=L)
        # correlation = irfft(Fw * np.conj(Fa), n=L)[:len(w)+len(a)-1]
        corr_full = np.fft.irfft(Fw * np.conj(Fa), n=L)
        # slice central part
        start_idx = len(w) - 1
        grad = 2.0 * corr_full[start_idx : start_idx + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a + step
        # enforce nonnegative
        cand = np.maximum(cand, 0.0)
        # project to simplex
        cand = _project_simplex(cand)

        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            # maybe reset Adam states?
            # but keep.

        # optionally accept improvement for a (like cur)
        # We'll set a = cand if cand_sc <= best_sc or random acceptance.
        # Let's use simple: if cand_sc <= best_sc: a=cand else occasional acceptance.
        if cand_sc <= best_sc or rng.random() < 0.05:
            a = cand

        # random mass-transfer
        if rng.random() < 0.01:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n-1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.3
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best = tmp.copy()
                    best_sc = tmp_sc
                if tmp_sc < _score(a):
                    a = tmp

        # increase smoothing parameter gradually up to max
        if beta_smooth < beta_max:
            beta_smooth = min(beta_max, beta_smooth * 1.005)

        # early exit
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]

import time
import numpy as np

# ------------------------------------------------------------
# Simple projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution via FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using a pre‑chosen FFT length L."""
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# Objective (same as evaluate_sequence when sum==1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.size
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function vector with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.4)   # small safety margin

    # ------------------------------------------------------------
    # initialise a – either the supplied parent or a simple analytic start
    # ------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a very short fallback (sqrt‑series) – will never be used in the official run
        n = 256
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            if k == 1:
                s = 0.0
            else:
                s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
            a[k] = (1.0 - s) / 2.0

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # FFT length for all convolutions

    # ------------------------------------------------------------
    # bookkeeping of the best solution seen
    # ------------------------------------------------------------
    best_vec = a.copy()
    best_sc = _score(best_vec, L)

    # ------------------------------------------------------------
    # optimisation state (adam‑style moments)
    # ------------------------------------------------------------
    m = np.zeros_like(a)        # first moment
    v = np.zeros_like(a)        # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.08                    # base learning rate
    it = 0

    # smoothing parameter for the soft‑max approximation of the max‑conv
    beta_s = 5.0                 # start low (smooth)
    beta_target = 40.0
    beta_inc = 1.003            # very gentle increase per iteration

    # ------------------------------------------------------------
    # main loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ------- current convolution & loss -------
        Fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]

        cur_sc = 2.0 * n * conv.max() / (a.sum() ** 2)

        # ------- soft‑max weights for a smooth loss -------
        tmp = beta_s * conv
        tmp -= tmp.max()                       # stabilise exp
        w = np.exp(tmp)
        w /= w.sum()                           # now sum_i w_i = 1

        # ------- gradient of the smooth loss -------
        # correlation r[k] = Σ_i w_i * a_{i-k}; we need k = 0..n-1
        Fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(Fw * np.conj(Fa), n=L)   # length L
        start_idx = len(w) - 1
        grad = 2.0 * corr_full[start_idx : start_idx + n]

        # ------- adam update -------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a + step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        # evaluate candidate (full convolution, exact score)
        cand_sc = _score(cand, L)

        # keep the best ever seen
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

        # accept the candidate if it improves or with a tiny probability
        if cand_sc < cur_sc or rng.random() < 0.02:
            a = cand

        # ------- occasional random mass‑transfer move (exploration) -------
        if rng.random() < 0.015:                 # ≈1.5 % per iteration
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.35
                tmp_vec = a.copy()
                tmp_vec[i] -= delta
                tmp_vec[j] += delta
                tmp_vec = _project_simplex(tmp_vec)
                tmp_sc = _score(tmp_vec, L)
                if tmp_sc < best_sc:

                if tmp_sc < cur_sc:
                    a = tmp_vec

        # ------- jittered restart towards the best solution -------
        if it % 4000 == 0 and rng.random() < 0.22:

            tmp_vec = best_vec + jitter
            tmp_vec = np.maximum(tmp_vec, 0.0)
            tmp_vec = _project_simplex(tmp_vec)
            tmp_sc = _score(tmp_vec, L)
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp_vec.copy()
            if tmp_sc < cur_sc:
                a = tmp_vec

        # ------- anneal smoothing parameter -------
        if beta_s < beta_target:
            beta_s = min(beta_target, beta_s * beta_inc)

        # early exit when we have already beaten the target
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------
    # return the best vector discovered (plain Python list)
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
