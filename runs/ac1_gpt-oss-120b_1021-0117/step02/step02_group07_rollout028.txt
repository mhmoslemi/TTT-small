# Helper functions – must be top‑level (no closures)

def _initial_binomial(n: int) -> np.ndarray:
    """Decreasing binomial seed a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_seq(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution via FFT, length 2·len(seq)-1."""
    f = np.fft.rfft(seq, n=fft_len)
    return np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
