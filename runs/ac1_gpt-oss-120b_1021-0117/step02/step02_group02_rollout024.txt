import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # shift by the mean and sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation (exponentiated‑gradient + jitter) for the
    AC1 upper bound.  Works with a supplied parent ``height_sequence_1`` or
    falls back to a uniform start.  Returns the best sequence found before
    the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. time budget
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. base sequence
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = None

    if parent is not None:
        a = np.clip(parent, 0.0, 1000.0)
        n = a.size
        if a.sum() < 0.01:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a.sum()                     # simplex (sum = 1)
    else:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3. keep the best seen so far
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # a cheap uniform alternative (might be better if parent is poor)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score = float(evaluate_sequence(uniform.tolist()))
    if uni_score < best_score:
        best, best_score = uniform.copy(), uni_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 4. pre‑compute FFT layout (linear convolution via zero‑padded FFT)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 5. optimisation loop (exponentiated‑gradient with occasional jitter)
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.8                     # initial stepsize for EG
    jitter_period = 70
    jitter_scale = 0.004          # std‑dev of log‑normal jitter

    while time.monotonic() < deadline:
        it += 1

        # ---- convolution ------------------------------------------------
        Fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]

        # ---- current objective -------------------------------------------
        kmax = int(np.argmax(conv))
        max_conv = conv[kmax]
        cur_score = 2.0 * n * max_conv          # sum(a) == 1 → denominator = 1

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            # early success check – 2 is the theoretical lower bound
            if best_score <= 2.0:
                break

        # ---- sub‑gradient of the active autocorrelation entry ------------
        offset = kmax - (n - 1)                  # signed shift relative to centre
        g = np.zeros_like(a)

        if offset >= 0:
            i = np.arange(0, n - offset)
            j = i + offset
        else:
            d = -offset
            i = np.arange(0, n - d)
            j = i + d

        # each product a[i]*a[j] contributes to the gradient of both indices
        g[i] += a[j]
        g[j] += a[i]

        # ---- exponentiated‑gradient step ---------------------------------
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)                 # stay non‑negative
        a_sum = a.sum()
        if a_sum == 0.0:                         # avoid division by zero
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum                         # re‑normalise → simplex

        # ---- occasional multiplicative jitter to escape plateaus ---------
        if it % jitter_period == 0:
            jitter = np.exp(rng.normal(0.0, jitter_scale, size=n))
            a = a * jitter
            a = a / a.sum()                       # renormalise

    # ------------------------------------------------------------------
    # 6. final return (plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
