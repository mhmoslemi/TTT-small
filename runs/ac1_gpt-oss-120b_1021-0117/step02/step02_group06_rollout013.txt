def evaluate_sequence(sequence: list) -> float:
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

import time
import numpy as np

# ---------- utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b using real FFT (no extra memory)."""
    n = a.shape[0] + b.shape[0] - 1
    fft_len = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    B = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)
    return conv[:n]


def _self_conv(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Convolution a*a, full length 2*len(a)-1, using pre‑computed FFT length."""
    A = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(A * A, n=fft_len_self)
    return conv[: 2 * a.shape[0] - 1]


def _cross_corr(a_rev: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Cross‑correlation a ⋆ w = sum_j a_j w_{i+j} via convolution a_rev * w."""
    A = np.fft.rfft(a_rev, n=fft_len_grad)
    W = np.fft.rfft(w, n=fft_len_grad)
    conv = np.fft.irfft(A * W, n=fft_len_grad)
    total_len = a_rev.shape[0] + w.shape[0] - 1  # = 3n-2
    return conv[:total_len]


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _initial_vectors(n: int, parent_arr) -> list:
    """Create a handful of plausible start vectors (already summed to 1)."""
    vecs = []

    # parent if present
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            vecs.append(p / s)

    # uniform
    vecs.append(np.full(n, 1.0 / n))

    # sine window
    i = np.arange(n, dtype=np.float64)
    sin_w = np.sin(np.pi * (i + 0.5) / n)
    vecs.append(sin_w / sin_w.sum())

    # arcsine‑type (Beta(0.5,0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    vecs.append(arcs / arcs.sum())

    # Catalan shape
    cat = _catalan_weights(n)
    vecs.append(cat / cat.sum())

    # simple power‑law base = (i+1)*(n-i)
    base = (i + 1) * (n - i)
    for p in (0.5, 1.0, 1.5, 2.0):
        v = base ** p
        vecs.append(v / v.sum())

    return vecs


# ---------- main routine ----------

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises the
    AC1 upper bound used by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # safety margin

    # ---- retrieve optional parent ----
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256

    # ---- pre‑compute FFT lengths ----
    conv_len = 2 * n - 1
    fft_len_self = 1 << ((conv_len - 1).bit_length())
    grad_len = n + conv_len - 1                     # = 3n-2
    fft_len_grad = 1 << ((grad_len - 1).bit_length())

    # ---- pick a good starting vector from a small pool ----
    cand_vecs = _initial_vectors(n, parent_arr)
    def _score_of(x):
        b = _self_conv(x, fft_len_self)
        return 2.0 * n * b.max()                 # same as evaluate_sequence when sum(x)=1

    best_a = None
    best_score = np.inf
    for v in cand_vecs:
        sc = _score_of(v)
        if sc < best_score:
            best_score = sc
            best_a = v.copy()
    if best_a is None:                # fallback (should never happen)
        best_a = np.full(n, 1.0 / n)

    # early exit if already excellent
    TARGET = 1.50274349
    if best_score <= TARGET:
        return best_a.tolist()

    # ---- optimiser state ----
    a = best_a.copy()
    momentum = np.zeros_like(a)

    beta = 5.0                 # start with a smooth approximation of max
    beta_max = 80.0
    beta_step = 5.0

    step_size = 0.15
    min_step = 5e-4

    no_improve = 0
    iter_cnt = 0

    while time.monotonic() < deadline:
        # increase smoothing temperature slowly
        if iter_cnt % 250 == 0 and beta < beta_max:
            beta = min(beta + beta_step, beta_max)

        # ----- compute convolution and smooth weights -----
        b = _self_conv(a, fft_len_self)          # length 2n-1
        max_b = b.max()
        # softmax over b (shifted for numeric stability)
        shift = b - max_b
        exp_shift = np.exp(beta * shift)
        w = exp_shift / exp_shift.sum()          # length 2n-1, sum=1

        # ----- gradient of the smoothed objective -----
        a_rev = a[::-1]
        corr = _cross_corr(a_rev, w, fft_len_grad)   # length 3n-2
        grad = 2.0 * corr[(n - 1):(n - 1 + n)]       # slice to length n

        # ----- Nesterov‑type update with projection -----
        momentum = 0.9 * momentum - step_size * grad
        cand = a + momentum
        cand = _project_onto_simplex(cand)

        # occasional random perturbation to escape flat regions
        if rng.random() < 0.012:
            cand = cand * np.exp(rng.normal(0.0, 0.012, size=n))
            cand = _project_onto_simplex(cand)

        cand_score = _score_of(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            no_improve = 0
            # if we beat target we can stop early
            if best_score <= TARGET:
                break
        else:
            no_improve += 1

        # adapt step size on stagnation
        if no_improve > 180:
            step_size = max(min_step, step_size * 0.5)
            momentum.fill(0.0)
            a = best_a.copy()
            no_improve = 0
        else:
            a = cand

        iter_cnt += 1

    # final safety projection (sum may deviate by tiny eps)
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = best_a / best_a.sum()
    return best_a.tolist()
