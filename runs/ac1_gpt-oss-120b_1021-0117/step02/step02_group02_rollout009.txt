import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimizer for the AC1 upper bound.
    Uses exponentiated mirror descent with momentum, occasional jitter,
    and random restarts / sampling.  Returns the best non‑negative
    step‑function (list of Python floats) found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 2.  Base sequence (parent if available, else uniform)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:
        parent = None

    if parent is not None:
        a = np.clip(parent, 0.0, 1000.0)
    else:
        a = np.ones(256, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:                                   # degenerate parent
        a = np.full(a.shape[0], 1.0 / a.shape[0], dtype=np.float64)
    else:
        a = a / s                                   # normalize to sum = 1

    n = a.size
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power‑of‑two

    # ------------------------------------------------------------------
    # 3.  Initialise best candidate
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))  # exact score

    # cheap uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 4.  Optimisation loop
    # ------------------------------------------------------------------
    it = 0
    eta0 = 1.0          # initial learning rate
    beta = 0.9          # momentum coefficient
    momentum = np.zeros_like(a)

    last_improve = 0
    restart_interval = 5000
    target_score = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution (FFT) ------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # ----- current objective -------------------------------------------------
        kmax = int(np.argmax(conv))
        max_conv = conv[kmax]
        cur_score = 2.0 * n * max_conv                # = evaluate_sequence for sum=1

        # ----- record improvement -----------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            last_improve = it
            if best_score <= target_score:
                break

        # ----- sub‑gradient of the active convolution term -----------------------
        offset = kmax - (n - 1)                       # shift relative to centre
        g = np.zeros_like(a)
        if offset >= 0:
            idx = np.arange(offset, n)
            partner = kmax - idx
            g[idx] = 2.0 * a[partner]
        else:
            idx = np.arange(0, n + offset)           # offset is negative
            partner = kmax - idx
            g[idx] = 2.0 * a[partner]

        # ----- momentum update ---------------------------------------------------
        eta = eta0 / np.sqrt(it)
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- exponentiated mirror‑descent step -------------------------------
        a = a * np.exp(-eta * momentum)

        # ----- renormalise to the simplex ---------------------------------------
        total = a.sum()
        if total <= 0.0 or not np.isfinite(total):
            a = uniform.copy()
        else:
            a = a / total

        # ----- occasional multiplicative jitter ---------------------------------
        if it % 200 == 0:
            jitter = np.exp(rng.normal(0.0, 0.004, size=n))
            a *= jitter
            a /= a.sum()

        # ----- random restart if stagnated --------------------------------------
        if it - last_improve > restart_interval:
            a = rng.random(n)
            a /= a.sum()
            momentum.fill(0.0)
            last_improve = it

    # ------------------------------------------------------------------
    # 5.  Final random sampling (if time permits)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = rng.random(n)
        cand /= cand.sum()
        fa = np.fft.rfft(cand, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score:
            best = cand.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

    # ------------------------------------------------------------------
    # 6.  Return best sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
