    L = 1 << ((2 * n - 1).bit_length())

def _grad_weighted(a, w):
    n = a.shape[0]
    L2 = 1 << ((n + len(w) - 1).bit_length())
    fv_a = np.fft.rfft(a, n=L2)
    fv_w_rev = np.fft.rfft(w[::-1], n=L2)
    conv_rev = np.fft.irfft(fv_a * fv_w_rev, n=L2)[: n + len(w) - 1]
    shift = len(w) - 1
    grad = 2.0 * conv_rev[shift - np.arange(n)]
    return grad

import time

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast linear (full) convolution via FFT
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

# ------------------------------------------------------------
def _grad_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of  ∑ w_k (a*a)_k  w.r.t. a (w sums to 1)."""
    n = a.shape[0]
    L2 = 1 << ((n + len(w) - 1).bit_length())
    fv_a = np.fft.rfft(a, n=L2)

    rev = np.fft.irfft(fv_a * fv_w, n=L2)[: n + len(w) - 1]
    shift = len(w) - 1
    grad = 2.0 * rev[shift - np.arange(n)]
    return grad
