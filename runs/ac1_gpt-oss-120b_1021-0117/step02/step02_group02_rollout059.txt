def correlation_w_a(w, a):
    # w shape (2n-1), a shape (n)
    # compute correlation w ⋆ a (valid length n)
    rev_a = a[::-1]
    size = 1 << ((len(w) + len(rev_a) - 1 - 1).bit_length())
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(rev_a, n=size)
    conv_full = np.fft.irfft(fw * fa_rev, n=size)
    start = len(a) - 1
    return conv_full[start:start+len(a)]

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 bound.
    Uses a soft‑max weighted gradient derived from the full convolution,
    applied with an exponentiated‑gradient (mirror descent) update on the simplex.
    Returns the best coefficient list found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. Base sequence (use provided parent if possible)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:
        parent = None

    if parent is not None:
        a = parent.copy()
    else:
        # default small uniform vector
        n = 256
        a = np.ones(n, dtype=np.float64)

    # ensure feasibility and normalise to sum = 1 (simplex)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.full_like(a, 1.0 / a.size, dtype=np.float64)
    else:
        a = a / s

    n = int(a.size)

    # ------------------------------------------------------------------
    # 3. Pre‑computed FFT lengths (matrix‑free, linear memory)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)          # for a*a
    corr_len = conv_len + n - 1                  # len(w)+len(a_rev)-1 = 3n-2
    fft_len_corr = _next_pow2(corr_len)         # for w * rev(a)

    # ------------------------------------------------------------------
    # 4. Initialise best candidate
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # also keep uniform as a cheap alternative
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349
    eps = 1e-12                       # keep positivity

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.6                        # initial learning rate for EG
    beta_max = 12.0                  # max soft‑max sharpness
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution a * a (FFT) ---------------------------------
        fa = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(fa * fa, n=fft_len_conv)[:conv_len]

        max_c = conv_full.max()
        cur_score = 2.0 * n * max_c      # = evaluate_sequence because sum(a)=1
        if cur_score < best_score:

            best_score = cur_score
            if best_score <= target_score:
                break

        # ----- soft‑max weights over convolution entries -----------------
        # beta grows linearly until beta_max is hit
        beta = min(beta_max, 0.01 * it)
        # stable softmax
        exp_vals = np.exp(beta * (conv_full - max_c))
        w = exp_vals / exp_vals.sum()

        # ----- gradient = 2 * (w ⋆ a)  (correlation via FFT) ----------
        rev_a = a[::-1]
        fw = np.fft.rfft(w, n=fft_len_corr)
