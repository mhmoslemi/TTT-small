import time
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    # Implements the algorithm of Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, u.size + 1)) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / v.size)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises the
    AC1 upper bound as evaluated by ``evaluate_sequence``.
    The algorithm is deliberately simple: it starts from a uniform vector,
    optionally improves it by sampling random points on the simplex, and keeps
    the best candidate seen before the time limit expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine length (use the provided parent if available)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        # fallback length
        n = 256

    # ------------------------------------------------------------------
    # 2. Initial candidates: uniform and (if present) the parent sequence
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # normalise the parent (clip before normalising, just as the reference
    # implementation does)
    try:
        parent = np.clip(parent, 0.0, 1000.0)
        s = parent.sum()
        if s < 0.01:
            parent = uniform.copy()
        else:
            parent = parent / s
    except Exception:
        parent = uniform.copy()

    # evaluate both
    best = uniform.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    parent_score = float(evaluate_sequence(parent.tolist()))
    if parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # ------------------------------------------------------------------
    # 3. Random simplex sampling (very cheap, bounded by the time budget)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    target = 1.50274349

    while time.monotonic() < deadline:
        # Draw a random point on the simplex via exponential variables
        x = rng.exponential(scale=1.0, size=n)
        x_sum = x.sum()
        if x_sum == 0.0:
            continue
        x = x / x_sum

        # Occasionally project to mitigate numerical drift
        x = _simplex_projection(x)

        score = float(evaluate_sequence(x.tolist()))
        if score < best_score:
            best = x.copy()
            best_score = score
            if best_score <= target:
                break

    # ------------------------------------------------------------------
    # 4. Return the best candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
