import time
import numpy as np

# ------------------------------------------------------------
# Simple projection onto the non‑negative simplex (sum = 1)
# ------------------------------------------------------------
def _simplex_normalize(v: np.ndarray) -> np.ndarray:
    """Clip negatives to zero and renormalise to sum‑one."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v.fill(1.0 / v.size)
    else:
        v /= s
    return v


# ------------------------------------------------------------
# Linear convolution via real FFT (full mode)
# ------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    L = 1 << ((n - 1).bit_length())     # next power of two
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:n]
    return conv


# ------------------------------------------------------------
# Score for a vector that already satisfies sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(a * a)  (sum(a) == 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()

# ------------------------------------------------------------
