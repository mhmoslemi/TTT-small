import time, math, numpy as np

def _project_onto_simplex(v):
    # as defined earlier
    # returns projected vector onto simplex sum=1, nonnegative.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _fft_conv(x, y, fft_len):
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    Z = X * Y
    out = np.fft.irfft(Z, n=fft_len)
    return out[:len(x) + len(y) - 1]

def _conv_self(a, fft_len):
    return _fft_conv(a, a, fft_len)

def _initial_candidates(n, parent):
    cands = []
    if parent is not None and parent.shape[0] == n:
        p = np.clip(parent, 0, 1000)
        cands.append(p)   # we'll normalize later
    # uniform
    cands.append(np.ones(n))
    # power shape
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for p in (0.5, 1.0, 1.5, 2.0):
        cands.append(base**p)
    # sin power
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        cands.append(sin_vals**p)
    # half cosine
    center = (n - 1) / 2.0
    cos_shape = np.cos(np.pi * (idx - center) / (n - 1))
    cos_shape = np.maximum(cos_shape, 0.0)
    cands.append(cos_shape)
    # gaussian
    sigma = n / 5.0
    gauss = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    cands.append(gauss)
    # random Dirichlet
    rng = np.random.default_rng(12345)
    cands.append(rng.dirichlet(np.ones(n)))
    return cands

# ---------- linear‑algebra helpers ----------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – uniform distribution
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_conv(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution via FFT (real input)."""
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    Z = X * Y
    out = np.fft.irfft(Z, n=fft_len)
    return out[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a."""
    return _fft_conv(a, a, fft_len)
