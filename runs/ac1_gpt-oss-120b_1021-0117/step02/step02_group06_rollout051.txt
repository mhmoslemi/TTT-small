def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # allocate small margin
    rng = np.random.default_rng(seed)

    # Determine n
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception
    except Exception:
        parent_arr = None
    n = parent_arr.shape[0] if parent_arr is not None else 256

    # FFT length
    fft_len = 1 << ((2*n - 1).bit_length())

    def conv_self(a):
        A = np.fft.rfft(a, n=fft_len)
        B = A * A
        full = np.fft.irfft(B, n=fft_len)
        return full[: 2*n - 1]

    def score_from_conv(conv):
        return 2.0 * n * float(conv.max())

    # projection onto simplex sum=1
    def proj_simplex(v):
        # Implementation like earlier
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
        if rho.size == 0:
            # all nonpositive
            return np.full_like(v, 1.0 / len(v))
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # initial candidates: list of vectors (sum normalized to 1)
    cand_vecs = []

    # parent
    if parent_arr is not None:
        v = np.clip(parent_arr, 0.0, 1000.0)
        s = v.sum()
        if s>0:
            cand_vecs.append(v / s)

    # deterministic shapes: Catalan, power, sin, triangular
    # predefine functions to compute shape given n
    def catalan_weights(nn):
        w = np.empty(nn, dtype=np.float64)
        w[0] = 1.0
        for i in range(1, nn):
            w[i] = w[i-1] * ((2*i)*(2*i-1))/(i*i*4.0)
        return w

    w = catalan_weights(n)
    cand_vecs.append(w / w.sum())

    idx = np.arange(n, dtype=np.float64)
    base = (idx+1)*(n-idx)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        v = base**p
        cand_vecs.append(v / v.sum())

    theta = np.pi*(idx+0.5)/n
    sin_vals = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sin_vals**p
        cand_vecs.append(v / v.sum())

    # triangular shape
    tri = np.minimum(idx+1, n-idx)
    cand_vecs.append(tri / tri.sum())

    # random dirichlet initializations
    for _ in range(5):
        # concentration parameter can be <1 for spiky
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
        cand_vecs.append(a)

    best_vec = None
    best_score = np.inf
    # evaluate initial candidates via conv (not using evaluator)
    for vec in cand_vecs:
        conv = conv_self(vec)
        sc = score_from_conv(conv)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
        if best_score <= 1.50274349:
            return best_vec.tolist()

    # core optimization loop
    a = best_vec.copy()
    # momentum
    v = np.zeros_like(a)
    beta = 0.9
    iter_no_improve = 0
    t0 = time.monotonic()
    # parameters
    step0 = 0.5
    min_alpha = 1e-5
    # main loop
    while time.monotonic() < deadline:
        conv = conv_self(a)
        max_b = float(conv.max())
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            iter_no_improve += 1

        # find argmax index
        k = int(conv.argmax())
        # compute gradient g
        g = np.zeros_like(a)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        if i_start <= i_end:
            idxs = np.arange(i_start, i_end+1, dtype=np.int64)
            g[idxs] = 2.0 * a[k - idxs]

        # adaptive step size
        # Using simple schedule: step = step0 / np.sqrt(iter_no_improve+1)
        step = max(min_alpha, step0 / np.sqrt(iter_no_improve+1))

        # momentum update
        v = beta * v - step * g
        cand = proj_simplex(a + v)

        # Evaluate candidate
        conv_cand = conv_self(cand)
        cand_score = 2.0 * n * float(conv_cand.max())

        # Accept if improvement else maybe revert momentum (discard update)
        if cand_score < cur_score:
            a = cand
            # keep momentum as is (not resetting)
        else:
            # reject: keep a, but decay momentum? set v to zeros
            v = np.zeros_like(v)
            # occasional random jitter
            if rng.random() < 0.02:
                jitter = np.exp(rng.normal(0.0, 0.02, size=n))
                a = proj_simplex(a * jitter)

        # occasional random restart if stuck
        if iter_no_improve > 2000:
            # restart from best with perturbation
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            a = proj_simplex(best_vec * perturb)
            v = np.zeros_like(v)
            iter_no_improve = 0

    return best_vec.tolist()

import time
import numpy as np

# ---------- helper utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT.
    Returns a length 2*len(a)-1 array (the “full” convolution).
    """
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * a.shape[0] - 1]


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """Generate a list of plausible start vectors (already L1‑normalised)."""
    cand = []

    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # Catalan style
    w = _catalan_weights(n)
    cand.append(w / w.sum())

    # power‑law hills
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        v = base ** p
        cand.append(v / v.sum())

    # sin‑power hills
    theta = np.pi * (idx + 0.5) / n
    sv = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sv ** p
        cand.append(v / v.sum())

    # triangular shape
    tri = np.minimum(idx + 1, n - idx)
    cand.append(tri / tri.sum())

    # a few random Dirichlet draws (spiky)
    rng = np.random.default_rng(12345)
    for _ in range(5):
        a = rng.gamma(0.5, 1.0, size=n)
        cand.append(a / a.sum())

    return cand
