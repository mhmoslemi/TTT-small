def move_random_transfer(a, transfer_max):

    cand[i] -= delta
    cand[j] += delta
    return cand

def _fast_ac1_score(a, L):
    n = a.shape[0]
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]

    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan-type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _fast_ac1_score(a: np.ndarray, L: int) -> float:
    """Fast evaluation of AC1 bound via FFT (same formula as the evaluator)."""
    n = a.shape[0]
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic optimisation of a non‑negative step function for the AC1 bound.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------------------
    # initialise sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        cur = parent.copy()
    except Exception:
        n = 256
        cur = _gen_binomial(n)

    L = _next_pow2(2 * n - 1)

    # ensure a tiny positive mass to avoid division by zero
    cur = np.clip(cur, 1e-12, 1000.0)

    best = cur.copy()
    best_score = _fast_ac1_score(best, L)
    cur_score = best_score

    # ----------------------------------------------------------------------
    # hyper‑parameters
    temperature = 2.0
    decay = 0.99995
    grad_eps = 0.03
    min_block = max(2, n // 250)
    max_block = max(2, n // 20)
    transfer_max = 0.5          # max fraction moved in random transfer
    scale_sigma = 0.20          # std. dev. for block scaling factor
    restart_prob = 0.001
    target_score = 1.50274349

    # main optimisation loop
    while time.monotonic() < deadline:
        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) gradient guided mass transfer (≈40%)
        if move_type < 0.40:
            # convolution of current vector (needed for kmax)
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)

            # exact sub‑gradient of the max entry:
            # derivative = a_{kmax-i} + a_i (if both terms exist)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]] + cur[mask]   # adds both contributions; self‑pair gives 2*a_i

            # direction that preserves total mass (sum(d)=0) and tends to reduce the max
            d = g.mean() - g
            cand = cur + grad_eps * d

        # --------------------------------------------------------------
        # 2) block scaling (≈25%)
        elif move_type < 0.65:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, scale_sigma))
            cand = cur.copy()
            cand[start:start + width] *= factor

        # --------------------------------------------------------------
        # 3) random mass transfer (≈15%)
        elif move_type < 0.80:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * transfer_max
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # --------------------------------------------------------------
        # 4) swap two entries (≈10%)
        elif move_type < 0.90:
            i, j = rng.choice(n, size=2, replace=False)
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # --------------------------------------------------------------

        elif move_type < 0.98:

            smooth = np.convolve(cand if 'cand' in locals() else cur,

            cand = (1.0 - alpha) * (cand if 'cand' in locals() else cur) + alpha * smooth

        # --------------------------------------------------------------

        np.clip(cand, 0.0, 1000.0, out=cand)

        if accept:

        # occasional restart from best
        if rng.random() < restart_prob:
            cur = best.copy()
            cur_score = best_score
            temperature = 2.0

        # adapt step size a little – increase if many acceptances, decrease otherwise

        if move_type < 0.40 and accept:

            grad_eps = max(0.005, grad_eps * 0.99)

    for _ in range(2000):

        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        kmax = int(np.argmax(conv))
        idx = np.arange(n)
        rev = kmax - idx
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(cur)
        g[mask] = cur[rev[mask]] + cur[mask]
        d = g.mean() - g
        cand = cur + 0.005 * d
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _fast_ac1_score(cand, L)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

    # ----------------------------------------------------------------------
    # return best candidate as plain Python floats
    return [float(x) for x in best]
