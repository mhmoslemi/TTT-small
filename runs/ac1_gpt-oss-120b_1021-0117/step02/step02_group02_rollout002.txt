import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # shift by max to improve numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # Degenerate case – return uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Uses projected (momentum) gradient descent on the simplex,
    averaging gradients over the K largest autocorrelation entries.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------

        parent = None

    if parent is not None:
        n = int(parent.size)
        a = parent.copy()
        s = a.sum()
        if s < 0.01:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s                     # normalise to sum = 1
    else:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Baselines and best‑so‑far bookkeeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # compare with uniform baseline (often 2.0, but keep for safety)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two

    # ------------------------------------------------------------------
    # 3. Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    momentum = np.zeros_like(a)
    mu = 0.85                     # momentum coefficient
    eta0 = 0.6                    # initial step magnitude
    K = 5                         # number of top autocorrelation shifts to average
    jitter_interval = 70
    jitter_sigma = 0.012
    it = 0

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        # Compute autocorrelation via FFT (full convolution)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # Current objective (sum_a == 1)
        max_idx = int(np.argmax(conv))
        cur_score = float(2.0 * n * conv[max_idx])

        # Record best
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # Determine the K largest shifts (including the absolute maximum)
        if K == 1:
            top_idxs = [max_idx]
        else:
            # argpartition is O(n), cheap for our sizes
            part = np.argpartition(conv, -K)[-K:]
            top_idxs = part.tolist()

        # Gradient = average of gradients for the selected shifts
        g = np.zeros_like(a)
        for k in top_idxs:
            offset = k - (n - 1)
            if offset >= 0:
                idx = np.arange(offset, n, dtype=int)
                partner = k - idx
                g[idx] += 2.0 * a[partner]
            else:
                idx = np.arange(0, n + offset, dtype=int)   # offset is negative
                partner = k - idx
                g[idx] += 2.0 * a[partner]
        g /= float(len(top_idxs))

        # Diminishing step size
        eta = eta0 / np.sqrt(it)

        # Momentum update (projected after the step)
        momentum = mu * momentum - eta * g
        a_new = a + momentum
        a = _proj_simplex(a_new)

        # Occasionally inject a small multiplicative jitter to escape plateaus
        if it % jitter_interval == 0:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = _proj_simplex(a * jitter)

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
