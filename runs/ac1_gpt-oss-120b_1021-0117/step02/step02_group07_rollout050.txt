        conv_cand = _conv(cand)
        maxb = conv_cand.max()
        cand_score = 2.0 * n * maxb / (cand.sum() ** 2)

        # Acceptance (simulated‑annealing style)
        elapsed = time.monotonic() - start
        total = deadline - start
        temperature = max(0.0, 0.1 * (1.0 - elapsed / total))  # linear cooling
        accept = False
        if cand_score < current_score:
            accept = True
        elif temperature > 0.0:
            prob = np.exp((current_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
