    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # initialize a using height_sequence_1 if available else uniform
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0, 1000)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            raise Exception
    except Exception:
        a = None
    if a is None:
        n = 256
        a = np.full(n, 1.0)
    else:
        n = a.size
    a = np.maximum(a, 0) + 1e-12
    a = a / a.sum()  # normalize to sum 1
    # start from initial best
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # main loops
    # compute initial temperature schedule
    temp0 = 1e-2
    temp_decay = 0.9999
    temp = temp0
    max_move = 0.05   # max fraction moved per step

    # precompute fft_len for convolution
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    while time.monotonic() < deadline:
        # propose new candidate
        # choose i, j distinct
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # amount to move: delta = epsilon * a[i] where epsilon random in [0, max_move]
        # ensure a[i] - delta >= 0
        eps = rng.random() * max_move
        delta = eps * a[i]
        if delta <= 0: continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # if a_new[i] negative (should not) skip
        # evaluate
        # Use FFT for fast convolution
        fa = np.fft.rfft(a_new, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max   # sum=1
        # acceptance criteria
        if cur_score < best_score or rng.random() < np.exp((best_score - cur_score) / max(temp, 1e-12)):
            a = a_new
            if cur_score < best_score:
                best = a_new.copy()
                best_score = cur_score
        # schedule temperature
        temp *= temp_decay
        # optional jitter
        if rng.random() < 0.001:
            jitter = np.exp(rng.normal(0, 0.001, n))
            a = np.clip(a * jitter, 0, None)
            a = a / a.sum()

def _proj_simplex(v):
    # Euclidean projection onto probability simplex sum=1, v>=0
    # following algorithm by Wang & Carreira-Perpinan
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time, math, numpy as np
from scipy import sparse
from scipy.optimize import linprog

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
