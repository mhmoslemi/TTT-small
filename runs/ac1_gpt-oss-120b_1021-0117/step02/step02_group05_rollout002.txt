    g = np.zeros_like(arr)

    g[self_mask] *= 2.0
    return g

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _fast_score(arr: np.ndarray, n: int, L: int) -> float:
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    mx = conv.max()
    s = arr.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * mx / (s * s)

            best = pl.copy()
            best_score = pl_score

    cur = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # parameters for the stochastic search
    temperature = 1.0
    temp_decay = 0.9996
    min_block = max(2, n // 200)
    max_block = max(2, n // 12)
    eps_vals = (0.04, 0.08, 0.12, 0.16)   # step sizes for gradient moves
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        move_type = rng.random()

        # -------------------------------------------------------------
        # 1) gradient‑guided mass transfer (≈45 %)
        if move_type < 0.45:
            # compute current convolution and location of max
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            self_mask = mask & (rev != idx)
            g[self_mask] *= 2.0

            mean_g = g.mean()
            d = mean_g - g   # direction preserving total mass

            # try a few step sizes and keep the best candidate this round
            cand = None
            cand_score = float('inf')
            for eps in eps_vals:
                cand_try = cur + eps * d
                # enforce non‑negativity then renormalise to original sum
                np.clip(cand_try, 0.0, None, out=cand_try)
                total = cand_try.sum()
                if total > 0:
                    cand_try *= (sum_target / total)
                sc = _fast_score(cand_try, n, L)
                if sc < cand_score:
                    cand = cand_try
                    cand_score = sc

        # -------------------------------------------------------------
        # 2) block scaling (≈35 %)
        elif move_type < 0.80:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.28))
            cand = cur.copy()
            cand[start:start + width] *= factor
            # renormalise
            np.clip(cand, 0.0, None, out=cand)
            total = cand.sum()
            if total > 0:
                cand *= (sum_target / total)
            cand_score = _fast_score(cand, n, L)

        # -------------------------------------------------------------
        # 3) random mass transfer (≈20 %)
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * 0.45
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            np.clip(cand, 0.0, None, out=cand)
            total = cand.sum()
            if total > 0:
                cand *= (sum_target / total)
            cand_score = _fast_score(cand, n, L)

        # -------------------------------------------------------------
        # accept / store best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= temp_decay

        # occasional restart from the global best
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ------------------------------------------------------------------
    return [float(x) for x in best]
