                    break
        if not conflict:
            # add i
            chosen.append(i)
            used[i+i] = True

import numpy as np
import time
import math

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length for a linear convolution of length 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution seq * seq using FFT (real signals)."""
    f = np.fft.rfft(seq, n=fft_len)
    return np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]


def _score(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution, maximal entry)."""
    s = np.sum(seq)
    if s <= 0.0:
        return math.inf, None, None
    conv = _conv_fft(seq, fft_len)
    maxb = np.max(conv)
    score = 2.0 * seq.size * maxb / (s * s)
    return score, conv, maxb


def _gradient_for_max(seq: np.ndarray, kmax: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry w.r.t. seq."""
    n = seq.size
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    j = kmax - i
    g = np.zeros_like(seq)
    g[i] += seq[j]
    g[j] += seq[i]
    return g
