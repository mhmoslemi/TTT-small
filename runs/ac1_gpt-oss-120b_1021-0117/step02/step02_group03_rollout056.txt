import time
import math
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # Fast path
    if np.all(v >= 0) and abs(v.sum() - 1.0) <= 1e-12:
        return v.copy()
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Gradient of the active lag k of the full convolution a*a
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Maximum value of the full autocorrelation a*a using FFT
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Solve a small linear program:
#   maximize   sum(b)
#   subject to a * b (convolution) <= M on a chosen active set,
#              b >= 0
# Returns the optimal b as a dense array or None on failure.
# ------------------------------------------------------------
def _solve_lp(a: np.ndarray, M: float, active: np.ndarray, L: int) -> np.ndarray | None:
    n = a.shape[0]
    m = active.shape[0]
    if m == 0:
        return None
    # Build A_ub matrix (m x n)
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        A[row, j] = a[i]
    c = -np.ones(n, dtype=np.float64)          # maximize sum(b)  <=> minimize -sum(b)
    b_ub = np.full(m, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    # 'highs' is fast and does not require external solvers
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 0.5})
    except Exception:  # pragma: no cover
        return None
    if res.success:
        return np.asarray(res.x, dtype=np.float64)
    return None

                        if mixed_score < 2.0 * n * M:
                            a = mixed
                            break

        # ---- periodic reset to the best seen -----------------------
        if iteration % 3000 == 0:
            a = best.copy()

        # early exit if target already reached
        if best_score <= TARGET_SCORE:
            break

    # ------------------------------------------------------------
    # Return best solution as plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best]
```
