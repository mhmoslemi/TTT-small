from scipy.special import logsumexp

def _catalan_weights(n):
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
    return w

def _project_onto_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

    B = np.fft.rfft(b, fft_len)

    return np.fft.irfft(C, fft_len)

    C = A * A
    full = np.fft.irfft(C, fft_len)
    return full[: 2*n - 1]

        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0:

    cand.append(w / w.sum())

import time

from scipy.special import logsumexp

def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_convolve(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Convolution a*b using a real FFT (full length)."""
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    C = A * B
    return np.fft.irfft(C, fft_len)

    A = np.fft.rfft(a, fft_len)
