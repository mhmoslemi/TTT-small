def compute_score(a):
    n = a.shape[0]
    s = a.sum()
    # ensure s>0 else return inf
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    max_conv = conv.max()
    return 2.0 * n * max_conv / (s * s)
```

import time
import numpy as np

def _score_fft(a: np.ndarray) -> float:
    """Fast compute of the evaluator score (2*n*max(conv)/sum(a)^2)."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    max_conv = conv.max()
    return 2.0 * n * max_conv / (s * s)

def _build_sparse_set(order, n, max_mult=2):
    """Greedy construction of a set S ⊂ {0,…,n‑1}
    such that each possible sum i+j appears at most max_mult times."""
    sum_counts = [0] * (2 * n - 1)
    S = []
    for i in order:
        # self‑sum test
        if sum_counts[2 * i] >= max_mult:
            continue
        ok = True
        for j in S:
            if sum_counts[i + j] >= max_mult:
                ok = False
                break
        if not ok:
            continue
        # accept i
        sum_counts[2 * i] += 1
        for j in S:
            sum_counts[i + j] += 1
        S.append(i)
    return S, sum_counts
