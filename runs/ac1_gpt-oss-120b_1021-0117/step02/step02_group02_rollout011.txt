    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    # maintain best
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    best_score = float(best_score)

    # compute initial uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # FFT parameters for convolution
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # optimisation
    rng = np.random.default_rng(seed)
    it = 0
    eta0 = 0.15  # initial step
    mu = 0.9     # momentum
    v = np.zeros_like(a)
    K = 5       # number of top shifts to use for subgradient
    beta = 10.0 # softmax weighting parameter

    while time.monotonic() < deadline:
        it += 1
        # compute convolution via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # evaluate objective
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= 1.50274349:
                break

        # identify top-K shifts
        # get indices of K largest conv values
        if K >= conv_len:
            top_idx = np.arange(conv_len)
        else:
            # partition for top K
            part = np.argpartition(-conv, K-1)[:K]
            # sort these top indices descending (optional)
            top_idx = part[np.argsort(-conv[part])]

        # compute weighted gradient
        # compute weights via softmax over selected values
        top_vals = conv[top_idx]
        # avoid overflow
        max_val = top_vals.max()
        exp_vals = np.exp(beta * (top_vals - max_val))
        weights = exp_vals / exp_vals.sum()

        grad = np.zeros_like(a)
        for w, k in zip(weights, top_idx):
            # compute gradient for shift k
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # create view
            i = np.arange(i_start, i_end + 1)
            partner = k - i
            # accumulate
            grad[i] += w * 2.0 * a[partner]

    # Return best
    return [float(v) for v in best]

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x>=0, sum=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative: return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Heuristic optimisation of the AC1 upper bound.

    Uses a projected (momentum) sub‑gradient method that combines several
    active convolution shifts.  Starts from the provided `height_sequence_1`
    (if present) or from the uniform distribution.
    Returns the best sequence encountered as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. Base sequence (fallback if the parent is unavailable)
    # ------------------------------------------------------------------
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.clip(a, 0.0, 1000.0)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            raise ValueError
    except Exception:
        # Uniform fallback of modest size (will be resized later)
        a = None

    if a is None:
        n = 256
        a = np.ones(n, dtype=np.float64)
    else:
        n = int(a.size)

    # Normalise to sum = 1 (scale‐invariant for the objective)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    # ------------------------------------------------------------------
    # 3. Initialise bookkeeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Uniform baseline – sometimes already better
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # FFT helpers (linear convolution via real FFT)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.30               # initial step magnitude
    mu = 0.85                 # momentum coefficient
    v = np.zeros_like(a)      # momentum buffer
    K = 7                     # number of top shifts to combine
    beta = 12.0               # soft‑max sharpness for gradients
    target_score = 1.50274349

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution (FFT) ------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # ---- current objective (sum a = 1) ------------------------------------
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # ---- select K largest convolution entries -----------------------------
        if K >= conv_len:
            top_idx = np.arange(conv_len)
        else:
            part = np.argpartition(-conv, K - 1)[:K]
            top_idx = part[np.argsort(-conv[part])]   # descending order

        top_vals = conv[top_idx]
        # softmax weights (stable)
        max_val = top_vals.max()
        exp_vals = np.exp(beta * (top_vals - max_val))
        weights = exp_vals / exp_vals.sum()

        # ---- weighted sub‑gradient -------------------------------------------
        grad = np.zeros_like(a)
        for w, k in zip(weights, top_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1)
            partner = k - i
            grad[i] += w * 2.0 * a[partner]

        # ---- momentum step + projection ---------------------------------------
        eta = eta0 / np.sqrt(it)
        v = mu * v - eta * grad
        a_new = a + v
        a = _proj_simplex(a_new)

        # ---- occasional jitter to avoid stagnation ----------------------------
        if it % 250 == 0:
            jitter = np.exp(rng.normal(0.0, 0.012, size=n))
            a = _proj_simplex(a * jitter)

    # ------------------------------------------------------------------
    # 6. Return best sequence found
    # ------------------------------------------------------------------
    return [float(x) for x in best]
