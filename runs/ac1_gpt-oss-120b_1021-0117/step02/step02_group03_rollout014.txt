        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]

import time
import numpy as np

def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of a with itself, using FFT of length L."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * a.shape[0] - 1]

def _bound(a: np.ndarray, L: int) -> float:
    """Return the evaluator bound 2*n*max(conv) for a (sum(a) assumed = 1)."""
    n = a.shape[0]
    conv = _conv_full(a, L)
    return 2.0 * n * float(conv.max())

def _grad_sum(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """Aggregate gradient of the convolution value for all lag indices ks."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in ks:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=np.int64)

            cen = k // 2
            if 0 <= cen < n:
                grad[cen] -= a[cen] * 0.5   # after *2 becomes –a[cen]

    return grad
