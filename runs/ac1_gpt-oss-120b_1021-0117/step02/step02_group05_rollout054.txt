            g[mask] = 2.0 * cur[rev[mask]]               # ∂max/∂a_i

            mean_g = g.mean()
            d = mean_g - g                               # descent direction, sum(d)=0

            neg = d < 0
            if np.any(neg):
                max_alpha = np.min(cur[neg] / -d[neg])
            else:
                max_alpha = 1.0
            step_frac = min(0.5, temperature) * 0.8
            alpha = step_frac * max_alpha
            cand = cur + alpha * d

        elif move < grad_prob + block_prob:
            # ---------- block scaling ----------
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            sigma = 0.15 * temperature
            factor = math.exp(rng.normal(0.0, sigma))
            cand[start:start + width] *= factor

        else:
            # ---------- random mass transfer ----------
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * max_transfer
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta

        # enforce non‑negativity, bound and preserve total mass
        cand = _apply_constraints(cand, total)

        cand_score = _ac1_score(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.45:       # comfortably below the target
                break

        # simulated‑annealing acceptance for the current walk
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            accept = rng.random() < prob
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # occasional restart from the best solution seen so far
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ------------------------------------------------------------------
    return [float(x) for x in best]
