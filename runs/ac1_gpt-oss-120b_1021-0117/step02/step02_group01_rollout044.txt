import numpy as np

# ----------------------------------------------------------------------
# Generate the classic sqrt-series a such that (a * a)_k = 1 for
# k = 0 .. len(a)-1.  This gives max convolution = 1 and
# sum(a)^2 = 2*len(a)-1, achieving a score of 2*len(a)/(2*len(a)-1) ≈ 1.
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Return the non‑negative sequence of length n with flat autocorrelation."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        # s = sum_{i=1}^{k-1} a[i] * a[k-i]
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
