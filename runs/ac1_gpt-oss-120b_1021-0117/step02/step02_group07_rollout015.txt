import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of seq with itself using FFT (output length 2*len-1)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv_full = np.fft.irfft(f * f, n=fft_len)
    return conv_full[: 2 * seq.size - 1]


def _score(seq: np.ndarray, conv: np.ndarray, n: int) -> float:
    """Compute the AC1 upper‑bound score for a sequence of length n."""
    s = seq.sum()
    if s <= 0.0:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))

                cand[i + 1] = avg

            elif op == 4:  # boost early region, damp late region
                early_end = max(1, n // 15)
                late_start = n - early_end
                i = rng.integers(0, early_end)
                j = rng.integers(late_start, n)
                frac = rng.uniform(0.06, 0.20)
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta

            else:  # small random re‑normalisation of a random subset
                sz = rng.integers(2, 6)
                idxs = rng.choice(n, size=sz, replace=False)
                tot = cand[idxs].sum()
                if tot > 0:
                    cand[idxs] = tot / sz

            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_conv = _conv_fft(cand, fft_len)
            cand_score = _score(cand, cand_conv, n)

            if cand_score < current_score or rng.random() < 0.004:
                current, conv_cur, current_score = cand, cand_conv, cand_score
                if cand_score < best_score:
                    best, best_conv, best_score = cand.copy(), cand_conv.copy(), cand_score

        # ------------------------------------------------------------------
        # Time guard: break as soon as we are out of budget
        # ------------------------------------------------------------------
        if time.monotonic() >= deadline:
            break

    # Return the best candidate found as a plain Python list
    return [float(x) for x in best]
