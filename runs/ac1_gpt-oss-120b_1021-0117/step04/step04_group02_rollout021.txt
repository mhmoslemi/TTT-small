import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        cur = a0 / s               # now sum(cur)=1
    else:
        # fallback: exponential decay, length 2000
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        cur = np.exp(-t / 100.0)
        cur = cur / cur.sum()      # simplex

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length for full autocorrelation

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    eps = None
    eps_decay = 0.9985
    base_alpha = 0.8
    max_line = 8
    sym_interval = 400
    improve_target = 1.50274349
    beta = 0.8                # momentum factor
    prev = cur.copy()         # previous iterate (for momentum)

    # --------------------------------------------------------

    while time.monotonic() < deadline:
        it += 1

        # ---- Nesterov look‑ahead ----
        y = cur + beta * (cur - prev)   # may be slightly off‑simplex, ok for gradient

        # ---- autocorrelation of the look‑ahead point ----
        y_fft = np.fft.rfft(y, n=L)
        conv_full = np.fft.irfft(y_fft * y_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ---- smoothing / active‑set weight ----
        if eps is None:

        if eps > 1e-6:
