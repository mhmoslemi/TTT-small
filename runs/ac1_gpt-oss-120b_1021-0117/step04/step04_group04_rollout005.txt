import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Compute AC1 score (same as evaluator for sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(conv) where conv = a * a (linear)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initial sequence -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()

        a0 = np.exp(-np.arange(n0) / 200.0)

    a0 = _project_simplex(a0)          # sum == 1, non‑negative
    n = a0.shape[0]
    # FFT length for all convolutions (>= 3n-2)
    L = 1 << ((3 * n - 2).bit_length())

    best = a0.copy()
    best_sc = _score(best, L)

    cur = best.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------

    # ------------------------------------------------------------
    step0 = 0.8                # base step size
    p_max = 24                 # final power for Lp‑approximation
    eps_no_improve = 0
    it = 0

    # ------------------------------------------------------------
    # helper for adaptive power p (smooth → sharp)
    # ------------------------------------------------------------
    def _current_power(it):
        # exponential schedule, fast increase then plateau at p_max
        tau = 2000.0
        return min(p_max, 2.0 + (p_max - 2.0) * (1.0 - math.exp(-it / tau)))

    # ------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution & score
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_sc = float(2.0 * n * conv.max())

        # adaptive Lp weight (w_i = conv_i^{p-1})
        p = _current_power(it)
        # avoid negative or NaN values (theoretical conv >= 0)
        conv_clip = np.clip(conv, 0.0, None)
        w = np.power(conv_clip, p - 1.0)

        w_rev = w[::-1]

        idx = (2 * n - 2) - np.arange(n)               # slice for gradient
