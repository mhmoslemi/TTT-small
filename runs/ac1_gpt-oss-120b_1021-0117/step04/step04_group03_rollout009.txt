def descent(a, budget_deadline, start_best_score, start_best_vec):
    best_vec = a.copy()
    best_sc = start_best_score
    cur = a.copy()
    cur_sc = start_best_score
    # parameters
    eps = 0.5
    eps_decay = 0.9995
    eps_min = 1e-6
    lr = 0.5 # step0 scaling constant

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    t_iter = 0
    # track for random perturbation
    no_improve = 0
    rng = np.random.default_rng(12345) maybe use seed.

    while time.monotonic() < deadline:
        t_iter += 1
        # compute conv and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # eps schedule
        eps = max(eps_min, eps * eps_decay)
        # compute softmax weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.zeros_like(w)
            w[np.argmax(conv)] = 1.0

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # update candidate
        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate
        cand_sc = _score(cand)

        # accept if improved
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # random perturbation
        if no_improve > 2000:
            # random mass transfer
            i_idx = rng.integers(0, n)
            if cur[i_idx] > 0:
                delta = cur[i_idx] * rng.random() * 0.2
                j_idx = rng.integers(0, n)
                if j_idx == i_idx:
                    j_idx = (j_idx + 1) % n
                tmp = cur.copy()
                tmp[i_idx] -= delta
                tmp[j_idx] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # optionally symmetrize occasionally
        if (t_iter % 1000) == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)

        # early exit if best score target reached
        if best_sc < 1.50274349:
            break
    return best_vec, best_sc

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score assuming sum(a)=1  (exact AC1 bound)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a)) for sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the epsilon‑smoothed max of the autocorrelation
# ------------------------------------------------------------
def _grad_smoothed_max(a: np.ndarray, eps: float, L: int) -> np.ndarray:
    """
    Compute ∇ (eps * log sum exp(conv/eps)) using FFTs.
    Returns a gradient vector of the same length as a.
    """
    n = a.size
    a_fft = np.fft.rfft(a, n=L)

    # full autocorrelation
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    mx = conv.max()

    # softmax weights over the autocorrelation entries
    w = np.exp((conv - mx) / eps)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w[:] = 0.0
        w[np.argmax(conv)] = 1.0

    # convolution of a with reversed weights gives gradient
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    # pick the n relevant entries (see previous implementation)
    idx = (2 * n - 2) - np.arange(n)   # shape (n,)
    grad = 2.0 * grad_full[idx]
    return grad
