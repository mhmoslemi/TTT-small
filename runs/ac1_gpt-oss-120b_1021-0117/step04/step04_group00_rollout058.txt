def _project_nonneg_normalize(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1."""

    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
