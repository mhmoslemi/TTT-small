import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum=1, non‑neg.)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Simple score evaluation (2 * n * max(convolution))
# ------------------------------------------------------------
def _score_from_max(n: int, max_val: float) -> float:
    return 2.0 * n * float(max_val)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final conversion to list
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------- initialise ---------------------------

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # project onto simplex (sum = 1)

    n = a.size
    # FFT length sufficient for a full linear convolution
    L = 1 << ((3 * n - 2).bit_length())
