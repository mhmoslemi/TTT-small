import time, math, numpy as np

def _project_simplex(v):
    # Euclidean projection onto simplex sum=1, v>=0
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return np.maximum(v, 0.)  # if already feasible
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return np.maximum(v, 0.0)
    n = v.size
    u = np.sort(v)[::-1]

        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
