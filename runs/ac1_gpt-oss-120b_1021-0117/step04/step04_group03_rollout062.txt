import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score (exact evaluator up to a constant factor)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(conv) for a vector with sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the smoothed max objective
# ------------------------------------------------------------
def _grad(a: np.ndarray, eps: float, L: int) -> tuple[np.ndarray, float]:
    """
    Return (grad, max_conv) where grad is the gradient of
    smoothed max_{k} (a*a)[k] with smoothing parameter eps.
    L must be a power of two >= 3*n-2.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()

    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum != 0.0:
        w /= w_sum
    else:
        w.fill(1.0 / w.size)

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val
