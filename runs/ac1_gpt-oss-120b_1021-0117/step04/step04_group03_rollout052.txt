def _subgrad(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    mask = (conv >= max_val - 1e-12)  # bool mask
    # Convert mask to float array
    m = mask.astype(np.float64)
    # Compute gradient: 2 * convolution of a with reversed m, sliced.
    # Equivalent to: grad_full = irfft(af * rfft(m_rev)), then idx slice.
    m_rev = m[::-1]
    mf = np.fft.rfft(m_rev, n=L)
    grad_full = np.fft.irfft(af * mf, n=L)
    # extract indexes: idx = (2*n-2) - np.arange(n)
    idx = (2*n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, max_val

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
