import time
import math
import numpy as np

# -------------------------------------------------------------
# Euclidean projection onto the probability simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Score used by the previous baseline (max convolution * 2n)
def _score(a: np.ndarray, fft_len: int) -> float:
    """Return 2*n*max(conv(a,a)) for a vector with sum == 1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())
