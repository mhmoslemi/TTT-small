import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)   sum(x)=1, x>=0
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast convolution and score (sum is assumed = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """2*n*max(a*a) where a is assumed to sum to 1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative coefficient list with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.clip(a, 0.0, 1000.0)
    else:
        n_fallback = 1024
        a = np.exp(-np.arange(n_fallback, dtype=np.float64) / 30.0)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a_sum                     # normalise to sum = 1
        a = _project_simplex(a)           # ensure exact simplex

    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())   # FFT length (power of two)

    # -------------------- bookkeeping --------------------
    best_vec = a.copy()
    best_sc = _score(best_vec, L)
    cur = a.copy()
    cur_sc = best_sc

    # Adam moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.5                     # base learning rate
    max_step = 0.3                    # L2‑norm cap for each update

    # smoothing for soft‑max
    eps0 = 1.0
    eps_decay = 0.9995
    eps_min = 1e-6

    it = 0
    no_improve = 0

    # ------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (softmax version) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # cap norm to avoid too‑large jumps
        step_norm = np.linalg.norm(step) + 1e-12
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _score(cand, L)

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional uphill move (simulated annealing)
            prob = math.exp(-(cand_sc - cur_sc) / (max_step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:

                    no_improve = 0

        if no_improve > 3000:

                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------
    # final local refinement (small jitter)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best_vec[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp, L)
        if tmp_sc < best_sc:
            best_vec, best_sc = tmp, tmp_sc
            continue
        # occasional symmetrisation
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _project_simplex(sym)
        sym_sc = _score(sym, L)
        if sym_sc < best_sc:
            best_vec, best_sc = sym, sym_sc

    # return as plain Python list
    return [float(x) for x in best_vec]
