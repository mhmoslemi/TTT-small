import time, math
import numpy as np

# ------------------------------------------------------------
# Project onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.shape[0] + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
