import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score for sum‑1 vectors (same as evaluator up to scaling)

    """Return 2*n*max(convolution) for a (assumed sum(a)=1)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    n = a.shape[0]
    # FFT length sufficient for linear convolutions of length up to 2n‑1 and
    # for the gradient step (needs length >= n + (2n‑1) - 1 = 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Adam optimiser state (on the simplex)
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.5                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # ------------------------------------------------------------------
    # Track best solution
    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_sc = _score_sum1(best_vec, L)
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # Parameters for the smoothed max (soft‑max) weighting
    # ------------------------------------------------------------------
    eps0 = 0.5
    eps_decay = 0.9995
    eps_min = 1e-7

    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate (FFT) -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- soft‑max weights over convolution entries -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]                                         # reversed weights
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)           # cyclic conv
        idx = (2 * n - 2) - np.arange(n)                       # pick correct slice
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / (math.sqrt(it) + 1e-12)

        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = _project_simplex(a_new)

        # ----- evaluate -----
        new_sc = _score_sum1(a_new, L)

        # ----- accept/reject with occasional annealing -----
        if new_sc < cur_sc:
            a, cur_sc = a_new, new_sc
            if new_sc < best_sc:
                best_vec, best_sc = a.copy(), new_sc
                no_improve = 0
        else:
            prob = math.exp(-(new_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a, cur_sc = a_new, new_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps autocorrelation) -----
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_sum1(sym, L)
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_sum1(tmp, L)
                if tmp_sc < cur_sc:
                    a, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early stop if target reached -----
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Final local refinements (small random jitters)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best_vec[i] * rng.random() * 0.02
        if delta <= 0.0:
            continue
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        sc = _score_sum1(tmp, L)
        if sc < best_sc:
            best_vec, best_sc = tmp, sc
            continue
        # try symmetrised version as well
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _project_simplex(sym)
        sc_sym = _score_sum1(sym, L)
        if sc_sym < best_sc:
            best_vec, best_sc = sym, sc_sym

    # Return as plain Python list
    return [float(x) for x in best_vec]
