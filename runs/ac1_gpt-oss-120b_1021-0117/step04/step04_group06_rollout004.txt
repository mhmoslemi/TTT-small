    """Return softmax(z) as a probability vector (sum=1, non‑negative)."""

    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return exp_z / s


# ------------------------------------------------------------
# Score: 2*n*max(convolution) for a vector with sum=1
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray | None = None) -> float:
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative coefficient list with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # -------------------- initialise a --------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: exponential decay, length 1024
        n_fallback = 1024
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 20.0)
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s0

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())  # FFT length for all convolutions

    # Transform initial probabilities to log‑space for unconstrained optimisation
    z = np.log(a0 + 1e-12)

    # Adam optimiser state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    lr0 = 0.05

    # smoothing for soft‑max of convolution
    eps0 = 0.5
    eps_decay = 0.998
    eps_min = 1e-6

    # bookkeeping
    best_a = a0.copy()
    best_sc = _score(best_a)
    cur_a = a0.copy()
    cur_sc = best_sc

    no_improve = 0
    it = 0

    # ------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current probability vector -----
        a = _softmax(z)                         # sum=1, >=0
        # ----- convolution a*a -----
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smoothed weight for max (soft‑max) -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient wrt a (smooth max) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        g_a = 2.0 * grad_full[idx]

        # ----- gradient wrt log‑variables (softmax Jacobian) -----
        dot = np.dot(g_a, a)
        g_z = g_a * a - dot * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)               # simple decay
        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        z_new = z - step

        # ----- evaluate candidate -----
        a_new = _softmax(z_new)
        af_new = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(af_new * af_new, n=L)[: 2 * n - 1]
        cand_sc = float(2.0 * n * conv_new.max())

        # ----- accept if better -----
        if cand_sc < cur_sc:
            z = z_new
            cur_a = a_new
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_a = a_new.copy()
                best_sc = cand_sc
        else:
            # occasional simulated‑annealing acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                z = z_new
                cur_a = a_new
                cur_sc = cand_sc
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % 700 == 0:
            sym = (cur_a + cur_a[::-1]) * 0.5
            z_sym = np.log(sym + 1e-12)
            # evaluate symmetrised candidate
            af_sym = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(af_sym * af_sym, n=L)[: 2 * n - 1]
            sym_sc = float(2.0 * n * conv_sym.max())
            if sym_sc < cur_sc:
                z = z_sym
                cur_a = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_a = sym.copy()
                    best_sc = sym_sc
                no_improve = 0

        # ----- random perturbation when stuck -----
        if no_improve > 2500:
            # add Gaussian noise to log‑variables
            z += rng.normal(scale=0.4, size=n)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------
    # final jitter phase (local refinement)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best_a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        tmp = best_a.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _softmax(np.log(tmp + 1e-12))          # renormalise
        sc = _score(tmp)
        if sc < best_sc:
            best_a, best_sc = tmp, sc
            continue
        # occasional symmetrisation
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _softmax(np.log(sym + 1e-12))
        sc_sym = _score(sym)
        if sc_sym < best_sc:
            best_a, best_sc = sym, sc_sym

    # return plain Python list
    return [float(x) for x in best_a]
