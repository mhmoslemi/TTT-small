            a[mask] = 0.0

            a = a / a.sum()
            # reset Adam state
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            no_improve = 0

        # ----- occasional symmetrisation (helps the anti‑symmetric optimum) -----
        if it % 2500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = sym / sym.sum()
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * float(sym_conv.max())
            if sym_sc < cur_sc:
                a = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = a.copy()
                    no_improve = 0

        # early exit if the target is already beaten
        if best_sc < 1.50274349:
            break

    # ---------- fine‑tuning with tiny Gaussian tweaks ----------
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.004, size=n)
        cand = best_vec + delta
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand = cand / s
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * float(cand_conv.max())
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

    # return plain Python list
    return [float(x) for x in best_vec]
