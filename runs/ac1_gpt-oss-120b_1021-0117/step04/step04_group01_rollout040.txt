def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0,budget_s - 0.5) # margin

    # init
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = np.ones(n) / n
    # ensure positivity, sum=1
    a = np.maximum(a,0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0/a.size)
    else:
        a = a / s

    n = a.size
    L = 1 << ((3*n - 2).bit_length())  # zero padding length

    # helpers:
    def project_simplex(v):
        # using same algorithm as previous code
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        w = np.maximum(v - theta, 0.0)
        return w

    def score_vec(vec):
        # assume sum(vec)=1
        a_fft = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return 2.0 * n * float(conv.max())

    # compute best
    best_vec = a.copy()
    best_sc = score_vec(best_vec)

    # main loop
    # hyper parameters
    alpha0 = 0.2  # step size factor
    eps = 1e-12   # small to avoid division
    # track no improvements
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # compute convolution
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # find positions max (within tolerance)
        # tolerance relative to max.
        tol = 1e-12 * max_val + 1e-14
        mask = np.abs(conv - max_val) <= tol
        # count positions
        cnt = mask.sum()
        if cnt == 0:
            # fallback to mask where conv close to max (there might be rounding)
            cnt = 1
            mask[np.argmax(conv)] = True
        w = np.where(mask, 1.0 / cnt, 0.0)
        # gradient of max
        w_rev = w[::-1]

        grad = 2.0 * n * grad_max

        improved = False
        for _ in range(10):

            cand = project_simplex(cand)
            cand_sc = score_vec(cand)
            if cand_sc < best_sc - 1e-12:  # improvement
                a = cand
                best_vec = cand.copy()
                best_sc = cand_sc
                improved = True
                no_imp = 0
                break
            step *= 0.5
        if not improved:
            no_imp += 1
            # maybe random perturb
            if no_imp > 2000:
                # small random perturbation to escape
                delta = rng.normal(scale=0.01, size=n)
                a = a + delta

                a = project_simplex(a)

        if best_sc < 1.50274349:

            a_sym_sc = score_vec(a_sym)
            if a_sym_sc < best_sc:

                best_sc = a_sym_sc
                best_vec = a_sym.copy()

    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # 0.3 sec margin

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a = np.ones(n_fallback, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())

    def project_simplex(v):
        u = np.sort(v)[::-1]
        cs = np.cumsum(u) - 1.0
        rho = np.where(u - cs / np.arange(1, v.size + 1) > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = cs[rho] / (rho + 1)
        else:
            theta = 0.0
        return np.maximum(v - theta, 0.0)

    def obj_score(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return 2.0 * n * float(conv.max())
    # initial best
    best_vec = a.copy()
    best_sc = obj_score(best_vec)

    # gradient descent parameters
    alpha0 = 0.5
    momentum = 0.9
    vel = np.zeros_like(a)
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # compute convolution and gradient weight
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # tolerance for multiple maxima
        eps = max(1e-14, 1e-12 * abs(max_val))
        mask = np.abs(conv - max_val) <= eps
        cnt = int(mask.sum())
        if cnt == 0:
            mask[np.argmax(conv)] = True
            cnt = 1
        w = np.where(mask, 1.0 / cnt, 0.0)
        # gradient of max via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad_max = 2.0 * grad_full[idx]   # shape (n,)
        grad_obj = 2.0 * n * grad_max
        # step size
        step = alpha0 / math.sqrt(it)
        # momentum update
        vel = momentum * vel - step * grad_obj
        cand = a + vel
        cand = project_simplex(cand)
        cand_sc = obj_score(cand)
        if cand_sc < best_sc - 1e-12:
            a = cand
            best_vec = cand.copy()
            best_sc = cand_sc
            no_imp = 0
        else:
            no_imp += 1
            # fallback: try simple gradient step without momentum
            cand2 = a - step * grad_obj
            cand2 = project_simplex(cand2)
            cand2_sc = obj_score(cand2)
            if cand2_sc < best_sc - 1e-12:
                a = cand2
                best_vec = cand2.copy()
                best_sc = cand2_sc
                no_imp = 0
                vel = np.zeros_like(a)  # reset momentum
        # occasional symmetrization
        if it % 4000 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = project_simplex(sym)
            sym_sc = obj_score(sym)
            if sym_sc < best_sc:
                a = sym
                best_vec = sym.copy()
                best_sc = sym_sc
                vel = np.zeros_like(a)
                no_imp = 0
        # random perturb when stagnant
        if no_imp > 3000:
            # random reallocation of some mass
            delta = rng.normal(scale=0.01, size=n)
            tmp = a + delta
            tmp = np.clip(tmp, 0, None)
            tmp = project_simplex(tmp)
            tmp_sc = obj_score(tmp)
            if tmp_sc < best_sc:
                a = tmp
                best_vec = tmp.copy()
                best_sc = tmp_sc
                no_imp = 0
                vel = np.zeros_like(a)
            else:
                no_imp = 0  # reset anyway
        # early exit
        if best_sc < 1.50274349:
            break

    # final fine-tuning random local search
    while time.monotonic() < deadline - 0.1:
        delta = rng.normal(scale=0.003, size=n)
        cand = best_vec + delta
        cand = np.clip(cand, 0, None)
        cand = project_simplex(cand)
        cand_sc = obj_score(cand)
        if cand_sc < best_sc:
            best_vec = cand
            best_sc = cand_sc

    return [float(x) for x in best_vec]

import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of heights) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # ------------------------------------------------------------------
    # initialise a feasible point (simplex with sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:                              # fallback: uniform mass
        a = np.ones(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # normalise to L1 = 1

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length covering 3n‑2 entries

    # ------------------------------------------------------------------
    # helper utilities (simplex projection, fast objective)
    # ------------------------------------------------------------------
    def _project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection of v onto the probability simplex."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        return np.maximum(v - theta, 0.0)

    def _obj_score(x: np.ndarray) -> float:
        """2·n·max(conv(x,x)) – the exact quantity returned by evaluate_sequence."""
        af = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return 2.0 * n * float(conv.max())

    # initial best
    best_vec = a.copy()
    best_sc = _obj_score(best_vec)

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    eps0 = 1.0               # initial soft‑max temperature
    eps_decay = 0.9999
    eps_min = 1e-12
