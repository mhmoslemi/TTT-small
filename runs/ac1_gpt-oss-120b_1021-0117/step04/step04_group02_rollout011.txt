import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a) ≈ 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the AC1 bound for a non‑negative sequence."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    maxv = float(conv.max())
    s = float(a.sum())
    # guard against tiny sums – evaluator would reject them anyway
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * maxv / (s * s))

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑AC1 step‑function."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---------- initialise ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s0 = 1.0
        a0 = a0 / s0                     # normalize to sum = 1
    else:
        # fallback: exponential decay, length 2000
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length ≥ 2n‑1

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---------- optimisation parameters ----------
    it = 0
    no_improve = 0
    eps = None
    base_alpha = 0.8
    eps_decay = 0.9995
    target = 1.50274349
    sym_interval = 400

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smoothing weight (soft‑max) -----
        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:

        else:
            w /= w_sum

        # ----- gradient of the weighted sum -----

            w_max /= w_max.sum()

            beta = 0.4                     # mixing weight

        # ----- early exit if target reached -----

            cur = cur[rng.permutation(n)]

    # final optional symmetrisation (often beneficial)
    sym = (best_vec + best_vec[::-1]) * 0.5
    sym = _project_simplex(sym)
    if _score(sym) < best_sc:
        best_vec = sym

    # return plain Python list
    return [float(x) for x in best_vec]
