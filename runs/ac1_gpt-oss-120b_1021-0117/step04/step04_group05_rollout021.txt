def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # get initial sequence
    if 'height_sequence_1' in globals():
        cur = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 1024
        cur = np.exp(-np.arange(n0, dtype=np.float64) / (n0/10.0))
    cur = np.maximum(cur, 0.0)
    # normalize to sum 1 via simplex projection
    cur = cur / cur.sum()
    n = cur.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    # Ensure L is at least maybe 2*n for FFT speed, but above ok.
    # Precompute FFT of cur
    cur_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[:2*n-1]
    cur_max = conv_full.max()
    cur_score = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_score = cur_score
    # parameters
    step0 = 0.6
    eps_rel = 0.04
    eps_min = 1e-8
    beta = 0.9
    vel = np.zeros_like(cur)
    it = 0
    no_improve = 0
    # main loop
    while time.monotonic() < deadline:
        it += 1
        # symmetrize current vector to enforce symmetry (may help)
        cur = (cur + cur[::-1]) * 0.5
        # project after symmetrization (stay on simplex)
        cur = cur / cur.sum()
        # recompute conv and scores after symmetrization
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[:2*n-1]
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max
        # gradient weights
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((conv_full - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # compute gradient via convolution with w_rev
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L)
        # extract gradient for each coordinate: conv with reversed w
        # idx as before: idx = (2*n-2) - np.arange(n)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum update
        vel = beta * vel + (1 - beta) * g
        # step size schedule
        alpha = step0 / math.sqrt(it)
        # propose candidate
        cand = cur - alpha * vel
        # symmetrize candidate
        cand = (cand + cand[::-1]) * 0.5
        # project onto simplex
        cand = cand / cand.sum()
        # evaluate candidate
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[:2*n-1]
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max
        if cand_score < cur_score:
            cur = cand
            cur_fft = cand_fft
            cur_max = cand_max
            cur_score = cand_score
            vel = np.zeros_like(cur)  # reset momentum after accepted step maybe
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:
            # probabilistic acceptance
            prob = math.exp(-(cand_score - cur_score) / (alpha + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_fft = cand_fft
                cur_max = cand_max
                cur_score = cand_score
                # keep momentum
                no_improve = 0
            else:
                no_improve += 1
        # occasional random perturbation if stuck
        if no_improve > 3500:
            # perturb by adding Gaussian noise and reproject
            sigma = 0.02 * (1.0 + it / 5000.0)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = np.clip(pert, 0, None)
            pert = pert / pert.sum()
            pert_fft = np.fft.rfft(pert, n=L)
            pert_conv = np.fft.irfft(pert_fft * pert_fft, n=L)[:2*n-1]
            pert_score = 2.0 * n * pert_conv.max()
            if pert_score < cur_score:
                cur = pert
                cur_fft = pert_fft
                cur_max = pert_conv.max()
                cur_score = pert_score
                vel = np.zeros_like(cur)
                no_improve = 0
            if pert_score < best_score:
                best_score = pert_score
                best_vec = pert.copy()
        # early exit if beat target
        if best_score < 1.50274349:
            break
    # return best sequence as list of floats
    return [float(x) for x in best_vec]

def _normalize(v: np.ndarray) -> np.ndarray:
    """Make a non‑negative vector sum to 1 (simple simplex projection)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v.fill(1.0 / v.size)
    else:
        v /= s
    return v
