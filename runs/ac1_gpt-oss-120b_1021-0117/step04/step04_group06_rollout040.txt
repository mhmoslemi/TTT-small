import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u > css / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score for a vector that already has sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(convolution) assuming sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------

    a_sum = a.sum()
    if a_sum <= 0.0:

        a = a / a_sum                     # project onto simplex
    n = a.shape[0]

    # FFT length (next power of two >= 2n-1)

    # ------------------- bookkeeping -------------------------------------------------

    # ------------------- hyper‑parameters -------------------------------------------
    step0 = 1.0                # base step magnitude
    eps0 = 1.0                 # initial temperature for soft‑max

    it = 0

    target = 1.50274349

    while time.monotonic() < deadline:

        max_val = conv.max()
