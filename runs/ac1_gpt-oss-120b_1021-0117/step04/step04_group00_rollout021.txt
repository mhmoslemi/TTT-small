import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Closed‑form bound for a given vector (same as evaluate_sequence)
# ------------------------------------------------------------
def _bound(a: np.ndarray) -> float:
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:                    # safety – mirrors evaluator
        return float("inf")
    q = np.dot(a, a)                # sum of squares
    return 2.0 * n * q / (s * s)    # 2 n * max(conv) / sum^2, as max(conv)=q


# ------------------------------------------------------------
# Analytic gradient of the bound (before projection)
# ------------------------------------------------------------
def _grad(a: np.ndarray, s: float, q: float) -> np.ndarray:
    """Gradient of 2 n * q / s^2 w.r.t. a (before the simplex projection)."""
    n = a.shape[0]
    # dB/da = (4 n / s^3) * (a * s - q)
    factor = (4.0 * n) / (s * s * s)
    return factor * (a * s - q)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # leave a margin

    # ------------------------------------------------------------------
    # Initialise – use the parent if present; otherwise a random start.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # modest length, e.g. 1024
        a = rng.random(1024, dtype=np.float64)

    a = np.maximum(a, 0.0)          # enforce non‑negativity
    a = _project_simplex(a)         # sum = 1
    n = a.size

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_sc = _bound(best)

    # simple schedule
    step0 = 0.8
    it = 0

    # ------------------------------------------------------------------
    # Projected gradient descent (convex – converges to uniform)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        s = a.sum()
        q = np.dot(a, a)                # sum of squares

        # gradient of the bound (before projection)
        g = _grad(a, s, q)

        # diminishing step size
        alpha = step0 / math.sqrt(it)

        # take a gradient step and project back onto the simplex
        a_new = _project_simplex(a - alpha * g)

        sc = _bound(a_new)

        # accept if improvement, otherwise keep current (convex ⇒ always improves)
        if sc < best_sc:
            best = a_new.copy()
            best_sc = sc

        a = a_new

        # occasional random perturbation to avoid stagnation (unlikely needed)
        if it % 500 == 0:
            # inject a tiny random tweak and re‑project
            perturb = rng.normal(scale=1e-4, size=n)
            a = _project_simplex(a + perturb)

        # early exit if we have essentially reached the theoretical optimum
        if best_sc <= 2.000001:          # uniform gives exactly 2
            break

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
