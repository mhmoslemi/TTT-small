# ------------------------------------------------------------
# Simplex projection (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(convolution of a with itself)."""

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # additional starter: uniform and arcsine shape
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()

    # pick the best cheap starter
    starters = [base, uniform, arcsine]
    cur = min(starters, key=_score).copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Smoothed‑max gradient descent (many cheap iterations)
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())
    eps = 1.0
    eps_decay = 0.99985
    eps_min = 1e-6
    step0 = 0.6
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # current convolution and its maximum
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smoothed weights for the max
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # step size (scale‑invariant)
        grad_norm = np.linalg.norm(grad) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        cand = cur - alpha * grad

        # enforce constraints
        cand = _project_simplex(cand)

        # occasional symmetrisation
        if (it % 400) == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # random re‑allocation when stuck
            if no_improve > 2500:
                i_idx = rng.integers(n)
                if cur[i_idx] > 0.0:
                    delta = cur[i_idx] * rng.random() * 0.25
                    j_idx = rng.integers(n)
                    while j_idx == i_idx:
                        j_idx = rng.integers(n)
                    tmp = cur.copy()
                    tmp[i_idx] -= delta
                    tmp[j_idx] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Final local hill‑climbing refinement
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        i_idx = rng.integers(n)
        if best_vec[i_idx] <= 0.0:
            continue
        delta = best_vec[i_idx] * rng.random() * 0.15
        j_idx = rng.integers(n)
        while j_idx == i_idx:
            j_idx = rng.integers(n)
        cand = best_vec.copy()
        cand[i_idx] -= delta
        cand[j_idx] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return the best vector found as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
