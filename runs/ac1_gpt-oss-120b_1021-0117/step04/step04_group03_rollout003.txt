import time

import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum to 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast maximum of the autocorrelation (conv of a with itself)
# ------------------------------------------------------------
def _max_autocorr(a: np.ndarray, L: int) -> float:
    """Return max_k (a * a)[k] using FFT of length L (L ≥ 2n‑1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Smoothed max and its gradient (soft‑max temperature eps)
# ------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float, L: int):
    """
    Returns (max_val, grad) where grad = ∇_a (softmax_ε max conv).
    Uses FFT of length L (L ≥ 3n‑2).
    """
    n = a.size
    a_fft = np.fft.rfft(a, n=L)

    # convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()

    # soft‑max weights
    shifted = (conv - max_val) / eps
    # stability: max exponent is 0 → exp(0)=1, others ≤1
    exps = np.exp(shifted)
    w_sum = exps.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = exps / w_sum

    # gradient: 2 * (w ⋆ a) where ⋆ is cross‑correlation.
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
    idx = (2 * n - 2) - np.arange(n)          # map to proper offsets
    grad = 2.0 * grad_full[idx]
    return max_val, grad

                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best_score, best_vec = tmp_score, tmp.copy()
        # temperature decay
        eps_temp = max(eps_min, eps_temp * eps_decay)

        # safety break if time almost exhausted
        if time.monotonic() + 0.05 > deadline:
            break

    # --------------------------------------------------------
    # Return best found vector as plain Python list
    # --------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    return [float(x) for x in best_vec]
```
