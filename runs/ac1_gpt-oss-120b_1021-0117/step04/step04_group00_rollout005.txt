    cur = a0.astype(np.float64)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # power of two >= 2n-1

    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.3  # initial step size

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # smoothing temperature schedule
    T0 = 0.1
    T_decay = 0.9995
    T_min = 1e-8

    # progress tracking
    it = 0
    no_improve = 0
    target = 1.50274349
    # loop
    while True:
        tnow = time.monotonic()
        if tnow >= deadline:
            break

        it += 1

        # compute convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2 * n - 1]
        max_c = conv.max()

        # smoothing weight
        T = max(T_min, T0 * (T_decay ** it))
        w = np.exp((conv - max_c) / T)
        w_sum = w.sum()
        if w_sum == 0.0:

        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)

        g = 2.0 * grad_full[idx]

        beta1_pow *= beta1
        beta2_pow *= beta2

        v = beta2 * v + (1.0 - beta2) * (g * g)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – sum = 1, non‑negative
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_candidates = u - css / np.arange(1, n + 1) > 0
    if not np.any(rho_candidates):
        theta = 0.0
    else:
        rho = np.nonzero(rho_candidates)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point

    # ------------------------------------------------------------------
