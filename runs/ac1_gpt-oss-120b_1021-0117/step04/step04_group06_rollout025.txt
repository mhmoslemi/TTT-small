import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Softmax (numerically stable)
# ------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ------------------------------------------------------------
# Gradient of the smoothed max–convolution w.r.t. a
# ------------------------------------------------------------
def _grad_a_from_w(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """Given FFT(a) and weight vector w, return ∇_a ϕ where ϕ is the soft‑max of a*a."""
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * full[idx]
