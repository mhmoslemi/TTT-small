def get_max_conv(a):
    L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    max_val = conv[:2*n-1].max()
    return max_val

def _fast_score(a):
    n = a.shape[0]
    Lfft = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(af * af, n=Lfft)
    max_val = conv[: 2*n-1].max()
    return float(2 * n * max_val)

import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _fast_score(a: np.ndarray) -> float:
    """ 2 * n * max(conv(a,a))  (assumes sum(a)==1) """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return float(2.0 * n * conv[: 2 * n - 1].max())
