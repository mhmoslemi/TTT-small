# ----------------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        # ----- occasional whole‑vector reflection (helps escape symmetry) -----
        if it % 2500 == 0:
            refl = cur[::-1]
            refl = _project_simplex(refl)
            refl_fft = np.fft.rfft(refl, n=L_full)
            refl_conv = np.fft.irfft(refl_fft * refl_fft, n=L_full)[: 2 * n - 1]
            refl_sc = 2.0 * n * refl_conv.max()
            if refl_sc < cur_sc:
                cur = refl
                cur_fft = refl_fft
                conv_full = refl_conv
                cur_max = refl_conv.max()
                cur_sc = refl_sc
                if refl_sc < best_sc:
                    best_sc = refl_sc
                    best_vec = refl.copy()

        # early exit if target already reached
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------------
    # Swap‑based hill‑climbing phase (uses remaining time)
    # --------------------------------------------------------------
    cur = best_vec.copy()
    cur_fft = np.fft.rfft(cur, n=L_full)
    conv_full = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
    cur_sc = 2.0 * n * conv_full.max()
    no_improve_sw = 0

    while time.monotonic() < deadline:
        # random operation: swap or small transfer
        if rng.random() < 0.6:                         # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer
            i = rng.integers(0, n)
            if cur[i] == 0.0:
                continue
            delta = cur[i] * rng.random() * 0.2
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)               # keep feasibility

        cand_fft = np.fft.rfft(cand, n=L_full)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[: 2 * n - 1]
        cand_sc = 2.0 * n * cand_conv.max()

        if cand_sc < cur_sc - 1e-12:
            cur = cand
            cur_fft = cand_fft
            conv_full = cand_conv
            cur_sc = cand_sc
            no_improve_sw = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve_sw += 1

        if best_sc < 1.50274349:
            break
        # occasional escape move if stagnating
        if no_improve_sw > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.2
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    conv_full = tmp_conv
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve_sw = 0

    # --------------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
