# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(a*a) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the smoothed max (soft‑max with temperature eps)
# ------------------------------------------------------------
def _soft_gradient(a: np.ndarray, eps: float, L: int, n: int):
    """Return gradient of eps*log-sum-exp(conv) w.r.t. a."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient = 2 * (a * rev(w))  (convolution)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, maxc


# ------------------------------------------------------------
# Spectral “square‑root” of a uniform convolution (analytic seed)
# ------------------------------------------------------------
def _spectral_initial(n: int, L: int) -> np.ndarray:
    """Construct a seed by taking the inverse FFT of sqrt(FFT(uniform))."""
    m = 2 * n - 1
    uniform = np.full(m, 1.0 / m, dtype=np.float64)
    ft = np.fft.rfft(uniform, n=L)
    # principal square‑root (complex)
    sqrt_ft = np.sqrt(ft)
    w_full = np.fft.irfft(sqrt_ft, n=L)
    w = w_full[:n]
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # length and FFT size
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a_parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a_parent = np.full(n, 1.0 / n, dtype=np.float64)

    L = 1 << ((3 * n - 2).bit_length())      # for 3n‑2 convolution length

    # --------------------------------------------------------
    # Normalise the parent (or uniform fallback)
    # --------------------------------------------------------
    a_parent = np.maximum(a_parent, 0.0)
    s = a_parent.sum()
    if s > 0.0:
        a_parent = a_parent / s
    else:
        a_parent = np.full_like(a_parent, 1.0 / n)

    # --------------------------------------------------------
    # Build a list of diverse starting points
    # --------------------------------------------------------
    starts = []

    # 1. parent (if any)
    starts.append(a_parent.copy())

    # 2. uniform
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3. random Dirichlet‑like draws
    for _ in range(3):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        starts.append(v)

    # 4. arcsine‑shaped (Beta(0.5,0.5) analogue)
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    starts.append(arcs)

    # 5. power‑law family (U‑shaped)
    for p in (0.4, 0.8, 1.2):
        w = ((i + 1) * (n - i)) ** (-p)
        w = w / w.sum()
        starts.append(w)

    # 6. spectral square‑root seed
    starts.append(_spectral_initial(n, L))

    # --------------------------------------------------------
    # Keep track of the overall best
    # --------------------------------------------------------
    best_seq = None
    best_sc = np.inf

    # --------------------------------------------------------
    # Core optimisation routine (projected gradient with momentum)
    # --------------------------------------------------------
    for seed_vec in starts:
        cur = seed_vec.copy()
        cur_sc = _fast_score(cur, L, n)
        if cur_sc < best_sc:
            best_sc, best_seq = cur_sc, cur.copy()

        # momentum buffer
        mom = np.zeros_like(cur)
        beta = 0.9

        # schedule parameters
        eps = 0.02            # start temperature for soft‑max
        eps_min = 1e-7
        eps_decay = 0.9995
        step0 = 0.8
        it = 0
        no_improve = 0

        while time.monotonic() < deadline:
            it += 1

            # ----- gradient (soft‑max) -----
            g, _ = _soft_gradient(cur, eps, L, n)

            # ----- momentum update -----
            mom = beta * mom + (1.0 - beta) * g

            # ----- step size -----
            grad_norm = np.linalg.norm(mom) + 1e-12
            alpha = step0 / math.sqrt(it) / grad_norm

            # ----- candidate and projection -----
            cand = cur - alpha * mom
            cand = _project_simplex(cand)

            # ----- evaluate -----
            cand_sc = _fast_score(cand, L, n)

            # ----- accept / record -----
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_sc, best_seq = cand_sc, cand.copy()
                no_improve = 0
            else:
                # occasional uphill move (Metropolis‑like)
                if rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
                    cur, cur_sc = cand, cand_sc
                no_improve += 1

            # ----- temperature decay -----
            if eps > eps_min:
                eps = max(eps_min, eps * eps_decay)

            # ----- occasional random perturbation when stuck -----
            if no_improve > 2000:
                # draw a new random Dirichlet‑type vector and blend a bit
                rnd = rng.exponential(1.0, size=n)
                rnd = rnd / rnd.sum()
                cur = 0.7 * cur + 0.3 * rnd
                cur = _project_simplex(cur)
                mom = np.zeros_like(cur)
                no_improve = 0
                cur_sc = _fast_score(cur, L, n)
                if cur_sc < best_sc:
                    best_sc, best_seq = cur_sc, cur.copy()

            # ----- symmetrisation every few thousand steps (helps for symmetric problems) -----
            if it % 2500 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _fast_score(sym, L, n)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_seq = sym_sc, sym.copy()
                # reset momentum after projection
                mom = np.zeros_like(cur)

            # ----- early exit if target reached -----
            if best_sc < 1.50274349:
                break

        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Fallback: uniform if nothing better found (should never happen)
    # --------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------
    # Return the best vector as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_seq]
