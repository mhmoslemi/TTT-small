def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.5

    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate random initial sequence: exponential or uniform
        a = np.exp(-np.arange(2000) / 30.0)
    a = np.maximum(a, 0)
    # project onto simplex (sum=1)
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()
    n = a.size
    L = 1 << ((3 * n - 2).bit_length())

    best_vec = a.copy()
    best_sc = _score(best_vec)

    cur = a.copy()
    cur_sc = best_sc

    # parameters
    step = 0.5
    eps = 0.3
    eps_decay = 0.995
    min_eps = 1e-8
    # Adam variables
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    t_iter = 0

    while time.monotonic() < deadline:
        t_iter += 1
        # compute convolution via FFT
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        # softmax weights
        eps = max(min_eps, eps * eps_decay)
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        cur = cur - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        # project onto simplex
        cur = _project_simplex(cur)

        cur_sc = _score(cur)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()

        # periodic symmetrization
        if t_iter % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:

                best_vec = sym.copy()
                cur = sym.copy()
                cur_sc = sym_sc
        # random perturbation if stuck

            for _ in range(5):

                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    cur = tmp.copy()
                    cur_sc = tmp_sc

import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    # find the rho index
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to guard against tiny numerical drift
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


# ------------------------------------------------------------
# Fast score: 2 * n * max(conv(a, a))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int = None) -> float:
    """Assumes sum(a)=1. Returns 2*n*max(conv)."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # keep a small safety margin for the outer harness

    a = np.maximum(a, 0.0)

    n = a.shape[0]

    # -------------------- optimisation hyper‑parameters --------------------
    step_size = 0.25               # base learning rate for Adam

    eps_decay = 0.995
    eps_min = 1e-8

    # Adam moments
    m = np.zeros_like(cur)
