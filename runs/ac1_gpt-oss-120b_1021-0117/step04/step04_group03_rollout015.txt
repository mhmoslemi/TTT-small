import time

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    return float(2.0 * n * conv.max())

        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        # step size (scale‑invariant)
        grad_norm = np.linalg.norm(momentum) + 1e-12
        alpha = step0 / (np.sqrt(it) * grad_norm)

        # candidate update & projection onto simplex
        cand = y - alpha * momentum
        cand = _project_simplex(cand)

        # enforce symmetry (helps the bound)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc + 1e-12 < cur_sc:
            prev = cur
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
            no_improve = 0
        else:
            no_improve += 1
            # reset momentum after long stagnation
            if no_improve > 8000:
                momentum = np.zeros_like(cur)
                no_improve = 0

        # occasional random mass‑transfer when stuck
        if no_improve > 3000:
            tmp = cur.copy()
            for _ in range(5):
                i_idx = rng.integers(0, n)
                if tmp[i_idx] > 0.0:
                    delta = tmp[i_idx] * rng.random() * 0.4
                    j_idx = rng.integers(0, n)
                    if j_idx == i_idx:
                        j_idx = (j_idx + 1) % n
                    tmp[i_idx] -= delta
                    tmp[j_idx] += delta
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc + 1e-12 < cur_sc:
                prev = cur
                cur = tmp
                cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        # early exit if we reach the target bound
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # --------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    return [float(x) for x in best_vec]
