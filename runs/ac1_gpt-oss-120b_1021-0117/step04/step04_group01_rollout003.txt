import time
import math
import numpy as np

# ---------- Simplex projection ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # In the degenerate case all zeros, return uniform distribution
    if w.sum() == 0.0:
        w = np.full_like(v, 1.0 / n)
    return w

# ---------- Fast score (2*n*max(conv)) ----------
def _score(a: np.ndarray, fft_len: int) -> float:
    """Compute 2*n*max(convolution) for a simplex‑normalised vector."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())

# ---------- Main entry ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # safety margin

    # ----- initialise -----
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: moderate geometric decay
        n_fallback = 2000
        r = 0.996
        a = np.power(r, np.arange(n_fallback), dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    n = a.size
    # FFT length covering both full convolution (2n-1) and mask‑rev‑a convolution (3n-2)
    fft_len = 1 << ((max(2 * n - 1, 3 * n - 2)).bit_length())

    best_vec = a.copy()
    best_sc = _score(best_vec, fft_len)

    cur = a.copy()
    cur_sc = best_sc

    # ----- hyper‑parameters -----
    step0 = 0.5               # initial step magnitude
    max_backtrack = 12
    backtrack_factor = 0.5
    no_improve = 0
    it = 0

    # ----- optimisation loop -----
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution of current iterate ----
        a_fft = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()

        # ---- subgradient of max(conv) ----
        # mask for all entries attaining the maximal value (within tolerance)
        atol = max(1e-14, 1e-12 * max_val)
        mask_bool = np.isclose(conv, max_val, atol=atol, rtol=1e-12)
        mask_sum = mask_bool.sum()
        if mask_sum == 0:
            # fallback: single argmax
            mask_weights = np.zeros_like(conv)
            mask_weights[conv.argmax()] = 1.0
        else:
            mask_weights = mask_bool.astype(np.float64) / float(mask_sum)

        # gradient = 2 * (mask_weights ⋆ cur)  (cross‑correlation)
        mask_fft = np.fft.rfft(mask_weights, n=fft_len)
        rev_cur = cur[::-1]
        rev_fft = np.fft.rfft(rev_cur, n=fft_len)
        corr_full = np.fft.irfft(mask_fft * rev_fft, n=fft_len)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # slice of length n

        # ---- line search ----
        step = step0 / math.sqrt(it)
        improved = False
        for _ in range(max_backtrack):
            cand = cur - step * grad
            cand = _project_simplex(cand)
            cand_sc = _score(cand, fft_len)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_vec, best_sc = cand.copy(), cand_sc
                    no_improve = 0
                improved = True
                break
            step *= backtrack_factor

        if not improved:
            no_improve += 1

        # ----- occasional symmetrisation (often helpful) -----
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, fft_len)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, fft_len)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target achieved -----
        if best_sc < 1.50274349:
            break

    # ----- final fine‑tuning with tiny Gaussian noise -----
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.003, size=n)
        cand = best_vec + delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand, fft_len)
        if cand_sc < best_sc:
            best_vec, best_sc = cand, cand_sc

    # Return as plain Python list of floats
    return [float(x) for x in best_vec]
