def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)  # small safety margin

    # init
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate exponential decreasing distribution length 2000
        n = 2000
        a = np.exp(-np.arange(n) / 100.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= s

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())  # length for FFT

    # auxiliary functions
    def score(vec):
        # compute 2*n*max(conv) where sum(vec) == 1
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2*n - 1]
        return float(2.0 * n * conv.max())

    # initial best
    best_vec = a.copy()
    best_sc = score(best_vec)

    # Parameters
    # gradient descent phase
    max_iter_grad = 20000  # but stop by deadline
    step0 = 0.6
    beta = 0.5
    eps0 = 0.4
    eps_decay = 0.9994

    # track previous for momentum? Not needed

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # if we have exhausted gradient iterations beyond reasonable, break
        if it >= max_iter_grad:
            break

        # compute convolution and max index
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2*n - 1]
        max_val = conv.max()
        # get all indices with near max (within tolerance)
        # to avoid ties and noise
        tol = 1e-12
        max_idxs = np.where(conv >= max_val - tol)[0]
        # pick random among them
        k = rng.choice(max_idxs)

        # gradient of max (subgradient) corresponds to 2 * a shifted by k
        g = np.zeros_like(a)
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        idx = np.arange(start_i, end_i + 1)
        g[idx] = 2.0 * a[k - idx]

        # step size schedule
        step = step0 / ((it + 1) ** beta)

        # line search
        improved = False
        cand = None
        cand_sc = None
        for back in range(12):  # up to 12 backtracks
            cand_vec = a - step * g
            # projection onto simplex
            cand_vec = _project_simplex(cand_vec)
            cand_sc = score(cand_vec)
            if cand_sc < best_sc or cand_sc < score(a):
                cand = cand_vec
                improved = True
                break
            step *= 0.5  # reduce step

        if improved:
            a = cand
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random mass transfer if stuck
        if no_improve > 1500:
            # random pair transfer with a random fraction
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            # amount to move
            delta = a[i] * rng.random() * 0.5
            if delta > 0:
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)  # ensures sum =1
                tmp_sc = score(tmp)
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                if tmp_sc < score(a):
                    a = tmp
                no_improve = 0

        # early exit if target reached
        if best_sc < 1.50274349:
            break

        it += 1

    # After gradient phase we do random local search (simulated annealing)
    # Use decreasing temperature for delta size and acceptance

    temperature = 0.1
    while time.monotonic() < deadline:
        # propose random move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1

        # choose delta proportion of mass from i to j
        max_move = a[i] * 0.8
        delta = max_move * rng.random()
        if delta <= 0:
            continue
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep nonneg by projecting onto simplex (or simply ensure >=0)
        cand = _project_simplex(cand)
        cand_sc = score(cand)
        # acceptance: if better, accept; else accept with probability exp(-(cand_sc - best_sc)/temperature)
        if cand_sc < best_sc or rng.random() < math.exp(-(cand_sc - best_sc) / max(temperature, 1e-9)):
            best_vec, best_sc = cand, cand_sc
            # also update current a to continue exploring around it
            a = cand
        # cool down temperature slowly
        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    # Return best solution as list
    return [float(x) for x in best_vec]

def _project_simplex(v: np.ndarray) -> np.ndarray:

    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
