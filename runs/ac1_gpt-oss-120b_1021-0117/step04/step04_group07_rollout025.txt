def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / a.size)

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())

    def fast_score(arr):
        af = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    cur = a.copy()
    cur_sc = fast_score(cur)
    best = cur.copy()
    best_sc = cur_sc

    eps = 0.01
    eps_min = 1e-7
    eps_decay = 0.9996

    eta0 = 0.7
    it = 0
    stagn = 0

    while time.monotonic() < deadline:
        it += 1
        # FFT of current a
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        maxc = conv.max()
        # Softmax weights
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            eps = max(eps_min, eps * eps_decay)
        else:
            # at very low eps, use hard max indicator
            w = np.zeros_like(conv)
            w[int(conv.argmax())] = 1.0
        # compute gradient via convolution with reversed weights
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # step size
        eta = eta0 / math.sqrt(it)
        # exponentiated gradient update (multiply then renorm)
        # clip for numeric stability
        delta_exp = np.exp(-eta * g)
        # avoid overflow/underflow
        delta_exp = np.clip(delta_exp, 1e-12, 1e12)
        cand = cur * delta_exp
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand /= cand_sum

        cand_sc = fast_score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            stagn = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # occasionally accept worse to escape local minima
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (eta + 1e-12)):
                cur, cur_sc = cand, cand_sc
            stagn += 1

        # random mass transfer when stagnated
        if stagn > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                cand2 = cur.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2_sum = cand2.sum()
                if cand2_sum != 0.0:
                    cand2 /= cand2_sum
                cand2_sc = fast_score(cand2)
                if cand2_sc < cur_sc:
                    cur, cur_sc = cand2, cand2_sc
                if cand2_sc < best_sc:
                    best, best_sc = cand2.copy(), cand2_sc
                stagn = 0

        # occasional symmetrisation? maybe skip.

        # early exit if reached target
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]

import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (sum is assumed to be 1)
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑negative sequence that (hopefully) beats the target bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # reserve a tiny margin

    # --------------------------------------------------------
    # Initialise from parent if it exists, otherwise uniform.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        cur = np.full(n0, 1.0 / n0, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s > 0.0:
        cur /= s
    else:
        cur = np.full_like(cur, 1.0 / cur.size)

    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())   # length for FFT (power of two)

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Parameters for the optimisation loops
    # --------------------------------------------------------
    eps = 0.02                # soft‑max temperature
    eps_min = 1e-8
    eps_decay = 0.9997

    step0 = 0.6               # base step size for projected GD
    beta = 0.9                # momentum factor
    momentum = np.zeros_like(cur)

    it = 0
    stagn = 0                 # iterations without improvement

    # --------------------------------------------------------
    # Main projected‑gradient‑with‑momentum loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and max (used for the soft‑max) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- soft‑max weights (or hard max when eps tiny) -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            eps = max(eps_min, eps * eps_decay)
        else:
            w = np.zeros_like(conv)
            w[int(conv.argmax())] = 1.0

        # ----- gradient of the smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)       # slice that yields ∇_a
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (decreasing like 1/√t) -----
        alpha = step0 / math.sqrt(it)

        # ----- candidate via Euclidean step + projection -----
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)

        cand_sc = _fast_score(cand, L, n)

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            stagn = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # occasional uphill move (simulated‑annealing flavour)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
                cur, cur_sc = cand, cand_sc
            stagn += 1

        # ----- if stuck, perform a random mass‑transfer move -----
        if stagn > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _fast_score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                stagn = 0

        # ----- early exit if we already beat the target -----
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Final local random refinement (hill‑climbing) phase
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * 0.3
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _fast_score(cand, L, n)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc

    # --------------------------------------------------------
    # Return the best vector found as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
