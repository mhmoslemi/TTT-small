def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5) # leave 0.5 sec margin

    def project(v):
        return _project_simplex(v)

    def score(a):
        # compute 2*n*max(conv)
        n = a.shape[0]
        L = 1 << ((3 * n - 2).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af*af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    best_vec = None
    best_sc = float('inf')

    # optimization routine
    def run_opt(init_vec):
        nonlocal best_vec, best_sc
        cur = init_vec.copy()
        n = cur.size
        L = 1 << ((3 * n - 2).bit_length())
        momentum = np.zeros_like(cur)
        # hyperparameters
        step0 = 2.0
        beta = 0.9
        eps0 = 5.0
        eps_decay = 0.9995
        eps_min = 1e-8
        it = 0
        cur_sc = score(cur)
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc

        while True:
            # time check
            if time.monotonic() >= deadline:
                break
            it += 1

            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[:2*n-1]
            max_val = conv.max()
            # smoothing weight
            eps = max(eps_min, eps0 * (eps_decay ** it))
            # compute w
            # avoid overflow by subtract max_val
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # compute gradient
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2*n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
            # momentum
            momentum = beta * momentum + (1.0 - beta) * g

            # step size
            base_alpha = step0 / math.sqrt(it)
            # ensure nonzero maybe
            alpha = base_alpha
            cand = None
            cand_sc = None
            # backtracking up to 12 steps
            for _ in range(12):
                cand_tmp = cur - alpha * momentum
                cand_tmp = project(cand_tmp)
                cand_tmp_sc = score(cand_tmp)
                if cand_tmp_sc <= cur_sc:
                    cand, cand_sc = cand_tmp, cand_tmp_sc
                    break
                alpha *= 0.5
            if cand is None:
                cand, cand_sc = cand_tmp, cand_tmp_sc

            # accept
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_vec, best_sc = cand.copy(), cand_sc
            else:
                # uphill with small probability
                prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc

            # regular symmetrisation more often
            if it % 200 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = project(sym)
                sym_sc = score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_vec, best_sc = sym.copy(), sym_sc

            # occasional random mass transfer
            if it % 1500 == 0:
                # random perturbation small
                i = rng.integers(0, n)
                if cur[i] > 0:
                    delta = cur[i] * rng.random() * 0.3
                    j = rng.integers(0, n)
                    if j == i:
                        j = (j+1) % n
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = project(tmp)
                    tmp_sc = score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_vec, best_sc = tmp.copy(), tmp_sc

            # early stop if good enough
            if best_sc < 1.50274349:
                break

    # initial vector(s)
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        init = np.maximum(init, 0.0)
        init = _project_simplex(init)
        # symmetrical
        init = (init + init[::-1]) * 0.5
        init = _project_simplex(init)
        run_opt(init)
    else:
        # random init
        for _ in range(3):
            a = rng.random(2048)
            a = _project_simplex(a)
            run_opt(a)
            if time.monotonic() >= deadline:
                break

    # final fallback ensure we have something
    if best_vec is None:
        # produce simple uniform simplex
        n = 1024
        best_vec = np.full(n, 1.0/n)
        best_sc = score(best_vec)

    return [float(x) for x in best_vec]

def _score(a):
   n = a.shape[0]
   L = 1 << ((3 * n - 2).bit_length())
   af = np.fft.rfft(a, n=L)
   conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
   return float(2.0 * n * conv.max())

def _arcsine_init(n):
   i = np.arange(n, dtype=np.float64)
   vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
   # ensure finite
   vals = np.where(np.isfinite(vals), vals, 0.0)
   return _project_simplex(vals)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_vec = None
    best_sc = float('inf')

    def update_best(vec, sc):
        nonlocal best_vec, best_sc
        if sc < best_sc:
            best_vec = vec.copy()
            best_sc = sc

    def run_opt(init_vec):
        nonlocal best_vec, best_sc
        cur = init_vec.copy()
        n = cur.size
        L = 1 << ((3 * n - 2).bit_length())
        momentum = np.zeros_like(cur)

        step0 = 2.0
        beta = 0.9
        eps0 = 5.0
        eps_decay = 0.9995
        eps_min = 1e-8

        cur_sc = _score(cur)
        update_best(cur, cur_sc)

        it = 0
        while True:
            if time.monotonic() >= deadline:
                break
            it += 1

            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # smoothing weights
            eps = max(eps_min, eps0 * (eps_decay ** it))
            # compute exponent safely
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]

            # momentum
            momentum = beta * momentum + (1.0 - beta) * g

            # step size
            base_alpha = step0 / math.sqrt(it)
            alpha = base_alpha
            cand = None
            cand_sc = None
            # backtracking line search
            for _ in range(12):
                cand_tmp = cur - alpha * momentum
                cand_tmp = _project_simplex(cand_tmp)
                cand_tmp_sc = _score(cand_tmp)
                if cand_tmp_sc <= cur_sc:
                    cand, cand_sc = cand_tmp, cand_tmp_sc
                    break
                alpha *= 0.5
            if cand is None:
                cand, cand_sc = cand_tmp, cand_tmp_sc

            # accept/reject
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                update_best(cur, cur_sc)
            else:
                prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc

            # symmetrisation
            if it % 200 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    update_best(cur, cur_sc)

            # random perturbation
            if it % 1500 == 0:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.3
                    j = rng.integers(0, n)
                    if j == i:
                        j = (j + 1) % n
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        update_best(cur, cur_sc)

            # early exit if target reached
            if best_sc < 1.50274349:
                break

    # Build initial vectors
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.maximum(base, 0.0)
        base = _project_simplex(base)
        # arcsine shape same length
        arc = _arcsine_init(base.size)
        init = (base + arc) * 0.5
        init = _project_simplex(init)
        run_opt(init)
    else:
        # choose a length
        n0 = 2048
        init = _arcsine_init(n0)
        run_opt(init)

    # fallback if nothing found
    if best_vec is None:
        # uniform simplex of length 1024
        n_fallback = 1024
        best_vec = np.full(n_fallback, 1.0 / n_fallback)
        best_sc = _score(best_vec)

    # convert to plain list of floats
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Score used by the evaluator (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Produce a symmetric “arcsine‑like” init with exponent α
# ------------------------------------------------------------
def _init_power(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    # avoid log(0); all entries are positive for 0≤i≤n‑1
    vals = np.exp(-alpha * np.log(denom))
    return _project_simplex(vals)


# ------------------------------------------------------------
# Main optimisation routine (matrix‑free, O(n log n) per step)
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    best_vec = None
    best_sc = float('inf')

    def _update_best(vec, sc):
        nonlocal best_vec, best_sc
        if sc < best_sc:
            best_vec = vec.copy()
            best_sc = sc

    # --------------------------------------------------------
    # Determine problem size and construct a handful of seeds
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(np.asarray(height_sequence_1).size)
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.maximum(base, 0.0)
        base = _project_simplex(base)
    else:
        n = 2048
        base = None

    # a few analytical initialisations
    init_candidates = []
    # arcsine (α=0.5) and neighbours
    for a in (0.3, 0.4, 0.5, 0.6, 0.7):
        init_candidates.append(_init_power(n, a))
    # plain uniform simplex
    init_candidates.append(np.full(n, 1.0 / n))
    # if a parent is present, mix it with a symmetric arcsine shape
    if base is not None:
        arc = _init_power(n, 0.5)
        mixed = _project_simplex((base + arc) * 0.5)
        init_candidates.append(mixed)

    # evaluate seeds and keep the best one
    for cand in init_candidates:
        sc = _score(cand)
        if sc < best_sc:
            best_vec, best_sc = cand.copy(), sc

    # early exit if we already beat the target
    target = 1.50274349
    if best_sc < target:
        return [float(x) for x in best_vec]

    # --------------------------------------------------------
    # Optimisation loop (projected momentum descent)
    # --------------------------------------------------------
    def _optimize(start_vec):
        nonlocal best_vec, best_sc
        cur = start_vec.copy()
        cur_sc = _score(cur)
        _update_best(cur, cur_sc)

        L = 1 << ((3 * cur.size - 2).bit_length())
        momentum = np.zeros_like(cur)

        step0 = 2.0          # base step magnitude
        beta = 0.9           # momentum factor
        eps0 = 5.0           # initial smoothing
        eps_decay = 0.9995
        eps_min = 1e-8

        it = 0
        while time.monotonic() < deadline:
            it += 1

            # ----- current convolution -----
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * cur.size - 1]
            max_val = conv.max()

            # ----- soft‑max weights -----
            eps = max(eps_min, eps0 * (eps_decay ** it))
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ----- gradient of the smoothed max -----
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * cur.size - 2) - np.arange(cur.size)
            g = 2.0 * grad_full[idx]

            # ----- momentum update -----
            momentum = beta * momentum + (1.0 - beta) * g

            # ----- step size with back‑tracking -----
            base_alpha = step0 / math.sqrt(it)
            alpha = base_alpha
            cand = None
            cand_sc = None
            cand_tmp = None
            cand_tmp_sc = None
            for _ in range(12):
                cand_tmp = cur - alpha * momentum
                cand_tmp = _project_simplex(cand_tmp)
                cand_tmp_sc = _score(cand_tmp)
                if cand_tmp_sc <= cur_sc:
                    cand, cand_sc = cand_tmp, cand_tmp_sc
                    break
                alpha *= 0.5
            if cand is None:
                cand, cand_sc = cand_tmp, cand_tmp_sc

            # ----- accept / reject -----
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                _update_best(cur, cur_sc)
            else:
                prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc

            # ----- occasional symmetrisation -----
            if it % 200 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    _update_best(cur, cur_sc)

            # ----- random mass‑transfer to escape plateaus -----
            if it % 1500 == 0:
                i = rng.integers(0, cur.size)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.3
                    j = rng.integers(0, cur.size)
                    if j == i:
                        j = (j + 1) % cur.size
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        _update_best(cur, cur_sc)

            # ----- early exit -----
            if best_sc < target:
                break

    # --------------------------------------------------------
    # Run optimisation starting from the best seed found above
    # --------------------------------------------------------
    _optimize(best_vec)

    # --------------------------------------------------------
    # If time permits try a fresh random restart (jittered best)
    # --------------------------------------------------------
    if time.monotonic() < deadline and best_sc >= target:
        jitter = best_vec * (1.0 + 0.05 * (rng.random(best_vec.size) - 0.5))
        jitter = _project_simplex(jitter)
        _optimize(jitter)

    # --------------------------------------------------------
    # Fallback: a uniform simplex if nothing else succeeded
    # --------------------------------------------------------
    if best_vec is None:
        n_fallback = 1024
        best_vec = np.full(n_fallback, 1.0 / n_fallback)

    return [float(x) for x in best_vec]
