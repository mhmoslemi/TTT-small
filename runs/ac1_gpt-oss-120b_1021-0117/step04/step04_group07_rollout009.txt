import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound assuming sum = 1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2 * n * max(a * a) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of smoothed max using softmax weights
# ------------------------------------------------------------
def _softmax_grad(a: np.ndarray, eps: float, L: int):
    """Return (grad, max_conv) where grad is the smoothed sub‑gradient."""
    n = a.size
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    maxc = conv.max()
    # softmax weights
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # convolution a * reverse(w)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)          # selects proper slice
    g = 2.0 * grad_full[idx]
    return g, maxc


# ------------------------------------------------------------
# Sub‑gradient at the (non‑smoothed) max location
# ------------------------------------------------------------
def _subgrad_at_max(a: np.ndarray, conv: np.ndarray):
    """Return sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    k = int(conv.argmax())
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k + 1, n)
    idx = np.arange(start, end, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g, conv.max()


# ------------------------------------------------------------
# Simple semicircle initialization (fallback)
# ------------------------------------------------------------
def _semicircle_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    c = (n - 1) / 2.0
    r = c
    vals = np.sqrt(np.maximum(0.0, 1.0 - ((i - c) / r) ** 2))
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / vals.sum()
    return vals


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients with a low AC1 bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a0 = _semicircle_init(n_fallback)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                       # enforce simplex

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())   # power‑of‑two >= 3n‑2

    cur = a0.copy()
    cur_sc = _fast_score(cur, L, n)

    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    eps = 0.01                 # initial smoothing
    eps_min = 1e-9
    eps_decay = 0.99           # faster decay than before
    step0 = 0.8                # base step size
    it = 0
    no_improve = 0
    target_score = 1.50274349

    # --------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- compute gradient -----
        if eps > eps_min:
            g, maxc = _softmax_grad(cur, eps, L)
            eps = max(eps_min, eps * eps_decay)
        else:
            # compute raw convolution once for sub‑gradient
            a_fft = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            g, maxc = _subgrad_at_max(cur, conv)

        # ----- step size -----
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / (math.sqrt(it) * grad_norm)

        # ----- two candidate updates -----
        cand1 = cur - alpha * g
        cand1 = _project_simplex(cand1)

        exp_factor = np.exp(-alpha * g)
        cand2 = cur * exp_factor
        cand2_sum = cand2.sum()
        if cand2_sum <= 0.0:
            cand2 = np.full_like(cand2, 1.0 / n)
        else:
            cand2 = cand2 / cand2_sum

        # ----- evaluate candidates -----
        sc1 = _fast_score(cand1, L, n)
        sc2 = _fast_score(cand2, L, n)

        # pick best among cur, cand1, cand2
        if sc1 <= cur_sc and sc1 <= sc2:
            nxt, nxt_sc = cand1, sc1
        elif sc2 <= cur_sc:
            nxt, nxt_sc = cand2, sc2
        else:
            nxt, nxt_sc = cur, cur_sc

        # ----- update current -----
        if nxt_sc < cur_sc - 1e-12:
            no_improve = 0
        else:
            no_improve += 1
        cur, cur_sc = nxt, nxt_sc

        # ----- record best -----
        if cur_sc < best_sc:
            best, best_sc = cur.copy(), cur_sc
            no_improve = 0

        # ----- occasional symmetrisation -----
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- random perturbation when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _fast_score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc <= target_score:
            break

    # --------------------------------------------------------
    # Return best found as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
