import time

import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (sum of coefficients assumed 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(conv(a,a)) where conv is linear convolution."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of smoothed max (soft‑max) via FFT
# ------------------------------------------------------------
def _gradient(a: np.ndarray, eps: float, L: int) -> np.ndarray:
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    m = conv.max()
    w = np.exp((conv - m) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]
