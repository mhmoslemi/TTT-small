    # function to init vector

        # sample exponential random vector and project onto simplex
        v = rng.exponential(scale=1.0, size=n)
        return _project_simplex(v)

    # Initialize best
    if "height_sequence_1" in globals():

        cur = random_init()
    best = cur.copy()
    best_sc = _score(cur)

    # optimization loops
    L = 1 << ((3 * n - 2).bit_length()) # for conv with reversed w
    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1=0.9; beta2=0.999; eps=1e-8
    step0=0.6
    eps0=0.6; eps_decay=0.9999; eps_min=1e-6
    it=0
    no_improve=0

    while time.monotonic() < deadline:
        it+=1
        # compute conv of cur with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # smoothing weight
        eps_cur = max(eps_min, eps0 * (eps_decay**it))
        # compute weights as softmax over top entries; optionally use top K
        # Here compute full softmax:
        w = np.exp((conv - max_val) / eps_cur)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        # enforce symmetry
        cand = (cand + cand[::-1]) * 0.5

        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast score = 2 * n * max(convolution) (sum assumed 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a0 = np.exp(-np.arange(n0) / 200.0)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)                     # sum == 1, non‑negative
    n = a0.shape[0]

    # FFT length – enough for a with a and with a reversed
    L = 1 << ((3 * n - 2).bit_length())

    cur = a0.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # ----- optimisation hyper‑parameters ------------------------------
    step0 = 0.8                # base step size
    eps0 = 1.0                 # initial smoothing for log‑sum‑exp
    eps_decay = 0.9995
    eps_min = 1e-6
    momentum = 0.9
    vel = np.zeros_like(cur)   # momentum buffer
    no_improve = 0
    it = 0

    # ----- main loop -------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # linear convolution a*a (full length)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smoothed max via soft‑max
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)               # slice for gradient
        g = 2.0 * grad_full[idx]

        # normalised step

        # heavy‑ball update
