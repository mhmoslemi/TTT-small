def _project_simplex(v):
    # same as before
    n = v.shape[0]
    u = np.sort(v)[::-1]

    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]

    if rho.size == 0:

    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a short safety margin before the external timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise – use the provided parent if present, otherwise uniform
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a0 = np.full(n0, 1.0 / n0, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                      # normalise to the simplex

    n = a0.size
    # FFT length sufficient for convolutions up to length 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    cur = a0.copy()
    a_fft = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    cur_sc = float(2.0 * n * conv.max())
    best = cur.copy()
    best_sc = cur_sc

    # hyper‑parameters
    eps = 0.01               # initial smoothing for the soft‑max
    eps_min = 1e-9
    eps_decay = 0.9995
    step0 = 0.6              # base step size
    beta = 0.9               # momentum coefficient
    m = np.zeros_like(cur)   # momentum accumulator
    it = 0
    no_improve = 0
    target = 1.50274349      # stop when we beat the target

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of the current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- gradient of the (smoothed) max -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
            eps = max(eps_min, eps * eps_decay)
        else:
            k = int(conv.argmax())
            g = np.zeros_like(cur)
            start_i = max(0, k - (n - 1))
            end_i = min(k + 1, n)
            idx_range = np.arange(start_i, end_i)
            g[idx_range] = 2.0 * cur[k - idx_range]

        # ----- momentum and step -----
        m = beta * m + (1.0 - beta) * g
        step = step0 / math.sqrt(it) / (np.linalg.norm(m) + 1e-12)

        # ----- candidate update, projection onto simplex, symmetrisation -----
        cand = cur - step * m
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5

        # ----- evaluate candidate (fast, because sum=1) -----
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = float(2.0 * n * conv_cand.max())

        # ----- accept/reject (with occasional uphill moves) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (step + 1e-12)):
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- random mass‑transfer when stuck for a while -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp_fft = np.fft.rfft(tmp, n=L)
                conv_tmp = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = float(2.0 * n * conv_tmp.max())
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if we already beat the target -----
        if best_sc < target:
            break

    # return plain Python list
    return [float(x) for x in best]
