import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    """2 * n * max(convolution) for a non‑negative sequence a."""

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        # ----- convolution of the current iterate -----------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smoothed max (log‑sum‑exp) weights -----------------------
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # ----- gradient of the smoothed objective -----------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # relevant slice (see original reference implementation)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur + step
        cand = _project_simplex(cand)

        cand_score = _score_seq(cand, L)

        # ----- keep the best ever ---------------------------------------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        # ----- accept/reject (simulated‑annealing style) ---------------
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
        else:
            diff = cand_score - cur_score
            prob = math.exp(-diff / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # ----- occasional random mass‑transfer when stuck -------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_seq(tmp, L)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
                no_improve = 0

        # ----- periodic symmetrisation (helps stability) ---------------
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_seq(sym, L)
            if sym_score < best_score:
                best, best_score = sym.copy(), sym_score
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score

        # ----- early exit if we have already beaten the target ----------
        if best_score < 1.50274349:
            break

        # ----- safety check ------------------------------------------------
        if time.monotonic() + 0.5 >= deadline:
            break

    # return a plain Python list
    return [float(x) for x in best]
