import time, math, numpy as np

# ---------- convolution helpers ----------
def _fft_len(n: int) -> int:
    """Length of FFT that fits a convolution of a (len n) with a (len 2n‑1)."""
    return 1 << ((3 * n - 2).bit_length())   # >= n+(2n‑1)-1

def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (full length 2n‑1) using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv

def _grad_wrt_a(a: np.ndarray, conv: np.ndarray, eps: float, L: int) -> np.ndarray:
    """Gradient of the smoothed max‐convolution w.r.t. a."""
    max_val = conv.max()
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    # convolution a * rev(w)
    rev_w = w[::-1]
    a_fft = np.fft.rfft(a, n=L)
    w_fft = np.fft.rfft(rev_w, n=L)
    full = np.fft.irfft(a_fft * w_fft, n=L)      # length >= 3n‑2
    idx = (2 * a.size - 2) - np.arange(a.size)   # slice for gradient entries
    return 2.0 * full[idx]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the target AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # ----- initial vector (simplex normalised) -----
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:                                   # fallback: uniform vector
        n = 2000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    n = a0.size
    L = _fft_len(n)

    # ----- hyper‑parameters -----
    eps0 = 0.6            # initial smoothing
    eps_decay = 0.9997
    eps_min = 1e-12
    eta0 = 12.0           # initial stepsize for exponentiated descent
    eta_decay = 4e-4
    target = 1.50274349

    cur = a0.copy()
    best = cur.copy()
    # initial score (objective = 2·n·max(conv) because sum(cur)=1)
    conv = _conv_full(cur, L)
    best_score = 2.0 * n * conv.max()

    eps = eps0
    it = 0
    no_improve = 0

    # ----- main optimisation loop -----
    while time.monotonic() < deadline:
        it += 1

        # convolution & current score
        conv = _conv_full(cur, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best = cur_score, cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_score < target:
            break

        # smoothing schedule
        eps = max(eps_min, eps * eps_decay)

        # gradient of smoothed maximum
        g = _grad_wrt_a(cur, conv, eps, L)

        # project gradient onto the simplex tangent (subtract mean w.r.t. cur)
        mu = np.dot(g, cur)
        g_proj = g - mu

        # exponentiated‑gradient update (multiplicative)
        eta = eta0 / (1.0 + it * eta_decay)
        cur = cur * np.exp(-eta * g_proj)

        # renormalise to the simplex (guarantees positivity)
        cur_sum = cur.sum()
        if cur_sum == 0.0:
            cur = np.full_like(cur, 1.0 / cur.size)
        else:
            cur /= cur_sum

        # occasional symmetrisation (helps with the even‑odd pattern)
        if it % 4000 == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur /= cur.sum()

        # simple random mass‑transfer when stuck
        if no_improve > 6000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                cur[i] -= delta
                cur[j] += delta
                cur = np.maximum(cur, 0.0)
                cur /= cur.sum()
            no_improve = 0

    # ----- final random fine‑tuning (use remaining time) -----
    while time.monotonic() < deadline - 0.03:
        # Gaussian perturbation around the best found vector
        sigma = 0.008
        cand = best + rng.normal(scale=sigma, size=n)

        cand_sum = cand.sum()

        conv = _conv_full(cand, L)
        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score, best = cand_score, cand.copy()

    # return as plain list of floats
    return [float(x) for x in best]
