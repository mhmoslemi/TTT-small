import time
import math
import numpy as np

# ------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation (assumes sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution) for a vector with sum == 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# One‑step smoothed‑max gradient
# ------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of 2*n*softmax(conv) w.r.t. a."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    # softmax weights on the convolution entries
    w = np.exp((conv - max_val) / eps)
    w /= w.sum()
    # reverse weights to obtain correct correlation gradient
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    # extract the n‑dimensional gradient component
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g
