import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound assuming sum(sequence)=1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(a*a) under the simplex normalization."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initialise a feasible vector (use parent if present)
    # --------------------------------------------------------

    else:

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 /= s  # normalize to simplex

    n = a0.size
    # FFT length sufficient for full linear convolution (3n‑2 points)
    L = 1 << ((3 * n - 2).bit_length())

    cur = a0.copy()
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------

    eps = 0.05           # soft‑max temperature

    eps_decay = 0.9995

    no_improve = 0

    # --------------------------------------------------------

    while time.monotonic() < deadline:

        maxc = conv.max()

        # ----- gradient of the smoothed max -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)

            if w_sum == 0.0:
