def _run_descent(init_vec, L, deadline):
   cur = init_vec.copy()
   cur_sc = _score(cur)
   best = cur_sc
   best_vec = cur.copy()
   # parameters
   eps = 10.0
   eps_decay = 0.9995
   eps_min = 1e-4
   step0 = 1.0
   beta = 0.9  # momentum factor
   m = np.zeros_like(cur)
   it = 0
   while time.monotonic() < deadline:
        # compute gradient as before but using w smoothed weights.
        # compute FFT conv for cur
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*len(cur)-1]
        max_val = conv.max()
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        w /= w_sum
        # compute gradient: cross-correlation of cur with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*len(cur)-2) - np.arange(len(cur))
        g = 2 * grad_full[idx]
        # momentum
        m = beta * m + (1-beta) * g
        grad_norm = np.linalg.norm(m) + 1e-12
        alpha = step0 / math.sqrt(it+1) / grad_norm
        cand = cur - alpha * m
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best:
                best = cand_sc
                best_vec = cand.copy()
        else:
            # if no improvement for certain steps maybe random restart or perturb
            pass
        it += 1
        # optional early break
        if best < 1.50274349:
            break
   return best_vec, best
```

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector assumed to sum to 1 (exact evaluator)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(conv)  (sum(a) is assumed 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
