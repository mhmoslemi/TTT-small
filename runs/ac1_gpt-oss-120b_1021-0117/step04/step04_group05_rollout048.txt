import time, math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper to get next power of two (FFT length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

                cand_bt = _project_simplex(cand_bt)
                cand_bt_fft = rfft(cand_bt, n=L_full)
                cand_bt_conv = irfft(cand_bt_fft * cand_bt_fft, n=L_full)[: 2 * n - 1]
                cand_bt_sc = float(2.0 * n * cand_bt_conv.max())
                if cand_bt_sc < cur_sc:
                    cand = cand_bt
                    cand_fft = cand_bt_fft
                    cand_conv = cand_bt_conv
                    cand_sc = cand_bt_sc
                    improved = True
                    break

        if improved:
            # accept
            last_step = cand - cur
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_conv.max()
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # occasional probabilistic acceptance (simulated annealing flavour)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                last_step = cand - cur
                cur = cand
                cur_fft = cand_fft
                cur_conv = cand_conv
                cur_max = cand_conv.max()
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # ----- occasional random local perturbations when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.25
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = rfft(tmp, n=L_full)
                tmp_conv = irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_sc = float(2.0 * n * tmp_conv.max())
                if tmp_sc < cur_sc:
                    last_step = tmp - cur
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_sc = tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()

        # ----- light local refinement: tiny pairwise transfers -----
        for _ in range(3):
            i = rng.integers(0, n)
            if cur[i] <= 0.0:
                continue
            delta = cur[i] * rng.random() * 0.02
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_fft = rfft(tmp, n=L_full)
            tmp_conv = irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
            tmp_sc = float(2.0 * n * tmp_conv.max())
            if tmp_sc < cur_sc:
                last_step = tmp - cur
                cur = tmp
                cur_fft = tmp_fft
                cur_conv = tmp_conv
                cur_max = tmp_conv.max()
                cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0
                break

        # ----- periodic symmetrisation (helps with many runs) -----
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = rfft(sym, n=L_full)
            sym_conv = irfft(sym_fft * sym_fft, n=L_full)[: 2 * n - 1]
            sym_sc = float(2.0 * n * sym_conv.max())
            if sym_sc < cur_sc:
                last_step = sym - cur
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_conv.max()
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- early exit if we beat the target -----
        if best_sc < 1.50274349:
            break

        # ----- safety time check -----
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best candidate (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
