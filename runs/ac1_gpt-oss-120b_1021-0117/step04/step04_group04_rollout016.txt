# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

        delta = cur[i] * rng.random() * 0.32
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp, L)
        if tmp_sc < best_sc:
            best, best_sc = tmp.copy(), tmp_sc
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc

    # safeguard: if something went wrong, fall back to the initial simplex vector
    if best_sc == float('inf') or np.isnan(best_sc):
        best = a0.copy()

    return [float(x) for x in best]
