def _project_simplex(v):

    # shift to make nonneg

    n = v.shape[0]

    u = np.sort(v)[::-1]

    css = np.cumsum(u) - 1

    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]

    if len(rho) == 0:

        theta = 0

    else:

        rho = rho[-1]

        theta = css[rho] / (rho + 1)

    w = np.maximum(v - theta, 0)

    return w

import time

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# AC1 bound for a sequence already normalized to sum = 1
# ------------------------------------------------------------
def _score_simplex(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(convolution) for a probability vector a."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
