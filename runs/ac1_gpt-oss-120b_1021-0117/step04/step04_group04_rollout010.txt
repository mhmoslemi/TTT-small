def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # Determine n and initial sequence
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a0 = np.exp(-np.arange(n0) / 200.0)
    n = a0.shape[0]

    # Ensure sum=1: project onto simplex
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # Precompute FFT length for convolution (for score) and gradient (need 3n length)
    L_conv = 1 << ((2*n-1).bit_length())
    L_grad = 1 << ((3*n-2).bit_length())
    # Score function
    def calc_score(a):
        sum_a = a.sum()
        if sum_a < 1e-12: return np.inf
        af = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(af * af, n=L_conv)[:2*n-1]
        max_b = conv.max()
        return 2.0 * n * max_b / (sum_a*sum_a)

    best = a0.copy()
    best_sc = calc_score(best)
    cur = best.copy()
    cur_sc = best_sc

    # Hyperparameters
    eps = 0.8
    eps_decay = 0.9998
    eps_min = 1e-9
    step0 = 0.6
    it = 0
    no_improve = 0

    # Main loop
    while time.time() < deadline:
        it += 1

        # Compute convolution via FFT for gradient weighting
        a_fft = np.fft.rfft(cur, n=L_grad)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L_grad)[:2*n-1]
        max_val = conv_full.max()

        # Compute smoothed max weights
        eps_cur = max(eps_min, eps * (eps_decay ** it))
        w = np.exp((conv_full - max_val) / eps_cur)
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # Gradient w.r.t. objective: convolution of cur with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L_grad)
        # extract gradient indices: conv length for gradient is >= 3n-2, indexes from (2n-2) downto (n)
        # previous code: idx = (2 * n - 2) - np.arange(n)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Step size
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm

        # Candidate via gradient step
        cand = cur - alpha * g
        cand = _project_simplex(cand)
        cand_sc = calc_score(cand)

        # Accept if better; else SA
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # Simulated annealing probability
            temp = max(1e-6, 0.02 / math.sqrt(it))
            prob = math.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # Occasionally perform a large random mass transfer (big move) when stuck
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0:
                # move a random fraction up to half of its mass
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = calc_score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # Every 2000 iterations, prune small entries
        if it % 2000 == 0:
            # keep top entries that contain at least 0.99 of mass
            mass_target = 0.99
            sorted_idx = np.argsort(-cur)
            cumsum = np.cumsum(cur[sorted_idx])
            keep_len = np.searchsorted(cumsum, mass_target) + 1
            mask = np.zeros(n, dtype=bool)
            mask[sorted_idx[:keep_len]] = True
            cur_pruned = np.where(mask, cur, 0.0)
            cur_pruned = _project_simplex(cur_pruned)
            cur_sc = calc_score(cur_pruned)
            cur = cur_pruned
            if cur_sc < best_sc:
                best, best_sc = cur.copy(), cur_sc
                no_improve = 0

        # Symmetrisation occasional
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = calc_score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # Early exit if beat target
        if best_sc < 1.50274349:
            break

    # Return as Python list
    return [float(x) for x in best]

import time

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
