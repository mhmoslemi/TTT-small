def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)

        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_max
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        # occasional symmetrisation during refinement
        if rng.random() < 0.005:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_max = sym_conv.max()
            sym_score = 2.0 * n * sym_max
            if sym_score < cur_score:
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_max
                cur_score = sym_score
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()

        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
