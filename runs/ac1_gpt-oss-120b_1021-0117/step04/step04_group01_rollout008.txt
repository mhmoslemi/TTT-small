def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)  # margin

    # init a
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a = np.exp(-np.arange(n_fallback)/100.0)  # decaying exponential
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s  # normalise to sum = 1

    n = a.shape[0]
    L = 1 << ((3*n-2).bit_length())
    
    # scoring function
    def score(vec):
        # compute 2*n*max(conv)/sum^2; sum is 1 due to normalisation so denominator 1
        # we can compute directly for speed
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())  # sum=1
    best_vec = a.copy()
    best_sc = score(best_vec)
    
    # hyperparameters
    eps0 = 0.05
    eps_decay = 0.9998
    eps_min = 1e-12
    eta0 = 0.5
    # momentum not used for now
    # iteration counters
    it = 0
    no_improve = 0
    # main loop
    while time.monotonic() < deadline:
        it += 1
        # compute conv and w
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        maxb = conv.max()
        eps = max(eps_min, eps0 * (eps_decay ** it))
        # softmax weights
        w = np.exp((conv - maxb) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # step size
        eta = eta0 / math.sqrt(it)
        # exponentiated gradient update
        # Avoid overflow by scaling
        exp_arg = -eta * g
        max_arg = np.max(exp_arg)
        a_new = a * np.exp(exp_arg - max_arg)
        # renorm:
        a_new_sum = a_new.sum()
        if a_new_sum <= 0:
            # fallback to projection onto simplex of old a
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        # evaluate
        cur_sc = score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        # accept new a (whether improved or not) for next iteration to keep moving
        a = a_new
        
        # occasional random perturbation if no improvement for many iterations
        if no_improve > 4000:
            # swap small mass
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                a[i] -= delta
                a[j] += delta
                a = np.clip(a, 0, None)
                a = a / a.sum()
            no_improve = 0
        
        # early exit if reached target
        if best_sc < 1.50274349:
            break

    # final local search with small Gaussian noise using remaining time
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.001, size=n)
        cand = best_vec + delta
        cand = np.clip(cand, 0, None)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand = cand / cand_sum
        cand_sc = score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            if best_sc < 1.50274349:
                break

    # final return list
    return [float(x) for x in best_vec]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # ---------- initialise ----------
    if "height_sequence_1" in globals():

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # normalise to sum = 1

    n = a.size
    # FFT length that safely contains the convolution of length 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # ---------- scoring helper (sum is 1, so denominator = 1) ----------
    def _score(v: np.ndarray) -> float:
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    best_vec = a.copy()

    # ---------- optimisation parameters ----------
    eps0 = 0.05                # initial soft‑max temperature
    eps_decay = 0.9997
    eps_min = 1e-12
