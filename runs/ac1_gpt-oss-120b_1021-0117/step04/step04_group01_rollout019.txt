def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init a
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000; a = np.full(n, 1.0/n)
    a = np.maximum(a, 0)
    a /= a.sum()  # simplex

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())

    # hyperparams
    eps = 0.5
    eps_decay = 0.9998
    eps_min = 1e-12
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.05
    adam_m = np.zeros_like(a)
    adam_v = np.zeros_like(a)
    best = a.copy()
    best_sc = _score(best, fft_len=L)
    cur_sc = best_sc

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # conv
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        max_val = conv.max()
        # softmax weights
        eps_cur = max(eps_min, eps * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps_cur)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)  # length L
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        adam_m = beta1 * adam_m + (1 - beta1) * g
        adam_v = beta2 * adam_v + (1 - beta2) * (g * g)
        m_hat = adam_m / (1 - beta1 ** it)
        v_hat = adam_v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        cand = a - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand, fft_len=L)
        # if candidate drastically worse preserve cur?
        # Accept anyway but if not improve for many iteration bump noise
        if cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            # optionally reject
            # keep a unchanged
            no_improve += 1

        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            no_improve = 0

        # Random perturb if stuck
        if no_improve > 2000:
            # add small random noise and reproject
            noise = rng.normal(scale=0.01, size=n)
            cand2 = a + noise
            cand2 = _project_simplex(cand2)
            cand2_sc = _score(cand2, fft_len=L)
            if cand2_sc < best_sc:
                best = cand2.copy()
                best_sc = cand2_sc
                a = cand2
                cur_sc = cand2_sc
                no_improve = 0
            else:
                # revert to best
                a = best.copy()
                cur_sc = best_sc
                no_improve = 0

        # early exit
        if best_sc < 1.50274349:
            break

    # final fine-tune with small Gaussian perturbations
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.005, size=n)
        cand = best + delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand, fft_len=L)
        if cand_sc < best_sc:
            best = cand
            best_sc = cand_sc

    return [float(x) for x in best]

# Helper: Euclidean projection onto the probability simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr = 0.02                     # base learning rate
    eps_adam = 1e-8

    # soft‑max smoothing schedule
    eps0 = 0.5
    eps_decay = 0.9997
    eps_min = 1e-12

    it = 0
    no_improve = 0

    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur_vec, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights -----
        eps_cur = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps_cur)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # length >= 3n-2
        idx = (2 * n - 2) - np.arange(n)               # slice to obtain ∂/∂a_i
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur_vec - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand, fft_len=L)

        # ----- accept if improvement, otherwise keep current -----
        if cand_sc < cur_sc:
            cur_vec, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # ----- record best -----
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc
            no_improve = 0

        # ----- occasional random shake‑up when stuck -----
        if no_improve > 3000:
            # redistribute a modest fraction of mass
            delta = 0.25 * rng.random() * cur_vec
            idx_from = rng.integers(0, n, size=n)
            idx_to = rng.integers(0, n, size=n)
            cand2 = cur_vec.copy()
            np.subtract.at(cand2, idx_from, delta)
            np.add.at(cand2, idx_to, delta)
            cand2 = _project_simplex(cand2)
            cand2_sc = _score(cand2, fft_len=L)
            if cand2_sc < best_sc:
                best_vec, best_sc = cand2.copy(), cand2_sc
                cur_vec, cur_sc = cand2, cand2_sc
                no_improve = 0
            else:
                # fallback to best without change
                cur_vec, cur_sc = best_vec.copy(), best_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # final fine‑tuning with tiny Gaussian nudges
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.003, size=n)
        cand = best_vec + delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand, fft_len=L)
        if cand_sc < best_sc:
            best_vec, best_sc = cand, cand_sc

    # --------------------------------------------------------
    return [float(x) for x in best_vec]
