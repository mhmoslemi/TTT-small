    import time, numpy as np, math
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    rng = np.random.default_rng(seed)

    # Determine length

    else:

    # compute initial score via evaluate_sequence (but we will also compute via our own bound)
    best_a = a.copy()
    best_sc = evaluate_sequence(best_a.tolist())
    cur_a = a.copy()
    cur_sc = best_sc

    L = 1 << ((2 * n - 1).bit_length())
    def _score_vec(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2 * n - 1]
        maxc = conv.max()
        return 2.0 * n * maxc

    # hyperparameters
    step0 = 0.5
    backtrack = 0.5
    max_backtrack = 10
    no_improve = 0

    it = 0

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a,0)
    else:
        n = 2000
        a = np.full(n, 1.0/n)

    a = a / max(a.sum(), 1e-12)  # sum to 1

    n = a.size
    L = 1 << ((2*n-1).bit_length())

    def score(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        maxc = conv.max()
        return 2.0 * n * maxc

    best_a = a.copy()
    best_sc = score(best_a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.1

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # compute conv and max index
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        maxc = conv.max()
        k = int(conv.argmax())
        # gradient
        g = np.zeros_like(a)
        if k < n:
            g[:k+1] = a[k::-1]
        else:
            start_idx = k - (n - 1)
            g[start_idx:] = a[start_idx:][::-1]
        g *= 2.0

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = a - step
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0:
            cand = cand / s
        else:
            cand = np.full_like(cand, 1.0/n)

        cand_sc = score(cand)

        if cand_sc < best_sc:
            best_a, best_sc = cand.copy(), cand_sc
            # reset Adam moments maybe
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            it = 0  # reset iteration count for bias correction?

        # accept if improvement else maybe also accept if small increase? We'll adopt acceptance if cand_sc <= best_sc else keep old?
        # Here we only update a if cand_sc <= current (not just best). Let's do:
        if cand_sc <= score(a):
            a = cand
        # else possibly random restart after stagnation
        # Stagnation detection:
        if it % 5000 == 0:
            # random perturbation
            delta = rng.normal(scale=0.01, size=n)
            a = a + delta
            a = np.maximum(a, 0)
            a = a / a.sum()
        if best_sc < 1.50274349:
            break

    # Fine-tune small random steps
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.001, size=n)
        cand = best_a + delta
        cand = np.maximum(cand, 0)
        cand = cand / cand.sum()
        cand_sc = score(cand)
        if cand_sc < best_sc:
            best_a, best_sc = cand, cand_sc

    return [float(x) for x in best_a]

def _score_vec(v: np.ndarray, fft_len: int) -> float:
    """Return 2*n*max(conv(v,v)) for a vector (already normalised)."""
    n = v.shape[0]
    af = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())
