import time

# ------------------------------------------------------------
# Project onto the probability simplex (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (2 * a * w) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / math.sqrt(it)
        delta = step * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = a - delta
        cand = _project_simplex(cand)

        # ----- score of candidate (exact) -----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * cand_conv.max()

        # ----- accept if improved, else occasional Metropolis move -----
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps autocorrelation) -----
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta_m = a[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta_m
                tmp[j] += delta_m
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    a, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early stop if target reached -----
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Final local refinement (small random jitter)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.02
        if delta <= 0.0:
            continue
        tmp = best.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_fft = np.fft.rfft(tmp, n=L)
        tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
        tmp_sc = 2.0 * n * tmp_conv.max()
        if tmp_sc < best_sc:
            best, best_sc = tmp, tmp_sc
            continue
        # occasional symmetry check
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _project_simplex(sym)
        sym_fft = np.fft.rfft(sym, n=L)
        sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
        sym_sc = 2.0 * n * sym_conv.max()
        if sym_sc < best_sc:
            best, best_sc = sym, sym_sc

    # --------------------------------------------------------
    # Return as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
