import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (L2‑optimal) – forces sum(x)=1, x>=0
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast score (uses FFT, assumes non‑negative entries)
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray = None) -> float:
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Optional LP step – improve the current vector using a sparse LP
# ------------------------------------------------------------
try:
    from scipy.sparse import diags
    from scipy.optimize import linprog

    _SCIPY_AVAIL = True
except Exception:  # pragma: no cover
    _SCIPY_AVAIL = False


def _lp_improve(a: np.ndarray, M: float) -> np.ndarray | None:
    """One LP improvement step (returns a mixed vector or None)."""
    if not _SCIPY_AVAIL:
        return None
    n = a.shape[0]
    # Build sparse convolution matrix A where (A @ b)[k] = (a * b)[k]
    # Every diagonal i (0≤i<n) contains the constant a[i].
    # Use broadcast to avoid materialising a giant dense array.
    data = np.broadcast_to(a[:, None], (n, n))
    A = diags(data, offsets=np.arange(n), shape=(2 * n - 1, n), format="csc")

    # Linear program: maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
    except Exception:  # pragma: no cover
        return None
    if not res.success:
        return None

    b = res.x
    sum_b = b.sum()
    if sum_b < 1e-12:
        return None
    # Scale to keep the L1‑norm identical to that of a
    scale = a.sum() / sum_b
    g = b * scale
    # Mix a and the new vector – ½ is a safe step size
    mixed = 0.5 * a + 0.5 * g
    mixed = _project_simplex(mixed)
    return mixed
