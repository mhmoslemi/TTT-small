# ---------- Euclidean projection onto the probability simplex ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ---------- Main entry ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence that (hopefully) beats the target bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initialise -----
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform distribution of length ~2500
        n_fallback = 2500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # normalise to sum = 1
    n = a.size

    # FFT length covering convolution of length up to 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # helper to evaluate the objective (sum is 1 ⇒ score = 2·n·max(conv))
    def _score_vec(v: np.ndarray) -> float:
        vf = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(vf * vf, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # ----- bookkeeping -----
    best = a.copy()
    best_sc = _score_vec(best)

    # ----- Adam parameters (projected Adam) -----
    m = np.zeros_like(a)                 # first moment
    v_mom = np.zeros_like(a)             # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12                            # base learning rate
    eps0 = 0.01                           # initial soft‑max temperature
    eps_decay = 0.9998
    eps_min = 1e-12

    it = 0
    cur_sc = best_sc

    # ----- main optimisation loop (projected Adam) -----
    while time.monotonic() < deadline:
        it += 1

        # convolution of current iterate
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv_full.max()
        cur_sc = 2.0 * n * maxc

        # soft‑max weights over convolution entries
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv_full - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the smoothed max (2·a ∗ w_rev)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # slice aligning with a
        g = 2.0 * grad_full[idx]

        # Adam update (biased moments)
        m = beta1 * m + (1.0 - beta1) * g
        v_mom = beta2 * v_mom + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_mom / (1.0 - beta2 ** it)

        # step size (Adam‑style)
        lr = lr0 * math.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = a - step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_sc = _score_vec(cand)

        # accept if improvement over current iterate
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc

        # record globally best
        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            # reset Adam moments to avoid stale direction after a big jump
            m.fill(0.0)
            v_mom.fill(0.0)

        # early exit if we already beat the target
        if best_sc < 1.50274349:
            break

        # occasional restart from the best point (helps escape stagnation)
        if it % 4000 == 0:
            a = best.copy()
            m.fill(0.0)
            v_mom.fill(0.0)

    # ----- final refinement: random pairwise swaps -----
    # use any remaining time to explore simple combinatorial improvements
    while time.monotonic() < deadline - 0.05:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]          # swap two entries
        cand_sc = _score_vec(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc

    # ----- return result as plain Python list -----
    return [float(x) for x in best]
