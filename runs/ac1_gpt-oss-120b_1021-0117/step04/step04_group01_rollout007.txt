import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(vec, L):
    """Score = 2*n*max(conv(vec,vec)). Assumes sum(vec) = 1."""
    n = vec.size
    af = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())
