def _project_simplex(v):
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1,len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast evaluation of the AC1 bound (sum is assumed to be 1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray, L: int = None) -> float:
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ----------------------------------------------------------------------

    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size and initial guess
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)

        a0 = np.exp(-np.arange(2000) / 200.0)

    n = a0.shape[0]

    # generate a few cheap initial candidates and pick the best
    candidates = []

    # 1) given parent (projected)
    cand = _project_simplex(np.maximum(a0, 0.0))
    candidates.append(cand)

    # 2) uniform distribution on the simplex
    uniform = np.full(n, 1.0 / n)
    candidates.append(_project_simplex(uniform))

    # 3) random Dirichlet (alpha = 0.5)
    rnd = -np.log(rng.random(n) + 1e-16)   # exponential(1)
    candidates.append(_project_simplex(rnd))

    best = None
    best_sc = float("inf")
    for c in candidates:
        sc = _score(c)
        if sc < best_sc:
            best_sc = sc
            best = c.copy()
    cur = best.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    # FFT length large enough for all convolutions (gradient needs 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # Adam hyper‑parameters

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    lr0 = 0.08                     # base learning rate

    # smoothing for the soft‑max approximation of max
    eps0 = 0.7
    eps_decay = 0.9999
    eps_min = 1e-8

    it = 0
    no_improve = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:

        # ----- convolution of current iterate (needed both for max and grad) -----
        af = np.fft.rfft(cur, n=L)

        max_val = conv.max()
