def _mirror_update(a, grad, eta):
    # a: sum=1, positive
    # grad: maybe any real
    # compute new a
    exp_term = np.exp(-eta * grad)
    # avoid overflow/underflow
    exp_term = np.clip(exp_term, 1e-300, 1e300)
    a_new = a * exp_term
    total = a_new.sum()
    if total == 0 or np.isnan(total):
        a_new = a
        total = a_new.sum()
    a_new = a_new / total
    return a_new

def _mirror_update(a, grad, eta):
    # compute exponent shift for numeric stability
    log_factor = -eta * grad
    log_factor = log_factor - np.max(log_factor)  # shift to max 0
    factor = np.exp(log_factor)
    a_new = a * factor
    s = a_new.sum()
    if s == 0 or np.isnan(s):
        return a  # fallback
    return a_new / s

import time

import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Helper: mirror‑descent (exponential) update for simplex
# ------------------------------------------------------------
def _mirror_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Return a * exp(-eta*grad) renormalised to sum 1 (stable version)."""
    log_factor = -eta * grad
    # shift to avoid overflow/underflow
    log_factor -= np.max(log_factor)
    factor = np.exp(log_factor)
    a_new = a * factor
    s = a_new.sum()
    if s <= 0 or np.isnan(s):
        return a.copy()
    return a_new / s

# ------------------------------------------------------------
# Helper: score assuming sum(a)=1 (exact evaluator without division)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(conv) where conv = a * a and sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
