# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)

                        best = tmp.copy()
                        best_score = tmp_score
                    if tmp_score < cur_score:
                        cur = tmp
                        cur_score = tmp_score
                    no_improve = 0

        # ---- periodic symmetrisation -------------------------------
        if it % 3000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _fast_score(sym, L)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
            if sym_score < best_score:
                best = sym.copy()
                best_score = sym_score

        # ---- occasional restart to escape deep plateaus -----------
        if no_improve > 25000:
            # blend best with fresh random mass
            cur = best * 0.85 + rng.random(n) * 0.15
            cur = _project_simplex(cur)
            velocity.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_score < 1.50274349:
            break

    # ----------------------------------------------------------------
    # Fallback safety (should never happen)
    if best.size == 0:
        best = np.ones(1, dtype=np.float64)
        best = _project_simplex(best)

    return [float(x) for x in best]
