def random_step_sequence(n, rng):
    k = rng.integers(2, 9)  # number of intervals
    # breakpoints: sample k-1 sorted indices from 1..n-1
    breakpoints = np.sort(rng.choice(n-1, size=k-1, replace=False) + 1)  # ensure unique sorted
    # add endpoints
    splits = np.concatenate(([0], breakpoints, [n]))
    # heights: sample positive values e.g., rng.exponential(scale=1.0, size=k)
    heights = rng.exponential(scale=1.0, size=k)
    a = np.empty(n, dtype=np.float64)
    for i in range(k):
        a[splits[i]:splits[i+1]] = heights[i]
    # ensure nonnegative (already positive)
    # project onto simplex
    a = _project_simplex(a)
    # optional symmetrize
    if rng.random() < 0.5:
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a)
    return a

def compute_score(a, L_conv=None):
    n = a.shape[0]
    if L_conv is None:
        L_conv = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L_conv)
    conv = np.fft.irfft(af * af, n=L_conv)[:2*n-1]
    max_val = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_val / (sum_a**2)

import time, math, numpy as np

# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum == 1, non‑neg.)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 score (uses FFT, assumes sum > 0)
# ------------------------------------------------------------
def _score_fast(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT length L."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    return 2.0 * n * max_val / (s * s)


# ------------------------------------------------------------
# Generate a random step‑function sequence (piecewise constant)
# ------------------------------------------------------------
def _random_step_seq(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(2, 9)                     # 2‑8 intervals
    cuts = np.sort(rng.choice(n - 1, size=k - 1, replace=False) + 1)
    splits = np.concatenate(([0], cuts, [n]))
    heights = rng.exponential(scale=1.0, size=k)
    a = np.empty(n, dtype=np.float64)
    for i in range(k):
        a[splits[i] : splits[i + 1]] = heights[i]
    a = _project_simplex(a)
    # optional symmetrisation
    if rng.random() < 0.4:
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a)
    return a
