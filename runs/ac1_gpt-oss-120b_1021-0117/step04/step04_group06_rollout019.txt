    rng = np.random.default_rng(seed)
    start = time.time()

    a = (a + a[::-1]) * 0.5

    # scoring using FFT
    def score(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    # evaluate initial
    cur = a.copy()
    cur_sc = score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc
    # momentum
    vel = np.zeros_like(cur)
    # hyperparameters
    step0 = 0.8
    eps0 = 0.6
    eps_decay = 0.999
    eps_min = 1e-8
    beta = 0.9
    no_improve = 0
    it = 0
    # main loop
    while time.time() < deadline:
        it += 1
        # compute conv and softmax w
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        eps = max(eps_min, eps0 * (eps_decay ** it))
        # compute softmax weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # tiny renormalisation to avoid drift
    s = w.sum()
    if s > 0:
        w /= s
    return w


# ------------------------------------------------------------
# Score: 2 * n * max(conv(a,a))
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the smoothed max (soft‑max with temperature eps)
# ------------------------------------------------------------
def _grad_and_max(a: np.ndarray, eps: float, idx: np.ndarray, L: int):
    """Return (gradient of soft‑max objective, current max of convolution)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    m = conv.max()
    # soft‑max weights
    w = np.exp((conv - m) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    g = 2.0 * grad_full[idx]
    return g, m
