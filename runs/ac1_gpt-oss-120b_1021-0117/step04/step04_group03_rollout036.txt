    decay = 0.9995
    eps = 1e-2
    eps_decay = 0.999
    delta_frac = 0.001 # tolerance for near max indices

    iteration = 0
    while time.time check:
        # compute convolution:
        L = 1 << ((2*n-1).bit_length())
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # find indices near max:
        thresh = max_val - delta_frac * max_val
        near_idxs = np.where(conv >= thresh)[0]  # may be many
        # compute subgradient:
        g = np.zeros_like(a)
        # For each s in near_idxs, add contributions
        for s in near_idxs:
            # compute shift contributions
            # positions i where 0 <= s - i < n => i in [max(0, s-(n-1)), min(n-1, s)]
            i_start = max(0, s - (n - 1))
            i_end = min(n - 1, s)
            # For those i:
            # j = s - i
            # g[i] += 2 * a[j]
            # vectorized:
            idx_i = np.arange(i_start, i_end+1)
            idx_j = s - idx_i
            g[idx_i] += 2 * a[idx_j]
        # take average:
        g /= len(near_idxs) if len(near_idxs)>0 else 1

        # step size:
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / np.sqrt(1 + iteration) / grad_norm

        # update
        a_new = a - alpha * g
        # ensure nonnegativity via projection onto simplex:
        a_new = _project_simplex(a_new)  # defined earlier

        # evaluate

        if score_new < best_score:

            best_a = a_new.copy()

import time
import math
import numpy as np
import scipy.sparse as sp

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    # shift by mean to ensure numerical stability
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector with unit sum: 2 * n * max(conv)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    # length for exact cyclic convolution
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Build sparse matrix for (f * g)[s] <= M constraints
# ------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return sparse matrix A with A[s, j] = f[s-j] (zero out‑of‑range)."""
    n = f.size
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)
    pos = 0
    for j in range(n):
        rows[pos : pos + n] = np.arange(j, j + n, dtype=np.int32)
        cols[pos : pos + n] = j
        data[pos : pos + n] = f
        pos += n
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
