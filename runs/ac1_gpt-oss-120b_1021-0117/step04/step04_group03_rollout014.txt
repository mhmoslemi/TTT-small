import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector that is (or will be) normalized to sum‑1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(convolution) for a assumed to sum to 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -------------------- initial vectors --------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
    else:
        base = None

    if base is None:
        n = 1024
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = int(base.size)

    # helper to enforce symmetry and normalisation
    def _prep(v):
        v = np.maximum(v, 0.0)
        v = (v + v[::-1]) * 0.5
        return _project_simplex(v)

    # few cheap candidates
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)

    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()

    candidates = [_prep(base), _prep(uniform), _prep(arcsine)]
    cur = min(candidates, key=_score)
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    # -------------------- hyper‑parameters --------------------
    L = 1 << ((2 * n - 1).bit_length())
    eps = 1.0
    eps_decay = 0.9999
    eps_min = 1e-7
    step0 = 1.0          # initial step multiplier
    no_improve = 0
    it = 0
    target = 1.50274349

    # -------------------- optimisation loop --------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution (needed for gradient)
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smoothed weights for the max
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:

        else:
