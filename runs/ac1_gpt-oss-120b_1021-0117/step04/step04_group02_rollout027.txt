import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a: np.ndarray) -> float:
    """Fast AC1 bound for a vector with sum≈1 (uses FFT)."""
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # reserve a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -----------------------------------------------------------------
    # Initialise a feasible point on the simplex (prefer the provided one)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
    else:
        # fallback: symmetric bell shape
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        sigma = n_fallback / 6.0
        a0 = np.exp(-0.5 * ((t - n_fallback / 2) / sigma) ** 2)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # now sum(a0)=1

    # enforce symmetry (helps the bound)
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = _project_simplex(a0)

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length covering all needed convolutions

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # -----------------------------------------------------------------
    # Adam optimiser parameters (projected onto simplex, always symmetric)
    # -----------------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.2

    it = 0
    no_improve = 0
    eps = None                     # smoothing temperature for soft‑max
    temperature = 1.0
    improve_target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # -------------------------------------------------------------
        # convolution and soft‑max weights
        # -------------------------------------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:
            eps = max(eps * 0.995, max_val * 1e-6, 1e-12)

        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # -------------------------------------------------------------
        # gradient of the smoothed max (via convolution)
        # -------------------------------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # -------------------------------------------------------------
        # Adam update (projected, symmetric)
        # -------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # -------------------------------------------------------------
        # Accept if better, otherwise occasional uphill move (annealing)
        # -------------------------------------------------------------
        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            temperature = max(temperature * 0.9999, 1e-4)
            prob = math.exp(-(cand_sc - cur_sc) / temperature)
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0 if cand_sc < best_sc else no_improve + 1
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1

        # -------------------------------------------------------------
        # Random mass‑transfer perturbation when stuck
        # -------------------------------------------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            if best_vec[i] > 0.0:
                delta = best_vec[i] * rng.random() * 0.45
                j = rng.integers(0, n)
                tmp = best_vec.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                cur = tmp
                cur_sc = tmp_sc
                no_improve = 0

        # -------------------------------------------------------------
        # Early exit if target achieved
        # -------------------------------------------------------------
        if best_sc < improve_target:
            break

    # return plain Python floats
    return [float(x) for x in best_vec]
