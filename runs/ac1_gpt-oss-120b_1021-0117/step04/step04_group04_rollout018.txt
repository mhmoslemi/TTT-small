def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    # define FFT length (next power of 2 > 3n)
    L = 1 << ((3 * n - 2).bit_length())

    def project_simplex(v):
        # same as previous
        # ensure nonnegative and sum=1
        v = np.maximum(v, 0)
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, len(v) + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    def score_fast(a, a_fft=None):
        # returns 2n * max(conv)
        # assume sum =1
        if a_fft is None:
            a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return 2.0 * n * conv.max()
    ```

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    else:
        n = 2000
    L = 1 << ((3 * n - 2).bit_length())
    def project_simplex(v):

        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, len(v) + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)
    def fast_score(a, a_fft=None):
        if a_fft is None:
            a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return 2.0 * n * conv.max()
