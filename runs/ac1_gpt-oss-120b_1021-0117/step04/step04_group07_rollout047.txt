def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Linear convolution via FFT, returning max element
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max of the linear convolution a*a (zero‑padded to length L)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initialise from parent if available, else simple fallback
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a0 = np.full(n0, 1.0 / n0, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0.fill(1.0 / a0.size)
    else:
        a0 /= s                     # normalise to simplex

    n = a0.size
    # FFT length >= 3n‑2 (covers all convolutions we need)
    L = 1 << ((3 * n - 2).bit_length())

    cur = a0.copy()
    cur_sc = 2.0 * n * _conv_max(cur, L)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # hyper‑parameters
    # --------------------------------------------------------
    eps = 0.02                     # start smoothing for soft‑max
    eps_min = 1e-9
    eps_decay = 0.9995
    step0 = 0.8                    # base learning rate
    beta = 0.9                     # momentum coefficient
    vel = np.zeros_like(cur)       # momentum buffer
    it = 0
    no_improve = 0
    idx = (2 * n - 2) - np.arange(n)   # index mapping for gradient extraction

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- gradient of the smoothed max -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            # hard max (sub‑gradient)
            w = np.zeros_like(conv)
            w[int(conv.argmax())] = 1.0

        # reverse for convolution‑gradient trick
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # circular conv

        if best_sc < 1.50274349:

    # --------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best]
