# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=fft_len)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=fft_len)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        m_grad = momentum * m_grad + (1.0 - momentum) * g
        grad_norm = np.linalg.norm(m_grad) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm

        # two candidate step sizes (full & half)
        cand1 = cur - alpha * m_grad
        cand1 = _project_simplex(cand1)
        sc1 = _score(cand1, fft_len)

        cand2 = cur - (alpha * 0.5) * m_grad
        cand2 = _project_simplex(cand2)
        sc2 = _score(cand2, fft_len)

        cand, cand_sc = (cand1, sc1) if sc1 < sc2 else (cand2, sc2)

        # acceptance (deterministic improvement + occasional uphill)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- occasional random mass transfer when stuck -------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, fft_len)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- occasional symmetrisation (helps many instances) -------
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, fft_len)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # early exit if target already reached
        if best_sc < 1.50274349:
            break

    # ----- cheap post‑processing: try a few simple structured seeds ----
    # (executed only if time permits)
    simple_seeds = []

    # uniform
    uni = np.full(n, 1.0 / n)
    simple_seeds.append(uni)

    # exponential families
    for tau in (n / 6.0, n / 4.0, n / 2.0):
        expv = np.exp(-np.arange(n) / tau)
        simple_seeds.append(_project_simplex(expv))

    # half‑cosine bump
    x = np.arange(n)
    cosb = np.maximum(0.0, np.cos(np.pi * (x - (n - 1) / 2) / n))
    simple_seeds.append(_project_simplex(cosb))

    # Gaussian bump
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((x - (n - 1) / 2) / sigma) ** 2)
    simple_seeds.append(_project_simplex(gauss))

    # random Dirichlet samples (few, cheap)
    for _ in range(20):
        rnd = rng.gamma(shape=1.5, scale=1.0, size=n)
        simple_seeds.append(_project_simplex(rnd))

    for cand in simple_seeds:
        if time.monotonic() >= deadline:
            break
        sc = _score(cand, fft_len)
        if sc < best_sc:
            best, best_sc = cand.copy(), sc

    # ----- return result ---------------------------------------------
    return [float(x) for x in best]
