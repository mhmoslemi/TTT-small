import time, math, numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init a
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.clip(a0, 0.0, None)
    else:
        n_fallback = 2000
        a0 = rng.random(n_fallback)
    # ensure positive to log
    a0 = np.maximum(a0, 1e-12)
    sum0 = a0.sum()
    if sum0 <= 0:
        a0 = np.full_like(a0, 1.0/a0.size)
    else:
        a0 = a0 / sum0   # sum=1

    n = a0.size
    L = 1 << ((3*n - 2).bit_length())

    # initialize parameters for Adam
    z = np.log(a0 + 1e-12)  # shape (n,)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.1
    eps_adam = 1e-8

    # smoothing epsilon for softmax
    eps_s = 0.1
    eps_min = 1e-12
    eps_decay = 0.9995

    best_a = a0.copy()
    # compute initial fast score
    a_fft = np.fft.rfft(best_a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    maxc = conv.max()
    best_score = 2.0 * n * maxc  # sum=1
    # record improvement count
    no_improve = 0

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # compute a via softmax
        expz = np.exp(z - np.max(z))  # subtract max for numerical stability
        a = expz / expz.sum()
        # compute convolution
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        maxc = conv.max()
        # compute softmax weights
        w = np.exp((conv - maxc) / eps_s)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient wrt a
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # slice
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # gradient wrt z using softmax Jacobian
        dot_ag = (a * g).sum()
        grad_z = a * (g - dot_ag)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z * grad_z)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - step

        # evaluate current a
        cur_score = 2.0 * n * maxc  # sum=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # decay smoothing epsilon
        eps_s = max(eps_min, eps_s * eps_decay)

        # occasional random perturbation
        if no_improve >= 2000:
            # random mass transfer in softmax param space
            # propose new z by adding noise to a few indices
            # For simplicity, reinitialize from best with small noise
            noise = rng.normal(scale=0.1, size=n)
            z = np.log(best_a + 1e-12) + noise
            m.fill(0)
            v.fill(0)
            no_improve = 0
        # early exit if we reach target bound
        if best_score < 1.50274349:
            break

    # return best as list
    return [float(x) for x in best_a]

import time, math, numpy as np

def _fast_score(a: np.ndarray, L: int) -> float:
    """Fast evaluation of the AC1 bound for a vector with sum(a)=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------
    # initialise a (simplex, sum = 1)
    # ------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.clip(a0, 0.0, None)
    else:
        # fallback: Dirichlet(1) of modest length
        n_fallback = 2000
        a0 = rng.random(n_fallback)
    a0 = np.maximum(a0, 1e-12)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # simplex normalisation

    n = a0.size
    # FFT length supporting full convolution of a (len n) with a (len n) and
    # also with a weight vector of length 2n‑1.
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # Adam optimiser on the log‑softmax parameters
    # ------------------------------------------------------------
    # parameters: a_i = exp(z_i) / sum_j exp(z_j)
    z = np.log(a0 + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    lr = 0.07
    eps_adam = 1e-8

    # smoothing for the softmax approximation of the max
    eps_s = 0.12
    eps_min = 1e-12
    eps_decay = 0.9996

    best_a = a0.copy()
    best_score = _fast_score(best_a, L)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ---- transform z -> a (softmax, always on simplex) ----
        zi = z - np.max(z)          # stabilise exp
        expz = np.exp(zi)
        a = expz / expz.sum()

        # ---- convolution and max (exact for current a) ----
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ---- gradient of a smoothed max (soft‑max weights) ----
        if eps_s > 1e-8:
            w = np.exp((conv - maxc) / eps_s)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # pure sub‑gradient at the maximal index
            k = int(conv.argmax())
            w = np.zeros_like(conv)
            w[k] = 1.0

        # gradient w.r.t. a (using convolution w_rev * a)
        w_rev = w[::-1]
        wf = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * wf, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g_a = 2.0 * grad_full[idx]          # ∂L/∂a

        # ---- map to gradient w.r.t. z (softmax Jacobian) ----
        dot_ag = np.dot(a, g_a)
        g_z = a * (g_a - dot_ag)

        # ---- Adam update (minimise) ----
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z * g_z)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z -= step

        # ---- keep the best solution seen ----
        cur_score = float(2.0 * n * maxc)          # sum(a)=1 → exact AC1 bound
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- decay the soft‑max smoothing ----
        eps_s = max(eps_min, eps_s * eps_decay)

        # ---- occasional random perturbation when stuck ----
        if no_improve >= 4000:
            # re‑initialise z from the current best plus modest random noise
            noise = rng.normal(scale=0.15, size=n)
            z = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ---- early exit if target already reached ----
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # ------------------------------------------------------------
    return [float(x) for x in best_a]
