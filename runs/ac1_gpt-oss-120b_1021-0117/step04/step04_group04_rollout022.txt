import time
import math
import numpy as np

# ------------------------------------------------------------
# Simple helper: compute the AC1 upper bound (the judge's score)
# ------------------------------------------------------------
def _compute_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2 * n * max(convolution) for a non‑negative sequence a."""
    n = a.size
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Simple projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise in the unlikely case of numerical drift
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w

    """

    # --------------------------------------------------------

    # Work with a normalised version (sum = 1) – the score is scale‑invariant.
    a0 = _project_simplex(a0)

    n = a0.size
    # FFT length sufficient for linear convolution (≥ 2n‑1)
