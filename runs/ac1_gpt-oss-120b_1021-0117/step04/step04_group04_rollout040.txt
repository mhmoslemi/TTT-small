import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Approximate objective using FFT (fast, slightly inexact)
# ------------------------------------------------------------
def _approx_score(a: np.ndarray, a_fft: np.ndarray, L: int) -> float:
    """2 * n * max(conv) computed via FFT."""
    n = a.shape[0]
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence that (hopefully) improves
    on the previous best AC1 bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: simple exponential decay
        n0 = 2000
        a0 = np.exp(-np.arange(n0) / 200.0)

    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0.0:
        a0 = np.ones_like(a0)

    # normalise to sum == 1 (projection onto simplex)
    a0 = _project_simplex(a0)
    n = a0.shape[0]

    # FFT length – enough for 3n‑2 linear convolution length
    L = 1 << ((3 * n - 2).bit_length())

    cur = a0.copy()
    a_fft = np.fft.rfft(cur, n=L)
    cur_sc = _approx_score(cur, a_fft, L)

    best = cur.copy()
    best_sc = cur_sc

    # ----- optimisation hyper‑parameters ------------------------------
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    step0 = 0.6                     # base learning‑rate
    eps0 = 1.0                      # initial smoothing for the soft‑max
    eps_decay = 0.9998
    eps_min = 1e-9

    no_improve = 0
    it = 0

    # ----- main loop -------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # recompute FFT for the current iterate
        a_fft = np.fft.rfft(cur, n=L)

        # linear convolution of cur with itself (full)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smooth approximation of the max (log‑sum‑exp)
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # gradient of the smoothed objective:
        #   grad = 2 * (cur * w_rev)   (convolution in frequency domain)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # length >= 3n‑2
        idx = (2 * n - 2) - np.arange(n)               # slice for gradient
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 * m_hat / (np.sqrt(v_hat) + 1e-8)

        cand = cur - step
        cand = _project_simplex(cand)

        # fast approximate score for the candidate
        cand_fft = np.fft.rfft(cand, n=L)
        cand_sc = _approx_score(cand, cand_fft, L)

        # ----- acceptance -------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
            else:
                no_improve += 1
        else:
            # occasional Metropolis style acceptance (temperature ~1/it)
            prob = math.exp(-(cand_sc - cur_sc) * (0.1 / it + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- stochastic mass‑transfer when stuck ------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_sc = _approx_score(tmp, tmp_fft, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- symmetrisation ------------------------------------------------
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_sc = _approx_score(sym, sym_fft, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- early exit ----------------------------------------------------
        if best_sc < 1.50274349:
            break

    # final safety: make sure the returned list is non‑empty and finite
    # (the projection guarantees non‑negativity and sum == 1)
    return [float(x) for x in best]
