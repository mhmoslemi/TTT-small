import time
import math
import numpy as np
import cvxpy as cp

# ---------- Utilities ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a: np.ndarray, fft_len: int = None) -> float:
    """Return 2*n*max(conv) for a vector (sum does not affect the value)."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())

# ---------- Main ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.4)   # safety margin

    # ---- initialise ----
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a = np.exp(-np.arange(n0, dtype=np.float64) / 120.0)  # decaying
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # normalise to sum = 1

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length for gradient step

    best_vec = a.copy()
    best_sc = _score(best_vec, fft_len=L)

    # ---- gradient descent hyper‑params ----
    eps0 = 0.6
    eps_decay = 0.9995
    eps_min = 1e-12
    step0 = 0.9
    no_improve = 0
    it = 0
    max_backtrack = 12
    backtrack_factor = 0.5

    cur = a.copy()
    cur_sc = best_sc

    # ---- set up LP objects (once) ----
    use_lp = True
    try:
        param_a = cp.Parameter(n, value=cur)
        param_M = cp.Parameter(value=0.0)
        b_var = cp.Variable(n, nonneg=True)
        conv_expr = cp.conv(param_a, b_var)     # length 2n‑1
        lp_prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [conv_expr <= param_M])
    except Exception:
        use_lp = False

    lp_every = 250          # try LP each this many iterations
    lp_tau = 0.5            # mixing factor when LP gives improvement

    # ---- main optimisation loop ----
    while time.monotonic() < deadline:
        it += 1
        # ---- current convolution & max ----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * float(max_val)   # same as _score for sum=1

        # ---- smooth‑max weights (soft‑max) ----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ---- gradient of smoothed max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # length >= 3n‑2
        idx = (2 * n - 2) - np.arange(n)               # slice for gradient
        g = 2.0 * grad_full[idx]

        # ---- line search ----
        step = step0 / math.sqrt(it)
        improved = False
        for _ in range(max_backtrack):
            cand = cur - step * g
            cand = _project_simplex(cand)
            cand_sc = _score(cand, fft_len=L)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_vec, best_sc = cand.copy(), cand_sc
                    no_improve = 0
                improved = True
                break
            step *= backtrack_factor
        if not improved:
            no_improve += 1

        # ---- occasional LP jump ----
        if use_lp and it % lp_every == 0:
            try:
                param_a.value = cur
                param_M.value = max_val
                lp_prob.solve(solver=cp.ECOS, warm_start=True,
                               max_iters=200, abstol=1e-7, reltol=1e-7)
                if b_var.value is not None:
                    b_raw = np.maximum(b_var.value, 0.0)
                    s_b = b_raw.sum()
                    if s_b > 0.0:
                        b_norm = b_raw / s_b
                        b_sc = _score(b_norm, fft_len=L)
                        if b_sc < best_sc:

                            cur = (1 - lp_tau) * cur + lp_tau * b_norm
                            cur = _project_simplex(cur)
                            cur_sc = _score(cur, fft_len=L)
            except Exception:
                use_lp = False   # fall back to pure gradient if LP fails

        # ---- random perturbation when stuck ----
        if no_improve > 3500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, fft_len=L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ---- early exit if target reached ----
        if best_sc < 1.50274349:
            break

    # ---- final fine‑tuning with tiny Gaussian noise ----
    while time.monotonic() < deadline - 0.07:
        delta = rng.normal(scale=0.004, size=n)
        cand = best_vec + delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand, fft_len=L)
        if cand_sc < best_sc:
            best_vec, best_sc = cand, cand_sc

    # ---- return result ----
    return [float(x) for x in best_vec]
