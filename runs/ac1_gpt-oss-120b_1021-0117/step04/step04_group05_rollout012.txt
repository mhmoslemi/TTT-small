def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # initial vector
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 1024
        cur = np.exp(-np.arange(n, dtype=np.float64) / (n / 10.0))
    cur = np.maximum(cur, 0.0)
    # enforce symmetry to improve maybe
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)
    n = cur.shape[0]
    # FFT length
    L = 1 << ((2*n - 1).bit_length())
    # compute initial conv, score
    cur_fft = np.fft.rfft(cur, n=L)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L)[:2*n-1]
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max  # sum =1
    best_vec = cur.copy()
    best_score = cur_score
    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    alpha0 = 0.5
    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        it += 1
        # compute gradient using max index
        k0 = int(np.argmax(cur_conv))
        idx = k0 - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[idx[mask]]
        # Adam update: compute biased moment estimates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = alpha0 / (np.sqrt(v_hat) + adam_eps)
        cand = cur - step * m_hat
        # project onto simplex
        cand = _project_simplex(cand)
        # compute conv and score for candidate
        cand_fft = np.fft.rfft(cand, n=L)

        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max
        if cand_score < cur_score - 1e-12:  # improvement
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_max
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:

                best_vec = cand.copy()

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
