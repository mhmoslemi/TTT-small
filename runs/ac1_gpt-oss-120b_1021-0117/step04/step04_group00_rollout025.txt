def optimize_from(a0):
    cur = a0.copy()
    m = np.zeros_like(cur) # first moment
    v = np.zeros_like(cur) # second moment
    beta1 = 0.9
    beta2 = 0.999
    epsilon_adam = 1e-8
    step0 = 0.2  # base LR (docs)
    eps = 1.0
    eps_decay = 0.9995
    min_eps = 1e-6
    best_local_vec = cur.copy()
    best_local_sc = _score(cur)
    non_improve = 0
    for iteration in itertools.count(1):
        if time.monotonic() >= deadline: break
        # compute conv, softmax weights, gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n-1]
        max_val = conv.max()
        # compute softmax weights
        eps = max(min_eps, eps * eps_decay) if iteration>1 else eps
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = step0 / np.sqrt(iteration) * m_hat / (np.sqrt(v_hat) + epsilon_adam)
        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_local_sc:
            best_local_sc = cand_sc
            best_local_vec = cand.copy()
        # Accept if improvement else maybe occasional acceptance
        if cand_sc <= best_local_sc or rng.random() < np.exp(-(cand_sc - best_local_sc) / (step0 + 1e-12)):
            cur = cand
            # maybe reset non_improve?
        # periodic symmetrisation
        if iteration % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_local_sc:
                cur, best_local_sc = sym, sym_sc
                best_local_vec = cur.copy()
        # random mass-transfer when stuck

def _optimize(a0, L, n, proj, score_func, rng, deadline, target):

    beta2 = 0.999

    min_eps = 1e-6

    best_sc = score_func(cur)
    non_improve = 0

    for it in range(1, 1000000):
        if time.monotonic() >= deadline:
            break

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score (exact objective) – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(convolution(a,a)) for a normalized to sum=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the smoothed max‑convolution
# ------------------------------------------------------------
def _gradient(a_fft, w_rev, L, n):
    """Gradient of Σ_i w_i·(a*a)_i w.r.t. a (matrix‑free)."""
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]
