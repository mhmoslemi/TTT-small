def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.5)

    if 'height_sequence_1' in globals():

        base = np.maximum(base, 0.0)
        base = _project_simplex(base)  # sum = 1
        n = base.size
    else:

        base = None

    def arcsine_dist(n):
        i = np.arange(n, dtype=np.float64)
        w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
        w /= w.sum()
        return w

    # Prepare list of initial candidates (numpy arrays) of length n
    inits = []
    if base is not None:
        inits.append(base)

        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()
        return 2.0 * n * maxc, conv

    # iterate over initializations
    for init in inits:
        # check remaining time
        if time.time() >= deadline:

        beta = 0.9

        while time.time() < deadline:
            it += 1

            max_val = conv.max()
            eps = max(eps_min, eps * eps_decay)

            # smoothed gradient

    # --------------------------------------------------------
    # Loop over starters, performing a cheap momentum descent
    # --------------------------------------------------------
    for init in starters:
        if time.time() >= deadline:
            break

        cur = init.copy()
        # initial objective
        _, maxc, _ = _conv_grad(cur, eps=1.0, L=L)   # eps is irrelevant here
        cur_score = 2.0 * n * maxc
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()

        # descent parameters
        eps = 0.5
        eps_decay = 0.9975
        eps_min = 1e-7
        step0 = 0.7
        beta = 0.9                # momentum factor
        v = np.zeros_like(cur)    # momentum vector
        no_improve = 0
        it = 0

        while time.time() < deadline:
            it += 1

            # current smoothed gradient
            conv, maxc, g = _conv_grad(cur, eps, L)
            cur_score = 2.0 * n * maxc

            # decay smoothing
            eps = max(eps_min, eps * eps_decay)

            # momentum‑based step (scale‑invariant)
            norm_g = np.linalg.norm(g) + 1e-12
            alpha = step0 / math.sqrt(it) / norm_g
            v = beta * v - alpha * g
            cand = cur + v
            cand = _project_simplex(cand)

            # occasional symmetrisation (helps due to inherent symmetry)
            if (it % 400) == 0:
                cand = (cand + cand[::-1]) * 0.5
                cand = _project_simplex(cand)

            # evaluate candidate
            _, maxc_c, _ = _conv_grad(cand, eps, L)
            cand_score = 2.0 * n * maxc_c

            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                no_improve = 0
                if cur_score < best_score:
                    best_score = cur_score
                    best_vec = cur.copy()
            else:
                no_improve += 1
                if no_improve > 2000:
                    # random mass transfer to shake the state
                    i_idx = rng.integers(0, n)
                    if cur[i_idx] > 0.0:
                        delta = cur[i_idx] * rng.random() * 0.2
                        j_idx = rng.integers(0, n)
                        if j_idx == i_idx:
                            j_idx = (j_idx + 1) % n
                        tmp = cur.copy()
                        tmp[i_idx] -= delta
                        tmp[j_idx] += delta
                        tmp = _project_simplex(tmp)
                        _, maxc_tmp, _ = _conv_grad(tmp, eps, L)
                        tmp_score = 2.0 * n * maxc_tmp
                        if tmp_score < cur_score:
                            cur = tmp
                            cur_score = tmp_score
                            no_improve = 0
                            if tmp_score < best_score:
                                best_score = tmp_score
                                best_vec = tmp.copy()
                    else:
                        # small random restart to escape a flat region
                        cur = rng.random(n)
                        cur = _project_simplex(cur)

            # early stop if we have already reached the target bound
            if best_score < 1.50274349:
                break

        if best_score < 1.50274349:
            break

    # --------------------------------------------------------
    # Fallback (should never happen, but keep safety)
    # --------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # return plain Python list as required by the harness
    return [float(x) for x in best_vec]
