def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises the
    AC1 upper bound defined by ``evaluate_sequence``.
    The theoretical optimum under the given constraints is the uniform
    distribution (all entries equal, sum = 1).  We therefore return a uniform
    vector of the appropriate length.
    """
    # Determine the length of the candidate.
    # If a parent construction is available use its length, otherwise fall back
    # to a modest size (1024).
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 1024

    # Uniform vector with L1‑norm equal to 1.
    # (Any scaling leaves the score unchanged, but the evaluator normalises
    #  internally, so fixing sum = 1 is convenient.)
    uniform_vec = np.full(n, 1.0 / n, dtype=np.float64)
