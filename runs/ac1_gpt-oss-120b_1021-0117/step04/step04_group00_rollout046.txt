import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact evaluator (used for comparing candidates internally)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(convolution(a,a)) / (sum(a)^2)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    maxb = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * maxb / (s * s))


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function whose AC1 bound is small."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise from the provided parent or from a random start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2048, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    cur = _project_simplex(a0)          # sum = 1, non‑negative
    n = cur.size
    # FFT length sufficient for convolutions with a mask of length 2n‑1
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # --------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4                # base learning rate
    beta1_pow = 1.0
    beta2_pow = 1.0

    # --------------------------------------------------------
    # Smoothing schedule for the soft‑max weights
    # --------------------------------------------------------
    eps0 = 1.0
    eps_decay = 0.9997
    eps_min = 1e-9

    target = 1.50274349      # stop when we beat the target
    it = 0

    # --------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution (full) ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val               # because sum(cur)==1

        # ---- soft‑max weights approximating the arg‑max ----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ---- gradient of the smoothed objective ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # selects the needed slice
        g = 2.0 * grad_full[idx]

        # ---- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- tentative move and projection onto simplex ----
        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ---- acceptance (mostly greedy, occasional uphill) ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
        else:
            # tiny probability to accept a worse move, helps escape plateaus
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc

        # ---- keep the best ever seen ----
        if cand_sc < best_sc:
            best_vec = cand.copy()
            best_sc = cand_sc
            if best_sc < target:
                break

        # ---- occasional symmetrisation (exploits Toeplitz symmetry) ----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
                if best_sc < target:
                    break

        # ---- random mass‑transfer when progress stalls ----
        if it % 2000 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

    # --------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
