def compute_gradient(a, eps):
    n = a.size
    L = 1 << ((3 * n - 2).bit_length())  # ensures >2n-2 and >n etc.
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n - 1]
    maxval = conv.max()
    w = np.exp((conv - maxval) / eps)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g

    arcsine = 1.0 / np.sqrt(x * (1.0 - x))
    arcsine = arcsine / arcsine.sum()

    # ------------------------------------------------------------------
    # 3️⃣  Pick best among cheap candidates (score = 2·n·max(conv))
    # ------------------------------------------------------------------
    def _conv_max(v):
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(conv.max())

    best_vec = None
    best_score = np.inf
    for cand in (base, uniform, arcsine):
        maxv = _conv_max(cand)
        sc = 2.0 * n * maxv
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
    cur = best_vec.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # 4️⃣  Optimisation parameters
    # ------------------------------------------------------------------
    eps = 1.0
    eps_decay = 0.9965
    eps_min = 1e-6

    beta1, beta2 = 0.9, 0.999
    rho1, rho2 = 1.0, 1.0
    m = np.zeros(n)
    v = np.zeros(n)
    base_lr = 0.5               # learning‑rate factor inside the exponent
    L = 1 << ((3 * n - 2).bit_length())   # length for FFT‑based gradient
    iteration = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # 5️⃣  Main optimisation loop (matrix‑free, FFT‑based)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ----- convolution and soft‑max weights (smooth max) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam‑style moment updates -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        rho1 *= beta1
        rho2 *= beta2
        m_hat = m / (1.0 - rho1)
        v_hat = v / (1.0 - rho2)

        # ----- Exponentiated (mirror) update -----
        step = -base_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        step = np.clip(step, -20.0, 20.0)          # avoid overflow
        cur = cur * np.exp(step)

        # ----- Projection onto the simplex (just normalise) -----
        s = cur.sum()
        if s <= 0.0 or not np.isfinite(s):
            cur = uniform.copy()
            s = cur.sum()
        cur = cur / s

        # ----- occasional symmetrisation (helps due to inherent symmetry) -----
        if iteration % 2000 == 0:
            cur = (cur + cur[::-1]) * 0.5

        # ----- schedule for the smoothing parameter -----
        eps = max(eps_min, eps * eps_decay)

        # ----- current score (sum is 1, so score = 2·n·max(conv)) -----
        cur_score = 2.0 * n * max_val

        # ----- keep the best ever seen -----
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- random restart if stuck for a while -----
        if no_improve > 8000:
            cur = rng.exponential(scale=1.0, size=n)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            rho1, rho2 = 1.0, 1.0
            eps = 1.0
            no_improve = 0

        # ----- early exit if target reached -----
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # 6️⃣  Return best vector found (as plain Python list)
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    return [float(x) for x in best_vec]
