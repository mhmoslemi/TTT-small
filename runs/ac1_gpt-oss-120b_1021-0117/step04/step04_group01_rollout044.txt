import time, math, numpy as np

# ---------- helpers ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a: np.ndarray, fft_len: int) -> float:
    """Return the AC1 bound for vector a (sum may be arbitrary)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_val / (s * s))

        cand_sc = _score(cand, fft_len=L)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc

    # return as plain Python list
    return [float(x) for x in best]
