import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Scoring function – identical to the evaluator up to the sum=1 scaling
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a,a)) for a sum‑to‑1 vector."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Core optimisation routine (Adam + soft‑max smoothing)
# ------------------------------------------------------------
def _optimize(init_vec: np.ndarray, deadline: float, rng: np.random.Generator,
              target: float) -> tuple[np.ndarray, float]:
    """Run the Adam‑style optimiser from init_vec until deadline."""
    a = init_vec.copy()
    n = a.size
    L = 1 << ((3 * n - 2).bit_length())

    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8

    # smoothing schedule (soft‑max of the convolution)
    eps0, eps_decay, eps_min = 0.5, 0.9995, 1e-12

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_sc = _score(a)
    best_vec, best_sc = a.copy(), cur_sc
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and its maximum -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights (smooth approximation of max) -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # optional clipping of the step norm (helps stability)
        max_norm = 0.2
        norm = np.linalg.norm(step)
        if norm > max_norm:
            step = step * (max_norm / norm)

        cand = a - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- keep the best ever seen -----
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc

        # ----- acceptance rule (mostly greedy, occasional uphill) -----
        if cand_sc <= cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % 400 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    a, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- early exit if target achieved -----
        if best_sc < target:
            break

    return best_vec, best_sc

# Public entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialise from the parent if available ----
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2048
        a0 = rng.random(n, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)          # enforce sum = 1, non‑negative

    # ---- primary optimisation run ----
    best_vec, best_sc = _optimize(a0, deadline, rng, target=1.50274349)

    # ---- a few cheap random restarts if time permits ----
    for _ in range(3):
        if time.monotonic() >= deadline:
            break
        rnd = rng.random(n, dtype=np.float64)
        rnd = _project_simplex(rnd)
        vec, sc = _optimize(rnd, deadline, rng, target=1.50274349)
        if sc < best_sc:
            best_vec, best_sc = vec, sc

    # ---- final fallback (should never happen) ----
    if best_vec is None or best_vec.size == 0:
        best_vec = a0

    return [float(x) for x in best_vec]
