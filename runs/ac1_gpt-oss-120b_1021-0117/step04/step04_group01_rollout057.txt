import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny numerical drift
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / n
    return w

# ------------------------------------------------------------
# Fast computation of 2*n*max(conv(a,a)) (sum(a) assumed to be 1)
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

def _score(a: np.ndarray, L: int) -> float:
    """Exact score as used by the evaluator (sum(a) is 1)."""
    n = a.size
    return 2.0 * n * _max_conv(a, L)

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # -------------------- initialise length & FFT size --------------------
    if "height_sequence_1" in globals():
        n = int(height_sequence_1.shape[0])
    else:
        n = 2500                     # a reasonable default if no parent

    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑2 >= 2n‑1

    # -------------------- helper to normalise a vector --------------------
    def _normalise(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0:
            v = np.full_like(v, 1.0 / v.size)
        else:
            v = v / s
        return v

    # -------------------- candidate factory (draws) --------------------
    best_vec = None
    best_sc = math.inf

    # --- base candidate: copy of the parent if available ----------------
    if "height_sequence_1" in globals():
        a0 = _normalise(np.asarray(height_sequence_1, dtype=np.float64).copy())
        sc0 = _score(a0, L)
        best_vec, best_sc = a0, sc0

    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center) / center   # ∈ [0,1]

    # ----- simple parametric families (few dozen candidates) -----
    # 1) Geometric decreasing tails
    for r in (0.9995, 0.997, 0.995, 0.99, 0.98):
        a = np.power(r, i)
        a = _normalise(a)
        sc = _score(a, L)
        if sc < best_sc:
            best_vec, best_sc = a, sc

    # 2) Symmetric shapes defined by a distance‑to‑center transform
    #    (linear, cosine, power, exponential)
    # linear
    a = np.maximum(0.0, 1.0 - dist)
    a = _normalise(a)
    sc = _score(a, L)
    if sc < best_sc:
        best_vec, best_sc = a, sc
    # cosine window
    a = (1.0 + np.cos(np.pi * dist)) * 0.5
    a = _normalise(a)
    sc = _score(a, L)
    if sc < best_sc:
        best_vec, best_sc = a, sc
    # power family
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        a = np.power(np.maximum(0.0, 1.0 - dist), p)
        a = _normalise(a)
        sc = _score(a, L)
        if sc < best_sc:
            best_vec, best_sc = a, sc
    # exponential decay
    for a_exp in (0.2, 0.5, 1.0, 1.5):
        a = np.exp(-a_exp * dist)
        a = _normalise(a)
        sc = _score(a, L)
        if sc < best_sc:
            best_vec, best_sc = a, sc

    # 3) Sinusoidal family (non‑negative on (0,π))
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        a = np.sin(np.pi * (i + 0.5) / n) ** p
        a = _normalise(a)
        sc = _score(a, L)
        if sc < best_sc:
            best_vec, best_sc = a, sc

    # 4) Random Dirichlet draws (few of them)
    for alpha in (0.2, 0.5, 0.8):
        a = rng.dirichlet(np.full(n, alpha))
        sc = _score(a, L)
        if sc < best_sc:
            best_vec, best_sc = a, sc

    # --------------------------------------------------------------------
    # If for any reason we still have no vector, fall back to uniform
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_sc = _score(best_vec, L)

    # --------------------------------------------------------------------
    # Gradient‑descent refinement (projected sub‑gradient on the simplex)
    cur = best_vec.copy()
    cur_sc = best_sc

    # Hyper‑parameters
    eps0 = 0.4                     # initial smoothing for soft‑max
    eps_decay = 0.9998
    eps_min = 1e-14
    step0 = 0.9                    # base step magnitude
    max_backtrack = 8
    backtrack_factor = 0.5
    no_improve = 0
    it = 0
    idx = (2 * n - 2) - np.arange(n)   # index map used for gradient extraction

    while time.monotonic() < deadline:
        it += 1

        # ----- compute convolution and soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the smooth max (via convolution) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[idx]          # sub‑gradient approximation

        # ----- back‑tracking line search -----
        step_base = step0 / math.sqrt(it)
        step = step_base
        improved = False
        for _ in range(max_backtrack):
            cand = cur - step * g
            cand = _project_simplex(cand)
            # enforce symmetry each iteration (helps reduce max‑conv)
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)   # keep on simplex
            cand_sc = _score(cand, L)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_vec, best_sc = cand.copy(), cand_sc
                    no_improve = 0
                improved = True
                break
            step *= backtrack_factor

        if not improved:
            no_improve += 1

        # ----- occasional symmetrisation (cheap) -----
        if it % 3000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- random perturbation when stuck -----
        if no_improve > 3500:
            # small redistribution of mass between two random indices
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n - 1)
            if i2 >= i1:
                i2 += 1
            if cur[i1] > 0:
                delta = cur[i1] * rng.random() * 0.25
                tmp = cur.copy()
                tmp[i1] -= delta
                tmp[i2] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- early stop if we hit the target -----
        if best_sc < 1.50274349:
            break

    # ------------------- final fine‑tuning (Gaussian noise) -------------------
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.004, size=n)
        cand = best_vec + delta
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)
        if cand_sc < best_sc:
            best_vec, best_sc = cand, cand_sc

    # -------------------------------------------------------------------------
    # Return as plain Python list (ensuring finite numbers)
    return [float(x) for x in best_vec]
