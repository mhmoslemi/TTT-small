import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Compute max of the (linear) convolution a*a  (full length)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Gradient of the (approximate) max‑convolution.
# Here we average the true sub‑gradients of the `top_k` largest
# convolution entries.
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        # indices of the `top_k` largest entries (unsorted)
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]          # sub‑gradient of (a*a)[k]
    g /= max(1, top_k)
    return g
