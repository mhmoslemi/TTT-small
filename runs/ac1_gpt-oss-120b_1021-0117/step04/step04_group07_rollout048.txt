import time, math, numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector that already satisfies sum=1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a short safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initial vector (use parent if present)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s  # normalize to simplex

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())  # power of two for FFT

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    cur = a0.copy()
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------
    eps = 0.01          # soft‑max temperature
    eps_min = 1e-12
    eps_decay = 0.9995

    lr = 0.10           # base learning rate for Adam
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    it = 0
    no_improve = 0
    target_score = 1.50274349

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current vector -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- soft‑max weights for gradient -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0 or np.isnan(w_sum):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # pure sub‑gradient: put all mass on the maximal index
            w = np.zeros_like(conv)
            w[int(conv.argmax())] = 1.0

        # ----- gradient of weighted max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        bias_c1 = 1.0 - beta1 ** it
        bias_c2 = 1.0 - beta2 ** it
        m_hat = m / bias_c1
        v_hat = v / bias_c2

        step = (lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ----- candidate and projection onto simplex -----
        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _fast_score(cand, L, n)

        # ----- accept if improvement, else possibly anneal -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # simple simulated‑annealing acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- occasional symmetrisation (helps) -----
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _fast_score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- temperature decay -----
        eps = max(eps_min, eps * eps_decay)

        # ----- early exit if target reached -----
        if best_sc < target_score:
            break

    # --------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best]
