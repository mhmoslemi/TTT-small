import time, numpy as np

def _score_vec(v: np.ndarray) -> float:
    """Exact AC1 bound for a non‑negative vector with sum = 1."""
    n = v.size
    # max of convolution = sum of squares (always at lag 0)
    return float(2.0 * n * np.dot(v, v))

def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑negative step‑function (list of floats) with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # reserve a little margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- length -------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # reasonable fallback size

    # ----- start from the uniform distribution (optimal for sum‑squares) -----
    cur = np.full(n, 1.0 / n, dtype=np.float64)   # sum = 1, non‑negative
    best = cur.copy()
    best_sc = _score_vec(best)

    # ----- simple stochastic local search (unlikely to beat the optimum,
    # but safely explores the neighbourhood) -----------------------
    while time.monotonic() < deadline:
        # transfer a random fraction of mass from i to j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.5   # move at most 50 % of i's mass
        if delta == 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep feasibility
        if cand[i] < 0.0:
            continue
        sc = _score_vec(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
            cur = cand
            # tiny chance we reach a remarkable value – stop early
            if best_sc < 1.50274349:
                break
        # occasional reset to uniform to avoid stagnation
        if rng.random() < 0.001:
            cur = np.full(n, 1.0 / n, dtype=np.float64)

    # ----- return plain Python list --------------------------------
    return [float(x) for x in best]
