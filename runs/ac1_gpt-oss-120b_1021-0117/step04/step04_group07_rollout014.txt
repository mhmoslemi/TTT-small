    L = 1 << ((2*n - 1).bit_length())
    # could precompute indices for gradient slice

    # helper functions
    def conv_fft(vec):
        """Return linear convolution a*a using FFT length L."""
        a_fft = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n - 1]
        return conv

    def score(vec):
        conv = conv_fft(vec)
        return float(2.0 * n * conv.max())

    # initial evaluation
    cur = a0.copy()
    cur_sc = score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # parameters
    eps = 0.01  # start smoothing
    eps_min = 1e-8
    eps_decay = 0.9995

    step0 = 0.6
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    while time.monotonic() < deadline:
        it += 1
        # compute convolution and gradient
        a_fft = np.fft.rfft(cur, n=L)

            w = np.exp((conv - maxc) / eps)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only works on 1‑D arrays")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Core optimizer
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs) -> list:
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initial vector (use parent if available)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform on a modest length
        n_fallback = 2500
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                      # renormalise to sum = 1

    n = a0.size
    # FFT length for a linear convolution of two length‑n vectors
    L = 1 << ((2 * n - 1).bit_length())
    idx_slice = (2 * n - 2) - np.arange(n)   # indices for the gradient

    # --------------------------------------------------------
    # helper: linear convolution a*a via FFT (returns length 2n‑1)
    # --------------------------------------------------------
    def _conv_fft(vec: np.ndarray) -> np.ndarray:
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return conv

    # --------------------------------------------------------
    # helper: current AC1 bound (sum already = 1)
    # --------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        conv = _conv_fft(vec)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    cur = a0.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------

    eps_min = 1e-8
    eps_decay = 0.9995

    beta1 = 0.9                # Adam first‑moment decay
    beta2 = 0.999              # Adam second‑moment decay
    m = np.zeros_like(cur)     # first moment
    v = np.zeros_like(cur)     # second moment

    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------

    while time.monotonic() < deadline:
        it += 1

        # ---- convolution of current iterate ----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ---- gradient (soft‑max approximation) ----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)          # soft‑max weights
            w_sum = w.sum()
            if w_sum == 0.0:

        else:

            k = int(conv.argmax())

            g[idx] = 2.0 * cur[k - idx]

        # ---- Adam update ----
