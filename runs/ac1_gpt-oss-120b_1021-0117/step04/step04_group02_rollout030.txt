def propose_candidate(seed=42, budget_s=1100, **kwargs):
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    rng = np.random.default_rng(seed)

    if 'height_sequence_1' in globals():

    else:
        # fallback exponential
        n_fallback = 2500
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t/70)  # more aggressive? Choose -t/70 for slower decay? Not sure.
    a0 = np.maximum(a0, 0.0)
    if a0.sum() <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((3*n - 2).bit_length())
    cur = a0.copy()
    # Force symmetry
    cur = (cur + cur[::-1]) / 2
    cur = _project_simplex(cur)
    best = cur.copy()
    best_sc = _score(cur)
    # Pre-alloc variables
    m = np.zeros_like(cur)   # first moment
    v = np.zeros_like(cur)   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t_iter = 0

    # Smoothing parameter
    eps_smooth = None
    eps_decay = 0.9995

    no_improve = 0

    while time.monotonic() < deadline:
        t_iter += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n-1]
        max_val = conv_full.max()

        # smoothing weight
        if eps_smooth is None:
            eps_smooth = max(max_val * 0.2, 1e-9)
        else:
            eps_smooth = max(eps_smooth * eps_decay, 1e-9)
        # Compute w via softmax
        w = np.exp((conv_full - max_val) / eps_smooth)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient: conv of a with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)  # mapping convolution indices to gradient positions
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = 0.1 / (np.sqrt(v_hat) + eps_adam)   # base learning rate adapt
        # compute alpha scaling as in baseline: further scaling factor
        # We can compute raw update = step * m_hat
        update = step * m_hat

        # Apply update
        cand = cur - update
        # project onto simplex
        cand = _project_simplex(cand)
        # enforce symmetry
        cand = (cand + cand[::-1]) / 2.0
        cand = _project_simplex(cand)  # ensure sum 1

        cand_sc = _score(cand)
        cur_sc = _score(cur)
        # Accept if improvement or with probability (Simulated annealing) maybe?
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Metropolis-like acceptance
            T = max(0.01, 1.0 / (t_iter ** 0.5))
            prob = np.exp(-(cand_sc - cur_sc) / T)
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc
                no_improve += 0
            else:
                no_improve += 1

        # Periodic symmetrization (already always symmetric but keep)
        # maybe also random perturbation when stuck
        if no_improve > 2000:
            # Random perturbation: pick random indices and redistribute mass
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)

        # early stop if target reached?

# ------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a: np.ndarray, L: int) -> float:

    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
