import time

import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score (exact for sum=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of heights) with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initialise from parent or simple fallback
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
    else:
        # modest size fallback
        n = 1024
        base = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise onto the simplex
    base = _project_simplex(base)
    n = base.size

    # Additional cheap candidates (uniform, arcsine)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()

    candidates = [base, uniform, arcsine]
    best = None
    best_sc = float("inf")
    for cand in candidates:
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
    cur = best.copy()
    cur_sc = best_sc

    # --------------------------------------------------------
    # Parameters for the smoothed‑max descent
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())          # >= 3n-2 for gradient conv
    eps = 1.0
    eps_decay = 0.9985
    eps_min = 1e-6
    step0 = 0.6
    beta = 0.9                                     # momentum factor
    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution of cur with itself (FFT, exact linear conv) ----
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]

        # ---- smooth‑max weights (log‑sum‑exp) ----
        x = conv / eps
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        sumexp = exp_shift.sum()
        w = exp_shift / sumexp          # sum(w)=1, approximates argmax

        # ---- gradient of the smooth‑max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * g

        # ---- step size ----
        grad_norm = np.linalg.norm(momentum) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm

        # ---- candidate and projection ----
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)

        # occasional symmetrisation (helps with the inherent symmetry)
        if (it % 400) == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # try the symmetric version as a separate point
        cand_sym = (cand + cand[::-1]) * 0.5
        cand_sym = _project_simplex(cand_sym)

        # evaluate both (fast FFT score)
        sc1 = _score(cand)
        sc2 = _score(cand_sym)
        if sc2 < sc1:
            cand, sc1 = cand_sym, sc2

        # ---- accept if improvement ----
        if sc1 < cur_sc - 1e-12:
            cur, cur_sc = cand, sc1
            no_improve = 0
            if sc1 < best_sc:
                best_sc, best = sc1, cand.copy()
        else:
            no_improve += 1

        # ---- occasional random mass transfer when stuck ----
        if no_improve > 2500:
            # small perturbation of the best solution
            tmp = best.copy()
            i_idx = rng.integers(0, n)
            j_idx = rng.integers(0, n)
            if i_idx != j_idx:
                delta = tmp[i_idx] * rng.random() * 0.2
                tmp[i_idx] -= delta
                tmp[j_idx] += delta
            tmp = _project_simplex(tmp)
            sc_tmp = _score(tmp)
            if sc_tmp < best_sc:
                best_sc, best = sc_tmp, tmp.copy()
                cur, cur_sc = tmp, sc_tmp
                momentum = np.zeros_like(cur)
                no_improve = 0
                # continue with fresh start
                continue
            # fallback to a fresh random start near the best
            cur = best.copy()
            cur = _project_simplex(cur + rng.normal(scale=0.01, size=n))
            momentum = np.zeros_like(cur)
            no_improve = 0

        # ---- temperature decay ----
        eps = max(eps_min, eps * eps_decay)

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Final local random refinement (if time permits)
    # --------------------------------------------------------
    while time.monotonic() < deadline - 0.05:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.15
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best = sc, cand.copy()

    # --------------------------------------------------------
    # Return best vector as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
