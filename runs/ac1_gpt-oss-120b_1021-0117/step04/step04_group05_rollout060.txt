def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    best_vec = None
    best_sc = float('inf')
    
    # Determine length n
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2048  # fallback length; maybe use 1024
    # Precompute FFT length
    L_full = 1 << ((3 * n - 2).bit_length())
    
    def run_once(init_vec):
        nonlocal best_vec, best_sc
        cur = init_vec.copy()
        # ensure simplex:
        cur = _project_simplex(cur)
        v = np.zeros_like(cur)
        cur_fft = np.fft.rfft(cur, n=L_full)
        # compute initial conv and score
        conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
        eps_rel = 0.01
        eps_min = 1e-12
        step0 = 1.0
        mu = 0.9
        it = 0
        no_improve = 0
        while time.monotonic() < deadline:
            it += 1
            # weight smoothed max
            eps = max(eps_min, cur_max * eps_rel)
            w = np.exp((conv - cur_max) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum
            # gradient
            w_rev = w[::-1]
            w_rev_fft = np.fft.rfft(w_rev, n=L_full)
            grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_full)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
            # momentum step
            alpha = step0 / math.sqrt(it)
            v = mu * v - alpha * g
            cand = cur + v
            cand = _project_simplex(cand)
            cand_fft = np.fft.rfft(cand, n=L_full)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[: 2 * n - 1]

                cur_fft = cand_fft
                conv = cand_conv
                cur_max = cand_conv.max()
                cur_sc = cand_sc
                v = np.zeros_like(cur)  # reset momentum after success
                no_improve = 0

                    best_vec = cur.copy()

import time
import math
import numpy as np

# -------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a short safety margin before the hard timeout

    step0 = 1.0              # initial step magnitude
    max_backtrack = 8        # line‑search attempts
    no_improve_limit = 3000

    best_vec = None
    best_sc = float("inf")

    # -------------------------------------------------------------
    def optimise(init_vec: np.ndarray):
        """Run a (projected) gradient descent from init_vec."""
        nonlocal best_vec, best_sc
        cur = np.maximum(init_vec, 0.0)
