import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score (2 * n * max(conv) / sum(a)^2) using FFT
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int, n: int) -> float:
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    maxc = conv.max()
    s = a.sum()
    if s <= 0.0:
        return float('inf')
    return float(2.0 * n * maxc / (s * s))


# ------------------------------------------------------------
# Gradient of the soft‑max approximation of max(conv)
# ------------------------------------------------------------
def _grad(a: np.ndarray, a_fft: np.ndarray, conv: np.ndarray,
          maxc: float, eps: float, L: int, n: int) -> np.ndarray:
    """Return ∇_a (ε·log Σ exp(conv/ε))."""
    if eps > 1e-12:
        w = np.exp((conv - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
    else:
        # sub‑gradient: weight only the max entry
        w = np.zeros_like(conv)
        w[int(conv.argmax())] = 1.0
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

            no_improve = 0

        # ----- temperature decay -----
        eps = max(eps_min, eps * eps_decay)

    # -----------------------------------------------------------------
    # Return best vector found (as plain Python list)
    # -----------------------------------------------------------------
    return [float(x) for x in best]
