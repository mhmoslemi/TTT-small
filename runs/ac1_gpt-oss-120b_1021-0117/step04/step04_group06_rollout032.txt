import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex (non‑negative, sum = 1)."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a vector with sum == 1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv(a,a)) using an FFT of length L."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients with a low AC1 bound.
    The routine runs for at most `budget_s` seconds and keeps the best
    candidate found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()

        cur = np.exp(-np.arange(n_fallback, dtype=np.float64) / 20.0)
    cur = np.clip(cur, 0.0, 1000.0)

    # normalise to the simplex (sum = 1)
    total = cur.sum()
    if total <= 0.0:
        cur = np.full_like(cur, 1.0 / cur.size)

        cur = _project_simplex(cur)

    n = cur.shape[0]

    # current iterate

    # -------------------- hyper‑parameters --------------------
