    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.8)   # safety margin

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # exponential fallback, length 1024
        n_fallback = 1024
        x = np.arange(n_fallback, dtype=np.float64)
        cur = np.exp(-x / (n_fallback / 10.0))
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # sum == 1, non‑negative
    n = cur.shape[0]

    # ---- FFT lengths ------------------------------------------------
    L_full = 1 << ((3 * n - 2).bit_length())   # >= 3n‑2  (gradient)
    L_conv = 1 << ((2 * n - 1).bit_length())   # >= 2n‑1  (scoring)

    # ---- bookkeeping ------------------------------------------------
    cur_fft = np.fft.rfft(cur, n=L_full)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_sc = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---- optimisation parameters ------------------------------------
    step0 = 0.8
    beta = 0.92                     # momentum factor
    m = np.zeros_like(cur)          # momentum accumulator
    eps_start = 0.02
    eps_min_rel = 1e-4
    eps_min = 1e-9
    backtrack_max = 8
    no_improve = 0
    it = 0

    # ---- main loop -------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # adapt smoothing parameter
        eps_rel = max(eps_min_rel, eps_start * math.exp(-it / 5000.0))
        eps = max(eps_min, cur_max * eps_rel)

        # soft‑max weights over the convolution values
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the smoothed max (2 * f * w_rev)
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L_full)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_full)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        m = beta * m + (1.0 - beta) * g

        # line‑search with back‑tracking
        alpha = step0 / math.sqrt(it)
        accepted = False
        for _ in range(backtrack_max):
            cand = cur - alpha * m
            cand = _project_simplex(cand)
            cand_fft = np.fft.rfft(cand, n=L_full)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[: 2 * n - 1]
            cand_sc = 2.0 * n * cand_conv.max()
            if cand_sc < cur_sc:
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            # simulated‑annealing style acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                accepted = True

        if accepted:
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_conv.max()
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1

        # ---- periodic symmetrisation --------------------------------
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L_full)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L_full)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_conv.max()
                cur_sc = sym_sc
                no_improve = 0
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()

        # ---- random mass‑transfer when stuck -------------------------
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        # ---- early exit if target achieved ----------------------------
        if best_sc < 1.50274349:
            break

        # ---- time guard ------------------------------------------------
        if time.monotonic() >= deadline:
            break

    # -------------------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # -------------------------------------------------------------------
    return [float(x) for x in best_vec]
