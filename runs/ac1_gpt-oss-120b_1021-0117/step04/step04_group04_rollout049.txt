import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x>=0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny drift
    s = w.sum()
    if s == 0:
        return np.full_like(v, 1.0 / n)
    return w / s


# ------------------------------------------------------------
# Fast AC1 upper‑bound for a sequence with sum == 1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(convolution) assuming sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        n = len(height_sequence_1)
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)          # sum == 1, non‑negative
        best_seq = a0.copy()
        best_sc = _score(best_seq, 1 << ((2 * n - 1).bit_length()))
    else:
        # fallback – uniform distribution of length 2000
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_sc = _score(best_seq, 1 << ((2 * n - 1).bit_length()))

    # FFT length sufficient for all convolutions (up to length 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration
    # ------------------------------------------------------------------
    phase1_deadline = start + 0.30 * (deadline - start)  # ~30 % of total time
    while time.monotonic() < phase1_deadline:
        # pick a random generative model
        typ = rng.integers(0, 4)
        if typ == 0:                     # uniform
            a = np.full(n, 1.0 / n, dtype=np.float64)
        elif typ == 1:                   # exponential decay
            tau = rng.uniform(30.0, 500.0)
            a = np.exp(-np.arange(n, dtype=np.float64) / tau)
            a /= a.sum()
        elif typ == 2:                   # power‑law
            p = rng.uniform(0.2, 4.0)
            a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            a /= a.sum()
        else:                            # few equal spikes
            k = int(rng.integers(30, min(200, n)))
            idx = rng.choice(n, size=k, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[idx] = 1.0 / k

        # small Gaussian jitter to increase diversity
        if rng.random() < 0.3:
            a += rng.normal(scale=0.005, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

        sc = _score(a, L)
        if sc < best_sc:
            best_sc = sc
            best_seq = a.copy()
            # early exit if already under target
            if best_sc < 1.50274349:
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # Phase 2 – smoothed gradient descent (Adam) from the best found
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    step_alpha = 0.12

    # smoothing schedule for the log‑sum‑exp surrogate
    eps0, eps_decay, eps_min = 0.6, 0.9995, 1e-8

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current sequence with itself -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smooth approximation of the max  -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # length >= 3n‑2
        idx = (2 * n - 2) - np.arange(n)               # slice needed
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_alpha * m_hat / (np.sqrt(v_hat) + adam_eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # occasional symmetrisation (helps because the objective is symmetric)
        if it % 2000 == 0:
            sym = (cand + cand[::-1]) * 0.5
            sym = _project_simplex(sym)
            cand = sym

        cand_sc = _score(cand, L)

        # accept the new point (always move) – keep track of best ever
        cur = cand
        cur_sc = cand_sc
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_seq = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # if we stall for a long time, inject a small random perturbation
        if no_improve > 4000:
            pert = rng.normal(scale=0.01, size=n)
            cand = cur + pert
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cand_sc = _score(cand, L)
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_seq = cand.copy()
                cur = cand
                cur_sc = cand_sc
            no_improve = 0

        # early stop when target reached
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
