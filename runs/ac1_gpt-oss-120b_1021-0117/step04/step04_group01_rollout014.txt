# ----- helper: Euclidean projection onto the probability simplex -----
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum x = 1}."""
    # implementation of the algorithm by Wang & Carreira-Perpinán
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]

        theta = 0.0

                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, fft_len=L)
                if tmp_sc < cur_sc:
                    prev = cur
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early termination if target reached -----------------------
        if best_sc < 1.50274349:
            break

    # ----- final fine‑tuning with tiny Gaussian steps -------------------
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.003, size=n)
        cand = best + delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand, fft_len=L)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc

    # return as plain Python list
    return [float(x) for x in best]
