import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 bound using FFT (assumes a is a 1‑D NumPy array)
# ------------------------------------------------------------
def _ac1_score(a: np.ndarray, L_conv: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L_conv)
    conv = np.fft.irfft(af * af, n=L_conv)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

# ------------------------------------------------------------
# Core optimisation routine (gradient descent + ADAM)
# ------------------------------------------------------------
def _optimise(a0: np.ndarray, budget: float, target: float) -> np.ndarray:
    n = a0.shape[0]
    # FFT lengths
    L_conv = 1 << ((2 * n - 1).bit_length())      # for a*a
    L_full = 1 << ((3 * n - 2).bit_length())      # for a * w_rev

    # ADAM hyper‑parameters
    base_lr = 0.35
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    # smoothing (soft‑max) schedule
    eps0, eps_min = 0.5, 1e-7

    # state
    cur = a0.copy()
    cur_sc = _ac1_score(cur, L_conv)
    best = cur.copy()
    best_sc = cur_sc

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    start = time.monotonic()
    deadline = start + budget - 0.3   # a tiny safety margin
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() < deadline:
        it += 1
        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L_full)
        conv = np.fft.irfft(a_fft * a_fft, n=L_full)[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights -----
        eps = max(eps_min, eps0 / math.sqrt(it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L_full)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L_full)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- ADAM update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)

    """Return a non‑empty list of non‑negative coefficients with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + float(budget_s)

    # ---- initial candidate -------------------------------------------------

    # ---- convert to plain Python list ----------------------------------------
    return [float(x) for x in best_overall]
