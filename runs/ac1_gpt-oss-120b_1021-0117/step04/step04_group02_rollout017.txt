    cur = a0
    n = cur.size

    best_vec = cur.copy()
    best_sc = cur_sc

    eps = None
    eps_decay = 0.99997
    base_alpha = 1.2
    sym_interval = 250
    max_no_improve = 8000
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:

        a_fft = np.fft.rfft(cur, n=L)

        # compute smoothing weight w
        if eps is None:
            eps = max(max_val * 0.1, 1e-12)

            eps = max(eps * eps_decay, 1e-12)

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
