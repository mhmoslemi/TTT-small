import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(convolution(a,a)) – a is assumed to sum to 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Gradient of the smoothed max‑convolution (or exact sub‑gradient)
# ------------------------------------------------------------
def _gradient(a: np.ndarray, eps: float, L: int, idx_sel: np.ndarray) -> np.ndarray:
    """Return the gradient of the (soft) max‑convolution w.r.t. a."""
    a_fft = np.fft.rfft(a, n=L)

    # full auto‑convolution
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]

    # weights for the (soft) max
    max_val = conv_full.max()
    if eps < 1e-12:                         # exact sub‑gradient
        mask = np.isclose(conv_full, max_val, atol=1e-12)
        w = np.empty_like(conv_full)
        w.fill(0.0)
        w[mask] = 1.0 / mask.sum()
    else:                                   # smooth soft‑max
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

    # gradient = 2 * a * reversed(w) (convolution)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    g = 2.0 * grad_full[idx_sel]
    return g

# Candidate generators (non‑negative, not yet normalised)

def _gen_uniform(n):
    return np.ones(n, dtype=np.float64)

def _gen_hann(n):
    if n == 1:
        return np.ones(1, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return w

def _gen_cosbell(n):
    """sin²‑bell: sin(pi*(i+0.5)/n)², symmetric and zero at the ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** 2
    return w

def _gen_random(n, rng):
    return rng.random(n, dtype=np.float64)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return

    # --------------------------------------------------------
    # Determine problem size and a reasonable initial vector
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

    starters = []

            no_improve += 1

        if no_improve > 3000:

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
