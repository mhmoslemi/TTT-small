                    cur_sc = tmp_sc

                    best_sc = tmp_sc

            break

        if no_improve > 5000:

            pert += noise
            pert = np.maximum(pert, 0.0)

                cur = pert
                cur_sc = pert_sc
            if pert_sc < best_sc:
                best = pert.copy()
                best_sc = pert_sc
            no_improve = 0

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – fixes sum to 1, non‑negative
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score function used during optimisation (sum is 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """2 * n * max(convolution(a,a)) – assumes sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin before the sandbox kills the process
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start.
    # Sort decreasingly to give a good asymmetric baseline.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    a0 = np.sort(a0)[::-1]               # decreasing shape
    a0 = _project_simplex(a0)            # enforce sum = 1
    cur = a0.copy()
    n = cur.size

    # ------------------------------------------------------------
    # FFT length – power of two covering the full convolution length
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.9               # Adam first‑moment decay
    beta2 = 0.999            # Adam second‑moment decay
    m = np.zeros_like(cur)   # first moment
    v = np.zeros_like(cur)   # second moment
    lr = 0.2                 # base learning rate
    eps_start = 1.0
    eps_decay = 0.9992
    eps_min = 1e-8
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights over all shifts -----
        eps = max(eps_min, eps_start * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
