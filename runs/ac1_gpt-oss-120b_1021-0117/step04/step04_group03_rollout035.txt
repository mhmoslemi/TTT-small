# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

                    cand_bt = cur - alpha * factor * g
                    cand_bt = _project_simplex(cand_bt)
                    cand_bt_sc = _score(cand_bt)
                    if cand_bt_sc < cur_sc:
                        cur, cur_sc = cand_bt, cand_bt_sc
                        improved = True
                        break

        if improved:
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()
                no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                # random perturbation to escape a flat region
                i_idx = rng.integers(0, n)
                if cur[i_idx] > 0.0:
                    delta = cur[i_idx] * rng.random() * 0.3
                    j_idx = rng.integers(0, n)
                    if j_idx == i_idx:
                        j_idx = (j_idx + 1) % n
                    tmp = cur.copy()
                    tmp[i_idx] -= delta
                    tmp[j_idx] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()
                        no_improve = 0
                else:
                    no_improve = 0

        # occasional symmetrisation (exploits the intrinsic symmetry of the problem)
        if (it % 400) == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()

        # early stop if we already beat the target
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # --------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    # ensure a tiny safety margin on the sum (evaluator requires sum >= 0.01)
    if best_vec.sum() < 0.01:
        best_vec = best_vec / best_vec.sum() * 0.01
    return [float(x) for x in best_vec]
