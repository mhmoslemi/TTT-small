import time, math, numpy as np

def _project_simplex(v):
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())

import time
import math
import numpy as np

# ---------- Euclidean projection onto the probability simplex ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ---------- Score (2·n·max convolution) for a simplex‑normalised vector ----------
def _score(a: np.ndarray, fft_len: int = None) -> float:
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())
