# ------------------------------------------------------------
#  Simplex projection (sum == 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    #  Pick the best among the cheap starts
    # --------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for a in starts:
        val = fast_score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    # --------------------------------------------------------
    #  Gradient‑descent parameters (smoothed max)
    # --------------------------------------------------------
    eps0 = 0.5          # initial smoothing
    eps_decay = 0.9995
    eps_min = 1e-9
    step0 = 0.5         # base step size

    cur = best_seq.copy()
    cur_val = best_val
    it = 0
    no_improve = 0

    # --------------------------------------------------------
    #  Main optimisation loop (time‑limited)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current point -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- smooth‑max weights (log‑sum‑exp) -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_conv) / eps)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)          # length ≥3n‑2
        idx = (2 * n - 2) - np.arange(n)                     # slice that yields length n
        g = 2.0 * grad_full[idx]

        # ----- descent step with projection onto simplex -----
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        cand = cur - alpha * g
        cand = _project_simplex(cand)

        cand_val = fast_score(cand)

        # ----- accept / reject -----
        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best_seq, best_val = cand.copy(), cand_val
                no_improve = 0
        else:
            prob = math.exp(-(cand_val - cur_val) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_val = cand, cand_val
            no_improve += 1

        # ----- occasional random mass transfer (helps escape plateaus) -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_val = fast_score(tmp)
                if tmp_val < cur_val:
                    cur, cur_val = tmp, tmp_val
                if tmp_val < best_val:
                    best_seq, best_val = tmp.copy(), tmp_val
                no_improve = 0

        # ----- early exit if target already met -----
        if best_val < 1.50274349:
            break

    # --------------------------------------------------------
    #  If we ran out of time without any candidate, fall back to uniform
    # --------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n)

    # Return a plain Python list of floats
    return [float(x) for x in best_seq]
