import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
def _score_from_conv(conv: np.ndarray, n: int) -> float:
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative coefficients with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = (a + a[::-1]) * 0.5          # initial symmetrisation
    else:
        n0 = 1024
        a = np.exp(-np.arange(n0, dtype=np.float64) / 8.0)

    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()                  # project onto simplex (sum=1)

    n = a.size
    # FFT length ≥ 3n‑2, power of two
    L = 1 << ((3 * n - 2).bit_length())

    # ---- bookkeeping ----------------------------------------------------
    best_vec = a.copy()
    cur = a.copy()
    # initial score
    cur_fft = np.fft.rfft(cur, n=L)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
    best_sc = cur_sc = _score_from_conv(cur_conv, n)

    # ---- hyper‑parameters ------------------------------------------------
    step0 = 2.0
    eps = 0.5
    eps_decay = 0.998
    eps_min = 1e-7
    beta = 0.9                     # momentum
    v = np.zeros_like(cur)         # momentum buffer
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution of current iterate
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights over convolution entries
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of weighted max (via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        v = beta * v + (1.0 - beta) * g
        grad_norm = np.linalg.norm(v) + 1e-12
        alpha = step0 / (math.sqrt(it) * grad_norm)

        # candidate step
        cand = cur - alpha * v
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = _score_from_conv(cand_conv, n)

        # accept if better, otherwise Metropolis‑type acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # periodic symmetrisation (helps autocorrelation)
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = _score_from_conv(sym_conv, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # random mass‑transfer when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = _score_from_conv(tmp_conv, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # final local refinement with tiny random jitters
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best_vec[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_fft = np.fft.rfft(tmp, n=L)
        tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
        tmp_sc = _score_from_conv(tmp_conv, n)
        if tmp_sc < best_sc:
            best_vec, best_sc = tmp, tmp_sc
            continue
        # occasional symmetrisation check
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _project_simplex(sym)
        sym_fft = np.fft.rfft(sym, n=L)
        sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
        sym_sc = _score_from_conv(sym_conv, n)
        if sym_sc < best_sc:
            best_vec, best_sc = sym, sym_sc

    # return as plain Python list
    return [float(x) for x in best_vec]
