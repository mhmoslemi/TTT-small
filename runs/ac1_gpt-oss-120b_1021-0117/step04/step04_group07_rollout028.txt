    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initial
    ...

    cur = a0.copy()
    cur_score = _fast_score(cur, L, n)
    best = cur.copy()
    best_score = cur_score

    # parameters
    eps = 0.02
    eps_min = 1e-9
    eps_decay = 0.9997
    step0 = 0.8
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # compute convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]

        maxc = conv.max()

        # gradient
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            g = 2.0 * grad_full[idx]
            eps = max(eps_min, eps * eps_decay)
        else:
            # subgradient using all max indices
            max_idxs = np.where(conv == maxc)[0]
            g = np.zeros_like(cur)
            for k in max_idxs:
                start_k = max(0, k - (n - 1))
                end_k = min(k + 1, n)
                inds = np.arange(start_k, end_k)
                g[inds] += 2.0 * cur[k - inds]
            if max_idxs.size > 1:
                g /= max_idxs.size

        # step size
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / (math.sqrt(it) * grad_norm)

        cand = cur - alpha * g
        cand = _project_simplex(cand)

    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:

        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
