import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Score (same formula as evaluate_sequence when sum == 1)
# ----------------------------------------------------------------------
def _score_fft(vec: np.ndarray, L: int) -> float:
    """Fast score = 2*n*max(conv(vec,vec)) for sum(vec)=1."""
    n = vec.shape[0]
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative step‑function (list of floats) with a low
    AC1 upper bound.  The search adapts a smoothed‑max gradient descent with
    periodic symmetrisation, momentum and random mass‑transfer perturbations.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------------
    # Initial vector (use parent if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: decaying exponential, length ≈ 1024
        n_fallback = 1024
        x = np.arange(n_fallback, dtype=np.float64)
        cur = np.exp(-x / (n_fallback / 10.0))

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)                # enforce sum == 1, non‑negative
    n = cur.shape[0]

    # --------------------------------------------------------------
    # FFT infrastructure
    # --------------------------------------------------------------
    # Length for convolutions involving a (size n) and a weight vector (size 2n‑1)
    L_full = 1 << ((3 * n - 2).bit_length())  # >= 3n‑2, power‑of‑two for speed

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    cur_fft = np.fft.rfft(cur, n=L_full)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_sc = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_sc = cur_sc

    # hyper‑parameters
    step0 = 1.0           # initial step magnitude
    eps_rel = 0.018       # relative smoothing for the soft‑max
    eps_min = 1e-9
    max_backtrack = 7
    beta = 0.85           # momentum coefficient
    vel = np.zeros_like(cur)

    it = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- smoothed‑max weights -----
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (soft‑max) -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L_full)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_full)
        idx = (2 * n - 2) - np.arange(n)   # map to proper coordinates
        grad = 2.0 * grad_full[idx]

        # momentum update
        vel = beta * vel + (1.0 - beta) * grad

        # ----- line‑search step -----
        alpha = step0 / math.sqrt(it)
        accepted = False
        cand = None
        cand_fft = None
        cand_sc = None
        for _ in range(max_backtrack):
            cand = cur - alpha * vel
            cand = _project_simplex(cand)
            # enforce (optional) symmetry – helps reduce max‑convolution
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

            cand_fft = np.fft.rfft(cand, n=L_full)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[: 2 * n - 1]
            cand_sc = 2.0 * n * cand_conv.max()
            if cand_sc < cur_sc:
                accepted = True
                break
            alpha *= 0.5

        if accepted:
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_conv.max()
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # occasional uphill acceptance (simulated‑annealing flavour)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_fft = cand_fft
                cur_conv = cand_conv
                cur_max = cand_conv.max()
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # ----- periodic symmetrisation (every few hundred iters) -----
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L_full)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L_full)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc <= cur_sc:
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_conv.max()
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.33
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)

                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        # ----- lightweight random local search near the best -----
        if it % 500 == 0:
            # try a few cheap perturbations
            for _ in range(12):
                i, j = rng.integers(0, n, size=2)
                if i == j or best_vec[i] == 0.0:
                    continue
                delta = best_vec[i] * rng.random() * 0.2
                cand = best_vec.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
                cand = (cand + cand[::-1]) * 0.5
                cand = _project_simplex(cand)

                cand_fft = np.fft.rfft(cand, n=L_full)
                cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[: 2 * n - 1]
                cand_sc = 2.0 * n * cand_conv.max()
                if cand_sc < best_sc:
                    best_sc = cand_sc
                    best_vec = cand.copy()
                    # also make it current to guide further descent
                    cur = cand
                    cur_fft = cand_fft
                    cur_conv = cand_conv
                    cur_max = cand_conv.max()
                    cur_sc = cand_sc
                    no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
