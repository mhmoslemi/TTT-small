def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        a = rng.random(n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    n = a.size

    L = 1 << ((2*n - 1).bit_length())

    # projection onto simplex function
    ...
    
    # evaluate score function (fast)
    def _score(x):
        a_fft = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    
    cur = a.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # Adam parameters
    mt = np.zeros_like(cur)
    vt = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5  # learning rate constant

    it = 0
    no_improve = 0
    temperature = 0.1

    while time.monotonic() < deadline:
        it += 1
        # maybe linearly decay temperature
        temperature = max(1e-5, temperature * 0.9995)
        # compute FFT and convolution for current
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        maxc = conv.max()
        # gradient
        if temperature > 1e-6:
            # softmax
            w = np.exp((conv - maxc) / temperature)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # compute conv of a with reversed w
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2*n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
        else:
            # subgradient for max index
            k = int(conv.argmax())
            g = np.zeros_like(cur)
            start_idx = max(0, k - (n - 1))
            end_idx = min(k + 1, n)
            idx_range = np.arange(start_idx, end_idx)
            g[idx_range] = 2.0 * cur[k - idx_range]
        # Adam update
        mt = beta1 * mt + (1 - beta1) * g
        vt = beta2 * vt + (1 - beta2) * (g * g)
        m_hat = mt / (1 - beta1**it)
        v_hat = vt / (1 - beta2**it)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = cur - step_vec
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # accept with prob exp(-(cand_sc-cur_sc)/temp2)
            # use small acceptance to escape
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (0.01 + cur_sc)):
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # update best
        if cur_sc < best_sc:
            best, best_sc = cur.copy(), cur_sc
            no_improve = 0

        # periodic symmetrisation
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
            if sym_sc < best_sc:
                best, best_sc = sym.copy(), sym_sc

        # occasional random mass transfer when stuck
        if no_improve > 3000:
            # generate random perturbation

            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.3

                tmp = cur.copy()
                tmp[i] -= delta

                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:

                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # early exit if best below target

import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 score (scale‑invariant, uses FFT)
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
