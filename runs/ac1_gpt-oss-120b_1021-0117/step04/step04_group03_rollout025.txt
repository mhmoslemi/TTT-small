import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score assuming sum = 1 (exact evaluator without the extra checks)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution of a with itself)."""
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative coefficient list with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Build initial pool of candidates
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
    else:
        base = None

    candidates = []

    if base is not None:
        base = _project_simplex(base)               # sum = 1
        candidates.append(base)

    # uniform vector
    if base is not None:
        n = base.size
    else:
        n = 1024
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    candidates.append(uniform)

    # arcsine‑type distribution (often a good starter)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()
    candidates.append(arcsine)

    # a few random Dirichlet samples
    for _ in range(5):
        rnd = rng.gamma(1.0, 1.0, size=n)
        rnd = rnd / rnd.sum()
        candidates.append(rnd)

    # pick the best starter
    cur = None
    cur_sc = float("inf")
    for cand in candidates:
        sc = _score(cand)
        if sc < cur_sc:
            cur_sc = sc
            cur = cand.copy()
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # length for linear convolution via FFT
    eps = 1.0
    eps_decay = 0.9995
    eps_min = 1e-7

    momentum = np.zeros(n, dtype=np.float64)
    beta = 0.9                     # momentum decay
    base_lr = 0.8                  # base learning rate
    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # convolution of cur with itself (full)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----------------------------------------------------
        # Smoothed max weights
        # ----------------------------------------------------
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----------------------------------------------------
        # Gradient of the smoothed objective
        # ----------------------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # step size (scale‑invariant)
        grad_norm = np.linalg.norm(momentum) + 1e-12
        alpha = base_lr / math.sqrt(it) / grad_norm

        cand = cur - alpha * momentum
        cand = _project_simplex(cand)

        if (it % 500) == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # keep global best
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            no_improve = 0

        # accept if improvement, otherwise count stagnation
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:

                no_improve = 0

    # --------------------------------------------------------
    # Return the best found coefficients as a plain list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
