import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x>=0, sum=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score (2*n*max(conv)) – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int = None) -> float:
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Random symmetric initialization (Dirichlet + mirror)
# ------------------------------------------------------------
def _random_symmetric_init(rng, n):
    v = rng.dirichlet(np.ones(n))
    v = (v + v[::-1]) * 0.5
    v = v / v.sum()
    return v

def _optimize(init_vec, deadline, rng):
    cur = init_vec.copy()
    n = cur.shape[0]
    # FFT length sufficient for all circular convolutions we need
    L = 1 << ((3 * n - 2).bit_length())

    # hyper‑parameters
    step0 = 0.6
    eps0 = 0.2
    eps_decay = 0.9995
    eps_min = 1e-8
    beta = 0.9          # momentum factor
    target = 1.50274349

    m = np.zeros_like(cur)            # momentum accumulator
    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ---------- convolution ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ---------- smoothed weights ----------
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of smoothed max ----------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)

        g = 2.0 * grad_full[idx]

        m = beta * m + (1.0 - beta) * g
