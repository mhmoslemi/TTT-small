def gradient_softmax(a, eps):
    L = FFT length
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n - 1]
    maxc = conv.max()
    # compute weights
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w = w / w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad
```

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a0 = np.full(n0, 1.0 / n0, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s == 0:
        a0[:] = 1.0 / a0.size
    else:
        a0 /= s

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())

    # define projection etc.

    best = a0.copy()
    best_score = _fast_score(best, L, n)

    # parameters
    eps0 = 0.05
    eps = eps0
    eps_min = 1e-7
    eps_decay = 0.9993

    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)

    cur = best.copy()
    cur_score = best_score
    it = 0
    no_improve = 0

    # main loop
    while time.monotonic() < deadline:
        it += 1
        # compute gradient
        # gradient based on smoothed max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        maxc = conv.max()
        # compute softmax weights
        w = np.exp((conv - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_score = _fast_score(cand, L, n)

        # accept if better
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                no_improve = 0
        else:
            # accept with small probability (simulated annealing)
            if rng.random() < math.exp(-(cand_score - cur_score) / (lr + 1e-12)):
                cur, cur_score = cand, cand_score
            no_improve += 1

        # update eps
        eps = max(eps_min, eps * eps_decay)

        # symmetrize occasionally
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _fast_score(sym, L, n)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score
                    no_improve = 0

        # random perturb when stuck
        if no_improve > 3000:
            # random transfer between two indices
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _fast_score(tmp, L, n)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
                no_improve = 0

        # early exit if reached target
        if best_score < 1.50274349:
            break

    return [float(x) for x in best]

import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector that already satisfies sum=1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(a*a) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a short safety margin
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # --------------------------------------------------------
    # initial vector (use parent if present)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

    else:

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0.fill(1.0 / a0.size)
    else:
        a0 /= s                      # normalize to simplex

    n = a0.size
    # FFT length sufficient for 3n‑2 linear convolution
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    cur = a0.copy()
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()
    best_sc = cur_sc

    # algorithmic parameters
    eps = 0.06               # start temperature for soft‑max
    eps_min = 1e-7
    eps_decay = 0.9995

    lr = 0.04                # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and smoothed max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # soft‑max weights
        w = np.exp((conv - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad

        else:

                cur, cur_sc = cand, cand_sc

        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best]
