import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score_full(a: np.ndarray, L: int = None) -> float:
    """Exact AC1 score for vector a (non‑negative, any sum)."""
    n = a.shape[0]
    s = a.sum()
    if s == 0:
        return float('inf')
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    else:

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)                # sum == 1

    cur_sc = best_sc

    # ----- gradient descent phase ------------------------------------

    step0 = 0.35
    it = 0
