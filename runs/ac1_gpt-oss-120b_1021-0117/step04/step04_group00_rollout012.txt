import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Helper: evaluate the AC1 bound (sum of coefficients is assumed 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) for a sum‑to‑one vector a."""
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())        # power of two ≥ 3n‑2
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    cur = a0.copy()
    n = cur.size

    # ------------------------------------------------------------

    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # Adam hyper‑parameters

    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.3

    # smoothing for the soft‑max (epsilon) – fast decay
    eps0 = 0.5
    eps_min = 1e-6

    # bookkeeping
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    best_vec = cur.copy()

    no_improve = 0

    # ------------------------------------------------------------

    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        t += 1

        # ----- convolution of cur with itself and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights centred on the current maximum -----
        eps = max(eps_min, eps0 / math.sqrt(t))
