import time
import math
import numpy as np

# ----- Euclidean projection onto the probability simplex (sum = 1, non‑negative) -----
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative coefficient list with a low AC1 bound.
    The routine works directly on the simplex (sum = 1) – the score is
    invariant under scaling, so this normalisation is safe.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep margin

    # ---- initialise ---------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple geometric fall‑off if no parent is available
        n_fallback = 2500
        r = 0.995
        a = np.power(r, np.arange(n_fallback), dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # simplex normalisation

    n = a.size
    # FFT length that avoids circular wrap‑around for convolutions up to length 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # ---- bookkeeping --------------------------------------------------------
    best_vec = a.copy()
    a_fft = np.fft.rfft(best_vec, n=L)

    best_sc = 2.0 * n * float(conv.max())

    # ---- hyper‑parameters ---------------------------------------------------
    step0 = 0.5                # base step magnitude

    max_backtrack = 12
    no_improve = 0
    it = 0
    target_score = 1.50274349

    # ---- main optimisation loop ---------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & active set --------------------------------
        a_fft = np.fft.rfft(cur, n=L)

        max_val = conv_full.max()

        tol = max(1e-12, max_val * 1e-12)
