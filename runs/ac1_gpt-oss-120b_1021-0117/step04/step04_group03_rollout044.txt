import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum=1, x>=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector normalised to sum=1 (exact evaluator)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(conv(a,a))  (assumes sum(a)=1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # initialise candidates
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
        base = _project_simplex(base)
    else:
        base = None

    n = 2535 if base is not None else 1024
    if base is None:
        base = np.full(n, 1.0 / n, dtype=np.float64)

    # some extra analytic starts (always safe)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()
    sinus = np.sin(np.pi * (i + 0.5) / n)

    cand_list = [base, uniform, arcsine, sinus]
    best_vec = None
    best_sc = float("inf")
