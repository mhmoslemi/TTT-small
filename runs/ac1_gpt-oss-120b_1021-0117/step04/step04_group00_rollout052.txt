# ------------------------------------------------------------
# Helper utilities

def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _score(a: np.ndarray) -> float:
    """
    Evaluates the AC1 bound for a non‑negative sequence a with sum(a)=1.
    Returns 2 * n * max(conv(a,a)).
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑two FFT length
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
