import time, math, numpy as np

def _project_simplex(v):
    # as earlier, maybe use improved version.
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)

import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final conversion
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initial vectors -------------------------------------------------
    inits = []

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
        inits.append(a)

    # uniform fallback (different length if parent missing)
    if not inits:
        n0 = 2000
        a = np.full(n0, 1.0 / n0, dtype=np.float64)
        inits.append(a)

    # a simple triangular shape (often good)
    n_tri = inits[0].size
    tri = np.arange(1, n_tri // 2 + 1)
    if n_tri % 2 == 0:
        tri = np.concatenate([tri, tri[::-1]])
    else:
        tri = np.concatenate([tri, [n_tri // 2 + 1], tri[::-1]])
    tri = tri.astype(np.float64)
    tri = _project_simplex(tri)
    inits.append(tri)

    best_overall = None
    best_score = float('inf')

    # ---- run optimizer on each start (respect remaining time) -------------
    for vec in inits:
        if time.monotonic() >= deadline:
            break
        # allocate a slice of remaining time per start (at least 5 s)
        remaining = deadline - time.monotonic()
        sub_deadline = time.monotonic() + max(5.0, remaining / (len(inits) - inits.index(vec)))
        cand, sc = _optimize_once(vec, sub_deadline, rng)
        if sc < best_score:
            best_overall, best_score = cand, sc

    if best_overall is None or best_overall.size == 0:
        best_overall = np.full(1, 1.0, dtype=np.float64)

    return [float(x) for x in best_overall]
