import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    # If v already sums to 1 and nonnegative, return copy (quick path)
    # but we just compute robustly.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ------------------------------------------------------------------
    # Initialise vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 1024
        cur = np.exp(-np.arange(n_fallback, dtype=np.float64) / (n_fallback / 10.0))
    cur = np.maximum(cur, 0.0)
    # Normalise to sum == 1 (simplex)
    cur = _project_simplex(cur)

    n = cur.shape[0]

    # FFT length sufficient for full (3n‑2) linear convolution
    L = 1 << ((3 * n - 2).bit_length())

    # Pre‑compute slice indices for gradient extraction
    idx = (2 * n - 2) - np.arange(n)

    # Helper: compute full convolution (first 2n‑1 entries) and score
    def _conv_and_score(v):
        fv = np.fft.rfft(v, n=L)
        conv_full = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        return conv_full, 2.0 * n * max_val

    cur_conv, cur_score = _conv_and_score(cur)
    best_vec = cur.copy()
    best_score = cur_score

    # Parameters (tuned empirically)
    step0 = 0.8
    eps_rel = 0.02
    eps_min = 1e-8
    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop (gradient‑descent + occasional random moves)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --- smoothed‑max weights ------------------------------------
        cur_max = cur_conv.max()
        # Reduce smoothing slowly to sharpen gradient
        eps = max(eps_min, cur_max * eps_rel * (0.5 ** (it // 2000)))
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the smoothed objective ---------------------
        w_rev = w[::-1]
        cur_fft = np.fft.rfft(cur, n=L)
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L)
        g = 2.0 * grad_full[idx]          # gradient w.r.t. each a_i

        # --- line search --------------------------------------------
        alpha = step0 / math.sqrt(it)
        accepted = False
        for _ in range(6):

            cand = _project_simplex(cand)

            alpha *= 0.5

                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)

                    if tmp_score < cur_score:

            cur_conv = cand_conv
            cur_score = cand_score

    # --------------------------------------------------------------
    # Return best candidate as plain Python list
    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
