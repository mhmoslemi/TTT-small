# Simple projection onto the simplex {x >= 0, sum(x)=1}
# -------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v[:] = 1.0 / v.size
    return v
