import time, math, numpy as np

def _project_simplex(v):
    # Implementation as before but ensure robust
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a, conv=None):
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2*n-1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af*af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int=42, budget_s: float=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    if "height_sequence_1" in globals():

    # ensure nonnegative and clip
    a = np.maximum(a, 0.0)
    # normalize to sum 1 (project onto simplex)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full_like(a, 1.0/a.size)
    else:
        a = a / a_sum
    # now ensure sum=1 exactly via projection (handles rounding)
    a = _project_simplex(a)

    n = a.shape[0]
    # FFT length for convolutions
    L = 1 << ((3*n - 2).bit_length())

    # optimisation parameters

    eps_decay = 0.9995
    eps_min = 1e-7
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    m = np.zeros_like(a)

    cur = a.copy()
    cur_sc = best_sc
    no_improve = 0

    # Main loop

        # FFT of current vector

import time, math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray | None = None) -> float:
    """Return 2*n*max(convolution)."""
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
