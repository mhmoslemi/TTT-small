import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum x = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector that already sums to 1 (exact evaluator)
# ------------------------------------------------------------
def _score(a: np.ndarray, n: int) -> float:
    """2 * n * max(convolution) – the same quantity the evaluator returns for sum(a)=1."""
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Soft‑max (smooth max) of the convolution and its gradient
# ------------------------------------------------------------
def _softmax_conv_and_grad(a: np.ndarray, L: int, eps: float):
    """Return (conv, max_conv, grad) where grad = ∂ softmax_max / ∂a."""
    # FFT of a
    a_fft = np.fft.rfft(a, n=L)

    # full convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]

    # soft‑max weights
    max_c = conv.max()
    # stability: subtract max before exponentiation
    w = np.exp((conv - max_c) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (a * w_rev)  (via FFT)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    idx = (2 * a.size - 2) - np.arange(a.size)
    grad = 2.0 * grad_full[idx]
    return conv, max_c, grad


# ------------------------------------------------------------
# Random point on the simplex
# ------------------------------------------------------------
def _random_simplex(rng, n):
    x = rng.exponential(scale=1.0, size=n)
    return x / x.sum()

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total_budget = float(budget_s)
    # keep a small safety margin
    deadline = start + max(0.0, total_budget - 0.5)

    # -------------------- base vector --------------------
    if "height_sequence_1" in globals():

        base = np.maximum(base, 0.0)
        base = _project_simplex(base)
        n = base.size
    else:
        n = 1024
        base = np.full(n, 1.0 / n, dtype=np.float64)

    # -------------------- auxiliary starters --------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()

    # a few random points on the simplex
    rand1 = _random_simplex(rng, n)
    rand2 = _random_simplex(rng, n)

    candidates = [base, uniform, arcsine, rand1, rand2]
    best_vec = None
    best_sc = float("inf")
    for cand in candidates:
        sc = _score(cand, n)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------
    # Adam‑style projected gradient descent on the smooth max
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())
    eps = 0.5            # initial smoothing
    eps_decay = 0.9995
    eps_min = 1e-6

    # Adam parameters

    lr = 0.05

    v = np.zeros_like(cur)

        cur_sc = 2.0 * n * max_c

        beta1_pow *= beta1
        beta2_pow *= beta2

    if best_vec is None:
        best_vec = cur
    return [float(x) for x in best_vec]
