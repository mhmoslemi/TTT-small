import time

import scipy.sparse as sp

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector that already sums to 1
# ------------------------------------------------------------
def _score_simplex(a: np.ndarray) -> float:
    """Return 2 * n * max(a * a) for a simplex‑normalised vector."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Sparse matrix for (f * g)[s] <= M constraints (used by the LP step)
# ------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return A with A[s, j] = f[s-j] (zero when out of range)."""
    n = f.size
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)
    pos = 0
    for j in range(n):
        rows[pos: pos + n] = np.arange(j, j + n, dtype=np.int32)
        cols[pos: pos + n] = j
        data[pos: pos + n] = f
        pos += n
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
