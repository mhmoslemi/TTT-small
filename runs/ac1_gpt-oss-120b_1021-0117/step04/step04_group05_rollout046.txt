# Helper: simple non‑negative normalization (sum -> 1)

def _norm_one(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and rescale to have sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # uniform fallback
        return np.full_like(v, 1.0 / v.size)
    return v / s
