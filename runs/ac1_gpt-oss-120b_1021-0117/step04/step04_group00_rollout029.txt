def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 1.0)  # keep 1 sec headroom

    # Initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random initialization length maybe 1024
        a_len = 2048
        a = rng.random(a_len, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)  # sum=1
    n = a.size
    L = 1 << ((3 * n - 2).bit_length())

    # Score
    def _score_vec(v):
        # uses same n, norm = 1 -> bound = 2*n*max(conv)
        v_fft = np.fft.rfft(v, n=L)
        conv_full = np.fft.irfft(v_fft * v_fft, n=L)
        conv = conv_full[:2*n-1]
        return float(2.0 * n * conv.max())

    cur = a.copy()
    cur_sc = _score_vec(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.5
    eps_smooth = 1.0
    eps_decay = 0.9995
    eps_min = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        t += 1
        # compute convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # softmax weights
        eps = max(eps_min, eps_smooth * (eps_decay ** t))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)  # same as before
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        direction = m_hat / (np.sqrt(v_hat) + 1e-8)

        # line search scaling
        alpha = lr
        cand = None
        cand_sc = None
        for _ in range(6):
            cand_vec = cur - alpha * direction
            cand_vec = _project_simplex(cand_vec)
            cand_sc = _score_vec(cand_vec)
            if cand_sc <= cur_sc:
                cand = cand_vec
                break
            alpha *= 0.5
        if cand is None:
            cand = cand_vec
            cand_sc = cand_sc

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cur_sc < best_sc:
                best_vec = cur.copy()
                best_sc = cur_sc
        else:

            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score_vec(seq: np.ndarray) -> float:
    """Return 2 * n * max(convolution(seq,seq)).  No normalization."""
    n = seq.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(fft * fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------

    start = time.monotonic()
    # keep a small head‑room for the harness

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1, non‑negative

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
    cur_sc = best_sc

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # algorithmic parameters
    step0 = 0.6               # base step magnitude (decays like 1/sqrt(t))
    eps0 = 1.0                # initial smoothing for soft‑max
