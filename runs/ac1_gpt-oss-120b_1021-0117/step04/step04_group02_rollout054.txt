def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., budget_s - 0.5)

    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.clip(a, 0.0, 1000.0)
    else:
        n = 2000
        a = np.exp(-np.arange(n)/100.0)

    a = np.maximum(a, 0.0)
    S = a.sum()
    if S <= 1e-12:
        a = np.full_like(a, 1.0 / a.size)
        S = a.sum()
    a = a / S

    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    best = a.copy()
    best_score = _score(best)
    cur = a.copy()
    cur_score = best_score

    m = np.zeros_like(a)
    beta = 0.9
    eps = None
    base_alpha = 0.5
    improve_target = 1.50274349
    no_improve = 0
    it = 0
    sym_interval = 300
    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n-1]
        max_val = conv.max()
        if eps is None:
            eps = max(max_val*0.2, 1e-12)
        else:
            eps = max(eps*0.995, 1e-12)

        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum update
        m = beta*m + (1-beta)*g
        # step size
        g_norm = np.linalg.norm(m) + 1e-12
        alpha = base_alpha / math.sqrt(it) / g_norm

        # candidate
        cand = cur - alpha * m
        cand = _project_simplex(cand)

        cand_score = _score(cand)

        if cand_score < cur_score or rng.random() < 0.01:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                if sym_score < best_score:
                    best_score = sym_score
                    best = sym.copy()
                no_improve = 0

        if no_improve > 2500:
            # random reallocation mass
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                cand2 = cur.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                cand2_score = _score(cand2)
                if cand2_score < cur_score:
                    cur = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                no_improve = 0

        if best_score < improve_target:

import time, math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a: np.ndarray) -> float:
    """Fast AC1 bound assuming sum(a) == 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    else:
        # simple exponential fallback

        a_ref = np.exp(-np.arange(n_fallback, dtype=float) / 100.0)

    n = a_ref.size

    idx = (2 * n - 2) - np.arange(n)

    def run_one(init_vec: np.ndarray):
