    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)

    if 'height_sequence_1' in globals():
        cur = np.array(height_sequence_1, dtype=float)

    cur = _project_simplex(cur)   # sum=1

    n = cur.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    half_len = 2 * n - 1

    # precompute indices for gradient extraction
    idx = (2 * n - 2) - np.arange(n)

    # compute conv and score
    cur_fft = np.fft.rfft(cur, n=L)
    cur_conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[:half_len]
    cur_max = cur_conv_full.max()
    cur_sc = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_sc = cur_sc

    # Parameters
    step0 = 0.7
    eps_rel_initial = 0.02
    eps_rel_final = 0.003
    eps_rel = eps_rel_initial
    eps_min = 1e-8
    warmup_iters = 2000
    sym_interval = 500
    no_improve = 0

    it = 0
    while time.monotonic() < deadline:

        # adapt epsilon
        if it > warmup_iters:

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
