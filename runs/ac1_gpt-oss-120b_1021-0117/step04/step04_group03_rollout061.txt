def _project_simplex_sum(v, target_sum=1.0):
    # Projects onto {x >= 0, sum x = target_sum}
    # as per Duchi et al.
    n = v.size
    # shift for feasibility
    # Use same algorithm: sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - target_sum
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_general(a):
    n = a.shape[0]
    S = a.sum()
    if S <= 0: return float('inf')
    L = 1 << ((2 * n - 1).bit_length())

    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
    max_val = conv.max()

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection with arbitrary target sum
# ------------------------------------------------------------
def _project_simplex_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - target_sum
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (exact formulation)
# ------------------------------------------------------------
def _score_general(a: np.ndarray) -> float:
    """Return 2 * n * max(conv) / (sum(a)**2)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))
