def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 1500
        a = np.linspace(1, 0.1, n_fallback)  # some shape
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    L = 1 << ((2*n - 1).bit_length())   # next power of two >= 2n-1

    def _project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        n = v.shape[0]
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    def _score_vec(x):
        # compute 2*n*max conv after normalising sum?
        # Already sum(x) should be close to 1, but we compute directly
        n_local = x.shape[0]
        L_local = 1 << ((2*n_local - 1).bit_length())
        af = np.fft.rfft(x, n=L_local)
        conv = np.fft.irfft(af * af, n=L_local)[: 2 * n_local - 1]
        return 2.0 * n_local * float(conv.max())

    # initial score
    cur = a.copy()
    cur_sc = _score_vec(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # momentum
    m = np.zeros_like(cur)

    # hyperparameters
    step0 = 1.0
    backtrack_factor = 0.5
    max_backtrack = 10
    beta = 0.9
    eps0 = 0.5
    eps_decay = 0.9995
    eps_min = 1e-12

    no_improve = 0
    it = 0

        idx = (2 * n - 2) - np.arange(n)

        # update momentum
        m = beta * m + (1.0 - beta) * grad

        # step size (normalized)
        norm_m = np.linalg.norm(m) + 1e-15
        step = step0 / np.sqrt(it + 1) / norm_m

        improved = False
        for _ in range(max_backtrack):

            cand = _project_simplex(cand)

            if cand_sc < cur_sc:

                    no_improve = 0
                improved = True

import time
import math
import numpy as np

# ---------- helpers ---------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = v.astype(np.float64, copy=False)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray, L: int) -> float:
    """Return 2·n·max(conv(a,a)) using an FFT of length L."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())
