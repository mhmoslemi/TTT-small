# Euclidean projection onto the probability simplex (sum = 1)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                conv_tmp = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_score = 2.0 * n * float(conv_tmp.max())
                if tmp_score < cur_score:
                    a, conv, cur_max, cur_score = tmp, conv_tmp, float(conv_tmp.max()), tmp_score
                if tmp_score < best_score:
                    best_a, best_score = tmp.copy(), tmp_score
                no_improve = 0

        # ---- early exit if target reached ----
        if best_score < 1.50274349:
            break

    # ------------------- final fine‑tuning -----------------
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.005, size=n)
        cand = best_a + delta
        cand = _project_simplex(cand)
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * float(conv_cand.max())
        if cand_score < best_score:
            best_a, best_score = cand, cand_score

    # return plain Python list
    return [float(x) for x in best_a]
