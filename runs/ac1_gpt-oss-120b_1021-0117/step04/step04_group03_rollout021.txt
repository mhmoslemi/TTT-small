# Simplex projection (L2‐optimal onto {x>=0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            no_improve = 0
            if cand_sc < best_score:
                best_score, best_vec = cand_sc, cand.copy()
                if best_score < target:
                    break
        else:
            no_improve += 1

        # -----------------------------------------------------------------
        # Escape from stagnation via a tiny random mass transfer
        # -----------------------------------------------------------------
        if no_improve > 2000:
            i_idx = rng.integers(0, n)
            if cur[i_idx] > 0.0:
                delta = cur[i_idx] * rng.random() * 0.25
                j_idx = rng.integers(0, n)
                if j_idx == i_idx:
                    j_idx = (j_idx + 1) % n
                tmp = cur.copy()
                tmp[i_idx] -= delta
                tmp[j_idx] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                # cheap score
                aft = np.fft.rfft(tmp, n=Lconv)
                convt = np.fft.irfft(aft * aft, n=Lconv)[: 2 * n - 1]
                sc_t = 2.0 * n * float(convt.max()) / (tmp.sum() ** 2)
                if sc_t < cur_sc:
                    cur, cur_sc = tmp, sc_t
                    no_improve = 0
                    if sc_t < best_score:
                        best_score, best_vec = sc_t, tmp.copy()
                        if best_score < target:
                            break
                else:
                    no_improve = 0

    # --------------------------------------------------------
    # Final lightweight local search (if time permits)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        # pick two indices and move a tiny amount
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if cur[i] <= 1e-12:
            continue
        delta = cur[i] * rng.random() * 0.05
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp = (tmp + tmp[::-1]) * 0.5
        tmp = _project_simplex(tmp)

        af = np.fft.rfft(tmp, n=Lconv)
        conv = np.fft.irfft(af * af, n=Lconv)[: 2 * n - 1]
        sc = 2.0 * n * float(conv.max()) / (tmp.sum() ** 2)
        if sc < cur_sc:
            cur, cur_sc = tmp, sc
            if sc < best_score:
                best_score, best_vec = sc, tmp.copy()
                if best_score < target:
                    break

    # --------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # --------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    # safety final check: ensure non‑negative, finite, non‑empty
    best_vec = np.maximum(best_vec, 0.0)
    return [float(x) for x in best_vec]
