def _lp_max_sum(a, M):
    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    a_param = cp.Parameter(n, value=a, constant=True) # param
    conv = cp.conv(a_param, g)  # length 2n-1
    constraints = [conv <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    # after solve, return g.value if status = optimal
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return g.value
    else:
        return None

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _lp_optimize(a, M):
    try:
        import cvxpy as cp
    except Exception:
        return None
    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    a_param = cp.Parameter(n, value=a, constant=True)
    conv_expr = cp.conv(a_param, g)  # length 2n-1
    constraints = [conv_expr <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000, verbose=False)
    except Exception:
        return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    g_val = g.value
    if g_val is None:
        return None
    return np.asarray(g_val, dtype=np.float64)

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    no_improve = 0
    it = 0
    target = 1.50274349

    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        it += 1

        # convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # mask of maximal entries (hard max)
        eps_mask = 1e-12
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 1:
            w /= cnt

        # gradient of max (subgradient)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

                g_opt = _lp_optimize(cur, float(max_val))
                if g_opt is not None and g_opt.size == n:

                            best_vec, best_sc = lp_vec.copy(), lp_sc
                            cur, cur_sc = lp_vec, lp_sc
                            no_improve = 0

        # early exit
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
