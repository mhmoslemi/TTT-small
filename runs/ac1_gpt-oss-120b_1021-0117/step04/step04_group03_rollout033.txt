import time
import math
import numpy as np
import scipy.sparse as sp

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a simplex‑normalised vector (sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Sparse Toeplitz matrix representing convolution with f
# ------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    n = f.size
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)
    pos = 0
    for j in range(n):
        rows[pos : pos + n] = np.arange(j, j + n, dtype=np.int32)
        cols[pos : pos + n] = j
        data[pos : pos + n] = f
        pos += n
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Base vector (use the provided parent if present)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
    else:
        # a modest random start if no parent is available
        base = rng.random(1024)
    base = _project_simplex(base)
    n = base.size

    # --------------------------------------------------------
    # Some cheap alternatives (uniform, arcsine)
    # --------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()

    candidates = [base, uniform, arcsine]
    cur = candidates[0].copy()
    cur_score = _score(cur)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < cur_score:
            cur = cand.copy()
            cur_score = sc

    best_vec = cur.copy()
    best_score = cur_score

    # --------------------------------------------------------
    # First stage: a few LP improvements (matrix‑free but sparse)
    # --------------------------------------------------------
    lp_allowed = (deadline - time.monotonic()) > 5.0
    if lp_allowed:
        try:
            import cvxpy as cp
        except Exception:  # cvxpy not available – skip LP stage
            cp = None

        if cp is not None:
            # allocate about 30 % of the total budget for LP passes
            lp_deadline = start + 0.30 * float(budget_s)
            lp_iter = 0
            mix_const = 0.3                       # fixed mixing weight
            while time.monotonic() < lp_deadline:
                # current convolution and its max value M
                L = 1 << ((2 * n - 1).bit_length())
                a_fft = np.fft.rfft(cur, n=L)
                conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
                M = conv.max()

                # build the sparse convolution matrix for constraints
                A = _build_conv_matrix(cur)

                # linear programme: maximise sum(g) s.t. A@g <= M, g>=0
                g = cp.Variable(n, nonneg=True)
                constraints = [A @ g <= M, g <= 1000.0]
                prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

                try:
                    prob.solve(solver=cp.ECOS, warm_start=True,
                               max_iters=2000, verbose=False)
                except Exception:
                    try:
                        prob.solve(solver=cp.GLPK, warm_start=True,
                                   verbose=False,
                                   glpk={"msg_lev": "GLP_MSG_OFF"})
                    except Exception:
                        break

                if g.value is None:
                    break
                g_opt = np.asarray(g.value).flatten()
                s = g_opt.sum()
                if s <= 1e-12:
                    break

                # normalise the LP solution and mix it with the current vector
                g_norm = g_opt / s
                new_cur = (1.0 - mix_const) * cur + mix_const * g_norm
                new_cur = _project_simplex(new_cur)

                new_score = _score(new_cur)
                if new_score < cur_score - 1e-9:
                    cur = new_cur
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_vec = new_cur.copy()

                # clean up for the next iteration
                del A, g, g_opt
                lp_iter += 1
                if lp_iter >= 20:          # safeguard against too many passes
                    break

    # --------------------------------------------------------
    # Second stage: smoothed‑max gradient descent (many cheap iters)
    # --------------------------------------------------------
    eps = 1.0
    eps_decay = 0.9995
    eps_min = 1e-9
    step0 = 0.5
    it = 0
    no_improve = 0
    Lgd = 1 << ((3 * n - 2).bit_length())   # FFT length for gradient step

    while time.monotonic() < deadline:
        it += 1

        # convolution of the current iterate
        a_fft = np.fft.rfft(cur, n=Lgd)
        conv = np.fft.irfft(a_fft * a_fft, n=Lgd)[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights around the maximum
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=Lgd)
        grad_full = np.fft.irfft(a_fft * w_fft, n=Lgd)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # step size (scale‑invariant)
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        cand = cur - alpha * g
        cand = _project_simplex(cand)

        # occasional symmetrisation (helps because the problem is symmetric)
        if (it % 800) == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_score:
            cur, cur_score = cand, cand_sc
            if cand_sc < best_score:
                best_score, best_vec = cand_sc, cand.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                # small random mass displacement to escape plateaus
                i_idx = rng.integers(0, n)
                if cur[i_idx] > 0.0:
                    delta = cur[i_idx] * rng.random() * 0.3
                    j_idx = rng.integers(0, n)
                    if j_idx == i_idx:
                        j_idx = (j_idx + 1) % n
                    tmp = cur.copy()
                    tmp[i_idx] -= delta
                    tmp[j_idx] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_score:
                        cur, cur_score = tmp, tmp_sc
                        if tmp_sc < best_score:
                            best_score, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        if best_score < 1.50274349:
            break

    # --------------------------------------------------------
    # Final trimming of negligible tail entries (reduces n ⇒ better score)
    # --------------------------------------------------------
    mask = best_vec > 1e-12
    if mask.any():
        first, last = np.where(mask)[0][[0, -1]]
        trimmed = best_vec[first : last + 1]
        trimmed = _project_simplex(trimmed)
        trimmed_sc = _score(trimmed)
        if trimmed_sc < best_score:
            best_vec, best_score = trimmed, trimmed_sc

    # --------------------------------------------------------
    # Return as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
