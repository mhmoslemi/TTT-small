def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # little margin

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
    else:
        n = 1024
        a = rng.random(n, dtype=np.float64)
        a = _project_simplex(a)

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())

    # Initialize optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.4  # base learning rate
    eps_smooth0 = 2.0
    eps_decay = 0.9998
    eps_min = 1e-8

    best = a.copy()
    best_score = _score(best)
    cur = a.copy()
    cur_score = best_score

    it = 0
    no_improve = 0

    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # compute convolution of cur
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # smoothing parameter
        eps_smooth = max(eps_min, eps_smooth0 * (eps_decay ** it))

        # softmax weights
        w = np.exp((conv - max_conv) / eps_smooth)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size
        lr = step0 / math.sqrt(it)

        # candidate update with backtracking
        cand = None
        cand_score = None
        step_factor = 1.0
        for _ in range(7):
            delta = - lr * step_factor * (m_hat / (np.sqrt(v_hat) + eps_adam))
            cand_tmp = cur + delta
            cand_tmp = _project_simplex(cand_tmp)
            # compute score
            cand_fft = np.fft.rfft(cand_tmp, n=L)
            conv_tmp = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
            cand_tmp_score = 2.0 * n * conv_tmp.max()
            if cand_tmp_score <= cur_score:
                cand = cand_tmp
                cand_score = cand_tmp_score
                break
            step_factor *= 0.5
        if cand is None:
            # take the last tried candidate even if worse
            cand = cand_tmp
            cand_score = cand_tmp_score

        # accept
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # occasional uphill
            prob = math.exp(-(cand_score - cur_score) / (lr * step_factor + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score

import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a,a)), assuming sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
