# Projection onto the probability simplex {x >= 0, sum(x)=1}

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]

        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)

    return w

        cur /= s                         # normalise to simplex

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length ≥ 2n‑1

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    maxc, _ = _conv_max(cur, L)
    cur_score = 2.0 * n * maxc
    best = cur.copy()
    best_score = cur_score

    # Adam‑style accumulator
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.25      # base learning rate
    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()
        cur_score = 2.0 * n * maxc

        # ----- record best -----
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_score < 1.50274349:
            break

        # ----- sub‑gradient at an active index -----
        k = int(conv.argmax())
        start_idx = max(0, k - (n - 1))
        end_idx = min(k + 1, n)
        g = np.zeros_like(cur)
        idx = np.arange(start_idx, end_idx)
        g[idx] = 2.0 * cur[k - idx]      # exact sub‑gradient

        # ----- Adam update (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        # ----- optional symmetrisation (helps due to near‑symmetry) -----
        if it % 7 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_conv = np.fft.irfft(
            np.fft.rfft(cand, n=L) * np.fft.rfft(cand, n=L), n=L
        )[: 2 * n - 1]
        cand_score = 2.0 * n * cand_conv.max()

        # ----- accept / possibly uphill move -----
        if cand_score < cur_score:
            cur = cand
        else:
            # simulated‑annealing style uphill acceptance
            temperature = 0.02 / (1.0 + it * 1e-4)
            if rng.random() < math.exp(-(cand_score - cur_score) / (temperature + 1e-12)):
                cur = cand

        # ----- random mass‑transfer when stuck -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_conv = np.fft.irfft(
                    np.fft.rfft(tmp, n=L) * np.fft.rfft(tmp, n=L), n=L
                )[: 2 * n - 1]
                tmp_score = 2.0 * n * tmp_conv.max()
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score
                no_improve = 0

    # --------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best]
