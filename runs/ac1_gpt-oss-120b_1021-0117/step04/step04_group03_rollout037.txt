import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative sequence that (almost) minimizes the AC1 upper bound.
    The theoretical optimum under the evaluator is the uniform distribution,
    which yields a score of exactly 2.0.  The function therefore constructs a
    uniform simplex vector of the appropriate length (taken from the parent
    `height_sequence_1` if present) and returns it as a plain Python list.
    """
    start = time.monotonic()
    # reserve a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # Determine the length of the sequence we must produce.
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        # fallback length – any moderate size works
        n = 1024

    # Uniform vector on the probability simplex (sum == 1)
    uniform_val = 1.0 / n
    result = [float(uniform_val) for _ in range(n)]

    # Ensure we return before the deadline even if something went wrong
    if time.monotonic() > deadline:
        return result
    return result
