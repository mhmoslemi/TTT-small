import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Compute the AC1 upper bound for a given vector (sum is assumed to be 1)
# ----------------------------------------------------------------------
def _ac1_score(vec: np.ndarray, fft_len: int) -> float:
    """Return 2*n*max(conv) where conv = vec * vec (full linear convolution)."""
    n = vec.shape[0]
    vf = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(vf * vf, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Generate a decreasing geometric sequence and normalise it
# ----------------------------------------------------------------------
def _geometric_sequence(n: int, r: float) -> np.ndarray:
    """Return a length‑n non‑negative vector a_i = C * r**i, sum(a)=1."""
    if r <= 0.0:
        a = np.zeros(n)
        a[0] = 1.0
        return a
    if r >= 1.0 - 1e-12:
        # essentially uniform
        return np.full(n, 1.0 / n)
    t = 1.0 - r  # 1 - r
    # denominator 1 - r**n  (use log to avoid underflow)
    log_r = math.log(r)
    r_n = math.exp(log_r * n)
    denom = 1.0 - r_n
    c = t / denom if denom != 0.0 else 1.0 / n
    powers = np.exp(log_r * np.arange(n))
    a = c * powers
    return a

            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_score:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_score = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        # early exit if target already beaten
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
