import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 score assuming sum(a)=1 (returns 2*n*max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Compute 2*n*max(a*a) for a with sum≈1 using FFT length L."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    best feasible vector found.

            s = 1.0
        a0 = a0 / s                     # now sum(a0)=1
    else:

        n_fallback = 2000

    n = a0.size

    L = 1 << ((2 * n - 1).bit_length())

    cur = a0.copy()

    best_vec = cur.copy()
