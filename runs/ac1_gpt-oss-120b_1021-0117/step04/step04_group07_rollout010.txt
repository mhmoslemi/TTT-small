# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2 * n * max(a * a) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a little head‑room for the harness
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # --------------------------------------------------------
    # initial vector (use parent if present)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # optional mild smoothing to reduce sharp peaks
        k = 9  # kernel width
        a0 = np.convolve(a0, np.ones(k) / k, mode="same")
    else:
        # fallback: random positive entries
        n_fallback = 2000
        a0 = rng.random(n_fallback, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # project onto simplex
    a0 = _project_simplex(a0)          # guarantee exact simplex

    n = a0.size
    # FFT length for full convolution (>= 3n-2)
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    cur = a0.copy()
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()
    best_sc = cur_sc

    # algorithmic parameters
    eps = 0.2                # initial smoothing for soft‑max
    eps_min = 1e-8
    eps_decay = 0.997
    step0 = 0.6              # base step size
    beta = 0.9               # momentum factor
    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0
    stagnation_limit = 5000

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- gradient (smooth or sub‑gradient) -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
            eps = max(eps_min, eps * eps_decay)
        else:
            # sub‑gradient: equal weight on all maximal indices
            mask = np.isclose(conv, maxc, atol=1e-12)
            mk = mask.astype(np.float64)
            mk_rev = mk[::-1]
            mk_fft = np.fft.rfft(mk_rev, n=L)
            grad_full = np.fft.irfft(a_fft * mk_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step and projection -----
        grad_norm = np.linalg.norm(momentum) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _fast_score(cand, L, n)

        # ----- accept/reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional uphill acceptance
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- random mass transfer when stuck -----
        if no_improve > stagnation_limit:
            # mix best with a fresh random point
            rand_vec = rng.random(n)
            rand_vec = _project_simplex(rand_vec)
            cur = 0.5 * best + 0.5 * rand_vec
            cur = _project_simplex(cur)
            cur_sc = _fast_score(cur, L, n)
            momentum.fill(0.0)
            no_improve = 0
            eps = 0.2

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # Return the best vector found (as plain Python list)
    return [float(x) for x in best]
