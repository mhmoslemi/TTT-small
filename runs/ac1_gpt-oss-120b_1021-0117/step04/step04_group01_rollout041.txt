def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.sparse as sp, scipy.optimize as opt, math
    # setup random
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # init a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = np.ones(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total            # normalized sum = 1
    n = a.size
    # FFT length
    L = 1 << ((2 * n - 1).bit_length())
    # scoring function
    def calc_score(v):
        conv = np.fft.irfft(np.fft.rfft(v, n=L) * np.fft.rfft(v, n=L), n=L)[:2*n-1]
        maxb = conv.max()
        s = v.sum()
        if s < 0.01:
            return np.inf
        return 2.0 * n * maxb / (s * s)
    best_vec = a.copy()
    best_sc = calc_score(best_vec)
    # main loop
    it = 0
    t_mix = 0.5
    # configure number of LP steps before fallback
    while time.monotonic() < deadline:
        # compute current max conv and conv vector
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        M = conv.max()
        # Build constraint matrix
        # if time is low, break to final fine tuning
        row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
        col_idx = np.repeat(np.arange(n), n)
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        # Solve LP maximize sum(b) s.t. A @ b <= M, b >= 0
        c = -np.ones(n)   # minimize -sum
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        # using highs
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True})
        if not res.success:
            # fallback to gradient
            break
        b_opt = res.x
        sum_a = a.sum()
        if b_opt.sum() == 0:
            break
        g = b_opt * (sum_a / b_opt.sum())
        # mix
        a_new = (1 - t_mix) * a + t_mix * g
        a_new = _project_simplex(a_new)
        sc_new = calc_score(a_new)
        if sc_new < best_sc:
            best_vec, best_sc = a_new.copy(), sc_new
        # accept if improved
        if sc_new < best_sc * 0.9999:  # improvement
            a = a_new
            # maybe reduce t_mix gradually
            t_mix = max(0.1, t_mix * 0.9)
        else:
            # no improvement, reduce mixing factor
            t_mix = max(0.05, t_mix * 0.5)
        it += 1
        # stop condition early target
        if best_sc < 1.50274349:
            break
    # after LP loop, maybe gradient fine-tuning
    # Use same gradient descent as earlier but with best_vec as start
    cur = best_vec.copy()
    cur_sc = best_sc
    # gradient hyperparams
    eps0 = 0.5
    eps_decay = 0.9995
    step0 = 0.6
    no_improve = 0
    while time.monotonic() < deadline - 0.1:
        # compute conv and gradient via softmax as earlier
        af = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        max_val = conv.max()
        eps = max(1e-12, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum:
            w /= w_sum
        else:
            w[:] = 1.0 / w.size
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # line search
        step = step0 / math.sqrt(it + 1)
        found = False
        cand = None
        cand_sc = None
        for _ in range(8):
            cand = cur - step * g
            cand = _project_simplex(cand)
            cand_sc = calc_score(cand)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_vec, best_sc = cand.copy(), cand_sc
                found = True
                break
            step *= 0.5
        if not found:
            no_improve += 1
            if no_improve > 2000:
                # random perturb
                i = rng.integers(0, n)
                if cur[i] > 0:
                    delta = cur[i] * rng.random() * 0.2
                    j = rng.integers(0, n-1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = calc_score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0
        it += 1
        if best_sc < 1.50274349:
            break
    return [float(x) for x in best_vec]

import time, math, numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(v: np.ndarray, fft_len: int) -> float:
    """Return the AC1 bound for a simplex‑normalised vector."""
    n = v.shape[0]
    af = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(v.sum())
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple fallback: decreasing exponential, normalised to sum 1
        n_fallback = 2000
        r = 0.998
        a = np.power(r, np.arange(n_fallback), dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # simplex normalisation (sum = 1)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length covering full convolution

    # ----- bookkeeping ------------------------------------------------
    best_vec = a.copy()
    best_sc = _score(best_vec, L)

    # ----- pre‑compute the sparsity pattern for the LP ----------------
    idx = np.arange(n, dtype=np.int32)
    row_idx = (idx[:, None] + idx[None, :]).ravel()          # shape (n*n,)
    col_idx = np.repeat(idx, n)                              # shape (n*n,)

    # ----- phase 1 : LP‑guided descent --------------------------------
    t_mix = 0.5                # mixing weight between old and LP‑solution
    max_lp_iters = 15
    lp_iter = 0
    while time.monotonic() < deadline and lp_iter < max_lp_iters:
        # current maximal convolution value
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        M = float(conv.max())

        # build the Toeplitz matrix A where (A @ b)[k] = (a * b)[k]
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        # linear programme: maximise sum(b)  subject to A @ b <= M , b >= 0
        c = -np.ones(n, dtype=np.float64)                     # minimise -sum(b)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        try:
            res = opt.linprog(
                c, A_ub=A, b_ub=b_ub,
                bounds=(0, None),
                method='highs',
                options={'presolve': True}
            )
        except Exception:   # in case the solver is unavailable / fails
            break

        if not res.success:
            break

        b_opt = res.x
        if b_opt.sum() == 0.0:
            break

        # rescale to keep the total mass unchanged
        g = b_opt * (a.sum() / b_opt.sum())

        # convex combination and projection back onto the simplex
        a_new = (1 - t_mix) * a + t_mix * g
        a_new = _project_simplex(a_new)

        sc_new = _score(a_new, L)
        if sc_new < best_sc:
            best_vec, best_sc = a_new.copy(), sc_new

        # accept if it improves, otherwise shrink the mixing factor
        if sc_new < best_sc * 0.9999:
            a = a_new
            t_mix = max(0.1, t_mix * 0.9)
        else:
            t_mix = max(0.05, t_mix * 0.5)

        lp_iter += 1
        if best_sc < 1.50274349:
            break

    # ----- phase 2 : smooth‑max projected gradient descent ------------
    cur = best_vec.copy()
    cur_sc = best_sc
    eps0 = 0.5
    eps_decay = 0.9996
    step0 = 0.8
    no_improve = 0
    it = 0

    while time.monotonic() < deadline - 0.1:
        it += 1

        af = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights for a smooth approximation of the max
        eps = max(1e-12, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the smooth max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        idx_slice = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx_slice]

        # back‑tracking line search
        step = step0 / math.sqrt(it)
        improved = False
        for _ in range(8):
            cand = cur - step * g
            cand = _project_simplex(cand)
            cand_sc = _score(cand, L)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_vec, best_sc = cand.copy(), cand_sc
                improved = True
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve > 2000:
                # random mass exchange to escape stagnation
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.25
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp, L)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0
        else:
            no_improve = 0

        if best_sc < 1.50274349:
            break

    # ----- return result ------------------------------------------------
    return [float(x) for x in best_vec]
