import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast score for a vector that is already normalised to sum == 1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L_fft: int) -> float:
    """2 * n * max(full linear convolution of a with itself)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(af * af, n=L_fft)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------

    best_sc = _fast_score(best, 1 << ((2 * n - 1).bit_length()))
