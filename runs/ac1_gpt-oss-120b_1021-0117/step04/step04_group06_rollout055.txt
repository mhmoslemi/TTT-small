        w = np.exp((conv_full - max_val) / eps_temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        wf = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * wf, n=L)
        idx = (2 * n - 2) - np.arange(n)           # slice producing length‑n gradient
        g = 2.0 * grad_full[idx]

        # Adam update with simplex projection
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps_adam)
        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        cand_sc = _score_vec(cand, n, L)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional uphill move to avoid stagnation
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # enforce (optional) symmetry every few iterations
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # random mass transfer when progress stalls
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # early exit if the target bound is reached
        if best_sc < 1.50274349:
            break

    # ------------------ final local refinement ------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        tmp = best.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score_vec(tmp, n, L)
        if tmp_sc < best_sc:
            best, best_sc = tmp, tmp_sc
            continue
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _project_simplex(sym)
        sym_sc = _score_vec(sym, n, L)
        if sym_sc < best_sc:
            best, best_sc = sym, sym_sc

    # return as plain Python list
    return [float(x) for x in best]
