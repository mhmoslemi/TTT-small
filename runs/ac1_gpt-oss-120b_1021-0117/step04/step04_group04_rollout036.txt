def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a = np.exp(-np.arange(n0) / 200.0)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.shape[0]

    L3 = 1 << ((3 * n - 2).bit_length())
    L2 = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best, L2)

    cur = best.copy()
    cur_sc = best_sc

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    eps0 = 0.5
    eps_decay = 0.9995
    eps_min = 1e-8

    lr = 0.2  # base learning rate
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L3)
        conv = np.fft.irfft(a_fft * a_fft, n=L3)[:2 * n - 1]
        max_val = conv.max()

        # weights
        eps = eps0 * (eps_decay ** it)
        if eps < eps_min:
            eps = eps_min
        w = np.exp((conv - max_val) / eps)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L3)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L3)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand, L2)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional random acceptance
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        if no_improve > 2000:
            i1 = rng.integers(0, n)
            if cur[i1] > 0.0:
                delta = cur[i1] * rng.random() * 0.4
                i2 = rng.integers(0, n-1)
                if i2 >= i1:
                    i2 += 1
                tmp = cur.copy()
                tmp[i1] -= delta
                tmp[i2] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L2)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        if it % 3000 == 0:
            sym = (cur + cur[::-1]) * 0.5

            sym_sc = _score(sym, L2)

                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        if best_sc < target:
            break

    return [float(x) for x in best]

import time
import numpy as np

# ------------------------------------------------------------
# Simple normalization onto the probability simplex (sum = 1, >=0)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform if everything vanished
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ------------------------------------------------------------
# Fast AC1 upper‑bound score (assumes sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: moderate exponential decay
        n0 = 2000
        a = np.exp(-np.arange(n0, dtype=np.float64) / 200.0)

    a = _normalize(a)
    n = a.shape[0]

    # FFT lengths – enough for linear conv (2n‑1) and a⋆w (3n‑2)
    L3 = 1 << ((3 * n - 2).bit_length())
    L2 = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best, L2)

    cur = best.copy()
    cur_sc = best_sc

    # ----- optimisation hyper‑parameters ------------------------------
    beta1, beta2 = 0.9, 0.999          # Adam moments
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    eps_cur = 0.5                       # smoothing for log‑sum‑exp
    eps_decay = 0.9995
    eps_min = 1e-8
    lr = 0.2                            # base learning rate for Adam

    it = 0
    no_improve = 0
    target = 1.50274349

    # ----- main optimisation loop ------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and its maximum
        a_fft = np.fft.rfft(cur, n=L3)
        conv = np.fft.irfft(a_fft * a_fft, n=L3)[: 2 * n - 1]
        max_val = conv.max()

        # smoothed max weights
        eps_cur = max(eps_min, eps_cur * eps_decay)
        w = np.exp((conv - max_val) / eps_cur)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L3)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L3)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _normalize(cand)

        cand_sc = _score(cand, L2)

        # ----- acceptance ------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional uphill acceptance to escape plateaus
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- random mass‑transfer when stuck ---------------------------
        if no_improve > 2000:
            i1 = rng.integers(0, n)
            if cur[i1] > 0.0:
                delta = cur[i1] * rng.random() * 0.4
                i2 = rng.integers(0, n - 1)
                if i2 >= i1:
                    i2 += 1
                tmp = cur.copy()
                tmp[i1] -= delta
                tmp[i2] += delta
                tmp = _normalize(tmp)
                tmp_sc = _score(tmp, L2)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- symmetrisation from time to time -------------------------
        if it % 3000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            sym_sc = _score(sym, L2)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # ----- early exit ------------------------------------------------
        if best_sc < target:
            break

    # return plain Python list
    return [float(x) for x in best]
