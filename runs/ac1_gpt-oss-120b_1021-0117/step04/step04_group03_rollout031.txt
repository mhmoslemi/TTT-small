# Project onto the probability simplex {x >= 0, sum(x)=1}

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score for a vector that already sums to 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max( a * a )  (a is assumed to sum to 1)."""
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Determine length and initial vectors
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
        n = base.size
    else:
        n = 1024
        base = np.full(n, 1.0 / n, dtype=np.float64)

    base = _project_simplex(base)                     # sum = 1

    # additional cheap starters
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()

    # decreasing exponential (r close to 1)
    r = 1.0 - 1.0 / (n * 1.5)          # δ ≈ 2/3n
    exp_vec = np.exp(np.log(r) * np.arange(n))
    exp_vec = exp_vec / exp_vec.sum()

    # random Dirichlet start
    random_vec = rng.random(n)
    random_vec = random_vec / random_vec.sum()

    init_cands = [base, uniform, arcsine, exp_vec, random_vec]

    best_vec = None
    best_sc = float("inf")
    for cand in init_cands:
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------
    # Parameters for the smoothed‑max descent
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())
    step0 = 0.8
    eps = 1.0
    eps_decay = 0.9995
    eps_min = 1e-8
    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and smoothed max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- descent direction & adaptive step -----
        direction = -g
        grad_norm = np.linalg.norm(direction) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm

        # ----- try a few step multipliers (simple line‑search) -----
        best_local = cur
        best_local_sc = cur_sc
        for factor in (1.0, 0.5, 1.5, 2.0):
            cand = cur + factor * alpha * direction
            cand = _project_simplex(cand)
            sc = _score(cand)
            if sc < best_local_sc:
                best_local_sc, best_local = sc, cand

        # ----- occasional symmetrisation (exploits inherent symmetry) -----
        if it % 400 == 0:
            best_local = (best_local + best_local[::-1]) * 0.5
            best_local = _project_simplex(best_local)
            best_local_sc = _score(best_local)

        # ----- accept improvement -----
        if best_local_sc < cur_sc - 1e-12:
            cur, cur_sc = best_local, best_local_sc
            no_improve = 0
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()
        else:
            no_improve += 1

            # ----- random mass transfer when stuck -----
            if no_improve > 2000:
                i_idx = rng.integers(0, n)
                if cur[i_idx] > 0.0:
                    delta = cur[i_idx] * rng.random() * 0.4
                    j_idx = rng.integers(0, n)
                    if j_idx == i_idx:
                        j_idx = (j_idx + 1) % n
                    tmp = cur.copy()
                    tmp[i_idx] -= delta
                    tmp[j_idx] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        no_improve = 0
                        if cur_sc < best_sc:
                            best_sc, best_vec = cur_sc, cur.copy()

            # ----- occasional restart with a fresh random point -----
            if no_improve > 5000:
                cur = rng.random(n)
                cur = cur / cur.sum()
                cur_sc = _score(cur)
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return the best found vector as a plain Python list
    # --------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    return [float(x) for x in best_vec]
