def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    
    # init a
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_ = 2000
        a = rng.random(n_)
    a = np.maximum(a, 0)
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()
        
    n = a.size
    L = 1 << ((3*n - 2).bit_length())   # same as before: enough to not wrap
    
    # helpers
    def project_simplex(v):
        # same as before but implement quickly
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        ind = np.arange(1, len(v)+1)
        rho = np.where(u - css / ind > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)
    
    def score(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return 2.0 * n * float(conv.max())
    
    cur = a.copy()
    cur_sc = score(cur)
    best = cur.copy()
    best_sc = cur_sc
    
    # parameters
    eps0 = 0.2     # smoothing for softmax (small for better gradient)
    eps_decay = 0.9999
    eps_min = 1e-12
    step0 = 1.0
    beta = 0.9    # momentum
    vel = np.zeros_like(cur)
    no_improve = 0
    it = 0
    
    # loops
    while time.monotonic() < deadline:
        it += 1
        
        # FFT of cur
        af = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        max_val = conv.max()
        
        # active set mask for gradient weights:
        # tolerance relative
        tol = max_val * 1e-8 + 1e-12
        mask = conv >= (max_val - tol)
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum
            
        # compute gradient via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        
        # step size
        step = step0 / math.sqrt(it)
        # momentum update
        vel = beta * vel - step * g
        cand = cur + vel
        cand = project_simplex(cand)
        # optional symmetrisation every S steps
        if it % 600 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = project_simplex(cand)
        
        cand_sc = score(cand)
        
        # accept if better, else backtrack with reduced step (no momentum)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            # update best
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
            else:
                no_improve += 1
        else:
            # backtrack: try decreasing step without momentum up to attempts
            improved = False
            step_bt = step
            for _ in range(12):
                step_bt *= 0.7
                vel_bt = - step_bt * g      # ignore momentum for backtrack
                cand_bt = cur + vel_bt
                cand_bt = project_simplex(cand_bt)
                if it % 600 == 0:

                    cand_bt = project_simplex(cand_bt)
                cand_bt_sc = score(cand_bt)
                if cand_bt_sc < cur_sc:
                    cur, cur_sc = cand_bt, cand_bt_sc
                    # update best
                    if cand_bt_sc < best_sc:
                        best, best_sc = cand_bt.copy(), cand_bt_sc
                        no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
        
        # random perturbation if stuck
        if no_improve > 3000:
            # apply small random redistribution of mass
            i = rng.integers(0, n)

                delta = cur[i] * rng.random() * 0.2
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_sc = score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
        
        # early exit
        if best_sc < 1.50274349:
            break
    
    # final fine tuning random Gaussian perturbations while time left
    while time.monotonic() < deadline - 0.1:
        delta = rng.normal(scale=0.003, size=n)
        cand = best + delta
        cand = project_simplex(cand)
        cand_sc = score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
    
    # ensure nonempty: fallback to uniform if something went wrong
    if best_sc == float('inf') or not np.isfinite(best_sc):
        best = np.full(n, 1.0/n)
    
    return [float(x) for x in best]

import time
import math
import numpy as np

# ---------- helpers (top level) ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - css / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score_fft(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv(a,a)) using FFT of length L."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())
