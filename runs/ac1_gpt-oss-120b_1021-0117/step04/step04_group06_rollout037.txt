def _max_grad(a, conv, tol=1e-12):
    max_val = conv.max()
    # Determine tolerance for active indices
    eps = tol * max_val + tol
    idxs = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for m in idxs:
        lo = max(0, m - (n-1))
        hi = min(m, n-1)
        # length = hi-lo+1
        i = np.arange(lo, hi+1, dtype=np.int64)
        j = m - i
        grad[i] += 2.0 * a[j]
    # average
    if idxs.size > 0:
        grad /= idxs.size
    return max_val, grad

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _score_vec(v: np.ndarray, L: int) -> float:
    """Fast evaluation of the AC1 bound for a vector v (sum(v) assumed 1)."""
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * v.size - 1]
    return float(2.0 * v.size * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.monotonic()

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # simplex projection (sum=1)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFT

    best_vec = a.copy()
    best_sc = _score_vec(best_vec, L)
    cur = a.copy()
    cur_sc = best_sc

    step0 = 0.8          # base step magnitude
    no_improve = 0
    it = 0

    # ---- main optimisation loop ----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        max_val = conv.max()
