import time
import math
import numpy as np

# -------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, x >= 0)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# -------------------------------------------------------------
def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the full convolution a*a using FFT of length fft_len."""
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative list of coefficients with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------ initialise length -----------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = a.size
    else:
        n = 2500
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------ geometric start search -----------------
    # we search a simple geometric shape a_i = c * r**i (r in (0,1)).
    # The best r found is used as the starting point.
    L = 1 << ((3 * n - 2).bit_length())          # length for FFTs

    # uniform baseline (used later if it turns out better)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_a = uniform.copy()
    best_score = 2.0 * n * _conv_max(best_a, L)

    # coarse grid for r (log‑scale near 1)
    r_vals = np.linspace(0.90, 0.99999, 200)
    for r in r_vals:
        # compute geometric vector analytically to avoid overflow
        log_r = math.log(r)
        # denominator 1 - r**n
        r_n = math.exp(n * log_r)
        denom = 1.0 - r_n
        if denom == 0.0:
            continue
        c = (1.0 - r) / denom
        a_geo = c * np.exp(log_r * np.arange(n))
        # quick sanity: ensure non‑negative
        a_geo = np.maximum(a_geo, 0.0)
        # normalise to sum = 1 (project onto simplex is equivalent)
        a_geo /= a_geo.sum()

        if cur_score < best_score:

    # start from the best geometric (or uniform if that stayed best)
    a = best_a.copy()
    cur_score = best_score

    step0 = 0.9

    min_step = 1e-7

        tol = max_val * 1e-12 + 1e-15

        w = np.zeros_like(conv)

        w_rev = w[::-1]
