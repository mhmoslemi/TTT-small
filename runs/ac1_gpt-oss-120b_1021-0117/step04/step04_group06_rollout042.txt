import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x>=0, sum x = 1}. """
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative coefficients with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple exponential decay fallback
        n0 = 1024
        a = np.exp(-np.arange(n0, dtype=np.float64) / 20.0)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)               # ensures sum = 1
    n = a.shape[0]

    # FFT length: next power of two at least 2*n-1
    L = 1 << ((2 * n - 1).bit_length())

    # ---- bookkeeping ------------------------------------------------
    best = a.copy()
    best_score = float(2.0 * n * np.max(np.convolve(best, best)))
    cur = a.copy()
    cur_score = best_score

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.3

    # soft‑max (smooth max) parameters
    eps0, eps_decay, eps_min = 0.5, 0.9995, 1e-8

    # stagnation handling
    no_improve = 0
    it = 0

    idx = (2 * n - 2) - np.arange(n)      # slice to obtain gradient of length n

    while time.monotonic() < deadline:
        it += 1

        # ---- compute convolution and smooth weights -----------------
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_c = conv.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_c) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient (smooth‑max surrogate) -----------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[idx]

        # ---- Adam step --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        # ---- evaluate ---------------------------------------------
        cand_score = float(2.0 * n * np.max(np.convolve(cand, cand)))

        # ---- accept / store best ----------------------------------
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # occasional Metropolis acceptance
            prob = math.exp(-(cand_score - cur_score) / (np.linalg.norm(step) + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
            no_improve += 1

        # ---- periodic symmetrisation -------------------------------
        if it % 600 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = float(2.0 * n * np.max(np.convolve(sym, sym)))
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # ---- random mass‑transfer when stuck -----------------------
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = float(2.0 * n * np.max(np.convolve(tmp, tmp)))
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
            no_improve = 0

        # early exit if we already beat the target
        if best_score < 1.50274349:
            break

    # ---- final local jitter ---------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.07
        if delta <= 0.0:
            continue
        tmp = best.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_score = float(2.0 * n * np.max(np.convolve(tmp, tmp)))
        if tmp_score < best_score:
            best, best_score = tmp, tmp_score
            continue
        # occasional symmetry check
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _project_simplex(sym)
        sym_score = float(2.0 * n * np.max(np.convolve(sym, sym)))
        if sym_score < best_score:
            best, best_score = sym, sym_score

    return [float(x) for x in best]
