    def score(a):

        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    # Optimization routine: given initial vector, returns best
    def optimize(init):
        cur = init.copy()
        # momentum
        m = np.zeros_like(cur)
        beta = 0.9
        # hyperparameters
        step0 = 0.5
        eps = 1.0
        eps_decay = 0.998
        eps_min = 1e-7
        best_local = cur.copy()
        best_score = score(cur)
        it = 0
        while time.monotonic() < deadline:
            it += 1
            # compute conv & gradient:
            cur_fft = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(cur_fft * cur_fft, n=L)[:2*n-1]
            max_val = conv.max()
            # smoothing weight
            eps = max(eps_min, eps * eps_decay)
            w = np.exp((conv - max_val)/eps)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0/w.size)
            else:
                w /= w_sum
            # gradient via convolution with reversed w
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
            idx = (2*n - 2) - np.arange(n)

            # momentum update
            m = beta * m + (1.0 - beta) * g

                best_score = cand_score

                if np.random.rand() < 0.01:

            # early break if reached target

        tmp[i] -= delta
        tmp[j] += delta
        tmp = project_simplex(tmp)
        tmp_score = score(tmp)
        if tmp_score < cur_score:
            cur = tmp
            cur_score = tmp_score
            if cur_score < best_overall_score:
                best_overall = cur.copy()
                best_overall_score = cur_score
        # occasional symmetry check
        if rng.random() < 0.1:
            sym = (cur + cur[::-1]) * 0.5

            sym_score = score(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                if sym_score < best_overall_score:
                    best_overall = sym.copy()
                    best_overall_score = sym_score

    # Return list
    return [float(x) for x in best_overall]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast score: 2 * n * max(convolution)  (sum assumed = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int, n: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Gradient‑descent optimiser with momentum & symmetry
# ------------------------------------------------------------
def _optimize(
    init: np.ndarray,
    L: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    target: float = 1.50274349,
    max_iters: int = 25000,
) -> tuple[np.ndarray, float]:
    cur = init.copy()
    m = np.zeros_like(cur)          # momentum buffer
    beta = 0.9                      # momentum coefficient
    step0 = 0.5                     # base step size
    eps = 0.5                       # initial smoothing
    eps_decay = 0.998
    eps_min = 1e-7
    idx = (2 * n - 2) - np.arange(n)  # index map for gradient extraction

    # initial score
    cur_fft = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
    cur_sc = 2.0 * n * conv.max()

    best_vec = cur.copy()
    best_sc = cur_sc
    it = 0

    while it < max_iters and time.monotonic() < deadline:
        it += 1

        # ----- convolution & max -----
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # ----- smoothed weights (soft‑max) -----
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (via convolution with reversed weights) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        m = beta * m + (1.0 - beta) * g
        grad_norm = np.linalg.norm(m) + 1e-12
        alpha = step0 / (math.sqrt(it) * grad_norm)

        # ----- candidate step & projection -----
        cand = cur - alpha * m
        cand = _project_simplex(cand)

        # ----- enforce symmetry each iteration -----
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand, L, n)

        # ----- record best -----
        if cand_sc < best_sc:
            best_vec = cand.copy()
            best_sc = cand_sc

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
        else:
            # occasional uphill move
            if rng.random() < 0.01:
                cur = cand
                cur_sc = cand_sc

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    return best_vec, best_sc
