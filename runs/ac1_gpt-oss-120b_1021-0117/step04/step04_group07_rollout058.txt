import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Score = 2*n*max(conv) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.4)  # safety margin

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a0 = np.full(n0, 1.0 / n0, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 /= s0
    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())  # FFT length >= 3n-2

    # ----- evaluate simple parametric families ---------------------------
    best = a0.copy()
    best_sc = _fast_score(best, L, n)

    # geometric (decreasing)
    for r in np.linspace(0.90, 0.999, 22):
        v = np.power(r, np.arange(n, dtype=np.float64))
        v /= v.sum()
        sc = _fast_score(v, L, n)
        if sc < best_sc:
            best, best_sc = v, sc

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    sc = _fast_score(v, L, n)
    if sc < best_sc:
        best, best_sc = v, sc

    # quadratic decreasing
    v = np.square(np.arange(n, 0, -1, dtype=np.float64))
    v /= v.sum()
    sc = _fast_score(v, L, n)
    if sc < best_sc:
        best, best_sc = v, sc

    # sin shape (half‑period)
    v = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    v /= v.sum()
    sc = _fast_score(v, L, n)
    if sc < best_sc:
        best, best_sc = v, sc

    # cosine‑based bump
    v = 1.0 - np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1))
    v /= v.sum()
    sc = _fast_score(v, L, n)
    if sc < best_sc:
        best, best_sc = v, sc

    cur = best.copy()
    cur_sc = best_sc

    # ----- optimizer state ------------------------------------------------
    m = np.zeros_like(cur)          # first moment
    v2 = np.zeros_like(cur)         # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    it = 0
    no_improve = 0
    eps_s = 0.02          # temperature for soft‑max smoothing
    eps_decay = 0.9995
    eps_min = 1e-7
    target = 1.50274349

    idx_grad = (2 * n - 2) - np.arange(n)  # indices for gradient extraction

    # ----- main optimisation loop ----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()
        cur_sc = 2.0 * n * maxc

        # gradient (soft‑max or sub‑gradient)
        if eps_s > eps_min:
            w = np.exp((conv - maxc) / eps_s)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            g = 2.0 * grad_full[idx_grad]
            eps_s = max(eps_min, eps_s * eps_decay)
        else:
            k = int(conv.argmax())
            start = max(0, k - (n - 1))
            end = min(k + 1, n)
            g = np.zeros_like(cur)
            i_idx = np.arange(start, end)
            j_idx = k - i_idx
            g[i_idx] = 2.0 * cur[j_idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v2 = beta2 * v2 + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v2 / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_sc = _fast_score(cand, L, n)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # occasional uphill move
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (np.linalg.norm(step) + 1e-12)):
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # keep global best
        if cur_sc < best_sc:
            best, best_sc = cur.copy(), cur_sc

        # symmetrisation (helps because problem is nearly symmetric)
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sc_sym = _fast_score(sym, L, n)
            if sc_sym < cur_sc:
                cur, cur_sc = sym, sc_sym
                if sc_sym < best_sc:
                    best, best_sc = sym.copy(), sc_sym
                no_improve = 0

        # random mass transfer when stuck
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                sc_tmp = _fast_score(tmp, L, n)
                if sc_tmp < cur_sc:
                    cur, cur_sc = tmp, sc_tmp
                    if sc_tmp < best_sc:
                        best, best_sc = tmp.copy(), sc_tmp
                    no_improve = 0

        # occasional restart from best with noise
        if no_improve > 8000:
            noise = rng.normal(scale=0.02, size=n)
            cur = best + noise
            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()
            cur = _project_simplex(cur)
            cur_sc = _fast_score(cur, L, n)
            m.fill(0.0)
            v2.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    return [float(x) for x in best]
