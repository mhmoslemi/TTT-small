    Returns np.inf if the input is invalid.

    if not isinstance(sequence, list):

    if not sequence:

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Scoring – the bound returned by the evaluator (sum is 1)
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, n: int, L: int) -> float:
    """Return 2*n*max(conv) for a with sum=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------

# ------------------------------------------------------------
def _soft_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Soft‑max weights for a given temperature eps."""
    m = conv.max()
    w = np.exp((conv - m) / eps)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= s
    return w
