import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast score = 2 * n * max(convolution)     (sum of a is 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray = None) -> float:
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9

    eps_adam = 1e-8
    lr = 0.05                         # base learning rate

    # ---- smoothing for soft‑max approximation -----------------------
    eps0 = 0.5

    no_improve = 0

    # ---- main optimisation loop ------------------------------------
    while time.monotonic() < deadline:

        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]

        if w_sum == 0.0:
