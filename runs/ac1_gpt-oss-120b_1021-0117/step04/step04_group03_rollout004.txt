    # compute gradient using smoothing epsilon
    fft_len = 1 << ((3*n - 2).bit_length())
    a_fft = np.fft.rfft(cur, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
    max_val = conv.max()
    eps = max(min_eps, eps * eps_decay)
    # compute softmax weights
    w = np.exp((conv - max_val)/eps)
    w_sum = w.sum()
    if w_sum == 0 or np.isinf(w_sum) or np.isnan(w_sum):
        w = np.full_like(w, 1.0/w.size)
    else:
        w /= w_sum
    # grad
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=fft_len)
    grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]

    # Adam update
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1**iteration)
    v_hat = v / (1 - beta2**iteration)
    direction = m_hat / (np.sqrt(v_hat) + eps_adam)
    lr = lr0 / np.sqrt(iteration)  # decaying learning rate
    cand = cur - lr * direction
    cand = _project_simplex(cand)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector assumed to sum to 1 (exact evaluator)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(convolution)  (sum(a) is assumed 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
