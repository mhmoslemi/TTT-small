import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast max‑convolution score for vectors with sum == 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(a * a) assuming sum(a) == 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the max‑convolution (sub‑gradient)
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of  max(conv(a,a))  w.r.t. a.
    conv: full convolution a*a (length 2n‑1) already computed.
    """
    n = a.size
    max_val = conv.max()
    # indices where conv is (almost) maximal
    tol = max(1e-12, 1e-8 * max_val)
    w = np.where(conv >= max_val - tol, 1.0, 0.0)
    cnt = w.sum()
    if cnt == 0:
        w = np.ones_like(conv) / conv.size
    else:
        w = w / cnt  # average over tied maxima

    # convolution w * reversed(a)
    a_rev = a[::-1]
    L = 1 << ((3 * n - 2).bit_length())
    w_fft = np.fft.rfft(w, n=L)
    rev_fft = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(w_fft * rev_fft, n=L)
    # gradient = 2 * (w * a_rev) evaluated on positions n‑1 … 2n‑2
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return grad

# Optional LP improvement (single cheap step)
# ------------------------------------------------------------
def _lp_improve(cur: np.ndarray, max_val: float, rng: np.random.Generator) -> np.ndarray:
    """Solve the LP from the reference implementation and mix a bit."""
    try:
        import cvxpy as cp
    except Exception:
        return cur

    n = cur.size
    # build dense Toeplitz matrix A (2n‑1) × n  – memory ≈ n², OK for n≈2500
    rows = np.repeat(np.arange(2 * n - 1), n)
    cols = np.tile(np.arange(n), 2 * n - 1)
    data = np.empty_like(rows, dtype=np.float64)
    pos = 0
    for j in range(n):
        data[pos:pos + 2 * n - 1] = np.concatenate(
            (np.zeros(j),
             cur,
             np.zeros((2 * n - 1) - j - n))
        )
        pos += 2 * n - 1
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= max_val, g <= 1000.0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.GLPK, warm_start=True, verbose=False,
               glpk={"msg_lev": "GLP_MSG_OFF"})
    if g.value is None:
        return cur
    g_opt = np.asarray(g.value).ravel()
    s = g_opt.sum()
    if s < 1e-12:
        return cur
    g_norm = g_opt / s
    # small mixing
    t = 0.15
    new = (1.0 - t) * cur + t * g_norm
    new = _project_simplex(new)
    return new

        Ltmp = 1 << ((2 * n - 1).bit_length())
        cur_fft = np.fft.rfft(cur, n=Ltmp)
        cur_conv = np.fft.irfft(cur_fft * cur_fft, n=Ltmp)[: 2 * n - 1]
        cur_max = cur_conv.max()
        cur = _lp_improve(cur, cur_max, rng)
        # recompute score after mixing
        cur_sc = _score(cur)
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

    # --------------------------------------------------------
    # 3. Sub‑gradient descent (major work)
    # --------------------------------------------------------
    L_full = 1 << ((3 * n - 2).bit_length())
    step0 = 0.7
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # current convolution and max
        cur_fft = np.fft.rfft(cur, n=L_full)
        cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
        cur_max = cur_conv.max()
        cur_sc = float(2.0 * n * cur_max)

        # sub‑gradient at current point
        grad = _subgradient(cur, cur_conv)

        # step size (scale‑invariant)
        grad_norm = np.linalg.norm(grad) + 1e-12
        eta = step0 / (math.sqrt(it) * grad_norm)

        # additive trial step
        cand = cur - eta * grad
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        # evaluate trial
        cand_fft = np.fft.rfft(cand, n=L_full)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[: 2 * n - 1]
        cand_sc = float(2.0 * n * cand_conv.max())

        improved = False
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            improved = True
        else:
            # try a more conservative half step
            cand2 = cur - 0.5 * eta * grad
            cand2 = np.maximum(cand2, 0.0)
            cand2 = _project_simplex(cand2)
            cand2_fft = np.fft.rfft(cand2, n=L_full)
            cand2_conv = np.fft.irfft(cand2_fft * cand2_fft, n=L_full)[: 2 * n - 1]
            cand2_sc = float(2.0 * n * cand2_conv.max())
            if cand2_sc < cur_sc:
                cur, cur_sc = cand2, cand2_sc
                improved = True

        if improved:
            no_improve = 0
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()
        else:
            no_improve += 1
            # occasional random mass transfer when stuck
            if no_improve > 2500:
                i_src = rng.integers(0, n)
                if cur[i_src] > 0:
                    delta = cur[i_src] * rng.random() * 0.25
                    i_dst = rng.integers(0, n)
                    if i_dst == i_src:
                        i_dst = (i_dst + 1) % n
                    tmp = cur.copy()
                    tmp[i_src] -= delta
                    tmp[i_dst] += delta
                    tmp = _project_simplex(tmp)
                    tmp_fft = np.fft.rfft(tmp, n=L_full)
                    tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                    tmp_sc = float(2.0 * n * tmp_conv.max())
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        # periodic symmetrisation – the problem is symmetric under reversal
        if (it % 600) == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            cur_fft = np.fft.rfft(cur, n=L_full)
            cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
            cur_sc = float(2.0 * n * cur_conv.max())
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()

        # early termination if target reached
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # 4. Return best found vector (plain Python list)
    # --------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    # convert to plain list of floats for the harness
    return [float(x) for x in best_vec]
