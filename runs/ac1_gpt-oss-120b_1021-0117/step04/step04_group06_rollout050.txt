import cvxpy as cp

def lp_step(a, M, active_indices):
    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    # constraints: for each i in active_indices: (a conv g)[i] <= M
    # compute expression: sum_{j} a_j * g_{i - j} (valid indices)
    # We'll build each constraint using slicing:
    constraints = []
    for i in active_indices:
        # Compute sum_j a_j * g_{i - j}
        # Determine j range
        lo = max(0, i - (n-1))
        hi = min(i, n-1)
        # g indices: i - j from lo..hi => g indices from i - lo down to i - hi? Actually j asc.
        # For each j in [lo..hi], term a_j * g_{i - j}
        # Build linear expression: a[lo:hi+1] @ g[i - np.arange(lo,hi+1)]
        # compute slice of g using index vector
        # Instead of using cp.sum, we'll use cp.multiply and cp.sum
        j_range = np.arange(lo, hi+1)
        a_slice = a[j_range]
        g_idx = i - j_range  # indices in g
        # construct expression as cp.sum(cp.multiply(a_slice, g[g_idx]))
        # But cp.multiply expects elementwise; need to index variable g
        expr = cp.sum(cp.multiply(a_slice, cp.vstack([g[k] for k in g_idx])))
        constraints.append(expr <= M)
    # Objective maximize sum(g)
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return None
    return np.array(g.value).flatten()

import numpy as np

# Optional LP support
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_full(a: np.ndarray, conv: np.ndarray | None = None) -> float:
    """Exact AC1 bound for a (no assumptions on sum)."""
    n = a.shape[0]
    s = a.sum()
    if s == 0.0:
        return float("inf")
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def _subgrad_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max‑convolution at a using active indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        if k < n:
            # indexes 0..k receive reversed a[0:k+1]
            g[: k + 1] += 2.0 * a[k :: -1]
        else:
            L = 2 * n - 1 - k          # length of overlapping part
            start = k - n + 1
            # reversed slice of the last L elements of a
            g[start : start + L] += 2.0 * a[n - L ::][::-1]
    return g


def _lp_step(a: np.ndarray, M: float, active: np.ndarray):
    """Solve  max sum(g)  s.t. a * g <= M on active indices, g>=0."""
    if not _HAS_CVX:
        return None
    n = a.shape[0]
    rows = active.size
    if rows == 0:
        return None
    # Build constraint matrix A (rows x n)
    A = np.zeros((rows, n), dtype=np.float64)
    for r, i in enumerate(active):
        lo = max(0, i - (n - 1))
        hi = min(i, n - 1)
        j = np.arange(lo, hi + 1)
        A[r, i - j] = a[j]
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:  # pragma: no cover
        return None
    if g.value is None:
        return None
    g_val = np.maximum(np.asarray(g.value, dtype=np.float64).flatten(), 0.0)
    if g_val.sum() == 0.0:
        return None
    return g_val
