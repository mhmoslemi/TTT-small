import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 bound as evaluated by ``evaluate_sequence``.
    The theoretical optimum is attained by a uniform distribution,
    giving a score of exactly 2.0.
    """
    # Determine the length of the candidate sequence.
    # Prefer the length of the provided parent construction if available.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        # fallback length – any positive integer works; 2000 is a reasonable size.
        n = 2000

    # Uniform vector with sum = 1 (the projection used by the optimiser).
    # This yields the minimal possible value of the AC1 bound.
    uniform_value = 1.0 / n
    a = np.full(n, uniform_value, dtype=np.float64)

    # Convert to a plain Python list of floats for the evaluator.
    return [float(x) for x in a]
