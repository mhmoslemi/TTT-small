import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex using the algorithm of Duchi et al."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.nan_to_num(a0, nan=0.0, posinf=1000.0, neginf=0.0)
        a0 = np.clip(a0, 0.0, 1000.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s                     # now sum=1
    else:
        # fallback: exponential decay, length 2000
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()               # simplex

    n = a0.size
    # enforce symmetry from the start
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = _project_simplex(a0)

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())   # FFT length >= 2n-1
    momentum = np.zeros_like(cur)
    mu = 0.9                               # momentum coefficient
    base_alpha = 2.0                       # base step size
    eps = None
    eps_decay = 0.9996
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smoothing weight (soft‑max) -----
        if eps is None:
            eps = max(max_val * 0.2, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)

        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map to gradient components
        g = 2.0 * grad_full[idx]                  # gradient of weighted sum
        g_norm = np.linalg.norm(g) + 1e-12

        # ----- step size -----
        alpha = base_alpha / math.sqrt(it) / g_norm

        # ----- momentum update & candidate -----
        momentum = mu * momentum - alpha * g
        cand = cur + momentum

        # ----- enforce symmetry and simplex -----
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            # ----- fallback: plain gradient step without momentum -----
            cand2 = cur - (alpha * 0.5) * g
            cand2 = (cand2 + cand2[::-1]) * 0.5
            cand2 = _project_simplex(cand2)
            cand2_sc = _score(cand2)
            if cand2_sc < cur_sc:
                cand = cand2
                cand_sc = cand2_sc
                improved = True
                momentum = np.zeros_like(cur)   # discard accumulated momentum

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # gradually damp momentum on repeated failures
            if it % 500 == 0:
                momentum *= 0.5

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                no_improve = 0

        # ----- occasional restart from the best with tiny noise -----
        if no_improve > 8000:
            noise = rng.normal(scale=0.005, size=n)
            tmp = best_vec + noise
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0
            else:
                no_improve = 0
                momentum = np.zeros_like(cur)

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # Return as plain Python list
    return [float(x) for x in best_vec]
