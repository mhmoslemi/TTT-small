def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialize a
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t/100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())

    cur = a0.copy()
    cur_sc = _score_simple(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # parameters
    base_alpha = 1.0
    beta = 0.9
    eps_decay = 0.9995
    eps = None
    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0
    improve_target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        cur_sc = float(2.0 * n * max_val)  # recompute current score (simple)
        # also check best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # update epsilon
        if eps is None:
            eps = max(max_val * 0.2, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # compute softmax weights
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        g_norm = np.linalg.norm(g) + 1e-12

        # step size
        alpha = base_alpha / math.sqrt(it) / g_norm

        # momentum update
        direction = beta * momentum - alpha * g
        cand = cur + direction
        # project onto simplex
        cand = _project_simplex(cand)
        cand_sc = _score_simple(cand)

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            momentum = direction
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # fallback: try reducing step size (line search)
            improved = False
            for _ in range(3):
                alpha *= 0.5
                direction = beta * momentum - alpha * g
                cand = cur + direction
                cand = _project_simplex(cand)
                cand_sc = _score_simple(cand)
                if cand_sc < cur_sc:
                    cur = cand
                    cur_sc = cand_sc
                    momentum = direction
                    improved = True
                    no_improve = 0
                    if cand_sc < best_sc:
                        best_sc = cand_sc
                        best_vec = cand.copy()
                    break
            if not improved:

                prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
                if rng.random() < prob:
                    cur = cand
                    cur_sc = cand_sc
                    momentum = direction
                    no_improve = 0
                else:
                    no_improve += 1

        # symmetrisation every 250 iter
        if it % 250 == 0:

            sym = _project_simplex(sym)
            sym_sc = _score_simple(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                momentum = np.zeros_like(cur)  # reset momentum after sym
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass transfer when stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:

                tmp = cur.copy()
                tmp[i] -= delta

                if tmp_sc < cur_sc:

                    momentum = np.zeros_like(cur)
                    if tmp_sc < best_sc:

                    no_improve = 0

                    no_improve = 0

            no_improve = 0

        # early exit if target achieved
        if best_sc < improve_target:

    # return result as list of floats
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector normalised to sum=1
# ------------------------------------------------------------
def _score_simple(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a)) for sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---------- initialise ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s                         # normalise to sum = 1
    else:
        # fallback: exponential decay
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())      # FFT length for gradient steps

    cur = a0.copy()
    cur_sc = _score_simple(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---------- optimisation parameters ----------
    base_alpha = 1.0          # initial step‑size factor
    beta = 0.9                # momentum coefficient
    eps_decay = 0.9995
    eps = None                # smoothing parameter for soft‑max
    momentum = np.zeros_like(cur)

    it = 0
    no_improve = 0
    improve_target = 1.50274349

    # ---------- main loop ----------
    while time.monotonic() < deadline:
        it += 1

        # convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = float(2.0 * n * max_val)

        # keep best so far
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # smoothing weight (soft‑max)

            eps = max(max_val * 0.2, 1e-9)
        else:

        if w_sum == 0.0:
