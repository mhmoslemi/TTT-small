import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast convolution (linear) via FFT
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length L (L >= 2*len(a)-1)."""
    af = np.fft.rfft(a, n=L)
    return np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the exact evaluator score for a vector with sum==1."""
    return float(2.0 * n * conv.max())
