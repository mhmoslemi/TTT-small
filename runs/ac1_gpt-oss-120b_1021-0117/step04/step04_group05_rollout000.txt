import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    cur_sc = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_sc = cur_sc

    # optimisation parameters
    step0 = 0.9
    eps_rel = 0.02
    eps_min = 1e-7
    beta = 0.9                     # momentum factor
    grad_acc = np.zeros_like(cur)   # accumulated gradient
    it = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # smoothed‑max weights
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L_full)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_full)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        grad_acc = beta * grad_acc + (1.0 - beta) * g

        # line search
        alpha = step0 / math.sqrt(it)
        accepted = False
        for _ in range(8):
            cand = cur - alpha * grad_acc
            cand = _project_simplex(cand)
            cand_fft = np.fft.rfft(cand, n=L_full)
            cand_conv_full = np.fft.irfft(cand_fft * cand_fft, n=L_full)
            cand_conv = cand_conv_full[: 2 * n - 1]
            cand_max = cand_conv.max()
            cand_sc = 2.0 * n * cand_max
            if cand_sc < cur_sc:
                accepted = True
                break
            alpha *= 0.5

        if accepted:
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_max
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # occasional probabilistic acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_fft = cand_fft
                cur_conv = cand_conv
                cur_max = cand_max
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # periodic symmetrisation
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L_full)
            sym_conv_full = np.fft.irfft(sym_fft * sym_fft, n=L_full)
            sym_conv = sym_conv_full[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_conv.max()
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # random mass‑transfer when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv_full = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)
                tmp_conv = tmp_conv_full[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_sc = tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------------
    # Return best candidate found
    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
