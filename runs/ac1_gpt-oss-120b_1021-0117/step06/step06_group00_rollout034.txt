def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    # random generator
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # small margin

    # initialization
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random vector of size ~2000
        n0 = 2000
        a0 = rng.random(n0, dtype=np.float64)
    # Ensure nonnegative (already)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    
    # Keep best
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam optimizer parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # Step size schedule
    lr0 = 0.5
    # Softmax beta schedule
    beta_soft = 0.5
    beta_max = 20.0
    beta_inc = (beta_max - beta_soft) / 5000  # up to ~5000 iterations
    it = 0
    no_improve = 0
    target = 1.50274349
    # main loop
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # compute FFT of current a
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        maxc = conv.max()

        # compute softmax weights
        # scale conv for numerical stability
        # Using current beta_soft
        # To avoid overflow, subtract maxc
        w_raw = np.exp(beta_soft * (conv - maxc))
        w = w_raw / w_raw.sum()

        # compute gradient via weight
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 / (np.sqrt(it) + 1)  # decaying step maybe sqrt
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - update)
        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # occasional acceptance (simulated annealing)
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # increase softmax beta gradually
        if beta_soft < beta_max:
            beta_soft = min(beta_max, beta_soft + beta_inc)

        # symmetrize periodically
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # random perturbation if stuck
        if no_improve > 2000:
            # random small perturbation: move mass from one index to another
            i = rng.integers(0, n)

                tmp[i] -= delta
                tmp[j] += delta

                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc

        # early exit
        if best_sc < target:
            break

    # return best vector
    return [float(x) for x in best_vec]

import time, numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a deterministic non‑negative step‑function (as a list of floats)
    that achieves a low AC1 upper bound.
    The construction is a linearly increasing sequence whose sum is 1.
    """
    # time budget handling – the computation is trivial, but we respect the API
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # Determine length: use the provided parent if present, otherwise a default.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)          # parent length
    else:
        n = 2000                             # reasonable fallback size

    # Linear ramp: a_i = 2*(i+1) / (n*(n+1))  (sum equals 1)
    idx = np.arange(n, dtype=np.float64)     # 0 … n‑1
    a = 2.0 * (idx + 1.0) / (n * (n + 1.0))

    # Normalise precisely (guard against very tiny rounding error)
    a /= a.sum()

    # Guarantee we return before the deadline
    if time.monotonic() >= deadline:
        return [float(x) for x in a]

    return [float(x) for x in a]
