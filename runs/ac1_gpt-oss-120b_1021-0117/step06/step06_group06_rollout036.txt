import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast convolution max (FFT, assumes sum(a)=1)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int):
    """Return (max(conv), full convolution array, FFT of a)."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv_full.max(), conv_full, a_fft


# ------------------------------------------------------------
# Core optimizer (Adam + simplex projection)
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # small safety margin

    # ---------- initialise length ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = a0.size
    else:
        n = 2500
        a0 = np.exp(-np.arange(n, dtype=np.float64) / 80.0)

    # ---------- generate a few cheap candidates ----------
    L = 1 << ((2 * n - 1).bit_length())
    candidates = []

    # 1) provided sequence (if any)
    if "height_sequence_1" in globals():
        candidates.append(a0.copy())

    # 2) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) pure exponential decays
    for rate in (0.001, 0.003, 0.005, 0.01, 0.02, 0.05):
        cand = np.exp(-rate * np.arange(n, dtype=np.float64))
        candidates.append(cand)

    # 4) symmetric exponential (distance from centre)
    centre = (n - 1) / 2.0
    dists = np.abs(np.arange(n, dtype=np.float64) - centre)
    for rate in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand = np.exp(-rate * dists)
        candidates.append(cand)

    # 5) sine shape (positive half‑wave)
    cand = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    candidates.append(cand)

    # 6) power‑law tails
    for p in (0.5, 0.8, 1.0, 1.2, 1.5):
        cand = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        candidates.append(cand)

    # ---------- evaluate candidates and keep the best ----------
    best_vec = None
    best_score = float('inf')
    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s                     # simplex
        max_val, _, _ = _conv_max(cand, L)
        score = 2.0 * n * max_val           # same as evaluate_sequence
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
        # time guard
        if time.monotonic() >= deadline:
            break

    # If for any reason we have no candidate, fall back to uniform
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * np.max(np.convolve(best_vec, best_vec, mode='full'))

    # ---------- Adam optimisation ----------
    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.15               # base learning rate
    eps_factor = 0.2        # temperature multiplier
    eps_decay = 0.995
    it = 0
    stall = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # ----- convolution & score -----
        max_val, conv_full, a_fft = _conv_max(a, L)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            stall = 0
            if best_score < target:
                break
        else:
            stall += 1

        # ----- temperature & weight vector -----
        eps = max(max_val * eps_factor, 1e-16)
        if eps_factor < 5e-5:
            mask = (conv_full == max_val)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w = w / w_sum
        else:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w = w / w_sum

        # ----- gradient (subgradient of smoothed max) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = _project_simplex(a_new)

        a = a_new

        # ----- schedule update -----
        eps_factor = max(eps_factor * eps_decay, 5e-5)

        # ----- occasional symmetrisation -----
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            a = sym

        # ----- random perturbation when stuck -----
        if stall > 3500:
            a = best_vec + 0.01 * rng.random(n)
            a = _project_simplex(a)
            m.fill(0)
            v.fill(0)
            stall = 0

        # ----- safety check -----
        if time.monotonic() >= deadline:
            break

    # ---------- final cheap local refinement ----------
    while time.monotonic() < deadline - 0.02:
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.random() * 0.05
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        max_val, _, _ = _conv_max(cand, L)
        cand_score = 2.0 * n * max_val
        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()
            a = cand
        # optional early break if excellent
        if best_score < target:
            break

    return [float(x) for x in best_vec]
