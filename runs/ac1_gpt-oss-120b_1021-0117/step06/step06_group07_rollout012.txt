def _grad_max(a, conv):
    n = a.size
    max_val = conv.max()
    idx_max = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in idx_max:
        i_start = max(0, k-(n-1))
        i_end   = min(n-1, k)
        i_arr = np.arange(i_start, i_end+1)
        g[i_arr] += a[k - i_arr]
        if k % 2 == 0:
            i_mid = k // 2
            if i_start <= i_mid <= i_end:
                g[i_mid] += a[i_mid]
    # Scale
    if idx_max.size > 0:
        g /= float(idx_max.size)
    return g

def _build_conv_matrix(f):
    n = f.shape[0]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)
    T = sp.diags(data, offsets, shape=(2*n-1, n), format='csc')
    return T

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # leave headroom

    cur_sc = best_sc

    no_improve = 0

import time
import math
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries summing to Z."""
    if Z < 0:
        raise ValueError("Simplex target sum Z must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Maximum of a self‑convolution using FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution (exact on the active set)
# ------------------------------------------------------------
def _grad_max(a: np.ndarray, conv_full: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max_k (a*a)[k] (averaged over all active k)."""
    n = a.size
    max_val = conv_full.max()
    idx_max = np.flatnonzero(conv_full == max_val)
    g = np.zeros_like(a)
    for k in idx_max:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_arr = np.arange(i_start, i_end + 1, dtype=int)
        g[i_arr] += a[k - i_arr]
        if k % 2 == 0:
            i_mid = k // 2
            if i_start <= i_mid <= i_end:
                g[i_mid] += a[i_mid]          # extra contribution for diagonal term
    if idx_max.size > 0:
        g /= float(idx_max.size)
    return g

# ------------------------------------------------------------
# Sparse Toeplitz‑like convolution matrix (used only for LP)
# ------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    n = f.shape[0]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                     # 0 … n‑1
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return T

    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The algorithm combines exact sub‑gradient descent, occasional
    symmetrisation and sparse LP refinement while respecting the time budget.
    """

    best_sc = 2.0 * n * conv_full.max()

                            no_improve = 0

    # Return best candidate found (as plain Python list)
    # --------------------------------------------------------
