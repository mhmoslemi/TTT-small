    deadline = start + max(0.0, float(budget_s) - 0.2)

    cur = (cur + cur[::-1]) * 0.5

    cur = _project_simplex(cur) # ensure sum 1

    init_eps_frac = 0.2

    beta2 = 0.999

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    it = 0
    # thresholds for restarts
    no_improve = 0
    improve_target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        # Initialize eps if needed
        if eps is None:
            eps = max_val * init_eps_frac + 1e-12
        else:
            eps = max(eps * eps_decay, 1e-12)

        cand = _project_simplex(cand)

        cand_score = _score(cand, L)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum x = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute the AC1 bound for a vector whose sum is (approximately) 1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initialise a feasible point on the simplex -----
    if "height_sequence_1" in globals():
