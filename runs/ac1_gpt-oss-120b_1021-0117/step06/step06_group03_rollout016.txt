def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + float(budget_s) - 0.2
    # initialize
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t/100.0)
        a0 /= a0.sum()
    cur = a0
    n = cur.size
    L = 1 << ((2*n - 1).bit_length())
    # Helper functions
    def _score_vec(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    def _project_simplex(v):
        # from previous code
        n_ = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n_+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w
    cur_score = _score_vec(cur)
    best_vec = cur.copy()
    best_score = cur_score
    # Adam parameters
    m = np.zeros_like(cur)
    v_sq = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.5   # initial learning rate scale
    t_iter = 0
    # softmax temperature
    temp = None
    temp_decay = 0.998
    # stagnation tracking
    no_improve = 0
    sym_interval = 100
    # main loop
    while time.monotonic() < deadline:
        t_iter += 1
        # FFT of current
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # temperature
        if temp is None:
            temp = max_val * 0.15 + 1e-12
        else:
            temp = max(temp * temp_decay, 1e-12)
        # softmax weights
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v_sq = beta2 * v_sq + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v_sq / (1 - beta2**t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # propose new
        cand = cur - step
        cand = _project_simplex(cand)
        cand_score = _score_vec(cand)
        # accept if better
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            # optional uphill acceptance with small prob
            prob = math.exp(-(cand_score - cur_score) * 0.1)  # scaled
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score
                no_improve = 0
            else:
                no_improve += 1
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()
        # symmetrize periodically
        if t_iter % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_vec(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()
        # random mass transfer when stagnating
        if no_improve > 500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_vec(tmp)
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_vec = tmp.copy()
        # restart from best with small noise if hopeless
        if no_improve > 2000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v_sq.fill(0.0)
            t_iter = 0
            temp = None
            no_improve = 0
        # early exit
        if best_score < 1.50274349:
            break
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a sequence whose sum ≈ 1:
#   2 * n * max(convolution)
# ------------------------------------------------------------
def _score_vec(v: np.ndarray) -> float:
    """Return 2*n*max(a*a) assuming sum(v)≈1."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + float(budget_s) - 0.2  # safety margin

    # --------------------------------------------------------
    # Initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
    else:  # simple fallback
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    # Symmetrise the start (helps flatten the autocorrelation)
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = _project_simplex(a0)

    cur = a0
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    cur_score = _score_vec(cur)
    best_vec = cur.copy()
    best_score = cur_score

    # --------------------------------------------------------
    # Adam‑style optimisation parameters
    # --------------------------------------------------------
    m = np.zeros_like(cur)          # first‑moment estimate
    v_sq = np.zeros_like(cur)       # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.6                   # base learning‑rate scale
    beta1_pow = 1.0
    beta2_pow = 1.0

    # temperature for soft‑max smoothing of the max
    temp = None
    temp_decay = 0.998

    # control variables
    it = 0
    no_improve = 0
    sym_interval = 200
    target_score = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- temperature update (smooth max) -----
        if temp is None:
            temp = max_val * 0.15 + 1e-12
        else:
            temp = max(temp * temp_decay, 1e-12)

        # ----- soft‑max weights -----
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (FFT‑based) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v_sq = beta2 * v_sq + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v_sq / (1.0 - beta2_pow)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # optional clipping of step length (prevents huge jumps)
        step_norm = np.linalg.norm(step)
        max_step = 0.12
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        # ----- propose new candidate -----
        cand = cur - step
        cand = _project_simplex(cand)
        cand_score = _score_vec(cand)

        # ----- acceptance test -----
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            # occasional uphill acceptance (simulated‑annealing style)
            prob = math.exp(-(cand_score - cur_score) * 0.08)
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score
                no_improve = 0
            else:
                no_improve += 1

        # ----- keep the overall best seen -----
        if cur_score < best_score:

            best_vec = cur.copy()

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:

            sym = _project_simplex(sym)
            sym_score = _score_vec(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()

        # ----- random mass transfer when stuck -----
        if no_improve > 600:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_vec(tmp)
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_vec = tmp.copy()
                else:
                    no_improve = 0  # give it a chance to move on

        # ----- restart from the best with tiny noise if hopeless -----
        if no_improve > 2500:
            cur = best_vec + 0.008 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v_sq.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            temp = None
            no_improve = 0

        # ----- early exit if target achieved -----
        if best_score < target_score:
            break

        # safety check for deadline
        if time.monotonic() >= deadline:
            break

    # Return the best sequence found as plain Python floats
    return [float(x) for x in best_vec]
