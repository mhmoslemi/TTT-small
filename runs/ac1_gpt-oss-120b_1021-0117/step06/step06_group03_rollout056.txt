import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------
    # initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:  # simple fallback
        n_fallback = 2000
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())      # length for zero‑padded FFT

    # --------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # --------------------------------------------------------
    # temperature schedule for smoothed max
    # --------------------------------------------------------
    eps_temp = None
    eps_decay = 0.9995
    min_eps = 1e-12

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    best_a = a.copy()
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = float(2.0 * n * conv_full.max())
    stagnation = 0
    it = 0
    target_sc = 1.50274349

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = float(2.0 * n * max_val)

        # ----- keep the best seen -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # ----- temperature update -----
        if eps_temp is None:
            eps_temp = max(max_val * 0.15, 1e-12)
        else:
            eps_temp = max(eps_temp * eps_decay, min_eps)

        # ----- softmax weights for a smoothed max -----
        w = np.exp((conv_full - max_val) / eps_temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (FFT‑based) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)                # slice for gradient
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- take step and re‑project onto simplex -----
        a = a - step
        a = _project_simplex(a)

        # ----- random perturbation when stuck -----
        if stagnation > 2000:
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0

            if sym_sc < cur_sc:
                a = sym

        # ----- early exit if target reached -----
        if best_sc < target_sc:

    # return the best sequence as plain Python floats
    return [float(x) for x in best_a]
