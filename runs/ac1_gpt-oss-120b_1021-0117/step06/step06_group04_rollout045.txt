def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    """Return (max value, index of max, full convolution a*a)."""

    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            no_improve = 0

        # -------------------------------------------------------------
        # early exit if target reached
        # -------------------------------------------------------------
        if best_sc < target_score:
            break

    # Return the best candidate as a plain Python list
    return [float(x) for x in best_vec]
