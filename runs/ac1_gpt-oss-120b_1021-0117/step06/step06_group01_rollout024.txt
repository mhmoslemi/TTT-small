import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector normalized to sum = 1
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        cand = (1 - t) * best_vec + t * rand
        _try_candidate(cand)

    # --------------------------------------------------------
    # Gradient‑based refinement (smooth‑max surrogate)
    # --------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    m = np.zeros_like(cur)            # Adam first moment
    v = np.zeros_like(cur)            # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.4                         # base learning rate
    eps_smooth = None
    eps_decay = 0.996
    no_improve = 0
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ---- convolution & smooth‑max weights ----
        L = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps_smooth is None:
            eps_smooth = max(max_val * 0.3, 1e-9)
        else:
            eps_smooth = max(eps_smooth * eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps_smooth)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ---- gradient of smoothed max (via convolution) ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]        # shape (n,)

        # ---- Adam update ----
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / (math.sqrt(it) + 1e-12)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ---- accept if better, otherwise keep current ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # ---- keep global best ----
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
            no_improve = 0

        # ----------------------------------------------------
        # periodic symmetrisation (helps a lot)
        # ----------------------------------------------------
        if it % 40 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()

        # ----------------------------------------------------
        # random mass‑transfer when stuck
        # ----------------------------------------------------
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        # ----------------------------------------------------
        # occasional random convex combination with fresh exponential
        # ----------------------------------------------------
        if it % 120 == 0:
            alpha = 10 ** rng.uniform(-5, -0.2)
            rand = np.exp(-alpha * np.arange(n, dtype=np.float64))
            t = rng.random()
            cand = (1 - t) * best_vec + t * rand
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # ----------------------------------------------------
        # restart from best with small noise if hopeless
        # ----------------------------------------------------
        if no_improve > 6000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps_smooth = None
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # Return plain Python list
    return [float(x) for x in best_vec]
