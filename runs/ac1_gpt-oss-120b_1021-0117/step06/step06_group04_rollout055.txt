import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return (max value, index of max, full convolution a*a up to length 2n‑1)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # -----------------------------------------------------------------
    # initialise a feasible point on the simplex
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / a0.size)
    else:
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback)

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    best_vec = a0.copy()
    best_max, _, _ = _conv_max_and_index(best_vec, L)
    best_sc = 2.0 * n * best_max / (best_vec.sum() ** 2)

    # -----------------------------------------------------------------
    # helper to possibly improve the current best
    # -----------------------------------------------------------------
    def try_candidate(cand):
        nonlocal best_sc, best_vec
        if time.monotonic() >= deadline:
            return False, np.inf
        max_val, _, _ = _conv_max_and_index(cand, L)
        sc = 2.0 * n * max_val / (cand.sum() ** 2)
        improved = sc < best_sc
        if improved:
            best_sc = sc
            best_vec = cand.copy()
        return improved, sc

    # -----------------------------------------------------------------
    # 1. systematic exploration of low‑dimensional families
    # -----------------------------------------------------------------
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)          # distance to the nearest edge

    # power‑law families (symmetric)
    for p in np.linspace(0.0, 6.0, 61):
        a = (dist + 1.0) ** (-p)
        a = a / a.sum()
        try_candidate(a)
        if time.monotonic() >= deadline:
            break

    # exponential families (symmetric)
    for lam in np.linspace(0.0, 5.0, 51):
        a = np.exp(-lam * dist)
        a = a / a.sum()
        try_candidate(a)
        if time.monotonic() >= deadline:
            break

    # asymmetric power‑law
    for p in np.linspace(0.0, 6.0, 61):
        a = (idx + 1.0) ** (-p)
        a = a / a.sum()
        try_candidate(a)
        if time.monotonic() >= deadline:
            break

    # asymmetric exponential
    for lam in np.linspace(0.0, 5.0, 51):
        a = np.exp(-lam * idx)
        a = a / a.sum()
        try_candidate(a)
        if time.monotonic() >= deadline:
            break

    # piecewise constant blocks at the ends

        a[-k:] = 0.5 / k
        a = a / a.sum()
        try_candidate(a)

            break
        a = rng.random(n)

        try_candidate(a)

    # -----------------------------------------------------------------
    # 2. Local refinement starting from the current best
    # -----------------------------------------------------------------
    cur = best_vec.copy()

                cur = sym
                cur_max = sym_max
