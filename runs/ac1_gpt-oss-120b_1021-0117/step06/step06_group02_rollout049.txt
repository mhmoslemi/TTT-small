import time

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    sym_interval = 300
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- update smoothing epsilon -----
        if eps is None:
            eps = max(max_val * 0.5, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # ----- soft‑max weights -----
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- tentative move and projection -----
        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- accept if improvement, else occasional uphill move -----
        if cand_sc < cur_sc:
            improved = True
        else:
            # simple temperature schedule
            T = max(0.001, (cur_sc - target) * 0.02)
            prob = math.exp(-(cand_sc - cur_sc) / T) if T > 0 else 0.0
            improved = rng.random() < prob

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- aggressive random mass‑transfer when stuck -----
        if no_improve > 1200:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.7
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    # very occasional acceptance of a non‑improving jump
                    if rng.random() < 0.02:
                        cur = tmp
                        cur_sc = tmp_sc
                        no_improve = 0

        # ----- restart from the best with small noise if hopeless -----
        if no_improve > 2500:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- periodic deadline check -----
        if it % 400 == 0 and time.monotonic() >= deadline:
            break

    # Return as plain Python list
    return [float(x) for x in best_vec]
```

This implementation respects all constraints, uses only linear‑time storage, runs a projected‑Adam optimiser with soft‑max smoothing of the max‑convolution, includes symmetrisation and stochastic mass‑transfer moves, and safeguards against exceeding the allotted runtime.
