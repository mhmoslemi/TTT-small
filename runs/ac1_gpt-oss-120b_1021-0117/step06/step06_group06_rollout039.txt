import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 score (sum assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # small safety margin

    # ---------- initial vector ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:                                   # simple fallback
        n_fallback = 2500
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 120.0)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s = 1.0
    a0 = a0 / s                         # normalize to sum = 1

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    cur = a0.copy()
    cur_sc = _score(cur, L)

    best = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------
    temp = cur_sc * 0.12          # initial temperature (fraction of score)
    temp_min = 1e-6
    decay = 0.99997               # exponential decay per iteration
    max_no_imp = 1500             # restart after many non‑improvements
    no_imp = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Main annealing loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # propose a small mass transfer between two positions
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.25   # up to 25 % of mass at i
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep non‑negativity (clip tiny negatives caused by rounding)
        if cand[i] < 0.0:
            continue

        cand_sc = _score(cand, L)

        # acceptance test
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_sc = cand_sc
            no_imp = 0
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
                # early exit if we beat the target
                if best_sc < target:
                    break
        else:
            no_imp += 1

        # temperature decay
        temp = max(temp_min, temp * decay)

        # occasional symmetrisation (helps for many kernels)
        if it % 2500 == 0:

            cur_sc = _score(cur, L)
            if cur_sc < best_sc:

                best_sc = cur_sc

            cur = best + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_sc = _score(cur, L)
            no_imp = 0
            temp = cur_sc * 0.12

    # ------------------------------------------------------------
    # Final tiny local refinement (few gradient steps)
    # ------------------------------------------------------------
    # Use a simple sub‑gradient descent for a brief period
    while time.monotonic() < deadline - 0.02:
        # compute sub‑gradient at the current max‑lag
        af = np.fft.rfft(best, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        k = int(conv_full.argmax())
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[k] = 1.0
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        grad = 2.0 * grad_full[(2 * n - 2) - np.arange(n)]

        cand = best - step * grad

        if cand_sc < best_sc:
            best = cand
            best_sc = cand_sc
        it += 1
        if best_sc < target:
            break

    # return as plain Python list
    return [float(x) for x in best]
