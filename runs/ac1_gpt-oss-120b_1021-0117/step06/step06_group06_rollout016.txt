import time, math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list) with a low AC1 bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---------- initial sequence ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # reasonable fallback: exponential decay
        n_fallback = 2500
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s = 1.0
    a0 = a0 / s                     # normalize to sum = 1
    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    # keep best overall solution
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.25          # base learning rate
    temp_decay = 0.995

    # outer loop – multiple restarts if time permits
    while time.monotonic() < deadline:
        # start from a (possibly shuffled) copy of the initial vector
        cur = a0.copy()
        rng.shuffle(cur)
        cur = cur / cur.sum()
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        it = 0
        no_improve = 0
        temp = None

        # inner optimisation loop
        while time.monotonic() < deadline:
            it += 1

            # ----- convolution of cur with itself -----
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            max_val = conv_full.max()
            cur_sc = 2.0 * n * max_val   # same as _score(cur)

            # ----- temperature for smoothed max -----
            if temp is None:
                temp = max_val * 0.15 + 1e-12
            else:
                temp = max(temp * temp_decay, max_val * 1e-12)

            # ----- weight vector (softmax or hard mask) -----
            if temp < max_val * 1e-6:
                mask = (conv_full == max_val)
                w = mask.astype(np.float64)
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(conv_full, 1.0 / conv_full.size)
                else:
                    w /= w_sum
            else:
                w = np.exp((conv_full - max_val) / temp)
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(conv_full, 1.0 / conv_full.size)
                else:
                    w /= w_sum

            # ----- gradient of smoothed max (matrix‑free) -----
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            cand = cur - step
            cand = _project_simplex(cand)
            cand_sc = _score(cand)

            improved = cand_sc < cur_sc

                cur_sc = cand_sc

                    best_vec = cand.copy()
            else:

            if it % 300 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:

                    cur_sc = sym_sc
                    no_improve = 0

            if no_improve > 800:

                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)

                        cur_sc = tmp_sc

            if best_sc < 1.5028:
                return [float(x) for x in best_vec]

    # fallback: return the best vector found (or the initial one)
    if best_vec is None:
        best_vec = a0
    return [float(x) for x in best_vec]
