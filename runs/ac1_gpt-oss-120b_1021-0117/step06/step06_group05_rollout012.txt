import time

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (works for any positive sum)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact AC1 bound used by the evaluator (sum is not assumed to be 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # leave margin

    # --------------------------------------------------------
    # Initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s                                   # sum = 1
    else:                                               # fallback
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())                # length for FFTs

    # --------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)
    beta1_pow, beta2_pow = 1.0, 1.0
    step_counter = 0

    cur = a0.copy()
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    eps_factor = 0.02          # initial smoothing factor (as fraction of max)
    eps_decay = 0.995
    base_lr = 0.7
    sym_interval = 400
    target_score = 1.50274349
    rand_move_prob = 0.01

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        step_counter += 1

        # ---- convolution and max ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ---- smoothing temperature (soft‑max) ----
        if step_counter == 1:
            eps = max(max_val * eps_factor, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed max (via FFT) ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update ----
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = base_lr / (math.sqrt(step_counter) + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ---- accept improvement or occasional uphill move ----
        cur_sc = _score(cur)
        if cand_sc < cur_sc or rng.random() < 0.001:
            cur = cand
            cur_sc = cand_sc

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

        # ---- occasional random mass transfer ----
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc

        # ---- symmetrisation every few iterations ----
        if step_counter % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc

        # ---- early exit if target reached ----
        if best_sc < target_score:
            break

    # --------------------------------------------------------
    # Light post‑processing: try a few Gaussian smoothings
    # --------------------------------------------------------
    def _smooth_and_project(x, sigma):
        radius = int(3 * sigma)
        t = np.arange(-radius, radius + 1, dtype=np.float64)
        kern = np.exp(-t * t / (2 * sigma * sigma))
        kern /= kern.sum()
        y = np.convolve(x, kern, mode='same')
        y = np.maximum(y, 0.0)
        y = _project_simplex(y)
        return y

    for sigma in (0.7, 1.2, 2.0):
        sm = _smooth_and_project(best_vec, sigma)
        sc = _score(sm)
        if sc < best_sc:
            best_sc = sc
            best_vec = sm

    # --------------------------------------------------------
    # Return the best vector found as a plain Python list
    # --------------------------------------------------------
    return [float(v) for v in best_vec]
