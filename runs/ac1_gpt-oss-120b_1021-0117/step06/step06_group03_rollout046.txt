def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            n = len(a0) # fallback
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        n = 2000
        t = np.arange(n, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()
    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    
    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    momentum = np.zeros_like(cur)
    eps = None
    beta = 0.95
    base_alpha = 1.0
    eps_decay = 0.995
    max_backtrack = 5
    no_improve = 0
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # FFT and conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # temperature
        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)
        # softmax weights
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        # step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m
        # backtracking line search
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_backtrack):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            # enforce symmetry to reduce variance maybe afterwards
            # but for evaluation just compute.
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5
        # if not improved, maybe accept uphill with small probability
        if not improved:
            # try one more candidate
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            # Simulated annealing probability based on difference and iteration
            prob = math.exp(-(cand_sc - cur_sc) / (1.0 + it * 0.001))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Symmetrize every iteration (or every few)
        cur = (cur + cur[::-1]) * 0.5
        cur = _project_simplex(cur)
        cur_sc = _score(cur)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # Random mass transfer when stagnating
        if no_improve > 800:
            # perturb best_vec a bit
            perturb = best_vec + 0.01 * rng.standard_normal(n)
            perturb = np.maximum(perturb, 0.0)
            perturb = _project_simplex(perturb)
            pert_sc = _score(perturb)
            if pert_sc < best_sc:
                best_sc = pert_sc
                best_vec = perturb.copy()
                cur = perturb.copy()
                cur_sc = pert_sc
                momentum = np.zeros_like(cur)
                eps = None
                no_improve = 0
            else:
                # random swap mass
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.5
                    j = rng.integers(0, n)
                    if j == i: 
                        j = (j + 1) % n
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur = tmp
                        cur_sc = tmp_sc
                        if tmp_sc < best_sc:
                            best_sc = tmp_sc
                            best_vec = tmp.copy()
                        no_improve = 0
                    else:
                        # no_improve stays
                        pass

        # Early stop
        if best_sc < target:
            break

        # Check time
        if time.monotonic() >= deadline:
            break

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of 2*n*max(convolution) assuming sum == 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
