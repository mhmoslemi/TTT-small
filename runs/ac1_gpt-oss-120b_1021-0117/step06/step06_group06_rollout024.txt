def gradient(a, eps, L):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv.max()
    # compute weight mask
    if eps is None:
        w = (conv == max_val).astype(np.float64)
        sum_w = w.sum()
        if sum_w == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w / sum_w
    else:
        w = np.exp((conv - max_val) / eps)
        sum_w = w.sum()
        if sum_w == 0.0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w / sum_w
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val, conv

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Gradient (or sub‑gradient) of the soft‑maxed autocorrelation
# ------------------------------------------------------------
def _gradient_and_max(a: np.ndarray, eps: float | None, L: int):
    """
    Returns (grad, max_val) where grad is the (sub‑)gradient of the
    smoothed maximum autocorrelation and max_val = max_k (a*a)[k].
    eps == None -> hard max (sub‑gradient).
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    # full linear convolution a * a
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()

    # weights for the (soft) max
    if eps is None:
        w = (conv == max_val).astype(np.float64)
        s = w.sum()
        if s == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= s
    else:
        w = np.exp((conv - max_val) / eps)
        s = w.sum()
        if s == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= s

    # gradient = 2 * (a * w_rev) restricted to the n relevant entries
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val


# ------------------------------------------------------------
# Simple initial shapes
# ------------------------------------------------------------
def _init_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _init_exponential(n: int, tau: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / tau)
    return a / a.sum()


def _init_random_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    a = rng.gamma(alpha, 1.0, size=n)
    s = a.sum()
    if s == 0.0:
        a.fill(1.0 / n)
    else:
        a /= s
    return a
