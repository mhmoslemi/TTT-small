import time, math, numpy as np, cvxpy as cp, sys
# but we may not need cvxpy, but may keep for occasional LP.

def _project_simplex(v, Z=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def _max_conv(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af*af, n=L)[:2*n-1]
    return float(conv.max()), conv

def _gradient_exact(a, conv, max_val):
    n = a.shape[0]
    mask = conv >= max_val - 1e-12
    idxs = np.nonzero(mask)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # i indices
        i = np.arange(i_start, i_end+1, dtype=int)
        g[i] += 2.0 * a[k - i]
    # optionally divide by number of mask positions? Not needed for subgradient.
    return g

import time, math
import numpy as np
import scipy.sparse as sp

# ------------------------------------------------------------
# Simplex projection (Euclidean, sum = Z, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Max of linear convolution a*a using FFT (full length)
# ------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> tuple[float, np.ndarray]:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv


# ------------------------------------------------------------
# Exact sub‑gradient of max‑conv (sum over all maximal shifts)
# ------------------------------------------------------------
def _grad_exact(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    n = a.shape[0]
    mask = conv >= max_val - 1e-12
    idxs = np.nonzero(mask)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    return g
