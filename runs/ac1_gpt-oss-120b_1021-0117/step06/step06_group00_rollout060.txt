import time, math, numpy as np
from scipy.optimize import linprog  # optional, may fall back if not used

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to guard against tiny numerical drift
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / n)
    return w / s


def _score(a, L=None):
    """Score = 2 * n * max(conv(a,a)) for a that already sums to 1."""
    n = a.size
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    deadline = start + max(0.0, float(budget_s) - 0.3)   # tiny safety margin

    # -------------------------- initialization -------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # Adam optimizer state
    m = np.zeros_like(cur)          # first moment
    v = np.zeros_like(cur)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5                       # base learning rate
    it = 0
    no_improve = 0
    target = 1.50274349

    # -------------------------- main loop ----------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution of cur with itself (for score) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights over convolution entries -----
        # shift so that max becomes 0, avoiding overflow
        shift = conv - max_val
        beta_temp = 1.0 + 9.0 * min(1.0, it / 10000.0)   # smooth → sharp
        w = np.exp(beta_temp * shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of weighted max‑convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / (np.sqrt(v_hat) + eps)
        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _score(cand, L)

        # ----- accept if better, otherwise keep current -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # occasional simulated‑annealing acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (lr0 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation (helps stability) -----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ----- random perturbation when stuck -----
        if no_improve > 2500:
            # Gaussian jitter on a random subset
            mask = rng.random(n) < 0.02  # ~2% of entries
            jitter = rng.normal(scale=0.01, size=n) * mask
            pert = cur + jitter
            pert = _project_simplex(np.maximum(pert, 0.0))
            pert_sc = _score(pert, L)
            if pert_sc < cur_sc:
                cur, cur_sc = pert, pert_sc
                no_improve = 0
                if pert_sc < best_sc:
                    best_vec, best_sc = pert.copy(), pert_sc

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
