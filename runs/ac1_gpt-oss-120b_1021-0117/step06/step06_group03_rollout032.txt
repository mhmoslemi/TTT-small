import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a: np.ndarray) -> float:
    """2*n*max(conv) for a non‑negative vector a (sum≈1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ---------- initialise ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)
    else:
        n_fallback = 2048
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 200.0)
        a0 = a0 / a0.sum()

    n = a0.size
    cur = a0
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---------- optimisation state ----------
    eps = None
    eps_decay = 0.996
    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t_iter = 0
    no_improve = 0
    target_score = 1.50274349
    max_norm = 0.5           # max L2‑norm of a step
    sym_interval = 250

    L = 1 << ((2 * n - 1).bit_length())   # FFT length (constant)

    while time.monotonic() < deadline:
        t_iter += 1

        # ----- convolution and smoothed max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = 0.07 * m_hat / (np.sqrt(v_hat) + adam_eps)

        # clip step norm
        norm_step = np.linalg.norm(step)
        if norm_step > max_norm:
            step = step * (max_norm / norm_step)

        # ----- backtracking line search -----

        cand = None

            if cand_sc_try < cur_sc:

                improved = True
                break

            cand_try = _project_simplex(cand_try)
            cand_sc_try = _score(cand_try)
            prob = math.exp(-(cand_sc_try - cur_sc) / (factor + 1e-12))

                improved = True

                no_improve = 0
            else:
                no_improve += 1
        else:

                no_improve = 0

                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)

                if tmp_sc < cur_sc:

                    no_improve = 0
                else:
                    no_improve = 0

        # ----- full restart after long stagnation -----
        if no_improve > 3000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            eps = None
            no_improve = 0

        # ----- early exit -----
        if best_sc < target_score:
            break

    # Return best found sequence as plain Python floats
    return [float(x) for x in best_vec]
