        else:
            a = a / s
    else:
        n = 2000
        a = np.exp(-np.arange(n)/120.0)
        a /= a.sum()
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    best_a = a.copy()
    best_sc = _score(best_a)
    cur_sc = best_sc

    # hyperparams
    base_eta = 0.8
    eps = None
    eps_decay = 0.9995
    eta_decay = 0.9995
    sym_interval = 200
    no_improve = 0

    it = 0
    while time.monotonic() < deadline and it < 20000:
        it += 1
        # compute conv and gradient
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        if eps is None:
            eps = max_val * 0.5
        else:
            eps = max(eps * eps_decay, max_val * 1e-12)  # ensure >0

        # softmax weights
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via convolution with w_rev
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[grad_idx]

        # exponentiated gradient update
        eta = base_eta / math.sqrt(it)
        # ensure e^( - eta*grad ) limited to avoid overflow
        # Clip grad to some range
        grad_clipped = np.clip(grad, -1e3, 1e3)

        a = _project_simplex(a)

# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
