import time, math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Return the projection of v onto the simplex of sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Approximate gradient of max‑convolution using the top‑k entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    return g / max(1, top_k)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initial vector (use the provided parent if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = rng.random(2000, dtype=np.float64)
    a0 = np.clip(a0, 0.0, 1000.0)
    if a0.sum() < 1e-12:
        a0[:] = 1.0
    a = a0 / a0.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    cur_fft = np.fft.rfft(a, n=L)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
    best_sc = 2.0 * n * cur_conv.max()
    best_vec = a.copy()

    # optimisation parameters
    step0 = 0.8
    beta = 0.96
    momentum = np.zeros_like(a)
    top_k = 20
    sym_interval = 80
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & bound -----
        cur_fft = np.fft.rfft(a, n=L)
        cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        cur_sc = 2.0 * n * cur_conv.max()

        # ----- keep the best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit -----
        if best_sc < target:
            break

        # ----- gradient step (projected SGD) -----
        g = _grad_topk(a, cur_conv, top_k)
        momentum = beta * momentum + (1.0 - beta) * g
        alpha = step0 / math.sqrt(it)
        a_tmp = a - alpha * momentum
        a_tmp = _project_simplex(a_tmp)   # stay on simplex (sum = 1)

        # ----- evaluate trial -----
        tmp_fft = np.fft.rfft(a_tmp, n=L)
        tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * tmp_conv.max()

        # ----- accept if improvement, else occasional uphill move -----
        if cand_sc < cur_sc:
            a, cur_sc = a_tmp, cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = a_tmp, cand_sc

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_fft = np.fft.rfft(a_sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- occasional random perturbation when stuck -----
        if no_improve > 4000:
            # small random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-12:
                delta = a[i] * rng.random() * 0.3
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                momentum = np.zeros_like(a)
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
