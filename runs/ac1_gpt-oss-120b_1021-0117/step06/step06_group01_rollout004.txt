import time

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best = tmp.copy()

        # ---------- restart with noise ----------
        if no_improve > restart_interval:
            cur = best + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps_soft = None
            beta1_pow = beta2_pow = 1.0
            no_improve = 0
            it = 0  # reset learning‑rate schedule

        # ---------- early stop ----------
        if best_sc < target:
            break

    # -----------------------------------------------------------------
    # final random perturbation sweep (if time permits)
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best + 0.01 * rng.standard_normal(n)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
        # limit loop quickly
        if time.monotonic() > deadline - 0.01:
            break

    return [float(x) for x in best]
