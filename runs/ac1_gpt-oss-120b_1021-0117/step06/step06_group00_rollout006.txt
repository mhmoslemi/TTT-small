import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score assuming sum(a)==1 (the denominator becomes 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # ----- initialization -------------------------------------------------

        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random vector of length ~2500
        a0 = rng.random(2500, dtype=np.float64)

    best_sc = _score_vec(best_vec)

    no_improve = 0
    target = 1.50274349

    # temperature for soft‑max weighting of the max‑convolution
    temp_factor = 1e-2          # relative to current max value
    temp_decay = 0.9995         # decay per iteration
    min_temp_factor = 1e-6

    # ----- main loop ------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # forward FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # full convolution a*a
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weighting over lag indices
        temp = max_val * max(temp_factor, min_temp_factor)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient of weighted max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        step = step0 / math.sqrt(it)

        # candidate step
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_vec(cand)

        # accept if better, else occasional probabilistic acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # periodic symmetrisation (helps for many optimal patterns)
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # random local redistribution when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
            else:
                # occasional full random restart
                rnd = _project_simplex(rng.random(n))
                rnd_sc = _score_vec(rnd)
                if rnd_sc < cur_sc:
                    cur, cur_sc = rnd, rnd_sc
                    momentum.fill(0.0)

        # decay temperature factor
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # early exit if we reached the target bound
        if best_sc < target:
            break

        # keep an eye on time every few thousand iterations
        if it % 5000 == 0 and time.monotonic() >= deadline:
            break

    # Return best-found vector as plain Python floats
    return [float(x) for x in best_vec]
