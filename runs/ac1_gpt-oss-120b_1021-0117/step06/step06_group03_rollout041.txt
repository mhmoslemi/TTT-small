def evaluate_sequence(sequence):
    if not isinstance(sequence, list): return inf
    if not sequence: return inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)): return inf
        if np.isnan(x) or np.isinf(x): return inf
    # convert to floats, clamp to nonnegative, clamp at 1000
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01: return inf
    return float(2 * n * max_b / (sum_a**2))

def solve_lp_for_g(a, M, active_idx):
    # a is np array length n
    # active_idx list of indices k where conv larger.
    # Build constraint matrix A shape (len(active_idx), n)
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_idx):
        # for each i where a[i] >0 and j = k - i in [0, n-1]
        # The term (f*g)[k] = sum_i a[i] * g[k-i]
        # So coefficients for g[j] = a[i] where i = k - j.
        # Equivalent: for each j, coefficient = a[k-j] if 0 <= k-j < n else 0.
        # So we fill for j in max(0, k-(n-1)) to min(n-1, k)
        start_j = max(0, k - (n - 1))
        end_j = min(n - 1, k)
        # fill for j from start_j to end_j:
        # coeff = a[k - j]
        # we can vectorize: j_range = np.arange(start_j, end_j+1); coeffs = a[k - j_range]
        # Append to rows, cols, data
        js = np.arange(start_j, end_j + 1, dtype=int)
        coeffs = a[k - js]
        rows.extend([row] * len(js))
        cols.extend(js.tolist())
        data.extend(coeffs.tolist())
    # Build sparse matrix
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    # Objective: maximize sum(g) => minimize -sum(g)
    c = np.full(n, -1.0)
    # inequality: A @ g <= M
    b_ub = np.full(len(active_idx), M)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return uniform
        return np.full_like(a, 1.0 / a.size)
    g = res.x
    # res.x may be zero; ensure sum>0
    if g.sum() == 0:
        g = np.full_like(a, 1.0 / a.size)
    return g

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of 2*n*max(conv) for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray, n: int, L: int) -> float:
    """Return 2*n*max(a*a) assuming sum(a)≈1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the smoothed max (FFT‑based)
# ------------------------------------------------------------
def _grad(a: np.ndarray, eps: float, n: int, L: int) -> tuple:
    """
    Returns (gradient, max_val, a_fft, conv_full) where
    gradient = 2 * (a * w_rev) and w is the soft‑max weight vector.
    """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    # soft‑max weights
    w = np.exp((conv_full - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient via convolution with reversed weights
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)   # slice that aligns with a
    grad = 2.0 * grad_full[idx]
    return grad, max_val, a_fft, conv_full


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # tiny safety margin

    # --------------------------------------------------------
    # initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        # fallback: random Dirichlet‑like vector
        n_fallback = 2500
        a0 = rng.random(n_fallback)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for convolution

    cur = a0.copy()
    cur_sc = _score(cur, n, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    beta1 = 0.9                # Adam first‑moment decay
    beta2 = 0.999              # Adam second‑moment decay
    eps_adam = 1e-8
    base_lr = 0.2              # base learning‑rate for Adam
    max_step_norm = 0.5        # cap on step length
    eps_decay = 0.997
    min_eps = 1e-12

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # smoothing temperature

    # stagnation counters
    no_improve = 0

        grad, max_val, a_fft, conv_full = _grad(cur, eps_temp if eps_temp else max(conv_full.max() * 0.5, 1e-12), n, L)
