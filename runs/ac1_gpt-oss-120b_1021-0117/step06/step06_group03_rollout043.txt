    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0,float(budget_s)-0.2)

    # init a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s<=0:
            a = np.full_like(a, 1.0/a.size)
        else:
            a = a / s
    else:
        n = 2000
        a = np.exp(-np.arange(n)/100.0)
        a /= a.sum()
    n = a.size

    # maintain best
    cur = a.copy()
    cur_score = _score(cur)   # defined as 2*n*max(conv)
    best = cur.copy()
    best_score = cur_score

    # parameters
    y = np.log(np.maximum(cur, 1e-12))    # log softmax parameters
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step_lr = 0.5    # initial learning rate
    it = 0
    no_improve = 0

    # function to compute a from y
    def _softmax(y):
        maxy = np.max(y)
        expy = np.exp(y - maxy)
        return expy / expy.sum()
    
    # main loop
    while time.monotonic() < deadline:
        it += 1

        a = _softmax(y)
        # compute convolution via FFT
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit
        if best_score < 1.50274349:
            break

        # gradient for max
        w = np.zeros_like(conv)
        w[max_idx] = 1.0
        # compute gradient wrt a using FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        grad_a = 2.0 * grad_full[:n]   # shape n

        # gradient wrt y (softmax)
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        y -= step_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional random perturbations to escape stagnation
        if no_improve > 2000:
            # add small Gaussian noise to y
            y += 0.01 * rng.standard_normal(n)
            no_improve = 0

        # periodic symmetrization of a (optional)
        if it % 150 == 0:
            a_sym = (a + a[::-1]) * 0.5
            # put back into y via log
            y = np.log(np.maximum(a_sym, 1e-12))

    # final answer: best sequence as list
    return [float(x) for x in best]

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def _score(a: np.ndarray) -> float:
    """Return 2·n·max(convolution) for a vector with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _grad_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the (unsmoothed) maximum convolution term
    with respect to a, for the shift k where the maximum occurs.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        i = np.arange(lo, hi + 1)
        g[i] += a[k - i]
    if k < n:
        i2 = np.arange(k, n)
        g[i2] += a[i2 - k]
    return 2.0 * g

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    target = 1.50274349

    # ------------------------------------------------------------------
    # Determine problem size and an initial feasible vector (sum≈1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)

        else:
            a0 = a0 / s
    else:                                   # fallback simple shape
        n = 2000
        a0 = np.exp(-np.arange(n, dtype=np.float64) / 100.0)
        a0 = a0 / a0.sum()
    n = a0.shape[0]

    # ------------------------------------------------------------------
    # Simple parametrised families – quick coarse search
    # ------------------------------------------------------------------
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    idx = np.arange(1, n + 1, dtype=np.float64)

    # power law a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.0, 3.0, 31):
        a = idx ** (-p)
        a /= a.sum()
        sc = _score(a)
        if sc < best_sc:

        a_rev = a[::-1]
        sc_rev = _score(a_rev)
        if sc_rev < best_sc:

    # exponential law a_i ∝ exp(-q·i)
    for q in np.linspace(0.0, 0.02, 31):
        a = np.exp(-q * np.arange(n, dtype=np.float64))
        a /= a.sum()
        sc = _score(a)
        if sc < best_sc:

        a_rev = a[::-1]
        sc_rev = _score(a_rev)
        if sc_rev < best_sc:

    if "height_sequence_1" in globals():

            y = np.log(np.maximum(a_sym, 1e-12))

    return [float(x) for x in best_vec]
