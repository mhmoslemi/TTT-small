import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum == 1, xi >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence attempting to minimise the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.exp(-np.arange(a.size, dtype=np.float64) / 100.0)
            s = a.sum()
        a = a / s                     # normalize to sum == 1
    else:                              # fallback initialisation
        n0 = 2000
        t = np.arange(n0, dtype=np.float64)
        a = np.exp(-t / 100.0)
        a = a / a.sum()

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # length for FFTs (power of two)

    # Adam optimiser state
    m = np.zeros_like(a)
    v_adam = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.08

    # temperature for smoothed max (softmax)
    temp_decay = 0.9995
    min_temp = 1e-12

    # bookkeeping
    it = 0
    no_improve = 0
    best_vec = a.copy()
    # compute first score
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_score = cur_score

    # hyper‑parameters
    sym_interval = 250
    smooth_interval = 250
    rand_move_prob = 0.003
    restart_thresh = 3000
    target_score = 1.50274349

    # ------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- softmax weights for smoothed max -----
        if it == 1:
            temp = max(cur_max * 0.2, 1e-12)
        else:
            temp = max(temp * temp_decay, min_temp)

        w = np.exp((conv - cur_max) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # ----- occasional symmetrisation -----
        if it % sym_interval == 0:
            a_new = (a_new + a_new[::-1]) * 0.5
            a_new = _project_simplex(a_new)

        # ----- occasional smoothing -----
        if it % smooth_interval == 0:
            # simple 3‑point smoothing kernel
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            a_new = np.convolve(a_new, kernel, mode='same')
            a_new = _project_simplex(a_new)

        # ----- evaluate new candidate -----
        a_fft_new = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(a_fft_new * a_fft_new, n=L)[: 2 * n - 1]
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # ----- accept if improvement (or tiny uphill with probability) -----
        improved = False
        if new_score < cur_score:
            improved = True
        else:
            # very occasional uphill acceptance
            if rng.random() < math.exp(-(new_score - cur_score) / (temp + 1e-12)):
                improved = True

        if improved:
            a = a_new
            a_fft = a_fft_new
            conv = conv_new
            cur_max = new_max
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_vec = a_new.copy()
        else:
            no_improve += 1

        # ----- random mass‑transfer move (exploration) -----
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                # evaluate quickly using existing FFT machinery
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_score = 2.0 * n * tmp_conv.max()
                if tmp_score < cur_score:
                    a = tmp
                    a_fft = tmp_fft
                    conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_score = tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_vec = tmp.copy()

        # ----- restart if stuck for a long time -----
        if no_improve > restart_thresh:
            a = best_vec + 0.01 * rng.random(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            v_adam.fill(0.0)
            it = 0
            # recompute state for the restarted point
            a_fft = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            no_improve = 0
            continue

        # ----- early exit if target reached -----
        if best_score < target_score:
            break

    # ------------------------------------------------------------
    # Return best sequence found (as plain Python floats)
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
