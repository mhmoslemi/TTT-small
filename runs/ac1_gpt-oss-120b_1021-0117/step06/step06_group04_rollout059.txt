                    if sc_sym < best_sc:
                        best_sc, best_vec = sc_sym, sym.copy()

    # -----------------------------------------------------------------
    # fall back to a simple uniform vector if everything failed
    # -----------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # Return a plain Python list (the harness will feed it to the evaluator)
    return [float(x) for x in best_vec]
