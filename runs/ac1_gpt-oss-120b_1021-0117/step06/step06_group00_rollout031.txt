import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score = 2 * n * max(convolution(a, a)).  Assumes sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.25)  # margin for cleanup

    # ----------------------------------------------------------------------
    # Initialise a feasible probability vector
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.maximum(base, 0.0)
        base = _project_simplex(base)
        n = base.size
        init_candidates = [base]
    else:
        n = 2000
        init_candidates = []

    # uniform start of the same length
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    init_candidates.append(uniform)

    # pick the best of the few starters
    cur = min(init_candidates, key=_score).copy()
    cur_sc = _score(cur)

    best = cur.copy()
    best_sc = cur_sc

    # ----------------------------------------------------------------------
    # Adam‑style optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_alpha = 0.6          # base step magnitude
    beta_temp_start, beta_temp_end = 1.0, 12.0   # smoothing schedule
    it = 0
    no_improve = 0
    target = 1.50274349

    L = 1 << ((2 * n - 1).bit_length())

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---------- convolution and smoothed max ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # temperature (smoothness) increases slowly
        beta_temp = beta_temp_start + (beta_temp_end - beta_temp_start) * min(1.0, it / 6000.0)
        w = np.exp(beta_temp * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the smoothed max ----------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
