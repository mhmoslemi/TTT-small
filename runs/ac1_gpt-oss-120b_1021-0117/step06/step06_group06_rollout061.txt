def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    # init
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a = np.exp(-np.arange(n_fallback, dtype=np.float64) / 200.0)  # slower decay
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full_like(a, 1.0/a.size)
        s = a.size * (1.0/a.size)
    a = a / s
    n = a.size
    # precompute FFT length
    L = 1 << ((2*n - 1).bit_length())
    # function to compute score (bounded by sum=1)
    def compute_score(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af*af, n=L)[: 2*n - 1]
        return float(2.0 * n * conv.max())
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    # parameters
    temp = 0.1
    temp_decay = 0.99995
    move_factor = 0.3
    it = 0
    stagnation = 0
    # main loop
    while time.monotonic() < deadline:
        it += 1
        # propose move
        i = rng.integers(n)
        if cur[i] == 0.0:
            continue
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        delta = cur[i] * move_factor * rng.random()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # compute score
        cand_sc = compute_score(cand)
        delta_sc = cand_sc - cur_sc
        if delta_sc < 0 or rng.random() < math.exp(-delta_sc / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best = cand.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1
        # temperature decay
        temp = max(temp * temp_decay, 1e-6)
        # occasional symmetrization
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym_sc = compute_score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best = sym.copy()
                stagnation = 0
        # occasional random restart if stagnation high
        if stagnation > 10000:
            # random perturbation from best
            cur = best + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_sc = compute_score(cur)
            stagnation = 0
        # early exit if below target
        if best_sc < 1.50274349:
            break
    # final fine local search: small random swaps
    while time.monotonic() < deadline - 0.05:
        i = rng.integers(n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        delta = cur[i] * 0.05 * rng.random()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = compute_score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best = cand.copy()
            cur = cand
            cur_sc = cand_sc
    return [float(x) for x in best]

import time, math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (sum of coefficients = 1)
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a tiny safety margin
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # ---------- initialise candidate ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:                                   # fallback: exponential decay
        n_fallback = 2500
        cur = np.exp(-np.arange(n_fallback, dtype=np.float64) / 200.0)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s <= 0.0:
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur = cur / s                         # enforce sum = 1
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())      # FFT length (power of two)

    # ---------- score utilities ----------
    cur_sc = _score_vec(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---------- Adam optimiser parameters ----------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.05        # learning‑rate scale
    eps_smooth = None
    eps_decay = 0.97

    it = 0
    no_improve = 0
    target = 1.50274349

    # ---------- main optimisation loop (Adam) ----------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & max value -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- smoothing parameter for soft‑max -----
        if eps_smooth is None:
            eps_smooth = max_val * 0.15 + 1e-12
        else:
            eps_smooth = max(eps_smooth * eps_decay, max_val * 1e-12)

        # ----- weights for smoothed max (softmax) -----
        if eps_smooth < max_val * 1e-8:
            w = (conv_full == max_val).astype(np.float64)
            tot = w.sum()
            if tot > 0:
                w /= tot
        else:
            w = np.exp((conv_full - max_val) / eps_smooth)
            tot = w.sum()
            if tot > 0:
                w /= tot

        # ----- gradient of the smoothed max (matrix‑free) -----
        w_rev = w[::-1]

        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)

        step = base_alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

            m.fill(0.0)
            v.fill(0.0)
            eps_smooth = None
            no_improve = 0

        # early exit if target achieved
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Simple parametric search: truncated geometric shapes
    # ------------------------------------------------------------
    # a_i = C * r^i,  r = 1 - eps,   C = eps / (1 - r**n)
    # scan eps on a log‑scale
    eps_vals = np.geomspace(1e-5, 2e-2, num=200)
    for eps in eps_vals:
        r = 1.0 - eps
        if r <= 0.0:
            continue
        # compute r**n safely
        log_r = math.log1p(-eps)               # log(r)

        denom = 1.0 - r_n
        if denom <= 0.0:
            continue

        # generate vector
        idx_arr = np.arange(n, dtype=np.float64)
        a = coeff * np.exp(log_r * idx_arr)
        # already sums to 1 (up to rounding)
        a_sc = _score_vec(a, L)
        if a_sc < best_sc:
            best_sc = a_sc
            best_vec = a.copy()
            if best_sc < target:
                break
        # check time
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------
    # Final cheap random swaps for fine‑tuning
    # ------------------------------------------------------------
    while time.monotonic() < deadline - 0.02:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * 0.05 * rng.random()
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score_vec(cand, L)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            cur_sc = cand_sc

    # ------------------------------------------------------------
    # Return result as plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
