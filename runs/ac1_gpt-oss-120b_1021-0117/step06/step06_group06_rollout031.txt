import time, math

# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a) = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the smoothed max of the convolution.
# ------------------------------------------------------------
def _gradient(a: np.ndarray, L: int, eps: float | None):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    # full convolution
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()

    # update smoothing parameter
    if eps is None:
        eps_new = max_val * 0.15 + 1e-12
    else:
        eps_new = max(eps * 0.97, max_val * 1e-12)

    # softmax weights over convolution entries
    if eps_new < max_val * 1e-6:
        w = np.zeros_like(conv)
        w[conv == max_val] = 1.0
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum
    else:
        w = np.exp((conv - max_val) / eps_new)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

    # gradient via convolution with the reversed weight vector
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, eps_new, max_val
