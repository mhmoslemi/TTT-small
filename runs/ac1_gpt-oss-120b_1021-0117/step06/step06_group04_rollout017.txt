def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # Determine n and get initial vector(s)
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: generate a simple vector of size 2000
        n_fallback = 2000
        a0 = np.ones(n_fallback, dtype=np.float64)
    # ensure nonnegative
    a0 = np.maximum(a0, 0.0)
    sum0 = a0.sum()
    if sum0 <= 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        sum0 = 1.0
    a0 = a0 / sum0

    n = a0.shape[0]
    L = 1 << ((2*n - 1).bit_length())

    # function to compute max conv
    def conv_max(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af*af, n=L)[:2*n-1]
        return conv.max(), conv

    # Multi-start: generate some random initial vectors
    best_vec = a0
    max_val, conv = conv_max(best_vec)
    best_sc = 2 * n * max_val
    # try some variations
    for p in np.linspace(0.0, 4.0, 9):
        v = (np.arange(n, 0, -1, dtype=np.float64) ** p)
        v = np.maximum(v, 0.0)
        v = v / v.sum()
        maxv, _ = conv_max(v)
        sc = 2 * n * maxv
        if sc < best_sc:
            best_sc = sc
            best_vec = v.copy()
    # maybe also random uniform
    for _ in range(5):
        v = rng.random(n)
        v = v / v.sum()
        maxv, _ = conv_max(v)
        sc = 2 * n * maxv

            best_vec = v.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # gradient descent parameters
    beta = 0.9
    base_alpha = 1.0
    top_k = 5
    momentum = np.zeros_like(cur)
    no_improve = 0
    # thresholds
    transfer_thresh = 300
    reinit_thresh = 800
    target_score = 1.50274349

    while time.monotonic() < deadline:

        cur_sc = 2 * n * max_val

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_full(vec: np.ndarray, L: int) -> np.ndarray:
    """Full convolution vec*vec using FFT of length L."""
    af = np.fft.rfft(vec, n=L)
    return np.fft.irfft(af * af, n=L)


def _conv_max(vec: np.ndarray, n: int, L: int):
    """Return max value and the full convolution (length 2n‑1)."""
    conv = _conv_full(vec, L)[: 2 * n - 1]
    return float(conv.max()), conv
