import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=Z}."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Smoothed max gradient via soft‑max weights and cross‑correlation
# ------------------------------------------------------------
def _smooth_grad(a: np.ndarray, beta: float, Lc: int, Lr: int):
    """
    Returns (grad, max_conv) where grad is the sub‑gradient of the
    smoothed max of the convolution a*a using soft‑max weight parameter beta.
    Lc is FFT length for convolution (>=2n‑1), Lr is FFT length for
    correlation (>=3n‑2).
    """
    n = a.size
    # convolution a*a
    a_fft = np.fft.rfft(a, n=Lc)
    conv = np.fft.irfft(a_fft * a_fft, n=Lc)[: 2 * n - 1]
    max_c = conv.max()

    # soft‑max weights emphasizing large entries
    diff = max_c - conv                     # non‑negative
    w = np.exp(-beta * diff)                # may underflow for far entries
    w_sum = w.sum()
    if w_sum == 0.0:                        # all underflowed -> uniform
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation w ⋆ a (valid mode) via FFT
    w_fft = np.fft.rfft(w, n=Lr)
    a_fft_r = np.fft.rfft(a, n=Lr)
    corr_full = np.fft.irfft(w_fft * np.conj(a_fft_r), n=Lr)
    # length of full correlation = len(w)+len(a)-1 = 3n-2
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]   # valid part, length n
    return grad, max_c
