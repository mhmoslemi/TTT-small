def _project_simplex(v):
    # ensure nonneg, sum 1 after projection
    # but typical algorithm works with any vector; we produce vector that sum = 1 and all >=0
    # Implementation:

    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:                                    # fallback: exponential decay
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 80.0)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s = 1.0
    a0 = a0 / s                                 # sum = 1

    n = a0.size
    # FFT length (power of two) – enough for all convolutions
    L = 1 << ((2 * n - 1).bit_length())

    cur = a0.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    momentum = np.zeros_like(cur)
    beta = 0.95               # momentum coefficient
    base_alpha = 0.4          # base step size
    eps = None                # temperature for softmax
    eps_decay = 0.995
    no_improve = 0
    target = 1.50274349
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        it += 1

        # ----- convolution and softmax weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max_val * 0.10 + 1e-12
        else:
            eps = max(eps * eps_decay, max_val * 1e-12)

        if eps < max_val * 1e-8:
            mask = (conv_full == max_val)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w /= w_sum
        else:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- adaptive step size -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / norm_m

        # ----- candidate generation & projection -----
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- backtracking if not improving -----
        if cand_sc >= cur_sc:
            improved = False
            for factor in (0.5, 0.25, 0.125):
                candb = cur - alpha * factor * momentum
                candb = _project_simplex(candb)
                candb_sc = _score(candb)
                if candb_sc < cur_sc:
                    cand = candb
                    cand_sc = candb_sc
                    improved = True
                    break
            if not improved:
                # occasional uphill acceptance (simulated annealing)
                temp = max(1e-3, 1.0 / math.sqrt(it))
                prob = math.exp(-(cand_sc - cur_sc) / temp)
                if rng.random() < prob:
                    # accept the uphill move
                    pass
                else:
                    cand = cur
                    cand_sc = cur_sc

        # ----- accept if better -----
        if cand_sc < cur_sc - 1e-12:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc

        # ----- random mass transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.30
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
                cand_sc = _score(cand)
                if cand_sc < cur_sc:
                    cur = cand
                    cur_sc = cand_sc
                    no_improve = 0
                    if cand_sc < best_sc:
                        best = cand.copy()
                        best_sc = cand_sc

        # ----- restart if hopeless -----
        if no_improve > 5000:
            cur = best + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # occasional tiny random swap (cheap refinement)
        if rng.random() < 0.005:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.05
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
                cand_sc = _score(cand)
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    cur = cand
                    cur_sc = cand_sc

    # --------------------------------------------------------
    # Final cheap local refinement (time‑limited)
    # --------------------------------------------------------
    while time.monotonic() < deadline - 0.1:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.07
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            cur = cand
            cur_sc = cand_sc

    # --------------------------------------------------------
    # Optional final symmetrisation
    # --------------------------------------------------------
    sym_final = (best + best[::-1]) * 0.5
    sym_final = _project_simplex(sym_final)
    sym_sc = _score(sym_final)
    if sym_sc < best_sc:
        best = sym_final
        best_sc = sym_sc

    # Return as plain Python list
    return [float(x) for x in best]
