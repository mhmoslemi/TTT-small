import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries summing to 1."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative list of coefficients with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness

    # Initialise a (simplex normalized)
    # --------------------------------------------------------

        s = a.sum()
        if s < 1e-12:
            a[:] = 1.0
            s = a.sum()
        a = a / s

    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    def _obj(x: np.ndarray) -> float:
        xf = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(xf * xf, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()
