# ------------------------------------------------------------
# Projection onto the probability simplex  (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if ind.size == 0:
        theta = 0.0
    else:
        rho = ind[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- random swap of two coordinates (cheap exploration) -----
        if rng.random() < random_swap_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                tmp = cur.copy()
                tmp[i], tmp[j] = tmp[j], tmp[i]           # swap
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0

        # ----- occasional re‑sorting (decreasing) -----
        if rng.random() < sort_prob:
            sorted_vec = np.sort(cur)[::-1]
            sorted_sc = _score(sorted_vec)
            if sorted_sc < best_sc:
                best_sc = sorted_sc
                best_vec = sorted_vec.copy()
            if sorted_sc < cur_sc:
                cur = sorted_vec
                cur_sc = sorted_sc
                no_improve = 0

        # ----- periodic symmetrisation (helps a lot) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                no_improve = 0

        # ----- restart from the best solution if we are stuck -----
        if no_improve > restart_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # return plain Python list
    return [float(x) for x in best_vec]
