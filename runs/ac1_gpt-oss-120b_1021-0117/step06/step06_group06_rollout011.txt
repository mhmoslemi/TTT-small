import time, math
import numpy as np

# ------------------------------------------------------------
# Utility: projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: ensure sum is exactly 1
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector with arbitrary sum
# ------------------------------------------------------------
def _ac1_score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv)/sum(a)^2 where conv = a*a (full convolution)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s <= 0.0:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


# ------------------------------------------------------------
# Gradient of the (soft) maximum of the convolution
# ------------------------------------------------------------
def _grad_and_score(x: np.ndarray, it: int, L: int) -> tuple:
    """
    Compute (gradient, score) for vector x.
    Gradient is w.r.t. the smoothed maximum of the self‑convolution.
    """
    n = x.shape[0]
    # FFT of the current vector
    af = np.fft.rfft(x, n=L)

    # full self‑convolution
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()

    # temperature schedule (smooth → hard max)
    # start with ~20% of max and decay exponentially
    temp_factor = max(0.2 * (0.999 ** it), 1e-6)
    temp = max_val * temp_factor

    if temp < max_val * 1e-8:
        # hard max, possibly multiple ties
        mask = (conv == max_val)
        cnt = mask.sum()
        if cnt == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = mask.astype(float) / cnt
    else:
        w = np.exp((conv - max_val) / temp)
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= ws

    # gradient via FFT (matrix‑free)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * w_fft, n=L)[: 2 * n - 1]
    # extract the slice that corresponds to ∂/∂a_j
    grad = 2.0 * grad_full[(2 * n - 2) - np.arange(n)]

    # score (consistent with evaluate_sequence)
    s = x.sum()
    score = float('inf') if s <= 0.0 else 2.0 * n * max_val / (s * s)
    return grad, score

    m = np.zeros_like(a)
