This is a convex optimization problem: f is convex (max of linear functions), domain is simplex (convex). So global optimum can be found via convex programming (LP). Indeed the earlier description mentions solving an LP for best g given f, iteratively.

But solving the LP directly may be expensive due to many constraints (2n-1 constraints). However each constraint (f*g)(k) ≤ M is a linear inequality where (f*g)(k) = sum_i f_i * g_{k-i}. For fixed f, optimizing g to maximize sum g_j given these constraints is a linear program. Similarly, we could formulate the primary problem (min M) directly as linear program: minimize M subject to nonnegative a_i, sum a_i >= epsilon (or sum arbitrary), and for each k, ∑_{i+j=k} a_i a_j ≤ M. However note that constraints are quadratic in a (product). This is not linear; but the previous approach uses alternating minimization: given f, solve linear program for g to maximize sum (increase mass while keeping convolution bound). Then combine.

def _max_conv(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2 * n - 1]

def _linear_decreasing(n):
    # decreasing linear sequence that sums to 1
    arr = np.arange(n, 0, -1, dtype=np.float64)
    return 2.0 * arr / (n * (n + 1))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.25)

    # determine length
    if 'height_sequence_1' in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000

    # base construction: linear decreasing sequence
    cur = _linear_decreasing(n)

    # evaluate
    best_vec = cur.copy()
    best_score = 2.0 * n * _max_conv(best_vec)

    target = 1.50274349

    # if already good, return
    if best_score <= target:
        return [float(x) for x in best_vec]

    # fallback optimizer: simple projected gradient descent with momentum
    # (reuse earlier code)
    L = 1 << ((2 * n - 1).bit_length())
    cur = _project_simplex(cur)  # already simplex but ensure numeric
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_alpha = 0.5
    t = 0
    cur_score = best_score
    no_improve = 0

    while time.monotonic() < deadline:
        t += 1
        # convolution and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # mask of max entries
        delta = max_val * 1e-12 + 1e-15
        w = (conv >= max_val - delta).astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step_alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # compute new score
        max_cand = _max_conv(cand)
        cand_score = 2.0 * n * max_cand

        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur = cand
        cur_score = cand_score

        # occasional symmetrisation
        if t % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = 2.0 * n * _max_conv(sym)
            if sym_score < best_score:
                best_score = sym_score
                best_vec = sym.copy()
                cur = sym
                cur_score = sym_score
                no_improve = 0

        # random perturb when stuck
        if no_improve > 2000:
            # random redistribution of a small amount
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.3

                    best_vec = tmp.copy()
                cur = tmp
                cur_score = tmp_score
            no_improve = 0

        if best_score <= target:
            break

    return [float(x) for x in best_vec]

import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _max_conv(a):
    """Maximum entry of the linear convolution a*a."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv.max()

def _linear_decreasing(n):
    """Return a decreasing linear sequence of length n that sums to 1."""
    # values proportional to (n-i) for i=0..n-1
    arr = np.arange(n, 0, -1, dtype=np.float64)       # n, n-1, ..., 1
    return 2.0 * arr / (n * (n + 1))
