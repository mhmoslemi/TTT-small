import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Fast AC1 bound assuming sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _enforce_sym_decr(x: np.ndarray) -> np.ndarray:
    """Make the vector symmetric and non‑increasing."""
    sym = (x + x[::-1]) * 0.5
    dec = np.maximum.accumulate(sym[::-1])[::-1]   # decreasing
    return dec

    n = a0.size
    idx = (2 * n - 2) - np.arange(n)               # index map for gradient
    L = 1 << ((2 * n - 1).bit_length())           # FFT length (next power of two)

    # --------------------------------------------------------

    # --------------------------------------------------------
    a0_sum = a0.sum()
    if a0_sum <= 0.0:

    else:

    a0 = _enforce_sym_decr(a0)

    # --------------------------------------------------------

    def descent(start_vec):

        m = np.zeros_like(cur)            # first moment (Adam)

        it = 0
