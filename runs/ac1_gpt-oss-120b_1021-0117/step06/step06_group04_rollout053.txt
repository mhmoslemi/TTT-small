import time

import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    return conv

    target_score = 1.50274349

    best_a = a.copy()
    conv = _conv_fft(a, L)
    best_sc = 2.0 * n * conv.max() / (a.sum() ** 2)   # sum(a)=1 ⇒ denominator=1

    no_improve = 0
    it = 0

    # -----------------------------------------------------------------
    # main optimisation loop
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current objective
        conv = _conv_fft(a, L)
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target_score:
            break

        # -------------------------------------------------------------
        # smooth approximation of the max via soft‑max weights
        # -------------------------------------------------------------
        gamma = min(gamma_max, gamma0 + gamma_inc * it)
        w = np.exp(gamma * (conv - cur_max))          # max entry becomes 1.0
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # -------------------------------------------------------------
        # gradient of the smooth surrogate: 2 * correlation(w, a)
        # -------------------------------------------------------------
        corr = np.correlate(w, a, mode='full')        # length 3n‑2
        grad = 2.0 * corr[n - 1:n - 1 + n]            # lags 0,…,n‑1

        # -------------------------------------------------------------
        # backtracking line‑search on the simplex
        # -------------------------------------------------------------
        eta = eta0 / math.sqrt(it)
        step = eta
        improved = False
        for _ in range(5):
            a_try = a - step * grad
            a_try = _project_simplex(a_try)
            conv_try = _conv_fft(a_try, L)
            sc_try = 2.0 * n * conv_try.max() / (a_try.sum() ** 2)
            if sc_try < cur_sc:
                a = a_try
                improved = True
                break
            step *= 0.5

        # occasional uphill move (simulated‑annealing style)
        if not improved:
            a_try = a - step * grad
            a_try = _project_simplex(a_try)
            conv_try = _conv_fft(a_try, L)
            sc_try = 2.0 * n * conv_try.max() / (a_try.sum() ** 2)
            prob = math.exp(-(sc_try - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = a_try
                improved = True

        # -------------------------------------------------------------
        # periodic symmetrisation (helps keep the shape regular)
        # -------------------------------------------------------------
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a = _project_simplex(a_sym)

        # -------------------------------------------------------------
        # random mass transfer when progress stalls
        # -------------------------------------------------------------
        if no_improve > transfer_thresh:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a = _project_simplex(a_tmp)
                no_improve = 0

    # return the best vector found as a plain Python list
    return [float(x) for x in best_a]
