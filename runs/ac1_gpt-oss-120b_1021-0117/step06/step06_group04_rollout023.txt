def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # margin
    # Determine n and initial array
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        total = a0.sum()
        if total == 0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            total = 1.0
        a0 = a0 / total
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)
    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    # function to compute score
    def score_of(x):
        # assuming sum=1? but compute anyway
        af = np.fft.rfft(x, L)
        conv = np.fft.irfft(af * af, L)[: 2 * n - 1]
        maxc = conv.max()
        # denominator sum^2, but sum=1 typical
        denom = x.sum()
        return 2.0 * n * maxc / (denom * denom)
    # Evaluate a few simple patterns to get good starting point
    best_vec = a0.copy()
    best_sc = score_of(best_vec)

    # also try uniform and power law
    uniform = np.full(n, 1.0 / n)
    sc_u = score_of(uniform)
    if sc_u < best_sc:
        best_sc, best_vec = sc_u, uniform.copy()

    # try power law family a_i ∝ (i+1)^p for p in [-1, 4] steps maybe 0.5
    for p in np.linspace(-1.5, 4.0, 12):
        a = (np.arange(1, n+1, dtype=np.float64) ** p)
        a = a / a.sum()
        sc = score_of(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # current solution = best_vec
    cur = best_vec.copy()
    cur_sc = best_sc

    # parameters
    T0 = 0.1 * best_sc  # initial temperature relative to score
    T = T0
    decay = 0.9995
    # Step sizes
    delta_factor = 0.2
    cauchy_scale = 0.02
    grad_lr = 0.5
    # Precompute indices array
    idx_arr = np.arange(n)

    # main loop
    iter = 0

import time, math, numpy as np

def _score(a: np.ndarray, n: int, L: int) -> float:
    """Return the AC1 bound for a (sum may be ≠1)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex. """
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)
