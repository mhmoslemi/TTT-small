def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    return conv

    cur = a0.copy()
    indices = np.arange(n)

    # current score
    conv_cur = _full_conv(cur, L)
    cur_max = conv_cur.max()
    cur_sc = 2.0 * n * cur_max / (cur.sum() ** 2)

    best_vec = cur.copy()
    best_sc = cur_sc

    # -------------------- hyper‑parameters --------------------
    base_alpha = 0.6          # base step‑size factor
    top_k = 5                 # number of top convolution lags used for sub‑gradient
    max_backtrack = 6
    sym_interval = 250
    target_score = 1.50274349
    stagnation = 0

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # recompute convolution (needed for fresh gradient)
        conv_cur = _full_conv(cur, L)
        cur_max = conv_cur.max()
        cur_sc = 2.0 * n * cur_max / (cur.sum() ** 2)

        # ---- sub‑gradient from the top‑k convolution entries ----
        if top_k < conv_cur.size:
            top_idxs = np.argpartition(conv_cur, -top_k)[-top_k:]
        else:
            top_idxs = np.arange(conv_cur.size)

        grad = np.zeros_like(cur)
        for lag in top_idxs:
            j = lag - indices
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * cur[j[mask]]

        # ---- step size (diminishing with iteration) ----
        norm_g = np.linalg.norm(grad) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_g)

        # ---- back‑tracking line search ----
        improved = False
        cand = None
        cand_sc = None
        alpha_try = alpha
        for _ in range(max_backtrack):
            cand_try = cur - alpha_try * grad
            cand_try = _project_simplex(cand_try)
            conv_cand = _full_conv(cand_try, L)
            max_cand = conv_cand.max()
            sc_cand = 2.0 * n * max_cand / (cand_try.sum() ** 2)
            if sc_cand < cur_sc:
                cand, cand_sc = cand_try, sc_cand
                improved = True
                break
            alpha_try *= 0.5

        # occasional uphill acceptance (simulated‑annealing style)
        if not improved:
            # use the last trial for probability calculation
            prob = math.exp(-(sc_cand - cur_sc) / (alpha_try + 1e-12))
            if rng.random() < prob:
                cand, cand_sc = cand_try, sc_cand
                improved = True

        if improved:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # ---- periodic symmetrisation (helps regularise shape) ----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            conv_sym = _full_conv(sym, L)
            sc_sym = 2.0 * n * conv_sym.max() / (sym.sum() ** 2)
            if sc_sym < cur_sc:
                cur, cur_sc = sym, sc_sym
                if sc_sym < best_sc:
                    best_sc, best_vec = sc_sym, sym.copy()
                stagnation = 0

        # ---- random mass transfer when progress stalls ----
        if stagnation > 1200:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                conv_tmp = _full_conv(tmp, L)
                sc_tmp = 2.0 * n * conv_tmp.max() / (tmp.sum() ** 2)
                if sc_tmp < cur_sc:
                    cur, cur_sc = tmp, sc_tmp
                    if sc_tmp < best_sc:
                        best_sc, best_vec = sc_tmp, tmp.copy()
                    stagnation = 0
                else:
                    stagnation = max(0, stagnation - 200)

        # ---- restart from the best point after long stagnation ----
        if stagnation > 2500:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            stagnation = 0

        # ---- early exit if target reached ----
        if best_sc < target_score:
            break

    # Return as plain Python list
    return [float(x) for x in best_vec]
