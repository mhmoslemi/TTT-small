def _grad_max(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv.max()
    # mask for max indices
    mask = conv == max_val
    w = mask.astype(np.float64)
    w /= w.sum()

    w_rev = w[::-1]

    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

def _gradient_max(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv.max()
    mask = conv == max_val
    w = mask.astype(np.float64)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0/conv.size)
    else:
        w /= w_sum
    # compute gradient = 2 * convolution of a with w_rev
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

import time, math
import numpy as np

# ------------------------------------------------------------
# Helpers
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score_norm(a: np.ndarray) -> float:
    """Score assuming sum(a)=1: 2*n*max(convolution)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _gradient_softmax(a: np.ndarray, eps: float):
    """Gradient of the (smoothed) maximum of a*a."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()

    if eps <= 0.0:
        w = (conv == max_val).astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum
    else:
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val


def _conv_and_max(a: np.ndarray):
    """Return full convolution and its maximum."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv, conv.max()
