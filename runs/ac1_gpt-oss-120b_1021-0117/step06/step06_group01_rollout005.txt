import time

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        # initialise / decay the smoothing parameter
        if eps_smooth is None:
            eps_smooth = max_val * 0.2
        else:
            eps_smooth = max(eps_smooth * 0.995, max_val * 1e-6)

        # ---- Adam update --------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- line‑search like trial (few scale factors) ------------------------
        improved = False
        for scale in (0.5, 1.0, 2.0):
            cand = cur - scale * step
            cand = _project_simplex(cand)
            sc = _score_normalised(cand, L)
            if sc < cur_sc:
                cur = cand
                cur_sc = sc
                improved = True
                break

        if improved:
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- occasional symmetrisation (helps a lot) ---------------------------
        if it % 75 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sc_sym = _score_normalised(sym, L)
            if sc_sym < cur_sc:
                cur = sym
                cur_sc = sc_sym
                if sc_sym < best_sc:
                    best_sc = sc_sym
                    best_vec = sym.copy()
                no_improve = 0

        # ---- random mass transfer when progress stalls -------------------------
        if no_improve > 1500:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.30
                j = rng.integers(0, n)
                if j == i:
                    j = (j + 1) % n
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                sc_tmp = _score_normalised(tmp, L)
                if sc_tmp < cur_sc:
                    cur = tmp
                    cur_sc = sc_tmp
                    if sc_tmp < best_sc:
                        best_sc = sc_tmp
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ---- restart from the best with a little noise if hopeless -------------
        if no_improve > 4000:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps_smooth = None
            no_improve = 0

        # ---- early exit if target achieved ------------------------------------
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Fallback safety: ensure something is returned
    # --------------------------------------------------------
    if best_vec is None or best_vec.size == 0:
        best_vec = a0.copy()

    return [float(x) for x in best_vec]
