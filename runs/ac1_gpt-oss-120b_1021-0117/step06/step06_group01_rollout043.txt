import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: exact evaluation of the AC1 bound (uses the given evaluator)
# ------------------------------------------------------------
def _exact_score(vec: np.ndarray) -> float:
    """Return the exact score using the provided evaluate_sequence."""
    # The evaluator is imported in the execution environment.
    return evaluate_sequence(vec.tolist())


# ------------------------------------------------------------
# Helper: project a vector onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative step heights that (approximately)
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # --------------------------------------------------------
    # Determine length and initialise a feasible point (sum = 1)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        # fallback: a uniform distribution of modest size
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    n = a0.size
    best_vec = a0.copy()
    best_sc = _exact_score(best_vec)

    # --------------------------------------------------------
    # Helper to test a candidate and keep the best so far
    # --------------------------------------------------------
    def _try_candidate(vec: np.ndarray):
        nonlocal best_vec, best_sc
        sc = _exact_score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    # --------------------------------------------------------
    # Small parametric families (cheap exhaustive search)
    # --------------------------------------------------------
    # exponential decay
    for _ in range(150):
        alpha = 10 ** rng.uniform(-4, 0)               # 1e‑4 … 1
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand = cand / cand.sum()
        _try_candidate(cand)

    # symmetric exponential (decay from both ends)
    for _ in range(150):
        alpha = 10 ** rng.uniform(-4, 0)
        d = np.minimum(np.arange(n, dtype=np.float64),
                       n - 1 - np.arange(n, dtype=np.float64))
        cand = np.exp(-alpha * d)
        cand = cand / cand.sum()
        _try_candidate(cand)

    # Gaussian‑like blobs
    mid = (n - 1) / 2.0
    for _ in range(150):
        sigma = rng.uniform(0.5, 50.0)
        cand = np.exp(-((np.arange(n, dtype=np.float64) - mid) ** 2) /
                       (2 * sigma ** 2))
        cand = cand / cand.sum()
        _try_candidate(cand)

    # cosine‑squared (high at ends, low in centre)
    for _ in range(150):
        cand = np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** 2
        cand = cand / cand.sum()
        _try_candidate(cand)

    # --------------------------------------------------------
    # Simple stochastic hill‑climbing with occasional restarts
    # --------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    no_improve = 0
    while time.monotonic() < deadline:
        # propose a random mass transfer between two positions
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑project to the simplex (keeps sum = 1 and non‑negative)
        cand = _project_simplex(cand)
        cand_sc = _exact_score(cand)

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # if we get stuck, restart from the current best with a little noise
        if no_improve > 3000:

            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_sc = _exact_score(cur)
            no_improve = 0

    # --------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
