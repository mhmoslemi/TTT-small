# ------------------------------------------------------------
# Helper: Euclidean projection onto probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Fast evaluation of the AC1 bound (works for any sum)

def _score(a: np.ndarray) -> float:
    """Return the AC1 upper bound for a (no modification of a)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

        grad = 2.0 * grad_full[idx]

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- step size (norm‑scaled) ----
        norm_m = np.linalg.norm(momentum) + 1e-12
        step = 0.6 / norm_m                     # base step, tuned experimentally

        # ---- candidate generation and projection ----
        cand = cur - step * momentum
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        cand_sc = _score(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur = cand

        # ---- periodic symmetrisation (helps a lot) ----
        if iter_cnt % 120 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = np.maximum(sym, 0.0)
            sym /= sym.sum()
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                no_improve = 0
            cur = sym

        # ---- random mass‑transfer when stuck ----
        if no_improve > 1500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp /= tmp.sum()
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    no_improve = 0
                cur = tmp

        # ---- restart from best with small perturbation if hopeless ----
        if no_improve > 4000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Return best candidate found (as plain Python list)
    # --------------------------------------------------------
    if best_vec.size == 0:                     # safety fallback
        best_vec = np.full(n, 1.0 / n)
    return [float(x) for x in best_vec]
