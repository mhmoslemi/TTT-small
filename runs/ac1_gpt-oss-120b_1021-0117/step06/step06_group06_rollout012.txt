def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # small safety

    # Initialize vector
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        a = np.exp(-np.arange(n) / 100.0)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
        s = 1.0
    a = a / s  # sum=1

    n = a.size
    L = 1 << ((2*n - 1).bit_length())  # power of two >= 2n-1

    # evaluation
    def evaluate(arr):
        af = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2 * n * conv.max())

    cur = a.copy()
    cur_score = evaluate(cur)

    best = cur.copy()
    best_score = cur_score

    # hyperparameters
    base_alpha = 0.2
    max_temp = 0.02
    cooling = 0.9997
    no_improve = 0
    iter_cnt = 0

    # Main loop
    while time.monotonic() < deadline:
        iter_cnt += 1

        # compute convolution and max idx
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[:2*n-1]
        max_idx = int(conv_full.argmax())
        max_val = float(conv_full[max_idx])
        cur_score = 2.0 * n * max_val

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # gradient for hard max
        grad = np.zeros_like(cur)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        i_range = np.arange(lo, hi + 1)
        j_range = max_idx - i_range
        grad[i_range] = 2.0 * cur[j_range]

        # step size
        alpha = base_alpha / (math.sqrt(iter_cnt) + 1.0)

        cand = cur - alpha * grad
        cand = _project_simplex(cand)

        cand_score = evaluate(cand)

        # acceptance
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            # simulated annealing
            temp = max_temp * (cooling ** iter_cnt)
            if temp < 1e-6:
                temp = 1e-6
            prob = math.exp(-(cand_score - cur_score) / temp)
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score
                no_improve = 0

        # occasional random perturbation
        if no_improve > 3000:
            # random redistribute some mass
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = evaluate(tmp)
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                    no_improve = 0

        # periodic symmetrization
        if iter_cnt % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = evaluate(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score
                no_improve = 0

    # final refinement: small random swaps for remaining time
    while time.monotonic() < deadline - 0.05:
        i = rng.integers(0, n)
        if cur[i] <= 0:
            continue
        delta = cur[i] * rng.random() * 0.05
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_score = evaluate(cand)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            cur = cand
            cur_score = cand_score

    return [float(x) for x in best]

# ------------------------------------------------------------
# Simple projection onto the simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to unit sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s

# Fast AC1 bound for a vector that already sums to 1
# (returns 2 * n * max(conv) )
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
