import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return max(convolution a*a), its index, and the full convolution."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_max, _, _ = _conv_max_and_index(cand, L)
        cand_score = 2.0 * n * cand_max / (cand.sum() ** 2)

        # ---------------------------------------------------------
        # accept / reject (with simulated annealing)
        # ---------------------------------------------------------
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            # annealing probability
            frac = t / 50000.0
            T = max(Tmin, T0 * (1.0 - frac))
            prob = math.exp(-(cand_score - cur_score) / (T + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
                no_improve += 1
            else:
                no_improve += 1

        # ---------------------------------------------------------
        # periodic symmetrisation
        # ---------------------------------------------------------
        if t % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_max, _, _ = _conv_max_and_index(sym, L)
            sym_score = 2.0 * n * sym_max / (sym.sum() ** 2)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best_vec, best_score = sym.copy(), sym_score
                no_improve = 0

        # ---------------------------------------------------------
        # random mass transfer when stagnating
        # ---------------------------------------------------------
        if no_improve > 1200:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_max, _, _ = _conv_max_and_index(tmp, L)
                tmp_score = 2.0 * n * tmp_max / (tmp.sum() ** 2)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    if tmp_score < best_score:
                        best_vec, best_score = tmp.copy(), tmp_score
                    no_improve = 0
                else:
                    no_improve = max(0, no_improve - 300)

        # ---------------------------------------------------------
        # restart from best with perturbation if really stuck
        # ---------------------------------------------------------
        if no_improve > 5000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        # early exit if target reached
        if best_score < 1.50274349:
            break

    # return Python list of floats
    return [float(x) for x in best_vec]
