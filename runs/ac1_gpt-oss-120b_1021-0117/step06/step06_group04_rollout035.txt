import time, math, numpy as np

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max_and_full(a, L):

    max_val = float(conv.max())
    idx = int(conv.argmax())

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_full(a: np.ndarray, L: int):
    """Full convolution a*a via FFT, returning max value, its index and the full array."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    max_val = float(conv.max())
    idx = int(conv.argmax())
    return max_val, idx, conv
