def _project_simplex(v):
    """Project onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:

    else:

        if not improved:
            cand_try = cur - alpha_try * momentum
            cand_try = _project_simplex(cand_try)
            max_val, _, _ = _conv_max_and_index(cand_try, L)
            sc = 2.0 * n * max_val / (cand_try.sum() ** 2)
            prob = math.exp(-(sc - cur_sc) / (alpha_try + 1e-12))
            if rng.random() < prob:
                cand, cand_sc = cand_try, sc
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            # recompute full convolution for the next iteration's gradient
            cur_max, cur_idx, cur_conv = _conv_max_and_index(cur, L)
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---------------------------------------------------------
        # periodic symmetrisation (helps keep shape regular)
        # ---------------------------------------------------------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            max_val, _, sym_conv = _conv_max_and_index(sym, L)
            sym_sc = 2.0 * n * max_val / (sym.sum() ** 2)
            if sym_sc < cur_sc:
                cur, cur_sc, cur_conv = sym, sym_sc, sym_conv
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ---------------------------------------------------------
        # random mass transfer when progress stalls
        # ---------------------------------------------------------
        if no_improve > transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                max_val, _, tmp_conv = _conv_max_and_index(tmp, L)
                tmp_sc = 2.0 * n * max_val / (tmp.sum() ** 2)
                if tmp_sc < cur_sc:
                    cur, cur_sc, cur_conv = tmp, tmp_sc, tmp_conv
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
                else:
                    no_improve = max(no_improve - 200, 0)

        # ---------------------------------------------------------
        # restart from the best point if stuck too long
        # ---------------------------------------------------------
        if no_improve > reinit_thresh:
            cur = best_vec + 0.005 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_max, cur_idx, cur_conv = _conv_max_and_index(cur, L)
            cur_sc = 2.0 * n * cur_max / (cur.sum() ** 2)
            momentum.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_sc < target_score:
            break

    # return plain Python list
    return [float(x) for x in best_vec]
