import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of 2*n*max(convolution) for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(conv) for a sequence whose sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution (exact, non‑smooth)
# ------------------------------------------------------------
def _subgrad_maxconv(a: np.ndarray, conv_full: np.ndarray, max_val: float) -> np.ndarray:
    """Compute a sub‑gradient of max(conv) w.r.t. a (a is non‑negative, sum=1)."""
    n = a.shape[0]
    # indices where the convolution attains its maximum
    idx = np.flatnonzero(np.isclose(conv_full, max_val, atol=1e-12, rtol=0.0))
    grad = np.zeros_like(a)
    offset = n - 1
    for k in idx:
        shift = k - offset
        if shift >= 0:
            # contributes to positions shift … n‑1
            grad[shift:] += 2.0 * a[: n - shift]
        else:
            s = -shift
            grad[: n - s] += 2.0 * a[s:]
    return grad
