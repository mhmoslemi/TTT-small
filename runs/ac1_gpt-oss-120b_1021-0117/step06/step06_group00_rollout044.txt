import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # enforce exact sum=1 (numerical)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score for a vector already normalized to sum=1 (2*n*max conv)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.9          # momentum coefficient
    step0 = 1.0         # initial step factor
    alpha = 8.0         # soft‑max temperature (larger => sharper)
    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main loop ------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and gradient (soft‑max weighted) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights over all convolution entries
        w_raw = np.exp(alpha * (conv - max_val))   # subtract max for stability
        w = w_raw / w_raw.sum()

        # gradient of weighted sum of convolution entries
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum & step ---------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        # ----- candidate generation -----------------------------
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score(cand)

        # ----- acceptance ---------------------------------------
