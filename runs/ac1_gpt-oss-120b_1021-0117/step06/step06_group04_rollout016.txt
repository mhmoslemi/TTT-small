    """Return max value of the linear (non‑circular) convolution a*a."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * n - 1]
    return float(conv.max())

    """Euclidean projection of v onto the probability simplex."""
    n = v.size

    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)

    # ----------------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)
    beta = 0.9
    base_alpha = 0.4
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # compute max‑convolution index and sub‑gradient
        conv = np.fft.irfft(np.fft.rfft(cur, 1 << ((2 * n - 1).bit_length())) ** 2)
        conv = conv[: 2 * n - 1]
        max_val = conv.max()
        max_idx = int(conv.argmax())

        j = max_idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[j[mask]]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        norm_m = np.linalg.norm(momentum) + 1e-12
        step = base_alpha / (math.sqrt(it) * norm_m) * momentum

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = 2.0 * n * _conv_max(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (0.01 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        if best_sc < 1.50274349:
            break

    # ----------------------------------------------------------------------
    # return the best found sequence as plain Python list
    # ----------------------------------------------------------------------
    return [float(x) for x in best_vec]
