import time, math, numpy as np

# ------------------------------------------------------------
# helper utilities
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max(a: np.ndarray, L: int):
    """Return (max value, index of max, full convolution)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def _score(a: np.ndarray, L: int) -> float:
    """Exact score used by the evaluator (2*n*max/ sum^2)."""
    n = a.shape[0]
    mx, _, _ = _conv_max(a, L)
    s = a.sum()
    # sum is never zero because we keep it normalized
    return 2.0 * n * mx / (s * s)
