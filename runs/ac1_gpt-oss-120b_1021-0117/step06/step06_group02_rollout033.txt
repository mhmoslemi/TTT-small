import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum == 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a) ≈ 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Compute 2·n·max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    # power‑of‑two length comfortably larger than 2n‑1
    L = 1 << ((2 * n - 1).bit_length() + 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
