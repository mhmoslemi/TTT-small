import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int, n: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Softmax (stable) used for the log‑parameterisation
# ------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ------------------------------------------------------------
# Compute full convolution, its maximum and the indices where it is attained
# ------------------------------------------------------------
def _conv_max_idxs(a: np.ndarray, L: int):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    max_val = conv.max()
    # tolerance to capture ties caused by floating point noise
    eps = max_val * 1e-12 if max_val != 0 else 1e-12
    idxs = np.where(conv >= max_val - eps)[0]
    return conv, max_val, idxs


# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution (averaged over all active indices)
# ------------------------------------------------------------
def _subgrad(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    n = a.size
    g = np.zeros_like(a)
    cnt = len(max_idxs)
    for k in max_idxs:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        j = np.arange(start, end + 1)
        g[j] += 2.0 * a[k - j]
    if cnt > 1:
        g /= cnt
    return g
