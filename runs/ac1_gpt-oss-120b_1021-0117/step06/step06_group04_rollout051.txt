    # determine length n (use the parent if it exists)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000                     # fallback length

    # -----------------------------------------------------------------
    # build a linear ramp a_i ∝ i+1, normalized to sum = 1
    # -----------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)      # 1 … n
    a = 2.0 * idx / (n * (n + 1))                    # proportional to i+1
    a /= a.sum()                                     # enforce exact sum = 1

    # -----------------------------------------------------------------
    # (optional) tiny random jitter if time permits – kept extremely small
    # -----------------------------------------------------------------
    if time.monotonic() < deadline:
        rng = np.random.default_rng(seed)
        jitter = (rng.random(n) - 0.5) * 1e-6        # ±5e‑7 perturbation
        a = a + jitter
        a = np.clip(a, 0.0, None)                    # keep non‑negative
        if a.sum() > 0.0:
            a /= a.sum()                             # renormalise

    # -----------------------------------------------------------------
    # return plain Python list
    # -----------------------------------------------------------------
    return [float(v) for v in a]
