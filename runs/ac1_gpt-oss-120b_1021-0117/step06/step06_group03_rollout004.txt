import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound.

    The function runs until the time budget expires or a target bound is reached.
    It works directly on the simplex (sum of coefficients = 1) and uses
    exponentiated‑gradient updates driven by the exact sub‑gradient of the
    maximum convolution value.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initial vector -------------------------------------------------

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

        else:
            a = a / s
    else:       # generic fallback
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a = np.exp(-t / 100.0)
        a = a / a.sum()

    n = a.size
