import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (L1 = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # enforce exact sum = 1 (numerical safety)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / n
    return w

# ------------------------------------------------------------
# Self‑convolution via FFT (real inputs)
# ------------------------------------------------------------
def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return conv

# ------------------------------------------------------------
# Objective (same formula as evaluate_sequence)
# ------------------------------------------------------------

    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))

                cur_score = sym_score
                cur_conv = sym_conv
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass transfer when stuck -----
        if no_improve > mass_move_thr:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                trial = a.copy()
                trial[i] -= delta
                trial[j] += delta
                trial = _project_simplex(trial)
                trial_conv = _conv_self(trial, L)
                trial_score = _objective(trial, trial_conv)
                if trial_score < cur_score:
                    a = trial
                    cur_score = trial_score
                    cur_conv = trial_conv
                    if trial_score < best_score:
                        best_score = trial_score
                        best_vec = trial.copy()
                    no_improve = 0

        # ----- restart from best if hopeless -----
        if no_improve > restart_thr:
            a = best_vec + 0.01 * rng.random(n)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- early exit on target -----
        if best_score < target:
            break

        # time guard
        if time.monotonic() >= deadline:
            break

    # ----- final local tweaks (cheap random swaps) -----
    while time.monotonic() < deadline - 0.03:
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.random() * 0.07
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        trial = a.copy()
        trial[i] -= delta
        trial[j] += delta
        trial = _project_simplex(trial)
        trial_conv = _conv_self(trial, L)
        trial_score = _objective(trial, trial_conv)
        if trial_score < best_score:
            best_score = trial_score
            best_vec = trial.copy()
            a = trial
            cur_score = trial_score
            cur_conv = trial_conv

    # Return as plain Python list
    return [float(x) for x in best_vec]
