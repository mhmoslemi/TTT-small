import time, math, numpy as np

def _project_simplex(v):
    # Euclidean projection onto simplex (nonnegative sum to 1)
    # simple version: clip negatives and renorm
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        return v / s
    # fallback uniform
    n = v.shape[0]
    return np.full_like(v, 1.0 / n)

def _score(a):

                        cur = tmp
                        cur_score = tmp_score
                        no_improve = 0

        # ----- restart from best with a little noise if hopelessly stuck -----
        if no_improve > 3500:
            cur = best + 0.015 * rng.random(n)
            cur = _project_simplex(cur)
            m.fill(0.0)
            v.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_score < target:
            break

        # ----- periodic time guard -----
        if it % 400 == 0 and time.monotonic() >= deadline:
            break

    # --------------------------------------------------------
    # return the best candidate found (as plain Python floats)
    # --------------------------------------------------------
    return [float(x) for x in best]
```
