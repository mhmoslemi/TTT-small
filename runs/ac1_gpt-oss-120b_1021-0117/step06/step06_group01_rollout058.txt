import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Various parametric families (generated algorithmically)
# ----------------------------------------------------------------------
def _exp_vec(alpha: float, n: int) -> np.ndarray:
    vec = np.exp(-alpha * np.arange(n, dtype=np.float64))
    return vec / vec.sum()

def _sym_exp_vec(alpha: float, n: int) -> np.ndarray:
    d = np.minimum(np.arange(n, dtype=np.float64),
                   n - 1 - np.arange(n, dtype=np.float64))
    vec = np.exp(-alpha * d)
    return vec / vec.sum()

def _gauss_vec(sigma: float, n: int) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    vec = np.exp(-((x - mid) ** 2) / (2.0 * sigma ** 2))
    return vec / vec.sum()

def _cos2_vec(n: int) -> np.ndarray:
    vec = np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** 2
    return vec / vec.sum()

def _random_mix(best: np.ndarray, rng: np.random.Generator, n: int) -> np.ndarray:
    alpha = 10 ** rng.uniform(-4, 0)
    rnd = _exp_vec(alpha, n)
    t = rng.random()
    cand = (1 - t) * best + t * rnd
    return _project_simplex(cand)
