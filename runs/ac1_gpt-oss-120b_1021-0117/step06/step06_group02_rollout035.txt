    rng = np.random.default_rng(seed)

    deadline = start + max(0.0, float(budget_s) - 0.2)   # margin

            s = 1.0
        a = a / s
    else:

        t = np.arange(n_fallback, dtype=np.float64)
        a = np.exp(-t / 120.0)
        a = a / a.sum()
    n = a.shape[0]
    L = 1 << ((3*n - 2).bit_length())
    grad_idx = (2*n - 2) - np.arange(n, dtype=np.int64)
    best_a = a.copy()
    best_score = float('inf')
    # compute initial score
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft*a_fft, n=L)[:2*n-1]
    best_score = 2.0 * n * conv.max()
    best_a[:] = a
    # parameters
    eps = None
    eps_decay = 0.999   # slower decay
    base_lr = 0.2   # learning rate for exponentiated gradient
    # maybe use Adam: but for simplicity use exponentiated with decaying lr.
    # We'll also compute momentum track (optional).
    # We'll also maintain some algorithm to random perturb when stuck.
    no_improve = 0
    iter = 0
    # main loop
    while time.monotonic() < deadline:
        iter += 1
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft*a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # compute or update epsilon
        if eps is None:
            eps = max(max_val * 0.5, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)
        # smoothed weights
        w = np.exp((conv_full - max_val) / eps)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]
        # learning rate schedule
        lr = base_lr / math.sqrt(iter)
        # update a multiplicatively
        a = a * np.exp(-lr * g)
        # ensure positivity and avoid underflow
        a = np.maximum(a, 1e-15)
        a_sum = a.sum()

            a_sum = a.sum()
        a = a / a_sum

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0

            if sym_score < cur_score:
