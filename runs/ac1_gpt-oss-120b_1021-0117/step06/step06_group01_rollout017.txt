def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # init
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n0 = 2000
        a0 = np.exp(-np.arange(n0)/100.0)

    a0 = np.maximum(a0, 0.0)
    # normalize sum=1
    sum0 = a0.sum()
    if sum0 < 1e-12:
        a0 = np.full_like(a0, 1.0/a0.size)
    else:
        a0 = a0 / sum0

    # enforce symmetry
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = _project_simplex(a0)

    n = a0.size
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    # helper
    def try_candidate(vec):
        nonlocal best_vec, best_sc
        sc = _score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    # exploration families (increase loops)
    for _ in range(800):
        alpha = 10 ** rng.uniform(-4, 0)
        cand = np.exp(-alpha * np.arange(n))
        cand = cand / cand.sum()
        cand = (cand + cand[::-1]) * 0.5
        try_candidate(_project_simplex(cand))

    # other families similarly bigger loops
    for _ in range(800):
        alpha = 10 ** rng.uniform(-4,0)
        d = np.minimum(np.arange(n), n-1-np.arange(n))
        cand = np.exp(-alpha * d)
        cand = cand / cand.sum()
        cand = (cand + cand[::-1]) * 0.5
        try_candidate(_project_simplex(cand))

    for _ in range(800):
        sigma = rng.uniform(0.5, 80.0)
        mid = (n-1)/2
        cand = np.exp(-((np.arange(n)-mid)**2)/(2*sigma**2))
        cand = cand / cand.sum()
        cand = (cand + cand[::-1]) * 0.5
        try_candidate(_project_simplex(cand))

    for _ in range(800):
        cand = np.cos(np.pi * np.arange(n)/(n-1))**2
        cand = cand / cand.sum()
        cand = (cand + cand[::-1]) * 0.5
        try_candidate(_project_simplex(cand))

    # gradient descent loop
    cur = best_vec.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)
    beta = 0.9
    base_alpha = 1.0
    eps = None
    eps_decay = 0.998
    sym_interval = 1   # apply each iteration
    max_trials = 30
    it = 0
    no_improve = 0
    improve_target = 1.50274349
    # store previous vector for Nesterov?
    prev = cur.copy()

    while time.monotonic() < deadline:
        it += 1

        # Nesterov extrapolation
        y = cur + beta * (cur - prev)
        # Ensure nonneg and sum=1? We'll compute gradient at y, then project after step.
        # But y may be out of simplex; gradient step will project.

        # Compute conv of y
        L = 1 << ((2*n - 1).bit_length())
        y_fft = np.fft.rfft(y, n=L)
        conv_full = np.fft.irfft(y_fft * y_fft, n=L)[:2*n-1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.2, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        momentum = beta * momentum + (1.0 - beta) * g

        # still symmetrize (already done in projection)

                j = rng.integers(0, n - 1)

                    j += 1

                tmp[i] -= delta

                tmp = (tmp + tmp[::-1]) * 0.5

                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    prev = cur
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        if no_improve > 6000:
            # restart
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            prev = cur.copy()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        if best_sc < improve_target:
            break

    # Return list
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
