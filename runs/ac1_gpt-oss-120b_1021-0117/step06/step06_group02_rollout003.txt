def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)

            prob = math.exp(-(cand_score - cur_score) / (max_val * 0.01 + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1

        # ----------------------------------------------------
        # periodic symmetrisation (helps because the problem is
        # often near‑symmetric)
        # ----------------------------------------------------
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score
                no_improve = 0

        # ----------------------------------------------------
        # random mass‑transfer when stuck
        # ----------------------------------------------------
        if no_improve > 1800:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score(tmp)
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                    if tmp_score < best_score:
                        best = tmp.copy()
                        best_score = tmp_score
                no_improve = 0

        # ----------------------------------------------------
        # restart from best with tiny noise if hopelessly stuck
        # ----------------------------------------------------
        if no_improve > 3500:
            cur = best + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps = None
            no_improve = 0

        # ----------------------------------------------------
        # early exit
        # ----------------------------------------------------
        if best_score < target:
            break

        # occasional time check
        if it % 500 == 0 and time.monotonic() >= deadline:
            break

    # return as plain Python floats
    return [float(x) for x in best]
