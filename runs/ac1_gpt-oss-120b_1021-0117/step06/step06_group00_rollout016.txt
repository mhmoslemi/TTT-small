def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score assuming sum(a)=1 (the evaluator divides by sum²)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a little margin for the return handling
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---------- initialization ----------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2500, dtype=np.float64)
    a = _project_simplex(np.maximum(a, 0.0))
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # ---------- Adam optimizer state ----------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                     # base learning rate
    beta1_pow = 1.0
    beta2_pow = 1.0

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution and sub‑gradient -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # mask of entries attaining the max (within tolerance)
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # gradient = 2 * (a * w_rev)  (convolution)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr / (np.sqrt(v_hat) + eps)

        cand = a - step * m_hat
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # accept candidate as current point (always move)
        a = cand

        # track best ever
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # ----- occasional symmetrisation -----
        if it % 1000 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
            a = sym

        # ----- random perturbation when stuck -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)
                sc = _score(a)
                if sc < best_sc:
                    best = a.copy()
                    best_sc = sc
                no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # final symmetrisation (helps a bit more)
    final = (best + best[::-1]) * 0.5
    final = _project_simplex(final)
    final_sc = _score(final)
    if final_sc < best_sc:
        best = final
        best_sc = final_sc

    return [float(x) for x in best]
