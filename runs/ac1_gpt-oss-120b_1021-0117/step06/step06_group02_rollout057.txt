    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)      # power‑of‑two ≥2n
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # --------------------------------------------------------
    # initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        t = np.arange(n0, dtype=np.float64)
        cur = np.exp(-t / 120.0)                         # a smooth decay

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s <= 0.0:
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur = cur / s                                      # sum = 1

    n = cur.shape[0]
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # --------------------------------------------------------
    # simple random‑transfer hill climbing
    # --------------------------------------------------------
    max_no_imp = 3000
    no_imp = 0
    while time.monotonic() < deadline:
        # pick two distinct indices and transfer a small amount
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.07   # up to 7 % of mass at i
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)          # ensure feasibility
        sc = _score(cand)

        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
            cur = cand
            no_imp = 0
        else:
            no_imp += 1
            # occasional uphill move to escape flat regions
            if rng.random() < 0.001:
                cur = cand

        # early exit if we beat the historic target
        if best_sc < 1.50274349:
            break

        # restart if we are stuck for a long time
        if no_imp > max_no_imp:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            no_imp = 0

    return [float(x) for x in best_vec]
