import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper functions (matrix‑free, linear memory)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max(v: np.ndarray, L: int) -> float:
    """Maximum of the auto‑convolution of v (real FFT)."""
    af = np.fft.rfft(v, L)
    conv = np.fft.irfft(af * af, L)[: 2 * v.shape[0] - 1]
    return float(conv.max())


def _conv_max_and_index(v: np.ndarray, L: int):
    """Maximum value and its index of the auto‑convolution of v."""
    af = np.fft.rfft(v, L)
    conv = np.fft.irfft(af * af, L)[: 2 * v.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

    # length & FFT size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            n = a0.size
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / s
        n = a0.size
    else:
        n = 2000
        a0 = rng.random(n)
        a0 = a0 / a0.sum()

    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    def _score(v):
        """Score = 2·n·max(conv) (sum(v) = 1)."""
        return 2.0 * n * _conv_max(v, L)

    # --------------------------------------------------------------
    # baseline – the supplied construction if present
    # --------------------------------------------------------------
    best_vec = a0.copy()
    best_score = _score(best_vec)

    # --------------------------------------------------------------
    # cheap parametric exploration (few shaped families)
    # --------------------------------------------------------------
    idx_arr = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # 1) sin‑power
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        a = np.sin(np.pi * (idx_arr + 0.5) / n) ** p
        if a.sum() == 0:
            continue
        a = a / a.sum()
        sc = _score(a)
        if sc < best_score:
            best_score, best_vec = sc, a.copy()

    # 2) parabola‑power (symmetric, zero at edges)
    for p in (0.5, 1.0, 2.0, 3.0):
        a = ((idx_arr + 1) * (n - idx_arr)) ** p
        if a.sum() == 0:
            continue
        a = a / a.sum()
        sc = _score(a)
        if sc < best_score:
            best_score, best_vec = sc, a.copy()

    # 3) power‑law descending (left‑heavy)
    for p in (0.5, 1.0, 2.0, 3.0):
        a = (n - idx_arr) ** p
        a = a / a.sum()
        sc = _score(a)
        if sc < best_score:
            best_score, best_vec = sc, a.copy()

    # 4) symmetric exponential decay
    for alpha in (0.001, 0.002, 0.005, 0.01, 0.02):
        a = np.exp(-alpha * np.abs(idx_arr - mid))
        a = a / a.sum()
        sc = _score(a)
        if sc < best_score:
            best_score, best_vec = sc, a.copy()

    # --------------------------------------------------------------
    # Simulated‑annealing style global search
    # --------------------------------------------------------------
    target = 1.50274349
    temp = 0.02                     # initial temperature
    decay = 0.9995                  # geometric cooling
    min_temp = 1e-7
    it = 0
    cur = best_vec.copy()
    cur_score = best_score

    while time.monotonic() < deadline:
        it += 1

        # ---- propose a move ------------------------------------------------
        if rng.random() < 0.68:                     # mass transfer (most frequent)
            i = rng.integers(0, n)
            if cur[i] <= 1e-15:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            # heavy‑tailed fraction of the mass at i
            frac = rng.random() ** 0.5
            delta = cur[i] * frac * 0.9
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
        else:                                        # Gaussian jitter
            sigma = 0.015
            cand = cur + rng.normal(scale=sigma, size=n)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()

        cand_score = _score(cand)
        dscore = cand_score - cur_score

        # ---- Metropolis acceptance ----------------------------------------
        accept = dscore < 0 or rng.random() < math.exp(-dscore / max(temp, 1e-12))
        if accept:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand.copy()
                if best_score < target:
                    break

        # ---- occasional symmetrisation ------------------------------------
        if it % 480 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best_score, best_vec = sym_score, sym.copy()
                    if best_score < target:
                        break

        # ---- cooling -------------------------------------------------------
        temp = max(min_temp, temp * decay)

        # ---- time guard ----------------------------------------------------
        if it % 1500 == 0 and time.monotonic() >= deadline:
            break

    # --------------------------------------------------------------
    # Final projected sub‑gradient refinement (few hundred iter.)
    # --------------------------------------------------------------
    cur = best_vec.copy()
    cur_score = best_score
    momentum = np.zeros_like(cur)
    beta = 0.9
    base_alpha = 0.6

    for it in range(1, 1201):
        max_val, max_idx = _conv_max_and_index(cur, L)
        # sub‑gradient of the peak
        grad = np.zeros_like(cur)
        j = max_idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * cur[j[mask]]
        # momentum / adaptive step
        momentum = beta * momentum + (1.0 - beta) * grad
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)
        cand_score = _score(cand)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand.copy()
                if best_score < target:
                    break
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # ------------------------------------------------------------------
    best_vec = _project_simplex(best_vec)   # final safety projection
    return [float(x) for x in best_vec]
