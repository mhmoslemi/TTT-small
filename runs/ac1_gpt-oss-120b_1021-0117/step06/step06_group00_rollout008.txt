import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # ensure exact sum=1 up to floating error
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast version of the evaluator (no clipping, assumes a>=0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # Determine length and FFT size
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Build a collection of deterministic initial candidates
    xs = np.arange(n, dtype=np.float64)
    candidates = []

    # 1) Uniform
    candidates.append(np.full(n, 1.0 / n))

    # 2) Triangular (peak in the centre)
    tri = np.minimum(xs + 1, n - xs)
    candidates.append(tri / tri.sum())

    # 3) Sine bump (zero at ends, peak centre)
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    candidates.append(sine / sine.sum())

    # 4) Raised cosine bump
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    candidates.append(bump / bump.sum())

    # 5) Exponential decays
    for alpha in (0.001, 0.003, 0.006):
        exp = np.exp(-alpha * xs)
        candidates.append(exp / exp.sum())

    # 6) Power laws
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        candidates.append(pw / pw.sum())

    # 7) Random (deterministic seed)
    rand = rng.random(n)
    candidates.append(rand / rand.sum())

    # 8) Parent if available
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            candidates.append(parent)

    # ------------------------------------------------------------------
    # Pick the best among the prepared candidates
    best_vec = None
    best_sc = float('inf')
    for vec in candidates:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # Optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.9               # momentum coefficient
    step0 = 0.8              # base step size
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---- convolution and sub‑gradient mask ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # ---- gradient (sub‑gradient of max‑convolution) ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- step size (inverse sqrt decay) ----
        step = step0 / math.sqrt(it)

        # ---- line search over a few scale factors ----
        local_best_vec = cur
        local_best_sc = cur_sc
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - step * factor * momentum)
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < local_best_sc:
                local_best_sc = cand_sc
                local_best_vec = cand

        # ---- acceptance ----
        if local_best_sc < cur_sc:
            cur, cur_sc = local_best_vec, local_best_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional simulated‑annealing acceptance
            prob = math.exp(-(local_best_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = local_best_vec, local_best_sc

        # ---- update global best ----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # ---- periodic symmetrisation ----
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ---- random redistribution when stuck ----
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
