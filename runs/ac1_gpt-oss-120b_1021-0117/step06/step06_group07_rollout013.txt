def _max_conv(a, L):
    af = np.fft.rfft(a, n=L)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Utility: maximal value of the (full) self‑convolution
# ------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max_{k} (a*a)[k] using an FFT of length L (L >= 2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return float(conv.max())
