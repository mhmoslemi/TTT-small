import time, math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return max value of a*a, its index, and the full convolution."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s                     # now sum = 1
    else:                               # simple fallback
        n_fallback = 2500
        p = 1.2
        a0 = (np.arange(n_fallback, 0, -1, dtype=np.float64) ** p)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length for a*a

    cur = a0.copy()
    cur_max, cur_idx, _ = _conv_max_and_index(cur, L)
    cur_sc = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_sc = cur_sc

    # -----------------------------------------------------------------
    # optimisation parameters
    # -----------------------------------------------------------------
    beta = 0.9                 # momentum
    base_alpha = 0.2           # base step size
    T = 1e-5                   # initial temperature for SA
    decay = 0.9999             # temperature decay per iteration
    sym_interval = 250
    no_improve = 0
    momentum = np.zeros_like(cur)

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # -------------------------------------------------------------
        # gradient based update
        # -------------------------------------------------------------
        cur_max, cur_idx, _ = _conv_max_and_index(cur, L)

        j = cur_idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur)
        grad[mask] = cur[j[mask]]
        if cur_idx % 2 == 0:
            mid = cur_idx // 2
            if mid < n:
                grad[mid] = 2.0 * cur[mid]

        momentum = beta * momentum + (1.0 - beta) * grad
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)

        cand = cur - alpha * momentum
        cand = _project_simplex(cand)
        cand_max = _conv_max_and_index(cand, L)[0]
        cand_sc = 2.0 * n * cand_max

        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / max(T, 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1

        # -------------------------------------------------------------
        # occasional random mass transfer (exploration)
        # -------------------------------------------------------------
        if rng.random() < 0.25:   # 25 % chance each iteration
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.6
                j = rng.integers(0, n)
                if j == i:
                    continue
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp_max = _conv_max_and_index(tmp, L)[0]
                tmp_sc = 2.0 * n * tmp_max
                accept = False
                if tmp_sc < cur_sc:
                    accept = True
                else:
                    prob = math.exp(-(tmp_sc - cur_sc) / max(T, 1e-12))
                    if rng.random() < prob:
                        accept = True
                if accept:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # -------------------------------------------------------------
        # periodic symmetrisation (helps stability)
        # -------------------------------------------------------------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_max = _conv_max_and_index(sym, L)[0]
            sym_sc = 2.0 * n * sym_max
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # -------------------------------------------------------------
        # restart from best with small noise if stuck too long
        # -------------------------------------------------------------
        if no_improve > 5000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            no_improve = 0

        # -------------------------------------------------------------
        # temperature schedule
        # -------------------------------------------------------------
        T *= decay

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # -----------------------------------------------------------------
    # Return the best vector found, as plain Python list
    # -----------------------------------------------------------------
    return [float(x) for x in best_vec]
