import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum == 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ------------------------------------------------------------
# Fast AC1 bound for a vector with sum ≈ 1 (no division by sum)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(convolution) for a sequence whose sum is ~1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)          # power‑of‑two > 2n‑1
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    Return a non‑negative step‑function (as a Python list of floats)
    that yields a low AC1 upper bound.
    """

    # leave a safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

    sym_interval = 250
    rand_interval = 300
    eps_decay = 0.9985

    split_time = start + (deadline - start) * 0.55
