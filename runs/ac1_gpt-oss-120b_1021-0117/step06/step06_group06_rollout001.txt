Implementation of _project_simplex:

def _project_simplex(v):
    # v: np.ndarray
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)  # leave margin

    # Determine initial length
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.size
        # Normalize to sum 1
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s == 0:
            init = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            init /= s
    else:
        n = 2000
        init = np.exp(-np.arange(n, dtype=np.float64) / 100.0)
        init = init / init.sum()

    # Parameters
    popsize = 15
    F = 0.8
    CR = 0.9
    # FFT length
    L = 1 << ((2 * n - 1).bit_length())

    # Initialize population
    pop = np.empty((popsize, n), dtype=np.float64)
    scores = np.empty(popsize, dtype=np.float64)
    for i in range(popsize):
        if i == 0:
            vec = init.copy()

            vec = rng.exponential(size=n)
            vec = np.maximum(vec, 0)
            vec /= vec.sum()

    best_vec = pop[best_idx].copy()

    target_score = 1.50274349
    # Main DE loop
    while time.monotonic() < deadline:
        # Generation

import time

# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum = 1, x ≥ 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# Fast AC1 bound (assumes sum(a) = 1)

def _score_vec(a: np.ndarray, L: int, n: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
