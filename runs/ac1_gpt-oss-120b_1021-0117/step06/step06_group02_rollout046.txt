import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Score of a sequence (the AC1 bound)
# ------------------------------------------------------------
def _ac1_score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv(a,a)) / (sum(a)^2) using FFT of length L."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    return float(2.0 * n * max_val / (s * s + 1e-30))

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # safety margin

        n_fallback = 2000

    cur = a0.copy()
    n = cur.shape[0]
    # FFT length (power of two ≥ 2*n)
    L = 1 << ((2 * n - 1).bit_length() + 1)
    # index map for gradient extraction

    cur_sc = _ac1_score(cur, L)
