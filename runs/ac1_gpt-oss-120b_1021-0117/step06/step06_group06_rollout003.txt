import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 score (sum of coefficients is assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(conv(a,a)) for a normalized to sum=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.25)

    # ---------- initial feasible vector ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: exponential decay
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)

    a0 = np.maximum(a0, 0.0)
    # early symmetrisation (helps the objective)
    a0 = (a0 + a0[::-1]) * 0.5
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s = 1.0
    a0 = a0 / s                     # sum = 1

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---------- optimisation hyper‑parameters ----------
    momentum = np.zeros_like(cur)
    beta = 0.95                     # momentum coefficient
    base_alpha = 0.9               # base step size
    eps = None
    eps_decay = 0.96
    max_trials = 10                # back‑tracking attempts
    sym_interval = 200
    no_improve = 0
    target = 1.50274349
    it = 0

    # ---------- main optimisation loop ----------
    while time.monotonic() < deadline:
        it += 1

        # convolution of cur with itself (via FFT)
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # temperature for smoothed max
        if eps is None:
            eps = max_val * 0.20 + 1e-12
        else:
            eps = max(eps * eps_decay, max_val * 1e-12)

        # compute soft‑max weights
        if eps < max_val * 1e-6 or it > 5000:
            mask = (conv_full == max_val)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w /= w_sum
        else:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w /= w_sum

        # gradient of the smoothed max (matrix‑free via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # pick the slice that corresponds to the gradient vector
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)

        # back‑tracking line search
        improved = False
        cand = None
        cand_sc = None
        step = alpha
        for _ in range(max_trials):
            cand_try = cur - step * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            step *= 0.5

        # occasional uphill move (simulated annealing)
        if not improved:
            cand_try = cur - step * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            temp = max(1e-4, 1.0 / math.sqrt(it))
            prob = math.exp(-(cand_sc - cur_sc) / temp) if cand_sc > cur_sc else 1.0
            if rng.random() < prob:
                cand = cand_try
                improved = True

        # accept if improvement (or accepted uphill)
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---------- periodic symmetrisation ----------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ---------- random mass transfer when stuck ----------
        if no_improve > 1000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.40
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # ---------- restart from best if hopeless ----------
        if no_improve > 3000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

        # safety time check
        if time.monotonic() >= deadline:
            break

    # ---------- final local refinement (many cheap random moves) ----------
    while time.monotonic() < deadline - 0.02:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.05
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            cur_sc = cand_sc

    # return as plain Python list
    return [float(x) for x in best_vec]
