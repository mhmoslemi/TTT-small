def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]

    sym_interval = 300
    restart_thresh = 4000
    improve_target = 1.50274349

    # --------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and smoothing weight -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * eps_factor, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient via FFT -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + it * 1e-4)      # gentle decay
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur + step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- acceptance -----
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # occasional uphill move (simulated annealing style)
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                no_improve = 0
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()

        # ----- random mass‑transfer when stagnating -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()

        # ----- full restart if hopelessly stuck -----
        if no_improve > restart_thresh:
            cur = best_vec + 0.005 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # return plain Python list
    return [float(x) for x in best_vec]
