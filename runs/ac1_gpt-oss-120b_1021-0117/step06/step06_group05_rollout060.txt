import time

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector with sum≈1 (fast via FFT)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv) for a with sum 1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of a smoothed max (log‑sum‑exp) of the convolution
# ------------------------------------------------------------
def _grad_and_eps(a: np.ndarray, L: int, eps: float | None,
                  eps_factor: float) -> tuple[np.ndarray, float, float]:
    """
    Compute gradient of the smoothed max of a*a and update epsilon.
    Returns (grad, max_conv, new_eps).
    """
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)

    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()

    if eps is None:
        eps = max_val * eps_factor
    else:
        eps = max(eps * 0.995, 1e-12)

    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)

    idx = (2 * n - 2) - np.arange(n)  # reverse‑index selection
    grad = 2.0 * grad_full[idx]
    return grad, max_val, eps
