import time

import numpy as np

# ---------- convolution helper (max of full convolution) ----------
def _max_conv(x: np.ndarray, L: int) -> float:
    """Return max of full convolution x*x using FFT of length L."""
    xf = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(xf * xf, n=L)[: 2 * x.size - 1]
    return float(conv.max())
