def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def ac1_score(a: np.ndarray) -> float:
    """
    Compute the AC1 bound 2*n*max(conv(a,a))/(sum(a)^2) efficiently via FFT.
    The vector may be any non‑negative numbers; the function returns np.inf
    if the sum is effectively zero.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Return a soft‑max weighting of conv (sum to 1)."""
    max_val = conv.max()
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def gradient_softmax(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max defined by weights w.
    Returns g of length n where n = len(a).
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

        g = gradient_softmax(cur, w, L_full)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # step size (decreasing with iteration count & gradient norm)
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # ----- backtracking line search -----
        improved = False
        alpha = alpha0
        for _ in range(max_trials):
            cand = cur - alpha * momentum
            cand = project_simplex(cand)
            cand_sc = ac1_score(cand)
            if cand_sc < cur_sc:
                improved = True
                break
            alpha *= 0.5

        # occasional uphill move if no improvement
        if not improved:
            cand = cur - alpha * momentum
            cand = project_simplex(cand)
            cand_sc = ac1_score(cand)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = project_simplex(sym)
            sym_sc = ac1_score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_sc = ac1_score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- restart with noise if hopeless -----
        if no_improve > 5000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ----- early exit -----
        if best_sc < target:
            break

        # safety check for time
        if time.monotonic() >= deadline:
            break

    # ----- final local random search (tight time budget) -----
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if best_vec[i] <= 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.3
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_simplex(cand)
        cand_sc = ac1_score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
        # break if we already beat the target
        if best_sc < target:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
