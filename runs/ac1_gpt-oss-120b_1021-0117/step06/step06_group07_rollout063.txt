import time

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Compute full linear convolution via FFT (length L >= 2n‑1)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv

# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    conv = _conv_full(a, L)
    return 2.0 * a.size * conv.max()

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                 # project onto simplex with sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_sc = _score(a, L)

    # optimisation hyper‑parameters
    step0 = 0.8
    beta = 0.9
    momentum = np.zeros_like(a)
    max_k = 50                # number of top convolution entries used for sub‑gradient
    sym_interval = 200        # how often to enforce symmetry
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its max -----
        conv_full = _conv_full(a, L)
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best observed vector -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- stop early if we beat the target -----
        if best_sc < target:
            break

        # ----- sub‑gradient based on the largest convolution entries -----
        # select indices of the top‑max_k values (or all if fewer)
        if max_k >= conv_full.size:
            idxs = np.arange(conv_full.size)
        else:
            idxs = np.argpartition(-conv_full, max_k - 1)[:max_k]

        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
        if idxs.size:
            g /= max(1, idxs.size)

        # ----- momentum update and projected step -----
        momentum = beta * momentum + (1.0 - beta) * g
        alpha = step0 / math.sqrt(it)
        a_tmp = a - alpha * momentum
        a_tmp = _project_simplex(a_tmp)          # keep on simplex (sum = 1)

        # ----- evaluate the trial vector -----
        cand_sc = _score(a_tmp, L)

        # ----- accept if it improves, else Metropolis‑style acceptance -----
        if cand_sc < cur_sc:
            a, cur_sc = a_tmp, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, a_tmp.copy()
                no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = a_tmp, cand_sc

        # ----- periodic symmetrisation (helps to flatten peaks) -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_sc = _score(a_sym, L)
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- occasional random mass transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate discovered (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
