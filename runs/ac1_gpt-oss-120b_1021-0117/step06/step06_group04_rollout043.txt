import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max(vec: np.ndarray, L: int):
    """Return (max value, index of max) of the linear convolution vec*vec."""
    af = np.fft.rfft(vec, L)
    conv = np.fft.irfft(af * af, L)[: 2 * vec.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv
