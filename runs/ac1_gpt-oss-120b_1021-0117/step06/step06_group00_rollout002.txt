import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a, L):
    """Return max of full autocorrelation of a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return float(conv.max())

        a0 = rng.random(n_fallback, dtype=np.float64)

    n = cur.size

    cur = (cur + cur[::-1]) * 0.5

    # best tracking
    best_vec = cur.copy()
    best_sc = 2.0 * n * _conv_max(best_vec, L)

    cur_sc = best_sc

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.6
    no_improve = 0
    it = 0
    target = 1.50274349

        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

        else:
            no_improve += 1
            # occasional simulated‑annealing acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ---------------- random perturbation when stuck --------------
        if no_improve > 3500:
            # add a small random jitter then re‑project
            jitter = rng.normal(scale=0.02, size=n)
            tmp = cur + jitter
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = 2.0 * n * _conv_max(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # ---------------- early exit ----------------------------------
        if best_sc < target:
            break

        # ----------------------------------------------------------------
        # safety check for time
        if time.monotonic() >= deadline:
            break

    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
