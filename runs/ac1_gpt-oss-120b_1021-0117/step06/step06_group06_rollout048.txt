import time, math, numpy as np

def _score_fft(a, L):
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    s = a.sum()
    return 2.0 * n * max_val / (s*s)
```

import time, math
import numpy as np

# ------------------------------------------------------------
# Utility functions (FFT based score, simple symmetrization)
# ------------------------------------------------------------
def _fft_score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    maxv = conv.max()
    s = a.sum()
    if s == 0.0:
        return float('inf')
    return 2.0 * n * maxv / (s * s)


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Make the vector symmetric (v + rev(v))/2."""
    return (v + v[::-1]) * 0.5


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Candidate generators (few parametric families)
# ------------------------------------------------------------
def _exp_family(n, r):
    """Geometric decay a_i = r**i, normalized."""
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _sym_exp_family(n, r):
    """Symmetric geometric decay."""
    half = (n + 1) // 2
    left = r ** np.arange(half, dtype=np.float64)
    if n % 2 == 0:
        full = np.concatenate([left, left[::-1]])
    else:
        full = np.concatenate([left, left[-2::-1]])
    full /= full.sum()
    return full


def _gauss_family(n, sigma):
    """Centered Gaussian, normalized."""
    x = np.arange(n, dtype=np.float64)
    cen = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - cen) / sigma) ** 2)
    a /= a.sum()
    return a


def _triangular_family(n, width):
    """Triangular bump, normalized."""
    x = np.arange(n, dtype=np.float64)
    cen = (n - 1) / 2.0
    a = np.maximum(0.0, 1.0 - np.abs(x - cen) / width)
    a /= a.sum()
    return a
