import time, math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries with sum Z."""
    v = np.asarray(v, dtype=np.float64)
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Maximum value of full self‑convolution using FFT
# ------------------------------------------------------------
def _max_conv_fft(a: np.ndarray, L: int) -> float:
    """Return max of a*a (full linear convolution) using an FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv  # also return full array when needed


# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution (taken at all indices attaining the max)
# ------------------------------------------------------------
def _subgrad_max(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a (using all maximising indices)."""
    n = a.size
    tol = max(1e-12, max_val * 1e-12)
    idxs = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    if idxs.size > 1:
        g /= idxs.size
    return g
