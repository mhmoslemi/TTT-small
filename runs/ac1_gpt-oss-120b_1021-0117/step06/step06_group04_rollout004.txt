import time
import math
import numpy as np

# ---------- helper utilities ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return max value, its index and the full convolution a*a (length 2n‑1)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def _conv_max(a: np.ndarray, L: int) -> float:
    """Fast max of a*a (a is already on the simplex)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    return float(conv.max())


# ---------- main entry point ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # -----------------------------------------------------------------
    # basic problem data
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s                     # sum = 1
    else:
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = a0.copy()
    best_score = 2.0 * n * _conv_max(best_vec, L)

    # -----------------------------------------------------------------
    # deterministic grid searches (quick, cheap)
    # -----------------------------------------------------------------
    idx_arr = np.arange(1, n + 1, dtype=np.float64)   # 1‑based indices

    # pure power laws
    for p in np.linspace(0.0, 5.0, 101):
        vec = idx_arr ** (-p)
        vec = vec / vec.sum()
        sc = 2.0 * n * _conv_max(vec, L)
        if sc < best_score:
            best_score, best_vec = sc, vec.copy()

    # pure exponentials
    for q in np.linspace(0.0, 0.02, 101):
        vec = np.exp(-q * (idx_arr - 1))
        vec = vec / vec.sum()
        sc = 2.0 * n * _conv_max(vec, L)
        if sc < best_score:
            best_score, best_vec = sc, vec.copy()

    # mixture of power and exponential
    for _ in range(300):
        p = rng.uniform(0.0, 5.0)
        q = rng.uniform(0.0, 0.02)
        w = rng.uniform(0.0, 1.0)
        vec = w * (idx_arr ** (-p)) + (1 - w) * np.exp(-q * (idx_arr - 1))
        vec = vec / vec.sum()
        sc = 2.0 * n * _conv_max(vec, L)
        if sc < best_score:
            best_score, best_vec = sc, vec.copy()

    # -----------------------------------------------------------------
    # stochastic search (random families)
    # -----------------------------------------------------------------
    # allocate a fraction of the remaining time to random search
    # (keep ~30 s for a final gradient refinement)
    random_deadline = deadline - 30.0
    if random_deadline < time.monotonic():
        random_deadline = time.monotonic() + 5.0   # ensure a tiny search

    while time.monotonic() < random_deadline:
        family = rng.integers(0, 4)   # 0: Dirichlet, 1: power, 2: expo, 3: mix
        if family == 0:  # Dirichlet / exponential‑Gamma
            vec = rng.exponential(scale=1.0, size=n)
        elif family == 1:  # power law
            p = rng.uniform(0.0, 5.0)
            vec = idx_arr ** (-p)
        elif family == 2:  # pure exponential
            q = rng.uniform(0.0, 0.02)
            vec = np.exp(-q * (idx_arr - 1))
        else:  # mixed power‑exponential
            p = rng.uniform(0.0, 5.0)
            q = rng.uniform(0.0, 0.02)
            w = rng.uniform(0.0, 1.0)
            vec = w * (idx_arr ** (-p)) + (1 - w) * np.exp(-q * (idx_arr - 1))

        # normalise onto the simplex
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        sc = 2.0 * n * _conv_max(vec, L)
        if sc < best_score:
            best_score, best_vec = sc, vec.copy()

        # occasional forced symmetrisation (helps regularity)
        if rng.random() < 0.01:
            sym = (vec + vec[::-1]) * 0.5
            sym = sym / sym.sum()
            sc = 2.0 * n * _conv_max(sym, L)
            if sc < best_score:
                best_score, best_vec = sc, sym.copy()

    # -----------------------------------------------------------------
    # local refinement via projected sub‑gradient steps
    # -----------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_score
    best_local = cur.copy()
    best_local_sc = cur_sc

    # allow at most ~30 s for this phase
    refine_deadline = time.monotonic() + min(30.0, deadline - time.monotonic())
    it = 0
    momentum = np.zeros_like(cur)
    beta = 0.85       # momentum coefficient
    base_alpha = 0.7 # step‑size factor

    while time.monotonic() < refine_deadline:
        it += 1
        cur_max, cur_idx, _ = _conv_max_and_index(cur, L)

        # sub‑gradient of max‑convolution
        grad = np.zeros_like(cur)
        j = cur_idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * cur[j[mask]]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)

        # trial step
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)
        cand_sc = 2.0 * n * _conv_max(cand, L)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_local_sc:
                best_local_sc, best_local = cand_sc, cand.copy()
            continue

        # occasional random mass transfer when no improvement
        i = rng.integers(0, n)
        if cur[i] > 0.0:
            delta = cur[i] * rng.random() * 0.25
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = 2.0 * n * _conv_max(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_local_sc:
                    best_local_sc, best_local = tmp_sc, tmp.copy()
                continue

        # very occasional symmetrisation
        if rng.random() < 0.02:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = 2.0 * n * _conv_max(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_local_sc:
                    best_local_sc, best_local = sym_sc, sym.copy()

    # -----------------------------------------------------------------
    # pick the best among all candidates found
    # -----------------------------------------------------------------
    final_vec = best_vec
    # compare with refined version
    if best_local_sc < best_score:
        final_vec = best_local

    # final safety check (evaluate_sequence will clamp values)
    try:
        # ensure list of floats
        out = [float(x) for x in final_vec]
        # one final call to evaluator to guard against hidden issues
        _ = evaluate_sequence(out)
        return out
    except Exception:
        # fallback to uniform vector if something went wrong
        fallback = [1.0 / n] * n
        return fallback
