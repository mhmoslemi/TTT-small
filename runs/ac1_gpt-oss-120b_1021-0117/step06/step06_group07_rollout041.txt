    return raw * scale

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Generate a step‑function heights vector with a low AC1 bound."""
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # length to use – prefer the provided parent if present
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2048                     # reasonable fallback size

    # ---------- primary construction (linear decreasing) ----------
    best_vec = _decreasing_sequence(n)
    best_score = evaluate_sequence(best_vec.tolist())

    # ---------- compare with the parent construction if available ----------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.size == n:        # same length -> direct comparison
            parent_score = evaluate_sequence(parent.tolist())
            if parent_score < best_score:
                best_vec = parent.copy()
                best_score = parent_score

    # ---------- cheap local refinement while time permits ----------
    rng = np.random.default_rng(seed + 1)
    it = 0
    # move a tiny amount of mass between two positions; accept only improvements
    while time.monotonic() < deadline and it < 2000:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if best_vec[i] <= 0.0:
            it += 1
            continue
        # up‑to‑2 % of the source entry is relocated
        delta = best_vec[i] * rng.random() * 0.02
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_vec = cand
            best_score = score
        it += 1

    # ---------- return the best sequence found ----------
    return [float(v) for v in best_vec]
