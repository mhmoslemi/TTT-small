def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    # find rho
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score_simple(a, L=None):
    """Fast AC1 score for a vector already normalised to sum=1."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # -------------------- initialization --------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        # start from a symmetrised version (helps the autocorrelation)
        a0 = (a0 + a0[::-1]) * 0.5
    else:
        # fallback: moderate‑size random vector
        a0 = rng.random(2000, dtype=np.float64)

    cur = _project_simplex(a0)          # sum = 1, non‑negative
    n = cur.shape[0]
    # FFT length large enough for the (a * w_rev) convolution (n + (2n‑1) -1 = 3n‑2)
    L = 1 << ((3 * n - 1).bit_length())

    momentum = np.zeros_like(cur)
    beta_mom = 0.9           # momentum coefficient
    step0 = 0.9              # initial step size factor
    beta_soft = 10.0         # starting inverse temperature for soft‑max weights
    beta_soft_inc = 5e-4     # linear increase per iteration (≈ 5 after 10k iters)

    best_vec = cur.copy()
    best_sc = _score_simple(best_vec, L)
    cur_sc = best_sc
    it = 0
    no_improve = 0
    target = 1.50274349

    # -------------------- optimisation loop --------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- current convolution & soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # gradually increase the softness (beta_soft) to focus on the true maximum
        beta_cur = beta_soft + beta_soft_inc * it
        # soft‑max over all lags
        w = np.exp(beta_cur * (conv_full - max_val))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to uniform weight on the (near) maximum entries
            eps = max_val * 1e-12 + 1e-15
            mask = conv_full >= max_val - eps
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the weighted max‑convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # extracts the needed slice
        grad = 2.0 * grad_full[idx]               # gradient of the objective

        # ----- momentum update -----
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        # ----- step size (sqrt‑decay) -----
        step = step0 / math.sqrt(it)

        # ----- two candidate updates: Euclidean‑proj & exponentiated -----
        # (a) Euclidean projection onto the simplex
        cand_proj = _project_simplex(cur - step * momentum)

        # (b) Exponentiated‑gradient (mirror descent) – always stays non‑negative
        exp_arg = -step * grad
        # clip to avoid overflow

        conv_proj = np.fft.irfft(cand_proj_fft * cand_proj_fft, n=L)[: 2 * n - 1]
        cand_proj_sc = float(2.0 * n * conv_proj.max())

        conv_exp = np.fft.irfft(cand_exp_fft * cand_exp_fft, n=L)[: 2 * n - 1]

                    best_vec, best_sc = sym.copy(), sym_sc

        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                conv_tmp = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]

                if tmp_sc < cur_sc:

        # ----- early termination if target reached -----

            break

        # safety guard against runaway loops
