def _score_vector(vec):
    """Utility to compute the AC1 bound using the provided evaluator."""
    # The evaluator is pre‑imported in the execution environment.
    return evaluate_sequence(vec)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to minimize
    the AC1 upper bound defined by `evaluate_sequence`.

    The implementation is deliberately lightweight:
      * If a parent construction `height_sequence_1` is available, it is
        evaluated.
      * A uniform probability distribution of the same length is also evaluated.
      * The better of the two candidates is returned.

    The whole routine respects the time budget and exits well before the
    sandbox timeout.
    """
    start = time.monotonic()
    # leave a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # Determine problem size
    if "height_sequence_1" in globals():
        parent = [float(x) for x in height_sequence_1]
        n = len(parent)
        have_parent = True
    else:
        # fallback size – any reasonable length works
        n = 2000
        parent = None
        have_parent = False

    # ------------------------------------------------------------------
    # Candidate 1: uniform distribution (sum = 1)
    uniform = [1.0 / n] * n

    # ------------------------------------------------------------------
    # Evaluate candidates (only if time permits)
    best_vec = uniform
    best_score = _score_vector(uniform)

    if have_parent and time.monotonic() < deadline:
        parent_score = _score_vector(parent)
        if parent_score < best_score:
            best_vec, best_score = parent, parent_score

    # ------------------------------------------------------------------
    # Return the best candidate discovered

    return best_vec
