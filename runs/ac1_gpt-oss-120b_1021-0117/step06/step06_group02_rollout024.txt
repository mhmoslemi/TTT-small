import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a) ~ 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(convolution) for a sequence a (sum≈1)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # initialise a feasible point on the simplex (symmetrized)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:  # fallback: exponential decay
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    # enforce symmetry initially
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = _project_simplex(a0)

    # --------------------------------------------------------
    # FFT parameters
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length() + 1)   # power‑of‑two > 2n‑1
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    # --------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.5
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)
    beta1_pow = 1.0
    beta2_pow = 1.0

    cur = a0.copy()
    cur_score = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_score

    # smoothing parameter for soft‑max gradient
    eps = None

    no_improve = 0

    while time.monotonic() < deadline:
        t_iter += 1
        # ----------------------------------------------------
        # convolution of current iterate
        # ----------------------------------------------------
        cur_fft = np.fft.rfft(cur, n=L)

        max_val = conv_full.max()

                        cur_score = tmp_sc
                        no_improve = 0

        # ----------------------------------------------------
        if no_improve > 3000:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)

        # ----------------------------------------------------

            break

    # return the best found sequence as a plain Python list
    return [float(x) for x in best_vec]
