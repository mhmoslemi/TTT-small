import time

# ------------------------------------------------------------
# Simplex projection (Euclidean)  sum(x)=1, x>=0
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast AC1 score assuming sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # cheap parametric families (randomised)
    # ------------------------------------------------------------------
    for _ in range(200):
        # single exponential
        alpha = 10 ** rng.uniform(-4, 0)
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand /= cand.sum()
        _maybe_update(cand)

    for _ in range(200):
        # symmetric exponential (distance from nearest end)
        alpha = 10 ** rng.uniform(-4, 0)
        d = np.minimum(np.arange(n, dtype=np.float64),
                       n - 1 - np.arange(n, dtype=np.float64))
        cand = np.exp(-alpha * d)
        cand /= cand.sum()
        _maybe_update(cand)

    for _ in range(200):
        # power‑law tail
        p = rng.uniform(0.0, 2.0)
        cand = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        cand /= cand.sum()
        _maybe_update(cand)

    for _ in range(200):
        # beta‑like shape
        a = rng.uniform(0.0, 5.0)
        b = rng.uniform(0.0, 5.0)
        i = np.arange(n, dtype=np.float64)
        cand = (i ** a) * ((n - 1 - i) ** b)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        _maybe_update(cand)

    for _ in range(200):
        # random Dirichlet draw
        gam = rng.gamma(1.0, size=n)
        cand = gam / gam.sum()
        _maybe_update(cand)

    # ------------------------------------------------------------------
    # Gradient‑based refinement (soft‑max smoothing)
    # ------------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)

    beta = 0.9                     # momentum coefficient
    base_alpha = 0.8               # step‑size scaling
    eps = None
    eps_decay = 0.996
    sym_interval = 30
    max_trials = 12
    it = 0
    no_improve = 0
    target = 1.50274349

    L = 1 << ((2 * n - 1).bit_length())   # FFT length (constant)

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.2, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)           # alignment for gradient
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (decaying) -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # ----- backtracking line search -----
        improved = False
        cand = None
        alpha = alpha0
        for _ in range(max_trials):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                cur_sc = cand_sc
                improved = True
                break
            alpha *= 0.5

        # ----- occasional uphill move (Metropolis) -----
        if not improved:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                cur_sc = cand_sc
                improved = True

        if improved:
            cur = cand
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps a lot) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- restart from best with small noise if hopeless -----
        if no_improve > 6000:
            cur = best_vec + 0.008 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return as plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
