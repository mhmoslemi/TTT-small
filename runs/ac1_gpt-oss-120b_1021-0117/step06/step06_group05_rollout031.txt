def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3) 

    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # FFT length
    L = 1 << ((2 * n - 1).bit_length())

    def score_vec(vec):
        # Ensure ndarray, maybe copy
        a = np.asarray(vec, dtype=np.float64)
        sum_a = a.sum()
        if sum_a < 0.01:
            return float('inf')
        # Use FFT for speed
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_b = conv.max()
        return float(2.0 * n * max_b / (sum_a * sum_a))

    # Initial best: parent if present
    best_vec = None
    best_score = float('inf')
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        # clip negative (should not be) and enforce small epsilon for sum
        parent = np.maximum(parent, 0.0)
        sc = score_vec(parent)
        best_vec = parent.copy()
        best_score = sc

    # Helper to evaluate exponential shapes
    def try_exponential(c, p=1.0):
        # avoids overflow
        idx = np.arange(n, dtype=np.float64)
        # compute exponent term
        # Exponential decay: r = exp(-c) per n? Actually we want a_i = exp(-c * (i/(n-1))**p)
        # Use normalized index t = i/(n-1)
        t = idx / (n - 1)
        a = np.exp(-c * np.power(t, p))
        a /= a.sum()
        return a

    # Search grid
    # Coarse grid for c
    for c in np.linspace(0.2, 6.0, 39):   # step ~0.15

        a = try_exponential(c, p=1.0)

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative coefficient list with a low AC1 bound.

    The routine searches over simple exponential families
    a_i ∝ exp(-c·i/(n-1)) (and a few variants) using an FFT‑based
    evaluation of the bound.  It keeps the best vector found before the
    time deadline and falls back to the provided parent construction if any.
    """
    start = time.monotonic()
    # keep a small safety margin for the harness timeout
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # ------------------------------------------------------------
    # problem size and FFT length
    # ------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        # generous fallback length
        n = 2000

    # length for a power‑of‑two FFT that covers the full linear convolution
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # fast bound evaluator (identical to evaluate_sequence but linear‑time)
    # ------------------------------------------------------------
    def _fast_score(vec: np.ndarray) -> float:
        a = np.asarray(vec, dtype=np.float64)
        s = a.sum()
        if s < 0.01:
            return float("inf")
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_b = conv.max()
        return float(2.0 * n * max_b / (s * s))

    # ------------------------------------------------------------
    # initial best (parent if present)
    # ------------------------------------------------------------
    best_vec = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        sc = _fast_score(parent)
        best_vec = parent.copy()
        best_score = sc

    # ------------------------------------------------------------
    # helper to build an exponential‑type candidate
    # ------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)  # reused index array

    def _make_exp(c: float, p: float = 1.0) -> np.ndarray:
        """a_i ∝ exp(-c·(i/(n‑1))**p).  Returned array is normalized to sum=1."""
        t = idx / (n - 1)                     # in [0,1]
        a = np.exp(-c * np.power(t, p))
        a_sum = a.sum()
        if a_sum == 0.0:
            return a  # degenerate case, will be rejected by scorer
        return a / a_sum

    # ------------------------------------------------------------
    # coarse search over decay parameter c (p=1)
    # ------------------------------------------------------------
    # range of c: from very mild decay up to about ln(n) * 2,
    # which comfortably contains the optimum.
    c_max = max(12.0, 2.0 * np.log(n))
    for c in np.linspace(0.1, c_max, 80):
        if time.monotonic() > deadline:
            break
        cand = _make_exp(c, p=1.0)
        sc = _fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            best_c = c
            best_p = 1.0

    # ------------------------------------------------------------
    # fine refinement around the best c found so far
    # ------------------------------------------------------------
    if best_vec is not None and time.monotonic() < deadline:
        # explore a tighter neighbourhood
        lo = max(0.01, best_c - 2.0)
        hi = best_c + 2.0
        for c in np.linspace(lo, hi, 60):
            if time.monotonic() > deadline:
                break
            cand = _make_exp(c, p=1.0)
            sc = _fast_score(cand)
            if sc < best_score:
                best_score = sc
                best_vec = cand.copy()
                best_c = c

    # ------------------------------------------------------------
    # additional exploration of power‑law exponents p
    # ------------------------------------------------------------
    if time.monotonic() < deadline:
        for p in (0.5, 0.8, 1.2, 1.5, 2.0):
            # coarse c sweep for each p
            for c in np.linspace(0.2, c_max, 50):
                if time.monotonic() > deadline:

                cand = _make_exp(c, p=p)

                    best_vec = cand.copy()
                    best_c, best_p = c, p

    # ------------------------------------------------------------

    # final sanity check – ensure the returned list is valid
    # ------------------------------------------------------------
    # enforce non‑negativity and a tiny lower bound on the sum
    best_vec = np.maximum(best_vec, 0.0)

    # return plain Python floats as required by the evaluator
    return [float(x) for x in best_vec]
