import cvxpy as cp

def solve_lp(a, M):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # constraints: convolution of a and b <= M elementwise.
    # For each shift k from 0 to 2n-2:
    #   sum_{i=0}^{n-1} a[i] * b[k-i] (where index j = k-i must be within [0,n-1]) <= M
    # Equivalent to: sum_{i=max(0,k-n+1)}^{min(k, n-1)} a[i] * b[k-i] <= M
    constraints = []
    for k in range(2*n-1):
        i_min = max(0, k - (n-1))
        i_max = min(k, n-1)
        # Build expression:
        # selection of b indices [k-i] for i in i_range
        # Could use cp.sum(cp.multiply(a[i_range], b_selected))
        idx_range = np.arange(i_min, i_max+1)
        b_idx = k - idx_range
        # Use cp.sum(cp.multiply(a[idx_range], b[b_idx]))
        constraints.append(cp.sum(cp.multiply(a[idx_range], b[b_idx])) <= M)
    # Objective = maximize sum(b)
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    # Solve with a fast solver; use ECOS or SCS.
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if b.value is None:
        return None
    return np.maximum(np.array(b.value, dtype=float), 0.0)
```

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Convolution + gradient of the smooth max (log‑sum‑exp)
# ------------------------------------------------------------
def _conv_and_gradient(a: np.ndarray, eps: float, L: int, n: int):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    # soft‑max weights
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return conv, max_val, grad

    af = np.fft.rfft(a, n=L)
