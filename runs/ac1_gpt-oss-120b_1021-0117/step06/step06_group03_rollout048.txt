import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex using the O(n log n) algorithm."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Gradient of a smoothed max of the self‑convolution
# ------------------------------------------------------------
def _grad_and_max(a: np.ndarray, eps: float, L: int, n: int):
    """Return (grad, max_conv) where grad is the smoothed‑max gradient."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()

    # softmax weights centred at the maximum
    w = np.exp((conv_full - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]

    return grad, max_val

# Max of the self‑convolution (no gradient)

def _max_conv(a: np.ndarray, L: int, n: int) -> float:
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(conv_full.max())
