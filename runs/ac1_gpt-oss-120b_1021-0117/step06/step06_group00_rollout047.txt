def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # initialization:
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random length maybe 2000
        cur = rng.random(2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    n = cur.size
    L = 1 << ((2*n - 1).bit_length())

    # best
    conv_full = np.fft.irfft(np.fft.rfft(cur, n=L) * np.fft.rfft(cur, n=L), n=L)[:2*n-1]
    best_sc = 2.0 * n * conv_full.max()
    best_vec = cur.copy()
    best_iter = 0

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eta = 0.05
    eps_adam = 1e-8
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # FFT of current
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_conv = conv_full.max()

        # gradient weights
        if it < 1500:
            alpha = min(80.0, 5.0 + it * 0.02)  # gradually increase
            # Softmax
            # subtract max for stability
            max_val = conv_full.max()
            exps = np.exp(alpha * (conv_full - max_val))
            w = exps / exps.sum()
        else:
            # hard max weight
            eps = max_conv * 1e-12 + 1e-15
            mask = conv_full >= max_conv - eps
            cnt = mask.sum()
            if cnt == 0:
                w = np.ones_like(conv_full) / conv_full.size
            else:
                w = mask.astype(np.float64) / cnt

        # compute gradient via convolution with w_rev
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = eta / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        # evaluate candidate score
        a_fft_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)[:2*n-1]
        cand_sc = 2.0 * n * conv_c.max()

        # acceptance (always accept if better)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            best_iter = it
            no_improve = 0
        else:

        # Update current for next iter (could be cand or keep cur)

            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                best_iter = it
                no_improve = 0
            if sym_sc < max_conv:
                cur = sym

        # occasional random perturbation if stuck

                    best_iter = it

            break

        # check time

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
