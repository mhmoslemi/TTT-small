def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.1)  # small margin

    # init sequence
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full_like(a, 1.0/a.size)
            s = 1.0
        a = a / s
    else:
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a = np.exp(-t/100.0)
        a = a / a.sum()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    # compute initial score
    cur = a.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # parameters
    eps = None
    eps_factor = 0.02
    eps_decay = 0.9995
    base_alpha = 0.5
    beta = 0.9
    momentum = np.zeros_like(cur)
    sym_interval = 200   # more frequent symmetrisation
    rand_move_prob = 0.02
    no_improve = 0
    # thresholds
    mass_transfer_thresh = 3000
    restart_thresh = 5000

    # main loop
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n - 1]
        max_val = conv_full.max()

        # smoothing temperature
        if eps is None:
            eps = max(max_val * eps_factor, 1e-15)
        else:
            eps = max(eps * eps_decay, 1e-15)

        # compute softmax weights for smoothed max gradient
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum
        momentum = beta * momentum + (1 - beta) * g

        # step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / math.sqrt(it) / norm_m

        # line search
        improved = False
        cand = None
        cand_sc = None
        # try decreasing alphas
        for _ in range(10):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # occasional uphill acceptance
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            # prob based on increase
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # symmetrization
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass transfer
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # mass transfer when stuck
        if no_improve > mass_transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                no_improve = 0

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a: np.ndarray) -> float:
    """AC1 bound for a vector with sum≈1 (uses FFT for speed)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
