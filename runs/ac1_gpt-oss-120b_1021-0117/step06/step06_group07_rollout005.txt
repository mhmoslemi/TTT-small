def _project_simplex(v, z=1.0):
    # v: np.array
    # returns projection onto {x >=0, sum(x)=z}
    # Use algorithm from Duchi et al.
    n = v.size
    if z < 0:
        raise ValueError('z must be non-negative')
    # Sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    # Find rho
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _subgrad_maxconv(a, conv, max_val, tol=1e-12):
    # conv: array length 2*n-1
    # returns gradient vector of shape n
    n = a.shape[0]
    # Determine indices where conv close to max_val

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        grad[i] += 2.0 * a[k - i]
    return grad

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Sub‑gradient of max‑convolution (using all maximal indices)
# ------------------------------------------------------------
def _subgrad_maxconv(a: np.ndarray, conv: np.ndarray, max_val: float,
                     tol: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of max_k (a*a)[k] at a.
    The sub‑gradient is 2 * sum_{k in argmax} a shifted by k.
    """
    n = a.shape[0]
    eps = tol * max_val if max_val > 0 else tol
    peak_idxs = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for k in peak_idxs:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        i = np.arange(i0, i1 + 1, dtype=np.int64)
        g[i] += 2.0 * a[k - i]
    return g

# ------------------------------------------------------------

    """

    yields a low AC1 bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

    a = a / a.sum()                     # enforce sum = 1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # current score (sum is 1, so denominator is 1)
    conv_a = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[: 2 * n - 1]
    cur_score = 2.0 * n * conv_a.max()
    best_vec = a.copy()
    best_sc = cur_score

    # optimisation parameters
    step0 = 0.6                # initial step magnitude
    beta = 0.9                 # Nesterov momentum factor
    it = 0
    no_improve = 0
    target = 1.50274349
    stagnation_limit = 3000
    sym_interval = 100        # enforced symmetry every sym_interval iters

    a_prev = a.copy()

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # Nesterov look‑ahead point
        y = a + beta * (a - a_prev)

        # Convolution at y
        y_fft = np.fft.rfft(y, n=L)
