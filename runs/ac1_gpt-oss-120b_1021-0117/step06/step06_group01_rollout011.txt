def _project_simplex(v):
    u = np.sort(v)[::-1]

    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]

        theta = 0

        rho = rho[-1]

    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector with sum == 1
# ------------------------------------------------------------
def _ac1_bound(a: np.ndarray, L: int, n: int) -> float:
    """Return 2 * n * max(conv(a,a)).  a is assumed non‑negative with sum 1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin for returning the result
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initialise a feasible point (sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:                                   # fallback: uniform distribution
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())      # power‑of‑two for FFT

    # ------------------------------------------------------------------
    # keep the best solution seen so far
    # ------------------------------------------------------------------
    best_vec = a0.copy()
    best_sc = _ac1_bound(best_vec, L, n)

    # ------------------------------------------------------------------
    # quick exploration of a few parametric families (cheap, dense)
    # ------------------------------------------------------------------
    def _try_candidate(vec: np.ndarray):
        nonlocal best_vec, best_sc
        sc = _ac1_bound(vec, L, n)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    # monotone exponential
    for _ in range(80):
        alpha = 10 ** rng.uniform(-4, 0)
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand /= cand.sum()
        _try_candidate(cand)

    # symmetric exponential (decay from both ends)
    for _ in range(80):
        alpha = 10 ** rng.uniform(-4, 0)
        d = np.minimum(np.arange(n, dtype=np.float64),
                       n - 1 - np.arange(n, dtype=np.float64))
        cand = np.exp(-alpha * d)
        cand /= cand.sum()
        _try_candidate(cand)

    # Gaussian‑like
    mid = (n - 1) / 2.0
    for _ in range(80):
        sigma = rng.uniform(0.5, 50.0)
        cand = np.exp(-((np.arange(n, dtype=np.float64) - mid) ** 2)
                      / (2 * sigma ** 2))
        cand /= cand.sum()
        _try_candidate(cand)

    # cosine‑squared
    for _ in range(80):
        cand = np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** 2
        cand /= cand.sum()
        _try_candidate(cand)

    # random convex combos of current best and a fresh exponential
    for _ in range(80):
        alpha = 10 ** rng.uniform(-4, 0)
        rand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        rand /= rand.sum()
        t = rng.random()
        cand = (1 - t) * best_vec + t * rand
        cand = _project_simplex(cand)
        _try_candidate(cand)

    # ------------------------------------------------------------------
    # Adam‑style gradient descent on the smoothed max (soft‑max surrogate)
    # ------------------------------------------------------------------
    cur = best_vec.copy()
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.25                     # base learning rate
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv_full.max()
        cur_sc = 2.0 * n * max_conv

        # ----- keep global best -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target:
            break

        # ----- soft‑max weights (temperature proportional to max) -----
        temperature = max(max_conv * 0.15, 1e-12)
        w = np.exp((conv_full - max_conv) / temperature)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)     # alignment for gradient entries
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        # occasionally inject tiny random noise (helps escape plateaus)
        if it % 400 == 0:
            noise = 0.001 * rng.standard_normal(n)
            cand_noise = cand + noise
            cand_noise = np.maximum(cand_noise, 0.0)
            cand_noise = _project_simplex(cand_noise)
            # quick check if noisy candidate is better
            sc_noise = _ac1_bound(cand_noise, L, n)
            if sc_noise < cur_sc:
                cand = cand_noise
                cur_sc = sc_noise

        cur = cand

        # ----- periodic symmetrisation (often reduces the bound) -----
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _ac1_bound(sym, L, n)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stuck for a long time -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _ac1_bound(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- occasional full restart from best with small perturbation -----
        if no_improve > 5000:
            cur = best_vec + 0.008 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
