def _simplex_normalize(x):
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s <= 0:
        # fallback to uniform
        return np.full_like(x, 1.0 / x.size)
    return x / s

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2) # safety margin.

    # Determine n.
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        # fallback: exponential decay length 2000.
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()
    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())

    # functions: simple projection, score
    def _project_simplex(v):
        # Euclidean projection onto simplex (optional)
        # Use simple scaling with non-negatives:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    def _score_vec(v):
        # v assumed sum=1
        a_fft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    
    cur = a0.copy()

    best_sc = cur_sc

    # hyperparameters

    sym_interval = 200
    max_backtrack = 15

    it = 0

    return [float(x) for x in best_vec]
