    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # Prepare length n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # FFT length
    L = 1 << ((2 * n - 1).bit_length())
    # Helper projection and score functions
    def proj(x):
        # project onto simplex sum=1
        # standard algorithm
        v = x.copy()
        # shift to make sum=1? Implementation using algorithm from original code
        # We'll use the same _project_simplex
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        return np.maximum(v - theta, 0.0)

    def score_vec(a):
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # Initialize candidate: use exponential shape search to get decent start
    best_vec = None
    best_sc = float('inf')
    # Simple search on epsilon space
    eps_vals = np.logspace(-4, -1.5, 80)  # from 1e-4 to about 0.0316
    indices = np.arange(n)
    half = (n - 1) / 2.0

    for eps in eps_vals:
        if time.monotonic() >= deadline:
            break
        r = 1.0 - eps
        a = np.power(r, indices, dtype=np.float64)
        a /= a.sum()
        sc = score_vec(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
        # symmetric version
        d = np.minimum(indices, n-1-indices)
        a2 = np.power(r, d, dtype=np.float64)
        a2 /= a2.sum()
        sc2 = score_vec(a2)
        if sc2 < best_sc:
            best_sc, best_vec = sc2, a2.copy()

    # If still not set, fallback uniform
    if best_vec is None:

        best_sc = score_vec(best_vec)

    cur = best_vec.copy()
    cur_sc = best_sc

        else:
            w = np.exp((conv_full - max_val) / eps_temp)

import time, math
import numpy as np

# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a vector whose sum = 1
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())
