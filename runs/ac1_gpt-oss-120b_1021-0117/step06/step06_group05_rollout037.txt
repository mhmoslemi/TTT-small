import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)≈1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a non‑negative sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Re‑order a vector so that large values sit at opposite ends
# ------------------------------------------------------------
def _antisymmetrize(v: np.ndarray) -> np.ndarray:
    """Place sorted values alternately at left and right ends."""
    s = np.sort(v)[::-1]                # descending
    out = np.empty_like(v)
    l, r = 0, v.size - 1
    toggle = True
    for val in s:
        if toggle:
            out[l] = val
            l += 1
        else:
            out[r] = val
            r -= 1
        toggle = not toggle
    return out


# ------------------------------------------------------------
# Gradient descent on a smoothed max‑convolution objective
# ------------------------------------------------------------
def _gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Return (best_vector, best_score) after a limited descent."""
    L = 1 << ((2 * n - 1).bit_length())
    cur = a0.copy()
    cur_sc = _score(cur)

    # hyper‑parameters (tuned a bit beyond the reference)
    eps_factor = 0.03
    eps_decay = 0.998
    beta = 0.9
    base_alpha = 0.8
    improve_target = 1.50274349

    eps = None
    momentum = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = cur_sc

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # ---- convolution of cur with itself ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ---- smoothing parameter ----
        if eps is None:
            eps = max(max_val * eps_factor, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # ---- soft‑max weights over all lags ----
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * g

        # ---- step size (decaying) ----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / math.sqrt(it) / norm_m

        # ---- backtracking line search ----
        improved = False
        cand = None
        cand_sc = None
        for _ in range(4):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # occasional uphill move (simulated‑annealing style)
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()

        # ---- periodic anti‑symmetrisation (helps to break symmetry) ----
        if it % 250 == 0:
            cur = _antisymmetrize(cur)
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = cur.copy()

        # ---- early exit if good enough ----
        if best_sc < improve_target:
            break

    return best_vec, best_sc


# ------------------------------------------------------------
# Simple local swap optimiser (hill‑climbing)
# ------------------------------------------------------------
def _local_swap_opt(
    a: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_swaps: int = 1500,
) -> tuple[np.ndarray, float]:
    """Random pairwise swaps, keep improvements."""
    cur = a.copy()
    cur_sc = _score(cur)
    for _ in range(max_swaps):
        if time.monotonic() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = _score(cur)
        if new_sc < cur_sc:
            cur_sc = new_sc
        else:
            cur[i], cur[j] = cur[j], cur[i]   # revert
    return cur, cur_sc
