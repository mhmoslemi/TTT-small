def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, cvxpy as cp, math

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    rng = np.random.default_rng(seed)

    # init a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12: a = np.full_like(a, 1.0 / a.size)
        else: a = a / s
    else:
        n = 2000
        a = np.exp(-np.arange(n)/100.0)
        a = a / a.sum()

    n = a.size
    # function to compute score
    def score(x):
        L = 1 << ((2*n - 1).bit_length())
        af = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max()), conv

    cur = a.copy()
    cur_sc, cur_conv = score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # hyperparameters
    t = 0.2
    # iteration loop
    it = 0
    no_imp = 0
    while time.monotonic() < deadline:
        it += 1

        # compute M
        M = cur_conv.max()

        # solve LP
        # if time low, break
        if time.monotonic() + 2.0 > deadline:  # leave time for final returns
            break

        # build and solve LP
        b = cp.Variable(n, nonneg=True)
        f_param = cp.Parameter(n, value=cur)  # constant copy of cur for this iteration
        conv_expr = cp.conv(f_param, b)  # shape 2n-1
        M_param = cp.Parameter()
        constraints = [conv_expr <= M_param]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            M_param.value = M
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
            b_opt = b.value
        except Exception:
            b_opt = None

        if b_opt is not None:
            sum_b = np.sum(b_opt)
            if sum_b > 0:
                # scale to sum 1
                g = b_opt / sum_b
                # convex combination
                new = (1 - t) * cur + t * g
                # enforce sum =1 (due to rounding)
                new = np.maximum(new, 0.0)
                new = new / new.sum()
                new_sc, new_conv = score(new)

                if new_sc < best_sc:
                    best_sc = new_sc
                    best = new.copy()
                    no_imp = 0
                else:
                    no_imp += 1

                cur = new
                cur_sc = new_sc
                cur_conv = new_conv

                # occasional symmetrize
                if it % 50 == 0:
                    sym = (cur + cur[::-1]) * 0.5
                    sym = sym / sym.sum()
                    sym_sc, sym_conv = score(sym)
                    if sym_sc < cur_sc:
                        cur = sym
                        cur_sc = sym_sc
                        cur_conv = sym_conv
                        if sym_sc < best_sc:
                            best_sc = sym_sc
                            best = sym.copy()
        else:
            # fallback: random perturb
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            new = np.maximum(new, 0.0)
            new = new / new.sum()
            new_sc, new_conv = score(new)
            if new_sc < best_sc:
                best_sc = new_sc
                best = new.copy()
                no_imp = 0
            cur = new
            cur_sc = new_sc
            cur_conv = new_conv

        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]

# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv
