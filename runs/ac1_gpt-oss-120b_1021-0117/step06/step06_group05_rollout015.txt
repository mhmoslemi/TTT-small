def run_opt(initial_a, max_iters, start_time, deadline):
    cur = initial_a copy
    n = len(cur)
    L = 1 << ((2*n - 1).bit_length())
    # Init hyperparameters
    m = np.zeros_like(cur) # first moment
    v = np.zeros_like(cur) # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.1   maybe adjust.
    eps_smooth = None
    eps_factor = 0.05
    eps_decay = 0.9995
    best_sc = _score(cur)
    best_vec = cur.copy()
    no_improve_counter = 0

    for it in range(1, max_iters+1):
        if time.monotonic() >= deadline:
            break
        # compute conv and smoothed gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        if eps_smooth is None:
            eps_smooth = max(max_val * eps_factor, 1e-12)
        else:
            eps_smooth = max(eps_smooth * eps_decay, 1e-12)
        w = np.exp((conv - max_val) / eps_smooth)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[:2*n-1]
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand = cur + step
        # project onto simplex
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_vec = cand.copy()
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        cur = cand
        # optionally periodic symmetrization or random moves
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                cur = sym

        # random perturbation when stuck
        if no_improve_counter > 2000:
            # generate random index pair and move mass
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i: j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    cur = tmp
                no_improve_counter = 0

    return best_vec, best_sc

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum≈1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution(a,a)) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Adam‑style optimizer for the smoothed max of the convolution
# ------------------------------------------------------------
def _optimize_run(start_vec: np.ndarray, rng: np.random.Generator, deadline: float):
    cur = start_vec.copy()
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.25

    # smoothing for the max
    eps = None
    eps_factor = 0.03
    eps_decay = 0.9995

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    no_improve = 0
    it = 0
    # main loop ------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and smoothed weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        if eps is None:
            eps = max(max_val * eps_factor, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-14)

        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        lr = lr0 / math.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur + step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- accept improvement or occasional uphill move -----
        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

                cur, cur_sc = tmp, tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

        # ----- safety break to avoid endless loops -----
        if it >= 30000:
