import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score_normed(a: np.ndarray) -> float:
    """2·n·max(a*a) for a with sum = 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    a0 = (a0 + a0[::-1]) * 0.5

    best_vec = a0.copy()
    best_sc = _score_normed(best_vec)

    # --------------------------------------------------------
    # helper to test a candidate and keep the best so far
    # --------------------------------------------------------
    def _try(vec: np.ndarray):
        nonlocal best_vec, best_sc
        sc = _score_normed(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    # --------------------------------------------------------
    # explore a few parametric families (cheap, dense search)
    # --------------------------------------------------------
    # monotone exponential
    for _ in range(300):
        alpha = 10 ** rng.uniform(-4, 0)
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand = cand / cand.sum()
        _try(cand)

    # symmetric exponential (decay from both ends)
    for _ in range(300):
        alpha = 10 ** rng.uniform(-4, 0)
        d = np.minimum(np.arange(n, dtype=np.float64),
                       n - 1 - np.arange(n, dtype=np.float64))
        cand = np.exp(-alpha * d)
        cand = cand / cand.sum()
        _try(cand)

    # Gaussian‑like
    mid = (n - 1) / 2.0
    for _ in range(300):
        sigma = rng.uniform(0.5, 80.0)
        cand = np.exp(-((np.arange(n, dtype=np.float64) - mid) ** 2)
                      / (2 * sigma ** 2))
        cand = cand / cand.sum()
        _try(cand)

    # cosine‑squared (high at ends, low in centre)
    for _ in range(300):
        cand = np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** 2
        cand = cand / cand.sum()
        _try(cand)

    # triangular shape
    for _ in range(300):
        d = np.abs(np.arange(n, dtype=np.float64) - mid) / mid
        cand = np.maximum(0.0, 1.0 - d)
        cand = cand / cand.sum()
        _try(cand)

    # beta‑like power law
    for _ in range(300):
        p = rng.uniform(0.2, 5.0)
        q = rng.uniform(0.2, 5.0)
        x = np.arange(n, dtype=np.float64) / (n - 1)
        cand = (x ** p) * ((1 - x) ** q)
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        _try(cand)

    # random convex combos of the current best and a fresh exponential
    for _ in range(500):
        alpha = 10 ** rng.uniform(-4, 0)
        rand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        rand = rand / rand.sum()
        t = rng.random()
        cand = (1 - t) * best_vec + t * rand
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5
        _try(cand)

    # --------------------------------------------------------
    # gradient‑based refinement (soft‑max surrogate of the max)
    # --------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)

    beta = 0.9                     # momentum coefficient
    base_alpha = 0.8               # initial step‑size scaling
    eps = None
    eps_decay = 0.995
    max_trials = 12
    it = 0
    no_improve = 0
    target = 1.50274349            # stop early if reached

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & soft‑max weights -----
        L = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.2, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-12)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)          # align for gradient
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (decaying) -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # ----- backtracking line search -----
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_try = (cand_try + cand_try[::-1]) * 0.5
            cand_sc = _score_normed(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # ----- occasional uphill move -----
        if not improved:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_try = (cand_try + cand_try[::-1]) * 0.5
            cand_sc = _score_normed(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- random mass‑transfer when stuck -----
        if no_improve > 1500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp_sc = _score_normed(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                cur = tmp
                cur_sc = tmp_sc
                no_improve = 0

        # ----- restart from best with small noise if hopeless -----
        if no_improve > 4000:
            cur = best_vec + 0.005 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur = (cur + cur[::-1]) * 0.5
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
