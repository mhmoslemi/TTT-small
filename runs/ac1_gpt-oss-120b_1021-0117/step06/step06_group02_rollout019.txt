import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum=1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Convolution (linear) – returns max value only
# ------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max of linear convolution a*a (full length)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.shape[0] - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Convolution + smoothed‑max gradient
# ------------------------------------------------------------
def _conv_max_grad(a: np.ndarray, L: int, eps: float | None):
    """Compute max convolution value and gradient of smoothed max."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = float(conv.max())

    # smoothing temperature
    if eps is None:
        eps_val = max(max_val * 0.5, 1e-9)
    else:
        eps_val = max(eps, 1e-12)

    # soft‑max weights
    w = np.exp((conv - max_val) / eps_val)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (a * rev(w))  (linear convolution)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    grad_idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[grad_idx]
    return max_val, grad
