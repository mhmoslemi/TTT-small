def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # Initial sequence
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        n0 = 2000
        t = np.arange(n0, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2*n - 1).bit_length())

    cur = a0.copy()
    cur_sc = _fft_score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Optimization hyperparameters
    base_alpha = 1.2   # step size scale
    beta = 0.95        # momentum
    momentum = np.zeros_like(cur)
    eps = None
    eps_decay = 0.998
    max_backtrack = 10
    stagnation = 0
    improve_target = 1.50274349

    while time.monotonic() < deadline:
        # conv and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n-1]
        max_val = conv.max()

        # temperature
        if eps is None:
            eps = max_val * 0.1 + 1e-12
        else:
            eps = max(eps * eps_decay, 1e-14)

        # softmax weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient: 2 * (a * w_rev) convolution (via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # indices to map to each a_i
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]   # shape n

        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad

        # step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / np.sqrt(time.monotonic() - start + 1) / norm_m  # alternative schedule
        # The denominator sqrt(time) ensures decreasing step.

        # line search
        improved = False
        cand = None
        cand_sc = None
        step_alpha = alpha
        for _ in range(max_backtrack):
            step = cur - step_alpha * momentum
            step = _project_simplex(step)
            sc = _fft_score(step)
            if sc < cur_sc:
                cand = step
                cand_sc = sc
                improved = True
                break
            step_alpha *= 0.5

        if not improved:
            # maybe random swap
            if rng.random() < 0.05:
                i = rng.integers(0, n)
                if cur[i] > 0:
                    delta = cur[i] * rng.random() * 0.3
                    j = rng.integers(0, n)
                    step = cur.copy()
                    step[i] -= delta
                    step[j] += delta
                    step = _project_simplex(step)
                    sc = _fft_score(step)
                    if sc < cur_sc:
                        cand = step
                        cand_sc = sc
                        improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # occasional random perturbation after stagnation
        if stagnation > 2000:
            # Heavy random perturbation
            perturb = rng.random(n) * 0.05
            tmp = cur + perturb
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            cur = tmp
            cur_sc = _fft_score(cur)
            stagnation = 0
            momentum = np.zeros_like(cur)
            eps = None

        # early exit
        if best_sc < improve_target:
            break

    # Return best_vector as list
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (ignoring the sum, which is 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence whose sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
