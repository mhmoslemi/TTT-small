import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (sum == 1, all >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score: 2 * n * max(convolution)  (assumes sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s                      # now sum == 1
    else:
        # Simple exponential decay if no parent is supplied
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    cur = a0.copy()
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())    # length for FFTs (next power of two)

    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    eps = None
    eps_factor = 0.06          # initial temperature fraction of current max

    base_alpha = 0.8

    momentum = np.zeros_like(cur)

    improve_target = 1.50274349

    while time.monotonic() < deadline:

        max_val = conv_full.max()

        # ----- temperature schedule (smooth approximation of max) -----
