    Prefer the provided parent `height_sequence_1`, otherwise use a smooth
    exponential decay.
    """
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
            s = 1.0
        a /= s
    else:
        n = 2500
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-t / 120.0)
        a /= a.sum()
    return a

def _score_from_max(n, max_val):
    """Score when the sum of coefficients equals 1."""
    return 2.0 * n * max_val

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that minimizes the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    cur = _init_sequence(rng)
    n = cur.shape[0]

    # FFT size – power of two ≥ 3n‑2 (for convolutions of length 2n‑1 with n)
    L = 1 << ((3 * n - 2).bit_length())

    # bookkeeping
    best_vec = cur.copy()
    # compute initial score
    cur_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    best_score = _score_from_max(n, max_val)

    # optimisation parameters
    it = 0
    no_improve = 0
    target_score = 1.50274349
    eta0 = 0.8                # initial learning‑rate factor
    sym_interval = 250        # how often to enforce symmetry
    random_interval = 800     # when to try a random mass transfer
    restart_interval = 2000   # when to re‑inject noise

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_score = _score_from_max(n, max_val)

        # ----- exact subgradient of the max (via mask) -----
        # mask of positions attaining the maximum (tiny tolerance for safety)
        tol = 1e-12
        mask = (conv_full >= max_val - tol).astype(np.float64)

        # convolution mask * reversed cur gives the subgradient
        cur_rev = cur[::-1]
        mask_fft = np.fft.rfft(mask, n=L)
        rev_fft = np.fft.rfft(cur_rev, n=L)
        grad_full = np.fft.irfft(mask_fft * rev_fft, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # length n

        # ----- mirror‑descent style update (exponential weights) -----
        eta = eta0 / math.sqrt(it)
        # normalise gradient to keep the exponent bounded (≤ eta)
        gmax = grad.max()
        if gmax > 0:
            step = eta * grad / gmax
        else:
            step = 0.0
        cand = cur * np.exp(-step)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = np.full_like(cur, 1.0 / n)
        else:
            cand /= cand_sum

        # ----- evaluate candidate -----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_score = _score_from_max(n, cand_conv.max())

        # ----- backtrack if not improving -----
        improved = cand_score < cur_score
        if not improved:
            # try a smaller step (half the learning rate)
            eta2 = eta * 0.5
            if gmax > 0:
                step2 = eta2 * grad / gmax
            else:
                step2 = 0.0
            cand2 = cur * np.exp(-step2)
            cand2 /= cand2.sum()
            cand2_fft = np.fft.rfft(cand2, n=L)
            cand2_conv = np.fft.irfft(cand2_fft * cand2_fft, n=L)[: 2 * n - 1]
            cand2_score = _score_from_max(n, cand2_conv.max())
            if cand2_score < cur_score:
                cand, cand_score = cand2, cand2_score
                improved = True

        # ----- accept or reject -----
        if improved:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()
        else:
            no_improve += 1

        # ----- periodic symmetrisation (often beneficial) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym /= sym.sum()
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_score = _score_from_max(n, sym_conv.max())
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()

        # ----- random mass transfer when stuck -----
        if no_improve > random_interval:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp /= tmp.sum()
            tmp_fft = np.fft.rfft(tmp, n=L)
            tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
            tmp_score = _score_from_max(n, tmp_conv.max())
            if tmp_score < cur_score:
                cur = tmp
                cur_score = tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best_score = tmp_score
                    best_vec = tmp.copy()

        # ----- restart from best with small noise when hopeless -----
        if no_improve > restart_interval:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()
            no_improve = 0

        # ----- early exit if target reached -----
        if best_score <= target_score:
            break

        # ----- time check every few hundred iterations -----
        if it % 500 == 0 and time.monotonic() >= deadline:
            break

    # Return plain Python list of floats
    return [float(x) for x in best_vec]
