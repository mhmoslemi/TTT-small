def _lp_refine(a, M, top_m, n):
    # a: current vector length n
    # M: current max conv
    # top_m: number of constraints to use (largest conv entries)
    # Build constraints matrix A
    idxs = np.argpartition(-conv, top_m - 1)[:top_m]
    data = []
    rows = []
    cols = []
    for row, k in enumerate(idxs):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # For each i in [i_start, i_end], column i has coeff a[k-i]
        col_indices = np.arange(i_start, i_end+1, dtype=int)
        coeffs = a[k - col_indices]
        rows.extend([row] * len(col_indices))
        cols.extend(col_indices.tolist())
        data.extend(coeffs.tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(top_m, n))
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M + 1e-12]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver='ECOS', warm_start=True, max_iters=1000)
    if g.value is not None:
        g_opt = np.maximum(g.value, 0)
        return g_opt
    else:
        return None
```

import time

import numpy as np
import scipy.sparse as sp

# ------------------------------------------------------------
# Simplex projection (Euclidean) onto {x >= 0, sum(x)=Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Compute max of linear convolution a*a (full) and exact
# sub‑gradient of the max (sum over all maximising indices)
# ------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray, L: int):
    """Return (max_val, grad) for a >= 0, sum(a)=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    max_val = conv.max()
    # indices attaining the maximum (within tolerance)
    idxs = np.where(np.abs(conv - max_val) <= 1e-12)[0]
    n = a.size
    grad = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        grad[i] += 2.0 * a[k - i]
    return float(max_val), grad

# ------------------------------------------------------------
# Small LP refinement using only the largest convolution entries
# ------------------------------------------------------------
def _lp_refine(a: np.ndarray, M: float, conv: np.ndarray, top_m: int, L: int):
    """Return a refined vector (or None) by solving a tiny LP."""
    n = a.size
    # select constraints at the top_m convolution values
    if top_m >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_m - 1)[:top_m]

    data, rows, cols = [], [], []
    for row, k in enumerate(idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        cols_row = np.arange(i_start, i_end + 1, dtype=int)
        coeffs = a[k - cols_row]
        rows.extend([row] * len(cols_row))
        cols.extend(cols_row.tolist())
        data.extend(coeffs.tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(idxs), n))

    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M + 1e-12])
    try:
        prob.solve(solver="ECOS", warm_start=True, max_iters=2000)
    except Exception:
        return None
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)
