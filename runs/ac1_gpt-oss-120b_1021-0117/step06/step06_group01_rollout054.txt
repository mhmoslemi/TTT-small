def _score(a):

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety

    # Initialize length n
    if "height_sequence_1" in globals():

        n = 2000

    best_vec = a0.copy()

        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    # initial family exploration - quick
    # exponential
    for _ in range(100):
        if not time.monotonic() < deadline: break
        alpha = 10 ** rng.uniform(-4, 0)
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand /= cand.sum()
        sc = _score(cand)
        _maybe_update(cand, sc)
    # symmetric exponential
    for _ in range(100):

# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
