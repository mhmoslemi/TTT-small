def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.25)  # small margin

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        a = rng.random(n, dtype=np.float64)
    # enforce nonneg and sum norm
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a[:] = 1.0
        total = a.sum()
    a = a / total

    n = a.size
    L = 1 << ((2*n-1).bit_length())   # power of two len for FFT

    # projection onto simplex
    def project_simplex(v, Z=1.0):
        # same as earlier
        if Z < 0:
            raise ValueError
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - Z
        rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # gradient topk
    def grad_topk(a, conv, top_k):
        n = a.size
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, top_k-1)[:top_k]
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            if i_start <= i_end:
                i = np.arange(i_start, i_end+1, dtype=int)
                # vectorized addition
                g[i] += 2.0 * a[k - i]
        g /= max(1, top_k)
        return g

    # Initialize best
    best_vec = a.copy()
    conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:2*n-1]
    best_sc = 2.0 * n * conv.max()
    cur_sc = best_sc

    # hyperparameters
    step0 = 0.5
    beta = 0.9
    momentum = np.zeros_like(a)
    top_k = 30
    sym_interval = 20   # more frequent symmetrization
    it = 0
    no_improve = 0
    target = 1.50274349

    # main loop
    while time.monotonic() < deadline:
        it += 1

        # convolution and max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=Z}."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Gradient from the K largest convolution entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    return g / max(1, top_k)
