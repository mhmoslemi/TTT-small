import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a candidate sequence for the AC1 bound.
    If the parent construction `height_sequence_1` is available we use it
    unchanged (it is already a high‑quality solution).  Otherwise we fall back
    to a simple uniform vector which yields the theoretical lower bound of 2
    for the evaluator.
    """
    # minimal latency – we ignore the random seed and the budget except for
    # the tiny safety margin required by the harness.
    _ = seed, kwargs  # unused but kept for the required signature.

    # ----------------------------------------------------------------------
    # use the provided parent sequence if present
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: a uniform distribution (score = 2)
        n = 2000
