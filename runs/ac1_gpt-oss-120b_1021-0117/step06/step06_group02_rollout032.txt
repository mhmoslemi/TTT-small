import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast score (exact AC1 bound) for a vector with sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a normalized (sum=1) sequence."""
    n = a.shape[0]
    # length for linear convolution: next power of two > 2n-1
    L = 1 << ((2 * n - 1).bit_length() + 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (approximately) minimizes the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------
    # initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    cur = np.full(n, 1.0 / n, dtype=np.float64)          # uniform start
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # simple stochastic improvement loop
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    sym_interval = 250
    max_iters = 5000

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # ----- random mass transfer between two positions -----
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cur[i] > 1e-12:
            delta = cur[i] * rng.random() * 0.6   # move up to 60 % of mass at i
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            cand_sc = _score(cand)

            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                cur = cand
                no_improve = 0
            else:
                no_improve += 1
                # occasional uphill step to escape plateaus
                if rng.random() < 0.03:
                    cur = cand

        # ----- periodic symmetrisation (often reduces the max) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                cur = sym
                no_improve = 0

        # ----- restart from the best if stuck for a while -----
        if no_improve > 2000:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            no_improve = 0

    # return as plain list of Python floats
    return [float(x) for x in best_vec]
