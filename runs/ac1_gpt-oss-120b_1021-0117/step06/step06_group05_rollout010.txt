import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum≈1, invariant to scaling)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    eps = None
    eps_factor = 0.5          # start with relatively smooth approximation
    eps_decay = 0.9995
    base_alpha = 0.9
    beta = 0.95
    momentum = np.zeros_like(cur)
    prev = cur.copy()
    sym_interval = 300
    improve_target = 1.50274349
    rand_move_prob = 0.015
    mass_transfer_thresh = 2000
    restart_thresh = 4000
    max_trials = 8
    sparse_trials = 30         # random sparse initialisations

    # --------------------------------------------------------
    # try a few random sparse starts before the main loop
    # --------------------------------------------------------
    for _ in range(sparse_trials):
        if time.monotonic() >= deadline:
            break
        k = rng.integers(5, 15)       # number of non‑zeros
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.random(k) + 1e-3
        vec = np.zeros(n, dtype=np.float64)
        vec[idx] = vals
        vec = vec / vec.sum()
        sc = _score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
            cur = vec.copy()
            cur_sc = sc
            momentum.fill(0.0)
            prev = cur.copy()
            eps = None

    # --------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & soft‑max temperature -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * eps_factor, 1e-14)
        else:
            eps = max(eps * eps_decay, 1e-14)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum (Nesterov‑style) -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # ----- candidate 1: Euclidean projected gradient -----
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            y = cur - alpha * momentum
            y = _project_simplex(y)
            sc = _score(y)
            if sc < cur_sc:
                cand = y
                cand_sc = sc
                improved = True
                break
            alpha *= 0.5

        # ----- candidate 2: multiplicative (mirror) update -----
        if not improved:
            step = alpha0
            a_exp = cur * np.exp(-step * grad)
            s_exp = a_exp.sum()
            if s_exp > 0.0:
                a_exp = a_exp / s_exp
                sc = _score(a_exp)
                if sc < cur_sc:
                    cand = a_exp
                    cand_sc = sc
                    improved = True

        # ----- uphill acceptance (simulated annealing) -----
        if not improved:
            y = cur - alpha * momentum
            y = _project_simplex(y)
            sc = _score(y)
            prob = math.exp(-(sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = y
                cand_sc = sc
                improved = True

        # ----- accept candidate -----
        if improved:
            prev = cur
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- occasional symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sc = _score(sym)
            if sc < cur_sc:
                prev = cur
                cur = sym
                cur_sc = sc
                if sc < best_sc:
                    best_sc = sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass transfer -----
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                sc = _score(tmp)
                if sc < cur_sc:
                    prev = cur
                    cur = tmp
                    cur_sc = sc
                    if sc < best_sc:
                        best_sc = sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # ----- mass‑transfer when stuck -----
        if no_improve > mass_transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                sc = _score(tmp)
                if sc < cur_sc:
                    prev = cur
                    cur = tmp
                    cur_sc = sc
                    if sc < best_sc:
                        best_sc = sc
                        best_vec = tmp.copy()
                no_improve = 0

        # ----- restart from best with small noise if hopeless -----
        if no_improve > restart_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            s = cur.sum()
            if s > 0.0:
                cur /= s
            prev = cur.copy()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # ------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
```
