import time, math
import numpy as np

# ------------------------------------------------------------
# Helper: compute the AC1 bound (sum of coefficients = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(convolution of a with itself).  Assumes sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Helper: gradient of a smoothed max(conv) w.r.t. a
# ------------------------------------------------------------
def _grad(a: np.ndarray, eps_factor: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad is ∂/∂a of the soft‑max of the
    convolution, using temperature eps_factor * max_conv.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max_conv * eps_factor + 1e-12

    # soft‑max weights for the convolution entries
    w = np.exp((conv - max_conv) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient of the weighted sum  w·conv   w.r.t. a
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    idx = (2 * n - 2) - np.arange(n)          # slice that yields ∂/∂a
    grad = 2.0 * grad_full[idx]
    return grad, max_conv

                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    no_improve = 0

        # ---------- periodic symmetrisation ----------
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            max_lc = np.log(sym + 1e-20).max()
            log_a = np.log(sym + 1e-20) - max_lc
            a = sym
            sym_score = _score(a)
            if sym_score < best_score:
                best_score = sym_score
                best_a = a.copy()
                no_improve = 0

        # ---------- random mass‑transfer when stuck ----------
        if no_improve > 1200:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp = tmp / tmp.sum()
                tmp_score = _score(tmp)
                if tmp_score < best_score:
                    best_score = tmp_score
                    best_a = tmp.copy()
                    a = tmp
                    max_lc = np.log(a + 1e-20).max()
                    log_a = np.log(a + 1e-20) - max_lc
                no_improve = 0

        # ---------- restart if hopeless ----------
        if no_improve > 3500:
            rand_vec = rng.exponential(scale=1.0, size=n)
            rand_vec = rand_vec / rand_vec.sum()
            max_lc = np.log(rand_vec + 1e-20).max()
            log_a = np.log(rand_vec + 1e-20) - max_lc
            a = rand_vec
            no_improve = 0

        # early exit if target already reached
        if best_score < 1.50274349:
            break

        # safety time check
        if time.monotonic() >= deadline:
            break

    # ---------- final local refinement (small random swaps) ----------
    a = best_a.copy()
    cur_score = best_score
    while time.monotonic() < deadline - 0.02:
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.random() * 0.05
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            continue
        cand = cand / cand.sum()
        cand_score = _score(cand)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

    # return plain Python list
    return [float(x) for x in best_a]
