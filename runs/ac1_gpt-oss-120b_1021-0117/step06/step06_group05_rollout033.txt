        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        if eps is None:
            eps = max(max_val * eps_factor, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)
        # softmax weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / math.sqrt(it) / norm_m
        # additive candidate
        cand_add = cur - alpha * momentum
        cand_add = _project_simplex(cand_add)
        cand_add_sc = _score(cand_add)
        # multiplicative candidate
        eta = mul_alpha / math.sqrt(it)
        cand_mul = cur * np.exp(-eta * g)
        # renormalize
        cand_mul = np.maximum(cand_mul, 0.0)
        sum_mul = cand_mul.sum()
        if sum_mul == 0:
            cand_mul = cur.copy()
        else:
            cand_mul = cand_mul / sum_mul
        cand_mul_sc = _score(cand_mul)
        # choose best among candidates
        if cand_add_sc < cur_sc or cand_mul_sc < cur_sc:
            if cand_add_sc <= cand_mul_sc:
                cur, cur_sc = cand_add, cand_add_sc
            else:
                cur, cur_sc = cand_mul, cand_mul_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # possibly accept uphill move (simulated annealing)
            uphill_prob = math.exp(-(cand_add_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < uphill_prob:
                cur, cur_sc = cand_add, cand_add_sc
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_vec = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # periodic symmetrization
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0
        
        # random small mass transfer
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
        
        # restart if stuck
        if no_improve > restart_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0
        
        if best_sc < target:
            break
    
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact AC1 bound for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(convolution) for a vector with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initial feasible point (normalised to sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback)

    # ------------------------------------------------------------------
    # Try a few random Dirichlet draws as extra seeds
    # ------------------------------------------------------------------
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    for _ in range(15):
        alpha = rng.uniform(0.1, 2.0)               # concentration parameter
        gamma = rng.gamma(alpha, 1.0, size=a0.size)
        cand = gamma / gamma.sum()
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    eps_factor = 0.20
    eps_decay = 0.9995
    eps = None

    beta = 0.90                     # momentum coefficient
    momentum = np.zeros_like(cur)

    base_alpha = 1.0                # additive step size factor
    mul_alpha = 0.5                 # multiplicative (exponential) step factor

    sym_interval = 200
    rand_move_prob = 0.02
    restart_thresh = 4000
    target = 1.50274349

    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution of cur with itself (FFT‑based)
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # temperature / smoothing for the max
        if eps is None:
            eps = max(max_val * eps_factor, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # soft‑max weights over the convolution entries
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:

        else:
            w /= w_sum

        # gradient of the smoothed max (via convolution with reversed weights)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]                # gradient vector

        # momentum update

        # ------------------------------------------------------------------

        if sum_mul == 0.0:
            cand_mul = cur.copy()

                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ------------------------------------------------------------------
        # Small random mass‑transfer moves to escape shallow minima
        # ------------------------------------------------------------------
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:

                    no_improve = 0

        # ------------------------------------------------------------------

        if no_improve > restart_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # ------------------------------------------------------------------
        # Early exit if we already beat the target
        # ------------------------------------------------------------------
        if best_sc < target:
            break

    # Return plain Python floats
    return [float(x) for x in best_vec]
