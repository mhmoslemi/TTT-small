import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, >=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    beta1, beta2 = 0.9, 0.999
    lr0 = 0.5
    epsadam = 1e-8
    eps_soft = None
    eps_decay = 0.995
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ---- convolution and soft‑max weights ----
        L = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps_soft is None:
            eps_soft = max(1e-9, max_val * 0.2)
        else:
            eps_soft = max(eps_soft * eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps_soft)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of smoothed max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)   # alignment
        grad = 2.0 * grad_full[idx]

        # ---- Adam step ----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + epsadam)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            # occasional uphill move (simulated annealing)
            temp = 1.0 / math.sqrt(it)
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- periodic symmetrisation ----
        if it % 100 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ---- random mass‑transfer when stuck ----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ---- restart from best with small jitter if hopeless ----
        if no_improve > 5000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.clip(cur, 0.0, None)
            cur = cur / cur.sum()
            m = np.zeros_like(cur)
            v = np.zeros_like(cur)
            eps_soft = None
            it = 0
            no_improve = 0

        # ---- early exit if target achieved ----
        if best_sc < target:
            break

    # ---------- final local random tweaks ----------
    for _ in range(300):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if best_vec[i] <= 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

    return [float(x) for x in best_vec]
