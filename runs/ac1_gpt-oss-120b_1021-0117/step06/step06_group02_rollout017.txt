        a0 = a0 / a0.sum()

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)   # ensure enough for convolution

    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    cur = a0.copy()

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    iter_cnt = 0
    no_imp = 0

    # smoothing temperature schedule
    temp = None

    while time.monotonic() < deadline:
        iter_cnt += 1

        # fft of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # temperature for softmax
        if temp is None:
            temp = max_val * 0.5
        else:
            temp = max(temp * 0.995, 1e-9)   # slowly decay

        # compute softmax weights
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:

import time, math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Fast AC1 bound for a sequence normalised to sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)   # power of two >2n
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
