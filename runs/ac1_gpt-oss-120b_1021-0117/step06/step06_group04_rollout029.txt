import time, math
import numpy as np

def _conv_max_and_idx(a: np.ndarray, L: int):
    """Return max value of a*a and the index where it occurs."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
