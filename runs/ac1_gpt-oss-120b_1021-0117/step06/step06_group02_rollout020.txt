        if epsilon is None:
            epsilon = max_val * 0.2
        else:

        # softmax weights
        w = np.exp((conv_full - max_val) / epsilon)
        w_sum = w.sum()

            cand = np.full_like(cur, 1.0 / n)
            cand_sum = 1.0
        cand = cand / cand_sum

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc

                    best_sc = sym_sc

                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                # remains non-negative
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best = tmp.copy()
                        best_sc = tmp_sc
                    no_improve = 0
                else:
                    # occasional acceptance
                    if rng.random() < 0.05:
                        cur = tmp
                        cur_sc = tmp_sc
                        no_improve = 0

            cur = best + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            no_improve = 0
            epsilon = None

        # early exit if reached target
        if best_sc < 1.50274349:
            break

    # convert to list of Python floats
    return [float(x) for x in best]

import numpy as np

# ------------------------------------------------------------
# AC1 bound (assumes any positive scaling of a is okay)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence a."""
    n = a.shape[0]
    # power‑of‑two length ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length() + 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
