import time

# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the AC1 bound (works for any non‑negative vector)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    return float(2.0 * n * max_val / (s * s))

                no_improve = 0

        # ----- targeted mass transfer to reduce the dominant autocorrelation -----
        if no_improve > 500:
            k_max = int(np.argmax(conv_full))
            i_low = max(0, k_max - n + 1)
            i_high = min(k_max, n - 1)
            idxs = np.arange(i_low, i_high + 1, dtype=int)
            if idxs.size:
                i_max = idxs[np.argmax(cur[idxs])]
                j_min = int(np.argmin(cur))
                if cur[i_max] > 0.0:
                    delta = cur[i_max] * rng.random() * 0.4
                    tmp = cur.copy()
                    tmp[i_max] -= delta
                    tmp[j_min] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur = tmp
                        cur_sc = tmp_sc
                        if tmp_sc < best_sc:
                            best_sc = tmp_sc
                            best_vec = tmp.copy()
                        no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- restart from best with small noise if hopeless -----
        if no_improve > 5000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

        # ----- deadline check (again) -----
        if time.monotonic() >= deadline:
            break

    # final symmetric polishing (optional)
    sym = (best_vec + best_vec[::-1]) * 0.5
    sym = _project_simplex(sym)
    if _score(sym) < best_sc:
        best_vec = sym

    # ------------------------------------------------------------
    # Return as a plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
