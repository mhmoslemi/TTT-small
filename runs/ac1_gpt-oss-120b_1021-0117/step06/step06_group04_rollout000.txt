def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.2)

    # init
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0:
            a = np.ones_like(a) / a.size
        else:
            a /= s
    else:
        n = 2000
        a = np.ones(n) / n
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    # compute initial score
    conv = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(a, L), L)[:2*n-1]
    max_conv = conv.max()
    best_score = 2.0 * n * max_conv / (a.sum()**2)
    best_vec = a.copy()

    # parameters
    beta = 0.2  # softmax temperature factor
    beta_max = 5.0
    eta0 = 0.5
    top_k = 10
    no_improve = 0
    improve_thresh = 2000  # after which we random perturb
    reinit_thresh = 5000
    # We'll track iteration count
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # compute convolution via FFT
        af = np.fft.rfft(a, L)
        conv = np.fft.irfft(af * af, L)[:2*n-1]
        # find top K indices
        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            part = np.argpartition(-conv, top_k-1)[:top_k]
            top_idx = part[np.argsort(-conv[part])]  # sorted descending
        # compute softmax weights
        vals = conv[top_idx]
        # scaling to avoid overflow
        maxv = vals.max()
        w = np.exp(beta * (vals - maxv))
        w /= w.sum()
        # compute gradient
        grad = np.zeros_like(a)
        indices = np.arange(n)
        for weight, idx in zip(w, top_idx):
            j = idx - indices
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]] * weight
        # exponentiated gradient update
        eta = eta0 / math.sqrt(it)
        # Clip gradient magnitude
        max_abs = np.max(np.abs(grad))
        if max_abs > 50:
            grad = grad * (50.0 / max_abs)
        a = a * np.exp(-eta * grad)
        # renormalize to simplex (sum=1)
        s = a.sum()
        if s == 0 or not np.isfinite(s):
            # fallback to uniform
            a = np.ones_like(a) / n
        else:
            a = a / s

        # compute new score
        af = np.fft.rfft(a, L)
        conv = np.fft.irfft(af * af, L)[:2*n-1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum()**2)

        # update best
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase softmax beta gradually
        if beta < beta_max:
            beta = min(beta_max, beta * 1.001)

        # occasional perturb
        if no_improve > improve_thresh:
            # random mass transfer between two random indices
            i = rng.integers(0, n)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                a[i] -= delta
                a[j] += delta
                # ensure positivity
                a = np.maximum(a, 0.0)
                a /= a.sum()
            no_improve = 0

        if no_improve > reinit_thresh:
            # reinitialize near best
            a = best_vec + 0.01 * rng.random(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

        # early stop
        if best_score < 1.50274349:
            break
    # return best as list
    return [float(x) for x in best_vec]

def _full_conv_self(a, L):
    af = np.fft.rfft(a, L)
    return np.fft.irfft(af * af, L)[:2*a.shape[0]-1]

def _full_conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a (length 2·len(a)-1) using an FFT of length L."""
    af = np.fft.rfft(a, L)
    return np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
