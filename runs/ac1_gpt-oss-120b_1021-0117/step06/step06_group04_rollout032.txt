def _conv_max(a, L):
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[:2*a.shape[0]-1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
        n = a0.shape[0]
    else:
        n = 2000
        a0 = np.full(n, 1.0/n)

    L = 1 << ((2*n - 1).bit_length())
    best_vec = a0.copy()
    max_val, max_idx, _ = _conv_max(best_vec, L)
    best_score = 2.0 * n * max_val / (best_vec.sum() ** 2)

    # start main loop
    eta0 = 1.0
    max_iter_per_restart = 5000
    restart_interval = 3000 # number of iterations before new restart
    total_iter = 0

    while time.monotonic() < deadline:
        # choose initial vector for this run
        if total_iter == 0:
            cur = a0.copy()
        else:
            # perturb around best
            cur = best_vec.copy()
            # add small random Dirichlet noise
            noise = rng.dirichlet(np.ones(n) * 0.01)
            cur = 0.9*cur + 0.1*noise
            cur = cur / cur.sum()
        # local descent
        for it in range(max_iter_per_restart):
            now = time.monotonic()
            if now >= deadline:
                break
            total_iter += 1
            max_val, max_idx, _ = _conv_max(cur, L)
            score = 2.0 * n * max_val / (cur.sum() ** 2)  # sum =1 so denominator 1

            if score < best_score:
                best_score = score
                best_vec = cur.copy()
                # early exit if target reached
                if best_score < 1.50274349:
                    return [float(x) for x in best_vec]

            # gradient
            j = max_idx - np.arange(n)
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(cur)

            # step size

    return [float(x) for x in best_vec]

import time
import math

def _conv_max_and_index(a: np.ndarray, L: int):
    """Return max value, its index and the full convolution a*a."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

    """Construct a non‑negative step‑function whose AC1 bound is very low."""
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # Length of the sequence (use the provided parent if present)
    if 'height_sequence_1' in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000                     # fallback length

    # Choose a geometric profile a_i ∝ r^i.
    # The optimal r solving n·r^{n‑1}=1 is r = exp(-log(n)/(n‑1)).
    r = math.exp(-math.log(n) / (n - 1))

    # Build the vector and renormalise to sum = 1 (the simplex).
    idx = np.arange(n, dtype=np.float64)
