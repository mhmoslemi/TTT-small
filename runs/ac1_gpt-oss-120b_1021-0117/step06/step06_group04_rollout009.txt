def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max(arr: np.ndarray, L: int):
    """Return max value, its index and full convolution a*a (FFT based)."""
    af = np.fft.rfft(arr, L)
    conv = np.fft.irfft(af * af, L)[: 2 * arr.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

    max_val, _, _ = _conv_max(arr, L)
    # sum(arr) is 1 after normalisation
    return 2.0 * n * max_val / (arr.sum() ** 2)

                frac = np.abs(rng.standard_normal()) / (np.abs(rng.standard_normal()) + 1e-9)
                frac = np.clip(frac, 0.0, 0.5)   # keep within [0,0.5]
                delta = cur[i] * frac
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score(tmp, n, L)
                if tmp_score < best_score:
                    best_score, best_vec = tmp_score, tmp.copy()
                    last_improve = 0
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score

        # ----- occasional symmetrisation (helps regularity) -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym, n, L)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best_score, best_vec = sym_score, sym.copy()
                    last_improve = 0

        # ----- restart if stuck too long -----
        if cur_score - best_score > 5e-4:   # stagnation detection
            last_improve += 1
        else:
            last_improve = 0

        if last_improve > 2000:
            # perturb best point and continue
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            last_improve = 0

        # early exit if we beat the target bound
        if best_score < 1.50274349:
            break

    # ---------------------------------------------------------
    # final safety check & return
    # ---------------------------------------------------------
    # ensure a valid simplex vector
    final = np.maximum(best_vec, 0.0)
    if final.sum() == 0.0:
        final = np.full_like(final, 1.0 / final.size)
    else:
        final = final / final.sum()
    return [float(x) for x in final]
