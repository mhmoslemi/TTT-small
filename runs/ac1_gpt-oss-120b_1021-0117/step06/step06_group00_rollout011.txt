def project_simplex(v):
    v = np.asarray(v)
    # ensure nonnegative maybe not needed after projection.
    # but we need from arbitrary real.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

def compute_score(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af*af, n=L)[:2*n-1]
    return 2.0 * n * conv.max()

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_from_max(n, max_val):
    """Score when the vector is already normalised (sum=1)."""
    return 2.0 * n * max_val

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a tiny safety margin for final handling
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    # FFT length – next power of two ≥ 2n‑1 (covers self‑convolution)
    L = 1 << ((2 * n - 1).bit_length())

    # initial score
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    cur_max = conv.max()
    cur_score = _score_from_max(n, cur_max)

    best_vec = cur.copy()
    best_score = cur_score

    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.6                     # base learning rate
    it = 0
    no_improve = 0
    target = 1.50274349           # desired bound

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---------- convolution & gradient ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_score = _score_from_max(n, max_val)

        # soft‑max weights on the convolution entries (temperature decays)
        tau = max(1e-6, 0.02 / math.sqrt(it))
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the weighted sum (sub‑gradient of max)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # selects the correct slice
        grad = 2.0 * grad_full[idx]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / math.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - update)
        # score of candidate
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_score = _score_from_max(n, cand_conv.max())

        # accept if improvement
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:
            no_improve += 1
            # occasional uphill move (tiny probability)
            if rng.random() < 0.01:
                cur, cur_score = cand, cand_score

        # update global best
        if cand_score < best_score:
            best_vec = cand.copy()
            best_score = cand_score

        # ---------- periodic symmetrisation ----------
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_score = _score_from_max(n, sym_conv.max())
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
            if sym_score < best_score:
                best_vec = sym.copy()
                best_score = sym_score

        # ---------- random local redistribution when stuck ----------
        if no_improve > 500:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_score = _score_from_max(n, tmp_conv.max())
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0
                if tmp_score < best_score:
                    best_vec = tmp.copy()
                    best_score = tmp_score

        # early exit if we reached the target bound
        if best_score < target:
            break

    # ------------------------------------------------------------------
    # final fallback random refinement (quick, cheap)
    while time.monotonic() < deadline:
        # small jitter on the best vector
        jitter = rng.normal(scale=0.005, size=n)
        cand = best_vec + jitter
        cand = _project_simplex(cand)
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_score = _score_from_max(n, cand_conv.max())
        if cand_score < best_score:
            best_vec = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

    # Return as a plain Python list of floats
    return [float(x) for x in best_vec]
