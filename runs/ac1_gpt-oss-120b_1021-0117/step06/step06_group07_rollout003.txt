import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Gradient from the set of (near‑maximal) convolution entries
# ------------------------------------------------------------
def _active_gradient(a: np.ndarray, conv: np.ndarray, eps_rel: float,
                     L: int, a_rev_fft: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient using all indices k where conv[k] is within
    eps_rel relative of the maximum.
    """
    maxb = conv.max()
    thr = maxb - eps_rel * maxb
    mask = (conv >= thr).astype(np.float64)
    # fallback: if mask is too dense we keep only a few top entries
    if mask.sum() > 50:
        top_k = 20
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
        mask[:] = 0.0
        mask[idxs] = 1.0
    # convolution mask * a_rev (full) via FFT
    mask_fft = np.fft.rfft(mask, n=L)
    full = np.fft.irfft(mask_fft * a_rev_fft, n=L)
    n = a.size
    # relevant slice gives gradient (factor 2 from derivative of a*a)
    grad = 2.0 * full[n - 1 : n - 1 + n]
    # average over the number of active constraints
    grad /= max(1.0, mask.sum())
    return grad
