import time, math, numpy as np

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    return np.maximum(v - theta, 0.0)

def _conv_max(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n-1).bit_length()+1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af*af, n=L)[:2*n-1]
    return conv.max()

def _score_fft(a, L=None):
    n = a.shape[0]
    maxc = _conv_max(a, L)
    return float(2.0 * n * maxc)

    return np.maximum(v - theta, 0.0)

def _conv_max(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return float(conv.max())

def _score_fft(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    return float(2.0 * n * _conv_max(a, L))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / a0.size)
    else:
        n = 2000
        a0 = np.exp(-np.arange(n) / 120.0)
        a0 = a0 / a0.sum()
    n = a0.shape[0]

    # Gaussian symmetric alternative
    sigma = n / 8.0
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    gauss = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    gauss = gauss / gauss.sum()

    L = 1 << ((3 * n - 2).bit_length())
    cur = a0 if _score_fft(a0, L) <= _score_fft(gauss, L) else gauss
    cur = _project_simplex(cur)
    cur_sc = _score_fft(cur, L)

    best_vec = cur.copy()
    best_sc = cur_sc

    # ---- optimisation parameters ------------------------------------
    base_alpha = 0.6
    beta = 0.9
    eps_decay = 0.999
    eps = None
    momentum = np.zeros_like(cur)
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)
    sym_interval = 150
    max_trials = 12
    no_improve = 0
    it = 0

    # ---- main loop -------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.5, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        momentum = beta * momentum + (1.0 - beta) * g
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / math.sqrt(it) / norm_m

        # line‑search
        improved = False
        for _ in range(max_trials):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _score_fft(cand, L)
            if cand_sc < cur_sc:
                improved = True
                break
            alpha *= 0.5

        if not improved:
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _score_fft(cand, L)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # symmetrisation
        if (no_improve % sym_interval) == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass transfer when stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.7
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp_sc = _score_fft(tmp, L)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # restart with tiny noise if hopeless
        if no_improve > 4000:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # cheap local perturbations while time permits
        if time.monotonic() < deadline - 5.0:
            for _ in range(5):
                i = rng.integers(0, n)
                if cur[i] < 1e-12:
                    continue
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                cand_sc = _score_fft(cand, L)
                if cand_sc < cur_sc:
                    cur = cand
                    cur_sc = cand_sc
                    if cand_sc < best_sc:
                        best_sc = cand_sc
                        best_vec = cand.copy()

        if time.monotonic() >= deadline:
            break

    best_vec = np.maximum(best_vec, 0.0)
    return [float(x) for x in best_vec]
