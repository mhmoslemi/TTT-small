import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a (length 2*len(a)-1) using FFT length L."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    return conv


def _smooth_gradient(a: np.ndarray, conv_full: np.ndarray,
                     beta: float, L: int) -> np.ndarray:
    """
    Gradient of a smooth approximation of max(conv) w.r.t. a.
    Uses softmax weights over the full convolution.
    """
    max_val = conv_full.max()
    # softmax weights (stable)
    w = np.exp(beta * (conv_full - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    w_rev = w[::-1]

    af = np.fft.rfft(a, L)
    wf = np.fft.rfft(w_rev, L)
    corr = np.fft.irfft(af * wf, L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)          # positions giving the correlation
    grad = 2.0 * corr[idx]
    return grad


def _score(a: np.ndarray, L_eval: int) -> float:
    """Exact score used by the evaluator (2*n*max(conv)/sum(a)^2)."""
    af = np.fft.rfft(a, L_eval)
    conv = np.fft.irfft(af * af, L_eval)[: 2 * a.shape[0] - 1]
    max_val = conv.max()
    s = a.sum()
    return float(2.0 * a.shape[0] * max_val / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ---- initialise a feasible point on the simplex -----------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
            s = 1.0
        a = a / s                     # now sum = 1
    else:                               # simple random fallback
        n0 = 2000
        a = rng.random(n0)
        a = a / a.sum()

    n = a.shape[0]
    # FFT lengths
    L_big = 1 << ((3 * n - 2).bit_length())   # for gradient (a * weight)
    L_small = 1 << ((2 * n - 1).bit_length())  # for score evaluation

    # ---- Adam optimiser state --------------------------------------
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # ---- keep the best solution ------------------------------------
    best_vec = a.copy()
    best_sc = _score(a, L_small)

    # ---- parameters ------------------------------------------------
    base_lr = 0.5
    smooth_beta = 12.0            # temperature for softmax
    sym_interval = 250
    transfer_thresh = 800         # trigger random mass transfer
    no_improve = 0
    target_score = 1.50274349

    while time.monotonic() < deadline:
        t_iter += 1
        # full convolution of a with itself
        conv_full = _conv_full(a, L_big)

        # smoothed gradient of max(conv)
        grad = _smooth_gradient(a, conv_full, smooth_beta, L_big)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = base_lr / math.sqrt(t_iter) * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(a - step)
        cand_sc = _score(cand, L_small)

        improved = False
        if cand_sc < best_sc:
            improved = True
        else:
            # occasional tiny uphill acceptance
            if rng.random() < 0.001:
                improved = True

        if improved:
            a = cand
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- periodic symmetrisation --------------------------------
        if t_iter % sym_interval == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L_small)
            if sym_sc < best_sc:
                a = sym
                best_sc = sym_sc
                best_vec = sym.copy()
                no_improve = 0

        # ---- random mass transfer when progress stalls --------------
        if no_improve > transfer_thresh:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L_small)
                if tmp_sc < best_sc:
                    a = tmp
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = max(0, no_improve - 200)

        # ---- early exit if target reached ----------------------------
        if best_sc < target_score:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
