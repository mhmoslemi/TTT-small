        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)

        conv_w_a = np.fft.irfft(w_fft * a_fft, n=L)[:3*n - 2]
        conv_w_ar = np.fft.irfft(w_fft * a_rev_fft, n=L)[:3*n - 2]

        term1 = conv_w_ar[(n-1):(n-1)+n]
        term2 = conv_w_a[:n]
        grad = term1 + term2

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a_tmp = a - step
        # project onto simplex
        a_tmp = _project_simplex(a_tmp, Z=1.0)
        # optional symmetrisation every 20 iters
        if it % 20 == 0:
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = _project_simplex(a_tmp, Z=1.0)

        a = a_tmp

        # update gamma
        if gamma < gamma_max:
            gamma = min(gamma_max, gamma * gamma_factor)

        # quick check for time
        if time.monotonic() >= deadline:
            break

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries with sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Return the maximum entry of the full linear convolution a*a
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------

    if "height_sequence_1" in globals():

    if a.sum() < 1e-12:

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_vec = a.copy()
    best_sc = 2.0 * n * _conv_max(a, L)
    target = 1.50274349

        # ------------------------------------------------------------

        max_conv = conv_full.max()
        cur_sc = 2.0 * n * max_conv

        # ------------------------------------------------------------
