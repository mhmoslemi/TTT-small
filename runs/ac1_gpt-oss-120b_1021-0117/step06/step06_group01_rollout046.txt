import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** iter_num)
        v_hat = v / (1 - beta2 ** iter_num)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = a - step
        cand = _project_simplex(cand)

        # enforce symmetry every iteration
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # occasional fresh perturbation when progress stalls
        if no_improve > 300:
            # generate a random exponential‑like shape
            alpha = 10 ** rng.uniform(-4, 0)
            rand = np.exp(-alpha * np.arange(n, dtype=np.float64))
            rand = rand / rand.sum()

            t = rng.random() * 0.3
            cand2 = (1 - t) * best + t * rand
            cand2 = _project_simplex(cand2)
            cand2 = (cand2 + cand2[::-1]) * 0.5
            cand2 = _project_simplex(cand2)

            cand2_sc = _score(cand2)
            if cand2_sc < best_sc:
                best_sc = cand2_sc
                best = cand2.copy()
                a = cand2.copy()
                m.fill(0.0)
                v.fill(0.0)
                eps = None
                no_improve = 0
                continue
            else:
                # reset to best with small Gaussian noise
                a = best + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = _project_simplex(a)
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(a)
                m.fill(0.0)
                v.fill(0.0)
                eps = None
                no_improve = 0
                continue

        a = cand

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # fallback in the unlikely event we are still far from target
    # --------------------------------------------------------
    if best_sc > 1.6:
        fallback = np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** 2
        fallback = fallback / fallback.sum()
        best = fallback

    # return plain Python list
    return [float(x) for x in best]
