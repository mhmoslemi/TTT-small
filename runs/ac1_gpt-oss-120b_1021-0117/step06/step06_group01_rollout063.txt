    deadline = start + max(0.0, float(budget_s) - 0.5)
    # get length

        base = np.maximum(base, 0.0)
        if base.sum() <= 0:
            base = np.full_like(base, 1.0 / base.size)
        else:
            base = base / base.sum()
    else:
        n = 2000
        base = np.exp(-np.arange(n) / 100.0)
        base = base / base.sum()
    n = base.size

    # fast score
    def _score(a):
        L = 1 << ((2*n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2*n - 1]
        return float(2.0 * n * conv.max())

    # projection simplex
    def _project(v):
        # implementation from earlier
        nloc = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, nloc+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    best_vec = base.copy()

        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a: np.ndarray) -> float:
    """Fast AC1 bound for a vector with sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # initialise a feasible point (sum = 1)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)
        a0 = a0 / a0.sum()
    n = a0.size

    best_vec = a0.copy()
    best_sc = _score(best_vec)

    # --------------------------------------------------------
    # helper to test a candidate and keep the best so far
    # --------------------------------------------------------
    def _try_candidate(vec: np.ndarray):
        nonlocal best_vec, best_sc
        sc = _score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
        return sc

    ar = np.arange(n, dtype=np.float64)

    # --------------------------------------------------------
    # explore a few parametric families (cheap, dense search)
    # --------------------------------------------------------
    # monotone exponential
    for eps in np.logspace(-5, -2, num=30):
        cand = np.exp(-eps * ar)
        cand = cand / cand.sum()
        _try_candidate(cand)

    # symmetric exponential (decay from both ends)
    d = np.minimum(ar, n - 1 - ar)
    for eps in np.logspace(-5, -2, num=30):
        cand = np.exp(-eps * d)
        cand = cand / cand.sum()
        _try_candidate(cand)

    # Gaussian‑like
    mid = (n - 1) / 2.0
    for sigma in np.linspace(0.5, n / 4.0, 30):
        cand = np.exp(-((ar - mid) ** 2) / (2 * sigma ** 2))
        cand = cand / cand.sum()
        _try_candidate(cand)

    # cosine‑squared (high at ends, low in centre)
    cand = np.cos(np.pi * ar / (n - 1)) ** 2
    cand = cand / cand.sum()
    _try_candidate(cand)

    # random convex combos of the current best and a fresh random vector
    for _ in range(200):
        t = rng.random()
        cand = t * best_vec + (1 - t) * rng.random(n)
        cand = _project_simplex(cand)
        _try_candidate(cand)

    # --------------------------------------------------------
    # gradient‑based refinement (Adam optimiser with simplex proj)
    # --------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    m = np.zeros_like(cur)       # first moment
    v = np.zeros_like(cur)       # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    temp = None
    temp_decay = 0.95
    no_improve = 0
    it = 0
    improve_target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ---- convolution and soft‑max weights ----
        L = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        if temp is None:
            temp = max_val * 0.2
        else:
            temp = max(temp * temp_decay, max_val * 1e-4, 1e-12)

        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.5 * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- occasional symmetrisation (helps a lot) ----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ---- random mass‑transfer when stuck ----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ---- restart from best with small noise if hopeless ----
        if no_improve > 8000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m = np.zeros_like(cur)
            v = np.zeros_like(cur)
            temp = None
            no_improve = 0

        # ---- early exit if target reached ----
        if best_sc < improve_target:
            break

    # final safety projection
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.size)
    else:
        best_vec = best_vec / total

    return [float(x) for x in best_vec]
