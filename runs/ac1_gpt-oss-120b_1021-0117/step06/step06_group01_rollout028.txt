import time

# ------------------------------------------------------------
# Project a vector onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(a*a) for sum(a)=1."""
    n = a.shape[0]

    af = scipy.fft.rfft(a, n=L)
    conv = scipy.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            alpha = 10 ** rng.uniform(-2, 0.5)   # ~0.01 … 3.2
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            cand = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))

        elif r < 0.20:                   # asymmetric Beta(alpha,beta)
            alpha = 10 ** rng.uniform(-2, 0.5)
            beta = 10 ** rng.uniform(-2, 0.5)
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            cand = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

        elif r < 0.30:                   # power‑law from both ends
            p = 10 ** rng.uniform(-2, 1)      # 0.01 … 10
            idx = np.arange(n, dtype=np.float64)
            cand = (idx + 1.0) ** (-p) + (n - idx) ** (-p)

        elif r < 0.40:                   # piecewise‑constant blocks
            K = rng.integers(2, 8)            # 2 … 7 blocks
            cuts = np.sort(rng.choice(n - 1, size=K - 1, replace=False) + 1)
            points = np.concatenate(([0], cuts, [n]))
            block_vals = rng.random(K)
            block_vals = block_vals / block_vals.sum()
            cand = np.empty(n, dtype=np.float64)
            for j in range(K):
                cand[points[j]:points[j + 1]] = block_vals[j]

        elif r < 0.50:                   # random Gaussian bump
            centre = rng.integers(0, n)
            sigma = rng.uniform(0.5, n / 2.0)
            cand = np.exp(-((np.arange(n, dtype=np.float64) - centre) ** 2) /
                          (2.0 * sigma ** 2))

        elif r < 0.60:                   # smoothed random noise
            raw = rng.random(n)
            win = rng.integers(3, 15)          # window length 3‑14
            kernel = np.ones(win) / win
            cand = np.convolve(raw, kernel, mode='same')

        elif r < 0.80:                   # convex combo of best & fresh exponential
            alpha = 10 ** rng.uniform(-4, 0)
            exp_vec = np.exp(-alpha * np.arange(n, dtype=np.float64))
            exp_vec = exp_vec / exp_vec.sum()
            t = rng.random()
            cand = (1.0 - t) * best_vec + t * exp_vec

        else:                            # pure random Dirichlet draw
            cand = rng.random(n)
        # --------------------------------------------------------------

        # normalise safely
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s

        # evaluate
        _try(cand)

        # --------------------------------------------------------------
        # occasional symmetrisation (helps a lot)
        # --------------------------------------------------------------
        if it % 50 == 0:
            sym = (best_vec + best_vec[::-1]) * 0.5
            sym = _project_simplex(sym)
            _try(sym)

        # --------------------------------------------------------------
        # occasional small random mass‑transfer on the current best
        # --------------------------------------------------------------
        if it % 20 == 0:
            i = rng.integers(0, n)
            if best_vec[i] > 0.0:
                delta = best_vec[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = best_vec.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                _try(tmp)

        # --------------------------------------------------------------
        # simple sub‑gradient step every few hundred iterations
        # --------------------------------------------------------------
        if it % 200 == 0:
            # compute full convolution and locate maximal entries
            af = scipy.fft.rfft(best_vec, n=L_fft)
            conv_full = scipy.fft.irfft(af * af, n=L_fft)[: 2 * n - 1]
            max_val = conv_full.max()
            # indices where conv is (practically) maximal
            eps = max_val * 1e-9
            idxs = np.where(conv_full >= max_val - eps)[0]
            if idxs.size:
                w = np.zeros_like(conv_full)
                w[idxs] = 1.0
                w_rev = w[::-1]
                w_fft = scipy.fft.rfft(w_rev, n=L_fft)
                grad_full = scipy.fft.irfft(af * w_fft, n=L_fft)[: 2 * n - 1]
                # align gradient for the original vector
                grad = 2.0 * grad_full[(2 * n - 2) - np.arange(n)]
                step = 0.6 / (np.linalg.norm(grad) + 1e-12)
                cand = best_vec - step * grad
                cand = _project_simplex(cand)
                _try(cand)

        # --------------------------------------------------------------
        # early exit if target already met
        # --------------------------------------------------------------
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # return the best found sequence as plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
