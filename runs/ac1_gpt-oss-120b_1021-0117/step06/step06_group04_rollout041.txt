import time
import math
import numpy as np

# ------------------------------------------------------------
# helper utilities
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return (max value, its index, full convolution) using a FFT of length L."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def _score_from_max(max_val: float, n: int) -> float:
    """Score given the maximal convolution value (sum is assumed 1)."""
    return 2.0 * n * max_val
