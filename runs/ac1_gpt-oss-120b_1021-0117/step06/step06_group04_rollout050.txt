def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (clip negatives, renormalise)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution, length 2*len(a)-1) using FFT of length L."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    return conv
