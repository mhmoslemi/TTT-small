import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    if v.ndim != 1:
        v = v.ravel()
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for sum≈1 (uses FFT)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv(a,a)) for a with sum≈1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            # enforce symmetry after projection
            trial = (trial + trial[::-1]) * 0.5
            trial = _project_simplex(trial)          # re‑project after averaging
            trial_sc = _score(trial, L)
            if trial_sc < cur_sc:
                cand = trial
                cand_sc = trial_sc
                improved = True
                break
            alpha *= 0.5

        # ---- occasional uphill move (simulated annealing) ----
        if not improved:
            trial = cur - alpha * momentum
            trial = _project_simplex(trial)
            trial = (trial + trial[::-1]) * 0.5
            trial = _project_simplex(trial)
            trial_sc = _score(trial, L)
            prob = math.exp(-(trial_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = trial
                cand_sc = trial_sc
                improved = True

        # ---- random symmetric mass transfer ----
        if not improved and rng.random() < rand_move_prob:
            i = rng.integers(0, n // 2)               # work on left half
            j = rng.integers(0, n // 2)
            if i != j and cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                # mirror the changes
                mi, mj = n - 1 - i, n - 1 - j
                tmp[mi] -= delta
                tmp[mj] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cand = tmp
                    cand_sc = tmp_sc
                    improved = True

        # ---- accept candidate if any ----
        if improved:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1

        # ---- restart from best with small jitter if stuck ----
        if no_improve > restart_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            momentum.fill(0.0)
            eps = None
            no_improve = 0
            # re‑score after restart
            cur_sc = _score(cur, L)

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    return best_vec
