import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the probability simplex of sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast max‑convolution via FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Sub‑gradient using the top‑k largest convolution entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    g /= float(top_k)
    return g

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # --------------------------------------------------------
    # Determine problem size
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000  # fallback length if parent not present

    # FFT length (next power of two)
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Helper to evaluate the objective (sum is forced to 1)
    # --------------------------------------------------------
    def _score(a_vec: np.ndarray) -> float:
        # a_vec should already be normalised to sum == 1
        max_b = _conv_max(a_vec, L)
        return 2.0 * n * max_b

    # --------------------------------------------------------
    # Initial search: geometric families (a_i ∝ r^i)
    # --------------------------------------------------------
    best_vec = None
    best_sc = float("inf")

    # try a few random starts as well
    for _ in range(3):
        a_rand = rng.random(n)
        a_rand /= a_rand.sum()
        sc = _score(a_rand)
        if sc < best_sc:
            best_sc, best_vec = sc, a_rand.copy()

    # geometric sweep – explore t = -log(r) which controls decay
    t_vals = np.concatenate((
        np.logspace(-5, -2, 150),          # fine near r≈1
        np.linspace(0.001, 0.05, 100),    # broader range

    idx = np.arange(n, dtype=np.float64)

    # fallback to uniform if something went wrong

                a[j] += delta
                a = a / a.sum()
                no_improve = 0

    # --------------------------------------------------------
    # Return the best candidate found (Python list of floats)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
