import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast convolution (full) via FFT, returns full array
# ------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution (average over top‑k indices)
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    """Return an approximate sub‑gradient using the top‑k convolution entries."""
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        length = i_end - i_start + 1
        if length <= 0:
            continue
        # slice of a that contributes
        a_slice = a[k - i_start : k - i_start + length]
        g[i_start : i_start + length] += 2.0 * a_slice
    # average over the selected entries
    g /= max(1, top_k)
    return g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # --------------------------------------------------------
    # Initialise
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a random vector of reasonable length
        a = rng.random(2000, dtype=np.float64)

    # enforce non‑negativity and a non‑trivial sum
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # simplex (sum = 1)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length = next power of two

    # --------------------------------------------------------
    # Optimisation bookkeeping
    # --------------------------------------------------------
    best_vec = a.copy()
    conv_full = _full_conv(a, L)
    best_sc = 2.0 * n * conv_full.max()
    cur_sc = best_sc

    # Adam‑style parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    top_k = 32                     # use a richer gradient estimate
    sym_interval = 200             # symmetrise every few hundred iterations
    no_improve = 0
    target_sc = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution and its maximum ----
        conv_full = _full_conv(a, L)
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ---- keep the best solution seen so far ----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit if target reached
        if best_sc < target_sc:
            break

        # ---- gradient step (Adam) ----
        g = _grad_topk(a, conv_full, top_k)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat
        a_candidate = _project_simplex(a_candidate)   # back to simplex

        # evaluate candidate
        cand_conv = _full_conv(a_candidate, L)
        cand_sc = 2.0 * n * cand_conv.max()

        # accept if improvement; otherwise keep current (no Metropolis move)
        if cand_sc < cur_sc:
            a = a_candidate
            cur_sc = cand_sc
            no_improve = 0

        # ---- periodic symmetrisation (helps to flatten the autocorrelation) ----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_sc = 2.0 * n * _full_conv(a_sym, L).max()
            if sym_sc < cur_sc:
                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = a_sym.copy()
                no_improve = 0

        # ---- occasional random mass transfer when stuck ----
        if no_improve > 3000:

            if j >= i:

            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
            no_improve = 0

    # --------------------------------------------------------
    # Return best candidate (as a plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
