    deadline = start + max(0.0, float(budget_s) - 0.2)

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        if a.sum() <= 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a.sum()
    else:
        n = 2500
        a = np.full(n, 1.0 / n)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    
    # compute initial convolution max and score
    cur_max, cur_idx, cur_conv = _conv_max_and_index(a, L)
    cur_sc = 2.0 * n * cur_max / (a.sum() ** 2)
    best_vec = a.copy()
    best_sc = cur_sc
    
    # parameters
    beta = 0.9
    base_alpha = 0.8
    max_trials = 20
    sym_interval = 200
    transfer_thresh = 200
    reinit_thresh = 1000
    target = 1.50274349
    
    momentum = np.zeros_like(a)
    indices = np.arange(n)
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # compute subgradient at max index
        cur_max, cur_idx, cur_conv = _conv_max_and_index(a, L)
        j = cur_idx - indices
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)
        # line search
        improved = False
        cand = None
        cand_sc = None
        alpha_try = alpha
        for _ in range(max_trials):
            cand_try = a - alpha_try * momentum
            cand_try = _project_simplex(cand_try)
            max_val, _, _ = _conv_max_and_index(cand_try, L)
            sc = 2.0 * n * max_val / (cand_try.sum() ** 2)
            if sc < cur_sc:
                cand, cand_sc = cand_try, sc
                improved = True
                break
            alpha_try *= 0.5
        # occasional uphill
        if not improved:
            cand_try = a - alpha_try * momentum
            cand_try = _project_simplex(cand_try)
            max_val, _, _ = _conv_max_and_index(cand_try, L)
            sc = 2.0 * n * max_val / (cand_try.sum() ** 2)
            prob = math.exp(-(sc - cur_sc) / (alpha_try + 1e-12))
            if rng.random() < prob:
                cand, cand_sc = cand_try, sc
                improved = True
        if improved:
            a, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # symmetrisation
        if it % sym_interval == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            max_val, _, _ = _conv_max_and_index(sym, L)
            sc = 2.0 * n * max_val / (sym.sum() ** 2)
            if sc < cur_sc:
                a, cur_sc = sym, sc
                if sc < best_sc:
                    best_sc, best_vec = sc, sym.copy()
                no_improve = 0
        
        # random mass transfer when stalled
        if no_improve > transfer_thresh:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                max_val, _, _ = _conv_max_and_index(tmp, L)
                sc = 2.0 * n * max_val / (tmp.sum() ** 2)
                if sc < cur_sc:
                    a, cur_sc = tmp, sc
                    if sc < best_sc:
                        best_sc, best_vec = sc, tmp.copy()
                    no_improve = 0
                else:
                    no_improve = max(0, no_improve - 100)
        
        # restart after long stall
        if no_improve > reinit_thresh:
            a = best_vec + 0.01 * rng.random(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            momentum = np.zeros_like(a)
            no_improve = 0
        
        if best_sc < target:
            break
        
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return (max value, its index, full convolution) of a*a."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv
