import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound for a sequence that already satisfies sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a (sum ≈ 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)   # power of two ≥ 2n‑1
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # --------------------------------------------------------
    # Initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: simple exponential decay
        n_fallback = 2500
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 120.0)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s = 1.0
    a = a0 / s                     # now sum = 1

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    # Adam optimizer state
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8                     # base learning rate
    eps_adam = 1e-8

    # bookkeeping
    cur = a.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc
    eps = None                    # smoothing parameter
    it = 0
    no_improve = 0
    sym_interval = 150
    restart_after = 4000

    while time.monotonic() < deadline:
        it += 1

        # ----------------------------------------------------
        # convolution, smoothed max weight, gradient
        # ----------------------------------------------------
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # smooth max weight
        if eps is None:
            eps = max(max_val * 0.5, 1e-9)
        else:
            eps = max(eps * 0.9995, 1e-12)

        if w_sum == 0.0:

            eps = None
            no_improve = 0

        # ----------------------------------------------------
        # early exit if we beat the target
        # ----------------------------------------------------
        if best_sc < 1.50274349:
            break

        # periodic time check
        if it % 500 == 0 and time.monotonic() >= deadline:
            break

    # Return as plain Python list
    return [float(x) for x in best]
