import time, math, numpy as np

def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=Z}."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv

            if sc < best_local_sc:
                best_local_sc, best_local = sc, a_tmp

        # ---- accept move ----
        if best_local_sc < cur_sc:
            a = best_local
            if best_local_sc < best_sc:
                best_sc, best_vec = best_local_sc, best_local.copy()
            no_improve = 0
        else:
            # occasional uphill move (Metropolis style)
            prob = math.exp(-(best_local_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a = best_local

        # ---- random perturbation if stuck ----
        if no_improve > 2500:
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            if i1 != i2:
                delta = a[i1] * rng.random() * 0.5
                a[i1] -= delta
                a[i2] += delta
                a /= a.sum()
                momentum.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # Return best sequence found as plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
```
