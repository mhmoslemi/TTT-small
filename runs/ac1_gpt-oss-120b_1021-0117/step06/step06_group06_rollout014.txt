import time, math
import numpy as np

# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of 2 * n * max(conv(a,a))  (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L_fft: int) -> float:
    """Return 2 * n * max(full convolution of a with itself)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(af * af, n=L_fft)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
