def _grad_topk(a, conv_full, top_k):
    n = a.shape[0]
    if top_k >= conv_full.size:
        idxs = np.arange(conv_full.size)
    else:
        idxs = np.argpartition(-conv_full, top_k-1)[:top_k]
    grad = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # i spans i_start..i_end inclusive
        # corresponding a indices j = k - i
        i_range = slice(i_start, i_end+1)
        j = np.arange(i_start, i_end+1)
        grad[i_range] += 2.0 * a[k - j]
    # average
    grad = grad / max(1, top_k)
    return grad

    # projection simplex function

            theta = css[rho] / (rho + 1)

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # Initialise a feasible vector (sum = 1, non‑negative)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random uniform vector of length ≈2500
        n = 2500
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0
        s = a.sum()
    a = a / s

    n = a.size
    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------------------
    # Helper utilities
    # ----------------------------------------------------------------------
    def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
        """Euclidean projection of v onto the simplex {x≥0, sum(x)=Z}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - Z
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    def _conv_max(arr: np.ndarray):
        """Return (max value, full convolution) of arr with itself."""
        af = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(conv.max()), conv

    def _grad_topk(arr: np.ndarray, conv_full: np.ndarray, top_k: int) -> np.ndarray:
        """Sub‑gradient using the top‑k convolution entries."""
        if top_k >= conv_full.size:
            idxs = np.arange(conv_full.size)
        else:
            idxs = np.argpartition(-conv_full, top_k - 1)[:top_k]
        g = np.zeros_like(arr)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            g[i] += 2.0 * arr[k - i]
        # average contribution
        g = g / max(1, top_k)
        return g

    # ----------------------------------------------------------------------
    # Book‑keeping
    # ----------------------------------------------------------------------
    cur = a.copy()
    cur_max, cur_conv = _conv_max(cur)
    cur_score = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_score = cur_score

    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.6                     # base learning rate
    t = 0

    top_k = 30
    sym_interval = 300
    no_improve = 0
    target = 1.50274349

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        t += 1

        # gradient of the smoothed max (via top‑k entries)
        grad = _grad_topk(cur, cur_conv, top_k)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # candidate point and projection onto simplex
        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_max, cand_conv = _conv_max(cand)
        cand_score = 2.0 * n * cand_max

        # acceptance test (mostly greedy, occasional uphill moves)
        if cand_score < cur_score:
            cur, cur_max, cur_conv, cur_score = cand, cand_max, cand_conv, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand.copy()
        else:
            # simulated‑annealing‑style probability
            T = max(0.001, 0.4 * math.exp(-t / 5000.0))
            prob = math.exp(-(cand_score - cur_score) / (T + 1e-12))
            if rng.random() < prob:
                cur, cur_max, cur_conv, cur_score = cand, cand_max, cand_conv, cand_score
            no_improve += 1

        # ------------------------------------------------------------------
        # Periodic symmetrisation (helps to keep the shape balanced)
        # ------------------------------------------------------------------
        if t % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_max, sym_conv = _conv_max(sym)
            sym_score = 2.0 * n * sym_max
            if sym_score < cur_score:
                cur, cur_max, cur_conv, cur_score = sym, sym_max, sym_conv, sym_score
                no_improve = 0
            if sym_score < best_score:
                best_score, best_vec = sym_score, sym.copy()

        # ------------------------------------------------------------------
        # Random mass‑transfer when stuck for a while
        # ------------------------------------------------------------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                cur[i] -= delta
                cur[j] += delta
                cur = cur / cur.sum()
                cur_max, cur_conv = _conv_max(cur)
                cur_score = 2.0 * n * cur_max
            no_improve = 0

        # early exit if we have already beaten the target
        if best_score < target:
            break

    # ----------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ----------------------------------------------------------------------
    return [float(x) for x in best_vec]
