import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max(seq: np.ndarray, L: int) -> float:
    """Return the maximum value of the linear convolution seq * seq."""
    af = np.fft.rfft(seq, L)
    conv = np.fft.irfft(af * af, L)[: 2 * seq.shape[0] - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # spare margin

    # ------------------------------------------------------------------
    # initial feasible point (simplex, sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / a.size)
            s = 1.0
        a = a / s
    else:
        # fallback: moderately sized uniform vector
        n_fallback = 2000
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # ------------------------------------------------------------------
    # score helper
    # ------------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        max_val = _conv_max(vec, L)
        return 2.0 * n * max_val / (vec.sum() ** 2)

    cur = a.copy()
    cur_score = _score(cur)

    best_vec = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # simple stochastic improvement loop
    # ------------------------------------------------------------------
    it = 0
    max_iters = 5000
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # pick two distinct indices and move a random fraction of mass
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = cur[i] * rng.random() * 0.4   # up to 40 % of the source mass
        if delta <= 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # simplex projection (keeps sum = 1, all non‑negative)
        cand = _project_simplex(cand)

        cand_score = _score(cand)

        # accept only if we improve
        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()
            cur = cand
            cur_score = cand_score
        else:
            # occasionally replace the current point with the global best
            # to keep the walk from stalling
            if rng.random() < 0.02:
                cur = best_vec.copy()
                cur_score = best_score

    # ------------------------------------------------------------------
    # return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
