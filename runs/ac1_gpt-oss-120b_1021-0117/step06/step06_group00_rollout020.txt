import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (non‑negative, sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # enforce sum=1 in case of tiny numerical drift
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _ac1_score(a, L):
    """Score = 2 * n * max(convolution(a,a)) for a already on the simplex."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    max_val = conv_full[: 2 * a.size - 1].max()
    return 2.0 * a.size * max_val

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small margin for final bookkeeping
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ----------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑2 >= 2n‑1

    cur_sc = _ac1_score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05                       # base learning rate
    it = 0
    no_improve = 0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ------------------------------------------------------------------
        # convolution of current vector (to get max index)
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        idx_max = int(np.argmax(conv))

        # ------------------------------------------------------------------
        # sub‑gradient of the max‑convolution (single active index)
        w_rev = np.zeros(2 * n - 1, dtype=np.float64)
        w_rev[(2 * n - 2) - idx_max] = 1.0
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[(2 * n - 2) - np.arange(n)]

        # ------------------------------------------------------------------
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------------
        # candidate point and its score
        cand = _project_simplex(cur - step)
        cand_sc = _ac1_score(cand, L)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # occasional symmetrisation (helps due to convolution symmetry)
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _ac1_score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ------------------------------------------------------------------
        # random redistribution when stuck for a long time
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _ac1_score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc

    # ----------------------------------------------------------------------
    # return the best vector found (as plain floats)
    return [float(x) for x in best_vec]
