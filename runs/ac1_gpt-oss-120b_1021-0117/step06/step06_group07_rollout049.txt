def _max_conv_and_grad(a, L):
    # compute convolution via real FFT
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[:2*a.size-1]
    max_val = conv_full.max()
    # tolerance for equality
    eps = 1e-12
    idxs = np.where(conv_full >= max_val - eps)[0]
    n = a.size
    grad = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # indices in i_start..i_end inclusive
        i = np.arange(i_start, i_end + 1)
        grad[i] += 2.0 * a[k - i]
    return max_val, grad

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

        a = np.maximum(a, 0.0)
    else:
        n = 2048
        a = rng.random(n, dtype=np.float64)

    # Normalize sum to 1
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()

    # Ensure symmetry
    a = (a + a[::-1]) * 0.5

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = a.copy()

    beta = 0.9
    momentum = np.zeros_like(a)
    no_improve = 0

def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    v = v.ravel()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv_and_grad(a: np.ndarray, L: int):
    """
    Compute the maximum of the autocorrelation (full convolution) of a
    with itself and the exact gradient of that maximum.
    Returns (max_val, grad) where grad has shape a.shape.
    """
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    max_val = conv_full.max()
    # indices achieving (almost) the maximum
    eps = 1e-12
    idxs = np.where(conv_full >= max_val - eps)[0]

    n = a.size
    grad = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        grad[i] += 2.0 * a[k - i]
    return max_val, grad
