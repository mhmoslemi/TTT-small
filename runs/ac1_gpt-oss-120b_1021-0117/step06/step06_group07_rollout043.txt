import time
import math
import numpy as np
import scipy.sparse as sp

# optional LP solver
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x >= 0, sum(x)=Z}. """
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, L: int) -> float:
    """Maximum entry of full linear convolution a*a using FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    """Sub‑gradient using the top‑k largest convolution entries."""
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    return g / max(1, top_k)


def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs) -> list:
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1 (simplex)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length, power of two

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_sc = 2.0 * n * _conv_max(a, L)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-12
    alpha0 = 0.4          # base step size

    # other heuristics
    top_k = 64
    no_improve = 0
    it = 0
    target = 1.50274349
    mix_interval = 500           # mix with uniform every so many steps
    mix_weight = 0.12            # how much uniform is mixed in
    rand_thresh = 1500           # when to perform random mass‑transfer

    uniform_vec = np.full_like(a, 1.0 / n)

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & current score -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target:
            break

        # ----- gradient step (Adam) -----
        g = _grad_topk(a, conv_full, top_k)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha0 / (np.sqrt(v_hat) + eps) * m_hat

        a_tmp = a - step
        a_tmp = _project_simplex(a_tmp)       # stay on simplex
        cand_sc = 2.0 * n * _conv_max(a_tmp, L)

        if cand_sc < cur_sc:
            a = a_tmp
            cur_sc = cand_sc

        # ----- periodic mixing with the uniform vector -----
        if it % mix_interval == 0:
            a_mix = (1.0 - mix_weight) * a + mix_weight * uniform_vec
            a_mix = _project_simplex(a_mix)
            mix_sc = 2.0 * n * _conv_max(a_mix, L)
            if mix_sc < cur_sc:
                a = a_mix
                cur_sc = mix_sc

        # ----- occasional random mass‑transfer when stuck -----
        if no_improve > rand_thresh:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

        # ----- cheap LP refinement (if solver available and time permits) -----
        if cp is not None and it % 3000 == 0 and (deadline - time.monotonic()) > 8.0:
            M = _conv_max(a, L)
            # Build a sparse convolution matrix (Toeplitz‑like)
            data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
            offsets = np.arange(n)
            T = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
            g_var = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)),
                              [T @ g_var <= M + 1e-12])
            try:
                prob.solve(solver="ECOS", warm_start=True, max_iters=1500, verbose=False)
            except Exception:  # pragma: no cover
                prob = None
            if prob is not None and g_var.value is not None:
                g_opt = np.maximum(g_var.value, 0.0)
                a_mix = 0.8 * a + 0.2 * g_opt
                a_mix = _project_simplex(a_mix)
                mix_sc = 2.0 * n * _conv_max(a_mix, L)
                if mix_sc < cur_sc:
                    a = a_mix
                    cur_sc = mix_sc
                    if mix_sc < best_sc:
                        best_sc, best_vec = mix_sc, a_mix.copy()
                        no_improve = 0

    # ------------------------------------------------------------------
    # Return best candidate found
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
