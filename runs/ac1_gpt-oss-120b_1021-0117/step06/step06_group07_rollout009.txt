import time, math, numpy as np

def _project_simplex(v, Z=1.0):
    # Euclidean projection onto simplex {x >= 0, sum = Z}
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    return 1 << (x-1).bit_length()
