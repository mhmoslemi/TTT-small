import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast computation of the AC1 bound for a vector with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    # length for a safe zero‑padded FFT
    L = 1 << ((2 * n - 1).bit_length() + 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient (soft‑max smoothed) of the max‑convolution
# ------------------------------------------------------------
def _grad(a: np.ndarray, eps: float, L: int, grad_idx: np.ndarray) -> np.ndarray:
    """Gradient of the smoothed max of a*a."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    mx = conv.max()
    # soft‑max weights
    w = np.exp((conv - mx) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # convolution with reversed weights gives the gradient
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    return 2.0 * grad_full[grad_idx]

                    no_improve = 0

        # ---- early exit if target reached ----
        if best_sc < target:
            break

        # ---- periodic deadline check ----
        if iteration % 400 == 0 and time.monotonic() >= deadline:
            break

    # Return plain Python list
    return [float(x) for x in best_vec]
