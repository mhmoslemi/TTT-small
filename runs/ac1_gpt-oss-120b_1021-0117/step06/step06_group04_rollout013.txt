def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.1)

    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        if s <= 0:
            cur = np.full_like(cur, 1.0/cur.size)
        else:
            cur = cur / s
    else:
        n = 2000
        cur = rng.random(n)
        cur = cur/cur.sum()
    n = cur.shape[0]
    L = 1 << ((2*n-1).bit_length())
    # function to compute objective. Since sum is 1.
    def objective(a):
        af = np.fft.rfft(a, L)
        conv = np.fft.irfft(af*af, L)[:2*n-1]
        return 2.0 * n * conv.max()
    cur_score = objective(cur)
    best = cur.copy()
    best_score = cur_score
    # algorithm parameters
    total_time = deadline - start
    # start temperature
    T0 = 0.05
    # heating schedule = linear decay
    it = 0
    # maintain step_factor max
    max_step = 0.5   # max fraction of mass to move

    # precompute indices array for gradient mapping
    indices = np.arange(n)

    while time.monotonic() < deadline:
        it += 1
        # compute max conv and its index
        af = np.fft.rfft(cur, L)
        conv = np.fft.irfft(af*af, L)[:2*n-1]
        max_val = float(conv.max())
        max_idx = int(conv.argmax())
        cur_score = 2.0 * n * max_val
        # gradient approximate
        shift = max_idx - indices
        mask = (shift >= 0) & (shift < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[shift[mask]]
        # top k indices
        k = 200 if n > 200 else n
        # get indices of largest gradient values
        # we can use np.argpartition for speed
        top_idxs = np.argpartition(-grad, k-1)[:k]  # larger first
        # choose i from top_idxs with probability proportional to gradient magnitude
        gvals = grad[top_idxs]
        if gvals.sum() > 0:
            probs = gvals / gvals.sum()
            i = rng.choice(top_idxs, p=probs)
        else:
            i = rng.integers(0, n)
        # choose j random not equal i
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # delta
        delta = cur[i] * (rng.random() * max_step)
        if delta <= 0:
            continue
        # ensure cur[i] - delta >= 0
        if cur[i] - delta < 0:
            delta = cur[i]
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # compute candidate score
        af_c = np.fft.rfft(cand, L)
        conv_c = np.fft.irfft(af_c*af_c, L)[:2*n-1]
        cand_score = 2.0 * n * float(conv_c.max())
        # acceptance
        elapsed = time.monotonic() - start
        T = T0 * max(0.0, 1.0 - elapsed/total_time)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # accepting worse with prob exp(-(cand_score - cur_score) / (T + 1e-12))
            if T > 0:
                prob = math.exp(-(cand_score - cur_score) / (T + 1e-12))
                if rng.random() < prob:
                    accept = True
        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        # occasional global perturbation every certain iterations
        if it % 500 == 0:
            # add small Gaussian noise to all entries and renormalize
            noise = rng.normal(scale=0.01, size=n)
            cand2 = cur + noise
            cand2 = np.maximum(cand2, 0.0)
            cand2 = cand2 / cand2.sum()
            af2 = np.fft.rfft(cand2, L)
            conv2 = np.fft.irfft(af2*af2, L)[:2*n-1]
            score2 = 2.0 * n * float(conv2.max())
            if score2 < best_score:
                best = cand2.copy()
                best_score = score2
                cur = cand2
                cur_score = score2
        # early termination if best_score target achieved
        if best_score < 1.50274349:
            break

    # final rescaling to satisfy evaluator constraints: sum may be 1, fine.
    # Ensure each < 1000, >0, done.
    return [float(x) for x in best]

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max(a: np.ndarray, L: int) -> float:
    """Maximum value of a*a (full convolution) using FFT of length L."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ------------------------------------------------------------------
    # 1) Initialise a feasible point on the simplex.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        if s <= 0.0:
            cur = np.full_like(cur, 1.0 / cur.size)
        else:
            cur = cur / s                       # now sum = 1
        n = cur.shape[0]
    else:
        # fallback: moderate length, random simplex point
        n = 2000
        cur = rng.random(n)
        cur = cur / cur.sum()

    # ------------------------------------------------------------------
    # 2) Scan a few simple parametric families to possibly improve the start.
    # ------------------------------------------------------------------
    best_start = cur.copy()
    best_score = 2.0 * n * _conv_max(cur, 1 << ((2 * n - 1).bit_length()))
    # power‑law family a_i ∝ (i+1)^p  (p in [0,3])
    for p in np.linspace(0.0, 3.0, 31):
        a = (np.arange(n, 0, -1, dtype=np.float64) ** p)
        a = a / a.sum()
        sc = 2.0 * n * _conv_max(a, 1 << ((2 * n - 1).bit_length()))
        if sc < best_score:
            best_score = sc
            best_start = a
    cur = best_start.copy()
    cur_score = best_score
    best_vec = cur.copy()
    best_sc = cur_score

    # ------------------------------------------------------------------
    # 3) Optimisation loop – gradient‑guided random mass transfers + SA.
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())   # FFT length for full convolution
    indices = np.arange(n)

    total_time = deadline - start
    it = 0
    # annealing / step‑size schedules
    T0 = 0.05
    max_step_initial = 0.5

    while time.monotonic() < deadline:
        it += 1
        # --------------------------------------------------------------
        # current convolution, max index and sub‑gradient
        # --------------------------------------------------------------
        af = np.fft.rfft(cur, L)
        conv = np.fft.irfft(af * af, L)[: 2 * n - 1]
        max_idx = int(conv.argmax())
        cur_score = 2.0 * n * float(conv.max())

        # sub‑gradient of max(conv) w.r.t. a
        shift = max_idx - indices
        mask = (shift >= 0) & (shift < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[shift[mask]]

        # --------------------------------------------------------------
        # choose a source index (large gradient) and a destination index
        # --------------------------------------------------------------
        k = 200 if n > 200 else n
        top_idxs = np.argpartition(-grad, k - 1)[:k]
        gvals = grad[top_idxs]
        if gvals.sum() > 0:
            probs = gvals / gvals.sum()
            i = rng.choice(top_idxs, p=probs)
        else:
            i = rng.integers(0, n)

        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # --------------------------------------------------------------
        # mass transfer magnitude, schedule‑driven
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start
        progress = elapsed / total_time
        step_scale = max_step_initial * (1.0 - progress) + 0.02
        delta = cur[i] * rng.random() * step_scale
        if delta <= 0.0:
            continue
        if cur[i] - delta < 0.0:
            delta = cur[i]

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # --------------------------------------------------------------
        # evaluate candidate
        # --------------------------------------------------------------
        af_c = np.fft.rfft(cand, L)
        conv_c = np.fft.irfft(af_c * af_c, L)[: 2 * n - 1]
        cand_score = 2.0 * n * float(conv_c.max())

        # acceptance (simulated annealing)
        # --------------------------------------------------------------
        T = T0 * max(0.0, 1.0 - progress)

        if cand_score < cur_score:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(cand_score - cur_score) / (T + 1e-12))
                if rng.random() < prob:
                    accept = True
        if accept:

            cur_score = cand_score

        if it % 1200 == 0:

            pert = cur + noise

                best_vec = pert.copy()
                cur = pert
                cur_score = sc_pert

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best found sequence as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
