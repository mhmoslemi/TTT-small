import time, math, numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Compute the AC1 bound (assuming sum(a)=1)
# ------------------------------------------------------------
def _score_fft(a: np.ndarray, L: int) -> float:
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return 2.0 * a.size * conv.max()

# ------------------------------------------------------------
# Subgradient of the (soft) maximum of the linear convolution
# ------------------------------------------------------------
def _subgrad(a: np.ndarray, a_fft: np.ndarray, conv: np.ndarray,
             max_val: float, eps_factor: float, L: int) -> np.ndarray:
    n = a.size
    eps = max(max_val * eps_factor, 1e-12)
    mask = conv >= max_val - eps
    if not mask.any():
        mask = conv == max_val
    w = np.zeros_like(conv)
    w[mask] = 1.0 / mask.sum()
    w_fft = np.fft.rfft(w[::-1], n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]
