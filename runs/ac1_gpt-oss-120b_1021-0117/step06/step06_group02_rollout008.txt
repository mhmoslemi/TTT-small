def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, math, time
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # margin 0.2

    # Initialize a0
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:
        # fallback: exponential decay length maybe 2000
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 = a0 / a0.sum()
    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)   # power of two > 2n-1

    # Initialize x = log(a0 + eps)
    eps_init = 1e-12
    x = np.log(a0 + eps_init)

    # Adam parameters
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    t_iter = 0

    best_a = None
    best_score = np.inf

    # Smoothing epsilon parameters
    eps = None
    eps_decay = 0.995   # decays each iteration
    min_eps = 1e-14

    # random perturb schedule after stuck?
    no_improve = 0
    # maybe keep last improvement iteration
    best_iter = 0

    # Compute initial score and set best
    a = np.exp(x - np.max(x))
    a = a / a.sum()
    cur_score = evaluate_sequence(a.tolist())
    best_a = a.copy()
    best_score = cur_score

    while time.monotonic() < deadline:
        t_iter += 1
        # recompute a from x each iteration
        # Softmax
        x_shift = x - np.max(x)   # for numeric stability
        a = np.exp(x_shift)
        a = a / a.sum()

        # Convolution
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()

        # epsilon smoothing
        if eps is None:
            eps = max(max_val * 0.5, 1e-8)
        else:
            eps = max(eps * eps_decay, min_eps)

        # compute weights w
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t a
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)
        g_a = 2.0 * grad_full[grad_idx]

        # compute avg for softmax gradient
        avg = np.dot(a, g_a)
        grad_x = a * (g_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)

        # Learning rate schedule: maybe lr = lr0 / sqrt(t_iter)
        lr = lr0 / math.sqrt(t_iter)

        x = x - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # Occasionally add random noise to escape shallow local minima
        if t_iter % 500 == 0:
            # small random gaussian to x
            x += 0.001 * rng.standard_normal(n)

        # compute current score
        a = np.exp(x - np.max(x))
        a = a / a.sum()
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            best_iter = t_iter
        else:
            no_improve += 1

        # If stuck too long, random restart around best

        # early exit if below a threshold
        if best_score < 1.49:   # maybe safe

        # periodic time check already each loop

    # Return best

import time, math, numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.25)  # safety margin

    # ---------- initialise a feasible point on the simplex ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s                     # now sum = 1
    else:                               # simple fallback
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    # length for FFT (power of two > 2n‑1, doubled for safety)
    L = 1 << ((2 * n - 1).bit_length() + 1)

    # ---------- parameterisation via softmax ----------
    eps_log = 1e-12
    x = np.log(a0 + eps_log)               # unconstrained variables
    m = np.zeros_like(x)                   # first moment (Adam)
    v = np.zeros_like(x)                   # second moment (Adam)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0

    # ---------- optimisation hyper‑parameters ----------
    lr0 = 0.5                               # base learning rate
    eps_smooth = None
    eps_decay = 0.995
    min_eps = 1e-14
    best_score = np.inf
    best_vec = None
    no_improve = 0
    it = 0
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    # ---------- evaluate initial point ----------
    a = np.exp(x - np.max(x))
    a = a / a.sum()
    best_score = float(evaluate_sequence(a.tolist()))
    best_vec = a.copy()

    # ---------- main optimisation loop ----------
    while time.monotonic() < deadline:
        it += 1

        # softmax to obtain current a (sum = 1, a ≥ 0)
        x_shift = x - np.max(x)
        a = np.exp(x_shift)
        a = a / a.sum()

        # convolution a*a (full)
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # smoothing parameter for the soft‑max of the convolution
        if eps_smooth is None:
            eps_smooth = max(max_val * 0.5, 1e-8)
        else:
            eps_smooth = max(eps_smooth * eps_decay, min_eps)

        # weights for the smooth max
        w = np.exp((conv_full - max_val) / eps_smooth)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max w.r.t. a
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g_a = 2.0 * grad_full[grad_idx]

        # compose with soft‑max Jacobian
        avg = np.dot(a, g_a)
        grad_x = a * (g_a - avg)

        # Adam update on x
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        lr = lr0 / math.sqrt(it)
        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- occasional small random jitter -----
        if it % 500 == 0:
            x += 0.001 * rng.standard_normal(n)

        # ----- evaluate current a -----
        a = np.exp(x - np.max(x))
        a = a / a.sum()
        cur_score = float(evaluate_sequence(a.tolist()))

        # keep best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- try a symmetric mirror (no impact on x) -----
        a_sym = (a + a[::-1]) * 0.5
        if a_sym.sum() > 0.0:
            a_sym = a_sym / a_sym.sum()
            sym_score = float(evaluate_sequence(a_sym.tolist()))
            if sym_score < best_score:
                best_score = sym_score
                best_vec = a_sym.copy()

        # ----- restart when stuck -----
        if no_improve > 3000:
            # re‑initialise near the best solution
            x = np.log(best_vec + eps_log) + 0.01 * rng.standard_normal(n)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0
            eps_smooth = None
            no_improve = 0
            it = 0
            continue

        # ----- early exit if we beat a sensible target -----
        if best_score < 1.495:
            break

        # ----- time check -----
        if it % 200 == 0 and time.monotonic() >= deadline:
            break

    # fallback safety
    if best_vec is None:
        best_vec = a0

    return [float(v) for v in best_vec.tolist()]
