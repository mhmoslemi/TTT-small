import time

import numpy as np

# ------------------------------------------------------------
# Simplex projection ( Euclidean projection onto {x >= 0, sum(x)=1} )
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score: 2*n*max(convolution) for a vector with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(a*a) where a is (approximately) on the simplex."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of smoothed max (log-sum-exp style) using FFTs
# ------------------------------------------------------------
def _grad_smoothed_max(a: np.ndarray, a_fft, conv, max_val, eps, L):
    """Gradient of the smoothed max w.r.t. a."""
    # softmax weights (log‑sum‑exp smoothing)
    w = np.exp((conv - max_val) / eps)

    idx = (2 * a.shape[0] - 2) - np.arange(a.shape[0])
    g = 2.0 * grad_full[idx]

        # ----- convolution of current iterate (FFT) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- update smoothing factor -----
        eps_factor = max(eps_factor * eps_decay, 0.001)
        eps = max(max_val * eps_factor, eps_min)

        # ----- gradient of smoothed max -----
        g = _grad_smoothed_max(cur, a_fft, conv_full, max_val, eps, L)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_alpha * m_hat / (np.sqrt(v_hat) + adam_eps)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand, L)

        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            # occasional uphill move (tiny probability)
            prob = math.exp(-(cand_sc - cur_sc) / (base_alpha + 1e-12))
            if rng.random() < prob * 0.001:   # very low chance
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps flatten the profile) -----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random mass transfer when progress stalls -----
        if no_improve > 800:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        # ----- restart from best with small perturbation if hopeless -----
        if no_improve > 3000:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.3
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # return the best sequence found (as plain Python floats)
    return [float(x) for x in best_vec]
