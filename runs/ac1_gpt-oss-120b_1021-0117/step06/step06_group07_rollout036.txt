import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) onto {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum Z."""
    if Z <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Gradient from a set of convolution indices
# ------------------------------------------------------------
def _grad_from_indices(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max‑convolution using the supplied index set."""
    n = a.size
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    # average so magnitude stays comparable when many indices are used
    g /= max(1, idxs.size)
    return g
