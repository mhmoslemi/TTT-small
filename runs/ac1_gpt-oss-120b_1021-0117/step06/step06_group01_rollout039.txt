# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalized to sum=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            if j >= i:
                j += 1

            delta = cur[i] * rng.random() * 0.45
            if delta <= 0:
                continue
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            if tmp[i] < 0:
                continue
            sc = _score(tmp)
            if sc < cur_sc:
                cur, cur_sc = tmp, sc
                if sc < best_sc:
                    best_vec, best_sc = tmp.copy(), sc
            else:
                # occasional uphill acceptance (simulated annealing style)
                if rng.random() < math.exp(-(sc - cur_sc) * 0.1):
                    cur, cur_sc = tmp, sc

        else:
            # ---- small random perturbation plus projection ----
            pert = cur + 0.02 * rng.standard_normal(n)
            pert = np.maximum(pert, 0.0)
            pert = _project_simplex(pert)
            sc = _score(pert)
            if sc < cur_sc:
                cur, cur_sc = pert, sc
                if sc < best_sc:
                    best_vec, best_sc = pert.copy(), sc

        # ----------------------------------------------------
        # occasional symmetrisation (helps to balance the shape)
        # ----------------------------------------------------
        if rng.random() < 0.04:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sc = _score(sym)
            if sc < cur_sc:
                cur, cur_sc = sym, sc
                if sc < best_sc:
                    best_vec, best_sc = sym.copy(), sc

        # early exit if we beat the target
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # final safety clipping (the evaluator also clips)
    # --------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # return plain Python list for the harness
    return [float(x) for x in best_vec]
