    while time.time() < time_limit:
        t += 1
        # compute convolution
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        max_val = conv.max()
        # epsilon for softmax
        temp = max_val * 0.1 + 1e-12
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()

        else:

import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # safeguard sum (numerical)
    s = w.sum()
    if s <= 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / s
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (sum assumed = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a normalized (sum=1) non‑negative vector."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Adam optimiser with FFT‑based gradient of a smoothed maximum
# ------------------------------------------------------------
def _adam_opt(a_init: np.ndarray, time_budget: float, rng: np.random.Generator,
              target: float = 1.50274349) -> tuple[np.ndarray, float]:
    a = a_init.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.07                     # base learning rate
    t = 0
    best_a = a.copy()
    best_sc = _score(a)

    deadline = time.monotonic() + max(0.0, time_budget)
    while time.monotonic() < deadline:
        t += 1
        # ----- convolution and soft‑max weights -----
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        mval = conv.max()
        eps_temp = max(mval * 0.08, 1e-12)
        w = np.exp((conv - mval) / eps_temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed maximum -----
        w_rev = w[::-1]
        wf = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * wf, n=L)
        idx = (2 * n - 2) - np.arange(n)          # slice that lines up with a
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)
        sc_new = _score(a_new)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

            sc_sym = _score(sym)

        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
            cur = cand
        # optional early stop
        if best_sc < target:
            break
    return best

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list) with a low AC1 bound.
    The function respects the time budget and strives to beat the target.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ----- initial feasible vector (simplex) -----
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # exponential decay fallback
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # enforce sum = 1

    n = a0.size
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    target = 1.50274349
    # Allocate most of the time to repeated Adam optimisation, leave a bit for final refinement
    refine_time = 0.3
    while time.monotonic() < deadline - refine_time:
        # start from the current best with a tiny random perturbation
        cur = best_vec.copy()
        cur += rng.normal(scale=0.001, size=n)
        cur = np.maximum(cur, 0.0)
        cur /= cur.sum()

        # give each Adam run roughly 0.6 s (adjusted to remaining time)
        remaining = deadline - time.monotonic() - refine_time
        run_time = min(0.6, max(0.05, remaining * 0.5))
        cur, sc = _adam_opt(cur, run_time, rng, target)

        if sc < best_sc:
            best_sc = sc
            best_vec = cur.copy()
            if best_sc < target:
                break

    # ----- final local random refinement -----
    final_time = deadline - time.monotonic()
    if final_time > 0.0:
        best_vec = _local_refine(best_vec, final_time, target)

    # Return as plain Python list of floats
    return [float(x) for x in best_vec]
