def propose_candidate(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.1 (safe margin)
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full_like(a, 1.0/a.size)
        a = a / a.sum()
    else:
        n = 2000
        # generate uniform
        a = np.full(n, 1.0/n)

    n = a.size
    L = 1 << ((2*n-1).bit_length())

    best = a.copy()
    best_score = _score(a)  # use same function but also return conv? We'll compute conv each time.

    # parameters
    eps_factor = 0.05
    eps = None
    eta = 0.5  # initial learning rate
    eta_decay = 0.9995
    sym_interval = 200
    random_move_prob = 0.005

    while time.monotonic() < deadline:
        # compute convolution
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val / (1.0**2)  # sum is 1, so just 2n*max

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score < 1.50274349:  # early exit
                break

        # compute smoothed weights
        if eps is None:
            eps = max_val * eps_factor
            eps = max(eps, 1e-12)
        else:
            eps = max(eps * 0.999, 1e-12)  # slowly decay
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * conv(a, w_rev)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        # slice to length n as earlier:
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # exponentiated gradient update
        a = a * np.exp(-eta * g)
        # avoid all zeros
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # decay learning rate
        eta = max(eta * eta_decay, 1e-6)

        # periodic symmetrization
        if int(time.monotonic()*1000) % sym_interval == 0:  # Could use iteration count

            af_sym = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(af_sym * af_sym, n=L)[:2*n-1]
            max_sym = conv_sym.max()
            sym_score = 2.0 * n * max_sym
            if sym_score < cur_score:

# ------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ------------------------------------------------------------
def _full_conv(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _score_from_conv(conv: np.ndarray, total: float) -> float:
    """AC1 bound from a convolution array and total mass."""
    if total == 0.0:
        return float("inf")
    return float(2.0 * (len(conv) // 2 + 1) * conv.max() / (total * total))
