def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start=time.monotonic()
    deadline=start+budget_s-0.3  # small margin

    # init
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64) + 1e-6

    a = np.maximum(a, 0)
    if a.sum() < 1e-12:
        a[:] = 1
    a = a / a.sum()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # best tracking
    conv_full = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:2*n-1]
    best_conv = conv_full.max()
    best_sc = 2 * n * best_conv
    best_a = a.copy()

    # parameters
    eta0 = 0.5   # initial multiplicative step
    add_step0 = 0.5   # additive step magnitude for projected SGD fallback
    beta = 0.9
    momentum = np.zeros_like(a)
    top_k = 30
    it = 0
    no_improve = 0
    sym_interval = 200
    target_sc = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # compute convolution and max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_conv = conv_full.max()
        cur_sc = 2 * n * max_conv

        # update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target_sc:
            break

        # compute gradient on top-k entries
        if top_k >= conv_full.size:
            idxs = np.arange(conv_full.size)
        else:
            idxs = np.argpartition(-conv_full, top_k-1)[:top_k]

        grad = np.zeros_like(a)
        # vectorized accumulate using loops
        # For each k in idxs:
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # i indices
            i = np.arange(i_start, i_end + 1, dtype=int)
            grad[i] += 2.0 * a[k - i]

        # multiplicative update
        eta = eta0 / math.sqrt(it)  # decreasing
        a_new = a * np.exp(-eta * grad)
        # renormalize to sum 1
        a_new_sum = a_new.sum()
        if a_new_sum == 0 or np.isnan(a_new_sum):
            # fallback to additive SGD step
            g = grad / max(1, top_k)
            momentum = beta * momentum + (1 - beta) * g
            alpha = add_step0 / math.sqrt(it)
            a_tmp = a - alpha * momentum
            # project onto simplex
            a_new = _project_simplex(a_tmp)   # use defined projection
        else:
            a_new = a_new / a_new_sum

        # evaluate candidate
        a_fft_new = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(a_fft_new * a_fft_new, n=L)[:2*n-1]
        cand_sc = 2 * n * conv_new.max()

        # accept if improvement or with prob
        if cand_sc < cur_sc:
            a = a_new
            # no need to recompute cur_sc; will be recomputed next loop
            no_improve = 0
        else:
            # Simulated annealing accept with small prob
            prob = math.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                a = a_new

        # periodic symmetrisation
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            a_fft_sym = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_fft_sym * a_fft_sym, n=L)[:2*n-1]
            sym_sc = 2 * n * conv_sym.max()
            if sym_sc < cur_sc:
                a = a_sym

        # occasional random transfer when stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

        # occasional cheap LP refinement
        if it % 5000 == 0 and (deadline - time.monotonic()) > 10.0:
            # Solve LP to find g maximizing sum with convolution constraint
            M = max_conv
            # Build convolution matrix for current a
            # Build using sparse diags as before
            data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
            offsets = np.arange(n)
            A = sp.diags(data, offsets, shape=(2*n-1, n), format="csc")
            g = cp.Variable(n, nonneg=True)
            constraints = [A @ g <= M + 1e-12]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
            try:
                prob.solve(solver="ECOS", warm_start=True, max_iters=2000)
            except Exception:
                prob = None
            if g.value is not None:
                g_opt = np.maximum(g.value, 0)
                t_mix = 0.3
                a_mix = (1 - t_mix) * a + t_mix * g_opt
                a_mix = a_mix / a_mix.sum()
                # evaluate
                a_fft_mix = np.fft.rfft(a_mix, n=L)
                conv_mix = np.fft.irfft(a_fft_mix * a_fft_mix, n=L)[:2*n-1]
                mix_sc = 2 * n * conv_mix.max()
                if mix_sc < cur_sc:
                    a = a_mix

    # final sanity: ensure nonnegative and sum>0
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0:
        best_a = np.ones_like(best_a) / best_a.size
    else:
        best_a = best_a / best_a.sum()
    return [float(x) for x in best_a]

import time
import math
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries with sum Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # small safety margin

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64) + 1e-6

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    # force symmetry – empirical optimum is symmetric
    a = (a + a[::-1]) * 0.5
    a = _project_simplex(a)

    n = a.size
    # length for FFT (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_conv = conv_full.max()
    best_sc = 2.0 * n * best_conv
    best_vec = a.copy()

    # optimisation parameters
    eta0 = 0.8               # base multiplicative step size
    add_step0 = 0.6          # base additive step (fallback)
    beta = 0.85              # momentum coefficient
    momentum = np.zeros_like(a)
    top_k = 30               # number of convolution entries used for gradient
    sym_interval = 180       # how often to re‑symmetrise explicitly
    no_improve = 0
    it = 0
    target_sc = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target_sc:
            break

        # ----- gradient on the top‑k convolution entries -----
        if top_k >= conv_full.size:
            idxs = np.arange(conv_full.size)
        else:
            idxs = np.argpartition(-conv_full, top_k - 1)[:top_k]

        grad = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1, dtype=int)
            grad[i] += 2.0 * a[k - i]

        # ----- multiplicative (exponential) update -----
        eta = eta0 / math.sqrt(it)
        a_candidate = a * np.exp(-eta * grad)
        s = a_candidate.sum()
        if s <= 0.0 or np.isnan(s):
            # fallback to projected additive SGD
            g = grad / max(1, top_k)
            momentum = beta * momentum + (1.0 - beta) * g
            alpha = add_step0 / math.sqrt(it)
            a_tmp = a - alpha * momentum
            a_candidate = _project_simplex(a_tmp)
        else:
            a_candidate = a_candidate / s

        # ----- enforce symmetry and simplex -----

        a_candidate = _project_simplex(a_candidate)

            no_improve = 0

            offsets = np.arange(n)
            A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
            g = cp.Variable(n, nonneg=True)
            prob_lp = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M + 1e-12])
            try:
                prob_lp.solve(solver="ECOS", warm_start=True, max_iters=1500)
            except Exception:
                prob_lp = None
            if g.value is not None:
                g_opt = np.maximum(g.value, 0.0)
                tmix = 0.25
                a_mix = (1 - tmix) * a + tmix * g_opt
                a_mix = a_mix / a_mix.sum()
                a_mix = (a_mix + a_mix[::-1]) * 0.5
                a_mix = _project_simplex(a_mix)
                mix_fft = np.fft.rfft(a_mix, n=L)
                conv_mix = np.fft.irfft(mix_fft * mix_fft, n=L)[: 2 * n - 1]
                mix_sc = 2.0 * n * conv_mix.max()
                if mix_sc < cur_sc:
                    a = a_mix
                    cur_sc = mix_sc

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    if best_vec.sum() == 0.0:
        best_vec = np.ones_like(best_vec) / best_vec.size
    else:
        best_vec = best_vec / best_vec.sum()
    return [float(x) for x in best_vec]
