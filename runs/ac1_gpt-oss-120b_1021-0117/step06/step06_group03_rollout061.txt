import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast max of full self‑convolution (linear convolution) via FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max(conv(a,a)) using a power‑of‑two FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of 2)

    cur = a0.copy()
    cur_sc = 2.0 * n * _conv_max(cur, L)   # current score = 2*n*max(conv)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------
    beta1, beta2 = 0.9, 0.999            # Adam moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    eps_adam = 1e-8
    base_alpha = 1.0
    eps = None
    eps_decay = 0.995
    it = 0
    no_improve = 0
    improve_target = 1.50274349          # early‑stop if we ever beat this
    max_backtrack = 6
    sym_interval = 250

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and smoothed‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (matrix‑free) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- try a step with backtracking line‑search -----
        improved = False
        cand = None
        cand_sc = None
        alpha = 1.0
        for _ in range(max_backtrack):
            cand_try = cur - alpha * step
            cand_try = _project_simplex(cand_try)
            cand_sc = 2.0 * n * _conv_max(cand_try, L)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # ----- occasional uphill move (simulated annealing) -----
        if not improved:
            cand_try = cur - step * alpha
            cand_try = _project_simplex(cand_try)
            cand_sc = 2.0 * n * _conv_max(cand_try, L)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        # ----- accept improvement -----
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps flatten the profile) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = 2.0 * n * _conv_max(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when progress stalls -----
        if no_improve > 800:
            # pick two distinct indices and move a fraction of mass
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = 2.0 * n * _conv_max(tmp, L)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- restart from best with tiny perturbation if hopeless -----
        if no_improve > 2500:
            cur = best_vec + 0.005 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
