import time

# ------------------------------------------------------------
# Projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Fast AC1 bound for sequences with sum≈1 (score = 2*n*max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)  # safety margin

    # ------------------- initialise -------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    L = 1 << ((3 * n - 2).bit_length())      # >= 3n-2 for linear conv
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------- hyper‑parameters -------------------
    base_eta = 0.25               # base step size for exponentiated gradient
    eps_decay = 0.9995            # cooling for the smoothed max
    eps_min = 1e-12
    eps = None                    # will be set from first iteration
    sym_interval = 250
    no_improve = 0
    max_no_improve_restart = 3000
    clip_grad = 20.0              # avoid overflow in exp()
    target_score = 1.50274349

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----------- convolution and smoothed max -----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # initialise / decay epsilon
        if eps is None:
            eps = max_val * 0.5
        else:
            eps = max(eps * eps_decay, eps_min)

        # soft‑max weights
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----------- gradient of smoothed max -------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # ----------- exponentiated gradient step ----------
        norm_g = np.linalg.norm(g) + 1e-12
        eta = base_eta / math.sqrt(it) / norm_g
        g_clip = np.clip(g, -clip_grad, clip_grad)
        cur = cur * np.exp(-eta * g_clip)
        cur_sum = cur.sum()
        if cur_sum <= 0.0:
            cur = np.full_like(cur, 1.0 / cur.size)
        else:
            cur = cur / cur_sum

        # evaluate new point
        cur_sc = _score(cur)

        # keep the best ever seen
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # --------------- periodic symmetrisation ----------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # --------------- aggressive random mass‑transfer -
        if no_improve > 1500:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.6
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                elif rng.random() < 0.05:
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0

        # --------------- restart from best if stuck -------
        if no_improve > max_no_improve_restart:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            eps = None
            no_improve = 0

        # --------------- early exit if target reached ----
        if best_sc < target_score:
            break

        # --------------- periodic time check ------------
        if it % 500 == 0 and time.monotonic() >= deadline:
            break

    # return the best vector as plain Python floats
    return [float(x) for x in best_vec]
