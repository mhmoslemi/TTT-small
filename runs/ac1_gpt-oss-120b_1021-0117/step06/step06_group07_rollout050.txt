def _conv_and_mask(a, L, eps=1e-12):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*a.size-1]
    max_val = conv.max()
    mask = np.where(conv >= max_val - eps)[0]
    return max_val, mask
```

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries summing to Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    # shift by the mean to speed up sorting when possible
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = css[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Convolution (full) via FFT together with mask of maximal entries
# ------------------------------------------------------------
def _conv_and_mask(a: np.ndarray, L: int, eps: float = 1e-12):
    """Return (max value of a*a, indices where it is attained)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    max_val = conv.max()
    mask = np.where(conv >= max_val - eps)[0]
    return float(max_val), mask

# ------------------------------------------------------------
# Exact sub‑gradient of max_k (a*a)[k]
# ------------------------------------------------------------
def _subgrad(a: np.ndarray, mask):
    """Sub‑gradient of max‑convolution w.r.t. a (sum of 2*a shifted)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in mask:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g

    max_val, _ = _conv_and_mask(a, L)

    step0 = 0.6            # base step size

    beta = 0.9             # momentum factor
    momentum = np.zeros_like(a)
    sym_interval = 200
    smooth_interval = 400
