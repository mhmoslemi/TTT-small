def propose_candidate(seed=42, budget_s=1100.0, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # Determine n
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    else:
        n = 2048  # fallback? Actually we can set 2000 maybe
        base = rng.random(n)
    # Normalize base to sum=1
    base = np.maximum(base, 0.0)
    if base.sum() < 1e-12:
        base[:] = 1.0
    base = base / base.sum()

    # Compute L for FFT: power of 2 >= 2n-1
    L = 1 << ((2 * n - 1).bit_length())

    # Compute initial best score
    best_vec = base.copy()
    best_score = compute_score(best_vec, L, n)

    # Population initialization: size pop_size
    pop_size = 30
    # allocate matrix shape (pop_size, n)
    pop = np.empty((pop_size, n), dtype=np.float64)
    # first individual is best
    pop[0] = best_vec
    # Others random Dirichlet: sample exponential and normalize.
    for i in range(1, pop_size):
        # sample from exponential distribution
        x = rng.exponential(scale=1.0, size=n)
        # ensure not all zero
        pop[i] = x / x.sum()
    # compute scores
    scores = np.empty(pop_size, dtype=np.float64)
    for i in range(pop_size):
        scores[i] = compute_score(pop[i], L, n)
        if scores[i] < best_score:
            best_score, best_vec = scores[i], pop[i].copy()

    # main DE loop
    gen = 0
    # parameters
    F = 0.8
    CR = 0.9
    # maybe adapt gradually: reduce F or CR over time

    while time.monotonic() < deadline:
        gen += 1

        # adapt parameters occasionally
        if gen % 30 == 0:
            # Slightly randomize to keep diversity
            F = 0.5 + 0.5 * rng.random()
            CR = 0.5 + 0.5 * rng.random()

        # For each target
        for i in range(pop_size):
            # mutation: best + F*(r1 - r2)
            # pick r1, r2 distinct from each other and from i
            idxs = rng.choice(pop_size, size=2, replace=False)
            while idxs[0] == i or idxs[1] == i:
                idxs = rng.choice(pop_size, size=2, replace=False)
            r1, r2 = pop[idxs[0]], pop[idxs[1]]
            mutant = best_vec + F * (r1 - r2)
            # clip to nonnegative (positive)
            np.maximum(mutant, 0.0, out=mutant)
            # optionally renorm to sum=1 to keep scale stable
            s = mutant.sum()
            if s < 1e-12:
                mutant = rng.exponential(scale=1.0, size=n)
                s = mutant.sum()

def _project_simplex(v, Z=1.0):
    # ensure nonnegative projection onto simplex sum Z.
    # If all non-positive, return uniform vector Z/n.
    # Implementation uses sorting approach.
    n = v.size
    # In case Z==0 return zeros.
    if Z <= 0:
        return np.zeros_like(v)
    # Sorting
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # in rare case sum zero due to rounding, adjust
    s = w.sum()
    if s == 0:
        w[:] = Z / n
    return w

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: simple renormalisation to the probability simplex (non‑negative,
# sum = 1).  Fast – no sorting.
# ----------------------------------------------------------------------
def _simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


# ----------------------------------------------------------------------
# Objective for a given vector (no scaling needed – the expression is
# invariant under positive scaling).
# ----------------------------------------------------------------------
def _score(a: np.ndarray, L: int, n: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution term (uses the index of the maximal
# entry only – a valid sub‑gradient).
# ----------------------------------------------------------------------
def _subgrad_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    k = int(np.argmax(conv))
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i = np.arange(i_start, i_end + 1, dtype=int)
    g = np.zeros_like(a)
    g[i] = 2.0 * a[k - i]
    return g


# ----------------------------------------------------------------------
# A few projected‑gradient steps starting from `vec`.  Stops early if the
# global deadline is near.
# ----------------------------------------------------------------------
def _refine(vec: np.ndarray, L: int, n: int, deadline: float,
            max_iters: int = 200) -> tuple[np.ndarray, float]:
    step0 = 0.7
    beta = 0.9
    momentum = np.zeros_like(vec)
    cur_sc = _score(vec, L, n)
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        it += 1
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        g = _subgrad_max(vec, conv)
        momentum = beta * momentum + (1.0 - beta) * g
        alpha = step0 / math.sqrt(it)
        cand = vec - alpha * momentum
        cand = _simplex(cand)
        cand_sc = _score(cand, L, n)
        if cand_sc < cur_sc:
            vec, cur_sc = cand, cand_sc
            momentum.fill(0.0)          # restart momentum after improvement
    return vec, cur_sc

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Base sequence (may be supplied by the harness)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

    base = _simplex(base)          # ensure feasibility

    n = base.size
    L = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two for FFT

    # ------------------------------------------------------------------
    # Keep the best solution discovered
    # ------------------------------------------------------------------
    best_vec = base.copy()
    best_sc = _score(best_vec, L, n)

    # ------------------------------------------------------------------
    # Differential‑Evolution style population
    # ------------------------------------------------------------------
    pop_size = 50
    pop = np.empty((pop_size, n), dtype=np.float64)
    pop[0] = best_vec
    for i in range(1, pop_size):

        pop[i] = x / x.sum()

    for i in range(pop_size):
        scores[i] = _score(pop[i], L, n)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    gen = 0

        if gen % 30 == 0:

            CR = 0.5 + 0.5 * rng.random()

        for i in range(pop_size):

            # pick two distinct individuals different from i
            idx = rng.choice(pop_size, size=2, replace=False)
            while i in idx:
                idx = rng.choice(pop_size, size=2, replace=False)
            r1, r2 = pop[idx[0]], pop[idx[1]]
            mutant = best_vec + F * (r1 - r2)
            mutant = _simplex(mutant)

            # --- crossover -------------------------------------------
            mask = rng.random(n) < CR
            if not np.any(mask):

            trial = np.where(mask, mutant, pop[i])
            trial = _simplex(trial)

            # --- selection -------------------------------------------
            sc = _score(trial, L, n)
            if sc < scores[i]:
                pop[i] = trial
                scores[i] = sc

                # replace the worst individual with the refined one

                pop[worst] = refined
                scores[worst] = refined_sc

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
