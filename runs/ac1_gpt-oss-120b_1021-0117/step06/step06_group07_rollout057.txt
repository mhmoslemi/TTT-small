import time
import math
import numpy as np

# ------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex:
#   {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution by FFT (length L >= 2n‑1)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * a.size - 1]


# ------------------------------------------------------------
# Sub‑gradient of max‑convolution (exact, using the active set)
# ------------------------------------------------------------
def _subgrad_maxconv(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """Compute a sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    # indices where the maximum is attained (allow a tiny tolerance)
    idxs = np.where(conv >= max_val - 1e-12)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        g[i] += a[j]           # derivative from a_i * a_{k-i}
        g[j] += a[i]           # derivative from a_{k-i} * a_i
    return g

    n = a.size

    best_vec = a.copy()
    cur_max = _conv_full(a, L).max()
    best_score = 2.0 * n * cur_max      # corresponds to evaluate_sequence with sum=1

    # optimisation parameters
