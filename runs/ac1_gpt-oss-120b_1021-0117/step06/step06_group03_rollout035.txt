import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast computation of the objective 2*n*max(conv) for sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        direction = m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- step‑size (decaying) -----
        norm_dir = np.linalg.norm(direction) + 1e-12
        alpha = 1.0 / (math.sqrt(it) * norm_dir)

        # ----- candidate and acceptance test -----
        cand = cur - alpha * direction
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps flatten profile) -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random perturbation when progress stalls -----
        if no_improve > 1500:
            # restart from a random Dirichlet point
            cur = rng.dirichlet(np.ones(n))
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target_score:
            break

        # ----- safety check for deadline -----
        if time.monotonic() >= deadline:
            break

    # --------------------------------------------------------
    # final local refinement: tiny mass‑transfer moves
    # --------------------------------------------------------
    refin_iter = 0
    while time.monotonic() < deadline and refin_iter < 300:
        refin_iter += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or best_vec[i] == 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.1
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        sc = _score(tmp)
        if sc < best_sc:
            best_sc, best_vec = sc, tmp.copy()
            # reset stagnation counter for potential further refinements
            no_improve = 0

    # return the best sequence as plain Python floats
    return [float(x) for x in best_vec]
