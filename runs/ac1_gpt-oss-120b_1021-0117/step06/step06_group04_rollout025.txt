import time, math, numpy as np

# ----------------------------------------------------------------------
# helper functions (no closures)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Convolution a*a via FFT, return (max, argmax, full conv)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def _score_and_conv(a: np.ndarray, n: int, L: int):
    """Return score, max conv value, full conv and its index."""
    max_val, idx, conv = _conv_max_and_index(a, L)
    sc = 2.0 * n * max_val / (a.sum() ** 2)
    return sc, max_val, conv, idx


def _optimize_from(a0: np.ndarray, n: int, L: int, rng: np.random.Generator,
                   deadline: float, target_score: float = 1.50274349):
    """Gradient‑like descent starting from a0, returns best vector and its score."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                                   # simplex projection

    best_a = a.copy()
    best_sc, _, _, _ = _score_and_conv(a, n, L)
    cur_a = a
    cur_sc = best_sc

    momentum = np.zeros_like(a)
    beta = 0.9
    base_alpha = 0.6
    max_trials = 6
    no_improve = 0
    sym_interval = 300
    transfer_thresh = 800
    smooth_interval = 500
    kernel = np.full(5, 0.2, dtype=np.float64)     # simple smoothing kernel

    while time.monotonic() < deadline:
        # current score and index of maximal convolution entry
        cur_sc, _, _, idx = _score_and_conv(cur_a, n, L)

        # sub‑gradient of the maximal entry
        k = idx
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        grad = np.zeros_like(cur_a)
        if hi >= lo:
            j = np.arange(lo, hi + 1)
            grad[j] = 2.0 * cur_a[k - j]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(no_improve + 1) * norm_m)

        # back‑tracking line search
        improved = False
        alpha_try = alpha
        for _ in range(max_trials):
            cand = cur_a - alpha_try * momentum
            cand = _project_simplex(cand)
            cand_sc, _, _, _ = _score_and_conv(cand, n, L)
            if cand_sc < cur_sc:
                improved = True
                break
            alpha_try *= 0.5

        # occasional uphill move
        if not improved:
            cand = cur_a - alpha_try * momentum
            cand = _project_simplex(cand)
            cand_sc, _, _, _ = _score_and_conv(cand, n, L)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha_try + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur_a = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # periodic symmetrisation
        if (no_improve % sym_interval) == 0:
            sym = (cur_a + cur_a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc, _, _, _ = _score_and_conv(sym, n, L)
            if sym_sc < cur_sc:
                cur_a, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_a = sym.copy()
                no_improve = 0

        # random mass transfer when progress stalls
        if no_improve > transfer_thresh:
            i = rng.integers(0, n)
            if cur_a[i] > 0.0:
                delta = cur_a[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur_a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc, _, _, _ = _score_and_conv(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur_a, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_a = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = max(0, no_improve - 200)

        # occasional smoothing
        if (no_improve % smooth_interval) == 0:
            smooth = np.convolve(cur_a, kernel, mode='same')
            cur_a = _project_simplex(smooth)

        # early exit if target reached
        if best_sc < target_score:
            break

    return best_a, best_sc

# main entry point required by the harness

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # build a handful of diverse starting points
    # ------------------------------------------------------------------
    starts = []

    if "height_sequence_1" in globals():
        starts.append(np.asarray(height_sequence_1, dtype=np.float64).copy())

    # random simplex points
    for _ in range(3):
        rnd = rng.random(n)
        starts.append(_project_simplex(rnd))

    # exponential decays with a few rates
    for c in (5e-4, 1e-3, 2e-3):
        vec = np.exp(-c * np.arange(n, dtype=np.float64))
        starts.append(_project_simplex(vec))

    best_vec = None
    best_sc = np.inf

    # ------------------------------------------------------------------
    # run optimisation from each start, respecting the global deadline
    # ------------------------------------------------------------------
    for a0 in starts:
        now = time.monotonic()
        if now >= deadline - 5.0:
            break
        # give each run at most 200 s or the remaining budget, whichever is smaller
        remaining = deadline - now
        local_deadline = now + min(200.0, remaining * 0.5)
        vec, sc = _optimize_from(a0, n, L, rng, local_deadline)
        if sc < best_sc:
            best_sc, best_vec = sc, vec.copy()
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # final fine‑tuning by tiny random transfers
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_sc, _, _, _ = _score_and_conv(best_vec, n, L)

    cur = best_vec.copy()
    cur_sc = best_sc
    while time.monotonic() < deadline - 0.1:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.2
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc, _, _, _ = _score_and_conv(tmp, n, L)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
        if best_sc < 1.50274349:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
