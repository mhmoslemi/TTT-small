def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)  # safety
    # init
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = np.exp(-np.arange(n) / 100.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 0.01:
        a = np.full_like(a, 1.0 / a.size)
        total = 1.0
    a = a / total  # sum=1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    # compute score
    def score(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()
    cur = a.copy()
    cur_score = score(cur)
    best = cur.copy()
    best_score = cur_score
    momentum = np.zeros_like(cur)
    beta = 0.9
    base_alpha = 0.8
    # iterate
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # compute gradient of max(conv) using softmax smoothing
        af = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv.max()
        # smoothing factor
        eps = max_val * 0.1 + 1e-12
        # softmax weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        w /= w_sum
        # gradient: convolution of cur with reversed weights
        w_rev = w[::-1]
        wf = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * wf, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        # step size
        norm = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (np.sqrt(it) * norm)
        # propose update
        cand = cur - alpha * momentum
        # project onto simplex
        cand = _project_simplex(cand)
        cand_score = score(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        # occasional symmetrization
        if it % 250 == 0:

import time, numpy as np, math

def _project_simplex(v):
    # same as before
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a: np.ndarray) -> float:
    """Fast AC1 bound for a vector whose sum is (approximately) 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power‑of‑2 ≥ 2n‑1
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
