import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # guard against pathological all‑zero output
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w


# ----------------------------------------------------------------------
# Convolution via FFT (full mode)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[: len(x) + len(y) - 1]

        # occasional smoothing to dampen high‑frequency spikes
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            # re‑normalise to stay on the simplex
            if smooth.sum() > 0.0:
                smooth /= smooth.sum()
                conv_s = _fft_convolve(smooth, smooth, L)
                score_s = 2.0 * n * conv_s.max()
                if score_s < cur_score:
                    a = smooth
                    cur_score = score_s
                elif rng.random() < math.exp(-(score_s - cur_score) / max(T, 1e-12)):
                    a = smooth
                    cur_score = score_s

    # --------------------------------------------------------------
    # Return the best solution discovered
    return [float(x) for x in best]
