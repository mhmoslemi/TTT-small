import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum=1, all >=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a sequence that already sums to 1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv) assuming sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------- initial vector -----------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: Dirichlet‑like random positive vector
        n_fallback = 2000
        a0 = rng.exponential(1.0, size=n_fallback)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)               # normalise to sum=1

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())     # FFT length >= 3n‑2

    # ------------------- optimisation state -------------------
    cur = a0.copy()
    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam‑style moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    base_lr0 = 0.5

    # smoothing schedule for the max
    eps0, eps_decay, eps_min = 5.0, 0.9993, 1e-6

    no_improve = 0
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smoothed weights -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]      # gradient of 2*n*max(conv) w.r.t. a

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        # bias‑correction
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = (base_lr0 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + adam_eps)
        cand = cur - step

        # keep nonnegativity, enforce symmetry, re‑project onto simplex
        cand = np.maximum(cand, 0.0)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        # ----- evaluation -----
        cand_sc = _score(cand, L)

        # ----- acceptance -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # occasional Metropolis acceptance to escape plateaus
            prob = math.exp(-(cand_sc - cur_sc) / (base_lr0 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- random mass transfer if stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- periodic full symmetrisation (helps numerical noise) -----
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- early exit -----
        if best_sc < target:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
