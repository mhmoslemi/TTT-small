import time
import numpy as np

def _topk_gradient(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    """
    Approximate sub‑gradient of max(conv) using the top_k largest convolution values.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    size = conv.shape[0]
    # indices of the largest entries
    if size > top_k:
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    else:
        idxs = np.arange(size)
    grad = np.zeros_like(a)
    for k in idxs:
        offset = k - (n - 1)          # shift of the contributing pairs
        if offset >= 0:
            i_start = offset
            i_end = n
            partner = k - np.arange(i_start, i_end, dtype=np.int64)
            grad[i_start:i_end] += a[partner]
        else:
            i_start = 0
            i_end = n + offset
            partner = k - np.arange(i_start, i_end, dtype=np.int64)
            grad[i_start:i_end] += a[partner]
    # each pair appears twice in the convolution sum
    grad = 2.0 * grad / max(1, len(idxs))
    return grad


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Uses an exponentiated‑gradient descent on the simplex with
    occasional jitter and restart on stagnation.
    Returns the best coefficient list found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time management (leave a small safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initial sequence (fallback to uniform if no parent is provided)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)

        a = np.maximum(parent, 0.0)
        s = a.sum()
        if s < 1e-12:

        else:
            a = a / s                     # normalise to sum = 1
    except Exception:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Pre‑computations
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    top_k = 3                                     # number of active peaks
    eta0 = 1.5                                    # initial step magnitude
    jitter_interval = 200                        # how often to perturb
    jitter_std = 0.025                           # strength of perturbation
    max_stale = 5000                             # restart after this many fails
    target_score = 1.50274349

    # ------------------------------------------------------------------
    # Helper to evaluate current score (sum is 1, so denominator = 1)
    # ------------------------------------------------------------------
    def _score_from_conv(c):
        return float(2.0 * n * c.max())

    # ------------------------------------------------------------------
    # Initial best candidate
    # ------------------------------------------------------------------
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best = a.copy()
    best_score = _score_from_conv(conv)

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0
    while time.monotonic() < deadline:

        cur_score = _score_from_conv(conv)
