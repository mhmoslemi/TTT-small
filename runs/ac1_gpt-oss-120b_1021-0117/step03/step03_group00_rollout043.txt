import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >=0, sum=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast objective for vectors with sum ≈ 1: 2*n*max(conv) / sum^2
# ------------------------------------------------------------
def _fast_score(vec: np.ndarray, L: int) -> float:
    """Return the AC1 bound for a non‑negative vector (sum ≈ 1)."""
    n = vec.size
    af = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = vec.sum()
    if s == 0.0:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple uniform fallback
        n_fallback = 2500
        cur = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    if cur.sum() == 0.0:
        cur.fill(1.0 / cur.size)
    else:
        cur = _project_simplex(cur)

    n = cur.size
    # FFT length sufficient for linear convolution of length up to 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    best_vec = cur.copy()
    best_sc = _fast_score(best_vec, L)

    cur_sc = best_sc

    # ---- optimisation parameters ----------------------------------------
    eps_cur = 0.5                 # initial smoothing
    eps_decay = 0.998             # exponential decay per iteration
    eps_min = 1e-9
    step0 = 1.0                  # base step magnitude
    beta = 0.9                   # momentum coefficient
    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0
    target = 1.50274349

    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate ----------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smoothed max weights --------------------------------------
        eps_cur = max(eps_min, eps_cur * eps_decay)
        w = np.exp((conv - max_val) / eps_cur)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)  # length >= 3n‑2
        idx = (2 * n - 2) - np.arange(n)               # map to proper slice
        g = 2.0 * grad_full[idx]

        # ----- momentum & step -------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * g
        alpha = step0 / math.sqrt(it)
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)

        # ----- evaluate --------------------------------------------------
        cand_sc = _fast_score(cand, L)

        # ----- accept if improvement ------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                if best_sc < target:
                    break
        else:
            no_improve += 1

        # ----- occasional symmetrisation ---------------------------------
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                    if best_sc < target:
                        break

        # ----- random mass transfer if stuck -----------------------------
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _fast_score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # safety break if time is almost up
        if time.monotonic() > deadline - 0.1:
            break

    # return the best vector found as a plain Python list
    return [float(x) for x in best_vec]
