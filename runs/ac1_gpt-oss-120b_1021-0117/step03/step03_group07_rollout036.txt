import time, numpy as np

def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n - 1]
    return conv

def _score(a): # assume a is normalized sum=1
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())  # denominator 1

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective used by the evaluator (sum of a is assumed = 1)."""
    n = a.shape[0]

    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()

    # logistic random trials
    for _ in range(200):
        c = rng.uniform(0, n - 1)
        scale = rng.exponential(scale=n / 10.0) + 1e-6
        cand = _make_logistic(n, c, scale)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    # random dense draws (uniform Dirichlet‑like)
    for _ in range(200):
        cand = rng.random(n)
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # -----------------------------------------------------------------
    # 2) Multi‑lag exponentiated‑gradient descent (momentum)
    # -----------------------------------------------------------------
    eta0 = 0.8
    beta = 0.9
    momentum = np.zeros_like(cur)
    it = 0
    target = 1.50274349
    top_k = 5

    while time.monotonic() < deadline:
        it += 1
        conv = _conv_fft(cur)

        # indices of the largest convolution values
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient of the weighted sum of the chosen lags
        g = np.zeros_like(cur)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # exponentiated‑gradient step
        eta = eta0 / np.sqrt(it)
        delta = -eta * momentum
        delta = np.clip(delta, -700, 700)          # avoid overflow
        cand = cur * np.exp(delta)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand /= cand_sum
        else:
            cand = np.full_like(cur, 1.0 / n)

        cand_sc = _score(cand)

        # simple improvement rule + occasional annealing
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break
        else:
            T = 0.02 / (np.sqrt(it) + 1e-12)
            if rng.random() < np.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # occasional jitter for escaping plateaus
        if it % 2000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0:
                tmp /= tmp_sum
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                        if best_sc < target:
                            break

        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # 3) Simple random mass‑transfer polishing
    # -----------------------------------------------------------------
    it2 = 0
    while time.monotonic() < deadline:
        it2 += 1
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n - 1))
        if j >= i:
            j += 1
        max_delta = cur[i] * 0.2
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break
        if it2 % 5000 == 0:
            # tiny jitter to keep exploration alive
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0:
                tmp /= tmp_sum
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                        if best_sc < target:
                            break
        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # final safeguard
    # -----------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
