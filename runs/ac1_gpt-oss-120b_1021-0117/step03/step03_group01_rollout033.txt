# Simplex projection (onto probability simplex, sum = 1)
# -----------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:      # guard against pathological case
        w[:] = 1.0 / n
    return w

# -----------------------------------------------------------
# Convolution (full) via FFT, return maximal entry
# -----------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    return float(conv.max())

# -----------------------------------------------------------
# Score for a sequence whose sum is 1 (scale‑invariant)
# -----------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    return 2.0 * n * _max_convolution(a)

# -----------------------------------------------------------
# Solve LP: max sum(b)  s.t.  b >= 0 ,  conv(f,b) <= M
# -----------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float):
    """Return feasible b (numpy array) or None."""
    try:
        import cvxpy as cp
    except Exception:               # cvxpy not available
        return None

    n = f.size
    b = cp.Variable(n, nonneg=True)
    fpar = cp.Parameter(n)
    fpar.value = f
    conv_expr = cp.conv(fpar, b)          # length 2n‑1
    # small tolerance to avoid numerical infeasibility
    constraints = [conv_expr <= M + 1e-12]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        return None
    if b.value is None:
        return None
    return np.array(b.value, dtype=np.float64)

# -----------------------------------------------------------
# Analytic sqrt series (a*a == 1 + … + x^{2n‑2})
# -----------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# -----------------------------------------------------------
# Main optimisation loop using alternating LP updates
# -----------------------------------------------------------
def _lp_optimize(start: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _score(a)

    # mixing coefficient (will decay)
    t = 0.5
    min_t = 0.05
    decay = 0.95

    while time.monotonic() < deadline:
        # 1) current max of autocorrelation
        M = _max_convolution(a)

        # 2) solve LP for a better partner b
        b = _solve_lp(a, M)
        if b is None:
            break
        sumb = b.sum()
        if sumb <= 0.0:
            break
        g = b / sumb                     # renormalise to sum=1

        # 3) convex combination and re‑project
        a = (1.0 - t) * a + t * g
        a = _project_simplex(a)

        # 4) evaluate
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            # optional: tighten mixing as we improve
            t = max(min_t, t * decay)

        # 5) early‑exit guard
        if time.monotonic() + 0.5 > deadline:
            break

    return best, best_score

# -----------------------------------------------------------
# Random local refinement (tiny perturbations)
# -----------------------------------------------------------
def _local_refine(vec: np.ndarray, deadline: float, rng: np.random.Generator):
    best = vec.copy()
    best_score = _score(best)
    n = vec.shape[0]
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=0.02, size=n)
        cand = np.clip(cand, 0.0, None)
        cand = _project_simplex(cand)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best = cand
        if time.monotonic() + 0.01 > deadline:
            break
    return best

# -----------------------------------------------------------
# Public entry point
# -----------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # initialise -------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = _sqrt_series(256)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # primary optimisation ---------------------------------------
    best, best_score = _lp_optimize(a0, deadline, rng)

    # optional second random start if time permits ---------------
    if time.monotonic() + 5.0 < deadline:
        rand_vec = rng.random(a0.shape)
        rand_vec = _project_simplex(rand_vec)
        alt, alt_score = _lp_optimize(rand_vec, deadline, rng)
        if alt_score < best_score:
            best, best_score = alt, alt_score

    # final local refinement ------------------------------------
    if time.monotonic() + 0.5 < deadline:
        best = _local_refine(best, deadline, rng)

    # return plain python list ----------------------------------
    return [float(x) for x in best]
