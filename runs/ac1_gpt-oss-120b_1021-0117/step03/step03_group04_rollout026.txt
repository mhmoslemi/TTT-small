def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,budget_s-0.5)
    # get length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 256
        base = None
    # projection onto simplex
    def proj_simplex(v):
        # using sorting
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, len(v)+1)
        mask = u - cssv/ind > 0
        if not mask.any():
            return np.full_like(v, 1.0/len(v))
        rho = np.where(mask)[0][-1]
        theta = cssv[rho]/(rho+1)
        w = np.maximum(v - theta, 0.0)
        return w
    # init candidates
    a = np.full(n, 1.0/n)
    best = a.copy()
    best_score = float('inf')
    # fast scoring via FFT
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len-1).bit_length())
    def fast_score(x):
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()
    # evaluate initial uniform
    cur = fast_score(a)
    best_score = cur
    best = a.copy()
    # optional extra candidates
    # tent shape
    idx = np.arange(n, dtype=float)
    tent = np.minimum(idx+1, n-idx) - 1
    tent = np.maximum(tent, 0)
    if tent.sum()>0:
        tent = tent/tent.sum()
        sc = fast_score(tent)
        if sc < best_score:
            best_score = sc
            best = tent.copy()
    # random Dirichlet
    # iterate while time
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # sample Dirichlet via exponential
        rand = rng.exponential(scale=1.0, size=n)

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(x: np.ndarray, n: int, fft_len: int) -> float:
    """Fast surrogate for the AC1 bound (sum(x)=1, so denominator=1)."""
    fa = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()
