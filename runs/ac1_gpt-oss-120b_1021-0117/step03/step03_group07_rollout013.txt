import time
import numpy as np

# ----- convolution and scoring -------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum not forced to 1)."""
    s = a.sum()
    if s == 0:
        return float("inf")
    conv = _conv_fft(a)
    return float(2.0 * a.shape[0] * conv.max() / (s * s))


# ----- families of candidate sequences ----------------------------------------
def _make_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r^i, normalized to sum 1."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalized to sum 1."""
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_linear(n: int) -> np.ndarray:
    """Strictly decreasing linear profile."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    """Strictly decreasing quadratic profile."""
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_sqrt_series(n: int) -> np.ndarray:
    """Iterative construction from the reference code."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a
