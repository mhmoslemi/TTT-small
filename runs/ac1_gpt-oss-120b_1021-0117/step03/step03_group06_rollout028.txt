import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using FFT."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]

def _corr_fft(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """Correlation of w (len 2n-1) with a (len n) via FFT.
    Returns the 'valid' part of length n."""
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(fw * fa, n=L)
    # valid part starts at n-1, length = n
    start = a_rev.size - 1
    return full[start:start + a_rev.size]

                tmp = np.maximum(tmp, 0.0)
                tmp /= tmp.sum()
                tmp_score = 2.0 * n * _conv_fft(tmp, L_conv).max()
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score
                    cur = tmp
                    no_improve = 0

        # ----- cooling of the smoothing parameter -----
        mu = max(min_mu, mu * mu_decay)

        # ----- restart after long stagnation -----
        if no_improve > 6000:
            jitter = rng.normal(scale=0.02, size=n)
            cur = best + jitter
            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()
            no_improve = 0

        # ----- early exit if target reached -----
        if best_score < 1.50274349:
            break

    # Return the best vector found as a plain Python list
    return [float(v) for v in best]
