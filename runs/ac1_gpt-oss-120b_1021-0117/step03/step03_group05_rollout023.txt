import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine respects a time budget and retains the best solution found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # short safety margin

    # --------------------------------------------------------------
    # 1.  Initial vector (use provided parent if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a simple analytic starter: short cosine‑shaped bump
        n0 = 256
        x = np.arange(n0)
        a0 = np.cos(np.pi * (x - n0 / 2) / n0) ** 2
    # project onto the simplex (non‑negative, sum = 1)
    def _project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto {x >= 0, sum x = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    a0 = np.maximum(a0, 0.0)
    cur = _project_simplex(a0)
    n = cur.size

    # --------------------------------------------------------------
    # 2.  Helper utilities (FFT convolution, score)
    # --------------------------------------------------------------
    _L = 1 << ((2 * n - 1).bit_length())           # power‑of‑two length for FFT

    def _conv_fft(v: np.ndarray) -> np.ndarray:
        fv = np.fft.rfft(v, n=_L)
        conv = np.fft.irfft(fv * fv, n=_L)
        return conv[: 2 * n - 1]

    def _score(v: np.ndarray) -> float:
        conv = _conv_fft(v)
        # because sum(v) == 1 after projection we can skip the denominator
        return float(2.0 * n * conv.max())

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------------
    # 3.  Optimisation parameters
    # --------------------------------------------------------------
    it = 0
    no_improve = 0
    K = 30                     # active set size for sub‑gradient
    add_alpha0 = 1.2           # base additive step size
    mul_eta0 = 0.8             # base multiplicative step size
    beta = 0.7                 # momentum coefficient
    direction = np.zeros_like(cur)   # momentum buffer
    target = 1.50274349        # stop once beaten

    # --------------------------------------------------------------
    # 4.  Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # (a) convolution and active set
        conv = _conv_fft(cur)
        max_idx = int(np.argmax(conv))
        # choose K largest indices (including the absolute maximum)
        if K >= conv.size:
            active = np.arange(conv.size)
        else:
            # np.argpartition is O(n) and returns unsorted K‑largest indices
            active = np.argpartition(-conv, K - 1)[:K]

        # (b) sub‑gradient from active indices
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]          # derivative of (a*a)_k

        # (c) momentum update
        direction = beta * direction + (1.0 - beta) * g

        # ----------------------------------------------------------

        # ----------------------------------------------------------
        alpha = add_alpha0 / np.sqrt(it + 0.5)

            cand_add_sc = _score(cand_add)
            ls_it += 1

        # ----------------------------------------------------------

        # ----------------------------------------------------------
        eta = mul_eta0 / np.sqrt(it + 0.5)
