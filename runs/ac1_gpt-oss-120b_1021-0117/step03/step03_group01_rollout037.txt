def _conv_max_and_index(vec, L=None):
    n = len(vec)
    if L is None or L < 2*n - 1:
        L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx
```

import time
import math
import numpy as np
import scipy.sparse as sp

# optional cvxpy import
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum == 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w


# ----------------------------------------------------------------------
# Convolution (via FFT) – return the maximal value and its index
# ----------------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int = None):
    n = vec.size
    if L is None or L < 2 * n - 1:
        L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Sparse matrix that computes convolution a * b  (a fixed, b variable)
# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> sp.csc_matrix:
    n = a.size
    # each diagonal offset i (0 … n‑1) is filled with a[i]
    diags = [np.full(n, ai, dtype=np.float64) for ai in a]
    offsets = np.arange(n, dtype=np.int32)
    return sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")


# ----------------------------------------------------------------------
# Linear program: max sum(b)   s.t.   a * b ≤ M   ,  b ≥ 0
# ----------------------------------------------------------------------
def _solve_lp(a: np.ndarray, M: float):
    if not _HAS_CVX:
        return None
    n = a.size
    C = _conv_matrix(a)
    b = cp.Variable(n)
    constraints = [b >= 0,
                   C @ b <= M * np.ones(2 * n - 1, dtype=np.float64)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, abstol=1e-9, reltol=1e-9)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, max_iters=2500, verbose=False)
        except Exception:
            return None
    if b.value is None:
        return None
    return np.array(b.value, dtype=np.float64)


# ----------------------------------------------------------------------
# Simple projected gradient descent on the current maximal convolution entry
# ----------------------------------------------------------------------
def _gradient_optimize(start: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.4
    min_step = 1e-8
    decay = 0.9998
    prob_random = 0.02
    prob_smooth = 0.005

    while time.monotonic() < deadline:
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] = 2.0 * a[max_idx - j]

        cand = a - step * g
        cand = _project_simplex(cand)

        cand_max, cand_idx = _conv_max_and_index(cand, L)
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            max_idx = cand_idx
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            if rng.random() < math.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                max_idx = cand_idx

        # random mass transfer
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max2, idx2 = _conv_max_and_index(cand2, L)
                score2 = 2.0 * n * max2
                if score2 < cur_score:
                    a = cand2
                    cur_score = score2
                    max_idx = idx2
                    if score2 < best_score:
                        best = cand2.copy()
                        best_score = score2

        # occasional smoothing
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode="same")
            smooth = _project_simplex(smooth)
            max_s, idx_s = _conv_max_and_index(smooth, L)
            score_s = 2.0 * n * max_s
            if score_s < cur_score:
                a = smooth
                cur_score = score_s
                max_idx = idx_s
                if score_s < best_score:
                    best = smooth.copy()
                    best_score = score_s

        step = max(min_step, step * decay)

        if time.monotonic() + 0.5 > deadline:
            break
    return best, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # initialise ---------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: uniform distribution of appropriate length (2535)
        n_fallback = 2535
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # Phase 1 – long gradient descent ------------------------------------
    phase1_deadline = total_deadline - 30.0  # reserve time for LP & final polish
    best, best_score = _gradient_optimize(a0, phase1_deadline, rng)

    # Phase 2 – LP driven updates (if cvxpy available) -------------------
    if _HAS_CVX and time.monotonic() + 5.0 < total_deadline:
        a = best.copy()
        n = a.size
        L = 1 << ((2 * n - 1).bit_length())
        while time.monotonic() + 4.0 < total_deadline:
            max_val, _ = _conv_max_and_index(a, L)
            M = max_val
            b_opt = _solve_lp(a, M)
            if b_opt is None:
                break
            b_opt = np.maximum(b_opt, 0.0)
            s = b_opt.sum()
            if s == 0.0:
                break
            g = b_opt / s                     # rescale to sum == 1
            t = 0.5                           # convex‑combination weight
            cand = (1 - t) * a + t * g
            cand = _project_simplex(cand)
            cand_max, _ = _conv_max_and_index(cand, L)
            cand_score = 2.0 * n * cand_max
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                a = cand
            else:
                # if no improvement, shrink the step and try again
                t *= 0.6
                if t < 1e-3:
                    break

    # Phase 3 – final short gradient refinement -------------------------
    if time.monotonic() + 0.5 < total_deadline:
        final_deadline = total_deadline
        best, _ = _gradient_optimize(best, final_deadline, rng)

    # final safety clamp -------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)

    return [float(x) for x in best]
