import time

# ----------------------------------------------------------------------
# Core utilities (no closures, all top‑level)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _piecewise_exponential(n: int, groups: int, r: float) -> np.ndarray:
    """
    Construct a piecewise constant sequence with `groups` blocks.
    Height of block i is r**i, repeated to fill length n.
    """
    if groups <= 0:
        groups = 1
    base_len = n // groups
    rem = n - base_len * groups
    lengths = np.full(groups, base_len, dtype=int)
    if rem:
        lengths[:rem] += 1
    heights = r ** np.arange(groups, dtype=np.float64)
    a = np.repeat(heights, lengths)
    # Guard against rounding errors (might be off by one element)
    if a.size > n:
        a = a[:n]
    elif a.size < n:
        a = np.concatenate([a, np.full(n - a.size, 0.0, dtype=np.float64)])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / a.size)
    return a


def _simple_family(n: int, rng: np.random.Generator):
    """Generate a few well‑known families and keep the best."""
    best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    best_sc = _score(best_vec)

    # exponential
    for r in np.linspace(0.90, 0.9999, 30):
        a = np.power(r, np.arange(n, dtype=np.float64))
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        sc = _score(a)
        if sc < best_sc:

    for p in np.linspace(0.2, 4.0, 30):
        a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # linear descending
    a = np.arange(n, 0, -1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    sc = _score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # quadratic descending
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    sc = _score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # sqrt‑based recursion (same as previous code)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    sc = _score(a)
    if sc < best_sc:

            a = _piecewise_exponential(n, groups, r)
            sc = _score(a)
            if sc < best_sc:
                best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc
