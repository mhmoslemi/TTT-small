import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    if v.ndim != 1:
        raise ValueError("simplex projector expects a 1‑d array")
    n = v.size
    # shift by mean to guarantee sum=1 after clipping
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgrad_max_conv(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """ Sub‑gradient of max(conv) w.r.t a, averaged over the active indices. """
    n = a.size
    g = np.zeros_like(a)
    for k in idxs:
        # i runs over valid indices where 0 <= i < n and 0 <= k-i < n
        i_start = 0 if k < n else k - (n - 1)
        i_end = k if k < n else n - 1
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        # each pair (i,j) contributes a[j] to grad[i]; later we double all contributions
        g[i] += a[j]
    g *= 2.0
    if len(idxs) > 1:
        g /= len(idxs)
    return g


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Simple projected (momentum) sub‑gradient descent on the AC1 bound.
    Uses the exact active set of convolution peaks for the sub‑gradient.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. base sequence (fallback if parent not present)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # project to feasible simplex (sum=1, non‑negative)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    # ------------------------------------------------------------------
    # 3. pre‑compute FFT size (constant throughout)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 4. optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 1.0          # initial step size
    beta = 0.9          # momentum decay
    jitter_every = 200  # iterations between random jitter
    jitter_std = 0.02   # log‑normal jitter strength
    stagnation_limit = 2500   # iterations without improvement => restart
    target_score = 1.50274349

    v = np.zeros_like(a)   # momentum buffer
    best = a.copy()
    best_score = float(2.0 * n * np.max(np.convolve(best, best)))
    it = 0
    stagn = 0

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # convolution via FFT (linear convolution, first 2n‑1 entries)
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # --------------------------------------------------------------
        # update best solution
        # --------------------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stagn = 0
            if best_score <= target_score:
                break
        else:
            stagn += 1

        # --------------------------------------------------------------
        # sub‑gradient from the active maximal convolution entries
        # --------------------------------------------------------------
        active_idxs = np.flatnonzero(np.isclose(conv, max_val, atol=1e-15))
        g = _subgrad_max_conv(a, active_idxs)

        # --------------------------------------------------------------
        # step size (inverse‑sqrt schedule) and momentum
        # --------------------------------------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a_new = _proj_simplex(a - eta * v)

        # --------------------------------------------------------------
        # occasional jitter to escape plateaus
        # --------------------------------------------------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a_new = _proj_simplex(a_new * jitter)
            v[:] = 0.0
            stagn = 0

        # --------------------------------------------------------------
        # restart if stagnating for too long
        # --------------------------------------------------------------
        if stagn >= stagnation_limit:
            a_new = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            a_new = _proj_simplex(a_new * perturb)
            v[:] = 0.0
            stagn = 0

        a = a_new

    # ------------------------------------------------------------------
    # 5. return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
