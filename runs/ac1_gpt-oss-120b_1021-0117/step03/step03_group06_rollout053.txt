def project_simplex(v):
    # v is a 1d np array
    n = v.shape[0]
    # shift to ensure positive solution
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho: largest i where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all nonpositive -> return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # adjust small numeric errors
    if w.sum() == 0:
        return np.full_like(v, 1.0 / n)
    return w / w.sum()

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    return np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]

def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact score used by the evaluator."""
    return 2.0 * n * conv.max() / (sum_a * sum_a)

def _subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry (at index k)."""
    n = a.size

    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k - idx]
    return g

            cand, cand_sc = cand_add, cand_add_sc
        else:
            cand, cand_sc = cand_mul, cand_mul_sc

        # --------------------------------------------------------------
        # 6) accept if improvement, otherwise Metropolis move
        # --------------------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            temp = base_temp / (1.0 + 0.0004 * it)   # slowly cooling temperature
            if rng.random() < math.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
                cur, cur_sc = cand, cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # --------------------------------------------------------------
        # 7) keep global best
        # --------------------------------------------------------------
        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()

        # --------------------------------------------------------------
        # 8) directed mass‑transfer move (uses the sub‑gradient information)
        # --------------------------------------------------------------
        if rng.random() < mass_move_prob:
            i = int(np.argmax(g))                       # index that contributes most to the max
            # choose a low‑mass index that is far from i (to break symmetry)
            low_idxs = np.where(cur < cur.mean())[0]
            if low_idxs.size > 0:
                j = int(rng.choice(low_idxs))
                if i != j and cur[i] > 1e-12:
                    delta = cur[i] * rng.random() * 0.28   # move at most ~30 % of source mass
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = np.maximum(tmp, 0.0)
                    tmp = tmp / tmp.sum()
                    tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best = tmp_sc, tmp.copy()

        # --------------------------------------------------------------
        # 9) periodic jittered restart from the best point
        # --------------------------------------------------------------
        if it % restart_interval == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            s_tmp = tmp.sum()
            if s_tmp > 0.0:
                tmp = tmp / s_tmp
                tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()
                no_improve = 0

        # --------------------------------------------------------------
        # 10) restart after long stagnation
        # --------------------------------------------------------------
        if no_improve > stagnation_limit:
            jitter = rng.normal(scale=jitter_scale * 5, size=n)
            cur = best + jitter
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_sc = _score_from_conv(_conv_fft(cur, L), n, cur.sum())
            no_improve = 0

        # --------------------------------------------------------------
        # 11) early exit if target reached
        # --------------------------------------------------------------
        if best_sc < 1.50274349:
            break

    # Return a plain Python list as required by the harness
    return [float(x) for x in best]
