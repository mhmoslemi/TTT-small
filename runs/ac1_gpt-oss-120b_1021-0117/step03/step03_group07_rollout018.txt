def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    c = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return c

def _score(a):
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

# Basic convolution / scoring utilities (identical to the reference code)

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (real input)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum is not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Simple analytic families – useful as cheap initial candidates
# ----------------------------------------------------------------------
def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Build the (dense‑but‑sparse‑stored) convolution matrix for a fixed f
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Returns the (2n‑1)×n matrix A such that (A @ g) = f * g (full convolution).
    The matrix is stored in CSR format and contains n² non‑zero entries.
    """
    n = f.size
    # each diagonal offset = -d (d = 0..n‑1) holds the constant value f[d]
    offsets = -np.arange(n, dtype=int)               # -0, -1, …, -(n‑1)
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Core optimisation loop – LP + mixing
# ----------------------------------------------------------------------
def _lp_opt_step(cur: np.ndarray, solver_time_limit: float) -> np.ndarray | None:
    """
    One LP step: minimise T subject to (cur * g)[k] ≤ T,  Σg = 1, g ≥ 0.
    Returns the optimal g (already normalised) or None on failure.
    """
    n = cur.size
    A = _conv_matrix(cur)                     # (2n‑1)×n sparse matrix
    g = cp.Variable(n)
    T = cp.Variable()
    constraints = [
        A @ g <= T,
        cp.sum(g) == 1,
        g >= 0,
    ]
    prob = cp.Problem(cp.Minimize(T), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=2000, feastol=1e-9, abstol=1e-9,
                   reltol=1e-9, verbose=False)
    except Exception:
        return None

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None
    g_val = g.value
    if g_val is None or np.isnan(g_val).any():
        return None
    # Numerical round‑off – enforce the normalisation explicitly
    s = g_val.sum()
    if s <= 0:
        return None
    return (g_val / s).astype(np.float64)

    best_sc = _score(best_vec)

    # --------------------------------------------------------------
    # Cheap analytic sweep – look for a better starting point
    # --------------------------------------------------------------
    for r in np.linspace(0.90, 0.9999, 30):
        cand = _make_exponential(n, r)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 4.0, 30):
        cand = _make_powerlaw(n, p)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for maker in (_make_linear, _make_quadratic, _make_sqrt_series):
        cand = maker(n)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    target = 1.50274349
    # --------------------------------------------------------------
    # Main loop: alternating LP step + simple mixing / jitter
    # --------------------------------------------------------------
    iter_count = 0
    max_iter = 25                     # generous upper bound; loop also checks time
    while time.monotonic() < deadline and iter_count < max_iter:
        iter_count += 1
        # ----- LP step -------------------------------------------------
        # allocate a per‑iteration time budget (≈ 1 s) to stay safe
        lp_deadline = time.monotonic() + 1.0
        lp_g = _lp_opt_step(cur, solver_time_limit=lp_deadline - time.monotonic())
        if lp_g is None:
            # LP failed – fall back to a simple random perturbation and continue
            jitter = rng.normal(scale=1e-4, size=n)
            cand = cur + jitter
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            cand_sc = _score(cand)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            continue

        # ----- Evaluate LP candidate alone -------------------------------
        g_sc = _score(lp_g)
        # Accept if it already improves the score
        if g_sc < cur_sc:
            cur, cur_sc = lp_g, g_sc
            if g_sc < best_sc:
                best_sc, best_vec = g_sc, lp_g.copy()
            if best_sc < target:
                break
            continue

        # ----- Otherwise try convex combinations -------------------------
        # try a few mixing weights; pick the best improvement
        best_mix_sc = cur_sc
        best_mix = None
        for t in (0.6, 0.4, 0.2, 0.1, 0.05):
            cand = (1.0 - t) * cur + t * lp_g
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            sc = _score(cand)
            if sc < best_mix_sc:
                best_mix_sc, best_mix = sc, cand.copy()
        if best_mix is not None and best_mix_sc < cur_sc:
            cur, cur_sc = best_mix, best_mix_sc
            if best_mix_sc < best_sc:
                best_sc, best_vec = best_mix_sc, best_mix.copy()
            if best_sc < target:
                break
            continue

        # ----- Small random jitter as a last resort ----------------------
        if rng.random() < 0.15:
            jitter = rng.normal(scale=5e-5, size=n)
            cand = cur + jitter
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            sc = _score(cand)
            if sc < cur_sc:
                cur, cur_sc = cand, sc
                if sc < best_sc:
                    best_sc, best_vec = sc, cand.copy()
                if best_sc < target:
                    break

        # time check – exit early if close to the hard deadline
        if time.monotonic() >= deadline:
            break

    # --------------------------------------------------------------
    # Safety fallback – never return an empty or invalid sequence
    # --------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # convert to plain Python list as required by the harness
    return [float(v) for v in best_vec]
