```
def _project_simplex(v):
    # Implementation O(n log n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
```

            cand_s = cur - alpha * g_s
            cand_s = _project_simplex(cand_s)
            conv_s = _conv(cand_s)
            score_s = _objective_from_conv(conv_s)
            candidates.append((cand_s, conv_s, score_s))

        # --------------------------------------------------------------
        # pick the best candidate
        # --------------------------------------------------------------
        cand_best, conv_best, best_sc = min(candidates, key=lambda t: t[2])

        # --------------------------------------------------------------
        # accept (with occasional uphill moves)
        # --------------------------------------------------------------
        if best_sc < cur_score:
            cur, conv_cur, cur_score = cand_best, conv_best, best_sc
        else:
            diff = best_sc - cur_score
            prob = np.exp(-diff / (alpha + 1e-12))
            if rng.random() < prob:
                cur, conv_cur, cur_score = cand_best, conv_best, best_sc

        # --------------------------------------------------------------
        # update global best
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_vec, best_score = cur.copy(), cur_score
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # random mass‑transfer move when stuck
        # --------------------------------------------------------------
        if no_improve > 3000 and rng.random() < 0.3:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                conv_cand = _conv(cand)
                score_cand = _objective_from_conv(conv_cand)
                if score_cand < cur_score:
                    cur, conv_cur, cur_score = cand, conv_cand, score_cand
                    if score_cand < best_score:
                        best_vec, best_score = cand.copy(), score_cand
                        no_improve = 0

        # --------------------------------------------------------------
        # jittered restart toward current best
        # --------------------------------------------------------------
        if it % 5000 == 0 and rng.random() < 0.25:
            jitter = rng.normal(scale=1e-4, size=n)
            cand = best_vec + jitter
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            conv_cand = _conv(cand)
            score_cand = _objective_from_conv(conv_cand)
            if score_cand < cur_score:
                cur, conv_cur, cur_score = cand, conv_cand, score_cand
                if score_cand < best_score:
                    best_vec, best_score = cand.copy(), score_cand
                    no_improve = 0

        # --------------------------------------------------------------
        # early exit if target reached
        # --------------------------------------------------------------
        if best_score < target:
            break

        # --------------------------------------------------------------
        # time guard
        # --------------------------------------------------------------
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # final short refinement (random transfers)
    # ------------------------------------------------------------------
    refine_deadline = time.monotonic() + min(0.4, deadline - time.monotonic())
    while time.monotonic() < refine_deadline:
        i = rng.integers(0, n)
        if cur[i] == 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j == i:
            continue
        delta = cur[i] * rng.random() * 0.3
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        conv_cand = _conv(cand)
        score_cand = _objective_from_conv(conv_cand)
        if score_cand < cur_score:
            cur, conv_cur, cur_score = cand, conv_cand, score_cand
            if score_cand < best_score:
                best_vec, best_score = cand.copy(), score_cand

    return [float(x) for x in best_vec]
