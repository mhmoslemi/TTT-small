import time
import math
import numpy as np

# -------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# -------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:  # safeguard
        w[:] = 1.0 / n
    return w

# -------------------------------------------------------
# Full self‑convolution via FFT (real)
# -------------------------------------------------------
def _conv_full(vec: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    return conv

# -------------------------------------------------------
# Gradient from all maximal convolution indices
# -------------------------------------------------------
def _gradient_from_peaks(a: np.ndarray, peaks: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in peaks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] += 2.0 * a[k - idx]
    return g

# Simple sqrt‑series used when no parent is present

def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
