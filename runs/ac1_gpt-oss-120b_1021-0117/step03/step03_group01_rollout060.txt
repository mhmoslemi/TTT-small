import time, math, numpy as np

# -------------------------------------------------------------
# Simple projection onto the probability simplex (sum = 1)
# -------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a non‑negative vector onto the simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:                     # safety fallback
        w[:] = 1.0 / len(v)
    return w

# -------------------------------------------------------------
# Fast convolution (full) using FFT – returns full array
# -------------------------------------------------------------
def _full_conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * len(a) - 1]
    return conv

# -------------------------------------------------------------
# Core optimiser – projected sub‑gradient descent with a few
# stochastic moves. Works on the simplex (∑a_i = 1, a_i ≥ 0).
# -------------------------------------------------------------
def _optimise(start: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑two FFT length

    conv = _full_conv_fft(a, L)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    step = 0.5
    min_step = 1e-8
    decay = 0.9997
    beta = 0.9
    mom = np.zeros_like(a)

    while time.monotonic() < deadline - 0.5:
        # ---- active‑set gradient (top‑k peaks) ----
        k = 3                                          # number of peaks to use
        if conv.size <= k:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -k)[-k:]
        g = np.zeros_like(a)
        for idx in idxs:
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            if lo <= hi:
                j = np.arange(lo, hi + 1)
                g[j] += 2.0 * a[idx - j]
        if len(idxs) > 1:
            g /= len(idxs)

        # ---- momentum + candidate ----
        mom = beta * mom + (1.0 - beta) * g
        cand = a - step * mom
        cand = _proj_simplex(cand)

        conv_c = _full_conv_fft(cand, L)
        cand_score = 2.0 * n * conv_c.max()

        if cand_score < cur_score:
            a, conv, cur_score = cand, conv_c, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # occasional uphill acceptance (simulated annealing)
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a, conv, cur_score = cand, conv_c, cand_score

        # ---- random mass transfer (exploration) ----
        if rng.random() < 0.04:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a[i] * frac
                t = a.copy()
                t[i] -= delta
                t[j] += delta
                t = _proj_simplex(t)
                conv_t = _full_conv_fft(t, L)
                sc = 2.0 * n * conv_t.max()
                if sc < cur_score:
                    a, conv, cur_score = t, conv_t, sc
                    if sc < best_score:
                        best, best_score = t.copy(), sc

        # ---- occasional smoothing ----
        if rng.random() < 0.008:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _proj_simplex(smooth)
            conv_s = _full_conv_fft(smooth, L)
            sc = 2.0 * n * conv_s.max()
            if sc < cur_score:
                a, conv, cur_score = smooth, conv_s, sc
                if sc < best_score:
                    best, best_score = smooth.copy(), sc

        # ---- step‑size decay ----
        step = max(min_step, step * decay)

    return best, best_score

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- initialise ----------------------------------------------------

    return [float(x) for x in best]
