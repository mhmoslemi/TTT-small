import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # 3. Best candidate bookkeeping
    # ------------------------------------------------------------------
    best = a.copy()
    # initial score (same as evaluator when sum=1)
    best_score = float(2.0 * n * np.max(np.convolve(best, best)))
    target_score = 1.50274349

    # ------------------------------------------------------------------
    # 4. FFT preparation
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 5. Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    eta0 = 0.6          # base step size
    beta1 = 0.9         # Adam first‑moment decay
    beta2 = 0.999       # Adam second‑moment decay
    eps = 1e-8
    jitter_every = 180   # steps between random multiplicative jitter
    jitter_std = 0.015
    max_stale = 2500    # steps without improvement before restart
    it = 0
    stale = 0
    m = np.zeros_like(a)   # Adam first moment
    v = np.zeros_like(a)   # Adam second moment

    # ------------------------------------------------------------------
    # 6. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution via FFT (full linear convolution) -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- keep the best solution so far -----
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
            if best_score <= target_score:
                break
        else:
            stale += 1

        # ----- active set for gradient (all indices within a fraction of max) -----
        thr = max_val * 0.85
        active = np.where(conv >= thr)[0]
        if active.size == 0:          # fallback to the true maximum
            active = np.array([int(conv.argmax())])

        # ----- sub‑gradient for the active set -----
        g = np.zeros_like(a)
        for k in active:
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            idx = np.arange(start, end + 1, dtype=np.int64)
            partner = k - idx
            g[idx] += a[partner]
        # each pair appears twice (i,j) and (j,i); scale accordingly
        g = 2.0 * g / max(1, active.size)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (eta0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----- projection onto the simplex (sum=1, non‑negative) -----
        a = _proj_simplex(a)

        # ----- occasional jitter to disturb stagnation -----
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            m.fill(0.0)
            v.fill(0.0)
            stale = 0

        # ----- long stagnation → restart around the best found -----
        if stale >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            m.fill(0.0)
            v.fill(0.0)
            stale = 0

    # ------------------------------------------------------------------
    # 7. Return the best sequence (plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
