def _optimize(start_vec, deadline, rng):
    a = start_vec.copy()
    n = a.shape[0]
    best = a.copy()
    # compute current score
    conv_self = _full_conv_fft(a)
    max_val = conv_self.max()
    cur_score = 2.0*n*max_val
    best_score = cur_score
    # momentum
    momentum = np.zeros_like(a)
    step = 0.5
    beta = 0.9
    min_step = 1e-7
    decay = 0.9995 # adjust
    prob_random = 0.03
    prob_smooth = 0.01
    # may use an early exit if time near.
    
    while time.monotonic() < deadline:
        # compute gradient
        max_val, grad = _max_and_grad(a)
        # gradient descent
        momentum = beta * momentum + (1 - beta) * grad
        cand = a - step * momentum
        cand = _project_simplex(cand)
        # evaluate cand
        conv_cand = _full_conv_fft(cand)
        max_cand = conv_cand.max()
        cand_score = 2.0 * n * max_cand
        # accept if improvement
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # occasional uphill w/ temperature
            T = step  # or some schedule
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand
                cur_score = cand_score
        # random mass transfer 
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                # move portion
                frac = rng.random() * 0.5
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                conv2 = _full_conv_fft(cand2)
                max2 = conv2.max()
                score2 = 2.0*n*max2
                if score2 < cur_score:
                    a = cand2
                    cur_score = score2
                    if score2 < best_score:
                        best_score = score2
                        best = cand2.copy()
        # smoothing
        if rng.random() < prob_smooth:
            # 3-point smoothing kernel
            smooth = np.convolve(a, np.array([0.25, 0.5, 0.25]), mode='same')
            smooth = _project_simplex(smooth)
            conv_s = _full_conv_fft(smooth)
            max_s = conv_s.max()
            score_s = 2.0 * n * max_s
            if score_s < cur_score:
                a = smooth
                cur_score = score_s
                if score_s < best_score:
                    best_score = score_s
                    best = smooth.copy()
        # step decay
        step = max(min_step, step * decay)
        # cutoff early
        if time.monotonic() + 0.5 > deadline:
            break
    return best, best_score

        # step decay
        step = max(min_step, step * decay)

        # early termination guard

import time
import math
import numpy as np

def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

def _full_conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv

def _softmax_gradient(a, lam):
    # convolution of a with itself
    conv = _full_conv_fft(a)
    max_val = conv.max()
    # softmax weights on the convolution entries
    w = np.exp(lam * (conv - max_val))
    ws = w.sum()
    if ws == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= ws
    # gradient = 2 * (w * a_rev)_{center}
    a_rev = a[::-1]
    L2 = 1 << ((w.size + a_rev.size - 1).bit_length())
    fw = np.fft.rfft(w, n=L2)
    fres = np.fft.rfft(a_rev, n=L2)
    conv2 = np.fft.irfft(fw * fres, n=L2)
    start = a.shape[0] - 1
    grad = 2.0 * conv2[start:start + a.shape[0]]
    return max_val, grad

def _optimize(start_vec, deadline, rng):
    a = _project_simplex(start_vec.copy())
    n = a.shape[0]
    best = a.copy()
    # initial evaluation
    cur_max, _ = _softmax_gradient(a, lam=1.0)  # lam dummy, we only need max
    cur_score = 2.0 * n * cur_max
    best_score = cur_score

    momentum = np.zeros_like(a)
    step = 0.5
    decay = 0.9995
    min_step = 1e-7
    beta = 0.9
    prob_random = 0.03
    prob_smooth = 0.01
    iteration = 0

    while True:
        now = time.monotonic()
        if now >= deadline or now + 0.1 > deadline:
            break

        lam = 1.0 + iteration * 0.02
        if lam > 30.0:
            lam = 30.0

        max_val, grad = _softmax_gradient(a, lam)

        # momentum descent
        momentum = beta * momentum + (1.0 - beta) * grad
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # evaluate candidate
        conv_c = _full_conv_fft(cand)
        max_c = conv_c.max()
        cand_score = 2.0 * n * max_c

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand
                cur_score = cand_score

        # random mass transfer
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1

                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta

                if score_s < best_score:
                    best_score = score_s
                    best = smooth.copy()

    return best, best_score

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w


# ----------------------------------------------------------------------
# Full convolution a*a using an FFT of a power‑of‑two length
# ----------------------------------------------------------------------
def _full_conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (temperature‑controlled) gradient of the max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, lam: float):
    """
    Returns (max_conv_value, gradient) where the gradient is the
    derivative of a smooth approximation to max(conv(a,a)).
    """
    conv = _full_conv_fft(a)
    max_val = conv.max()

    # weights ∝ exp(lam * (conv - max_val)), sum to 1
    w = np.exp(lam * (conv - max_val))
    ws = w.sum()
    if ws == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= ws

    a_rev = a[::-1]
    L2 = 1 << ((w.size + a_rev.size - 1).bit_length())
    fw = np.fft.rfft(w, n=L2)
    fr = np.fft.rfft(a_rev, n=L2)
    conv2 = np.fft.irfft(fw * fr, n=L2)

    start = a.shape[0] - 1
    grad = 2.0 * conv2[start:start + a.shape[0]]
    return max_val, grad


# ----------------------------------------------------------------------
# Core optimiser: momentum descent on the soft‑max gradient
# ----------------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = _project_simplex(start_vec.copy())
    n = a.shape[0]

    # initial score
    cur_max, _ = _softmax_gradient(a, lam=1.0)
    cur_score = 2.0 * n * cur_max
    best = a.copy()
    best_score = cur_score

    momentum = np.zeros_like(a)

    step = 0.5
    decay = 0.9995
    min_step = 1e-7
    beta = 0.9
    prob_random = 0.03
    prob_smooth = 0.01
    iteration = 0

    while True:
        now = time.monotonic()
        if now >= deadline or now + 0.1 > deadline:
            break

        # temperature schedule for the soft‑max
        lam = 1.0 + iteration * 0.02
        if lam > 30.0:
            lam = 30.0

        max_val, grad = _softmax_gradient(a, lam)

        # momentum‑driven step
        momentum = beta * momentum + (1.0 - beta) * grad
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # evaluate candidate
        conv_c = _full_conv_fft(cand)
        max_c = conv_c.max()
        cand_score = 2.0 * n * max_c

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # occasional uphill acceptance (simulated annealing)
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand
                cur_score = cand_score

        # random mass transfer between two positions
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)

                frac = rng.random() * 0.5
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                conv2 = _full_conv_fft(cand2)
                max2 = conv2.max()
                score2 = 2.0 * n * max2
                if score2 < cur_score:
                    a = cand2
                    cur_score = score2
                    if score2 < best_score:
                        best_score = score2
                        best = cand2.copy()
