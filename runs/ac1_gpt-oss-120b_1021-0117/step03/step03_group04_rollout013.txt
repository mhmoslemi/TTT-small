def _fast_objective(x: np.ndarray) -> float:
    """Compute 2·n·max(convolution(x,x)) efficiently via FFT."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return 2.0 * n * conv.max()

            best_score = new_score

        # accept only if the move strictly improves the current sequence
        if new_score < cur_score:
            a, cur_score = a_new, new_score

    # Return plain Python floats as required by the harness.
    return [float(v) for v in best]
