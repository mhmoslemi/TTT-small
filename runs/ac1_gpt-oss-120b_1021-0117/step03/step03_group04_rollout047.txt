import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    # Implementation of the standard O(n) simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all zeros – return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Simple optimizer for the AC1 bound.
    Starts from a few elementary candidates, optionally refines the best
    with a few projected‑gradient steps, and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time budget
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 2. Length of the problem
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
        n = int(parent_arr.size)
    except Exception:
        # fallback length if the parent is unavailable
        n = 256

    # ------------------------------------------------------------------
    # 3. Base candidates (all normalised to sum = 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    tent = np.maximum(tent, 0.0)
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:
        tent = uniform.copy()

    rand = rng.random(n)
    rand = np.maximum(rand, 0.0)
    rand = rand / rand.sum()

    candidates = [uniform, tent, rand]

    # include the parent if it looks usable (just for length, not shape)
    try:
        if parent_arr.ndim == 1 and parent_arr.size == n:
            p = np.clip(parent_arr, 0.0, 1000.0)
            s = p.sum()
            if s >= 0.01:
                p = p / s
                candidates.append(p)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fast scoring (FFT based, uses the fact that sum = 1)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 5. Pick the best initial candidate
    # ------------------------------------------------------------------
    best = None
    best_val = float("inf")
    for c in candidates:
        val = fast_score(c)
        if val < best_val:
            best_val = val
            best = c.copy()

    # ------------------------------------------------------------------
    # 6. Very small projected‑gradient refinement (optional)
    # ------------------------------------------------------------------
    # a few iterations are cheap; we stop well before the deadline.
    grad_steps = 200
    eta0 = 0.5
    mu = 0.8
    v = np.zeros_like(best)

    it = 0
    while time.monotonic() < deadline and it < grad_steps:
        it += 1

        # compute full convolution via FFT
        fa = np.fft.rfft(best, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best = best.copy()

        # gradient of max‑convolution (simple approximation using all shifts)
        # For each shift we accumulate a_i * a_{i+shift}
        # The sub‑gradient w.r.t. a_i is 2 * sum_{j} a_j where j aligns with i.
        # We compute it efficiently using the convolution itself.
        grad = 2.0 * np.fft.irfft(fa * np.conj(fa), n=fft_len)[:n]
        # momentum update
        v = mu * v + (1.0 - mu) * grad
        # diminishing step size
        eta = eta0 / np.sqrt(it)
        # projected gradient step (onto simplex to keep sum = 1)
        best = _proj_simplex(best - eta * v)

    # ------------------------------------------------------------------
    # 7. Return result as plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]
