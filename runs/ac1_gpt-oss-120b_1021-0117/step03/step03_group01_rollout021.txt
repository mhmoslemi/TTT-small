import time
import math
import numpy as np

# --------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, >=0)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:               # safety net – should never happen
        w[:] = 1.0 / n
    return w

# --------------------------------------------------------------
def _conv_full(vec: np.ndarray, L: int) -> np.ndarray:
    """Return the linear convolution a*a (length 2*len(a)-1)."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * len(vec) - 1]

        # fallback: simple sqrt‑series of modest length then padded
        def _sqrt_series(n):
            a = np.empty(n, dtype=np.float64)
            a[0] = 1.0
            for k in range(1, n):
                if k == 1:
                    s = 0.0
                else:
                    s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
                a[k] = (1.0 - s) / 2.0
            return a
        a0 = _sqrt_series(256)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # ---- primary optimisation ---------------------------------------
    best, best_score = _optimize_vec(a0, deadline, rng)

    # ---- optional secondary start if time permits ------------------
    if time.monotonic() + 5.0 < deadline:
        rand_vec = rng.random(a0.shape)
        rand_vec = _project_simplex(rand_vec)
        alt_best, alt_score = _optimize_vec(rand_vec, deadline, rng)
        if alt_score < best_score:
            best = alt_best
            best_score = alt_score

    # ---------------------------------------------------------------
    return [float(x) for x in best]
