import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Uses exponentiated gradient descent on the simplex with
    a sub‑gradient taken from the top convolution peaks.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time management
    # ------------------------------------------------------------------
    # Leave a few seconds margin for clean exit.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2.  Initialise sequence (fallback if the parent is unavailable)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = int(parent.size)
        a = parent.copy()
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s <= 1e-12:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
    except Exception:
        # generic fallback: uniform distribution of size 256
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Avoid exact zeros (they would stay zero forever under EG)
    eps = 1e-14
    a = np.maximum(a, eps)
    a = a / a.sum()

    # ------------------------------------------------------------------
    # 3.  Convolution setup
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 4.  Hyper‑parameters
    # ------------------------------------------------------------------
    top_k = max(5, int(np.sqrt(n)))            # number of active peaks
    eta0 = 0.5                                 # initial step size for EG
    jitter_every = 200                         # how often to jitter
    jitter_std = 0.015                         # std‑dev of log‑normal jitter
    max_stale = 5000                           # restart after this many non‑improvements
    target_score = 1.50274349                  # stop if we reach the target

    # ------------------------------------------------------------------
    # 5.  Helper for exponential‑gradient update on the simplex
    # ------------------------------------------------------------------
    def eg_update(x, grad, eta):
        """Exponentiated‑gradient step followed by renormalisation."""
        # ensure positivity
        factor = np.exp(-eta * grad)
        x = x * factor
        x = np.maximum(x, eps)
        total = x.sum()
        if total == 0.0:
            x[:] = 1.0 / x.size
        else:
            x /= total
        return x

    # ------------------------------------------------------------------
    # 6.  Track the best sequence seen so far
    # ------------------------------------------------------------------
    # initial score
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best_score = float(2.0 * n * conv.max())
    best = a.copy()
    no_improve = 0
    it = 0

    # ------------------------------------------------------------------
    # 7.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution (FFT) -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- best‑so‑far tracking -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ----- active‑set indices (top‑k peaks) -----
        if conv.size > top_k:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size)

        # ----- sub‑gradient of the max‑convolution -----
        g = np.zeros_like(a)
        a_rev = a[::-1]               # reversed view for negative offsets
        for k in idxs:
            offset = int(k) - (n - 1)  # centre index is n-1
            if offset >= 0:
                length = n - offset
                if length > 0:
                    g[offset:] += a[:length]
            else:
                d = -offset
                length = n - d
                if length > 0:
                    g[:length] += a_rev[:length]
        # average over the active set and the factor 2 from the derivative
        g = (2.0 / max(1, len(idxs))) * g

        # ----- EG step -----
        it += 1
        eta = eta0 / np.sqrt(it)
        a = eg_update(a, g, eta)

        # ----- occasional jitter to escape plateaus -----
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = eg_update(a, np.zeros_like(a), 0.0)   # just to renormalise after jitter
            a *= jitter
            a = eg_update(a, np.zeros_like(a), 0.0)   # re‑normalise
            no_improve = 0

        # ----- long stagnation → restart around best -----
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a *= perturb
            a = eg_update(a, np.zeros_like(a), 0.0)   # renormalise
            no_improve = 0

    # ------------------------------------------------------------------
    # 8.  Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
