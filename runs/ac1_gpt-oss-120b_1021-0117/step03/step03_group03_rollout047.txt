def gradient_from_max(a, k):
    n = len(a)
    g = np.zeros_like(a)
    start = max(0, k - (n-1))
    end = min(n-1, k)   # inclusive
    # For j in [start, end], add 2*a[k-j]
    idx = np.arange(start, end+1)
    g[idx] = 2 * a[k - idx]
    return g

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    # Reference: https://arxiv.org/abs/1101.6081
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
