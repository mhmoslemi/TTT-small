```python
import time
import numpy as np

def _proj_simplex(v):
    """ Euclidean projection onto simplex {x >=0, sum=1} """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find rho = max {j | u_j - cssv_j / j > 0}
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # result is uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_convolve_self(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full convolution of a with itself via FFT (real inputs)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Returns a list of non‑negative coefficients that (empirically) lower the
    bound computed by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time budget
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Base sequence (fallback if no parent is supplied)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = parent.size
        a0 = parent.copy()
    except Exception:
        n = 256
        a0 = np.ones(n, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a0 / s0                     # normalise to sum = 1
    a = _proj_simplex(a)

    # ------------------------------------------------------------------
    # Baseline candidates (uniform and parent) – keep the best so far
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def _score(arr):
        conv = _fft_convolve_self(arr, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    best = a.copy()
    best_score = _score(best)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    decay = 0.9                     # RMSProp decay
    eps = 1e-8
    sq_mean = np.zeros_like(a)      # second‑moment accumulator
    eta0 = 0.3                      # base step size
    it = 0
    no_improve = 0
    max_stale = 2500
    jitter_every = 150
    jitter_std = 0.025
    target_score = 1.50274349
    tol = 1e-12

    # Small stochastic kick to avoid the uniform fixed point
    a = _proj_simplex(a * np.exp(rng.normal(0.0, 0.05, size=n)))

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        conv = _fft_convolve_self(a, fft_len, conv_len)
        B = conv.max()
        cur_score = 2.0 * n * B

        # track best
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # exact sub‑gradient of the maximal convolution entry
        mask = conv >= B - tol
        idxs = np.where(mask)[0]
        g = np.zeros_like(a)

        # accumulate contributions from each maximising shift
        for k in idxs:
            # indices i where a_i participates in (a*a)[k]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo > hi:
                continue
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            partner = k - i_range
            g[i_range] += a[partner]

        # gradient of the objective (2*n factor included)
        g = 2.0 * n * g

        # RMSProp update
        sq_mean = decay * sq_mean + (1.0 - decay) * (g * g)
        it += 1
        eta = eta0 / np.sqrt(it)
        step = eta * g / (np.sqrt(sq_mean) + eps)
        a = a - step
        a = _proj_simplex(a)

        # occasional jitter to escape plateaus
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            sq_mean.fill(0.0)
            no_improve = 0

        # restart around the best if stuck too long
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            sq_mean.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
