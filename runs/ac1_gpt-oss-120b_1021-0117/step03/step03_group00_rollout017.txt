# ------------------------------------------------------------
# Utility: project onto probability simplex (optional fallback)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Exact score (same formula as evaluate_sequence)
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray = None) -> float:
    n = a.shape[0]
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initial vector (use parent if available)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random positive vector, length 2500 (any reasonable size)
        a = rng.random(2500, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a.fill(1.0 / a.size)
    else:
        a /= s

    # enforce symmetry – the objective is invariant under reversal
    a = (a + a[::-1]) * 0.5
    a /= a.sum()

    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())          # FFT length for linear conv up to 3n‑2
    idx = (2 * n - 2) - np.arange(n)               # slice to extract gradient

    # --------------------------------------------------------
    # optimisation state
    # --------------------------------------------------------
    cur = a.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    eps = 1.0                     # smoothing parameter for max
    eps_decay = 0.9995
    eta0 = 0.8                    # base step magnitude for multiplicative update
    stagnation = 0
    it = 0

    # --------------------------------------------------------
    # main loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smooth weights for the max -----
        eps = max(1e-7, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)        # all <= 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # linear conv length >= 3n‑2
        g = 2.0 * grad_full[idx]                       # gradient on simplex

        # ----- multiplicative (exponentiated) update -----
        eta = eta0 / math.sqrt(it)                     # decreasing step
        exp_arg = -eta * g
        exp_arg = np.clip(exp_arg, -700, 700)          # avoid overflow
        factor = np.exp(exp_arg)
        cand = cur * factor
        cand_sum = cand.sum()
        if cand_sum == 0.0 or np.isnan(cand_sum):
            # fallback to additive step with projection
            cand = cur - eta * g
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
        else:
            cand = cand / cand_sum

        # enforce symmetry after the update
        cand = (cand + cand[::-1]) * 0.5
        cand = cand / cand.sum()

        # ----- evaluate candidate -----
        cand_sc = _score(cand)

        # ----- accept or reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                stagnation = 0
            else:
                stagnation += 1
        else:
            # occasional Metropolis‑type acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            stagnation += 1

        # ----- periodic random redistribution if stuck -----
        if stagnation > 5000:
            # move a random fraction of mass symmetrically
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                cur[i] -= delta
                cur[j] -= delta
                # distribute to two random symmetric positions
                k = rng.integers(0, n // 2)
                l = n - 1 - k
                cur[k] += delta / 2.0
                cur[l] += delta / 2.0
            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()
            cur_sc = _score(cur)
            stagnation = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return best vector found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best]
