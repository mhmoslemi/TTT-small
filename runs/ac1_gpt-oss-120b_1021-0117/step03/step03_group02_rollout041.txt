    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,float(budget_s)-1.0)
    if 'height_sequence_1' in globals():

    else:

    n = init.size if init is not None else 2000
    L = 1 << ((2*n - 1).bit_length())

    # define projection
    def project(v):
        # expects nonnegative after projection
        # standard Euclidean projection onto simplex sum=1
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho_arr = u - css / np.arange(1, v.size + 1)
        rho = np.where(rho_arr > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        return np.maximum(v - theta, 0.0)

    def metric(a):
        max_val, _ = _conv_full_max(a, L)
        return 2.0 * n * max_val

import time

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if v.size == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def _conv_full_max(a: np.ndarray, L: int):
    """Return (max(conv), full convolution) for a*a."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv
