def _optimize(start_vec, deadline, rng):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    # compute conv and max
    conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:2*n-1]
    max_val = conv.max()
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score
    # momentum
    momentum = np.zeros_like(a)
    step = 1.0
    decay = 0.9995
    beta = 0.9
    K = 5
    # tolerance for top
    while time.monotonic() < deadline:
        # compute conv and get top K
        fv = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        # find top K indices
        if K < len(conv_full):
            # argpartition for top K
            idxs = np.argpartition(-conv_full, K-1)[:K]
        else:
            idxs = np.arange(len(conv_full))
        # compute gradient as average
        g = np.zeros_like(a)
        for idx in idxs:
            lo = max(0, idx - (n-1))
            hi = min(n-1, idx)
            # compute slice length hi - lo + 1
            sl_len = hi - lo + 1
            # a segment to be used
            segment = a[idx - np.arange(lo, hi+1)]  # shape length
            g[lo:hi+1] += 2.0 * segment
        g /= len(idxs)
        # update momentum
        momentum = beta * momentum + (1 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)
        # evaluate cand
        fv_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(fv_c * fv_c, n=L)[:2*n-1]
        max_c = conv_c.max()
        cand_score = 2.0 * n * max_c
        # acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
            a = cand
            cur_score = cand_score
            # track best
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        # sometimes random mass transfer
        if rng.random() < 0.01:
            # choose i with mass to move
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.5 # move fraction up to 0.5
                new = a.copy()
                new[i] -= delta
                new[j] += delta
                new = _project_simplex(new)
                fv_new = np.fft.rfft(new, n=L)
                conv_new = np.fft.irfft(fv_new * fv_new, n=L)[:2*n-1]
                new_score = 2.0 * n * conv_new.max()
                if new_score < cur_score:
                    a = new
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best = new.copy()
        # occasionally smoothing
        if rng.random() < 0.005:
            kernel = np.array([0.25, 0.5, 0.25])
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            fv_s = np.fft.rfft(smooth, n=L)
            conv_s = np.fft.irfft(fv_s * fv_s, n=L)[:2*n-1]
            s_score = 2.0 * n * conv_s.max()
            if s_score < cur_score:
                a = smooth

                    best_score = s_score
                    best = smooth.copy()
        # decay step

        # early exit guard
        if time.monotonic() + 0.5 > deadline:

import time
import math
import numpy as np

# --------------------------------------------------------------
# Euclidean projection onto the probability simplex (L1 = 1)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w


# --------------------------------------------------------------
# Simple analytic start (square‑root series) – rarely used
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# --------------------------------------------------------------
# Core optimizer – multi‑peak sub‑gradient with momentum
# --------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    # FFT length (next power of two >= 2n-1)
    L = 1 << ((2 * n - 1).bit_length())

    # initial score
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    momentum = np.zeros_like(a)
    step = 1.0
    decay = 0.9995
    beta = 0.9
    top_k = 5                # number of active autocorrelation peaks
    prob_random = 0.01
    prob_smooth = 0.005

    while time.monotonic() < deadline:
        # ------- current convolution and top‑k peaks -------
        fv = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

        # indices of the largest K entries
        if top_k < conv_full.size:
            idxs = np.argpartition(-conv_full, top_k - 1)[:top_k]
        else:
            idxs = np.arange(conv_full.size)

        # ------- sub‑gradient (average of active constraints) -------
        g = np.zeros_like(a)
        for idx in idxs:
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            jrng = np.arange(lo, hi + 1)
            g[lo : hi + 1] += 2.0 * a[idx - jrng]
        g /= len(idxs)

        # ------- momentum update and candidate -------
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # ------- evaluate candidate -------
        fv_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(fv_c * fv_c, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_c.max()

        # ------- acceptance -------
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(step, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()

        # ------- occasional random mass transfer -------
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta_frac = rng.random() * 0.5
                delta = a[i] * delta_frac
                alt = a.copy()
                alt[i] -= delta
                alt[j] += delta
                alt = _project_simplex(alt)
                fv_alt = np.fft.rfft(alt, n=L)
                conv_alt = np.fft.irfft(fv_alt * fv_alt, n=L)[: 2 * n - 1]
                alt_score = 2.0 * n * conv_alt.max()
                if alt_score < cur_score:
                    a = alt
                    cur_score = alt_score
                    if alt_score < best_score:
                        best_score = alt_score
                        best = alt.copy()

        # ------- occasional smoothing -------
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode="same")
            smooth = _project_simplex(smooth)
            fv_s = np.fft.rfft(smooth, n=L)
            conv_s = np.fft.irfft(fv_s * fv_s, n=L)[: 2 * n - 1]
            smooth_score = 2.0 * n * conv_s.max()
            if smooth_score < cur_score:
                a = smooth
                cur_score = smooth_score
                if smooth_score < best_score:
                    best_score = smooth_score
                    best = smooth.copy()

        # ------- step‑size decay -------
        step = max(1e-7, step * decay)

        # ------- early exit guard -------
        if time.monotonic() + 0.5 > deadline:
            break

    return best, best_score


# --------------------------------------------------------------
# Public entry point required by the harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # initialise from parent if available, otherwise fall back to a simple series
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = _sqrt_series(256)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    best, best_score = _optimize(a0, deadline, rng)

    # second start if time permits
    if time.monotonic() + 5.0 < deadline:
        rand_start = rng.random(a0.shape)
        rand_start = _project_simplex(rand_start)
        alt_best, alt_score = _optimize(rand_start, deadline, rng)
        if alt_score < best_score:
            best, best_score = alt_best, alt_score

    # safeguard – return the fallback if something went wrong
    if not np.isfinite(best_score):
        best = a0.copy()

    return [float(x) for x in best]
