import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection onto {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Full convolution via FFT (returns max and whole array)
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    """Return maximum value of a*a (full convolution) and the array."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv


# ------------------------------------------------------------
# Gradient of the soft‑max‑approximated maximum of the convolution
# ------------------------------------------------------------
def _gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given current vector a (length n) and weight vector w (length 2n‑1) that
    approximates the max via a soft‑max, compute the gradient of
    sum_k w_k * (a*a)_k.
    The formula reduces to a convolution of w with the reversed a.
    """
    rev_a = a[::-1]
    # linear convolution (size 3n‑2)
    c = np.convolve(w, rev_a)
    n = a.size
    # g_j = 2 * c[n‑1‑j] for j = 0..n‑1
    g = 2.0 * c[n - 1 :: -1][:n]
    return g

    n = a.size
