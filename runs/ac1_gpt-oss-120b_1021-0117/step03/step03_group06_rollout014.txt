import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of x with itself via FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    return np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]

def _score(x: np.ndarray, n: int, L: int) -> float:
    """Exact score used by the evaluator (sum of x is assumed positive)."""
    c = _conv_fft(x, L)
    return 2.0 * n * c.max() / (x.sum() ** 2)

        cur_sc = best_shape_sc

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop – entropic mirror descent with a
    #     multi‑active‑set sub‑gradient, occasional stochastic moves
    #     and jittered restarts.
    # ------------------------------------------------------------------
    best = cur.copy()
    best_sc = cur_sc
    it = 0
    no_improve = 0

    alpha0 = 0.9          # step‑size prefactor for mirror descent
    top_frac = 0.012      # fraction of convolution entries used for the sub‑gradient
    jitter_small = 1e-5   # jitter scale for periodic restarts
    jitter_big = 5e-5

    while time.monotonic() < deadline:
        it += 1

        # ----- compute full convolution and pick the most relevant indices -----
        conv = _conv_fft(cur, L)
        max_val = conv.max()
        ktop = max(1, int(top_frac * conv.size))
        # indices of the ktop largest convolution values
        idx_top = np.argpartition(conv, -ktop)[-ktop:]

        # ----- build a sub‑gradient that aggregates contributions from the top entries -----
        g = np.zeros_like(cur)
        for k in idx_top:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # ----- mirror‑descent (exponentiated) update -----
        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum <= 0.0 or not np.isfinite(cand_sum):
            cand = np.full_like(cur, 1.0 / n)
        else:
            cand = cand / cand_sum

        cand_sc = _score(cand, n, L)

        # ----- accept if improvement, otherwise Metropolis‑style move -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # temperature decays like 1/√it
            temp = 0.12 / math.sqrt(it)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (temp + 1e-12)):
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()

        # ----- occasional random mass‑transfer move (5 attempts per trigger) -----
        if rng.random() < 0.09:
            for _ in range(5):
                i = rng.integers(0, n)
                if cur[i] == 0.0:
                    continue
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp /= tmp.sum()
                tmp_sc = _score(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()

        # ----- jittered restart from the current best every few thousand iterations -----
        if it % 3500 == 0:
            jitter = rng.normal(scale=jitter_small, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp /= tmp.sum()
            tmp_sc = _score(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best = tmp_sc, tmp.copy()
            no_improve = 0

        # ----- long‑stagnation restart with larger jitter -----
        if no_improve > 13000:
            jitter = rng.normal(scale=jitter_big, size=n)
            cur = np.maximum(best + jitter, 0.0)
            cur /= cur.sum()
            cur_sc = _score(cur, n, L)
            no_improve = 0

        # ----- early exit if a target bound is already beaten -----
        if best_sc < 1.50274349:
            break

    # Return the best vector found as a plain Python list.
    return [float(v) for v in best]
