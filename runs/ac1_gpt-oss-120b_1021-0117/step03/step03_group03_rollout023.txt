def max_conv(a):
    n=len(a)
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len-1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
    return conv.max()

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # get n
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = parent.size
    except Exception:
        n = 256
    # compute conv FFT length
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    # define function to evaluate score
    def eval_score(a):
        s = a.sum()
        if s < 0.01:
            return math.inf
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max() / (s * s))
    # initial candidate: uniform
    a_uni = np.full(n, 1.0/n, dtype=np.float64)
    best = a_uni.copy()
    best_score = eval_score(best)
    # also consider parent normalized
    try:
        parent = np.clip(parent, 0.0, 1000.0)
        s = parent.sum()
        if s >= 0.01:
            parent_norm = parent / s
            score_parent = eval_score(parent_norm)
            if score_parent < best_score:
                best = parent_norm.copy()
                best_score = score_parent
    except Exception:
        pass
    # random search
    # parameters
    min_frac = 0.25  # min fraction of n for m
    max_frac = 0.75  # max fraction
    # maybe start with larger m and gradually decrease for fine tuning
    while time.monotonic() < deadline:
        # random m
        frac = rng.uniform(min_frac, max_frac)
        m = max(1, int(round(frac * n)))
        # choose random indices distinct
        idx = rng.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0/m
        score = eval_score(a)
        if score < best_score:
            best = a.copy()
            best_score = score
            # early exit if target reached
            if best_score <= 1.50274349:
                break
        # optionally try local perturbations around best
        # generate a mutated version: add some jitter to masses
        # but for simplicity we may skip
    # return best as list of python floats
    return [float(v) for v in best]

import time
import math
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """
    Heuristic optimisation for the AC1 bound.
    Combines an Adam‑style projected gradient descent on the simplex
    (using a sub‑gradient built from the top convolution peaks) with
    occasional random restarts and a simple random‑sampling phase.
    Returns the best non‑negative sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time management (leave a small safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2.  Problem dimensions
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = int(parent.size)
    except Exception:
        n = 256
        parent = None

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 3.  Helper to evaluate the AC1 score (mirrors evaluate_sequence)
    # ------------------------------------------------------------------
    def _score(a: np.ndarray) -> float:
        s = a.sum()
        if s < 0.01:
            return math.inf
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max() / (s * s))

    # ------------------------------------------------------------------
    # 4.  Initial candidate (uniform) and optional parent fallback
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)  # uniform simplex point
    best = a.copy()
    best_score = _score(best)

    if parent is not None:
        parent = np.clip(parent, 0.0, 1000.0)
        s = parent.sum()
        if s >= 0.01:
            parent_norm = parent / s
            sc = _score(parent_norm)
            if sc < best_score:
                best = parent_norm.copy()
                best_score = sc
                a = parent_norm.copy()

    # ------------------------------------------------------------------
    # 5.  Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15          # base learning rate
    t = 0
    no_improve = 0
    max_stale = 2500   # restart after this many non‑improving iterations
    target_score = 1.50274349

    # ------------------------------------------------------------------
    # 6.  Main optimisation loop (projected Adam on the simplex)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- convolution (FFT) ---------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        # ---- keep the best solution ---------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ---- sub‑gradient from top‑k convolution peaks ---------------
        top_k = 15
        if conv.size <= top_k:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                i = np.arange(offset, n, dtype=np.int64)
                j = k - i
                g[i] += a[j]
            else:
                i = np.arange(0, n + offset, dtype=np.int64)
                j = k - i
                g[i] += a[j]
        g = 2.0 * g / max(1, len(idxs))

        # ---- Adam update --------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection back onto simplex ----------------------------
        a = _proj_simplex(a)

        # ---- restart if stuck ---------------------------------------
        if no_improve >= max_stale:
            scale = 0.04
            jitter = np.exp(rng.normal(0.0, scale, size=n))
            a = _proj_simplex(best * jitter)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # 7.  Random‑sampling phase (quickly explore diverse shapes)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        frac = rng.uniform(0.2, 0.9)
        msize = max(1, int(round(frac * n)))
        idx = rng.choice(n, size=msize, replace=False)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = 1.0 / msize
        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            if best_score <= target_score:
                break

    # ------------------------------------------------------------------
    # 8.  Return the best sequence found (plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
