def compute_gradient(a, conv, top_k):
    n = len(a)
    g = np.zeros_like(a)
    # get indices
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    for k in idxs:
        offset = k - (n - 1)
        if offset >= 0:
            # contributions from a[offset:] with a[:n-offset] reversed
            g[offset:] += a[:n - offset]
        else:
            # offset negative => shift left
            o = -offset
            g[:n - o] += a[o:]
    # factor 2 because gradient of conv entry w.r.t a is 2*shift
    return 2.0 * g / max(1, len(idxs))

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Returns the best non‑negative step‑function heights found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. Base sequence (fallback if the parent is unavailable)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = parent.size
        a = parent.copy()
    except Exception:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # normalise to the simplex (sum = 1, non‑negative)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    a = _proj_simplex(a)

    # ------------------------------------------------------------------
    # 3. Pre‑computations
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 2.0               # initial step magnitude
    beta = 0.8               # momentum decay
    top_k = 30               # number of peaks to smooth
    jitter_every = 200       # steps between random jitter
    jitter_std = 0.015       # std‑dev for multiplicative jitter
    max_stale = 3000         # restart after this many non‑improving steps
    target_score = 1.50274349

    v = np.zeros_like(a)      # momentum buffer
    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # 5. Helper to evaluate current score
    # ------------------------------------------------------------------
    def _score(vec):
        fa = np.fft.rfft(vec, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max(), conv

    # ------------------------------------------------------------------
    # 6. Initialise best candidate
    # ------------------------------------------------------------------
    best_score, conv = _score(a)
    best = a.copy()

    # ------------------------------------------------------------------
    # 7. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- compute convolution and current score --------------------
        cur_score, conv = _score(a)

        # ---- record improvement ---------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        if top_k >= conv.size:

        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        g = np.zeros_like(a)
        for k in idxs:
            offset = int(k) - (n - 1)
            if offset >= 0:
                # shift to the right
                limit = n - offset
                if limit > 0:
                    g[offset:] += a[:limit]
            else:
                # shift to the left
                o = -offset
                limit = n - o
                if limit > 0:
                    g[:limit] += a[o:]
        # sub‑gradient of max (we step opposite to the gradient)
        g = 2.0 * g / max(1, len(idxs))

        # ---- momentum step & projection -------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v)

            a = _proj_simplex(a * jitter)

            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # 8. Return the best sequence (plain python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
