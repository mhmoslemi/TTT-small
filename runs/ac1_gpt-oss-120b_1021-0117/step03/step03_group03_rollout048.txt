def optimize_from(start, max_iters, top_k):
    a = start.copy()
    v = np.zeros_like(a)
    best_local = a.copy()
    best_local_score = compute_score(a)
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        # compute conv & score
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        # update best_local
        if cur_score < best_local_score:
            best_local = a.copy()
            best_local_score = cur_score
        # gradient
        idxs = np.argpartition(conv, -top_k)[-top_k:] if conv.size > top_k else np.arange(conv.size)
        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                idx = np.arange(offset, n, dtype=np.int64)
                partner = k - idx
                g[idx] += a[partner]
            else:
                idx = np.arange(0, n + offset, dtype=np.int64)  # offset negative => n+offset lower bound
                partner = k - idx
                g[idx] += a[partner]
        g = 2.0 * g / max(1, len(idxs))
        # update
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1 - beta) * g
        a = _proj_simplex(a - eta * v)
        # jitter occasionally
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v.fill(0.0)
        # optionally detection of stagnation but maybe not needed within this sub-run
        # early break if cur_score target reached
        if cur_score <= target_score:
            break
    return best_local, best_local_score, it

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
