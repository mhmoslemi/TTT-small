def _proj_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size+1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)

            w /= w_sum

        # ---- gradient of the smoothed max (2·(w*a) convolution) ----
        fw = np.fft.rfft(w, n=fft_len_wa)
        fa2 = np.fft.rfft(a, n=fft_len_wa)
        conv_wa = np.fft.irfft(fw * fa2, n=fft_len_wa)[:conv_len_wa]
        start_idx = len_w - 1
        g = 2.0 * conv_wa[start_idx:start_idx + n]

        # ---- momentum & step ----
        v = mu * v + (1.0 - mu) * g
        eta = eta0 / math.sqrt(it)
        a = _proj_simplex(a - eta * v)

        # ---- anneal temperature and occasional jitter ----
        beta *= beta_inc
        if it % 300 == 0:
            jitter = np.exp(rng.normal(0.0, 0.005, size=n))
            a = _proj_simplex(a * jitter)

    # --------------------------------------------------------------
    # 5. Return the best discovered sequence
    # --------------------------------------------------------------
    return [float(x) for x in best]
