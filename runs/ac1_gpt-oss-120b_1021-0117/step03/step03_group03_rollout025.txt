import time
import math
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.full_like(v, 1.0 / n)
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimiser for the AC1 upper bound.
    Returns the best non‑negative step‑function (as a list) found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. initialise sequence (fallback if parent not present)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = int(parent.size)
        a = np.clip(parent, 0.0, 1000.0)
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    a = _proj_simplex(a)

    # ------------------------------------------------------------------
    # 2. keep the best candidate seen so far
    # ------------------------------------------------------------------
    best = a.copy()
    # (for a short vector we can afford a dense convolution here)
    best_score = 2.0 * n * np.max(np.convolve(best, best))

    # also test the uniform distribution as a cheap fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = 2.0 * n * np.max(np.convolve(uniform, uniform))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 3. pre‑compute FFT parameters
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 4. optimiser hyper‑parameters
    # ------------------------------------------------------------------
    eta0 = 1.0            # base learning rate
    beta = 0.9            # momentum coefficient
    momentum = np.zeros_like(a)

    top_k = max(5, n // 40)          # number of convolution peaks used for the gradient
    jitter_every = 250               # steps between random jitter
    jitter_std = 0.02                # jitter standard deviation (log‑normal)
    max_stale = 5000                 # restart after this many non‑improving steps
    target_score = 1.50274349

    it = 0
    no_improve = 0

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 5. main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- convolution via FFT -------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        M = conv.max()
        S = a.sum()
        cur_score = 2.0 * n * M / (S * S)

        # ---- keep the best -------------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ---- gradient approximation (top‑k peaks) -------------------------------
        if conv.size > top_k:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size)

        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                idx = np.arange(offset, n, dtype=np.int64)
                partner = k - idx
                g[idx] += a[partner]
            else:
                idx = np.arange(0, n + offset, dtype=np.int64)
                partner = k - idx
                g[idx] += a[partner]

        if idxs.size:
            g = 2.0 * g / float(idxs.size)     # average contribution, factor 2 for ∂(a*a)

        # ---- momentum + step ----------------------------------------------------
        it += 1
        eta = eta0 / math.sqrt(it)
        momentum = beta * momentum + (1.0 - beta) * g
        a = _proj_simplex(a - eta * momentum)

        # ---- occasional jitter to escape plateaus --------------------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            momentum[:] = 0.0
            no_improve = 0

        # ---- restart if stagnating ------------------------------------------------
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            a = _proj_simplex(a * perturb)
            momentum[:] = 0.0
            no_improve = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:

        if i == j:

        if a[i] <= 1e-12:

            best_score = cand_score
            a = a_candidate

    # 7. return the best sequence found (plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
