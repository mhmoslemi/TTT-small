    import time, math, numpy as np
    rng= np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.5)   # safety

    # init
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = np.ones(n, dtype=np.float64)
    # enforce non-neg & normalize
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # compute initial score
    cur = a.copy()
    cur_conv = _conv_fft(cur, L)
    cur_sc = _score_from_conv(cur_conv, n, cur.sum())
    best = cur.copy()
    best_sc = cur_sc

    it = 0
    # hyperparameters
    alpha0 = 0.5   # maybe smaller for smoother descent
    eps_move = 0.2
    jitter_scale = 1e-5
    t_swap = 0.001      # small probability for swapping?
    # also we may compute temperature for Metropolis approx
    temp0 = 0.1
    no_improve = 0

    while time.monotonic() < deadline:
        # compute gradient step
        conv = _conv_fft(cur, L)
        # maybe compute top K positions for gradient
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # exponentiated update
        it += 1
        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum
        else:
            cand = np.full_like(cur, 1.0 / n)

        cand_conv = _conv_fft(cand, L)
        cand_sc = _score_from_conv(cand_conv, n, cand.sum())
        # accept if better or metropolis
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # acceptance probability decreasing with it
            # compute diff
            prob = math.exp(-(cand_sc - cur_sc) / max(eta, 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()

        if rng.random() < 0.07:

            j = rng.integers(0, n-1)

                tmp[i] -= delta
                tmp[j] += delta

                    tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())

                        no_improve = 0
                    if tmp_sc < best_sc:

import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution x*x using a power‑of‑two FFT length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv

def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact same formula as `evaluate_sequence`."""
    return 2.0 * n * conv.max() / (sum_a * sum_a)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # Initial vector: use the supplied parent if present, otherwise uniform.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

        a = np.full(2000, 1.0 / 2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # normalise to a probability simplex

    n = a.size
