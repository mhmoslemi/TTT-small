def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of x with itself using an FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    return np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]

def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Score = 2*n*max(conv) when the vector sums to 1."""
    return 2.0 * n * conv.max()

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            w = w / w_sum

        # --------------------------------------------------------------
        # 2.  Gradient (smooth sub‑gradient) via correlation
        # --------------------------------------------------------------
        W = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(A * np.conj(W), n=L)   # correlation of cur and w
        g = 2.0 * corr[:n]                         # smooth sub‑gradient

        # --------------------------------------------------------------
        # 3.  Multiplicative (mirror‑descent) candidate
        # --------------------------------------------------------------
        eta_mul = step_mul0 / math.sqrt(it)
        mul_factor = np.exp(np.clip(-eta_mul * g, -50, 50))
        cand_mul = cur * mul_factor
        cand_mul_sum = cand_mul.sum()
        if cand_mul_sum <= 0.0 or not np.isfinite(cand_mul_sum):
            cand_mul = np.full_like(cur, 1.0 / n)
        else:
            cand_mul = cand_mul / cand_mul_sum

        # --------------------------------------------------------------
        # 4.  Additive (projected‑gradient) candidate
        # --------------------------------------------------------------
        eta_add = step_add0 / math.sqrt(it)
        cand_add = cur - eta_add * g
        cand_add = _project_simplex(cand_add)

        # --------------------------------------------------------------
        # 5.  Evaluate candidates
        # --------------------------------------------------------------
        # evaluate multiplicative candidate
        A_mul = np.fft.rfft(cand_mul, n=L)
        sc_mul = _score_from_conv(np.fft.irfft(A_mul * A_mul, n=L)[: 2 * n - 1], n)

        # evaluate additive candidate
        A_add = np.fft.rfft(cand_add, n=L)
        sc_add = _score_from_conv(np.fft.irfft(A_add * A_add, n=L)[: 2 * n - 1], n)

        # pick the better of the two
        if sc_mul < sc_add:
            cand, cand_sc = cand_mul, sc_mul
        else:
            cand, cand_sc = cand_add, sc_add

        # --------------------------------------------------------------
        # 6.  Acceptance (deterministic improvement + occasional uphill)
        # --------------------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (temp0 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # keep global best
        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()
            no_improve = 0

        # --------------------------------------------------------------
        # 7.  Random mass‑transfer move (small perturbation)
        # --------------------------------------------------------------
        if rng.random() < 0.07:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.25
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp = _project_simplex(tmp)
                # evaluate
                A_tmp = np.fft.rfft(tmp, n=L)
                sc_tmp = _score_from_conv(np.fft.irfft(A_tmp * A_tmp, n=L)[: 2 * n - 1], n)
                if sc_tmp < cur_sc:
                    cur, cur_sc = tmp, sc_tmp
                if sc_tmp < best_sc:
                    best_sc, best = sc_tmp, tmp.copy()
                    no_improve = 0

        # --------------------------------------------------------------
        # 8.  Periodic jittered restart from the best point
        # --------------------------------------------------------------
        if it % 3000 == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            A_tmp = np.fft.rfft(tmp, n=L)
            sc_tmp = _score_from_conv(np.fft.irfft(A_tmp * A_tmp, n=L)[: 2 * n - 1], n)
            if sc_tmp < cur_sc:
                cur, cur_sc = tmp, sc_tmp
            if sc_tmp < best_sc:
                best_sc, best = sc_tmp, tmp.copy()
                no_improve = 0

        # --------------------------------------------------------------
        # 9.  Restart after stagnation
        # --------------------------------------------------------------
        if no_improve > 8000:
            cur = best + rng.normal(scale=jitter_scale * 5.0, size=n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            A_cur = np.fft.rfft(cur, n=L)
            cur_sc = _score_from_conv(np.fft.irfft(A_cur * A_cur, n=L)[: 2 * n - 1], n)
            no_improve = 0

        # --------------------------------------------------------------
        # 10. Early exit if target achieved
        # --------------------------------------------------------------
        if best_sc < 1.50274349:
            break

    # Return a plain Python list
    return [float(x) for x in best]
