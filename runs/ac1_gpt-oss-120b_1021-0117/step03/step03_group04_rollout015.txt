import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT (full length)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _subgradient_from_conv(a: np.ndarray, conv: np.ndarray, n: int,
                           rng: np.random.Generator, max_active: int = 10) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (a on the simplex)."""
    maxv = conv.max()
    tol = 1e-12 * maxv
    idx = np.where(conv >= maxv - tol)[0]
    if idx.size > max_active:
        idx = rng.choice(idx, size=max_active, replace=False)
    w = 1.0 / idx.size
    g = np.zeros_like(a)
    for shift in idx:
        m = shift - (n - 1)          # offset in [-n+1, n-1]
        if m >= 0:
            # a_{j-m} term
            g[m:] += w * a[: n - m]
            # a_{j+m} term
            g[: n - m] += w * a[m:]
        else:
            p = -m
            # a_{j-m} = a_{j+p}
            g[: n - p] += w * a[p:]
            # a_{j+m} = a_{j-p}
            g[p:] += w * a[: n - p]
    return g

            start_seq = fresh / fresh.sum()

    # ------------------------------------------------------------------
    # 10. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_global_seq]
