import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a simplex‑normalised vector (sum = 1)
# ------------------------------------------------------------
def _score_simplex(a: np.ndarray, fft_len: int) -> float:
    """2 * n * max(conv) for a vector whose entries sum to 1."""
    n = a.size
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    rng = np.random.default_rng(seed)

    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------- initialise -------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)

            a0 = a0 / s                     # simplex normalisation
    else:

    n = a0.size
    # FFT length sufficient for all convolutions we need (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    cur = best_vec.copy()

    # ------------------- optimisation parameters -------------------
    eps0 = 0.5               # initial smoothing for soft‑max
