import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation (sum of coefficients is assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray | None = None) -> float:
    """Return 2*n*max(conv); if conv not supplied, compute via FFT."""
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative coefficient list with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # small safety margin

    # -------------------- initial vector --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: exponential decay, length 512
        n_fallback = 512

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a_sum           # project onto the simplex (sum=1)

    n = a.shape[0]
    # FFT length sufficient for both 2n‑1 and 3n‑2 linear convolutions
    L = 1 << ((3 * n - 2).bit_length())

    # -------------------- bookkeeping --------------------
    best_vec = a.copy()
    best_sc = _score(best_vec)
    cur = a.copy()
    cur_sc = best_sc

    # hyper‑parameters
    step0 = 0.8                # base step magnitude
    eps0 = 0.5                 # initial smoothing for soft‑max
    eps_decay = 0.998          # exponential decay per iteration
    eps_min = 1e-7
    no_improve = 0
    it = 0

    # -------------------- optimisation loop --------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
