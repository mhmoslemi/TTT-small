import time
import numpy as np

def _sqrt_series(n: int) -> np.ndarray:
    """Return length‑n non‑negative series a satisfying a*a = 1+…+x^{2n‑2}
    (up to floating‑point error)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a[i] * a[k-i]
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) * 0.5
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 upper bound
    is close to the theoretical optimum.  The routine ignores the seed (the
    construction is deterministic) but respects the time budget.
    """
    # reserve a tiny margin for the final return
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Desired length – use the parent if available, otherwise a modest default
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256

    # Build the exact square‑root series for the given length
    a = _sqrt_series(n)

    # Clip possible tiny negatives caused by rounding
    a = np.maximum(a, 0.0)

    # Normalise to unit sum (the objective is scale‑invariant)
    total = a.sum()
    if total < 1e-12:               # safety fallback
        a = np.full(n, 1.0 / n, dtype=np.float64)
        total = 1.0
    else:
        a = a / total

    # Ensure we return before the deadline (nothing else to do)
    # Convert to plain Python floats as required by the evaluator
    return [float(v) for v in a]
