import time
import numpy as np

# ----------------------------------------------------------------------
# Generate the classic “sqrt‑series” a such that (a*a)[k]=1 for k=0..n‑1
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Return the non‑negative sequence a of length n with a*a = 1+…+x^{2n‑2}
    on the first n coefficients (the classical square‑root series)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        # compute s = Σ_{i=1}^{k‑1} a[i] * a[k‑i]
        if k == 1:
            s = 0.0
        else:
            # inner product of a[1:k] with reversed a[1:k]
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    The construction uses the sqrt‑series, scaled to unit L1 norm."""

        n = 256

    # Build the sqrt series of the desired length
    a = _sqrt_series(n)

    # Ensure non‑negativity (the recursion already guarantees it, but guard)
    a = np.maximum(a, 0.0)

    if total <= 0.0:

    np.clip(a, 0.0, 1000.0, out=a)

    # Return as plain Python floats
    return [float(x) for x in a]
