import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by ``evaluate_sequence``.
    The theoretical lower bound of the score is 2, attained by a uniform
    distribution (after normalising the ℓ¹‑norm to 1).  This implementation
    therefore returns a uniform vector of the appropriate length.
    """
    # Determine the length to use – prefer the provided parent sequence if
    # present, otherwise fall back to a modest size.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 256   # any reasonable default when no parent is given

    # Uniform ℓ¹‑normalised vector: each entry = 1/n.
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # Convert to plain Python floats for the evaluator.
    return [float(x) for x in uniform]
