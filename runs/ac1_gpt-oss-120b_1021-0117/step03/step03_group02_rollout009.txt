import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Full convolution via FFT (returns max value and the array)
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv

# ------------------------------------------------------------
# Subgradient of the maximal convolution entry
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Return a subgradient for max_{k} (a*a)[k] (full convolution)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        if k < n:
            # left part: i in [0, k]  -> add a[k - i]
            g[: k + 1] += a[k::-1]
            # right part: i in [k, n-1] -> add a[i - k]
            g[k:] += a[: n - k]
        else:
            lo = k - (n - 1)
            # only the left contribution exists
            # indices lo..n-1 map to a reversed slice n-1..lo
            g[lo:] += a[n - 1 : lo - 1 : -1]
    return g

# Main entry point
# ------------------------------------------------------------

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)  # keep 2 s margin

    if 'height_sequence_1' in globals():

        a = rng.random(2500, dtype=np.float64)

    # normalise to the simplex (sum = 1)
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two length for FFT

    # evaluate initial solution
    maxb, conv = _conv_full_max(a, L)
    cur_score = 2.0 * n * maxb
    best = a.copy()
    best_score = cur_score

    # ------- optimisation parameters -------
    step = 0.3            # initial step magnitude
    min_step = 1e-6
    decay = 0.9997
    stall_limit = 400    # restart after this many non‑improving iterations
    prob_rand = 0.025    # chance of a random mass‑transfer perturbation

    stall = 0

    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- current convolution & max indices ----
        maxb, conv = _conv_full_max(a, L)
        cur_score = 2.0 * n * maxb

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            stall = 0
        else:
            stall += 1

        # ---- subgradient of the maximal entry ----
        max_idx = np.where(conv == maxb)[0]
        g = _subgradient(a, max_idx)

        # normalise gradient direction
        norm = np.linalg.norm(g) + 1e-12
        direction = g / norm

        # ---- gradient step with projection ----
        cand = a - step * direction
        cand = _project_simplex(cand)

        # evaluate candidate

        cand_score = 2.0 * n * maxb_c

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            # (best will be updated at top of loop)

        if rng.random() < prob_rand:

            stall = 0

        # ---- early exit safeguard ----
        if time.monotonic() + 0.5 > deadline:
            break

    return [float(x) for x in best]
