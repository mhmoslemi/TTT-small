def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math, cvxpy as cp
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    # initialize a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        idx = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((idx+1)*(n-idx))
    a = np.maximum(a, 0.0)
    n = a.size
    S_target = math.sqrt(2.0 * n)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, S_target / n)
    else:
        a = a * (S_target / a_sum)
    # compute initial score
    L = 1 << ((2 * n - 1).bit_length())
    def conv_fft(x):
        fx = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fx * fx, n=L)[:2*n-1]
        return conv
    cur = a.copy()
    cur_conv = conv_fft(cur)
    cur_sc = float(2 * n * cur_conv.max() / (cur.sum() ** 2))
    best = cur.copy()
    best_sc = cur_sc
    it=0
    # parameters
    alpha0=0.8
    t_mix=0.4
    # main loop
    while time.monotonic() < deadline:
        it += 1
        # compute conv and max index
        conv = conv_fft(cur)
        k = int(conv.argmax())
        # subgradient
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]
        # mirror descent step
        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum > 0 and np.isfinite(cand_sum):
            cand = cand * (S_target / cand_sum)
        else:
            cand = cur.copy()
        # evaluate candidate
        cand_sc = float(2 * n * conv_fft(cand).max() / (cand.sum() ** 2))
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
        else:
            # occasional acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
        # keep best
        if cur_sc < best_sc:
            best_sc, best = cur_sc, cur.copy()
        # occasional LP refinement
        if it % 50 == 0:
            # solve LP to get g_opt
            M = conv.max()
            f_const = cur
            g_var = cp.Variable(n, nonneg=True)
            constraints = [cp.conv(f_const, g_var) <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            try:

            except Exception:
                # fallback: skip
                g_opt = None
            else:
                if g_var.value is not None:
                    g_opt = np.maximum(g_var.value, 0.0)
                else:
                    g_opt = None
            if g_opt is not None:
                # scale to target sum
                g_opt_sum = g_opt.sum()
                if g_opt_sum > 0:
                    g_opt = g_opt * (S_target / g_opt_sum)
                    # mix with current

                    new_sc = float(2 * n * conv_fft(new).max() / (new.sum() ** 2))
                    if new_sc < cur_sc:

        # occasional random mass transfer

                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.2
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0)
                tmp = tmp * (S_target / tmp.sum())

                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if cur_sc < best_sc:

        # check termination
        if best_sc < 1.50274349:
            break
        # safety: if too many iterations, break
        if it > 20000:
            break

    # ensure we return list of floats
    return [float(x) for x in best]

import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using FFT (length L)."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv


def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clamp to non‑negative and rescale to a given L1 sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0 or not np.isfinite(s):
        # fall back to uniform distribution
        return np.full_like(v, target / v.size)
    return v * (target / s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # Initialise: use the provided parent if present, otherwise a simple uniform.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest length (≈2000) uniform start – the exact length is irrelevant.
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    n = a.size
    target_sum = math.sqrt(2.0 * n)          # the “natural” normalisation
    a = _project_to_sum(a, target_sum)

    # pre‑allocate FFT length (next power of two for speed)
    L = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------------------
    # Scoring helpers (the same formula used inside evaluate_sequence)
    # ----------------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        conv = _conv_fft(vec, L)
        max_conv = conv.max()
        s = vec.sum()
        return float(2.0 * n * max_conv / (s * s))

    cur = a.copy()
    cur_sc = _score(cur)

    best = cur.copy()
    best_sc = cur_sc

    # ----------------------------------------------------------------------
    # Parameters for the search.
    # ----------------------------------------------------------------------
    it = 0
    alpha0 = 1.0           # initial step multiplier for the exponentiated update
    eps_move = 0.25        # max fraction of mass moved in a random transfer
    jitter_scale = 1e-4    # Gaussian jitter scale for occasional restarts
    top_k = 3              # number of top convolution entries used for a richer sub‑gradient
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # ----- current convolution and top‑k indices -----
        conv = _conv_fft(cur, L)
        # indices of the largest few entries (including possible ties)
        top_idx = np.argpartition(-conv, range(min(top_k, conv.size)))[:top_k]

        # ----- sub‑gradient of the max‑conv (sum over the chosen tops) -----
        g = np.zeros_like(cur)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]          # contribution of each active term

        # ----- exponentiated (mirror) step -----
        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand = _project_to_sum(cand, target_sum)
        cand_sc = _score(cand)

        # ----- acceptance of the candidate -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0

            prob = math.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
