import time
import numpy as np

def _conv_full(v: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of v with itself using FFT."""
    n = v.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(v, n=L)
    prod = fv * fv
    conv = np.fft.irfft(prod, n=L)[: 2 * n - 1]
    return conv


def _score(v: np.ndarray) -> float:
    """Score defined by the evaluator (2 * n * max(conv) / sum(v)^2)."""
    n = v.size
    conv = _conv_full(v)
    return float(2.0 * n * conv.max() / (v.sum() ** 2))


def _simplex_normalize(v: np.ndarray) -> np.ndarray:
    """
    Project onto the simplex {x >= 0, sum(x) = 1}.
    Simple Euclidean projection (O(n log n)).
    """
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / v.size)
    return w

        g = np.zeros_like(cur)
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * cur[max_idx - idx]          # sub‑gradient of max term

        # ------------------------------------------------------------------
        # additive (projected‑gradient) step
        # ------------------------------------------------------------------
        alpha = add_alpha0 / np.sqrt(it)
        cand_add = _simplex_normalize(cur - alpha * g)
        cand_add_sc = _score(cand_add)

        # ------------------------------------------------------------------
        # multiplicative (exponential‑weights) step
        # ------------------------------------------------------------------
        eta = mul_eta0 / np.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = _simplex_normalize(cur * exp_factor)
        cand_mul_sc = _score(cand_mul)

        # ------------------------------------------------------------------
        # pick the best candidate (with a tiny chance of uphill move)
        # ------------------------------------------------------------------
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc
        else:
            # occasional uphill acceptance (simulated‑annealing style)
            temp = max(1e-6, 0.02 / np.sqrt(it))
            prob = np.exp(-(min(cand_add_sc, cand_mul_sc) - cur_sc) / temp)
            if rng.random() < prob:
                if cand_add_sc < cand_mul_sc:
                    cur, cur_sc = cand_add, cand_add_sc
                else:
                    cur, cur_sc = cand_mul, cand_mul_sc

        # ------------------------------------------------------------------
        # bookkeeping: best solution so far
        # ------------------------------------------------------------------
        if cur_sc < best_sc:
            best, best_sc = cur.copy(), cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # random mass‑transfer move if progress stalls
        # ------------------------------------------------------------------
        if no_improve > 2000 and rng.random() < 0.3:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _simplex_normalize(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # ------------------------------------------------------------------
        # jittered restart toward the best known solution
        # ------------------------------------------------------------------
        if it % 5000 == 0 and rng.random() < 0.4:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = _simplex_normalize(best + jitter)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # early exit if the target is already beaten
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # final short refinement using cheap random transfers
    # ----------------------------------------------------------------------
    refin_deadline = time.monotonic() + min(0.3, deadline - time.monotonic())
    while time.monotonic() < refin_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.4
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _simplex_normalize(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best, best_sc = tmp.copy(), tmp_sc

    # return the best vector found as a plain Python list
    return [float(x) for x in best]
