def optimize(initial, deadline, rng):

    a = initial.copy()
    n = a.shape[0]
    nfft = 1 << ((2*n - 1).bit_length())   # nearest power of two >2n-2
    # Initialize Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5   # initial learning rate
    lr_decay = 0.9995  # per iteration (slower maybe)
    iteration = 1
    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # Could pre-calc compute score using conv maybe faster: we have conv_max value and sum=1 => score=2*n*max_val
    # We'll compute using conv_max for speed.

    while time.monotonic() < deadline:
        # compute conv using FFT
        fv = np.fft.rfft(a, n=nfft)
        conv_full = np.fft.irfft(fv * fv, n=nfft)[:2*n-1]
        max_val = conv_full.max()
        peak_idxs = np.flatnonzero(np.abs(conv_full - max_val) <= 1e-12)
        # compute gradient
        grad = np.zeros_like(a)
        for idx in peak_idxs:
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            js = np.arange(lo, hi+1)
            grad[js] += 2.0 * a[idx - js]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        # projection onto simplex (L1=1)
        cand = _project_simplex(cand)

        # Evaluate cand
        fv2 = np.fft.rfft(cand, n=nfft)
        conv2 = np.fft.irfft(fv2 * fv2, n=nfft)[:2*n-1]
        cand_max = conv2.max()
        cand_score = 2.0 * n * cand_max

        if cand_score < best_score - 1e-12:
            best = cand.copy()
            best_score = cand_score

        # Accept if improves, else maybe accept with probability
        cur_score = 2.0 * n * max_val
        if cand_score < cur_score:
            a = cand
            # keep m and v as is (no revert)
        else:
            # acceptance with probability exp(-(cand_score - cur_score) / temperature)
            # temperature may be related to lr (like step size)
            temp = max(lr, 1e-4)
            if rng.random() < math.exp(-(cand_score - cur_score) / temp):
                a = cand

        # Random perturbation with probability p_random
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                frac = rng.random() * 0.5  # move up to 50% of a[i]
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                # project onto simplex (though sum unchanged and no negative)
                cand2 = _project_simplex(cand2)
                # evaluate
                fv3 = np.fft.rfft(cand2, n=nfft)
                conv3 = np.fft.irfft(fv3 * fv3, n=nfft)[:2*n-1]
                cand2_max = conv3.max()
                cand2_score = 2.0 * n * cand2_max
                if cand2_score < best_score:
                    best = cand2.copy()
                    best_score = cand2_score
                if cand2_score < cur_score:
                    a = cand2
                    # cur_score updates later

        # smoothing with probability p_smooth
        if rng.random() < 0.005:
            kernel = np.array([0.25, 0.5, 0.25])
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            fv4 = np.fft.rfft(smooth, n=nfft)
            conv4 = np.fft.irfft(fv4 * fv4, n=nfft)[:2*n-1]
            smooth_max = conv4.max()
            smooth_score = 2.0 * n * smooth_max
            if smooth_score < best_score:
                best = smooth.copy()
                best_score = smooth_score
            if smooth_score < cur_score:
                a = smooth

        # update step size and iteration
        lr = max(1e-5, lr * lr_decay)
        iteration += 1

        # early exit if time near goal
        if time.monotonic() + 0.1 > deadline:
            break

    return best, best_score

import time
import math
import numpy as np

# -------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_idx = np.where(rho_arr > 0)[0]
    if rho_idx.size:
        rho = rho_idx[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:          # extremely unlikely, but guard against it
        w[:] = 1.0 / n
    return w


# -------------------------------------------------------------
# Fast convolution (FFT) returning maximum value and its indices
# -------------------------------------------------------------
def _conv_max_and_peaks(vec: np.ndarray, nfft: int, eps: float = 1e-12):
    """Full convolution of vec with itself using FFT, returning
    max value, indices attaining the max (within eps), and the full conv."""
    fv = np.fft.rfft(vec, n=nfft)
    conv = np.fft.irfft(fv * fv, n=nfft)[: 2 * vec.size - 1]
    max_val = conv.max()
    peak_idxs = np.flatnonzero(np.abs(conv - max_val) <= eps)
    return float(max_val), peak_idxs, conv


# -------------------------------------------------------------
# Optional analytic sqrt‑series used when no parent is present
# -------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# -------------------------------------------------------------
# Core optimisation routine (Adam‑style gradient on the current peak)
# -------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    nfft = 1 << ((2 * n - 1).bit_length())

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr = 0.5                # initial learning rate
    lr_decay = 0.9995

    # initial best
    cur_max, _, _ = _conv_max_and_peaks(a, nfft)
    best = a.copy()
    best_score = 2.0 * n * cur_max
    iteration = 1

    while True:
        if time.monotonic() >= deadline:
            break

        # ---- current state ----
        cur_max, peak_idxs, _ = _conv_max_and_peaks(a, nfft)
        cur_score = 2.0 * n * cur_max

        # ---- gradient of the (possibly multiple) peaks ----
        grad = np.zeros_like(a)
        for idx in peak_idxs:
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            js = np.arange(lo, hi + 1, dtype=np.intp)
            grad[js] += 2.0 * a[idx - js]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step_vec
        cand = _project_simplex(cand)

        # ---- evaluate candidate ----
        cand_max, _, _ = _conv_max_and_peaks(cand, nfft)
        cand_score = 2.0 * n * cand_max

        # ---- accept / reject ----
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            temp = max(lr, 1e-4)
            if rng.random() < math.exp(-(cand_score - cur_score) / temp):
                a = cand
                cur_score = cand_score

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # ---- occasional random mass transfer ----
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.5
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                c2_max, _, _ = _conv_max_and_peaks(cand2, nfft)
                c2_score = 2.0 * n * c2_max

                    best_score = c2_score
                if c2_score < cur_score:
                    a = cand2

        if rng.random() < 0.005:

            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            s_max, _, _ = _conv_max_and_peaks(smooth, nfft)

            if s_score < best_score:

                best_score = s_score
            if s_score < cur_score:
                a = smooth
