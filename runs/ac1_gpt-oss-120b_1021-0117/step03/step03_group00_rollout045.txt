    else:

    best = a0.copy()
    best_sc = _score(best) # compute full
    cur = best.copy()
    cur_sc = best_sc

    eps0 = 0.5
    eps_decay = 0.997
    eps_min = 1e-9
    step0 = 0.8
    mu = 0.9
    v = np.zeros_like(cur)

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # compute FFT of cur
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        # compute w by smooth or exact mask
        # choose smooth if eps > some threshold else mask
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-4:
            # smooth
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            mask = conv >= max_val - 1e-12  # tolerance
            cnt = mask.sum()
            if cnt == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = mask.astype(np.float64) / cnt

        # compute gradient via convolution of cur with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector with unit sum: 2 * n * max(convolution)
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray = None) -> float:
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # --------------------------------------------------------
    # initial vector (use parent if available)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

    else:

        n_fallback = 2000

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s == 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # normalise to simplex

    n = a0.size
    # FFT length large enough for both 2n‑1 and 3n‑2 linear convolutions
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------

    best_vec = a0.copy()
    best_sc = _score(best_vec)

    # parameters
    eps0 = 0.5
    eps_decay = 0.997
    eps_min = 1e-9
    step0 = 0.8
    mu = 0.9                     # momentum factor
    v = np.zeros_like(cur)      # momentum buffer

        max_val = conv.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
