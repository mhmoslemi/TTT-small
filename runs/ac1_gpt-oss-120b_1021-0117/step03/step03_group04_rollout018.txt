    best = None

import time
import math
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimizer for the AC1 upper bound.
    Uses an Adam‑style projected sub‑gradient method with top‑K
    autocorrelation terms and occasional jitter.
    Returns the best coefficient list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Load optional parent construction
    # ------------------------------------------------------------------
    parent_raw = globals().get('height_sequence_1', None)
    if parent_raw is not None:
        try:
            parent_arr = np.asarray(parent_raw, dtype=np.float64)
            if (parent_arr.ndim != 1 or parent_arr.size == 0 or
                    not np.isfinite(parent_arr).all()):
                raise ValueError
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            s = parent_arr.sum()
            if s >= 0.01:
                parent_arr = parent_arr / s
            else:
                parent_arr = None
        except Exception:
            parent_arr = None
    else:
        parent_arr = None

    # ------------------------------------------------------------------
    # 2. Length and basic candidates
    # ------------------------------------------------------------------
    n = int(parent_arr.size) if (parent_arr is not None) else 256

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    tent = np.clip(tent, 0.0, None)
    if tent.sum() > 0.0:
        tent /= tent.sum()
    else:
        tent = uniform.copy()

    rand_vec = rng.random(n)
    rand_vec /= rand_vec.sum()

    candidates = [uniform, tent, rand_vec]
    if parent_arr is not None:
        candidates.append(parent_arr)

    # ------------------------------------------------------------------
    # 3. Fast objective (sum of coefficients is 1, so bound = 2*n*max conv)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 4. Pick the best starter
    # ------------------------------------------------------------------
    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    # ------------------------------------------------------------------
    # 5. Adam‑style projected sub‑gradient optimisation
    # ------------------------------------------------------------------
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5                 # base learning rate
    K = 30                    # number of top autocorrelation terms
    target = 1.50274349

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution and score ----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = 2.0 * n * conv.max()

        # ---- record best ----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = it
            if best_score <= target:
                break

        # ---- top‑K indices and weights for the sub‑gradient ----
        if K < conv_len:
            top_idx = np.argpartition(-conv, K)[:K]
        else:
            top_idx = np.arange(conv_len, dtype=np.int64)
        top_vals = conv[top_idx]
        wsum = top_vals.sum()
        if wsum > 0:
            weights = top_vals / wsum
        else:
            weights = np.full_like(top_vals, 1.0 / top_vals.size)

        # ---- sub‑gradient of the max‑convolution term ----
        g = np.zeros_like(a)
        for shift, w in zip(top_idx, weights):
            offset = shift - (n - 1)
            if offset >= 0:
                idx_range = np.arange(offset, n, dtype=np.int64)
                partner = shift - idx_range
            else:
                idx_range = np.arange(0, n + offset, dtype=np.int64)
                partner = shift - idx_range
            g[idx_range] += w * 2.0 * a[partner]

        # ---- Adam update (learning rate decays with sqrt(it)) ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        step = (lr0 / math.sqrt(it)) * (m / (np.sqrt(v) + eps))
        a = _proj_simplex(a - step)

        # ---- occasional jitter to avoid plateaus ----
        if it % 500 == 0:

            a = _proj_simplex(best * pert)
            # reset moments to avoid stale directions
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it

    # ------------------------------------------------------------------
    # 6. Return the best sequence discovered
    # ------------------------------------------------------------------
    return [float(v) for v in best]
