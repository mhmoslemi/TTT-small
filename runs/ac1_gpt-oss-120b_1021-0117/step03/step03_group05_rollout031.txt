import time
import numpy as np

# ----------------------------------------------------------------------
# helper utilities (FFT convolution, score, simplex projection)
# ----------------------------------------------------------------------
def _fft_len(n):
    """next power of two for length of linear convolution of size n."""
    return 1 << ((2 * n - 1).bit_length())

def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution via FFT (real, length L)."""
    fv = np.fft.rfft(v, n=L)
    cv = np.fft.irfft(fv * fv, n=L)
    return cv[: 2 * len(v) - 1]

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, len(v) + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(v: np.ndarray, n: int, L: int) -> float:
    """Return the AC1 bound for vector v (assumes sum(v) != 0)."""
    conv = _conv_fft(v, L)
    maxb = conv.max()
    s = v.sum()
    # s is never zero (simplex projection guarantees >= 1e-12)
    return 2.0 * n * maxb / (s * s)

    The routine respects the time budget and keeps the best solution found.
    """

    # keep a small safety margin for the final fallback

    n = a.size
    L = _fft_len(n)                     # FFT length (power of two)
