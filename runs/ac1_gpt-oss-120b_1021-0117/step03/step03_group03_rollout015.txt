    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0., float(budget_s)-1.)
    # initial a:
    try:
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0., 1000.)
    except Exception:
        n = 256
        a0 = np.ones(n, dtype=np.float64)

    n = a0.size
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1./n)
    else:
        a = a / s

    # keep best
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    # Precompute FFT length
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len-1).bit_length())
    # Optimization parameters
    eta0 = 0.2  # initial step size for gradient descent (since we use projection, step may be large)
    # choose small because gradient magnitude can be large
    momentum_beta = 0.9
    v = np.zeros_like(a)
    it = 0

    # re-sidual tolerance for max detection
    eps = 1e-12

    while time.monotonic() < deadline:
        it += 1
        # compute convolution via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        maxval = conv.max()
        # compute gradient exactly for all max indices
        idxs = np.where(conv >= maxval - eps)[0]
        # compute gradient
        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                # add a[i+offset] to g[i] for i in 0..n-1-offset
                # i runs over positions that have partner index i+offset < n
                i_end = n - offset
                # vectorized slice
                g[:i_end] += a[offset:]
            else:
                # offset negative => partner is i+offset (i+offset >=0) meaning i runs offset..n-1?

                i_start = -offset
                # length = n - i_start

        # gradient factor 2 from derivative
        g *= 2.0

        # Optionally normalize gradient magnitude
        # Use step size decaying over iterations
        eta = eta0 / np.sqrt(it)

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Returns the best non‑negative step‑function (unit‑sum) found before time runs out.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1. Initialise a feasible starting point
    # ------------------------------------------------------------------
    try:
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
    except Exception:
        a = np.ones(256, dtype=np.float64)

    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
    else:
        a[:] = a / s                     # normalise to unit sum

    # ------------------------------------------------------------------
    # 2. Pre‑compute FFT size for linear convolution
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 3. Evaluate the initial candidate
    # ------------------------------------------------------------------
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best = a.copy()
    best_score = 2.0 * n * conv.max()

    # also keep uniform as a cheap fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    fu = np.fft.rfft(uniform, n=fft_len)
    conv_u = np.fft.irfft(fu * fu, n=fft_len)[:conv_len]
    uniform_score = 2.0 * n * conv_u.max()
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 0.5           # base step size
    beta = 0.9           # momentum decay
    momentum = np.zeros_like(a)
    it = 0
    no_improve = 0
    max_stale = 4000
    tol_max = 1e-12
    target_score = 1.50274349
    jitter_interval = 150
    jitter_std = 0.02

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and current score ---------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- keep best ----------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ----- exact sub‑gradient from all maximal convolution entries ------
        idxs = np.where(conv >= max_val - tol_max)[0]
        g = np.zeros_like(a)
        for k in idxs:
            if k < n:                                   # left half (including centre)
                # reversed slice a[0:k+1][::-1] added to g[0:k+1]
                g[:k+1] += a[:k+1][::-1]
            else:                                        # right half
                start = k - (n - 1)                      # first index receiving contribution
                length = n - start                       # length of overlapping segment
                g[start:] += a[:length][::-1]
        g *= 2.0                                         # factor from derivative of a_i a_j

        # ----- step size (square‑root decay) --------------------------------
        eta = eta0 / np.sqrt(it)

        # ----- momentum update ------------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- back‑tracking line search ------------------------------------
        step = eta
        for _ in range(8):
            cand = _proj_simplex(a - step * momentum)

            # evaluate candidate max convolution
            fc = np.fft.rfft(cand, n=fft_len)
            conv_c = np.fft.irfft(fc * fc, n=fft_len)[:conv_len]
            max_c = conv_c.max()
            if max_c <= max_val:                # improvement (or no worsening)
                a = cand
                break
            step *= 0.5                         # shrink step and retry
        else:
            # no successful step; keep current a unchanged
            pass

        # ----- occasional jitter to escape plateaus -------------------------
        if it % jitter_interval == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            momentum[:] = 0.0
            no_improve = 0

        # ----- restart after long stagnation --------------------------------
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            a = _proj_simplex(a * perturb)
            momentum[:] = 0.0
            no_improve = 0

    # ------------------------------------------------------------------
    # 6. Final short random‑sampling phase (use remaining time)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        jitter = np.exp(rng.normal(0.0, 0.04, size=n))
        cand = _proj_simplex(best * jitter)
        fc = np.fft.rfft(cand, n=fft_len)
        conv_c = np.fft.irfft(fc * fc, n=fft_len)[:conv_len]
        score_c = 2.0 * n * conv_c.max()
        if score_c < best_score:
            best = cand
            best_score = score_c

    # ------------------------------------------------------------------
    # 7. Return as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
