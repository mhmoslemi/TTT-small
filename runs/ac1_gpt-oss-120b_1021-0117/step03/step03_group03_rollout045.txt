    a = np.exp(w)
    a /= a.sum()
    # compute conv
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    max_val = conv.max()
    cur_score = 2 * n * max_val   # same as evaluate if sum=1
    if cur_score < best_score:
        best = a.copy()
        best_score = cur_score
        no_improve = 0
        if best_score <= target_score:
            break
    else:
        no_improve += 1

    # gradient
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]

    g = np.zeros_like(a)
    for k in idxs:
        offset = k - (n - 1)
        if offset >= 0:
            i = np.arange(offset, n, dtype=np.int64)
            partner = k - i
            g[i] += 2.0 * a[partner]
        else:
            i = np.arange(0, n + offset, dtype=np.int64)
            partner = k - i
            g[i] += 2.0 * a[partner]
    g /= max(1.0, len(idxs))  # average

    # step size schedule
    it += 1
    eta = eta0 / np.sqrt(it)   # decaying
    # update log-weights
    w -= eta * g   # descent

    if it % 200 == 0:

import time
import numpy as np

def _active_indices(conv: np.ndarray, top_k: int, rel_tol: float) -> np.ndarray:
    """Return up to `top_k` indices where conv is close to its maximum."""
    max_val = conv.max()
    # indices whose value is within rel_tol of the maximum
    idxs = np.where(conv >= max_val * (1 - rel_tol))[0]
    if idxs.size > top_k:
        # keep only the largest `top_k` values
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    return idxs


def _gradient_from_active(a: np.ndarray, idxs: np.ndarray, n: int) -> np.ndarray:
    """Sub‑gradient of max(conv) given active convolution indices."""
    g = np.zeros_like(a)
    for k in idxs:
        offset = k - (n - 1)
        if offset >= 0:
            i = np.arange(offset, n, dtype=np.int64)
            partner = k - i
            g[i] += 2.0 * a[partner]
        else:
            i = np.arange(0, n + offset, dtype=np.int64)
            partner = k - i
            g[i] += 2.0 * a[partner]
    if idxs.size:
        g /= float(idxs.size)
    return g


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation of the AC1 upper bound.
    Returns a list of non‑negative coefficients.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Load initial sequence (fallback to uniform)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a0 = parent.copy()
    except Exception:
        n = 256
        a0 = np.ones(n, dtype=np.float64)

    # normalise to sum = 1 (objective is scale‑invariant)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s < 1e-12:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a0 = a0 / s

    # logarithmic parametrisation (exponentiated gradient)
    eps = 1e-30
    w = np.log(a0 + eps)          # log‑weights

    # ------------------------------------------------------------------
    # 2.  Pre‑compute FFT size (reuse each iteration)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 3.  Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    eta0 = 2.0                     # base learning rate
    top_k = 50                     # max active peaks
    rel_tol = 1e-3                 # tolerance for "active" peaks
    jitter_every = 250
    jitter_std = 0.015
    max_stale = 3000
    target_score = 1.50274349

    best = None
    best_score = np.inf
    no_improve = 0
    it = 0

    # deadline a few seconds before the hard limit
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # a) reconstruct a from log‑weights and normalise
        # --------------------------------------------------------------
        a = np.exp(w)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
            w = np.log(a + eps)
            a_sum = 1.0
        a /= a_sum

        # --------------------------------------------------------------
        # b) convolution via FFT and current score
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a_sum * a_sum)   # sum == 1 → 2 n max_conv

        # --------------------------------------------------------------
        # c) record best solution
        # --------------------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = float(cur_score)
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # d) gradient of the max‑convolution peak(s)
        # --------------------------------------------------------------
        idxs = _active_indices(conv, top_k, rel_tol)
        grad = _gradient_from_active(a, idxs, n)

        # --------------------------------------------------------------
        # e) learning‑rate schedule & exponentiated‑gradient step
        # --------------------------------------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        w -= eta * grad

        # --------------------------------------------------------------
        # f) occasional random jitter to escape plateaus
        # --------------------------------------------------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, 1000.0)
            a /= a.sum()
            w = np.log(a + eps)

        # --------------------------------------------------------------
        # g) restart if no improvement for a long stretch
        # --------------------------------------------------------------
        if no_improve >= max_stale:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, 1000.0)
            a /= a.sum()
            w = np.log(a + eps)
            no_improve = 0

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    if best is None:
        best = a0
    return [float(v) for v in best]
