def max_shift_gradient(a):
    n = a.shape[0]
    conv = np.fft.irfft(np.fft.rfft(a, n=fft_len)**2, n=fft_len)[:2*n-1]
    s = np.argmax(conv)
    r = s - (n-1)
    # compute gradient
    g = np.zeros_like(a)
    if r >= 0:
        # indices i where i-r >=0
        g[r:] += a[:n-r]
        # indices i where i+r < n
        g[:n-r] += a[r:]
    else:
        r = -r
        g[:n-r] += a[r:]
        g[r:] += a[:n-r]
    # For r==0, this adds a twice
    return g

def gradient_max_shift(a):
    n = a.size

    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    s = np.argmax(conv)
    offset = s - (n-1)
    g = np.zeros_like(a)

        g[:n-offset] += a[offset:]
    else:
        r = -offset
        g[:n-r] += a[r:]
        g[r:] += a[:n-r]
    return g

import time
import numpy as np

# ----------------------------------------------------------------------
#  Simple Euclidean projection onto the probability simplex (sum = 1, a_i >= 0)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    # shift by the mean and clip – this is a fast O(n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # degenerate: return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    Search for a step‑function that minimizes the AC1 upper bound.
    The routine stays within the allocated time budget and returns the
    best sequence discovered (as plain Python floats).
    """

    # uniform simplex point (sum = 1)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    idx = np.arange(n, dtype=np.float64)
    sigma = n / 8.0

    else:

    # pool of initial candidates
    candidates = [uniform, gauss]
    if parent is not None:
        candidates.append(parent)

    # --------------------------------------------------------------
    # 2. Helper: fast evaluation of the AC1 bound for sum‑one vectors
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two

    def fast_score(a: np.ndarray) -> float:
        """2 * n * max(convolution)  (valid when sum(a)=1)."""
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # 3. Initialise the best candidate
    # --------------------------------------------------------------
    best = None
    best_val = float("inf")
    for cand in candidates:
        val = fast_score(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()

    a = best.copy()                     # current iterate
    v = np.zeros_like(a)                # momentum vector
    mu = 0.93                            # momentum decay
    eta0 = 1.0                           # initial step magnitude

    # --------------------------------------------------------------
    # 4. Main optimisation loop (projected sub‑gradient)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

        idx_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        if offset >= 0:
            g[offset:] += a[: n - offset]
            g[: n - offset] += a[offset:]
        else:
            off = -offset
            g[: n - off] += a[off:]
            g[off:] += a[: n - off]

        # ----- momentum update -----

        # ----- step size (diminishing) -----
        eta = eta0 / np.sqrt(it + 1.0)

    # 5. Return best sequence as a plain Python list
    # --------------------------------------------------------------
    return [float(x) for x in best]
