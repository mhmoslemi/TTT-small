import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 bound.
    Returns a plain Python list of coefficients.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ----------------------------------------------------------------------
    # Initialise sequence (parent if available, else uniform)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = np.full(2000, 1.0 / 2000, dtype=np.float64)

    # force feasibility
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # put onto probability simplex

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    # ----------------------------------------------------------------------
    # Adam parameters (mirror‑descent on log‑variables)
    # ----------------------------------------------------------------------
    epsilon = 1e-12
    a = np.maximum(a, epsilon)
    loga = np.log(a)

    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr = 0.12                     # base learning rate for log‑variables
    eps_adam = 1e-8

    best = a.copy()
    # initial score
    conv = _conv_fft(a, L)
    best_sc = 2.0 * n * conv.max() / (a.sum() ** 2)

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) evaluate current candidate
        # --------------------------------------------------------------
        conv = _conv_fft(a, L)
        max_val = conv.max()

        cur_sc = 2.0 * n * max_val / (a.sum() ** 2)

        if cur_sc < best_sc:

            best = a.copy()

        else:

        # early exit if target reached
        if best_sc < 1.50274349:

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        lo = max(0, k - (n - 1))

        g = np.zeros_like(a)
