def _gradient_smooth_max(a, alpha, L=None):
    n = len(a)
    m = 2*n - 1
    # compute convolution a*a as before to get conv vector c
    if L is None: L = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fa * fa, n=L)[:2*n-1]  # length m
    max_c = conv_full.max()
    # compute p as softmax weights
    # to avoid overflow, subtract max_c:
    exp_vals = np.exp(alpha * (conv_full - max_c))
    p = exp_vals / exp_vals.sum()
    # compute gradient: 2 * conv(p_rev, a)
    p_rev = p[::-1]
    fp = np.fft.rfft(p_rev, n=L)
    grad_full = np.fft.irfft(fa * fp, n=L)  # circular conv of a and p_rev (length L)
    # Need linear conv lengths so ensure zero padding: but using same L (>= n+m-1) ensures circular corresponds to linear (since we zero pad at start).
    # However we used same L as earlier, which is >= (n+m-1). So we can safely extract linear part.
    # The linear conv result length up to n+m-1 = 3n-2. Since we used rfft length L, we get that many effective points. We need slice start = m-1 (since p_rev length m).
    start = m - 1
    grad_slice = grad_full[start:start+n]
    g = 2.0 * grad_slice
    return conv_full, max_c, g

def _smooth_adam_optimize(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    # choose L large enough for both conv length (2n-1) and p_rev conv (3n-2).
    L = 1 << ((3*n-2).bit_length())
    m = 2*n - 1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    mmt = np.zeros_like(a)
    vmt = np.zeros_like(a)
    best = a.copy()
    best_score = float('inf')
    t = 0
    # schedule for alpha
    alpha = 1.0
    # initial step size baseline
    base_eta = 0.5  # may adjust
    while time.monotonic() < deadline - 0.1:
        t += 1
        # compute conv and gradient
        fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[:m]
        max_c = conv_full.max()
        score = 2.0 * n * max_c
        # update best
        if score < best_score:
            best_score = score
            best = a.copy()
        # compute softmax weights
        exp_vals = np.exp(alpha * (conv_full - max_c))
        p = exp_vals / exp_vals.sum()
        # gradient
        p_rev = p[::-1]
        fp = np.fft.rfft(p_rev, n=L)
        grad_full = np.irfft(fa * fp, n=L)  # length L
        # extract slice for gradient length n
        start = m - 1  # index offset
        g = 2.0 * grad_full[start:start+n]  # shape n
        # Adam update
        mmt = beta1 * mmt + (1 - beta1) * g
        vmt = beta2 * vmt + (1 - beta2) * (g * g)
        m_hat = mmt / (1 - beta1**t)
        v_hat = vmt / (1 - beta2**t)
        eta = base_eta / (np.sqrt(t) + 1)  # decaying step size
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasionally random perturb
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a_new[i] > 0:
                j = rng.integers(0, n-1)

                delta = a_new[i] * frac
                a_new[i] -= delta
                a_new[j] += delta

        # acceptance
        # compute score for candidate? Might be expensive; we could just accept always as we have adaptive step
        a = a_new

        alpha = min(alpha * 1.02, 30.0)  # up to 30 typical
        # early exit if near deadline
        if time.monotonic() + 0.3 > deadline:

def _generate_initial(n, rng):
   # generate shape param vectors and also a random vector.
   best = None; best_score = float('inf')
   # define param loops
   exponents = [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]
   for typ in ('sine', 'cosine', 'parabola', 'ellipse'):
       for p in exponents:
           if typ == 'sine':
               x = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
           elif typ == 'cosine':
               # center
               center = (n - 1)/2
               # compute offset scaled to [-pi/2, pi/2]
               offset = (np.arange(n) - center) * (np.pi / n)
               x = np.cos(offset)
           elif typ == 'parabola':
               center = (n - 1)/2
               width = (n - 1)/2
               x = 1.0 - ((np.arange(n) - center) / width) ** 2
           elif typ == 'ellipse':
               center = (n - 1)/2
               radius = (n - 1)/2
               x = np.maximum(0.0, 1.0 - ((np.arange(n) - center) / radius) ** 2) ** 0.5
           else:
               continue
           x = np.maximum(x, 0.0)
           x = x ** p
           if np.all(x == 0):
               continue
           x = x / x.sum()
           # evaluate score quickly via conv max (function). Use time-check after each evaluation maybe.
           max_val, _ = _conv_max_and_index(x, 1 << ((2*n-1).bit_length()))
           score = 2.0 * n * max_val
           if score < best_score:
               best_score = score
               best = x.copy()
   # also random vector
   rand_vec = rng.random(n)
   rand_vec = _project_simplex(rand_vec)
   max_val, _ = _conv_max_and_index(rand_vec, 1 << ((2*n-1).bit_length()))
   score = 2.0 * n * max_val
   if score < best_score:
       best_score = score
       best = rand_vec.copy()
   # also use height_sequence_1 if exists.
   if 'height_sequence_1' in globals():
       a0 = np.asarray(height_sequence_1, dtype=np.float64)
       a0 = np.maximum(a0, 0.0)
       a0 = _project_simplex(a0)

       if score < best_score:
           best_score = score
           best = a0.copy()
   return best

def _create_shape(n, shape_type, exponent):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    if shape_type == 'sine':
        val = np.sin(np.pi * (idx + 1) / (n + 1))
    elif shape_type == 'cosine':
        # offset range [-pi/2, pi/2]
        val = np.cos(np.pi * (idx - center) / n)
    elif shape_type == 'parabola':
        width = (n - 1) / 2.0
        val = 1.0 - ((idx - center) / width) ** 2
    elif shape_type == 'ellipse':
        width = (n - 1) / 2.0
        val = np.sqrt(np.maximum(0.0, 1.0 - ((idx - center) / width) ** 2))
    else:
        val = np.ones(n)
    val = np.maximum(val, 0.0)
    val = val ** exponent
    return val

def _initial_candidate(n, rng):

def _smooth_adam_optimize(a0, deadline, rng):
    a = a0.copy()
    n = a.size
    m = 2*n - 1
    L = 1 << ((3*n - 2).bit_length())
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    mmt = np.zeros_like(a)
    vmt = np.zeros_like(a)
    best = a.copy()
    # compute initial score
    # compute conv_max of a (just using function)
    max_val, _ = _conv_max_and_index(a, 1 << ((2*n-1).bit_length()))
    best_score = 2.0 * n * max_val
    t = 0
    alpha = 0.5

    # main loop
    while time.monotonic() < deadline - 0.1:
        t += 1
        # compute FFT of a
        fa = np.fft.rfft(a, n=L)
        # compute convolution a*a (full) using same L but we need only first m entries
        conv_full = np.fft.irfft(fa * fa, n=L)[:m]
        max_c = conv_full.max()
        cur_score = 2.0 * n * max_c

        # compute softmax probabilities
        exp_vals = np.exp(alpha * (conv_full - max_c))
        p = exp_vals / exp_vals.sum()

        # compute correlation gradient
        fp = np.fft.rfft(p, n=L)
        corr_full = np.fft.irfft(fa * np.conj(fp), n=L)[: n + m - 1]  # linear correlation length
        g = 2.0 * corr_full[n-1 : n-1 + n]  # gradient length n

        # Adam update
        mmt = beta1 * mmt + (1.0 - beta1) * g
        vmt = beta2 * vmt + (1.0 - beta2) * (g * g)
        m_hat = mmt / (1.0 - beta1 ** t)
        v_hat = vmt / (1.0 - beta2 ** t)
        eta = 0.5 / (np.sqrt(t) + 1.0)  # decaying step size
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)

        # occasionally random mass transfer perturbation
        if rng.random() < 0.01:
            i = rng.integers(0, n)
            if a_new[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a_new[i] * frac
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_simplex(a_new)

        # accept new point (always accept)
        a = a_new

import time
import math
import numpy as np

# --------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:                # fallback – uniform distribution
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# --------------------------------------------------------------
# Convolution utilities (FFT, max value, index)
# --------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    """Return (max(conv), argmax) for the linear convolution vec*vec."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# --------------------------------------------------------------
# Shape‑based initialisation
# --------------------------------------------------------------
def _shape_vector(n: int, kind: str, exp: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0                     # centre
    if kind == "sine":
        base = np.sin(np.pi * (i + 1) / (n + 1))
    elif kind == "cosine":
        base = np.cos(np.pi * (i - c) / n)          # positive in the middle
    elif kind == "parabola":
        w = c
        base = 1.0 - ((i - c) / w) ** 2
    elif kind == "ellipse":
        w = c
        base = np.sqrt(np.maximum(0.0, 1.0 - ((i - c) / w) ** 2))
    else:
        base = np.ones_like(i)
    base = np.maximum(base, 0.0) ** exp
    return base


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a promising start vector (L1‑normalised)."""
    L = 1 << ((2 * n - 1).bit_length())
    best_vec = None
    best_score = np.inf

    exps = (0.5, 1.0, 1.5, 2.0, 3.0, 4.0)
    for kind in ("sine", "cosine", "parabola", "ellipse"):
        for e in exps:
            a = _shape_vector(n, kind, e)
            if a.sum() == 0.0:
                continue
            a = _project_simplex(a)
            mx, _ = _conv_max_and_index(a, L)
            score = 2.0 * n * mx
            if score < best_score:
                best_score = score
                best_vec = a.copy()

    # random start
    rand = rng.random(n)
    rand = _project_simplex(rand)
    mx, _ = _conv_max_and_index(rand, L)
    s = 2.0 * n * mx
    if s < best_score:
        best_score = s
        best_vec = rand.copy()

    # parent construction if available
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        parent = _project_simplex(parent)
        mx, _ = _conv_max_and_index(parent, L)
        s = 2.0 * n * mx
        if s < best_score:
            best_score = s
            best_vec = parent.copy()

    return best_vec


# --------------------------------------------------------------
# Smooth‑max Adam optimiser (gradient on a soft‑max of the convolution)
# --------------------------------------------------------------
def _smooth_adam_optimize(start_vec: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    m = 2 * n - 1                      # length of self‑convolution
    L = 1 << ((3 * n - 2).bit_length())   # size for FFT (covers n+m‑1)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    mmt = np.zeros_like(a)
    vmt = np.zeros_like(a)

    best = a.copy()
    mx, _ = _conv_max_and_index(a, 1 << ((2 * n - 1).bit_length()))
    best_score = 2.0 * n * mx

    t = 0
    alpha = 0.5                        # smoothing parameter

    while time.monotonic() < deadline - 0.1:
        t += 1

        # ---- convolution of a with itself (full) ----
        fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[:m]
        max_c = conv_full.max()
        cur_score = 2.0 * n * max_c

        # keep best seen so far
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # ---- soft‑max over convolution entries ----
        exp_vals = np.exp(alpha * (conv_full - max_c))   # stable
        p = exp_vals / exp_vals.sum()                   # probabilities, length m

        # ---- gradient of soft‑max (correlation a ⋆ p) ----
        fp = np.fft.rfft(p, n=L)
        corr_full = np.fft.irfft(fa * np.conj(fp), n=L)[: n + m - 1]
        grad = 2.0 * corr_full[n - 1: n - 1 + n]        # length n

        # ---- Adam update ----
        mmt = beta1 * mmt + (1.0 - beta1) * grad
        vmt = beta2 * vmt + (1.0 - beta2) * (grad * grad)
        m_hat = mmt / (1.0 - beta1 ** t)
        v_hat = vmt / (1.0 - beta2 ** t)
        step = 0.5 / (np.sqrt(t) + 1.0)                # decaying magnitude
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)

        # ---- occasional random mass transfer (exploration) ----
        if rng.random() < 0.01:
            i = rng.integers(0, n)
            if a_new[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                fraction = rng.random() * 0.4
                delta = a_new[i] * fraction
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_simplex(a_new)

        a = a_new

        # increase smoothing (approach true max)
        alpha = min(alpha * 1.02, 30.0)

        # early‑exit guard
        if time.monotonic() + 0.3 > deadline:
            break

    # final check – one extra evaluation of the last point
    mx, _ = _conv_max_and_index(a, 1 << ((2 * n - 1).bit_length()))
    final_score = 2.0 * n * mx
    if final_score < best_score:
        best_score = final_score
        best = a.copy()

    return best, best_score


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    n = len(height_sequence_1) if "height_sequence_1" in globals() else 256
    start = _initial_candidate(n, rng)

    # primary optimisation
    best_vec, best_val = _smooth_adam_optimize(start, deadline, rng)

    # safety fallback – uniform distribution if something went wrong
    if best_vec is None or len(best_vec) == 0:
        best_vec = np.ones(n, dtype=np.float64) / n

    # return plain Python list
    return [float(x) for x in best_vec]
