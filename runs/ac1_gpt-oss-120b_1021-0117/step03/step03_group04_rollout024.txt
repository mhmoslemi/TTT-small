import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if v.ndim != 1:
        raise ValueError
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    # a few Gaussian blobs of varying width
    gauss_list = []
    for sigma_frac in (0.07, 0.12, 0.20, 0.35):
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((idx - (n - 1) / 2) / sigma) ** 2)
        if g.sum() > 0:
            gauss_list.append(g / g.sum())

    # random Dirichlet‑like vector
    rand = rng.random(n)
    if rand.sum() > 0:
        rand = rand / rand.sum()
    else:
        rand = uniform.copy()

    candidates = [uniform, tent, sine, rc, rand] + gauss_list
    if parent_ok:
        p = np.clip(parent, 0.0, 1000.0)
        s = p.sum()
        if s >= 0.01:
            candidates.append(p / s)

    # ------------------------------------------------------------------
    # 4. Pick the best starting point
    # ------------------------------------------------------------------
    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = _fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    # ------------------------------------------------------------------
    # 5. Main search loop – projected momentum gradient descent
    # ------------------------------------------------------------------
    v = np.zeros_like(a)        # momentum vector
    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # current convolution & score
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = 2.0 * n * conv.max()

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = it

        # ---------- sub‑gradient from maximal autocorrelation peaks ----------
        max_val = conv.max()
        # consider all indices whose value is within 1e-12 relative of the max
        close = conv >= max_val * (1.0 - 1e-12)
        peaks = np.flatnonzero(close)

        g = np.zeros_like(a)
        for shift in peaks:
            offset = shift - (n - 1)
            if offset >= 0:
                i = np.arange(offset, n, dtype=np.int64)
                j = shift - i
            else:
                i = np.arange(0, n + offset, dtype=np.int64)
                j = shift - i
            g[i] += 2.0 * a[j]
        if peaks.size:
            g /= peaks.size

        # ---------- momentum update ----------
        beta = 0.9
        v = beta * v + (1.0 - beta) * g

        # ---------- step size (diminishing) ----------
        eta = 0.5 / np.sqrt(it)

        # ---------- projected update ----------
        a = _proj_simplex(a - eta * v)

        # ---------- occasional stochastic jitter ----------
        if it % 50 == 0:
            jitter = np.exp(rng.normal(0.0, 0.012, size=n))
            a = _proj_simplex(a * jitter)

        # ---------- restart if stagnated ----------
        if it - last_improve > 2500:
            # re‑initialize around the current best with moderate noise
            noise = np.exp(rng.normal(0.0, 0.18, size=n))
            a = _proj_simplex(best * noise)
            v.fill(0.0)
            last_improve = it

    # ------------------------------------------------------------------
    # 6. Return the best candidate found (as plain Python floats)
    # ------------------------------------------------------------------
    if best is None:
        best = uniform.copy()
    return [float(x) for x in best]
