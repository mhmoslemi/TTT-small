import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    if v.ndim != 1:
        v = v.ravel()
    n = v.shape[0]
    # shift by median to improve stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        # ----- convolution -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- keep best -----
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ----- sub‑gradient of max‑convolution -----
        # mask of near‑max indices (relative tolerance 0.1%)
        tol = max(max_val * 0.001, 1e-12)
        M = (conv >= max_val - tol).astype(np.float64)

        # gradient = 2 * (M * a_rev) convolved
        a_rev = a[::-1]
        fM = np.fft.rfft(M, n=fft_len)
        f_rev = np.fft.rfft(a_rev, n=fft_len)
        tmp = np.fft.irfft(fM * f_rev, n=fft_len)
        start = n - 1
        g = 2.0 * tmp[start:start + n]

        # ----- momentum step & projection -----
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + g
        a = _proj_simplex(a - eta * v)

        # ----- occasional jitter to escape plateaus -----
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v[:] = 0.0
            no_improve = 0

        # ----- restart if stagnating -----
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v[:] = 0.0
            no_improve = 0

    # ------------------------------------------------------------------
    # Return best sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(x) for x in best]
