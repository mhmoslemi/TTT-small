import time
import math
import numpy as np
from scipy.signal import correlate

# ------------------- utilities -------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (real input)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ------------------- candidate families -------------------
def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_hann(n: int) -> np.ndarray:
    if n == 1:
        a = np.ones(1)
    else:
        a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    return a / a.sum()


def _make_hamming(n: int) -> np.ndarray:
    if n == 1:
        a = np.ones(1)
    else:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    return a / a.sum()


def _make_blackman(n: int) -> np.ndarray:
    if n == 1:
        a = np.ones(1)
    else:
        a = (
            0.42
            - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
            + 0.08 * np.cos(4 * np.pi * np.arange(n) / (n - 1))
        )
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _make_sine(n: int) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1)))
    return a / a.sum()


def _make_cos2(n: int) -> np.ndarray:
    a = np.cos(np.pi * (np.arange(n, dtype=np.float64) - (n - 1) / 2) / n) ** 2
    a = np.maximum(a, 0.0)
    return a / a.sum()


# ------------------- main search -------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialization -----
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # ----- explore analytic families -----
    families = []

    families.append(_make_linear(n))
    families.append(_make_quadratic(n))
    families.append(_make_hann(n))
    families.append(_make_hamming(n))
    families.append(_make_blackman(n))
    families.append(_make_sine(n))
    families.append(_make_cos2(n))

    for r in np.linspace(0.90, 0.9999, 30):
        families.append(_make_exponential(n, r))
    for p in np.linspace(0.2, 4.0, 30):
        families.append(_make_powerlaw(n, p))

    for cand in families:
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    target = 1.50274349

    # ----- softmax‑gradient descent (phase A) -----
    it = 0
    step0 = 0.5
    alpha = 5.0
    while time.monotonic() < deadline:
        it += 1
        conv = _conv_fft(cur)
        max_conv = conv.max()
        cur_sc = 2.0 * n * max_conv / (cur.sum() ** 2)
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            if best_sc < target:
                break

        # soft‑max weights on the convolution values
        diffs = conv - max_conv                     # ≤0
        w = np.exp(alpha * diffs)
        w_sum = w.sum()
        if w_sum == 0.0:  # numerical safety
            w = np.zeros_like(w)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # gradient = 2 * correlate(w, cur)  (valid mode -> length n)
        g = 2.0 * correlate(w, cur, mode="valid", method="fft")

        eta = step0 / math.sqrt(it)
        cand = cur - eta * g
        cand = _project_simplex(cand)

        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break
        else:
            # occasional uphill acceptance
            T = 1e-4 * cur_sc
            if rng.random() < math.exp(-(cand_sc - cur_sc) / T):
                cur, cur_sc = cand, cand_sc

        # schedule updates
        alpha = min(alpha * 1.015, 5000.0)

        # small jitter every few hundred steps
        if it % 400 == 0:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

    # ----- targeted random mass‑transfer moves (phase B) -----
    # keep current convolution for cheap max queries
    conv = _conv_fft(cur)
    cur_sc = _score(cur)

    while time.monotonic() < deadline:
        # most violating lag
        k = int(np.argmax(conv))

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = cur[idx] * cur[k - idx]
        i = int(idx[np.argmax(prod)]) if prod.size else int(rng.integers(0, n))
        j = int(np.argmin(cur))
        if j == i:
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1

        max_delta = cur[i] * 0.5
        if max_delta < 1e-12:
            continue
        delta = max_delta * rng.random()

        # create candidate by moving delta from i to j
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, conv, cur_sc = cand, _conv_fft(cand), cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break
        else:
            # occasional tiny perturbations
            if rng.random() < 0.001:
                jitter = rng.normal(scale=2e-5, size=n)
                tmp = cur + jitter
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, conv, cur_sc = tmp, _conv_fft(tmp), tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                        if best_sc < target:
                            break

    # final fallback safety
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
