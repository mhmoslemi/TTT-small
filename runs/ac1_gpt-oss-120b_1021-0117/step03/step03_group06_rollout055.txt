def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution x*x using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]

    """Score = 2*n*max(conv)/sum_a**2 (identical to evaluate_sequence)."""
    return 2.0 * n * conv.max() / (sum_a * sum_a)
