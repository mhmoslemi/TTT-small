import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        a = a / total                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # 3. Best candidate tracking
    # ------------------------------------------------------------------
    best = a.copy()
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    max_val = conv.max()
    best_score = float(2.0 * n * max_val)   # sum = 1 ⇒ denominator = 1

    # ------------------------------------------------------------------
    # 4. Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    eta0 = 1.0               # initial step magnitude
    beta = 0.9               # momentum decay
    top_k = 20               # number of convolution peaks used for gradient
    jitter_every = 120       # steps between multiplicative jitter
    jitter_std = 0.015       # std‑dev of log‑normal jitter
    target_score = 1.50274349
    v = np.zeros_like(a)     # momentum buffer
    it = 0
    stale = 0
    max_stale = 2500

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution (FFT) -----------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)   # sum(a) == 1 by construction

        # ----- keep the best sequence ------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
            if best_score <= target_score:
                break
        else:
            stale += 1

        # ----- gradient from the most significant convolution peaks -----
        if conv.size > top_k:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size, dtype=np.int64)

        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                i = np.arange(offset, n, dtype=np.int64)
                j = i - offset
                g[i] += a[j]
                g[j] += a[i]
            else:
                shift = -offset
                i = np.arange(0, n - shift, dtype=np.int64)
                j = i + shift
                g[i] += a[j]
                g[j] += a[i]

        # average over the selected peaks to keep the step size stable
        g = g / max(1, idxs.size)

        # ----- momentum and descent step ---------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = a - eta * v

        # ----- projection onto the simplex (sum=1, non‑negative) -------
        a = _proj_simplex(a)

        # ----- occasional jitter to escape plateaus --------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v.fill(0.0)
            stale = 0

        # ----- restart if no improvement for a long time ---------------
        if stale >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.01, size=n))
            a = _proj_simplex(a * perturb)
            v.fill(0.0)
            stale = 0

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
