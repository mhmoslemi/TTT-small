    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # initial evaluation
    max_val, conv = _conv_full_max(a, L)

    # parameters
    step0 = 0.5
    min_step = 1e-7
    decay = 0.998
    random_move_prob = 0.025
    no_improve = 0
    max_no_improve = 3000

    it = 0
    while time.monotonic() < deadline:
        # compute conv and max
        max_val, conv = _conv_full_max(a, L)
        cur_score = 2.0 * n * max_val

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # subgradient at max indices
        eps = 1e-12
        max_idx = np.where(conv >= max_val - eps)[0]
        g = np.zeros_like(a)
        for m in max_idx:
            lo = max(0, m - (n - 1))
            hi = min(n - 1, m)
            if lo <= hi:
                idx_range = np.arange(lo, hi + 1)
                g[idx_range] += a[m - idx_range]
        if max_idx.size > 0:
            g = 2.0 * g / max_idx.size

        # step size
        step = max(min_step, step0 / np.sqrt(it + 1))

        # gradient step
        cand = a - step * g
        cand = _project_simplex(cand)

        # evaluate candidate
        max_val_c, _ = _conv_full_max(cand, L)
        cand_score = 2.0 * n * max_val_c

        if cand_score < cur_score - 1e-15:
            a = cand
            cur_score = cand_score
            # keep best may be updated at start of loop
        else:
            # occasional uphill acceptance
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):

                cur_score = cand_score

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Full linear convolution via FFT (returns max and whole array)
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    """Return (max(conv), full convolution) of a with itself."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improve the
    AC1 upper bound.  The routine runs a projected sub‑gradient descent
    on the quantity 2·n·max(conv) with a simplex constraint.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin before the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ----- initialise -------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: a short uniform vector (will be resized later)
        a = np.ones(1, dtype=np.float64)

    # project onto simplex (sum = 1, non‑negative)
    a = _project_simplex(a / (a.sum() if a.sum() != 0 else 1.0))

    n = a.size
    # power‑of‑two length for FFT, stays constant
    L = 1 << ((2 * n - 1).bit_length())

    # initial evaluation
    max_val, _ = _conv_full_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # ----- optimisation parameters ------------------------------------
    step0 = 0.5                # initial step size scale
    min_step = 1e-7
    random_move_prob = 0.025   # occasional mass‑transfer perturbation
    max_no_improve = 3000      # restart after this many stalls
    no_improve = 0
    it = 0

    # ----- main optimisation loop ------------------------------------
    while time.monotonic() < deadline:
        # current convolution and score
        max_val, conv = _conv_full_max(a, L)
        cur_score = 2.0 * n * max_val

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- sub‑gradient of the max‑convolution --------------------
        eps = 1e-12
        max_idx = np.where(conv >= max_val - eps)[0]
        g = np.zeros_like(a)
        for m in max_idx:
            lo = max(0, m - (n - 1))
            hi = min(n - 1, m)
            if lo <= hi:
                idx_range = np.arange(lo, hi + 1)
                g[idx_range] += a[m - idx_range]
        if max_idx.size:
            g = 2.0 * g / max_idx.size  # average over all active lags

        # ---- step size (diminishing) -------------------------------
        step = max(min_step, step0 / np.sqrt(it + 1))

        # ---- gradient step + projection -----------------------------
        cand = a - step * g
        cand = _project_simplex(cand)

        # evaluate candidate
        max_val_c, _ = _conv_full_max(cand, L)
        cand_score = 2.0 * n * max_val_c

        # accept if improvement; occasionally accept uphill moves
        if cand_score < cur_score - 1e-15:
            a = cand
            cur_score = cand_score
        else:
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score

        # ---- random mass‑transfer perturbation ----------------------
        if rng.random() < random_move_prob:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max_val2, _ = _conv_full_max(cand2, L)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:

                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()

        # ---- restart if we are stuck --------------------------------
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = _project_simplex(a)
            no_improve = 0

        it += 1

        # safety break if only a short time remains
        if time.monotonic() + 0.5 > deadline:
            break

    # return the best solution found as a plain Python list
    return [float(x) for x in best]
