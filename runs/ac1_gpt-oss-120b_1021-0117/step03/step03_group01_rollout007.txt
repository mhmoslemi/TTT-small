def _conv_max_and_index(vec, L):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*len(vec)-1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx
```

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:  # safety
        w[:] = 1.0 / n
    return w


# ----------------------------------------------------------------------
# Convolution via FFT – return max value and its index
# ----------------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    """Return (max convolution value, index) for vec * vec."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Simple analytic sqrt series (fallback initializer)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# contributes to the current worst autocorrelation toward a small one.

    # largest entry inside the active region
    region = a[lo : hi + 1]

    i = lo + i_rel
    # smallest entry overall (prefer a different index)
    j = int(np.argmin(a))
    if j == i:
        # second smallest

    # transfer a modest fraction
    delta = min(a[i] * 0.35, 0.1)
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    return _project_simplex(cand)
