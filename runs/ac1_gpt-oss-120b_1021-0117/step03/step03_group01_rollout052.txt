def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine length n
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 256
        a0 = _sqrt_series(n)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())

    # initial best
    cur = a0.copy()
    cur_max, cur_idx = _conv_max_and_index(cur, L)
    cur_score = 2.0 * n * cur_max
    best = cur.copy()
    best_score = cur_score

    # momentum and step
    momentum = np.zeros_like(cur)
    step = 0.5
    min_step = 1e-7
    decay = 0.9999
    prob_mass = 0.04
    prob_smooth = 0.015
    prob_swap = 0.01
    prob_noise = 0.015
    beta = 0.9

    while time.monotonic() < deadline:
        # Choose operation
        r = rng.random()
        cand = None
        if r < 0.45:  # gradient descent
            lo = max(0, cur_idx - (n - 1))
            hi = min(n - 1, cur_idx)
            g = np.zeros_like(cur)
            if lo <= hi:
                j = np.arange(lo, hi + 1)
                g[j] = 2.0 * cur[cur_idx - j]
            momentum = beta * momentum + (1.0 - beta) * g
            cand = cur - step * momentum
            cand = _project_simplex(cand)
        elif r < 0.45 + prob_mass:  # random mass transfer
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
        elif r < 0.45 + prob_mass + prob_smooth:  # smoothing
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(cur, kernel, mode='same')
            cand = _project_simplex(smooth)
        elif r < 0.45 + prob_mass + prob_smooth + prob_swap:  # swap
            i, j = rng.integers(0, n, size=2)
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.45 + prob_mass + prob_smooth + prob_swap + prob_noise:  # add Gaussian noise
            sigma = step * 0.2
            cand = cur + rng.normal(scale=sigma, size=n)
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
        else:
            # occasionally restart from best with a small perturbation
            if rng.random() < 0.001:
                cand = best + rng.normal(scale=0.01, size=n)
                cand = np.maximum(cand, 0.0)
                cand = _project_simplex(cand)

        if cand is None:
            continue

        cand_max, cand_idx = _conv_max_and_index(cand, L)
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            cur_idx = cand_idx
            # update momentum to keep direction? keep as is; maybe reset momentum
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # uphill acceptance
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                cur = cand
                cur_score = cand_score
                cur_idx = cand_idx

        # decay step size
        step = max(min_step, step * decay)

        # early exit guard
        if time.monotonic() + 0.5 > deadline:
            break

    return [float(x) for x in best]

import time
import math
import numpy as np

# --------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    pos = np.where(rho_arr > 0)[0]
    if pos.size:
        rho = pos[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:          # safeguard
        w[:] = 1.0 / n
    return w

# --------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1+…+x^{2n‑2})
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# --------------------------------------------------------------
# Convolution via FFT – return maximal entry and its index
# --------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())

    cur = a0.copy()
    cur_max, cur_idx = _conv_max_and_index(cur, L)
    cur_score = 2.0 * n * cur_max

    best = cur.copy()
    best_score = cur_score

    momentum = np.zeros_like(cur)
    step = 0.5

    # operation probabilities
    prob_grad = 0.45
    prob_mass = 0.04
    prob_smooth = 0.015
    prob_swap = 0.01
    prob_noise = 0.015
    prob_restart = 0.001

    # ----------------------------------------------------------

    while time.monotonic() < deadline:

        cand = None

        # ------------------------------------------------------

        if r < prob_grad:
