import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin

    # ------------------------------------------------------------------
    # initialise a
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: simple Gaussian window of length 2000
        n_fallback = 2000
        x = np.arange(n_fallback, dtype=np.float64)
        sigma = n_fallback / 6.0

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # normalise to sum=1

    n = a0.size
    # FFT size (power of two) sufficient for convolutions up to length 3n-2
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)    # slice selector for gradient

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    cur = a0.copy()

    beta2 = 0.999

        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----

        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        g = 2.0 * grad_full[idx]      # shape (n,)

        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # small safety margin

    # ------------------------------------------------------------------
    # initialise a (use the parent if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a Gaussian window of moderate length
        n_fallback = 2000
        x = np.arange(n_fallback, dtype=np.float64)
        sigma = n_fallback / 6.0
        a0 = np.exp(-0.5 * ((x - n_fallback / 2) / sigma) ** 2)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # normalise to sum = 1

    n = a0.size
    # FFT length – power of two, large enough for convolutions up to length 3n‑2
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)    # slice indices needed for gradient

    # --------------------------------------------------------------
    # Helper: returns (score, full convolution, rfft of the vector)
    # --------------------------------------------------------------
    def _score_and_fft(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv.max()
        return float(2.0 * n * max_val), conv, af

    # --------------------------------------------------------------

    cur = a0.copy()

    v = np.zeros_like(cur)
    beta1 = 0.9

    lr = 0.25                         # base learning rate for Adam

    # smoothing for the soft‑max approximation of max(conv)

    eps_decay = 0.9997

        max_val = conv.max()

        if w_sum == 0.0:

        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
