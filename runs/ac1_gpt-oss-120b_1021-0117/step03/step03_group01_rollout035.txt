import time
import math
import numpy as np

# ---------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = 1)
# ---------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:      # degenerate case, fall back to uniform
        w[:] = 1.0 / n
    return w

# ---------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1+…+x^{2n‑2})
# ---------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ---------------------------------------------------------------
# FFT based maximal entry of a*a (full convolution)
# ---------------------------------------------------------------
def _conv_max(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(conv.max())

# ---------------------------------------------------------------
# Full convolution (value + index) – needed for gradient step
# ---------------------------------------------------------------
def _conv_full(vec: np.ndarray) -> np.ndarray:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

# ---------------------------------------------------------------
# Score for a vector (the exact evaluator’s formula, but cheap)
# ---------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    s = vec.sum()
    if s < 1e-12:
        return float('inf')
    maxc = _conv_max(vec)
    n = vec.size
    return 2.0 * n * maxc / (s * s)

# ---------------------------------------------------------------
# Very light local refinement (projected gradient on the current max)
# ---------------------------------------------------------------
def _refine(start: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    a = _project_simplex(start.copy())
    n = a.size
    momentum = np.zeros_like(a)
    step = 0.5
    decay = 0.9995
    min_step = 1e-7
    beta = 0.9

    while time.monotonic() < deadline:
        conv = _conv_full(a)
        idx = int(conv.argmax())
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        g = np.zeros_like(a)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            g[js] = 2.0 * a[idx - js]
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # occasional re‑ordering – descending often helps
        if rng.random() < 0.1:
            cand = np.sort(cand)[::-1]

        if _score(cand) < _score(a):
            a = cand
        step = max(min_step, step * decay)

        if time.monotonic() + 0.5 > deadline:
            break
    return a

# ---------------------------------------------------------------

    except Exception:

    if best_vec is None:
