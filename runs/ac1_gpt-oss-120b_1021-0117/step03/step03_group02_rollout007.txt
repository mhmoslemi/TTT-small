# Simplex projection onto {x >= 0, sum(x) = 1}

def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / n
    return w

                cand2_score = 2.0 * n * max2
                if cand2_score < cur_score:
                    cand = cand2
                    cur_score = cand2_score
                    max_val = max2

        # ----- accept the new candidate --------------------------------------
        if not np.array_equal(cand, a):
            a = cand

        # ----- update global best -------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- stagnation handling -----------------------------------------
        if no_improve >= no_improve_limit:
            # perturb the best solution slightly and continue
            a = best.copy()
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            if a.sum() == 0.0:
                a = rng.random(n)
            a = _project_simplex(a)
            cur_score = 2.0 * n * _conv_max(a, L)
            no_improve = 0
            step = max(min_step, step * 0.5)

        # ----- safety break -------------------------------------------------
        if time.monotonic() + 0.5 > deadline:
            break

    # ----- final tiny refinement (optional, cheap) -------------------------
    # one last pass with a very small step to polish the result
    for _ in range(50):
        if time.monotonic() + 0.1 > deadline:
            break
        conv_full = _conv_full(a, L)
        max_val = float(conv_full.max())
        g = _gradient_from_top_lags(a, conv_full, max_val, K, alpha)
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        cand = _project_simplex(a - 0.001 * (g / norm_g))
        cand_score = 2.0 * n * _conv_max(cand, L)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
        else:
            break

    return [float(x) for x in best]
