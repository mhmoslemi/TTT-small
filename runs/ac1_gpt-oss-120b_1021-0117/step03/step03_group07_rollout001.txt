import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = css[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using FFT (real input)."""
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]

def _score(a: np.ndarray, L: int) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = a.size
    conv = _conv_fft(a, L)
    maxb = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * maxb / (s * s))

def _make_exponential(n: int, r: float) -> np.ndarray:
    """Geometric decreasing distribution, normalised to sum = 1."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    return a

def _make_powerlaw(n: int, p: float) -> np.ndarray:
    """Power‑law tail, normalised."""
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    return a
