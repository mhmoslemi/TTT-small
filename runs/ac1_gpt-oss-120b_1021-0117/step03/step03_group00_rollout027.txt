import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Smoothed max gradient (log‑sum‑exp) for the convolution objective.
# Returns the maximal convolution value and the gradient.
# ------------------------------------------------------------
def _smoothed_grad(a: np.ndarray, eps: float, L: int):
    """Compute max(conv(a,a)) and the gradient of the smoothed max."""
    n = a.size
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()

    # soft‑max weights (stable)
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (a * reverse(w)) restricted to n entries
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)          # descending slice
    grad = 2.0 * grad_full[idx]
    return max_val, grad


# ------------------------------------------------------------
# Full convolution‑based score (sum is assumed 1, i.e. 2*n*max)
# ------------------------------------------------------------
def _score_fft(a: np.ndarray, L: int, n: int) -> float:
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()

# ------------------------------------------------------------

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ---------- initial vectors ----------
    init_vecs = []

    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        s = parent.sum()
        if s > 0.0:

        else:

        n = parent.size
        init_vecs.append(parent)
    else:
        # fallback length (unlikely to happen)
        n = 1000
        parent = np.full(n, 1.0 / n)
        init_vecs.append(parent)

    # a couple of random simplexes
    for _ in range(2):

        v /= v.sum()
        init_vecs.append(v)

    # FFT length sufficient for all convolutions
    L = 1 << ((3 * n - 2).bit_length())

    # ---------- hyper‑parameters ----------
    eps0 = 5.0                 # initial smoothing

    step0 = 0.8                # base step magnitude
