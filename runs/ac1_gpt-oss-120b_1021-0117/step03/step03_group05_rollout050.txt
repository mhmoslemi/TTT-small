def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5) # -0.5 to have spare

    # Initialize
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple random init of length 2000? we need some length predetermined? may set n=2000?
        n = 2000
        a = rng.random(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    # FFT length: ensure >= 3n-2 for correlation
    L = 1 << ((3*n - 2).bit_length())

    # Helpers
    def proj_simplex(v):
        # Euclidean projection onto simplex sum=1, v>=0
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    best = a.copy()
    # compute initial score
    Fa = np.fft.rfft(best, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[:2*n-1]
    best_score = float(2 * n * conv.max())

    cur = best.copy()
    cur_score = best_score

    # iteration params
    it = 0
    # step schedule constants
    add_alpha0 = 0.6
    mul_eta0 = 0.6
    beta0 = 8.0   # smoothing parameter start
    beta_inc = 0.001 # incremental increase per iter
    max_no_improve = 5000
    no_improve = 0

    target = 1.50274349
    while time.monotonic() < deadline:
        it += 1
        # compute convolution and gradient
        Fa = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(Fa * Fa, n=L)[:2*n-1]
        M = conv_full.max()

        # smoothing parameter
        beta = beta0 + beta_inc * it

        # compute softmax weights (stable)
        # subtract max from conv (M) to avoid overflow
        exp_arg = beta * (conv_full - M)  # <=0
        # clip exp_arg to avoid underflow? not needed
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to using max index set
            active = np.flatnonzero(conv_full == M)
            w = np.zeros_like(conv_full)
            w[active] = 1.0 / active.size
            w_sum = 1.0
        w = w / w_sum

        # compute gradient via correlation (a ⋆ w)
        Fw = np.fft.rfft(w, n=L)
        # correlation using conj(Fa)*Fw
        corr_full = np.fft.irfft(np.conj(Fa) * Fw, n=L)[:3*n-2]  # length >= n+len(w)-1
        # gradient entries correspond to lag 0..n-1
        grad = 2.0 * corr_full[:n]

        # additive update
        alpha = add_alpha0 / np.sqrt(it)
        cand_a = cur - alpha * grad
        cand_a = proj_simplex(cand_a)
        # compute its score
        Fa2 = np.fft.rfft(cand_a, n=L)
        conv2 = np.fft.irfft(Fa2 * Fa2, n=L)[:2*n-1]
        cand_sc = float(2 * n * conv2.max())

        # multiplicative update
        eta = mul_eta0 / np.sqrt(it)
        exp_factor = np.exp(-eta * grad)
        cand_m = cur * exp_factor
        s = cand_m.sum()
        if s == 0.0:
            cand_m = np.full_like(cur, 1.0 / n)
        else:
            cand_m = cand_m / s
        # evaluate
        Fa3 = np.fft.rfft(cand_m, n=L)
        conv3 = np.fft.irfft(Fa3 * Fa3, n=L)[:2*n-1]
        cand_m_sc = float(2 * n * conv3.max())

        # choose best among cur, cand_a, cand_m
        if cand_a_sc < cur_score and cand_a_sc <= cand_m_sc:
            cur, cur_score = cand_a, cand_a_sc
        elif cand_m_sc < cur_score:
            cur, cur_score = cand_m, cand_m_sc
        else:
            # maybe occasional random acceptance if improvement small (like simulated annealing)
            # but we skip for simplicity
            pass

        # update best
        if cur_score < best_score:
            best, best_score = cur.copy(), cur_score
            no_improve = 0
        else:
            no_improve += 1

        # random mass transfer move if stagnation
        if no_improve > 3000:
            i = rng.integers(n)
            if cur[i] > 0.0:
                j = rng.integers(n-1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = proj_simplex(tmp)
                # evaluate
                Fa_tmp = np.fft.rfft(tmp, n=L)
                conv_tmp = np.fft.irfft(Fa_tmp * Fa_tmp, n=L)[:2*n-1]
                tmp_sc = float(2 * n * conv_tmp.max())
                if tmp_sc < cur_score:
                    cur, cur_score = tmp, tmp_sc
                    if tmp_sc < best_score:
                        best, best_score = tmp.copy(), tmp_sc
                        no_improve = 0

        # jittered restart toward best solution
        if it % 5000 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = proj_simplex(tmp)
            # evaluate
            Fa_tmp = np.fft.rfft(tmp, n=L)
            conv_tmp = np.fft.irfft(Fa_tmp * Fa_tmp, n=L)[:2*n-1]
            tmp_sc = float(2 * n * conv_tmp.max())
            if tmp_sc < cur_score:
                cur, cur_score = tmp, tmp_sc
                if tmp_sc < best_score:
                    best, best_score = tmp.copy(), tmp_sc
                    no_improve = 0

        # early exit if target achieved
        if best_score < target:
            break

    # final local refinement (random transfers) for remaining time
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = proj_simplex(tmp)
        Fa_tmp = np.fft.rfft(tmp, n=L)
        conv_tmp = np.fft.irfft(Fa_tmp * Fa_tmp, n=L)[:2*n-1]
        tmp_sc = float(2 * n * conv_tmp.max())
        if tmp_sc < cur_score:
            cur, cur_score = tmp, tmp_sc
            if tmp_sc < best_score:
                best, best_score = tmp.copy(), tmp_sc

    # return best as python list
    return [float(x) for x in best]

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ------------------------------------------------------------------
    # initialise (use provided parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random simplex vector of moderate length
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size

    # FFT length sufficient for linear convolution and correlation
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # helpers
    # ------------------------------------------------------------------
    def proj_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto {x >= 0, sum x = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    # initial best
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
    best = a.copy()
    best_score = float(2 * n * conv.max())

    cur = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0
    add_alpha0 = 0.6
    mul_eta0 = 0.6
    beta0 = 8.0            # start of smoothing (softmax) parameter
    beta_inc = 0.001       # slow increase, focuses on the max later
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # ----- autocorrelation -----
        Fa = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
        M = conv_full.max()

        # ----- smooth weights (soft‑max) over the autocorrelation -----
        beta = beta0 + beta_inc * it
        exp_arg = beta * (conv_full - M)          # ≤ 0, prevents overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                          # numerical safeguard
            active = np.flatnonzero(conv_full == M)
            w = np.zeros_like(conv_full)
            w[active] = 1.0 / active.size
            w_sum = 1.0
        w /= w_sum

        # ----- gradient = 2 * (a ⋆ w) for lags 0 … n‑1 -----
        Fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(Fa) * Fw, n=L)[: 3 * n - 2]
        grad = 2.0 * corr_full[:n]

        # ----- additive (projected gradient) step -----
        alpha = add_alpha0 / np.sqrt(it)
        cand_add = cur - alpha * grad
        cand_add = proj_simplex(cand_add)
        Fa_add = np.fft.rfft(cand_add, n=L)
        conv_add = np.fft.irfft(Fa_add * Fa_add, n=L)[: 2 * n - 1]
        cand_add_sc = float(2 * n * conv_add.max())

        # ----- multiplicative (exponential‑weights) step -----
        eta = mul_eta0 / np.sqrt(it)
        exp_factor = np.exp(-eta * grad)
        cand_mul = cur * exp_factor
        s_mul = cand_mul.sum()
        if s_mul == 0.0:
            cand_mul = np.full_like(cur, 1.0 / n)
        else:
            cand_mul = cand_mul / s_mul
        Fa_mul = np.fft.rfft(cand_mul, n=L)
        conv_mul = np.fft.irfft(Fa_mul * Fa_mul, n=L)[: 2 * n - 1]
        cand_mul_sc = float(2 * n * conv_mul.max())

        # ----- pick the best among current and the two candidates -----
        if cand_add_sc < cur_score and cand_add_sc <= cand_mul_sc:
            cur, cur_score = cand_add, cand_add_sc
        elif cand_mul_sc < cur_score:
            cur, cur_score = cand_mul, cand_mul_sc

        # ----- keep global best -----
        if cur_score < best_score:
            best, best_score = cur.copy(), cur_score
            no_improve = 0
        else:
            no_improve += 1

        # ----- occasional random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(n)
            if cur[i] > 0.0:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = proj_simplex(tmp)
                Fa_tmp = np.fft.rfft(tmp, n=L)
                conv_tmp = np.fft.irfft(Fa_tmp * Fa_tmp, n=L)[: 2 * n - 1]
                tmp_sc = float(2 * n * conv_tmp.max())
                if tmp_sc < cur_score:
                    cur, cur_score = tmp, tmp_sc
                    if tmp_sc < best_score:
                        best, best_score = tmp.copy(), tmp_sc
                        no_improve = 0

        # ----- jittered restart toward the best solution -----
        if it % 5000 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = proj_simplex(tmp)
            Fa_tmp = np.fft.rfft(tmp, n=L)
            conv_tmp = np.fft.irfft(Fa_tmp * Fa_tmp, n=L)[: 2 * n - 1]
            tmp_sc = float(2 * n * conv_tmp.max())
            if tmp_sc < cur_score:
                cur, cur_score = tmp, tmp_sc
                if tmp_sc < best_score:
                    best, best_score = tmp.copy(), tmp_sc
                    no_improve = 0

        # ----- early exit if target reached -----
        if best_score < target:
            break

    # ------------------------------------------------------------------
    # final short local refinement while time permits
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = proj_simplex(tmp)
        Fa_tmp = np.fft.rfft(tmp, n=L)
        conv_tmp = np.fft.irfft(Fa_tmp * Fa_tmp, n=L)[: 2 * n - 1]
        tmp_sc = float(2 * n * conv_tmp.max())
        if tmp_sc < cur_score:
            cur, cur_score = tmp, tmp_sc
            if tmp_sc < best_score:
                best, best_score = tmp.copy(), tmp_sc

    # ------------------------------------------------------------------
    # return the best vector found as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]
