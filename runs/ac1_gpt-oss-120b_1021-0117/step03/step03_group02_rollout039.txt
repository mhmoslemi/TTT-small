import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        v = v.ravel()
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# Power‑of‑two helper
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # -------------------- initialise --------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # simple fallback: uniform vector
        a = np.ones(1, dtype=np.float64)

    # normalise to the probability simplex (sum = 1, non‑negative)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = _project_simplex(a / a_sum)

    n = a.size
    # single FFT length sufficient for both convolutions (≥ 3n‑2)
    L = _next_pow2(3 * n - 2)

    # -------------------- parameters --------------------
    alpha = 200.0          # soft‑max sharpness
    step = 0.5             # initial step size
    min_step = 1e-8
    decay = 0.9999
    prob_random = 0.015    # chance for a random mass‑transfer move
    no_improve_limit = 500
    no_improve = 0

    # initial score (exact evaluation)
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    cur_score = float(2.0 * n * conv_full.max())
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- compute full convolution and soft‑max weights -----
        Af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        max_conv = conv_full.max()
        cur_score = float(2.0 * n * max_conv)

        # soft‑max over the whole convolution
        w_raw = np.exp(alpha * (conv_full - max_conv))
        w = w_raw / w_raw.sum()

        # ----- gradient (smooth sub‑gradient of max) -----
        rev_a = a[::-1]
        wf = np.fft.rfft(w, n=L)
        rev_af = np.fft.rfft(rev_a, n=L)
        g_full = np.fft.irfft(wf * rev_af, n=L)
        # linear convolution length = (2n‑1) + n‑1 = 3n‑2
        g_full = g_full[: 3 * n - 2]
        grad = 2.0 * g_full[n - 1 : n - 1 + n]   # size n

        # ----- try a gradient step -----
        cand = a - step * grad
        cand = _project_simplex(cand)

        Af_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(Af_c * Af_c, n=L)[: 2 * n - 1]
        cand_score = float(2.0 * n * conv_c.max())

        improved = False
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            improved = True
        else:
            # backtrack: halve step a few times
            tmp_step = step * 0.5
            for _ in range(3):
                if tmp_step < min_step:
                    break
                cand_bt = a - tmp_step * grad
                cand_bt = _project_simplex(cand_bt)
                Af_bt = np.fft.rfft(cand_bt, n=L)
                conv_bt = np.fft.irfft(Af_bt * Af_bt, n=L)[: 2 * n - 1]
                bt_score = float(2.0 * n * conv_bt.max())
                if bt_score < cur_score - 1e-12:
                    a = cand_bt
                    cur_score = bt_score
                    step = tmp_step
                    improved = True
                    break
                tmp_step *= 0.5
            # occasional uphill acceptance (simulated annealing flavour)
            if not improved and rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                improved = True

        # ----- keep global best -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            if not improved:
                no_improve += 1
            else:
                no_improve = 0

        # ----- occasional random mass transfer -----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.2
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                Af2 = np.fft.rfft(cand2, n=L)
                conv2 = np.fft.irfft(Af2 * Af2, n=L)[: 2 * n - 1]
                cand2_score = float(2.0 * n * conv2.max())
                if cand2_score < cur_score - 1e-12:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                        no_improve = 0

        # ----- step decay & restart logic -----
        step = max(min_step, step * decay)
        if no_improve >= no_improve_limit:
            # random restart on the simplex
            a = rng.random(n)
            a = a / a.sum()
            a = _project_simplex(a)
            step = 0.5
            no_improve = 0

        # ----- safeguard against time overrun -----
        if time.monotonic() + 0.5 > deadline:
            break

    # final sanity check via the provided evaluator (ensures clipping etc.)
    final_seq = [float(x) for x in best]
    # The evaluator may clip or renormalise; we keep the raw numbers.
    return final_seq
