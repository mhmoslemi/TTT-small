    n = a.size

    while time.time() < deadline:
        it += 1

        conv = _conv_fft(cur, L)
        max_val = conv.max()
        # indices of max
        mask = conv == max_val
        # Typically there may be multiple maxima due to symmetry.
        # Compute subgradient as sum of contributions from all max positions.

import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full autocorrelation of x using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    prod = fx * fx
    conv = np.fft.irfft(prod, n=L)[: 2 * x.size - 1]
    return conv

def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Score = 2 * n * max(conv) / sum_a**2."""
    return 2.0 * n * conv.max() / (sum_a * sum_a)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ---- initialise -------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback uniform of moderate length
        n0 = 2000
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # normalise to sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two

    cur = a.copy()
    cur_conv = _conv_fft(cur, L)
    cur_sc = _score_from_conv(cur_conv, n, cur.sum())
    best, best_sc = cur.copy(), cur_sc

    # ---- hyper‑parameters -------------------------------------------------
    it = 0
    alpha0 = 0.8            # initial step size multiplier
    eps_move = 0.35         # fraction of mass moved in random transfers
    jitter_scale = 4e-5     # jitter amplitude for restarts
    temp = 0.25             # Metropolis temperature start
    decay_temp = 0.9997     # exponential cooling per iteration
    restart_interval = 2500
    random_move_prob = 0.12

    # ---- main optimisation loop -------------------------------------------
    while time.time() < deadline:
        it += 1

        # compute convolution and the set of maximal lags
        conv = _conv_fft(cur, L)
        max_val = conv.max()
        active = np.nonzero(conv == max_val)[0]

        # sub‑gradient for all active lags
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # mirror (exponential) update
        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum <= 0.0 or not np.isfinite(cand_sum):
            cand = np.full_like(cur, 1.0 / n)
        else:
            cand = cand / cand_sum

        cand_sc = _score_from_conv(_conv_fft(cand, L), n, cand.sum())

        # accept if improvement or with Metropolis probability
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
        temp *= decay_temp

        # record best
        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()

        # ---- occasional random mass‑transfer move -------------------------
        if rng.random() < random_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * eps_move
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp_sum = tmp.sum()
                if tmp_sum > 0.0:
                    tmp = tmp / tmp_sum
                    tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp_sum)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best = tmp_sc, tmp.copy()

        # ---- jittered restart around the current best --------------------
        if it % restart_interval == 0:

            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0.0:
                tmp = tmp / tmp_sum
                tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp_sum)
                if tmp_sc < cur_sc:

                if tmp_sc < best_sc:

    # Return a plain Python list (the evaluator expects a list)
    return [float(x) for x in best]
