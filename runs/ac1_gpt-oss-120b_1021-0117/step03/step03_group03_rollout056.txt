import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    conv_len = 2 * n - 1
    # FFT length sufficient for a * a and a * w_rev (length 3n‑2)
    fft_len = 1 << ((3 * n - 2 - 1).bit_length())

    # ----- best‑so‑far --------------------------------------------------------
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    best = a.copy()
    best_score = float(2.0 * n * conv.max())
    # compare with uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    A_u = np.fft.rfft(uniform, n=fft_len)
    conv_u = np.fft.irfft(A_u * A_u, n=fft_len)[:conv_len]
    uniform_score = float(2.0 * n * conv_u.max())
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ----- optimisation parameters --------------------------------------------
    beta = 0.9                     # momentum decay
    eta0 = 0.6                     # initial step magnitude
    jitter_every = 250
    jitter_std = 0.015
    max_stale = 4000
    target_score = 1.50274349

    # smoothing (soft‑max) parameters
    alpha_start = 1.0
    alpha_max = 22.0
    alpha = alpha_start

    v = np.zeros_like(a)           # momentum buffer
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # ----- convolution ----------------------------------------------------
        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)

        # ----- update best ----------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ----- soft‑max weights for the peak ---------------------------------
        # w sums to 1, concentrates on large convolution values
        diffs = conv - max_val                # <=0
        w = np.exp(alpha * diffs)             # in (0,1]
        w_sum = w.sum()
        if w_sum == 0.0:                      # numerical safeguard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient via convolution with reversed weight -------------------
        w_rev = w[::-1]                       # reversed
        W_rev = np.fft.rfft(w_rev, n=fft_len)
        full = np.fft.irfft(A * W_rev, n=fft_len)
        full = full[:n + conv_len - 1]        # length 3n‑2
        idx = conv_len - 1 - np.arange(n)     # mapping to gradient indices
        g = 2.0 * full[idx]                  # sub‑gradient of smoothed max

        # ----- momentum step & projection ------------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v)

        # ----- adaptive smoothing parameter ----------------------------------
        # increase alpha gradually (linear schedule)
        if it < 5000:
            alpha = alpha_start + (alpha_max - alpha_start) * (it / 5000.0)
        else:
            alpha = alpha_max

        # ----- occasional jitter ---------------------------------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v[:] = 0.0
            no_improve = 0

        # ----- restart if stagnated ------------------------------------------
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v[:] = 0.0
            no_improve = 0

    # ----- return best sequence ------------------------------------------------
    return [float(x) for x in best]
