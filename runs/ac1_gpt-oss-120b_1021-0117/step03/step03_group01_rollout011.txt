def _optimize(start_vec, deadline, rng):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    cur_max, cur_idx = _conv_max_and_index(a, L)
    cur_score = 2 * n * cur_max
    best = a.copy()
    best_score = cur_score

    step = 1.0
    min_step = 1e-8
    decay = 0.9995
    beta = 0.9
    momentum = np.zeros_like(a)

    iter_count = 0
    # thresholds for additional operations
    prob_rand = 0.04
    prob_smooth = 0.008
    prob_noise = 0.02

    while time.monotonic() < deadline:
        # gradient
        lo = max(0, cur_idx - (n - 1))
        hi = min(n - 1, cur_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] = 2.0 * a[cur_idx - j]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _simplex_projection(cand)

        cand_max, cand_idx = _conv_max_and_index(cand, L)
        cand_score = 2 * n * cand_max

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            cur_idx = cand_idx
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # simulated annealing acceptance based on step temperature
            temperature = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(temperature, 1e-12)):
                a = cand
                cur_score = cand_score
                cur_idx = cand_idx

        # random mass transfer
        if rng.random() < prob_rand:
