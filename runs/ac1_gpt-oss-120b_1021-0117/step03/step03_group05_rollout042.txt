import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities (FFT convolution, simplex projection, scoring)
# --------------------------------------------------------------
def _conv_fft(v: np.ndarray, _L: int) -> np.ndarray:
    """Linear convolution using FFT (full length 2*len(v)-1)."""
    fv = np.fft.rfft(v, n=_L)
    conv = np.fft.irfft(fv * fv, n=_L)
    m = 2 * v.size - 1
    return conv[:m]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    # based on the algorithm of Wang & Carreira‑Perpiñán (2013)
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    rho = np.where(u - css / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise (numerical safety)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / v.size
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Score = 2 * n * max(conv)  (sum of a is 1 after simplex projection)."""
    return 2.0 * n * np.max(conv)

            break

    # --------------------------------------------------------------
    # Phase 2 – simple simulated‑annealing style random transfers
    # --------------------------------------------------------------
    sa_deadline = deadline
    temp_start = 0.15
    temp_end = 0.001
