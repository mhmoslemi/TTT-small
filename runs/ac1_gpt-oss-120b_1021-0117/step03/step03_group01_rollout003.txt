def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # initial a
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = _sqrt_series(256)
    a0 = np.maximum(a0, 0.0)
    a = _project_simplex(a0)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # initial conv
    max_val, max_idx, conv = _conv_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.5
    min_step = 1e-8
    decay = 0.9998
    beta = 0.9
    momentum = np.zeros_like(a)

    prob_random = 0.015
    prob_smooth = 0.01

    eps = 1e-12

    while time.monotonic() < deadline:
        # active set
        active = np.where(conv >= max_val - eps)[0]
        # compute gradient
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            j = np.arange(lo, hi + 1, dtype=int)
            g[j] += 2.0 * a[k - j]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        cand_max, cand_idx, cand_conv = _conv_max(cand, L)
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            a = cand
            conv = cand_conv
            max_val = cand_max
            max_idx = cand_idx
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # occasional uphill acceptance
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / (max(T, 1e-12))):
                a = cand
                conv = cand_conv
                max_val = cand_max
                max_idx = cand_idx
                cur_score = cand_score

        # random mass transfer
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                # move a fraction of mass from i to j
                frac = rng.random() * 0.4  # up to 40% of a[i]
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max2, idx2, conv2 = _conv_max(cand2, L)
                score2 = 2.0 * n * max2
                if score2 < cur_score:
                    a = cand2
                    conv = conv2
                    max_val = max2
                    max_idx = idx2
                    cur_score = score2
                    if score2 < best_score:
                        best = cand2.copy()
                        best_score = score2

        # smoothing
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            max_s, idx_s, conv_s = _conv_max(smooth, L)
            score_s = 2.0 * n * max_s
            if score_s < cur_score:
                a = smooth
                conv = conv_s
                max_val = max_s
                max_idx = idx_s
                cur_score = score_s
                if score_s < best_score:
                    best = smooth.copy()
                    best_score = score_s

        # step decay
        step = max(min_step, step * decay)

        # early exit if time short
        if time.monotonic() + 0.2 > deadline:
            break

    # fallback: ensure best is finite
    if not np.isfinite(best_score) or best_score > 1e9:
        # fallback to simple uniform distribution
        best = np.full(n, 1.0 / n)
        best_score = 2.0 * n * np.max(np.convolve(best, best, mode='full')[:2*n-1])

    return [float(x) for x in best]

def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w
```

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w


# ------------------------------------------------------------
# Helper: max value and index of a self‑convolution (FFT)
# ------------------------------------------------------------
def _conv_max(vec: np.ndarray, L: int):
    """Return (max value, index of max, full convolution array) of vec*vec."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    max_val = float(conv.max())
    max_idx = int(conv.argmax())
    return max_val, max_idx, conv

# Core optimizer
# ------------------------------------------------------------
def _optimize_from(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    max_val, max_idx, conv = _conv_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.5
    min_step = 1e-8
    decay = 0.9998
    beta = 0.9
    momentum = np.zeros_like(a)

    prob_random = 0.015
    prob_smooth = 0.01
    eps = 1e-12

    while time.monotonic() < deadline:
        # --------------------------------------------------------
        # Subgradient from all active (near‑max) autocorrelation points
        # --------------------------------------------------------
        active = np.where(conv >= max_val - eps)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # --------------------------------------------------------
        # Gradient step with momentum and simplex projection
        # --------------------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

                max_idx = cand_idx
