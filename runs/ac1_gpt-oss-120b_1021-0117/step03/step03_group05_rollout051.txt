import time
import numpy as np

def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Full convolution v*v using an FFT of length L."""
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * v.size - 1]
    return conv

def _score(v: np.ndarray, L: int) -> float:
    """Compute the AC1 upper bound for vector v."""
    n = v.size
    conv = _conv_fft(v, L)
    return float(2.0 * n * conv.max() / (v.sum() ** 2))

def _subgradient(v: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Exact subgradient of max_k (v*v)_k for the given active indices."""
    n = v.size

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[lo:hi + 1] += 2.0 * v[k - idx]
    return g
