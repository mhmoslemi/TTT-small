def _grad_smoothed_max(a, alpha):
    n = a.shape[0]
    conv = _conv_fft(a)  # length 2n-1
    # Compute weight vector w (probability distribution) via softmax
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    w_rev = w[::-1]  # reversed

    # Compute convolution a * w_rev via FFT
    L = 1 << ((n + (2*n-1) - 1).bit_length())
    # Actually we need length >= n + (2n-1) - 1 = 3n-2.
    # So compute L = next power of 2 >= 3n-2
    L = 1 << ((3*n - 2).bit_length())
    # Compute FFT
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    conv_aw = np.fft.irfft(fa * fw, n=L)
    # conv_aw length L, but we need first 3n-2 entries
    # Compute g
    idx = 2*n - 2 - np.arange(n)
    g = 2.0 * conv_aw[idx]
    return g

import time

# ----- basic utilities -------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    a = a / a.sum()
    return a


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    a = a / a.sum()
    return a


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a
