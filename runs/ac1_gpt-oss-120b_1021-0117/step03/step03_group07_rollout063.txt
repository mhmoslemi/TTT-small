import time
import numpy as np

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full) via FFT, real input."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact score used by the evaluator."""
    n = a.shape[0]
    conv = _conv_full(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initial vector (parent if available, otherwise a few heuristics)

        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # a few analytic families – keep the best
    def _make_exponential(r):
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
        return a

    def _make_powerlaw(p):
        a = 1.0 / np.arange(1, n + 1, dtype=np.float64) ** p
        a /= a.sum()
        return a

    for r in np.linspace(0.90, 0.9999, 30):
        cand = _make_exponential(r)
        sc = _score(cand)
        if sc < best_sc:

    cand = np.arange(n, 0, -1, dtype=np.float64)

    if _score(cand) < best_sc:
