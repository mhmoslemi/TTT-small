import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum to 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # Guard against all‑zero output
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w


# ------------------------------------------------------------
# Convolution utilities (FFT, max value & indices)
# ------------------------------------------------------------
def _conv_max_and_indices(a: np.ndarray):
    """Return (max_val, array_of_indices_achieving_max) for a*a convolution."""
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    max_val = float(conv.max())
    # tolerance to collect all maximal points
    eps = 1e-12 * max_val if max_val != 0 else 1e-12
    idxs = np.where(conv >= max_val - eps)[0]
    return max_val, idxs


def _gradient_from_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t a (averaged over all maximal indices)."""
    n = a.size
    g = np.zeros_like(a)
    for idx in max_idxs:
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        j = np.arange(lo, hi + 1, dtype=np.int64)
        g[j] += 2.0 * a[idx - j]
    if max_idxs.size > 1:
        g /= max_idxs.size
    return g
