import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (project onto probability simplex, convolution via FFT)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real inputs)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Analytic families for a quick warm‑start
# ----------------------------------------------------------------------
def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_tent(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape."""
    idx = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(idx - c) / (c + 1e-12)
    a = np.maximum(a, 0.0)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth gradient of the p‑norm of the convolution (p>1)
# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, p: float) -> np.ndarray:
    """Gradient of (∑ (a*a)^p)^{1/p} w.r.t. a on the simplex."""
    n = a.shape[0]
    conv = _conv_fft(a)                     # length 2n‑1
    conv_pow = conv ** p
    s = conv_pow.sum()
    if s == 0.0:
        return np.zeros_like(a)
    factor = s ** ((1.0 - p) / p)            # (∑c^p)^{(1‑p)/p}
    w = conv ** (p - 1.0)                    # weight for each lag

    # linear convolution a * w  (len 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w, n=L)
    conv_full = np.fft.irfft(af * wf, n=L)[: 3 * n - 2]
    # correlation a ⋆ w   (len 3n‑2)
    corr_full = np.fft.irfft(af * np.conj(wf), n=L)[: 3 * n - 2]

    grad = factor * (conv_full[:n] + corr_full[:n])
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise: use parent if present, else fall back to a few families
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256

    n = cur.size

    best_sc = _score(best_vec)

    target = 1.50274349

            tmp_sc = _score(tmp)

    return [float(v) for v in best_vec]
