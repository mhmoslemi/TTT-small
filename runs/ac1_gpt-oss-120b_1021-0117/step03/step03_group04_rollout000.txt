import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimizer for the AC1 upper bound.
    Returns a non‑negative sequence (list of floats) whose AC1 value is as low
    as possible within the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time handling
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 2. Base length and initial candidates

        n = int(parent_raw.size)
        parent = parent_raw.copy()
        s = parent.sum()
        if s >= 0.01:
            parent /= s
        else:
            parent = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        # fallback length if the parent is unavailable
        n = 256
        parent = None

    # Uniform distribution
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # Tent (triangular) shape, symmetric
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    tent = np.maximum(tent, 0.0)
    if tent.sum() > 0:
        tent /= tent.sum()
    else:
        tent = uniform.copy()

    # Random Dirichlet‑like start
    rand_vec = rng.random(n)
    rand_vec /= rand_vec.sum()

    candidates = [uniform, tent, rand_vec]
    if parent is not None:
        candidates.append(parent)

    # ------------------------------------------------------------------
    # 3. Fast evaluation via FFT (sum is always 1 after projection)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(a: np.ndarray) -> float:
        """2*n*max(conv(a,a)) – sum(a) == 1 by construction."""
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # pick best initial candidate
    best = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    mu = 0.9                     # momentum coefficient
    eta0 = 0.5                   # initial step magnitude
    K = 5                        # number of top shifts for sub‑gradient
    target = 1.50274349
    v = np.zeros_like(a)        # momentum buffer
    it = 0
    last_improve_it = 0

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and current surrogate score -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve_it = it
            if best_score <= target:
                break

        # ----- sub‑gradient from top K shifts -----
        if K < conv_len:
            top_idx = np.argpartition(-conv, K)[:K]
        else:
            top_idx = np.arange(conv_len)
        top_vals = conv[top_idx]
        wsum = top_vals.sum()
        if wsum > 0:
            weights = top_vals / wsum
        else:
            weights = np.full_like(top_vals, 1.0 / top_vals.size)

        g = np.zeros_like(a)
        for shift, wgt in zip(top_idx, weights):
            if shift < n:
                # affect entries 0..shift (inclusive)
                g[: shift + 1] += wgt * 2.0 * a[shift :: -1]
            else:
                offset = shift - (n - 1)
                # affect entries offset..n-1
                g[offset:] += wgt * 2.0 * a[offset:][::-1]

        # ----- momentum update -----
        v = mu * v + (1.0 - mu) * g

        # ----- step size (diminishing) -----

        a = _proj_simplex(a - eta * v)

        # ----- enforce symmetry (helps reduce peak) -----
        a = (a + a[::-1]) * 0.5

            a = rng.random(n)
            a /= a.sum()
            v.fill(0.0)
            last_improve_it = it

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
