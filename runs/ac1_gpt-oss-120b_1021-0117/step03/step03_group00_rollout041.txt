def compute_score(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    maxv = conv.max()
    sum_a = a.sum()
    # avoid division by zero
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * maxv / (sum_a * sum_a))

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def _score(a: np.ndarray, L: int = None) -> float:
    """Exact AC1 bound (2*n*max(conv)/(sum^2))."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    maxv = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * maxv / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 1024
        a0 = np.full(n0, 1.0 / n0, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())

    best = a0.copy()
    best_sc = _score(best, L)
    cur = best.copy()
    cur_sc = best_sc

    eps0, eps_decay, eps_min = 0.5, 0.9997, 1e-8
    step0 = 0.5
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution ----------------------------------------
        af = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- smoothed weights -------------------------------------------
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed max ------------------------------------
        w_rev = w[::-1]
        wf = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * wf, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- step & projection -------------------------------------------
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        cand = cur - alpha * g
        cand = _project_simplex(cand)

        # ----- evaluate ----------------------------------------------------
        cand_sc = _score(cand, L)

        # ----- accept / revert ---------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc

        else:
