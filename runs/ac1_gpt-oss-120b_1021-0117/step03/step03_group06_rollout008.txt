def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)  # small safety

    # init sequence
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_default = 2000
        a = np.full(n_default, 1.0/n_default, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0/a.size)
    else:
        a = a / s

    n = a.size
    # FFT length: next power of two >= 2*n - 1
    L = 1 << ((2*n - 1).bit_length())
    # helper functions
    def conv_fft(x):
        fx = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fx * fx, n=L)[:2*n-1]
        return conv
    def score_of(x):
        conv = conv_fft(x)
        return 2.0 * n * conv.max() / (x.sum()**2)

    cur = a.copy()
    cur_score = score_of(cur)
    best = cur.copy()
    best_score = cur_score

    # hyperparameters
    alpha0 = 0.7
    eps_move = 0.15   # fraction of mass moved in targeted move
    jitter_scale = 5e-5
    restart_interval = 2500
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # compute conv and active index
        conv = conv_fft(cur)
        max_b = conv.max()
        k = int(conv.argmax())
        # compute active index range
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        other = k - idx
        # gradient (mirror)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[other]   # as used previously

        # exponentiated gradient step
        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum <= 0.0 or not np.isfinite(cand_sum):
            cand = np.full_like(cur, 1.0/n)
        else:
            cand = cand / cand_sum

        cand_score = 2.0 * n * conv_fft(cand).max() / (cand.sum()**2)

        # accept if better
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            # with small probability accept bad step to escape plateaus
            if rng.random() < 0.005:
                cur = cand
                cur_score = cand_score
                no_improve += 1
            else:
                no_improve += 1

        # targeted mass move (occasionally)
        if rng.random() < 0.1:
            # compute contributions in current max conv location
            contributions = cur[idx] * cur[other]
            i_max = idx[contributions.argmax()]
            # random target not equal to i_max
            j = rng.integers(0, n-1)
            if j >= i_max:
                j += 1
            delta = cur[i_max] * eps_move
            tmp = cur.copy()
            tmp[i_max] -= delta
            tmp[j] += delta
            # ensure non-negativity
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            tmp_score = 2.0 * n * conv_fft(tmp).max() / (tmp.sum()**2)
            if tmp_score < cur_score:
                cur = tmp
                cur_score = tmp_score
                no_improve = 0
            if tmp_score < best_score:
                best = tmp.copy()
                best_score = tmp_score

        # jitter restart from best occasionally
        if it % restart_interval == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            tmp_score = 2.0 * n * conv_fft(tmp).max() / (tmp.sum()**2)
            if tmp_score < cur_score:
                cur = tmp
                cur_score = tmp_score
                no_improve = 0
            if tmp_score < best_score:
                best = tmp.copy()
                best_score = tmp_score

        # update best if cur improved further
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score

        # early exit if target reached
        if best_score < 1.50274349:
            break

    # Return best sequence converted to Python list
    return [float(x) for x in best]
