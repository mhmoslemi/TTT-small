def _conv_fft(a):
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    return conv

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    # start time
    start = time.monotonic()
    budget = float(budget_s)
    deadline = start + max(0.0, budget - 0.5)  # safety margin
    # initial vector
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        if s > 0:
            cur = cur / s
        else:
            cur = np.full_like(cur, 1.0/cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0/n, dtype=np.float64)

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # scoring function
    def _score_vec(v):
        s = v.sum()
        if s < 0.01:
            return np.inf
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv*fv, n=L)[:2*n-1]
        return float(2.0 * n * conv.max() / (s*s))

    # best so far
    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)

    # --------- families ----------
    def make_exponential(r):
        a = np.power(r, np.arange(n, dtype=np.float64))
        total = a.sum()
        if total == 0:
            return np.full(n, 1.0/n)
        return a/total

    def make_powerlaw(p):
        a = 1.0 / np.power(np.arange(1, n+1, dtype=np.float64), p)
        total = a.sum()
        if total == 0:
            return np.full(n, 1.0/n)
        return a/total

    def make_linear():
        a = np.arange(n,0,-1, dtype=np.float64)
        return a/ a.sum()

    def make_quadratic():
        a = np.square(np.arange(n,0,-1, dtype=np.float64))
        return a/ a.sum()

    def make_sqrt_series():
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            if k == 1:
                s = 0.0
            else:
                s = np.dot(a[1:k], a[k-1:0:-1])
            a[k] = (1.0 - s) / 2.0
        a = np.maximum(a, 0.0)
        tot = a.sum()
        if tot == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / tot
        return a

    def make_u_shape(p):
        idx = np.arange(n, dtype=np.float64) + 0.5
        x = idx / n
        eps = 1e-12
        a = np.power(x * (1.0 - x) + eps, p)
        a = np.maximum(a, 0.0)
        tot = a.sum()
        if tot == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / tot
        return a

    # Explore families
    for r in np.linspace(0.90, 0.9999, 30):
        cand = make_exponential(r)
        sc = _score_vec(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 4.0, 30):
        cand = make_powerlaw(p)
        sc = _score_vec(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    cand = make_linear()
    sc = _score_vec(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()
    cand = make_quadratic()
    sc = _score_vec(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()
    cand = make_sqrt_series()
    sc = _score_vec(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(-0.8, -0.1, 14):
        cand = make_u_shape(p)
        sc = _score_vec(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    # start from best
    cur = best_vec.copy()
    cur_sc = best_sc

    # random generator
    rng = np.random.default_rng(seed if isinstance(seed, (int, np.integer)) else 42)

    # -------- Phase 1: Exponential gradient descent ----------
    eta0 = 0.5
    it = 0
    # allocate a buffer for gradient to avoid repeated allocation maybe
    while time.monotonic() < deadline * 0.65:   # 65% of time
        it += 1
        fv = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fv*fv, n=L)[:2*n-1]
        k = int(np.argmax(conv))

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        g = np.zeros_like(cur)
        rev = k - idx
        g[idx] = 2.0 * cur[rev]

        eta = eta0 / np.sqrt(it)
        cand = cur * np.exp(-eta * g)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum
        else:
            cand = np.full_like(cur, 1.0/n)

        # evaluate candidate
        fv_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(fv_c*fv_c, n=L)[:2*n-1]
        cand_sc = float(2.0 * n * conv_c.max() / (cand.sum()**2))

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional acceptance to escape local minima
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc

        # jitter occasionally
        if it % 4000 == 0:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            fv_t = np.fft.rfft(tmp, n=L)
            conv_t = np.fft.irfft(fv_t*fv_t, n=L)[:2*n-1]
            tmp_sc = float(2.0 * n * conv_t.max() / (tmp.sum()**2))
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        if time.monotonic() >= deadline:
            break

    # -------- Phase 2: Projected Euclidean subgradient ----------
    # Simplex projection
    def _project_simplex(v):
        n = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho_arr = np.where(u - css / np.arange(1, n+1) > 0)[0]
        if rho_arr.size == 0:
            theta = 0.0
        else:
            rho = rho_arr[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    alpha0 = 0.8
    it2 = 0
    while time.monotonic() < deadline:
        it2 += 1
        fv = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fv*fv, n=L)[:2*n-1]
        k = int(np.argmax(conv))

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        g = np.zeros_like(cur)
        rev = k - idx
        g[idx] = 2.0 * cur[rev]

        alpha = alpha0 / np.sqrt(it2)
        cand = cur - alpha * g
        cand = _project_simplex(cand)

        # evaluate
        fv_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(fv_c*fv_c, n=L)[:2*n-1]
        cand_sc = float(2.0 * n * conv_c.max() / (cand.sum()**2))

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional random exploration
            if rng.random() < 0.01:
                # random mass shift
                idx_nonzero = np.where(cur > 1e-12)[0]
                if idx_nonzero.size:
                    i = int(rng.choice(idx_nonzero))
                    j = int(rng.integers(0, n-1))
                    if j == i:
                        j = (j+1) % n
                    delta = cur[i] * rng.random() * 0.3
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    fv_tmp = np.fft.rfft(tmp, n=L)
                    conv_tmp = np.fft.irfft(fv_tmp*fv_tmp, n=L)[:2*n-1]
                    tmp_sc = float(2.0 * n * conv_tmp.max() / (tmp.sum()**2))
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()
        # occasional jitter
        if it2 % 2500 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            fv_tmp = np.fft.rfft(tmp, n=L)
            conv_tmp = np.fft.irfft(fv_tmp*fv_tmp, n=L)[:2*n-1]
            tmp_sc = float(2.0 * n * conv_tmp.max() / (tmp.sum()**2))
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
        if time.monotonic() >= deadline:
            break

    # -------- final safety fallback ----------
    if best_sc == float('inf') or best_vec.size == 0:
        best_vec = np.full(n, 1.0/n, dtype=np.float64)

    return [float(v) for v in best_vec]

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real input)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score_vec(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum is not forced)."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _conv_fft(a)
    return float(2.0 * a.size * conv.max() / (s * s))


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        s = 0.0 if k == 1 else np.dot(a[1:k], a[k - 1 : 0 : -1])
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _make_u_shape(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 0.5
    x = idx / n
    eps = 1e-12
    a = np.power(x * (1.0 - x) + eps, p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)

    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    start = time.monotonic()
    total_allowed = max(0.0, float(budget_s) - 0.5)
    deadline = start + total_allowed
    phase1_end = start + 0.65 * total_allowed

    # -----------------------------------------------------------------
    # initialise from parent if present, otherwise uniform
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
