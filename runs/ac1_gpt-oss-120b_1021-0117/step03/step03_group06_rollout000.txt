def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # initialize base sequence
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        base = np.full(2000, 1.0/2000, dtype=np.float64)
    # Ensure nonnegative, normalize to sum 1
    base = np.maximum(base, 0.0)
    s = base.sum()
    if s <= 0:
        base.fill(1.0 / base.size)
    else:
        base /= s
    n = base.size
    L = 1 << ((2*n-1).bit_length())
    # function
    def conv_fft(a):
        f = np.fft.rfft(a, n=L)
        c = np.fft.irfft(f * f, n=L)
        return c[:2*n-1]
    # evaluate utility
    def score(a, conv=None):
        if conv is None:
            conv = conv_fft(a)
        return 2.0 * n * conv.max()
    # baseline best
    cur = base.copy()
    cur_conv = conv_fft(cur)
    cur_score = score(cur, cur_conv)
    best = cur.copy()
    best_score = cur_score
    # parameters
    it = 0
    alpha0 = 1.0   # step multiplier
    jitter_scale = 5e-5
    temp0 = 0.1
    mass_move_prob = 0.12
    mass_scale = 0.3
    restart_interval = 2500
    # main loop
    while time.monotonic() < deadline:
        it += 1
        # compute conv & gradient
        conv = conv_fft(cur)
        k = int(conv.argmax())
        # compute gradient indices
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        # gradient vector
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]   # as before
        # step size
        eta = alpha0 / math.sqrt(it)
        # exponentiated update
        mult = np.exp(-eta * g)
        cand = cur * mult
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand /= cand_sum
        else:
            cand = np.full_like(cur, 1.0/n)
        cand_conv = conv_fft(cand)
        cand_score = score(cand, cand_conv)
        # metropolis acceptance
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
        else:
            # compute probability using temperature decreasing
            temp = temp0 / math.sqrt(it)
            prob = math.exp(-(cand_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
        # update best
        if cur_score < best_score:
            best, best_score = cur.copy(), cur_score
        # random mass transfer
        if rng.random() < mass_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * mass_scale
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp_sum = tmp.sum()
                if tmp_sum > 0:
                    tmp /= tmp_sum
                    tmp_conv = conv_fft(tmp)
                    tmp_score = score(tmp, tmp_conv)
                    if tmp_score < cur_score:
                        cur, cur_score = tmp, tmp_score
                    if tmp_score < best_score:
                        best, best_score = tmp.copy(), tmp_score
        # jittered restart
        if it % restart_interval == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0:
                tmp /= tmp_sum
                tmp_conv = conv_fft(tmp)
                tmp_score = score(tmp, tmp_conv)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
        # early exit if good enough
        if best_score < 2.0 * n * (1.50274349 / (2 * n)):
            # Actually best_score is already 2*n * max_conv.
            # But target for best_score is target_score itself, not scaled.
            # So compare directly
            if best_score < 1.50274349:
                break
    # Return list
    return [float(x) for x in best]

import time, math, numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a non‑negative step function that lowers the AC1 bound.
    Uses exponentiated gradient updates on the active autocorrelation index,
    occasional random mass‑transfer moves, and periodic jittered restarts.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----------------------------------------------------------------------
    # Initialise from the provided parent or a uniform distribution.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = np.full(2000, 1.0 / 2000, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 <= 0.0:
        a0.fill(1.0 / a0.size)
    else:
        a0 /= s0                     # enforce sum = 1

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    # ----------------------------------------------------------------------
    # Helper: full self‑convolution via FFT (returns length 2*n‑1 array).
    # ----------------------------------------------------------------------
    def conv_fft(x: np.ndarray) -> np.ndarray:
        fx = np.fft.rfft(x, n=L)
        c = np.fft.irfft(fx * fx, n=L)
        return c[: 2 * n - 1]

    # ----------------------------------------------------------------------
    # Helper: objective = 2*n*max(conv)  (sum is 1).
    # ----------------------------------------------------------------------
    def obj_val(conv: np.ndarray) -> float:
        return 2.0 * n * conv.max()

    # ----------------------------------------------------------------------
    # Baseline best solution.
    # ----------------------------------------------------------------------
    cur = a0.copy()
    cur_conv = conv_fft(cur)
    cur_sc = obj_val(cur_conv)

    best = cur.copy()
    best_sc = cur_sc

    # ----------------------------------------------------------------------
    # Search hyper‑parameters.
    # ----------------------------------------------------------------------
    it = 0
    alpha0 = 1.0          # initial step multiplier for exponentiated update
    jitter_scale = 5e-5   # scale of Gaussian jitter during restarts
    temp0 = 0.1           # initial temperature for Metropolis acceptance
    mass_move_prob = 0.12 # probability of a random mass‑transfer move
    mass_scale = 0.30     # maximal fraction of mass moved in a transfer
    restart_interval = 2500
    stagnation_limit = 8000

    no_improve_best = 0   # counts iterations without improving the best score

    # ----------------------------------------------------------------------
    # Main optimisation loop.
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- 1. Gradient step on the active convolution index -----
        conv = conv_fft(cur)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]            # gradient of conv[k] w.r.t. a

        eta = alpha0 / math.sqrt(it)          # diminishing step size
        mult = np.exp(-eta * g)
        cand = cur * mult
        cand_sum = cand.sum()
        if cand_sum > 0.0:
            cand /= cand_sum
        else:
            cand.fill(1.0 / n)

        cand_conv = conv_fft(cand)
        cand_sc = obj_val(cand_conv)

        # ----- 2. Acceptance (deterministic descent + Metropolis) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
        else:
            temp = temp0 / math.sqrt(it)
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- 3. Random mass‑transfer move (exploration) -----
        if rng.random() < mass_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * mass_scale
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp_sum = tmp.sum()
                if tmp_sum > 0.0:
                    tmp /= tmp_sum
                    tmp_sc = obj_val(conv_fft(tmp))
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc

        # ----- 4. Update best solution -----
        if cur_sc < best_sc:
            best, best_sc = cur.copy(), cur_sc
            no_improve_best = 0
        else:
            no_improve_best += 1

        # ----- 5. Periodic jittered restart from the best point -----
        if it % restart_interval == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0.0:
                tmp /= tmp_sum
                tmp_sc = obj_val(conv_fft(tmp))
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve_best = 0

        # ----- 6. Random restart after long stagnation -----
        if no_improve_best > stagnation_limit:
            cur = rng.random(n)
            cur /= cur.sum()
            cur_sc = obj_val(conv_fft(cur))
            no_improve_best = 0

        # ----- 7. Early exit if target achieved -----
        if best_sc < 1.50274349:
            break

    # Return the best found sequence as a plain Python list.
    return [float(x) for x in best]
