def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of a real vector with itself using an FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]


def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact score used by the evaluator."""
    return float(2.0 * n * conv.max() / (sum_a * sum_a))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----------------------------------------------------------------------
    # Initialise: use the provided parent if present, otherwise a smooth sine.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 0.5) / n)          # strictly positive, smooth
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        n = a.size
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total

    n = a.size
    L = _next_pow2(2 * n - 1)

    # ----------------------------------------------------------------------
    # Current best solution tracking
    # ----------------------------------------------------------------------
    cur_conv = _conv_fft(a, L)
    cur_score = _score_from_conv(cur_conv, n, a.sum())
    best = a.copy()
    best_score = cur_score

    # ----------------------------------------------------------------------
    # Hyper‑parameters (tuned for fast descent and occasional exploration)
    # ----------------------------------------------------------------------
    it = 0
    alpha0 = 0.5           # base step size for exponentiated update
    active_k = 7           # how many largest convolution entries to use
    eps_move = 0.2         # max fraction of mass shuffled in a random move
    jitter_scale = 5e-4    # magnitude of Gaussian jitter for restarts
    temp0 = 0.02           # temperature floor for Metropolis jumps
    no_improve = 0

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- compute current convolution and score -----------------------
        cur_conv = _conv_fft(a, L)
        cur_sum = a.sum()
        cur_score = _score_from_conv(cur_conv, n, cur_sum)

        # ----- active set: top‑k convolution entries ----------------------
        ksize = active_k if active_k < cur_conv.size else cur_conv.size
        top_idx = np.argpartition(-cur_conv, ksize - 1)[:ksize]

        # ----- sub‑gradient from the active set ---------------------------
        g = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            ids = np.arange(lo, hi + 1, dtype=int)
            g[ids] += 2.0 * a[k - ids]          # contribution of this constraint

        # ----- exponentiated mirror‑descent step -------------------------
        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        mul = np.clip(mul, 1e-20, 1e20)          # protect against overflow
        cand = a * mul
        s = cand.sum()
        if s > 0.0 and np.isfinite(s):
            cand = cand / s
        else:
            cand = np.full_like(a, 1.0 / n)

        cand_conv = _conv_fft(cand, L)
        cand_score = _score_from_conv(cand_conv, n, cand.sum())

        # ----- acceptance (greedy + occasional Metropolis) ---------------
        if cand_score < cur_score:
            a, cur_score = cand, cand_score
            no_improve = 0
        else:
            prob = math.exp(-(cand_score - cur_score) / (temp0 + eta))
            if rng.random() < prob:
                a, cur_score = cand, cand_score
            no_improve += 1

        if cand_score < best_score:
            best_score, best = cand_score, cand.copy()

        # ----- random mass‑transfer move (exploration) --------------------
        if rng.random() < 0.07:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * eps_move
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                s = tmp.sum()
                if s > 0.0:
                    tmp = tmp / s
                    tmp_score = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
                    if tmp_score < cur_score:
                        a, cur_score = tmp, tmp_score
                        no_improve = 0
                    if tmp_score < best_score:
                        best_score, best = tmp_score, tmp.copy()

        # ----- jittered restart from the best point ----------------------
        if it % 3000 == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            s = tmp.sum()
            if s > 0.0:
                tmp = tmp / s
                tmp_score = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
                if tmp_score < cur_score:
                    a, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best_score, best = tmp_score, tmp.copy()
                no_improve = 0

        # ----- restart after long stagnation ------------------------------
        if no_improve > 12000:
            cur = best + rng.normal(scale=jitter_scale * 8, size=n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_score = _score_from_conv(_conv_fft(cur, L), n, cur.sum())
            a = cur
            no_improve = 0

        # ----- early exit if target reached --------------------------------
        if best_score < 1.50274349:
            break

    # Return the best sequence found as a plain Python list
    return [float(v) for v in best]
