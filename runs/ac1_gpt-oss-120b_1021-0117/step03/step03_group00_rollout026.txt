def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialization
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: simple triangular distribution
        n_fallback = 2000   # may adjust
        a = np.arange(1, n_fallback+1)
        a = np.concatenate((a, a[::-1][1:]))   # symmetrical triangular.
        # trim to length 2535 maybe? We'll just use length a.
    # ensure nonnegative
    a = np.maximum(a, 0.0)
    # Normalize to sum=1
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    # Precompute FFT size
    L = 1 << ((3 * n - 2).bit_length())
    # Initialize optimizer state for Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    # smoothing schedule
    eps0 = 1.0
    eps_decay = 0.9992  # slightly slower; we can experiment
    # However ensure eps doesn't vanish too quickly.
    # Learning rate schedule
    lr0 = 0.05

    best_vec = a.copy()
    best_sc = _score(best_vec, L)  # need to compute conv? Provide L

    # Define functions for scoring and gradient
    # Already have _score

    # Additional helper to compute conv and max
    def compute_conv_and_max(vec):
        # compute conv via FFT
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv.max()
        return conv, max_val

    it = 0
    no_improve = 0

    # Main loop
    while time.monotonic() < deadline:
        # Decide whether to do LP step occasionally
        if it % 5000 == 0 and it > 0:  # every 5000 iterations
            # compute current M and conv
            conv, M = compute_conv_and_max(a)
            # Build sparse matrix for constraints A_ub
            # Build using vectorized approach
            # However building each LP step may be expensive, check remaining time.
            remaining = deadline - time.monotonic()
            if remaining < 5.0:
                # insufficient time for LP
                pass
            else:
                # Build A_ub
                # Use np.repeat and np.tile
                i = np.arange(n, dtype=np.int32)
                j = np.arange(n, dtype=np.int32)
                rows = np.repeat(i, n) + np.tile(j, n)
                cols = np.tile(j, n)
                data = np.repeat(a, n)
                A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
                c = -np.ones(n)  # minimize -sum(g)
                b_ub = np.full(2 * n - 1, M, dtype=np.float64)
                # Solve LP
                try:
                    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                                bounds=[(0, None)] * n,
                                                method='highs',
                                                options={'presolve': True})
                except Exception:
                    res = None
                if res is not None and res.success:
                    g = res.x
                    sum_g = g.sum()
                    if sum_g > 0:
                        g = g / sum_g
                    else:
                        g = np.full_like(a, 1.0 / n)
                    # mix
                    t_mix = 0.2
                    cand_vec = (1 - t_mix) * a + t_mix * g
                    cand_vec = _project_simplex(cand_vec)  # ensure sum=1
                    cand_sc = _score(cand_vec, L)
                    if cand_sc < best_sc:
                        best_vec, best_sc = cand_vec.copy(), cand_sc
                        a = cand_vec.copy()
                        cur_sc = cand_sc
                        # Reset Adam moments?
                        m.fill(0.0)
                        v.fill(0.0)
                        no_improve = 0
        # Gradient descent step (Adam)
        it += 1
        # compute conv and max
        conv, max_val = compute_conv_and_max(a)
        # smoothing epsilon
        eps = max(1e-7, eps0 * (eps_decay ** it))
        # compute weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum
        # compute gradient using convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        a_fft = np.fft.rfft(a, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + adam_eps)
        cand = a - step
        # projection onto simplex
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc
            a = cand.copy()
            no_improve = 0
        else:
            no_improve += 1
            # maybe occasional random mass transfer if stuck
            if no_improve > 3000:
                # random mass transfer
                i_idx = rng.integers(0, n)
                if a[i_idx] > 0:
                    delta = a[i_idx] * rng.random() * 0.4
                    j_idx = rng.integers(0, n - 1)
                    if j_idx >= i_idx:
                        j_idx += 1
                    tmp = a.copy()
                    tmp[i_idx] -= delta
                    tmp[j_idx] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp, L)
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        a = tmp.copy()
                    no_improve = 0
        # periodic symmetrization
        if it % 2500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
                a = sym.copy()
                no_improve = 0

        # early exit if achieve target
        if best_sc < 1.50274349:
            break

    # Convert best vector to list
    return [float(x) for x in best_vec]

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(vec: np.ndarray, L_fft: int) -> float:
    """Exact AC1 bound for a vector assumed to sum to 1."""
    n = vec.size
    af = np.fft.rfft(vec, n=L_fft)
    conv = np.fft.irfft(af * af, n=L_fft)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # small analytic starter (sqrt‑series tiled)
        def _sqrt_series(m):
            a = np.empty(m, dtype=np.float64)
            a[0] = 1.0
            for k in range(1, m):
                if k == 1:
                    s = 0.0
                else:
                    s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
                a[k] = (1.0 - s) / 2.0
            return a

        a0 = np.tile(_sqrt_series(64), 40)[:2535]

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                       # simplex normalisation

    n = a0.size
    # FFT length sufficient for all convolutions we need (≥3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # ---- optimisation state --------------------------------------------
    cur = a0.copy()
    cur_sc = _score(cur, L)
    best = cur.copy()
    best_sc = cur_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.05

    # smoothing schedule for the max
    eps0 = 1.0
    eps_decay = 0.9992
    it = 0
    no_improve = 0

    # parameters for periodic linear‑programming step
    lp_interval = 2000          # how many gradient iterations between LP calls
    lp_mix = 0.1                # convex‑mixing weight when LP yields a candidate

    while time.monotonic() < deadline:
        it += 1

        # ---------- convolution of current iterate ----------
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ---------- smoothed max gradient ----------
        eps = max(1e-7, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + adam_eps)
        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand, L)

        # ---------- accept/reject ----------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:

                no_improve = 0
        else:

            prob = math.exp(-(cand_sc - cur_sc) / (np.linalg.norm(step) + 1e-12))
            if rng.random() < prob:

            if time.monotonic() < deadline - 3.0:

                        v.fill(0.0)

                    if cand_lp_sc < best_sc:
                        best, best_sc = cand_lp.copy(), cand_lp_sc
                        no_improve = 0

        # ---------- symmetrisation ----------
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ---------- random mass transfer if stuck ----------
        if no_improve > 4000:
            i_idx = rng.integers(0, n)
            if cur[i_idx] > 0.0:

                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0
                else:
                    no_improve = 0

        # ---------- early exit ----------
        if best_sc < 1.50274349:
            break

    # Return the best vector found as a plain Python list
    return [float(x) for x in best]
