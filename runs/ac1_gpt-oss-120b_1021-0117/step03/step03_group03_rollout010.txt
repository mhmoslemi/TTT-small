import cvxpy as cp

def solve_g(f, M):
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0, cp.conv(f, g) <= M]  # conv length 2n-1
    # Actually cp.conv returns length (len(f)+len(g)-1) = 2n-1.
    # We need M constant for each index.
    constraints.append(cp.conv(f, g) <= M) # Already.
    # Also nonneg on all conv entries: conv <= M; this is vector inequality.

    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True) # maybe fast.

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # fallback: return uniform maybe
        return np.full(n, 1.0/n)
    g_val = g.value
    # Ensure nonneg and clip
    g_val = np.maximum(g_val, 0)
    # In case sum zero, fallback to uniform
    if g_val.sum() < 1e-12:
        g_val = np.full(n, 1.0/n)
    # Rescale to sum 1
    g_val = g_val / g_val.sum()
    return g_val

def propose_candidate(seed=42,...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5

    # init f
    try: parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        f = parent.copy()
    except:
        n = 256
        f = np.ones(n)

    n = len(f)
    # Clip to [0,1000]; ensure nonnegative
    f = np.clip(f, 0.0, 1000.0)
    # Normalize sum to 1
    s = f.sum()
    if s < 1e-12:
        f = np.full(n, 1.0/n)
    else:
        f = f / s

    # Set initial best
    best_f = f.copy()
    best_score = 2.0 * n * np.max(np.convolve(best_f, best_f))
    # CVXPY problem setup
    import cvxpy as cp
    f_param = cp.Parameter(n)
    M_param = cp.Parameter()
    g = cp.Variable(n)
    conv_expr = cp.conv(f_param, g)
    constraints = [g >= 0, conv_expr <= M_param]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # pre-solve warm_start init
    # Not needed.

    # hyperparameters
    t = 0.5   # mixing weight
    max_iter = 1000
    iteration = 0
    no_improve = 0
    max_stale = 30   # after this many iterations with no improvement, jitter
    jitter_std = 0.02
    # start loop
    while time.monotonic() < deadline and iteration < max_iter:
        iteration += 1

        # compute M
        conv_f = np.fft.irfft(np.fft.rfft(f, n=1 << ((2*n-1 - 1).bit_length())) * np.fft.rfft(f, n=1 << ((2*n-1 - 1).bit_length())))[:2*n-1]  # compute conv
        M = conv_f.max()

        # Solve LP for g
        f_param.value = f
        M_param.value = M
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)   # maybe set some limits
        except Exception as e:
            # fallback: break
            break

        if g.value is None:
            # fallback: break
            break
        g_val = np.maximum(g.value, 0.0)
        sum_g = g_val.sum()
        if sum_g < 1e-12:
            g_val = np.full(n, 1.0/n)
            sum_g = 1.0
        g_val = g_val / sum_g   # normalize

        # update f
        f = (1 - t) * f + t * g_val
        # renorm to sum=1 (should already)
        f = f / f.sum()

        # evaluate score
        cur_score = 2.0 * n * np.max(np.convolve(f, f))
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
            # optionally reduce t
            if best_score <= 1.50274349:
                break
        else:
            no_improve += 1

        # jitter if stagnation
        if no_improve >= max_stale:
            # random perturbation multiplicative jitter
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            f = np.clip(f * jitter, 0.0, 1000.0)
            f = f / f.sum()
            no_improve = 0

    return [float(v) for v in best_f]

    def max_conv(x):

        m = len(x)

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    # shift by max to improve numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv(x: np.ndarray) -> float:
    """Maximum value of the (linear) convolution of x with itself."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Uses a momentum‑accelerated sub‑gradient method with
    adaptive active‑set gradients, occasional jitter,
    and a final sparse LP refinement (if cvxpy is available).
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. Initialise sequence (fallback if the parent is unavailable)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    best = a.copy()
    best_score = 2.0 * n * _max_conv(best)

    # ------------------------------------------------------------------
    # 3. Pre‑compute FFT size for the inner loop
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 4. Optimisation parameters (gradient stage)
    # ------------------------------------------------------------------
    eta0 = 0.6          # initial step magnitude
    beta = 0.9          # momentum decay
    top_thresh = 0.9995  # fraction of max to define the active set
    jitter_every = 180   # steps between jitter
    jitter_std = 0.015   # std‑dev of multiplicative jitter
    v = np.zeros_like(a)  # momentum buffer
    it = 0
    no_improve = 0
    max_stale = 3000

    # ------------------------------------------------------------------
    # 5. Main gradient‑descent loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---------- convolution (FFT‑based) ----------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ---------- update best ----------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            no_improve += 1

        # ---------- active‑set gradient ----------
        thresh = max_val * top_thresh
        act_idx = np.where(conv >= thresh)[0]
        if act_idx.size == 0:
            act_idx = np.array([int(conv.argmax())], dtype=np.int64)

        g = np.zeros_like(a)
        for k in act_idx:
            offset = k - (n - 1)
            if offset >= 0:
                idx = np.arange(offset, n, dtype=np.int64)
                partner = k - idx
                g[idx] += a[partner]
            else:
                idx = np.arange(0, n + offset, dtype=np.int64)
                partner = k - idx
                g[idx] += a[partner]
        # each pair counted twice except the centre; factor 2 normalises
        g = 2.0 * g / max(1, act_idx.size)

        # ---------- momentum step & simplex projection ----------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v)

        # ---------- occasional jitter ----------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v[:] = 0.0
            no_improve = 0

        # ---------- long stagnation → restart ----------
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v[:] = 0.0
            no_improve = 0

    # ------------------------------------------------------------------
    # 6. Optional LP refinement (if time permits and cvxpy is present)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline - 5.0:  # keep a few seconds margin

            while time.monotonic() < deadline - 1.0 and lp_iters < 30:

                cur_M = _max_conv(best)

                if s_g < 1e-12:
                    break
                g_val = g_val / s_g

                cand = (1.0 - t_mix) * best + t_mix * g_val
                cand = cand / cand.sum()
                cand_score = 2.0 * n * _max_conv(cand)

                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                else:
                    # if no improvement, shrink the mixing weight
                    t_mix *= 0.5
                lp_iters += 1
        except Exception:
            # cvxpy unavailable or solver failed – silently skip refinement
            pass

    # ------------------------------------------------------------------
    # 7. Return the best sequence (as plain python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
