    L = p.size

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    # based on the algorithm of Wang & Carreira-Perpinán (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Gradient‑based search for a step‑function that minimizes the AC‑1 bound.
    Uses a smooth approximation of the max‑convolution (soft‑max) and
    projected gradient descent with momentum.  All operations are O(N log N)
    and keep auxiliary memory linear in the sequence length.
    """

    if parent_raw is not None:
