def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.5)
    # load parent or init
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        if s > 0:
            cur = cur / s
        else:
            cur = np.full_like(cur, 1.0/cur.size)
    else:
        n0 = 256
        cur = np.full(n0, 1.0/n0, dtype=np.float64)
    n = cur.size
    L = 1 << ((2*n-1).bit_length())
    def conv_fft(a):
        fv = np.fft.rfft(a, n=L)
        return np.fft.irfft(fv * fv, n=L)[:2*n-1]
    def score(a):
        # using objective formula (scale-invariant)
        conv = conv_fft(a)
        return float(2.0 * n * conv.max() / (a.sum()**2))
    cur = cur.astype(np.float64)
    cur_sc = score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

def _conv_fft(a, L):
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[:2*a.shape[0]-1]

    conv = _conv_fft(a, L)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution a_i * a_{k-i} for each i in the active window
        contrib = cur[idx] * cur[k - idx]
        i = int(idx[np.argmax(contrib)])
        p = k - i

        # choose a low‑value destination (different from i and p)
        order = np.argsort(cur)
        for cand_idx in order:
            if cand_idx != i and cand_idx != p:
                j = int(cand_idx)
                break
        else:
            j = (i + 1) % n
            if j == p:
                j = (j + 1) % n

        # transfer a fraction of mass from i (and possibly p) to j
        delta_i = cur[i] * step_frac * rng.random()
        cand = cur.copy()
        cand[i] -= delta_i
        added = delta_i
        if p != i:
            delta_p = cur[p] * step_frac * rng.random()
            cand[p] -= delta_p
            added += delta_p
        cand[j] += added

        cand_sc = _score(cand, L)
        # Metropolis acceptance
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break

        # scheduling
        if it % 500 == 0:
            step_frac = max(0.01, step_frac * 0.985)
            temp = max(1e-8, temp * 0.985)

        # occasional random jitter to escape plateaus
        if it % 2500 == 0 and rng.random() < 0.2:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            s = tmp.sum()
            if s > 0:
                tmp = tmp / s
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # Phase 2 – mirror / exponentiated‑gradient refinement
    # -----------------------------------------------------------------
    eta0 = 0.6
    it2 = 0
    while time.monotonic() < deadline:
        it2 += 1
        conv = _conv_fft(cur, L)
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]
        eta = eta0 / np.sqrt(it2)
        cand = cur * np.exp(-eta * g)
        s = cand.sum()
        if s > 0:
            cand = cand / s
        cand_sc = _score(cand, L)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break

        if it2 % 2000 == 0:
            # occasional random redistribution move
            nz = np.where(cur > 1e-12)[0]
            if nz.size:
                i = int(rng.choice(nz))
                j = int(rng.integers(0, n))
                if j == i:
                    j = (j + 1) % n
                delta = cur[i] * 0.3 * rng.random()
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                        if best_sc < target:
                            break

        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # Fallback (should never trigger)
    # -----------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
