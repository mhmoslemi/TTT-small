import time, math, numpy as np

def _full_self_conv(a: np.ndarray, P: int) -> np.ndarray:
    """Full linear convolution a * a using FFT length P."""
    Fa = np.fft.rfft(a, n=P)
    conv = np.fft.irfft(Fa * Fa, n=P)[: 2 * a.size - 1]
    return conv.real


def _grad_smoothed(a: np.ndarray, conv: np.ndarray, P: int, t: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximated max‑convolution.
    Returns a vector of length len(a).
    """
    n = a.size
    # soft‑max weights (stable)
    m = conv.max()
    w = np.exp(t * (conv - m))
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # convolution of w with reversed a gives the needed sums
    a_rev = a[::-1]
    Wa = np.fft.rfft(w, n=P)
    Fa_rev = np.fft.rfft(a_rev, n=P)
    corr = np.fft.irfft(Wa * Fa_rev, n=P).real  # linear correlation (no wrap‑around)
    # indices n-1 … n-1+n‑1 correspond to i = 0 … n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score used by the evaluator (sum of a is 1)."""
    return 2.0 * n * conv.max()
