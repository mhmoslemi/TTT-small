def _proj_simplex(v):
    """Project onto simplex (sum=1, nonnegative). Return projected array."""
    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0

        theta = 0
    else:

    w = np.maximum(v - theta, 0.0)

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Simple Euclidean projection onto the probability simplex:
    all entries non‑negative and sum to 1.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

    """
    Search for a non‑negative step‑function minimizing the AC1 bound.
    Uses a small parametric sweep, random Dirichlet trials and a
    momentum‑gradient refinement.
    """
    rng = np.random.default_rng(seed)

    except Exception:

            if sc < best_score:

    # ------------------------------------------------------------------

    i = np.arange(n, dtype=np.float64)
