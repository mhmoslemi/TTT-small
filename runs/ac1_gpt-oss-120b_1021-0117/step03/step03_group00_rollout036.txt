def _score(a: np.ndarray, conv: np.ndarray = None) -> float:
    """Return 2*n*max(conv) (sum(a) is assumed 1)."""
    n = a.shape[0]
    if conv is None:
        # linear convolution via FFT (full length 2n-1)
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
```

import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Exact score (assuming sum(a) = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    """

    """

    # ------------------- initialise -------------------

        a = np.maximum(a, 0.0)
    else:

    s = a.sum()
    if s <= 0.0:

    else:
        a = a / s                     # normalise to simplex

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length for gradient step

    # ------------------- bookkeeping -----------------

    best = a.copy()
    cur_sc = best_sc = _score(cur)

    # Adam parameters
