import time, numpy as np
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initial vector
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random Dirichlet: exponential distribution then normalize
        n0 = 1024
        a0 = rng.exponential(scale=1.0, size=n0)
    # enforce non-neg
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s == 0.0:
        a0.fill(1.0 / a0.size)
    else:
        a0 /= s

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())

    def conv_fft(v):
        fv = np.fft.rfft(v, n=L)
        return np.fft.irfft(fv * fv, n=L)[:2*n-1]

    def score(v):
        conv = conv_fft(v)
        return 2.0 * n * conv.max() / (v.sum() ** 2)

    def project_simplex(v):
        # Efficient O(nlogn) Euclidean projection onto simplex
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    cur = a0.copy()
    cur_sc = score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # momentum
    m = np.zeros_like(cur)
    momentum_beta = 0.9

    it = 0
    no_improve = 0
    target = 1.50274349
    # step schedule constants
    add_alpha0 = 0.8
    mul_eta0 = 0.8

    while time.monotonic() < deadline:
        it += 1
        # compute convolution and active indices (max)
        conv = conv_fft(cur)
        max_val = conv.max()
        # active indices (allow tiny tolerance)
        active = np.where(conv >= max_val - 1e-12)[0]

        # gradient of max (subgradient)
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # momentum update
        m = momentum_beta * m + (1.0 - momentum_beta) * g

        # additive candidate
        alpha = add_alpha0 / np.sqrt(it)
        cand_add = cur - alpha * m
        cand_add = project_simplex(cand_add)
        cand_add_sc = score(cand_add)

        # multiplicative (exponential) candidate
        eta = mul_eta0 / np.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        s_mul = cand_mul.sum()
        if s_mul == 0.0:
            cand_mul = np.full_like(cur, 1.0 / n)
        else:
            cand_mul = cand_mul / s_mul
        cand_mul_sc = score(cand_mul)

        # choose best candidate among cur, additive, multiplicative
        # (cur already has cur_sc)
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc

        # possibly accept worse move with small probability (Metropolis)
        else:
            # probability decays with iteration
            prob = np.exp(-(min(cand_add_sc, cand_mul_sc) - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                if cand_add_sc < cand_mul_sc:
                    cur, cur_sc = cand_add, cand_add_sc
                else:
                    cur, cur_sc = cand_mul, cand_mul_sc

        # update best
        if cur_sc < best_sc:
            best, best_sc = cur.copy(), cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # random mass transfer when stagnating
        if no_improve > 2000 and rng.random() < 0.3:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_sc = score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # jittered restart towards best periodically
        if it % 5000 == 0 and rng.random() < 0.2:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = project_simplex(tmp)
            tmp_sc = score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc

        if best_sc < target:
            break

    # final local refinement: a few random pair moves
    refin_deadline = time.monotonic() + min(0.4, deadline - time.monotonic())
    while time.monotonic() < refin_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.4
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = project_simplex(tmp)
        tmp_sc = score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best, best_sc = tmp.copy(), tmp_sc

    # return best as list
    return [float(x) for x in best]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list) with a low AC1 bound.
    The method uses projected (momentum) sub‑gradient descent on the
    maximum autocorrelation, with occasional random perturbations.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep margin

    # initialise

    else:

        a0 = rng.exponential(scale=1.0, size=1024)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s == 0.0:
        a0.fill(1.0 / a0.size)
    else:
        a0 /= s

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # ----------------------------------------------------------------------
    # helper utilities
    # ----------------------------------------------------------------------
    def _conv_fft(v: np.ndarray) -> np.ndarray:
        fv = np.fft.rfft(v, n=L)
        return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

    def _score(v: np.ndarray) -> float:
        conv = _conv_fft(v)
        return 2.0 * n * conv.max() / (v.sum() ** 2)

    def _project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto {x>=0, sum x = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    cur = a0.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    m = np.zeros_like(cur)
