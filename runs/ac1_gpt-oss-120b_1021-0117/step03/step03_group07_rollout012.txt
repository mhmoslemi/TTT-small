def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # get initial vector
    if "height_sequence_1" in globals():
        cur0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur0 = np.maximum(cur0, 0.0)
        s = cur0.sum()
        if s > 0:
            cur0 = cur0 / s
        else:
            cur0 = np.full_like(cur0, 1.0 / cur0.size)
    else:
        n = 256
        cur0 = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur0.size

    # start best
    best_vec = cur0.copy()
    best_sc = _score(best_vec)

    # search families
    # exponential
    for r in np.linspace(0.9, 0.9999, 250):
        a = np.power(r, np.arange(n, dtype=np.float64))
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a / a_sum
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    # power law
    for p in np.linspace(0.2, 4.0, 250):
        a = 1.0 / np.power(np.arange(1, n+1, dtype=np.float64), p)
        a = a / a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    # linear descending
    a = np.arange(n, 0, -1, dtype=np.float64)
    a = a / a.sum()
    sc = _score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # quadratic
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    a = a / a.sum()
    sc = _score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # sqrt series from previous fn maybe not needed, but we can.

    # set cur = best_vec
    cur = best_vec.copy()
    cur_sc = best_sc

    # gradient descent loops
    it = 0
    eta0 = 0.6
    alpha0 = 1.0

    while time.monotonic() < deadline:
        it += 1
        # compute conv and max index
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))

        # gradient for max index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # g_i = 2 * cur[k - i]
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # choose update type
        if it % 2 == 0:
            # exponentiated gradient (mirror descent)
            eta = eta0 / np.sqrt(it)
            cand = cur * np.exp(-eta * g)
            # projection onto simplex via scaling (ensuring sum=1)
            s = cand.sum()
            if s > 0:
                cand = cand / s
            else:
                cand = np.full_like(cand, 1.0 / n)
        else:
            # Euclidean proj
            alpha = alpha0 / np.sqrt(it)
            cand = cur - alpha * g
            cand = _project_simplex(cand)

        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional acceptance with temperature schedule
            T = eta0 / (np.sqrt(it) * 5.0 + 1e-12)
            if rng.random() < np.exp(-(cand_sc - cur_sc) / max(T, 1e-12)):
                cur, cur_sc = cand, cand_sc

        # occasional targeted pair moves

            # compute idx and prod for current k
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            prod = cur[idx] * cur[k - idx]
            i = int(idx[np.argmax(prod)]) if prod.size else int(rng.integers(0, n))

            low_idxs = np.argsort(cur)[: max(1, n // 100)]
            j = int(rng.choice(low_idxs))
            if j == i:
                j = (j + 1) % n

            # compute delta

import time

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]

    fv = np.fft.rfft(a, n=L)

    conv = np.fft.irfft(prod, n=L)[: 2 * n - 1]
