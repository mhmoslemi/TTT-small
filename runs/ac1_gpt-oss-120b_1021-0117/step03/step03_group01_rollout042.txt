import time
import numpy as np

# ------------------------------------------------------------
# simplex projection (L1 = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:        # degenerate safeguard
        w[:] = 1.0 / n
    return w


# ------------------------------------------------------------
# full convolution via FFT (returns the 2n‑1 vector)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# max value of the convolution and its index
# ------------------------------------------------------------
def _conv_max_and_index(a: np.ndarray, L: int):
    conv = _conv_full(a, L)
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


# ------------------------------------------------------------
# L2‑gradient of the flat‑convolution objective
# ------------------------------------------------------------
def _l2_grad(a: np.ndarray, conv: np.ndarray,
             avg: float, L2: int) -> np.ndarray:
    """
    Gradient of  sum_k (conv_k - avg)^2  w.r.t. a.
    Uses a_rev * (conv-avg) convolution.
    """
    diff = conv - avg                     # length 2n‑1
    a_rev = a[::-1]
    fa = np.fft.rfft(a_rev, n=L2)
    fd = np.fft.rfft(diff, n=L2)
    rev = np.fft.irfft(fa * fd, n=L2)[: 3 * a.size - 2]
    # slice = indices n‑1 … n‑1+n‑1  (length n)
    grad = 2.0 * rev[a.size - 1: a.size - 1 + a.size]
    return grad

        rand_vec = _project_simplex(rand_vec)
        # a short second optimisation run
        a2 = rand_vec.copy()
        momentum = np.zeros_like(a2)
        step = 0.5
        cur_max, cur_idx, cur_conv = _conv_max_and_index(a2, L)
        cur_score = 2.0 * n * cur_max
        while time.monotonic() < deadline - 0.5:
            lo = max(0, cur_idx - (n - 1))
            hi = min(n - 1, cur_idx)
            g_max = np.zeros_like(a2)
            if lo <= hi:
                idxs = np.arange(lo, hi + 1)
                g_max[idxs] = 2.0 * a2[cur_idx - idxs]

            if rng.random() < prob_l2:
                g_l2 = _l2_grad(a2, cur_conv, avg, L2)
                g = (1.0 - alpha_l2) * g_max + alpha_l2 * g_l2
            else:
                g = g_max

            momentum = beta * momentum + (1.0 - beta) * g
            cand = a2 - step * momentum
            cand = _project_simplex(cand)

            cand_max, cand_idx, cand_conv = _conv_max_and_index(cand, L)
            cand_score = 2.0 * n * cand_max

            if cand_score < cur_score:
                a2 = cand
                cur_score = cand_score
                cur_idx = cand_idx
                cur_conv = cand_conv
                if cand_score < best_score:
                    best_score = cand_score
                    best = cand.copy()
            else:
                T = step
                if rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                    a2 = cand
                    cur_score = cand_score
                    cur_idx = cand_idx
                    cur_conv = cand_conv

            step = max(min_step, step * decay)
            if time.monotonic() + 0.5 > deadline:
                break

    # ------------------------------------------------------------------
    # return the best candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]
