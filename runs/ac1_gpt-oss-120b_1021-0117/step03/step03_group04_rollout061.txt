import time
import math
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # degenerate case – return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Compact optimizer for the AC1 upper bound.
    Returns the best coefficient list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Prepare time limit
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 2. Load parent if available, otherwise fall back to uniform size
    # ------------------------------------------------------------------
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or not np.isfinite(parent_raw).all():
            raise ValueError
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
    except Exception:
        parent_raw = None

    if parent_raw is not None:
        n = int(parent_raw.size)
        a_parent = parent_raw.copy()
        s = a_parent.sum()
        if s < 0.01:
            a_parent = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a_parent = a_parent / s
    else:
        n = 256
        a_parent = None

    # ------------------------------------------------------------------
    # 3. Build a handful of initial candidates
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    tent = np.clip(tent, 0.0, None)
    if tent.sum() > 0:
        tent /= tent.sum()
    else:
        tent = uniform.copy()

    rand_vec = rng.random(n)
    rand_vec /= rand_vec.sum()

    candidates = [uniform, tent, rand_vec]
    if a_parent is not None:
        candidates.append(a_parent)

    # ------------------------------------------------------------------
    # 4. Fast AC1 bound for a vector whose sum equals 1
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # pick the best starter
    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()                      # current iterate
    v = np.zeros_like(a)                 # momentum vector

    # ------------------------------------------------------------------
    # 5. Optimisation loop
    # ------------------------------------------------------------------
    target = 1.50274349
    mu = 0.9          # momentum decay
    eta0 = 1.0        # initial step magnitude
    it = 0
