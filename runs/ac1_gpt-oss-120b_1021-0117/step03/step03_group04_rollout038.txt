def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-5.0)
    # initialize length n
    try: parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
         n = parent_raw.size
         a = np.clip(parent_raw, 0, 1000)
    except Exception: n = 256; a = np.full(n, 1.0/n)
    # ensure nonzero sum
    sum_a = a.sum()
    if sum_a < 1e-12: a = np.full(n, 1.0/n)
    # normalize to sum 1
    a = a / a.sum()
    # compute score
    fft_len = 1 << ((2*n-2).bit_length())
    def score_vec(v):
        fa = np.fft.rfft(v, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
        return 2.0 * n * conv.max()
    cur_score = score_vec(a)
    best_a = a.copy()
    best_score = cur_score
    # annealing schedule
    temp0 = 1.0  # initial temperature relative to score scale
    # choose schedule: temperature = temp0 * (1 - t/T) where T is budget proportion.
    # We'll loop while time left.
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # pick two indices i, j distinct
        i, j = rng.integers(0, n, size=2, endpoint=False)
        while i==j:
            j = rng.integers(0, n)
        # propose move: move some mass from i to j
        # delta sample from Gaussian scaled by step size.
        # step size = alpha * (1/sqrt(it))
        step = 0.01 / np.sqrt(it)  # adjust later
        delta = step * rng.standard_normal()
        # adjust to ensure a[i] - delta >= 0 and a[j] + delta >= 0
        # we keep sum constant.
        delta = np.clip(delta, -a[i], a[j])  # can't go beyond positivity.
        if abs(delta) < 1e-12:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure bounds [0,1000] though values already low, fine.
        # compute candidate score
        new_score = score_vec(a_new)
        # acceptance
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            # temperature decreasing with iteration
            # compute acceptance probability
            # scale temperature relative to cur_score magnitude
            temp = max(0.001, 0.3 * (1.0 - it / 1000000)) # adjust
            if rng.random() < np.exp(-(new_score - cur_score) / temp):
                a = a_new
                cur_score = new_score
        # occasional re-randomization if stuck

            a = a * jitter

            cur_score = score_vec(a)
    # after loop, return best_a as list of floats
    return [float(x) for x in best_a]

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimizer that minimizes the AC1 upper bound.
    Uses a projected Adam‑style update based on the most
    influential convolution shifts.  Returns the best
    coefficient list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Timing
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. Length & initial candidate(s)
    # ------------------------------------------------------------------
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0:
            raise ValueError
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
    except Exception:
        parent_raw = None

    if parent_raw is not None:
        n = int(parent_raw.size)
        a_parent = np.maximum(parent_raw, 0.0)
        s = a_parent.sum()
        a_parent = a_parent / s if s >= 0.01 else np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 256
        a_parent = None

    # Uniform / random starts
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    rand_dir = rng.random(n)
    rand_dir /= rand_dir.sum()
    candidates = [uniform, rand_dir]
    if a_parent is not None:
        candidates.append(a_parent)

    # ------------------------------------------------------------------
    # 3. Fast scoring (max of self‑convolution)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def _fast_maxconv(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return conv.max()

    # pick the best start
    best = None
    best_score = float("inf")
    for cand in candidates:
        m = _fast_maxconv(cand)
        sc = 2.0 * n * m
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    # ------------------------------------------------------------------
    # 4. Optimisation parameters (Adam‑style)
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.8          # base learning rate
    K = 15             # number of top shifts used for the sub‑gradient
    jitter_prob = 0.02
    target = 1.50274349

    m_vec = np.zeros_like(a)   # first moment
    v_vec = np.zeros_like(a)   # second moment

    it = 0
    last_improve = 0

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & current score -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = it
            if best_score <= target:
                break

        # ----- sub‑gradient from top K shifts -----
        if K < conv_len:
            top_idx = np.argpartition(-conv, K)[:K]
        else:
            top_idx = np.arange(conv_len)
        top_vals = conv[top_idx]
        wsum = top_vals.sum()
        if wsum > 0:
            weights = top_vals / wsum
        else:
            weights = np.full_like(top_vals, 1.0 / top_vals.size)

        grad = np.zeros_like(a)
        for shift, w in zip(top_idx, weights):
            offset = shift - (n - 1)
            if offset >= 0:
                idx_range = np.arange(offset, n, dtype=np.int64)
                partner = shift - idx_range
            else:
                idx_range = np.arange(0, n + offset, dtype=np.int64)
                partner = shift - idx_range
            grad[idx_range] += w * 2.0 * a[partner]

        # ----- Adam update -----
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
        m_hat = m_vec / (1.0 - beta1 ** it)
        v_hat = v_vec / (1.0 - beta2 ** it)
        step = lr0 / (np.sqrt(it) + 1.0)          # modest decay
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a = _proj_simplex(a_new)

        # ----- occasional jitter to escape plateaus -----
        if it % 250 == 0 and rng.random() < jitter_prob:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * jitter)

        # ----- restart if stuck for a long while -----
        if it - last_improve > 5000:
            # re‑initialise from a fresh random Dirichlet draw
            a = rng.random(n)
            a = _proj_simplex(a)
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            last_improve = it

    # ------------------------------------------------------------------
    # 6. Return best candidate (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
