    beta2 = 0.999
    t = 0

    eps_smooth0 = 2.0

    eps_min = 1e-6

    no_improve = 0
    while time.monotonic() < deadline:
        t += 1
        # conv of cur
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2*n-1]
        max_val = conv.max()

        w = np.exp((conv - max_val)/eps_smooth)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast score (assumes sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence whose AC1 upper bound
    (as measured by the supplied evaluator) is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------- initialise -------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple random start if no parent is present
        a0 = rng.random(2000)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # project onto simplex (sum = 1)

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length >= 2n‑1

    cur = a0.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # ------------------- optimisation state -------------------
    m = np.zeros_like(cur)          # first moment (Adam)
    v = np.zeros_like(cur)          # second moment (Adam)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    it = 0

    eps0 = 2.0          # initial smoothing for soft‑max
    eps_decay = 0.9997
    eps_min = 1e-7
    lr0 = 0.3           # base learning rate
    no_improve = 0

    # ------------------- main loop -------------------
    while time.monotonic() < deadline:
        it += 1

        # conv of current iterate
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # slice that aligns with a
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # accept if better; occasional uphill moves to escape plateaus
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (lr0 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # periodic symmetrisation (helps with the autocorrelation shape)
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        if no_improve > 4000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                cur[i] -= delta
                cur[j] += delta
                cur = _project_simplex(cur)
                cur_sc = _score(cur)
                if cur_sc < best_sc:
                    best, best_sc = cur.copy(), cur_sc
                no_improve = 0

        # early exit if we beat the target
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------
    # Return plain Python list for the evaluator
    # ------------------------------------------------------------
    return [float(x) for x in best]
