import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute the AC1 bound assuming the vector sums to 1.
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray = None) -> float:
    """Return 2*n*max(conv) for sum(a)=1."""
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                if k == 1:
                    s = 0.0
                else:
                    s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
                a[k] = (1.0 - s) / 2.0
            return a
        a0 = np.tile(_sqrt_series(64), 40)[:2535]

    a0 = np.maximum(a0, 0.0)
    total = a0.sum()

    n = a0.size
    # FFT length that comfortably accommodates a * w (length 3n‑2)
