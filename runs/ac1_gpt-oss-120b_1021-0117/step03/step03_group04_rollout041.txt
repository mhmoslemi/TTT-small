def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)
    # Determine n and parent if exists
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim ==1 and parent_raw.size>0 and np.isfinite(parent_raw).all():
            n = parent_raw.size
            parent_seq = np.clip(parent_raw, 0.0, 1000.0)
            parent_seq = np.maximum(parent_seq, 0.0)
            s = parent_seq.sum()
            if s>=0.01:
                parent_seq = parent_seq / s
            else:
                parent_seq = np.full(n, 1.0/n)
        else:
            raise ValueError
    except Exception:
        n = 256
        parent_seq = None

    # Build list of initial candidates
    init_candidates = []

    # Uniform
    init_candidates.append(np.full(n, 1.0/n))

    # Tent shape
    idx = np.arange(n, dtype=np.float64)
    # linear ramp up to mid, down after mid. For general n:
    mid = (n-1)/2.0
    tent = np.minimum(idx, n-1-idx) + 1  # Actually start from 1? We'll generate positive shape.
    # Use shape: a_i = (i+1) for i <= mid else (n-i)
    # Ensure edges zero? Actually we can set edges minimal positive. Let's generate as:
    tent = np.where(idx <= mid, idx+1, n-idx)
    # Ensure non-negative
    tent = np.maximum(tent, 0.0)
    # Normalize
    tent_sum = tent.sum()
    if tent_sum>0:
        tent = tent / tent_sum
    init_candidates.append(tent)

    # Sine shape
    sine = np.sin(np.pi * (idx + 1) / (n + 1))
    sine = np.maximum(sine, 0)
    sine_sum = sine.sum()
    if sine_sum>0:
        sine = sine / sine_sum
    init_candidates.append(sine)

    # Gaussian shape
    sigma = n / 6.0  # width such that covers many points
    gauss = np.exp(-0.5 * ((idx - (n-1)/2.0) / sigma)**2)
    gauss = np.maximum(gauss, 0)
    gauss = gauss / gauss.sum()
    init_candidates.append(gauss)

    # Random Dirichlet
    randvec = rng.random(n) + 1e-12
    randvec = randvec / randvec.sum()
    init_candidates.append(randvec)

    # Parent sequence if available
    if parent_seq is not None:
        init_candidates.append(parent_seq)

    # Evaluate initial candidates quickly to select best start
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    def fast_score_vec(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        return max_val

    best = None

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x >= 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Input must be 1‑D")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # should not happen, but fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # ------------------------------------------------------------------
    return [float(v) for v in best_overall]
