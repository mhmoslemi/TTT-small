import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = parent.shape[0]
        a = parent.copy()
    except Exception:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # 2. Baseline best candidate (uniform also considered)
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(best, best))

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = 2.0 * n * np.max(np.convolve(uniform, uniform))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 3. FFT configuration
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1                     # length of a*a
    # convolution a * w_rev needs length 3n-2, pick power‑of‑two ≥ that
    fft_len = 1 << ((3 * n - 2 - 1).bit_length())

    # ------------------------------------------------------------------
    # 4. Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    eta0 = 0.8                # initial learning‑rate factor
    beta = 0.9                # momentum decay
    jitter_every = 250        # steps between random jitter
    jitter_std = 0.02         # std‑dev of multiplicative log‑normal jitter
    max_stale = 4000          # steps without improvement before restart
    target_score = 1.50274349
    alpha = 200.0             # temperature for soft‑max approximation of max

    v = np.zeros_like(a)      # momentum buffer
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # ---- 4.1 convolution a*a (full linear convolution) ----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_c = conv.max()
        cur_score = 2.0 * n * max_c

        # ---- 4.2 record best solution ----
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ---- 4.3 smooth‑max weights ----
        # w_i = exp(alpha * (conv_i - max_c))   (max_c subtracted for stability)
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        # normalized weights (softmax)
        w_norm = w / w_sum

        # ---- 4.4 gradient of smoothed max ----
        # g = 2 * sum_k w_k * a shifted by k   => can be obtained by convolution
        w_rev = w_norm[::-1]                     # reversed weight vector
        fw = np.fft.rfft(w_rev, n=fft_len)
        conv_aw = np.fft.irfft(fa * fw, n=fft_len)[: (n + conv_len - 1)]
        # indices to extract gradient: (2n-2) - i  for i=0..n-1
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * conv_aw[idx]

        # ---- 4.5 momentum step and projection onto simplex ----
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v)

        # ---- 4.6 occasional stochastic jitter ----
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v.fill(0.0)
            no_improve = 0

        # ---- 4.7 restart if stagnated ----
        if no_improve >= max_stale:
            a = best.copy()
            pert = np.exp(rng.normal(0.0, 0.01, size=n))
            a = _proj_simplex(a * pert)
            v.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # 5. Return the best sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(x) for x in best]
