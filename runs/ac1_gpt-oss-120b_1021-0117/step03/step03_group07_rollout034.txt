def _conv_fft(a):

  n = a.shape[0]

  L = 1 << ((2*n-1).bit_length())

  # compute real FFT of a padded to L

  fv = np.fft.rfft(a, n=L)

  conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]

  return conv

def _project_simplex(v):

    u = np.sort(v)[::-1]

    css = np.cumsum(u) - 1

    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (fast)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2*n*max(conv)/sum(a)^2 ; here sum(a)≈1 after projection."""
    n = a.shape[0]
    conv = _conv_fft(a)
    maxb = conv.max()
    s = a.sum()
    if s == 0:
        return float("inf")
    return 2.0 * n * maxb / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ------------------------------------------------------------------
    # Base length and initial vector (parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 2500
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate and possibly update the best one
    # ------------------------------------------------------------------
    def try_candidate(vec):
        nonlocal best_vec, best_sc
        sc = _score(vec)
        if sc < best_sc:
            best_sc, best_vec = sc, vec.copy()
        return sc

    # ------------------------------------------------------------------
    # Exhaustive parametric search (cheap families)
    # ------------------------------------------------------------------
    # Uniform (already covered by cur if parent is uniform)
    try_candidate(np.full(n, 1.0 / n, dtype=np.float64))

    # Exponential families a_i ∝ r^i
    for delta in np.logspace(-6, -1, 120):  # r = 1 - delta
        r = 1.0 - delta
        if r <= 0 or r >= 1:
            continue
        a = np.power(r, np.arange(n, dtype=np.float64))
        a /= a.sum()
        try_candidate(a)
        try_candidate(a[::-1])          # reversed version

    # Power‑law families a_i ∝ 1/(i+1)^p
    for p in np.linspace(0.01, 6.0, 150):
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / np.power(idx, p)
        a /= a.sum()
        try_candidate(a)
        try_candidate(a[::-1])

    # Stretched exponential a_i ∝ exp(-α * i^β)
    betas = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]
    for beta in betas:
        # target value at the last index varies from tiny to moderate
        for r_last in np.logspace(-4, -0.1, 80):
            # α solves exp(-α*(n-1)^β) = r_last
            alpha = -np.log(r_last) / ((n - 1) ** beta)
            i = np.arange(n, dtype=np.float64)
            a = np.exp(-alpha * (i ** beta))
            a_sum = a.sum()
            if a_sum == 0:

            a /= a_sum
            try_candidate(a)
            try_candidate(a[::-1])

    mu = (n - 1) / 2.0
    for sigma_frac in np.linspace(0.1, 0.9, 30):
        sigma = sigma_frac * n
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
        a /= a.sum()
        try_candidate(a)
        try_candidate(a[::-1])

    for _ in range(8):

        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        try_candidate(a)

    # ------------------------------------------------------------------
    # Gradient‑based refinement (Adam with simplex projection)
    # ------------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.6

    while time.monotonic() < deadline:
        it += 1
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution term
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size decays like 1/√t
        step = (lr0 / np.sqrt(it)) * (m_hat / (np.sqrt(v_hat) + eps))

        # update and project back to simplex
        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

            if best_sc < target:
                break

        # occasional jitter to escape plateaus
        if it % 3000 == 0 and rng.random() < 0.15:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
                cur, cur_sc = tmp, tmp_sc

        # time check
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final safeguard: if something went wrong return a simple uniform vector
    # ------------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
