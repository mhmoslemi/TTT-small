def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initial vector a0
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple analytic: maybe parabola shape: a_i = (i+1)*(n-i) ... not sure
        n = 2535
        idx = np.arange(n)
        a0 = (idx+1)*(n-idx)  # triangular, symmetric but not controlling sum
    # ensure non-neg
    a0 = np.maximum(a0, 0)
    # normalize sum to 1
    s = a0.sum()
    if s <= 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s

    # enforce symmetry
    a0 = (a0 + a0[::-1]) * 0.5

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())
    # Pre-allocate arrays for momentum
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)
    beta1 = 0.9
    beta2 = 0.999
    epsilon_opt = 1e-8

    # parameters
    lr0 = 0.5  # base learning rate
    eps0 = 0.2  # initial smoothing epsilon
    eps_decay = 0.9997
    eps_min = 1e-7

    # best
    best_vec = a0.copy()
    best_sc = float('inf')
    # compute initial score
    af = np.fft.rfft(best_vec, n=L)
    conv0 = np.fft.irfft(af * af, n=L)[:2*n-1]
    best_sc = 2.0 * n * conv0.max()
    cur = a0
    cur_sc = best_sc
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # compute convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()

        # smoothing epsilon
        eps = max(eps_min, eps0 * (eps_decay ** it))

        # compute softmax weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient: convolve a with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # extract indices for gradient length n
        # conv of a (len n) with w_rev (len 2n-1) yields length n+2n-2 = 3n-2
        # But we used FFT length L >= 3n-2.
        # The slice corresponding to positions (n-1) to (2n-2) inclusive gives length n.
        start_idx = n - 1
        end_idx = start_idx + n
        grad_slice = grad_full[start_idx:end_idx]
        g = 2.0 * grad_slice

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Simple score (2*n*max(conv)) for a vector whose sum is 1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

def _lp_step(f: np.ndarray, M: float, t: float, L: int):
    """Find a better partner vector via a tiny linear program."""
    try:
        import cvxpy as cp
    except Exception:
        return None
