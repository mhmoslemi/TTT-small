        idxs = np.where(np.isclose(conv, max_val, atol=1e-12))[0]   # may use =

        for t in idxs:

            start = max(0, t - (n-1))
            end = min(n-1, t)
            # slice indices = np.arange(start, end+1)
            idx_range = np.arange(start, end+1)
            # compute contributions
            # a[t - idx_range] indexes reverse slice
            g[idx_range] += 2.0 * a[t - idx_range]

        # average if multiple shifts
        if idxs.size > 1:

        # momentum
        v = mu * v + (1 - mu) * g

        # step size
        eta = eta0 / np.sqrt(it)

        # update and project
        a = _proj_simplex(a - eta * v)

        if it % 250 == 0:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))

    return best, best_score

    rng = np.random.default_rng(seed)

    # Get parent if available

    except Exception:

    if parent_raw is not None and parent_raw.ndim == 1 and parent_raw.size > 0 and np.isfinite(parent_raw).all():

        if s >= 0.01:

        else:

    else:
        n = 256
        a_parent = None

import time
import numpy as np


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def _max_conv_score(a: np.ndarray) -> float:
    """Score = 2·n·max(convolution) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    return 2.0 * n * conv.max()


def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max (conv) w.r.t. a (average over all maximal shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    eps = max(1e-12, max_val * 1e-12)
    idxs = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for t in idxs:
        start_i = max(0, t - (n - 1))
        end_i = min(n - 1, t)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        g[i_range] += 2.0 * a[t - i_range]
    if idxs.size > 1:
        g /= idxs.size
    return g
