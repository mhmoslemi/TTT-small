def _project_simplex(v):
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    pos = np.where(rho_arr > 0)[0]
    if pos.size:
        rho = pos[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def _project_simplex(v: np.ndarray) -> np.ndarray:

    pos = np.where(rho_arr > 0)[0]
    if pos.size:
        rho = pos[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0

        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_f = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_f * cand_f, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_cand.max()

        improved = False
        if cand_score < cur_score - 1e-14:
            a = cand
            cur_score = cand_score
            improved = True
            step = min(step * 1.01, 1.0)               # modest step increase
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                no_improve = 0
        else:
            # occasional uphill acceptance (simulated annealing flavour)
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                improved = True

        # ----- random mass‑transfer perturbation -----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                cand2_f = np.fft.rfft(cand2, n=L)
                conv2 = np.fft.irfft(cand2_f * cand2_f, n=L)[: 2 * n - 1]
                cand2_score = 2.0 * n * conv2.max()
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                        no_improve = 0

        # ----- step decay -----
        step = max(min_step, step * decay)

        # ----- restart if stuck -----
        if no_improve >= max_no_improve:
            a = best.copy()
            a = a + rng.random(n) * 1e-3                # tiny noise to break symmetry
            a = _project_simplex(a)
            step = 0.4
            no_improve = 0

        # ----- early exit safeguard -----
        if time.monotonic() + 0.5 > deadline:
            break

    # Return the best sequence encountered as a plain Python list
    return [float(x) for x in best]
