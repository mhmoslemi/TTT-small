def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution x*x using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]

def _conv_fft_two(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:

            cur_fft = np.fft.rfft(cur, n=L)
            cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
            cur_sc = 2.0 * n * cur_conv.max()
            no_improve = 0

        # ----- temperature schedule -----
        beta = min(max_beta, beta * beta_inc)

    return [float(v) for v in best]
