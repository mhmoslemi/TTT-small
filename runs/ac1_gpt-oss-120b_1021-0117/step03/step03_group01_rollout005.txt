   deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

   n = None
   if 'height_sequence_1' in globals():
        a_initial = np.asarray(height_sequence_1, dtype=np.float64)

            best = c.copy()

   momentum = np.zeros_like(a)

   prob_smooth = 0.01

                    cur_score = sc

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the non‑negative simplex of sum 1."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_candidates = np.where(rho_arr > 0)[0]
    if rho_candidates.size:
        rho = rho_candidates[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # Guard against a zero vector (should not happen)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# ------------------------------------------------------------
# Helper: convolution via FFT, returning full array, max value, argmax
# ------------------------------------------------------------
def _conv_full(vec: np.ndarray, L: int) -> np.ndarray:
    """Return the linear convolution of vec with itself (length 2*len-1)."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    return conv

def _conv_max_and_arr(vec: np.ndarray, L: int):
    conv = _conv_full(vec, L)
    idx = int(conv.argmax())
    return conv, float(conv[idx]), idx

# ------------------------------------------------------------
# Helper: sqrt series (a*a = 1+…+x^{2n‑2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# Core optimizer

def _optimize(a_start: np.ndarray, deadline: float, rng: np.random.Generator,
              target: float):
    """Run a gradient‑based search from a_start, respecting the deadline."""
    a = a_start.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑2 FFT length
    # initial score
    conv, max_val, _ = _conv_max_and_arr(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # hyper‑parameters
    step = 0.25
    decay = 0.9998
    beta = 0.95
    prob_random = 0.04
    prob_smooth = 0.01
    max_transfer_frac = 0.5
    momentum = np.zeros_like(a)

    while time.monotonic() < deadline:
        # ---- gradient from all (near) maximal peaks ----
        conv, max_val, _ = _conv_max_and_arr(a, L)
        tol = max_val * 1e-12 + 1e-15
        peaks = np.where(conv >= max_val - tol)[0]

        g = np.zeros_like(a)
        for p in peaks:
            lo = max(0, p - (n - 1))
            hi = min(p, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] += 2.0 * a[p - idx]

        # ---- momentum update & candidate ----
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        conv_c, max_val_c, _ = _conv_max_and_arr(cand, L)
        cand_score = 2.0 * n * max_val_c

        # ---- accept if improvement or via simulated annealing ----
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (step + 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                if best_score <= target:
                    break

        # ---- occasional random mass transfer ----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:

                if j >= i:

                delta = a[i] * rng.random() * max_transfer_frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)

            if score_s < cur_score:

    TARGET_SCORE = 1.50274349

    best_vec = None
    best_score = np.inf
