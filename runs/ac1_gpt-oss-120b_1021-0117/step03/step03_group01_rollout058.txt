import time
import math
import numpy as np

# --------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:                      # safety fallback
        w[:] = 1.0 / n
    return w


# --------------------------------------------------------------
# Convolution (FFT) – return maximal value and its index
# --------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    """Return (max value, index of max) of full convolution vec*vec."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# --------------------------------------------------------------
# Geometric decreasing start (sum = 1)
# --------------------------------------------------------------
def _geometric_start(n: int, r: float = 0.999):
    """Return a normalized geometric sequence a_i = C * r^i, sum = 1."""
    powers = r ** np.arange(n, dtype=np.float64)
    C = (1.0 - r) / (1.0 - r ** n)
    return C * powers

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- initialise ----------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:

        a0 = np.ones(256, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # ---- geometric start ------------------------------------------------
    n = a0.size
    a_geo = _geometric_start(n, r=0.999)
    a_geo = _project_simplex(a_geo)

    # ---- main optimisation on first start --------------------------------
    best, best_score = _optimize(a0, deadline, rng)

    # ---- optional second start if time permits -------------------------
    now = time.monotonic()
    if now + 5.0 < deadline:
        alt_best, alt_score = _optimize(a_geo, deadline, rng)

        if alt2_score < best_score:
            best, best_score = alt2_best, alt2_score

    # ---- return as plain Python list ------------------------------------
