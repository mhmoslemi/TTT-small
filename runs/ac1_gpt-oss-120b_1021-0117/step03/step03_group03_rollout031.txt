import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    if v.ndim != 1:
        v = v.ravel()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # fallback uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # ensure sum is exactly 1 (numerical)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Returns a non‑negative sequence (as list of floats) that aims to
    minimise 2·n·max(conv(a,a))/sum(a)^2.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 2. Base sequence (fallback if the parent is unavailable)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = int(parent.size)
        a = np.clip(parent, 0.0, 1000.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s                     # normalise to sum = 1
    except Exception:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3. Initialise best candidate
    # ------------------------------------------------------------------
    best = a.copy()
    # direct objective (sum = 1)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(best, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best_score = float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 0.7                 # initial step magnitude
    beta = 0.9                 # momentum factor
    top_k = max(5, int(n * 0.025))   # number of leading convolution peaks
    jitter_every = 180        # steps between stochastic perturbations
    jitter_std = 0.03         # log‑normal jitter std‑dev
    target_score = 1.50274349
    v_buf = np.zeros_like(a)   # momentum buffer
    it = 0
    no_improve = 0
    max_stale = 2500

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---------- convolution (FFT) ----------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val  # sum(a)=1

        # ---------- best‑so‑far ----------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ---------- sub‑gradient from top‑k peaks ----------
        if top_k < conv.size:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size)
        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                i = np.arange(offset, n, dtype=np.int64)
                partner = k - i
                g[i] += a[partner]
            else:
                i = np.arange(0, n + offset, dtype=np.int64)
                partner = k - i
                g[i] += a[partner]
        if len(idxs):
            g = 2.0 * g / len(idxs)

        # ---------- momentum step & projection ----------
        it += 1
        eta = eta0 / np.sqrt(it)
        v_buf = beta * v_buf + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v_buf)

        # ---------- occasional stochastic jitter ----------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v_buf[:] = 0.0
            no_improve = 0

        # ---------- restart if stagnated ----------
        if no_improve >= max_stale:
            a = best.copy()
            pert = np.exp(rng.normal(0.0, 0.05, size=n))
            a = _proj_simplex(a * pert)
            v_buf[:] = 0.0
            no_improve = 0

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
