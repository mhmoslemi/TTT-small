def propose_candidate(seed=42, budget_s=1100, **kwargs):

    import time, numpy as np
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 0.5)

    rng = np.random.default_rng(seed)

    # init a
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = np.ones(1, dtype=np.float64) # fallback

    # project to simplex sum=1
    a = _project_simplex(a / max(a.sum(), 1e-16))

    n = a.size
    L = 1 << ((2*n-1).bit_length())

    # Compute initial score
    conv = _conv_full(a, L)  # full conv
    max_b = conv.max()
    cur_score = 2 * n * max_b  # since sum=1 after projection
    best = a.copy()
    best_score = cur_score

    # optimization parameters
    step = 0.2
    decay = 0.9996
    min_step = 1e-8
    grad_eps = 1e-12   # tolerance for active constraints
    max_no_improve = 250
    no_improve = 0
    iter_count = 0

    # Adam parameters
    beta1=0.9
    beta2=0.999
    eps=1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # main loop
    while time.monotonic() < deadline:
        # compute conv & max
        conv = _conv_full(a, L)
        M = conv.max()
        # compute gradient subgradient
        tol = max(grad_eps, M * 1e-9) # relative tolerance
        mask = conv >= M - tol
        idxs = np.nonzero(mask)[0]
        # compute subgradient as average over idxs
        g = np.zeros_like(a)
        for idx in idxs:
            lo = max(0, idx - (n-1))
            hi = min(n-1, idx)
            js = np.arange(lo, hi+1, dtype=int)
            # add 2*a[idx - js]
            g[js] += 2.0 * a[idx - js]
        # average if multiple active indexes
        if idxs.size > 1:
            g /= idxs.size
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** (iter_count + 1))
        v_hat = v / (1 - beta2 ** (iter_count + 1))
        update = step * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - update
        # project onto simplex
        cand = _project_simplex(cand)

        # evaluate candidate
        conv_c = _conv_full(cand, L)
        max_c = conv_c.max()
        cand_score = 2 * n * max_c
        improved = False

        if cand_score < cur_score - 1e-14:
            a = cand
            cur_score = cand_score
            improved = True
        else:
            # occasional uphill acceptance
            # compute temperature proportional to step?
            # Accept with prob exp(-(cand_score-cur_score)/temp)
            temp = max(step, 1e-6)
            if rng.random() < np.exp(-(cand_score - cur_score) / temp):
                a = cand
                cur_score = cand_score
                improved = True

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            no_improve = 0
        else:
            if not improved:
                no_improve += 1
            else:
                no_improve = 0

        # random mass-transfer perturbation
        if rng.random() < 0.03:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.5
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                # proj
                cand2 = _project_simplex(cand2)
                conv2 = _conv_full(cand2, L)
                max2 = conv2.max()
                cand2_score = 2 * n * max2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                        no_improve = 0

        # step decay
        step = max(min_step, step * decay)

        # restart if stuck
        if no_improve >= max_no_improve:
            # perturb current best slightly
            a = best + rng.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            m.fill(0.0); v.fill(0.0)
            step = 0.2
            no_improve = 0

        iter_count += 1

        # if near deadline, break
        if time.monotonic() + 0.5 > deadline:
            break

    # return best as list
    return [float(x) for x in best]

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    # make a copy to avoid modifying the original
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # renormalise explicitly to avoid tiny drift
    w_sum = w.sum()
    if w_sum == 0:
        return w
    return w / w_sum

# ------------------------------------------------------------
# Full linear convolution via FFT (returns full array)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, length: int) -> np.ndarray:
    """Full (zero‑padded) convolution of a with itself using FFT."""
    fv = np.fft.rfft(a, n=length)
    conv = np.fft.irfft(fv * fv, n=length)
    return conv[: 2 * a.size - 1]

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # -------------------- initialise --------------------
    if 'height_sequence_1' in globals():

    else:
        # simple fallback: decreasing geometric series, then simplex‑projected
        n_fallback = 256
        a = np.exp(-np.arange(n_fallback, dtype=np.float64) / 20.0)
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two FFT length

    # initial score (sum already = 1)
    conv = _conv_full(a, L)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------
    step = 0.35                     # base learning‑rate for Adam
    min_step = 1e-6
    decay = 0.9999                  # multiplicative decay of the base step
    beta1, beta2 = 0.9, 0.999       # Adam moments
    eps = 1e-8
    m = np.zeros_like(a)            # first‑moment accumulator
    v = np.zeros_like(a)            # second‑moment accumulator
    it = 0

    prob_random = 0.015             # chance for a large random transfer
