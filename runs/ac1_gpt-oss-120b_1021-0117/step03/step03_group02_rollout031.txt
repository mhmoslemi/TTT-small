import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Convolution via FFT (full, returns max and the whole array)
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    """Return (max(convolution), full convolution) where length = 2*len(a)-1."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv


# ------------------------------------------------------------
# Gradient of the weighted max (soft‑max weight w) w.r.t. a
# ------------------------------------------------------------
def _grad_wrt_a(a: np.ndarray, w: np.ndarray, Lg: int):
    """Compute 2 * cross‑correlation of a with weight w (size 2n‑1)."""
    # reverse a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=Lg)
    fw = np.fft.rfft(w, n=Lg)
    conv = np.fft.irfft(fa_rev * fw, n=Lg)
    n = a.size
    # slice corresponds to indices n-1 … n-1+n-1
    g = 2.0 * conv[n - 1 : n - 1 + n].real
    return g

# Main entry point
# ------------------------------------------------------------

    start = time.monotonic()
    # leave a small safety margin for finalisation

    # -------------------- initialise --------------------

        idx = np.arange(n0, dtype=np.float64)

    if a.size == 0:

    # project onto simplex (sum = 1, non‑negative)

    n = a.size
    # FFT lengths

    def score_from_max(max_val):
        # sum is 1 after projection, so denominator = 1
        return 2.0 * n * max_val

    best = a.copy()

    # -------------------- optimisation parameters --------------------

    decay = 0.9998
