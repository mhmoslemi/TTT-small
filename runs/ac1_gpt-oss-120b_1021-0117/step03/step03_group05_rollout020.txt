import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(v: np.ndarray) -> np.ndarray:
    """Full linear convolution v * v using an FFT of length power‑of‑two."""
    n = v.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(v: np.ndarray) -> float:
    """AC1 bound for a non‑negative vector v (assume sum(v)=1)."""
    n = v.size
    maxc = _conv_fft(v).max()
    return 2.0 * n * maxc


def _gradient(v: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_{k∈active} (v*v)[k] w.r.t. v."""
    n = v.size

        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * v[k - idx]
    return g
