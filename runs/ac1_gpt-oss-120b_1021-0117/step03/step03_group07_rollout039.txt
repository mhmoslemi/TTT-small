def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum need not be 1)."""
    n = a.shape[0]

    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

            cand[hi_idx] -= delta
            cand[lo_idx] += delta
            sc = _score(cand)
            if sc < cur_sc:
                cur, cur_sc = cand, sc
                if sc < best_sc:
                    best_sc, best_vec = sc, cand.copy()
                    if best_sc < target:
                        break

        # small jitter to avoid stagnation
        if it % 1500 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            cand = cur + jitter
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand /= s
            else:
                cand = cur.copy()
            sc = _score(cand)
            if sc < cur_sc:
                cur, cur_sc = cand, sc
                if sc < best_sc:
                    best_sc, best_vec = sc, cand.copy()
                    if best_sc < target:
                        break

    # final safety fallback (should never be needed)
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
