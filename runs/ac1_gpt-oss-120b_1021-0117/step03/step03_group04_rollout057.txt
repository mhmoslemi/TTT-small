import time
import math
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    if v.ndim != 1:
        v = v.ravel()
    # Shift by median to improve numeric stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the target bound.

    The routine runs a small linear‑programming based iteration:
    given a current sequence a (∑a = 1) it solves
        max  Σ g_j    s.t.  g ≥ 0 ,  a * g ≤ M,
    where M = max(a * a) and * is linear convolution.
    The solution g is mixed back into a and projected onto the simplex.
    Repeating this quickly drives down the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Choose a working length and initialise a
    # ------------------------------------------------------------------
    max_n = 512                     # size kept modest for speed + memory
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        if raw.ndim != 1 or raw.size == 0 or not np.isfinite(raw).all():
            raise ValueError
        raw = np.clip(raw, 0.0, 1000.0)
        n = min(int(raw.size), max_n)
        a = np.maximum(raw[:n], 0.0)
        s = a.sum()
        a = a / s if s >= 0.01 else np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        # fall back to a simple decreasing exponential start
        n = max_n
        r0 = 0.5
        a = np.power(r0, np.arange(n, dtype=np.float64))
        a = a / a.sum()

    # ------------------------------------------------------------------
    # 2. Helpers
    # ------------------------------------------------------------------
    def bound_of(seq: np.ndarray) -> float:
        """2 * n * max(conv(seq, seq))  (sum(seq) == 1)."""
        conv = np.convolve(seq, seq)
        return 2.0 * n * conv.max()

    best = a.copy()
    best_score = bound_of(best)

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (LP + simplex mixing)
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # current convolution bound
        conv = np.convolve(a, a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= 1.50274349:
                break

        # ------------------------------------------------------------------
        # Build the convolution matrix F (a * g <= M)
        # ------------------------------------------------------------------
        i = np.arange(n, dtype=np.int64)
        j = np.arange(n, dtype=np.int64)
        row = (i[:, None] + j[None, :]).ravel()          # shape (n*n,)
        col = np.tile(j, n)                              # shape (n*n,)
        data = np.repeat(a.astype(np.float64), n)        # each a_i repeated n times
        F = sp.csr_matrix((data, (row, col)), shape=(2 * n - 1, n))

        # ------------------------------------------------------------------
        # Linear programme: maximise sum(g)  s.t.  g>=0,  F@g <= M
        # ------------------------------------------------------------------
        g = cp.Variable(n)
        constraints = [g >= 0, F @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                       abstol=1e-9, reltol=1e-9, feastol=1e-9,
                       verbose=False)
        except Exception:
            break                     # solver failure – exit early

        if g.value is None:
            break                     # infeasible or aborted

        g_vec = np.array(g.value, dtype=np.float64).ravel()
        g_vec = np.maximum(g_vec, 0.0)
        sum_g = g_vec.sum()
        if sum_g == 0.0:
            break

        # normalise to keep sum = 1 (simplex)
        g_norm = g_vec / sum_g

        # simple mixing (step size decays with iterations)
        eta = 0.3 / math.sqrt(it)        # decreasing step size
        a = _proj_simplex((1.0 - eta) * a + eta * g_norm)

        # occasional jitter to escape tiny plateaus
        if it % 15 == 0:
            jitter = np.exp(rng.normal(0.0, 0.012, size=n))
            a = _proj_simplex(a * jitter)

    # ------------------------------------------------------------------
    # 4. Return best found sequence as plain Python floats
    # ------------------------------------------------------------------
    best = _proj_simplex(best)          # safety projection
    return [float(v) for v in best]
