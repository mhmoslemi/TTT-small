    max_c = conv.max()
    # softmax weights
    w = np.exp(beta * (conv - max_c))
    sum_w = w.sum()
    if sum_w == 0 or not np.isfinite(sum_w):
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / sum_w
    # reversed p
    p_rev = p[::-1]
    # correlation via convolution
    A_fft = np.fft.rfft(a, n=L)
    P_rev_fft = np.fft.rfft(p_rev, n=L)

    # pick entries
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * corr[idx]
    return g

    cur_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
    cur_sc = 2.0 * n * conv_full.max() / (cur_sum * cur_sum)

    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    it = 0
    no_improve = 0
    alpha0 = 1.0          # initial step size factor for exponentiated update
    beta0 = 2.0          # start softmax temperature
    beta_max = 30.0      # final softmax temperature
    eps_move = 0.25      # max fraction of mass shifted in a random move
    jitter_scale = 3e-4  # scale of Gaussian jitter for restarts
    temp_factor = 0.1    # temperature for occasional uphill moves

    while time.monotonic() < deadline:
        it += 1

        # ---------------------------------------------------------
        # Current convolution and smooth max‑gradient
        # ---------------------------------------------------------
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_c = conv_full.max()

        # increasing softmax temperature
        beta = min(beta_max, beta0 + 1e-4 * it)

        # softmax weights over convolution entries
        w = np.exp(beta * (conv_full - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            p = np.full_like(w, 1.0 / w.size)
        else:
            p = w / w_sum                     # probabilities, sum to 1

        # correlation‑based gradient (g_i = 2 * Σ_j cur_j * p_{i+j})
        p_rev = p[::-1]
        p_rev_fft = np.fft.rfft(p_rev, n=L)
        corr_full = np.fft.irfft(cur_fft * p_rev_fft, n=L)
        g = 2.0 * corr_full[idx_corr]        # shape (n,)

        # ---------------------------------------------------------
        # Exponentiated (mirror) step
        # ---------------------------------------------------------
        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum <= 0.0 or not np.isfinite(cand_sum):
            cand = np.full_like(cur, 1.0 / n)
        else:
            cand = cand / cand_sum

        # score of candidate
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * cand_conv.max() / (cand_sum * cand_sum)

        # ---------------------------------------------------------
        # Acceptance (deterministic improvement or Metropolis)
        # ---------------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (temp_factor + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # update global best
        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()
            no_improve = 0

        # ---------------------------------------------------------
        # Random mass‑transfer move
        # ---------------------------------------------------------
        if rng.random() < 0.12:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * eps_move
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp_sum = tmp.sum()
                if tmp_sum > 0.0:
                    tmp = tmp / tmp_sum
                    tmp_fft = np.fft.rfft(tmp, n=L)
                    tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                    tmp_sc = 2.0 * n * tmp_conv.max() / (tmp_sum * tmp_sum)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best = tmp_sc, tmp.copy()

        # ---------------------------------------------------------
        # Jittered restart from the best point (periodic)
        # ---------------------------------------------------------
        if it % 2500 == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0.0:
                tmp = tmp / tmp_sum
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max() / (tmp_sum * tmp_sum)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()
                no_improve = 0

        # ---------------------------------------------------------
        # Stagnation restart
        # ---------------------------------------------------------
        if no_improve > 15000:
            cur = best + rng.normal(scale=jitter_scale * 5.0, size=n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_fft = np.fft.rfft(cur, n=L)
            cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
            cur_sc = 2.0 * n * cur_conv.max() / (cur.sum() * cur.sum())
            no_improve = 0

        # ---------------------------------------------------------
        # Early exit if target reached
        # ---------------------------------------------------------
        if best_sc < 1.50274349:
            break

    # Return as plain Python list of floats
    return [float(x) for x in best]
