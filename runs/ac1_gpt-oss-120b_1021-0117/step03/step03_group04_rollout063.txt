def propose_candidate(seed=42,budget_s=1100,**kwargs):

    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # leaves headroom.

    # Get n & parent:
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0 or not np.isfinite(parent_raw).all():
            raise ValueError
        n = int(parent_raw.size)
        parent = np.clip(parent_raw, 0.0, 1000.0)
        # normalize to sum 1
        s = parent.sum()
        if s < 0.01:
            parent = np.full(n, 1./n)
        else:
            parent = parent / s
    except Exception:
        n = 256
        parent = None

    # candidate generation
    candidates = []

    # Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Tent shape:
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1, n - idx) - 1
    tent = np.maximum(tent, 0.0)
    s = tent.sum()
    if s > 0:
        candidates.append(tent / s)

    # Gaussian shapes:
    mids = (n-1)/2.0
    for sigma_frac in [0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
        sigma = sigma_frac * n
        gauss = np.exp(-0.5 * ((idx - mids) / sigma)**2)
        # zero edges automatically small but keep >0.
        gauss_sum = gauss.sum()
        if gauss_sum > 0:
            candidates.append(gauss / gauss_sum)

    # Power shape: a_i = (i*(n-1-i))**p  (zero at ends)
    for p in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
        power = (idx * (n-1 - idx))**p
        # set edges to zero (already zero due to factor i=0 or n-1-i=0)
        s = power.sum()
        if s > 0:
            candidates.append(power / s)

    # Beta shape: (i/(n-1))**alpha * (1 - i/(n-1))**alpha
    norm_idx = idx / (n-1)
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        beta = (norm_idx**alpha) * ((1 - norm_idx)**alpha)
        s = beta.sum()
        if s > 0:
            candidates.append(beta / s)

    # Sine shape
    for q in [1.0, 2.0, 3.0]:
        sin_shape = np.sin(np.pi * (idx + 0.5) / n)**q
        s = sin_shape.sum()
        if s > 0:
            candidates.append(sin_shape / s)

    # Random Dirichlet
    rand = rng.random(n)
    rand = np.maximum(rand, 0.0)
    if rand.sum() > 0:
        candidates.append(rand / rand.sum())

    if parent is not None:
        candidates.append(parent)

    # Helper to compute fast AC1 (score) via FFT
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x):
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        # compute bound
        return float(2.0 * n * conv.max() / (x.sum()**2))

    best_a = None
    best_score = float('inf')
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    # subgradient refinement
    # Use projection onto simplex
    # Pre-compute FFT length for subsequent loops.

    # Subgradient step parameters:
    step0 = 1.0
    it = 0

    # hold best found during refinement
    while time.monotonic() < deadline:
        it += 1
        # compute conv, etc.

import time

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
