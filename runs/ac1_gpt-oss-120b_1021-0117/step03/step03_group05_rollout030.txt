        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total

    n = a.size

    # precompute FFT length: next power of two >= 2*n
    fft_len = 1 << ((2 * n - 1).bit_length())

    def conv_fft(v):
        fv = np.fft.rfft(v, n=fft_len)
        conv = np.fft.irfft(fv * fv, n=fft_len)[:2 * n - 1]
        return conv

        # conv is array of length 2*n-1
        # bound = 2*n * max(conv) / (sum**2), where sum = 1 (we keep normalized)
        return 2.0 * n * conv.max()  # since sum=1

    def project_simplex(v):
        # Euclidean projection onto simplex sum=1, v>=0
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    # initial score
    conv = conv_fft(a)
    cur_score = score_from_conv(conv)
    best_vec = a.copy()
    best_score = cur_score

    # set parameters
    it = 0
