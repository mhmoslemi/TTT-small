import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    # shift by mean to improve numerical stability
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # renormalise explicitly to avoid tiny drift
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w

    while time.monotonic() < deadline:
        # ---- full autocorrelation (FFT) ---------------------------------
        a_f = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_f * a_f, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- soft‑max weights (smooth approximation of max) ------------
        shift = conv_full - max_val                # ≤ 0
        exp_vals = np.exp(beta * shift)           # stable, max=1
        w = exp_vals / exp_vals.sum()             # probabilities sum to 1

        # ---- gradient of smoothed max via convolution -------------------
        a_rev = a[::-1]
        a_rev_f = np.fft.rfft(a_rev, n=L)
        w_f = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(a_rev_f * w_f, n=L)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # length‑n gradient

        # ---- momentum step ------------------------------------------------
        vel = momentum * vel - step * grad
        cand = a + vel
        cand = _project_simplex(cand)

        # ---- occasional random perturbation -------------------------------
        if rng.random() < prob_perturb:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = cand[i] * rng.random() * 0.25
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)

        # ---- accept candidate ------------------------------------------------
        a = cand

        # ---- schedule updates -----------------------------------------------
        beta = min(beta_max, beta + beta_inc)
        step = max(min_step, step * decay)

        # ---- restart if stuck ------------------------------------------------
        if no_improve >= restart_thr:
            a = rng.random(n)
            a = _project_simplex(a)
            vel.fill(0.0)
            step = 0.12
            beta = 0.5
            no_improve = 0

        # ---- early exit safety ------------------------------------------------
        if time.monotonic() + 0.4 > deadline:
            break

    # ------------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------------
    return [float(x) for x in best]
