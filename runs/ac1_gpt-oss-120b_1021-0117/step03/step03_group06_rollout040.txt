    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound is
    as low as possible within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    global_best_seq = None
    global_best_score = float('inf')

    # lengths we will try – include the parent length if it exists
    lengths = [2500, 3000, 3500, 4000, 5000]
    if "height_sequence_1" in globals():
        lengths.append(len(height_sequence_1))

    while time.monotonic() < deadline:
        # pick a length (occasionally random)
        n = int(rng.choice(lengths))
        # optionally vary it a little
        if rng.random() < 0.2:
            n = int(np.clip(n + rng.integers(-200, 201), 2000, 6000))

        # ----- initial vector -----
        if "height_sequence_1" in globals() and n == len(height_sequence_1) and rng.random() < 0.5:

                init = init / s

        if global_best_score < 1.50274349:
            break

    # fallback – uniform distribution if nothing succeeded
    if global_best_seq is None:
        n = 2500
        global_best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in global_best_seq]
