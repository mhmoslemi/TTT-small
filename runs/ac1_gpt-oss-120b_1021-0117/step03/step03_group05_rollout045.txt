def project_simplex(v):
    # v is np.ndarray
    # return projection onto simplex sum=1, v>=0
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) with a low AC1 bound.
    The routine respects the time budget, keeps the best solution seen and
    performs a simple sub‑gradient descent on the max‑convolution objective.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a small margin

    # ------------------------------------------------------------------
    # initialise a feasible vector (use parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
    else:
        # fallback: random Dirichlet vector of moderate length
        n0 = 1024
        a = rng.random(n0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                                     # simplex (sum = 1)

    n = a.size
    # FFT length – next power of two >= 2*n-1
    _L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # helpers: convolution via FFT, score (our own cheap version), simplex proj
    # ------------------------------------------------------------------
    def _conv_fft(v: np.ndarray) -> np.ndarray:
        fv = np.fft.rfft(v, n=_L)
        conv = np.fft.irfft(fv * fv, n=_L)[: 2 * n - 1]
        return conv

    def _score(v: np.ndarray) -> float:
        c = _conv_fft(v)
        return 2.0 * n * c.max() / (v.sum() ** 2)

    def _project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto {x >= 0, sum x = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    cur = a.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # algorithm parameters
    it = 0
    no_improve = 0
    target = 1.50274349      # stop early if attained
    max_active = 20          # cap active indices for gradient
    add_alpha0 = 0.9
    mul_eta0 = 0.9

    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) evaluate convolution and collect active (max) indices
        # --------------------------------------------------------------
        conv = _conv_fft(cur)
        max_val = conv.max()
        # tolerance relative to max to capture ties
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]
        if active.size > max_active:
            # keep only the largest few
            idxs = np.argpartition(-conv[active], max_active - 1)[:max_active]
            active = active[idxs]

        # --------------------------------------------------------------
        # 2) sub‑gradient of the max‑convolution (sum over active indices)
        # --------------------------------------------------------------
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # contribution 2 * cur[k - idx] (mirrored slice)
            g[idx] += 2.0 * cur[k - idx]

        # --------------------------------------------------------------
        # 3) additive (projected‑gradient) candidate
        # --------------------------------------------------------------
        alpha = add_alpha0 / np.sqrt(it)
        cand_add = cur - alpha * g
        cand_add = _project_simplex(cand_add)
        cand_add_sc = _score(cand_add)

        # --------------------------------------------------------------
        # 4) multiplicative (exponential‑weights) candidate
        # --------------------------------------------------------------
        eta = mul_eta0 / np.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        s_mul = cand_mul.sum()
        if s_mul == 0.0:
            cand_mul = np.full_like(cur, 1.0 / n)
        else:
            cand_mul = cand_mul / s_mul
        cand_mul_sc = _score(cand_mul)

        # --------------------------------------------------------------
        # 5) pick the best of the three
        # --------------------------------------------------------------
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc

        # --------------------------------------------------------------
        # 6) targeted mass‑swap move (always cheap)
        # --------------------------------------------------------------
        if it % 200 == 0:
            i_max = int(g.argmax())
            i_min = int(g.argmin())
            if cur[i_max] > 0.0 and i_max != i_min:
                delta = cur[i_max] * 0.3 * rng.random()
                tmp = cur.copy()
                tmp[i_max] -= delta
                tmp[i_min] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc

        # --------------------------------------------------------------
        # 7) occasional jittered restart toward best solution
        # --------------------------------------------------------------
        if it % 5000 == 0 and rng.random() < 0.4:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc

        # --------------------------------------------------------------
        # 8) bookkeeping
        # --------------------------------------------------------------
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 9) early exit if target reached
        # --------------------------------------------------------------
        if best_sc < target:
            break

        # --------------------------------------------------------------
        # 10) escape stagnation – random redistribution
        # --------------------------------------------------------------
        if no_improve > 4000 and rng.random() < 0.25:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

    # ------------------------------------------------------------------
    # final lightweight refinement (few random swaps) before returning
    # ------------------------------------------------------------------
    refin_end = time.monotonic() + min(0.3, deadline - time.monotonic())
    while time.monotonic() < refin_end:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

    # Return the best sequence as a plain Python list of floats
    return [float(x) for x in best_vec]
