        # ---- step size (sqrt schedule) and line‑search ----
        it += 1
        eta = eta0 / math.sqrt(it)
        for _ in range(5):                     # at most 5 back‑tracking attempts
            a_candidate = _proj_simplex(a - eta * g)
            cand_score = 2.0 * n * _conv_self(a_candidate).max()
            if cand_score <= cur_score:
                a = a_candidate
                cur_score = cand_score
                break
            eta *= 0.5                         # shrink step if no improvement

        # ---- occasional random jitter to escape plateaus ----
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = a * jitter
            a = a / a.sum()
            stale = 0

        # ---- restart from the best point after long stagnation ----
        if stale >= stale_limit:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.01, size=n))
            a = a * perturb
            a = a / a.sum()
            stale = 0

    # ------------------------------------------------------------------
    # short random‑move refinement while time permits
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = rng.random() * 0.005   # move up to 0.5 % of the total mass
        if a[i] < eps:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        a_new = a_new / a_new.sum()
        new_score = 2.0 * n * _conv_self(a_new).max()
        if new_score < best_score:
            best = a_new.copy()
            best_score = new_score
            a = a_new.copy()

    # ------------------------------------------------------------------
    # return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
