import time, math, numpy as np

def _conv_self(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution using a real FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv

def _score_from_conv(conv: np.ndarray, n: int, s: float) -> float:
    """Score = 2·n·max(conv) / s² as required by the evaluator."""
    return 2.0 * n * conv.max() / (s * s)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    total = a.sum()
    if total <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= total                                 # put onto the probability simplex

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())           # power‑of‑two length for FFT

    cur = a.copy()
    cur_conv = _conv_self(cur, L)
    cur_score = _score_from_conv(cur_conv, n, cur.sum())

    best = cur.copy()
    best_score = cur_score

    # ----------------------------------------------------------------------
    # Search hyper‑parameters
    # ----------------------------------------------------------------------
    it = 0
    alpha0 = 0.85               # initial step scale for mirror descent
    top_k = 4                    # number of largest convolution entries to use
    jitter_scale = 6e-4
    temp = 0.12                  # temperature for occasional uphill moves
    stagnation = 0
    max_stagnation = 12000

    while time.monotonic() < deadline:
        it += 1

        # ---- convolution and active indices --------------------------------
        conv = _conv_self(cur, L)
        max_val = conv.max()
        # select up to top_k indices with largest values
        if conv.size <= top_k:
            active = np.arange(conv.size, dtype=np.int64)
        else:
            # argpartition gives unsorted top‑k efficiently
            active = np.argpartition(-conv, top_k - 1)[:top_k]

        # ---- sub‑gradient (mirror‑descent) ---------------------------------
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        eta = alpha0 / math.sqrt(it)
        # exponentiated (mirrored) update, numerically safe
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum > 0.0 and np.isfinite(cand_sum):
            cand /= cand_sum
        else:
            cand = np.full_like(cur, 1.0 / n)

        cand_conv = _conv_self(cand, L)
        cand_score = _score_from_conv(cand_conv, n, cand.sum())

        # ---- acceptance ----------------------------------------------------
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            stagnation = 0
        else:
            prob = math.exp(-(cand_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
            stagnation += 1

        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score

        # ---- occasional random mass‑transfer move -------------------------
        if rng.random() < 0.14:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                s = tmp.sum()
                if s > 0.0:
                    tmp /= s
                    tmp_conv = _conv_self(tmp, L)
                    tmp_score = _score_from_conv(tmp_conv, n, s)
                    if tmp_score < cur_score:
                        cur, cur_score = tmp, tmp_score
                    if tmp_score < best_score:
                        best, best_score = tmp.copy(), tmp_score

        # ---- jittered restart from best every few thousand iterations -----
        if it % 4200 == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            s = tmp.sum()
            if s > 0.0:
                tmp /= s
                tmp_conv = _conv_self(tmp, L)
                tmp_score = _score_from_conv(tmp_conv, n, s)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
                stagnation = 0

        # ---- long‑stagnation restart ---------------------------------------
        if stagnation > max_stagnation:
            # restart from a slightly perturbed uniform distribution
            cur = np.full_like(cur, 1.0 / n)
            cur_score = _score_from_conv(_conv_self(cur, L), n, 1.0)
            stagnation = 0

        # ---- early exit ----------------------------------------------------
        if best_score < 1.50274349:
            break

    # Return plain Python list (evaluator expects a list, not a NumPy array)
    return [float(x) for x in best]
