def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # spare

    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate simple decreasing distribution: a_i = 1/(i+1)^p normalized
        n = 2000 maybe use 1024? but we want matching length.
        n = 1024
        i = np.arange(n)
        a0 = 1.0 / (i+1)**0.5
    # enforce nonnegative and normalize sum=1
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s == 0:
        a0 = np.full_like(a0, 1.0/a0.size)
    else:
        a0 = a0 / s
    n = a0.size
    L = 1 << ((2*n - 1).bit_length())

    # conv and score function using FFT
    def conv_fft(v):
        fv = np.fft.rfft(v, n=L)
        return np.fft.irfft(fv*fv, n=L)[:2*n-1]

    def score(v):
        # assume sum(v)==1
        conv = conv_fft(v)
        return 2.0 * n * conv.max()
        # we could also compute max directly each update.

    # Simplex projection (same as earlier)
    def project_simplex(v):
        """Euclidean projection onto {x>=0, sum=1}."""
        # Implementation using sorting.
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho+1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    cur = a0.copy()
    cur_sc = score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # parameters
    max_iter = 10**7 # not used.
    it = 0
    no_improve = 0

    # hyperparameters
    K = 30    # more active indices
    add_alpha0 = 1.0
    mul_eta0 = 1.0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # compute convolution once
        conv = conv_fft(cur)
        # Get top K indices; If many ties, they will still choose via argpartition.
        if K >= conv.size:
            active = np.arange(conv.size)
        else:
            # use argpartition for top K
            active = np.argpartition(-conv, K-1)[:K]

        # compute subgradient
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # i indexes: i = lo..hi; term a_{k - i}
            # vectorized:
            idx = np.arange(lo, hi+1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]   # each derivative 2*a_{k-i}
        # (Potentially g may be huge; but fine)

        # additive update
        alpha = add_alpha0 / math.sqrt(it)
        cand_add = cur - alpha * g
        cand_add = project_simplex(cand_add)
        cand_add_sc = score(cand_add)

        # multiplicative update
        eta = mul_eta0 / math.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        s_mul = cand_mul.sum()
        if s_mul == 0:
            cand_mul = np.full_like(cur, 1.0/cur.size)
        else:
            cand_mul = cand_mul / s_mul
        cand_mul_sc = score(cand_mul)

        # pick best among cur, add, mul
        # Evaluate cur_sc already known
        # Choose smallest
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc
        else:
            # acceptance probability for maybe slightly worse

            # Might as well do small random acceptance for exploring.
            pass

        # Update best
        if cur_sc < best_sc:
            best_vec = cur.copy()
            best_sc = cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # random mass transfer if stuck
        if no_improve > 5000:
            # random attempt
            i = rng.integers(0, n)
            if cur[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp) # ensures sum=1
                tmp_sc = score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec = tmp.copy()
                        best_sc = tmp_sc
                        no_improve = 0

        # occasional jitter restart
        if it % 5000 == 0:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)

            if tmp_sc < cur_sc:

                    best_vec = tmp.copy()
                    best_sc = tmp_sc

import time, math, numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (as a list) with a low AC1 bound.
    The routine respects the time budget and keeps the best solution found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ----------------------------------------------------------------------
    # initialise the vector (use the provided parent if it exists)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple analytic starter: decreasing power law, then normalised
        n0 = 2048
        idx = np.arange(n0, dtype=np.float64) + 1.0
        a = 1.0 / np.sqrt(idx)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                      # normalise to sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # ----------------------------------------------------------------------
    # helper utilities
    # ----------------------------------------------------------------------
    def conv_fft(v: np.ndarray) -> np.ndarray:
        """Full (linear) convolution of v with itself, length 2n‑1."""
        fv = np.fft.rfft(v, n=L)
        return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

    def score(v: np.ndarray) -> float:
        """AC1 bound for a vector normalised to sum = 1."""
        c = conv_fft(v)
        return 2.0 * n * c.max()

    def project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto {x≥0, sum x = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0.0] = 0.0
        return w

    # ----------------------------------------------------------------------
    # bookkeeping
    # ----------------------------------------------------------------------
    cur = a.copy()
    cur_sc = score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # algorithmic parameters
    it = 0
    no_improve = 0
    K = max(5, int(0.02 * n))          # active convolution indices
    add_alpha0 = 0.8
    mul_eta0 = 0.8
    target = 1.50274349                # stop once we beat this bound

    # ----------------------------------------------------------------------
    # main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        conv = conv_fft(cur)

        # active set (K largest convolution entries)
        if K >= conv.size:
            active = np.arange(conv.size)
        else:
            active = np.argpartition(-conv, K - 1)[:K]

        # subgradient of the max‑convolution (sum over active indices)
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # ---------- additive (projected‑gradient) update ----------
        alpha = add_alpha0 / math.sqrt(it)
        cand_add = project_simplex(cur - alpha * g)
        cand_add_sc = score(cand_add)

        # if not improving, try a smaller step
        if cand_add_sc >= cur_sc:
            cand_small = project_simplex(cur - 0.2 * alpha * g)
            sc_small = score(cand_small)
            if sc_small < cand_add_sc:
                cand_add, cand_add_sc = cand_small, sc_small

        # ---------- multiplicative (exponential‑weights) update ----------
        eta = mul_eta0 / math.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        sm = cand_mul.sum()
        if sm == 0.0:
            cand_mul = np.full_like(cur, 1.0 / n)
        else:
            cand_mul = cand_mul / sm
        cand_mul_sc = score(cand_mul)

        # ---------- candidate selection ----------
        # pick the best among current, additive and multiplicative
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc
        else:
            # occasional Metropolis‑style acceptance to escape plateaus
            prob = math.exp(-(min(cand_add_sc, cand_mul_sc) - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                if cand_add_sc < cand_mul_sc:
                    cur, cur_sc = cand_add, cand_add_sc
                else:
                    cur, cur_sc = cand_mul, cand_mul_sc

        # ---------- bookkeeping ----------
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # ---------- random mass‑transfer when stalled ----------
        if no_improve > 4000 and rng.random() < 0.25:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_sc = score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # ---------- jittered restart towards the best solution ----------
        if it % 7000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = project_simplex(tmp)
            tmp_sc = score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # ---------- early exit ----------
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # final refinement: few local random transfers (short time slice)
    # ----------------------------------------------------------------------
    refin_end = time.monotonic() + min(0.3, deadline - time.monotonic())
    while time.monotonic() < refin_end:
        i = rng.integers(0, n)
        if cur[i] == 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.35
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = project_simplex(tmp)
        tmp_sc = score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

    # return the best vector found as a plain Python list
    return [float(x) for x in best_vec]
