import time, math
import numpy as np

# --------------------------------------------------------------
# Simplex projection (Euclidean) : ensure non‑negative sum = 1
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:          # safeguard – should never happen
        w[:] = 1.0 / n
    return w

# --------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1+…+x^{2n‑2}) used as a good start
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# --------------------------------------------------------------
# Full convolution (via FFT) – returns array, max value, argmax
# --------------------------------------------------------------
def _full_conv(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    return conv

def _conv_max_and_idx(conv: np.ndarray):
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# --------------------------------------------------------------
# Gradient from a set of peak indices (sub‑gradient of max)
# --------------------------------------------------------------
def _gradient_from_peaks(a: np.ndarray, peaks):
    n = a.size
    g = np.zeros_like(a)
    for idx in peaks:
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        j = np.arange(lo, hi + 1, dtype=int)
        g[j] += 2.0 * a[idx - j]
    if peaks.size > 0:
        g /= peaks.size
    return g
