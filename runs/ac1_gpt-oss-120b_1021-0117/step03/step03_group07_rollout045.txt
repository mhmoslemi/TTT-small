import time
import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (no sum‑normalisation required)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _make_exponential(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a / a.sum()
    return a

def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.arange(1, n + 1, dtype=np.float64) ** p
    a = a / a.sum()
    return a

def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()

def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()

def _smooth_grad(a: np.ndarray, alpha: float, conv: np.ndarray, L_fft: int) -> np.ndarray:
    """Gradient of the soft‑max smoothed max‑convolution objective."""
    n = a.shape[0]
    mx = conv.max()
    w = np.exp(alpha * (conv - mx))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a (via FFT)
    fw = np.fft.fft(w, n=L_fft)
    frev = np.fft.fft(a[::-1], n=L_fft)
    conv_rev = np.fft.ifft(fw * frev).real
    grad_raw = conv_rev[n - 1: n - 1 + n]      # length n slice
    return 4.0 * n * grad_raw                  # factor from the objective

    else:

    n = cur.size
    best_vec = cur.copy()
