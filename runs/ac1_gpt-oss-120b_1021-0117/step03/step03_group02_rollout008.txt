import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Full self‑convolution via FFT (returns max and full array)
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv

# Gradient from arbitrary non‑negative weights (mask or softmax)
# ------------------------------------------------------------
def _grad_from_weights(weights: np.ndarray, a: np.ndarray, L: int,
                       start: int, stop: int) -> np.ndarray:
    """g = 2 * (weights * a_rev) convolution, sliced to length n."""
    a_rev = a[::-1]
    wf = np.fft.rfft(weights, n=L)
    arf = np.fft.rfft(a_rev, n=L)
    g_full = np.fft.irfft(wf * arf, n=L)
    return 2.0 * g_full[start:stop]
