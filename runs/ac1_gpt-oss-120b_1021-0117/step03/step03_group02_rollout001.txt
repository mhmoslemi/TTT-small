import time, numpy as np

def _project_simplex(v):
    # same as previous, with sum=1 but accepting any sum
    # We'll implement generic: ensure nonneg, sum to 1
    n = v.shape[0]
    if n == 0:
        return v
    # shift to make sum 1
    # Compute sorted
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0)
    # renormalize to sum 1 (should already but due to numerical errors)
    sumw = w.sum()
    if sumw > 0:
        w /= sumw
    return w
```

def _project_simplex(v):
    # Input vector v (np.ndarray)
    # Projects onto simplex {x>=0, sum(x)=1}
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # renormalize
    s = w.sum()
    if s > 0:
        w /= s
    return w

import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # keep a small safety margin before the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initialise a feasible sequence
    # --------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: simple decreasing series
        a = np.linspace(1.0, 0.0, 256, dtype=np.float64)
    a = _project_simplex(a / (a.sum() if a.sum() != 0 else 1.0))
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())        # power‑of‑two FFT length

    # --------------------------------------------------------
    # compute initial score (sum is 1 after projection)
    # --------------------------------------------------------
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    cur_score = 2.0 * n * conv_full.max()
    best = a.copy()
    best_score = cur_score

    # --------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------
    lr = 0.3                  # learning rate
    min_lr = 1e-5
    decay = 0.9995
    alpha = 8.0               # soft‑max sharpness (higher → closer to max)
    no_improve = 0
    max_no_improve = 300
    prob_random = 0.02

    # --------------------------------------------------------
    # main loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution of current a (for max and weights) -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- soft‑max weights over all lags -----
        # use max_val for numerical stability
        exp_arg = alpha * (conv_full - max_val)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient = 2 * (w ⋆ reverse(a)) -----
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # length‑n gradient

        # ----- gradient step + simplex projection -----
        cand = a - lr * grad
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_cand.max()

        improved = False
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            improved = True
        else:
            # occasional uphill acceptance (simulated annealing flavour)
            if rng.random() < np.exp(-(cand_score - cur_score) / max(lr, 1e-12)):
                a = cand
                cur_score = cand_score
                improved = True

        # ----- keep global best -----
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            no_improve = 0
        else:
            if not improved:
                no_improve += 1
            else:
                no_improve = 0

        # ----- random mass‑transfer perturbation (exploration) -----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                frac = rng.random() * 0.3
                delta = a[i] * frac
                pert = a.copy()
                pert[i] -= delta
                pert[j] += delta
                pert = _project_simplex(pert)
                pert_fft = np.fft.rfft(pert, n=L)
                conv_pert = np.fft.irfft(pert_fft * pert_fft, n=L)[: 2 * n - 1]
                pert_score = 2.0 * n * conv_pert.max()
                if pert_score < best_score:
                    best = pert.copy()
                    best_score = pert_score
                    a = pert
                    cur_score = pert_score
                    no_improve = 0

        # ----- learning‑rate decay -----
        lr = max(min_lr, lr * decay)

        # ----- restart if stuck -----
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = _project_simplex(a)
            lr = 0.3
            no_improve = 0
            # recompute current score for the new point
            a_fft = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            cur_score = 2.0 * n * conv_full.max()

        # ----- time guard -----
        if time.monotonic() + 0.5 > deadline:
            break

    # return the best sequence found as plain Python floats
    return [float(x) for x in best]
