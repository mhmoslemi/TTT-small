import time, numpy as np

def propose_candidate(...):
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64) 
        init_seq = np.maximum(init_seq, 0)
        init_seq_sum = init_seq.sum()
        if init_seq_sum > 0:
            init_seq = init_seq / init_seq_sum
        else:
            init_seq = np.full_like(init_seq, 1.0 / init_seq.size)
        n = init_seq.size
    else:
        n = 256
        init_seq = np.full(n, 1.0/n, dtype=np.float64)

    # Set initial logits as log(init_seq) (plus small epsilon for zeros)
    eps = 1e-12
    w_logits = np.log(init_seq + eps)

    # Precompute index range for convolution size
    L = 1 << ((3 * n - 2).bit_length())
    # We'll reuse for both convolutions

    # function to compute a from logits
    def softmax(logits):
        # subtract max for stability
        m = np.max(logits)
        exps = np.exp(logits - m)
        s = exps.sum()
        return exps / s

    # Helper compute conv
    def conv_full(a):
        # compute full convolution a * a using FFT of size L
        # using rfft
        fv = np.fft.rfft(a, n=L)
        prod = fv * fv
        conv = np.fft.irfft(prod, n=L)[:2*n-1]
        return conv

    # initial evaluation
    a = softmax(w_logits)
    best_vec = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # Some target threshold
    target = 1.50274349

    # parameters
    lam = 5.0   # initial smoothing parameter
    lam_inc = 1.01  # multiplicative increase each iteration
    max_lam = 500.0
    eta0 = 0.2

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # compute a
        a = softmax(w_logits)
        # compute conv
        conv = conv_full(a)   # length 2n-1
        # compute smoothed max
        # scaling to avoid overflow
        maxc = np.max(conv)
        exp_term = np.exp(lam * (conv - maxc))
        sum_exp = exp_term.sum()
        M = maxc + np.log(sum_exp) / lam

        # compute softmax weights
        w = exp_term / sum_exp   # shape 2n-1

        # compute gradient wrt a: g_a = 2 * sum_k w_k * shift
        # compute convolution of w with reversed a
        rev_a = a[::-1]
        # compute convolution via FFT
        fw = np.fft.rfft(w, n=L)
        frev = np.fft.rfft(rev_a, n=L)
        conv_wa = np.fft.irfft(fw * frev, n=L)  # length L; we need slice
        # conv_wa length = 2n-1 + n -1 = 3n-2 => extract indices n-1 to n-1 + n
        start_idx = n - 1
        g_a = 2.0 * conv_wa[start_idx : start_idx + n]
        # compute dot product
        dot = np.dot(a, g_a)

        # gradient wrt logits
        g_w = a * (g_a - dot)

        # step size
        eta = eta0 / np.sqrt(it)
        w_logits -= eta * g_w

        # optional jitter occasionally to escape local minima
        if it % 5000 == 0:
            jitter = np.random.default_rng().normal(scale=1e-4, size=n)
            w_logits += jitter

        # evaluate exact objective periodically
        if it % 2000 == 0 or it == 1:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score - 1e-8:
                best_score = cur_score
                best_vec = a.copy()
                if best_score < target:
                    break

        # increase lambda
        lam = min(max_lam, lam * lam_inc)

    # final ensure best
    return [float(v) for v in best_vec]

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ------------------------------------------------------------------
    # base sequence (parent if present, otherwise uniform)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    eps = 1e-12

    # logits for a soft‑max parametrisation (guarantees simplex and positivity)
    logits = np.log(cur + eps)

    # ------------------------------------------------------------------
    # utility functions (FFT‑based convolution, soft‑max)
    # ------------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())          # power‑of‑two size covering 3n‑2

    def softmax(x):
        m = np.max(x)
        e = np.exp(x - m)
        return e / e.sum()

    def conv_full(a):
        """full convolution a*a using FFT (real input)"""
        fa = np.fft.rfft(a, n=L)
        prod = fa * fa
        conv = np.fft.irfft(prod, n=L)[: 2 * n - 1]
        return conv

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.08                # base learning rate (Adam rescales automatically)
    beta1_pow, beta2_pow = 1.0, 1.0
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    # smoothing parameter for a soft‑max over the convolution coefficients
    lam = 5.0                 # initial smoothness
    lam_mult = 1.012          # increase each iteration (approaches true max)
    lam_max = 600.0

    best_vec = cur.copy()
    best_sc = float("inf")
    evaluate = evaluate_sequence

    eval_interval = 250       # how often to call the expensive scorer

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----------------------- current probabilities --------------------
        a = softmax(logits)

        # ----------------------- convolution & smoothing -----------------
        conv = conv_full(a)                # length 2n‑1
        max_c = conv.max()

        # soft‑max weights over the convolution entries
        exp_term = np.exp(lam * (conv - max_c))
        w = exp_term / exp_term.sum()      # shape (2n‑1,)

        # gradient w.r.t. a (smoothened max)
        rev_a = a[::-1]
        fw = np.fft.rfft(w, n=L)
        fr = np.fft.rfft(rev_a, n=L)
        tmp = np.fft.irfft(fw * fr, n=L)   # length L, contains full convolution
        # needed slice: indices n‑1 … n‑1+n‑1
        g_a = 2.0 * tmp[n - 1 : n - 1 + n]  # shape (n,)

        # ----------------------- convert to gradient on logits ----------
        dot = np.dot(a, g_a)
        g_logits = a * (g_a - dot)          # Jacobian of soft‑max

        # ----------------------- Adam update ----------------------------
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * g_logits
        v = beta2 * v + (1.0 - beta2) * (g_logits * g_logits)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / (1.0 + it * 1e-5)        # tiny decay to keep steps bounded
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=5e-6, size=n)
            logits += jitter

        # increase smoothing (approach true max)
        lam = min(lam_max, lam * lam_mult)

        # ----------------------- occasional exact scoring ---------------
        if it % eval_interval == 0:
            sc = evaluate(a.tolist())
            if sc < best_sc:
                best_sc, best_vec = sc, a.copy()
                # early exit if we already beat the target
                if best_sc < 1.50274349:
                    break

        # safety break if we ran an excessive number of iterations
        if it >= 800000:
            break

    # ------------------------------------------------------------------
    # final fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
