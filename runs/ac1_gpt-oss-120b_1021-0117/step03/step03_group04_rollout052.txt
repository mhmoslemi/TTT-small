import time
import math
import numpy as np

# Optional: cvxpy may be unavailable; we import lazily.
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum = 1, v >= 0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a using FFT (real inputs)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    Heuristic optimizer for the AC1 upper bound.
    Uses a blend of soft‑max sub‑gradient descent and a final linear‑program
    refinement. Returns the best coefficient list found before the deadline.
    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    try:

    except Exception:
        parent_raw = None

    # ------------------------------------------------------------------
    # 2. Determine problem size and target sum (sum = sqrt(2n))
    # ------------------------------------------------------------------
    n = int(parent_raw.size) if parent_raw is not None else 256

    # ------------------------------------------------------------------

    candidates = []

    # uniform

    candidates.append(uniform)

    # tent (triangular) shape

    tent = np.maximum(tent, 0.0)
    if tent.sum() > 0:

    else:

    candidates.append(tent)

    gauss = gauss / gauss.sum() * sum_target
    candidates.append(gauss)

    # parent if valid

        s = a_parent.sum()
        if s >= 0.01:

        else:

        candidates.append(a_parent)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    conv_len = 2 * n - 1

    # pick the best start
