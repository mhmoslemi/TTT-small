    deadline = start + max(0.0, float(budget_s)-0.1)

    rng = np.random.default_rng(seed)

    cur = np.maximum(cur, 0.0)

    n = cur.size

    L = 1 << ((2*n - 1).bit_length())
    sqrt_n = np.sqrt(n) # maybe used

    best_score = _score(cur)

    # parameters
    add_alpha0 = 1.0
    topK = 30  # active set size
    eps_active = 1e-4

    it = 0
    while time.time() < deadline:
        it += 1

        # compute conv
        fv = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        # find max conv
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv  # sum=1

        # early stop if better than target
        if cur_score < 1.50274349:
            best = cur.copy()
            best_score = cur_score
            break

        # active set: indices where conv >= max_conv * (1 - eps_active)
        # gather indices
        threshold = max_conv * (1 - eps_active)
        active = np.where(conv >= threshold)[0]
        # optionally limit to topK
        if active.size > topK:

            # use argpartition
            act_candidates = np.argpartition(-conv, topK-1)[:topK]

        g = np.zeros_like(cur)

def project_simplex(v):
    # ensure sum to 1 and nonnegative
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    # renormalize to exactly sum 1 (due to numeric)
    s = w.sum()
    if s > 0:
        w = w / s
    else:
        # fallback uniform
        w = np.full_like(v, 1.0 / v.size)
    return w
```

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the AC1 bound.
    Returns the best vector (as a list) found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # a small safety margin

    # ----------------------------------------------------------------------
    # problem size and FFT helpers
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        cur = rng.random(n)

    # ensure feasibility: non‑negative and sum = 1 (simplex)
    cur = np.maximum(cur, 0.0)

    def _project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto {x >= 0, sum x = 1}."""
        # sort descending
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        s = w.sum()
        if s > 0:
            w /= s
        else:
            w[:] = 1.0 / v.size
        return w

    cur = _project_simplex(cur)

    # fixed FFT length (next power of two)
    _L = 1 << ((2 * n - 1).bit_length())

    def _conv_max(v: np.ndarray) -> float:
        fv = np.fft.rfft(v, n=_L)
        conv = np.fft.irfft(fv * fv, n=_L)[: 2 * n - 1]
        return float(conv.max())

    def _score(v: np.ndarray) -> float:
        s = v.sum()
        if s <= 0.0:
            return float("inf")
        return 2.0 * n * _conv_max(v) / (s * s)

    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # ----------------------------------------------------------------------
    # search parameters
    # ----------------------------------------------------------------------
    K = 30                     # size of active convolution set
    add_alpha0 = 1.0           # base additive step size
    target = 1.50274349        # early‑exit threshold
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # current convolution and score
        fv = np.fft.rfft(cur, n=_L)
        conv = np.fft.irfft(fv * fv, n=_L)[: 2 * n - 1]
        max_conv = float(conv.max())
        cur_sc = 2.0 * n * max_conv / (cur.sum() ** 2)

        # keep global best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit
        if best_sc < target:
            break

        # ------------------------------------------------------------------
        # active‑set subgradient (top‑K convolution entries)
        # ------------------------------------------------------------------
        if K >= conv.size:
            active = np.arange(conv.size)
        else:
            active = np.argpartition(-conv, K - 1)[:K]

        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]   # derivative of (a*a)_k

        # ------------------------------------------------------------------
        # additive projected‑gradient step with tiny line‑search
        # ------------------------------------------------------------------
        alpha0 = add_alpha0 / np.sqrt(it)
        cand_best = cur
        cand_best_sc = cur_sc
        for factor in (1.0, 0.5, 2.0):
            alpha = alpha0 * factor
            cand = cur - alpha * g
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < cand_best_sc:
                cand_best, cand_best_sc = cand, cand_sc

        if cand_best_sc < cur_sc:
            cur, cur_sc = cand_best, cand_best_sc
            continue

        # ------------------------------------------------------------------
        # occasional random mass‑transfer move (exploratory)
        # ------------------------------------------------------------------
        if rng.random() < 0.15:
            i = np.argmax(cur)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            delta = cur[i] * rng.random() * 0.4
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
                continue

        # ------------------------------------------------------------------
        # restart from a fresh random sparse subset when stuck
        # ------------------------------------------------------------------
        if no_improve > 3500:
            m = rng.integers(70, 120)            # size of sparse support
            idx = rng.choice(n, size=m, replace=False)
            cur = np.zeros(n, dtype=np.float64)
            cur[idx] = 1.0 / m
            cur = _project_simplex(cur)
            no_improve = 0
            continue

    # fallback to uniform distribution if nothing found
    if best_sc == float("inf"):
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
