def compute_gradient(a, idxs):
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in idxs:
        offset = k - (n - 1)
        if offset >= 0:
            i_start = offset
            i_end = n
            # i range offset..n-1 inclusive
            i_range = np.arange(i_start, i_end, dtype=np.int64)
            partner = k - i_range
            g[i_range] += a[partner]
        else:
            i_start = 0
            i_end = n + offset
            i_range = np.arange(i_start, i_end, dtype=np.int64)
            partner = k - i_range
            g[i_range] += a[partner]
    g *= 2.0
    return g
```

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all entries will be equal
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimizes the
    AC1 upper bound.  The algorithm uses a smooth‑max gradient of the
    self‑convolution, Adam‑style updates and projection onto the simplex.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1️⃣  Time budget
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2️⃣  Initial sequence (fallback to uniform if no parent)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalize to sum = 1

    # ------------------------------------------------------------------
    # 3️⃣  Keep the best solution seen so far
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))  # pre‑imported
    # a quick uniform fallback (may already be better)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 4️⃣  FFT sizes (large enough for both convolution and correlation)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1                     # length of linear convolution
    total_len = n + conv_len - 1              # length needed for correlation
    L = 1 << ((total_len - 1).bit_length())  # next power of two

    # ------------------------------------------------------------------
    # 5️⃣  Optimisation parameters
    # ------------------------------------------------------------------
    alpha_exp = 6.0               # controls smooth‑max sharpness
    base_lr = 0.08                # Adam base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    it = 0
    no_improve = 0
    max_stale = 20000
    jitter_every = 200
    jitter_std = 0.02
    target_score = 1.50274349

    # ------------------------------------------------------------------
    # 6️⃣  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution (via FFT) ----------------------------------
        fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)
        conv = conv_full[:conv_len]
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val / (a.sum() ** 2))

        # ----- record best -------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ----- smooth‑max weights ------------------------------------
        # avoid division‑by‑zero (max_val > 0 because sum(a)>0)
        rel = conv / max_val               # in [0,1]
        w = np.exp(alpha_exp * (rel - 1.0))  # w[max]=1, others<1

        # ----- gradient via correlation (a * w) ----------------------
        fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(fa) * fw, n=L)
        g = 2.0 * corr_full[:n]               # sub‑gradient of max‑conv

        # ----- Adam update -------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _proj_simplex(a)                  # enforce non‑negativity & sum=1

        # ----- occasional random jitter -------------------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- restart if progress stalls ------------------------------
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # 7️⃣  Return the best sequence (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
