def conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for j in range(n):
        rows.extend(range(j, j+n))
        cols.extend([j]*n)
        data.extend(a.tolist())
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    data = np.array(data, dtype=np.float64)
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    deadline = time.monotonic() + max(0.0, float(budget_s)-2.0)
    # init vector
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = np.full(n, 1.0/n) # uniform
    # ensure nonnegative
    a = np.maximum(a, 0.0)
    # project onto simplex (normalize sum=1) using similar _project_simplex
    a = _project_simplex(a)  # ensures sum=1
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    # main loop
    while time.monotonic() < deadline:
        # compute conv max and M
        conv = np.convolve(a, a)
        M = conv.max()
        # build matrix A
        n = a.size
        rows = 2*n - 1
        A = np.empty((rows, n), dtype=np.float64)
        # fill via loop
        for j in range(n):
            A[j:j+n, j] = a
        # set up cvxpy
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8, feastol=1e-8, max_iters=5000)
        except Exception:
            # fallback to gradient optimizer
            break
        if g.value is None:
            break
        g_opt = np.array(g.value, dtype=np.float64)
        # ensure nonnegative and finite
        g_opt = np.maximum(g_opt, 0.0)
        sum_g = g_opt.sum()
        if sum_g <= 0:
            break
        g_norm = g_opt / sum_g  # project onto simplex (sum=1)
        # mixing
        t = 0.2   # mixing step
        a_new = (1 - t) * a + t * g_norm
        # project onto simplex again
        a_new = _project_simplex(a_new)
        # evaluate

    # we can run previous gradient optimizer for leftover time

        a_gd, score_gd = _optimize(best, deadline, rng)

import time, math, numpy as np, cvxpy as cp

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# ----------------------------------------------------------------------
# Convolution max via FFT (returns max value and its index)
# ----------------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# ----------------------------------------------------------------------
# Projected gradient descent on the current peak (same as previous baseline)
# ----------------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.5
    min_step = 1e-7
    decay = 0.9999
    prob_random = 0.02
    prob_smooth = 0.005
    beta = 0.9
    momentum = np.zeros_like(a)

    while time.monotonic() < deadline:
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:

            g[j] = 2.0 * a[max_idx - j]

        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        cand_max, cand_idx = _conv_max_and_index(cand, L)
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:

            max_idx = cand_idx
            if cand_score < best_score:

        else:

        UB = 10.0

        constraints = [A @ g <= M, g <= UB]
