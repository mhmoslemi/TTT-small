import time, math, numpy as np

def _project_simplex(v):
    sorted_v = np.sort(v)[::-1]
    css = np.cumsum(sorted_v) - 1.0
    rho = np.where(sorted_v - css/np.arange(1, v.size+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    return float(2.0 * n * conv.max()), conv, a_fft

    rng = np.random.default_rng(seed)
    start = time.monotonic()

    a0 = np.maximum(a0, 0.0)

    if s < 1e-12:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:

    n = a0.size
    L_fft = 1 << ((3 * n - 2).bit_length())  # for gradient conv length
    # for scoring conv we need at least 2n-1, but L_fft suffices.

    cur = a0.copy()
    cur_sc, cur_conv, cur_fft = _score_vec(cur, L=L_fft)

    beta2 = 0.999

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a vector with sum = 1
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(convolution) for a (sum(a) assumed 1)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    n = a0.size

    L = 1 << ((3 * n - 2).bit_length())
