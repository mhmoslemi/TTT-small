import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Gradient‑descent search for a step‑function whose AC1 upper bound is small.
    Uses a smooth approximation of the max‑convolution (soft‑max) and projects
    onto the simplex after each update.  Returns the best vector encountered.
    """
    rng = np.random.default_rng(seed)
    # --------------------------------------------------------------
    # 1. Time management
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------
    # 2. Initial sequence (use the supplied parent if present)
    # --------------------------------------------------------------
    src = globals().get("height_sequence_1")
    if src is not None:
        a = np.asarray(src, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        n = int(a.size)
    else:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # project onto simplex (sum = 1)

    # --------------------------------------------------------------
    # 3. Store the best solution seen so far
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_conv = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len_conv)
    conv = np.fft.irfft(fa * fa, n=fft_len_conv)[:conv_len]
    best = a.copy()
    best_score = float(2.0 * n * conv.max())

    # check uniform vector as a cheap fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    fu = np.fft.rfft(uniform, n=fft_len_conv)
    conv_u = np.fft.irfft(fu * fu, n=fft_len_conv)[:conv_len]
    uniform_score = float(2.0 * n * conv_u.max())
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # --------------------------------------------------------------
    # 4. Parameters for the optimisation loop
    # --------------------------------------------------------------
    fft_len_full = 1 << ((3 * n - 2 - 1).bit_length())   # to convolve a (len n) with w (len 2n-1)
    eta0 = 0.5
    beta = 0.9
    gamma = 8.0                     # smoothness of the soft‑max
    jitter_every = 150
    jitter_std = 0.03
    target_score = 1.50274349
    v = np.zeros_like(a)            # momentum buffer
    it = 0
    no_improve = 0
    max_stale = 3000

    # --------------------------------------------------------------
    # 5. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution a*a (full) -----
        fa = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(fa * fa, n=fft_len_conv)[:conv_len]

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- keep best solution -----
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ----- smooth approximation of the max (soft‑max) -----
        # w sums to 1 and concentrates around the largest convolution entries
        exp_vals = np.exp(gamma * (conv - cur_max))
        w = exp_vals / exp_vals.sum()

        # ----- gradient of the smooth max -----
        w_rev = w[::-1]                       # reverse for the Toeplitz product
        fb = np.fft.rfft(a, n=fft_len_full)
        fw = np.fft.rfft(w_rev, n=fft_len_full)
        grad_full = np.fft.irfft(fb * fw, n=fft_len_full)

        start = 2 * n - 2                     # index of w_rev[0] in the full convolution
        idx = start - np.arange(n)            # slice that yields gradient entries
        grad = 2.0 * grad_full[idx]

        # ----- momentum + projected gradient step -----
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * grad
        a = _proj_simplex(a - eta * v)

        # ----- occasional random jitter to escape plateaus -----
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v[:] = 0.0
            no_improve = 0

        # ----- restart if stuck for a long time -----
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v[:] = 0.0
            no_improve = 0

    # --------------------------------------------------------------
    # 6. Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------
    return [float(x) for x in best]
