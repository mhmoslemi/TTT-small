        ind = np.arange(1, u.size+1)

def subgrad(a, conv, top_K):
    # conv: array of length conv_len
    # returns gradient vector shape (n,)
    conv_len = conv.shape[0]
    # find indices of top_K
    if top_K < conv_len:
        idxs = np.argpartition(-conv, top_K)[:top_K]
    else:
        idxs = np.arange(conv_len)
    vals = conv[idxs]
    total = vals.sum()
    if total > 0:
        weights = vals / total
    else:
        weights = np.full_like(vals, 1.0/vals.size)
    g = np.zeros_like(a)
    for shift, w in zip(idxs, weights):
        offset = shift - (n-1)
        if offset >= 0:
            i = np.arange(offset, n, dtype=np.int64)
            partner = shift - i
            g[i] += w * 2.0 * a[partner]
        else:
            i = np.arange(0, n + offset, dtype=np.int64)
            partner = shift - i
            g[i] += w * 2.0 * a[partner]
    return g

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
