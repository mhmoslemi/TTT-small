    def proj_simplex(v):
        # ensure nonnegative and sum to 1
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        # renorm to avoid numerical drift
        s = w.sum()
        if s > 0:
            w /= s
        else:
            w = np.full_like(v, 1.0 / v.size)

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine respects a time budget and keeps the best solution found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # initialise the vector (use the provided parent if it exists)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a cheap analytic starter – a short sqrt‑series, then padded with zeros
        n0 = 256
        a0 = np.empty(n0, dtype=np.float64)
        a0[0] = 1.0
        for k in range(1, n0):
            if k == 1:
                s = 0.0
            else:
                s = float(np.dot(a0[1:k], a0[k - 1:0:-1]))
            a0[k] = (1.0 - s) / 2.0
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / total                     # normalise to sum = 1

    cur = a0.copy()
    n = cur.size

    # ----------------------------------------------------------------------
    # helper utilities
    # ----------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power‑of‑two)

    def conv_full(v: np.ndarray) -> np.ndarray:
        """Full linear convolution v * v using a real FFT."""
        fv = np.fft.rfft(v, n=L)
        c = np.fft.irfft(fv * fv, n=L)
        return c[: 2 * n - 1]

    def score_of(v: np.ndarray) -> float:
        """Exact AC1 bound for a (non‑negative) vector v."""
        s = v.sum()
        if s <= 0.0:
            return np.inf
        conv = conv_full(v)
        return 2.0 * n * conv.max() / (s * s)

    def proj_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto the simplex {x>=0, sum x = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        s = w.sum()
        if s > 0:
            w /= s
        else:
            w = np.full_like(v, 1.0 / v.size)
        return w

    # ----------------------------------------------------------------------
    # bookkeeping
    # ----------------------------------------------------------------------
    cur_sc = score_of(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    it = 0
    no_improve = 0
    target = 1.50274349          # stop once we beat this bound
    add_alpha0 = 2.5             # initial additive step size
    mul_eta0 = 2.5               # initial multiplicative step size
    K = 20                       # number of “active” convolution indices

    # ----------------------------------------------------------------------
    # main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        conv = conv_full(cur)
        max_conv = conv.max()

        # active set: K largest convolution entries
        if K >= conv.size:
            active = np.arange(conv.size)
        else:
            active = np.argpartition(-conv, K - 1)[:K]

        # sub‑gradient of the maximal entries
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # ------------------------------------------------------------------
        # additive (projected‑gradient) update
        # ------------------------------------------------------------------
        alpha = add_alpha0 / np.sqrt(it)
        cand_add = cur - alpha * g
        cand_add = proj_simplex(cand_add)
        cand_add_sc = score_of(cand_add)

        # ------------------------------------------------------------------
        # multiplicative (exponential‑weights) update
        # ------------------------------------------------------------------
        eta = mul_eta0 / np.sqrt(it)
        expfac = np.exp(-eta * g)
        cand_mul = cur * expfac
        s_mul = cand_mul.sum()
        if s_mul > 0.0:
            cand_mul = cand_mul / s_mul
        else:
            cand_mul = np.full_like(cur, 1.0 / n)
        cand_mul_sc = score_of(cand_mul)

        # ------------------------------------------------------------------
        # pick the best among current, additive and multiplicative candidates
        # ------------------------------------------------------------------
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc
        else:
            # very occasional uphill move to escape flat plateaus
            if rng.random() < 0.001:
                if cand_add_sc < cand_mul_sc:
                    cur, cur_sc = cand_add, cand_add_sc
                else:
                    cur, cur_sc = cand_mul, cand_mul_sc

        # ------------------------------------------------------------------
        # bookkeeping: keep best solution seen
        # ------------------------------------------------------------------
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # occasional random mass‑transfer move if progress stalls
        # ------------------------------------------------------------------
        if no_improve > 4000 and rng.random() < 0.3:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = proj_simplex(tmp)
                tmp_sc = score_of(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # ------------------------------------------------------------------
        # jittered restart toward the best known solution
        # ------------------------------------------------------------------
        if it % 7000 == 0 and rng.random() < 0.25:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = proj_simplex(tmp)
            tmp_sc = score_of(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # ------------------------------------------------------------------
        # early exit if we have already beaten the target
        # ------------------------------------------------------------------
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # final local refinement (few random transfers) while time permits
    # ----------------------------------------------------------------------
    refin_end = time.monotonic() + min(0.4, deadline - time.monotonic())
    while time.monotonic() < refin_end:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = proj_simplex(tmp)
        tmp_sc = score_of(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

    # return the best vector found as a plain Python list
    return [float(x) for x in best_vec]
