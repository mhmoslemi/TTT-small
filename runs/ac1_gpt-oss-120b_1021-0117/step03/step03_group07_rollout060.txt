import time
import math
import numpy as np

def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (real input)."""
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]


def _score(a: np.ndarray, L: int) -> float:
    """Exact objective used by the evaluator (sum not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, L)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _make_exponential(n: int, r: float) -> np.ndarray:
    """Geometric decreasing sequence r**i, normalised to sum 1."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = (i+1)^(-p) normalised."""
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that minimises the AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # -----------------------------------------------------------------
    # Base vector (parent if available, otherwise a uniform start)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        if s < 1e-12:
            cur = np.full_like(cur, 1.0 / cur.size)
        else:
            cur = cur / s
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    # FFT sizes (next powers of two)
    L_conv = 1 << ((2 * n - 1).bit_length())          # for a*a
    M_grad = 1 << ((3 * n - 2).bit_length())          # for a ⋆ w

    best_vec = cur.copy()
    best_sc = _score(best_vec, L_conv)

    # -----------------------------------------------------------------
    # Quick scan of simple analytic families for a better start
    # -----------------------------------------------------------------
    for r in np.linspace(0.85, 0.9999, 80):
        cand = _make_exponential(n, r)
        sc = _score(cand, L_conv)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 5.0, 80):
        cand = _make_powerlaw(n, p)
        sc = _score(cand, L_conv)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    # simple linear taper
    lin = np.arange(n, 0, -1, dtype=np.float64)
    cand = lin / lin.sum()
    sc = _score(cand, L_conv)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # -----------------------------------------------------------------
    # Smooth‑max gradient descent (exponentiated‑gradient)
    # -----------------------------------------------------------------
    eta0 = 0.5               # base step size for exponentiated update
    beta = 5.0               # initial smoothing parameter
    beta_max = 200.0
    beta_inc = 1.5
    beta_every = 5000        # increase beta every this many iterations
    target = 1.50274349

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # ----- convolution and score -----
        conv = _conv_fft(cur, L_conv)
        max_conv = conv.max()
        cur_sc = 2.0 * n * max_conv / (cur.sum() ** 2)
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            if best_sc < target:
                break

        # ----- soft‑max weighting -----
        # use max_conv for numerical stability
        w = np.exp(beta * (conv - max_conv))
        w /= w.sum()
        w_rev = w[::-1]                     # reverse for correlation

        # ----- gradient via correlation (FFT) -----
        fa = np.fft.rfft(cur, n=M_grad)
        fw = np.fft.rfft(w_rev, n=M_grad)
        corr_full = np.fft.irfft(fa * fw, n=M_grad)   # length M_grad
        L = 2 * n - 1
        idx = L - 1 - np.arange(n)           # map to required indices
        g = 2.0 * corr_full[idx]             # gradient of the smoothed max

        # ----- exponentiated‑gradient step -----
        eta = eta0 / math.sqrt(it)
        step = np.exp(-eta * g)
        # avoid overflow/underflow
        step = np.exp(np.clip(-eta * g, -50.0, 50.0))
        cur = cur * step
        s = cur.sum()
        if s == 0.0:
            cur = np.full_like(cur, 1.0 / n)
        else:
            cur /= s

        # ----- occasional jitter to escape plateaus -----
        if it % 2000 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            cand = cur + jitter
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum
                cand_sc = _score(cand, L_conv)
                if cand_sc < cur_sc:
                    cur, cur_sc = cand, cand_sc
                    if cand_sc < best_sc:
                        best_sc, best_vec = cand_sc, cand.copy()
                        if best_sc < target:
                            break

        # ----- increase smoothing (approach true max) -----
        if it % beta_every == 0:
            beta = min(beta * beta_inc, beta_max)

        # ----- time check -----
        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # Final random mass‑transfer refinement (cheap, few thousand moves)
    # -----------------------------------------------------------------
    it2 = 0
    while time.monotonic() < deadline and it2 < 5000:
        it2 += 1
        # pick donor from a relatively large coefficient, receiver from a small one
        large_idx = int(rng.choice(np.where(best_vec > best_vec.mean())[0]))
        small_idx = int(rng.choice(np.where(best_vec < best_vec.mean())[0]))
        if large_idx == small_idx:
            continue
        max_delta = best_vec[large_idx] * 0.3
        delta = max_delta * rng.random()
        cand = best_vec.copy()
        cand[large_idx] -= delta
        cand[small_idx] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        cand_sc = _score(cand, L_conv)
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

    # -----------------------------------------------------------------
    # Return best vector found (as plain Python floats)
    # -----------------------------------------------------------------
    if best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    return [float(v) for v in best_vec]
