def _conv_self(v):

    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

    conv = _conv_self(v)

import time
import numpy as np


def _conv_self(v: np.ndarray) -> np.ndarray:
    """Linear convolution of v with itself (full length)."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score(v: np.ndarray) -> float:
    """AC1 upper bound for a vector whose entries sum to 1."""
    n = v.shape[0]
    conv = _conv_self(v)
    return 2.0 * n * conv.max()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1, dtype=np.float64)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _grad_logsumexp(v: np.ndarray, beta: float):
    """
    Gradient of the smooth approximation s_beta(v) = (1/beta) log sum_k exp(beta*conv_k)
    where conv = v * v (full linear convolution).
    Returns gradient (size n), full convolution array, and its maximum.
    """
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    conv = _conv_self(v)
    max_conv = conv.max()
    # stable softmax weights
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    M = 1 << ((n + L - 1).bit_length())
    fv = np.fft.rfft(v, n=M)
    fw = np.fft.rfft(w_rev, n=M)
