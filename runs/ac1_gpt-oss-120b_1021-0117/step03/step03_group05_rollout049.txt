def _project_simplex(v):
    # Euclidean projection onto simplex {x>=0, sum=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0.0] = 0.0
    return w


def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Full auto‑convolution of v (length n) using FFT, returning length 2n‑1."""
    n = v.shape[0]
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """AC1 bound from a convolution array and total mass."""
    return 2.0 * n * conv.max() / (total * total)


def _grad_smooth(v: np.ndarray, conv: np.ndarray, beta: float, L: int, n: int) -> np.ndarray:
    """Gradient of the smoothed max (conv) = log‑sum‑exp(beta·conv)/beta."""
    maxc = conv.max()

        # -------------------- occasional random mass‑transfer move --------------------
        if no_improve > 3000 and rng.random() < 0.2:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                conv_tmp = _conv_fft(tmp, L)
                tmp_sc = _score_from_conv(conv_tmp, tmp.sum(), n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        # -------------------- jittered restart toward the best known solution --------------------
        if it % 8000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            conv_tmp = _conv_fft(tmp, L)
            tmp_sc = _score_from_conv(conv_tmp, tmp.sum(), n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # early exit if target already reached
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # final short local refinement (random mass transfers) if time permits
    # ----------------------------------------------------------------------
    refin_end = time.monotonic() + min(0.4, deadline - time.monotonic())
    while time.monotonic() < refin_end:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        conv_tmp = _conv_fft(tmp, L)
        tmp_sc = _score_from_conv(conv_tmp, tmp.sum(), n)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()

    # return the best vector found as a plain Python list
    return [float(x) for x in best_vec]
