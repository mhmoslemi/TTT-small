import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Main candidate generator
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # leave a little margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: uniform random vector of moderate length
        a = rng.random(2000, dtype=np.float64)

    # normalise to the simplex
    a = a / (a.sum() if a.sum() > 0 else 1.0)
    a = _project_simplex(a)

    n = a.size
    # FFT length sufficient for full correlation (size 3n-2)
    L = 1 << ((3 * n - 2).bit_length())

    # ---- evaluation of the initial point ---------------------------
    fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    cur_score = 2.0 * n * conv_full.max()
    best = a.copy()
    best_score = cur_score

    # ---- optimisation hyper‑parameters -----------------------------
    step_lr = 0.2                 # base learning‑rate for Adam
    min_step = 1e-6
    beta1, beta2 = 0.9, 0.999    # Adam moments
    eps = 1e-8
    m = np.zeros_like(a)         # first moment
    v = np.zeros_like(a)         # second moment
    t = 0                         # iteration counter
    alpha = 12.0                  # soft‑max sharpness
    no_improve = 0
    max_no_improve = 6000        # restart after this many stalls
    prob_perturb = 0.004          # occasional random mass‑transfer

    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        t += 1
        # ---- current convolution and score -------------------------
        fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_conv = conv_full.max()
        cur_score = 2.0 * n * max_conv

        # keep the global best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- soft‑max weights over all lags ------------------------
        w = np.exp(alpha * (conv_full - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient via correlation (linear‑time via FFT) --------
        fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(fa * np.conj(fw), n=L)
        grad = 2.0 * corr_full[n - 1 : 2 * n - 1]   # shape (n,)

        # ---- Adam update -------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        delta = step_lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- safeguard for the deadline -----------------------------

    # return the best found sequence as plain Python floats
    return [float(x) for x in best]
