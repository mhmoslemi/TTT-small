import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if v.ndim != 1:
        raise ValueError
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimizer for the AC1 upper bound.
    Uses a projected gradient descent with a top‑K sub‑gradient,
    momentum and occasional jitter. Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. time management
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2. problem size and optional parent
    # --------------------------------------------------------------
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0:
            raise ValueError
        # sanity‑check
        if not np.isfinite(parent_raw).all():
            raise ValueError
        n = int(parent_raw.size)
        a_parent = np.maximum(parent_raw, 0.0)
        s = a_parent.sum()
        a_parent = a_parent / s if s >= 0.01 else np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        # fallback length
        n = 256
        a_parent = None

    # --------------------------------------------------------------
    # 3. generate a few diverse starting points
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # linearly decreasing (tent‑like but one‑sided)
    idx = np.arange(n, dtype=np.float64)
    linear = (n - idx).astype(np.float64)
    linear = linear / linear.sum()

    # quadratic decreasing
    quad = ((n - idx) ** 2).astype(np.float64)
    quad = quad / quad.sum()

    # random Dirichlet‑like start
    rand = rng.random(n)
    rand = rand / rand.sum()

    candidates = [uniform, linear, quad, rand]
    if a_parent is not None:
        candidates.append(a_parent)

    # --------------------------------------------------------------
    # 4. fast scoring (FFT based), works because sum(a)=1
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # pick the best initial candidate
    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    # --------------------------------------------------------------
    # 5. optimisation parameters
    # --------------------------------------------------------------
    eta0 = 0.7          # initial step size
    mu = 0.95           # momentum factor
    K = 15              # number of top convolution entries for sub‑gradient
    target = 1.50274349
    v = np.zeros_like(a)   # momentum accumulator
    it = 0

    # --------------------------------------------------------------
    # 6. main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and current score -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= target:
                break

        # ----- sub‑gradient from top K shifts -----
        if K < conv_len:
            top_idx = np.argpartition(-conv, K)[:K]
        else:
            top_idx = np.arange(conv_len, dtype=np.int64)
        top_vals = conv[top_idx]
        wsum = top_vals.sum()
        if wsum > 0:
            weights = top_vals / wsum
        else:
            weights = np.full_like(top_vals, 1.0 / top_vals.size)

        g = np.zeros_like(a)
        for shift, w in zip(top_idx, weights):
            offset = int(shift) - (n - 1)
            if offset >= 0:
                idx_range = np.arange(offset, n, dtype=np.int64)

            else:

                partner = shift - idx_range

    return [float(v) for v in best]
