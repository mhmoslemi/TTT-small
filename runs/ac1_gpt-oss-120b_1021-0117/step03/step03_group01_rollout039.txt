def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-2.0)

    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = _sqrt_series(256) # fallback
    a0 = np.maximum(a0,0)
    a0 = _project_simplex(a0)   # ensures sum=1

    # evaluate
    best = a0.copy()
    best_score = _score(best)

    # main loop
    a = best.copy()
    step = 0.5
    min_step = 1e-6
    decay = 0.9995
    K = 5
    eps = 1e-6 # tolerance for selecting top active indices

    # set L for conv
    n = a.size
    L = 1 << ((2*n-1).bit_length())

    while time.monotonic() < deadline:
        # compute conv
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0*n*max_val
        # if improved
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # compute subgradient: top K indices
        top_idx = np.argpartition(conv, -K)[-K:]
        g = np.zeros_like(a)
        for p in top_idx:
            lo = max(0, p - (n-1))
            hi = min(n-1, p)
            # corresponding indices j = lo..hi
            # g[lo:hi+1] += 2*a[p - np.arange(lo,hi+1)]
            j = np.arange(lo, hi+1)
            g[j] += 2.0 * a[p - j]
        # optionally normalize gradient
        # step direction
        cand = a - step * g
        cand = _project_simplex(cand)
        # compute candidate score quickly: compute conv max
        fv_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(fv_c * fv_c, n=L)[:2*n-1]
        cand_score = 2.0*n*conv_c.max()
        # accept if better
        if cand_score < cur_score:
            a = cand
        else:
            # maybe accept with probability
            if rng.random() < 0.02:
                a = cand
        # step decay
        step = max(min_step, step*decay)

        # occasional random mass transfer
        if rng.random() < 0.01:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i: j += 1
                frac = rng.random()*0.2
                delta = a[i]*frac
                a2 = a.copy()
                a2[i] -= delta
                a2[j] += delta
                a2 = _project_simplex(a2)
                a = a2

        # early exit: if near deadline break

    # return best as list
    return [float(x) for x in best]

def _optimize(vec, deadline, rng):
    n = len(vec)
    L = 1 << ((2*n-1).bit_length())
    a = vec.copy()
    best = a.copy()
    best_score = _score(a, L)
    cur_score = best_score

    step = 0.5
    min_step = 1e-7
    decay = 0.9995
    beta = 0.9
    momentum = np.zeros_like(a)
    K = 5
    max_active = 20

    while time.monotonic() < deadline:
        # compute conv for a if not already done
        conv = _full_convolution(a, L)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val / (a.sum()**2) # same as _score

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # active indices
        eps = max_val * 1e-4
        active = np.where(conv >= max_val - eps)[0]
        if active.size > max_active:
            # choose random subset
            active = rng.choice(active, size=max_active, replace=False)

        # compute subgradient
        g = np.zeros_like(a)
        for p in active:
            lo = max(0, p - (n - 1))
            hi = min(n - 1, p)
            j = np.arange(lo, hi + 1)
            g[j] += 2.0 * a[p - j]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_score = _score(cand, L)

        # accept improvement or with probability
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            # annealing
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand
                cur_score = cand_score

        # random mass transfer occasionally
        if rng.random() < 0.015:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random()*0.3
                delta = a[i] * frac
                a2 = a.copy()
                a2[i] -= delta
                a2[j] += delta
                a2 = _project_simplex(a2)
                a = a2

        # occasional smoothing
        if rng.random() < 0.008:
            kernel = np.array([0.25,0.5,0.25])
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            a = smooth

        # step decay
        step = max(min_step, step * decay)

        # early exit guard
        if time.monotonic() + 0.5 > deadline:
            break

    return best, best_score

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-2.0)

        rand = _project_simplex(rand)
        alt, alt_score = _optimize(rand, deadline, rng)

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:          # degenerate case – give uniform weight
        w[:] = 1.0 / n
    return w

# ----------------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1+…+x^{2n‑2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ----------------------------------------------------------------------
# Full linear convolution using FFT (real‑to‑real)
# ----------------------------------------------------------------------
def _full_convolution(vec: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    return conv

# ----------------------------------------------------------------------
# Score according to the official evaluator (sum is assumed ≈1)
# ----------------------------------------------------------------------
def _score(vec: np.ndarray, L: int) -> float:
    s = vec.sum()
    if s < 0.01:
        return float('inf')
    conv = _full_convolution(vec, L)
    return float(2.0 * len(vec) * conv.max() / (s * s))

# ----------------------------------------------------------------------
# Core optimisation routine – projected sub‑gradient descent
# ----------------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    best = a.copy()
    best_score = _score(best, L)

    # algorithmic parameters
    step = 0.5
    min_step = 1e-9
    decay = 0.9995
    beta = 0.9                                 # momentum coefficient
    momentum = np.zeros_like(a)
    max_active = 20
    eps_factor = 1e-4                          # active‑set tolerance

    while time.monotonic() < deadline:
        # current convolution & score
        conv = _full_convolution(a, L)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # ---- active set (indices where conv is close to its maximum) ----
        max_val = conv.max()
        eps = max_val * eps_factor
        active = np.where(conv >= max_val - eps)[0]
        if active.size > max_active:
            active = rng.choice(active, size=max_active, replace=False)

        # ---- sub‑gradient (average over active indices) ----
        g = np.zeros_like(a)
        for p in active:
            lo = max(0, p - (n - 1))
            hi = min(n - 1, p)
            j = np.arange(lo, hi + 1)
            g[j] += 2.0 * a[p - j]              # contribution of index p

        # normalise gradient direction
        g_norm = np.linalg.norm(g) + 1e-12
        direction = -g / g_norm                 # descent direction

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * direction

        # ---- candidate generation & projection ----
        cand = _project_simplex(a + step * momentum)

        # ---- evaluate candidate ----
        cand_score = _score(cand, L)

        # ---- accept if improvement, otherwise occasional uphill move ----
        if cand_score < cur_score:
            a = cand
        else:
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand

        # ---- occasional random mass transfer (exploration) ----
        if rng.random() < 0.015:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = a[i] * frac
                a2 = a.copy()
                a2[i] -= delta
                a2[j] += delta
                a = _project_simplex(a2)

        # ---- occasional smoothing (helps to spread spikes) ----
        if rng.random() < 0.008:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            a = _project_simplex(smooth)

        # ---- step‑size decay ----
        step = max(min_step, step * decay)

        # ---- early‑exit guard (leave time for return) ----
        if time.monotonic() + 0.5 > deadline:
            break

    return best, best_score

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- initialise ----------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = _sqrt_series(256)                     # fallback if parent absent
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)                      # enforce sum = 1

    # ---- primary optimisation -------------------------------------------
    best, best_score = _optimize(a0, deadline, rng)

    # ---- secondary start if time permits --------------------------------
    if time.monotonic() + 5.0 < deadline:
        rand_vec = rng.random(a0.shape)
        rand_vec = _project_simplex(rand_vec)
        alt_best, alt_score = _optimize(rand_vec, deadline, rng)
        if alt_score < best_score:
            best = alt_best
            best_score = alt_score

    # ---- return as plain Python list ------------------------------------
    return [float(x) for x in best]
