import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Maximum of the linear convolution (autocorrelation)
# ------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max_i (a*a)[i] using FFT of length L (L >= 2*len(a)-1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Main entry point

    """Return a non‑empty non‑negative sequence with a low AC1 bound."""

    start = time.monotonic()

    # Initial vector (use parent if present)
    # --------------------------------------------------------

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                         # normalize to sum = 1

    n = a.size
    # FFT length – next power of two ≥ 2n‑1

    # optimization hyper‑parameters

        max_val = float(conv.max())
