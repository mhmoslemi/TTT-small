    # Hyperparameters

    beta = 0.9
    jitter_interval = 200
    jitter_std = 0.02
    max_stale = 2000

    v = np.zeros_like(a)

        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
        else:
            stale += 1

        # if very good target met break
        if best_score <= 1.50274349:
            break

        # compute subgradient based on indices achieving max
        eps = max_val * 1e-12 + 1e-12
        idxs = np.where(conv >= max_val - eps)[0]
        # compute gradient accumulator
        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            # vector addition per offset
            if offset >= 0:
                # first part
                g[:n - offset] += a[offset:]
                # second part
                g[offset:] += a[:n - offset]
            else:
                off = -offset
                # offset negative
                g[off:] += a[:n - off]
                g[:n - off] += a[off:]
        # average over number of max indices
        if len(idxs) > 1:
            g /= len(idxs)

        # momentum update
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g

        a = _proj_simplex(a - eta * v)

        if it % jitter_interval == 0:

            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))

def gradient_from_peaks(a, conv, top_k):
    n = a.size
    idxs = np.argpartition(conv, -top_k)[-top_k:] if conv.size > top_k else np.arange(conv.size)
    g = np.zeros_like(a)
    for k in idxs:
        offset = k - (n - 1)
        if offset >= 0:
            g[:n - offset] += a[offset:]
            g[offset:] += a[:n - offset]
        else:
            off = -offset
            g[off:] += a[:n - off]
            g[:n - off] += a[off:]
    g /= max(1, len(idxs))
    return g

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        n = 256
        a = np.ones(n, dtype=np.float64) / n
    else:
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = np.maximum(parent, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

    # Helper projection
    def proj_simplex(v):
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        rho = np.where(u - cssv / np.arange(1, len(u)+1) > 0)[0]
        if rho.size == 0:
            theta = 0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    best = a.copy()
    # compute initial score
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best_score = float(2.0 * n * conv.max())

    # hyperparameters
    eta0 = 0.9
    beta = 0.93
    top_k = 20
    jitter_interval = 200
    jitter_std = 0.015
    max_stale = 3000

    v = np.zeros_like(a)
    it = 0
    stale = 0

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:
        # compute convolution
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
            # early exit if passes target
            if best_score <= 1.50274349:
                break
        else:
            stale += 1

        # compute gradient from top_k peaks
        if conv.size > top_k:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size)

        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                g[:n - offset] += a[offset:]
                g[offset:] += a[:n - offset]
            else:
                off = -offset
                g[off:] += a[:n - off]
                g[:n - off] += a[off:]
        g /= max(1, len(idxs))

        # momentum update
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = proj_simplex(a - eta * v)

        # jitter
        if it % jitter_interval == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = proj_simplex(a * jitter)
            v[:] = 0.0
            stale = 0

        # restart after stagnation
        if stale >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = proj_simplex(a * perturb)
            v[:] = 0.0
            stale = 0

    return [float(x) for x in best]

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # fallback to uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Returns the best non‑negative step‑function (list of floats) found
    before the allocated time expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise sequence (fallback to uniform if parent missing)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = np.maximum(parent, 0.0)
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # 2.  Prepare convolution helpers
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 3.  Track the best candidate seen so far
    # ------------------------------------------------------------------
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best = a.copy()
    best_score = float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # 4.  Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    eta0 = 0.9          # initial step magnitude
    beta = 0.93         # momentum coefficient
    top_k = 20          # number of convolution peaks used for gradient
    jitter_every = 200  # steps between random log‑normal perturbations
    jitter_std = 0.015  # jitter standard deviation
    max_stale = 3000    # steps without improvement before a restart

    v = np.zeros_like(a)   # momentum buffer
    it = 0
    stale = 0

    # ------------------------------------------------------------------
    # 5.  Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 6.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution (FFT) ---------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- keep best -----------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
            if best_score <= 1.50274349:
                break
        else:
            stale += 1

        # ----- gradient from top‑k convolution peaks --------------------
        if conv.size > top_k:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size)

        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                g[:n - offset] += a[offset:]
                g[offset:] += a[:n - offset]
            else:
                off = -offset
                g[off:] += a[:n - off]
                g[:n - off] += a[off:]
        g /= max(1, len(idxs))

        # ----- momentum step & projection -------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v)

        # ----- occasional jitter to escape plateaus --------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v[:] = 0.0
            stale = 0

        # ----- restart if stagnation persists ---------------------------
        if stale >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v[:] = 0.0
            stale = 0

    # ------------------------------------------------------------------
    # 7.  Optional short local‑search phase (time permitting)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best * np.exp(rng.normal(0.0, 0.005, size=n))
        cand = _proj_simplex(cand)
        fa = np.fft.rfft(cand, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        score = 2.0 * n * conv.max()
        if score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # 8.  Return the best sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
