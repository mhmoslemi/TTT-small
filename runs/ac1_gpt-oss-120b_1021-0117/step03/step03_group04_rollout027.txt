import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Projected sub‑gradient/Adam optimizer for the AC1 bound.
    Returns the best coefficient list found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Base length and parent (if any)
    # ------------------------------------------------------------------
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        n = raw.size
        parent = np.maximum(raw, 0.0)
    except Exception:          # no parent supplied
        n = 256
        parent = None

    # ------------------------------------------------------------------
    # 2. Helper to normalise a vector to the simplex (sum = 1)
    # ------------------------------------------------------------------
    def _normalise(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-12:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    # ------------------------------------------------------------------
    # 3. Initial candidates (algorithmic, no huge literals)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    cand_uniform = np.full(n, 1.0 / n)
    cand_tent = np.minimum(idx + 1, n - idx)          # symmetric triangle, edges zero
    cand_tent = _normalise(cand_tent)

    # exponential decay, several rates
    cand_exp1 = _normalise(np.power(0.999, idx))
    cand_exp2 = _normalise(np.power(0.995, idx))

    # power‑law
    cand_pow = _normalise(1.0 / (idx + 1))

    # half‑cosine (monotonically decreasing)
    cand_cos = _normalise(np.cos(np.pi * idx / (2 * n)))  # starts at 1, ends at 0

    # Gaussian centred near the start
    sigma = n / 12.0
    cand_gauss = _normalise(np.exp(-0.5 * (idx / sigma) ** 2))

    # random Dirichlet‑like start
    cand_rand = _normalise(rng.random(n))

    candidates = [cand_uniform, cand_tent, cand_exp1, cand_exp2,
                  cand_pow, cand_cos, cand_gauss, cand_rand]
    if parent is not None:
        candidates.append(_normalise(parent))

    # ------------------------------------------------------------------
    # 4. Fast AC1 evaluation (sum is already 1)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    best = None
    best_score = float("inf")
    for c in candidates:
        sc = fast_score(c)
        if sc < best_score:
            best_score = sc
            best = c.copy()
    a = best.copy()

    # ------------------------------------------------------------------
    # 5. Adam‑style optimisation of the max‑convolution sub‑gradient
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    eta = 0.02                     # base step size
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    it = 0
    target = 1.50274349
    stagnation = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and current AC1 -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            stagnation = 0
            if best_score <= target:
                break
        else:
            stagnation += 1

        # ----- sub‑gradient of the maximal convolution entry -----
        # handle numerical ties
        tol = max(1e-12, cur_max * 1e-12)
        max_idx = np.flatnonzero(np.abs(conv - cur_max) <= tol)

        g = np.zeros_like(a)
        for k in max_idx:
            # gradient: g_j = 2 * a_{k-j} for valid indices
            j = np.arange(n)

            ok = (partner >= 0) & (partner < n)
