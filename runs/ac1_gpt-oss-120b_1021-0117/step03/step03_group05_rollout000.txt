def project_simplex(v):
    # v >= arbitrary, sum to 1, nonnegative.
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initial vector (use provided parent if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: small uniform vector
        a0 = np.full(256, 1.0, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s0   # normalise to sum=1

    cur = a0.copy()
    n = cur.size
    # FFT size (next power of two for convolution length)
    _L = 1 << ((2 * n - 1).bit_length())

    def _conv_self(v):
        fv = np.fft.rfft(v, n=_L)
        conv = np.fft.irfft(fv * fv, n=_L)[: 2 * n - 1]
        return conv

    def _score(v):
        conv = _conv_self(v)
        maxb = conv.max()
        sm = v.sum()
        if sm < 0.01:
            return float('inf')
        return 2.0 * n * maxb / (sm * sm)

    # initialise scores
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # algorithmic parameters
    target = 1.50274349
    it = 0
    no_improve = 0
    # step sizes (tuned)
    add_alpha0 = 0.7
    mul_eta0 = 0.7
    # active set size

        it += 1
        conv = _conv_self(cur)
        maxb = conv.max()
        # choose active indices: top K of conv

            active = np.arange(conv.size)
        else:
            # argpartition gives K largest (unsorted)
            active = np.argpartition(-conv, K_max - 1)[:K_max]

        # gradient from active terms
        g = np.zeros_like(cur)
        # vectorised accumulation: loop over active indices
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # indices of cur that affect conv[k]
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # additive projected‑gradient step
        # ------------------------------------------------------------------

        cand_add = _project_simplex(cand_add)
        cand_add_sc = _score(cand_add)

        # ------------------------------------------------------------------

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
