import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimiser for the AC1 upper bound.
    Uses a momentum based sub‑gradient method driven by the
    largest convolution peaks.  The best sequence seen before the
    time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 2. Base sequence (fallback to uniform if the parent is absent)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    # enforce non‑negativity and normalise to a probability simplex
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
        a = _proj_simplex(a)

    # ------------------------------------------------------------------
    # 3. Initialise best candidate
    # ------------------------------------------------------------------
    def _score(vec):
        """score = 2 * n * max(conv)  (sum(vec) == 1)"""
        conv = np.convolve(vec, vec)
        return 2.0 * n * float(conv.max())

    best = a.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 4. FFT parameters (used each iteration)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 5. Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    eta0 = 0.7            # initial step size
    beta = 0.85           # momentum decay
    top_k = 6             # number of convolution peaks used for the gradient
    jitter_every = 250    # steps between multiplicative jitter
    jitter_std = 0.015    # jitter standard deviation (log‑normal)
    restart_patience = 600
    max_iters = 200000    # hard cap, will be cut by time limit

    v = np.zeros_like(a)  # momentum buffer
    it = 0
    stale = 0

    # ------------------------------------------------------------------
    # 6. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline and it < max_iters:
        # ---- convolution (FFT, linear) ----------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ---- keep the best ------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
        else:
            stale += 1

        # ---- gradient from the top‑k convolution peaks --------------------
        # offsets relative to centre (index n-1)
        if conv.size > top_k:

    # ------------------------------------------------------------------
