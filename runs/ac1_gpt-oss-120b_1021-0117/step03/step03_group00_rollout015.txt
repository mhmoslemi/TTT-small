def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    # optional RNG
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialize a
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform distribution length 2000? Or arbitrary
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # make nonnegative and normalize sum=1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / a.size)

    n = int(a.shape[0])
    # FFT length
    L = 1 << ((3 * n - 2).bit_length())

    # projection onto simplex
    def _project_simplex(v):
        # Euclidean projection onto simplex (sum=1, nonneg)
        # using algorithm from Duchi et al.
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # initial objective
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    cur_sc = 2.0 * n * conv.max()
    best_sc = cur_sc
    best_vec = a.copy()

    # optimization parameters
    eps0 = 1.0
    decay = 0.996
    eps_min = 1e-8
    eta0 = 0.5
    # momentum optional
    # momentum = np.zeros_like(a)
    # beta = 0.9

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # recompute a_fft and conv (current)
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()

        # smoothing weight
        eps = eps0 * (decay ** it)
        if eps < eps_min:
            eps = eps_min
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:

        else:

        # gradient: conv of a with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # learning rate

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise a (sum = 1, non‑negative)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform vector of length 2000
        a = np.full(2000, 1.0 / 2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a.fill(1.0 / a.size)

    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())          # FFT length >= 3n‑2

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    eps0 = 1.0            # initial smoothing for soft‑max
    decay = 0.996         # exponential decay of eps
    eps_min = 1e-6
    eta0 = 0.5            # base learning rate
    sym_freq = 10         # how often to enforce symmetry
    stagnation_limit = 5000
    target_score = 1.50274349

    best_vec = a.copy()
    # initial score (2·n·max(conv) because sum(a)=1)
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = float(2.0 * n * conv.max())

    it = 0
    stagnated = 0

    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # current convolution and score
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_sc = float(2.0 * n * cur_max)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            stagnated = 0
        else:
            stagnated += 1

        if best_sc < target_score:
            break

        # --------------------------------------------------------------
        # smoothed weight vector w (approximate arg‑max)
        # --------------------------------------------------------------
        eps = eps0 * (decay ** it)
        if eps < eps_min:
            eps = eps_min

        max_idx = int(conv.argmax())
        if eps <= eps_min:
            w = np.zeros_like(conv)
            w[max_idx] = 1.0
        else:
            w = np.exp((conv - cur_max) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum

        # --------------------------------------------------------------
        # gradient of the smoothed objective
        # --------------------------------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # length >= 3n‑2
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # simple projected gradient step (learning rate ∝ 1/√it)
        # --------------------------------------------------------------
        eta = eta0 / math.sqrt(it)
        cand = a - eta * grad
        cand = _project_simplex(cand)

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        if it % sym_freq == 0:

            cand = _project_simplex(cand)

        a = cand

        # occasional random mass transfer to escape stagnation
        # --------------------------------------------------------------
        if stagnated > stagnation_limit:

                if j >= i:
                    j += 1
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)
            stagnated = 0

    # return best found sequence as plain Python list
    # ------------------------------------------------------------------
    if best_vec.size == 0:
        best_vec = a
    return [float(x) for x in best_vec]
