def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes the
    AC1 upper bound defined by `evaluate_sequence`.  The method is a
    projected‑gradient descent with momentum, using FFT‑based convolution to
    obtain the max autocorrelation value efficiently.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise a – use the parent if it exists, otherwise a short sqrt‑series
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: the analytic sqrt series (used by the reference solution)
        n0 = 256
        a = np.empty(n0, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n0):
            if k == 1:
                s = 0.0
            else:
                s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
            a[k] = (1.0 - s) / 2.0

    a = np.maximum(a, 0.0)               # enforce non‑negativity
    total = a.sum()
    if total == 0.0:                     # degenerate guard
        a[0] = 1.0
        total = 1.0
    a = a / total                         # normalise to L¹ = 1 (scaling invariant)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    # ------------------------------------------------------------------
    # convolution helper (full linear convolution via FFT)
    # ------------------------------------------------------------------
    def _conv_full(x: np.ndarray) -> np.ndarray:
