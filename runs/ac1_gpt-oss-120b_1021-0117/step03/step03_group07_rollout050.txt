import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities (all top‑level, no closures)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (real input)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum is forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

        a[1] = 0.0

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _make_cosine_window(n: int, power: float = 1.0) -> np.ndarray:
    """Hann‑like window (zero at the ends) raised to a power."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** power
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _grad_smooth_max(a: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of a smoothed max (log‑sum‑exp) of a*a."""
    n = a.shape[0]
    # convolution a*a
    conv = _conv_fft(a)                     # length 2n‑1
    # soft‑max weights, numerically stable
    log_max = alpha * conv.max()
    w = np.exp(alpha * conv - log_max)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient = 2 * (a * rev(w)) shifted
    w_rev = w[::-1]
    L2 = 1 << ((3 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=L2)
    fw = np.fft.rfft(w_rev, n=L2)
    conv2 = np.fft.irfft(fa * fw, n=L2)[: 3 * n - 2]
    m = 2 * n - 1
    grad = 2.0 * conv2[m - 1 - np.arange(n)]
    return grad

        grad = _grad_smooth_max(cur, alpha)
        eta = eta0 / np.sqrt(it)
        cand = cur - eta * grad
        cand = _project_simplex(cand)

        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break
        else:
            # occasional acceptance of a slight uphill move (simulated annealing)
            if rng.random() < 0.02:
                cand2 = cur + eta * grad
                cand2 = _project_simplex(cand2)
                cand2_sc = _score(cand2)
                if cand2_sc < cur_sc:
                    cur, cur_sc = cand2, cand2_sc
                    if cand2_sc < best_sc:
                        best_sc, best_vec = cand2_sc, cand2.copy()
                        if best_sc < target:
                            break

        # small jitter every few thousand iterations
        if it % 4000 == 0:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

    # ------------------------------------------------------------------
    # Phase B – targeted random mass‑transfer refinements
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick two indices, move a bit of mass
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n - 1))
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.4
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break

        # occasional aggressive redistribution
        if rng.random() < 0.001:
            jitter = rng.normal(scale=2e-4, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

    # ------------------------------------------------------------------
    # Final safeguard
    # ------------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
