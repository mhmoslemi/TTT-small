def gradient_from_conv(a, conv, idxs):
    n = len(a)
    g = np.zeros_like(a)
    for k in idxs:
        offset = k - (n - 1)
        if offset >= 0:
            # i from offset .. n-1
            i = np.arange(offset, n)
            j = k - i  # this yields indices j from n-1 ... offset
            # add a[j] to g[i]
            g[i] += a[j]
        else:
            # offset negative => shift left
            i = np.arange(0, n + offset)  # because offset negative
            j = k - i
            g[i] += a[j]
    return 2.0 * g / len(idxs)  # average gradient

import time, numpy as np

def _proj_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
