def _conv_fft(x, L):
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]

    n = x.size
    conv = _conv_fft(x, L)
    max_b = conv.max()
    sum_a = x.sum()
    if sum_a <= 0:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback to uniform of length 2000
        n0 = 2000
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size, dtype=np.float64)
    else:
        a = a / s

    L = 1 << ((2 * n - 1).bit_length())

    cur = a.copy()
    cur_score = _score_of(cur, L)

    best = cur.copy()
    best_score = cur_score

    cooling = 0.99995
    move_frac = 0.1
    iter_cnt = 0
    no_improve = 0

    # Main loop
    while time.monotonic() < deadline:
        iter_cnt += 1

        # --- Random transfer move ---
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue

import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself via FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]

def _score_of(x: np.ndarray, L: int) -> float:
    """Score = 2·n·max(conv) / (sum(x)²)."""
    n = x.size
    conv = _conv_fft(x, L)
    max_b = conv.max()
    s = x.sum()
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

    # ----------------------------------------------------------------------
    cur = a.copy()
    cur_score = _score_of(cur, L)

    best = cur.copy()
    best_score = cur_score

    # ----------------------------------------------------------------------
    # Simulated‑annealing style search parameters.
    # ----------------------------------------------------------------------
    T = 0.5                # initial temperature
    cooling = 0.99995      # exponential cooling factor per iteration
    move_frac = 0.10       # max fraction of mass moved in a basic step
    it = 0
    no_improve = 0

    # ----------------------------------------------------------------------
    # Main optimisation loop.
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- Random mass‑transfer move (basic SA step) -----
        i = rng.integers(0, n)
        if cur[i] > 0.0:
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * move_frac * rng.random()
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum > 0.0:
                cand /= cand_sum
                cand_score = _score_of(cand, L)

                diff = cand_score - cur_score
                if diff < 0.0 or rng.random() < math.exp(-diff / (T + 1e-12)):
                    cur = cand
                    cur_score = cand_score
                    no_improve = 0
                else:
                    no_improve += 1

                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                    no_improve = 0

        # ----- Periodic multiplicative (mirror‑descent) step -----
        if it % 30 == 0:
            conv = _conv_fft(cur, L)
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g = np.zeros_like(cur)
            g[idx] = 2.0 * cur[k - idx]
            eta = 0.6 / math.sqrt(it)        # decaying step size
            cand2 = cur * np.exp(-eta * g)
            cand2_sum = cand2.sum()
            if cand2_sum > 0.0:
                cand2 /= cand2_sum
                cand2_score = _score_of(cand2, L)
                if cand2_score < cur_score:
                    cur = cand2
                    cur_score = cand2_score
                if cand2_score < best_score:
                    best = cand2.copy()
                    best_score = cand2_score

        # ----- Temperature schedule -----
        T = max(T * cooling, 1e-6)

        # ----- Restart after long stagnation -----
        if no_improve > 5000:
            jitter = rng.normal(scale=0.02, size=n)
            cur = best + jitter
            cur = np.maximum(cur, 0.0)
            cur_sum = cur.sum()
            if cur_sum > 0.0:
                cur /= cur_sum
                cur_score = _score_of(cur, L)
            no_improve = 0

        # ----- Early exit if target reached -----
        if best_score < 1.50274349:
            break

    # Return the best sequence as a plain Python list of floats.
    return [float(v) for v in best]
