import time, numpy as np

def _proj_simplex(v):
    """ Euclidean projection of v onto the probability simplex """
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    # condition: u_i - cssv_i / i > 0
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all zero
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # compare with uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    fu = np.fft.rfft(uniform, n=fft_len)
    conv_u = np.fft.irfft(fu * fu, n=fft_len)[:conv_len]
    uniform_score = float(2.0 * n * conv_u.max())
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()
        v[:] = 0.0

    no_improve = 0
    it = 0
    center = n - 1

    # ------------------------------------------------------------------
    # 6. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution (FFT) ----------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- keep best -------------------------------------------------
        if cur_score < best_score - 1e-15:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ----- gradient from top‑k convolution peaks --------------------
        if conv.size > top_k:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size)

        g = np.zeros_like(a)
        for k in idxs:
            offset = int(k - center)
            if offset >= 0:
                g[offset:] += a[:n - offset]
            else:
                g[:n + offset] += a[-offset:]
        g *= 2.0
        if len(idxs):
            g /= len(idxs)               # average contribution

        # ----- momentum step & projection -------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = a - eta * v
        a = _proj_simplex(a)

        # ----- occasional random jitter ---------------------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v[:] = 0.0
            no_improve = 0

        # ----- restart after long stagnation ----------------------------
        if no_improve >= restart_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v[:] = 0.0
            no_improve = 0

    # ------------------------------------------------------------------
    # 7. Return best sequence (plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
