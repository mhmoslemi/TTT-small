def _make_exponential(n,r):
    a = np.power(r, np.arange(n,dtype=np.float64))
    a_sum = a.sum()
    return a / a_sum if a_sum>0 else np.full(n,1.0/n)

def _make_powerlaw(n,p):
    a = 1.0/np.power(np.arange(1,n+1,dtype=np.float64),p)
    a_sum = a.sum()
    return a / a_sum if a_sum>0 else np.full(n,1.0/n)

def _score_direct(a):
    # compute using convolution with fft
    n = a.size
    L = 1 << ((2*n-1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[:2*n-1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv)

def propose_candidate(seed=42, budget_s=1100, **kwargs):

import time
import numpy as np

# -------------------------------------------------------------
# Simple families for a quick initial guess
# -------------------------------------------------------------
def _make_exponential(n, r):
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_powerlaw(n, p):
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_linear(n):
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n):
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


# -------------------------------------------------------------
# Exact objective (identical to the evaluator's formula)
# -------------------------------------------------------------
def _score_vec(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)

    # ----------- explore a few analytic families -------------------------------
    for r in np.linspace(0.90, 0.9999, 60):
        cand = _make_exponential(n, r)
        sc = _score_vec(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 4.0, 60):
        cand = _make_powerlaw(n, p)
        sc = _score_vec(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    cand = _make_linear(n)
    sc = _score_vec(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()
    cand = _make_quadratic(n)
    sc = _score_vec(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # ----------- parameters for the refined search ---------------------------
    target = 1.50274349
    eta0 = 0.5               # base step size for mirror descent
    gamma = 5.0              # start smoothing factor
    gamma_max = 5000.0
    gamma_mul = 1.001        # slow increase each iteration
    eval_every = 5           # recompute exact score this often
    L_conv = 1 << ((2 * n - 1).bit_length())          # length for self‑convolution
    L_corr = 1 << ((3 * n - 2).bit_length())          # length for correlation (≥ n+2n‑2)

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ---------- convolution of current vector ----------
        Fa = np.fft.rfft(cur, n=L_corr)
        conv_full = np.fft.irfft(Fa * Fa, n=L_corr)[: 2 * n - 1]
        max_conv = conv_full.max()
        cur_sc = 2.0 * n * max_conv

        # keep the best ever seen
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            if best_sc < target:
                break

        # ---------- smoothed max (log‑sum‑exp) ----------
        exp_arg = gamma * (conv_full - max_conv)          # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        w_norm = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # ---------- gradient = correlation(cur, w_norm) ----------
        Fw = np.fft.rfft(w_norm, n=L_corr)
        corr_full = np.fft.irfft(Fa * np.conj(Fw), n=L_corr)
        g = corr_full[:n]                # no extra γ factor (already inside w)

        # ---------- mirror‑descent update ----------
        eta = eta0 / np.sqrt(it)
        cand = cur * np.exp(-eta * g)    # stay non‑negative
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand /= cand_sum
        else:
            cand = np.full_like(cur, 1.0 / n)

        # ---------- evaluate candidate (every few iterations) ----------
        if it % eval_every == 0:
            cand_sc = _score_vec(cand, L=L_conv)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
                    if best_sc < target:
                        break
            else:
                # occasional Metropolis‑style acceptance
                T = 0.01 / (np.sqrt(it) + 1e-12)
                if rng.random() < np.exp(-(cand_sc - cur_sc) / T):
                    cur, cur_sc = cand, cand_sc
                    if cand_sc < best_sc:
                        best_sc, best_vec = cand_sc, cand.copy()
                        if best_sc < target:
                            break
        else:
            # without a fresh exact score we still move forward
            cur = cand

        # ---------- slowly increase smoothing ----------
        gamma = min(gamma * gamma_mul, gamma_max)

        # ---------- occasional random perturbations ----------
        if it % 2000 == 0:
            # tiny Gaussian jitter
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0:
                tmp /= tmp_sum
                tmp_sc = _score_vec(tmp, L=L_conv)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc

            # random mass‑transfer move
            if rng.random() < 0.3:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n - 1))
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp /= tmp.sum()
                tmp_sc = _score_vec(tmp, L=L_conv)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc

        # ---------- time check ----------
        if time.monotonic() >= deadline:
            break

    # safeguard – should never happen, but keep a valid fallback
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
