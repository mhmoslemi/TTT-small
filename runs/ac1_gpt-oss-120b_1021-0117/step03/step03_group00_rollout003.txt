def _score(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2*n -1]
    return float(2.0 * n * conv.max())
```

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.arange(1, n + 1)
    rho = np.where(u - css / rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Score for a vector with sum == 1 (the denominator cancels)
# ----------------------------------------------------------------------
def _score(a: np.ndarray, fft_len: int = None) -> float:
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Shape generators (non‑negative, zero at the ends)
# ----------------------------------------------------------------------
def _sine_shape(n):
    i = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (i + 1) / (n + 1))


def _sine_power_shape(n, p):
    a = _sine_shape(n)
    return np.power(a, p)


def _triangular_shape(n):
    i = np.arange(n, dtype=np.float64)
    return (i + 1) * (n - i)


def _gaussian_shape(n, sigma):
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def _normalize(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Determine length and a baseline vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = a0.size
        a0 = _normalize(a0)
    else:
        # fallback length (chosen to be modest)
        n = 2000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # Scan a few analytic families for a good start
    # --------------------------------------------------------------
    best_vec = a0.copy()
    best_sc = _score(best_vec)
    candidates = []

    # sine
    candidates.append(_normalize(_sine_shape(n)))
    # sine powers
    for p in (2, 3, 4):
        candidates.append(_normalize(_sine_power_shape(n, p)))
    # triangular
    candidates.append(_normalize(_triangular_shape(n)))
    # gaussians with a few sigmas
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        candidates.append(_normalize(_gaussian_shape(n, sigma)))

    # Evaluate the analytic candidates
    for cand in candidates:
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()

    # --------------------------------------------------------------
    # Parameters for the smoothed‑max descent
    # --------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    L_fft = 1 << ((3 * n - 2).bit_length())   # enough for gradient convolutions
    L_score = 1 << ((2 * n - 1).bit_length())  # for score evaluations

    eps0 = 0.25
    eps_decay = 0.9996
    eps_min = 1e-7
    step0 = 0.5
    no_improve = 0
    it = 0
    target = 1.50274349

    # --------------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution and max ----
        a_fft = np.fft.rfft(cur, n=L_fft)
        conv = np.fft.irfft(a_fft * a_fft, n=L_fft)[: 2 * n - 1]
        max_val = conv.max()

        # ---- smoothed weights (softmax) ----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # ---- gradient of the smoothed objective ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L_fft)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L_fft)
        idx = (2 * n - 2) - np.arange(n)   # map to positions of a
        g = 2.0 * grad_full[idx]

        # ---- step with adaptive magnitude ----
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / (math.sqrt(it) * grad_norm)
        cand = cur - alpha * g
        cand = _project_simplex(cand)

        # ---- evaluate candidate ----
        cand_sc = _score(cand, fft_len=L_score)

        # ---- accept / reject ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
        else:
            # occasional Metropolis acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ---- periodic symmetrisation (helps with evenness) ----
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, fft_len=L_score)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                    no_improve = 0

        # ---- random mass transfer if stuck ----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, fft_len=L_score)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ---- early exit if we beat the target ----
        if best_sc < target:
            break

    # Return best vector as plain Python list
    return [float(x) for x in best_vec]
