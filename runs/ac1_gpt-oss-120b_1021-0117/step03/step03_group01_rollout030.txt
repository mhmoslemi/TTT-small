# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = target)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """Project a vector onto the non‑negative simplex of a given total mass."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - target_sum
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    # Renormalise exactly to target_sum (numerical safety)
    if w_sum > 0:
        w *= target_sum / w_sum
    else:
        w[:] = target_sum / n
    return w

# Helper: maximum of the autocorrelation via FFT (a*a)
# ----------------------------------------------------------------------
def _conv_max(vec: np.ndarray) -> float:
    """Return max_{k} (vec * vec)[k] using an FFT based convolution."""
    n = vec.size
    # length of FFT – next power of two covering 2n‑1 points
    L = 1 << ((2 * n - 2).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(conv.max())
