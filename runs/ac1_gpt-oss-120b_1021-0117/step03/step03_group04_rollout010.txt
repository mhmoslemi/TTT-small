import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fast_score(x: np.ndarray) -> float:
    """Fast evaluation of the AC1 upper bound (ignores the sum denominator)."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())

def _subgradient(a: np.ndarray, top_idx: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max‑convolution approximation."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for shift, w in zip(top_idx, weights):
        offset = shift - (n - 1)
        if offset >= 0:
            idx = np.arange(offset, n, dtype=np.int64)
            partner = shift - idx
        else:
            idx = np.arange(0, n + offset, dtype=np.int64)
            partner = shift - idx
        g[idx] += w * 2.0 * a[partner]
    return g

        else:
            a = uniform.copy()

        # ----- occasional jitter to escape flat regions -----
        if it % 200 == 0:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a *= jitter
            a /= a.sum()

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
```
