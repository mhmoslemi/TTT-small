def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

# Simplex projection (L1 = 1, non‑negative)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a non‑negative vector onto the simplex sum=1."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # safeguard against numerical zero‑sum
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

            score_s = 2.0 * n * max_s
            if score_s < cur_score:
                a = smooth
                cur_score = score_s
                max_idx = idx_s
                if score_s < best_score:
                    best_score = score_s
                    best = smooth.copy()

        # ---- step‑size decay ----
        step = max(min_step, step * decay)

        # ---- early exit guard to keep time for return ----
        if time.monotonic() + 0.3 > deadline:
            break

    return best, best_score


# --------------------------------------------------------------
# Public entry point required by the harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # initialise – use the supplied parent if present, otherwise sqrt_series
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = _sqrt_series(256)

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()               # project onto the simplex (sum = 1)

    # primary optimisation run
    best, best_score = _optimize(a0, deadline, rng)

    # secondary random start if time permits
    if time.monotonic() + 4.0 < deadline:
        n = a0.size
        rand_vec = rng.random(n)
        rand_vec = rand_vec / rand_vec.sum()
        alt_best, alt_score = _optimize(rand_vec, deadline, rng)
        if alt_score < best_score:
            best, best_score = alt_best, alt_score

    # final fallback (uniform) – should never be needed but guarantees validity
    if best_score > 2.0:
        n = a0.size
        best = np.ones(n) / n

    return [float(x) for x in best]
