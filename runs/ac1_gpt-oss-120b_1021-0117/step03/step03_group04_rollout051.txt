def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 upper bound.
    Uses a smoothed‑max gradient with momentum, simplex projection and occasional jitter.
    Returns the best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Length and parent handling
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        n = parent_arr.size
        parent = np.clip(parent_arr, 0.0, 1000.0)
        s = parent.sum()
        if s >= 0.01:
            parent = parent / s
        else:
            parent = np.full(n, 1.0 / n, dtype=np.float64)
        have_parent = True
    except Exception:
        n = 256
        parent = None
        have_parent = False

    # ------------------------------------------------------------------
    # Initialize a diverse set of candidates
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1, n - idx) - 1.0
    tent = np.clip(tent, 0.0, None)
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:
        tent = uniform.copy()

    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((idx - (n - 1) / 2) / sigma) ** 2)
    gauss = gauss / gauss.sum()

    rand = rng.random(n)
    rand = rand / rand.sum()

    candidates = [uniform, tent, gauss, rand]
    if have_parent:
        candidates.append(parent)

    # ------------------------------------------------------------------
    # Helper: fast AC1 score using FFT (sum is 1)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    L = 1 << ((conv_len + n - 1).bit_length())  # >= conv_len + n -1

    def fast_score(a: np.ndarray) -> float:
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[:conv_len]
        return 2.0 * n * conv.max()

    # Pick the best starter
    best = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    # ------------------------------------------------------------------
    # Optimization parameters
    # ------------------------------------------------------------------
    mu = 0.9           # momentum decay
    eta0 = 1.0         # initial step magnitude
    beta = 10.0        # soft‑max temperature
    target = 1.50274349
    jitter_interval = 500
    max_iters_restart = 4000
    v = np.zeros_like(a)

    # ------------------------------------------------------------------
    # Main search loop (multiple restarts)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # Choose a fresh start: a jittered copy of the current best or random Dirichlet
        if rng.random() < 0.3:
            a = best.copy()
            a = a * np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a)
        else:
            a = rng.random(n)
            a = a / a.sum()

        v[:] = 0.0
        it = 0

        # Gradient descent with momentum on this start
        while time.monotonic() < deadline and it < max_iters_restart:
            it += 1

            # Convolution a*a via FFT
            A = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(A * A, n=L)[:conv_len]
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # Record best ever
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                if best_score <= target:
                    return [float(x) for x in best]

            # Smoothed max weights (soft‑max)
            w = np.exp(beta * (conv - cur_max))
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum

            # Gradient = 2 * (w_rev * a) evaluated on the central n entries
            w_rev = w[::-1]
            W_rev = np.fft.rfft(w_rev, n=L)
            cross = np.fft.irfft(A * W_rev, n=L)[:conv_len + n - 1]
            g = 2.0 * cross[n - 1:n - 1 + n]

            # Momentum update
            v = mu * v + (1.0 - mu) * g

            # Diminishing step size
            eta = eta0 / np.sqrt(it)

            # Projected update
            a = _proj_simplex(a - eta * v)

            # Occasional jitter to escape plateaus
            if it % jitter_interval == 0:
                a = a * np.exp(rng.normal(0.0, 0.015, size=n))
                a = _proj_simplex(a)

    # ------------------------------------------------------------------
    # Return best found (fallback to uniform if somehow missing)
    # ------------------------------------------------------------------
    if best is None:
        best = uniform
    return [float(x) for x in best]
