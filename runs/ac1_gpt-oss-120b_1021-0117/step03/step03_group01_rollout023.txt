def improve_via_lp(a, n_iters, t, deadline, rng):
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    for it in range(n_iters):
        # compute conv and M
        L = 1 << ((2 * a.size - 1).bit_length())
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*a.size-1]
        M = conv.max()
        # Build A
        A = np.empty((2*a.size-1, a.size), dtype=np.float64)
        # fill
        n = a.size
        for j in range(n):
            A[j:j+n, j] = a
        # Solve LP: max sum(b) -> min -sum(b)
        c = -np.ones(n, dtype=np.float64)
        # bounds for b >=0
        bounds = [(0, None)] * n
        # Note: numpy pass A_ub and b_ub
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b == 0:
            break
        b = b / sum_b  # normalize to sum 1
        # convex combo
        a = (1-t) * a + t * b
        # evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best = a.copy()
        # adjust t maybe gradually reduce?
        # Also maybe early exit if time near deadline.

        if time.monotonic() + 1.0 > deadline:
            break
    return best, best_score

import time, math, numpy as np

def _project_simplex(v):
    # Euclidean projection onto simplex of sum=1 (non-negative)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0:
        w[:] = 1.0 / len(v)
    return w

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum == 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]

        w[:] = 1.0 / n
    return w

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    alpha = 0.2            # initial step size
    min_alpha = 1e-6
    decay = 0.9995
    it = 0

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # sub‑gradient of the infinity‑norm (sum contributions of all active peaks)
        # ------------------------------------------------------------------
        tol = max(1e-12, M * 1e-10)
        active = np.where(conv >= M - tol)[0]

        g = np.zeros_like(a)
        for idx in active:
            lo = idx - (n - 1)
            if lo < 0:
                lo = 0
            hi = idx
            if hi > n - 1:
                hi = n - 1
            j = np.arange(lo, hi + 1)
            g[j] += 2.0 * a[idx - j]

        # ------------------------------------------------------------------
        # Adam update + projection
        # ------------------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = _project_simplex(cand)

        # evaluate candidate
        fv_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(fv_c * fv_c, n=L)[: 2 * n - 1]
        M_c = conv_c.max()
        cand_score = 2.0 * n * M_c

        if cand_score < cur_score:
            a, conv, M, cur_score = cand, conv_c, M_c, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # occasional uphill acceptance
            if rng.random() < 0.005:
                a, conv, M, cur_score = cand, conv_c, M_c, cand_score

        # ------------------------------------------------------------------
        # random mass transfer (exploration)
        # ------------------------------------------------------------------
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.4
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                fv2 = np.fft.rfft(cand2, n=L)
                conv2 = np.fft.irfft(fv2 * fv2, n=L)[: 2 * n - 1]
                score2 = 2.0 * n * conv2.max()
                if score2 < cur_score:
                    a, conv, M, cur_score = cand2, conv2, conv2.max(), score2
                    if score2 < best_score:
                        best, best_score = cand2.copy(), score2

        # ------------------------------------------------------------------
        # occasional smoothing
        # ------------------------------------------------------------------
        if rng.random() < 0.005:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            fv_s = np.fft.rfft(smooth, n=L)
            conv_s = np.fft.irfft(fv_s * fv_s, n=L)[: 2 * n - 1]
            score_s = 2.0 * n * conv_s.max()
            if score_s < cur_score:
                a, conv, M, cur_score = smooth, conv_s, conv_s.max(), score_s
                if score_s < best_score:
                    best, best_score = smooth.copy(), score_s

        # ------------------------------------------------------------------
        # step‑size decay
        # ------------------------------------------------------------------
        alpha = max(min_alpha, alpha * decay)

        # early‑exit guard
        if time.monotonic() + 0.5 > deadline:
            break

    return best, best_score


# ----------------------------------------------------------------------
# One LP refinement: maximise Σb under the current peak bound
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, M: float, mix: float, rng: np.random.Generator):
    """
    Solve  max Σb   s.t.   a ∗ b ≤ M ,   b ≥ 0 .
    Return the convex combination (1‑mix)·a + mix·b̂,
    where b̂ is the normalised LP solution (sum = 1).
    """
    import scipy.optimize

    n = a.size
    # Build dense Toeplitz matrix (2n‑1) × n  :  (a∗b)[k] = Σ_j a[k‑j]·b_j
    m_rows = 2 * n - 1
    A = np.empty((m_rows, n), dtype=np.float64)
    A[:] = 0.0
    for j in range(n):
        A[j : j + n, j] = a

    c = -np.ones(n, dtype=np.float64)               # minimise –Σb  ⇔ maximise Σb
    b_ub = np.full(m_rows, M, dtype=np.float64)    # right‑hand side of a∗b ≤ M
    bounds = [(0.0, None)] * n

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=bounds, method='highs')
    if not res.success:
        return a
    b = res.x
    s = b.sum()
    if s == 0.0:
        return a
    b /= s                                          # enforce Σb = 1
    comb = (1.0 - mix) * a + mix * b
    return _project_simplex(comb)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)

    # ------------------------------------------------------------------
    # initial vector (use the provided construction when present)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # simple fallback: a short smooth shape, then tiled
        base = np.linspace(1.0, 0.0, 256, dtype=np.float64) ** 0.5
        a0 = np.tile(base, int(np.ceil(2560 / base.size)))[:2560]

    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 == 0.0:
        a0[:] = 1.0 / a0.size
    else:
        a0 = a0 / s0                                   # simplex normalisation

    # ------------------------------------------------------------------
    # main Adam optimisation
    # ------------------------------------------------------------------
    best, best_score = _adam_optimize(a0, deadline, rng)

    # ------------------------------------------------------------------
    # optional LP refinement (if time permits)
    # ------------------------------------------------------------------
    if time.monotonic() + 5.0 < deadline:
        # compute current bound M from the best candidate
        n = best.size
        L = 1 << ((2 * n - 1).bit_length())
        fv = np.fft.rfft(best, n=L)
        conv_best = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        M_best = conv_best.max()
        refined = _lp_refine(best, M_best, mix=0.5, rng=rng)
        # evaluate refined score (formula is cheap)
        fv_r = np.fft.rfft(refined, n=L)
        conv_r = np.fft.irfft(fv_r * fv_r, n=L)[: 2 * n - 1]
        refined_score = 2.0 * n * conv_r.max()
        if refined_score < best_score:
            best, best_score = refined, refined_score

    # ------------------------------------------------------------------
    # secondary random start if still time left
    # ------------------------------------------------------------------
    if time.monotonic() + 5.0 < deadline:
        rand = rng.random(a0.shape, dtype=np.float64)
        rand = _project_simplex(rand)
        alt, alt_score = _adam_optimize(rand, deadline, rng)
        if alt_score < best_score:
            best, best_score = alt, alt_score

    return [float(x) for x in best]
