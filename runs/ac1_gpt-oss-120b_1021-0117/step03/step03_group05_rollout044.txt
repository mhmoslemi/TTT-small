            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc

        # --------------------------------------------------------------
        # jittered restart towards the best known solution
        # --------------------------------------------------------------
        if it % 2000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

    # ----------------------------------------------------------------------
    # final local refinement (short random swaps) within remaining time
    # ----------------------------------------------------------------------
    refin_end = time.monotonic() + min(0.3, deadline - time.monotonic())
    while time.monotonic() < refin_end:
        i = rng.integers(0, n)
        if cur[i] == 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.25
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

    # ----------------------------------------------------------------------
    # return the best vector found as a plain Python list
    # ----------------------------------------------------------------------
    return [float(x) for x in best_vec]
