        fa = np.fft.rfft(a, n=fft_len)
        conv_self = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_max = conv_self.max()
        cur_score = 2.0 * n * cur_max

        # ----- keep the best --------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
            if best_score <= 1.50274349:          # target reached
                break
        else:
            no_improve += 1

        # ----- soft‑max weights over convolution peaks -------------------------
        tau = max(1e-6, tau0 / np.sqrt(iteration + 1))
        # stability: subtract max before exponentiation
        exp_vals = np.exp((conv_self - cur_max) / tau)
        w = exp_vals / exp_vals.sum()               # shape (conv_len,)

        # ----- gradient of weighted sum of convolutions -----------------------
        w_rev = w[::-1]                              # reverse for correlation
        fw = np.fft.rfft(w_rev, n=fft_len)
        grad_full = np.fft.irfft(fa * fw, n=fft_len)[:grad_len]
        # extract the n‑dimensional gradient (central part)
        g = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- momentum -------------------------------------------------------
        v = beta * v + (1.0 - beta) * g

        # ----- exponentiated gradient step (mirror descent on simplex) -------
        eta = eta0 / np.sqrt(iteration + 1)
        delta = -eta * v
        delta = np.clip(delta, -50, 50)              # avoid overflow/underflow
        a = a * np.exp(delta)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        iteration += 1

        # ----- occasional jitter to escape plateaus ---------------------------
        if iteration % 200 == 0:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = a * jitter
            a = a / a.sum()
            v.fill(0.0)
            no_improve = 0

        # ----- restart if stalled ---------------------------------------------
        if no_improve >= max_no_improve:
            a = best.copy()
            pert = np.exp(rng.normal(0.0, 0.05, size=n))
            a = a * pert
            a = a / a.sum()
            v.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # 4.  Return result (plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
