import time
import numpy as np

def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of v with itself using FFT of length L."""
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * v.size - 1]

def _score(v: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 upper bound for a non‑negative vector v."""
    s = v.sum()
    if s == 0.0:
        return float("inf")
    conv = _conv_fft(v, L)
    return 2.0 * n * conv.max() / (s * s)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x >= 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + float(budget_s) - 0.5  # spare margin for cleanup

    # ------------------------------------------------------------------
    # problem size & initial vector (use provided parent if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base_vec = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base_vec = np.maximum(base_vec, 0.0)
        tot = base_vec.sum()
        if tot <= 0.0:
            base_vec = np.full(n, 1.0 / n)
        else:
            base_vec = base_vec / tot
    else:
        n = 2000
        base_vec = np.full(n, 1.0 / n)

    L = 1 << ((2 * n - 1).bit_length())
    best_vec = base_vec.copy()
    best_sc = _score(best_vec, n, L)

    # ------------------------------------------------------------------
    # random sparse search parameters
    # ------------------------------------------------------------------
    low_frac, high_frac = 0.07, 0.12          # fraction of n used as support size
    k_low = max(2, int(low_frac * n))
    k_high = max(k_low + 1, int(high_frac * n))
    target = 1.50274349
    it = 0

    # ------------------------------------------------------------------
    # main search loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- random support, equal weights ---------------------------------
        k = rng.integers(k_low, k_high + 1)
        idx = rng.choice(n, size=k, replace=False)
        v = np.zeros(n, dtype=np.float64)
        w = 1.0 / k
        v[idx] = w
        sc = _score(v, n, L)

        if sc < best_sc:
            best_sc = sc
            best_vec = v.copy()
            if best_sc < target:
