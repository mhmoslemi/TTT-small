def _conv_fft(a):
    # Real FFT convolution same as previous.
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    return conv_full

def _score(a):
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _make_exponential(n, r):
    a = np.power(r, np.arange(n, dtype=np.float64))
    a = a / a.sum()
    return a

def _make_powerlaw(n, p):
    a = 1.0 / np.power(np.arange(1, n+1, dtype=np.float64), p)
    a = a / a.sum()
    return a

def _make_linear(n):
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()

def _make_quadratic(n):
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()

def _make_sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0/n)
    else:
        a = a / s
    return a

import numpy as np
import time

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a
