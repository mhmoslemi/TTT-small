            active = np.arange(conv_vals.size)
        else:
            active = np.argpartition(-conv_vals, K - 1)[:K]

        # compute sub‑gradient of the max (sum of gradients of active terms)
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # derivative of Σ a_i a_{k‑i} w.r.t. a_i is 2·a_{k‑i}
            g[idx] += 2.0 * cur[k - idx]

        # -------------------- additive projected‑gradient step --------------------
        alpha = add_alpha0 / math.sqrt(it)
        cand_add = cur - alpha * g
        cand_add = _project_simplex(cand_add)
        cand_add_sc = _score(cand_add)

        # -------------------- multiplicative (exponential‑weights) step ----------
        eta = mul_eta0 / math.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        s = cand_mul.sum()
        if s <= 0.0:
            cand_mul = uniform.copy()
        else:
            cand_mul = cand_mul / s
        cand_mul_sc = _score(cand_mul)

        # -------------------- accept the best candidate -------------------------
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc
        else:
            # occasional Metropolis‑style acceptance to escape plateaus
            if rng.random() < 0.03:
                if cand_add_sc < cand_mul_sc:
                    cur, cur_sc = cand_add, cand_add_sc
                else:
                    cur, cur_sc = cand_mul, cand_mul_sc

        # -------------------- keep global best ---------------------------------
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc

        # -------------------- random mass‑transfer moves -----------------------
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc

        # -------------------- jittered restart toward best --------------------
        if it % 7500 == 0 and rng.random() < 0.25:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
                cur, cur_sc = tmp.copy(), tmp_sc

        # -------------------- early exit if target achieved --------------------
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # final light refinement (few random transfers) if time permits
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.25
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

    # return the best vector found as a plain Python list
    return [float(x) for x in best_vec]
