import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection onto the probability simplex (sum=1, nonnegative) """
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

    n = v.size

    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast score: 2 * n * max( a * a )  (sum(a) is assumed 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int = None) -> float:
    n = a.shape[0]
    if L is None:
        L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # compact analytic fallback (sqrt‑series tiled)
        def _sqrt_series(m):
            a = np.empty(m, dtype=np.float64)
            a[0] = 1.0
            for k in range(1, m):
                if k == 1:
                    s = 0.0
                else:
                    s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
                a[k] = (1.0 - s) / 2.0
            return a

        base = _sqrt_series(64)
        a0 = np.tile(base, 40)[:2535]   # length matches typical parent

    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / total                     # normalise to the simplex

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length for all convolutions
    idx_slice = (2 * n - 2) - np.arange(n)   # indices for gradient extraction

    best = a0.copy()
    best_score = _score(best, L=L)
    cur = best.copy()
    cur_score = best_score

    step0 = 0.6
    step_decay = 0.9999
    no_improve = 0
    it = 0
    target = 1.50274349

    # ----- main optimisation loop ------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # active set (indices attaining the maximum, with tolerance)
        tol = 1e-12 * max_val if max_val != 0 else 1e-12
        w = (conv >= max_val - tol).astype(np.float64)

        # sub‑gradient of the max (via convolution with reversed indicator)
        w_rev_fft = np.fft.rfft(w[::-1], n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        grad = 2.0 * grad_full[idx_slice]

        # unit descent direction
        gnorm = np.linalg.norm(grad) + 1e-12
        direction = -grad / gnorm

        # step‑size schedule
        step_len = step0 * (step_decay ** it)

        # line‑search over a few scalings
        scales = (0.2, 0.5, 1.0, 2.0, 5.0)
        cand_best = None
        cand_best_sc = None
        for s in scales:
            cand = cur + step_len * s * direction
            cand = _project_simplex(cand)
            sc = _score(cand, L=L)
            if cand_best_sc is None or sc < cand_best_sc:
                cand_best_sc = sc
                cand_best = cand

        # accept improvement if any
        if cand_best_sc is not None and cand_best_sc < cur_score:
            cur, cur_score = cand_best, cand_best_sc
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
                no_improve = 0
        else:
            # occasional Metropolis‑type acceptance
            prob = math.exp(-(cand_best_sc - cur_score) / (step_len + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand_best, cand_best_sc
            no_improve += 1

        # periodic symmetrisation
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L=L)
            if sym_sc < cur_score:
                cur, cur_score = sym, sym_sc
                if sym_sc < best_score:
                    best, best_score = sym.copy(), sym_sc
                    no_improve = 0

        # random mass‑transfer when stagnating
        if no_improve >= 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L=L)
                if tmp_sc < cur_score:
                    cur, cur_score = tmp, tmp_sc
                if tmp_sc < best_score:
                    best, best_score = tmp.copy(), tmp_sc
                no_improve = 0

        # early exit if target reached
        if best_score < target:
            break

    # return plain Python list
    return [float(x) for x in best]
