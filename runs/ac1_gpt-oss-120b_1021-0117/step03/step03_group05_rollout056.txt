import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _max_convolution(v: np.ndarray, L: int) -> tuple[float, int]:
    """Return (max(conv(v,v)), index_of_max) using an FFT of length L."""
    fv = np.fft.rfft(v, n=L)
    c = np.fft.irfft(fv * fv, n=L)[: 2 * v.shape[0] - 1]
    idx = int(np.argmax(c))
    return float(c[idx]), idx


def _gradient_max(v: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the convolution entry k = argmax(conv(v,v))."""
    n = v.shape[0]
    lo = max(0, k - (n - 1))

    j = k - i
    g = np.zeros_like(v)
    np.add.at(g, i, v[j])
    np.add.at(g, j, v[i])
    return g
