def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.sparse as sp, cvxpy as cp, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.1

    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        a = np.full(n, 1.0/n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12: a = np.full_like(a, 1.0/a.size)
    a = a / a.sum()  # normalized to sum=1

    n = a.size

    # Compute initial metrics
    def conv_fft(x):
        L = 1 << ((2*n - 1).bit_length())
        fx = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fx * fx, n=L)[:2*n-1]
        return conv
    def max_conv(x):
        return conv_fft(x).max()
    cur_max = max_conv(a)
    cur_score = 2.0 * n * cur_max  # sum=1, denominator=1

    best_a = a.copy()
    best_score = cur_score

    # Precompute pattern offsets for building A each iteration quickly:
    offsets = np.arange(n)
    # for building A each iteration we will create list of diags
    # This will allocate n arrays each iteration.

    # parameters
    max_iter = 200  # limit number of LP solves
    tau = 0.4
    # maybe gradually reduce tau
    for it in range(max_iter):
        now = time.monotonic()
        if now > deadline:
            break

        # Build A for current a
        # each diagonal is constant equal to a[i]
        diags = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
        A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csc')
        # Solve LP
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= cur_max]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.GLPK, glpk={'msg_lev': 'GLP_MSG_OFF'})
        except Exception as e:
            # fallback: break if LP fails
            break
        if g.value is None:
            break
        g0 = np.maximum(g.value, 0.0)
        sumg = g0.sum()
        if sumg <= 0:
            # no improvement, random jitter
            g0 = np.full(n, 1.0/n)
            sumg = 1.0
        g_norm = g0 / sumg  # sum=1

        # update a
        a_new = (1 - tau) * a + tau * g_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        # evaluate new max conv
        new_max = max_conv(a_new)
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        # Accept new a if improves, else keep old a (maybe incorporate acceptance)
        if new_score <= cur_score:
            a = a_new
            cur_max = new_max
            cur_score = new_score
        else:
            # possible occasional acceptance with probability
            # Use simple temperature schedule
            T = max(0.01, (max_iter - it) / max_iter) # decreasing
            if rng.random() < math.exp(-(new_score - cur_score) / T):
                a = a_new
                cur_max = new_max
                cur_score = new_score

        # optionally adjust tau gradually: reduce as iteration proceeds.
        tau = max(0.05, tau * 0.99)

    # Return best solution as list
    return [float(x) for x in best_a]
