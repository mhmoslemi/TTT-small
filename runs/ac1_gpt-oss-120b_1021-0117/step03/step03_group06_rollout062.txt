```python
def proj_simplex(v):
    # from https://github.com/inducer/pareto-optimal/blob/master/project_simplex
    # Compute the projection of v onto the simplex {x >= 0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1          # zero‑based index
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function that achieves a very low AC1 bound.
    The construction consists of two equal mass blocks placed at the ends
    of the sequence, leaving a small zero gap in the middle.
    """
    # Determine the length to use
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000                     # fallback length if no parent is provided

    # Trivial cases
    if n <= 2:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # gap of 1 for odd length, 2 for even length
        gap = 1 if (n % 2 == 1) else 2
        left_len = (n - gap) // 2
        if left_len <= 0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            # each block carries half of the total mass
            block_val = 0.5 / left_len
            left = np.full(left_len, block_val, dtype=np.float64)
            middle = np.zeros(gap, dtype=np.float64)
            a = np.concatenate([left, middle, left])

    # Ensure exact normalisation (sum = 1) – protects against round‑off
    total = a.sum()
    if total > 0:
        a = a / total
    else:
        # Fallback to uniform if something went wrong
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Return a plain Python list as required by the evaluator
    return [float(x) for x in a]
