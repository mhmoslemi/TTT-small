def propose_candidate(seed=42, budget_s=1100, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # get n and parent

        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim !=1 or parent_arr.size==0:
            raise
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
        n = parent_arr.size
    except Exception:
        n = 256
        parent_arr = None

    # base candidates:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    # Tent shape:
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx+1, n-idx) - 1.0
    tent = np.maximum(tent, 0.0)
    if tent.sum()>0:
        tent /= tent.sum()
    else:
        tent = uniform.copy()
    # random Dirichlet:
    rand_vec = rng.random(n)
    rand_vec /= rand_vec.sum()
    candidates = [uniform, tent, rand_vec]
    if parent_arr is not None:
        # normalize parent to sum=1
        s = parent_arr.sum()
        if s >= 0.01:
            parent_norm = parent_arr / s
        else:
            parent_norm = uniform.copy()
        candidates.append(parent_norm)

    # fast scoring function:
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    def max_conv(seq):
        fa = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return conv.max()
    # pick best initial:
    best_score = float('inf')
    best_seq = None
    for cand in candidates:

        sc = 2 * n * m

            best_seq = cand.copy()

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """
    Heuristic optimization for the AC1 upper bound.
    Uses a projected sub‑gradient descent with exact gradient of the
    active autocorrelation shift, momentum, and occasional stochastic jumps.
    Returns the best coefficient list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------

    try:

        n = int(parent_raw.size)
        a_parent = np.maximum(parent_raw, 0.0)
        s = a_parent.sum()
        a_parent = a_parent / s if s >= 0.01 else np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        n = 256
        a_parent = None

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    tent = np.maximum(tent, 0.0)
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:

    rand_vec = rng.random(n)
    rand_vec = np.maximum(rand_vec, 0.0)
    rand_vec = rand_vec / rand_vec.sum()

    candidates = [uniform, tent, rand_vec]
    if a_parent is not None:
