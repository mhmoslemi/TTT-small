import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# ------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1+…+x^{2n‑2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------
# Convolution utilities (FFT, power‑of‑two length)
# ------------------------------------------------------------
def _make_fft_len(n: int) -> int:
    return 1 << ((2 * n - 1).bit_length())

def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

def _conv_max(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    return float(conv.max())

def _score(vec: np.ndarray, L: int) -> float:
    s = float(vec.sum())
    if s == 0.0:
        return float('inf')
    m = _conv_max(vec, L)
    n = vec.size
    return 2.0 * n * m / (s * s)
