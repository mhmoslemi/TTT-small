import time

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Analytic decreasing starter (the “sqrt‐series” of the paper)
# ------------------------------------------------------------
def _sqrt_series(m: int) -> np.ndarray:
    a = np.empty(m, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, m):
        if k == 1:
            s = 0.0
        else:
            # dot product of the tail with the reversed prefix
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------

    """return (2*n*max(conv), full convolution) for sum(v)=1."""
    n = v.shape[0]
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv
