import time, math, numpy as np

def _conv_self_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution of x using an FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]

def _score_from_max(conv_max: float, n: int) -> float:
    """Score = 2 * n * max(conv) (sum of coefficients is kept = 1)."""
    return 2.0 * n * conv_max

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    # --------------------------------------------------------------

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
    else:
        n0 = 2000
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    cur = a.copy()
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # FFT length for self‑convolution

    # initial score (sum(cur) == 1)
    conv_cur = _conv_self_fft(cur, L)

    # --------------------------------------------------------------

    eps_move = 0.2       # max fraction of mass shuffled in a random move

    # --------------------------------------------------------------

    while time.monotonic() < deadline:
