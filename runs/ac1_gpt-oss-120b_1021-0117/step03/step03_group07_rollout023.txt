import time

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (no dense matrices)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    A = np.fft.rfft(a, n=L)
    B = A * A                                   # point‑wise product (a * a)
    return np.fft.irfft(B, n=L)[: 2 * n - 1]    # linear convolution slice


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))
