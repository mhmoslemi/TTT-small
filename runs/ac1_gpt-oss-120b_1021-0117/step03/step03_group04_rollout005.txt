def compute_subgradient(a, K, n, conv_len):
    # compute convolution via FFT once
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    # We want subgrad w.r.t a of max of conv entries. Use weighted selection.
    # Compute top K indexes.
    if K < conv_len:
        top_idx = np.argpartition(-conv, K)[:K]
    else:
        top_idx = np.arange(conv_len)
    # weights proportional to conv value.
    vals = conv[top_idx]
    wsum = vals.sum()
    if wsum > 0:
        weights = vals / wsum
    else:
        weights = np.full_like(vals, 1.0/len(vals))
    # Initialize gradient zero
    grad = np.zeros_like(a)
    # for each shift offset compute contributions
    for s, w in zip(top_idx, weights):
        offset = s - (n-1)
        if offset >=0:
            # indices where i+offset < n
            i_start = 0
            i_end = n - offset - 1
            # i range 0.. n-offset-1
            # gradient[i] += w*2*a[i+offset]
            # gradient[i+offset] += w*2*a[i]
            # we can vectorize slices.
            grad[i_start:i_end+1] += w * 2.0 * a[offset: offset + (i_end - i_start + 1)]
            grad[offset: offset + (i_end - i_start + 1)] += w * 2.0 * a[i_start:i_end+1]
        else:
            # offset negative: shift left
            # offset = -d, d>0. Then i+d < n.
            d = -offset
            i_start = d
            i_end = n - 1
            # i range d..n-1, partner i-d
            # grad[i] += w*2*a[i-d]; grad[i-d] += w*2*a[i]
            grad[i_start:i_end+1] += w * 2.0 * a[i_start - d: i_end - d + 1]
            grad[i_start - d: i_end - d + 1] += w * 2.0 * a[i_start:i_end+1]
    return grad, conv

import time
import numpy as np

# -------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# -------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    if v.ndim != 1:
        raise ValueError
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Candidate generation (all vectors are normalized to sum = 1)

def _init_candidates(n: int, parent_norm: np.ndarray, rng: np.random.Generator):
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # triangular (tent) shape
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    tent = np.clip(tent, 0.0, None)
    if tent.sum() > 0:
        tent = tent / tent.sum()
        cand.append(tent)

    # Gaussian window
    sigma = n / 8.0
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    gauss = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    gauss = gauss / gauss.sum()
    cand.append(gauss)

    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

def _subgrad(a: np.ndarray, K: int, n: int, fft_len: int, conv_len: int):
    """Return (gradient, current max autocorrelation) for vector a."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

    # choose up to K dominant shifts
    if K < conv_len:
        top_idx = np.argpartition(-conv, K)[:K]
    else:
        top_idx = np.arange(conv_len)

    vals = conv[top_idx]
    wsum = vals.sum()
    if wsum > 0:
        weights = vals / wsum
    else:
        weights = np.full_like(vals, 1.0 / vals.size)

    grad = np.zeros_like(a)

    for s, w in zip(top_idx, weights):
        offset = s - (n - 1)
        if offset >= 0:
            L = n - offset                 # length of overlapping region
            # grad[i] for i in [0, L)
            grad[:L] += w * a[offset:offset + L]
            # grad[i+offset] for i in [0, L)
            grad[offset:offset + L] += w * a[:L]
        else:
            d = -offset
            L = n - d
            # grad[i] for i in [d, n)
            grad[d:] += w * a[:L]
            # grad[i-d] for i in [d, n) -> indices [0, L)
            grad[:L] += w * a[d:]

    return grad, conv.max()


# -------------------------------------------------------------
# Main entry point required by the harness
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimizer for the AC1 upper bound.
    Returns a list of non‑negative coefficients (sum is irrelevant for the
    final bound because the objective is homogeneous).
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # keep a safety margin

    # ---------------------------------------------------------
    # Determine problem size and (optionally) a parent vector
    # ---------------------------------------------------------
    parent_norm = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0 or not np.isfinite(parent_raw).all():
            raise ValueError
        n = int(parent_raw.size)
        parent_raw = np.maximum(parent_raw, 0.0)
        s = parent_raw.sum()
        if s >= 0.01:
            parent_norm = parent_raw / s
    except Exception:
        # Fallback length if the parent is unavailable
        n = 256

    # ---------------------------------------------------------
    # Prepare FFT parameters (fixed for the whole run)
    # ---------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ---------------------------------------------------------
    # Initialise a diverse set of candidates
    # ---------------------------------------------------------
    candidates = _init_candidates(n, parent_norm, rng)

    # Pick the best of the initial pool (fast evaluation via FFT)
    best = None
    best_score = float("inf")
    for cand in candidates:
        fa = np.fft.rfft(cand, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        sc = 2.0 * n * conv.max()          # objective for sum‑normalised vectors
        if sc < best_score:
            best_score = sc
            best = cand.copy()

    a = best.copy()

    # ---------------------------------------------------------
    # Adam optimiser parameters (projected onto the simplex)
    # ---------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                 # base learning rate
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    t = 0
    K = 15                    # number of top shifts used for the sub‑gradient
    target = 1.50274349

    # ---------------------------------------------------------
    # Main optimisation loop
    # ---------------------------------------------------------
    while time.monotonic() < deadline:
        t += 1

        # sub‑gradient of the current maximal entry
        g, cur_max = _subgrad(a, K, n, fft_len, conv_len)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _proj_simplex(a)                     # maintain feasibility

        cur_score = 2.0 * n * cur_max            # exact objective for sum‑normalised a
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= target:
                break

        # occasional multiplicative jitter to escape shallow plateaus
        if t % 400 == 0:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * jitter)

    # ---------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # ---------------------------------------------------------
    return [float(v) for v in best]
