def _conv_max(v, L):
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*len(v)-1]
    return conv.max()

    else:

        a = _sqrt_series(n)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    # ensure sum=1 exactly? after projection sum=1 (since proj to simplex).
    # adapt scale factor?
    best = a.copy()
    best_score = 2.0 * n * _conv_max(best, L)
    cur = a.copy()
    cur_score = best_score

    # parameters
    T0 = 0.1  # initial temperature
    T_decay = 0.9995
    eps0 = 0.1  # initial move magnitude relative to sum (since sum=1)
    eps_decay = 0.9995

    momentum = np.zeros_like(a)  # optional

    while time.monotonic() < deadline:
        # decide operation type
        op = rng.random()
        if op < 0.6:
            # random mass transfer
            i = rng.integers(0, n)
            if cur[i] <= 1e-12:
                continue
            j = rng.integers(0, n)
            if j == i:
                continue
            eps = eps0 * rng.random()
            eps = min(eps, cur[i])
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps
            # ensure nonneg (should hold)
            # project to simplex? The sum remains 1 exactly, because we moved mass, not changed sum.
            # But small floating errors? Hardly.
        elif op < 0.8:
            # smoothing
            kernel = np.array([0.25, 0.5, 0.25])
            cand = np.convolve(cur, kernel, mode='same')
            # after smoothing, sum may change; project to simplex
            cand = _project_simplex(cand)
        else:
            # gradient step
            # compute max index
            fv = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
            idx = int(conv.argmax())
            # gradient
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            g = np.zeros_like(cur)
            if lo <= hi:
                j = np.arange(lo, hi+1)
                g[j] = 2.0 * cur[idx - j]
            # apply momentum update
            momentum = 0.9 * momentum + 0.1 * g
            step = eps0 * 0.5
            cand = cur - step * momentum
            cand = _project_simplex(cand)
        # evaluate candidate
        maxc = _conv_max(cand, L)
        cand_score = 2.0 * n * maxc
        # accept condition:
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score)/max(T0, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        # update temperature and epsilon
        T0 *= T_decay
        eps0 *= eps_decay
        # break if time close
        if time.monotonic() + 0.5 > deadline:
            break

    # final fallback: possibly evaluate with evaluate_sequence in case something messed
    return [float(x) for x in best]

import time
import math
import numpy as np

# -------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # guard against numerical all‑zero
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# -------------------------------------------------------------
# Max of autocorrelation (a * a) via FFT
# -------------------------------------------------------------
def _conv_max(vec: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    return float(conv.max())

# -------------------------------------------------------------
# Optional analytic sqrt series (fallback if no parent)
# -------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
