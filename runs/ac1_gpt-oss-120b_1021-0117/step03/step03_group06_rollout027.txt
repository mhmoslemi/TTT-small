        # 2) Build a sub‑gradient from the selected active indices.
        # --------------------------------------------------------------
        g = np.zeros_like(cur)
        for k in active_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # each contribution is 2 * a[k - idx]
            g[idx] += 2.0 * cur[k - idx]

        # --------------------------------------------------------------
        # 3) Mirror‑descent (exponentiated) update.
        # --------------------------------------------------------------
        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if not np.isfinite(cand_sum) or cand_sum <= 0.0:
            cand = np.full_like(cur, 1.0 / n)
        else:
            cand = cand / cand_sum

        cand_conv = conv_full(cand)
        cand_score = score_of(cand, cand_conv)

        # --------------------------------------------------------------
        # 4) Acceptance test (strict improvement + occasional uphill).
        # --------------------------------------------------------------
        improved = False
        if cand_score < cur_score:
            improved = True
        else:
            # Metropolis‑style acceptance with a slowly cooling temperature.
            temp = 0.12 / math.sqrt(it)
            prob = math.exp(-(cand_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_conv = cand_conv
            cur_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 5) Update global best.
        # --------------------------------------------------------------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # --------------------------------------------------------------
        # 6) Guided random mass‑transfer move (biased by gradient magnitude).
        # --------------------------------------------------------------
        if rng.random() < 0.06:
            # source index – favour entries that contribute most to the active set
            src_weights = g + 1e-12
            src_weights /= src_weights.sum()
            i = rng.choice(n, p=src_weights)

            if cur[i] > 0.0:
                # destination – favour currently small entries, avoid neighbours
                dst_weights = 1.0 / (cur + 1e-12)
                dst_weights[i] = 0.0
                # suppress a small neighbourhood around i
                nb = max(1, n // 120)        # ~0.8 % of the length
                lo_nb = max(0, i - nb)
                hi_nb = min(n, i + nb + 1)
                dst_weights[lo_nb:hi_nb] = 0.0
                total = dst_weights.sum()
                if total == 0.0:
                    dst_weights = np.ones_like(cur) / n
                else:
                    dst_weights /= total
                j = rng.choice(n, p=dst_weights)

                delta = cur[i] * rng.random() * eps_move
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp_sum = tmp.sum()
                if tmp_sum > 0.0:
                    tmp /= tmp_sum
                    tmp_conv = conv_full(tmp)
                    tmp_score = score_of(tmp, tmp_conv)
                    if tmp_score < cur_score:
                        cur = tmp
                        cur_conv = tmp_conv
                        cur_score = tmp_score
                        no_improve = 0
                    if tmp_score < best_score:
                        best = tmp.copy()
                        best_score = tmp_score

        # --------------------------------------------------------------
        # 7) Small jittered restart from the current best (periodic).
        # --------------------------------------------------------------
        if it % 2500 == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0.0:
                tmp /= tmp_sum
                tmp_conv = conv_full(tmp)
                tmp_score = score_of(tmp, tmp_conv)
                if tmp_score < cur_score:
                    cur = tmp
                    cur_conv = tmp_conv
                    cur_score = tmp_score
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score
                no_improve = 0

        # --------------------------------------------------------------
        # 8) Bigger random restart after long stagnation.
        # --------------------------------------------------------------
        if no_improve > 8000:
            jitter = rng.normal(scale=jitter_scale * 5, size=n)
            cur = best + jitter
            cur = np.maximum(cur, 0.0)
            cur_sum = cur.sum()
            if cur_sum > 0.0:
                cur /= cur_sum
            else:
                cur = np.full_like(cur, 1.0 / n)
            cur_conv = conv_full(cur)
            cur_score = score_of(cur, cur_conv)
            no_improve = 0

        # --------------------------------------------------------------
        # 9) Early exit if target achieved.
        # --------------------------------------------------------------
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best solution discovered (as plain Python floats).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)   # defensive clipping for the evaluator
    return [float(v) for v in best]
