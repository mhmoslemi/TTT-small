import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)
    # Get initial vector
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = rng.random(n)
    # enforce nonneg and normalize sum to 1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= s
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    
    def conv_fft(v):
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        return conv

    def score(v):
        conv = conv_fft(v)
        return 2.0 * n * conv.max() / (v.sum() ** 2)

    def project_simplex(v):
        # Euclidean projection onto nonnegative simplex sum=1
        # algorithm: sort decreasing, find rho, theta
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho_arr = u - css / np.arange(1, v.size+1) > 0
        if not np.any(rho_arr):
            theta = 0.0
        else:
            rho = np.nonzero(rho_arr)[0][-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        # renormalize to exactly sum=1 (might be slightly off due to rounding)
        total = w.sum()
        if total > 0:
            w /= total
        else:
            w = np.full_like(v, 1.0 / v.size)
        return w

    cur = a.copy()
    cur_sc = score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc
    
    # Parameter schedule
    it = 0
    no_improve = 0
    target = 1.50274349
    # for gradient we will use active set determined by threshold factor
    eps_ratio = 1e-3  # to include near-max elements
    
    while time.monotonic() < deadline:
        it += 1
        # compute convolution and active indices
        conv = conv_fft(cur)
        maxc = conv.max()
        # select active indices where conv within eps_ratio * maxc
        thresh = maxc - eps_ratio * maxc
        active = np.where(conv >= thresh)[0]
        # compute gradient approximation by looping (if active size moderate)
        g = np.zeros_like(cur)
        # compute contributions efficiently: for each k in active
        # we can loop
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # indices i from lo..hi => contributions to g[i] += 2 * cur[k-i]
            idx = np.arange(lo, hi+1)
            g[idx] += 2.0 * cur[k - idx]
        # step sizes
        alpha = 0.8 / np.sqrt(it)   # additive step factor
        eta = 0.8 / np.sqrt(it)     # multiplicative factor

        # additive candidate
        cand_add = cur - alpha * g
        cand_add = project_simplex(cand_add)
        cand_add_sc = score(cand_add)

        # multiplicative candidate
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        s_mul = cand_mul.sum()
        if s_mul > 0:
            cand_mul = cand_mul / s_mul
        else:
            cand_mul = np.full_like(cur, 1.0 / n)
        cand_mul_sc = score(cand_mul)

        # pick best
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc
        # else keep cur

        # track best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # occasional random mass transfer moves
        if no_improve > 2000:
            # swap mass between two random indices
            i = rng.integers(0, n)
            if cur[i] > 0:
                j = rng.integers(0, n)
                if j == i:
                    continue
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_sc = score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                        no_improve = 0

        # restart with jitter towards best
        if it % 5000 == 0:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = project_simplex(tmp)
            tmp_sc = score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        if best_sc < target:
            break

    # final local random search refinement (short)
    refine_deadline = time.monotonic() + min(0.5, deadline - time.monotonic())
    while time.monotonic() < refine_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta

        tmp = project_simplex(tmp)
        tmp_sc = score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
    # return best vector as list

import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine respects the time budget and keeps the best vector seen.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # spare margin

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback random simplex point
        n_fallback = 2000
        a = rng.random(n_fallback)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # ------------------------------------------------------------------
    # helper utilities
    # ------------------------------------------------------------------
    def conv_fft(v: np.ndarray) -> np.ndarray:
        """Full convolution v*v using real FFT."""
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        return conv

    def score(v: np.ndarray) -> float:
        """Exact AC1 bound for vector v (assumes sum(v) > 0)."""
        conv = conv_fft(v)
        return 2.0 * n * conv.max() / (v.sum() ** 2)

    def project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto {x >= 0, sum(x) = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0.0] = 0.0
        tot = w.sum()
        if tot > 0.0:
            w /= tot
        else:
            w = np.full_like(v, 1.0 / v.size)
        return w

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_sc = score(best_vec)

        sc = score(cand)
        if sc < best_sc:

            best_vec = cand.copy()

    # ------------------------------------------------------------------
    cur = best_vec.copy()

    it = 0

    target = 1.50274349      # stop if we beat this bound
