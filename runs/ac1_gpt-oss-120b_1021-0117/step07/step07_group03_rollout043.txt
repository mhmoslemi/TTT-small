def _conv_smooth_max_and_grad(a, L, tau):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    # compute smoothed max
    max_val = conv.max()
    # compute softmax weights
    # shift for stability
    shift = max_val
    exp_vals = np.exp((conv - shift) / tau)
    sum_exp = exp_vals.sum()
    weights = exp_vals / sum_exp
    # gradient
    w_rev = weights[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    # return smooth max = shift + tau * log(sum_exp)
    smooth_max = shift + tau * np.log(sum_exp)
    return smooth_max, grad

import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max(a, L):
    """Maximum value of the linear convolution a*a (full mode)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    return conv_full[: 2 * n - 1].max()


def _conv_smooth_max_and_grad(a, L, tau):
    """
    Smoothed max (log‑sum‑exp) of a*a and its gradient.
    a : non‑negative vector, sum(a)=1
    tau : smoothing temperature (tau→0 gives the true max)
    Returns (smooth_max, gradient) where gradient has shape a.shape.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    # full convolution
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    max_val = conv.max()

    # soft‑max weights (numerically stable)
    if tau < 1e-12:
        eps = max_val * 1e-12 + 1e-15
        mask = (conv >= max_val - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt > 0:
            mask /= cnt
        weights = mask
        smooth_max = max_val
    else:
        shift = max_val
        exp_vals = np.exp((conv - shift) / tau)
        sum_exp = exp_vals.sum()
        weights = exp_vals / sum_exp
        smooth_max = shift + tau * np.log(sum_exp)

    # gradient = 2 * (a * w_rev)  (matrix‑free via FFT)
    w_rev = weights[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return smooth_max, grad

        t += 1

        # gradient of smoothed max
        smooth_max, grad = _conv_smooth_max_and_grad(a, L, tau)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = _project_simplex(cand)

        # occasional symmetrisation (helps flatten the autocorrelation)
        if t % 250 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # evaluate candidate (exact max)
        cur_max = _conv_max(cand, L)
        cur_sc = 2.0 * n * cur_max

        # keep the best ever seen
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = cand                     # progress to next iterate

        # decay smoothing temperature
        tau = max(tau * 0.995, 1e-12)

        # random perturbation when stuck
        if no_improve > 5000:
            sigma = 0.08 / math.sqrt(t + 1.0)
            pert = a + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert_sc = 2.0 * n * _conv_max(pert, L)
            if pert_sc < best_sc:
                best_sc = pert_sc
                best_vec = pert.copy()
                a = pert
                no_improve = 0
            else:
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
