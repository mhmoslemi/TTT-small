import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max_and_grad_smooth(a, L, tau):
    """
    Returns (max_convolution, gradient) where gradient is derived from a
    smooth approximation of the max using a softmax with temperature tau.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    max_val = conv.max()

    # softmax weights focusing on large values
    shift = max_val
    exp_vals = np.exp((conv - shift) / tau)
    w = exp_vals / exp_vals.sum()

    # gradient of the smooth max
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

        # step size adapts with iteration count and current temperature
        step = step0 / math.sqrt(it) * (tau / (max_val + 1e-12))

        # candidate update and forced symmetry
        cand = a - step * momentum
        cand = (cand + cand[::-1]) * 0.5          # keep the vector symmetric
        cand = _project_simplex(cand)

        # evaluate true objective (max convolution) for the candidate
        cand_max, _ = _conv_max_and_grad_smooth(cand, L, tau)
        cand_score = 2.0 * n * cand_max

        # simple greedy acceptance (with occasional Metropolis fallback)
        if cand_score < cur_score:
            a, cur_score = cand, cand_score
            no_improve = 0
        else:
            # occasional uphill move to escape shallow wells
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                a, cur_score = cand, cand_score
                no_improve = 0
            else:
                no_improve += 1

        # record best ever seen
        if cand_score < best_score:
            best_score, best_vec = cand_score, cand.copy()
            no_improve = 0

        # stochastic perturbation when stuck
        if no_improve > 2500:
            sigma = 0.02 / math.sqrt(it)
            pert = a + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert_max, _ = _conv_max_and_grad_smooth(pert, L, tau)
            pert_score = 2.0 * n * pert_max
            if pert_score < best_score:
                best_score, best_vec = pert_score, pert.copy()
                a, cur_score = pert, pert_score
            momentum.fill(0.0)
            no_improve = 0

        # occasional mixing with the uniform distribution (exploration)
        if it % 500 == 0:
            uniform = np.full(n, 1.0 / n, dtype=np.float64)
            a = _project_simplex(0.9 * a + 0.1 * uniform)
            cur_score = 2.0 * n * _conv_max_and_grad_smooth(a, L, tau)[0]

        # temperature annealing
        tau = max(min_tau, tau * 0.9995)

        # early exit if target reached
        if best_score < target:
            break

    # return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
