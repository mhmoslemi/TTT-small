import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score_from_vec(a, L):
    """Score for a vector with sum(a)=1 using FFT convolution."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    return float(2.0 * a.size * conv_full[: 2 * a.size - 1].max())

    else:
        a0 = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score_from_vec(best_vec, L)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.8   # initial learning‑rate factor
    it = 0
    no_improve = 0
    target = 1.50274349

    # temperature for soft‑max weighting of the max‑convolution

    temp_decay = 0.9992

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & soft‑max ------------------------------------
        a_fft = np.fft.rfft(cur, n=L)
