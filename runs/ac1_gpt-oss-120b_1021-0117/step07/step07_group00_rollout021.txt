def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # leave margin

    # Initialize vector
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2500, dtype=np.float64)

    # ensure nonnegative
    a = np.maximum(a, 0.0)
    # project to simplex (sum=1). Use function _project_simplex
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    # Precompute index mapping for gradient extraction
    idx = (2 * n - 2) - np.arange(n)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Temperature schedule: start with some fraction of max conv
    temp = None
    decay = 0.998
    min_temp_factor = 1e-4

    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    # Keep track of no improvement to random perturb
    no_improve = 0

    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # Compute convolution via FFT
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # temperature
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # compute softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # Compute gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # extract gradient elements: conv of a with w_rev, but need indexing as earlier
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step_size = 0.5 / math.sqrt(it)   # maybe start bigger; adjust factor
        # maybe adapt step_size to scaling
        # compute step
        step = step_size * m_hat / (np.sqrt(v_hat) + eps)

        # propose candidate
        cand = _project_simplex(a - step)
        cand_score = _score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()

        # accept if improvement else maybe accept with small prob
        if cand_score < _score(a):
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # optional probabilistic acceptance similar to SA
            prob = math.exp(-(cand_score - _score(a)) / (step_size + 1e-12))
            if rng.random() < prob:
                a = cand

        # random mass redistribution when stuck
        if no_improve > 2500:
            # do small random perturbation
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i: j += 1
            delta = a[i] * rng.random() * 0.5
            tmp = a.copy()
            tmp[i] -= delta

import time, math, numpy as np

def _project_simplex(v):
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score (same formula as the official evaluator)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))
