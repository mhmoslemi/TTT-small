def _score_fft(a, n, L):
    s = a.sum()
    if s < 1e-12: return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    return float(2.0 * n * conv.max() / (s * s))

def _subgrad(a, n, L):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    cnt = mask.sum()
    if cnt == 0:
        w = np.zeros_like(mask)
    else:
        w = mask.astype(np.float64) / cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad

def _optimize(initial, n, L, deadline, best_score, best_vec):
    cur = _project_simplex(initial.copy())
    cur_sc = _score_fft(cur, n, L)
    # If initial better than best_score, update.
    if cur_sc < best_score:
        best_score = cur_sc; best_vec = cur.copy()
    # hyperparameters
    beta = 0.5
    step0 = 0.4
    decay = 1e-3
    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0
    # thresholds
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1
        grad = _subgrad(cur, n, L)
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / (1.0 + decay * it)
        # evaluate candidates for scaling factors
        local_best_vec = cur
        local_best_sc = cur_sc
        for factor in (0.6, 1.0, 1.4):
            cand = _project_simplex(cur - step * factor * momentum)
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < local_best_sc:
                local_best_sc = cand_sc
                local_best_vec = cand
        # acceptance
        if local_best_sc < cur_sc:

import time, math, numpy as np

TARGET = 1.50274349  # known target bound


def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w


def _score_fft(a, n, L):
    """Exact evaluator but using FFT (assumes a>=0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def _subgrad(a, n, L):
    """Sub‑gradient of max‑convolution with respect to a (scale‑invariant)."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    cnt = mask.sum()
    if cnt == 0:
        w = np.zeros_like(mask, dtype=np.float64)
    else:
        w = mask.astype(np.float64) / cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad


def _optimize(initial, n, L, deadline, best_score, best_vec, rng, target):
    """Adam‑style sub‑gradient descent with simplex projection."""
    cur = _project_simplex(initial.copy())
    cur_sc = _score_fft(cur, n, L)
    if cur_sc < best_score:
        best_score, best_vec = cur_sc, cur.copy()

    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    step0 = 0.4
    it = 0
    no_improve = 0
    scale_factors = (0.6, 1.0, 1.4)

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # sub‑gradient
        g = _subgrad(cur, n, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + 1e-8)

        # line‑search over a few scalings
        local_best_vec = cur
        local_best_sc = cur_sc
        for factor in scale_factors:
            cand = _project_simplex(cur - step * factor)
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < local_best_sc:
                local_best_sc = cand_sc
                local_best_vec = cand

        # acceptance
        if local_best_sc < cur_sc:
            cur, cur_sc = local_best_vec, local_best_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional uphill move (tiny probability)
            if rng.random() < 0.001:
                cur, cur_sc = local_best_vec, local_best_sc

        # global best update
        if cur_sc < best_score:
            best_score, best_vec = cur_sc, cur.copy()

        # periodic symmetrisation
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_score:
                    best_score, best_vec = sym_sc, sym.copy()

        # random redistribution when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.05
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_score:
                    best_score, best_vec = tmp_sc, tmp.copy()

        # early exit
        if best_score < target:
            break

    return best_score, best_vec

    # leave a small safety margin

    # length of the sequence

    candidates.append(np.full(n, 1.0 / n))

    tri = np.minimum(xs + 1, n - xs)
    candidates.append(tri / tri.sum())

    # 3) Sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    candidates.append(sine / sine.sum())

    # 4) Raised cosine bump
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    candidates.append(bump / bump.sum())

    # 5) Exponential decays
    for alpha in (0.001, 0.003, 0.006):
        exp = np.exp(-alpha * xs)
        candidates.append(exp / exp.sum())

    # 6) Power laws
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        candidates.append(pw / pw.sum())

    # 7) Random Dirichlet‑like
    rand = rng.random(n)
    candidates.append(rand / rand.sum())

    # 8) Parent if present
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            candidates.append(parent)

    # ------------------------------------------------------------------
    # Pick the best deterministic start
    best_score = float('inf')
    best_vec = None
    for vec in candidates:
        sc = _score_fft(vec, n, L)
        if sc < best_score:
            best_score, best_vec = sc, vec.copy()

    # ------------------------------------------------------------------
    # Run a focused optimisation from the best starter
    if best_vec is not None:

        best_vec = np.full(n, 1.0 / n)
    return [float(x) for x in best_vec]
