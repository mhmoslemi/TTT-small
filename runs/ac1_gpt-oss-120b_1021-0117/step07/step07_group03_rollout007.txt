import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _conv_max_and_grad(a, L):
    """
    Full convolution a*a (via FFT) and a subgradient of its maximum.
    Returns (max_value, gradient_vector) where gradient is 2*a convolved
    with the uniform mask on the maximal entries.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def _soft_obj_and_grad(a, L, tau):
    """
    Smooth approximation of max(a*a) via softmax with temperature tau.
    Returns (smooth_obj, gradient, max_val) where max_val is the exact max.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()

    shift = (conv - max_val) / tau
    w = np.exp(shift)
    sum_w = w.sum()
    if sum_w == 0 or not np.isfinite(sum_w):
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        sum_w = 1.0

    w_fft = np.fft.rfft(w_rev, n=L)

    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]

    obj = max_val + tau * np.log(sum_w)
    return obj, grad, max_val

                best_score = pert_score
                best_vec = pert.copy()
                cur = pert
                cur_score = pert_score
                no_improve = 0
            else:
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        # early exit if target reached
        if best_score < target:
            break

    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
