def _compute_score(vec, n):

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    # standard algorithm O(n log n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0

        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
