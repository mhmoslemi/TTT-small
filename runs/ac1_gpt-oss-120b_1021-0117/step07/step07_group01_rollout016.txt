import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w = np.full_like(v, 1.0 / n)
    return w

def _score_vec(a: np.ndarray, L: int) -> float:

    conv_full = np.fft.irfft(af * af, n=L)
    return float(2.0 * n * conv_full[: 2 * n - 1].max())
