Implementation: (see Duchi et al. 2008) algorithm:

def project_simplex(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

        best_score = float("inf")

    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        s = parent.sum()
        if s > 0:
            parent = parent / s
            try:
                pscore = evaluate_sequence(parent.tolist())
                if pscore < best_score:
                    best_score = pscore
                    best_vec = parent.copy()
            except Exception:
                pass

    # --------------------------------------------------------------
    # initialise search
    cur = best_vec.copy()
    cur_score = best_score
    rng = np.random.default_rng(seed)

    max_iters = 50000
    temp = 1.0               # simulated‑annealing temperature
    temp_decay = 0.9995
    max_frac = 0.5           # maximal transferred fraction of source mass
    frac_decay = 0.9999
    grad_src_prob = 0.3      # probability to pick source weighted by gradient
    perturb_interval = 2000
    perturb_scale = 0.005

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # convolution of current point (needed for gradient‑based source choice)
        conv_cur = np.convolve(cur, cur)
        max_idx = int(conv_cur.argmax())

        # sub‑gradient for the active shift
        idx_arr = max_idx - np.arange(n)
        mask = (idx_arr >= 0) & (idx_arr < n)
        g = np.zeros(n, dtype=np.float64)
        g[mask] = 2.0 * cur[idx_arr[mask]]

        # decide source index
        if rng.random() < grad_src_prob and g.sum() > 0:
            probs = g / g.sum()
        else:
            probs = cur
        if probs.sum() == 0:
            probs = np.full(n, 1.0 / n)
        i = rng.choice(n, p=probs)

        if cur[i] <= 0:
            continue

        # pick a distinct target
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a random fraction of mass
        delta = cur[i] * rng.random() * max_frac
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # occasional random perturbation to escape plateaus
        if (it % perturb_interval) == 0:
            noise = rng.normal(scale=perturb_scale, size=n)
            cand = cand + noise
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

        # evaluate the candidate
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        # acceptance rule (simulated annealing)
        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()

        # decay schedule
        temp *= temp_decay
        max_frac *= frac_decay

    # final safety checks
    if best_vec is None or best_score == float("inf"):
        best_vec = uniform
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    if best_vec.sum() < 0.01:
        s = best_vec.sum()
        best_vec = best_vec / s if s > 0 else uniform

    return [float(x) for x in best_vec]
