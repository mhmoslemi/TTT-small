import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution(a, L, fft_a=None):
    """Return max of a*a (full convolution) using FFT of length L."""
    if fft_a is None:
        fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return conv.max(), conv, fft_a

def _score_vec(a, L):
    """Score for a simplex‑normalised vector (sum(a)=1)."""
    mx, _, _ = _max_convolution(a, L)
    return float(2.0 * a.size * mx)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final conversion
    deadline = start + max(0.0, float(budget_s) - 0.25)

    # ---- initial vector ----------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random length between 1500 and 3000 if no parent is provided
        n = rng.integers(1500, 3000)
        a0 = rng.random(n, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec, L)
    cur_sc = best_sc

    # ---- optimiser state --------------------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.9               # momentum decay
    step0 = 1.0              # initial step length
    it = 0
    no_improve = 0
    target = 1.50274349      # stop when we beat the target

    # temperature for the soft‑max that approximates the max‑convolution
    temp_factor = 1e-3       # relative to current max value
    temp_decay = 0.9995
    min_temp_factor = 1e-8

    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # forward FFT of the current vector
        a_fft = np.fft.rfft(cur, n=L)

        # full convolution a*a and its maximum
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---- soft‑max weighting of the convolution  ----------------------
        temp = max(max_val * temp_factor, max_val * min_temp_factor)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient of the weighted max‑convolution (matrix‑free)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)    # pick the correct slice
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- temperature decay --------------------------------------------
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # early exit if target already reached
        if best_sc < target:

        # periodic time check (cheap)
        if it % 5000 == 0 and time.monotonic() >= deadline:
            break

    # ----------------------------------------------------------------------
    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
