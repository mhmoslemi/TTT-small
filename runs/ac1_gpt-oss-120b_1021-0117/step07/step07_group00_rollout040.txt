import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast score: 2*n*max(conv) for a vector with sum≈1 (no denominator)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # FFT size (power of two)

    # best-so-far (true score via evaluate_sequence)
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())

    # fast surrogate score for the current iterate
    cur_sc = _fast_score(cur, n, L)

    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.92            # momentum decay
    step0 = 0.9            # base step size
    it = 0
    no_improve = 0
    temp = None
    decay = 0.998
    min_temp_factor = 1e-4

    # for occasional exact evaluations (costly)
    exact_eval_interval = 2000

    while time.time() < deadline:
        it += 1

        # ---------- convolution and temperature ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ---------- soft‑max weighting ----------
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ---------- gradient of weighted max ----------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---------- momentum & adaptive step ----------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        # try a few step scalings and keep the best candidate
        best_cand = None
        best_cand_sc = float('inf')
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - step * factor * momentum)
            cand_sc = _fast_score(cand, n, L)
            if cand_sc < best_cand_sc:
                best_cand_sc = cand_sc
                best_cand = cand

        cand = best_cand
        cand_sc = best_cand_sc

        # ---------- acceptance ----------
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            # occasional exact evaluation to update the true best
            if it % exact_eval_interval == 0:
                true_sc = evaluate_sequence(cand.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, cand.copy()
        else:
            no_improve += 1

        # ---------- random redistribution when stuck ----------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                true_sc = evaluate_sequence(tmp.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, tmp.copy()
        # ---------- symmetrisation ----------
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                true_sc = evaluate_sequence(sym.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, sym.copy()
        # ---------- early stop if we beat an ambitious target ----------
        if best_sc < 1.45:   # well below the historic target
            break

    # final sanity check – ensure the returned vector yields a finite score
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc) or final_sc >= 1e9:
            # fallback to uniform distribution (guaranteed finite)
            best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
