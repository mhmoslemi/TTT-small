import time, math, numpy as np

# ------------------------------------------------------------
# helper: Euclidean projection onto the probability simplex
def _project_simplex(v):
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    start = time.monotonic()
    # keep a little slack for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------
    # initialise ------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        cur = _project_simplex(np.maximum(a0, 0.0))
    else:
        # exponential decay as a generic start
        n0 = 2500
        idx = np.arange(n0, dtype=np.float64)
        a0 = np.exp(-idx / (n0 / 5.0))
        cur = _project_simplex(a0)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFT

    # best solution so far
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    best_sc = _score_from_max(max_val, n)
    best_vec = cur.copy()

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 1.0               # base learning rate
    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- compute sub‑gradient of the max‑convolution ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # mask for all indices achieving (or very close to) the max
        tol = 1e-12
        mask = (conv >= max_val - tol).astype(np.float64)

        # gradient = 2 * (cur * mask_rev)   (matrix‑free via FFT)
        mask_rev = mask[::-1]
        mask_fft = np.fft.rfft(mask_rev, n=L)
        grad_full = np.fft.irfft(a_fft * mask_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map to size‑n slice
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        # ----- candidate score --------------------------------------
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv_full = np.fft.irfft(cand_fft * cand_fft, n=L)
        cand_conv = cand_conv_full[: 2 * n - 1]
        cand_max = cand_conv.max()
        cand_sc = _score_from_max(cand_max, n)

        # ----- accept if better ------------------------------------
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
        else:
            no_improve += 1
            # occasional stochastic acceptance (simulated annealing style)
            if rng.random() < math.exp(-(cand_sc - best_sc) / (lr0 + 1e-12)):
                cur = cand
                # keep momentum (no reset) – helps escape shallow pits
            # else keep cur unchanged

        # ----- periodic symmetrisation (helps many optimal patterns)
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv_full = np.fft.irfft(sym_fft * sym_fft, n=L)
            sym_max = sym_conv_full[: 2 * n - 1].max()
            sym_sc = _score_from_max(sym_max, n)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                cur = sym
                m.fill(0.0)
                v.fill(0.0)

        # ----- stagnation handling ---------------------------------
        if no_improve > 3000:
            # jitter around the best solution
            jitter = rng.normal(scale=0.01, size=n)
            cur = best_vec + jitter
            cur = _project_simplex(np.maximum(cur, 0.0))
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- early exit -----------------------------------------
        if best_sc < target:
            break

        # ----- time check (every few thousand iterations) ----------
        if it % 5000 == 0 and time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------
    # return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
