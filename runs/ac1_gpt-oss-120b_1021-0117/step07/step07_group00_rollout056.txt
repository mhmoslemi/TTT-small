import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _triangular_initial(n):
    """Simple symmetric triangular shape, then projected onto the simplex."""
    half = n // 2
    if n % 2 == 0:
        up = np.arange(1, half + 1)
        down = np.arange(half, 0, -1)
    else:
        up = np.arange(1, half + 2)
        down = np.arange(half + 1, 0, -1)
    seq = np.concatenate([up, down]).astype(np.float64)
    return _project_simplex(seq)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.25)   # safety margin

    # ------------------------------------------------------------------
    # initial vector
    if "height_sequence_1" in globals():
        cur = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        # fall‑back length if no parent is supplied
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.2

    # temperature schedule for smooth max
    temp = None
    decay = 0.997
    min_temp_factor = 1e-4

    it, no_improve = 0, 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # ----- convolution & soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the (soft) max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)

        # ----- Adam update -----

        cand = _project_simplex(np.maximum(cand, 0.0))

            no_improve = 0
        else:

                cur, cur_sc = cand, cand_sc
                no_improve = 0

        if cand_sc < best_sc:

        # ----- periodic symmetrisation -----

        if no_improve > 2500:

            break

    tri = _triangular_initial(n)
    tri_sc = _score(tri)
    if tri_sc < best_sc:
        best_sc, best_vec = tri_sc, tri.copy()

    # cosine window (smooth, symmetric)
    cos = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    cos = _project_simplex(cos)
    cos_sc = _score(cos)
    if cos_sc < best_sc:
        best_sc, best_vec = cos_sc, cos.copy()

    # final safety projection
    best_vec = _project_simplex(np.maximum(best_vec, 0.0))

    return [float(x) for x in best_vec]
