import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast objective for a vector already on the simplex (sum≈1, a≥0)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    start = time.time()

    # Initial vector
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()

        cur = rng.random(2000, dtype=np.float64)

    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5

    it = 0
    no_improve = 0
    temp = None
    decay = 0.997
    min_temp_factor = 5e-4
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:

        it += 1

        # convolution via FFT

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule (softmax smoothing)

            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
