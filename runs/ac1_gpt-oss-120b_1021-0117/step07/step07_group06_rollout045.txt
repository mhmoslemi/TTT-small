import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a, n, L):
    """Return 2*n*max(conv) for a already on the simplex."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate a smooth initial shape if no parent is provided
        n0 = 2000
        xx = np.arange(n0)
        a = np.sin(np.pi * (xx + 0.5) / n0) + 1e-6
    a = _project_simplex(np.maximum(a, 0.0))
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # optimizer state (Adam)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12
    beta1_pow, beta2_pow = beta1, beta2
    it = 0
    no_improve = 0
    best_vec = a.copy()
    best_sc = _score(best_vec, n, L)
    cur_sc = best_sc

    target = 1.50274349   # early stop target

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- FFT of current vector -----
        a_fft = np.fft.rfft(a, n=L)

        # ----- convolution (auto‑correlation) -----
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- subgradient via active set -----
        # tolerance to catch multiple near‑max entries
        tol = max_val * 1e-12 + 1e-15
        mask = conv >= (max_val - tol)
        cnt = mask.sum()
        if cnt == 0:
            # fallback: use exact max index
            mask = np.zeros_like(conv, dtype=bool)
            mask[conv.argmax()] = True
            cnt = 1
        w = mask.astype(np.float64) / float(cnt)

        # ----- gradient (matrix‑free via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr / (np.sqrt(v_hat) + eps)
        cand = a - step * m_hat
        cand = _project_simplex(np.maximum(cand, 0.0))

        cand_sc = _score(cand, n, L)

        # ----- accept if improvement -----
        if cand_sc < cur_sc:

            no_improve = 0

    if sine_sc < best_sc:

    # Return best vector as plain Python floats
    return [float(x) for x in best_vec]
