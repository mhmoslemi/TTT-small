import time, math, numpy as np


def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def _score_vec(v, L):
    """Score for a vector already normalized to sum=1."""
    a_fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * len(v) - 1]
    return float(2 * len(v) * conv.max())

                        best_vec, best_sc = tmp.copy(), tmp_sc

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best vector found (plain Python floats)
    return [float(x) for x in best_vec]
