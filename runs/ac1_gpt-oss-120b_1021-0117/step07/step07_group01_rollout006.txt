import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Fast AC1 score for a vector already (approximately) on the simplex."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    n = a.size
    L_conv = 1 << ((2 * n - 1).bit_length())
    L_grad = 1 << ((3 * n - 2).bit_length())

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # tracking best solution
    best = a.copy()
    best_sc = _score_vec(best)

    # control parameters
    target = 1.50274349
    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of a with itself -----
        a_fft = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=L_conv)[: 2 * n - 1]
        conv_max = conv.max()

        # ----- smooth approximation of the max (softmax) -----
        # α grows slowly from 5 to ~200
        alpha = 5.0 + min(195.0, it * 0.02)
        maxv = conv_max
        exp_term = np.exp(alpha * (conv - maxv))
        w = exp_term / exp_term.sum()

        # ----- gradient of the smooth max via convolution with w -----
        a_fft_g = np.fft.rfft(a, n=L_grad)
        w_fft = np.fft.rfft(w, n=L_grad)
        grad_full = np.fft.irfft(a_fft_g * w_fft, n=L_grad)
        grad = 2.0 * grad_full[:n]

        # ----- Adam update (projected) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate decays very gently
        lr = 0.5 / (1.0 + it * 0.0005)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = _project_simplex(cand)

        # ----- scoring -----
        cand_sc = _score_vec(cand)
        cur_sc = _score_vec(a)

        # ----- acceptance logic -----
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < cur_sc:
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                a = cand

        # ----- occasional symmetrisation -----
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                no_improve = 0

        # ----- random mass redistribution when stuck -----
        if no_improve > 2000:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
            no_improve = 0

        # ----- restart from a mixed random point -----
        if no_improve > 4000:
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # return the best candidate as a plain Python list
    return [float(x) for x in best]
