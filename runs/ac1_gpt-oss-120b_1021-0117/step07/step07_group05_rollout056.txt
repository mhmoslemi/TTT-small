Pseudo:

def simplex_proj(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum==1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("input must be 1‑d")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _build_sidon_set(n, aim=None):
    """
    Greedy construction of a Sidon set S ⊂ {0,…,n‑1}:
    all pairwise sums i+j (including i+i) are distinct.
    `aim` optionally caps the size of the set.
    """
    used = set()
    S = []
    for x in range(n):
        # quick stop if we already reached the desired size
        if aim is not None and len(S) >= aim:
            break
        # check conflict of sums with existing elements
        if 2 * x in used:
            continue
        conflict = False
        for y in S:
            if x + y in used:
                conflict = True
                break
        if conflict:
            continue
        # accept x
        for y in S:
            used.add(x + y)
        used.add(2 * x)
        S.append(x)
    return S

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as measured by ``evaluate_sequence``.
    The method builds a small Sidon set of indices, places equal mass on them,
    and falls back to a simple stochastic improvement if time permits.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ----------------------------------------------------------------------
    # Determine length n and an initial candidate (parent or uniform)
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        total = parent.sum()
        if total > 0:
            parent = parent / total
        else:
            parent = None
        n = parent.size
    else:
        parent = None
        n = 2000
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Build a Sidon set and initialise a candidate from it
    # Target size ≈ √(2n) – this gives a provably good score.
    target_k = int(np.floor(np.sqrt(2 * n)))
    sidon_idx = _build_sidon_set(n, aim=target_k)
    k = len(sidon_idx)
    if k >= 2:
        sidon = np.zeros(n, dtype=np.float64)
        v = 1.0 / k
        sidon[sidon_idx] = v
        best = sidon
        conv = np.convolve(best, best)
        best_score = 2.0 * n * conv.max()          # sum == 1 ⇒ denominator = 1
    else:
        # Fallback to uniform if Sidon construction fails
        best = uniform.copy()
        conv = np.convolve(best, best)
        best_score = 2.0 * n * conv.max()

    # ----------------------------------------------------------------------
    # Compare with the supplied parent (if any) and keep the best so far
    if parent is not None:
        conv_parent = np.convolve(parent, parent)
        parent_score = 2.0 * n * conv_parent.max()
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # ----------------------------------------------------------------------
    # Simple stochastic refinement (hill‑climbing + occasional annealed moves)
    rng = np.random.default_rng(seed)
    cur = best.copy()
    cur_score = best_score
    T = 1.0                # temperature for occasional uphill moves
    decay = 0.99995        # geometric cooling
    min_T = 1e-6

    # Helper: compute score for a vector with sum ≈ 1 (exactly 1 after our moves)
    def _score(vec):
        c = np.convolve(vec, vec)
        return 2.0 * n * c.max()

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        # transfer a random fraction of mass from i to j (scaled by temperature)
        delta = cur[i] * rng.random() * T
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _score(cand)

        # accept if better, or with a small uphill probability
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # cool down
        T = max(min_T, T * decay)

        # occasional random restart to escape possible plateaus
        if it % 5000 == 0:
            rand_vec = rng.exponential(1.0, size=n)
            rand_vec /= rand_vec.sum()
            rand_score = _score(rand_vec)
            if rand_score < cur_score:
                cur = rand_vec
                cur_score = rand_score
                if rand_score < best_score:
                    best = rand_vec.copy()
                    best_score = rand_score

    # ----------------------------------------------------------------------
    # Return the best vector found as plain Python floats
    return [float(x) for x in best]
