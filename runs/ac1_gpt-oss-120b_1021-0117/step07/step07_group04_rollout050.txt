def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # correct sum due to numeric errors
    diff = w.sum() - 1.0
    if abs(diff) > 1e-12:
        # adjust uniformly
        w -= diff / n
        w = np.maximum(w, 0.0)
        # renormalize if needed
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
    return w

def _score_fft(v, n, L):
    s = v.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    return float(2.0 * n * conv.max() / (s * s))

def _subgrad(a, n, L):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv_full.max()
    # mask tolerance
    eps = max_val * 1e-8 + 1e-12
    mask = conv_full >= max_val - eps
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt
    # reverse w
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)  # mapping for gradient positions
    grad = 2.0 * grad_full[idx]
    return grad, max_val

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)  # small headroom

    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    L = 1 << ((2*n - 1).bit_length())

    # Build initial candidates
    candidates = []

    # Uniform
    candidates.append(np.full(n, 1.0 / n))

    xs = np.arange(n, dtype=np.float64)

    # Triangular
    tri = np.minimum(xs + 1, n - xs)
    candidates.append(tri / tri.sum())

    # Sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    candidates.append(sine / sine.sum())

    # Cosine bump (raised)
    cosb = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    candidates.append(cosb / cosb.sum())

    # Gaussian families
    for sigma_factor in (0.05, 0.08, 0.12):
        sigma = sigma_factor * n
        gauss = np.exp(-0.5 * ((xs - (n-1)/2) / sigma) ** 2)
        candidates.append(gauss / gauss.sum())

    # Power law
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        candidates.append(pw / pw.sum())

    # Random Dirichlet-like (exponential of random)
    for _ in range(3):
        rand = rng.random(n)
        cand = np.exp(rand)  # positive
        candidates.append(cand / cand.sum())

    # Parent if present
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            candidates.append(parent)

    # Evaluate initial candidates
    best_vec = None
    best_sc = float('inf')
    for vec in candidates:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # Optimization parameters
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349
    # For random restarts schedule
    restart_interval = 2000  # iterations
    # For periodic symmetrization
    sym_interval = 150

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # Compute gradient and max conv
        grad, max_val = _subgrad(cur, n, L)

        # momentum update
        momentum = beta * momentum + (1 - beta) * grad

        # step size schedule
        step = step0 / math.sqrt(it)

        # try several scaling factors
        best_local_vec = cur
        best_local_sc = cur_sc
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - step * factor * momentum)
            sc = _score_fft(cand, n, L)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local_vec = cand

        # acceptance
        if best_local_sc < cur_sc:
            cur, cur_sc = best_local_vec, best_local_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional acceptance with probability
            prob = math.exp(-(best_local_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = best_local_vec, best_local_sc

        # update global best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            # Early exit condition
            if best_sc < target:
                break

        # periodic symmetrization
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                    if best_sc < target:
                        break

        # random redistribution if stuck
        if no_improve > 3000:
            # perform small random perturbation to escape local minima
            perturb = cur * (rng.normal(scale=0.01, size=n))
            perturb = np.maximum(perturb, -cur)  # ensure nonnegative after addition
            cand = cur + perturb
            cand = _project_simplex(cand)
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
                    if best_sc < target:
                        break

        # occasional restart with new random candidate
        if restart_interval > 0 and it % restart_interval == 0:
            # generate random Dirichlet candidate and continue from there
            rand = rng.random(n)
            cand = rand / rand.sum()
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                cur, cur_sc = cand, cand_sc
                momentum.fill(0)
                no_improve = 0

    # return best vector as list
    return [float(x) for x in best_vec]

import time, math, numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # tiny adjustment for numerical drift
    diff = w.sum() - 1.0
    if abs(diff) > 1e-12:
        w -= diff / n
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s > 0:
            w /= s
    return w

# ------------------------------------------------------------
# Fast scoring using FFT (assumes non‑negative entries)
def _score_fft(a: np.ndarray, n: int, L: int) -> float:
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution (with a mild tolerance)
def _subgrad(a: np.ndarray, n: int, L: int):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    eps = max_val * 1e-6 + 1e-12          # a bit wider than pure max
    mask = conv_full >= max_val - eps
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)      # mapping to gradient entries
    grad = 2.0 * grad_full[idx]
    return grad, max_val

# ------------------------------------------------------------
# Symmetrize a vector and re‑project onto the simplex
def _symmetrize_proj(v: np.ndarray) -> np.ndarray:
    sym = (v + v[::-1]) * 0.5
    return _project_simplex(sym)
