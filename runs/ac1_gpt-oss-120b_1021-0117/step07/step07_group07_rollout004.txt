def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # length determination
    if "height_sequence_1" in globals():
        init = np.maximum(np.asarray(height_sequence_1, dtype=np.float64), 0.0)
        cur = _project_simplex(init)
        n = cur.shape[0]
    else:
        n = rng.integers(2000, 3000)
        cur = _project_simplex(rng.random(n))

    # Precompute pad length for FFT
    L = 1 << ((2*n - 1).bit_length())

    # Best record
    best = cur.copy()
    best_sc = _score(best)

    # Adam params
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 1.0
    temp = 0.2
    temp_factor = 1.003
    temp_max = 200.0

    it = 0
    no_improve = 0

    # main loop
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        it += 1

        # FFT + conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # Weighted gradient via softmax
        # adjust temperature if tiny differences
        w = np.exp(temp * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # reverse for convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)   # reverse indexing to align conv product
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 / (np.sqrt(v_hat) + eps)
        cand = cur - step * m_hat

        # enforce nonnegativity and simplex
        cand = _project_simplex(cand)
        # mirror symmetrization
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # small probabilistic accept
            if rng.random() < 0.005:
                cur, cur_sc = cand, cand_sc
            else:
                no_improve += 1

        # temperature schedule
        if temp < temp_max:
            temp *= temp_factor

        # periodic symmetrization already done above; maybe extra every 300

        # random mix move for diversification every 200 iterations
        if it % 200 == 0:
            rand = rng.random(n)
            rand = _project_simplex(rand)
            # direction combine
            alpha = 0.2  # mixing weight
            mix = (1 - alpha) * cur + alpha * rand
            mix = (mix + mix[::-1]) * 0.5
            mix = _project_simplex(mix)
            mix_sc = _score(mix)
            if mix_sc < cur_sc:
                cur, cur_sc = mix, mix_sc
                no_improve = 0
                if mix_sc < best_sc:
                    best, best_sc = mix.copy(), mix_sc

        # stochastic local move when stuck

                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:

        # early stop if target satisfied
        if best_sc < 1.50274349:

    # final evaluation using evaluator to ensure validity (not needed)
    # but we can compute again maybe not needed.

    return [float(x) for x in best]

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score for a probability vector (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for returning the result
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1) Initialization
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        cur = _project_simplex(np.maximum(a0, 0.0))
    else:
        # random Dirichlet start
        n = rng.integers(2000, 3000)
        cur = rng.random(n)
        cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------------
    # 2) Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.8

    # soft‑max temperature for max‑approximation
    temp = 0.2
    temp_factor = 1.003
    temp_max = 250.0

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution & weighted gradient (soft‑max) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # weights focused on the current maximal entries
        w = np.exp(temp * (conv - max_val))
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)

        # gradient for the weighted sum 2*a*w_rev (see derivation)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

    return [float(x) for x in best]
