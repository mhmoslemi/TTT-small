import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper bound measured by ``evaluate_sequence``.
    The routine works with any existing global ``height_sequence_1``;
    otherwise it falls back to a uniform distribution.
    """
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # --------------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # reasonable fallback size

    # --------------------------------------------------------------
    # Local score evaluator – identical to the provided ``evaluate_sequence``
    def _score_and_idx(a: np.ndarray):
        """Return (score, index_of_max) for a non‑negative vector a."""
        total = a.sum()
        if total <= 0.0:
            return float("inf"), -1
        b = np.convolve(a, a)               # full convolution
        max_b = b.max()
        idx = int(b.argmax())
        score = 2.0 * len(a) * max_b / (total * total)
        return score, idx

    # --------------------------------------------------------------
    # Initialise – parent if available, otherwise uniform
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        s = parent.sum()
        cur = parent / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    cur_score, cur_idx = _score_and_idx(cur)
    best = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Hyper‑parameters (all linear‑memory)
    step_initial = 0.5          # initial move magnitude
    step_decay = 3000.0         # decay constant for step size
    min_step = 1e-4

    temp_initial = max(best_score * 0.02, 1e-4)   # starting temperature
    temp_decay = 5000.0        # exponential cooling

    no_improve = 0
    iteration = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    while time.monotonic() < deadline:
        iteration += 1

        # Adaptive step size and temperature
        step = max(min_step, step_initial * np.exp(-iteration / step_decay))
        temp = temp_initial * np.exp(-iteration / temp_decay)

        # ------------------------------------------------------------------
        # Choose a move – most of the time we target the current worst shift
        if rng.random() < 0.7:                   # targeted move
            i_low = max(0, cur_idx - (n - 1))
            i_high = min(cur_idx, n - 1)
            if i_low <= i_high:
                i = rng.integers(i_low, i_high + 1)
                # pick j outside the overlap interval
                while True:
                    j = rng.integers(0, n)
                    if not (i_low <= j <= i_high):
                        break
            else:                                 # fallback to weighted random
                i = rng.choice(n, p=cur / cur.sum())
                j = rng.integers(0, n - 1)
                while j == i:
                    j = rng.integers(0, n - 1)
        else:                                     # weighted random move
            i = rng.choice(n, p=cur / cur.sum())
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)

        # Transfer a fraction of mass from i to j
        delta = cur[i] * rng.random() * step
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # Guard against tiny negative values caused by rounding
        if cand[i] < 0.0:
            cand[j] += cand[i]      # cand[i] is negative here
            cand[i] = 0.0

        # Re‑normalise to unit sum (protect against rounding drift)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s

        # Score the candidate
        cand_score, cand_idx = _score_and_idx(cand)

        # Acceptance test (simulated annealing)
        accept = False
        if cand_score < cur_score:
            accept = True
        elif temp > 0.0 and rng.random() < np.exp(-(cand_score - cur_score) / temp):
            accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            cur_idx = cand_idx
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # Periodic smoothing step – cheap way to flatten the profile
        if iteration % 300 == 0:
            k = max(3, int(n ** 0.20))               # kernel width ≈ n^0.2
            kernel = np.ones(k, dtype=np.float64) / k
            smooth = np.convolve(cur, kernel, mode='same')
            smooth = np.clip(smooth, 0.0, None)
            sm_sum = smooth.sum()
            if sm_sum > 0.0:
                smooth /= sm_sum
                smooth_score, smooth_idx = _score_and_idx(smooth)
                if smooth_score < cur_score:
                    cur = smooth
                    cur_score = smooth_score
                    cur_idx = smooth_idx
                    no_improve = 0
                    if smooth_score < best_score:
                        best = smooth.copy()
                        best_score = smooth_score

        # ------------------------------------------------------------------
        # Random restart if we have not improved for a long stretch
        if no_improve >= 5000:
            rand = rng.exponential(scale=1.0, size=n)
            rand /= rand.sum()
            rand_score, rand_idx = _score_and_idx(rand)
            cur = rand
            cur_score = rand_score
            cur_idx = rand_idx
            no_improve = 0
            if rand_score < best_score:
                best = rand.copy()
                best_score = rand_score

        # ------------------------------------------------------------------
        # Light time check every few thousand iterations
        if iteration % 2000 == 0 and time.monotonic() >= deadline:
            break

    # --------------------------------------------------------------
    # Final safety – ensure feasibility and fall back if needed
    best = np.clip(best, 0.0, 1000.0)
    total = best.sum()
    if total > 0.0:
        best /= total
    try:
        final_score = evaluate_sequence(best.tolist())
        if not np.isfinite(final_score):
            best = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        best = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best]
