import time, numpy as np

def _project_simplex(v):
    # Euclidean projection onto the probability simplex
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    idx = (2 * n - 2) - np.arange(n)

    def _score_fft(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv.max()
        s = v.sum()
        if s <= 0:
            return np.inf
        return 2.0 * n * max_val / (s * s)

    cur_score = _score_fft(cur)
    best_vec = cur.copy()
    best_score = cur_score

    # Adam state
    m = np.zeros_like(cur)
    v_hat = np.zeros_like(cur)  # rename to avoid confusion with variable v
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.5

    # schedule for hard mask tolerance (temperature)
    # we will gradually tighten the mask (smaller tolerance)
    tol_factor = 0.99
    tol = 1e-8  # relative tolerance initial

    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # self‑convolution
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # mask of entries near the maximum (subgradient of the max)
        # decreasing tolerance makes mask shrink as we progress
        cur_tol = max_val * max(tol, 1e-15)
        mask = (conv >= max_val - cur_tol).astype(np.float64)

        # gradient = 2 * (a * mask_rev)  (cross‑correlation)
        w_rev = mask[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat_corr) + eps)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        cand_score = _score_fft(cand)

            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
        else:

            if rng.random() < 0.002:

import time, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w
