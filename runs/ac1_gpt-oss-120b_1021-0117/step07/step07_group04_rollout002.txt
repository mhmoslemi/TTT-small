import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # fix possible rounding drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------------
    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349
    max_no_improve = 2000

    # --------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution & active set -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        tol = max_val * 1e-8 + 1e-12
        mask = conv >= max_val - tol
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # ----- gradient (sub‑gradient of max‑convolution) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        direction = m_hat / (np.sqrt(v_hat) + eps)

        step = step0 / math.sqrt(it)

        # ----- line search over several scaling factors -----
        local_best_vec = cur
        local_best_sc = cur_sc
        for factor in (0.5, 0.75, 1.0, 1.25, 1.5):
            cand = _project_simplex(cur - step * factor * direction)
            sc = _score_fft(cand, n, L)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_vec = cand

        # ----- acceptance -----
        if local_best_sc < cur_sc - 1e-12:
            cur, cur_sc = local_best_vec, local_best_sc
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(local_best_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = local_best_vec, local_best_sc
                no_improve = 0

        # ----- global best update -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            no_improve = 0

        # ----- occasional symmetrisation -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
                cur, cur_sc = sym, sym_sc
                no_improve = 0

        # ----- random perturbation when stuck -----
        if no_improve >= max_no_improve:
            sigma = 0.1 / math.sqrt(it)
            pert = rng.normal(scale=sigma, size=n)
            tmp = best_vec + pert
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, n, L)
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
            else:
                # simple mass exchange as a cheap alternative
                i = rng.integers(0, n)
                if cur[i] > 0:
                    delta = cur[i] * rng.random() * 0.5
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score_fft(tmp, n, L)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()
                        no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
