    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.05)

        seq = np.asarray(height_sequence_1, dtype=np.float64)
    else:

    # Clip and ensure nonneg

    # Ensure sum >=0.01; normalize to sum=1 for convenience (scales out)

        seq = seq + (0.01 - total)/len(seq)
        total = np.sum(seq)
    seq = seq / total   # sum=1
    n = len(seq)
    # objective function
    def obj(v):
        c = np.convolve(v, v)
        return 2.0 * n * np.max(c)  # sum=1
    cur_score = obj(seq)
    best_seq = seq.copy()
    best_score = cur_score
    rng = np.random.default_rng(seed)
    max_iters = 200000  # adjust based on time
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # Randomly choose operation type: 0 = random mass transfer, 1 = targeted
        if rng.random() < 0.7:
            # random mass transfer like before but can use larger fraction
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = seq[i] * rng.random() * 0.2 + 1e-12
            cand = seq.copy()
            cand[i] -= delta
            cand[j] += delta
            # projection onto simplex (sum=1)
            # we can just renormalize since sum unchanged, but ensure nonneg.
            # Renormalization after small negative may produce negative; use projection.
            cand = _project_simplex(cand)
        else:
            # targeted move
            conv = np.convolve(seq, seq)
            k = int(np.argmax(conv))
            # compute contributions
            idx = np.arange(n)
            j_idx = k - idx
            mask = (j_idx >= 0) & (j_idx < n)
            prod = np.zeros(n)
            prod[mask] = seq[mask] * seq[j_idx[mask]]
            imax = int(np.argmax(prod))
            jmax = k - imax
            # pick a target index different from imax,jmax
            t = rng.integers(0, n)
            while t == imax or t == jmax:

            cand[jmax] -= delta
            cand[t] += 2*delta
            # ensure nonnegative (already)

        if cand_score < cur_score:
            seq = cand
            cur_score = cand_score
