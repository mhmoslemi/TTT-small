```python
import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # compute sorted v desc
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    # find rho
    cond = u - css / ind > 0
    if cond.any():
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _max_conv(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return conv

def _score_from_conv(conv, n):
    # sum=1 assumed
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # small slack

    # initialization
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random of length ~2500
        cur = rng.random(2500, dtype=np.float64)
    # Ensure nonnegative
    cur = np.maximum(cur, 0.0)
    # Project onto simplex (sum=1)
    cur = _project_simplex(cur)
    # enforce symmetry
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    n = cur.shape[0]
    L = 1 << ((2*n - 1).bit_length())

    # initial scoring
    conv = _max_conv(cur, L)
    best_sc = _score_from_conv(conv, n)
    best_vec = cur.copy()
    best_sc = float(best_sc)

    # Adam variables
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.05  # base learning rate

    # temperature for softmax
    temp_factor = 0.02   # relative factor of max
    temp_decay = 0.9995
    min_temp_factor = 1e-6

    it = 0
    # main loop
    while time.monotonic() < deadline:
        it += 1

        # FFT of cur
        a_fft = np.fft.rfft(cur, n=L)

        # full convolution
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()

        # compute softmax weights
        temp = max(max_val * max(temp_factor, min_temp_factor), 1e-12)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient via weighted sum of conv
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx_start = 2*n - 2
        grad = 2.0 * grad_full[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        # project onto simplex and enforce symmetry
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        # compute candidate score
        conv_cand = _max_conv(cand, L)
        cand_sc = _score_from_conv(conv_cand, n)

        # accept if improvement
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
        else:
            # occasional random perturb for escape
            if it % 5000 == 0:
                # small random shuffle of mass
                idx = rng.integers(0, n)
                delta = cur[idx] * rng.random() * 0.3
                jdx = rng.integers(0, n-1)
                if jdx >= idx:
                    jdx += 1
                tmp = cur.copy()
                tmp[idx] -= delta
                tmp[jdx] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                conv_tmp = _max_conv(tmp, L)
                tmp_sc = _score_from_conv(conv_tmp, n)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    cur = tmp

        # temperature decay
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # early stop if time near deadline
        if time.monotonic() > deadline - 0.1:
            break

    # Return best vector as list of Python floats
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if cond.any():
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def _conv_max(a, L):
    """Return full convolution a*a (length 2n-1) using FFT with padding L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv
