import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # fix tiny sum error
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, L):
    """Fast score computation (assumes a>=0)."""
    s = a.sum()
    if s < 1e-12:

    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max() / (s * s))

            no_improve = 0
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()
        else:
            no_improve += 1
            # occasional uphill acceptance (simulated annealing)
            prob = math.exp(-(best_local_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = best_local_vec, best_local_sc

        # ----- periodic symmetrisation -----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sc_sym = _score_fft(sym, L)
            if sc_sym < cur_sc:
                cur, cur_sc = sym, sc_sym
                no_improve = 0
                if sc_sym < best_sc:
                    best_sc, best_vec = sc_sym, sym.copy()

        # ----- random perturbation when stuck -----
        if no_improve > 3000:
            delta = rng.normal(scale=0.01, size=n)
            cand = cur + delta
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            sc = _score_fft(cand, L)
            if sc < cur_sc:
                cur, cur_sc = cand, sc
                no_improve = 0
                if sc < best_sc:
                    best_sc, best_vec = sc, cand.copy()
            else:
                # fallback to a fresh random simplex point
                cand = rng.random(n)
                cand = cand / cand.sum()
                sc = _score_fft(cand, L)
                if sc < cur_sc:
                    cur, cur_sc = cand, sc
                    no_improve = 0
                    if sc < best_sc:
                        best_sc, best_vec = sc, cand.copy()

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------------------
    # Return best found sequence as a plain list
    return [float(x) for x in best_vec]
