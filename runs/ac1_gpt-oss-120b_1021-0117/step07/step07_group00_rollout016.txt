import time, math, numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step sequence that minimizes the AC1 upper bound.
    Uses an iterative LP‑guided mixing scheme (the LP maximises the sum of a
    partner sequence under the current autocorrelation bound).  The best found
    vector is returned as a plain Python list.
    """
    # --------------------------------------------------------------
    # optional LP solver
    try:
        import cvxpy as cp
    except Exception:          # pragma: no cover
        cp = None

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # --------------------------------------------------------------
    # initialise from parent if present
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.clip(cur, 0.0, 1000.0)

    # normalise sum to sqrt(2n) – the standard normalisation for the score
    n = cur.size
    target_sum = math.sqrt(2.0 * n)
    s = cur.sum()
    if s > 0.0:
        cur *= target_sum / s
    else:
        cur.fill(target_sum / n)

    # --------------------------------------------------------------
    # helper: fast evaluation of the score (identical to the official one)
    def fast_score(v):
        v = np.asarray(v, dtype=np.float64)
        n_ = v.shape[0]
        L = 1 << ((2 * n_ - 1).bit_length())
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n_ - 1]
        max_val = conv.max()
        s_ = v.sum()
        if s_ < 0.01:
            return float('inf')
        return 2.0 * n_ * max_val / (s_ * s_)

    best_vec = cur.copy()
    best_sc = fast_score(best_vec)
    cur_sc = best_sc

    # --------------------------------------------------------------
    # parameters
    t = 0.2                # mixing factor
    min_t = 0.05
    it = 0
    no_improve = 0
    target_score = 1.50274349

    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---- current autocorrelation bound --------------------------------
        L = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        M = conv.max()

        # ---- LP step (if possible) ----------------------------------------
        if cp is not None:
            g = cp.Variable(n, nonneg=True)
            f_const = cp.Constant(cur)
            conv_fg = cp.conv(f_const, g)                # length 2n‑1
            prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_fg <= M])
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=200, verbose=False)
            except Exception:                            # pragma: no cover
                # solver failure → fall back to a cheap perturbation
                cp = None
                prob = None
            if prob is not None and g.value is not None:
                g_opt = np.maximum(g.value, 0.0)
                sum_g = g_opt.sum()
                if sum_g <= 0.0:
                    g_opt = cur.copy()
                    sum_g = g_opt.sum()
                # rescale to the fixed total mass
                g_scaled = g_opt * (target_sum / sum_g)

                # ---- mixing -------------------------------------------------
                cand = (1.0 - t) * cur + t * g_scaled
                cand = np.clip(cand, 0.0, 1000.0)
                cand_sum = cand.sum()
                if cand_sum > 0.0:
                    cand *= target_sum / cand_sum

                cand_sc = fast_score(cand)

                # ---- greedy acceptance ------------------------------------
                if cand_sc < cur_sc:
                    cur, cur_sc = cand, cand_sc
                    no_improve = 0
                else:
                    no_improve += 1
                    if rng.random() < 0.03:            # occasional uphill move
                        cur, cur_sc = cand, cand_sc
                        no_improve = 0

                if cur_sc < best_sc:
                    best_sc, best_vec = cur_sc, cur.copy()
                if best_sc < target_score:
                    break
                # ---- step‑size decay ---------------------------------------
                if it % 10 == 0:
                    t = max(min_t, t * 0.96)
                # ---- random redistribution when stuck ----------------------
                if no_improve > 250:
                    i = rng.integers(0, n)
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    delta = cur[i] * rng.random() * 0.4
                    cur[i] -= delta
                    cur[j] += delta
                    cur = np.clip(cur, 0.0, 1000.0)
                    cur_sum = cur.sum()
                    if cur_sum > 0.0:
                        cur *= target_sum / cur_sum
                    cur_sc = fast_score(cur)
                    if cur_sc < best_sc:
                        best_sc, best_vec = cur_sc, cur.copy()
                    no_improve = 0
                continue      # next optimisation round

        # ---- fallback: simple stochastic redistribution --------------------
        # (executed only when cvxpy is unavailable or failed)
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.35
        cur[i] -= delta
        cur[j] += delta
        cur = np.clip(cur, 0.0, 1000.0)
        cur_sum = cur.sum()
        if cur_sum > 0.0:
            cur *= target_sum / cur_sum
        cur_sc = fast_score(cur)
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
        # early exit check
        if best_sc < target_score:
            break

    # --------------------------------------------------------------
    # return the best sequence found
    return [float(x) for x in best_vec]
