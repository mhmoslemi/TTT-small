import time, math, numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
def _project_simplex(v):
    """Project v onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ------------------------------------------------------------
# Helper: convolution via FFT (full linear convolution).
def _conv_fft(a, L):
    """Return full linear convolution a*a (length 2*len(a)-1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * len(a) - 1]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative heights aiming at a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # tiny safety margin

    # --------------------------------------------------------
    # length of the sequence
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = int(rng.integers(2000, 3000))

    # FFT length (next power of two >= 2*n-1)
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # candidate generators (shape → normalized to sum=1)
    def _uniform():
        a = np.ones(n, dtype=np.float64)
        return a / a.sum()

    def _linear():
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()

    def _triangular():
        c = (n - 1) / 2.0
        d = np.abs(np.arange(n) - c)
        a = c + 1 - d
        a[a < 0] = 0.0
        return a / a.sum()

    def _hann():
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
        return a / a.sum()

    def _gaussian(sigma):
        c = (n - 1) / 2.0
        d = np.arange(n) - c
        a = np.exp(-0.5 * (d / sigma) ** 2)
        return a / a.sum()

    def _exponential(tau):
        c = (n - 1) / 2.0
        d = np.abs(np.arange(n) - c)
        a = np.exp(-d / tau)
        return a / a.sum()

    def _raised_cosine(p):
        a = (0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))) ** p
        return a / a.sum()

    generators = [
        _uniform,
        _linear,
        _triangular,
        _hann,
    ]

    # add a few parameterised families
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        generators.append(lambda s=sigma: _gaussian(s))
    for tau in (n / 8.0, n / 6.0, n / 4.0):
        generators.append(lambda t=tau: _exponential(t))
    for p in (0.5, 1.0, 2.0, 3.0):
        generators.append(lambda pp=p: _raised_cosine(pp))

    # include the provided parent if available
    if "height_sequence_1" in globals():
        def _parent():
            a = np.asarray(height_sequence_1, dtype=np.float64)
            return _project_simplex(np.maximum(a, 0.0))
        generators.append(_parent)

    # --------------------------------------------------------
    # Evaluate initial candidates
    best_vec = None
    best_sc = math.inf

    for gen in generators:
        a = gen()

        sc = float(2.0 * n * conv.max())
        if sc < best_sc:
            best_sc = sc
            best_vec = a.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    eps = 1e-12

        conv = _conv_fft(cur, L)
        max_val = conv.max()
        # indices attaining the maximum (normally just one)

        grad = np.zeros_like(cur)
        for k in argmax:
