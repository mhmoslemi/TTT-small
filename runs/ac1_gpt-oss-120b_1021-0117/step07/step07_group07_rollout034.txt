def max_convolution(vec):
    n = len(vec)
    L = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(vec, n=L)

    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = rng.integers(2000, 3000)
        cur = rng.random(n)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    # compute initial score
    cur_max = _max_conv(cur, L, n)
    cur_score = 2.0 * n * cur_max / (cur.sum() ** 2)
    best = cur.copy()
    best_score = cur_score
    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    # temperature for softmax
    alpha = 0.5
    alpha_factor = 1.005
    alpha_max = 300.0
    # step magnitude base
    step0 = 0.3
    # counters
    it = 0
    no_improve = 0
    # loops
    while time.monotonic() < deadline:
        it += 1
        # FFT compute
        A = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # compute softmax weights
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):

        else:

        # gradient via weighted sum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(A * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        m = beta1 * m + (1.0 - beta1) * grad

import time
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_simplex(v):
    """Project a vector onto the simplex {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort decreasing
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Helper: maximum entry of the linear convolution a*a
def _max_conv(a, L, n):
    """Return max_{k} (a * a)[k] using FFT (no extra memory)."""
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv.max()

# ------------------------------------------------------------

    rng = np.random.default_rng(seed)

    L = 1 << ((2 * n - 1).bit_length())

    # Adam optimizer state

    alpha = 0.5               # initial exponent
