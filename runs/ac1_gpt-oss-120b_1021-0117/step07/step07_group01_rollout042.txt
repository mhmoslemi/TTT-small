def _simplex_projection(v):
    # Standard projection onto simplex (sum=1, v>=0)
    # Using algorithm O(n log n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    step0 = 0.6
    it = 0
    no_improve = 0

    # thresholds
    sym_interval = 200
    random_interval = 2500
    smooth_interval = 500

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # FFT of current a
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # subgradient weights
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad

            a = 0.5 * best + 0.5 * rand_vec
            a = _simplex_projection(a)
            no_improve = 0

    # return best vector as list
    return [float(x) for x in best]

import time, math, numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a non‑negative vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    # find the largest index where condition holds
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_fft(a: np.ndarray, L: int, n: int) -> float:
    """Score used by the evaluator (sum of coefficients is assumed to be 1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # ----------------------------------------------------------------------
    # initialise a feasible sequence
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a random simplex of length ~2500
        a = rng.random(2500, dtype=np.float64)
    a = _simplex_projection(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    best = a.copy()
    best_sc = _score_fft(best, L, n)

    # Adam‑like state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.6
    it = 0
    no_improve = 0

    # parameters for occasional operations
    sym_interval = 200          # symmetrisation frequency
    smooth_interval = 500       # smoothing frequency
    rand_interval = 2500        # random redistribution threshold

    # ----------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- current convolution and maximal value -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- sub‑gradient of the max‑convolution (uniform over active indices) -----
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # ----- gradient of the weighted max‑convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)

        step = step0 / (math.sqrt(it) + 1.0)

        # pick the better proposal
        if cand_add_sc < cand_mul_sc:

        else:
            cand, cand_sc = cand_mul, cand_mul_sc

        # ----- acceptance logic -----
        if cand_sc < best_sc:

            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < _score_fft(a, L, n):
            a = cand

                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta

                a = tmp
            no_improve = 0

        # ----- restart from a mixed random vector after longer stagnation -----
        if no_improve > 2 * rand_interval:
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _simplex_projection(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _simplex_projection(a)
            no_improve = 0

    # ----------------------------------------------------------------------
    # final short hill‑climbing pass using only cheap two‑index moves
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * 0.05
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _simplex_projection(cand)
        sc = _score_fft(cand, L, n)
        if sc < best_sc:
            best = cand
            best_sc = sc

    return [float(x) for x in best]
