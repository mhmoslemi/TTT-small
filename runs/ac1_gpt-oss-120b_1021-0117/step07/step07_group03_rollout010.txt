import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ----------------------------------------------------------------------
def _conv_max_and_grad(a, L):
    """
    Compute (max_convolution, subgradient) of full convolution a*a.
    a is a non‑negative vector (not necessarily normalized).
    Returns max value and gradient of the max w.r.t. a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt                       # average over active entries
    w_rev = mask[::-1]                    # reversed mask for gradient
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)      # map to gradient components
    grad = 2.0 * grad_full[idx]
    return max_val, grad

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------------
    # Initial vector
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # light smoothing to reduce extreme spikes
        if a0.size > 5:
            k = 5
            kernel = np.ones(k) / k
            a0 = np.convolve(a0, kernel, mode="same")
        cur = _project_simplex(np.maximum(a0, 0.0))
    else:
        # fallback: Dirichlet‑like random start
        cur = rng.random(2000, dtype=np.float64)
        cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # best solution tracker
    max_val, _ = _conv_max_and_grad(cur, L)
    best_score = 2.0 * n * max_val
    best_vec = cur.copy()

    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.2                # base learning rate (will be scaled)
    t = 0

    # control parameters
    sym_period = 250             # how often to symmetrise
    stagnation = 0
    stagn_limit = 4000
    restart_prob = 0.001
    perturb_prob = 0.005
    perturb_scale = 0.05

    while True:
        if time.time() >= deadline:
            break
        t += 1

        # gradient of current point
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_score = 2.0 * n * max_val

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # adaptive step
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # occasional symmetrisation (helps flatten the convolution peak)
        if t % sym_period == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # acceptance (pure descent; occasional random jump)
        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            stagnation = 0
        else:
            stagnation += 1
            # small chance to accept a worsening move (simulated annealing)
            if rng.random() < math.exp(-(cand_score - cur_score) / (base_lr + 1e-12)):
                cur, cur_score = cand, cand_score
                stagnation = 0

        # update global best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()
            stagnation = 0

        # random perturbation when stuck
        if stagnation > stagn_limit:
            if rng.random() < restart_prob:
                # full restart from a fresh random point
                cur = rng.random(n, dtype=np.float64)
            else:
                # gentle noise injection
                cur = cur + perturb_scale * rng.standard_normal(size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0
            continue

        # occasional tiny perturbation to escape flat regions
        if rng.random() < perturb_prob:
            cur = cur + perturb_scale * rng.standard_normal(size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))

    # Return the best found coefficients as plain Python floats
    return [float(x) for x in best_vec]
