import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # adjust floating error
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

                    no_improve = 0

                        best_sc, best_vec = tmp_sc, tmp.copy()
        # early exit if target achieved
        if best_sc < target:
            break

    return [float(x) for x in best_vec]

            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
