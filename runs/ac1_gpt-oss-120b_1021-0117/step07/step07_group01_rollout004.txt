import time, math, numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto probability simplex (sum=1, >=0)
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------
# Helper: score for a vector already normalized to sum=1
def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a tiny safety buffer
    deadline = start + max(0.0, float(budget_s) - 0.25)

    # ------------------ initialization --------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: modest size random vector
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)               # sum = 1, non‑negative
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # -------------------- Adam state ---------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.8

    # soft‑max temperature for the max‑convolution
    beta_soft = 8.0

    it = 0
    no_improve = 0
    target = 1.50274349

    # -------------------- main loop ---------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(a, n=L)

        # full convolution a*a
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # soft‑max weights over convolution entries
        w = np.exp(beta_soft * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of weighted sum 2*a * w_rev (convolution)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        # two candidate proposals: additive (projected) and multiplicative (mirror)
        cand_add = _project_simplex(a - update)

        # multiplicative update, safely clipped
        grad_clip = np.clip(grad, -40.0 / max(step, 1e-12), 40.0 / max(step, 1e-12))
        exp_factor = np.exp(-step * grad_clip)
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()

        # pick better candidate
        sc_add = _score(cand_add)
        sc_mul = _score(cand_mul)
        if sc_add <= sc_mul:
            cand, cand_sc = cand_add, sc_add
        else:
            cand, cand_sc = cand_mul, sc_mul

        # acceptance logic
        cur_sc = _score(a)
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
            # slowly sharpen soft‑max to focus on the true peak
            if beta_soft < 40.0:
                beta_soft *= 1.015
        elif cand_sc < cur_sc:
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic uphill move
            prob = math.exp(-(cand_sc - best_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # periodic symmetrisation (helps reduce peak)
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                no_improve = 0

        # random redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
            no_improve = 0

        # hard restart if very long stagnation
        if no_improve > 5200:
            rnd = rng.random(n, dtype=np.float64)
            rnd = _project_simplex(rnd)
            a = 0.5 * best + 0.5 * rnd
            a = _project_simplex(a)
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # -------------------- local refinement --------------------
    # simple pairwise mass transfer hill‑climb
    while time.monotonic() < deadline - 0.12:
        i = rng.integers(0, n)
        if best[i] <= 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.07   # up to 7 % shift
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best = cand
            best_sc = cand_sc

        # occasional symmetrisation inside refinement
        if rng.random() < 0.008:
            sym = (best + best[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best = sym
                best_sc = sym_sc

    # ----------------------------------------------------------
    # Return as plain Python list of floats
    return [float(x) for x in best]
