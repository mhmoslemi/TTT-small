import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to minimise
    the AC1 upper bound measured by ``evaluate_sequence``.
    """
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ----------------------------------------------------------------------
    # Problem size and initial vector
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
        cur = np.clip(cur, 0.0, 1000.0)
    else:
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)  # uniform fallback

    n = cur.shape[0]

    # Normalise to sum == 1 (objective is scale‑invariant)
    s = cur.sum()
    if s < 1e-12:
        cur = np.full_like(cur, 1.0 / n)
    else:
        cur = cur / s

    # ----------------------------------------------------------------------
    # Helper to compute the score (no extra convolution)
    def _score(x, conv_cache=None):
        # sum == 1 by construction, but compute robustly
        sm = x.sum()
        if sm < 0.01:
            return float("inf")
        if conv_cache is None:
            c = np.convolve(x, x)
        else:
            c = conv_cache
        return float(2.0 * n * c.max() / (sm * sm))

    # initial score and convolution
    conv = np.convolve(cur, cur)
    cur_score = _score(cur, conv)
    best_vec = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng(seed)

    # Simulated‑annealing parameters
    T0 = max(0.1 * cur_score, 1e-4)          # initial temperature
    step_factor = 0.5                         # initial fraction of moved mass
    it = 0
    max_iters = 200_000

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # --------------------------------------------------------------
        # Identify the lag giving the current maximum autocorrelation
        idx_max = int(conv.argmax())

        # Contributions of each index to that lag
        i_start = max(0, idx_max - (n - 1))
        i_end = min(n - 1, idx_max)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = idx_max - i_vals
        contrib = cur[i_vals] * cur[j_vals]

        # source index – where product is largest
        src_local = int(np.argmax(contrib))
        src = int(i_vals[src_local])

        # destination index – random but different from source
        dest = int(rng.integers(0, n))
        while dest == src:
            dest = int(rng.integers(0, n))

        # Amount of mass to move (fraction of source)
        delta = cur[src] * step_factor * rng.random()
        if delta <= 0.0:
            continue

        # Perform the tentative move
        cur[src] -= delta
        cur[dest] += delta
        if cur[src] < 0.0:          # safeguard against numerical drift
            cur[src] += delta
            cur[dest] -= delta
            continue

        # New convolution and score
        conv_new = np.convolve(cur, cur)
        new_score = _score(cur, conv_new)

        # Temperature schedule (linear cooling)
        elapsed = time.monotonic() - start
        frac = elapsed / (deadline - start) if deadline > start else 1.0
        temperature = T0 * (1.0 - frac)
        if temperature < 1e-12:
            temperature = 1e-12

        # Acceptance rule (Metropolis)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = cur.copy()
        else:
            # revert the move
            cur[src] += delta
            cur[dest] -= delta

        if step_factor < 0.01:
