import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # ---------- initial sequence ----------
    if 'height_sequence_1' in globals():
        a0 = np.maximum(np.asarray(height_sequence_1, dtype=np.float64), 0.0)
        n = a0.size
    else:
        n = 2500
        a0 = None

    # Gaussian guess of same length (always created)
    x = np.arange(n, dtype=np.float64)
    sigma = n / 6.0
    a1 = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
    a1 /= a1.sum()

    # pick better of the two initial points
    L = 1 << ((2 * n - 1).bit_length())
    def _init_score(a):
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()
    if a0 is not None:
        a0 /= a0.sum()
        if _init_score(a0) < _init_score(a1):
            a = a0.copy()
        else:
            a = a1.copy()
    else:
        a = a1.copy()

    best = a.copy()
    best_sc = _init_score(best)

    # ---------- optimisation parameters ----------
    eta0 = 0.6          # initial step size
    tau = 5.0           # soft‑max temperature (fixed)
    it = 0
    no_improve = 0
    target = 1.50274349

    # ---------- main loop ----------
    while time.monotonic() < deadline:
        it += 1

        # forward FFT of current a
        a_fft = np.fft.rfft(a, n=L)

        # full autocorrelation
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if best_sc < target:
            break

        # ----- soft‑max weights for the max entry -----
        # shift by max for numerical stability (all args ≤ 0)
        w = np.exp(tau * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient via convolution with reversed weights -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)   # pick the linear part
        grad = 2.0 * grad_full[idx]

        # ----- mirror‑descent (multiplicative) update -----
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)

        # re‑normalise onto the simplex (positivity is guaranteed)
        total = a.sum()
        if total <= 0.0 or not np.isfinite(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        # ----- occasional symmetrisation -----
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym /= sym.sum()

            sc_sym = 2.0 * n * conv_sym.max()

                best_sc = sc_sym
                best = sym.copy()
                a = sym
                no_improve = 0
            else:
                a = sym

            a = 0.5 * best + 0.5 * rnd
            a = a / a.sum()
            no_improve = 0

    return [float(x) for x in best]
