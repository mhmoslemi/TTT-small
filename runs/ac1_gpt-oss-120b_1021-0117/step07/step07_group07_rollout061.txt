import time, numpy as np

# ----------------------------------------------------------------------
# helper: project onto probability simplex (sum = 1, non‑negative)
def _project_simplex(v):
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# helper: fast score (identical to evaluate_sequence but without clipping)
def _fast_score(a):
    """Score 2*n*max(conv)/sum(a)^2 using FFT, no safety clipping."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)
