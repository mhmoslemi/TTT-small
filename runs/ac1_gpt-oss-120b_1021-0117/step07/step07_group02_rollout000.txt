import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_fast(a):
    """Score for a vector already normalized to sum=1: 2*n*max(conv)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            if sym_score < best_score:
                best_score = sym_score
                best_vec = sym.copy()
                best_no_improve = 0

        # random local perturbation if stuck
        if best_no_improve >= 4000:
            # redistribute a random amount between two positions
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_fast(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best_score = tmp_score
                    best_vec = tmp.copy()
            else:
                # full random restart with a bias towards current shape
                rnd = _project_simplex(rng.random(n))
                rnd_score = _score_fast(rnd)
                if rnd_score < best_score:
                    cur, cur_score = rnd, rnd_score
                    best_vec = rnd.copy()
            best_no_improve = 0

        # decay temperature for smooth max
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # early stop if target reached
        if best_score < target:
            break

        # occasional time checkpoint
        if it % 2000 == 0 and time.monotonic() >= deadline:
            break

    # Return the best-found vector as plain Python floats
    return [float(x) for x in best_vec]
