import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalize to sum exactly 1
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2 * n - 1]
    return float(2.0 * n * conv.max())

    L = 1 << ((2 * n - 1).bit_length())

    best_sc = _score(best_vec)
    cur_sc = best_sc

    beta2 = 0.999
    eps = 1e-8

    step0 = 0.8
    alpha0 = 1.0
    alpha_max = 200.0
    target = 1.50274349
    it = 0

    # ---------- main optimisation loop ----------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # FFT of current vector

        conv = conv_full[:2 * n - 1]

        # grow alpha quickly, then cap
        alpha = min(alpha_max, alpha0 + 0.02 * it)   # linear growth

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def _score(a):
    """Score for a vector already normalised to sum = 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
