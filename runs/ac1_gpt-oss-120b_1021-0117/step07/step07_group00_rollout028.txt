    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # keep margin
    # get initial vector
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        cur = rng.random(n)
    # project to simplex (sum to 1)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())   # compute exact score
    cur_sc = best_sc

    # algorithm parameters
    momentum = np.zeros_like(cur)
    beta = 0.85
    step0 = 1.0
    it = 0
    no_improve = 0
    temp = None
    decay = 0.995
    target = 1.50274349
    # main loop
    while time.time() < deadline:
        it += 1

        # compute convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        max_val = conv.max()
        # temperature: set based on max
        if temp is None:
            temp = max_val * 0.06
        else:
            temp = max(temp * decay, max_val * 1e-4)

        # softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # compute gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size adaptation
        step = step0 / math.sqrt(it)

        # candidate
        cand = _project_simplex(cur - step * momentum)
        cand_sc = evaluate_sequence(cand.tolist())
        # acceptance
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (step + 1e-12)):
            cur, cur_sc = cand, cand_sc
            no_improve = 0

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

        # ----- gradient of the weighted max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size -----
        step = step0 / math.sqrt(it)

        # ----- candidate vector (projected) -----
        cand = _project_simplex(cur - step * momentum)

        # ----- score of candidate (exact) -----
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_max = conv_cand.max()
        cand_sum = cand.sum()
        cand_sc = 2.0 * n * cand_max / (cand_sum * cand_sum)

        # ----- acceptance test (simulated annealing) -----
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (step + 1e-12)):
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1

        # ----- random redistribution when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)

            tmp_fft = np.fft.rfft(tmp, n=L)
            conv_tmp = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
            tmp_max = conv_tmp.max()
            tmp_sum = tmp.sum()
            tmp_sc = 2.0 * n * tmp_max / (tmp_sum * tmp_sum)

            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- periodic symmetrisation -----
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)

            sym_fft = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_max = conv_sym.max()
            sym_sum = sym.sum()
            sym_sc = 2.0 * n * sym_max / (sym_sum * sym_sum)

            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # safety fallback (should never happen)
    if best_vec.size == 0:
        best_vec = cur

    return [float(x) for x in best_vec]
