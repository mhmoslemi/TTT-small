import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score assuming sum(a)==1 (the denominator becomes 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback length chosen to be similar to the parent size
        a0 = rng.random(2500, dtype=np.float64)

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.9                # momentum decay
    step0 = 1.0               # initial step magnitude
    it = 0
    no_improve = 0
    target = 1.50274349

    # temperature parameters for soft‑max weighting (annealed)
    temp_factor = 0.10        # relative to current max value
    temp_decay = 0.995        # per‑iteration decay
    min_temp_factor = 1e-8

    # ----- main loop ------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # full convolution a*a
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weighting over the convolution entries
        temp = max_val * max(temp_factor, min_temp_factor)
        # avoid division by zero when max_val is zero (won't happen for sum=1)
        w = np.exp((conv - max_val) / (temp + 1e-20))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient of the weighted maximum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule (diminishing)
        step = step0 / math.sqrt(it)

        # candidate move
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_vec(cand)

        # acceptance logic
        cur_sc = _score_vec(cur)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # update global best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # periodic symmetrisation (helps many optimal patterns)
        if it % 300 == 0:

            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < best_sc:

                cur, cur_sc = sym, sym_sc
                no_improve = 0
            elif sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0

        # random local redistribution when we seem stuck

            if cur[i] > 0.0:

                cur, cur_sc = rnd, rnd_sc
                momentum.fill(0.0)

        # early exit if target reached

            break

    # Return best-found vector as plain Python floats
    return [float(x) for x in best_vec]
