def optimize(init_vec, time_budget, start_time):
   n = len(init_vec)
   L = 1 << ((2 * n - 1).bit_length())
   cur = _project_simplex(init_vec)
   # ensure symmetry
   cur = (cur + cur[::-1]) / 2.0
   cur = _project_simplex(cur)
   best_vec = cur.copy()
   max_val, _ = _conv_max_and_grad(cur, L)
   best_score = 2.0 * n * max_val
   step0 = 1.0
   beta = 0.9
   momentum = np.zeros_like(cur)
   it = 0
   no_improve = 0
   while time.time() < start_time + time_budget:
        it += 1
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_score = 2.0 * n * max_val
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        cand = cur - step * momentum
        # enforce bounds positivity before projection
        cand = _project_simplex(cand)
        # symmetry
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)
        # evaluate
        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_score = 2.0 * n * cand_max
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            # accept with prob
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score
                no_improve = 0
            else:
                no_improve += 1

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()
            no_improve = 0

        # random perturb if stuck
        if no_improve > 5000:
            sigma = 0.05 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert = (pert + pert[::-1]) * 0.5
            pert = _project_simplex(pert)
            pert_max, _ = _conv_max_and_grad(pert, L)
            pert_score = 2.0 * n * pert_max
            if pert_score < cur_score:
                cur = pert
                cur_score = pert_score
                if pert_score < best_score:
                    best_score = pert_score
                    best_vec = pert.copy()
                no_improve = 0
            else:
                momentum.fill(0.0)
                no_improve = 0

   return best_vec, best_score

import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of a non‑negative vector onto the simplex {x≥0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _conv_max_and_grad(a, L):
    """
    Returns (max convolution value, sub‑gradient) of a*a (full convolution).
    a must be a non‑negative 1‑d array.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

# ----------------------------------------------------------------------
def _optimize_vector(init_vec, time_budget, rng, global_deadline):
    """
    Projected sub‑gradient descent with momentum, enforced symmetry and occasional
    stochastic perturbations. Returns best vector found and its score.
    """
    target = 1.50274349
    # initial projection + symmetrisation
    cur = _project_simplex(np.maximum(init_vec, 0.0))
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    max_val, _ = _conv_max_and_grad(cur, L)
    best_score = 2.0 * n * max_val
    cur_score = best_score

    momentum = np.zeros_like(cur)
    beta = 0.9          # momentum decay
    step0 = 0.8         # initial step factor
    it = 0
    no_improve = 0
    start = time.time()

    while True:
        now = time.time()
        if now >= global_deadline or (now - start) >= time_budget:
            break
        it += 1

        max_val, grad = _conv_max_and_grad(cur, L)
        cur_score = 2.0 * n * max_val

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # decreasing step size
        step = step0 / math.sqrt(it)

        # candidate move
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_score = 2.0 * n * cand_max

        # accept if improvement, else Metropolis‑type acceptance
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
                no_improve = 0
            else:
                no_improve += 1

        # record best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()
            no_improve = 0

        # stochastic perturbation if stuck
        if no_improve > 5000:
            sigma = 0.05 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert = (pert + pert[::-1]) * 0.5
            pert = _project_simplex(pert)
            pert_max, _ = _conv_max_and_grad(pert, L)
            pert_score = 2.0 * n * pert_max
            if pert_score < cur_score:
                cur, cur_score = pert, pert_score
                if pert_score < best_score:
                    best_score = pert_score
                    best_vec = pert.copy()
                no_improve = 0
            else:
                momentum.fill(0.0)
                no_improve = 0

        if best_score < target:
            break

    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # length of the problem (use parent if present)
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)

    # 3. half‑sine shape

    # 4. quadratic (peak in the centre)
    init_vecs.append((x + 1) * (n - x))

    # 5. Gaussian‑like bump centred
    centre = (n - 1) / 2.0
    alpha = 4.0 / (n ** 2)
    init_vecs.append(np.exp(-alpha * (x - centre) ** 2))

    # 6. random Dirichlet‑like vector
    init_vecs.append(rng.random(n, dtype=np.float64))

    # ------------------------------------------------------------------
    best_global_vec = None
    best_global_score = float("inf")
    remaining_initialisations = len(init_vecs)

    for idx, base in enumerate(init_vecs):
        if time.time() >= deadline:
            break
        # allocate time slice proportionally to remaining restarts
        time_left = deadline - time.time()
        time_budget = time_left / (remaining_initialisations - idx)

        # normalise and symmetrise the start vector
        vec = np.maximum(base, 0.0)
        if vec.sum() == 0:
            vec = np.full_like(vec, 1.0 / vec.size)
        else:
            vec = vec / vec.sum()
        vec = (vec + vec[::-1]) * 0.5
        if vec.sum() == 0:

        # run the optimiser on this start point
        cand_vec, cand_score = _optimize_vector(vec, time_budget, rng, deadline)

        if cand_score < best_global_score:

            best_global_vec = cand_vec.copy()
            if best_global_score < 1.50274349:

    # fallback – uniform vector (should never be needed)
    if best_global_vec is None:
        best_global_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_global_vec]
