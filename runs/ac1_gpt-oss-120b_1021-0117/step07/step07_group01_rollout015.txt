import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(v, L):
    """Score assuming sum(v)=1, using FFT convolution."""
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * v.size - 1]
    return float(2.0 * v.size * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------

    else:

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    # enforce symmetry from the start (does not change sum)
    a = (a + a[::-1]) * 0.5
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score_vec(best, L)

    # Adam / momentum state
    m = np.zeros_like(a)
    v_ad = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # hyper‑parameters
    step0 = 0.6                     # base learning rate
    tau = 0.2                       # smoothing temperature for max
    tau_decay = 0.9995
    target = 1.50274349

    it = 0
    no_improve = 0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # --------------------------------------------------------------

        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # --------------------------------------------------------------
        # smoothed max weights (softmax over convolution values)
        # avoid overflow: subtract max_conv before exponentiation

        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):

            mask = conv >= max_conv - max_conv * 1e-12

        w /= w_sum

        idx = (2 * n - 2) - np.arange(n)
