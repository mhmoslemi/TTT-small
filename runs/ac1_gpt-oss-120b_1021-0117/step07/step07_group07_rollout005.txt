import time, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (nonnegative, sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score assuming sum(a)=1 (i.e., 2*n*max_conv)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    start = time.monotonic()

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # Adam moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    step0 = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349
    sym_interval = 200

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution and max index -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        k_star = int(conv.argmax())

        # ----- gradient of the maximal convolution entry -----
        idx = np.arange(n)
        j = k_star - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[j[mask]]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if it % 1000 == 0:
            step0 *= 0.95

        # ----- early termination -----
        if best_sc < target:
            break

    # Return as plain Python list
    return [float(x) for x in best]
