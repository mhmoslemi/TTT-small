import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # Sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound as evaluated by ``evaluate_sequence``.
    The routine performs a short random‑search around a uniform baseline,
    respecting the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Determine length n
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # Uniform baseline (sum = 1)
    base_val = 1.0 / n
    cur = np.full(n, base_val, dtype=np.float64)

    # Track the best candidate seen so far
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())

    # Simple schedule for perturbation magnitude
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Diminishing step size
        step = 0.5 / math.sqrt(it)

        # Random Gaussian perturbation, then enforce non‑negativity
        noise = rng.standard_normal(n) * step
        cand = cur + noise
        cand = np.maximum(cand, 0.0)

        # Project back onto the simplex (sum = 1)
        cand = _project_simplex(cand)

        # Evaluate exact score
        sc = evaluate_sequence(cand.tolist())
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

        # Accept the candidate as the new current point if it improves
        # (simple greedy acceptance)
        if sc < evaluate_sequence(cur.tolist()):
            cur = cand

        # Periodic reset to uniform to avoid stagnation
        if it % 500 == 0:
            cur = np.full(n, base_val, dtype=np.float64)

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
