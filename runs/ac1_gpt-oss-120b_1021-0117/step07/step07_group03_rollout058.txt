import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max_and_grad(a, L):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def _conv_max(a, L):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return conv.max()

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialization
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # use Dirichlet to get sum=1
        cur = rng.dirichlet(np.ones(2000), size=1)[0]
    # ensure nonnegative and project to simplex (sum=1)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # evaluate initial
    max_val, grad = _conv_max_and_grad(cur, L)
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()
    momentum = np.zeros_like(cur)
    beta = 0.95
    step0 = 0.8  # larger initial step
    it = 0
    no_improve = 0
    target = 1.50274349

    # main loop
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # gradient at current point
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        step = step0 / math.sqrt(it + 1.0)

        # candidate via Euclidean projection
        cand_euc = _project_simplex(cur - step * momentum)

        # candidate via entropic update
        # Clip gradient to avoid overflow
        grad_clipped = np.clip(grad, -1e3, 1e3)
        exp_factor = np.exp(-step * grad_clipped)
        cand_ent = cur * exp_factor
        s = cand_ent.sum()
        if s > 0:
            cand_ent = cand_ent / s
        else:
            cand_ent = _project_simplex(cand_ent)

        # optionally symmetrize candidates
        cand_euc = (cand_euc + cand_euc[::-1]) * 0.5
        cand_euc = _project_simplex(cand_euc)
        cand_ent = (cand_ent + cand_ent[::-1]) * 0.5
        cand_ent = _project_simplex(cand_ent)

        # evaluate both
        cand_sc_euc = 2.0 * n * _conv_max(cand_euc, L)
        cand_sc_ent = 2.0 * n * _conv_max(cand_ent, L)

        # pick better candidate
        if cand_sc_euc <= cand_sc_ent:
            cand = cand_euc
            cand_sc = cand_sc_euc
        else:
            cand = cand_ent
            cand_sc = cand_sc_ent

        # acceptance
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (step + 1e-12)):
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # random perturbation if stuck
        if no_improve > 3000:
            sigma = 0.1 / math.sqrt(it + 1.0)
            pert = cur + sigma * rng.standard_normal(size=n)

            # enforce symmetry

            else:

                no_improve = 0

        # early stop if target met
        if best_sc < target:

    return [float(x) for x in best_vec]

import time, math, numpy as np

# -------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L):
    """
    Returns (max_convolution, subgradient) of the full convolution a*a.
    a is assumed non‑negative; sum constraint is handled externally.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad


def _conv_max(a, L):
    """Maximum value of the full convolution a*a."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return conv.max()
