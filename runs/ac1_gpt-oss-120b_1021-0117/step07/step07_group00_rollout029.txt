import time
import math
import numpy as np

def _project_simplex(v):
    """Project a vector onto the probability simplex (sum == 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # Sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, L):
    """Score used during the search (sum(a) is assumed to be 1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound as measured by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Determine length
    if "height_sequence_1" in globals():
        n = int(height_sequence_1.shape[0])
    else:
        n = 2000                     # fallback length

    # FFT size (next power of two >= 2*n-1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Start from the simplex‑projected uniform vector (theoretical optimum)
    cur = np.full(n, 1.0 / n, dtype=np.float64)
    cur = _project_simplex(cur)      # just in case of rounding

    best_vec = cur.copy()
    best_sc = _fast_score(best_vec, L)

    # Simple stochastic refinement – harmless if it never improves.
    momentum = np.zeros_like(cur)
    beta = 0.85
    step0 = 1.0
    it = 0
    temp = None
    decay = 0.996
    min_temp_factor = 1e-4

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Compute convolution via FFT (full length)
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # Temperature for soft‑max weighting
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # Soft‑max over the convolution values
        w = np.exp((conv - max_val) / (temp + 1e-12))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # Gradient of the weighted max (FFT‑based)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        # Candidate step & projection onto simplex
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _fast_score(cand, L)

        # Accept if better, otherwise occasionally accept worse
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc
            cur = cand
        else:
            # Simulated‑annealing style acceptance
            prob = math.exp(-(cand_sc - best_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand

        # --------------------------------------------------------------
        # Occasionally try a random mass‑swap to escape plateaus
        if it % 1200 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, L)
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
                cur = tmp

        # --------------------------------------------------------------
        # Symmetrisation every few hundred iterations (often helpful)
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L)
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
                cur = sym

        # Early exit if we reach the theoretical optimum (score == 2.0)
        # (allow a tiny tolerance for numerical noise)
        if best_sc <= 2.0 + 1e-9:
            break

    # ------------------------------------------------------------------
    # Final safety clipping (required by the evaluator)
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    if best_vec.sum() < 0.01:            # ensure the evaluator does not reject
        best_vec[0] = 0.01

    # Return as plain Python list
    return [float(x) for x in best_vec]
