def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import necessary modules.
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # init:
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    # compute score
    def compute_score(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    best = a.copy()
    best_score = compute_score(best)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 1.0
    it = 0
    no_improve = 0

    # maybe also maintain random candidate pool
    # simple loop
    while time.monotonic() < deadline:
        it += 1
        # compute conv/grad
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # mask for entries within epsilon of max
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        # compute gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        step = step0 / (np.sqrt(it) + 1.0)   # or step0 / (1+it*0.001)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        # Candidate additive
        cand_add = _project_simplex(a - update)

        # Candidate multiplicative (mirror descent)
        cand_mul = a * np.exp(-step * grad)
        s = cand_mul.sum()
        if s > 0:
            cand_mul /= s
        else:
            cand_mul = a.copy()

        # evaluate both
        sc_add = compute_score(cand_add)
        sc_mul = compute_score(cand_mul)

        # pick better
        if sc_add < sc_mul:
            cand, cand_sc = cand_add, sc_add
        else:
            cand, cand_sc = cand_mul, sc_mul

        # acceptance: if better than best update best; else possibly accept if better than current a
        if cand_sc < best_score:
            best = cand.copy()
            best_score = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < compute_score(a):
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # probabilistic acceptance
            prob = np.exp(-(cand_sc - best_score) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # symmetrize occasionally
        if it % 250 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = compute_score(sym)
            if sym_sc < best_score:
                best = sym.copy()
                best_score = sym_sc
                a = sym
                no_improve = 0

        # random perturbation if stuck
        if no_improve > 2000:
            # shift some mass
            idxi = rng.integers(0, n)
            if a[idxi] > 0:
                delta = a[idxi] * rng.random() * 0.5
                idxj = rng.integers(0, n-1)
                if idxj >= idxi: idxj += 1
                tmp = a.copy()
                tmp[idxi] -= delta
                tmp[idxj] += delta
                tmp = _project_simplex(tmp)
                a = tmp
            no_improve = 0

        # random restart if still stuck
        if no_improve > 4000:
            rand = rng.random(n)
            rand = _project_simplex(rand)
            a = 0.5 * best + 0.5 * rand
            a = _project_simplex(a)
            no_improve = 0

        # occasional random global search: try a new random Dirichlet candidate
        if it % 500 == 0:
            rand = rng.random(n)
            rand = _project_simplex(rand)
            sc_rand = compute_score(rand)
            if sc_rand < best_score:
                best = rand.copy()
                best_score = sc_rand
                a = rand
                no_improve = 0

        # early exit if target reached
        if best_score < 1.50274349:
            break

    # final fallback ensure best valid
    if best_score == float('inf') or not np.isfinite(best_score):
        best = np.array([1.0])  # trivial fallback
    return [float(x) for x in best]

import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a, L):
    """Fast internal score (assumes sum(a)=1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

    Return a list of non‑negative coefficients whose AC1 upper bound
