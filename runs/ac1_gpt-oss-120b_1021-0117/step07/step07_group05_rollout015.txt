import time
import numpy as np

# ------------------------------------------------------------
# Helper: project a vector onto the probability simplex (sum == 1, >=0)
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Helper: linear convolution of a vector with itself (FFT, no dense matrices)
def _self_convolve(x, n):
    # length for a linear convolution without wrap‑around
    m = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(fx * fx, n=m)[: 2 * n - 1]
    return conv

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper bound as measured by ``evaluate_sequence``.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    rng = np.random.default_rng(seed)

    # ---------- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        # if no parent is supplied, start from a random Dirichlet point
        n = 2000
        a = rng.random(n)
    # project onto the simplex (sum == 1, non‑negative)
    a = _project_simplex(a)
    n = a.shape[0]

    # ---------- keep the best candidate seen so far -----------------------
    try:
        best_score = evaluate_sequence(a.tolist())
    except Exception:
        best_score = float("inf")
    best_vec = a.copy()

    cur_score = best_score

    # ---------- optimisation parameters ------------------------------------
    max_iters = 200000               # safety cap
    step_base = 0.02                 # base step size (will be normalised)
    no_improve = 0
    restart_interval = 3000

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # ---- convolution and the active (max) index -----------------------
        conv = _self_convolve(a, n)
        max_b = conv.max()
        k = int(conv.argmax())          # lag where the maximum occurs

        # ---- sub‑gradient of max_b w.r.t. a -------------------------------
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            g[idx] = a[k - idx]        # a_{k-i}
        g *= 4.0 * n                    # gradient of 2*n*max_b (objective)

        # ---- normalise step to keep updates moderate -----------------------
        norm = np.linalg.norm(g) + 1e-12
        step = step_base / norm

        # ---- tentative update ---------------------------------------------
        a_candidate = a - step * g

        # occasional small random jitter to escape flats (including uniform)
        if rng.random() < 0.008:
            jitter = rng.normal(scale=0.005, size=n)
            a_candidate += jitter

        # ---- project back onto the simplex ---------------------------------
        a_candidate = _project_simplex(a_candidate)

        # ---- evaluate ------------------------------------------------------
        try:
            cand_score = evaluate_sequence(a_candidate.tolist())
        except Exception:
            continue

        # ---- accept if improvement (monotone descent) --------------------
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            no_improve = 0
        else:

        # ---- keep global best ---------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_vec = a_candidate.copy()

        # ---- occasional restart if stuck ----------------------------------
        if no_improve >= restart_interval:
            a = _project_simplex(rng.random(n))
            cur_score = evaluate_sequence(a.tolist())
            no_improve = 0

    # --------------------------------------------------------------------
    # Return the best vector found (as a plain Python list of floats)
    return [float(x) for x in best_vec]
