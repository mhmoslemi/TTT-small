def compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    # Clip
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    n = seq.size
    # compute convolution via FFT
    size = 1 << ((2*n - 1).bit_length())
    # Using rfft for real transforms
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    max_b = conv[:2*n-1].max()
    return 2 * n * max_b / (sum_a * sum_a)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # Determine n
    if "height_sequence_1" in globals():
        init_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = init_seq.size
    else:
        n = 2000
        init_seq = None

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    # compute score for uniform and parent
    def score_of(seq):
        seq = np.clip(seq, 0.0, 1000.0)
        sum_a = seq.sum()
        if sum_a < 0.01:
            return float('inf')
        size = 1 << ((2*n - 1).bit_length())
        fft_seq = np.fft.rfft(seq, n=size)
        conv = np.fft.irfft(fft_seq * fft_seq, n=size)
        max_b = conv[:2*n-1].max()
        return float(2 * n * max_b / (sum_a * sum_a))

    best_seq = uniform
    best_score = score_of(best_seq)

    if init_seq is not None:
        parent_score = score_of(init_seq)
        if parent_score < best_score:
            best_seq = init_seq.copy()
            best_score = parent_score

    # set current seq for hill climbing
    cur = best_seq.copy()
    cur_score = best_score

    rng = np.random.default_rng(seed)

    # parameters for annealing
    temperature = 1.0
    decay = 0.9999  # per iteration
    iteration = 0
    # We'll run until deadline
    while time.monotonic() < deadline:
        iteration += 1
        # generate move: swap two indices or transfer
        if rng.random() < 0.5:
            # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = cur[i] * rng.random() * 0.1  # up to 10% of source
            if delta <= 0:
                continue
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            # ensure non-neg
            if cand[i] < 0:
                continue

        # compute score
        cand_score = score_of(cand)

        # accept if better or with probability
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / max(temperature,1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score

        temperature *= decay

        # occasional random restart if no improvement for many iterations
        if iteration % 5000 == 0 and time.monotonic() < deadline:
            # random perturbation to escape
            cur = best_seq.copy()
            # add small gaussian noise and renormalize to keep sum unchanged
            noise = rng.normal(scale=0.01, size=n)
            cand = cur + noise
            cand = np.clip(cand, 0.0, None)
            # renorm to original sum (which stays same)
            sum_c = cand.sum()
            if sum_c > 0:
                cand *= cur.sum() / sum_c
                cur = cand
                cur_score = score_of(cur)

    # final candidate as list of floats
    return [float(x) for x in best_seq]

    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    max_b = conv.max()
    score = float(2 * n * max_b / (sum_a * sum_a))
    return score, max_b, conv


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    yields a low AC1 upper bound as measured by ``evaluate_sequence``.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)   # safety margin

    # ----- problem size -------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # fallback size

    rng = np.random.default_rng(seed)

    # ----- initial candidates -------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score, _, _ = _evaluate(best_seq)

    # parent (if present) – keep it as‑is (clipped) and test
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        parent_score, _, _ = _evaluate(parent)
        if parent_score < best_score:
            best_seq = parent.copy()
            best_score = parent_score

    # current solution for the local search
    cur_seq = best_seq.copy()
    cur_score, _, cur_conv = _evaluate(cur_seq)

    # ---- search parameters ---------------------------------------------
    temperature = 1.0
    decay = 0.9995
    no_improve = 0
    max_no_improve = 5000
    max_iters = 10_000_000   # soft upper bound, real stop is deadline

    # ----- main loop ----------------------------------------------------
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # use the current peak of the convolution to guide a move
        peak_idx = int(np.argmax(cur_conv))

        # identify the pair (i, j) that contributes most to this peak
        partners = peak_idx - np.arange(n)
        mask = (partners >= 0) & (partners < n)
        contrib = np.zeros_like(cur_seq)
        contrib[mask] = cur_seq[mask] * cur_seq[partners[mask]]
        i_max = int(np.argmax(contrib))

        # pick a random destination different from i_max and its partner
        partner_i = partners[i_max] if mask[i_max] else -1
        j = rng.integers(0, n)
        while j == i_max or j == partner_i:
            j = rng.integers(0, n)

        # move a fraction of mass from i_max to j
        if cur_seq[i_max] <= 0.0:
            continue
        delta = cur_seq[i_max] * rng.random() * 0.2   # ≤20 % of source mass

        # ensure we stay in the allowed range
        if cur_seq[j] + delta > 1000.0:
            delta = max(0.0, 1000.0 - cur_seq[j])
        if delta <= 0.0:
            continue

        cand_seq = cur_seq.copy()
        cand_seq[i_max] -= delta
        cand_seq[j] += delta

        cand_score, _, cand_conv = _evaluate(cand_seq)

        # acceptance rule (hill‑climb with occasional uphill steps)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = cand_seq
            cur_score = cand_score
            cur_conv = cand_conv
            temperature *= decay
            no_improve = 0
            if cand_score < best_score:
                best_seq = cand_seq.copy()
                best_score = cand_score
        else:
            no_improve += 1
            temperature *= decay

        # occasional random restart if stuck
        if no_improve >= max_no_improve:
            # draw a fresh Dirichlet vector (sum = 1) and continue
            fresh = rng.gamma(1.0, size=n)
            fresh /= fresh.sum()
            cur_seq = fresh
            cur_score, _, cur_conv = _evaluate(cur_seq)
            no_improve = 0

    # Return the best sequence found as plain Python floats
    return [float(x) for x in best_seq]
