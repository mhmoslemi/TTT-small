def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # helper
    def project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
            return v
        n = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    def fast_score(a, conv=None):
        # assumes sum(a)=1; returns 2*n*max(conv)
        if conv is None:
            n = a.shape[0]
            L = 1 << ((2 * n - 1).bit_length())
            af = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * a.shape[0] * conv.max())
```

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a):
    """Score = 2*n*max(conv) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
