import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            if rng.random() < 0.002:  # occasional acceptance
                cur, cur_sc = cand, cand_sc

        # temperature cooling
        temp = max(0.5, temp * temp_factor)

        # periodic symmetrisation
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # occasional sparsification (zero tiny entries)
        if it % 300 == 0:
            thresh = np.percentile(cur, 5)
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc

        # random mass redistribution when stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc

        # early exit if target reached
        if best_sc < target:
            break

    # ----- random local search on remaining budget ------------------------
    while time.monotonic() < deadline - 0.1:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.1
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc

    return [float(x) for x in best]
