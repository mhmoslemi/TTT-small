import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_and_fft(a, L):
    """Return full convolution a*a (length 2n-1), its maximum, and FFT of a."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    n = a.shape[0]
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    return conv, max_val, a_fft

def _grad_from_max(a, a_fft, conv, max_val, L):
    """Sub‑gradient of max(conv) w.r.t. a using the exact max mask."""
    # weight 1 on all positions attaining the maximum
    mask = np.isclose(conv, max_val, atol=1e-12)
    cnt = mask.sum()
    if cnt == 0:         # fallback (should not happen)
        mask = conv == max_val
        cnt = mask.sum()
    w = np.zeros_like(conv)
    w[mask] = 1.0 / cnt
    # reverse for convolution
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad

    else:
        n = rng.integers(2000, 3000)
        cur = np.abs(rng.standard_normal(n))
        cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power of two >= 2n-1

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 1.0

    # best solution seen so far
    conv, max_val, a_fft = _conv_and_fft(cur, L)
    best = cur.copy()
    best_sc = 2.0 * n * max_val
    cur_sc = best_sc

    it = 0
    no_improve = 0

    # parameters for heuristic actions
    sym_interval = 200          # how often to symmetrise
    stagnation_limit = 3000     # when to try a random redistribution
    random_accept_prob = 0.004  # occasional acceptance of worse move
