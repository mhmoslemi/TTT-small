import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if np.sum(v) == 0.0:
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_convolve_full(x, y, size):
    """Full convolution of real vectors x and y using a fixed FFT size."""
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:len(x) + len(y) - 1]

def _bound_from_conv(a, conv):
    """Exact AC1 bound for a given vector a and its self‑convolution conv."""
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    return float(2.0 * len(a) * np.max(conv) / (s * s))

def _smooth_weights(conv, target_exp=8.0):
    """
    Compute soft‑max weights approximating the max of conv.
    Returns (smooth_max, weights) where weights sum to 1.
    """
    max_c = np.max(conv)
    if max_c <= 0:
        # avoid division by zero – uniform weights
        w = np.full_like(conv, 1.0 / conv.size)
        return 0.0, w
    s = target_exp / (max_c + 1e-12)          # scale so max(s*conv) ≈ target_exp
    z = s * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)                # stable soft‑max
    w = exp_z / np.sum(exp_z)
    smooth = np.dot(w, conv)
    # exact gradient of smooth‑max: w * (1 + s*(conv - smooth))
    w = w * (1.0 + s * (conv - smooth))
    return smooth, w

def _gradient(a, conv, weights, size):
    """
    Gradient of the smooth max objective.
    a   : current vector (len=n)
    conv: a*a convolution (len=2n-1)
    weights: same length as conv, sum to 1, already
    size: FFT size (>=3n)
    Returns gradient of shape (n,).
    """
    w_rev = weights[::-1]
    prod = _fft_convolve_full(a, w_rev, size)
    idx = (2 * len(a) - 2) - np.arange(len(a))
    return 2.0 * prod[idx]

    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as measured by ``evaluate_sequence``.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # FFT size (next power of two >= 3n)
    fft_len = 1 << int(np.ceil(np.log2(3 * n)))

    # ------------------------------------------------------------------
    # initial candidates
    best_vec = None
    best_score = float("inf")

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    conv_u = _fft_convolve_full(uniform, uniform, fft_len)
    score_u = _bound_from_conv(uniform, conv_u)
    best_vec, best_score = uniform, score_u

    # parent, if present
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        parent = _project_simplex(parent)          # enforce sum=1
