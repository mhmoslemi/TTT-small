def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # a little safety

    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    L = 1 << ((2*n-1).bit_length())   # power of two

    xs = np.arange(n, dtype=np.float64)

    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0/n))

    # triangular
    tri = np.minimum(xs+1, n - xs)
    candidates.append(tri / tri.sum())

    # raised cosine
    bump = (1 - np.cos(2*np.pi*xs/(n-1))) / 2
    candidates.append(bump / bump.sum())

    # exponential
    for alpha in (0.001, 0.003, 0.006):
        exp = np.exp(-alpha * xs)
        candidates.append(exp / exp.sum())

    # power law
    for p in (0.5, 1.0, 1.5):
        pw = (xs+1)**(-p)
        candidates.append(pw / pw.sum())

    # Gaussian with sigma varying (sigma as fraction of n)
    for sigma_frac in (0.1, 0.2, 0.3):
        sigma = sigma_frac * n
        gauss = np.exp(-0.5 * ((xs - (n-1)/2)/sigma)**2)
        candidates.append(gauss / gauss.sum())

    # random
    rand = rng.random(n)
    candidates.append(rand / rand.sum())

    # parent if exists
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            candidates.append(parent)

    # evaluate each using fast score
    def _score(a):
        s = a.sum()
        if s < 1e-12:
            return np.inf
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return float(2.0 * n * conv.max() / (s * s))

    best_sc = np.inf
    best_vec = None
    for vec in candidates:
        sc = _score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2   # base learning rate for simplex-projected update

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # Compute convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        # gradient: convolution of cur with reversed w, doubled
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update (unprojected)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # Accept if improvement else maybe rare acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional simulated annealing acceptance
            if rng.random() < 0.001:  # small chance
                cur, cur_sc = cand, cand_sc

        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            no_improve = 0

        # Symmetrize periodically
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # Random perturbation when stuck
        if no_improve > 2000:
            # generate small random redistribution
            delta = rng.random(n) * cur * 0.1
            idx_i = rng.integers(n, size=n)
            idx_j = rng.integers(n, size=n)
            # apply delta transfer
            tmp = cur.copy()
            # subtract from i, add to j
            np.add.at(tmp, idx_i, -delta)
            np.add.at(tmp, idx_j, delta)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0
            else:
                no_improve = 0  # reset anyway

        # Early break if target reached
        if best_sc < target:
            break

    return [float(x) for x in best_vec]

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum = 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, L):
    """Fast evaluation of the objective for a non‑negative vector a."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * len(a) - 1]
    return float(2.0 * len(a) * conv.max() / (s * s))
