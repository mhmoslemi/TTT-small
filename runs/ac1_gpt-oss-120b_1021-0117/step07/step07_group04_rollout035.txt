def _score_fft(a, n, L):
    """Fast evaluation of the objective (assumes a >= 0)."""
    s = a.sum()
    if s < 1e-12:

    return float(2.0 * n * conv.max() / (s * s))

    cand_list.append(np.full(n, 1.0 / n))

    # triangular (peak centre)
    tri = np.minimum(xs + 1, n - xs)
    cand_list.append(tri / tri.sum())

    # sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    cand_list.append(sine / sine.sum())

    # raised cosine bump
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    cand_list.append(bump / bump.sum())

    # exponential decays
    for a in (0.001, 0.003, 0.006):
        exp = np.exp(-a * xs)
        cand_list.append(exp / exp.sum())

    # power laws
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        cand_list.append(pw / pw.sum())

    # linear decreasing
    lin = (n - xs)
    cand_list.append(lin / lin.sum())

    # parent if available (sorted decreasing)
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            parent = np.sort(parent)[::-1]
            cand_list.append(parent)

    # ----- pick the best deterministic start ---------------------------------
    best_vec = None
    best_sc = float('inf')
    for vec in cand_list:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # ----- Adam optimiser -----------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.5

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # convolution and maximum
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weighting of active indices (sub‑gradient)
        tau = max_val * 0.01 + 1e-12
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the max‑convolution term
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = base_lr / (math.sqrt(it) + 1)
        upd = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - upd
        cand = np.maximum(cand, 0.0)

        sm = cand.sum()
        if sm <= 0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / sm

        # enforce decreasing order (rearrangement inequality)
        cand = np.sort(cand)[::-1]

        cand_sc = _score_fft(cand, n, L)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # global best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            s = tmp.sum()
            if s > 0:
                tmp = tmp / s
                tmp = np.sort(tmp)[::-1]
                tmp_sc = _score_fft(tmp, n, L)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

        # safety guard
        if it >= 200000:
            break

    return [float(v) for v in best_vec]
