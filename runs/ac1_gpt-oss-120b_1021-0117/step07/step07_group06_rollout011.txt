import time, math, numpy as np

def _project_simplex(v):
    # Project onto simplex sum=1 v>=0
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    # renormalize due to rounding
    s = w.sum()
    if s > 0:
        w = w / s
    else:
        w = np.full_like(v, 1.0/n)
    return w

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.6
    it = 0

    # scoring (same scaling as evaluator when sum=1)
    def _score_vec(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    cur_score = _score_vec(cur)
    best = cur.copy()
    best_score = cur_score

    # soft‑max temperature schedule
    alpha_max = 120.0
    alpha0 = 1.0

    no_improve = 0

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- FFT of current vector -----
        a_fft = np.fft.rfft(cur, n=L)

        # ----- autocorrelation (full) -----
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # ----- soft‑max weighting -----
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            mask = conv >= max_conv - 1e-12 * abs(max_conv) - 1e-15
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv, 1.0 / conv.size)
                w_sum = 1.0
        w /= w_sum

        # ----- gradient of weighted max‑convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - update
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / s

        cand_score = _score_vec(cand)

        # ----- acceptance -----
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1
            if no_improve % 2000 == 0:
                # gentle random jitter
                noise = rng.normal(scale=0.01, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur = cur / cur.sum()
                cur_score = _score_vec(cur)

        # ----- periodic symmetrisation -----
        if it % 4000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_vec(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score

        # ----- restart when stuck -----
        if no_improve > 10000:
            cur = best + rng.normal(scale=0.02, size=n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_score = _score_vec(cur)
            no_improve = 0

        # ----- early exit if very good -----
        if best_score < 1.502:
            break

    # ------------------------------------------------------------------
    return [float(x) for x in best]
