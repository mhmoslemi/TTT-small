import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                    if best_sc < target:
                        break

        # ----- random redistribution when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _fast_score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                        if best_sc < target:
                            break

        # ----- lightweight hill‑climbing moves (every 50 iters) -----
        if it % 50 == 0:
            for _ in range(5):
                i = rng.integers(0, n)
                if cur[i] == 0.0:
                    continue
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                move = cur[i] * rng.random() * 0.1
                new_vec = cur.copy()
                new_vec[i] -= move
                new_vec[j] += move
                # non‑negativity already guaranteed
                new_sc = _fast_score(new_vec)
                if new_sc < cur_sc:
                    cur, cur_sc = new_vec, new_sc
                    no_improve = 0
                    if new_sc < best_sc:
                        best_sc, best_vec = new_sc, new_vec.copy()
                        if best_sc < target:
                            break

    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
