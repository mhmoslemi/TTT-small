def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # length
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        init_parent = np.asarray(height_sequence_1, dtype=np.float64)
        init_parent = np.maximum(init_parent, 0.0)
    else:
        n = 2000
        init_parent = None

    # determine FFT size
    L = 1 << ((2*n - 1).bit_length())

    # build initial candidates list (list of np arrays)
    xs = np.arange(n, dtype=np.float64)
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0/n))
    # triangular
    tri = np.minimum(xs+1, n-xs)
    candidates.append(tri / tri.sum())
    # sine bump
    sine = np.sin(np.pi * (xs+1) / (n+1))
    candidates.append(sine / sine.sum())
    # raised cosine
    bump = (1 - np.cos(2*np.pi*xs/(n-1))) / 2
    candidates.append(bump / bump.sum())
    # exponential decays
    for alpha in (0.001, 0.005, 0.01):
        exp = np.exp(-alpha*xs)
        candidates.append(exp / exp.sum())
    # power law
    for p in (0.5, 1.0, 1.5):
        pw = (xs+1)**(-p)
        candidates.append(pw / pw.sum())
    # random
    rand = rng.random(n)
    candidates.append(rand / rand.sum())
    # parent if available
    if init_parent is not None and init_parent.sum() > 0:
        parent_norm = init_parent / init_parent.sum()
        candidates.append(parent_norm)

    # symmetrize each candidate and project onto simplex (should already be)
    # Actually need to ensure they sum to 1 (they are normalized). Then symmetrize:
    for i, cand in enumerate(candidates):
        sym = (cand + cand[::-1]) * 0.5
        candidates[i] = _project_simplex(sym)

    # evaluate each candidate quickly using FFT for max conv
    best_vec = None
    best_score = float('inf')
    for cand in candidates:
        a_fft = np.fft.rfft(cand, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        score = 2.0 * n * max_val / (cand.sum()**2)
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
    cur = best_vec.copy()
    cur_score = best_score

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.05  # base learning rate for Adam
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    # patience counters
    no_improve = 0
    target = 1.50274349
    sym_interval = 200  # symmetrizations

    while time.monotonic() < deadline:
        t += 1
        # compute convolution and gradient for cur
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val / (cur.sum()**2)

        # compute mask of active entries
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # gradient
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        idx = (2*n - 2) - np.arange(n)

import time
import numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Fix possible numerical drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

    candidates.append(rand / rand.sum())

    # 8) Parent if available
    if parent is not None and parent.sum() > 0.0:
        candidates.append(parent / parent.sum())

    # Symmetrise and project each start onto the simplex
    for i, c in enumerate(candidates):
        sym = (c + c[::-1]) * 0.5
        candidates[i] = _project_simplex(sym)

    # --------------------------------------------------------------
    # Choose the best deterministic start (fast evaluation)
    best_vec = None
    best_score = float('inf')
    for cand in candidates:
        a_fft = np.fft.rfft(cand, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        score = float(2.0 * n * max_val / (cand.sum() ** 2))
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
    cur = best_vec.copy()
    cur_score = best_score

    # --------------------------------------------------------------
    # Adam optimiser parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha_lr = 0.08          # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    # Counters and thresholds
    no_improve = 0
    target = 1.50274349
    sym_interval = 250        # enforce symmetry regularly
    jitter_patience = 4000    # steps without improvement before a shake

    # Soft‑max temperature for a smooth sub‑gradient
    soft_temp = 20.0

    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        t += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val / (cur.sum() ** 2))

        # ----- smooth weight vector (soft‑max over convolution) -----
        w_exp = np.exp(soft_temp * (conv - max_val))
        w = w_exp / w_exp.sum()          # sum‑to‑1, non‑negative

        # ----- gradient of the soft‑maxed objective -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]      # sub‑gradient for max‑type objective

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = alpha_lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- propose a new candidate (symmetrised & projected) -----
        cand = cur - step
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        max_cand = conv_cand.max()
        cand_score = float(2.0 * n * max_cand / (cand.sum() ** 2))

        # ----- acceptance -----
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            # occasional probabilistic acceptance to escape plateaus
            prob = np.exp(-(cand_score - cur_score) / (alpha_lr + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score
                no_improve = 0
            else:
                no_improve += 1

        # ----- keep global best -----
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()

        # ----- periodic symmetrisation (helps stability) -----
        if t % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            max_sym = conv_sym.max()
            sym_score = float(2.0 * n * max_sym / (sym.sum() ** 2))
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()

        # ----- random jitter if we are stuck too long -----
        if no_improve > jitter_patience:
            noise = rng.normal(scale=0.006, size=n)
            cand = cur + noise
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)
            cand_fft = np.fft.rfft(cand, n=L)
            conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
            cand_score = float(2.0 * n * conv_cand.max() / (cand.sum() ** 2))
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                no_improve = 0
                if cand_score < best_score:
                    best_score = cand_score
                    best_vec = cand.copy()
            else:
                no_improve = 0   # reset counter even if not accepted

        # ----- early exit if target reached -----
        if best_score < target:
            break

    # --------------------------------------------------------------
    # Return the best vector found (clipped to the allowed range)
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
