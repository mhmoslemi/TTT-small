        a = rng.random(2000, dtype=np.float64) + 1e-3
    # ensure nonneg
    a = np.maximum(a, 0.0)
    # normalize to sum 1
    a = a / np.sum(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    # compute initial
    max_val, grad = _conv_max_and_grad(a, L)
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = a.copy()
    # optimization parameters
    eta0 = 0.5
    eps = 1e-12
    it = 0
    no_improve=0
    target=1.50274349
    while time.time() < deadline:
        it += 1
        # compute gradient
        max_val, grad = _conv_max_and_grad(a, L)
        cur_sc = 2.0*n*max_val
        # adaptive learning rate
        eta = eta0 / math.sqrt(it)
        # exponentiated gradient update
        a = a * np.exp(-eta * grad)
        # renormalize
        s = np.sum(a)
        if s == 0 or not np.isfinite(s):
            # reinitialize to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        # symmetrize occasionally
        if it % 100 == 0:
            a = (a + a[::-1]) * 0.5
            a = a / np.sum(a)
        # evaluate
        max_val2, _ = _conv_max_and_grad(a, L)
        cand_sc = 2.0*n*max_val2
        # keep best
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        # random perturb if stuck
        if no_improve > 2000:
            sigma = 0.02 / math.sqrt(it)
            perturb = a + sigma * rng.standard_normal(size=n)
            perturb = np.maximum(perturb, 0)
            perturb = perturb / np.sum(perturb)

def _conv_max_and_grad(a, L):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]   # this is weighting for gradient? same as previous
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    # index mapping:
    # gradient of convolution at k w.r.t a_i: derivative is ... need mapping as earlier code:
    idx = (2*n - 2) - np.arange(n)  # from original code
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

def _project_simplex(v):
    """ Euclidean projection of a vector onto the probability simplex. """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_and_grad(a, L):
    """
    Compute the maximal value of the full self‑convolution of a
    and a sub‑gradient of that maximal value.
    a must be a 1‑D non‑negative array.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt

    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad
