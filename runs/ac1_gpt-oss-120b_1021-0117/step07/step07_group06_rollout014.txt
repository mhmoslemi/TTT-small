import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[0] = 1.0
    return w

def _score(a):
    """Score for a vector already normalized to sum=1 (2*n*max(conv))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        cur = _project_simplex(a0)
    else:
        n0 = 2000
        cur = _project_simplex(rng.random(n0) + 1e-3)

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5

    it = 0
    no_improve = 0
    target = 1.50274349
    stuck_restart = 5000
    perturb_every = 500
    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # self‑convolution via FFT
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # sub‑gradient for the max (uniform over argmax positions)
        tol = max_val * 1e-12 + 1e-15
        mask = conv >= (max_val - tol)
        cnt = mask.sum()
        if cnt == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = mask.astype(np.float64) / cnt

        # gradient = 2 * (a * rev(w))   – matrix‑free via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)                 # map to size‑n gradient
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # keep best ever
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

        # accept if improves current
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # periodic symmetrisation (helps many instances)
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                no_improve = 0

        # random local perturbation when stuck
        if it % perturb_every == 0 and no_improve > 0:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0

        # restart if no progress for a long time
        if no_improve > stuck_restart:
            cur = _project_simplex(rng.random(n))
            cur_sc = _score(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

        # safeguard against overrunning the deadline
        if time.monotonic() >= deadline:
            break

    # return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
