def _score(vec, n):
    """Exact replica of the evaluator's formula (without its clipping safeguards)."""
    s = np.sum(vec)
    # sum is guaranteed positive by the caller
    conv = np.convolve(vec, vec)
    return 2.0 * n * np.max(conv) / (s * s)

            continue

        # (6) hill‑climbing acceptance
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_vec = cand.copy()
                best_score = cand_score

        # occasional global perturbation to escape plateaus
        if it % 1000 == 0:
            noise = rng.normal(scale=0.002, size=n)
            pert = cur + noise
            pert = np.clip(pert, 0.0, None)
            s = np.sum(pert)
            if s > 0:
                pert = pert / s               # keep the scale
                try:
                    pert_score = _score(pert, n)
                except Exception:
                    continue
                if pert_score < cur_score:
                    cur = pert
                    cur_score = pert_score
                    if pert_score < best_score:
                        best_vec = pert.copy()
                        best_score = pert_score

    # ----------------------------------------------------------------------
    # Final sanity – make sure the sum is at least the evaluator’s lower bound
    if np.sum(best_vec) < 0.01:
        best_vec = best_vec * (0.01 / np.sum(best_vec))

    # Return a plain Python list of floats
    return [float(x) for x in best_vec]
