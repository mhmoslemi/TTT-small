import time, numpy as np, math

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = v - theta
    w[w<0] = 0.0
    return w

def _score_from_max(n, max_val):
    return 2.0 * n * max_val

        # ---------- convolution of extrapolated point ----------
        y_fft = np.fft.rfft(y, n=L)
        conv_full = np.fft.irfft(y_fft * y_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val_y = conv.max()

        # ---------- temperature schedule ----------
        if it == 1:
            temp = max_val_y * 0.08
        else:
            temp = max(temp * temp_decay, max_val_y * min_temp_factor)

        # ---------- smoothed‑max weights ----------
        w = np.exp((conv - max_val_y) / temp)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = (conv == max_val_y).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ---------- gradient of the smoothed max ----------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(y_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---------- mirror‑descent (exponential) update ----------
        # clip gradient to avoid overflow in the exponent
        grad_clipped = np.clip(grad, -1e3, 1e3)
        exp_arg = -step * grad_clipped
        exp_arg = np.clip(exp_arg, -700, 700)   # safe range for np.exp
        new = y * np.exp(exp_arg)

        # renormalise to sum=1 and enforce simplex constraints
        s = new.sum()
        if s <= 0.0 or not np.isfinite(s):
            new = _project_simplex(np.maximum(y - step * grad, 0.0))
        else:
            new = new / s
        new = _project_simplex(new)

        # ---------- evaluate new point ----------
        new_fft = np.fft.rfft(new, n=L)
        conv_full_new = np.fft.irfft(new_fft * new_fft, n=L)
        max_val_new = conv_full_new[: 2 * n - 1].max()
        new_sc = _score_from_max(n, max_val_new)

        # ---------- acceptance ----------
        if new_sc < cur_sc:
            prev, cur = cur, new
            cur_sc = new_sc
            no_improve = 0
            step = min(step * step_inc, 1.0)
            if new_sc < best_sc:
                best_sc = new_sc
                best_vec = new.copy()
        else:
            no_improve += 1
            prob = math.exp(-(new_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                prev, cur = cur, new
                cur_sc = new_sc
            step = max(step * step_dec, 1e-4)

        # ---------- periodic symmetrisation ----------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            conv_full_sym = np.fft.irfft(sym_fft * sym_fft, n=L)
            max_val_sym = conv_full_sym[: 2 * n - 1].max()
            sym_sc = _score_from_max(n, max_val_sym)
            if sym_sc < cur_sc:
                prev, cur = cur, sym
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ---------- random redistribution when stalled ----------
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_fft = np.fft.rfft(tmp, n=L)
            conv_full_tmp = np.fft.irfft(tmp_fft * tmp_fft, n=L)
            max_val_tmp = conv_full_tmp[: 2 * n - 1].max()
            tmp_sc = _score_from_max(n, max_val_tmp)
            if tmp_sc < cur_sc:
                prev, cur = cur, tmp
                cur_sc = tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()

        # ---------- early exit ----------
        if best_sc < target:
            break

    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
