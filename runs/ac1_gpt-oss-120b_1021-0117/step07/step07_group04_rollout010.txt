import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluation of the objective (assumes a>=0 and sum(a)>0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------------
    # length and FFT size
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # deterministic seed candidates
    xs = np.arange(n, dtype=np.float64)
    cand_list = []

    cand_list.append(np.full(n, 1.0 / n))                         # uniform
    tri = np.minimum(xs + 1, n - xs)                            # triangular
    cand_list.append(tri / tri.sum())
    sine = np.sin(np.pi * (xs + 1) / (n + 1))                    # sine bump
    cand_list.append(sine / sine.sum())
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2           # raised cosine
    cand_list.append(bump / bump.sum())
    for alpha in (0.001, 0.003, 0.006):
        exp = np.exp(-alpha * xs)
        cand_list.append(exp / exp.sum())
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        cand_list.append(pw / pw.sum())
    rand = rng.random(n)
    cand_list.append(rand / rand.sum())
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            cand_list.append(parent)

    # --------------------------------------------------------------
    # pick best deterministic start
    best_vec = None
    best_sc = float('inf')
    for vec in cand_list:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------------
    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha0 = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution and active mask -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # ----- gradient (sub‑gradient of max‑convolution) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha0 / math.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - update
        cand = _project_simplex(cand)
        cand_sc = _score_fft(cand, n, L)

        # ----- accept if better -----
        if cand_sc < cur_sc - 1e-12:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            if rng.random() < 0.01:
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # ----- keep global best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            if best_sc < target:
                break

        # ----- random perturbation when stuck -----
        if no_improve > 5000:
            # small Gaussian jitter
            tmp = cur + rng.normal(scale=0.005, size=n)
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
            else:
                # fallback redistribution
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.5
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score_fft(tmp, n, L)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        no_improve = 0
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()

    return [float(x) for x in best_vec]
