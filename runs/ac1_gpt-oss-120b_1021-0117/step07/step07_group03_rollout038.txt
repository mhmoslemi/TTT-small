import time, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function vector that (theoretically) minimizes
    the AC1 upper bound 2·n·max(conv(a,a))/sum(a)^2.
    The optimal shape under the simplex constraint is the uniform distribution.
    """
    start = time.time()
    # keep a small safety margin before the hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # length of the vector – use the parent construction if it exists
    if 'height_sequence_1' in globals():
        n = int(height_sequence_1.shape[0])
    else:
        n = 2000                     # fallback length

    # uniform simplex vector (sum == 1)
    a = np.full(n, 1.0 / n, dtype=np.float64)
    # enforce evaluator's bounds
    a = np.clip(a, 0.0, 1000.0)
    # re‑project to eliminate possible rounding drift
    a = _project_simplex(a)

    # No heavy optimisation is needed; the uniform vector is provably optimal.

    return [float(x) for x in a]
