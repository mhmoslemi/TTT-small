    def score(v):
        v = np.asarray(v, dtype=np.float64)
        # no need to clip because we keep within bounds
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2*n - 1]
        return float(2.0 * n * conv.max())

import time, math, numpy as np
import scipy.optimize
import scipy.sparse as sp
from scipy.fft import rfft, irfft

def _proj_simplex(v):
    """Simple projection onto the probability simplex (sum=1, v>=0)."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s > 0.0:
        v = v / s
    else:
        v = np.full_like(v, 1.0 / v.size)
    return v

def _score_vec(v, L=None):
    """Exact score used by the evaluator (sum is assumed to be 1)."""
    n = v.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = rfft(v, n=L)
    conv = irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    cur = _proj_simplex(a0)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    idx_map = (2 * n - 2) - np.arange(n)               # for gradient indexing

    best = cur.copy()
    best_sc = _score_vec(best, L)

    # ------------------------------------------------------------------
    # optional DPSS warm‑start (positive, nicely tapered)

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
