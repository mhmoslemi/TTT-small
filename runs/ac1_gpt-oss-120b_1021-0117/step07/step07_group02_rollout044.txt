import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score for a vector a assumed to sum to 1 (denominator = 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ---- initialization ----------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
    else:
        # reasonable fallback length
        n_fallback = 2500
        a = rng.random(n_fallback, dtype=np.float64)
    a = _project_simplex(a)                     # enforce sum=1, non‑negativity
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # ---- optimizer state --------------------------------------------------
    m = np.zeros_like(a)        # first moment
    v = np.zeros_like(a)        # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha0 = 0.5                # base step size for Adam
    it = 0
    best_vec = a.copy()
    best_sc = _score_vec(best_vec)
    no_improve = 0
    target = 1.50274349

    # ---- main loop --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # FFT of current iterate
        a_fft = np.fft.rfft(a, n=L)

        # full convolution a*a and its maximum
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val          # score for current a (sum is 1)

        # subgradient weights: uniform over all maximizing lags
        mask = conv == max_val
        cnt = mask.sum()
        if cnt == 0:                         # should not happen, fallback to softmax
            w = np.exp((conv - max_val) / (max_val * 1e-4 + 1e-12))
            w /= w.sum()
        else:
            w = mask.astype(np.float64) / float(cnt)

        # gradient = 2 * (a * w_rev) evaluated via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(a - step)
        cand_sc = _score_vec(cand)

        # keep best ever
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # accept if improves current iterate
        if cand_sc < cur_sc:
            a = cand
        else:
            # occasional exploratory perturbation
            if rng.random() < 0.001:
                i = rng.integers(0, n)
                if a[i] > 0:
                    delta = a[i] * rng.random() * 0.2
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = a.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    a = _project_simplex(tmp)

        # periodic symmetrisation (helps many optimal patterns)
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0
            if sym_sc < cur_sc:
                a = sym

        # random restart when stuck
        if no_improve > 15000:
            a = best_vec + rng.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # return the best-found vector as plain Python floats
    return [float(x) for x in best_vec]
