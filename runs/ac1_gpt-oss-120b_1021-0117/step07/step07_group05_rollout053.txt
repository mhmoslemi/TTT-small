def project_simplex(v):
    # Projects onto the simplex {x >=0, sum=1}
    # reference: http://arxiv.org/abs/1101.6081
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def full_convolution(x):
    n = len(x)
    n2 = 1 << ((2*n - 1).bit_length())
    fx = np.fft.rfft(x, n=n2)
    c = np.fft.irfft(fx * fx, n=n2)[:2*n-1]  # real part truncated
    return c

import time
import numpy as np

# ------------------------------------------------------------
# Helper: convolution (full) using FFT – O(n log n), low memory
def _conv_full(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    c = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return c


# Helper: objective = 2 * n * max(conv) / (sum(a)^2)
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_full(a)
    max_b = conv.max()
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_obj = _objective(best_seq)

    # parent, if present
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.sum() > 0.0:
            parent = parent / parent.sum()          # normalise – scale does not matter
        else:
            parent = uniform.copy()
        parent_obj = _objective(parent)
        if parent_obj < best_obj:
            best_seq = parent.copy()
            best_obj = parent_obj

    # a few random Dirichlet starts for diversity
    rng = np.random.default_rng(seed)
    for _ in range(3):
        if time.monotonic() >= deadline:
            break
        rnd = rng.random(n)
        rnd /= rnd.sum()
        rnd_obj = _objective(rnd)
        if rnd_obj < best_obj:
            best_seq = rnd.copy()
            best_obj = rnd_obj

    # --------------------------------------------------------
    # greedy improvement – move mass from the biggest contribution to a
    # location that does not participate in the current maximal lag.
    cur = best_seq.copy()
    cur_obj = best_obj

    it = 0
    max_iters = 200_000                     # hard cap, real stop is time limit
    while time.monotonic() < deadline and it < max_iters:
        # convolution and location of the current peak
        conv = _conv_full(cur)
        max_b = conv.max()
        k = int(conv.argmax())               # lag where the peak occurs

        # indices that contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1)
        contrib = cur[idxs] * cur[k - idxs]   # element‑wise products

        # source index: the one with the largest product
        imax_local = int(np.argmax(contrib))
        i_src = i_start + imax_local

        # choose a target index that is *outside* the overlapping window
        mask = np.ones(n, dtype=bool)
        mask[i_start:i_end + 1] = False
        outside = np.flatnonzero(mask)
        if outside.size > 0:
            i_tgt = int(rng.choice(outside))
        else:
            # all indices overlap – pick the weakest contributor instead
            imin_local = int(np.argmin(contrib))
            i_tgt = i_start + imin_local
            if i_tgt == i_src:
                i_tgt = int(rng.integers(0, n))
                while i_tgt == i_src:
                    i_tgt = int(rng.integers(0, n))

        # fraction of source mass to move (decreases slowly)
        eps_frac = max(0.01, 0.2 / (1.0 + it / 500.0))
        epsilon = eps_frac * cur[i_src]
        if epsilon <= 0.0:
            break

        # tentative move
        cur[i_src] -= epsilon
        cur[i_tgt] += epsilon

        new_obj = _objective(cur)

        if new_obj < cur_obj:                       # accept improvement
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
        else:
            # revert – the move was not beneficial
            cur[i_src] += epsilon
            cur[i_tgt] -= epsilon

        # occasional re‑normalisation to combat drift
        if (it + 1) % 1000 == 0:
            cur = np.clip(cur, 0.0, None)
            s = cur.sum()
            if s > 0.0:
                cur /= s

        it += 1

    # --------------------------------------------------------
    # final safety checks and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:                       # keep evaluator happy
        best_seq *= 0.01 / max(best_seq.sum(), 1e-12)

    return [float(x) for x in best_seq]
