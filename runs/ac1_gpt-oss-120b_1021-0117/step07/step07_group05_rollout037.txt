def propose_candidate(seed=42, budget_s=1100, **kwargs):

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)
    rng = np.random.default_rng(seed)

    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # Simple uniform initial
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = evaluate_sequence(best_seq.tolist())
    # Optionally parent candidate
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        # normalise? Not necessary, but we can evaluate raw
        parent_score = evaluate_sequence(parent.tolist())
        if parent_score < best_score:
            best_seq = parent
            best_score = parent_score

    # Function to evaluate and update best
    def try_seq(seq):
        nonlocal best_seq, best_score
        try:
            sc = evaluate_sequence(seq.tolist())
        except Exception:
            return
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # Exponential search
    # generate eps values
    eps_vals = np.concatenate([
        np.logspace(-4, -2, 120),   # 0.0001 to 0.01
        np.logspace(-2, -1, 80)    # 0.01 to 0.1
    ])
    # maybe include some small eps near zero? Already includes 1e-4 which is enough.

    for eps in eps_vals:
        if time.monotonic() >= deadline:
            break
        if eps <= 0 or eps >= 1:
            continue
        r = 1.0 - eps
        # compute seq
        i = np.arange(n, dtype=np.float64)
        # use log(r) for exponent
        seq = np.exp(i * np.log(r))
        total = seq.sum()
        if total == 0:
            continue
        seq /= total
        # Clip not needed but for safety
        seq = np.clip(seq, 0.0, 1000.0)
        try_seq(seq)

    # Local refinement around best epsilon using finer grid if time allows
    # Determine epsilon from best_seq if it is exponential pattern may not be exact.
    # We can approximate best epsilon from ratio of successive elements: r_est = best_seq[1]/best_seq[0] if best_seq[0] != 0.
    # Then do fine search around r_est.
    if time.monotonic() < deadline:
        # approximate r_est
        if best_seq[0] > 0:
            r_est = best_seq[1] / best_seq[0] if best_seq[1] > 0 else 0.999
            eps_est = 1.0 - r_est
        else:
            eps_est = 0.001
        # fine grid around eps_est +/- factor
        for factor in [0.5, 0.8, 0.9, 1.0, 1.1, 1.2, 1.5]:

            if eps <= 0 or eps >= 1:

            r = 1.0 - eps
            i = np.arange(n, dtype=np.float64)
            seq = np.exp(i * np.log(r))
            seq /= seq.sum()

    # Simple stochastic hill climbing from best_seq
    # Use projection onto simplex (sum=1) to maintain sum
    def project_simplex(v):
        # implementation from earlier code (efficient)
        v = np.asarray(v, dtype=np.float64)
        nlen = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho_arr = np.where(u - css / np.arange(1, nlen+1) > 0)[0]
        if rho_arr.size == 0:
            theta = 0.0
        else:
            rho = rho_arr[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    cur = best_seq.copy()

    it = 0

        if tot == 0:

            new_vals = rand * tot

import numpy as np
import time

def _evaluate(seq):
    """Safely evaluate a sequence with the provided evaluator."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return float('inf')


def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _exponential_seq(eps, n):
    """Return a non‑negative step‑function of length n with exponential decay  r=1‑eps, normalised to sum = 1."""
    if eps <= 0.0 or eps >= 1.0:
        return None
    r = 1.0 - eps
    # use log for numerical stability when r is close to 1
    log_r = np.log(r)
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(i * log_r)
    total = seq.sum()
    if total == 0.0 or not np.isfinite(total):
        return None
    seq /= total
    return seq
