import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score_fast(a, L):
    """Fast evaluation of 2*n*max(conv) with a normalized to sum=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _max_and_mask(a, L):
    """Return max value of the full autocorrelation and a 0/1 mask of
    positions attaining the max (with a tiny tolerance)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    tol = max(1e-12, max_val * 1e-12)
    mask = (conv >= max_val - tol).astype(np.float64)
    return max_val, mask

def _grad_from_mask(a, mask, n, L):
    """Exact sub‑gradient of max(conv) w.r.t. a using the mask of maximal
    positions. Implements 2*conv_rev[n-1:n-1+n] where conv_rev is a_rev * mask."""
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=L)
    mask_fft = np.fft.rfft(mask, n=L)
    conv_rev = np.fft.irfft(a_rev_fft * mask_fft, n=L)
    start = n - 1
    return 2.0 * conv_rev[start:start + n]
