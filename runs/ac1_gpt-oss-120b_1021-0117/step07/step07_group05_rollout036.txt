Implementation of simplex projection standard:

```
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv_and_index(a: np.ndarray) -> tuple[float, int]:
    """Return (max value, index) of the full convolution a*a."""
    conv = np.convolve(a, a)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound measured by ``evaluate_sequence``.
    The routine works with any existing ``height_sequence_1``; otherwise it
    starts from a uniform distribution and improves it by a projected
    sub‑gradient descent followed by a simple stochastic local search.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ------------------------------------------------------------------
    # Initialise a probability vector (sum == 1)
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.shape[0]
    a = _project_simplex(a)          # enforce sum == 1, non‑negativity

    # ------------------------------------------------------------------
    # Keep the best vector seen so far
    max_val, _ = _max_conv_and_index(a)
    best_score = 2.0 * n * max_val    # same as evaluate_sequence for sum==1
    best_vec = a.copy()

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent on the max‑convolution
    max_iters = 15000
    base_step = 8.0                   # large initial step, will decay
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        # current max‑convolution and its index
        max_val, idx = _max_conv_and_index(a)

        # sub‑gradient of the active entry (2 * a[idx‑j] on its support)
        start_j = max(0, idx - n + 1)
        end_j = min(n - 1, idx)
        g = np.zeros_like(a)
        js = np.arange(start_j, end_j + 1, dtype=np.int64)
        g[js] = 2.0 * a[idx - js]

        # diminishing step size
        step = base_step * (decay ** it)

        # try a few scaled steps and keep the best improvement
        best_local_score = 2.0 * n * max_val
        best_local_a = a
        for factor in (0.5, 1.0, 2.0):
            cand = a - step * factor * g
            cand = _project_simplex(cand)
            cand_max, _ = _max_conv_and_index(cand)
            cand_score = 2.0 * n * cand_max
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local_a = cand

        # record globally best
        if best_local_score < best_score:
            best_score = best_local_score
            best_vec = best_local_a.copy()

        # accept the local improvement (or stay put)
        if best_local_score <= 2.0 * n * max_val:
            a = best_local_a

        # occasional random jitter to avoid stagnation
        if it % 500 == 0 and rng.random() < 0.12:
            jitter = rng.normal(scale=0.02, size=n)
            a = _project_simplex(np.clip(a + jitter, 0.0, 1000.0))

    # ------------------------------------------------------------------
    # Phase 2 – simple stochastic pairwise mass transfers
    cur = best_vec.copy()
    cur_score = best_score
    max_iters2 = 6000
    it2 = 0
    while time.monotonic() < deadline and it2 < max_iters2:
        it2 += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random fraction (up to 30 %) of mass from i to j
        delta = cur[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)

        cand_max, _ = _max_conv_and_index(cand)
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand

    # ------------------------------------------------------------------
    # Return the best candidate found as a plain Python list of floats.
    return [float(x) for x in best_vec]
