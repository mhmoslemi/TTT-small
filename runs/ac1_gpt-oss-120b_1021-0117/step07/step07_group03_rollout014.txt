    start = time.time()
    deadline = start + max(0.0, budget_s - 0.2)  # 0.2 safety

    # init best via given height_sequence_1 or random
    if "height_sequence_1" in globals():

    else:
        a0 = rng.random(2000, dtype=np.float64)

    a0 = a0 / a0.sum()
    n = a0.size
    L = 1 << ((2*n - 1).bit_length())
    cur = a0.copy()
    best = cur.copy()
    max_val, _ = _conv_max_and_grad(cur, L)
    best_sc = 2.0 * n * max_val
    cur_sc = best_sc

    # optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_max = 0.5
    step_min = 0.05
    period = 200
    it = 0
    no_improve = 0

    target = 1.50274349

    while time.time() < deadline:
        it += 1
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)

        # cyclic learning rate
        cycle = (it % period) / period
        step = step_min + (step_max - step_min) * 0.5 * (1 + math.cos(math.pi * cycle))

        # propose new point

    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
