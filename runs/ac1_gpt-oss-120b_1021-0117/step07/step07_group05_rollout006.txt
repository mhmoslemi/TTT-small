Pseudo:

def project_simplex(v):
    """Project onto simplex {x >=0, sum x = 1}"""
    v = np.asarray(v, dtype=np.float64)
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np

def _project_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex:
    {x >= 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
