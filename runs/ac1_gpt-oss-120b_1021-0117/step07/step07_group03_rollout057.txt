import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex (non‑negative, sum=1). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L):
    """
    Returns (max_convolution, subgradient) of max(a * a) (full convolution).
    a is assumed non‑negative; the function works for any a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad


def _optimize_sequence(x0, n, L, deadline, target, rng,
                       max_iter=5000, sym_interval=200):
    """
    Adam‑style projected optimisation from a given start point.
    Returns the best vector found and its score.
    """
    # initialise
    x = _project_simplex(x0.copy())
    max_val, _ = _conv_max_and_grad(x, L)
    cur_score = 2.0 * n * max_val
    best_score = cur_score
    best_vec = x.copy()

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        max_val, grad = _conv_max_and_grad(x, L)
        cur_score = 2.0 * n * max_val

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        step = lr / (np.sqrt(v_hat) + eps)

        x_new = _project_simplex(x - step * m_hat)

        # occasional symmetrisation (helps flatten the convolution)
        if sym_interval and (it % sym_interval == 0):
            x_new = (x_new + x_new[::-1]) * 0.5
            x_new = _project_simplex(x_new)

        max_val_new, _ = _conv_max_and_grad(x_new, L)
        new_score = 2.0 * n * max_val_new

        x = x_new  # always move (even if not improving)

        if new_score < best_score:
            best_score = new_score
            best_vec = x.copy()
            if best_score < target:
                break

    return best_vec, best_score
