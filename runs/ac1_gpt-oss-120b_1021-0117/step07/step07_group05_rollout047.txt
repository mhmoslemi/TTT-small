Projection algorithm (efficient O(n log n)):

def project_simplex(v):
    """Project onto the probability simplex: nonnegative, sum=1."""
    v = np.asarray(v, dtype=np.float64)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    # Sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return a non‑negative vector with the same L1 norm (=1) as v."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s

def _conv_max_fft(a: np.ndarray) -> tuple[float, int]:
    """Maximum value and index of the full convolution a*a using FFT."""
    n = a.shape[0]
    fsize = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, fsize)
    c = np.fft.irfft(fa * fa, fsize)[: 2 * n - 1]   # linear convolution
    max_val = float(np.max(c))
    max_idx = int(np.argmax(c))
    return max_val, max_idx

def _objective(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) (sum(a) is 1 by construction)."""
    n = a.shape[0]
    max_val, _ = _conv_max_fft(a)
    return 2.0 * n * max_val

def _subgradient(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of max(a*a) at the index ``idx``.
    For j in the active window derivative = 2·a[idx‑j], else 0.
    """
    n = a.shape[0]
    low = max(0, idx - (n - 1))
    high = min(idx, n - 1)
    grad = np.zeros_like(a)
    if low <= high:
        rng = np.arange(low, high + 1, dtype=int)
        grad[rng] = 2.0 * a[idx - rng]
    return grad
