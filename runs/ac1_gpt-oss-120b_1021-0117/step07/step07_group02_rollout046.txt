import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a <= 0:
        return float('inf')
    return float(2.0 * n * max_val / (sum_a * sum_a))

def _project_simplex(v):
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score_vec(a):
    """Exact AC1 upper‑bound score for a given coefficient vector."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a <= 0.0:
        return float('inf')
    return float(2.0 * n * max_val / (sum_a * sum_a))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initial vector -------------------------------------------------
    if 'height_sequence_1' in globals():

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # ---- Adam optimizer state -------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.25                      # base learning rate

    # ---- soft‑max temperature for gradient approximation ---------------
    temp_factor = 1e-2
    temp_decay = 0.9996
    min_temp = 1e-6

    # ---- best‑so‑far tracking -------------------------------------------
    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)

    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # full convolution a*a and its maximum
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights over all lags
        temp = max_val * max(temp_factor, min_temp)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient of weighted sum of convolutions
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
        cand_sc = _score_vec(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            no_improve = 0
        else:
            # occasional stochastic move to keep exploration alive
            if rng.random() < 0.005:
                cur = cand
            no_improve += 1

        # symmetrisation (helps for many optimal patterns)
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                cur = sym
                no_improve = 0

        # random local redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    cur = tmp
                    no_improve = 0
            else:
                if rng.random() < 0.02:          # rare full restart
                    cur = _project_simplex(rng.random(n))
                    m.fill(0.0)
                    v.fill(0.0)
                    no_improve = 0

        # temperature decay
        temp_factor = max(temp_factor * temp_decay, min_temp)

        # early termination if target achieved
        if best_sc < target:
            break

        # periodic time check
        if it % 4000 == 0 and time.monotonic() >= deadline:
            break

    return [float(x) for x in best_vec]
