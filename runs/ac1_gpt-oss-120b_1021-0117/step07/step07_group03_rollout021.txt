def _smooth_max_and_grad(a, L, T):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2*n - 1]
    # compute softmax weights
    if T <= 0:
        # fallback to hard max
        w = np.zeros_like(conv)
        max_val = conv.max()
        mask = (conv == max_val)
        w[mask] = 1.0 / mask.sum()
        smooth = max_val
    else:
        max_conv = conv.max()
        # subtract max for numeric stability
        exp_vals = np.exp((conv - max_conv) / T)
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp
        smooth = np.dot(w, conv)  # weighted sum approximates max
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return smooth, grad

import time, math, numpy as np

# --------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L):
    """Hard‑max version used only for a quick initial temperature estimate."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    mask = (conv == max_val).astype(np.float64)
    mask /= mask.sum()
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad


def _smooth_max_and_grad(a, L, T):
    """Smooth approximation of max via softmax with temperature T."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    if T <= 0:
        # fallback to exact max
        max_val = conv.max()
        mask = (conv == max_val).astype(np.float64)
        mask /= mask.sum()
        w = mask
        smooth = max_val
    else:
        max_val = conv.max()
        # stability: subtract max before exponentiation
        shifted = (conv - max_val) / T
        exp_vals = np.exp(shifted)          # all ≤ 1
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp
        smooth = float(np.dot(w, conv))

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return smooth, grad
