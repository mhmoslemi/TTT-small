def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # efficient O(n log n) projection
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)

    # correct possible rounding errors
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:

        w = np.maximum(w, 0.0)
