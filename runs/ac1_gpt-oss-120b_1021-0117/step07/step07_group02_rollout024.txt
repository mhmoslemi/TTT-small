    a = _project_simplex(a)

        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n - 1]
        return float(2.0 * n * conv.max())
    
    best_a = a.copy()
    best_score = score(best_a)
    # optimization parameters
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 1.0
    it = 0

    # Main loop
    while time.monotonic() < deadline:
        it += 1

        af = np.fft.rfft(a, n=L)

        # step size
        step = step0 / math.sqrt(it)
        cand = _project_simplex(a - step * momentum)
        cand_score = score(cand)
        
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # acceptance criteria (simulated annealing)
        # If cand better, accept; else accept with prob
        if cand_score < score(a) or rng.random() < math.exp(-(cand_score - score(a)) / (step + 1e-12)):

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a, L=None):
    """Score assuming sum(a)==1 (denominator becomes 1)."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    a = _project_simplex(a)                       # sum == 1
    n = a.shape[0]

    # optimizer state (Adam)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 1.0
    it = 0

    # temperature for soft‑max weighting of the max‑convolution
    temp_factor = 1e-2          # relative to current max value
    temp_decay = 0.9995
    min_temp_factor = 1e-6

    cur_score = _score_vec(a, L)
    best_a = a.copy()
    best_score = cur_score
    no_improve = 0
    target = 1.50274349

    # ----- main loop ------------------------------------------------------
    while time.monotonic() < deadline:

        # forward FFT and full convolution
        a_fft = np.fft.rfft(a, n=L)

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weighting over all lags

        if w_sum > 0:

        w_rev = w[::-1]

    return [float(x) for x in best_a]
