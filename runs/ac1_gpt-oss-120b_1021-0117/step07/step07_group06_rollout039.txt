def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init vector
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64) + 1e-3

    cur = np.maximum(cur, 0.0)
    # normalize to simplex
    cur = cur / cur.sum()

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    def project(v):
        # Euclidean projection onto simplex
        v = np.maximum(v, 0.0)
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
        if rho.size > 0:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        w = np.maximum(v - theta, 0.0)
        # correct rounding
        s = w.sum()
        if s > 0:
            w /= s
        return w

    def score(a):
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    best = cur.copy()
    best_sc = score(best)
    cur_sc = best_sc

    m = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.6

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # mask of near-maximum entries
        eps = max_val * 1e-9 + 1e-12
        mask = conv >= max_val - eps
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt == 0:
            # fallback to single max index
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            cnt = 1
        w /= cnt

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        m = beta * m + (1 - beta) * grad

        # step size
        step = step0 / math.sqrt(it)

        cand = project(cur - step * m)
        cand_sc = score(cand)

        # accept if improvement or with small prob
        if cand_sc < cur_sc or rng.random() < math.exp((cur_sc - cand_sc) / (step + 1e-12)):
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc

        # periodic symmetrization
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = project(sym)
            sym_sc = score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # random mass shift when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:

                tmp = project(tmp)
                tmp_sc = score(tmp)
                if tmp_sc < cur_sc:

                    no_improve = 0
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc

        # early exit
        if best_sc < target:
            break

    # return best as list of floats
    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    v = np.maximum(v, 0.0)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def _score(a):
    """Score for a vector (sum assumed = 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialization

        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64) + 1e-3

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.6
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        a_fft = np.fft.rfft(cur, n=L)

        mask = conv >= max_val - eps
        w = mask.astype(np.float64)

        w /= cnt

        w_rev = w[::-1]
