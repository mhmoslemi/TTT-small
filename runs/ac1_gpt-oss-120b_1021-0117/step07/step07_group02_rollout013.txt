def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.1) # leave margin

    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = int(2500)
        a0 = rng.random(n)
    # Ensure nonnegative, project onto simplex
    cur = _project_simplex(np.maximum(a0, 0.0))

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # function to compute max conv
    def _max_conv(x):
        a_fft = np.fft.rfft(x, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        return conv_full.max(), conv_full

    max_val, conv_full = _max_conv(cur)
    cur_score = float(2.0 * n * max_val)

    best_vec = cur.copy()
    best_score = cur_score

    # parameters
    step0 = 1.0
    temp_factor = 1e-2
    temp_decay = 0.9998
    min_temp_factor = 1e-8
    target_score = 1.50274349 # we want lower

    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # compute conv and max
        max_val, conv_full = _max_conv(cur)
        cur_score = float(2.0 * n * max_val)

        # temperature
        temp = max_val * max(temp_factor, min_temp_factor)

        # choose weighting
        # use softmax while temp_factor > 1e-4 else exact mask
        if temp_factor > 1e-4:
            w = np.exp((conv_full - max_val) / temp)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
        else:
            # mask near max
            tol = 1e-12 * max(1.0, max_val)  # absolute tolerance
            mask = conv_full >= max_val - tol
            mask_count = np.count_nonzero(mask)
            if mask_count == 0:
                w = np.zeros_like(conv_full)
                w[conv_full.argmax()] = 1.0
            else:
                w = np.zeros_like(conv_full)
                w[mask] = 1.0 / mask_count

        # compute gradient
        w_rev = w[::-1]
        a_fft = np.fft.rfft(cur, n=L)
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # step size
        step = step0 / math.sqrt(it)

        # candidate
        cand = _project_simplex(cur - step * grad)

        # evaluate candidate
        cand_max, _ = _max_conv(cand)
        cand_score = float(2.0 * n * cand_max)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:
            no_improve += 1

        # occasional random perturb
        if no_improve > 3000:
            # shift small amount
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_max,_ = _max_conv(tmp)
                tmp_score = float(2.0 * n * tmp_max)
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_vec = tmp.copy()
                else:
                    # maybe random restart if still no improvement
                    if rng.random() < 0.01:
                        cur = _project_simplex(rng.random(n))
                        cur_score = float(2.0 * n * _max_conv(cur)[0])
                        no_improve = 0
            else:
                cur = _project_simplex(rng.random(n))
                cur_score = float(2.0 * n * _max_conv(cur)[0])
                no_improve = 0

        # decay temperature
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # early stop
        if best_score < target_score:
            break

        # Periodic symmetrization
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_max,_ = _max_conv(sym)
            sym_score = float(2.0 * n * sym_max)

                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()

        # check time
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break

    # Return best solution as Python floats
    return [float(x) for x in best_vec]

import time, math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w
