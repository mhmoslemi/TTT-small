import time, math, numpy as np

def _project_simplex(v):
    """Projection onto the probability simplex (sum=1, nonnegative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # shift by max for numerical stability
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # due to rounding errors, renormalize
    if w.sum() > 0:
        w = w / w.sum()
    return w

def _conv_max_and_grad(a, L):
    """Return max of a*a convolution and its (sub)gradient."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def _adam_optimize(init, L, deadline, target, best_score, best_vec, rng):
    """Run an Adam‑like optimization from `init` until deadline."""
    a = init.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2          # base learning rate
    max_step = 0.5
    it = 0
    no_improve = 0

    while True:
        if time.time() >= deadline:
            break
        it += 1

        max_val, grad = _conv_max_and_grad(a, L)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        step = step / np.sqrt(it)            # extra decay
        step = np.clip(step, -max_step, max_step)

        a_new = _project_simplex(a - step)

        # occasional symmetrisation
        if it % 200 == 0:
            a_new = (a_new + a_new[::-1]) * 0.5
            a_new = _project_simplex(a_new)

        a = a_new

        # random perturbation if stuck
        if no_improve >= 3000:
            sigma = 0.05 / np.sqrt(it)
            pert = a + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            max_val_pert, _ = _conv_max_and_grad(pert, L)
            pert_score = 2.0 * n * max_val_pert
            if pert_score < cur_score:
                a = pert
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
            else:
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        if best_score < target:
            break

    return best_score, best_vec

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    start = time.time()
    # keep a small safety margin so we can return before forced kill

    target = 1.50274349

    # initial vector
    if "height_sequence_1" in globals():

    else:
        # generate a random vector of length ~2500
        length = rng.integers(2000, 3000)

    init = _project_simplex(np.maximum(init, 0.0))

    n = init.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best_score = float('inf')
    best_vec = None

    # allocate most of the time to a run that starts from the known construction
    # (if it exists this is already a good starting point)
    primary_deadline = start + 0.75 * (deadline - start)
    best_score, best_vec = _adam_optimize(init, L, primary_deadline,
                                          target, best_score, best_vec, rng)

    # If time remains, launch a few short random restarts

        if best_score < target:

    if best_vec is None:
        best_vec = init

    # Return as plain Python list
    return [float(x) for x in best_vec]

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _conv_max_and_grad(a, L):
    """Return max of a*a convolution and its sub‑gradient."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def _adam_optimize(init, L, deadline, target, best_score, best_vec, rng):
    """Adam‑style optimization starting from `init` until `deadline`."""
    a = init.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.2          # base learning rate
    max_step = 0.5
    it = 0
    no_improve = 0

    while True:
        if time.time() >= deadline:
            break
        it += 1

        max_val, grad = _conv_max_and_grad(a, L)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        step = step / np.sqrt(it)          # extra decay
        step = np.clip(step, -max_step, max_step)

        # candidate based on the raw step
        cand = _project_simplex(a - step)

        # occasional line‑search over scaled steps
        if it % 50 == 0:
            best_local = cand
            best_local_score = 2.0 * n * _conv_max_and_grad(cand, L)[0]
            for scale in (0.5, 2.0):
                step_s = np.clip(step * scale, -max_step, max_step)
                cand_s = _project_simplex(a - step_s)
                sc = 2.0 * n * _conv_max_and_grad(cand_s, L)[0]
                if sc < best_local_score:
                    best_local = cand_s
                    best_local_score = sc
            a = best_local
        else:
            a = cand

        # occasional symmetrisation (helps to lower peaks)
        if it % 200 == 0:
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)

        # random perturbation if stagnating
        if no_improve >= 3000:
            sigma = 0.05 / np.sqrt(it)
            pert = a + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert_score = 2.0 * n * _conv_max_and_grad(pert, L)[0]
            if pert_score < cur_score:
                a = pert
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
            else:
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        if best_score < target:
            break

    return best_score, best_vec

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin
    target = 1.50274349

    # ------------------------------------------------------------------
    # initialise – use the provided construction if available
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        length = rng.integers(2000, 3000)
        init = rng.random(length, dtype=np.float64)

    init = _project_simplex(np.maximum(init, 0.0))

    n = init.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best_score = float('inf')
    best_vec = None

    # primary run (most of the budget) starting from the known construction
    primary_deadline = start + 0.75 * (deadline - start)
    best_score, best_vec = _adam_optimize(
        init, L, primary_deadline, target, best_score, best_vec, rng
    )

    # secondary runs – short random restarts with the remaining time
    while True:
        now = time.time()
        if now >= deadline:
            break
        remaining = deadline - now
        if remaining < 5.0:
            break
        # allocate at most 15 s or half the remaining budget for a restart
        sub_deadline = now + min(15.0, remaining * 0.5)
        rnd = rng.random(n, dtype=np.float64)
        rnd = _project_simplex(rnd)
        best_score, best_vec = _adam_optimize(
            rnd, L, sub_deadline, target, best_score, best_vec, rng
        )
        if best_score < target:
            break

    if best_vec is None:
        best_vec = init

    # return plain Python floats
    return [float(x) for x in best_vec]
