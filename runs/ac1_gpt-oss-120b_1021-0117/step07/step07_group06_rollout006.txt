def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin small

    # Load initial vector
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # For fallback: random init length ~ 2000
        n = 2000
        a0 = rng.random(n, dtype=np.float64) + 1e-3

    # Normalize to simplex (sum=1)
    cur = _project_simplex(np.maximum(a0, 0.0))

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # optimization state
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    it = 0
    no_improve = 0

    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # FFT of current
        a_fft = np.fft.rfft(cur, n=L)
        # auto-correlation via convolution (full)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # gradient: subgradient of max conv
        rev_a = cur[::-1]
        offset = n - 1 - max_idx
        grad = np.zeros_like(cur)
        if offset >= 0:
            # first part
            length = n - offset
            grad[:length] = 2.0 * rev_a[offset:]
        else:
            # offset negative => shift left
            off = -offset
            length = n - off
            grad[off:] = 2.0 * rev_a[:length]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        step = step0 / math.sqrt(it)

        # candidate vector
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score(cand)

        # accept if better
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:

            # occasional probabilistic acceptance (small temperature)
            temp = step * 0.5 + 1e-12
            prob = math.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # symmetrization periodically

            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score assuming a sums to 1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_vec = cur.copy()
