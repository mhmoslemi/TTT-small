import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _softmax(theta):
    """Stable softmax returning a probability vector (sum=1, all >0)."""
    mx = np.max(theta)
    e = np.exp(theta - mx)
    s = e.sum()
    return e / s

            m.fill(0.0)
            v.fill(0.0)
            pow_beta1, pow_beta2 = 1.0, 1.0

        # ---- FFT based convolution ----------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        # ---- smooth max (log‑sum‑exp) ------------------------------------
        # increase smoothing gradually
        if beta_smooth < beta_max and t % 1000 == 0:
            beta_smooth = min(beta_max, beta_smooth + beta_inc)

        scaled = beta_smooth * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)   # stabilize
        w = exp_scaled / exp_scaled.sum()          # weights over conv entries

        # ---- gradient w.r.t a (using convolution with reversed weights) --
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad_a = 2.0 * grad_full[idx]               # size n

        # ---- chain rule to obtain gradient w.r.t theta --------------------
        dot = np.dot(a, grad_a)
        grad_theta = a * (grad_a - dot)

        # ---- Adam update --------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)
        pow_beta1 *= beta1
        pow_beta2 *= beta2
        m_hat = m / (1.0 - pow_beta1)
        v_hat = v / (1.0 - pow_beta2)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        theta -= step

        # ---- occasional random perturbation if stuck ----------------------
        if no_improve > 3000:
            sigma = 0.1 / math.sqrt(t)
            theta += sigma * rng.standard_normal(size=n)
            no_improve = 0

        # ---- evaluate true bound periodically -----------------------------
        if t % eval_interval == 0:
            cur_sc = 2.0 * n * conv.max()
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if best_sc < target:
                break

    # final fallback: evaluate best vector (ensure finite)
    if best_sc == np.inf:
        best_vec = _project_simplex(np.maximum(cur, 0.0))
    return [float(x) for x in best_vec]
