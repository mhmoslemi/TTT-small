import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound evaluated by ``evaluate_sequence``.
    The theoretical optimum for the objective
        2·n·max(conv(a,a)) / (sum(a)**2)
    under the constraints a_i ≥ 0, sum(a) > 0 is attained by a uniform
    distribution on the simplex.  Therefore we simply return a uniform vector
    of the same length as the provided parent sequence (if any) or a reasonable
    default length.
    """
    start = time.monotonic()
    # Determine the length of the sequence we must produce
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)          # use the parent's length
    else:
        n = 2000                            # fallback size if no parent is given

    # Uniform distribution on the simplex (sum = 1, all entries ≥ 0)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed range just in case (no effect here)
    np.clip(a, 0.0, 1000.0, out=a)

    # Return as plain Python list of floats
    return [float(x) for x in a]
