import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a, L, n):
    """Score for a vector already normalized to sum=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a tiny safety margin for the harness timeout
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # initial sequence
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)          # sum = 1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    best = a.copy()
    best_sc = _score(best, L, n)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 1.0          # larger initial step
    it = 0
    no_improve = 0
    target = 1.50274349
    sym_interval = 250   # how often to symmetrize
    # exponentiation factor for extra candidate (pushes mass to the front)
    pow_factor = 1.12

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ------------------------------------------------------------------
        # convolution and gradient of the max‑convolution
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # mask of (almost) maximal entries
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[:] = 0.0

        # sub‑gradient of max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ------------------------------------------------------------------
        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand_add = _project_simplex(a - update)

        # multiplicative (exponential) update
        exp_factor = np.exp(-step * grad)
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()

        # power‑law candidate (pushes mass to larger entries)
        cand_pow = np.power(a, pow_factor, out=np.empty_like(a))
        cand_pow = _project_simplex(cand_pow)

        # evaluate candidates
        cand_add_sc = _score(cand_add, L, n)
        cand_mul_sc = _score(cand_mul, L, n)
        cand_pow_sc = _score(cand_pow, L, n)

        # choose the best among the three
        cand, cand_sc = min(
            [(cand_add, cand_add_sc), (cand_mul, cand_mul_sc), (cand_pow, cand_pow_sc)],
            key=lambda kv: kv[1],
        )

        # ------------------------------------------------------------------
        # acceptance logic
        cur_sc = _score(a, L, n)
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < cur_sc:
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic acceptance to escape flat regions
            prob = math.exp(-(cand_sc - best_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # ------------------------------------------------------------------
        # periodic symmetrisation (often beneficial for the autocorrelation)
        if it % sym_interval == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L, n)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                no_improve = 0

        # ------------------------------------------------------------------
        # random redistribution when stuck
        if no_improve > 1800:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
            no_improve = 0

        # ------------------------------------------------------------------
        # hard restart from a mixed random state if still no improvement
        if no_improve > 5000:
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            no_improve = 0

        # ------------------------------------------------------------------
        # early exit if we have already beaten the target
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # Return the best vector found (as a plain Python list of floats)
    return [float(x) for x in best]
