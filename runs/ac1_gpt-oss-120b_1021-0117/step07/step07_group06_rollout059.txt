import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(v, L=None):
    """Return 2·n·max(conv(v,v)) using FFT (no explicit convolution)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
