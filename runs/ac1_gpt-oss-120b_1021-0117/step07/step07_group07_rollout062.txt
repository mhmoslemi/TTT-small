import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - css / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a, L):
    """Return max of full self‑convolution of a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return conv.max(), conv

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑2 FFT size
