import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny rounding loss
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """2·n·max convolution (a is assumed to sum to 1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # exponential decay from the left
    exp_left = np.exp(-0.001 * np.arange(n, dtype=np.float64))
    _try_candidate(_project_simplex(exp_left))

    # exponential decay from the right
    exp_right = np.exp(-0.001 * (n - 1 - np.arange(n, dtype=np.float64)))
    _try_candidate(_project_simplex(exp_right))

    # sin‑shaped bump
    sin = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 1) / (n + 1))
    _try_candidate(_project_simplex(sin))

    # ------------------------------------------------------------------
    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
