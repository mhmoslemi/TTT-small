def project_simplex(v):
    v = np.asarray(v)

import time, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum == 1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that seeks to
    minimise the AC1 upper bound measured by ``evaluate_sequence``.
    The routine works with any pre‑existing ``height_sequence_1``; otherwise
    it starts from a uniform distribution.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ----------------------------------------------------------------------
    # Determine problem size and initialise a feasible point (sum == 1)
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a = _project_simplex(a)
    else:
        n = 2000                     # fallback size
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform, already on simplex

    # ----------------------------------------------------------------------
    # Evaluate the initial candidate
    try:
        best_score = evaluate_sequence(a.tolist())
    except Exception:
        best_score = float("inf")
    best_vec = a.copy()
    cur = a.copy()
    cur_score = best_score

    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Parameters for the projected gradient descent
    step = 0.5               # initial step size
    min_step = 1e-6
    max_step = 5.0
    decay = 0.5
    grow = 1.05
    max_iters = 20000

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # convolution and its maximum
        conv = np.convolve(cur, cur)
        M = conv.max()
        idx = int(conv.argmax())

        # build the term a_{idx - j} for each j (zero outside the support)
        j = np.arange(n)
        idx_minus = idx - j
        term = np.zeros_like(cur)
        valid = (idx_minus >= 0) & (idx_minus < n)
        term[valid] = cur[idx_minus[valid]]

        # descent direction: increase where term < M, decrease where term > M
        direction = M - term

        # tentative update and projection back onto the simplex
        cand = cur + step * direction
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand)

        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            # evaluation failure -> shrink step and continue
            step = max(step * decay, min_step)
            continue

        # accept if improvement, otherwise shrink step
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            step = min(step * grow, max_step)
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:
            step = max(step * decay, min_step)

        # occasional random perturbation to escape shallow minima
        if it % 500 == 0:
            i, j_idx = rng.integers(0, n, size=2)
            if i != j_idx:
                delta = cur[i] * rng.random() * 0.1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j_idx] += delta
                tmp = np.clip(tmp, 0.0, 1000.0)
                tmp = _project_simplex(tmp)
                try:
                    tmp_score = evaluate_sequence(tmp.tolist())
                except Exception:
                    tmp_score = float("inf")
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                    step = min(step * grow, max_step)
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_vec = tmp.copy()

    # ----------------------------------------------------------------------
    # Return the best vector discovered as a plain Python list
    return [float(x) for x in best_vec]
