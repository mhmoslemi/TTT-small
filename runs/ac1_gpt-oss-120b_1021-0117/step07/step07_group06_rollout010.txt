import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score for a vector already normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    deadline = start + max(0.0, float(budget_s) - 0.5)

    init_vectors = []

        a0 = np.asarray(height_sequence_1, dtype=np.float64)

        a0 = _project_simplex(a0)
        init_vectors.append(a0)

    # length to use for further initializations
    if init_vectors:
        n = init_vectors[0].size
    else:
        n = 2000

    x = np.arange(n, dtype=np.float64)

    init_vectors.append(_project_simplex(cos_vec))

    # half‑sine start

    # ------------------------------------------------------------------
    best_vec = None
