import numpy as np

def _initial_sequence():
    """Return a normalized non‑negative sequence of the appropriate length."""
    if "height_sequence_1" in globals():
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = None
    else:
        a = None
    if a is None:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)
    return a

def _score_from_seq(a):
    """Fast score: 2*n*max(conv)/sum(a)^2 (sum is assumed >0)."""
    n = a.shape[0]
    max_c = np.max(np.convolve(a, a))
    return 2.0 * n * max_c / (np.sum(a) ** 2)

def _move_towards_flat(a, rng, step_factor=0.5):
    """
    Perform one directed move that reduces the maximum convolution entry.
    Returns a new array and the new score, or (None, None) if no move is possible.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    k0 = int(conv.argmax())

    # valid indices that actually participate in the max‑lag sum
    i_start = max(0, k0 - (n - 1))
    i_end = min(n - 1, k0)
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)

    if idx.size == 0:
        return None, None

    partner = a[k0 - idx]                     # a_{k0-i}
    product = a[idx] * partner                # contribution of each pair

    # pick source index with largest contribution
    src_rel = int(np.argmax(product))
    i_src = int(idx[src_rel])
    i_dst = int(k0 - i_src)                  # its partner index

    # if partner coincides (i_src == i_dst) pick the next best
    if i_dst == i_src:
        if product.size < 2:
            return None, None
        # second best
        sorted_rel = np.argsort(-product)    # descending
        for rel in sorted_rel[1:]:
            cand = int(idx[rel])
            if k0 - cand != cand:
                i_src = cand
                i_dst = k0 - cand
                break
        else:
            return None, None

    # compute a safe transfer amount
    delta = a[i_src] * rng.random() * step_factor
    if delta <= 0.0:
        return None, None

    a_new = a.copy()
    a_new[i_src] -= delta
    a_new[i_dst] += delta
    # numerical safety – enforce non‑negativity
    a_new[i_src] = max(a_new[i_src], 0.0)

    new_score = _score_from_seq(a_new)
    return a_new, new_score

def _random_perturb(a, rng):
    """Small random mass exchange between two distinct indices."""
    n = a.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a, _score_from_seq(a)
    delta = a[i] * rng.random() * 0.3
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2[i] = max(a2[i], 0.0)
    return a2, _score_from_seq(a2)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that seeks to
    minimise the AC1 upper bound as measured by ``evaluate_sequence``.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    rng = np.random.default_rng(seed)

    a = _initial_sequence()
    n = a.shape[0]

    # ensure a has unit ℓ₁ norm (objective is scale‑invariant)
    s = a.sum()
    if s > 0.0:
        a = a / s
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = a.copy()
    best_score = _score_from_seq(best_seq)

    # ------------------------------------------------------------------
    it = 0
    max_iters = 200000               # hard cap; loop also stops at deadline
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # try a directed move that specifically targets the current maximum
        a_candidate, cand_score = _move_towards_flat(a, rng, step_factor=0.5)
        if a_candidate is not None and cand_score < best_score - 1e-12:
            a = a_candidate
            best_seq = a_candidate.copy()
            best_score = cand_score
            continue

        # occasional random perturbation to escape plateaus

                best_seq = a_tmp.copy()
                best_score = tmp_score

            total = a[idxs].sum()
            if total > 0:

                pert = pert / pert.sum() * total * 0.02   # 2 % of their mass
