import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score for a vector whose sum is (approximately) 1:
       2 * n * max(convolution)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_idx = int(conv_full.argmax())
        max_val = conv_full[max_idx]

        # one‑hot mask at the maximal entry (soft‑max with huge temperature)
        w = np.zeros_like(conv_full)
        w[max_idx] = 1.0
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        # gradient of the (approximate) max‑convolution
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map to size‑n gradient
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size decays like 1/sqrt(it)
        step = step0 / math.sqrt(it)

        # candidate and projection onto the simplex
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score(cand)

        # accept if improvement; else occasional probabilistic jump
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # small Metropolis‑like acceptance
            temp = step * 0.4 + 1e-12
            prob = math.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # periodic symmetrisation (helps many instances)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # if stuck for a while, perform a lightweight random redistribution
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.6
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc

        # occasional aggressive reset (keeps diversity)
        if it % 3000 == 0:
            # blend current best with a fresh random simplex point
            rnd = _project_simplex(rng.random(n))
            t = rng.random() * 0.2          # small mixing weight
            cur = (1 - t) * best_vec + t * rnd
            cur = _project_simplex(cur)
            cur_sc = _score(cur)

    # ----------------------------------------------------------------------
    # Return best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
