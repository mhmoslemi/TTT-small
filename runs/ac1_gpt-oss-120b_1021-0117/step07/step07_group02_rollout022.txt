import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score for a vector already normalised to sum=1 (denominator drops)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _subgradient(a, conv, max_val):
    """Exact subgradient of max(conv) w.r.t. a (sum normalised to 1)."""
    n = a.shape[0]
    # indices attaining the maximum (within a tight tolerance)
    idxs = np.flatnonzero(np.isclose(conv, max_val, atol=1e-12, rtol=0.0))
    grad = np.zeros_like(a)
    for k in idxs:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i = np.arange(start, end + 1)
        grad[i] += 2.0 * a[k - i]          # derivative of conv[k] w.r.t a[i]
    grad *= 2.0 * n                       # chain rule for objective 2*n*max
    return grad
