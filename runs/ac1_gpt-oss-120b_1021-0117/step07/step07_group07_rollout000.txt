def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = _project_simplex(np.maximum(a0, 0.0))
    else:
        n = rng.integers(2000, 3000)
        cur = _project_simplex(rng.random(n))

    n = cur.size
    L = 1 << ((2*n - 1).bit_length())
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1=0.9; beta2=0.999; eps=1e-8

    temp = 1.0
    temp_factor = 1.01
    temp_max = 5000.0

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # FFT of current
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # identify active indices within epsilon
        eps_active = max_val * 1e-8  # small relative tolerance
        mask = conv_full >= max_val - eps_active
        active_cnt = np.count_nonzero(mask)
        if active_cnt == 0:
            w = np.zeros_like(conv_full)
            w[conv_full.argmax()] = 1.0
        else:
            w = np.where(mask, 1.0/active_cnt, 0.0)

        # gradient via convolution with reversed weight
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[(2*n-2) - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad**2)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        step = (0.5 / (np.sqrt(v_hat) + eps))   # step0 = 0.5

        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1

        # temperature increase (optional, not used in w)
        temp = min(temp_max, temp * temp_factor)

        # periodic symmetrization
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # random perturb if stuck
        if no_improve > 2500:
            # small random redistribution
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0

        if best_sc < target:   # target defined earlier

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # Sort v decreasingly
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a):
    """Score when sum(a)=1: 2·n·max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
