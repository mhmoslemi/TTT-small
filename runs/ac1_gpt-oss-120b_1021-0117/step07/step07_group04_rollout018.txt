def optimize(start_vec, max_iter, rng):
    cur = start_vec.copy()
    cur = _project_simplex(cur)
    cur_sc = 2.0 * n * _max_conv(cur)  # compute via FFT conv
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    best_local_cur = cur.copy()
    best_local_sc = cur_sc
    for it in range(1, max_iter+1):
        if time.monotonic() >= deadline:
            break
        # compute FFT and conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # compute mask
        eps = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        # gradient subgrad
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # step size schedule
        step = step0 / np.sqrt(it)
        # try steps
        cand_best_sc = cur_sc
        cand_best = cur
        for factor in (0.5, 1.0, 1.5):
            cand = cur - step * factor * momentum
            # project simplex and symmetrize
            cand = _project_simplex(cand)
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)
            # compute score
            cand_fft = np.fft.rfft(cand, n=L)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[:2*n-1]
            cand_sc = 2.0 * n * cand_conv.max()
            if cand_sc < cand_best_sc:
                cand_best_sc = cand_sc
                cand_best = cand
        if cand_best_sc < cur_sc - 1e-12:

            cur_sc = cand_best_sc
            # reset stagnation counter

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0., float(budget_s) - 0.5)

    # Determine length n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        parent_vec = np.asarray(height_sequence_1, dtype=np.float64)
        parent_vec = np.maximum(parent_vec, 0.0)
        parent_vec = parent_vec / parent_vec.sum() if parent_vec.sum()>0 else np.full_like(parent_vec, 1.0/n)
    else:
        n = 2000
        parent_vec = None

    L = 1 << ((2*n-1).bit_length())

    # Project onto simplex
    def project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        # similar to _project_simplex from previous code
        nlen = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, nlen+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        # adjust small errors
        excess = w.sum() - 1.0
        if abs(excess) > 1e-12:
            w -= excess / nlen
            w = np.maximum(w, 0.0)
        return w

    # compute score (max conv) quickly
    def compute_score_vec(a):
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # sum close to 1
        s = a.sum()
        if s < 1e-12:
            return float('inf')
        return 2.0 * n * max_val / (s*s)

    # Build initial candidates
    xs = np.arange(n, dtype=np.float64)
    candidates = []

    # Uniform
    candidates.append(np.full(n, 1.0/n))

    # Triangular
    tri = np.minimum(xs + 1, n - xs)
    candidates.append(tri / tri.sum())

    # Sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    candidates.append(sine / sine.sum())

    # Raised cosine
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w
