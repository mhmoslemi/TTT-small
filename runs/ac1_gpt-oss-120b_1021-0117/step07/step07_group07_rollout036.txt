    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):

    else:
        w /= w_sum

    # gradient
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]

    # Adam update
    m = beta1 * m + (1-beta1) * grad
    v = beta2 * v + (1-beta2) * (grad **2)
    it += 1
    m_hat = m / (1 - beta1 ** it)
    v_hat = v / (1 - beta2 ** it)
    step = step0 / (np.sqrt(v_hat) + eps)

    cand = cur - step * m_hat
    # Ensure non-negative before projection: candidate after proj will handle

    cand = _project_simplex(cand)

    cand_sc = _score(cand)

    if cand_sc < cur_sc:

        no_improve = 0
        if cand_sc < best_sc:

import time
import numpy as np


def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a):
    """Score for a vector already normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        n = len(height_sequence_1)
