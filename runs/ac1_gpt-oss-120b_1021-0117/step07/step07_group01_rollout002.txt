import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score for a vector already normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _gradient(a, L):
    """Sub‑gradient of max{a*a} w.r.t. a (assumes sum(a)=1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * len(a) - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * len(a) - 2) - np.arange(len(a))
    return 2.0 * grad_full[idx]

    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                     # enforce sum = 1
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # random sampling phase (quickly hunt for a better start)
    rand_trials = 300
    for _ in range(rand_trials):
        if time.monotonic() >= deadline:
            break
        r = rng.random(n, dtype=np.float64)
        r = _project_simplex(r)
        sc = _score(r)
        if sc < best_sc:
            best = r.copy()
            best_sc = sc
    a = best.copy()

    # ------------------------------------------------------------------
    # Adam optimizer with occasional extra moves
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # compute gradient and current score
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # gradient of the max‑convolution
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base step size
        base_step = 0.8 / (math.sqrt(it) + 1.0)

        # generate a few trial candidates
        cand_list = []

        # Adam‑based additive update with three scales
        for scale in (0.5, 1.0, 2.0):
            upd = base_step * scale * m_hat / (np.sqrt(v_hat) + eps)
            cand = _project_simplex(a - upd)
            cand_list.append(cand)

        # pure gradient step (also three scales)
        for scale in (0.5, 1.0, 2.0):
            cand = _project_simplex(a - base_step * scale * grad)
            cand_list.append(cand)

        # evaluate candidates (skip evaluating current a)
        cand_scores = [_score(c) for c in cand_list]
        min_idx = int(np.argmin(cand_scores))
        cand_best = cand_list[min_idx]
        cand_best_sc = cand_scores[min_idx]

        # accept if we improved the global best
        if cand_best_sc < best_sc - 1e-12:
            best = cand_best.copy()
            best_sc = cand_best_sc
            a = cand_best.copy()
            no_improve = 0
        else:
            # otherwise maybe move to a slightly better point
            if cand_best_sc < cur_sc:
                a = cand_best.copy()
                no_improve = 0
            else:
                no_improve += 1

        # ------------------------------------------------------------------
        # occasional symmetrisation
        if it % 400 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym.copy()
                no_improve = 0

        # ------------------------------------------------------------------
        # random pair transfer when stagnating
        if no_improve > 500 and it % 50 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.2
            tmp = a.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            sc_tmp = _score(tmp)
            if sc_tmp < best_sc:
                best = tmp.copy()
                best_sc = sc_tmp
                a = tmp.copy()
                no_improve = 0

        # ------------------------------------------------------------------
        # restart from a random blend after long stagnation
        if no_improve > 2000:
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = _project_simplex(0.5 * best + 0.5 * rand_vec)
            no_improve = 0

        # ------------------------------------------------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    return [float(x) for x in best]
