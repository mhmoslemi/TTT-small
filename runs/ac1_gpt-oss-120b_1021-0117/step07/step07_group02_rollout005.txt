import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - css / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score assuming a sums to 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score_vec(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # occasional random perturbation to escape plateaus
            if no_improve % 400 == 0:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.5
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score_vec(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        no_improve = 0
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()
                else:
                    # full random restart
                    rnd = _project_simplex(rng.random(n))
                    rnd_sc = _score_vec(rnd)
                    if rnd_sc < cur_sc:
                        cur, cur_sc = rnd, rnd_sc
                        m.fill(0.0)
                        v.fill(0.0)
                        no_improve = 0
                        if rnd_sc < best_sc:
                            best_sc, best_vec = rnd_sc, rnd.copy()

        # periodic symmetrisation (helps for many optimal patterns)
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # decay temperature factor
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # early exit if we reached the target bound
        if best_sc < target:
            break

        # checkpoint for deadline
        if it % 5000 == 0 and time.monotonic() >= deadline:
            break

    # Return best‑found vector as plain Python floats
    return [float(x) for x in best_vec]
