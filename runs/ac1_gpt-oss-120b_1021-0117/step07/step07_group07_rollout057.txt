def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # initialization
    if 'height_sequence_1' in globals():
        cur = project_simplex(np.maximum(height_sequence_1, 0.0))
    else:
        n = rng.integers(2000, 3000)
        cur = project_simplex(rng.random(n))

    n = cur.size
    L = 1 << ((2*n - 1).bit_length())
    best = cur.copy()
    best_score = _score(best)
    cur_score = best_score

    # ADAM state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    temp = 0.1
    temp_factor = 1.006
    temp_max = 500.0

    step0 = 0.5

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # compute convolution and softmax gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n - 1]
        max_val = conv.max()

        # softmax weights
        # compute w = exp(temp * (conv - max_val))
        # Clip large exponents?
        w = np.exp(temp * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)   # reversed indices
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        cand = project_simplex(cand)

        cand_score = _score(cand)

        # acceptance
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            no_improve += 1
            # occasional acceptance
            if rng.random() < 0.001:
                cur, cur_score = cand, cand_score

        # temperature schedule
        if temp < temp_max:
            temp = min(temp_max, temp * temp_factor)

        # periodic symmetrization
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = project_simplex(sym)
            sym_score = _score(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # restart when stuck
        if no_improve > 2500:
            # generate small random perturbation
            pert = cur.copy()
            # choose a small subset indices to shuffle mass
            idxs = rng.choice(n, size=10, replace=False)
            # random values to assign that still sum to same (project)
            pert[idxs] = rng.random(len(idxs))
            pert = project_simplex(pert)
            pert_score = _score(pert)
            if pert_score < cur_score:
                cur, cur_score = pert, pert_score
                no_improve = 0
                if pert_score < best_score:
                    best, best_score = pert.copy(), pert_score
            else:
                # if still bad, reinitialize a bit
                cur = project_simplex(rng.random(n))
                cur_score = _score(cur)
                m.fill(0)
                v.fill(0)
                it = 0
                no_improve = 0

        # early exit if we beat target
        if best_score < 1.50274349:

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Score for a non‑negative vector a with sum(a)=1 (i.e. 2*n*max(conv))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    rng = np.random.default_rng(seed)
