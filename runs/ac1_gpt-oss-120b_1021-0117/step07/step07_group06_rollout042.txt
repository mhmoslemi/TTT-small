import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renorm
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score_vec(v, L):
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2*v.size -1]
    return float(2.0 * v.size * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # safety margin

    # Determine length
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000

    # arcsine-like initialization
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    # safe replace non-finite
    a = np.where(np.isfinite(a), a, 0.0)
    cur = a / a.sum()
    cur = _project_simplex(cur)  # ensure exact simplex

    # precompute FFT length
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_sc = _score_vec(best, L)

    # optimizer parameters
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    it = 0
    no_improve = 0

    # softmax schedule
    alpha_max = 80.0
    alpha0 = 1.0

    target = 1.45   # early stopping target below required bound

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # convolution via FFT
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # softmax weights
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback to uniform mask of near-maximum entries
            mask = conv >= max_val * (1.0 - 1e-12)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(conv)
                w_sum = w.size
        w /= w_sum

        # gradient: 2 * (a convolve reversed w)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # map to size n gradient (valid part)
        grad = 2.0 * grad_full[(n-1): (2*n-1)]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size decay
        step = step0 / math.sqrt(it)

        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_vec(cand, L)

        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            no_improve = 0
        if cand_sc < cur_sc if (cur_sc := _score_vec(cur, L)) else True:
            cur, cur_sc = cand, cand_sc
            no_improve = 0

            if rng.random() < math.exp(-(cand_sc - cur_sc) / temp):

        if it % 300 == 0:

            sym_sc = _score_vec(sym, L)

                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        if no_improve > 2000:
            idx = rng.integers(0, n)

                j = rng.integers(0, n-1)

                    if tmp_sc < best_sc:

                no_improve = 0

        if best_sc < target:

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def _score_vec(v, L):
    """Score = 2·n·max(conv(v,v)) for a vector already summing to 1."""
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * v.size - 1]
    return float(2.0 * v.size * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin before the sandbox kills the process
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Length of the problem
    n = len(height_sequence_1) if "height_sequence_1" in globals() else 2000

    # ------------------------------------------------------------------
    # 1️⃣  Arcsine‑like initialization (continuous solution that makes the
    #     self‑convolution flat).  x are mid‑points of the n sub‑intervals.
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.where(np.isfinite(a), a, 0.0)
    cur = a / a.sum()
    cur = _project_simplex(cur)               # exact simplex projection

    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())       # power‑of‑two FFT length
    cur_sc = _score_vec(cur, L)
    best = cur.copy()
    best_sc = cur_sc

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.9                # momentum coefficient
    step0 = 0.8               # initial step magnitude
    it = 0
    no_improve = 0

    # soft‑max schedule for the “max” term
    alpha0 = 1.0
    alpha_max = 80.0

    # early‑stop target (well below the required 1.5027…)
    target = 1.45

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution of the current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- smooth approximation of the max (soft‑max) -----
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0.0:
            # fall back to a uniform mask of the near‑max entries
            mask = conv >= max_val * (1.0 - 1e-12)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(conv)
                w_sum = w.size
        w /= w_sum

        # ----- gradient of the smooth max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map linear convolution → gradient
        grad = 2.0 * grad_full[idx]

        # ----- momentum update & step size -----
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        # ----- candidate and its score -----
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_vec(cand, L)

        # ----- keep the best overall solution -----
        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            no_improve = 0

        # ----- accept/reject for the current iterate -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic acceptance (simulated‑annealing flavour)
            temp = step * 0.5 + 1e-12
            if rng.random() < math.exp(-(cand_sc - cur_sc) / temp):
                cur, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation (helps for many instances) -----
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # ----- random local redistribution when stuck -----
        if no_improve > 2000:
            i0 = rng.integers(0, n)
            if cur[i0] > 0.0:
                delta = cur[i0] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i0:
                    j += 1
                tmp = cur.copy()
                tmp[i0] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if we have already passed the target -----
        if best_sc < target:
            break

    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best]
