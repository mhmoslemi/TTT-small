import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum == 1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("simplex projection expects 1‑D vector")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _full_convolution(x, size=None):
    """Return full convolution x*x using real FFT (no dense Toeplitz)."""
    n = x.shape[0]
    if size is None:
        size = 1 << ((2 * n - 2).bit_length())  # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
def _objective(x, n, conv_size):
    """Compute the AC1 upper bound for a non‑negative vector x (sum may be any)."""
    conv = _full_convolution(x, conv_size)
    max_b = conv.max()
    sum_a = x.sum()
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that (approximately)
    minimizes the AC1 upper bound as measured by ``evaluate_sequence``.
    The routine starts from the supplied parent ``height_sequence_1`` (if any)
    or from the uniform distribution, then performs a mixture of
    gradient‑guided and random simplex moves under a strict time budget.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ------------------------------------------------------------------
    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a.sum()                 # normalise to sum == 1
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # pre‑compute FFT size (same for all convolutions)
    conv_size = 1 << ((2 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # initialise best
    cur = a.copy()
    cur_score = _objective(cur, n, conv_size)
    best_vec = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng(seed)

    # parameters
    step_size = 0.08          # initial gradient step magnitude
    min_step = 1e-5
    move_frac = 0.25          # max fraction of source mass to move in random step
    prob_grad = 0.35          # chance to perform a gradient move each iteration

    # ------------------------------------------------------------------
    # main optimisation loop
    while time.monotonic() < deadline:
        if rng.random() < prob_grad:
            # ---- gradient‑guided move ---------------------------------
            conv = _full_convolution(cur, conv_size)
            max_c = conv.max()
            argmax = np.flatnonzero(conv == max_c)
            # sub‑gradient of the max‑convolution (sum of shifted copies)
            grad = np.zeros_like(cur)
            for k in argmax:
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                idx = np.arange(i_start, i_end + 1)
                grad[idx] += 2.0 * cur[k - idx]
            # back‑tracking line search on the simplex
            step = step_size
            improved = False
            while step >= min_step:
                cand = cur - step * grad
                cand = _project_simplex(cand)
                cand_score = _objective(cand, n, conv_size)
                if cand_score < cur_score:
                    improved = True
                    break
                step *= 0.5
            if improved:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_vec = cand.copy()
                # adapt step size
                step_size = min(step_size * 1.1, 1.0)
            else:
                step_size = max(step_size * 0.7, min_step)
        else:
            # ---- random simplex move ----------------------------------
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = cur[i] * rng.random() * move_frac
            if delta <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            cand_score = _objective(cand, n, conv_size)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_vec = cand.copy()

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # Ensure the returned values are plain floats within the allowed range.
    return [float(x) for x in best_vec]
