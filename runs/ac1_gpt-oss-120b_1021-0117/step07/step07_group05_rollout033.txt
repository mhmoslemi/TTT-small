        if height_seq.sum() == 0:
            # fallback to uniform
            n = height_seq.shape[0]
            cur = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            cur = height_seq.copy()
            cur = cur / cur.sum()   # normalize to sum = 1
            n = cur.shape[0]
    else:
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    # Initial evaluation
    try:
        cur_score = evaluate_sequence(cur.tolist())
    except Exception:
        cur_score = float('inf')
    best_vec = cur.copy()
    best_score = cur_score

    # Helper: compute objective and max index
    def compute_max_and_score(a):
        # a already normalized sum=1 (but not required)
        conv = np.convolve(a, a)
        max_val = conv.max()
        # compute score
        score = float(2 * n * max_val / (a.sum() ** 2))
        return max_val, conv, score

    max_frac = 0.5   # max fraction of source mass to move
    # Temperature schedule for SA
    init_temp = 0.1
    final_temp = 1e-5

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Compute current conv and max index
        conv = np.convolve(cur, cur)
        max_val = conv.max()

        # For each i where j = k_max - i is valid
        # vectorized:
        idx = np.arange(n)
        j_idx = k_max - idx
        valid = (j_idx >= 0) & (j_idx < n)
        impact = np.zeros(n, dtype=np.float64)
        impact[valid] = cur[idx[valid]] * cur[j_idx[valid]]

        # Choose source index i proportional to impact (if any)
        if impact.sum() > 0:

            i = rng.choice(n, p=probs)

        temp = init_temp * ((final_temp / init_temp) ** progress)

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if temp > 0.0:
                prob = np.exp(-(cand_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()

        # --------------------------------------------------------------
        # Occasionally inject a small global perturbation
        if it % 5000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            pert = cur + noise
            pert = np.maximum(pert, 0.0)
            sum_pert = pert.sum()
            if sum_pert > 0.0:
                pert = pert / sum_pert
                try:
                    pert_score = evaluate_sequence(pert.tolist())
                except Exception:
                    pert_score = float("inf")
                if pert_score < best_score:
                    best_score = pert_score
                    best_vec = pert.copy()
                if pert_score < cur_score:
                    cur = pert
                    cur_score = pert_score

    # Final safety clamp (the evaluator also clamps, but we keep it tidy)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    return [float(x) for x in best_vec]
