def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # small margin

    # initialization
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random positive vector of size 2000
        n = 2000
        a0 = rng.random(n, dtype=np.float64) + 1e-3

    # project to simplex (sum to 1)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n-1

    # best solution storage
    best_vec = cur.copy()
    best_sc = 2.0 * n * _conv_max(cur, L)  # we need a function to compute max only.

    # optimizer parameters
    m = np.zeros_like(cur)  # first moment
    v = np.zeros_like(cur)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.1
    alpha = 20.0  # softmax sharpness
    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # compute convolution and weighted gradient
        max_val, grad = _conv_softmax_grad(cur, L, alpha)
        cur_sc = 2.0 * n * max_val

        # update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
            if best_sc < target:
                break

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)

        # occasional symmetrize
        if it % 250 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # evaluate candidate
        cand_max, cand_grad = _conv_softmax_grad(cand, L, alpha)  # need gradient? but we only need max
        cand_sc = 2.0 * n * cand_max

        # acceptance: if better accept; else accept with prob
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            # simulated annealing acceptance probability decaying
            T = 0.01 * (1.0 - it / 50000)  # degrade
            prob = math.exp(-(cand_sc - cur_sc) / max(T, 1e-6))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # if stuck for many iterations, random perturb
        if no_improve > 5000:
            sigma = 0.05 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert_max, _ = _conv_softmax_grad(pert, L, alpha)
            pert_sc = 2.0 * n * pert_max
            if pert_sc < cur_sc:
                cur = pert
                cur_sc = pert_sc
                no_improve = 0
            else:
                # reset momentum
                m.fill(0)
                v.fill(0)
                no_improve = 0

    # Return best after converting to list
    return [float(x) for x in best_vec]

def _conv_softmax_grad(a, L, alpha):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2 * n - 1]
    max_val = conv.max()
    # compute softmax weights
    # subtract max for numeric stability
    # w = np.exp(alpha * (conv - max_val))
    # Normalize w
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # Compute gradient: conv of a with reversed w (since gradient of weighted sum)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    # gradient indices mapping: as in previous code: idx = (2*n-2)-np.arange(n)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_softmax_grad(a, L, alpha=30.0):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    # compute softmax weights over conv entries
    # shift to avoid overflow: conv - max_val <=0
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # compute gradient
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def _conv_max(a, L):
    """Maximum value of the full convolution a*a using FFT."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return conv.max()


def _conv_softmax_grad(a, L, alpha):
    """
    Return (max_convolution, gradient) where gradient is the derivative of the
    soft‑max approximated max (with parameter alpha) with respect to a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()

    # soft‑max weights (stable)
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient of weighted sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad
