import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_normed(a, L):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # small safety buffer

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback length similar to typical parent length
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # ensure sum = 1 and non‑negative

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())       # power of two for FFT

    # optional initial symmetrisation (helps many instances)
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    best_vec = cur.copy()
    best_sc = _score_normed(best_vec, L)
    cur_sc = best_sc

    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.9               # momentum decay
    eta0 = 1.0               # base step size
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # ----- convolution and max index -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        idx = int(conv.argmax())

        # ----- gradient of (soft) max -----
        # use a tiny epsilon‑band around the maximum for stability
        eps = max_val * 1e-8
        mask = conv >= max_val - eps
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0:
            w[idx] = 1.0
            w_sum = 1.0
        w /= w_sum                     # now w is a probability vector
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[(2 * n - 2) - np.arange(n)]

        # ----- momentum and step size -----
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = eta0 / math.sqrt(it)

        # ----- two candidate updates -----
        # 1. simple projected gradient step
        cand1 = cur - eta * momentum
        cand1 = _project_simplex(np.maximum(cand1, 0.0))
        cand1_sc = _score_normed(cand1, L)

        # 2. exponentiated‑gradient step (always stays non‑negative)
        ex = -eta * momentum
        ex = np.clip(ex, -50, 50)            # avoid overflow
        cand2 = cur * np.exp(ex)
        s2 = cand2.sum()
        if s2 > 0:
            cand2 = cand2 / s2
        else:
            cand2 = cand1                      # fallback
        cand2_sc = _score_normed(cand2, L)

        # ----- pick the best among current and candidates -----
        if cand1_sc < cur_sc and cand1_sc <= cand2_sc:
            cur, cur_sc = cand1, cand1_sc
            no_improve = 0
        elif cand2_sc < cur_sc:
            cur, cur_sc = cand2, cand2_sc
            no_improve = 0
        else:
            # occasional annealed acceptance of a worse move
            no_improve += 1
            temp = max_val * 0.01 + 1e-12
            prob = math.exp(-(min(cand1_sc, cand2_sc) - cur_sc) / temp)
            if rng.random() < prob:
                if cand1_sc < cand2_sc:
                    cur, cur_sc = cand1, cand1_sc
                else:
                    cur, cur_sc = cand2, cand2_sc
                no_improve = 0

        # ----- record best seen so far -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # ----- periodic symmetrisation -----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_normed(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random redistribution when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_normed(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- early exit if target achieved -----
        if best_sc < target:
            break

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
