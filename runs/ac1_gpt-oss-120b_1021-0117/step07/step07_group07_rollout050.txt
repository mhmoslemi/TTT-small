def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.5)

    # Init
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
        cur = np.maximum(cur, 0.0)
        cur = _project_simplex(cur)
    else:
        n = rng.integers(2000, 3000)
        cur = _project_simplex(rng.random(n))
    n = cur.size
    L = 1 << ((2*n - 1).bit_length())
    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.8
    temp = 0.5
    temp_factor = 1.005
    temp_max = 300.0

    it = 0
    no_improve = 0
    # parameters
    swap_frac_max = 0.05    # max fraction of source mass moved
    swap_attempts = 0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # Decide operation
        if no_improve < 3000:
            # gradient step
            # compute a_fft, conv, etc.
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()
            # softmax weights
            w = np.exp(temp * (conv - max_val))
            w_sum = w.sum()
            if w_sum == 0.0 or np.isnan(w_sum):
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum
            # gradient via convolution with reversed w
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = step0 / (np.sqrt(v_hat) + eps)

            cand = cur - step * m_hat
            cand = _project_simplex(cand)

            cand_sc = _score(cand)

            # acceptance
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best, best_sc = cand.copy(), cand_sc
            else:
                no_improve += 1
                # occasional random acceptance
                if rng.random() < 0.003:
                    cur, cur_sc = cand, cand_sc

            # temperature schedule
            temp = min(temp_max, temp * temp_factor)

        else:
            # swap phase
            # random transfer
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                frac = rng.random() * swap_frac_max
                delta = cur[i] * frac
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                # No need to project as sum unchanged and nonneg
                cand_sc = _score(cand)
                if cand_sc < cur_sc:
                    cur, cur_sc = cand, cand_sc
                    no_improve = 0
                    if cand_sc < best_sc:
                        best, best_sc = cand.copy(), cand_sc
                else:
                    # occasional acceptance
                    if rng.random() < 0.001:
                        cur, cur_sc = cand, cand_sc
                        no_improve = 0

        # periodic symmetrisation
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # random redistribution when stuck long
        if no_improve > 5000:
            # add small Gaussian jitter to whole vector
            noise = rng.normal(scale=0.01, size=n)
            cand = cur + noise
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best, best_sc = cand.copy(), cand_sc

        # early exit if best below target
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]

import time, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score = 2*n*max(conv(a,a)) for a already on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # initial vector
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        cur = _project_simplex(cur)
    else:
        n0 = rng.integers(2000, 3000)
        cur = _project_simplex(rng.random(n0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # Adam state for gradient steps
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.8

    # temperature for soft‑max weighting of the convolution
    temp = 0.5
    temp_factor = 1.005
    temp_max = 300.0

    it = 0
    no_improve = 0
    swap_frac_max = 0.05          # maximal fraction of mass moved in swaps
    target_score = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ------------------------------------------------------------------
        # Decide which move to perform
        if no_improve < 3000:
            # ------------------- gradient / Adam step --------------------
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # soft‑max weighting of convolution entries
            w = np.exp(temp * (conv - max_val))
            w_sum = w.sum()
            if w_sum == 0.0 or np.isnan(w_sum):
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

            # gradient of weighted sum (2 * a * w_rev convolution)
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = step0 / (np.sqrt(v_hat) + eps)

            cand = cur - step * m_hat
            cand = _project_simplex(cand)

            cand_sc = _score(cand)

            # acceptance

        # Early termination if we already beat the target
        if best_sc < target_score:

    # Return the best vector found as a plain Python list of floats
    return [float(x) for x in best]
