    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Temperature schedule for softmax gradient
    temp_factor = 1e-2  # relative to max conv
    temp_decay = 0.9995
    min_temp_factor = 1e-6

    best_a = a.copy()
    best_score = _score_vec_fast(best_a)   # compute via FFT

    it = 0
    no_improve_counter = 0

    while time.monotonic() < deadline:
        it += 1

        # compute conv and max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n-1]
        max_val = conv_full.max()

        # compute weighted gradient via softmax
        temp = max_val * max(temp_factor, min_temp_factor)
        # Avoid overflow/underflow: subtract max_val before exp
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum != 0:
            w /= w_sum
        else:
            w.fill(0.0)

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - css / ind > 0
    rho = np.where(cond)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec_fast(a):
    """Fast computation of the AC1 upper bound (sum assumed positive)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s <= 0.0:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))
