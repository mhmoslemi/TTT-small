import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a):
    """Score for a vector already normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _grad_max(a, k):
    """Sub‑gradient of max‑convolution w.r.t. a for shift k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if k < n:
        g[k:] = a[: n - k]
        g[: n - k] += a[k:]
    return g

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.4)  # safety margin

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():

        cur = _project_simplex(np.maximum(a0, 0.0))
    else:
        n0 = int(rng.integers(2000, 3000))
        cur = _project_simplex(rng.random(n0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam‑style accumulator (optional, we will use simple decay)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # learning‑rate schedule parameters
    base_lr = 0.5
    decay = 5e-4

    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # main gradient‑descent phase (sub‑gradient of the max)
    while True:

        it += 1

        # convolution and arg‑max
        a_fft = np.fft.rfft(cur, n=L)

        conv = conv_full[: 2 * n - 1]
        kmax = int(np.argmax(conv))

        g = _grad_max(cur, kmax)

        # Adam update (still matrix‑free)
