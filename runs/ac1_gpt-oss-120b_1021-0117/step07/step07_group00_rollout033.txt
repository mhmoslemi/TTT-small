def _softmax_weights(conv, temp):
   w = np.exp((conv - conv.max())/temp)
   w_sum = w.sum()
   if w_sum ==0: w = (conv==conv.max()).astype(float); w_sum = w.sum()

def propose_candidate(seed=42, budget_s=1100, **kwargs):
   rng = np.random.default_rng(seed)
   start = time.time()
   deadline = start + max(0.0, float(budget_s) - 0.2) # small margin

   if "height_sequence_1" in globals():
       cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
   else:
       cur = rng.random(2000, dtype=np.float64)
   # ensure non-negative
   cur = np.maximum(cur, 0.0)
   # project onto simplex sum=1
   cur = _project_simplex(cur)

   n = cur.size
   L = 1 << ((2*n - 1).bit_length())

   best = cur.copy()
   # compute initial score via evaluate_sequence
   try:
       best_sc = evaluate_sequence(best.tolist())
   except Exception:
       best_sc = float('inf')
   # We also keep best_conv but not needed.

   # optimizer params
   m = np.zeros_like(cur)
   v = np.zeros_like(cur)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   alpha = 0.8 # base learning rate
   temp = None
   temp_decay = 0.997
   min_temp_factor = 1e-4

   no_improve = 0
   it = 0

   while time.time() < deadline:
       it += 1

       # Compute convolution via FFT
       a_fft = np.fft.rfft(cur, n=L)
       conv_full = np.fft.irfft(a_fft * a_fft, n=L)
       conv = conv_full[:2*n-1]
       max_val = conv.max()

       # temperature
       if temp is None:
           temp = max_val * 0.07
       else:
           temp = max(temp * temp_decay, max_val * min_temp_factor)

       # compute weights w
       w = np.exp((conv - max_val) / temp)
       w_sum = w.sum()
       if w_sum == 0.0:
           w = (conv == max_val).astype(np.float64)
           w_sum = w.sum()
       w /= w_sum

       # gradient via convolution with reversed w
       w_rev = w[::-1]
       w_fft = np.fft.rfft(w_rev, n=L)
       grad_full = np.fft.irfft(a_fft * w_fft, n=L)
       idx = (2*n - 2) - np.arange(n)
       grad = 2.0 * grad_full[idx]  # gradient of weighted max

       # Adam updates
       m = beta1 * m + (1 - beta1) * grad
       v = beta2 * v + (1 - beta2) * (grad ** 2)
       m_hat = m / (1 - beta1 ** it)
       v_hat = v / (1 - beta2 ** it)

       cand_sc = evaluate_sequence(cand.tolist())  # compute exact

       if cand_sc < best_sc:
           best, best_sc = cand.copy(), cand_sc
           no_improve = 0
       else:

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fast(a):
    """Fast computation of the evaluator's score (2*n*max(conv)/(sum^2))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))
