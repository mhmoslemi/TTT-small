import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_metrics(a, L):
    """Full convolution a*a (via FFT) and return (conv, max, sum, score)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    score = 2.0 * n * max_val / (s * s) if s > 0 else np.inf
    return conv, max_val, s, score

def _softmax_grad(a, conv, alpha, L):
    """Gradient of soft‑max weighted max‑convolution."""
    n = a.shape[0]
    maxv = conv.max()
    w = np.exp(alpha * (conv - maxv))
    w_sum = w.sum()
    if w_sum == 0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    a_fft = np.fft.rfft(a, n=L)
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.4)   # safety margin

    # --- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2500, dtype=np.float64)
    a = _project_simplex(np.maximum(a, 0.0))
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # best solution tracking
    conv, max_val, s, cur_score = _conv_metrics(a, L)
    best_score, best_a = cur_score, a.copy()

    # optimizer state
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.8
    alpha0, alpha_max = 0.5, 30.0
    ramp_iters = 2000
    it = 0
    stagn = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # schedule for soft‑max sharpness
        alpha = alpha0 + (alpha_max - alpha0) * min(1.0, it / ramp_iters)

        # gradient at current point
        conv, max_val, s, cur_score = _conv_metrics(a, L)
        grad = _softmax_grad(a, conv, alpha, L)

        # momentum and step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        # candidate update
        cand = _project_simplex(a - step * momentum)

        # occasional symmetrisation
        if it % 250 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # evaluate candidate
        _, _, _, cand_score = _conv_metrics(cand, L)

        # acceptance rule
        if cand_score < cur_score:
            a, cur_score = cand, cand_score
            stagn = 0
        else:
            temp = max(1e-4, 0.02 / math.sqrt(it))
            prob = math.exp(-(cand_score - cur_score) / temp)
            if rng.random() < prob:
                a, cur_score = cand, cand_score
                stagn = 0
            else:
                stagn += 1

        # record best
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            stagn = 0

        # random perturbation when stuck
        if stagn > 4000:
            sigma = 0.15 / math.sqrt(it)
            pert = a + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(np.maximum(pert, 0.0))
            _, _, _, pert_score = _conv_metrics(pert, L)
            if pert_score < cur_score:
                a, cur_score = pert, pert_score
                if pert_score < best_score:
                    best_score, best_a = pert_score, pert.copy()
                stagn = 0
            else:
                momentum.fill(0.0)
                stagn = 0

        # early exit if target reached
        if best_score < target:
            break

    return [float(x) for x in best_a]
