def _project_simplex(v):
    # ensure nonneg and sum=1
    v = np.maximum(v,0)
    if v.sum() == 0:
        # assign uniform
        return np.full_like(v, 1.0 / v.size)
    # Use sorting method:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

import time, math, numpy as np

def _project_simplex(v):
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
