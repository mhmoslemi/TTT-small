def propose_candidate(seed=42, budget_s=1100, **kwargs):

    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # small safety margin

    # initialization
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate random vector length ~2000
        n = 2000
        a0 = rng.random(n) + 1e-3

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    # set FFT length >= 3n-2
    L = 1 << ((3*n - 1).bit_length())
    # preallocate optimizer states
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2
    # softmax schedule
    alpha0 = 0.5
    alpha_max = 150.0

    best_vec = cur.copy()
    best_sc = _score(best_vec)

    cur_sc = best_sc

    it = 0
    no_improve = 0

    # target bound for early exit (score expects sum=1)
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # FFT of current
        a_fft = np.fft.rfft(cur, n=L)

        # autocorrelation via convolution
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # softmax weights
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        # shift to avoid overflow
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback uniform mask of near max entries
            mask = conv >= max_val - (abs(max_val) * 1e-12 + 1e-15)
            w = mask.astype(np.float64)
            cnt = w.sum()
            if cnt > 0:
                w /= cnt
        else:
            w /= w_sum

        # gradient using convolution of cur and w_rev
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic acceptance
            # temperature proportional to step magnitude
            temp = lr0 * 0.1 + 1e-12
            prob = math.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc
                no_improve = 0

        # update best
        if cand_sc < best_sc:
            best_vec = cand.copy()
            best_sc = cand_sc

        # periodic symmetrisation
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                no_improve = 0
            if sym_sc < best_sc:
                best_vec = sym.copy()
                best_sc = sym_sc

        # random redistribution if stuck
        if no_improve > 4000:
            # pick two distinct indices and move some mass
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_vec = tmp.copy()
                    best_sc = tmp_sc

        # early exit if target reached
        if best_sc < target:
            break

    # Return best vector as Python list
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalize to sum exactly 1 (numerical)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    start = time.monotonic()

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    # FFT length must accommodate convolution of cur (n) with weight (2n-1)
    L = 1 << ((3 * n - 1).bit_length())

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    eps = 1e-8
    lr0 = 0.15

    # soft‑max temperature
    alpha0 = 5.0
    alpha_max = 200.0

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    it = 0
    no_improve = 0

    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # autocorrelation
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights (smooth approximation of max)
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            mask = conv >= max_val - (abs(max_val) * 1e-12 + 1e-15)
            w = mask.astype(np.float64)
            cnt = w.sum()
            if cnt > 0:
                w /= cnt
        else:
            w /= w_sum

        # gradient of weighted max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad

        beta1_pow *= beta1
        beta2_pow *= beta2

                    best_sc = tmp_sc

            break

    # final safety: ensure result is a plain list of Python floats
    return [float(x) for x in best_vec]

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
