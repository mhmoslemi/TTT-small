def _softmax_grad(a, L, beta):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_conv = conv.max()
    # compute w = exp(beta*(conv - max_conv))
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    w /= w_sum
    # gradient = 2 * conv of a with reversed w
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, conv.max()  # return grad and max (approx? actual max unchanged)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # small margin
    # init
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = rng.random(2500, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    # best record
    max_val, _ = _conv_max_and_grad(a, L)
    best_sc = float(2.0 * n * max_val)
    best_a = a.copy()
    # optimizer states
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349

    # Phase 1: gradient descent with momentum
    while time.time() < deadline:
        it += 1
        max_val, grad = _conv_max_and_grad(a, L)
        cur_sc = 2.0 * n * max_val

        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        step = step0 / math.sqrt(it)
        cand = _project_simplex(a - step * momentum)

        # symmetrize occasionally
        if it % 50 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_sc = 2.0 * n * cand_max

        if cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0

        # stagnation handling
        if no_improve > 2000:
            # random perturbation
            sigma = 0.02 / math.sqrt(it)
            pert = a + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert_max, _ = _conv_max_and_grad(pert, L)
            pert_sc = 2.0 * n * pert_max
            if pert_sc < cur_sc:
                a = pert
                cur_sc = pert_sc
                if pert_sc < best_sc:
                    best_sc = pert_sc
                    best_a = pert.copy()
                no_improve = 0
            else:
                momentum.fill(0.0)
                no_improve = 0

        if best_sc < target:
            break

        # optional switch to hill climbing after some iterations
        if it % 1000 == 0 and time.time() < deadline - 5:
            # start hill climbing
            a_hc = best_a.copy()
            # run hill climbing loop
            hc_iters = 0
            while time.time() < deadline - 5:
                hc_iters += 1
                # pick two indices
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                # move delta from i to j
                delta = min(a_hc[i], 0.01 * rng.random())
                if delta <= 0:
                    continue
                a_new = a_hc.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # optional symmetrization after move to maintain symmetry
                if hc_iters % 200 == 0:
                    a_new = (a_new + a_new[::-1]) * 0.5
                    # ensure nonnegativity (projection not needed because sum unchanged)
                    a_new = np.maximum(a_new, 0.0)
                    # reproject onto simplex to adjust rounding errors
                    a_new = _project_simplex(a_new)

                new_max = _conv_max(a_new, L)
                new_sc = 2.0 * n * new_max
                if new_sc < best_sc:
                    best_sc = new_sc
                    best_a = a_new.copy()
                    a_hc = a_new
                else:
                    # accept with small probability to escape
                    if rng.random() < 0.001:
                        a_hc = a_new
                # break if reached target
                if best_sc < target:
                    break
            # after hill climbing, return to main loop using best as a
            a = best_a.copy()
            # reset momentum
            momentum.fill(0.0)
            # continue
    # final result: best_a
    return [float(x) for x in best_a]

import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L):
    """
    Returns (max_convolution, subgradient) of the maximum value of a*a (full convolution).
    a is assumed non‑negative; the function works for any a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

    conv_full = np.fft.irfft(a_fft * a_fft, n=L)

            else:
                momentum.fill(0.0)
                no_improve = 0

        if best_sc < target:
            break

    return best_vec, best_sc


def _hill_climb(start_vec, L, n, rng, deadline, best_sc, best_vec):
    """
    Simple random mass‑transfer hill climbing.
    """
    cur = start_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] <= 0.0:
            continue
        delta = min(cur[i], 0.01 * rng.random())
        if delta <= 0.0:
            continue
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # occasional symmetrisation (keeps shape balanced)
        if rng.random() < 0.01:
            new = (new + new[::-1]) * 0.5
            new = _project_simplex(new)

        new_max = _conv_max(new, L)
        new_sc = 2.0 * n * new_max

        if new_sc < cur_sc:
            cur, cur_sc = new, new_sc
            if new_sc < best_sc:
                best_sc, best_vec = new_sc, new.copy()
        else:
            if rng.random() < 0.001:
                cur, cur_sc = new, new_sc

        if best_sc < 1.50274349:
            break
    return best_vec, best_sc

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # safety margin

    # --------------------------------------------------------------
    # Determine length and FFT size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n‑1

    # --------------------------------------------------------------
    # Build a few diverse initial vectors (all will be projected later)
    inits = []

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        inits.append(a0)

    # triangular window
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    tri = 1.0 - np.abs(idx - mid) / (mid + 1.0)
    tri[tri < 0] = 0.0
    inits.append(tri)

    # Hann (cosine) window
    cos = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    inits.append(cos)

    # random start
    inits.append(rng.random(n, dtype=np.float64))

    best_vec = None
    best_sc = float("inf")
    # reserve a few seconds for the final hill‑climb phase
    reserve_hc = max(5.0, 0.05 * (deadline - start_time))

    # --------------------------------------------------------------
    # Phase 1 – run the momentum optimizer on each initializer
    for init in inits:
        if time.time() >= deadline - reserve_hc:
            break
        # give each run at most a few minutes (or less if total time is low)
        phase_deadline = min(deadline - reserve_hc, time.time() + 300.0)
        vec, sc = _run_optimizer(init, L, n, rng, phase_deadline)
        if sc < best_sc:
            best_sc, best_vec = sc, vec
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------------
    # Phase 2 – hill climbing around the best found point
    if best_vec is not None and time.time() < deadline:
        best_vec, best_sc = _hill_climb(best_vec, L, n, rng, deadline,
                                        best_sc, best_vec)

    # --------------------------------------------------------------
    # Fallback (should never be needed)
    if best_vec is None:
        best_vec = _project_simplex(rng.random(n))

    # Return plain Python list of floats
    return [float(x) for x in best_vec]
