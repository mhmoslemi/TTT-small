import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a, L):
    """Score assuming sum(a)=1: 2*n*max(conv) computed via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # initial vector (use parent if available)
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    idx = (2 * n - 2) - np.arange(n)             # indices for gradient extraction

    # best-so-far bookkeeping
    best_vec = cur.copy()
    best_sc = _score_norm1(best_vec, L)
    cur_sc = best_sc

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05
    t_iter = 0

    # temperature schedule
    temp = None
    decay_temp = 0.9995
    min_temp_factor = 0.005

    # stagnation detection
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        t_iter += 1

        # ---- convolution and temperature ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * decay_temp, max_val * min_temp_factor)

        # ---- softmax weights for smooth max ----
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:                         # fallback to hard max
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ---- gradient of the smooth max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
        cand_sc = _score_norm1(cand, L)

        # ---- accept if improvement ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic acceptance
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (lr + 1e-12)):
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # ---- update global best ----
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc

        # ---- occasional symmetrisation ----
        if t_iter % 600 == 0:

            sym_sc = _score_norm1(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ---- random redistribution when stuck ----
        if no_improve > 3000:

            j = rng.integers(0, n - 1)

    return [float(x) for x in best_vec]
