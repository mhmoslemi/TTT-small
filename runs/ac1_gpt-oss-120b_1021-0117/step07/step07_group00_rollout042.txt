def build_sidon_set(n):
    sums = set()
    S = []
    for i in range(n):
        # compute sums of i with each s in S, plus i+i
        conflict = False
        for s in S:
            if (i + s) in sums:
                conflict = True
                break
        if conflict or (i + i) in sums:
            continue
        # no conflict; add i
        for s in S:
            sums.add(i + s)
        sums.add(i + i)
        S.append(i)
    return S

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _sidon_set_greedy(n, rng):
    """Greedy construction of a Sidon set (all pairwise sums distinct)."""
    order = rng.permutation(n)
    sums = set()
    S = []
    for i in order:
        if (i + i) in sums:
            continue
        conflict = False
        for s in S:
            if (i + s) in sums:
                conflict = True
                break
        if conflict:
            continue
        # add i to the set
        for s in S:
            sums.add(i + s)
        sums.add(i + i)
        S.append(i)
    return S

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # length of the sequence to produce
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000                     # fallback length

    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    target = 1.50274349
    best_vec = None
    best_sc = float("inf")

    # internal fast score (sum is 1 after projection)
    def _score(a):
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # try several random greedy Sidon constructions
    while time.time() < deadline:
        S = _sidon_set_greedy(n, rng)
        m = len(S)
        if m < 2:
            continue                     # degenerate, try again
        vec = np.zeros(n, dtype=np.float64)
        vec[S] = 1.0 / m                 # equal weights, sum = 1
        # although already on the simplex, project for safety
        vec = _project_simplex(vec)

        sc = _score(vec)
        if sc < best_sc:
            best_sc, best_vec = sc, vec.copy()
            if best_sc < target:
                break

    # fallback: uniform distribution (guaranteed feasible)
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # final safety projection (ensures non‑negative, sum≥0.01)
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    best_vec = _project_simplex(best_vec)

    return [float(x) for x in best_vec]
