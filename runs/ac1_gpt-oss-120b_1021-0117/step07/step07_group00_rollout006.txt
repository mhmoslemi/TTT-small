import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Score = 2*n*max(convolution(a,a)) for sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _gradient(cur, temp, L):
    """Weighted gradient of the smoothed max of the autocorrelation."""
    n = cur.size
    a_fft = np.fft.rfft(cur, n=L)

    # autocorrelation
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()

    # softmax weights
    w = np.exp((conv - max_val) / temp)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = (conv == max_val).astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    # gradient via convolution with reversed weights
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

            no_improve += 1
            # occasional uphill acceptance (simulated annealing)
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # --------------------------------------------------------------
        # random redistribution when stuck
        if no_improve > 4000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # --------------------------------------------------------------
        # symmetrisation every few hundred iterations
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # early exit if target achieved
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # final local random tweaks (if time permits)
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best_vec[i] * rng.random() * 0.2
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp, L)
        if tmp_sc < best_sc:
            best_vec, best_sc = tmp, tmp_sc

    # Return as plain Python floats
    return [float(x) for x in best_vec]
