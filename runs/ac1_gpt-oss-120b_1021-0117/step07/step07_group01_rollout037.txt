import time, math, numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            if prob is not None and g.value is not None:
                g0 = np.maximum(g.value, 0.0)
                s = g0.sum()
                if s > 0:
                    g0 = g0 / s
                else:
                    g0 = a.copy()
                # convex combination and re‑projection
                cand = (1.0 - t_mix) * a + t_mix * g0
                cand = _project_simplex(cand)
                cand_sc = _score(cand)

                # ----- acceptance logic -----
                if cand_sc < best_sc:
                    best, best_sc = cand.copy(), cand_sc
                    a = cand
                    continue
                if cand_sc < _score(a):
                    a = cand
                    continue
                # occasional stochastic jump
                if rng.random() < 0.05:
                    a = cand
                continue  # go to next iteration

        # ------------------------------------------------------------------
        # If LP is unavailable or failed, fall back to Adam‑style updates
        # ----- sub‑gradient of the max‑convolution -----
        eps_mask = M * 1e-12 + 1e-15
        mask = conv >= M - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # gradient of weighted max‑convolution (see original code)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.ifft(af * w_fft).real
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- simple Adam state (kept across iterations) -----
        if it == 1:
            m_ad = np.zeros_like(a)
            v_ad = np.zeros_like(a)
            beta1, beta2 = 0.9, 0.999
            eps_ad = 1e-12
        m_ad = beta1 * m_ad + (1.0 - beta1) * grad
        v_ad = beta2 * v_ad + (1.0 - beta2) * (grad * grad)
        m_hat = m_ad / (1.0 - beta1 ** it)
        v_hat = v_ad / (1.0 - beta2 ** it)
        step = 0.85 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps_ad)

        # ----- two candidates: additive (projected) and multiplicative -----
        cand_add = _project_simplex(a - update)

        exp_factor = np.exp(-step * grad)
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        cand_mul = cand_mul / s if s > 0 else a.copy()

        cand_add_sc = _score(cand_add)
        cand_mul_sc = _score(cand_mul)

        if cand_add_sc < cand_mul_sc:
            cand, cand_sc = cand_add, cand_add_sc
        else:
            cand, cand_sc = cand_mul, cand_mul_sc

        # ----- accept if improvement, else occasional stochastic move -----
        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            a = cand
            continue
        if cand_sc < _score(a):
            a = cand
            continue
        if rng.random() < math.exp(-(cand_sc - best_sc) / (step + 1e-12)):
            a = cand

        # ----- occasional symmetrisation (helps convergence) -----
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best, best_sc = sym.copy(), sym_sc
                a = sym

        # ----- random perturbation when stagnating -----
        if it % 1500 == 0:
            perturb = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + perturb, 0.0)
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best]
