def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.5

    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a0 = rng.random(n) + 1e-3

    # Ensure within bounds:
    a0 = np.clip(a0, 0.0, 1000.0)

    # Helper to compute score (same as evaluator but faster)
    def score_vec(v):
        n = v.shape[0]
        L = 1 << ((2 * n - 1).bit_length())
        vf = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(vf * vf, n=L)[: 2 * n - 1]
        maxb = conv.max()
        summ = v.sum()
        if summ < 0.01:
            return np.inf
        return 2.0 * n * maxb / (summ * summ)

    best = a0
    best_sc = score_vec(best)

    cur = best.copy()
    cur_sc = best_sc

    # simple local search: random swaps
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # pick two indices with probability proportional to current values (to move mass)
        i = rng.integers(0, cur.size)
        j = rng.integers(0, cur.size)
        if i == j or cur[i] == 0.0:
            continue
        # move a fraction
        frac = rng.random() * 0.2  # up to 20% of mass at i
        delta = cur[i] * frac
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep within bounds
        cand[i] = max(0.0, min(1000.0, cand[i]))
        cand[j] = max(0.0, min(1000.0, cand[j]))
        cand_sc = score_vec(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # accept with small probability decaying with time
            temp = max(0.01, 0.1 * (1 - it / 1e6))
            if rng.random() < math.exp(-(cand_sc - cur_sc) / temp):
                cur, cur_sc = cand, cand_sc
    return [float(x) for x in best]

import time, math, numpy as np

def _score_vec(v: np.ndarray) -> float:
    """Exact score used by the evaluator (fast, no clipping)."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(vf * vf, n=L)[: 2 * n - 1]
    maxb = conv.max()
    sm = v.sum()
    if sm < 0.01:
        return np.inf
    return 2.0 * n * maxb / (sm * sm)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # initialise from the parent if supplied, otherwise random
    if "height_sequence_1" in globals():
        cur = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        n0 = 2000
        cur = np.clip(rng.random(n0) * 999.0 + 1.0, 0.0, 1000.0)

    # ------------------------------------------------------------------
    best = cur.copy()
    best_sc = _score_vec(best)
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # simple stochastic local search with occasional symmetrisation
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # pick two distinct indices, move a fraction of mass from i to j
        i = rng.integers(0, cur.size)
        j = rng.integers(0, cur.size - 1)
        if j >= i:
            j += 1
        if cur[i] <= 0.0:
            continue
        frac = rng.random() * 0.2          # up to 20 % move
        delta = cur[i] * frac
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # respect the hard clipping limits
        cand[i] = max(0.0, min(1000.0, cand[i]))
        cand[j] = max(0.0, min(1000.0, cand[j]))

        cand_sc = _score_vec(cand)

        # greedy acceptance, with a tiny annealed probability of uphill moves
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            temp = 0.02 * max(0.0, 1.0 - it / 2e5) + 1e-12
            if rng.random() < math.exp(-(cand_sc - cur_sc) / temp):
                cur, cur_sc = cand, cand_sc

        # every few thousand iterations, mirror the vector (helps many instances)
        if it % 4000 == 0:
            mir = (cur + cur[::-1]) * 0.5
            mir = np.clip(mir, 0.0, 1000.0)
            mir_sc = _score_vec(mir)
            if mir_sc < cur_sc:
                cur, cur_sc = mir, mir_sc
                if mir_sc < best_sc:
                    best, best_sc = mir.copy(), mir_sc

        # early stop if we already hit the (theoretically) optimal value 2.0
        if best_sc <= 2.0 + 1e-12:
            break

    # ------------------------------------------------------------------
    # Return the best vector found as a plain Python list of floats
    return [float(x) for x in best]
