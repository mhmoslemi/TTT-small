import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score_fft(a, L):
    """Score = 2*n*max(conv) assuming sum(a)=1."""
    n = a.size
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        # ----- candidate 2: multiplicative (mirror descent) -----
        cand_mul = a * np.exp(-lr * grad)
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()

        # ----- evaluate both candidates -----
        def _eval_candidate(c):
            cf = np.fft.rfft(c, n=L)
            conv_c = np.fft.irfft(cf * cf, n=L)[: 2 * n - 1]
            return float(2.0 * n * conv_c.max())

        sc_add = _eval_candidate(cand_add)
        sc_mul = _eval_candidate(cand_mul)

        if sc_add < sc_mul:
            cand, cand_sc = cand_add, sc_add
        else:
            cand, cand_sc = cand_mul, sc_mul

        # ----- accept if improvement -----
        if cand_sc < cur_sc:
            a = cand
        else:
            # occasional uphill move (tiny probability)
            if rng.random() < 1e-4:
                a = cand

        # ----- increase smoothing -----
        alpha = min(alpha_max, alpha + alpha_inc)

        # ----- periodic symmetrisation -----
        if it % 400 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _eval_candidate(sym)
            if sym_sc < best_sc:
                best_sc, best = sym_sc, sym.copy()
                no_improve = 0
            a = sym

        # ----- random perturb when stagnating -----
        if no_improve > 3000:
            # move a random chunk of mass
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                a = _project_simplex(tmp)
            no_improve = 0

        # ----- restart from random blend if really stuck -----
        if no_improve > 7000:
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    return [float(x) for x in best]
