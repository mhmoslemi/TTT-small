    """ Euclidean projection onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluation of the AC1 upper bound (assumes a>=0, sum>0)."""
    s = a.sum()
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def _grad(a, n, L, rel_tol=1e-6):
    """Sub‑gradient of max‑convolution w.r.t. a (a assumed non‑negative, sum=1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * rel_tol + 1e-12
    mask = conv >= max_val - eps
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Determine problem size and FFT length
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())   # next power of two

    # --------------------------------------------------------------
    # Assemble a few deterministic seeds
    xs = np.arange(n, dtype=np.float64)
    cand_list = []

    # 1) Uniform
    cand_list.append(np.full(n, 1.0 / n))

    # 2) Triangular peak
    tri = np.minimum(xs + 1, n - xs)
    cand_list.append(tri / tri.sum())

    # 3) Sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    cand_list.append(sine / sine.sum())

    # 4) Raised cosine bump
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    cand_list.append(bump / bump.sum())

    # 5) Exponential decays
    for alpha in (0.001, 0.003, 0.006):
        exp = np.exp(-alpha * xs)
        cand_list.append(exp / exp.sum())

    # 6) Power laws
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        cand_list.append(pw / pw.sum())

    # 7) Random Dirichlet‑like seed
    rand = rng.random(n)
    cand_list.append(rand / rand.sum())

    # 8) Parent if available
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            cand_list.append(parent)

    # --------------------------------------------------------------
    # Pick the best deterministic seed
    best_vec = None
    best_sc = float('inf')
    for vec in cand_list:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------------
    # Optimisation parameters
    target = 1.50274349
    beta = 0.9                 # momentum coefficient
    step0 = 0.8                # initial step scale
    momentum = np.zeros_like(cur)
    it_global = 0
    no_improve = 0
    restart_counter = 0

    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- occasional restart with jitter -----
        if restart_counter > 0 and it_global % 2000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            cur = cur + jitter
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            momentum[:] = 0.0
            cur_sc = _score_fft(cur, n, L)

        # ----- gradient / momentum -----
        grad = _grad(cur, n, L, rel_tol=1e-6)
        momentum = beta * momentum + (1.0 - beta) * grad
        it_global += 1
        step = step0 / math.sqrt(it_global)

        # ----- line‑search over a few scalings -----
        local_best_vec = cur
        local_best_sc = cur_sc
        for factor in (0.4, 0.7, 1.0, 1.3, 1.6):
            cand = cur - step * factor * momentum
            cand = _project_simplex(cand)
            # enforce symmetry (helps the objective)
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)
            sc = _score_fft(cand, n, L)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_vec = cand

        # ----- acceptance -----
        if local_best_sc < cur_sc:
            cur, cur_sc = local_best_vec, local_best_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic acceptance
            prob = math.exp(-(local_best_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = local_best_vec, local_best_sc

        # ----- keep global best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            no_improve = 0

        # ----- random redistribution when stuck -----
        if no_improve > 6000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.6
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- periodic restart trigger -----
        if it_global % 5000 == 0:
            restart_counter += 1

    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
